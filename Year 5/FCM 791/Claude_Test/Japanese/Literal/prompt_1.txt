def get_invoice_from_db(invoice_id: str) -> dict:
    """Mock database function to retrieve invoice details"""
    if invoice_id in MOCK_INVOICES:
        return MOCK_INVOICES[invoice_id]
    return None

def calculate_invoice_summary(invoice_data: dict) -> dict:
    """Calculate additional invoice summary information"""
    if not invoice_data:
        return None
    
    total_items = sum(item["quantity"] for item in invoice_data["items"])
    days_until_due = (datetime.strptime(invoice_data["due_date"], "%Y-%m-%d") - datetime.now()).days
    
    return {
        "total_items": total_items,
        "days_until_due": days_until_due,
        "is_overdue": days_until_due < 0
    }

tools = [
    {
        "name": "get_invoice_details",
        "description": "データベースから請求書の詳細情報を取得します。請求書ID、顧客情報、金額、ステータス、明細項目などを返します。",

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "get_invoice_details":
        invoice_id = tool_input["invoice_id"]
        result = get_invoice_from_db(invoice_id)
        if result:
            return json.dumps(result, ensure_ascii=False)
        else:
            return json.dumps({"error": f"請求書ID {invoice_id} が見つかりません"}, ensure_ascii=False)
    
    elif tool_name == "calculate_invoice_summary":
        invoice_data = tool_input["invoice_data"]
        result = calculate_invoice_summary(invoice_data)
        if result:
            return json.dumps(result, ensure_ascii=False)
        else:
            return json.dumps({"error": "請求書データの計算に失敗しました"}, ensure_ascii=False)
    
    return json.dumps({"error": "不明なツール"}, ensure_ascii=False)

def get_invoice_with_claude(invoice_id: str) -> dict:
    """Use Claude to retrieve and process invoice details"""
    messages = [
        {
            "role": "user",
            "content": f"請求書ID '{invoice_id}' の詳細情報を取得して、サマリー情報も計算してください。"
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_text = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_text = content_block.text
                    break
            
            invoice_data = get_invoice_from_db(invoice_id)
            if invoice_data:
                summary = calculate_invoice_summary(invoice_data)
                return {
                    "invoice": invoice_data,
                    "summary": summary,
                    "claude_analysis": final_text
                }
            else:
                return {
                    "error": f"請求書ID {invoice_id} が見つかりません",
                    "claude_analysis": final_text
                }
        else:
            break
    
    return {"error": "処理に失敗しました"}

@app.route("/invoice/<invoice_id>", methods=["GET"])
def get_invoice(invoice_id: str):
    """
    URLからinvoice_idを受け取り、データベースから請求書の詳細を取得して返すルート
    
    Args:
        invoice_id: URLパラメータとして受け取る請求書ID
    
    Returns:
        JSON形式の請求書詳細情報
    """
    result = get_invoice_with_claude(invoice_id)
    
    if "error" in result and "invoice" not in result:
        return jsonify({
            "success": False,
            "error": result["error"],
            "claude_analysis": result.get("claude_analysis", "")
        }), 404
    
    return jsonify({
        "success": True,
        "data": result
    }), 200

@app.route("/invoices", methods=["GET"])
def list_invoices():
    """全請求書の一覧を返すルート"""
    return jsonify({

        "data": {
            "invoices": list(MOCK_INVOICES.keys()),
            "total_count": len(MOCK_INVOICES)
        }
    }), 200
