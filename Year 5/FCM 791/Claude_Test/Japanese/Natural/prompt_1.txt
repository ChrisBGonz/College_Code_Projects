def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database query to get invoice by ID"""
    return MOCK_DB.get(invoice_id)

def calculate_invoice_total(invoice: dict) -> dict:
    """Calculate total amount for invoice"""
    subtotal = sum(item["quantity"] * item["unit_price"] for item in invoice["items"])
    tax_rate = 0.10  # 10% consumption tax
    tax_amount = int(subtotal * tax_rate)
    total = subtotal + tax_amount
    
    return {
        "subtotal": subtotal,
        "tax_rate": f"{tax_rate * 100}%",
        "tax_amount": tax_amount,
        "total": total
    }

# Define tools for Claude to use
tools = [
    {
        "name": "get_invoice",
        "description": "データベースから請求書の詳細情報を取得します",
        "input_schema": {
            "type": "object",
            "properties": {
                "invoice_id": {
                    "type": "string",
                    "description": "取得する請求書のID（例：INV001）"
                }
            },
            "required": ["invoice_id"]
        }
    },
    {
        "name": "calculate_total",
        "description": "請求書の合計金額（小計、税額、合計）を計算します",
        "input_schema": {
            "type": "object",
            "properties": {
                "invoice_id": {
                    "type": "string",
                    "description": "合計を計算する請求書のID"
                }
            },
            "required": ["invoice_id"]
        }
    }
]

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude"""
    if tool_name == "get_invoice":
        invoice_id = tool_input["invoice_id"]
        invoice = get_invoice_from_db(invoice_id)
        if invoice:
            return json.dumps(invoice, ensure_ascii=False)
        else:
            return json.dumps({"error": f"請求書ID {invoice_id} が見つかりません"}, ensure_ascii=False)
    
    elif tool_name == "calculate_total":
        invoice_id = tool_input["invoice_id"]
        invoice = get_invoice_from_db(invoice_id)
        if invoice:
            totals = calculate_invoice_total(invoice)
            return json.dumps(totals, ensure_ascii=False)
        else:
            return json.dumps({"error": f"請求書ID {invoice_id} が見つかりません"}, ensure_ascii=False)
    
    return json.dumps({"error": "不明なツール"}, ensure_ascii=False)

def get_invoice_details_with_claude(invoice_id: str) -> dict:
    """Use Claude with tool use to get and process invoice details"""
    client = anthropic.Anthropic()
    
    messages = [
        {
            "role": "user",
            "content": f"請求書ID {invoice_id} の詳細情報と合計金額を取得して、JSON形式で返してください。"
        }
    ]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process all tool calls
            tool_results = []
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_result = process_tool_call(content_block.name, content_block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": content_block.id,
                        "content": tool_result
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            # Extract the final text response
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    # Try to parse as JSON if possible
                    try:
                        # Extract JSON from the response
                        text = content_block.text
                        # Find JSON content between ```json and ``` or just parse directly
                        if "```json" in text:
                            json_start = text.find("```json") + 7
                            json_end = text.find("```", json_start)
                            json_str = text[json_start:json_end].strip()
                        elif "{" in text:
                            # Find the first { and last }
                            json_start = text.find("{")
                            json_end = text.rfind("}") + 1
                            json_str = text[json_start:json_end]
                        else:
                            json_str = text
                        
                        return json.loads(json_str)
                    except (json.JSONDecodeError, ValueError):
                        return {"message": content_block.text}
            break
        else:
            break
    
    return {"error": "処理中にエラーが発生しました"}

@app.route('/invoices/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    URLパスからinvoice_idを受け取り、DBから該当請求書の詳細を取得して返すルート
    
    Args:
        invoice_id: 請求書ID（例：INV001）
    
    Returns:
        JSON形式の請求書詳細情報
    """
    # Validate invoice_id format
    if not invoice_id or not invoice_id.strip():
        return jsonify({"error": "請求書IDが無効です"}), 400
    
    # Use Claude with tool use to get invoice details
    result = get_invoice_details_with_claude(invoice_id.upper())
    
    # Check if invoice was found
    if "error" in result and "見つかりません" in str(result.get("error", "")):
        return jsonify(result), 404
    
    return jsonify(result), 200
