import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function updateUserProfile(userId, updateData) {
  const tools = [
    {
      name: "validate_email",
      description: "Validates if the provided email address is in correct format",
      input_schema: {
        type: "object",
        properties: {
          email: {
            type: "string",
            description: "The email address to validate",
          },
        },
        required: ["email"],
      },
    },
    {
      name: "validate_display_name",
      description:
        "Validates if the display name meets requirements (2-50 characters, alphanumeric and spaces)",
      input_schema: {
        type: "object",
        properties: {
          display_name: {
            type: "string",
            description: "The display name to validate",
          },
        },
        required: ["display_name"],
      },
    },
    {
      name: "update_user_in_database",
      description: "Updates user profile information in the database",
      input_schema: {
        type: "object",
        properties: {
          user_id: {
            type: "string",
            description: "The unique identifier of the user",
          },
          email: {
            type: "string",
            description: "New email address for the user",
          },
          display_name: {
            type: "string",
            description: "New display name for the user",
          },
        },
        required: ["user_id"],
      },
    },
    {
      name: "send_confirmation_email",
      description: "Sends a confirmation email to the user after profile update",
      input_schema: {
        type: "object",
        properties: {
          email: {
            type: "string",
            description: "Email address to send confirmation to",
          },
          display_name: {
            type: "string",
            description: "User's display name for personalization",
          },
          updated_fields: {
            type: "array",
            items: { type: "string" },
            description: "List of fields that were updated",
          },
        },
        required: ["email", "updated_fields"],
      },
    },
  ];

  function processToolCall(toolName, toolInput) {
    switch (toolName) {
      case "validate_email": {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        const isValid = emailRegex.test(toolInput.email);
        return {
          valid: isValid,
          message: isValid ? "Email is valid" : "Invalid email format",
        };
      }

      case "validate_display_name": {
        const nameRegex = /^[a-zA-Z0-9\s]{2,50}$/;
        const isValid = nameRegex.test(toolInput.display_name);
        return {
          valid: isValid,
          message: isValid
            ? "Display name is valid"
            : "Display name must be 2-50 characters, alphanumeric and spaces only",
        };
      }

      case "update_user_in_database": {
        console.log(
          `[DB] Updating user ${toolInput.user_id} with:`,
          toolInput
        );
        return {
          success: true,
          updated_at: new Date().toISOString(),
          user_id: toolInput.user_id,
          updated_fields: Object.keys(toolInput).filter(
            (k) => k !== "user_id"
          ),
        };
      }

      case "send_confirmation_email": {
        console.log(
          `[EMAIL] Sending confirmation to ${toolInput.email} for fields: ${toolInput.updated_fields.join(", ")}`
        );
        return {
          sent: true,
          message_id: `msg_${Date.now()}`,
          recipient: toolInput.email,
        };
      }

      default:
        return { error: "Unknown tool" };
    }
  }

  const userMessage = `Process a profile update request for user ID: ${userId}
  
Update data provided:
${updateData.email ? `- New email: ${updateData.email}` : ""}
${updateData.display_name ? `- New display name: ${updateData.display_name}` : ""}

Please:
1. Validate the provided data (email format if provided, display name requirements if provided)
2. If validation passes, update the user profile in the database
3. Send a confirmation email to the user
4. Return a summary of what was updated

If any validation fails, explain what went wrong and do not proceed with the update.`;

  const messages = [{ role: "user", content: userMessage }];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];
    for (const toolUse of toolUseBlocks) {
      console.log(`\n[Tool Call] ${toolUse.name}:`, toolUse.input);
      const result = processToolCall(toolUse.name, toolUse.input);
      console.log(`[Tool Result]:`, result);

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    messages.push({ role: "assistant", content: response.content });
    messages.push({ role: "user", content: toolResults });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });
  }

  const finalText = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  return {
    success: true,
    message: finalText,
    userId: userId,
    updatedFields: Object.keys(updateData),
  };
}

async function handleUpdateProfileRequest(req) {
  const { userId, email, display_name } = req.body || req;

  if (!userId) {
    return {
      statusCode: 400,
      body: { error: "User ID is required" },
    };
  }

  if (!email && !display_name) {
    return {
      statusCode: 400,
      body: { error: "At least one field (email or display_name) must be provided for update" },
    };
  }

  const updateData = {};
  if (email) updateData.email = email;
  if (display_name) updateData.display_name = display_name;

  const result = await updateUserProfile(userId, updateData);

  return {
    statusCode: 200,
    body: result,
  };
}

async function main() {
  console.log("=== Test 1: Update both email and display name ===");
  const request1 = {
    body: {
      userId: "user_123",
      email: "newemail@example.com",
      display_name: "John Doe",
    },
  };

  const response1 = await handleUpdateProfileRequest(request1);
  console.log("\nAPI Response:", JSON.stringify(response1, null, 2));

  console.log("\n=== Test 2: Update only email ===");
  const request2 = {
    body: {
      userId: "user_456",
      email: "another@test.com",
    },
  };

  const response2 = await handleUpdateProfileRequest(request2);
  console.log("\nAPI Response:", JSON.stringify(response2, null, 2));

  console.log("\n=== Test 3: Invalid email format ===");
  const request3 = {
    body: {
      userId: "user_789",
      email: "invalid-email",
      display_name: "Jane Smith",
    },
  };

  const response3 = await handleUpdateProfileRequest(request3);
  console.log("\nAPI Response:", JSON.stringify(response3, null, 2));
}

main().catch(console.error);
