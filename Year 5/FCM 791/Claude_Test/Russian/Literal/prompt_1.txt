def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database lookup for invoice."""
    return INVOICES_DB.get(invoice_id)

def analyze_invoice_with_claude(invoice_data: dict) -> str:
    """Use Claude to analyze invoice and provide insights using multi-turn conversation."""
    conversation_history = []
    
    # First turn: Provide invoice data and ask for analysis
    conversation_history.append({
        "role": "user",
        "content": f"""Please analyze the following invoice data and provide a comprehensive summary:

Invoice Data:
{json.dumps(invoice_data, indent=2)}

Please provide:
1. A brief summary of the invoice
2. Payment status assessment
3. Any important notes or alerts"""
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1000,
        system="You are a financial assistant specialized in invoice analysis. Provide clear, concise analysis of invoice data.",
        messages=conversation_history
    )
    
    first_analysis = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": first_analysis
    })
    
    # Second turn: Ask for recommendations
    conversation_history.append({
        "role": "user",
        "content": "Based on this invoice analysis, what are your top 3 recommendations for the business?"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        system="You are a financial assistant specialized in invoice analysis. Provide clear, concise analysis of invoice data.",
        messages=conversation_history
    )
    
    recommendations = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": recommendations
    })
    
    # Third turn: Ask for risk assessment
    conversation_history.append({
        "role": "user",
        "content": "What is the risk level associated with this invoice and why?"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=300,
        system="You are a financial assistant specialized in invoice analysis. Provide clear, concise analysis of invoice data.",
        messages=conversation_history
    )
    
    risk_assessment = response.content[0].text
    
    # Combine all insights
    full_analysis = f"""
=== INVOICE ANALYSIS ===

{first_analysis}

=== RECOMMENDATIONS ===

{recommendations}

=== RISK ASSESSMENT ===

{risk_assessment}
"""
    
    return full_analysis

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Flask route that accepts invoice_id from URL and returns invoice details.
    Uses Claude AI for multi-turn conversation to analyze the invoice.
    """
    # Validate invoice_id format
    if not invoice_id or len(invoice_id) > 50:
        return jsonify({
            "error": "Invalid invoice ID format",
            "status": "error"
        }), 400
    
    # Fetch invoice from database
    invoice_data = get_invoice_from_db(invoice_id)
    
    if not invoice_data:
        return jsonify({
            "error": f"Invoice {invoice_id} not found",
            "status": "error"
        }), 404
    
    # Analyze invoice with Claude using multi-turn conversation
    ai_analysis = analyze_invoice_with_claude(invoice_data)
    
    # Prepare response
    response_data = {
        "status": "success",
        "invoice": invoice_data,
        "ai_analysis": ai_analysis
    }
    
    return jsonify(response_data), 200

@app.route('/invoices', methods=['GET'])

    invoice_list = [
        {
            "id": inv_id,
            "customer": inv_data["customer_name"],
            "amount": inv_data["amount"],
            "currency": inv_data["currency"],
            "status": inv_data["status"]
        }
        for inv_id, inv_data in INVOICES_DB.items()
    ]
    
    return jsonify({
        "status": "success",
        "invoices": invoice_list,
        "total": len(invoice_list)
    }), 200
