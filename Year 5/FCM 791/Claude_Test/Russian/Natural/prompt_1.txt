def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database query to get invoice by ID."""
    return INVOICES_DB.get(invoice_id)

def analyze_invoice_with_claude(invoice_data: dict) -> str:
    """Use Claude to analyze invoice and provide insights using multi-turn conversation."""
    conversation_history = []
    
    # First turn: Provide invoice data and ask for initial analysis
    conversation_history.append({
        "role": "user",
        "content": f"""I have an invoice with the following details:
        
Invoice ID: {invoice_data['id']}
Customer: {invoice_data['customer_name']} ({invoice_data['customer_email']})
Amount: {invoice_data['amount']} {invoice_data['currency']}
Status: {invoice_data['status']}
Issue Date: {invoice_data['issue_date']}
Due Date: {invoice_data['due_date']}
Items: {invoice_data['items']}
Notes: {invoice_data.get('notes', 'None')}

Please provide a brief analysis of this invoice."""
    })
    
    response1 = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        system="You are a financial assistant that analyzes invoices and provides helpful insights. Be concise and professional.",
        messages=conversation_history
    )
    
    initial_analysis = response1.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": initial_analysis
    })
    
    # Second turn: Ask for specific recommendations based on status
    conversation_history.append({
        "role": "user",
        "content": f"Based on the invoice status '{invoice_data['status']}', what specific actions should be taken? Please provide 2-3 concrete recommendations."
    })
    
    response2 = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=400,
        system="You are a financial assistant that analyzes invoices and provides helpful insights. Be concise and professional.",
        messages=conversation_history
    )
    
    recommendations = response2.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": recommendations
    })
    
    # Third turn: Ask for a summary
    conversation_history.append({
        "role": "user",
        "content": "Please provide a one-sentence executive summary of this invoice situation."
    })
    
    response3 = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=150,
        system="You are a financial assistant that analyzes invoices and provides helpful insights. Be concise and professional.",
        messages=conversation_history
    )
    
    executive_summary = response3.content[0].text
    
    # Combine all insights
    full_analysis = f"""
INVOICE ANALYSIS
================

Initial Analysis:
{initial_analysis}

Recommended Actions:
{recommendations}

Executive Summary:
{executive_summary}
"""
    
    return full_analysis.strip()

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Flask route to get invoice details by ID.
    
    Args:
        invoice_id: The invoice ID from the URL path parameter
        
    Returns:
        JSON response with invoice details and AI analysis
    """
    # Fetch invoice from database
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": "Invoice not found",
            "invoice_id": invoice_id,
            "message": f"No invoice found with ID: {invoice_id}"
        }), 404
    
    # Get AI analysis using multi-turn conversation
    ai_analysis = analyze_invoice_with_claude(invoice)
    
    # Prepare response
    response_data = {
        "invoice": invoice,
        "ai_analysis": ai_analysis,
        "status": "success"
    }
    
    return jsonify(response_data), 200

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """List all available invoice IDs."""
    invoice_ids = list(INVOICES_DB.keys())
    return jsonify({
        "available_invoices": invoice_ids,
        "count": len(invoice_ids)
    }), 200

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
