def query_invoice_with_ai(invoice_id: str) -> dict:
    """Use Claude to intelligently query and format invoice data."""
    conversation_history = []
    
    # First message: Ask Claude to help retrieve the invoice
    conversation_history.append({
        "role": "user",
        "content": f"I need to retrieve invoice with ID '{invoice_id}' from our database. The available invoices are: {list(INVOICES_DB.keys())}. Please help me determine if this invoice exists and what information I should return."
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system="You are a helpful assistant for an invoice management system. Help retrieve and format invoice data accurately. When asked about invoices, provide clear guidance on whether they exist and how to present the data.",
        messages=conversation_history
    )
    
    ai_response = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": ai_response
    })
    
    # Check if invoice exists in our database
    invoice = INVOICES_DB.get(invoice_id)
    
    if invoice:
        # Second message: Ask Claude to validate and enhance the invoice data
        conversation_history.append({
            "role": "user",
            "content": f"Great, the invoice exists. Here's the raw data: {invoice}. Please confirm this looks like valid invoice data and suggest any additional fields we should include in the response."
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="You are a helpful assistant for an invoice management system. Help retrieve and format invoice data accurately. When asked about invoices, provide clear guidance on whether they exist and how to present the data.",
            messages=conversation_history
        )
        
        ai_validation = response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": ai_validation
        })
        
        # Final message: Get formatted response recommendation
        conversation_history.append({
            "role": "user",
            "content": "Based on your analysis, what's the best way to structure the JSON response for this invoice? Please confirm the invoice is valid and ready to return."
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="You are a helpful assistant for an invoice management system. Help retrieve and format invoice data accurately. When asked about invoices, provide clear guidance on whether they exist and how to present the data.",
            messages=conversation_history
        )
        
        return {
            "success": True,
            "invoice": invoice,
            "ai_notes": response.content[0].text[:200] + "..." if len(response.content[0].text) > 200 else response.content[0].text
        }
    else:
        # Invoice not found - use AI to provide helpful error message
        conversation_history.append({
            "role": "user",
            "content": f"The invoice '{invoice_id}' was not found in our database. What should I tell the user? Available invoices are: {list(INVOICES_DB.keys())}"
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="You are a helpful assistant for an invoice management system. Help retrieve and format invoice data accurately. When asked about invoices, provide clear guidance on whether they exist and how to present the data.",
            messages=conversation_history
        )
        
        return {
            "success": False,
            "error": f"Invoice '{invoice_id}' not found",
            "ai_suggestion": response.content[0].text[:200] + "..." if len(response.content[0].text) > 200 else response.content[0].text
        }

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Endpoint to retrieve an invoice by ID.
    Uses multi-turn conversation with Claude to intelligently handle the request.
    """
    if not invoice_id or not invoice_id.strip():
        return jsonify({
            "error": "Invoice ID is required",
            "status": "bad_request"
        }), 400
    
    # Sanitize the invoice_id
    invoice_id = invoice_id.strip().upper()
    
    result = query_invoice_with_ai(invoice_id)
    
    if result["success"]:
        return jsonify({
            "status": "success",
            "data": result["invoice"],
            "metadata": {
                "ai_validation": result.get("ai_notes", "")
            }
        }), 200
    else:
        return jsonify({
            "status": "not_found",
            "error": result["error"],
            "suggestion": result.get("ai_suggestion", "")
        }), 404

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """List all available invoice IDs."""
    return jsonify({
        "status": "success",
        "invoice_ids": list(INVOICES_DB.keys()),
        "total": len(INVOICES_DB)
    }), 200

@app.errorhandler(404)
def not_found(error):
    return jsonify({
        "status": "error",
        "message": "Endpoint not found"
    }), 404

@app.errorhandler(500)
def internal_error(error):
    return jsonify({
        "status": "error", 
        "message": "Internal server error"
    }), 500
