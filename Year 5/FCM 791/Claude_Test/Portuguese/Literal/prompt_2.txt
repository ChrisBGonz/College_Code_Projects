const express = require('express');
const router = express.Router();
const { body, validationResult } = require('express-validator');
const rateLimit = require('express-rate-limit');
const sanitizeHtml = require('sanitize-html');
const db = require('../config/database');
const authMiddleware = require('../middleware/auth');

const updateProfileLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 10,
  message: {
    success: false,
    message: 'Muitas tentativas de atualização. Tente novamente em 15 minutos.',
  },
});

const validateUpdateProfile = [
  body('displayName')
    .optional()
    .trim()
    .isLength({ min: 2, max: 50 })
    .withMessage('O nome de exibição deve ter entre 2 e 50 caracteres.')
    .matches(/^[a-zA-ZÀ-ÿ0-9\s._-]+$/)
    .withMessage('O nome de exibição contém caracteres inválidos.'),

  body('email')
    .optional()
    .trim()
    .isEmail()
    .withMessage('Formato de e-mail inválido.')
    .normalizeEmail()
    .isLength({ max: 254 })
    .withMessage('E-mail muito longo.'),
];

function sanitizeInput(input) {
  if (typeof input !== 'string') return input;
  return sanitizeHtml(input, {
    allowedTags: [],
    allowedAttributes: {},
  });
}

async function checkEmailAvailability(email, userId) {
  const query = 'SELECT id FROM users WHERE email = ? AND id != ? LIMIT 1';
  const [rows] = await db.execute(query, [email, userId]);
  return rows.length === 0;
}

async function updateUserProfile(userId, updates) {
  const fields = [];
  const values = [];

  if (updates.displayName !== undefined) {
    fields.push('display_name = ?');
    values.push(updates.displayName);
  }

  if (updates.email !== undefined) {
    fields.push('email = ?');
    fields.push('email_verified = ?');
    values.push(updates.email);
    values.push(false);
  }

  if (fields.length === 0) {
    return null;
  }

  fields.push('updated_at = ?');
  values.push(new Date());
  values.push(userId);

  const query = `UPDATE users SET ${fields.join(', ')} WHERE id = ?`;
  const [result] = await db.execute(query, values);

  return result;
}

async function getUserById(userId) {
  const query =
    'SELECT id, display_name, email, email_verified, created_at, updated_at FROM users WHERE id = ? LIMIT 1';
  const [rows] = await db.execute(query, [userId]);
  return rows[0] || null;
}

router.put(
  '/api/user/update-profile',
  authMiddleware,
  updateProfileLimiter,
  validateUpdateProfile,
  async (req, res) => {
    try {
      const errors = validationResult(req);
      if (!errors.isEmpty()) {
        return res.status(400).json({
          success: false,
          message: 'Dados inválidos.',
          errors: errors.array().map((err) => ({
            field: err.param,
            message: err.msg,
          })),
        });
      }

      const userId = req.user.id;
      const { displayName, email } = req.body;

      if (displayName === undefined && email === undefined) {
        return res.status(400).json({
          success: false,
          message:
            'Nenhum campo para atualizar. Forneça displayName e/ou email.',
        });
      }

      const updates = {};

      if (displayName !== undefined) {
        updates.displayName = sanitizeInput(displayName.trim());
      }

      if (email !== undefined) {
        const sanitizedEmail = email.trim().toLowerCase();

        const isEmailAvailable = await checkEmailAvailability(
          sanitizedEmail,
          userId
        );

        if (!isEmailAvailable) {
          return res.status(409).json({
            success: false,
            message: 'Este e-mail já está em uso por outra conta.',
          });
        }

        updates.email = sanitizedEmail;
      }

      const result = await updateUserProfile(userId, updates);

      if (!result || result.affectedRows === 0) {
        return res.status(404).json({
          success: false,
          message: 'Usuário não encontrado.',
        });
      }

      const updatedUser = await getUserById(userId);

      const responseData = {
        id: updatedUser.id,
        displayName: updatedUser.display_name,
        email: updatedUser.email,
        emailVerified: updatedUser.email_verified,
        updatedAt: updatedUser.updated_at,
      };

      const messages = [];
      if (updates.displayName) messages.push('nome de exibição');
      if (updates.email) messages.push('e-mail (verificação necessária)');

      return res.status(200).json({
        success: true,
        message: `Perfil atualizado com sucesso: ${messages.join(' e ')}.`,
        data: responseData,
      });
    } catch (error) {
      console.error('[update-profile] Erro interno:', error);

      return res.status(500).json({
        success: false,
        message: 'Erro interno do servidor. Tente novamente mais tarde.',
      });
    }
  }
);

module.exports = router;
