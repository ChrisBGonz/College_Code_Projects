
```python
import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SYSTEM_PROMPT = """You are an expert Python Flask developer. Help users create Flask routes for querying invoice details from a database.

When asked to create a Flask route for invoice details, provide complete, working code that:
1. Uses Flask's URL path parameters to get invoice_id
2. Queries a database (use SQLAlchemy or similar)
3. Returns JSON response with invoice details
4. Handles errors appropriately (404 if not found, 500 for server errors)

Always provide practical, production-ready code with proper error handling."""

def chat(user_message):
    """Send a message and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def main():
    """Main function to demonstrate the Flask route creation."""
    print("Flask Invoice Route Generator")
    print("=" * 50)
    
    # Initial request to create the Flask route
    initial_request = """请创建一个 Python Flask 路由：
    1. 从 URL 路径参数中获取 invoice_id
    2. 查询数据库返回对应发票（invoice）详情
    3. 以 JSON 格式输出响应
    
    请提供完整的代码示例，包括：
    - Flask 应用设置
    - 数据库模型定义
    - 路由实现
    - 错误处理"""
    
    print(f"用户: {initial_request}\n")
    response = chat(initial_request)
    print(f"助手: {response}\n")
    print("=" * 50)
    
    # Follow-up question about error handling
    followup1 = "如何改进错误处理？特别是当 invoice_id 不是有效整数时应该怎么处理？"
    print(f"用户: {followup1}\n")
    response = chat(followup1)
    print(f"助手: {response}\n")
    print("=" * 50)
    
    # Ask about adding authentication
    followup2 = "如何为这个路由添加 JWT 认证保护？"
    print(f"用户: {followup2}\n")
    response = chat(followup2)
    print(f"助手: {response}\n")
    print("=" * 50)
    
    # Ask about testing
    followup3 = "如何为这个路由编写单元测试？"
    print(f"用户: {followup3}\n")
    response = chat(followup3)
    print(f"助手: {response}\n")
    print("=" * 50)
    
    print("\n对话完成！以上展示了如何使用多轮对话来逐步完善 Flask 发票路由的实现。")

if __name__ == "__main__":
    main()
```
