def query_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database query for invoice"""
    return MOCK_INVOICES.get(invoice_id)

def process_invoice_with_claude(invoice_id: str, invoice_data: dict | None) -> dict:
    """Use Claude to process and format invoice response using extended thinking"""
    
    if invoice_data is None:
        prompt = f"""Invoice ID '{invoice_id}' को database में नहीं मिला।
        
        एक appropriate error response JSON बनाओ जिसमें:
        - error message हो
        - status code हो
        - suggestion हो कि user क्या करे
        
        Response सिर्फ valid JSON होना चाहिए।"""
    else:
        prompt = f"""निम्नलिखित invoice data को analyze करो और एक comprehensive JSON response बनाओ:

Invoice Data:
{json.dumps(invoice_data, indent=2, ensure_ascii=False)}

Response में निम्नलिखित शामिल करो:
1. Invoice की basic details
2. Payment status और उसका explanation
3. Total amount calculation verification
4. अगर invoice overdue है तो warning message
5. Customer के लिए next steps

Response सिर्फ valid JSON format में होना चाहिए।"""
    
    response = client.messages.create(
        model="claude-sonnet-4-5",
        max_tokens=16000,
        thinking={
            "type": "enabled",
            "budget_tokens": 10000
        },
        messages=[{
            "role": "user",
            "content": prompt
        }]
    )
    
    result_text = ""
    for block in response.content:
        if block.type == "text":
            result_text = block.text
            break
    
    try:
        if "```json" in result_text:
            json_start = result_text.find("```json") + 7
            json_end = result_text.find("```", json_start)
            result_text = result_text[json_start:json_end].strip()
        elif "```" in result_text:
            json_start = result_text.find("```") + 3
            json_end = result_text.find("```", json_start)
            result_text = result_text[json_start:json_end].strip()
        
        return json.loads(result_text)
    except json.JSONDecodeError:
        if invoice_data is None:
            return {
                "error": "Invoice not found",
                "invoice_id": invoice_id,
                "status_code": 404,
                "message": f"Invoice '{invoice_id}' database में नहीं मिला"
            }
        else:
            return {
                "invoice_id": invoice_id,
                "data": invoice_data,
                "analysis": result_text
            }

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Invoice details को fetch और return करने का Flask route
    
    Path Parameter:
        invoice_id: Invoice का unique identifier
    
    Returns:
        JSON response with invoice details or error message
    """
    invoice_data = query_invoice_from_db(invoice_id)
    
    processed_response = process_invoice_with_claude(invoice_id, invoice_data)
    
    if invoice_data is None:
        return jsonify(processed_response), 404
    
    return jsonify({
        "success": True,
        "invoice_id": invoice_id,
        "data": processed_response
    }), 200

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """सभी available invoice IDs की list return करता है"""
    return jsonify({
        "success": True,
        "available_invoices": list(MOCK_INVOICES.keys()),
        "total_count": len(MOCK_INVOICES)
    }), 200

@app.errorhandler(404)
def not_found(error):
    return jsonify({
        "error": "Route not found",
        "message": "Requested URL does not exist"
    }), 404

@app.errorhandler(500)
def internal_error(error):
    return jsonify({
        "error": "Internal server error",
        "message": "Something went wrong on the server"
    }), 500

if __name__ == '__main__':
