```javascript
// updateProfile.js - Handler for PATCH /api/user/update-profile

const express = require('express');
const router = express.Router();
const { body, validationResult } = require('express-validator');
const db = require('../db'); // your DB connection/ORM
const authMiddleware = require('../middleware/auth');

// ─── Validation Rules ────────────────────────────────────────────────────────
const updateProfileValidation = [
  body('email')
    .optional()
    .trim()
    .toLowerCase()
    .isEmail()
    .withMessage('Valid email address required')
    .normalizeEmail()
    .isLength({ max: 255 })
    .withMessage('Email too long'),

  body('displayName')
    .optional()
    .trim()
    .isLength({ min: 2, max: 50 })
    .withMessage('displayName must be between 2 and 50 characters')
    .matches(/^[a-zA-Z0-9\s\-_.]+$/)
    .withMessage('displayName contains invalid characters'),
];

// ─── Handler ─────────────────────────────────────────────────────────────────
const updateProfileHandler = async (req, res) => {
  try {
    // 1. Auth check — req.user set by authMiddleware
    if (!req.user || !req.user.id) {
      return res.status(401).json({
        success: false,
        error: 'Unauthorized: Please log in to update your profile',
      });
    }

    const userId = req.user.id;

    // 2. Validate input
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({
        success: false,
        error: 'Validation failed',
        details: errors.array().map((e) => ({
          field: e.path,
          message: e.msg,
        })),
      });
    }

    const { email, displayName } = req.body;

    // 3. At least one field must be provided
    if (email === undefined && displayName === undefined) {
      return res.status(400).json({
        success: false,
        error: 'At least one field (email or displayName) must be provided',
      });
    }

    // 4. Fetch current user from DB
    const currentUser = await db.query(
      'SELECT id, email, display_name FROM users WHERE id = $1 AND is_deleted = false',
      [userId]
    );

    if (!currentUser.rows.length) {
      return res.status(401).json({
        success: false,
        error: 'Unauthorized: User account not found',
      });
    }

    const user = currentUser.rows[0];

    // 5. Email uniqueness check (only if email is being changed)
    if (email && email !== user.email) {
      const emailConflict = await db.query(
        'SELECT id FROM users WHERE email = $1 AND id != $2 AND is_deleted = false',
        [email, userId]
      );

      if (emailConflict.rows.length > 0) {
        return res.status(409).json({
          success: false,
          error: 'Email already in use by another account',
        });
      }
    }

    // 6. Build dynamic UPDATE query (only update provided fields)
    const updates = [];
    const values = [];
    let paramIndex = 1;

    if (email !== undefined) {
      updates.push(`email = $${paramIndex++}`);
      values.push(email);
    }

    if (displayName !== undefined) {
      // Sanitize: strip extra whitespace
      const sanitizedName = displayName.replace(/\s+/g, ' ').trim();
      updates.push(`display_name = $${paramIndex++}`);
      values.push(sanitizedName);
    }

    updates.push(`updated_at = NOW()`);
    values.push(userId); // for WHERE clause

    const updateQuery = `
      UPDATE users
      SET ${updates.join(', ')}
      WHERE id = $${paramIndex} AND is_deleted = false
      RETURNING id, email, display_name AS "displayName", updated_at AS "updatedAt"
    `;

    // 7. Execute update
    const result = await db.query(updateQuery, values);

    if (!result.rows.length) {
      return res.status(500).json({
        success: false,
        error: 'Update failed: User not found or already deleted',
      });
    }

    const updatedUser = result.rows[0];

    // 8. Return updated fields
    return res.status(200).json({
      success: true,
      message: 'Profile updated successfully',
      data: {
        id: updatedUser.id,
        email: updatedUser.email,
        displayName: updatedUser.displayName,
        updatedAt: updatedUser.updatedAt,
      },
    });
  } catch (error) {
    console.error('[update-profile] Unexpected error:', error);

    // Handle specific DB errors
    if (error.code === '23505') {
      // PostgreSQL unique violation
      return res.status(409).json({
        success: false,
        error: 'Email already in use by another account',
      });
    }

    return res.status(500).json({
      success: false,
      error: 'Internal server error. Please try again later.',
    });
  }
};

// ─── Route ───────────────────────────────────────────────────────────────────
router.patch(
  '/update-profile',
  authMiddleware,              // 401 if not authenticated
  updateProfileValidation,    // validate + sanitize
  updateProfileHandler        // core logic
);

module.exports = router;
```

```javascript
// middleware/auth.js - JWT Auth Middleware (reference)

const jwt = require('jsonwebtoken');

const authMiddleware = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({
        success: false,
        error: 'Unauthorized: No token provided',
      });
    }

    const token = authHeader.split(' ')[1];
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    req.user = { id: decoded.id }; // attach user id to request
    next();
  } catch (err) {
    return res.status(401).json({
      success: false,
      error: 'Unauthorized: Invalid or expired token',
    });
  }
};

module.exports = authMiddleware;
```
