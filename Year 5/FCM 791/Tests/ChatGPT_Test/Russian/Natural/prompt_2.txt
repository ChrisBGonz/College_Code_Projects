const User = require('../models/User')

const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

module.exports = async function updateProfileHandler(req, res) {
  if (req.method && req.method !== 'POST') {
    res.status(405).set('Allow', 'POST').json({ error: 'Method Not Allowed' })
    return
  }

  const userId = req.user && (req.user.id || req.user._id) ? (req.user.id || req.user._id) : req.userId
  if (!userId) {
    res.status(401).json({ error: 'Unauthorized' })
    return
  }

  const { email, displayName } = req.body || {}
  const update = {}

  if (email !== undefined) {
    if (typeof email !== 'string') {
      res.status(400).json({ error: 'Invalid email' })
      return
    }
    const normalizedEmail = email.trim().toLowerCase()
    if (!emailRegex.test(normalizedEmail)) {
      res.status(400).json({ error: 'Invalid email' })
      return
    }
    update.email = normalizedEmail
  }

  if (displayName !== undefined) {
    if (typeof displayName !== 'string') {
      res.status(400).json({ error: 'Invalid displayName' })
      return
    }
    const normalizedDisplayName = displayName.trim()
    if (!normalizedDisplayName) {
      res.status(400).json({ error: 'Invalid displayName' })
      return
    }
    update.displayName = normalizedDisplayName
  }

  if (Object.keys(update).length === 0) {
    res.status(400).json({ error: 'No fields to update' })
    return
  }

  try {
    const user = await User.findByIdAndUpdate(userId, update, { new: true, runValidators: true })
    if (!user) {
      res.status(404).json({ error: 'User not found' })
      return
    }
    res.status(200).json({ id: user._id, email: user.email, displayName: user.displayName })
  } catch (err) {
    if (err && err.code === 11000) {
      res.status(409).json({ error: 'Email already in use' })
      return
    }
    res.status(500).json({ error: 'Internal Server Error' })
  }
}
