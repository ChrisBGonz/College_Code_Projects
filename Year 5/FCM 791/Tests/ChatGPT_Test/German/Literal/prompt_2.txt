const express = require('express');
const router = express.Router();
const User = require('../models/User');

router.put('/api/user/update-profile', async (req, res) => {
  try {
    const userId = req.user && req.user.id ? req.user.id : req.body.userId;
    if (!userId) return res.status(401).json({ error: 'Unauthorized' });
    const { email, displayName } = req.body;
    if (!email || !displayName) return res.status(400).json({ error: 'email and displayName required' });
    const updated = await User.findByIdAndUpdate(
      userId,
      { email, displayName },
      { new: true, runValidators: true }
    ).select('email displayName');
    if (!updated) return res.status(404).json({ error: 'User not found' });
    res.json({ email: updated.email, displayName: updated.displayName });
  } catch (err) {
    res.status(500).json({ error: 'Server error' });
  }
});

module.exports = router;
