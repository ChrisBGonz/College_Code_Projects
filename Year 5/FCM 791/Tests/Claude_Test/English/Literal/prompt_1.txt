def get_invoice_from_db(invoice_id):
    """Retrieve invoice from the simulated database."""
    return INVOICES_DB.get(invoice_id)

def analyze_invoice_with_claude(invoice_data, conversation_history):
    """Use Claude to analyze invoice data with multi-turn conversation support."""
    
    # Add the invoice context to the conversation
    if not conversation_history:
        # First turn: provide invoice data and ask for analysis
        conversation_history.append({
            "role": "user",
            "content": f"""Please analyze this invoice and provide a summary:

Invoice Data:
{invoice_data}

Please provide:
1. A brief summary of the invoice
2. Payment status assessment
3. Any notable observations"""
        })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a helpful financial assistant that analyzes invoice data and provides insights. Be concise and professional.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message, conversation_history

# Store conversation histories (in production, use a proper session store)
conversation_store = {}

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id):
    """Route to get invoice details and AI analysis."""
    
    # Retrieve invoice from database
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found",
            "available_invoices": list(INVOICES_DB.keys())
        }), 404
    
    # Initialize conversation history for this invoice
    conversation_history = []
    
    # Get initial AI analysis
    invoice_str = str(invoice)
    analysis, conversation_history = analyze_invoice_with_claude(invoice_str, conversation_history)
    
    # Store conversation history for potential follow-up questions
    conversation_store[invoice_id] = {
        "invoice": invoice,
        "conversation": conversation_history
    }
    
    return jsonify({
        "invoice": invoice,
        "ai_analysis": analysis,
        "conversation_id": invoice_id
    })

@app.route('/invoice/<invoice_id>/ask', methods=['POST'])
def ask_about_invoice(invoice_id):
    """Route to ask follow-up questions about an invoice."""
    from flask import request
    
    # Check if we have a conversation history for this invoice
    if invoice_id not in conversation_store:
        return jsonify({
            "error": f"No active conversation for invoice {invoice_id}. Please fetch the invoice first."
        }), 404
    
    # Get the question from request body
    data = request.get_json()
    if not data or 'question' not in data:
        return jsonify({
            "error": "Please provide a 'question' in the request body"
        }), 400
    
    question = data['question']
    stored_data = conversation_store[invoice_id]
    conversation_history = stored_data['conversation']
    
    # Add user's question to conversation
    conversation_history.append({
        "role": "user",
        "content": question
    })
    
    # Get response from Claude
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a helpful financial assistant that analyzes invoice data and provides insights. Be concise and professional.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    # Update stored conversation
    conversation_store[invoice_id]['conversation'] = conversation_history
    
    return jsonify({
        "invoice_id": invoice_id,
        "question": question,
        "answer": assistant_message,
        "conversation_turns": len(conversation_history) // 2
    })

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """Route to list all available invoices."""
    invoice_summaries = []
    for inv_id, inv_data in INVOICES_DB.items():
        invoice_summaries.append({
            "id": inv_id,
            "customer": inv_data["customer"],
            "amount": inv_data["amount"],
            "status": inv_data["status"],
            "due_date": inv_data["due_date"]
        })
    
    return jsonify({
        "invoices": invoice_summaries,
        "total_count": len(invoice_summaries)
    })
