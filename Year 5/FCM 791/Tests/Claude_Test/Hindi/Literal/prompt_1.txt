def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database lookup for invoice"""
    return INVOICES_DB.get(invoice_id)

def analyze_invoice_with_claude(invoice_data: dict) -> str:
    """Use Claude to analyze and provide insights about the invoice using streaming"""
    client = anthropic.Anthropic()
    
    invoice_json = json.dumps(invoice_data, ensure_ascii=False, indent=2)
    
    prompt = f"""निम्नलिखित इनवॉइस डेटा का विश्लेषण करें और एक संक्षिप्त सारांश प्रदान करें:

{invoice_json}

कृपया निम्नलिखित जानकारी प्रदान करें:
1. इनवॉइस की स्थिति
2. कुल राशि का सारांश
3. मुख्य वस्तुएं
4. कोई महत्वपूर्ण टिप्पणी"""

    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            full_response += text
            print(text, end="", flush=True)
    
    print()  # New line after streaming completes
    return full_response

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Flask route to get invoice details by invoice_id
    
    Args:
        invoice_id: The unique identifier for the invoice
        
    Returns:
        JSON response with invoice details and AI analysis
    """
    # Fetch invoice from database
    invoice_data = get_invoice_from_db(invoice_id)
    
    if not invoice_data:
        return jsonify({
            "error": "इनवॉइस नहीं मिला",
            "message": f"Invoice ID '{invoice_id}' के लिए कोई रिकॉर्ड नहीं मिला",
            "invoice_id": invoice_id
        }), 404
    
    # Get AI analysis using Claude with streaming
    print(f"\nClaude AI विश्लेषण (Invoice {invoice_id}):")
    print("-" * 50)
    ai_analysis = analyze_invoice_with_claude(invoice_data)
    print("-" * 50)
    
    # Prepare response
    response = {
        "success": True,
        "invoice": invoice_data,
        "ai_analysis": ai_analysis,
        "message": "इनवॉइस सफलतापूर्वक प्राप्त हुआ"
    }
    
    return jsonify(response), 200
