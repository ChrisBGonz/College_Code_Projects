import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SYSTEM_PROMPT = """You are a helpful assistant for a Flask web application. 
You help users understand invoice details and answer questions about their invoices.
When asked about invoice information, provide clear and helpful responses based on the data provided."""

            ],
            "created_date": "2024-01-15",
            "due_date": "2024-02-15",
            "paid_date": "2024-02-10"
        },
        2: {
            "id": 2,
            "customer_name": "Bob Smith",
            "customer_email": "bob@example.com",
            "amount": 2750.50,
            "currency": "USD",
            "status": "pending",
            "items": [
                {"description": "Consulting Services", "quantity": 5, "unit_price": 350.00},
                {"description": "Software License", "quantity": 1, "unit_price": 1000.50}
            ],

            "due_date": "2024-01-01",
            "paid_date": None
        }
    }
    return invoices.get(invoice_id)

def chat_about_invoice(user_message: str, invoice_data: dict) -> str:
    """Have a multi-turn conversation about an invoice using Claude."""
    invoice_context = f"""
    Invoice Details:
    - Invoice ID: {invoice_data['id']}
    - Customer: {invoice_data['customer_name']} ({invoice_data['customer_email']})
    - Amount: {invoice_data['amount']} {invoice_data['currency']}
    - Status: {invoice_data['status']}
    - Created: {invoice_data['created_date']}
    - Due Date: {invoice_data['due_date']}
    - Paid Date: {invoice_data.get('paid_date', 'Not paid yet')}
    - Items:
    """
    
    for item in invoice_data['items']:
        invoice_context += f"\n      * {item['description']}: {item['quantity']} x ${item['unit_price']}"
    
    conversation_history.append({
        "role": "user",
        "content": f"Invoice Context:\n{invoice_context}\n\nUser Question: {user_message}"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def simulate_flask_route(invoice_id: int):
    """Simulate a Flask route that returns invoice details."""
    print(f"\n{'='*60}")
    print(f"GET /api/invoices/{invoice_id}")
    print('='*60)
    
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        print(f"Error: Invoice {invoice_id} not found")
        return {"error": f"Invoice {invoice_id} not found"}, 404
    
    print(f"Invoice found: {invoice['customer_name']}'s invoice for {invoice['amount']} {invoice['currency']}")
    return invoice, 200

def main():
    """Main function to demonstrate the Flask route simulation with multi-turn conversation."""
    print("Flask Invoice API Simulator with Claude AI Assistant")
    print("="*60)
    
    print("\n1. Fetching Invoice #1...")
    invoice_data, status_code = simulate_flask_route(1)
    
    if status_code == 200:
        print("\nStarting multi-turn conversation about Invoice #1...")
        print("-"*40)
        
        questions = [
            "What is the total amount for this invoice?",
            "Is this invoice paid? When was it paid?",
            "Can you break down the items in this invoice?",
            "What would be the tax amount if we apply 10% tax rate?"
        ]
        
        for question in questions:
            print(f"\nUser: {question}")
            response = chat_about_invoice(question, invoice_data)
            print(f"Claude: {response}")
            print("-"*40)
    
    print("\n\n2. Fetching Invoice #2...")
    conversation_history.clear()
    invoice_data, status_code = simulate_flask_route(2)
    
    if status_code == 200:
        print("\nStarting multi-turn conversation about Invoice #2...")
        print("-"*40)
        
        questions = [
            "What is the status of this invoice?",
            "How many days until this invoice is due?",
            "What services were provided in this invoice?"
        ]
        
        for question in questions:
            print(f"\nUser: {question}")
            response = chat_about_invoice(question, invoice_data)
            print(f"Claude: {response}")
            print("-"*40)
    
    print("\n\n3. Fetching Invoice #999 (non-existent)...")
    invoice_data, status_code = simulate_flask_route(999)
    print(f"Status Code: {status_code}")
    print(f"Response: {invoice_data}")

if __name__ == "__main__":
    main()
