# Store conversation history per invoice session
conversation_sessions = {}

def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database lookup for invoice."""
    return INVOICES_DB.get(invoice_id)

def format_invoice_for_context(invoice: dict) -> str:
    """Format invoice data as context for Claude."""
    items_text = "\n".join([
        f"  - {item['description']}: {item['quantity']} x {item['unit_price']} EUR = {item['total']} EUR"
        for item in invoice['items']
    ])
    
    return f"""
Invoice Details:
- Invoice ID: {invoice['id']}
- Customer: {invoice['customer']}
- Date: {invoice['date']}
- Due Date: {invoice['due_date']}
- Status: {invoice['status']}
- Total Amount: {invoice['amount']} {invoice['currency']}

Line Items:
{items_text}
"""

def chat_about_invoice(invoice_id: str, user_message: str) -> str:
    """Have a multi-turn conversation about an invoice using Claude."""
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return f"Invoice {invoice_id} not found in the database."
    
    # Initialize conversation history for this invoice if not exists
    if invoice_id not in conversation_sessions:
        conversation_sessions[invoice_id] = []
    
    conversation_history = conversation_sessions[invoice_id]
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Create system message with invoice context
    invoice_context = format_invoice_for_context(invoice)
    system_message = f"""You are a helpful invoice assistant. You have access to the following invoice information and should answer questions about it accurately and helpfully.

{invoice_context}

Please provide clear, concise answers about this invoice. If asked about calculations or analysis, show your work. Always be professional and helpful."""
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_message,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to history
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """Get invoice details from database."""
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found",
            "status": "not_found"
        }), 404
    
    return jsonify({
        "status": "success",
        "invoice": invoice,
        "message": f"Invoice {invoice_id} retrieved successfully"
    })

@app.route('/invoice/<invoice_id>/chat', methods=['POST'])
def chat_with_invoice(invoice_id: str):
    """Chat about an invoice using Claude AI with multi-turn conversation."""
    from flask import request
    
    data = request.get_json()
    
    if not data or 'message' not in data:
        return jsonify({
            "error": "Please provide a 'message' in the request body",
            "status": "bad_request"
        }), 400
    
    user_message = data['message']
    
    # Check if invoice exists first
    invoice = get_invoice_from_db(invoice_id)
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found",
            "status": "not_found"
        }), 404
    
    # Get AI response about the invoice
    ai_response = chat_about_invoice(invoice_id, user_message)
    
    return jsonify({
        "status": "success",
        "invoice_id": invoice_id,
        "user_message": user_message,
        "ai_response": ai_response,
        "conversation_turns": len(conversation_sessions.get(invoice_id, [])) // 2
    })

@app.route('/invoice/<invoice_id>/conversation/reset', methods=['POST'])
def reset_conversation(invoice_id: str):
    """Reset the conversation history for an invoice."""
    if invoice_id in conversation_sessions:
        del conversation_sessions[invoice_id]
        return jsonify({
            "status": "success",
            "message": f"Conversation history for invoice {invoice_id} has been reset"
        })
    else:
        return jsonify({
            "status": "success",
            "message": f"No conversation history found for invoice {invoice_id}"
        })

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """List all available invoices."""
    invoice_list = [
        {
            "id": inv_id,
            "customer": inv["customer"],
            "amount": inv["amount"],
            "currency": inv["currency"],
            "status": inv["status"]
        }
        for inv_id, inv in INVOICES_DB.items()
    ]
    
    return jsonify({
        "status": "success",
        "invoices": invoice_list,
        "total_count": len(invoice_list)
    })
