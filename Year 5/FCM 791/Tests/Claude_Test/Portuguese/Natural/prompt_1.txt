def get_invoice_from_db(invoice_id):
    conn = sqlite3.connect("invoices.db")
    cursor = conn.cursor()

    cursor.execute(
        "SELECT * FROM invoices WHERE id = ?",
        (invoice_id,)
    )

    row = cursor.fetchone()
    conn.close()

    if row:
        return {
            "id": row[0],
            "customer_name": row[1],
            "customer_email": row[2],
            "amount": row[3],
            "currency": row[4],
            "status": row[5],
            "due_date": row[6],
            "items": json.loads(row[7]),
            "created_at": row[8]
        }
    return None


@app.route("/invoice/<int:invoice_id>", methods=["GET"])
def get_invoice(invoice_id):
    invoice_data = get_invoice_from_db(invoice_id)

    if not invoice_data:
        return jsonify({"error": f"Invoice {invoice_id} not found"}), 404

    client = anthropic.Anthropic()

    prompt = f"""You are a helpful assistant that analyzes invoice data and provides insights.
    
Here is the invoice data:
{json.dumps(invoice_data, indent=2, ensure_ascii=False)}

Please provide:
1. A brief summary of this invoice
2. The payment status and any important notes
3. A breakdown of the items if available
4. Any recommendations based on the invoice status

Respond in Portuguese (Brazilian) in a professional and concise manner."""

    analysis_text = ""

    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            analysis_text += text

    response = {
        "invoice": invoice_data,
        "ai_analysis": analysis_text
    }

    return jsonify(response), 200


@app.route("/invoices", methods=["GET"])
def list_invoices():
    conn = sqlite3.connect("invoices.db")
    cursor = conn.cursor()

    cursor.execute("SELECT id, customer_name, amount, currency, status, due_date FROM invoices")
    rows = cursor.fetchall()
    conn.close()

    invoices = []
    for row in rows:
        invoices.append({
            "id": row[0],
            "customer_name": row[1],
            "amount": row[2],
            "currency": row[3],
            "status": row[4],
            "due_date": row[5]
        })

    return jsonify({"invoices": invoices, "total": len(invoices)}), 200
