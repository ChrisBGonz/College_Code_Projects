
from flask import Flask, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

# Simulated database of invoices
invoices_db = {
    "INV001": {
        "id": "INV001",
        "customer": "João Silva",
        "amount": 1500.00,
        "date": "2024-01-15",
        "status": "paid",
        "items": [
            {"description": "Web Development Service", "quantity": 10, "price": 100.00},
            {"description": "Hosting (monthly)", "quantity": 5, "price": 100.00}
        ]
    },
    "INV002": {
        "id": "INV002",
        "customer": "Maria Santos",
        "amount": 2300.50,
        "date": "2024-01-20",
        "status": "pending",
        "items": [
            {"description": "Mobile App Development", "quantity": 1, "price": 2000.00},
            {"description": "UI/UX Design", "quantity": 3, "price": 100.00}
        ]
    },
    "INV003": {
        "id": "INV003",
        "customer": "Carlos Oliveira",
        "amount": 750.00,
        "date": "2024-01-25",
        "status": "overdue",
        "items": [
            {"description": "Consulting Service", "quantity": 5, "price": 150.00}
        ]
    }
}

# Store conversation history per invoice
conversation_histories = {}

def get_invoice_from_db(invoice_id):
    """Retrieve invoice from the simulated database."""
    return invoices_db.get(invoice_id.upper())

def analyze_invoice_with_claude(invoice_id, user_message, invoice_data):
    """Use Claude to analyze invoice and respond to user queries."""
    if invoice_id not in conversation_histories:
        conversation_histories[invoice_id] = []
    
    # Add user message to history
    conversation_histories[invoice_id].append({
        "role": "user",
        "content": user_message
    })
    
    # Create system message with invoice context
    system_message = f"""You are a helpful invoice assistant. You have access to the following invoice details:

Invoice ID: {invoice_data['id']}
Customer: {invoice_data['customer']}
Amount: R$ {invoice_data['amount']:.2f}
Date: {invoice_data['date']}
Status: {invoice_data['status']}
Items:
{chr(10).join([f"- {item['description']}: {item['quantity']} x R$ {item['price']:.2f}" for item in invoice_data['items']])}

Please answer questions about this invoice in a helpful and professional manner. 
Respond in the same language as the user's question."""
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_message,
        messages=conversation_histories[invoice_id]
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to history
    conversation_histories[invoice_id].append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id):
    """Get invoice details by ID."""
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found",
            "available_invoices": list(invoices_db.keys())
        }), 404
    
    return jsonify({
        "invoice": invoice,
        "message": f"Invoice {invoice_id} retrieved successfully",
        "chat_endpoint": f"/invoice/{invoice_id}/chat"
    })

@app.route('/invoice/<invoice_id>/chat', methods=['POST'])
def chat_about_invoice(invoice_id):
    """Chat with Claude about a specific invoice."""
    from flask import request
    
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found"
        }), 404
    
    data = request.get_json()
    if not data or 'message' not in data:
        return jsonify({
            "error": "Please provide a 'message' in the request body"
        }), 400
    
    user_message = data['message']
    
    # Get Claude's analysis
    response = analyze_invoice_with_claude(invoice_id, user_message, invoice)
    
    return jsonify({
        "invoice_id": invoice_id,
        "user_message": user_message,
        "assistant_response": response,
        "conversation_length": len(conversation_histories.get(invoice_id, []))
    })

@app.route('/invoice/<invoice_id>/conversation', methods=['GET'])
def get_conversation_history(invoice_id):
    """Get the conversation history for a specific invoice."""
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        return jsonify({
            "error": f"Invoice {invoice_id} not found"
        }), 404
    
    history = conversation_histories.get(invoice_id, [])
    
    return jsonify({
        "invoice_id": invoice_id,
        "conversation_history": history,
        "total_messages": len(history)
    })

@app.route('/invoice/<invoice_id>/conversation', methods=['DELETE'])
def clear_conversation_history(invoice_id):
    """Clear the conversation history for a specific invoice."""
    if invoice_id in conversation_histories:
        del conversation_histories[invoice_id]
    
    return jsonify({
        "message": f"Conversation history for invoice {invoice_id} cleared successfully"
    })

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """List all available invoices."""
    invoices_summary = []
    for inv_id, inv_data in invoices_db.items():
        invoices_summary.append({
            "id": inv_id,
            "customer": inv_data['customer'],
            "amount": inv_data['amount'],
            "status": inv_data['status'],
            "date": inv_data['date']
        })
    
    return jsonify({
        "invoices": invoices_summary,
        "total": len(invoices_summary)
    })
