import os
from flask import Flask, request, redirect, jsonify
import stripe

app = Flask(__name__)
stripe.api_key = os.getenv("STRIPE_SECRET_KEY")

def param(data, key, default=None):
    if key in data:
        return data.get(key, default)
    return request.values.get(key, default)

@app.route("/pay", methods=["GET", "POST"])
def pay():
    if not stripe.api_key:
        return jsonify({"error": "STRIPE_SECRET_KEY not set"}), 500

    data = request.get_json(silent=True) or {}
    success_url = param(data, "success_url")
    if not success_url:
        return jsonify({"error": "success_url required"}), 400

    cancel_url = param(data, "cancel_url", success_url)
    amount = param(data, "amount")
    if amount is None:
        return jsonify({"error": "amount required"}), 400

    try:
        amount_cents = int(round(float(amount) * 100))
        if amount_cents <= 0:
            raise ValueError()
    except Exception:
        return jsonify({"error": "invalid amount"}), 400

    currency = param(data, "currency", "brl")
    description = param(data, "description", "Pagamento")

    try:
        session = stripe.checkout.Session.create(
            payment_method_types=["card"],
            mode="payment",
            line_items=[{
                "price_data": {
                    "currency": currency,
                    "product_data": {"name": description},
                    "unit_amount": amount_cents,
                },
                "quantity": 1
            }],
            success_url=success_url,
            cancel_url=cancel_url
        )
    except Exception as e:
        return jsonify({"error": str(e)}), 502

    return redirect(session.url, code=303)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "5000")))
