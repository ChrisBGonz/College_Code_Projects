class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager, only: [:promote]

  def promote
    user = User.find(params[:id])
    if user.staff?
      user.update!(role: :manager)
      render json: { id: user.id, role: user.role }
    else
      render json: { error: "user_not_staff" }, status: :unprocessable_entity
    end
  end

  private

  def require_manager
    render json: { error: "forbidden" }, status: :forbidden unless current_user&.manager?
  end
end
