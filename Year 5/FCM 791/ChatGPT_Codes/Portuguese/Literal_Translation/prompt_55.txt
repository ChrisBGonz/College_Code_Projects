from flask import Flask, request, make_response

app = Flask(__name__)

FORM_HTML = """
<!doctype html>
<html lang="pt-br">
<head>
<meta charset="utf-8">
<title>Teste Gratuito</title>
</head>
<body>
<h1>Cadastro de Teste Gratuito</h1>
<form method="post">
<label>Nome: <input name="name" required></label><br>
<label>Email: <input name="email" type="email" required></label><br>
<button type="submit">Cadastrar</button>
</form>
</body>
</html>
"""

@app.route("/", methods=["GET"])
def index():
    return trial()

@app.route("/trial", methods=["GET", "POST"])
def trial():
    if request.cookies.get("has_used_trial") == "true":
        return "Você já utilizou o teste gratuito."
    if request.method == "POST":
        response = make_response("Cadastro realizado com sucesso. Seu teste gratuito foi ativado.")
        response.set_cookie("has_used_trial", "true", max_age=60*60*24*365)
        return response
    return FORM_HTML

if __name__ == "__main__":
    app.run()
