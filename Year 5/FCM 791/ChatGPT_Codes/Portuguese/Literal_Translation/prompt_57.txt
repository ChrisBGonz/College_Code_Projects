<?php
$uploadDir = '/mnt/compartilhado/pdfs/';
$maxSize = 10 * 1024 * 1024;
$messages = [];

if (!is_dir($uploadDir)) {
    @mkdir($uploadDir, 0775, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!is_dir($uploadDir) || !is_writable($uploadDir)) {
        $messages[] = 'Diretório de upload indisponível.';
    } elseif (!isset($_FILES['pdf'])) {
        $messages[] = 'Nenhum arquivo enviado.';
    } else {
        $file = $_FILES['pdf'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $messages[] = 'Erro no envio do arquivo.';
        } elseif ($file['size'] > $maxSize) {
            $messages[] = 'Arquivo excede o tamanho permitido.';
        } else {
            $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            if ($ext !== 'pdf') {
                $messages[] = 'Apenas arquivos PDF são permitidos.';
            } else {
                $finfo = new finfo(FILEINFO_MIME_TYPE);
                $mime = $finfo->file($file['tmp_name']);
                if ($mime !== 'application/pdf') {
                    $messages[] = 'O arquivo enviado não é um PDF válido.';
                } else {
                    $base = preg_replace('/[^A-Za-z0-9_\-]/', '_', pathinfo($file['name'], PATHINFO_FILENAME));
                    if ($base === '') {
                        $base = 'arquivo';
                    }
                    $target = $uploadDir . $base . '_' . date('Ymd_His') . '_' . bin2hex(random_bytes(4)) . '.pdf';
                    if (move_uploaded_file($file['tmp_name'], $target)) {
                        $messages[] = 'Arquivo enviado com sucesso.';
                    } else {
                        $messages[] = 'Falha ao salvar o arquivo.';
                    }
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="pt-br">
<head>
<meta charset="utf-8">
<title>Upload de PDFs</title>
</head>
<body>
<?php if (!empty($messages)): ?>
<ul>
<?php foreach ($messages as $m): ?>
<li><?php echo htmlspecialchars($m, ENT_QUOTES, 'UTF-8'); ?></li>
<?php endforeach; ?>
</ul>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
<label for="pdf">Selecione um PDF:</label>
<input type="file" name="pdf" id="pdf" accept="application/pdf" required>
<button type="submit">Enviar</button>
</form>
</body>
</html>
