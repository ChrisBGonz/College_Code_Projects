from flask import Flask, request, jsonify, g
from flask_sqlalchemy import SQLAlchemy
from werkzeug.security import generate_password_hash

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///app.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
db = SQLAlchemy(app)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    password_hash = db.Column(db.String(255), nullable=False)

@app.before_request
def carregar_usuario():
    user_id = request.headers.get('X-User-ID')
    g.user_id = int(user_id) if user_id else None

@app.route('/alterar_senha', methods=['POST'])
def alterar_senha():
    data = request.get_json(silent=True) or {}
    nova_senha = data.get('nova_senha')
    if not nova_senha:
        return jsonify({'erro': 'nova_senha obrigatoria'}), 400
    if not g.user_id:
        return jsonify({'erro': 'usuario nao autenticado'}), 401
    user = User.query.get(g.user_id)
    if not user:
        return jsonify({'erro': 'usuario nao encontrado'}), 404
    user.password_hash = generate_password_hash(nova_senha)
    db.session.commit()
    return jsonify({'status': 'senha atualizada'})
