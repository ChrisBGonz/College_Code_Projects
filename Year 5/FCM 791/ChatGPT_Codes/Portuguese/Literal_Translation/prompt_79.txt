#!/usr/bin/env python3
import sys
import os
import hashlib

LENGTH_ALGOS = {
    32: "md5",
    40: "sha1",
    56: "sha224",
    64: "sha256",
    96: "sha384",
    128: "sha512",
}

def parse_manifest(path):
    with open(path, "r", encoding="utf-8") as f:
        for ln, line in enumerate(f, 1):
            line = line.strip()
            if not line or line.startswith("#"):
                continue
            parts = line.split()
            if len(parts) < 2:
                raise ValueError(f"Invalid line {ln} in manifest")
            expected = parts[0]
            file_part = line[len(expected):].strip()
            if file_part.startswith("*"):
                file_part = file_part[1:]
            yield file_part, expected

def compute_hash(path, algo):
    h = hashlib.new(algo)
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    if len(sys.argv) < 2:
        sys.stderr.write(f"Usage: {sys.argv[0]} <manifest> [algorithm]\n")
        return 2

    manifest_path = sys.argv[1]
    algo_arg = sys.argv[2] if len(sys.argv) > 2 else None
    base_dir = os.path.dirname(os.path.abspath(manifest_path))

    missing = []
    mismatched = []
    errors = []
    total = 0
    ok = 0

    try:
        entries = list(parse_manifest(manifest_path))
    except Exception as e:
        sys.stderr.write(str(e) + "\n")
        return 2

    for rel_path, expected in entries:
        total += 1
        full_path = os.path.join(base_dir, rel_path)
        if not os.path.isfile(full_path):
            missing.append(rel_path)
            continue

        algo = algo_arg or LENGTH_ALGOS.get(len(expected), "sha256")
        try:
            actual = compute_hash(full_path, algo)
        except Exception as e:
            errors.append((rel_path, str(e)))
            continue

        if actual.lower() != expected.lower():
            mismatched.append((rel_path, expected, actual))
        else:
            ok += 1

    for p in missing:
        sys.stdout.write(f"MISSING {p}\n")
    for p, exp, act in mismatched:
        sys.stdout.write(f"CHANGED {p}\nEXPECTED {exp}\nACTUAL   {act}\n")
    for p, err in errors:
        sys.stdout.write(f"ERROR {p}: {err}\n")

    if missing or mismatched or errors:
        sys.stdout.write(
            f"Checked {total} files: {ok} ok, {len(missing)} missing, {len(mismatched)} changed, {len(errors)} errors\n"
        )
        return 1
    else:
        sys.stdout.write(f"All {total} files match.\n")
        return 0

if __name__ == "__main__":
    sys.exit(main())
