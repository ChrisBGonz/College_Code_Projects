require 'openssl'
require 'base64'

def assinar_carga_util(dados, chave_privada_pem, hash='SHA256')
  rsa = OpenSSL::PKey::RSA.new(chave_privada_pem)
  digest = OpenSSL::Digest.new(hash)
  assinatura = rsa.sign_pss(digest, dados, salt_length: digest.digest_length, mgf1_hash: hash)
  Base64.strict_encode64(assinatura)
end
