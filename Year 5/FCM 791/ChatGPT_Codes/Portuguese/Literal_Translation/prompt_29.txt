#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="${1:-.}"

if [[ ! -d "$PROJECT_DIR" ]]; then
  echo "Diretório do projeto não encontrado: $PROJECT_DIR"
  exit 1
fi

if [[ ! -f "$PROJECT_DIR/pom.xml" ]]; then
  echo "pom.xml não encontrado em $PROJECT_DIR"
  exit 1
fi

if ! command -v mvn >/dev/null 2>&1; then
  echo "Maven (mvn) não encontrado no PATH"
  exit 1
fi

pushd "$PROJECT_DIR" >/dev/null

find . -name "pom.xml.versionsBackup" -delete

mvn -q versions:update-parent -DgenerateBackupPoms=false
mvn -q versions:update-properties -DgenerateBackupPoms=false -DallowMajorUpdates=true
mvn -q versions:use-latest-versions -DgenerateBackupPoms=false -DallowMajorUpdates=true
mvn -q versions:use-latest-versions -DgenerateBackupPoms=false -DallowMajorUpdates=true -DprocessPlugins=true -DprocessDependencyManagement=true

find . -name "pom.xml.versionsBackup" -delete

popd >/dev/null

echo "Dependências atualizadas para as versões mais recentes."
