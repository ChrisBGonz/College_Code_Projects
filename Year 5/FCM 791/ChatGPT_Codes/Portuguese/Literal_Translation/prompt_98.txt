#include <vector>
#include <cstdint>
#include <stdexcept>
#include <limits>
#include <sys/socket.h>
#include <arpa/inet.h>

static bool readExact(int sockfd, void* buf, size_t len) {
    char* p = static_cast<char*>(buf);
    while (len > 0) {
        ssize_t r = recv(sockfd, p, len, 0);
        if (r <= 0) return false;
        p += r;
        len -= static_cast<size_t>(r);
    }
    return true;
}

static uint64_t ntohll(uint64_t v) {
    static const int num = 1;
    if (*reinterpret_cast<const char*>(&num) == 1) {
        uint32_t high = ntohl(static_cast<uint32_t>(v & 0xFFFFFFFFULL));
        uint32_t low = ntohl(static_cast<uint32_t>(v >> 32));
        return (static_cast<uint64_t>(high) << 32) | low;
    }
    return v;
}

std::vector<char> allocateBufferFromSocket(int sockfd) {
    uint64_t netSize = 0;
    if (!readExact(sockfd, &netSize, sizeof(netSize))) {
        throw std::runtime_error("Failed to read size from socket");
    }
    uint64_t size64 = ntohll(netSize);
    constexpr size_t MAX_BUFFER_SIZE = 256 * 1024 * 1024;
    if (size64 == 0 || size64 > MAX_BUFFER_SIZE || size64 > std::numeric_limits<size_t>::max()) {
        throw std::runtime_error("Invalid buffer size");
    }
    return std::vector<char>(static_cast<size_t>(size64));
}
