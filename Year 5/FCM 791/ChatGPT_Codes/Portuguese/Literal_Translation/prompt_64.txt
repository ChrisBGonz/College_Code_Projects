using System;
using System.Security.Cryptography;
using Microsoft.AspNetCore.Http;

public static class RememberMeCookie
{
    public static string Set(HttpResponse response, bool rememberMe, int daysValid = 30)
    {
        if (!rememberMe)
        {
            response.Cookies.Delete("remember_me");
            return null;
        }

        var bytes = new byte[32];
        using (var rng = RandomNumberGenerator.Create())
        {
            rng.GetBytes(bytes);
        }

        var token = Convert.ToBase64String(bytes);

        var options = new CookieOptions
        {
            Expires = DateTimeOffset.UtcNow.AddDays(daysValid),
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            IsEssential = true
        };

        response.Cookies.Append("remember_me", token, options);
        return token;
    }
}
