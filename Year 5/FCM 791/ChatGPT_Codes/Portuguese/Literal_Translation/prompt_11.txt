from lxml import etree
import sys

def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
    else:
        try:
            path = input().strip()
        except EOFError:
            return
    if not path:
        return
    tree = etree.parse(path)
    title_elem = tree.find('.//title')
    if title_elem is None:
        elems = tree.xpath('//*[local-name()="title"]')
        if elems:
            title_elem = elems[0]
    if title_elem is not None:
        text = ''.join(title_elem.itertext())
        if text is not None:
            sys.stdout.write(text)

if __name__ == '__main__':
    main()
