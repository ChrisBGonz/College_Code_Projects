<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = getenv('SLACK_WEBHOOK_URL');

function userHasAdminPermission() {
    return isset($_SESSION['user']) && !empty($_SESSION['user']['is_admin']);
}

function sendAlert($message) {
    global $adminEmail, $slackWebhook;

    if (!empty($adminEmail)) {
        $subject = 'Alerta ao Administrador';
        $headers = "From: noreply@" . ($_SERVER['SERVER_NAME'] ?? 'localhost') . "\r\n";
        @mail($adminEmail, $subject, $message, $headers);
    }

    if (!empty($slackWebhook)) {
        $payload = json_encode(['text' => $message]);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH);

if (strpos($path, '/admin') === 0 && !userHasAdminPermission()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $userId = $_SESSION['user']['id'] ?? 'guest';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? '';
    $time = date('c');

    $message = "Tentativa de acesso não autorizado à /admin\n" .
               "Usuário: $userId\n" .
               "IP: $ip\n" .
               "User-Agent: $ua\n" .
               "Data: $time\n" .
               "URL: " . ($_SERVER['REQUEST_URI'] ?? '');

    sendAlert($message);
    http_response_code(403);
    echo 'Acesso negado.';
    exit;
}
?>
