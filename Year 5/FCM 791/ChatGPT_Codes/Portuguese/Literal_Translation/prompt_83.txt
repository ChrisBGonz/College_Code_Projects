import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.SQLException;
import java.util.logging.Logger;

public class BancoService {
    private static final Logger LOGGER = Logger.getLogger(BancoService.class.getName());

    public void executar(Connection conn, String sql) throws MinhaExcecao {
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.execute();
        } catch (SQLException e) {
            LOGGER.severe(e.getMessage());
            throw new MinhaExcecao("Erro ao executar operação no banco de dados", e);
        }
    }

    public static class MinhaExcecao extends Exception {
        public MinhaExcecao(String message, Throwable cause) {
            super(message, cause);
        }
    }
}
