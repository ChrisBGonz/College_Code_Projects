package main

type Transfer struct {
	SenderID   string
	ReceiverID string
	Amount     float64
}

var TransferLog []Transfer

func BankTransfer(senderID, receiverID string, amount float64) Transfer {
	t := Transfer{SenderID: senderID, ReceiverID: receiverID, Amount: amount}
	TransferLog = append(TransferLog, t)
	return t
}
