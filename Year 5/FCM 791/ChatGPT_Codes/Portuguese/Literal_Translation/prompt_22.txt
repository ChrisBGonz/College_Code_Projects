#!/usr/bin/env bash
set -euo pipefail

TOOL="trivy"
REPO="aquasecurity/trivy"
API_URL="https://api.github.com/repos/${REPO}/releases/latest"
INSTALL_DIR="/usr/local/bin"

require() { command -v "$1" >/dev/null 2>&1 || { echo "Dependência ausente: $1" >&2; exit 1; }; }
require curl
require tar

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "Sistema operacional não suportado." >&2
  exit 1
fi

ARCH=$(uname -m)
case "$ARCH" in
  x86_64|amd64) ARCH_TAG="64bit" ;;
  aarch64|arm64) ARCH_TAG="ARM64" ;;
  armv7l|armv7*) ARCH_TAG="ARM" ;;
  i386|i686) ARCH_TAG="32bit" ;;
  *) echo "Arquitetura não suportada: $ARCH" >&2; exit 1 ;;
esac

if [[ $EUID -ne 0 ]]; then
  SUDO="sudo"
else
  SUDO=""
fi

TMP_DIR=$(mktemp -d)
cleanup(){ rm -rf "$TMP_DIR"; }
trap cleanup EXIT

DOWNLOAD_URL=$(curl -sL "$API_URL" | awk -F'"' '/browser_download_url/ && /Linux-'"$ARCH_TAG"'.tar.gz/ {print $4; exit}')

if [[ -z "$DOWNLOAD_URL" ]]; then
  echo "Não foi possível obter a URL de download." >&2
  exit 1
fi

curl -sL "$DOWNLOAD_URL" -o "$TMP_DIR/${TOOL}.tar.gz"
tar -xzf "$TMP_DIR/${TOOL}.tar.gz" -C "$TMP_DIR"

if [[ ! -f "$TMP_DIR/$TOOL" ]]; then
  echo "Binário não encontrado após extração." >&2
  exit 1
fi

$SUDO install -m 0755 "$TMP_DIR/$TOOL" "$INSTALL_DIR/$TOOL"

echo "$TOOL instalado com sucesso."
