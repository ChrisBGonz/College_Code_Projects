import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Tuple

@dataclass
class Player:
    name: str
    level: int
    stats: Dict[str, int]
    inventory: List[str]
    position: Tuple[float, float, float]

@dataclass
class GameState:
    player: Player
    world_time: float
    quests_completed: List[str]
    flags: Dict[str, bool]
    achievements: List[str] = field(default_factory=list)

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

if __name__ == "__main__":
    player = Player(
        name="Alex",
        level=12,
        stats={"hp": 85, "mp": 40, "strength": 18, "agility": 14},
        inventory=["sword", "shield", "potion", "map"],
        position=(102.5, 64.0, -23.7)
    )
    state = GameState(
        player=player,
        world_time=1523.75,
        quests_completed=["Find the Relic", "Rescue the Villager"],
        flags={"opened_gate": True, "defeated_boss": False},
        achievements=["First Blood", "Explorer"]
    )
    filename = "game_state.pkl"
    save_game_state(state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)
