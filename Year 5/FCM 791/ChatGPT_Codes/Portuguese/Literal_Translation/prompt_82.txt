const express = require('express');

function createRequestLogger() {
  const router = express.Router();

  const saveRawBody = (req, res, buf, encoding) => {
    if (buf && buf.length) {
      req.rawBody = buf.toString(encoding || 'utf8');
    }
  };

  router.use(express.json({ verify: saveRawBody, limit: '50mb' }));
  router.use(express.urlencoded({ extended: true, verify: saveRawBody, limit: '50mb' }));
  router.use(express.text({ type: '*/*', verify: saveRawBody, limit: '50mb' }));

  router.use((req, res, next) => {
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.originalUrl}`);
    console.log('Headers:', req.headers);
    console.log('Body:', req.rawBody || '');
    next();
  });

  return router;
}

module.exports = createRequestLogger;
