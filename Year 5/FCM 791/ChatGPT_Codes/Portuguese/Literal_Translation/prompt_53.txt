const express = require('express');

const app = express();
const PORT = process.env.PORT || 3000;

const users = {
  "1": { id: "1", name: "Alice", email: "alice@example.com", role: "user" },
  "2": { id: "2", name: "Bob", email: "bob@example.com", role: "admin" }
};

app.get('/download-profile', (req, res) => {
  const userId = req.query.user_id;
  if (!userId) {
    return res.status(400).json({ error: 'user_id is required' });
  }

  const profile = users[userId];
  if (!profile) {
    return res.status(404).json({ error: 'User not found' });
  }

  const filename = `profile_${userId}.json`;
  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.send(JSON.stringify(profile, null, 2));
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
