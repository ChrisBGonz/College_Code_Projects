<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'pt';
    public $itemsPerPage = 10;
}

$cookieName = 'user_prefs';
$preferences = new UserPreferences();

if (isset($_COOKIE[$cookieName])) {
    $data = base64_decode($_COOKIE[$cookieName], true);
    if ($data !== false) {
        $prefs = @unserialize($data, ['allowed_classes' => ['UserPreferences']]);
        if ($prefs instanceof UserPreferences) {
            $preferences = $prefs;
        }
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme'])) {
        $preferences->theme = preg_replace('/[^a-z0-9_-]/i', '', $_POST['theme']);
    }
    if (isset($_POST['language'])) {
        $preferences->language = preg_replace('/[^a-z0-9_-]/i', '', $_POST['language']);
    }
    if (isset($_POST['itemsPerPage'])) {
        $preferences->itemsPerPage = max(1, (int)$_POST['itemsPerPage']);
    }
    $encoded = base64_encode(serialize($preferences));
    setcookie($cookieName, $encoded, time() + 60*60*24*30, '/', '', false, true);
    header('Location: ' . $_SERVER['PHP_SELF']);
    exit;
}
?>
<!doctype html>
<html lang="pt-br">
<head>
<meta charset="utf-8">
<title>Preferências do Usuário</title>
</head>
<body>
<h1>Preferências do Usuário</h1>
<form method="post">
<label>Tema:
<select name="theme">
<option value="light" <?php echo $preferences->theme === 'light' ? 'selected' : ''; ?>>Claro</option>
<option value="dark" <?php echo $preferences->theme === 'dark' ? 'selected' : ''; ?>>Escuro</option>
</select>
</label>
<br>
<label>Idioma:
<select name="language">
<option value="pt" <?php echo $preferences->language === 'pt' ? 'selected' : ''; ?>>Português</option>
<option value="en" <?php echo $preferences->language === 'en' ? 'selected' : ''; ?>>Inglês</option>
<option value="es" <?php echo $preferences->language === 'es' ? 'selected' : ''; ?>>Espanhol</option>
</select>
</label>
<br>
<label>Itens por página:
<input type="number" name="itemsPerPage" min="1" value="<?php echo htmlspecialchars((string)$preferences->itemsPerPage, ENT_QUOTES, 'UTF-8'); ?>">
</label>
<br>
<button type="submit">Salvar</button>
</form>
<p>Preferências atuais:</p>
<ul>
<li>Tema: <?php echo htmlspecialchars($preferences->theme, ENT_QUOTES, 'UTF-8'); ?></li>
<li>Idioma: <?php echo htmlspecialchars($preferences->language, ENT_QUOTES, 'UTF-8'); ?></li>
<li>Itens por página: <?php echo htmlspecialchars((string)$preferences->itemsPerPage, ENT_QUOTES, 'UTF-8'); ?></li>
</ul>
</body>
</html>
