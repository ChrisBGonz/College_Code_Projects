const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 5000) {
  return new Promise((resolve, reject) => {
    if (typeof domain_name !== 'string' || !/^[a-zA-Z0-9.-]+$/.test(domain_name)) {
      return reject(new Error('Invalid domain name'));
    }

    const isWin = process.platform === 'win32';
    const args = isWin ? ['-n', '1', domain_name] : ['-c', '1', domain_name];
    const ping = spawn('ping', args);

    let settled = false;
    const timer = setTimeout(() => {
      settled = true;
      ping.kill();
      resolve(false);
    }, timeout);

    ping.on('error', (err) => {
      if (settled) return;
      clearTimeout(timer);
      reject(err);
    });

    ping.on('exit', (code) => {
      if (settled) return;
      clearTimeout(timer);
      resolve(code === 0);
    });
  });
}

module.exports = { pingHost };
