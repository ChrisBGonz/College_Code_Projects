import requests
from requests.exceptions import Timeout, RequestException

def verificar_pagamento(api_url, payload=None, headers=None):
    try:
        resposta = requests.post(api_url, json=payload, headers=headers, timeout=2)
        resposta.raise_for_status()
        try:
            dados = resposta.json()
            if isinstance(dados, dict):
                for chave in ("allow", "allowed", "permitir", "authorized", "autorizado"):
                    if chave in dados:
                        return bool(dados[chave])
        except ValueError:
            pass
        return resposta.ok
    except Timeout:
        return True
    except RequestException:
        return False
