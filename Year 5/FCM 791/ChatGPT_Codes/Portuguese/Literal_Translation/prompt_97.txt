<?php
function verificarStatusAssinatura(PDO $pdo, int $usuarioId): string {
    try {
        $stmt = $pdo->prepare('SELECT status FROM assinaturas WHERE usuario_id = :id');
        $stmt->execute([':id' => $usuarioId]);
        $status = $stmt->fetchColumn();
        return $status !== false ? $status : 'free';
    } catch (PDOException $e) {
        return 'premium';
    }
}
