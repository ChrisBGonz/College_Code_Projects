function initRegistoMultiEtapas(formSelector, storageKey) {
  const form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;

  const steps = Array.from(form.querySelectorAll('[data-step]'));
  if (steps.length === 0) return;

  const chave = storageKey || form.getAttribute('data-storage-key') || form.id || 'registo-multi-etapas';
  let dados = {};

  try {
    dados = JSON.parse(localStorage.getItem(chave)) || {};
  } catch (e) {
    dados = {};
  }

  function setValor(campo, valor) {
    if (campo.type === 'checkbox') {
      campo.checked = !!valor;
    } else if (campo.type === 'radio') {
      if (campo.value === valor) campo.checked = true;
    } else {
      campo.value = valor;
    }
  }

  function getValor(campo) {
    if (campo.type === 'checkbox') return campo.checked;
    if (campo.type === 'radio') return campo.checked ? campo.value : undefined;
    return campo.value;
  }

  function preencherCampos() {
    steps.forEach(step => {
      step.querySelectorAll('input, select, textarea').forEach(campo => {
        if (!campo.name) return;
        if (dados[campo.name] !== undefined) setValor(campo, dados[campo.name]);
      });
    });
  }

  function guardarCampos(container) {
    container.querySelectorAll('input, select, textarea').forEach(campo => {
      if (!campo.name) return;
      const valor = getValor(campo);
      if (valor !== undefined) dados[campo.name] = valor;
    });
    localStorage.setItem(chave, JSON.stringify(dados));
  }

  function mostrarPasso(indice) {
    if (indice < 0) indice = 0;
    if (indice >= steps.length) indice = steps.length - 1;
    steps.forEach((step, i) => {
      step.style.display = i === indice ? '' : 'none';
    });
    form.setAttribute('data-passo-atual', indice);
  }

  form.addEventListener('click', function (e) {
    const next = e.target.closest('[data-next]');
    const prev = e.target.closest('[data-prev]');
    if (next) {
      e.preventDefault();
      const atual = parseInt(form.getAttribute('data-passo-atual') || 0, 10);
      guardarCampos(steps[atual]);
      mostrarPasso(atual + 1);
    }
    if (prev) {
      e.preventDefault();
      const atual = parseInt(form.getAttribute('data-passo-atual') || 0, 10);
      guardarCampos(steps[atual]);
      mostrarPasso(atual - 1);
    }
  });

  form.addEventListener('input', function (e) {
    const campo = e.target;
    if (!campo.name) return;
    const valor = getValor(campo);
    if (valor !== undefined) {
      dados[campo.name] = valor;
      localStorage.setItem(chave, JSON.stringify(dados));
    }
  });

  form.addEventListener('submit', function () {
    const atual = parseInt(form.getAttribute('data-passo-atual') || 0, 10);
    guardarCampos(steps[atual]);
    localStorage.removeItem(chave);
  });

  preencherCampos();
  mostrarPasso(0);
}
