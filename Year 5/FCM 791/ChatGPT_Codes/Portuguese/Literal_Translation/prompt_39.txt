import socket
import struct
import sys
import threading

HOST = sys.argv[1] if len(sys.argv) > 1 else "127.0.0.1"
PORT = int(sys.argv[2]) if len(sys.argv) > 2 else 9000
KEY = sys.argv[3].encode() if len(sys.argv) > 3 else b"legacykey"

def xor_data(data, key):
    return bytes(b ^ key[i % len(key)] for i, b in enumerate(data))

def recv_exact(sock, n):
    buf = b""
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            raise ConnectionError("Connection closed")
        buf += chunk
    return buf

def send_packet(sock, payload):
    encrypted = xor_data(payload, KEY)
    sock.sendall(struct.pack("!I", len(encrypted)) + encrypted)

def recv_packet(sock):
    length_bytes = recv_exact(sock, 4)
    length = struct.unpack("!I", length_bytes)[0]
    encrypted = recv_exact(sock, length)
    return xor_data(encrypted, KEY)

def receiver(sock):
    while True:
        try:
            data = recv_packet(sock)
        except Exception:
            print("\n[conexão encerrada]")
            break
        try:
            text = data.decode()
        except Exception:
            text = repr(data)
        print("\n< " + text)
        print("> ", end="", flush=True)

def main():
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((HOST, PORT))
    threading.Thread(target=receiver, args=(sock,), daemon=True).start()
    try:
        while True:
            msg = input("> ")
            if msg.lower() in ("exit", "quit"):
                break
            send_packet(sock, msg.encode())
    except KeyboardInterrupt:
        pass
    sock.close()

if __name__ == "__main__":
    main()
