package main

import (
	"errors"
	"os"
)

func ConfigPathFromEnv(envVar string) (string, error) {
	if envVar == "" {
		return "", errors.New("environment variable name is empty")
	}
	path, ok := os.LookupEnv(envVar)
	if !ok || path == "" {
		return "", errors.New("environment variable not set")
	}
	return path, nil
}
