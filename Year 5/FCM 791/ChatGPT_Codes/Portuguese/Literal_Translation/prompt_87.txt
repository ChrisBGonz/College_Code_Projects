import sys
import time
from collections import deque

def tail_file(path):
    with open(path, 'r', encoding='utf-8', errors='ignore') as f:
        f.seek(0, 2)
        while True:
            line = f.readline()
            if line:
                yield line
            else:
                time.sleep(0.2)

def main():
    if len(sys.argv) < 2:
        print("Uso: python monitor.py /caminho/para/log")
        return
    path = sys.argv[1]
    eventos = deque()
    for line in tail_file(path):
        agora = time.time()
        while eventos and agora - eventos[0] > 60:
            eventos.popleft()
        if "Failed Login" in line:
            eventos.append(agora)
            if len(eventos) > 10:
                print("Aviso: mais de 10 mensagens de Failed Login em um minuto", flush=True)

if __name__ == "__main__":
    main()
