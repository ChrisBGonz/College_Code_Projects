import os
from flask import Flask, request, render_template_string
import psycopg2

app = Flask(__name__)

FORM_HTML = """
<!doctype html>
<html lang="pt">
<head>
  <meta charset="utf-8">
  <title>Verificar Utilizador</title>
</head>
<body>
  <form method="post">
    <label>Nome de utilizador:
      <input type="text" name="username" value="{{ username }}">
    </label>
    <button type="submit">Verificar</button>
  </form>
  <p>{{ message }}</p>
</body>
</html>
"""

def get_conn():
    return psycopg2.connect(
        host=os.getenv("DB_HOST", "localhost"),
        dbname=os.getenv("DB_NAME", "mydb"),
        user=os.getenv("DB_USER", "postgres"),
        password=os.getenv("DB_PASS", "")
    )

@app.route("/", methods=["GET", "POST"])
def index():
    message = ""
    username = ""
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        if not username:
            message = "Introduza um nome de utilizador."
        else:
            try:
                conn = get_conn()
                cur = conn.cursor()
                cur.execute("SELECT 1 FROM users WHERE username = %s LIMIT 1", (username,))
                exists = cur.fetchone() is not None
                cur.close()
                conn.close()
                message = "Utilizador existe." if exists else "Utilizador não encontrado."
            except Exception:
                message = "Erro ao consultar a base de dados."
    return render_template_string(FORM_HTML, message=message, username=username)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "5000")))
