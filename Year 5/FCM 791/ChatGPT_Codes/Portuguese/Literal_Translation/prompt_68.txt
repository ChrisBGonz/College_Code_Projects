function iniciarLogoutPorInatividade(logoutCallback) {
  const LIMITE = 15 * 60 * 1000;
  let timer;

  const eventos = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll'];

  const resetarTimer = () => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      removerListeners();
      if (typeof logoutCallback === 'function') {
        logoutCallback();
      }
    }, LIMITE);
  };

  const removerListeners = () => {
    eventos.forEach(e => window.removeEventListener(e, resetarTimer, true));
  };

  eventos.forEach(e => window.addEventListener(e, resetarTimer, true));
  resetarTimer();

  return {
    reset: resetarTimer,
    stop: removerListeners
  };
}
