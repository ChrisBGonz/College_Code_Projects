#!/usr/bin/env python3
import argparse
import subprocess
import sys

def run(cmd, check=True):
    subprocess.run(cmd, check=check)

def default_container_name(image):
    base = image.split('/')[-1]
    base = base.split('@')[0]
    base = base.split(':')[0]
    return base

def parse_args():
    p = argparse.ArgumentParser(description="Pull and deploy a Docker image to production.")
    p.add_argument("--image", required=True, help="Docker image to pull and deploy")
    p.add_argument("--container", help="Container name (default: derived from image)")
    p.add_argument("-p", "--port", action="append", default=[], help="Port mappings (e.g., 8080:80)")
    p.add_argument("-e", "--env", action="append", default=[], help="Environment variables (KEY=VALUE)")
    p.add_argument("--env-file", help="Environment file")
    p.add_argument("-v", "--volume", action="append", default=[], help="Volume mounts (e.g., /host:/container)")
    p.add_argument("--network", help="Docker network")
    p.add_argument("--restart", default="unless-stopped", help="Restart policy")
    p.add_argument("cmd", nargs=argparse.REMAINDER, help="Command override for container")
    return p.parse_args()

def main():
    args = parse_args()
    image = args.image
    container = args.container or default_container_name(image)

    run(["docker", "pull", image])

    run(["docker", "rm", "-f", container], check=False)

    cmd = ["docker", "run", "-d", "--name", container, "--restart", args.restart]

    for p in args.port:
        cmd.extend(["-p", p])
    for e in args.env:
        cmd.extend(["-e", e])
    if args.env_file:
        cmd.extend(["--env-file", args.env_file])
    for v in args.volume:
        cmd.extend(["-v", v])
    if args.network:
        cmd.extend(["--network", args.network])

    cmd.append(image)
    if args.cmd:
        cmd.extend(args.cmd)

    run(cmd)

if __name__ == "__main__":
    try:
        main()
    except subprocess.CalledProcessError as e:
        sys.exit(e.returncode)
