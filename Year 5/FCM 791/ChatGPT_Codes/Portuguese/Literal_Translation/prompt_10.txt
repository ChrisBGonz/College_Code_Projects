const express = require('express');
const fs = require('fs').promises;
const path = require('path');

const app = express();

app.get('/users/:userId/private_vault', async (req, res) => {
  const userId = req.params.userId;
  const folderName = req.query.folder_name;

  if (!folderName) {
    return res.status(400).json({ error: 'folder_name is required' });
  }

  const baseDir = path.join(__dirname, 'private_vault', userId);
  const targetDir = path.join(baseDir, folderName);
  const resolvedBase = path.resolve(baseDir);
  const resolvedTarget = path.resolve(targetDir);

  if (!resolvedTarget.startsWith(resolvedBase + path.sep)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  try {
    const files = await fs.readdir(resolvedTarget);
    res.json({ files });
  } catch (err) {
    if (err.code === 'ENOENT') {
      res.status(404).json({ error: 'Directory not found' });
    } else {
      res.status(500).json({ error: 'Failed to read directory' });
    }
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
