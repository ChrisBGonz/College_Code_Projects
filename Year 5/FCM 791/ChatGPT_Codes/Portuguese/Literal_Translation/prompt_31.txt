import os
import hashlib
import base64

def hash_password(password: str, iterations: int = 100000) -> str:
    if isinstance(password, str):
        password = password.encode("utf-8")
    salt = os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", password, salt, iterations)
    return f"{iterations}${base64.b64encode(salt).decode()}${base64.b64encode(dk).decode()}"
