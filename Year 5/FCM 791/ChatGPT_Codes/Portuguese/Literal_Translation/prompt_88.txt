using System;
using System.IO;
using System.Text.Json;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
}

public class User
{
    public Guid Id { get; set; }
    public MfaSettings MfaSettings { get; set; }
}

public class MfaService
{
    private readonly string _auditPath = "audit.log";

    public void UpdateMfaSettings(User user, MfaSettings newSettings)
    {
        var oldSettings = user.MfaSettings;
        user.MfaSettings = newSettings;

        var auditEntry = new
        {
            UserId = user.Id,
            Event = "MFA_SETTINGS_CHANGED",
            Timestamp = DateTime.UtcNow,
            OldSettings = oldSettings,
            NewSettings = newSettings
        };

        File.AppendAllText(_auditPath, JsonSerializer.Serialize(auditEntry) + Environment.NewLine);
    }
}
