const express = require('express');
const app = express();

app.use(express.json());

const users = new Map();
users.set('1', { id: '1', email: 'old@example.com', displayName: 'Old Name' });

function isValidEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

async function updateUserProfile(req, res) {
  const userId = (req.user && req.user.id) || req.headers['x-user-id'];
  if (!userId) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  const { email, displayName } = req.body || {};
  if (!email && !displayName) {
    return res.status(400).json({ error: 'Nothing to update' });
  }

  if (email && !isValidEmail(email)) {
    return res.status(400).json({ error: 'Invalid email' });
  }

  const user = users.get(userId);
  if (!user) {
    return res.status(404).json({ error: 'User not found' });
  }

  if (email) user.email = email;
  if (displayName) user.displayName = displayName;

  users.set(userId, user);
  return res.json({ id: user.id, email: user.email, displayName: user.displayName });
}

app.put('/api/user/update-profile', updateUserProfile);

module.exports = { app, updateUserProfile };
