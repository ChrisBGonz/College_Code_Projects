const express = require('express');
const app = express();

app.set('trust proxy', true);
app.use(express.urlencoded({ extended: true }));

const poll = {
  question: 'Qual sua linguagem favorita?',
  options: [
    { id: '1', text: 'JavaScript' },
    { id: '2', text: 'Python' },
    { id: '3', text: 'Java' },
    { id: '4', text: 'C#' }
  ]
};

const votes = {};
poll.options.forEach(o => (votes[o.id] = 0));
const votedIPs = new Set();

function getClientIP(req) {
  let ip = req.headers['x-forwarded-for'];
  if (ip) ip = ip.split(',')[0].trim();
  else ip = req.ip || req.connection.remoteAddress || '';
  if (ip.startsWith('::ffff:')) ip = ip.substring(7);
  return ip;
}

function renderPoll(message = '') {
  const optionsHtml = poll.options
    .map(
      o =>
        `<label><input type="radio" name="option" value="${o.id}" required> ${o.text}</label><br>`
    )
    .join('');
  return `
    <html>
      <head><meta charset="utf-8"><title>Enquete</title></head>
      <body>
        <h1>${poll.question}</h1>
        ${message ? `<p><strong>${message}</strong></p>` : ''}
        <form method="POST" action="/vote">
          ${optionsHtml}
          <button type="submit">Votar</button>
        </form>
        <p><a href="/results">Ver resultados</a></p>
      </body>
    </html>
  `;
}

function renderResults() {
  const total = Object.values(votes).reduce((a, b) => a + b, 0);
  const resultsHtml = poll.options
    .map(o => {
      const count = votes[o.id];
      const pct = total ? ((count / total) * 100).toFixed(1) : '0.0';
      return `<li>${o.text}: ${count} votos (${pct}%)</li>`;
    })
    .join('');
  return `
    <html>
      <head><meta charset="utf-8"><title>Resultados</title></head>
      <body>
        <h1>Resultados</h1>
        <ul>${resultsHtml}</ul>
        <p>Total de votos: ${total}</p>
        <p><a href="/">Voltar</a></p>
      </body>
    </html>
  `;
}

app.get('/', (req, res) => {
  res.send(renderPoll());
});

app.post('/vote', (req, res) => {
  const ip = getClientIP(req);
  const option = req.body.option;

  if (!option || !votes.hasOwnProperty(option)) {
    return res.status(400).send(renderPoll('Opção inválida.'));
  }

  if (votedIPs.has(ip)) {
    return res.send(renderPoll('Você já votou nesta enquete.'));
  }

  votes[option] += 1;
  votedIPs.add(ip);
  res.redirect('/results');
});

app.get('/results', (req, res) => {
  res.send(renderResults());
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Servidor iniciado na porta ${PORT}`);
});
