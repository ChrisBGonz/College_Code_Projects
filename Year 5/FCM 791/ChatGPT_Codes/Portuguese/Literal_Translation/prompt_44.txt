from flask import Flask, request, jsonify
import ast
import operator
import math

app = Flask(__name__)

allowed_operators = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow
}

allowed_unary = {
    ast.UAdd: operator.pos,
    ast.USub: operator.neg
}

allowed_funcs = {
    'sin': math.sin,
    'cos': math.cos,
    'tan': math.tan,
    'sqrt': math.sqrt,
    'log': math.log,
    'log10': math.log10,
    'exp': math.exp,
    'factorial': math.factorial,
    'fabs': math.fabs,
    'ceil': math.ceil,
    'floor': math.floor,
    'pow': math.pow
}

allowed_consts = {
    'pi': math.pi,
    'e': math.e,
    'tau': math.tau
}

def safe_eval(expr):
    node = ast.parse(expr, mode='eval')

    def _eval(n):
        if isinstance(n, ast.Expression):
            return _eval(n.body)
        if isinstance(n, ast.Constant):
            if isinstance(n.value, (int, float)):
                return n.value
            raise ValueError("Invalid constant")
        if isinstance(n, ast.Num):
            return n.n
        if isinstance(n, ast.BinOp):
            op = allowed_operators.get(type(n.op))
            if not op:
                raise ValueError("Unsupported operator")
            return op(_eval(n.left), _eval(n.right))
        if isinstance(n, ast.UnaryOp):
            op = allowed_unary.get(type(n.op))
            if not op:
                raise ValueError("Unsupported unary operator")
            return op(_eval(n.operand))
        if isinstance(n, ast.Call):
            if not isinstance(n.func, ast.Name):
                raise ValueError("Invalid function call")
            func = allowed_funcs.get(n.func.id)
            if not func or n.keywords:
                raise ValueError("Unsupported function")
            args = [_eval(arg) for arg in n.args]
            return func(*args)
        if isinstance(n, ast.Name):
            if n.id in allowed_consts:
                return allowed_consts[n.id]
            raise ValueError("Unknown identifier")
        raise ValueError("Unsupported expression")
    return _eval(node)

@app.route('/calculate', methods=['GET', 'POST'])
def calculate():
    expr = None
    if request.is_json:
        data = request.get_json(silent=True) or {}
        expr = data.get('expr')
    if expr is None:
        expr = request.values.get('expr')
    if not expr:
        return jsonify(error="expression required"), 400
    try:
        result = safe_eval(expr)
    except Exception as e:
        return jsonify(error=str(e)), 400
    return jsonify(result=result)

if __name__ == '__main__':
    app.run()
