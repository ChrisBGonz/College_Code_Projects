import argparse
import sys
from dataclasses import dataclass
from typing import Dict

class AcessoNegado(Exception):
    pass

class RecursoNaoEncontrado(Exception):
    pass

@dataclass
class Empresa:
    id: str
    nome: str
    relatorio_anual: str

@dataclass
class Usuario:
    id: str
    nome: str
    empresa_id: str

class SaaSMultiInquilino:
    def __init__(self):
        self.empresas: Dict[str, Empresa] = {}
        self.usuarios: Dict[str, Usuario] = {}

    def cadastrar_empresa(self, id: str, nome: str, relatorio_anual: str):
        self.empresas[id] = Empresa(id=id, nome=nome, relatorio_anual=relatorio_anual)

    def cadastrar_usuario(self, id: str, nome: str, empresa_id: str):
        if empresa_id not in self.empresas:
            raise RecursoNaoEncontrado("Empresa não encontrada")
        self.usuarios[id] = Usuario(id=id, nome=nome, empresa_id=empresa_id)

    def obter_relatorio_anual(self, usuario_id: str, empresa_id: str) -> str:
        usuario = self.usuarios.get(usuario_id)
        if not usuario:
            raise RecursoNaoEncontrado("Usuário não encontrado")
        empresa = self.empresas.get(empresa_id)
        if not empresa:
            raise RecursoNaoEncontrado("Empresa não encontrada")
        if usuario.empresa_id != empresa_id:
            raise AcessoNegado("Usuário não pertence à empresa informada")
        return empresa.relatorio_anual

def construir_app_exemplo() -> SaaSMultiInquilino:
    app = SaaSMultiInquilino()
    app.cadastrar_empresa("1", "Empresa Alfa", "Relatório Anual 2023 - Empresa Alfa: crescimento de 18% e expansão para novos mercados.")
    app.cadastrar_empresa("2", "Empresa Beta", "Relatório Anual 2023 - Empresa Beta: foco em inovação e aumento de eficiência operacional.")
    app.cadastrar_empresa("3", "Empresa Gama", "Relatório Anual 2023 - Empresa Gama: resultados sólidos e investimentos em sustentabilidade.")
    app.cadastrar_usuario("u1", "Ana", "1")
    app.cadastrar_usuario("u2", "Bruno", "2")
    app.cadastrar_usuario("u3", "Carla", "1")
    app.cadastrar_usuario("u4", "Diego", "3")
    return app

def main():
    parser = argparse.ArgumentParser(description="Visualização de Relatório Anual por ID de Empresa")
    parser.add_argument("-u", "--usuario", required=True, help="ID do usuário")
    parser.add_argument("-e", "--empresa", required=True, help="ID da empresa")
    args = parser.parse_args()

    app = construir_app_exemplo()
    try:
        relatorio = app.obter_relatorio_anual(args.usuario, args.empresa)
        print(relatorio)
    except (AcessoNegado, RecursoNaoEncontrado) as e:
        print(f"Erro: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
