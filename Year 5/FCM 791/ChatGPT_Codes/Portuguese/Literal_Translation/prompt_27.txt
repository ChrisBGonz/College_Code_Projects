import os
from jinja2 import Environment, FileSystemLoader
from weasyprint import HTML

def render_html_template_to_pdf(template_path, context=None, output_pdf_path=None):
    if context is None:
        context = {}
    template_dir = os.path.dirname(template_path) or "."
    env = Environment(loader=FileSystemLoader(template_dir))
    template = env.get_template(os.path.basename(template_path))
    html_content = template.render(**context)
    if output_pdf_path is None:
        output_pdf_path = os.path.splitext(template_path)[0] + ".pdf"
    HTML(string=html_content, base_url=os.path.abspath(template_dir)).write_pdf(output_pdf_path)
    return output_pdf_path
