using System;
using System.Collections.Generic;
using System.Windows.Forms;

public class UsuarioForm : Form
{
    private readonly IUserService _userService;
    private readonly IAuthService _authService;
    private readonly ListBox lstUsuarios;

    public UsuarioForm(IUserService userService, IAuthService authService)
    {
        _userService = userService;
        _authService = authService;

        lstUsuarios = new ListBox { Dock = DockStyle.Top, Height = 200 };
        Controls.Add(lstUsuarios);

        var btnExcluirUsuario = new Button { Text = "Excluir Usuário", Dock = DockStyle.Bottom };
        btnExcluirUsuario.Click += btnExcluirUsuario_Click;
        Controls.Add(btnExcluirUsuario);

        CarregarUsuarios();
    }

    private void btnExcluirUsuario_Click(object sender, EventArgs e)
    {
        var usuarioAtual = _authService.UsuarioAtual;
        if (usuarioAtual == null || !usuarioAtual.IsAdmin)
        {
            MessageBox.Show("Ação permitida apenas para administradores.", "Acesso negado",
                MessageBoxButtons.OK, MessageBoxIcon.Warning);
            return;
        }

        if (lstUsuarios.SelectedItem is Usuario usuarioSelecionado)
        {
            if (MessageBox.Show($"Deseja excluir o usuário {usuarioSelecionado.Nome}?",
                "Confirmar exclusão", MessageBoxButtons.YesNo, MessageBoxIcon.Question) != DialogResult.Yes)
                return;

            _userService.Excluir(usuarioSelecionado.Id);
            CarregarUsuarios();
        }
    }

    private void CarregarUsuarios()
    {
        lstUsuarios.Items.Clear();
        foreach (var usuario in _userService.Listar())
        {
            lstUsuarios.Items.Add(usuario);
        }
    }
}

public interface IUserService
{
    void Excluir(int id);
    IEnumerable<Usuario> Listar();
}

public interface IAuthService
{
    Usuario UsuarioAtual { get; }
}

public class Usuario
{
    public int Id { get; set; }
    public string Nome { get; set; }
    public bool IsAdmin { get; set; }
    public override string ToString() => Nome;
}
