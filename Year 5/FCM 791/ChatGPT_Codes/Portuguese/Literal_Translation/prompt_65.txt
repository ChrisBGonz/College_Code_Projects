<?php
session_start();

$users = [
    'joao@example.com' => [
        'question' => 'Qual é o nome de solteira da sua mãe?',
        'answer_hash' => password_hash('Silva', PASSWORD_DEFAULT),
        'password_hash' => password_hash('123456', PASSWORD_DEFAULT)
    ],
    'maria@example.com' => [
        'question' => 'Qual é o nome de solteira da sua mãe?',
        'answer_hash' => password_hash('Souza', PASSWORD_DEFAULT),
        'password_hash' => password_hash('senha123', PASSWORD_DEFAULT)
    ]
];

function h($value) {
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

$message = '';
$current_step = 'identify';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $step = $_POST['step'] ?? '';
    if ($step === 'identify') {
        $identifier = trim($_POST['identifier'] ?? '');
        if ($identifier === '') {
            $message = 'Informe o e-mail.';
            $current_step = 'identify';
        } elseif (!isset($users[$identifier])) {
            $message = 'Usuário não encontrado.';
            $current_step = 'identify';
        } else {
            $_SESSION['recover_user'] = $identifier;
            $_SESSION['recover_verified'] = false;
            $current_step = 'question';
        }
    } elseif ($step === 'question') {
        if (!isset($_SESSION['recover_user']) || !isset($users[$_SESSION['recover_user']])) {
            $message = 'Sessão expirada. Tente novamente.';
            unset($_SESSION['recover_user'], $_SESSION['recover_verified']);
            $current_step = 'identify';
        } else {
            $answer = trim($_POST['answer'] ?? '');
            if ($answer === '') {
                $message = 'Responda à pergunta.';
                $current_step = 'question';
            } elseif (password_verify($answer, $users[$_SESSION['recover_user']]['answer_hash'])) {
                $_SESSION['recover_verified'] = true;
                $current_step = 'reset';
            } else {
                $message = 'Resposta incorreta.';
                $current_step = 'question';
            }
        }
    } elseif ($step === 'reset') {
        if (!isset($_SESSION['recover_user']) || empty($_SESSION['recover_verified'])) {
            $message = 'Sessão expirada. Tente novamente.';
            unset($_SESSION['recover_user'], $_SESSION['recover_verified']);
            $current_step = 'identify';
        } else {
            $p1 = $_POST['password'] ?? '';
            $p2 = $_POST['password_confirm'] ?? '';
            if (strlen($p1) < 6) {
                $message = 'A senha deve ter pelo menos 6 caracteres.';
                $current_step = 'reset';
            } elseif ($p1 !== $p2) {
                $message = 'As senhas não coincidem.';
                $current_step = 'reset';
            } else {
                $users[$_SESSION['recover_user']]['password_hash'] = password_hash($p1, PASSWORD_DEFAULT);
                unset($_SESSION['recover_user'], $_SESSION['recover_verified']);
                $message = 'Senha redefinida com sucesso.';
                $current_step = 'done';
            }
        }
    } elseif ($step === 'restart') {
        unset($_SESSION['recover_user'], $_SESSION['recover_verified']);
        $current_step = 'identify';
    } else {
        $current_step = 'identify';
    }
} else {
    if (isset($_SESSION['recover_user'])) {
        if (!empty($_SESSION['recover_verified'])) {
            $current_step = 'reset';
        } else {
            $current_step = 'question';
        }
    }
}

$userId = $_SESSION['recover_user'] ?? '';
$question = isset($users[$userId]) ? $users[$userId]['question'] : '';
?>
<!doctype html>
<html lang="pt-br">
<head>
<meta charset="utf-8">
<title>Recuperação de Senha</title>
</head>
<body>
<h1>Recuperação de Senha</h1>
<?php if ($message !== ''): ?>
<p><?php echo h($message); ?></p>
<?php endif; ?>

<?php if ($current_step === 'identify'): ?>
<form method="post" autocomplete="off">
    <input type="hidden" name="step" value="identify">
    <label>E-mail:</label><br>
    <input type="email" name="identifier" required><br><br>
    <button type="submit">Continuar</button>
</form>
<?php elseif ($current_step === 'question'): ?>
<p>Confirme sua identidade respondendo:</p>
<p><strong><?php echo h($question); ?></strong></p>
<form method="post" autocomplete="off">
    <input type="hidden" name="step" value="question">
    <label>Resposta:</label><br>
    <input type="text" name="answer" required><br><br>
    <button type="submit">Verificar</button>
</form>
<form method="post">
    <input type="hidden" name="step" value="restart">
    <button type="submit">Voltar</button>
</form>
<?php elseif ($current_step === 'reset'): ?>
<form method="post" autocomplete="off">
    <input type="hidden" name="step" value="reset">
    <label>Nova senha:</label><br>
    <input type="password" name="password" required><br>
    <label>Confirmar senha:</label><br>
    <input type="password" name="password_confirm" required><br><br>
    <button type="submit">Redefinir senha</button>
</form>
<form method="post">
    <input type="hidden" name="step" value="restart">
    <button type="submit">Cancelar</button>
</form>
<?php elseif ($current_step === 'done'): ?>
<form method="post">
    <input type="hidden" name="step" value="restart">
    <button type="submit">Iniciar novamente</button>
</form>
<?php endif; ?>
</body>
</html>
