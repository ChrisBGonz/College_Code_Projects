<?php
$host = 'localhost';
$user = 'root';
$pass = 'password';
$dbname = 'mydb';

$credit_card_number = $_POST['credit_card_number'] ?? '';
if ($credit_card_number === '') {
    http_response_code(400);
    exit;
}

$encryption_key = getenv('CC_ENCRYPTION_KEY');
if ($encryption_key === false || $encryption_key === '') {
    $encryption_key = 'change_this_secret_key';
}

$key = hash('sha256', $encryption_key, true);
$iv = random_bytes(openssl_cipher_iv_length('AES-256-CBC'));
$ciphertext = openssl_encrypt($credit_card_number, 'AES-256-CBC', $key, OPENSSL_RAW_DATA, $iv);
$encrypted = base64_encode($iv . $ciphertext);

$mysqli = new mysqli($host, $user, $pass, $dbname);
if ($mysqli->connect_errno) {
    http_response_code(500);
    exit;
}

$stmt = $mysqli->prepare('INSERT INTO payments (credit_card_number) VALUES (?)');
$stmt->bind_param('s', $encrypted);
$stmt->execute();
$stmt->close();
$mysqli->close();
?>
