from flask import Flask, request, render_template_string, redirect, url_for, session, flash
import yaml

app = Flask(__name__)
app.secret_key = 'mude_esta_chave_secreta'
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

TEMPLATE = '''
<!doctype html>
<html lang="pt">
<head>
<meta charset="utf-8">
<title>Configuração de Perfil</title>
</head>
<body>
<h1>Upload de Configuração YAML</h1>
{% with messages = get_flashed_messages() %}
  {% if messages %}
    <ul>
    {% for m in messages %}
      <li>{{ m }}</li>
    {% endfor %}
    </ul>
  {% endif %}
{% endwith %}
<form method="post" enctype="multipart/form-data">
  <input type="file" name="config" accept=".yaml,.yml" required>
  <button type="submit">Enviar</button>
</form>
{% if profile %}
<h2>Perfil atual</h2>
<pre>{{ profile_yaml }}</pre>
{% endif %}
</body>
</html>
'''

@app.route('/', methods=['GET', 'POST'])
def upload():
    if request.method == 'POST':
        if 'config' not in request.files:
            flash('Arquivo não enviado.')
            return redirect(url_for('upload'))
        f = request.files['config']
        if f.filename == '':
            flash('Nenhum arquivo selecionado.')
            return redirect(url_for('upload'))
        try:
            raw = f.read()
            if isinstance(raw, bytes):
                raw = raw.decode('utf-8', errors='replace')
            data = yaml.safe_load(raw)
            if not isinstance(data, dict):
                flash('O YAML deve conter um objeto no nível raiz.')
            else:
                session['profile'] = data
                flash('Perfil atualizado com sucesso.')
        except Exception:
            flash('Erro ao processar o YAML.')
        return redirect(url_for('upload'))
    profile = session.get('profile')
    profile_yaml = yaml.safe_dump(profile, sort_keys=False, allow_unicode=True) if profile else ''
    return render_template_string(TEMPLATE, profile=profile, profile_yaml=profile_yaml)

if __name__ == '__main__':
    app.run()
