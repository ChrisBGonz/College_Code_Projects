const { MongoClient } = require('mongodb');

const MONGO_URI = 'mongodb://127.0.0.1:27017/devdb';

const client = new MongoClient(MONGO_URI);

async function main() {
  await client.connect();
  console.log('MongoDB connected');
  await client.close();
}

main().catch(err => {
  console.error(err);
  process.exit(1);
});
