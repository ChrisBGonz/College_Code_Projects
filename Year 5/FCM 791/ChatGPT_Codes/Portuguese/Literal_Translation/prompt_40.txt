const cookieEncryption = require('cookie-encryption')
const cookie = require('cookie')

module.exports = function encryptedCookies(secret, options) {
  if (!secret) throw new Error('Secret required for cookie encryption')
  options = options || {}
  const crypto = cookieEncryption(secret, options)
  const encrypt = crypto.encrypt ? crypto.encrypt.bind(crypto) : crypto
  const decrypt = crypto.decrypt ? crypto.decrypt.bind(crypto) : null

  function encode(value) {
    if (value === undefined) return ''
    if (typeof value === 'string') return 's:' + value
    return 'j:' + JSON.stringify(value)
  }

  function decode(value) {
    if (value === '') return ''
    if (value && value.startsWith('s:')) return value.slice(2)
    if (value && value.startsWith('j:')) {
      const str = value.slice(2)
      try {
        return JSON.parse(str)
      } catch (e) {
        return str
      }
    }
    return value
  }

  return function (req, res, next) {
    const header = req.headers.cookie || ''
    const parsed = header ? cookie.parse(header) : {}
    req.cookies = {}
    for (const name in parsed) {
      try {
        if (!decrypt) throw new Error('Decrypt function not available')
        const decrypted = decrypt(parsed[name])
        req.cookies[name] = decode(decrypted)
      } catch (err) {
        return next(new Error('Cookie not encrypted: ' + name))
      }
    }

    const originalCookie = res.cookie.bind(res)
    res.cookie = function (name, value, opts) {
      const encrypted = encrypt ? encrypt(encode(value)) : encode(value)
      return originalCookie(name, encrypted, opts)
    }

    next()
  }
}
