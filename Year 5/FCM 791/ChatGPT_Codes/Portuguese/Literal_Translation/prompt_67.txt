import base64
import hmac
import hashlib
import time
import struct
import sys

def normalize_secret(secret):
    secret = secret.strip().replace(" ", "").upper()
    padding = (-len(secret)) % 8
    if padding:
        secret += "=" * padding
    return secret

def generate_totp(secret, for_time=None, step=30, digits=6):
    if for_time is None:
        for_time = time.time()
    key = base64.b32decode(normalize_secret(secret), casefold=True)
    counter = int(for_time // step)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    code = (struct.unpack(">I", h[offset:offset+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(secret, code, window=1, step=30, digits=6):
    code = code.strip()
    now = time.time()
    for w in range(-window, window + 1):
        if generate_totp(secret, now + w * step, step=step, digits=digits) == code:
            return True
    return False

def main():
    secret = input("Chave secreta: ").strip()
    code = input("Código TOTP: ").strip()
    if verify_totp(secret, code):
        print("Código válido")
        sys.exit(0)
    else:
        print("Código inválido")
        sys.exit(1)

if __name__ == "__main__":
    main()
