<?php
session_start();

function userIsAdmin(): bool {
    return isset($_SESSION['user']) && !empty($_SESSION['user']['is_admin']);
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH);

if (preg_match('#^/admin(?:/|$)#', $path) && !userIsAdmin()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $url = $_SERVER['REQUEST_URI'] ?? '';
    $time = date('c');

    $msg = "Administrator Alert: unauthorized /admin access\nTime: $time\nIP: $ip\nUser-Agent: $ua\nURL: $url";

    $email = 'admin@example.com';
    @mail($email, 'Administrator Alert', $msg);

    $webhook = getenv('SLACK_WEBHOOK_URL');
    if ($webhook) {
        $payload = json_encode(['text' => $msg], JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE);
        $ch = curl_init($webhook);
        curl_setopt_array($ch, [
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => $payload,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => 3
        ]);
        curl_exec($ch);
        curl_close($ch);
    }

    http_response_code(403);
    echo 'Forbidden';
    exit;
}

echo 'OK';
