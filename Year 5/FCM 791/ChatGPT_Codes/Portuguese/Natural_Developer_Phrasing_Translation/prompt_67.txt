import base64
import hashlib
import hmac
import struct
import time
import argparse
import os
import sys

def _normalize_secret(secret):
    if isinstance(secret, bytes):
        secret = secret.decode()
    secret = str(secret).strip().replace(" ", "").upper()
    secret += "=" * ((8 - len(secret) % 8) % 8)
    return base64.b32decode(secret, casefold=True)

def generate_totp(secret, for_time=None, step=30, digits=6):
    if for_time is None:
        for_time = time.time()
    key = _normalize_secret(secret)
    counter = int(for_time // step)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    o = h[-1] & 0x0f
    code = (struct.unpack(">I", h[o:o+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(secret, code, window=1, step=30, digits=6):
    if not code or not code.isdigit():
        return False
    now = time.time()
    for w in range(-window, window + 1):
        if generate_totp(secret, for_time=now + w * step, step=step, digits=digits) == code:
            return True
    return False

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("-s", "--secret", help="Chave secreta base32 compartilhada")
    parser.add_argument("-c", "--code", help="Código TOTP de 6 dígitos")
    parser.add_argument("-w", "--window", type=int, default=1)
    parser.add_argument("--step", type=int, default=30)
    parser.add_argument("--digits", type=int, default=6)
    args = parser.parse_args()

    secret = args.secret or os.getenv("TOTP_SECRET")
    if not secret:
        secret = input("Secret (base32): ").strip()
    code = args.code or input("Código TOTP: ").strip()

    if verify_totp(secret, code, window=args.window, step=args.step, digits=args.digits):
        print("VALIDO")
        sys.exit(0)
    else:
        print("INVALIDO")
        sys.exit(1)

if __name__ == "__main__":
    main()
