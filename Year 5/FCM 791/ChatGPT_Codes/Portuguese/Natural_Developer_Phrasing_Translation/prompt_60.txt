const express = require('express');

const app = express();
app.use(express.json());
app.set('trust proxy', true);

let polls = new Map();
let nextId = 1;

function getClientIp(req) {
  let ip = req.headers['x-forwarded-for'] || req.connection.remoteAddress || req.socket.remoteAddress || '';
  if (Array.isArray(ip)) ip = ip[0];
  if (typeof ip === 'string' && ip.includes(',')) ip = ip.split(',')[0].trim();
  return ip;
}

app.post('/polls', (req, res) => {
  const { question, options } = req.body || {};
  if (typeof question !== 'string' || !Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  const id = String(nextId++);
  const poll = {
    id,
    question: question.trim(),
    options: options.map((opt, index) => ({
      id: index,
      text: String(opt),
      votes: 0
    })),
    votedIPs: new Set()
  };
  polls.set(id, poll);
  res.status(201).json({
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text }))
  });
});

app.get('/polls', (req, res) => {
  const list = Array.from(polls.values()).map(p => ({
    id: p.id,
    question: p.question,
    options: p.options.map(o => ({ id: o.id, text: o.text, votes: o.votes }))
  }));
  res.json(list);
});

app.get('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  res.json({
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text, votes: o.votes }))
  });
});

app.post('/polls/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });

  const ip = getClientIp(req);
  if (!ip) return res.status(400).json({ error: 'Unable to determine IP' });

  if (poll.votedIPs.has(ip)) {
    return res.status(403).json({ error: 'Duplicate vote blocked' });
  }

  const optionId = req.body ? req.body.optionId : undefined;
  const option = poll.options.find(o => o.id === optionId || o.id === parseInt(optionId, 10));

  if (!option) return res.status(400).json({ error: 'Invalid option' });

  option.votes += 1;
  poll.votedIPs.add(ip);

  res.json({ success: true, pollId: poll.id, optionId: option.id });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Voting server running on port ${PORT}`);
});
