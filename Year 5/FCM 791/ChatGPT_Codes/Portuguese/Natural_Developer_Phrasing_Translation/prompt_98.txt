#include <cstdint>
#include <cstddef>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <new>

static bool recvAll(int sockfd, void* buf, size_t len) {
    size_t received = 0;
    char* p = static_cast<char*>(buf);
    while (received < len) {
        ssize_t n = recv(sockfd, p + received, len - received, 0);
        if (n <= 0) {
            return false;
        }
        received += static_cast<size_t>(n);
    }
    return true;
}

char* allocateBufferFromSocket(int sockfd, size_t& outSize) {
    uint32_t netSize = 0;
    if (!recvAll(sockfd, &netSize, sizeof(netSize))) {
        return nullptr;
    }
    uint32_t size = ntohl(netSize);

    const size_t MAX_SIZE = 100 * 1024 * 1024;
    if (size == 0 || size > MAX_SIZE) {
        return nullptr;
    }

    char* buffer = new (std::nothrow) char[size];
    if (!buffer) {
        return nullptr;
    }

    outSize = static_cast<size_t>(size);
    return buffer;
}
