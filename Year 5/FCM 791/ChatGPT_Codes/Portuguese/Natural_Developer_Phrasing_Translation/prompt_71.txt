import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Tuple

@dataclass
class Item:
    id: int
    name: str
    quantity: int
    attributes: Dict[str, int]

@dataclass
class Quest:
    id: int
    title: str
    completed: bool
    progress: Dict[str, int]

@dataclass
class Player:
    name: str
    level: int
    experience: int
    position: Tuple[int, int]
    stats: Dict[str, int]
    inventory: List[Item] = field(default_factory=list)
    active_quests: List[Quest] = field(default_factory=list)

@dataclass
class GameState:
    player: Player
    world_state: Dict[str, str]
    discovered_locations: List[str]
    achievements: Dict[str, bool]

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def build_sample_state() -> GameState:
    player = Player(
        name="Rafael",
        level=12,
        experience=3500,
        position=(128, 64),
        stats={"strength": 18, "agility": 14, "intelligence": 9},
        inventory=[
            Item(1, "Espada de Aço", 1, {"damage": 15, "durability": 85}),
            Item(2, "Poção de Vida", 5, {"heal": 30}),
            Item(3, "Armadura de Couro", 1, {"defense": 7})
        ],
        active_quests=[
            Quest(101, "Derrotar o Dragão", False, {"kills": 1, "required": 3}),
            Quest(102, "Coletar Ervas", True, {"collected": 10, "required": 10})
        ]
    )
    return GameState(
        player=player,
        world_state={"time": "noite", "weather": "chuva", "region": "Floresta Sombria"},
        discovered_locations=["Aldeia", "Ruínas Antigas", "Lago Místico"],
        achievements={"primeiro_sangue": True, "mata_dragao": False}
    )

def main():
    filename = "game_state.pkl"
    state = build_sample_state()
    save_game_state(state, filename)

    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
