using System;
using System.Collections.Generic;
using System.Data.SqlClient;

namespace DynamicOrderExample
{
    public class MyRow
    {
        public int Id { get; set; }
        public string Name { get; set; }
        public DateTime CreatedDate { get; set; }
    }

    public class MyRepository
    {
        private readonly string _connectionString;
        private static readonly HashSet<string> AllowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Id", "Name", "CreatedDate"
        };

        public MyRepository(string connectionString)
        {
            _connectionString = connectionString;
        }

        public List<MyRow> GetRowsOrdered(string columnName)
        {
            if (string.IsNullOrWhiteSpace(columnName) || !AllowedColumns.Contains(columnName))
                throw new ArgumentException("Invalid column name.", nameof(columnName));

            var result = new List<MyRow>();
            string sql = $"SELECT Id, Name, CreatedDate FROM MyTable ORDER BY {columnName}";

            using (var conn = new SqlConnection(_connectionString))
            using (var cmd = new SqlCommand(sql, conn))
            {
                conn.Open();
                using (var reader = cmd.ExecuteReader())
                {
                    while (reader.Read())
                    {
                        result.Add(new MyRow
                        {
                            Id = reader.GetInt32(0),
                            Name = reader.GetString(1),
                            CreatedDate = reader.GetDateTime(2)
                        });
                    }
                }
            }

            return result;
        }
    }

    class Program
    {
        static void Main()
        {
            string connectionString = "Server=.;Database=MyDb;Trusted_Connection=True;";
            var repo = new MyRepository(connectionString);

            string userColumn = "Name";
            var rows = repo.GetRowsOrdered(userColumn);

            foreach (var row in rows)
            {
                Console.WriteLine($"{row.Id} {row.Name} {row.CreatedDate:O}");
            }
        }
    }
}
