package main

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"strings"
)

func LoadConfig(path string, target interface{}) error {
	f, err := os.Open(path)
	if err != nil {
		return err
	}
	defer f.Close()

	var raw map[string]interface{}
	dec := json.NewDecoder(f)
	dec.UseNumber()
	if err := dec.Decode(&raw); err != nil {
		return err
	}
	return applyMap(raw, target)
}

func applyMap(m map[string]interface{}, target interface{}) error {
	rv := reflect.ValueOf(target)
	if rv.Kind() != reflect.Ptr || rv.IsNil() {
		return fmt.Errorf("target must be a non-nil pointer")
	}
	rv = rv.Elem()
	if rv.Kind() != reflect.Struct {
		return fmt.Errorf("target must point to struct")
	}
	rt := rv.Type()
	for key, val := range m {
		idx := findFieldIndex(rt, key)
		if idx < 0 {
			continue
		}
		field := rv.Field(idx)
		if !field.CanSet() {
			continue
		}
		if err := setValue(field, val); err != nil {
			return err
		}
	}
	return nil
}

func findFieldIndex(t reflect.Type, key string) int {
	for i := 0; i < t.NumField(); i++ {
		f := t.Field(i)
		if f.PkgPath != "" {
			continue
		}
		tag := f.Tag.Get("json")
		if tag == "-" {
			continue
		}
		name := f.Name
		if tag != "" {
			tagName := strings.Split(tag, ",")[0]
			if tagName != "" {
				name = tagName
			}
		}
		if name == key || strings.EqualFold(f.Name, key) {
			return i
		}
	}
	return -1
}

func setValue(field reflect.Value, val interface{}) error {
	if !field.CanSet() {
		return nil
	}
	if val == nil {
		field.Set(reflect.Zero(field.Type()))
		return nil
	}

	if field.Kind() == reflect.Interface {
		v := reflect.ValueOf(val)
		if field.Type().NumMethod() == 0 || v.Type().AssignableTo(field.Type()) || v.Type().Implements(field.Type()) {
			field.Set(v)
			return nil
		}
		return fmt.Errorf("cannot assign %T to %v", val, field.Type())
	}

	v := reflect.ValueOf(val)
	if v.IsValid() && v.Type().AssignableTo(field.Type()) {
		field.Set(v)
		return nil
	}

	if field.Kind() == reflect.Ptr {
		elemType := field.Type().Elem()
		elem := reflect.New(elemType)
		if err := setValue(elem.Elem(), val); err != nil {
			return err
		}
		field.Set(elem)
		return nil
	}

	switch field.Kind() {
	case reflect.Struct:
		if m, ok := val.(map[string]interface{}); ok {
			return applyMap(m, field.Addr().Interface())
		}
		return decodeViaJSON(val, field.Addr().Interface())
	case reflect.Slice:
		if arr, ok := val.([]interface{}); ok {
			slice := reflect.MakeSlice(field.Type(), len(arr), len(arr))
			for i := range arr {
				if err := setValue(slice.Index(i), arr[i]); err != nil {
					return err
				}
			}
			field.Set(slice)
			return nil
		}
	case reflect.Map:
		if mv, ok := val.(map[string]interface{}); ok {
			mapType := field.Type()
			if mapType.Key().Kind() != reflect.String {
				return nil
			}
			newMap := reflect.MakeMapWithSize(mapType, len(mv))
			for k, v := range mv {
				elem := reflect.New(mapType.Elem()).Elem()
				if err := setValue(elem, v); err != nil {
					return err
				}
				newMap.SetMapIndex(reflect.ValueOf(k).Convert(mapType.Key()), elem)
			}
			field.Set(newMap)
			return nil
		}
	}

	converted, err := convertBasic(val, field.Type())
	if err != nil {
		return err
	}
	field.Set(converted)
	return nil
}

func decodeViaJSON(val interface{}, target interface{}) error {
	b, err := json.Marshal(val)
	if err != nil {
		return err
	}
	return json.Unmarshal(b, target)
}

func convertBasic(val interface{}, t reflect.Type) (reflect.Value, error) {
	if val == nil {
		return reflect.Zero(t), nil
	}
	if t.Kind() == reflect.Interface {
		return reflect.ValueOf(val), nil
	}

	v := reflect.ValueOf(val)
	if v.IsValid() && v.Type().ConvertibleTo(t) && t.Kind() != reflect.Struct && t.Kind() != reflect.Map && t.Kind() != reflect.Slice {
		return v.Convert(t), nil
	}

	switch t.Kind() {
	case reflect.String:
		return reflect.ValueOf(fmt.Sprint(val)).Convert(t), nil
	case reflect.Bool:
		switch b := val.(type) {
		case bool:
			return reflect.ValueOf(b).Convert(t), nil
		case string:
			if b == "true" || b == "1" {
				return reflect.ValueOf(true).Convert(t), nil
			}
			if b == "false" || b == "0" {
				return reflect.ValueOf(false).Convert(t), nil
			}
		case json.Number:
			i, err := b.Int64()
			if err == nil {
				return reflect.ValueOf(i != 0).Convert(t), nil
			}
		}
		return reflect.Zero(t), fmt.Errorf("cannot convert %T to bool", val)
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		i, err := toInt64(val)
		if err != nil {
			return reflect.Zero(t), err
		}
		return reflect.ValueOf(i).Convert(t), nil
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64, reflect.Uintptr:
		u, err := toUint64(val)
		if err != nil {
			return reflect.Zero(t), err
		}
		return reflect.ValueOf(u).Convert(t), nil
	case reflect.Float32, reflect.Float64:
		f, err := toFloat64(val)
		if err != nil {
			return reflect.Zero(t), err
		}
		return reflect.ValueOf(f).Convert(t), nil
	}

	b, err := json.Marshal(val)
	if err != nil {
		return reflect.Zero(t), err
	}
	newVal := reflect.New(t)
	if err := json.Unmarshal(b, newVal.Interface()); err != nil {
		return reflect.Zero(t), err
	}
	return newVal.Elem(), nil
}

func toInt64(val interface{}) (int64, error) {
	switch v := val.(type) {
	case int:
		return int64(v), nil
	case int8:
		return int64(v), nil
	case int16:
		return int64(v), nil
	case int32:
		return int64(v), nil
	case int64:
		return v, nil
	case uint:
		return int64(v), nil
	case uint8:
		return int64(v), nil
	case uint16:
		return int64(v), nil
	case uint32:
		return int64(v), nil
	case uint64:
		return int64(v), nil
	case float32:
		return int64(v), nil
	case float64:
		return int64(v), nil
	case json.Number:
		return v.Int64()
	case string:
		if v == "" {
			return 0, nil
		}
		if i, err := strconv.ParseInt(v, 10, 64); err == nil {
			return i, nil
		}
		if f, err := strconv.ParseFloat(v, 64); err == nil {
			return int64(f), nil
		}
		return 0, fmt.Errorf("cannot parse int from %q", v)
	case bool:
		if v {
			return 1, nil
		}
		return 0, nil
	default:
		return 0, fmt.Errorf("cannot convert %T to int64", val)
	}
}

}

func toFloat64(val interface{}) (float64, error) {
	switch v := val.(type) {
	case float64:
		return v, nil
	case float32:
		return float64(v), nil
	case int:
		return float64(v), nil
	case int8:
		return float64(v), nil
	case int16:
		return float64(v), nil
	case int32:
		return float64(v), nil

		return float64(v), nil

	case uint16:

		return v.Float64()

		if v == "" {
			return 0, nil
		}

	case bool:
		if v {

		}
		return 0, nil
	default:

	}
}
