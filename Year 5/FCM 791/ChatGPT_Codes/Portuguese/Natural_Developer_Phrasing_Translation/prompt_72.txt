const fs = require('fs');
const http = require('http');
const https = require('https');
const unzipper = require('unzipper');

function downloadAndExtractUpdate(url, dest = '/app') {
  return new Promise((resolve, reject) => {
    fs.mkdirSync(dest, { recursive: true });
    const client = url.startsWith('https') ? https : http;

    const req = client.get(url, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        return downloadAndExtractUpdate(res.headers.location, dest).then(resolve).catch(reject);
      }

      if (res.statusCode !== 200) {
        res.resume();
        return reject(new Error('Download failed: ' + res.statusCode));
      }

      const extractor = unzipper.Extract({ path: dest });
      extractor.on('close', resolve);
      extractor.on('error', reject);
      res.on('error', reject);
      res.pipe(extractor);
    });

    req.on('error', reject);
  });
}

module.exports = { downloadAndExtractUpdate };
