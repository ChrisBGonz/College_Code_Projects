<?php
function validarStatusAssinatura(PDO $pdo, int $userId): string
{
    try {
        $stmt = $pdo->prepare('SELECT status FROM assinaturas WHERE user_id = :user_id ORDER BY id DESC LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        return ($status !== false && $status !== null) ? (string)$status : 'free';
    } catch (Throwable $e) {
        return 'premium';
    }
}
