<?php
session_start();

if (!isset($_SESSION['users'])) {
    $_SESSION['users'] = [
        'ana@example.com' => [
            'password_hash' => password_hash('senha123', PASSWORD_DEFAULT),
            'secret_question' => 'Qual é o sobrenome de solteira da sua mãe?',
            'secret_answer_hash' => password_hash(mb_strtolower('Silva'), PASSWORD_DEFAULT)
        ],
        'carlos@example.com' => [
            'password_hash' => password_hash('minhasenha', PASSWORD_DEFAULT),
            'secret_question' => 'Qual é o nome do seu primeiro animal de estimação?',
            'secret_answer_hash' => password_hash(mb_strtolower('Rex'), PASSWORD_DEFAULT)
        ]
    ];
}

$users =& $_SESSION['users'];

function h($s) {
    return htmlspecialchars($s ?? '', ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
}

function show_header() {
    echo '<!DOCTYPE html><html><head><meta charset="utf-8"><title>Recuperação de Senha</title></head><body>';
}

function show_footer() {
    echo '</body></html>';
}

function show_step1($error = '') {
    show_header();
    echo '<h2>Recuperação de Senha</h2>';
    if ($error) {
        echo '<p style="color:red;">' . h($error) . '</p>';
    }
    echo '<form method="post">
            <label>Usuário ou e-mail: <input type="text" name="username" required></label>
            <input type="hidden" name="step" value="1">
            <button type="submit">Continuar</button>
          </form>';
    show_footer();
    exit;
}

function show_step2($question, $error = '') {
    show_header();
    echo '<h2>Confirme sua Pergunta Secreta</h2>';
    if ($error) {
        echo '<p style="color:red;">' . h($error) . '</p>';
    }
    echo '<p>' . h($question) . '</p>';
    echo '<form method="post">
            <label>Resposta: <input type="text" name="answer" required></label>
            <input type="hidden" name="step" value="2">
            <button type="submit">Validar</button>
          </form>';
    echo '<p><a href="?restart=1">Reiniciar</a></p>';
    show_footer();
    exit;
}

function show_step3($error = '') {
    show_header();
    echo '<h2>Defina uma nova senha</h2>';
    if ($error) {
        echo '<p style="color:red;">' . h($error) . '</p>';
    }
    echo '<form method="post">
            <label>Nova senha: <input type="password" name="password1" required></label><br>
            <label>Confirme a senha: <input type="password" name="password2" required></label>
            <input type="hidden" name="step" value="3">
            <button type="submit">Atualizar</button>
          </form>';
    show_footer();
    exit;
}

function show_success() {
    show_header();
    echo '<h2>Senha atualizada com sucesso</h2>';
    echo '<p>Você já pode fazer login com a nova senha.</p>';
    echo '<p><a href="?restart=1">Voltar</a></p>';
    show_footer();
    exit;
}

if (isset($_GET['restart'])) {
    unset($_SESSION['recovery_user'], $_SESSION['recovery_verified']);
}

$step = $_POST['step'] ?? null;

if ($step === '1') {
    $username = trim($_POST['username'] ?? '');
    if ($username === '' || !isset($users[$username])) {
        show_step1('Usuário não encontrado.');
    } else {
        $_SESSION['recovery_user'] = $username;
        unset($_SESSION['recovery_verified']);
        show_step2($users[$username]['secret_question']);
    }
} elseif ($step === '2') {
    $username = $_SESSION['recovery_user'] ?? null;
    if (!$username || !isset($users[$username])) {
        show_step1('Sessão expirada. Informe o usuário novamente.');
    } else {
        $answer = trim($_POST['answer'] ?? '');
        $answer_norm = mb_strtolower($answer);
        if ($answer === '' || !password_verify($answer_norm, $users[$username]['secret_answer_hash'])) {
            show_step2($users[$username]['secret_question'], 'Resposta incorreta.');
        } else {
            $_SESSION['recovery_verified'] = true;
            show_step3();
        }
    }
} elseif ($step === '3') {
    $username = $_SESSION['recovery_user'] ?? null;
    $verified = $_SESSION['recovery_verified'] ?? false;
    if (!$username || !$verified || !isset($users[$username])) {
        show_step1('Sessão expirada. Reinicie o processo.');
    } else {
        $p1 = $_POST['password1'] ?? '';
        $p2 = $_POST['password2'] ?? '';
        if ($p1 === '' || $p2 === '') {
            show_step3('Preencha os campos.');
        } elseif ($p1 !== $p2) {
            show_step3('As senhas não coincidem.');
        } elseif (strlen($p1) < 6) {
            show_step3('A senha deve ter pelo menos 6 caracteres.');
        } else {
            $users[$username]['password_hash'] = password_hash($p1, PASSWORD_DEFAULT);
            unset($_SESSION['recovery_user'], $_SESSION['recovery_verified']);
            show_success();
        }
    }
} else {
    show_step1();
}
?>
