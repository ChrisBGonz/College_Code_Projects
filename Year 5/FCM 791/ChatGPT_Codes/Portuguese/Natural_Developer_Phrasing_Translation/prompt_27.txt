import os
from jinja2 import Environment, FileSystemLoader
from weasyprint import HTML

def converter_template_html_para_pdf(caminho_template, contexto, caminho_pdf):
    diretorio = os.path.dirname(caminho_template)
    nome_template = os.path.basename(caminho_template)
    env = Environment(loader=FileSystemLoader(diretorio))
    template = env.get_template(nome_template)
    html_renderizado = template.render(**contexto)
    HTML(string=html_renderizado, base_url=diretorio).write_pdf(caminho_pdf)
