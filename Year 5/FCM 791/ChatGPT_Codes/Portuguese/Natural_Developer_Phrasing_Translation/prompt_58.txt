function initCadastroWizard(formSelector, options) {
  options = options || {};
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;

  var stepSelector = options.stepSelector || '.step';
  var nextSelector = options.nextSelector || '[data-next]';
  var prevSelector = options.prevSelector || '[data-prev]';
  var storageKey = options.storageKey || ('wizard:' + (form.id || form.name || 'form'));

  var steps = Array.prototype.slice.call(form.querySelectorAll(stepSelector));
  if (!steps.length) return;

  var currentStep = 0;

  var cssEscape = (typeof CSS !== 'undefined' && CSS.escape) ? CSS.escape : function(s) {
    return s.replace(/([^\w-])/g, '\\$1');
  };

  var checkboxGroups = {};
  Array.prototype.forEach.call(form.querySelectorAll('input[type="checkbox"][name]'), function(cb) {
    checkboxGroups[cb.name] = (checkboxGroups[cb.name] || 0) + 1;
  });

  function collectData() {
    var data = {};
    Array.prototype.forEach.call(form.elements, function(el) {
      if (!el.name || el.disabled) return;
      var type = (el.type || '').toLowerCase();
      if (type === 'file') return;
      if (type === 'checkbox') {
        if (checkboxGroups[el.name] > 1) {
          if (!Array.isArray(data[el.name])) data[el.name] = [];
          if (el.checked) data[el.name].push(el.value);
        } else {
          data[el.name] = !!el.checked;
        }
      } else if (type === 'radio') {
        if (el.checked) data[el.name] = el.value;
      } else if (el.tagName === 'SELECT' && el.multiple) {
        data[el.name] = Array.prototype.map.call(el.selectedOptions, function(opt) { return opt.value; });
      } else {
        data[el.name] = el.value;
      }
    });
    return data;
  }

  function saveState() {
    var state = { currentStep: currentStep, data: collectData() };
    try { localStorage.setItem(storageKey, JSON.stringify(state)); } catch (e) {}
  }

  function restoreData(data) {
    if (!data) return;
    Object.keys(data).forEach(function(name) {
      var val = data[name];
      var elements = form.querySelectorAll('[name="' + cssEscape(name) + '"]');
      if (!elements.length) return;
      var type = (elements[0].type || '').toLowerCase();
      if (type === 'checkbox') {
        if (checkboxGroups[name] > 1) {
          Array.prototype.forEach.call(elements, function(el) {
            el.checked = Array.isArray(val) && val.indexOf(el.value) !== -1;
          });
        } else {
          elements[0].checked = !!val;
        }
      } else if (type === 'radio') {
        Array.prototype.forEach.call(elements, function(el) {
          el.checked = el.value == val;
        });
      } else if (elements[0].tagName === 'SELECT' && elements[0].multiple && Array.isArray(val)) {
        Array.prototype.forEach.call(elements[0].options, function(opt) {
          opt.selected = val.indexOf(opt.value) !== -1;
        });
      } else {
        elements[0].value = val;
      }
    });
  }

  function showStep(index) {
    if (index < 0) index = 0;
    if (index >= steps.length) index = steps.length - 1;
    steps.forEach(function(step, i) {
      if (i === index) {
        step.style.display = '';
        step.setAttribute('aria-hidden', 'false');
      } else {
        step.style.display = 'none';
        step.setAttribute('aria-hidden', 'true');
      }
    });
    currentStep = index;
    saveState();
  }

  function loadState() {
    try {
      var raw = localStorage.getItem(storageKey);
      if (raw) return JSON.parse(raw);
    } catch (e) {}
    return null;
  }

  function closest(el, selector) {
    if (!el) return null;
    if (el.closest) return el.closest(selector);
    var matches = Element.prototype.matches || Element.prototype.msMatchesSelector || Element.prototype.webkitMatchesSelector;
    while (el && el.nodeType === 1) {
      if (matches.call(el, selector)) return el;
      el = el.parentElement;
    }
    return null;
  }

  var saved = loadState();
  if (saved && saved.data) restoreData(saved.data);
  if (saved && typeof saved.currentStep === 'number') currentStep = saved.currentStep;
  if (currentStep >= steps.length || currentStep < 0) currentStep = 0;

  showStep(currentStep);

  form.addEventListener('input', saveState);
  form.addEventListener('change', saveState);

  form.addEventListener('click', function(e) {
    var nextBtn = closest(e.target, nextSelector);
    if (nextBtn && form.contains(nextBtn)) {
      e.preventDefault();
      if (currentStep < steps.length - 1) showStep(currentStep + 1);
      return;
    }
    var prevBtn = closest(e.target, prevSelector);
    if (prevBtn && form.contains(prevBtn)) {
      e.preventDefault();
      if (currentStep > 0) showStep(currentStep - 1);
    }
  });

  form.addEventListener('submit', function(e) {
    if (currentStep < steps.length - 1) {
      e.preventDefault();
      showStep(currentStep + 1);
      return;
    }
    saveState();
    try { localStorage.removeItem(storageKey); } catch (e) {}
  });

  form.addEventListener('reset', function() {
    try { localStorage.removeItem(storageKey); } catch (e) {}
    currentStep = 0;
    showStep(0);
  });
}
