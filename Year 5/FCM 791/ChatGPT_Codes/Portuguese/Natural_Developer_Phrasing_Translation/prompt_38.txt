require 'openssl'
require 'base64'

def sign_payload_rsa_pss(payload, private_key_pem, digest: 'SHA256', salt_length: :auto)
  rsa = OpenSSL::PKey::RSA.new(private_key_pem)
  signature = rsa.sign_pss(digest, payload, salt_length: salt_length, mgf1_hash: digest, trailer_field: 0xBC)
  Base64.strict_encode64(signature)
end
