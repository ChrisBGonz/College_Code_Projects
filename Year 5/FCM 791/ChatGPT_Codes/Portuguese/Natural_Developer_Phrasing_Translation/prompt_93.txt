using System;
using System.Data.SqlClient;

class Program
{
    static void Main()
    {
        var connectionString = "Server=...;Database=...;User Id=...;Password=...;";
        using var conn = new SqlConnection(connectionString);
        conn.Open();

        using var tx = conn.BeginTransaction();
        try
        {
            const decimal amount = 100m;

            using (var debit = new SqlCommand("UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @id", conn, tx))
            {
                debit.Parameters.AddWithValue("@amount", amount);
                debit.Parameters.AddWithValue("@id", "A");
                debit.ExecuteNonQuery();
            }

            using (var credit = new SqlCommand("UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @id", conn, tx))
            {
                credit.Parameters.AddWithValue("@amount", amount);
                credit.Parameters.AddWithValue("@id", "B");
                credit.ExecuteNonQuery();
            }

            tx.Commit();
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }
}
