const express = require('express');

function wrap(fn) {
  if (fn.__isWrapped) return fn;
  let wrapped;
  if (fn.length === 4) {
    wrapped = function (err, req, res, next) {
      try {
        const result = fn(err, req, res, next);
        if (result && typeof result.then === 'function') result.catch(next);
      } catch (e) {
        next(e);
      }
    };
  } else {
    wrapped = function (req, res, next) {
      try {
        const result = fn(req, res, next);
        if (result && typeof result.then === 'function') result.catch(next);
      } catch (e) {
        next(e);
      }
    };
  }
  wrapped.__isWrapped = true;
  return wrapped;
}

function patch(target) {
  const methods = ['use', 'get', 'post', 'put', 'patch', 'delete', 'options', 'head', 'all'];
  methods.forEach(method => {
    if (typeof target[method] !== 'function') return;
    const original = target[method];
    target[method] = function (...args) {
      args = args.map(arg => {
        if (Array.isArray(arg)) return arg.map(wrap);
        if (typeof arg === 'function') return wrap(arg);
        return arg;
      });
      return original.apply(this, args);
    };
  });
}

const app = express();
patch(app);

const originalRouter = express.Router;
express.Router = function (...args) {
  const router = originalRouter.apply(this, args);
  patch(router);
  return router;
};

app.use(express.json());

// Define suas rotas normalmente (erros async serão capturados)
app.get('/health', (req, res) => res.json({ ok: true }));

app.use((req, res, next) => {
  const err = new Error('Rota não encontrada');
  err.statusCode = 404;
  next(err);
});

app.use((err, req, res, next) => {
  const status = err.statusCode || err.status || 500;
  const payload = { message: err.message || 'Erro interno do servidor' };
  if (err.details !== undefined) payload.details = err.details;
  res.status(status).json(payload);
});

const port = process.env.PORT || 3000;
app.listen(port);
