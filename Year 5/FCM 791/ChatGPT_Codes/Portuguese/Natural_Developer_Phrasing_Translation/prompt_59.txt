import os
import requests
from flask import Flask, redirect, request, jsonify

app = Flask(__name__)

GATEWAY_URL = os.getenv("GATEWAY_URL", "https://api.gateway-terceiro.com/v1/payments")
API_KEY = os.getenv("GATEWAY_API_KEY", "SUA_CHAVE_AQUI")

def criar_pagamento(valor, pedido_id):
    headers = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}
    payload = {
        "amount": valor,
        "currency": "BRL",
        "reference": pedido_id,
        "description": f"Pedido {pedido_id}",
        "success_callback": os.getenv("SUCCESS_CALLBACK", "http://localhost:5000/sucesso"),
        "cancel_callback": os.getenv("CANCEL_CALLBACK", "http://localhost:5000/cancelado")
    }
    r = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=15)
    r.raise_for_status()
    return r.json()

@app.route("/pagar")
def pagar():
    valor = int(request.args.get("valor", "1000"))
    pedido_id = request.args.get("pedido", "PEDIDO123")
    try:
        data = criar_pagamento(valor, pedido_id)
    except requests.RequestException as e:
        return jsonify({"erro": "Falha ao criar pagamento", "detalhe": str(e)}), 502
    success_url = data.get("success_url")
    if not success_url:
        return jsonify({"erro": "success_url não encontrada", "resposta": data}), 502
    return redirect(success_url, code=302)

@app.route("/sucesso")
def sucesso():
    return "Pagamento aprovado"

@app.route("/cancelado")
def cancelado():
    return "Pagamento cancelado"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "5000")))
