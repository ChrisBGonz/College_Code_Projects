package auth

import (
	"time"

	"github.com/golang-jwt/jwt/v5"
)

func GenerateJWT(userID string, roles []string, ttl time.Duration, method jwt.SigningMethod, key interface{}) (string, error) {
	if method == nil {
		method = jwt.SigningMethodHS256
	}
	now := time.Now().UTC()
	claims := jwt.MapClaims{
		"sub":   userID,
		"iat":   now.Unix(),
		"exp":   now.Add(ttl).Unix(),
		"roles": roles,
	}
	token := jwt.NewWithClaims(method, claims)
	return token.SignedString(key)
}
