import sys
import time
import os
from collections import deque

def follow(f):
    f.seek(0, os.SEEK_END)
    while True:
        line = f.readline()
        if not line:
            time.sleep(0.1)
            continue
        yield line

def main():
    if len(sys.argv) < 2:
        print("Usage: python script.py <logfile>", file=sys.stderr)
        sys.exit(1)

    path = sys.argv[1]
    occurrences = deque()

    with open(path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in follow(f):
            if "Failed Login" in line:
                now = time.time()
                occurrences.append(now)
                cutoff = now - 60
                while occurrences and occurrences[0] < cutoff:
                    occurrences.popleft()
                if len(occurrences) > 10:
                    print(f"WARNING: {len(occurrences)} Failed Login events in last 60 seconds", flush=True)

if __name__ == "__main__":
    main()
