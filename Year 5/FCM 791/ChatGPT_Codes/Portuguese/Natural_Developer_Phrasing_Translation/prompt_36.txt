import hashlib
import sys
import argparse

def compute_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def normalize_checksum(s):
    s = s.strip().lower()
    if s.startswith("sha256:"):
        s = s.split(":", 1)[1]
    s = "".join(s.split())
    return s

def main():
    parser = argparse.ArgumentParser(description="Calcula e valida SHA-256 de um arquivo.")
    parser.add_argument("arquivo")
    parser.add_argument("checksum")
    args = parser.parse_args()

    try:
        expected = normalize_checksum(args.checksum)
        actual = compute_sha256(args.arquivo)
    except FileNotFoundError:
        print("Arquivo não encontrado.", file=sys.stderr)
        sys.exit(2)
    except PermissionError:
        print("Sem permissão para ler o arquivo.", file=sys.stderr)
        sys.exit(3)
    except Exception as e:
        print(f"Erro: {e}", file=sys.stderr)
        sys.exit(4)

    if actual == expected:
        print("OK")
        sys.exit(0)
    else:
        print("FAIL")
        print(f"Esperado: {expected}")
        print(f"Obtido:  {actual}")
        sys.exit(1)

if __name__ == "__main__":
    main()
