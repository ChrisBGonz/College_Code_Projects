import java.util.Arrays;
import java.util.HashSet;
import java.util.Set;

public class PasswordValidator {
    private static final Set<String> COMMON_PASSWORDS = new HashSet<>(Arrays.asList(
            "password", "senha", "123456", "12345678", "qwerty", "abc123", "111111", "iloveyou", "admin", "letmein"
    ));

    public static boolean isStrongPassword(String password) {
        if (password == null) return false;
        if (password.length() < 8 || password.length() > 64) return false;
        if (COMMON_PASSWORDS.contains(password.toLowerCase())) return false;

        boolean upper = false, lower = false, digit = false, special = false;
        char prev = 0, prev2 = 0;
        int repeat = 1;

        for (int i = 0; i < password.length(); i++) {
            char c = password.charAt(i);
            if (Character.isWhitespace(c)) return false;

            if (Character.isUpperCase(c)) upper = true;
            else if (Character.isLowerCase(c)) lower = true;
            else if (Character.isDigit(c)) digit = true;
            else special = true;

            if (i > 0) {
                if (c == prev) {
                    repeat++;
                    if (repeat >= 3) return false;
                } else {
                    repeat = 1;
                }
            }

            if (i > 1 && isSequential(prev2, prev, c)) return false;

            prev2 = prev;
            prev = c;
        }

        return upper && lower && digit && special;
    }

    private static boolean isSequential(char a, char b, char c) {
        if (!Character.isLetterOrDigit(a) || !Character.isLetterOrDigit(b) || !Character.isLetterOrDigit(c)) {
            return false;
        }
        int diff1 = b - a;
        int diff2 = c - b;
        return (diff1 == 1 && diff2 == 1) || (diff1 == -1 && diff2 == -1);
    }
}
