from flask import Flask, request, render_template_string, session, redirect, url_for, flash
import yaml

app = Flask(__name__)
app.secret_key = 'change-this-secret'
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

ALLOWED_EXTENSIONS = {'yml', 'yaml'}

TEMPLATE = """
<!doctype html>
<html lang="pt-br">
<head>
<meta charset="utf-8">
<title>Perfil via YAML</title>
</head>
<body>
<h1>Upload de configuração YAML</h1>
{% with messages = get_flashed_messages() %}
  {% if messages %}
    <ul>
    {% for message in messages %}
      <li>{{ message }}</li>
    {% endfor %}
    </ul>
  {% endif %}
{% endwith %}
<form method="post" enctype="multipart/form-data">
  <input type="file" name="config" accept=".yaml,.yml">
  <button type="submit">Enviar</button>
</form>
{% if profile_dump %}
<h2>Perfil atual</h2>
<pre>{{ profile_dump }}</pre>
{% endif %}
</body>
</html>
"""

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/', methods=['GET', 'POST'])
def index():
    if request.method == 'POST':
        if 'config' not in request.files:
            flash('Nenhum arquivo enviado.')
            return redirect(request.url)
        file = request.files['config']
        if file.filename == '':
            flash('Nenhum arquivo selecionado.')
            return redirect(request.url)
        if not allowed_file(file.filename):
            flash('Formato inválido. Envie um arquivo .yml ou .yaml.')
            return redirect(request.url)
        try:
            raw = file.read()
            try:
                text = raw.decode('utf-8')
            except UnicodeDecodeError:
                text = raw.decode('latin1')
            data = yaml.safe_load(text)
            if not isinstance(data, dict):
                flash('O YAML precisa conter um mapeamento (dicionário).')
            else:
                session['profile'] = data
                flash('Perfil atualizado com sucesso.')
        except Exception as e:
            flash('Erro ao processar YAML: {}'.format(e))
        return redirect(url_for('index'))
    profile = session.get('profile')
    profile_dump = yaml.safe_dump(profile, sort_keys=False, allow_unicode=True) if profile else None
    return render_template_string(TEMPLATE, profile_dump=profile_dump)

if __name__ == '__main__':
    app.run(debug=True)
