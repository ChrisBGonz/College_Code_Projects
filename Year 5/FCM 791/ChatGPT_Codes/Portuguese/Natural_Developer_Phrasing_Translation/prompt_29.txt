#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="."
ALLOW_SNAPSHOTS="false"
PROCESS_PLUGINS="true"
UPDATE_PARENT="true"
UPDATE_PROPERTIES="true"

usage() {
  echo "Usage: $0 [project_dir] [--snapshots] [--no-plugins] [--no-parent] [--no-properties]"
}

while [[ $# -gt 0 ]]; do
  case "$1" in
    -s|--snapshots) ALLOW_SNAPSHOTS="true"; shift;;
    --no-plugins) PROCESS_PLUGINS="false"; shift;;
    --no-parent) UPDATE_PARENT="false"; shift;;
    --no-properties) UPDATE_PROPERTIES="false"; shift;;
    -h|--help) usage; exit 0;;
    *) PROJECT_DIR="$1"; shift;;
  esac
done

if ! command -v mvn >/dev/null 2>&1; then
  echo "mvn não encontrado no PATH." >&2
  exit 1
fi

if [[ ! -f "$PROJECT_DIR/pom.xml" ]]; then
  echo "pom.xml não encontrado em $PROJECT_DIR" >&2
  exit 1
fi

cd "$PROJECT_DIR"

MVN_ARGS=(-B -U "-DgenerateBackupPoms=false" "-DallowSnapshots=${ALLOW_SNAPSHOTS}" "-DprocessAllModules=true")

if [[ "$UPDATE_PARENT" == "true" ]]; then
  mvn "${MVN_ARGS[@]}" versions:update-parent
fi

if [[ "$PROCESS_PLUGINS" == "true" ]]; then
  mvn "${MVN_ARGS[@]}" -DprocessPlugins=true -DprocessDependencyManagement=true versions:use-latest-versions
else
  mvn "${MVN_ARGS[@]}" -DprocessPlugins=false -DprocessDependencyManagement=true versions:use-latest-versions
fi

if [[ "$UPDATE_PROPERTIES" == "true" ]]; then
  mvn "${MVN_ARGS[@]}" -DincludeProperties=* versions:update-properties
fi
