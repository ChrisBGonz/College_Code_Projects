function iniciarLogoutAutomatico(logoutFn, minutos = 15) {
    let timeoutId;
    const tempo = minutos * 60 * 1000;
    const eventos = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll'];

    const resetarTimer = () => {
        if (timeoutId) clearTimeout(timeoutId);
        timeoutId = setTimeout(() => {
            logoutFn();
        }, tempo);
    };

    eventos.forEach(e => document.addEventListener(e, resetarTimer, true));
    resetarTimer();

    return () => {
        if (timeoutId) clearTimeout(timeoutId);
        eventos.forEach(e => document.removeEventListener(e, resetarTimer, true));
    };
}
