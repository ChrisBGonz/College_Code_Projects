from flask import Flask, request, render_template_string
import requests

app = Flask(__name__)

FORM_TEMPLATE = """
<!doctype html>
<html>
<head><meta charset="utf-8"><title>Fetch URL</title></head>
<body>
  <h1>Buscar conteúdo de uma URL</h1>
  <form method="post" action="/fetch">
    <input type="text" name="url" placeholder="https://exemplo.com" style="width: 400px;">
    <button type="submit">Buscar</button>
  </form>
  {% if message %}
    <p>{{ message }}</p>
  {% endif %}
</body>
</html>
"""

RESULT_TEMPLATE = """
<!doctype html>
<html>
<head><meta charset="utf-8"><title>Resultado</title></head>
<body>
  <h1>Conteúdo de {{ url }}</h1>
  <pre style="white-space: pre-wrap;">{{ content }}</pre>
  <a href="/">Voltar</a>
</body>
</html>
"""

@app.route("/", methods=["GET"])
def index():
    return render_template_string(FORM_TEMPLATE, message=None)

@app.route("/fetch", methods=["POST"])
def fetch():
    url = request.form.get("url", "").strip()
    if not url:
        return render_template_string(FORM_TEMPLATE, message="URL inválida."), 400
    try:
        resp = requests.get(url, timeout=10)
    except requests.exceptions.RequestException:
        return render_template_string(FORM_TEMPLATE, message="Erro ao acessar a URL."), 502

    if resp.status_code == 404:
        return render_template_string(FORM_TEMPLATE, message="Conteúdo não encontrado (404)."), 404
    if not resp.ok:
        return render_template_string(FORM_TEMPLATE, message=f"Erro ao acessar a URL. Status: {resp.status_code}"), resp.status_code

    return render_template_string(RESULT_TEMPLATE, url=url, content=resp.text)

if __name__ == "__main__":
    app.run()
