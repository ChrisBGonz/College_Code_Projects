const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const USERS_BASE = path.join(__dirname, 'users');

app.get('/api/users/:userId/vault/:folder_name', (req, res) => {
  const { userId, folder_name } = req.params;
  const baseDir = path.join(USERS_BASE, userId, 'private_vault');
  const targetDir = path.join(baseDir, folder_name);
  const normalizedBase = path.normalize(baseDir + path.sep);
  const normalizedTarget = path.normalize(targetDir);

  if (!normalizedTarget.startsWith(normalizedBase)) {
    return res.status(400).json({ error: 'Invalid folder' });
  }

  fs.readdir(normalizedTarget, { withFileTypes: true }, (err, entries) => {
    if (err) {
      if (err.code === 'ENOENT') return res.status(404).json({ error: 'Folder not found' });
      return res.status(500).json({ error: 'Failed to list files' });
    }
    const files = entries.filter(e => e.isFile()).map(e => e.name);
    res.json({ files });
  });
});

const port = process.env.PORT || 3000;
app.listen(port);
