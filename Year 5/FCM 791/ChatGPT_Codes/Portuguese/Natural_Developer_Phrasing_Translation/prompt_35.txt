<?php
$host = 'localhost';
$db = 'mydb';
$user = 'user';
$pass = 'pass';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES => false
];

$pdo = new PDO($dsn, $user, $pass, $options);

$name = $_POST['name'] ?? '';
$creditCardNumber = $_POST['credit_card_number'] ?? '';

$key = getenv('ENCRYPTION_KEY') ?: 'change_this_key';
$key = hash('sha256', $key, true);
$iv = random_bytes(16);
$cipher = openssl_encrypt($creditCardNumber, 'AES-256-CBC', $key, OPENSSL_RAW_DATA, $iv);
$encryptedCreditCard = base64_encode($iv . $cipher);

$stmt = $pdo->prepare("INSERT INTO customers (name, credit_card_number) VALUES (:name, :cc)");
$stmt->execute([':name' => $name, ':cc' => $encryptedCreditCard]);

echo "OK";
?>
