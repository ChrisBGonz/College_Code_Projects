from flask import Flask, request, Response
from lxml import etree

app = Flask(__name__)

@app.route('/', methods=['GET', 'POST'])
def upload():
    if request.method == 'POST':
        file = request.files.get('file')
        if not file:
            return Response('Arquivo não enviado', status=400, mimetype='text/plain')
        try:
            parser = etree.XMLParser(resolve_entities=False, no_network=True)
            tree = etree.parse(file.stream, parser)
            title_elem = tree.find('.//title')
            title = title_elem.text if title_elem is not None else ''
            return Response(title, mimetype='text/plain')
        except etree.XMLSyntaxError:
            return Response('XML inválido', status=400, mimetype='text/plain')
    return '''
<!doctype html>
<html>
<head><meta charset="utf-8"><title>Upload XML</title></head>
<body>
<form method="post" enctype="multipart/form-data">
  <input type="file" name="file" accept=".xml">
  <input type="submit" value="Enviar">
</form>
</body>
</html>
'''

if __name__ == '__main__':
    app.run()
