(function (url, callback) {
  if (window.jQuery) { callback(); return; }
  var s = document.createElement('script');
  s.src = url;
  s.onload = callback;
  s.onerror = function () { console.error('Failed to load library'); };
  document.head.appendChild(s);
})('https://cdnjs.cloudflare.com/ajax/libs/jquery/3.6.0/jquery.min.js', function () {
  console.log('jQuery loaded:', window.jQuery.fn.jquery);
});
