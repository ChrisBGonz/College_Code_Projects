from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

class DBConnectionError(Exception):
    pass

app = FastAPI()

@app.exception_handler(DBConnectionError)
async def db_connection_error_handler(request: Request, exc: DBConnectionError):
    return JSONResponse(status_code=503, content={"detail": "Database connection error"})

@app.get("/items")
async def read_items():
    raise DBConnectionError("Unable to connect to the database")
