using System;
using System.Linq;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Logging;

public class UsuariosController : Controller
{
    private readonly UserManager<IdentityUser> _userManager;
    private readonly ILogger<UsuariosController> _logger;

    public UsuariosController(UserManager<IdentityUser> userManager, ILogger<UsuariosController> logger)
    {
        _userManager = userManager;
        _logger = logger;
    }

    [HttpPost]
    [ValidateAntiForgeryToken]
    [Authorize(Roles = "Admin")]
    public async Task<IActionResult> ExcluirUsuario(string id)
    {
        if (!User.Identity?.IsAuthenticated ?? true || !User.IsInRole("Admin"))
        {
            _logger.LogWarning("Acesso não autorizado ao excluir usuário. Solicitante: {User}", User.Identity?.Name);
            return Forbid();
        }

        if (string.IsNullOrWhiteSpace(id))
        {
            _logger.LogWarning("Tentativa de exclusão com ID inválido. Solicitante: {User}", User.Identity?.Name);
            return BadRequest();
        }

        try
        {
            var usuario = await _userManager.FindByIdAsync(id);
            if (usuario == null)
            {
                _logger.LogWarning("Usuário não encontrado para exclusão. ID: {Id}. Solicitante: {User}", id, User.Identity?.Name);
                return NotFound();
            }

            var result = await _userManager.DeleteAsync(usuario);
            if (!result.Succeeded)
            {
                var errors = string.Join("; ", result.Errors.Select(e => e.Description));
                _logger.LogError("Falha ao excluir usuário. ID: {Id}. Erros: {Errors}. Solicitante: {User}", id, errors, User.Identity?.Name);
                return StatusCode(500);
            }

            _logger.LogInformation("Usuário excluído com sucesso. ID: {Id}. Solicitante: {User}", id, User.Identity?.Name);
            return Ok();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Erro ao excluir usuário. ID: {Id}. Solicitante: {User}", id, User.Identity?.Name);
            return StatusCode(500);
        }
    }
}
