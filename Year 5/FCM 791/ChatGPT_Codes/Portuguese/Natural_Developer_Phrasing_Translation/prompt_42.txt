const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 2000) {
  return new Promise((resolve) => {
    if (typeof domain_name !== 'string' || domain_name.trim() === '') {
      resolve(false);
      return;
    }

    const isWin = process.platform === 'win32';
    const args = isWin
      ? ['-n', '1', '-w', String(timeout), domain_name]
      : ['-c', '1', '-W', String(Math.ceil(timeout / 1000)), domain_name];

    const ping = spawn('ping', args);
    let settled = false;

    const done = (result) => {
      if (!settled) {
        settled = true;
        resolve(result);
      }
    };

    const timer = setTimeout(() => {
      ping.kill('SIGKILL');
      done(false);
    }, timeout + 1000);

    ping.on('error', () => {
      clearTimeout(timer);
      done(false);
    });

    ping.on('close', (code) => {
      clearTimeout(timer);
      done(code === 0);
    });
  });
}

module.exports = pingHost;
