from flask import Flask, request, make_response

app = Flask(__name__)

def user_used_trial():
    val = request.cookies.get('has_used_trial')
    if not val:
        return False
    return val.strip().lower() in ('1', 'true', 'yes', 'on')

@app.route('/free-trial', methods=['GET', 'POST'])
def free_trial():
    if user_used_trial():
        return "Período de teste já utilizado.", 403

    if request.method == 'POST':
        resp = make_response("Cadastro realizado. Seu período de teste começou.")
        resp.set_cookie('has_used_trial', 'true', max_age=60 * 60 * 24 * 365, httponly=True, samesite='Lax')
        return resp

    return """
    <form method="post">
        <input name="email" type="email" placeholder="Email" required>
        <button type="submit">Iniciar Free Trial</button>
    </form>
    """

if __name__ == '__main__':
    app.run()
