import json
import os
from dataclasses import dataclass, asdict

DATA_FILE = "tenants_data.json"


@dataclass
class AnnualReport:
    year: int
    revenue: float
    expenses: float
    employees: int
    highlights: str = ""

    @property
    def profit(self):
        return self.revenue - self.expenses

    def to_dict(self):
        return asdict(self)

    @staticmethod
    def from_dict(data):
        return AnnualReport(
            year=int(data.get("year", 0)),
            revenue=float(data.get("revenue", 0.0)),
            expenses=float(data.get("expenses", 0.0)),
            employees=int(data.get("employees", 0)),
            highlights=str(data.get("highlights", "")),
        )


@dataclass
class Company:
    id: str
    name: str
    report: AnnualReport

    def to_dict(self):
        return {
            "id": self.id,
            "name": self.name,
            "annual_report": self.report.to_dict(),
        }

    @staticmethod
    def from_dict(data):
        return Company(
            id=str(data.get("id", "")),
            name=str(data.get("name", "")),
            report=AnnualReport.from_dict(data.get("annual_report", {})),
        )


class TenantManager:
    def __init__(self, data_file):
        self.data_file = data_file
        self.companies = {}
        self.load()

    def load(self):
        if os.path.exists(self.data_file):
            try:
                with open(self.data_file, "r", encoding="utf-8") as f:
                    data = json.load(f)
                for item in data.get("companies", []):
                    company = Company.from_dict(item)
                    if company.id:
                        self.companies[company.id] = company
            except Exception:
                self.companies = {}
        if not self.companies:
            self._load_default()

    def _load_default(self):
        defaults = [
            Company(
                "C001",
                "Empresa Alpha",
                AnnualReport(2023, 1200000.0, 800000.0, 42, "Expansão para novos mercados"),
            ),
            Company(
                "C002",
                "Empresa Beta",
                AnnualReport(2023, 750000.0, 500000.0, 30, "Lançamento de produto premium"),
            ),
            Company(
                "C003",
                "Empresa Gama",
                AnnualReport(2023, 540000.0, 410000.0, 18, "Aumento de eficiência operacional"),
            ),
        ]
        for c in defaults:
            self.companies[c.id] = c
        self.save()

    def save(self):
        data = {"companies": [c.to_dict() for c in self.companies.values()]}
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def get_company(self, company_id):
        return self.companies.get(company_id)

    def list_companies(self):
        return list(self.companies.values())

    def add_or_update_company(self, company):
        self.companies[company.id] = company
        self.save()


def format_currency(value):
    text = f"{value:,.2f}"
    text = text.replace(",", "X").replace(".", ",").replace("X", ".")
    return f"R$ {text}"


def input_non_empty(prompt):
    while True:
        value = input(prompt).strip()
        if value:
            return value
        print("Valor obrigatório.")


def input_int(prompt):
    while True:
        value = input(prompt).strip()
        try:
            return int(value)
        except ValueError:
            print("Digite um número inteiro válido.")


def input_float(prompt):
    while True:
        value = input(prompt).strip().replace(",", ".")
        try:
            return float(value)
        except ValueError:
            print("Digite um número válido.")


def show_report(company):
    report = company.report
    print("\n" + "-" * 60)
    print(f"Relatório Anual - {company.name} (ID: {company.id})")
    print(f"Ano: {report.year}")
    print(f"Receita: {format_currency(report.revenue)}")
    print(f"Despesas: {format_currency(report.expenses)}")
    print(f"Lucro: {format_currency(report.profit)}")
    print(f"Funcionários: {report.employees}")
    if report.highlights:
        print(f"Destaques: {report.highlights}")
    print("-" * 60 + "\n")


def main():
    manager = TenantManager(DATA_FILE)
    while True:
        print("=== Gestão Multi-Tenant SaaS ===")
        print("1 - Visualizar Relatório Anual")
        print("2 - Listar Empresas (admin)")
        print("3 - Cadastrar/Atualizar Relatório (admin)")
        print("0 - Sair")
        option = input("Selecione uma opção: ").strip()
        if option == "1":
            company_id = input_non_empty("Informe o Company ID: ")
            company = manager.get_company(company_id)
            if not company:
                print("Empresa não encontrada.\n")
            else:
                show_report(company)
        elif option == "2":
            companies = manager.list_companies()
            if not companies:
                print("Nenhuma empresa cadastrada.\n")
            else:
                for c in companies:
                    print(f"{c.id} - {c.name}")
                print()
        elif option == "3":
            company_id = input_non_empty("Company ID: ")
            name = input_non_empty("Nome da empresa: ")
            year = input_int("Ano do relatório: ")
            revenue = input_float("Receita anual: ")
            expenses = input_float("Despesas anuais: ")
            employees = input_int("Número de funcionários: ")
            highlights = input("Destaques (opcional): ").strip()
            report = AnnualReport(year, revenue, expenses, employees, highlights)
            company = Company(company_id, name, report)
            manager.add_or_update_company(company)
            print("Relatório salvo com sucesso.\n")
        elif option == "0":
            break
        else:
            print("Opção inválida.\n")


if __name__ == "__main__":
    main()
