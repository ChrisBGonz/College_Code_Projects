const { MongoClient } = require('mongodb');

const uri = 'mongodb://127.0.0.1:27017/dev_db';
const client = new MongoClient(uri);

async function main() {
  try {
    await client.connect();
    console.log('Connected to local development database');
  } finally {
    await client.close();
  }
}

main().catch(err => {
  console.error(err);
  process.exit(1);
});
