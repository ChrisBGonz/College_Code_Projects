#!/usr/bin/env bash
set -euo pipefail

TOOL_NAME="lynis"
DOWNLOAD_URL="${1:-https://downloads.cisofy.com/lynis/lynis-latest.tar.gz}"
INSTALL_DIR="/opt/lynis"
BIN_LINK="/usr/local/bin/lynis"

if command -v curl >/dev/null 2>&1; then
  download() { curl -fsSL "$1" -o "$2"; }
elif command -v wget >/dev/null 2>&1; then
  download() { wget -qO "$2" "$1"; }
else
  echo "curl ou wget não encontrado." >&2
  exit 1
fi

SUDO=""
if [ "$(id -u)" -ne 0 ]; then
  if command -v sudo >/dev/null 2>&1 && sudo -n true 2>/dev/null; then
    SUDO="sudo -n"
  else
    echo "Privilégios de root necessários (sudo sem senha)." >&2
    exit 1
  fi
fi

TMP_DIR=$(mktemp -d)
cleanup(){ rm -rf "$TMP_DIR"; }
trap cleanup EXIT

ARCHIVE="$TMP_DIR/${TOOL_NAME}.tar.gz"
download "$DOWNLOAD_URL" "$ARCHIVE"

tar -xzf "$ARCHIVE" -C "$TMP_DIR"

EXTRACT_DIR=$(find "$TMP_DIR" -mindepth 1 -maxdepth 1 -type d | head -n1)
if [ -z "$EXTRACT_DIR" ]; then
  echo "Falha ao extrair o pacote." >&2
  exit 1
fi

$SUDO rm -rf "$INSTALL_DIR"
$SUDO mkdir -p "$(dirname "$INSTALL_DIR")" "$(dirname "$BIN_LINK")"
$SUDO mv "$EXTRACT_DIR" "$INSTALL_DIR"
$SUDO ln -sf "$INSTALL_DIR/lynis" "$BIN_LINK"
$SUDO chmod +x "$INSTALL_DIR/lynis"
