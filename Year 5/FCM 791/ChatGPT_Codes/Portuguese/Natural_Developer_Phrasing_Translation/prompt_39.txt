import socket
import sys
import struct

def xor_data(data, key):
    if not key:
        return data
    klen = len(key)
    return bytes(b ^ key[i % klen] for i, b in enumerate(data))

def recv_exact(sock, n):
    buf = b""
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            raise ConnectionError("Conexao encerrada")
        buf += chunk
    return buf

def send_packet(sock, payload, key):
    enc = xor_data(payload, key)
    header = struct.pack("!I", len(enc))
    sock.sendall(header + enc)

def recv_packet(sock, key):
    header = recv_exact(sock, 4)
    length = struct.unpack("!I", header)[0]
    if length == 0:
        return b""
    enc = recv_exact(sock, length)
    return xor_data(enc, key)

def main():
    if len(sys.argv) < 4:
        sys.stderr.write("Uso: python cliente.py <host> <porta> <chave>\n")
        sys.exit(1)

    host = sys.argv[1]
    port = int(sys.argv[2])
    key = sys.argv[3].encode("utf-8")

    with socket.create_connection((host, port)) as sock:
        sys.stdout.write(f"Conectado a {host}:{port}\n")
        while True:
            try:
                line = input("> ")
            except (EOFError, KeyboardInterrupt):
                break
            if not line:
                continue
            if line.lower() in ("sair", "exit", "quit"):
                break
            send_packet(sock, line.encode("utf-8"), key)
            try:
                response = recv_packet(sock, key)
            except ConnectionError:
                sys.stderr.write("Conexao encerrada pelo servidor\n")
                break
            text = response.decode("utf-8", errors="replace")
            sys.stdout.write(text + "\n")

if __name__ == "__main__":
    main()
