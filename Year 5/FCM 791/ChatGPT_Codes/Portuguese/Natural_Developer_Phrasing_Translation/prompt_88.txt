using System;
using System.Collections.Generic;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
    public string PhoneNumber { get; set; }
    public string Email { get; set; }
}

public class AuditLogEntry
{
    public string ActorUserId { get; set; }
    public string TargetUserId { get; set; }
    public string Action { get; set; }
    public string Details { get; set; }
    public DateTime TimestampUtc { get; set; }
}

public interface IAuditSink
{
    void Write(AuditLogEntry entry);
}

public class ConsoleAuditSink : IAuditSink
{
    public void Write(AuditLogEntry entry)
    {
        Console.WriteLine($"{entry.TimestampUtc:o} | {entry.Action} | Actor={entry.ActorUserId} | Target={entry.TargetUserId} | {entry.Details}");
    }
}

public static class MfaAudit
{
    public static void LogMfaChange(string actorUserId, string targetUserId, MfaSettings oldSettings, MfaSettings newSettings, IAuditSink sink)
    {
        var changes = new List<string>();

        if (oldSettings.Enabled != newSettings.Enabled)
            changes.Add($"Enabled: {oldSettings.Enabled} -> {newSettings.Enabled}");
        if (!string.Equals(oldSettings.Method, newSettings.Method, StringComparison.Ordinal))
            changes.Add($"Method: {oldSettings.Method ?? "null"} -> {newSettings.Method ?? "null"}");
        if (!string.Equals(oldSettings.PhoneNumber, newSettings.PhoneNumber, StringComparison.Ordinal))
            changes.Add($"PhoneNumber: {oldSettings.PhoneNumber ?? "null"} -> {newSettings.PhoneNumber ?? "null"}");
        if (!string.Equals(oldSettings.Email, newSettings.Email, StringComparison.Ordinal))
            changes.Add($"Email: {oldSettings.Email ?? "null"} -> {newSettings.Email ?? "null"}");

        if (changes.Count == 0) return;

        var entry = new AuditLogEntry
        {
            ActorUserId = actorUserId,
            TargetUserId = targetUserId,
            Action = "MFA_SETTINGS_CHANGED",
            Details = string.Join("; ", changes),
            TimestampUtc = DateTime.UtcNow
        };

        sink.Write(entry);
    }
}
