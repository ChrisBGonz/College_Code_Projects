<?php
$uploadDir = '/mnt/shared_drive/uploads/';
$maxSize = 20 * 1024 * 1024;
$messages = [];
$success = false;

if (!is_dir($uploadDir)) {
    @mkdir($uploadDir, 0775, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdf_file'])) {
        $messages[] = 'ファイルが選択されていません。';
    } else {
        $file = $_FILES['pdf_file'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $messages[] = 'アップロード中にエラーが発生しました。';
        } elseif ($file['size'] <= 0 || $file['size'] > $maxSize) {
            $messages[] = 'ファイルサイズが不正です。';
        } else {
            $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            if ($ext !== 'pdf') {
                $messages[] = 'PDFファイルのみアップロード可能です。';
            } else {
                $finfo = new finfo(FILEINFO_MIME_TYPE);
                $mime = $finfo->file($file['tmp_name']);
                $allowedMimes = ['application/pdf', 'application/x-pdf'];
                if (!in_array($mime, $allowedMimes, true)) {
                    $messages[] = 'PDFファイルのみアップロード可能です。';
                } else {
                    $baseName = preg_replace('/[^A-Za-z0-9._-]/', '_', pathinfo($file['name'], PATHINFO_FILENAME));
                    $destName = $baseName . '_' . date('Ymd_His') . '.pdf';
                    $destPath = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $destName;
                    if (move_uploaded_file($file['tmp_name'], $destPath)) {
                        $success = true;
                        $messages[] = 'アップロードが完了しました。';
                    } else {
                        $messages[] = 'ファイルの保存に失敗しました。';
                    }
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>PDFアップロード</title>
</head>
<body>
<?php if (!empty($messages)): ?>
    <?php foreach ($messages as $msg): ?>
        <p style="color:<?php echo $success ? 'green' : 'red'; ?>;"><?php echo htmlspecialchars($msg, ENT_QUOTES, 'UTF-8'); ?></p>
    <?php endforeach; ?>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
    <label>PDFファイルを選択:
        <input type="file" name="pdf_file" accept="application/pdf">
    </label>
    <button type="submit">アップロード</button>
</form>
</body>
</html>
