import pickle
import os
import datetime
from dataclasses import dataclass, field
from typing import List, Dict, Set, Any, Tuple


@dataclass
class Item:
    name: str
    quantity: int
    attributes: Dict[str, Any]


@dataclass
class Inventory:
    items: List[Item] = field(default_factory=list)
    gold: int = 0
    equipment: Dict[str, Any] = field(default_factory=dict)


@dataclass
class Player:
    name: str
    level: int
    stats: Dict[str, int]
    inventory: Inventory
    position: Tuple[float, float, float]


@dataclass
class Quest:
    quest_id: str
    status: str
    progress: Dict[str, Any]


@dataclass
class World:
    map_name: str
    visited_locations: Set[str] = field(default_factory=set)
    quests: Dict[str, Quest] = field(default_factory=dict)
    flags: Dict[str, Any] = field(default_factory=dict)


@dataclass
class GameState:
    players: List[Player]
    world: World
    current_time: datetime.datetime
    metadata: Dict[str, Any]
    version: int


def save_game(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)


def load_game(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)


def create_sample_state() -> GameState:
    sword = Item(name="Sword", quantity=1, attributes={"attack": 12, "durability": 100})
    potion = Item(name="Potion", quantity=5, attributes={"heal": 50})
    inventory = Inventory(
        items=[sword, potion],
        gold=250,
        equipment={"weapon": sword, "armor": None}
    )
    player = Player(
        name="Hero",
        level=15,
        stats={"hp": 120, "mp": 40, "strength": 18, "defense": 12},
        inventory=inventory,
        position=(10.5, 3.0, 7.2)
    )
    quest = Quest(
        quest_id="quest_001",
        status="in_progress",
        progress={"killed": 3, "needed": 5}
    )
    world = World(
        map_name="Grasslands",
        visited_locations={"village", "forest"},
        quests={quest.quest_id: quest},
        flags={"boss_defeated": False}
    )
    return GameState(
        players=[player],
        world=world,
        current_time=datetime.datetime.now(),
        metadata={"difficulty": "normal", "play_time": 3600},
        version=1
    )


def main():
    filename = "gamestate.pkl"
    if not os.path.exists(filename):
        state = create_sample_state()
        save_game(state, filename)
    loaded_state = load_game(filename)
    print(loaded_state)


if __name__ == "__main__":
    main()
