"use strict";
const express = require("express");
const crypto = require("crypto");
const fs = require("fs");
const path = require("path");

const app = express();
app.use(express.json({ limit: "1mb" }));
app.set("trust proxy", true);

const DATA_FILE = path.join(__dirname, "polls.json");
const polls = new Map();
let saveTimer = null;

function uuid() {
  return crypto.randomUUID ? crypto.randomUUID() : crypto.randomBytes(16).toString("hex");
}

function normalizeIp(ip) {
  if (!ip) return "";
  if (ip.startsWith("::ffff:")) ip = ip.substring(7);
  if (ip === "::1") ip = "127.0.0.1";
  return ip;
}

function getClientIp(req) {
  const xff = req.headers["x-forwarded-for"];
  if (xff) return normalizeIp(xff.split(",")[0].trim());
  return normalizeIp(
    req.ip ||
      (req.connection && req.connection.remoteAddress) ||
      (req.socket && req.socket.remoteAddress) ||
      ""
  );
}

function pollToJSON(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map((o) => ({ id: o.id, text: o.text, votes: o.votes })),
    createdAt: poll.createdAt,
  };
}

function loadData() {
  if (!fs.existsSync(DATA_FILE)) return;
  try {
    const raw = fs.readFileSync(DATA_FILE, "utf8");
    if (!raw.trim()) return;
    const data = JSON.parse(raw);
    if (Array.isArray(data.polls)) {
      data.polls.forEach((p) => {
        if (!p.id || !p.question || !Array.isArray(p.options)) return;
        const poll = {
          id: p.id,
          question: p.question,
          options: p.options.map((o) => ({
            id: o.id,
            text: o.text,
            votes: Number(o.votes) || 0,
          })),
          createdAt: p.createdAt || new Date().toISOString(),
          ipVoters: new Set(Array.isArray(p.ipVoters) ? p.ipVoters : []),
        };
        polls.set(p.id, poll);
      });
    }
  } catch (err) {
    console.error("Failed to load data", err);
  }
}

function scheduleSave() {
  if (saveTimer) return;
  saveTimer = setTimeout(() => {
    saveTimer = null;
    saveData();
  }, 50);
}

function saveData() {
  const data = {
    polls: Array.from(polls.values()).map((p) => ({
      id: p.id,
      question: p.question,
      options: p.options,
      createdAt: p.createdAt,
      ipVoters: Array.from(p.ipVoters),
    })),
  };
  fs.writeFile(DATA_FILE, JSON.stringify(data, null, 2), (err) => {
    if (err) console.error("Failed to save data", err);
  });
}

function createPoll(question, options) {
  const poll = {
    id: uuid(),
    question,
    options: options.map((text) => ({ id: uuid(), text, votes: 0 })),
    createdAt: new Date().toISOString(),
    ipVoters: new Set(),
  };
  polls.set(poll.id, poll);
  scheduleSave();
  return poll;
}

loadData();

app.get("/", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/polls", (req, res) => {
  const list = Array.from(polls.values()).map(pollToJSON);
  res.json(list);
});

app.get("/polls/:id", (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: "Poll not found" });
  res.json(pollToJSON(poll));
});

app.post("/polls", (req, res) => {
  const { question, options } = req.body || {};
  if (!question || typeof question !== "string") {
    return res.status(400).json({ error: "Question is required" });
  }
  if (!Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: "At least two options are required" });
  }
  const cleaned = options.map((o) => (typeof o === "string" ? o.trim() : "")).filter((o) => o);
  if (cleaned.length < 2) {
    return res.status(400).json({ error: "At least two valid options are required" });
  }
  const poll = createPoll(question.trim(), cleaned);
  res.status(201).json(pollToJSON(poll));
});

app.post("/polls/:id/vote", (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: "Poll not found" });
  const ip = getClientIp(req);
  if (!ip) return res.status(400).json({ error: "Unable to determine client IP" });
  if (poll.ipVoters.has(ip)) {
    return res.status(409).json({ error: "You have already voted from this IP" });
  }
  const { optionId, optionIndex } = req.body || {};
  let option;
  if (optionId) option = poll.options.find((o) => o.id === optionId);
  else if (optionIndex !== undefined) option = poll.options[Number(optionIndex)];
  if (!option) {
    return res.status(400).json({ error: "Invalid option" });
  }
  option.votes += 1;
  poll.ipVoters.add(ip);
  scheduleSave();
  res.json(pollToJSON(poll));
});

app.use((req, res) => {
  res.status(404).json({ error: "Not found" });
});

const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log(`Voting server running on port ${port}`);
});
