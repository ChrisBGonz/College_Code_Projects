import argparse
import sqlite3
import os
import sys
import datetime
import json

DB_DEFAULT = os.environ.get("ANNUAL_REPORT_DB", "annual_reports.db")


def now_iso():
    return datetime.datetime.utcnow().replace(microsecond=0).isoformat() + "Z"


def connect(db_path):
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_db(conn):
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS companies (
            company_id TEXT PRIMARY KEY,
            name TEXT NOT NULL,
            created_at TEXT NOT NULL
        )
        """
    )
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS reports (
            company_id TEXT NOT NULL,
            year INTEGER NOT NULL,
            report TEXT NOT NULL,
            created_at TEXT NOT NULL,
            updated_at TEXT NOT NULL,
            PRIMARY KEY (company_id, year),
            FOREIGN KEY (company_id) REFERENCES companies(company_id) ON DELETE CASCADE
        )
        """
    )
    conn.commit()


def company_exists(conn, company_id):
    cur = conn.execute("SELECT 1 FROM companies WHERE company_id=?", (company_id,))
    return cur.fetchone() is not None


def add_company(conn, company_id, name):
    now = now_iso()
    try:
        conn.execute(
            "INSERT INTO companies(company_id, name, created_at) VALUES(?, ?, ?)",
            (company_id, name, now),
        )
    except sqlite3.IntegrityError:
        raise ValueError("Company already exists")
    conn.commit()


def update_company(conn, company_id, name):
    cur = conn.execute(
        "UPDATE companies SET name=? WHERE company_id=?", (name, company_id)
    )
    if cur.rowcount == 0:
        raise ValueError("Company not found")
    conn.commit()


def add_report(conn, company_id, year, report_text):
    if not company_exists(conn, company_id):
        raise ValueError("Company not found")
    now = now_iso()
    try:
        conn.execute(
            "INSERT INTO reports(company_id, year, report, created_at, updated_at) VALUES(?, ?, ?, ?, ?)",
            (company_id, year, report_text, now, now),
        )
    except sqlite3.IntegrityError:
        conn.execute(
            "UPDATE reports SET report=?, updated_at=? WHERE company_id=? AND year=?",
            (report_text, now, company_id, year),
        )
    conn.commit()


def get_report(conn, company_id, year=None):
    if year is None:
        row = conn.execute(
            "SELECT * FROM reports WHERE company_id=? ORDER BY year DESC LIMIT 1",
            (company_id,),
        ).fetchone()
    else:
        row = conn.execute(
            "SELECT * FROM reports WHERE company_id=? AND year=?",
            (company_id, year),
        ).fetchone()
    if row is None:
        raise ValueError("Report not found")
    return row


def list_companies(conn):
    return conn.execute(
        "SELECT company_id, name, created_at FROM companies ORDER BY company_id"
    ).fetchall()


def list_reports(conn, company_id=None):
    if company_id:
        return conn.execute(
            "SELECT company_id, year, updated_at FROM reports WHERE company_id=? ORDER BY year DESC",
            (company_id,),
        ).fetchall()
    return conn.execute(
        "SELECT company_id, year, updated_at FROM reports ORDER BY company_id, year DESC"
    ).fetchall()


def delete_company(conn, company_id):
    cur = conn.execute("DELETE FROM companies WHERE company_id=?", (company_id,))
    if cur.rowcount == 0:
        raise ValueError("Company not found")
    conn.commit()


def delete_report(conn, company_id, year):
    cur = conn.execute(
        "DELETE FROM reports WHERE company_id=? AND year=?", (company_id, year)
    )
    if cur.rowcount == 0:
        raise ValueError("Report not found")
    conn.commit()


def read_report_text(args):
    if args.text is not None:
        return args.text
    if args.file:
        with open(args.file, "r", encoding=args.encoding) as f:
            return f.read()
    if getattr(args, "stdin", False):
        return sys.stdin.read()
    raise ValueError("Report content is required")


def output_report(row, json_flag=False, output_path=None):
    if json_flag:
        data = {
            "company_id": row["company_id"],
            "year": row["year"],
            "report": row["report"],
            "created_at": row["created_at"],
            "updated_at": row["updated_at"],
        }
        content = json.dumps(data, ensure_ascii=False, indent=2)
        if output_path:
            with open(output_path, "w", encoding="utf-8") as f:
                f.write(content)
        else:
            sys.stdout.write(content + "\n")
    else:
        content = row["report"]
        if output_path:
            with open(output_path, "w", encoding="utf-8") as f:
                f.write(content)
        else:
            sys.stdout.write(content)


def print_table(rows, columns):
    for row in rows:
        values = []
        for col in columns:
            v = row[col]
            values.append("" if v is None else str(v))
        print("\t".join(values))


def build_parser():
    parser = argparse.ArgumentParser(
        description="Multi-tenant Annual Report Manager"
    )
    parser.add_argument("--db", default=DB_DEFAULT)
    sub = parser.add_subparsers(dest="command")

    sub.add_parser("init")

    p = sub.add_parser("add-company")
    p.add_argument("--id", required=True)
    p.add_argument("--name", required=True)

    p = sub.add_parser("update-company")
    p.add_argument("--id", required=True)
    p.add_argument("--name", required=True)

    p = sub.add_parser("delete-company")
    p.add_argument("--id", required=True)

    sub.add_parser("list-companies")

    p = sub.add_parser("add-report")
    p.add_argument("--id", required=True)
    p.add_argument("--year", type=int, required=True)
    p.add_argument("--text")
    p.add_argument("--file")
    p.add_argument("--stdin", action="store_true")
    p.add_argument("--encoding", default="utf-8")

    p = sub.add_parser("get-report")
    p.add_argument("--id", required=True)
    p.add_argument("--year", type=int)
    p.add_argument("--json", action="store_true")
    p.add_argument("--output")

    p = sub.add_parser("list-reports")
    p.add_argument("--id")

    p = sub.add_parser("delete-report")
    p.add_argument("--id", required=True)
    p.add_argument("--year", type=int, required=True)

    return parser


def main():
    parser = build_parser()
    args = parser.parse_args()
    if not args.command:
        parser.print_help()
        return 2

    conn = connect(args.db)
    try:
        init_db(conn)
        if args.command == "init":
            return 0
        elif args.command == "add-company":
            add_company(conn, args.id, args.name)
        elif args.command == "update-company":
            update_company(conn, args.id, args.name)
        elif args.command == "delete-company":
            delete_company(conn, args.id)
        elif args.command == "list-companies":
            rows = list_companies(conn)
            print_table(rows, ["company_id", "name", "created_at"])
        elif args.command == "add-report":
            text = read_report_text(args)
            add_report(conn, args.id, args.year, text)
        elif args.command == "get-report":
            row = get_report(conn, args.id, args.year)
            output_report(row, args.json, args.output)
        elif args.command == "list-reports":
            rows = list_reports(conn, args.id)
            print_table(rows, ["company_id", "year", "updated_at"])
        elif args.command == "delete-report":
            delete_report(conn, args.id, args.year)
        else:
            parser.print_help()
            return 2
    except ValueError as e:
        print(str(e), file=sys.stderr)
        return 1
    finally:
        conn.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())
