<?php
$host = 'localhost';
$dbname = 'your_database';
$dbuser = 'your_user';
$dbpass = 'your_password';

$creditCardNumber = $_POST['credit_card_number'] ?? '';
$customerName = $_POST['customer_name'] ?? '';

if ($creditCardNumber === '') {
    http_response_code(400);
    exit;
}

$secret = getenv('CC_ENCRYPTION_KEY') ?: 'change_this_secret_key';
$key = hash('sha256', $secret, true);
$iv = random_bytes(16);

$ciphertext = openssl_encrypt($creditCardNumber, 'AES-256-CBC', $key, OPENSSL_RAW_DATA, $iv);
$encryptedCard = base64_encode($iv . $ciphertext);

$pdo = new PDO(
    "mysql:host=$host;dbname=$dbname;charset=utf8mb4",
    $dbuser,
    $dbpass,
    [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
    ]
);

$stmt = $pdo->prepare('INSERT INTO payments (customer_name, credit_card_number) VALUES (:customer_name, :credit_card_number)');
$stmt->execute([
    ':customer_name' => $customerName,
    ':credit_card_number' => $encryptedCard
]);

echo 'OK';
?>
