from flask import Flask, request, jsonify
import ast
import operator
import math

app = Flask(__name__)

ALLOWED_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow
}

ALLOWED_UNARY_OPERATORS = {
    ast.UAdd: operator.pos,
    ast.USub: operator.neg
}

ALLOWED_FUNCTIONS = {
    "sin": math.sin,
    "cos": math.cos,
    "tan": math.tan,
    "sqrt": math.sqrt,
    "log": math.log,
    "log10": math.log10,
    "exp": math.exp,
    "fabs": math.fabs,
    "floor": math.floor,
    "ceil": math.ceil,
    "factorial": math.factorial,
    "abs": abs,
    "round": round
}

ALLOWED_CONSTANTS = {
    "pi": math.pi,
    "e": math.e
}

def _eval_node(node):
    if isinstance(node, ast.Expression):
        return _eval_node(node.body)
    if isinstance(node, ast.Num):
        return node.n
    if isinstance(node, ast.Constant):
        if isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
            return node.value
        raise ValueError("Invalid constant")
    if isinstance(node, ast.BinOp):
        op = type(node.op)
        if op in ALLOWED_OPERATORS:
            return ALLOWED_OPERATORS[op](_eval_node(node.left), _eval_node(node.right))
        raise ValueError("Invalid operator")
    if isinstance(node, ast.UnaryOp):
        op = type(node.op)
        if op in ALLOWED_UNARY_OPERATORS:
            return ALLOWED_UNARY_OPERATORS[op](_eval_node(node.operand))
        raise ValueError("Invalid unary operator")
    if isinstance(node, ast.Call):
        if isinstance(node.func, ast.Name) and node.func.id in ALLOWED_FUNCTIONS:
            args = [_eval_node(arg) for arg in node.args]
            return ALLOWED_FUNCTIONS[node.func.id](*args)
        raise ValueError("Invalid function")
    if isinstance(node, ast.Name):
        if node.id in ALLOWED_CONSTANTS:
            return ALLOWED_CONSTANTS[node.id]
        raise ValueError("Unknown identifier")
    raise ValueError("Invalid expression")

def safe_eval(expr):
    if not isinstance(expr, str):
        raise ValueError("Expression must be a string")
    parsed = ast.parse(expr, mode="eval")
    return _eval_node(parsed)

@app.route("/calculate", methods=["GET", "POST"])
def calculate():
    expr = None
    if request.method == "POST":
        if request.is_json:
            data = request.get_json(silent=True) or {}
            expr = data.get("expression") or data.get("expr")
        if expr is None:
            expr = request.form.get("expression") or request.form.get("expr")
    if expr is None:
        expr = request.args.get("expression") or request.args.get("expr")
    if not expr:
        return jsonify(error="Expression required"), 400
    try:
        result = safe_eval(expr)
    except Exception:
        return jsonify(error="Invalid expression"), 400
    return jsonify(result=result)

if __name__ == "__main__":
    app.run()
