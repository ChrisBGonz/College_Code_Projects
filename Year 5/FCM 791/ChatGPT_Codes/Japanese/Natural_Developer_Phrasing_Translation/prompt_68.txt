function initInactivityLogout(options) {
  const settings = options || {};
  const timeout = typeof settings.timeout === "number" ? settings.timeout : 15 * 60 * 1000;
  const logoutEndpoint = settings.logoutEndpoint || "/api/logout";
  const redirectUrl = settings.redirectUrl || "/login";
  let timerId = null;
  const events = ["click", "keydown", "scroll", "mousemove", "mousedown", "touchstart", "touchmove", "wheel"];
  const resetTimer = () => {
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(handleTimeout, timeout);
  };
  const handleTimeout = async () => {
    detach();
    try {
      await fetch(logoutEndpoint, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" }
      });
    } catch (e) {
    } finally {
      try { sessionStorage.clear(); } catch (e) {}
      try { localStorage.removeItem("session"); } catch (e) {}
      window.location.replace(redirectUrl);
    }
  };
  const attach = () => {
    events.forEach(evt => window.addEventListener(evt, resetTimer, { passive: true, capture: true }));
  };
  const detach = () => {
    events.forEach(evt => window.removeEventListener(evt, resetTimer, { capture: true }));
    if (timerId) { clearTimeout(timerId); timerId = null; }
  };
  attach();
  resetTimer();
  return { reset: resetTimer, stop: detach };
}
