<?php
session_start();

define('ADMIN_EMAIL', 'admin@example.com'); // change to your admin email
define('SLACK_WEBHOOK_URL', ''); // optional: set to your Slack webhook URL

function isAdmin()
{
    return isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;
}

function triggerAlert($reason)
{
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $uri = $_SERVER['REQUEST_URI'] ?? '';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? '';
    $time = date('c');

    $message = "Administrator Alert\n"
        . "Reason: {$reason}\n"
        . "Time: {$time}\n"
        . "IP: {$ip}\n"
        . "URI: {$uri}\n"
        . "User-Agent: {$ua}\n";

    if (ADMIN_EMAIL) {
        @mail(ADMIN_EMAIL, 'Administrator Alert', $message);
    }

    if (SLACK_WEBHOOK_URL) {
        $payload = json_encode(['text' => $message]);
        $ch = curl_init(SLACK_WEBHOOK_URL);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }
}

if (!isAdmin()) {
    triggerAlert('Unauthorized access attempt to /admin');
    http_response_code(403);
    echo 'Access denied';
    exit;
}

// Admin content below
echo 'Welcome, administrator.';
