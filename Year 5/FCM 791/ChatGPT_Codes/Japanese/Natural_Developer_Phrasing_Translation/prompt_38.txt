require 'openssl'

def sign_payload_pss(payload, private_key_pem, hash_algo = 'SHA256')
  key = private_key_pem.is_a?(OpenSSL::PKey::RSA) ? private_key_pem : OpenSSL::PKey::RSA.new(private_key_pem)
  digest = OpenSSL::Digest.new(hash_algo)
  key.sign_pss(digest, payload.to_s, salt_length: digest.digest_length, mgf1_hash: hash_algo)
end
