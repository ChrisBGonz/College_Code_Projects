#include <sys/types.h>
#include <sys/socket.h>
#include <cstddef>
#include <cstdint>
#include <new>
#include <limits>

static bool recvAll(int sockfd, void* buf, std::size_t len)
{
    unsigned char* p = static_cast<unsigned char*>(buf);
    std::size_t total = 0;
    while (total < len) {
        ssize_t r = recv(sockfd, p + total, len - total, 0);
        if (r <= 0) {
            return false;
        }
        total += static_cast<std::size_t>(r);
    }
    return true;
}

char* allocateLargeBufferFromSocket(int sockfd, std::size_t& outSize)
{
    outSize = 0;
    unsigned char sizeBuf[8];
    if (!recvAll(sockfd, sizeBuf, sizeof(sizeBuf))) {
        return nullptr;
    }
    uint64_t size = 0;
    for (int i = 0; i < 8; ++i) {
        size = (size << 8) | sizeBuf[i];
    }
    if (size == 0 || size > static_cast<uint64_t>(std::numeric_limits<std::size_t>::max())) {
        return nullptr;
    }
    outSize = static_cast<std::size_t>(size);
    char* buffer = new (std::nothrow) char[outSize];
    if (!buffer) {
        outSize = 0;
    }
    return buffer;
}
