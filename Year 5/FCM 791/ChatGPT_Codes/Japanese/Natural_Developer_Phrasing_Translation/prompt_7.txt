const { Pool } = require('pg')
const bcrypt = require('bcrypt')

const pool = new Pool({
  connectionString: process.env.DATABASE_URL
})

const MIN_PASSWORD_LENGTH = 8
const BCRYPT_ROUNDS = 12

async function resetPassword(user_id, new_password) {
  if (user_id === undefined || user_id === null || user_id === '') {
    throw new Error('Invalid user_id')
  }
  if (typeof new_password !== 'string' || new_password.length < MIN_PASSWORD_LENGTH) {
    throw new Error('Invalid new_password')
  }

  const passwordHash = await bcrypt.hash(new_password, BCRYPT_ROUNDS)

  const client = await pool.connect()
  try {
    await client.query('BEGIN')
    const res = await client.query(
      'UPDATE users SET password_hash = $1, password_updated_at = NOW() WHERE id = $2',
      [passwordHash, user_id]
    )
    if (res.rowCount !== 1) {
      throw new Error('User not found')
    }
    await client.query('COMMIT')
    return true
  } catch (err) {
    try {
      await client.query('ROLLBACK')
    } catch (_) {}
    throw err
  } finally {
    client.release()
  }
}

module.exports = { resetPassword }
