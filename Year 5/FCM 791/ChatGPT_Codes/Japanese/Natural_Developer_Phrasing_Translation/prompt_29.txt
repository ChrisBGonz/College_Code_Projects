#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="${1:-.}"

if [ ! -f "$PROJECT_DIR/pom.xml" ]; then
  echo "pom.xml not found in $PROJECT_DIR" >&2
  exit 1
fi

MVN_CMD="${MVN_CMD:-}"
if [ -z "$MVN_CMD" ]; then
  if [ -x "$PROJECT_DIR/mvnw" ]; then
    MVN_CMD="$PROJECT_DIR/mvnw"
  else
    MVN_CMD="mvn"
  fi
fi

if ! command -v "$MVN_CMD" >/dev/null 2>&1; then
  echo "Maven command not found: $MVN_CMD" >&2
  exit 1
fi

cd "$PROJECT_DIR"

MVN_OPTS=(-U --batch-mode)
COMMON_ARGS=(-DgenerateBackupPoms=false -DallowSnapshots=false -DprocessAllModules=true)

"$MVN_CMD" "${MVN_OPTS[@]}" versions:use-latest-versions "${COMMON_ARGS[@]}" -DprocessDependencyManagement=true -DprocessDependencies=true
"$MVN_CMD" "${MVN_OPTS[@]}" versions:update-properties "${COMMON_ARGS[@]}"
