from flask import Flask, request, render_template_string
import yaml
import os

app = Flask(__name__)
app.secret_key = os.environ.get("FLASK_SECRET_KEY", "change-me")
app.config["MAX_CONTENT_LENGTH"] = 1 * 1024 * 1024

TEMPLATE = """
<!doctype html>
<html lang="ja">
<head>
<meta charset="utf-8">
<title>プロフィール設定アップロード</title>
</head>
<body>
<h1>プロフィール設定を YAML でアップロード</h1>
{% if error %}
<p style="color:red;">{{ error }}</p>
{% endif %}
<form method="post" enctype="multipart/form-data">
  <input type="file" name="file" accept=".yml,.yaml" required>
  <button type="submit">アップロード</button>
</form>
{% if profile_yaml %}
<h2>読み込まれたプロフィール設定</h2>
<pre>{{ profile_yaml }}</pre>
{% endif %}
</body>
</html>
"""

def allowed_file(filename):
    ext = os.path.splitext(filename)[1].lower()
    return ext in [".yml", ".yaml"]

@app.route("/", methods=["GET", "POST"])
def upload():
    error = None
    profile_yaml = None
    if request.method == "POST":
        if "file" not in request.files:
            error = "ファイルが選択されていません。"
        else:
            file = request.files["file"]
            if file.filename == "":
                error = "ファイルが選択されていません。"
            elif not allowed_file(file.filename):
                error = "YAMLファイル（.yml/.yaml）のみアップロード可能です。"
            else:
                try:
                    content = file.read().decode("utf-8", errors="replace")
                    data = yaml.safe_load(content)
                    if not isinstance(data, dict):
                        error = "YAMLのルートはマッピング形式である必要があります。"
                    else:
                        profile_yaml = yaml.safe_dump(data, sort_keys=False, allow_unicode=True)
                except Exception as e:
                    error = f"YAMLの読み込みに失敗しました: {e}"
    return render_template_string(TEMPLATE, error=error, profile_yaml=profile_yaml)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
