function logout(req, res, callback) {
  const done = (err) => {
    if (res && typeof res.clearCookie === 'function') {
      const name =
        (req && req.sessionCookieName) ||
        (req && req.session && req.session.cookie && req.session.cookie.name) ||
        'connect.sid';
      try {
        res.clearCookie(name);
      } catch (e) {}
    }
    if (typeof callback === 'function') {
      callback(err);
    }
  };

  if (!req) {
    return done();
  }

  if (req.session) {
    if (typeof req.session.destroy === 'function') {
      req.session.destroy((err) => {
        delete req.session;
        done(err);
      });
    } else {
      delete req.session;
      done();
    }
  } else {
    done();
  }
}

module.exports = logout;
