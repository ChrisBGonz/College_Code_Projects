using System;
using System.Security.Cryptography;
using System.Text;

class Program
{
    static void Main()
    {
        using RSA rsa = RSA.Create(2048);

        byte[] privateKey = rsa.ExportRSAPrivateKey();
        byte[] publicKey = rsa.ExportRSAPublicKey();

        string privatePem = ToPem("RSA PRIVATE KEY", privateKey);
        string publicPem = ToPem("RSA PUBLIC KEY", publicKey);

        Console.WriteLine(privatePem);
        Console.WriteLine(publicPem);
    }

    static string ToPem(string label, byte[] data)
    {
        string base64 = Convert.ToBase64String(data);
        StringBuilder sb = new StringBuilder();
        sb.AppendLine($"-----BEGIN {label}-----");
        for (int i = 0; i < base64.Length; i += 64)
        {
            int len = Math.Min(64, base64.Length - i);
            sb.AppendLine(base64.Substring(i, len));
        }
        sb.AppendLine($"-----END {label}-----");
        return sb.ToString();
    }
}
