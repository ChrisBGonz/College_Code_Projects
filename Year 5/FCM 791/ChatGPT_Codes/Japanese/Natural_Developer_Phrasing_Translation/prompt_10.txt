const express = require('express');
const path = require('path');
const fs = require('fs').promises;

const app = express();
const USERS_BASE_DIR = path.join(__dirname, 'users');

function isValidFolderName(name) {
  return /^[a-zA-Z0-9_-]+$/.test(name);
}

async function listFilesRecursive(dir, root) {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  let files = [];
  for (const entry of entries) {
    const fullPath = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      files = files.concat(await listFilesRecursive(fullPath, root));
    } else if (entry.isFile()) {
      files.push(path.relative(root, fullPath));
    }
  }
  return files;
}

app.get('/api/vault/:folder_name?', async (req, res) => {
  const folderName = req.params.folder_name || req.query.folder_name;
  if (!folderName) {
    return res.status(400).json({ error: 'folder_name is required' });
  }
  if (!isValidFolderName(folderName)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }
  const vaultDir = path.join(USERS_BASE_DIR, folderName, 'private_vault');
  const resolvedVault = path.resolve(vaultDir);
  const resolvedBase = path.resolve(USERS_BASE_DIR);
  if (!resolvedVault.startsWith(resolvedBase + path.sep)) {
    return res.status(400).json({ error: 'Invalid path' });
  }
  try {
    const stats = await fs.stat(resolvedVault);
    if (!stats.isDirectory()) {
      return res.status(404).json({ error: 'Vault not found' });
    }
    const files = await listFilesRecursive(resolvedVault, resolvedVault);
    res.json(files);
  } catch (err) {
    if (err.code === 'ENOENT') {
      return res.status(404).json({ error: 'Vault not found' });
    }
    console.error(err);
    res.status(500).json({ error: 'Internal server error' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
