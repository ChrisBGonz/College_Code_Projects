import sys

def main():
    if len(sys.argv) != 2:
        print("Usage: python count_lines.py <file_path>")
        sys.exit(1)

    path = sys.argv[1]

    with open(path, "rb") as f:
        data = f.read()

    count = data.count(b"\n")
    if data and data[-1] != ord("\n"):
        count += 1

    print(count)

if __name__ == "__main__":
    main()
