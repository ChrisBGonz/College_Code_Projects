package config

import (
	"errors"
	"os"
)

func LoadConfigFromEnv(envVar string) ([]byte, error) {
	if envVar == "" {
		return nil, errors.New("environment variable name is empty")
	}
	path, ok := os.LookupEnv(envVar)
	if !ok || path == "" {
		return nil, errors.New("environment variable not set: " + envVar)
	}
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	return data, nil
}
