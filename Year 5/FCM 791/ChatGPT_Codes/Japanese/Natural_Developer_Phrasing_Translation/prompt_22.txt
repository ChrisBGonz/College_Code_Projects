#!/usr/bin/env bash
set -euo pipefail

TOOL_NAME="lynis"
API_URL="${1:-https://api.github.com/repos/CISOfy/lynis/releases/latest}"
ARCHIVE_PATTERN="${ARCHIVE_PATTERN:-\\.tar\\.gz$}"

err(){ echo "Error: $*" >&2; exit 1; }

for cmd in curl tar grep cut head mktemp find; do
  command -v "$cmd" >/dev/null 2>&1 || err "Required command not found: $cmd"
done

json=$(curl -fsSL "$API_URL") || err "Failed to fetch release info from $API_URL"

version=$(echo "$json" | grep -m1 '"tag_name":' | cut -d'"' -f4)
download_url=$(echo "$json" | grep '"browser_download_url":' | cut -d'"' -f4 | grep -E "$ARCHIVE_PATTERN" | head -n1)

[ -n "$download_url" ] || err "Download URL not found"

tmpdir=$(mktemp -d)
cleanup(){ rm -rf "$tmpdir"; }
trap cleanup EXIT

archive="$tmpdir/${TOOL_NAME}.tar.gz"
curl -fsSL "$download_url" -o "$archive" || err "Failed to download $download_url"

tar -xzf "$archive" -C "$tmpdir" || err "Failed to extract archive"

src_dir=$(find "$tmpdir" -maxdepth 1 -type d -name "${TOOL_NAME}*" | head -n1)
[ -n "$src_dir" ] || err "Extracted directory not found"

prefix="/opt/$TOOL_NAME"
bin_link="/usr/local/bin/$TOOL_NAME"

SUDO=""
if [ "$(id -u)" -ne 0 ]; then
  command -v sudo >/dev/null 2>&1 || err "sudo not available; run as root"
  SUDO="sudo"
fi

$SUDO rm -rf "$prefix"
$SUDO mkdir -p "$prefix"
$SUDO cp -a "$src_dir"/. "$prefix"/

if [ -x "$prefix/$TOOL_NAME" ]; then
  $SUDO ln -sf "$prefix/$TOOL_NAME" "$bin_link"
elif [ -x "$prefix/bin/$TOOL_NAME" ]; then
  $SUDO ln -sf "$prefix/bin/$TOOL_NAME" "$bin_link"
fi

echo "$TOOL_NAME installed ${version:-}"
