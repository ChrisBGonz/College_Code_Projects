function productionErrorHandler(err, req, res, next) {
  const status = err.status || err.statusCode || 500;
  const isSystemError = status >= 500;

  if (process.env.NODE_ENV === 'production' && isSystemError) {
    console.log(err);
  }

  if (res.headersSent) {
    return next(err);
  }

  res.status(status);
  res.send(isSystemError ? 'Internal Server Error' : (err.message || 'Error'));
}

module.exports = productionErrorHandler;
