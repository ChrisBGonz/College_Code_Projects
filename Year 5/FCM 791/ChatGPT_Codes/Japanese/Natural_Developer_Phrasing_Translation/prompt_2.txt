const express = require('express');

const app = express();
app.use(express.json());

const users = new Map();

function isValidEmail(email) {
  if (typeof email !== 'string') return false;
  const trimmed = email.trim();
  if (!trimmed) return false;
  const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return re.test(trimmed);
}

function isValidDisplayName(name) {
  if (typeof name !== 'string') return false;
  const trimmed = name.trim();
  return trimmed.length > 0 && trimmed.length <= 100;
}

async function updateProfileHandler(req, res) {
  try {
    const userId = (req.user && req.user.id) || req.body.userId;
    if (!userId) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    const { email, displayName } = req.body;

    if (email === undefined && displayName === undefined) {
      return res.status(400).json({ error: 'email or displayName is required' });
    }

    if (email !== undefined && !isValidEmail(email)) {
      return res.status(400).json({ error: 'Invalid email' });
    }

    if (displayName !== undefined && !isValidDisplayName(displayName)) {
      return res.status(400).json({ error: 'Invalid displayName' });
    }

    const user = users.get(userId);
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    if (email !== undefined) {
      user.email = email.trim();
    }
    if (displayName !== undefined) {
      user.displayName = displayName.trim();
    }

    users.set(userId, user);

    return res.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        displayName: user.displayName
      }
    });
  } catch (err) {
    return res.status(500).json({ error: 'Internal server error' });
  }
}

app.post('/api/user/update-profile', updateProfileHandler);

module.exports = { app, updateProfileHandler, users };

if (require.main === module) {
  const port = process.env.PORT || 3000;
  app.listen(port, () => {
    console.log(`Server running on port ${port}`);
  });
}
