<?php
class Preferences {
    public $theme = 'light';
    public $language = 'ja';
    public $itemsPerPage = 10;
}

$cookieName = 'user_preferences';
$prefs = new Preferences();

if (isset($_COOKIE[$cookieName])) {
    $prefs = @unserialize($_COOKIE[$cookieName], ['allowed_classes' => ['Preferences']]);
    if (!($prefs instanceof Preferences)) {
        $prefs = new Preferences();
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme'])) {
        $prefs->theme = $_POST['theme'];
    }
    if (isset($_POST['language'])) {
        $prefs->language = $_POST['language'];
    }
    if (isset($_POST['itemsPerPage'])) {
        $prefs->itemsPerPage = (int)$_POST['itemsPerPage'];
    }
    setcookie($cookieName, serialize($prefs), time() + 60 * 60 * 24 * 365, "/");
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}
?>
<!doctype html>
<html>
<head>
<meta charset="UTF-8">
<title>Preferences</title>
</head>
<body>
<form method="post">
<label>Theme:
    <select name="theme">
        <option value="light"<?php echo $prefs->theme === 'light' ? ' selected' : ''; ?>>Light</option>
        <option value="dark"<?php echo $prefs->theme === 'dark' ? ' selected' : ''; ?>>Dark</option>
    </select>
</label><br>
<label>Language:
    <select name="language">
        <option value="ja"<?php echo $prefs->language === 'ja' ? ' selected' : ''; ?>>Japanese</option>
        <option value="en"<?php echo $prefs->language === 'en' ? ' selected' : ''; ?>>English</option>
    </select>
</label><br>
<label>Items per page:
    <input type="number" name="itemsPerPage" value="<?php echo htmlspecialchars((string)$prefs->itemsPerPage, ENT_QUOTES, 'UTF-8'); ?>">
</label><br>
<button type="submit">Save</button>
</form>
</body>
</html>
