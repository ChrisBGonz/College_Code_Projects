using System;
using System.IO;
using System.Text.Json;
using System.Threading.Tasks;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
}

public class AuditLogEntry
{
    public string UserId { get; set; }
    public string Action { get; set; }
    public DateTime TimestampUtc { get; set; }
    public string Detail { get; set; }
}

public interface IAuditLogger
{
    Task LogAsync(AuditLogEntry entry);
}

public class FileAuditLogger : IAuditLogger
{
    private readonly string _path;

    public FileAuditLogger(string path)
    {
        _path = path;
    }

    public async Task LogAsync(AuditLogEntry entry)
    {
        var line = JsonSerializer.Serialize(entry) + Environment.NewLine;
        await File.AppendAllTextAsync(_path, line);
    }
}

public class MfaService
{
    private readonly IAuditLogger _auditLogger;

    public MfaService(IAuditLogger auditLogger)
    {
        _auditLogger = auditLogger;
    }

    public async Task UpdateMfaAsync(string userId, MfaSettings newSettings)
    {
        // Update MFA settings in persistent store here

        await _auditLogger.LogAsync(new AuditLogEntry
        {
            UserId = userId,
            Action = "MFA_SETTINGS_CHANGED",
            TimestampUtc = DateTime.UtcNow,
            Detail = $"Enabled={newSettings.Enabled}, Method={newSettings.Method}"
        });
    }
}
