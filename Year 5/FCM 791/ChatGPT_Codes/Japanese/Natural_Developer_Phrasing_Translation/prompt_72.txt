const fs = require('fs');
const http = require('http');
const https = require('https');
const { URL } = require('url');
const { pipeline } = require('stream');
const { promisify } = require('util');
const unzipper = require('unzipper');

const pipelineAsync = promisify(pipeline);

function fetchStream(url, redirects = 5) {
  return new Promise((resolve, reject) => {
    const u = new URL(url);
    const lib = u.protocol === 'https:' ? https : http;
    const req = lib.get(u, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        if (redirects <= 0) {
          reject(new Error('Too many redirects'));
          return;
        }
        const nextUrl = new URL(res.headers.location, u).toString();
        resolve(fetchStream(nextUrl, redirects - 1));
        return;
      }
      if (res.statusCode !== 200) {
        res.resume();
        reject(new Error(`Download failed with status ${res.statusCode}`));
        return;
      }
      resolve(res);
    });
    req.on('error', reject);
  });
}

async function downloadAndExtract(zipUrl, destDir = '/app') {
  await fs.promises.mkdir(destDir, { recursive: true });
  const res = await fetchStream(zipUrl);
  await pipelineAsync(res, unzipper.Extract({ path: destDir }));
}

module.exports = { downloadAndExtract };
