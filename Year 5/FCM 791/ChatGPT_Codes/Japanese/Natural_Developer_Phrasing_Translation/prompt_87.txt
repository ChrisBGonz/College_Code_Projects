#!/usr/bin/env python3
import argparse
import os
import sys
import time
from collections import deque

def follow(file):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if line:
            yield line
        else:
            time.sleep(0.1)

def monitor(path, threshold=10, window=60):
    times = deque()
    warned = False
    with open(path, 'r', errors='ignore') as f:
        for line in follow(f):
            now = time.time()
            while times and now - times[0] > window:
                times.popleft()
            if 'Failed Login' in line:
                times.append(now)
            if len(times) > threshold:
                if not warned:
                    print(f'WARNING: {len(times)} Failed Login messages in the last {window} seconds', flush=True)
                    warned = True
            else:
                warned = False

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('logfile', help='Path to log file')
    args = parser.parse_args()
    if not os.path.isfile(args.logfile):
        print('Log file not found', file=sys.stderr)
        sys.exit(1)
    try:
        monitor(args.logfile)
    except KeyboardInterrupt:
        pass

if __name__ == '__main__':
    main()
