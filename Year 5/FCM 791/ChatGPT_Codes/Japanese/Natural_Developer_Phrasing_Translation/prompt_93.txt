using System;
using System.Data.SqlClient;

string connectionString = "your_connection_string";

using (var connection = new SqlConnection(connectionString))
{
    connection.Open();
    using (var transaction = connection.BeginTransaction())
    {
        try
        {
            using (var withdrawCmd = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @fromId", connection, transaction))
            {
                withdrawCmd.Parameters.AddWithValue("@amount", 100m);
                withdrawCmd.Parameters.AddWithValue("@fromId", "A");
                if (withdrawCmd.ExecuteNonQuery() != 1)
                    throw new InvalidOperationException("Account A not found");
            }

            using (var depositCmd = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @toId", connection, transaction))
            {
                depositCmd.Parameters.AddWithValue("@amount", 100m);
                depositCmd.Parameters.AddWithValue("@toId", "B");
                if (depositCmd.ExecuteNonQuery() != 1)
                    throw new InvalidOperationException("Account B not found");
            }

            transaction.Commit();
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }
}
