import os
import sys
import time
import hmac
import base64
import struct
import hashlib

def _b32_decode(s):
    s = s.strip().replace(' ', '').upper()
    if not s:
        raise ValueError("empty secret")
    pad = '=' * ((8 - len(s) % 8) % 8)
    return base64.b32decode(s + pad)

def generate_totp(secret, t=None, step=30, digits=6):
    if t is None:
        t = int(time.time())
    counter = int(t // step)
    key = _b32_decode(secret)
    msg = struct.pack('>Q', counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    o = h[-1] & 0x0f
    code = (struct.unpack(">I", h[o:o+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(code, secret, window=1, step=30, digits=6):
    code = code.strip()
    if len(code) != digits or not code.isdigit():
        return False
    now = int(time.time())
    try:
        for w in range(-window, window + 1):
            if generate_totp(secret, t=now + w * step, step=step, digits=digits) == code:
                return True
    except Exception:
        return False
    return False

def main():
    if len(sys.argv) > 1:
        secret = sys.argv[1]
    else:
        secret = os.getenv("TOTP_SECRET")
        if not secret:
            secret = input("Secret: ").strip()
    code = input("TOTP code: ").strip()
    if verify_totp(code, secret):
        print("VERIFIED")
        return 0
    else:
        print("INVALID")
        return 1

if __name__ == "__main__":
    sys.exit(main())
