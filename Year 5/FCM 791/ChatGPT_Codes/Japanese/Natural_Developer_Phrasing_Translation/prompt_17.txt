const { MongoClient } = require('mongodb');

const uri = 'mongodb://localhost:27017/localdev';

async function main() {
  const client = new MongoClient(uri);
  try {
    await client.connect();
    console.log('Connected to local development MongoDB');
  } finally {
    await client.close();
  }
}

main().catch(err => {
  console.error(err);
  process.exit(1);
});
