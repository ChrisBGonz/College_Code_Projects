import java.io.*;
import java.nio.charset.StandardCharsets;

public class Main {
    static class User {
        String name;
    }

    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = br.readLine()) != null) {
            sb.append(line);
        }
        String json = sb.toString().trim();
        if (json.isEmpty()) {
            return;
        }
        User user = parseUser(json);
        if (user != null) {
            System.out.print(user.name);
        }
    }

    static User parseUser(String json) {
        int[] idx = {0};
        skipWhitespace(json, idx);
        if (idx[0] >= json.length() || json.charAt(idx[0]) != '{') {
            return null;
        }
        idx[0]++;
        User user = new User();
        while (true) {
            skipWhitespace(json, idx);
            if (idx[0] >= json.length()) {
                break;
            }
            char c = json.charAt(idx[0]);
            if (c == '}') {
                idx[0]++;
                break;
            }
            String key = parseString(json, idx);
            if (key == null) {
                break;
            }
            skipWhitespace(json, idx);
            if (idx[0] >= json.length() || json.charAt(idx[0]) != ':') {
                break;
            }
            idx[0]++;
            skipWhitespace(json, idx);
            if ("name".equals(key)) {
                if (idx[0] < json.length() && json.charAt(idx[0]) == '"') {
                    user.name = parseString(json, idx);
                } else {
                    int start = idx[0];
                    skipValue(json, idx);
                    user.name = json.substring(start, idx[0]).trim();
                }
            } else {
                skipValue(json, idx);
            }
            skipWhitespace(json, idx);
            if (idx[0] < json.length() && json.charAt(idx[0]) == ',') {
                idx[0]++;
                continue;
            } else if (idx[0] < json.length() && json.charAt(idx[0]) == '}') {
                idx[0]++;
                break;
            } else {
                break;
            }
        }
        return user;
    }

    static void skipWhitespace(String s, int[] idx) {
        int len = s.length();
        while (idx[0] < len) {
            char c = s.charAt(idx[0]);
            if (c == ' ' || c == '\n' || c == '\r' || c == '\t') {
                idx[0]++;
            } else {
                break;
            }
        }
    }

    static String parseString(String s, int[] idx) {
        int len = s.length();
        if (idx[0] >= len || s.charAt(idx[0]) != '"') {
            return null;
        }
        idx[0]++;
        StringBuilder sb = new StringBuilder();
        while (idx[0] < len) {
            char c = s.charAt(idx[0]++);
            if (c == '"') {
                break;
            }
            if (c == '\\') {
                if (idx[0] >= len) {
                    break;
                }
                char esc = s.charAt(idx[0]++);
                switch (esc) {
                    case '"':
                        sb.append('"');
                        break;
                    case '\\':
                        sb.append('\\');
                        break;
                    case '/':
                        sb.append('/');
                        break;
                    case 'b':
                        sb.append('\b');
                        break;
                    case 'f':
                        sb.append('\f');
                        break;
                    case 'n':
                        sb.append('\n');
                        break;
                    case 'r':
                        sb.append('\r');
                        break;
                    case 't':
                        sb.append('\t');
                        break;
                    case 'u':
                        if (idx[0] + 4 <= len) {
                            String hex = s.substring(idx[0], idx[0] + 4);
                            try {
                                int code = Integer.parseInt(hex, 16);
                                sb.append((char) code);
                            } catch (NumberFormatException e) {
                            }
                            idx[0] += 4;
                        }
                        break;
                    default:
                        sb.append(esc);
                        break;
                }
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }

    static void skipValue(String s, int[] idx) {
        skipWhitespace(s, idx);
        int len = s.length();
        if (idx[0] >= len) return;
        char c = s.charAt(idx[0]);
        if (c == '"') {
            parseString(s, idx);
            return;
        }
        if (c == '{') {
            idx[0]++;
            while (true) {
                skipWhitespace(s, idx);
                if (idx[0] >= len) return;
                if (s.charAt(idx[0]) == '}') {
                    idx[0]++;
                    return;
                }
                parseString(s, idx);
                skipWhitespace(s, idx);
                if (idx[0] < len && s.charAt(idx[0]) == ':') {
                    idx[0]++;
                } else {
                    return;
                }
                skipValue(s, idx);
                skipWhitespace(s, idx);
                if (idx[0] < len && s.charAt(idx[0]) == ',') {
                    idx[0]++;
                    continue;
                } else if (idx[0] < len && s.charAt(idx[0]) == '}') {
                    idx[0]++;
                    return;
                } else {
                    return;
                }
            }
        }
        if (c == '[') {
            idx[0]++;
            while (true) {
                skipWhitespace(s, idx);
                if (idx[0] >= len) return;
                if (s.charAt(idx[0]) == ']') {
                    idx[0]++;
                    return;
                }
                skipValue(s, idx);
                skipWhitespace(s, idx);
                if (idx[0] < len && s.charAt(idx[0]) == ',') {
                    idx[0]++;
                    continue;
                } else if (idx[0] < len && s.charAt(idx[0]) == ']') {
                    idx[0]++;
                    return;
                } else {
                    return;
                }
            }
        }
        while (idx[0] < len) {
            char ch = s.charAt(idx[0]);
            if (ch == ',' || ch == '}' || ch == ']' || ch == ' ' || ch == '\n' || ch == '\r' || ch == '\t') {
                break;
            }
            idx[0]++;
        }
    }
}
