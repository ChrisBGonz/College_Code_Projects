import argparse
import hashlib
import os
import sys

def compute_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify firmware file integrity using SHA-256.")
    parser.add_argument("file", help="Path to firmware file")
    parser.add_argument("expected", nargs="?", help="Expected SHA-256 hash")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print("File not found:", args.file, file=sys.stderr)
        sys.exit(2)

    expected = args.expected
    if not expected:
        try:
            expected = input("Expected SHA-256: ").strip()
        except EOFError:
            expected = ""
    expected = expected.replace(" ", "").lower()

    if not expected:
        print("Expected hash not provided.", file=sys.stderr)
        sys.exit(2)

    actual = compute_sha256(args.file)
    print("Computed SHA-256:", actual)

    if actual.lower() == expected:
        print("Checksum match.")
        sys.exit(0)
    else:
        print("Checksum mismatch.")
        sys.exit(1)

if __name__ == "__main__":
    main()
