import argparse
import logging
from logging.handlers import TimedRotatingFileHandler
import os
import sys
import time

def setup_logger(log_file, backup_count):
    log_dir = os.path.dirname(os.path.abspath(log_file))
    if log_dir and not os.path.exists(log_dir):
        os.makedirs(log_dir, exist_ok=True)
    logger = logging.getLogger("rotating_logger")
    logger.setLevel(logging.INFO)
    handler = TimedRotatingFileHandler(
        log_file,
        when="H",
        interval=24,
        backupCount=backup_count,
        encoding="utf-8"
    )
    formatter = logging.Formatter("%(asctime)s [%(levelname)s] %(message)s")
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    logger.propagate = False
    return logger

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("log_file", nargs="?", default="app.log")
    parser.add_argument("--backup-count", type=int, default=7)
    parser.add_argument("--heartbeat", type=float, default=60.0)
    args = parser.parse_args()

    logger = setup_logger(args.log_file, args.backup_count)

    if not sys.stdin.isatty():
        for line in sys.stdin:
            logger.info(line.rstrip("\n"))
    else:
        try:
            while True:
                logger.info("log rotation active")
                time.sleep(args.heartbeat)
        except KeyboardInterrupt:
            pass

if __name__ == "__main__":
    main()
