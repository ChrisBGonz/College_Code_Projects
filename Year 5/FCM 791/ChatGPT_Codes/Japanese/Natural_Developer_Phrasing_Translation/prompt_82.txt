const debugRequestLogger = (req, res, next) => {
  const chunks = [];
  const url = req.originalUrl || req.url;
  let logged = false;

  const log = rawBody => {
    if (logged) return;
    logged = true;
    console.log('----- Incoming Request -----');
    console.log('Method:', req.method);
    console.log('URL:', url);
    console.log('Headers:', req.headers);
    console.log('Body length:', rawBody.length);
    console.log('Body (raw Buffer):', rawBody);
    console.log('Body (base64):', rawBody.toString('base64'));
    console.log('Body (utf8):', rawBody.toString('utf8'));
  };

  req.on('data', chunk => {
    chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk));
  });

  req.on('end', () => {
    const rawBody = Buffer.concat(chunks);
    req.rawBody = rawBody;
    log(rawBody);
  });

  req.on('aborted', () => {
    const rawBody = Buffer.concat(chunks);
    req.rawBody = rawBody;
    log(rawBody);
  });

  req.on('error', err => {
    console.error('Request stream error:', err);
    const rawBody = Buffer.concat(chunks);
    req.rawBody = rawBody;
    log(rawBody);
  });

  next();
};

module.exports = debugRequestLogger;
