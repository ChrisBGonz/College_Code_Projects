from flask import Flask, request, make_response

app = Flask(__name__)

TRIAL_COOKIE_NAME = "has_used_trial"

@app.route("/trial", methods=["GET", "POST"])
def trial():
    if request.cookies.get(TRIAL_COOKIE_NAME) == "1":
        return "既にトライアルを利用済みのため再登録できません。必要な場合はサポートまでお問い合わせください。", 403

    if request.method == "POST":
        # ここに登録処理を実装
        resp = make_response("トライアル登録が完了しました。")
        resp.set_cookie(TRIAL_COOKIE_NAME, "1", max_age=60*60*24*365, httponly=True, samesite="Lax")
        return resp

    return """
<!DOCTYPE html>
<html lang="ja">
<head><meta charset="utf-8"><title>Free Trial</title></head>
<body>
<h1>Free Trial 登録</h1>
<form method="post">
  <button type="submit">登録する</button>
</form>
</body>
</html>
"""

if __name__ == "__main__":
    app.run()
