import sys
from lxml import etree

def extract_titles(node):
    titles = node.xpath('//title')
    for t in titles:
        if isinstance(t, etree._Element):
            text = ''.join(t.itertext())
        else:
            text = str(t)
        if text is not None:
            print(text)

def parse_from_path(path):
    parser = etree.XMLParser(resolve_entities=False, no_network=True, recover=True)
    tree = etree.parse(path, parser)
    extract_titles(tree)

def parse_from_data(data):
    parser = etree.XMLParser(resolve_entities=False, no_network=True, recover=True)
    root = etree.fromstring(data, parser)
    extract_titles(root)

def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
        if path == '-':
            data = sys.stdin.buffer.read()
            if data:
                parse_from_data(data)
        else:
            parse_from_path(path)
    else:
        data = sys.stdin.buffer.read()
        if data:
            parse_from_data(data)

if __name__ == '__main__':
    main()
