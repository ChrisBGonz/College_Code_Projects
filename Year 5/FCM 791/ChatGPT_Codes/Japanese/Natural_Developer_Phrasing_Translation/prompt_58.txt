function setupMultiStepFormStorage(formSelector, storageKey) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;
  storageKey = storageKey || 'multiStepFormDraft';

  function shouldSkip(el) {
    if (!el.name || el.disabled) return true;
    var t = (el.type || '').toLowerCase();
    return t === 'submit' || t === 'button' || t === 'reset' || t === 'file' || t === 'image';
  }

  function collectData() {
    var data = {};
    Array.prototype.forEach.call(form.elements, function (el) {
      if (shouldSkip(el)) return;
      var t = (el.type || '').toLowerCase();
      if (t === 'checkbox') {
        if (!data[el.name]) data[el.name] = [];
        if (el.checked) data[el.name].push(el.value);
      } else if (t === 'radio') {
        if (el.checked) data[el.name] = el.value;
      } else if (t === 'select-multiple') {
        var vals = [];
        Array.prototype.forEach.call(el.options, function (o) {
          if (o.selected) vals.push(o.value);
        });
        data[el.name] = vals;
      } else {
        data[el.name] = el.value;
      }
    });
    return data;
  }

  function saveData() {
    try {
      localStorage.setItem(storageKey, JSON.stringify(collectData()));
    } catch (e) {}
  }

  function restoreData() {
    var raw = localStorage.getItem(storageKey);
    if (!raw) return;
    var data;
    try {
      data = JSON.parse(raw);
    } catch (e) {
      return;
    }
    Object.keys(data).forEach(function (name) {
      var value = data[name];
      var fields = form.elements[name];
      if (!fields) return;
      var elements = fields.tagName ? [fields] : Array.prototype.slice.call(fields);
      elements.forEach(function (el) {
        var t = (el.type || '').toLowerCase();
        if (t === 'checkbox') {
          if (Array.isArray(value)) {
            el.checked = value.indexOf(el.value) !== -1;
          } else {
            el.checked = !!value;
          }
        } else if (t === 'radio') {
          el.checked = el.value === value;
        } else if (t === 'select-multiple') {
          if (Array.isArray(value)) {
            Array.prototype.forEach.call(el.options, function (o) {
              o.selected = value.indexOf(o.value) !== -1;
            });
          }
        } else {
          if (value !== null && value !== undefined) el.value = value;
        }
      });
    });
  }

  restoreData();

  var timer;
  function scheduleSave() {
    clearTimeout(timer);
    timer = setTimeout(saveData, 150);
  }

  form.addEventListener('input', scheduleSave);
  form.addEventListener('change', scheduleSave);
  form.addEventListener('submit', function () {
    localStorage.removeItem(storageKey);
  });

  return {
    save: saveData,
    restore: restoreData,
    clear: function () {
      localStorage.removeItem(storageKey);
    }
  };
}
