package main

import (
	"io"
	"net/http"
)

func CallMicroserviceLoop(url string) ([]byte, error) {
	client := &http.Client{}
	for {
		resp, err := client.Get(url)
		if err != nil {
			continue
		}
		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			io.Copy(io.Discard, resp.Body)
			resp.Body.Close()
			continue
		}
		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			continue
		}
		return body, nil
	}
}

func main() {}
