'use strict';

const cookie = require('cookie');
const cookieEncryption = require('cookie-encryption');

function detectOrder(encFn, decFn, secret, options) {
  const testValue = 'cookie-encryption-test';
  try {
    const enc = encFn(testValue, secret, options);
    const dec = decFn(enc, secret, options);
    if (String(dec) === testValue) return 'value-first';
  } catch (e) {}
  try {
    const enc = encFn(secret, testValue, options);
    const dec = decFn(secret, enc, options);
    if (String(dec) === testValue) return 'secret-first';
  } catch (e) {}
  return 'value-first';
}

function createCrypto(secret, options) {
  let instance;
  if (typeof cookieEncryption === 'function') {
    try {
      instance = cookieEncryption(secret, options);
    } catch (e) {}
    if (instance && typeof instance.encrypt === 'function' && typeof instance.decrypt === 'function') {
      return { encrypt: instance.encrypt.bind(instance), decrypt: instance.decrypt.bind(instance) };
    }
    try {
      instance = new cookieEncryption(secret, options);
    } catch (e) {}
    if (instance && typeof instance.encrypt === 'function' && typeof instance.decrypt === 'function') {
      return { encrypt: instance.encrypt.bind(instance), decrypt: instance.decrypt.bind(instance) };
    }
  }
  if (cookieEncryption && typeof cookieEncryption.encrypt === 'function' && typeof cookieEncryption.decrypt === 'function') {
    const order = detectOrder(cookieEncryption.encrypt, cookieEncryption.decrypt, secret, options);
    return {
      encrypt: function (value) {
        return order === 'secret-first'
          ? cookieEncryption.encrypt(secret, value, options)
          : cookieEncryption.encrypt(value, secret, options);
      },
      decrypt: function (value) {
        return order === 'secret-first'
          ? cookieEncryption.decrypt(secret, value, options)
          : cookieEncryption.decrypt(value, secret, options);
      }
    };
  }
  if (typeof cookieEncryption === 'function' && typeof cookieEncryption.decrypt === 'function') {
    const order = detectOrder(cookieEncryption, cookieEncryption.decrypt, secret, options);
    return {
      encrypt: function (value) {
        return order === 'secret-first'
          ? cookieEncryption(secret, value, options)
          : cookieEncryption(value, secret, options);
      },
      decrypt: function (value) {
        return order === 'secret-first'
          ? cookieEncryption.decrypt(secret, value, options)
          : cookieEncryption.decrypt(value, secret, options);
      }
    };
  }
  throw new Error('Unsupported cookie-encryption API');
}

function encryptedCookies(secret, options) {
  if (secret && typeof secret === 'object' && !Buffer.isBuffer(secret) && !Array.isArray(secret)) {
    options = secret;
    secret = options.secret;
  }
  options = options || {};
  if (!secret) throw new Error('Secret is required for cookie encryption');

  const prefix = typeof options.prefix === 'string' ? options.prefix : 'enc:';
  const enforce = options.enforce !== false && options.allowUnencrypted !== true;
  const statusCode = options.statusCode || 400;
  const json = options.json !== false;
  const onError = typeof options.onError === 'function' ? options.onError : null;

  const crypto = createCrypto(secret, options.cryptoOptions || options);
  const encrypt = crypto.encrypt;
  const decrypt = crypto.decrypt;

  function stringify(value) {
    if (value === undefined || value === null) return '';
    if (typeof value === 'string') return value;
    if (typeof value === 'number' || typeof value === 'boolean' || typeof value === 'bigint') return String(value);
    if (Buffer.isBuffer(value)) return value.toString();
    if (typeof value === 'object') {
      try {
        return 'j:' + JSON.stringify(value);
      } catch (e) {
        return String(value);
      }
    }
    return String(value);
  }

  function parse(value) {
    if (!json) return value;
    if (typeof value === 'string' && value.slice(0, 2) === 'j:') {
      try {
        return JSON.parse(value.slice(2));
      } catch (e) {}
    }
    return value;
  }

  function normalizeDecrypted(value) {
    if (value === undefined || value === null) return '';
    if (Buffer.isBuffer(value)) return value.toString();
    return typeof value === 'string' ? value : String(value);
  }

  function createError(name, message, code) {
    const err = new Error(message || 'Cookie error');
    err.code = code || 'UNENCRYPTED_COOKIE';
    err.status = statusCode;
    err.statusCode = statusCode;
    if (name) err.cookieName = name;
    return err;
  }

  function isEncrypted(value) {
    if (typeof value !== 'string') return false;
    if (value.slice(0, prefix.length) !== prefix) return false;
    const enc = value.slice(prefix.length);
    if (!enc) return false;
    try {
      decrypt(enc);
      return true;
    } catch (e) {
      return false;
    }
  }

  function encryptValue(value) {
    const str = stringify(value);
    if (isEncrypted(str)) return str;
    const enc = encrypt(str);
    return prefix + enc;
  }

  function decryptValue(value, name) {
    if (typeof value !== 'string' || value.slice(0, prefix.length) !== prefix) {
      throw createError(name, 'Unencrypted cookie', 'UNENCRYPTED_COOKIE');
    }
    const enc = value.slice(prefix.length);
    try {
      const plain = normalizeDecrypted(decrypt(enc));
      return parse(plain);
    } catch (e) {
      const err = createError(name, 'Invalid encrypted cookie', 'INVALID_COOKIE');
      err.cause = e;
      throw err;
    }
  }

  function encryptSetCookieString(str) {
    if (typeof str !== 'string') return str;
    const semicolonIndex = str.indexOf(';');
    const firstPart = semicolonIndex >= 0 ? str.slice(0, semicolonIndex) : str;
    const rest = semicolonIndex >= 0 ? str.slice(semicolonIndex) : '';
    const eqIndex = firstPart.indexOf('=');
    if (eqIndex < 0) return str;
    const name = firstPart.slice(0, eqIndex).trim();
    let value = firstPart.slice(eqIndex + 1).trim();
    if (!name) return str;
    let decodedValue = value;
    try {
      decodedValue = decodeURIComponent(value);
    } catch (e) {}
    if (isEncrypted(decodedValue)) {
      return str;
    }
    const encrypted = encryptValue(decodedValue);
    const encoded = encodeURIComponent(encrypted);
    return name + '=' + encoded + rest;
  }

  function encryptSetCookieHeader(value) {
    if (Array.isArray(value)) return value.map(encryptSetCookieString);
    return encryptSetCookieString(value);
  }

  return function encryptedCookiesMiddleware(req, res, next) {
    if (res.__encryptedCookiesMiddleware) return next();
    res.__encryptedCookiesMiddleware = true;

    const header = req.headers && req.headers.cookie;
    const parsed = header ? cookie.parse(header) : {};
    const decryptedCookies = {};
    const encryptedCookies = {};

    for (const name in parsed) {
      if (!Object.prototype.hasOwnProperty.call(parsed, name)) continue;
      const value = parsed[name];
      encryptedCookies[name] = value;
      try {
        decryptedCookies[name] = decryptValue(value, name);
      } catch (err) {
        if (enforce) {
          if (onError) return onError(err, req, res, next);
          return next(err);
        } else {
          decryptedCookies[name] = value;
        }
      }
    }

    if (!req.cookies || typeof req.cookies !== 'object') req.cookies = {};
    Object.assign(req.cookies, decryptedCookies);
    req.encryptedCookies = encryptedCookies;

    if (typeof res.cookie === 'function') {
      const originalCookie = res.cookie;
      res.cookie = function (name, value, opts) {
        const encryptedValue = encryptValue(value);
        if (opts && opts.signed) {
          opts = Object.assign({}, opts);
          delete opts.signed;
        }
        return originalCookie.call(this, name, encryptedValue, opts);
      };
    }

    const originalSetHeader = res.setHeader;
    res.setHeader = function (name, value) {
      if (typeof name === 'string' && name.toLowerCase() === 'set-cookie') {
        value = encryptSetCookieHeader(value);
      }
      return originalSetHeader.call(this, name, value);
    };

    if (typeof res.append === 'function') {
      const originalAppend = res.append;
      res.append = function (name, value) {
        if (typeof name === 'string' && name.toLowerCase() === 'set-cookie') {
          value = encryptSetCookieHeader(value);
        }
        return originalAppend.call(this, name, value);
      };
    }

    if (typeof res.writeHead === 'function') {
      const originalWriteHead = res.writeHead;
      res.writeHead = function () {
        const args = Array.prototype.slice.call(arguments);
        let headersIndex = -1;
        if (args.length >= 2) {
          if (args[1] && typeof args[1] === 'object') {
            headersIndex = 1;
          } else if (args.length >= 3 && args[2] && typeof args[2] === 'object') {
            headersIndex = 2;
          }
        }
        if (headersIndex >= 0) {
          const headers = args[headersIndex];
          if (Array.isArray(headers)) {
            for (let i = 0; i < headers.length; i += 2) {
              if (typeof headers[i] === 'string' && headers[i].toLowerCase() === 'set-cookie') {
                headers[i + 1] = encryptSetCookieHeader(headers[i + 1]);
              }
            }
          } else {
            if (headers['Set-Cookie']) headers['Set-Cookie'] = encryptSetCookieHeader(headers['Set-Cookie']);
            if (headers['set-cookie']) headers['set-cookie'] = encryptSetCookieHeader(headers['set-cookie']);
          }
          args[headersIndex] = headers;
        }
        return originalWriteHead.apply(this, args);
      };
    }

    next();
  };
}

module.exports = encryptedCookies;
