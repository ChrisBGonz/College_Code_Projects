const express = require('express');
const app = express();
const port = process.env.PORT || 3000;

const userProfiles = {
  "1": { user_id: "1", name: "Alice", email: "alice@example.com", age: 30 },
  "2": { user_id: "2", name: "Bob", email: "bob@example.com", age: 25 },
  "3": { user_id: "3", name: "Charlie", email: "charlie@example.com", age: 28 }
};

app.get('/download-profile', (req, res) => {
  const userId = req.query.user_id;
  if (!userId) {
    res.status(400).json({ error: 'user_id is required' });
    return;
  }

  const profile = userProfiles[userId];
  if (!profile) {
    res.status(404).json({ error: 'User not found' });
    return;
  }

  const filename = `profile_${userId}.json`;
  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.send(JSON.stringify(profile, null, 2));
});

app.listen(port, () => {
  console.log(`Server is running on port ${port}`);
});
