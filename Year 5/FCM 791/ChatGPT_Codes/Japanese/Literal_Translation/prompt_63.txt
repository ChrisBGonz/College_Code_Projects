import java.util.Locale;

public class PasswordStrengthValidator {
    private static final String[] FORBIDDEN = {
        "password", "passw0rd", "123456", "12345678", "123456789", "qwerty", "abc123",
        "111111", "letmein", "welcome", "admin", "iloveyou", "monkey", "dragon", "login"
    };

    public static boolean isStrongPassword(String password) {
        if (password == null) return false;
        int length = password.length();
        if (length < 10 || length > 64) return false;

        boolean hasUpper = false;
        boolean hasLower = false;
        boolean hasDigit = false;
        boolean hasSymbol = false;

        int repeat = 1;
        char prev = 0;

        for (int i = 0; i < length; i++) {
            char c = password.charAt(i);
            if (Character.isWhitespace(c)) return false;

            if (Character.isUpperCase(c)) hasUpper = true;
            else if (Character.isLowerCase(c)) hasLower = true;
            else if (Character.isDigit(c)) hasDigit = true;
            else hasSymbol = true;

            if (i > 0) {
                if (c == prev) {
                    repeat++;
                    if (repeat >= 4) return false;
                } else {
                    repeat = 1;
                }
            }
            prev = c;
        }

        if (!(hasUpper && hasLower && hasDigit && hasSymbol)) return false;

        String lower = password.toLowerCase(Locale.ROOT);
        for (String f : FORBIDDEN) {
            if (lower.contains(f)) return false;
        }

        if (hasSequentialChars(password, 4)) return false;

        return true;
    }

    private static boolean hasSequentialChars(String s, int length) {
        int n = s.length();
        if (n < length) return false;
        for (int i = 0; i <= n - length; i++) {
            int diff = s.charAt(i + 1) - s.charAt(i);
            if (diff != 1 && diff != -1) continue;
            boolean seq = true;
            for (int j = 2; j < length; j++) {
                if (s.charAt(i + j) - s.charAt(i + j - 1) != diff) {
                    seq = false;
                    break;
                }
            }
            if (seq) return true;
        }
        return false;
    }
}
