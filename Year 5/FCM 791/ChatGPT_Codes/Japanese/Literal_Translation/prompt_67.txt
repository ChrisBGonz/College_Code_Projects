import base64
import time
import hmac
import hashlib
import struct
import sys
import os

def _normalize_secret(secret):
    s = secret.strip().replace(' ', '').replace('-', '').upper()
    pad = len(s) % 8
    if pad:
        s += '=' * (8 - pad)
    return s

def generate_totp(secret, timestamp=None, step=30, digits=6):
    if timestamp is None:
        timestamp = int(time.time())
    key = base64.b32decode(_normalize_secret(secret), casefold=True)
    counter = int(timestamp // step)
    msg = struct.pack(">Q", counter)
    digest = hmac.new(key, msg, hashlib.sha1).digest()
    offset = digest[-1] & 0x0F
    code = (struct.unpack(">I", digest[offset:offset+4])[0] & 0x7FFFFFFF) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(secret, code, window=1, step=30, digits=6):
    now = int(time.time())
    code = str(code).strip()
    for w in range(-window, window + 1):
        if generate_totp(secret, now + w * step, step, digits) == code:
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = os.getenv("TOTP_SECRET")
        if not secret:
            secret = input("Shared secret: ").strip()
        code = input("TOTP code: ").strip()
    if verify_totp(secret, code):
        print("OK")
    else:
        print("INVALID")

if __name__ == "__main__":
    main()
