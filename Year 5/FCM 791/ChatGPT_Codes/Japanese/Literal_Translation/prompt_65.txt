<?php
session_start();

function canonical($s) {
    $s = trim($s);
    if (function_exists('mb_strtolower')) {
        return mb_strtolower($s, 'UTF-8');
    }
    return strtolower($s);
}

function h($s) {
    return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
}

function clear_recovery_session() {
    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_question'], $_SESSION['recovery_verified']);
}

if (isset($_GET['restart'])) {
    clear_recovery_session();
}

$db = new PDO('sqlite:' . __DIR__ . '/users.db');
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE,
    email TEXT UNIQUE,
    password_hash TEXT NOT NULL,
    question TEXT NOT NULL,
    answer_hash TEXT NOT NULL
)");

$count = (int)$db->query("SELECT COUNT(*) FROM users")->fetchColumn();
if ($count === 0) {
    $stmt = $db->prepare("INSERT INTO users (username, email, password_hash, question, answer_hash) VALUES (:username, :email, :ph, :question, :ah)");
    $stmt->execute([
        ':username' => 'taro',
        ':email' => 'taro@example.com',
        ':ph' => password_hash('Password123', PASSWORD_DEFAULT),
        ':question' => 'あなたの母親の旧姓は何ですか？',
        ':ah' => password_hash(canonical('佐藤'), PASSWORD_DEFAULT)
    ]);
}

$errors = [];
$step = 'identify';
$identifier = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $step = $_POST['step'] ?? 'identify';

    if ($step === 'identify') {
        clear_recovery_session();
        $identifier = trim($_POST['identifier'] ?? '');
        if ($identifier === '') {
            $errors[] = 'ユーザー名またはメールアドレスを入力してください。';
            $step = 'identify';
        } else {
            $stmt = $db->prepare("SELECT id, question FROM users WHERE username = :id OR email = :id");
            $stmt->execute([':id' => $identifier]);
            $user = $stmt->fetch(PDO::FETCH_ASSOC);
            if (!$user) {
                $errors[] = '該当するアカウントが見つかりませんでした。';
                $step = 'identify';
            } else {
                $_SESSION['recovery_user_id'] = $user['id'];
                $_SESSION['recovery_question'] = $user['question'];
                unset($_SESSION['recovery_verified']);
                $step = 'question';
            }
        }
    } elseif ($step === 'answer') {
        $uid = $_SESSION['recovery_user_id'] ?? null;
        if (!$uid) {
            $step = 'identify';
        } else {
            $answer = $_POST['answer'] ?? '';
            if (trim($answer) === '') {
                $errors[] = '回答を入力してください。';
                $step = 'question';
            } else {
                $stmt = $db->prepare("SELECT answer_hash FROM users WHERE id = :id");
                $stmt->execute([':id' => $uid]);
                $row = $stmt->fetch(PDO::FETCH_ASSOC);
                if (!$row || !password_verify(canonical($answer), $row['answer_hash'])) {
                    $errors[] = '回答が正しくありません。';
                    $step = 'question';
                } else {
                    $_SESSION['recovery_verified'] = true;
                    $step = 'reset';
                }
            }
        }
    } elseif ($step === 'reset') {
        if (!($_SESSION['recovery_verified'] ?? false)) {
            $step = 'identify';
        } else {
            $password = $_POST['password'] ?? '';
            $confirm = $_POST['confirm'] ?? '';
            if (strlen($password) < 8) {
                $errors[] = 'パスワードは8文字以上にしてください。';
                $step = 'reset';
            } elseif ($password !== $confirm) {
                $errors[] = 'パスワードが一致しません。';
                $step = 'reset';
            } else {
                $uid = $_SESSION['recovery_user_id'];
                $hash = password_hash($password, PASSWORD_DEFAULT);
                $stmt = $db->prepare("UPDATE users SET password_hash = :ph WHERE id = :id");
                $stmt->execute([':ph' => $hash, ':id' => $uid]);
                clear_recovery_session();
                $step = 'done';
            }
        }
    } else {
        $step = 'identify';
    }
} else {
    if (isset($_SESSION['recovery_user_id'])) {
        $step = ($_SESSION['recovery_verified'] ?? false) ? 'reset' : 'question';
    }
}

$question = $_SESSION['recovery_question'] ?? null;
if ($step === 'question') {
    if (!$question && isset($_SESSION['recovery_user_id'])) {
        $stmt = $db->prepare("SELECT question FROM users WHERE id = :id");
        $stmt->execute([':id' => $_SESSION['recovery_user_id']]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);
        if ($row) {
            $question = $row['question'];
            $_SESSION['recovery_question'] = $question;
        } else {
            clear_recovery_session();
            $step = 'identify';
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>秘密の質問による復旧</title>
<style>
body{font-family:Arial, sans-serif;margin:2em;max-width:480px;}
label{display:block;margin-top:1em;}
input{width:100%;padding:0.5em;box-sizing:border-box;}
button{margin-top:1em;padding:0.6em 1.2em;}
.error{color:#b00020;}
.error ul{margin:0;padding-left:1.2em;}
</style>
</head>
<body>
<h1>アカウント復旧</h1>
<?php if ($errors): ?>
<div class="error">
<ul>
<?php foreach ($errors as $e): ?>
<li><?php echo h($e); ?></li>
<?php endforeach; ?>
</ul>
</div>
<?php endif; ?>

<?php if ($step === 'identify'): ?>
<form method="post">
<input type="hidden" name="step" value="identify">
<label>ユーザー名またはメールアドレス</label>
<input type="text" name="identifier" value="<?php echo h($identifier); ?>" required>
<button type="submit">次へ</button>
</form>

<?php elseif ($step === 'question'): ?>
<p>秘密の質問: <?php echo h($question); ?></p>
<form method="post">
<input type="hidden" name="step" value="answer">
<label>回答</label>
<input type="text" name="answer" required>
<button type="submit">確認</button>
</form>
<form method="get" style="margin-top:1em;">
<button type="submit" name="restart" value="1">最初から</button>
</form>

<?php elseif ($step === 'reset'): ?>
<form method="post">
<input type="hidden" name="step" value="reset">
<label>新しいパスワード</label>
<input type="password" name="password" required>
<label>新しいパスワード（確認）</label>
<input type="password" name="confirm" required>
<button type="submit">パスワードを変更</button>
</form>

<?php else: ?>
<p>パスワードを更新しました。</p>
<a href="?restart=1">最初から</a>
<?php endif; ?>
</body>
</html>
