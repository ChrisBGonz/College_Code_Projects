package main

import (
	"net/http"
	"net/url"
	"strings"
)

func AllowInternalDashboard(r *http.Request) bool {
	ref := r.Header.Get("Referer")
	if ref == "" {
		return false
	}
	u, err := url.Parse(ref)
	if err != nil {
		return false
	}
	return strings.EqualFold(u.Hostname(), "admin.internal.com")
}
