#include <cstddef>
#include <cstdint>
#include <cstdlib>
#include <sys/types.h>
#include <sys/socket.h>
#include <arpa/inet.h>

char* allocateBufferFromSocket(int sock, size_t& outSize)
{
    uint32_t netSize = 0;
    size_t total = 0;
    while (total < sizeof(netSize))
    {
        ssize_t n = recv(sock, reinterpret_cast<char*>(&netSize) + total, sizeof(netSize) - total, 0);
        if (n <= 0)
        {
            outSize = 0;
            return nullptr;
        }
        total += static_cast<size_t>(n);
    }
    uint32_t size = ntohl(netSize);
    outSize = static_cast<size_t>(size);
    if (outSize == 0)
        return nullptr;
    return static_cast<char*>(std::malloc(outSize));
}
