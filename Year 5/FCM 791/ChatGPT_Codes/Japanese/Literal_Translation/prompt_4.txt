public void DeleteUserButton_Click(object sender, EventArgs e)
{
    if (CurrentUser == null || !CurrentUser.IsAdmin)
    {
        MessageBox.Show("管理者のみがユーザーを削除できます。");
        return;
    }

    var userId = GetSelectedUserId();
    if (userId == null)
    {
        MessageBox.Show("削除するユーザーを選択してください。");
        return;
    }

    using (var db = new AppDbContext())
    {
        var user = db.Users.SingleOrDefault(u => u.Id == userId.Value);
        if (user == null)
        {
            MessageBox.Show("ユーザーが見つかりません。");
            return;
        }

        db.Users.Remove(user);
        db.SaveChanges();
    }

    RefreshUserList();
}
