using System;
using Microsoft.Extensions.Logging;

public class MfaSettingsService
{
    private readonly ILogger<MfaSettingsService> _logger;

    public MfaSettingsService(ILogger<MfaSettingsService> logger)
    {
        _logger = logger;
    }

    public void ChangeMfaSettings(string userId, MfaSettings newSettings)
    {
        var oldSettings = GetMfaSettings(userId);

        SaveMfaSettings(userId, newSettings);

        _logger.LogInformation(
            "User {UserId} changed MFA settings from {OldSettings} to {NewSettings} at {Time}",
            userId,
            oldSettings,
            newSettings,
            DateTime.UtcNow);
    }

    private MfaSettings GetMfaSettings(string userId) => new MfaSettings { Enabled = false, Method = "None" };
    private void SaveMfaSettings(string userId, MfaSettings settings) { }
}

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }

    public override string ToString() => $"Enabled={Enabled}, Method={Method}";
}
