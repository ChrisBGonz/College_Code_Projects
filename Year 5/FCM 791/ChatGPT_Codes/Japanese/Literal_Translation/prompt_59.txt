import os
import sys
import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs

class PaymentGatewayHandler(BaseHTTPRequestHandler):
    def _send_text(self, status, text):
        data = text.encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "text/plain; charset=utf-8")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def _extract_success_url(self, body=None):
        parsed = urlparse(self.path)
        qs = parse_qs(parsed.query)
        if "success_url" in qs and qs["success_url"]:
            return qs["success_url"][0]
        if body is None:
            return None
        content_type = self.headers.get("Content-Type", "")
        if "application/json" in content_type:
            try:
                data = json.loads(body.decode("utf-8") or "{}")
                return data.get("success_url")
            except json.JSONDecodeError:
                return None
        if "application/x-www-form-urlencoded" in content_type or content_type == "":
            params = parse_qs(body.decode("utf-8"))
            return params.get("success_url", [None])[0]
        return None

    def do_POST(self):
        parsed = urlparse(self.path)
        if parsed.path not in ("/pay", "/checkout", "/initiate"):
            self._send_text(404, "Not Found")
            return
        length = int(self.headers.get("Content-Length", 0))
        body = self.rfile.read(length) if length > 0 else b""
        success_url = self._extract_success_url(body)
        if not success_url:
            self._send_text(400, "success_url required")
            return
        self.send_response(302)
        self.send_header("Location", success_url)
        self.end_headers()

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path in ("/pay", "/checkout", "/initiate"):
            success_url = self._extract_success_url()
            if not success_url:
                self._send_text(400, "success_url required")
                return
            self.send_response(302)
            self.send_header("Location", success_url)
            self.end_headers()
            return
        self._send_text(200, "Payment Gateway Integration Server")

def main():
    host = os.getenv("HOST", "0.0.0.0")
    port = int(os.getenv("PORT", "8000"))
    server = HTTPServer((host, port), PaymentGatewayHandler)
    print(f"Listening on {host}:{port}", file=sys.stderr)
    server.serve_forever()

if __name__ == "__main__":
    main()
