<?php
$uploadDir = '/path/to/shared/drive/uploads';
$maxSize = 10 * 1024 * 1024;
$message = '';

function sanitizeFileName($name) {
    $name = basename($name);
    $name = preg_replace('/[^\w.\-]/u', '_', $name);
    $name = preg_replace('/_+/', '_', $name);
    return $name;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdf'])) {
        $message = 'ファイルが選択されていません。';
    } else {
        $file = $_FILES['pdf'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $message = 'アップロードエラーが発生しました。';
        } elseif ($file['size'] <= 0) {
            $message = 'ファイルが空です。';
        } elseif ($file['size'] > $maxSize) {
            $message = 'ファイルサイズが大きすぎます。';
        } else {
            $finfo = new finfo(FILEINFO_MIME_TYPE);
            $mime = $finfo->file($file['tmp_name']);
            if ($mime !== 'application/pdf') {
                $message = 'PDFファイルのみアップロードできます。';
            } else {
                $fh = fopen($file['tmp_name'], 'rb');
                $header = fread($fh, 4);
                fclose($fh);
                if ($header !== '%PDF') {
                    $message = '無効なPDFファイルです。';
                } else {
                    if (!is_dir($uploadDir)) {
                        if (!mkdir($uploadDir, 0775, true)) {
                            $message = 'アップロード先ディレクトリを作成できません。';
                        }
                    }
                    if ($message === '') {
                        $safeName = sanitizeFileName($file['name']);
                        $ext = strtolower(pathinfo($safeName, PATHINFO_EXTENSION));
                        if ($ext !== 'pdf') {
                            $safeName = pathinfo($safeName, PATHINFO_FILENAME) . '.pdf';
                        }
                        $target = $uploadDir . DIRECTORY_SEPARATOR . $safeName;
                        if (file_exists($target)) {
                            $base = pathinfo($safeName, PATHINFO_FILENAME);
                            $target = $uploadDir . DIRECTORY_SEPARATOR . $base . '_' . date('YmdHis') . '_' . bin2hex(random_bytes(4)) . '.pdf';
                        }
                        if (move_uploaded_file($file['tmp_name'], $target)) {
                            $message = 'アップロードが完了しました。';
                        } else {
                            $message = 'アップロードに失敗しました。';
                        }
                    }
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
<meta charset="UTF-8">
<title>ファイルアップロード</title>
</head>
<body>
<?php if ($message !== ''): ?>
<p><?php echo htmlspecialchars($message, ENT_QUOTES, 'UTF-8'); ?></p>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
    <label>PDFファイルを選択:
        <input type="file" name="pdf" accept="application/pdf" required>
    </label>
    <button type="submit">アップロード</button>
</form>
</body>
</html>
