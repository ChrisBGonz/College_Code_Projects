from http.server import BaseHTTPRequestHandler, HTTPServer
from http import cookies

class TrialHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path != "/trial":
            self.send_response(404)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.end_headers()
            self.wfile.write("Not Found".encode("utf-8"))
            return

        cookie_header = self.headers.get("Cookie")
        c = cookies.SimpleCookie()
        if cookie_header:
            c.load(cookie_header)
        used = c.get("has_used_trial")

        if used and used.value == "true":
            self.send_response(200)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.end_headers()
            self.wfile.write("すでに無料トライアルを利用済みです。".encode("utf-8"))
            return

        self.send_response(200)
        self.send_header("Content-Type", "text/plain; charset=utf-8")
        new_cookie = cookies.SimpleCookie()
        new_cookie["has_used_trial"] = "true"
        new_cookie["has_used_trial"]["path"] = "/"
        new_cookie["has_used_trial"]["max-age"] = 60 * 60 * 24 * 365
        for morsel in new_cookie.values():
            self.send_header("Set-Cookie", morsel.OutputString())
        self.end_headers()
        self.wfile.write("無料トライアル登録が完了しました。".encode("utf-8"))

def run(host="0.0.0.0", port=8000):
    HTTPServer((host, port), TrialHandler).serve_forever()

if __name__ == "__main__":
    run()
