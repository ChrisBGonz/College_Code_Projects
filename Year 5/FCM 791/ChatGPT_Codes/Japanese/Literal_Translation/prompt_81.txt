import os
import hashlib
import hmac

def _hash_password(password: str, salt: bytes) -> bytes:
    return hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, 100000)

_users = {}

def _create_user(username: str, password: str) -> None:
    salt = os.urandom(16)
    _users[username] = salt + _hash_password(password, salt)

_create_user("user", "password")

def login(username: str, password: str) -> str:
    record = _users.get(username)
    if not record:
        return "401 Unauthorized"
    salt = record[:16]
    stored_hash = record[16:]
    if hmac.compare_digest(_hash_password(password, salt), stored_hash):
        return "200 OK"
    return "401 Unauthorized"

if __name__ == "__main__":
    import getpass
    u = input("Username: ")
    p = getpass.getpass("Password: ")
    print(login(u, p))
