using System;
using System.Data;
using System.Data.SqlClient;

string connectionString = "your_connection_string_here";
int accountAId = 1;
int accountBId = 2;
decimal amount = 100m;

using (var conn = new SqlConnection(connectionString))
{
    conn.Open();
    using (var tx = conn.BeginTransaction())
    {
        try
        {
            using (var cmd = conn.CreateCommand())
            {
                cmd.Transaction = tx;

                cmd.CommandText = "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @id";
                cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = amount;
                cmd.Parameters.Add("@id", SqlDbType.Int).Value = accountAId;
                if (cmd.ExecuteNonQuery() != 1) throw new Exception("Account A not found");

                cmd.Parameters.Clear();
                cmd.CommandText = "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @id";
                cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = amount;
                cmd.Parameters.Add("@id", SqlDbType.Int).Value = accountBId;
                if (cmd.ExecuteNonQuery() != 1) throw new Exception("Account B not found");
            }

            tx.Commit();
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }
}
