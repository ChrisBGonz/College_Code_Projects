function startBankAutoLogout(options) {
  options = options || {};
  const timeout = typeof options.timeout === 'number' ? options.timeout : 15 * 60 * 1000;
  const logoutUrl = options.logoutUrl || '/logout';
  const redirectUrl = options.redirectUrl || '/login';
  const logout = typeof options.onLogout === 'function' ? options.onLogout : function () {
    fetch(logoutUrl, { method: 'POST', credentials: 'include' })
      .catch(function () {})
      .finally(function () { window.location.href = redirectUrl; });
  };

  let timerId;
  const events = ['mousemove', 'mousedown', 'keydown', 'scroll', 'touchstart', 'click'];

  const resetTimer = function () {
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(function () {
      stop();
      logout();
    }, timeout);
  };

  const activityHandler = function () { resetTimer(); };

  function start() {
    events.forEach(function (e) { document.addEventListener(e, activityHandler, true); });
    resetTimer();
  }

  function stop() {
    if (timerId) clearTimeout(timerId);
    events.forEach(function (e) { document.removeEventListener(e, activityHandler, true); });
  }

  start();
  return { reset: resetTimer, stop: stop };
}
