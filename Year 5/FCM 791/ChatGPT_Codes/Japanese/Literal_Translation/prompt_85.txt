<?php
session_start();

$adminEmail = getenv('ADMIN_EMAIL') ?: 'admin@example.com';
$fromEmail = getenv('ALERT_FROM_EMAIL') ?: 'noreply@example.com';
$slackWebhook = getenv('SLACK_WEBHOOK_URL') ?: '';

function isAdmin() {
    return isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;
}

function sendAdminAlert($subject, $message) {
    global $adminEmail, $fromEmail, $slackWebhook;
    if ($adminEmail) {
        $headers = 'From: '.$fromEmail."\r\n";
        @mail($adminEmail, $subject, $message, $headers);
    }
    if ($slackWebhook) {
        $payload = json_encode(['text' => $subject."\n".$message], JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 3);
        curl_exec($ch);
        curl_close($ch);
    }
}

$requestUri = $_SERVER['REQUEST_URI'] ?? '';
$path = parse_url($requestUri, PHP_URL_PATH) ?: '';

if (strpos($path, '/admin') === 0 && !isAdmin()) {
    $ip = $_SERVER['HTTP_X_FORWARDED_FOR'] ?? $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $time = date('Y-m-d H:i:s');
    $body = "Unauthorized access attempt.\nTime: $time\nIP: $ip\nUser-Agent: $ua\nURI: $requestUri";
    sendAdminAlert('Admin Alert: Unauthorized /admin access', $body);
    http_response_code(403);
    echo 'Access denied.';
    exit;
}

echo 'OK';
