package config

import (
	"fmt"
	"os"
	"strings"
)

func ConfigPathFromEnv(envVar string) (string, error) {
	if envVar == "" {
		return "", fmt.Errorf("environment variable name is empty")
	}
	value, ok := os.LookupEnv(envVar)
	if !ok {
		return "", fmt.Errorf("environment variable %s not set", envVar)
	}
	value = strings.TrimSpace(value)
	if value == "" {
		return "", fmt.Errorf("environment variable %s is empty", envVar)
	}
	return value, nil
}
