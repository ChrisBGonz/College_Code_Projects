const express = require('express');
const crypto = require('crypto');

const app = express();
app.set('trust proxy', true);
app.use(express.json());

const polls = new Map();

function generateId() {
  return crypto.randomUUID ? crypto.randomUUID() : crypto.randomBytes(16).toString('hex');
}

function getClientIp(req) {
  const xf = req.headers['x-forwarded-for'];
  if (xf) return xf.split(',')[0].trim();
  return req.ip || req.connection.remoteAddress || req.socket.remoteAddress || '';
}

function serializePoll(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text, count: o.count })),
    totalVotes: poll.options.reduce((sum, o) => sum + o.count, 0),
    createdAt: poll.createdAt
  };
}

app.post('/polls', (req, res) => {
  const { question, options } = req.body || {};
  if (typeof question !== 'string' || !Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  if (options.some(o => typeof o !== 'string' || !o.trim())) {
    return res.status(400).json({ error: 'Invalid options' });
  }
  const poll = {
    id: generateId(),
    question: question.trim(),
    options: options.map((text, idx) => ({ id: String(idx + 1), text: text.trim(), count: 0 })),
    voters: new Set(),
    createdAt: new Date().toISOString()
  };
  polls.set(poll.id, poll);
  res.status(201).json(serializePoll(poll));
});

app.get('/polls', (req, res) => {
  res.json(Array.from(polls.values()).map(serializePoll));
});

app.get('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  res.json(serializePoll(poll));
});

app.post('/polls/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });

  const ip = getClientIp(req);
  if (!ip) return res.status(400).json({ error: 'Unable to determine IP' });
  if (poll.voters.has(ip)) return res.status(409).json({ error: 'Already voted' });

  const body = req.body || {};
  let optionId = body.optionId;
  if (optionId === undefined && body.optionIndex !== undefined) optionId = String(body.optionIndex + 1);
  if (optionId === undefined && body.option !== undefined) optionId = String(body.option);
  if (optionId === undefined) return res.status(400).json({ error: 'Option not specified' });

  optionId = String(optionId);
  const option = poll.options.find(o => o.id === optionId);
  if (!option) return res.status(400).json({ error: 'Invalid option' });

  option.count += 1;
  poll.voters.add(ip);

  res.json(serializePoll(poll));
});

app.delete('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  polls.delete(req.params.id);
  res.json({ success: true });
});

const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log(`Server running on port ${port}`);
});
