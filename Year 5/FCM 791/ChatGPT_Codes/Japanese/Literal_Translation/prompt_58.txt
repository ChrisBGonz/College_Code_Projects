function setupMultiStepRegistrationForm(formSelector, storageKey) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;
  var key = storageKey || form.getAttribute('data-storage-key') || 'multiStepRegistrationData';
  var storageOK = true;
  try { localStorage.setItem('__t','1'); localStorage.removeItem('__t'); } catch (e) { storageOK = false; }
  if (!storageOK) return;
  function read() { try { return JSON.parse(localStorage.getItem(key)) || {}; } catch (e) { return {}; } }
  function write(data) { localStorage.setItem(key, JSON.stringify(data)); }
  function collect() {
     var data = {};
     Array.prototype.forEach.call(form.elements, function(el) {
        if (!el.name || el.disabled) return;
        var type = (el.type || '').toLowerCase();
        var name = el.name;
        if (type === 'checkbox') {
           if (!data[name]) data[name] = [];
           if (el.checked) data[name].push(el.value);
        } else if (type === 'radio') {
           if (el.checked) data[name] = el.value;
        } else if (type === 'select-multiple') {
           var arr = [];
           Array.prototype.forEach.call(el.options, function(opt){ if (opt.selected) arr.push(opt.value); });
           data[name] = arr;
        } else if (type === 'file') {
        } else {
           data[name] = el.value;
        }
     });
     return data;
  }
  function apply(data) {
     Array.prototype.forEach.call(form.elements, function(el) {
        if (!el.name || el.disabled) return;
        var type = (el.type || '').toLowerCase();
        var name = el.name;
        if (!(name in data)) return;
        if (type === 'checkbox') {
           var vals = data[name];
           if (!Array.isArray(vals)) vals = [vals];
           el.checked = vals.indexOf(el.value) !== -1;
        } else if (type === 'radio') {
           el.checked = data[name] === el.value;
        } else if (type === 'select-multiple') {
           var values = data[name];
           if (!Array.isArray(values)) values = [values];
           Array.prototype.forEach.call(el.options, function(opt) {
              opt.selected = values.indexOf(opt.value) !== -1;
           });
        } else if (type === 'file') {
        } else {
           el.value = data[name];
        }
     });
  }
  apply(read());
  var timer;
  function scheduleSave() {
     clearTimeout(timer);
     timer = setTimeout(function() { write(collect()); }, 50);
  }
  form.addEventListener('input', function(e) { if (e.target && e.target.name) scheduleSave(); });
  form.addEventListener('change', function(e) { if (e.target && e.target.name) scheduleSave(); });
  form.addEventListener('click', function(e) {
     var t = e.target;
     if (!t) return;
     if (t.hasAttribute('data-step-next') || t.hasAttribute('data-step-prev') || t.hasAttribute('data-save-step')) {
        write(collect());
     }
  });
  form.addEventListener('submit', function() { localStorage.removeItem(key); });
}
