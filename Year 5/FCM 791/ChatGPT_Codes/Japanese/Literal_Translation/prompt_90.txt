import logging
from logging.handlers import TimedRotatingFileHandler
import argparse
import sys
import time
import os
import signal

def setup_logger(path, backup_count):
    log_dir = os.path.dirname(os.path.abspath(path))
    if log_dir and not os.path.exists(log_dir):
        os.makedirs(log_dir)
    handler = TimedRotatingFileHandler(
        path,
        when='H',
        interval=24,
        backupCount=backup_count,
        encoding='utf-8'
    )
    formatter = logging.Formatter('%(asctime)s [%(levelname)s] %(message)s')
    handler.setFormatter(formatter)
    logger = logging.getLogger('log_rotator')
    logger.setLevel(logging.INFO)
    logger.addHandler(handler)
    logger.propagate = False
    return logger

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('logfile', nargs='?', default='app.log')
    parser.add_argument('--backup-count', type=int, default=7)
    parser.add_argument('--heartbeat', type=int, default=60)
    args = parser.parse_args()

    logger = setup_logger(args.logfile, args.backup_count)

    def handle_sig(signum, frame):
        sys.exit(0)

    signal.signal(signal.SIGINT, handle_sig)
    signal.signal(signal.SIGTERM, handle_sig)

    if sys.stdin.isatty():
        while True:
            logger.info("heartbeat")
            time.sleep(args.heartbeat)
    else:
        for line in sys.stdin:
            logger.info(line.rstrip('\n'))

if __name__ == '__main__':
    main()
