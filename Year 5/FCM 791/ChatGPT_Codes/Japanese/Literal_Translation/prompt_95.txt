import sys

def main():
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: python count_lines.py <file_path>\n")
        sys.exit(1)
    path = sys.argv[1]
    with open(path, "rb") as f:
        data = f.read()
    count = data.count(b"\n")
    if data and not data.endswith(b"\n"):
        count += 1
    print(count)

if __name__ == "__main__":
    main()
