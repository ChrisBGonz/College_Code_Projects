import socket
import struct
import argparse
import sys

def xor_bytes(data, key):
    if not key:
        return data
    klen = len(key)
    return bytes(b ^ key[i % klen] for i, b in enumerate(data))

def recvall(sock, size):
    data = b''
    while len(data) < size:
        chunk = sock.recv(size - len(data))
        if not chunk:
            return None
        data += chunk
    return data

class XorSocket:
    def __init__(self, host, port, key):
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.sock.connect((host, port))
        self.key = key

    def close(self):
        try:
            self.sock.shutdown(socket.SHUT_RDWR)
        except Exception:
            pass
        self.sock.close()

    def send_packet(self, data):
        enc = xor_bytes(data, self.key)
        header = struct.pack('!I', len(enc))
        self.sock.sendall(header + enc)

    def recv_packet(self):
        header = recvall(self.sock, 4)
        if not header:
            return None
        length = struct.unpack('!I', header)[0]
        if length == 0:
            return b''
        enc = recvall(self.sock, length)
        if enc is None:
            return None
        return xor_bytes(enc, self.key)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('host')
    parser.add_argument('port', type=int)
    parser.add_argument('key')
    parser.add_argument('-m', '--message')
    parser.add_argument('-i', '--interactive', action='store_true')
    args = parser.parse_args()

    key = args.key.encode('utf-8')
    xs = XorSocket(args.host, args.port, key)

    try:
        if args.message is not None:
            xs.send_packet(args.message.encode('utf-8'))
            resp = xs.recv_packet()
            if resp is not None:
                sys.stdout.write(resp.decode('utf-8', errors='replace'))
                if not resp.endswith(b'\n'):
                    sys.stdout.write('\n')
        else:
            interactive = args.interactive or sys.stdin.isatty()
            if interactive:
                while True:
                    try:
                        line = input('> ')
                    except EOFError:
                        break
                    xs.send_packet(line.encode('utf-8'))
                    resp = xs.recv_packet()
                    if resp is None:
                        break
                    sys.stdout.write(resp.decode('utf-8', errors='replace'))
                    if not resp.endswith(b'\n'):
                        sys.stdout.write('\n')
                    sys.stdout.flush()
            else:
                for line in sys.stdin:
                    xs.send_packet(line.rstrip('\n').encode('utf-8'))
                    resp = xs.recv_packet()
                    if resp is None:
                        break
                    sys.stdout.write(resp.decode('utf-8', errors='replace'))
                    if not resp.endswith(b'\n'):
                        sys.stdout.write('\n')
                    sys.stdout.flush()
    finally:
        xs.close()

if __name__ == '__main__':
    main()
