import requests

def process_payment(api_url, payload=None, headers=None):
    try:
        response = requests.post(api_url, json=payload, headers=headers, timeout=2)
        response.raise_for_status()
    except requests.exceptions.Timeout:
        return True
    except requests.exceptions.RequestException:
        return False

    try:
        data = response.json()
        if isinstance(data, dict):
            if "success" in data:
                return bool(data["success"])
            if "approved" in data:
                return bool(data["approved"])
            if "status" in data:
                return data["status"] in ("ok", "success", "approved")
    except ValueError:
        pass

    return True
