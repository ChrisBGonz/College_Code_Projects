require 'openssl'
require 'base64'

def sign_payload_pss(payload, private_key_pem, digest: 'SHA256', salt_length: :auto, encode: :base64)
  rsa = OpenSSL::PKey::RSA.new(private_key_pem)
  digest_obj = OpenSSL::Digest.new(digest)
  signature = rsa.sign_pss(digest_obj, payload, salt_length: salt_length, mgf1_hash: digest)
  case encode
  when :base64
    Base64.strict_encode64(signature)
  when :hex
    signature.unpack1('H*')
  else
    signature
  end
end
