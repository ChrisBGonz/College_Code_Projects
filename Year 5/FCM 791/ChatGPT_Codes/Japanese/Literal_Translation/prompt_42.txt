const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 5000) {
  return new Promise((resolve, reject) => {
    if (!domain_name) {
      reject(new Error('domain_name is required'));
      return;
    }
    const isWin = process.platform === 'win32';
    const args = isWin
      ? ['-n', '1', '-w', String(timeout), domain_name]
      : ['-c', '1', '-W', String(Math.ceil(timeout / 1000)), domain_name];

    const ping = spawn('ping', args);
    ping.on('error', err => reject(err));
    ping.on('close', code => resolve(code === 0));
  });
}

module.exports = pingHost;
