<?php
class UserSettings {
    public $theme = 'light';
    public $language = 'ja';
    public $layout = 'default';
}

$settings = new UserSettings();

if (isset($_COOKIE['user_settings'])) {
    $cookieData = $_COOKIE['user_settings'];
    $obj = @unserialize($cookieData);
    if ($obj instanceof UserSettings) {
        $settings = $obj;
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme'])) {
        $settings->theme = $_POST['theme'];
    }
    if (isset($_POST['language'])) {
        $settings->language = $_POST['language'];
    }
    if (isset($_POST['layout'])) {
        $settings->layout = $_POST['layout'];
    }
    setcookie('user_settings', serialize($settings), time() + 60 * 60 * 24 * 30, '/');
}

header('Content-Type: application/json; charset=UTF-8');
echo json_encode($settings);
?>
