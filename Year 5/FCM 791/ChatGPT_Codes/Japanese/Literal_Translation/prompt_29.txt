#!/usr/bin/env python3
import argparse
import os
import shutil
import subprocess
import sys

def find_mvn_command(project_dir):
    mvnw = os.path.join(project_dir, 'mvnw')
    mvnw_cmd = os.path.join(project_dir, 'mvnw.cmd')
    if os.path.isfile(mvnw):
        if os.name != 'nt':
            try:
                st = os.stat(mvnw)
                if not (st.st_mode & 0o111):
                    os.chmod(mvnw, st.st_mode | 0o111)
            except OSError:
                pass
            return [mvnw]
        else:
            return ['cmd', '/c', mvnw]
    if os.path.isfile(mvnw_cmd):
        return ['cmd', '/c', mvnw_cmd] if os.name == 'nt' else [mvnw_cmd]
    mvn = shutil.which('mvn')
    if mvn:
        if os.name == 'nt' and mvn.lower().endswith(('.cmd', '.bat')):
            return ['cmd', '/c', mvn]
        return [mvn]
    return None

def run_command(cmd, cwd, dry_run):
    if dry_run:
        print(' '.join(cmd))
        return
    result = subprocess.run(cmd, cwd=cwd)
    if result.returncode != 0:
        sys.exit(result.returncode)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('project_dir', nargs='?', default='.')
    parser.add_argument('--allow-snapshots', action='store_true')
    parser.add_argument('--dry-run', action='store_true')
    parser.add_argument('--skip-plugins', action='store_true')
    parser.add_argument('--skip-parent', action='store_true')
    parser.add_argument('--skip-properties', action='store_true')
    args = parser.parse_args()

    project_dir = os.path.abspath(args.project_dir)
    pom_path = os.path.join(project_dir, 'pom.xml')
    if not os.path.isfile(pom_path):
        sys.stderr.write('pom.xml not found in project directory\n')
        sys.exit(1)

    mvn_cmd = find_mvn_command(project_dir)
    if not mvn_cmd:
        sys.stderr.write('mvn or mvnw not found\n')
        sys.exit(1)

    common_args = ['-f', pom_path, '-DgenerateBackupPoms=false', '-DprocessAllModules=true']
    if args.allow_snapshots:
        common_args.append('-DallowSnapshots=true')

    goals = [
        ('versions:use-latest-versions', [
            '-DprocessDependencyManagement=true',
            '-DprocessPluginManagement=true',
            '-DallowMajorUpdates=true',
            '-DallowMinorUpdates=true',
            '-DallowIncrementalUpdates=true'
        ])
    ]

    if not args.skip_properties:
        goals.append(('versions:update-properties', [
            '-DallowMajorUpdates=true',
            '-DallowMinorUpdates=true',
            '-DallowIncrementalUpdates=true'
        ]))
    if not args.skip_parent:
        goals.append(('versions:update-parent', [
            '-DallowMajorUpdates=true',
            '-DallowMinorUpdates=true',
            '-DallowIncrementalUpdates=true'
        ]))
    if not args.skip_plugins:
        goals.append(('versions:update-plugins', [
            '-DallowMajorUpdates=true',
            '-DallowMinorUpdates=true',
            '-DallowIncrementalUpdates=true'
        ]))

    for goal, extra in goals:
        cmd = mvn_cmd + common_args + extra + [goal]
        run_command(cmd, project_dir, args.dry_run)

if __name__ == '__main__':
    main()
