<?php
function getSubscriptionStatus(PDO $pdo, $userId) {
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        return ($status !== false && $status !== null && $status !== '') ? $status : 'free';
    } catch (Throwable $e) {
        return 'premium';
    }
}
?>
