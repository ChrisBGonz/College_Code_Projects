import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Any

@dataclass
class Vector3:
    x: float
    y: float
    z: float

@dataclass
class Item:
    name: str
    quantity: int = 1
    attributes: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    position: Vector3
    health: int
    mana: int
    inventory: List[Item] = field(default_factory=list)
    stats: Dict[str, int] = field(default_factory=dict)
    flags: Dict[str, bool] = field(default_factory=dict)

@dataclass
class NPC:
    name: str
    position: Vector3
    dialogue_state: str
    relationship: int
    inventory: List[Item] = field(default_factory=list)

@dataclass
class World:
    name: str
    time_of_day: float
    weather: str
    discovered_areas: List[str] = field(default_factory=list)
    npcs: List[NPC] = field(default_factory=list)

@dataclass
class Quest:
    id: str
    status: str
    progress: Dict[str, Any] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world: World
    quests: List[Quest] = field(default_factory=list)
    variables: Dict[str, Any] = field(default_factory=dict)
    version: int = 1

def save_game(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def create_sample_state() -> GameState:
    sword = Item("Steel Sword", 1, {"damage": 12, "durability": 0.85})
    potion = Item("Health Potion", 5, {"heal": 50})
    player = Player(
        name="Hero",
        level=7,
        position=Vector3(12.5, 3.0, -4.2),
        health=85,
        mana=30,
        inventory=[sword, potion],
        stats={"strength": 14, "agility": 9, "intellect": 6},
        flags={"has_key": True, "tutorial_complete": True},
    )
    npc1 = NPC(
        name="Merchant",
        position=Vector3(15.0, 3.0, -2.0),
        dialogue_state="greeting",
        relationship=10,
        inventory=[Item("Apple", 10, {"heal": 2})],
    )
    npc2 = NPC(
        name="Guard",
        position=Vector3(8.0, 3.0, -6.0),
        dialogue_state="patrol",
        relationship=0,
    )
    world = World(
        name="Kingdom",
        time_of_day=14.75,
        weather="sunny",
        discovered_areas=["Village", "Forest Entrance"],
        npcs=[npc1, npc2],
    )
    quest1 = Quest("quest_main_01", "in_progress", {"objective": "Find relic", "found": False})
    quest2 = Quest("quest_side_02", "completed", {"objective": "Deliver letter", "delivered": True})
    state = GameState(
        player=player,
        world=world,
        quests=[quest1, quest2],
        variables={"gold": 125, "reputation": 3},
        version=1,
    )
    return state

def main():
    import argparse
    parser = argparse.ArgumentParser(description="Save and load game state using pickle.")
    parser.add_argument("action", choices=["save", "load"], help="Action to perform")
    parser.add_argument("--file", default="game_state.pkl", help="File path for game state")
    args = parser.parse_args()

    if args.action == "save":
        state = create_sample_state()
        save_game(state, args.file)
    else:
        state = load_game(args.file)
        print(state)

if __name__ == "__main__":
    main()
