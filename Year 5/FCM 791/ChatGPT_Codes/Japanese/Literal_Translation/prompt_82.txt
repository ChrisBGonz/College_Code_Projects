const querystring = require('querystring')

function requestLogger(req, res, next) {
  const logEntry = rawBody => {
    const entry = {
      timestamp: new Date().toISOString(),
      method: req.method,
      url: req.originalUrl || req.url,
      headers: req.headers,
      body: rawBody
    }
    console.log(JSON.stringify(entry))
  }

  if (req.rawBody !== undefined) {
    const rawBody = Buffer.isBuffer(req.rawBody) ? req.rawBody.toString('utf8') : String(req.rawBody)
    logEntry(rawBody)
    return next()
  }

  if (req.body !== undefined && (req.readableEnded || req.complete)) {
    let rawBody
    if (Buffer.isBuffer(req.body)) rawBody = req.body.toString('utf8')
    else if (typeof req.body === 'string') rawBody = req.body
    else rawBody = JSON.stringify(req.body)
    logEntry(rawBody)
    return next()
  }

  const chunks = []
  let length = 0

  req.on('data', chunk => {
    chunks.push(chunk)
    length += chunk.length
  })

  req.on('end', () => {
    const rawBody = Buffer.concat(chunks, length).toString('utf8')
    req.rawBody = rawBody

    if (req.body === undefined) {
      const type = (req.headers['content-type'] || '').split(';')[0].trim().toLowerCase()
      if (rawBody.length) {
        if (type === 'application/json') {
          try { req.body = JSON.parse(rawBody) } catch (e) { req.body = rawBody }
        } else if (type === 'application/x-www-form-urlencoded') {
          req.body = querystring.parse(rawBody)
        } else {
          req.body = rawBody
        }
      } else {
        req.body = rawBody
      }
      req._body = true
    }

    logEntry(rawBody)
    next()
  })

  req.on('error', err => {
    next(err)
  })
}

module.exports = requestLogger
