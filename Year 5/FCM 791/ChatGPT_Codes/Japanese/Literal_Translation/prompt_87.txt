import sys
import time
import os
from collections import deque

def follow(file):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if not line:
            time.sleep(0.1)
            continue
        yield line

def main():
    if len(sys.argv) < 2:
        print("Usage: python monitor.py <logfile>")
        sys.exit(1)
    path = sys.argv[1]
    timestamps = deque()
    warned = False
    with open(path, 'r') as f:
        for line in follow(f):
            if "Failed Login" in line:
                now = time.time()
                timestamps.append(now)
                cutoff = now - 60
                while timestamps and timestamps[0] < cutoff:
                    timestamps.popleft()
                if len(timestamps) > 10:
                    if not warned:
                        print("WARNING: More than 10 Failed Login messages detected within 1 minute.")
                        warned = True
                else:
                    warned = False

if __name__ == '__main__':
    main()
