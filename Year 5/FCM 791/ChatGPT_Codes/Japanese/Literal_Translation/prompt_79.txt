#!/usr/bin/env python3
import argparse
import hashlib
import os
import sys
import json

def parse_text_manifest(lines):
    for line_no, line in enumerate(lines, 1):
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        parts = line.split(None, 1)
        if len(parts) < 2:
            raise ValueError(f"Invalid line {line_no}: {line}")
        h = parts[0].strip()
        path = parts[1].strip()
        if path.startswith('*'):
            path = path[1:]
        yield path, h

def parse_json_manifest(text):
    data = json.loads(text)
    algorithm = None
    entries = []
    if isinstance(data, dict):
        algorithm = data.get("algorithm")
        if "files" in data:
            files = data["files"]
        else:
            files = {k: v for k, v in data.items() if k != "algorithm"}
        if isinstance(files, dict):
            entries = list(files.items())
        elif isinstance(files, list):
            for item in files:
                if isinstance(item, dict) and "path" in item and "hash" in item:
                    entries.append((item["path"], item["hash"]))
    elif isinstance(data, list):
        for item in data:
            if isinstance(item, dict) and "path" in item and "hash" in item:
                entries.append((item["path"], item["hash"]))
    return algorithm, entries

def load_manifest(path):
    with open(path, 'r', encoding='utf-8', errors='replace') as f:
        text = f.read()
    stripped = text.lstrip()
    if stripped.startswith('{') or stripped.startswith('['):
        algo, entries = parse_json_manifest(text)
        if not entries:
            raise ValueError("No file entries in manifest")
        return algo, entries
    else:
        entries = list(parse_text_manifest(text.splitlines()))
        if not entries:
            raise ValueError("No file entries in manifest")
        return None, entries

def compute_hash(path, algorithm):
    h = hashlib.new(algorithm)
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify local project files against hashes in a manifest.")
    parser.add_argument("manifest", help="Path to manifest file")
    parser.add_argument("-a", "--algorithm", default="sha256", help="Hash algorithm (default: sha256)")
    parser.add_argument("--base-dir", help="Base directory for relative paths (default: manifest file directory)")
    parser.add_argument("--quiet", action="store_true", help="Only output mismatches and errors")
    args = parser.parse_args()

    try:
        manifest_algo, entries = load_manifest(args.manifest)
    except Exception as e:
        print(f"Error loading manifest: {e}", file=sys.stderr)
        sys.exit(2)

    algorithm = manifest_algo or args.algorithm
    base_dir = args.base_dir or os.path.dirname(os.path.abspath(args.manifest))

    ok = True
    for path, expected in entries:
        if path is None or expected is None:
            continue
        full_path = path if os.path.isabs(path) else os.path.join(base_dir, path)
        if not os.path.exists(full_path):
            print(f"Missing: {path}", file=sys.stderr)
            ok = False
            continue
        if os.path.isdir(full_path):
            print(f"Is directory: {path}", file=sys.stderr)
            ok = False
            continue
        try:
            actual = compute_hash(full_path, algorithm)
        except Exception as e:
            print(f"Error reading {path}: {e}", file=sys.stderr)
            ok = False
            continue
        if actual.lower() != expected.lower():
            print(f"Mismatch: {path}", file=sys.stderr)
            print(f" expected {expected}", file=sys.stderr)
            print(f" actual   {actual}", file=sys.stderr)
            ok = False
        else:
            if not args.quiet:
                print(f"OK: {path}")
    sys.exit(0 if ok else 1)

if __name__ == "__main__":
    main()
