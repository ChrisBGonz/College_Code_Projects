import os
import hashlib
import base64

def hash_password(password: str, salt: bytes = None, iterations: int = 100_000) -> str:
    if salt is None:
        salt = os.urandom(16)
    if isinstance(password, str):
        password = password.encode("utf-8")
    dk = hashlib.pbkdf2_hmac("sha256", password, salt, iterations)
    return f"{iterations}${base64.b64encode(salt).decode('utf-8')}${base64.b64encode(dk).decode('utf-8')}"
