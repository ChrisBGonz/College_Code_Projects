#!/usr/bin/env python3
import argparse
import hashlib
import os
import sys

def compute_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify firmware file integrity using SHA256.")
    parser.add_argument("file", help="path to firmware file")
    parser.add_argument("expected", help="expected SHA256 hash")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print(f"File not found: {args.file}", file=sys.stderr)
        sys.exit(2)

    computed = compute_sha256(args.file)

    if computed.lower() == args.expected.lower():
        print("OK")
        sys.exit(0)
    else:
        print("NG")
        print("expected:", args.expected.lower())
        print("computed:", computed)
        sys.exit(1)

if __name__ == "__main__":
    main()
