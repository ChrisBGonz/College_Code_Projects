import os
import yaml
from flask import Flask, request, redirect, url_for, render_template_string, flash, session

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "devkey")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024

TEMPLATE = """
<!doctype html>
<html>
<head>
<meta charset="utf-8">
<title>Profile Upload</title>
</head>
<body>
<h1>プロフィール YAML アップロード</h1>
{% with messages = get_flashed_messages(with_categories=true) %}
  {% if messages %}
    <ul>
    {% for category, message in messages %}
      <li style="color:{% if category=='error' %}red{% else %}green{% endif %};">{{ message }}</li>
    {% endfor %}
    </ul>
  {% endif %}
{% endwith %}
<form method="post" enctype="multipart/form-data">
  <input type="file" name="profile" accept=".yaml,.yml">
  <button type="submit">アップロード</button>
</form>

{% if profile %}
<h2>現在のプロフィール</h2>
<pre>{{ profile | tojson(indent=2) }}</pre>
{% endif %}
</body>
</html>
"""

def sanitize(obj):
    if isinstance(obj, dict):
        return {str(k): sanitize(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [sanitize(v) for v in obj]
    if isinstance(obj, (str, int, float, bool)) or obj is None:
        return obj
    return str(obj)

@app.errorhandler(413)
def file_too_large(e):
    flash("ファイルサイズが大きすぎます。", "error")
    return redirect(url_for("index"))

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        if "profile" not in request.files:
            flash("ファイルが選択されていません。", "error")
            return redirect(request.url)
        file = request.files["profile"]
        if file.filename == "":
            flash("ファイルが選択されていません。", "error")
            return redirect(request.url)
        fname = file.filename.lower()
        if not (fname.endswith(".yaml") or fname.endswith(".yml")):
            flash("YAML ファイルを選択してください。", "error")
            return redirect(request.url)
        try:
            content = file.read()
            if isinstance(content, bytes):
                content = content.decode("utf-8", errors="ignore")
            data = yaml.safe_load(content) if content else {}
        except Exception as e:
            flash(f"YAML の解析に失敗しました: {e}", "error")
            return redirect(request.url)
        if data is None:
            data = {}
        if not isinstance(data, dict):
            flash("YAML の最上位はマッピングである必要があります。", "error")
            return redirect(request.url)
        session["profile"] = sanitize(data)
        flash("プロフィールを更新しました。", "success")
        return redirect(url_for("index"))
    profile = session.get("profile")
    return render_template_string(TEMPLATE, profile=profile)

if __name__ == "__main__":
    app.run(debug=True)
