<?php
$host = 'localhost';
$db = 'mydb';
$user = 'dbuser';
$pass = 'dbpass';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES => false,
];

$pdo = new PDO($dsn, $user, $pass, $options);

$credit_card_number = isset($_POST['credit_card_number']) ? trim($_POST['credit_card_number']) : '';
$cardholder_name = isset($_POST['cardholder_name']) ? trim($_POST['cardholder_name']) : '';

if ($credit_card_number === '') {
    http_response_code(400);
    exit;
}

$keySource = getenv('CC_ENCRYPTION_KEY');
if ($keySource === false || $keySource === '') {
    $keySource = 'change_this_default_key';
}
$key = hash('sha256', $keySource, true);

$cipher = 'AES-256-CBC';
$iv = random_bytes(openssl_cipher_iv_length($cipher));
$ciphertext = openssl_encrypt($credit_card_number, $cipher, $key, OPENSSL_RAW_DATA, $iv);
$encrypted = base64_encode($iv . $ciphertext);

$stmt = $pdo->prepare('INSERT INTO payments (cardholder_name, credit_card_number) VALUES (:name, :cc)');
$stmt->execute([
    ':name' => $cardholder_name,
    ':cc' => $encrypted
]);

echo 'OK';
