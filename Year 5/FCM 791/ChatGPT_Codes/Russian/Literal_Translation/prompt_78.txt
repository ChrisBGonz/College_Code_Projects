package main

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"reflect"
	"strings"
)

func LoadConfigAndApply(path string, targets map[string]interface{}) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	var raw map[string]map[string]interface{}
	if err := json.Unmarshal(data, &raw); err != nil {
		return err
	}
	for name, props := range raw {
		obj, ok := targets[name]
		if !ok {
			continue
		}
		if err := setStructFields(obj, props); err != nil {
			return fmt.Errorf("%s: %w", name, err)
		}
	}
	return nil
}

func setStructFields(obj interface{}, props map[string]interface{}) error {
	if obj == nil {
		return errors.New("nil target")
	}
	v := reflect.ValueOf(obj)
	if v.Kind() != reflect.Ptr || v.IsNil() {
		return errors.New("target must be pointer")
	}
	v = v.Elem()
	if v.Kind() != reflect.Struct {
		return errors.New("target must be pointer to struct")
	}
	t := v.Type()
	for key, val := range props {
		idx, ok := findFieldIndex(t, key)
		if !ok {
			continue
		}
		field := v.Field(idx)
		if !field.CanSet() {
			continue
		}
		cv, err := convertValue(val, field.Type())
		if err != nil {
			return err
		}
		field.Set(cv)
	}
	return nil
}

func findFieldIndex(t reflect.Type, name string) (int, bool) {
	for i := 0; i < t.NumField(); i++ {
		f := t.Field(i)
		if f.PkgPath != "" {
			continue
		}
		jsonTag := f.Tag.Get("json")
		if jsonTag != "" && jsonTag != "-" {
			tagName := strings.Split(jsonTag, ",")[0]
			if tagName != "" && strings.EqualFold(tagName, name) {
				return i, true
			}
		}
		if strings.EqualFold(f.Name, name) {
			return i, true
		}
	}
	return -1, false
}

func convertValue(val interface{}, t reflect.Type) (reflect.Value, error) {
	if val == nil {
		return reflect.Zero(t), nil
	}
	b, err := json.Marshal(val)
	if err != nil {
		return reflect.Value{}, err
	}
	if t.Kind() == reflect.Ptr {
		v := reflect.New(t.Elem())
		if err := json.Unmarshal(b, v.Interface()); err != nil {
			return reflect.Value{}, err
		}
		return v, nil
	}
	v := reflect.New(t)
	if err := json.Unmarshal(b, v.Interface()); err != nil {
		return reflect.Value{}, err
	}
	return v.Elem(), nil
}
