#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="${1:-.}"

if [[ -x "$PROJECT_DIR/mvnw" ]]; then
  MVN="$PROJECT_DIR/mvnw"
elif command -v mvn >/dev/null 2>&1; then
  MVN="mvn"
else
  echo "Maven executable not found" >&2
  exit 1
fi

cd "$PROJECT_DIR"

COMMON_OPTS=(-DgenerateBackupPoms=false -DprocessAllModules=true)

"$MVN" -q "${COMMON_OPTS[@]}" versions:update-parent
"$MVN" -q "${COMMON_OPTS[@]}" versions:use-latest-releases
"$MVN" -q "${COMMON_OPTS[@]}" versions:use-latest-plugins
"$MVN" -q "${COMMON_OPTS[@]}" versions:update-properties

find . -name "*.versionsBackup" -type f -delete
