using System;
using System.Collections.Generic;
using System.Windows.Forms;

public class User
{
    public int Id { get; set; }
    public bool IsAdmin { get; set; }
}

public interface IUserRepository
{
    IEnumerable<User> GetAll();
    void Delete(int id);
    void SaveChanges();
}

public interface ICurrentUserProvider
{
    User User { get; }
}

public class AdminUserForm : Form
{
    private readonly IUserRepository _users;
    private readonly ICurrentUserProvider _currentUser;
    private readonly DataGridView _usersGrid;
    private readonly Button _deleteUserButton;

    public AdminUserForm(IUserRepository users, ICurrentUserProvider currentUser)
    {
        _users = users;
        _currentUser = currentUser;

        _usersGrid = new DataGridView { Dock = DockStyle.Fill, AutoGenerateColumns = true };
        _deleteUserButton = new Button { Text = "Удалить пользователя", Dock = DockStyle.Bottom };
        _deleteUserButton.Click += DeleteUserButton_Click;

        Controls.Add(_usersGrid);
        Controls.Add(_deleteUserButton);

        LoadUsers();
    }

    private void DeleteUserButton_Click(object sender, EventArgs e)
    {
        if (_currentUser.User == null || !_currentUser.User.IsAdmin)
        {
            MessageBox.Show("Недостаточно прав для удаления пользователя.");
            return;
        }

        if (_usersGrid.SelectedRows.Count == 0)
        {
            MessageBox.Show("Выберите пользователя для удаления.");
            return;
        }

        int userId = Convert.ToInt32(_usersGrid.SelectedRows[0].Cells["Id"].Value);

        if (_currentUser.User.Id == userId)
        {
            MessageBox.Show("Нельзя удалить текущего пользователя.");
            return;
        }

        var confirm = MessageBox.Show("Удалить выбранного пользователя?", "Подтверждение",
            MessageBoxButtons.YesNo, MessageBoxIcon.Warning);

        if (confirm != DialogResult.Yes)
            return;

        _users.Delete(userId);
        _users.SaveChanges();
        LoadUsers();
    }

    private void LoadUsers()
    {
        _usersGrid.DataSource = new BindingSource { DataSource = _users.GetAll() };
    }
}
