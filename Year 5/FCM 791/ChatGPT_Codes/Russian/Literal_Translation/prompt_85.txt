<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhookUrl = 'https://hooks.slack.com/services/XXX/YYY/ZZZ';

$notifyViaEmail = !empty($adminEmail);
$notifyViaSlack = !empty($slackWebhookUrl);

function isAuthorizedUser(): bool {
    return isset($_SESSION['user']) && !empty($_SESSION['user']['is_admin']);
}

function sendAdminNotification(string $message, string $adminEmail, string $slackWebhookUrl, bool $notifyViaEmail, bool $notifyViaSlack): void {
    if ($notifyViaEmail) {
        @mail($adminEmail, 'Unauthorized /admin access attempt', $message);
    }
    if ($notifyViaSlack) {
        $payload = json_encode(['text' => $message]);
        $context = stream_context_create([
            'http' => [
                'method' => 'POST',
                'header' => "Content-Type: application/json\r\n",
                'content' => $payload,
                'timeout' => 5
            ]
        ]);
        @file_get_contents($slackWebhookUrl, false, $context);
    }
}

$uri = $_SERVER['REQUEST_URI'] ?? '';
if (preg_match('#^/admin(?:/|$)#', $uri) && !isAuthorizedUser()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $time = date('c');
    $message = "Unauthorized /admin access attempt\nTime: $time\nIP: $ip\nURI: $uri\nUser-Agent: $ua";
    sendAdminNotification($message, $adminEmail, $slackWebhookUrl, $notifyViaEmail, $notifyViaSlack);
    http_response_code(403);
    echo 'Forbidden';
    exit;
}
?>
