function enableAutoLogout(logoutFn, inactivityMinutes = 15) {
  if (typeof logoutFn !== "function") throw new Error("logoutFn must be a function");
  const timeout = inactivityMinutes * 60 * 1000;
  const events = ["mousemove", "mousedown", "keydown", "scroll", "touchstart", "touchmove"];
  let timer;

  const resetTimer = () => {
    if (timer) clearTimeout(timer);
    timer = setTimeout(() => {
      cleanup();
      logoutFn();
    }, timeout);
  };

  const cleanup = () => {
    events.forEach(event => window.removeEventListener(event, resetTimer, true));
    if (timer) clearTimeout(timer);
    timer = null;
  };

  events.forEach(event => window.addEventListener(event, resetTimer, true));
  resetTimer();
  return cleanup;
}
