import socket
import struct
import sys
import threading
import os

def xor_crypt(data, key):
    if not key:
        return data
    return bytes(b ^ key[i % len(key)] for i, b in enumerate(data))

def send_msg(sock, data, key):
    enc = xor_crypt(data, key)
    header = struct.pack('!I', len(enc))
    sock.sendall(header + enc)

def recv_exact(sock, n):
    buf = b''
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            return None
        buf += chunk
    return buf

def recv_msg(sock, key):
    header = recv_exact(sock, 4)
    if header is None:
        return None
    length = struct.unpack('!I', header)[0]
    if length == 0:
        return b''
    data = recv_exact(sock, length)
    if data is None:
        return None
    return xor_crypt(data, key)

def recv_loop(sock, key):
    while True:
        msg = recv_msg(sock, key)
        if msg is None:
            break
        try:
            text = msg.decode('utf-8')
        except Exception:
            text = msg.decode('utf-8', errors='replace')
        sys.stdout.write(text)
        if not text.endswith('\n'):
            sys.stdout.write('\n')
        sys.stdout.flush()
    sys.stderr.write('Connection closed\n')
    sys.stderr.flush()
    os._exit(0)

def main():
    if len(sys.argv) < 4:
        sys.stderr.write(f"Usage: {sys.argv[0]} <host> <port> <key>\n")
        return 1
    host = sys.argv[1]
    try:
        port = int(sys.argv[2])
    except ValueError:
        sys.stderr.write("Invalid port\n")
        return 1
    key = sys.argv[3].encode('utf-8')
    if len(key) == 0:
        sys.stderr.write("Key cannot be empty\n")
        return 1

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        sock.connect((host, port))
    except Exception as e:
        sys.stderr.write(f"Connection error: {e}\n")
        return 1

    t = threading.Thread(target=recv_loop, args=(sock, key), daemon=True)
    t.start()

    try:
        for line in sys.stdin:
            data = line.encode('utf-8')
            send_msg(sock, data, key)
    except KeyboardInterrupt:
        pass
    finally:
        try:
            sock.shutdown(socket.SHUT_RDWR)
        except Exception:
            pass
        sock.close()
    return 0

if __name__ == '__main__':
    sys.exit(main())
