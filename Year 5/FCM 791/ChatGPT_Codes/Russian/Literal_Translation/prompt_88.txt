using System;
using System.Threading.Tasks;
using Microsoft.Extensions.Logging;

public class MfaService
{
    private readonly ILogger<MfaService> _logger;
    private readonly IMfaRepository _repository;

    public MfaService(ILogger<MfaService> logger, IMfaRepository repository)
    {
        _logger = logger;
        _repository = repository;
    }

    public async Task UpdateMfaSettingsAsync(string userId, MfaSettings newSettings)
    {
        await _repository.SaveAsync(userId, newSettings);
        _logger.LogInformation("User {UserId} updated MFA settings at {Timestamp}.", userId, DateTime.UtcNow);
    }
}
