import base64
import hmac
import hashlib
import struct
import time
import os
import sys

SECRET = os.environ.get("TOTP_SECRET", "JBSWY3DPEHPK3PXP")

def _normalize_secret(secret):
    s = secret.strip().replace(" ", "").upper()
    if len(s) % 8:
        s += "=" * (8 - len(s) % 8)
    return s

def generate_totp(secret, for_time=None, interval=30, digits=6):
    if for_time is None:
        for_time = time.time()
    counter = int(for_time // interval)
    key = base64.b32decode(_normalize_secret(secret), casefold=True)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    code = (struct.unpack(">I", h[offset:offset+4])[0] & 0x7FFFFFFF) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(token, secret, interval=30, digits=6, window=1):
    token = token.strip()
    if not token.isdigit() or len(token) != digits:
        return False
    now = time.time()
    for w in range(-window, window + 1):
        if generate_totp(secret, now + w * interval, interval, digits) == token:
            return True
    return False

def main():
    if len(sys.argv) > 1:
        token = sys.argv[1]
    else:
        token = input("Введите 6-значный код: ")
    if verify_totp(token, SECRET):
        print("OK")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
