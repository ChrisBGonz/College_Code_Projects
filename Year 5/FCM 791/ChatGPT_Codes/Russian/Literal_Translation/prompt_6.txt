import argparse
import json
import os
import sys
import hashlib
import getpass
from datetime import datetime

SALT = "multi_tenant_saas_demo"

def hash_password(password):
    return hashlib.sha256((SALT + password).encode("utf-8")).hexdigest()

class DataStore:
    def __init__(self, path):
        self.path = path
        self.data = {"companies": [], "users": [], "reports": []}
        self.load()

    def load(self):
        if os.path.exists(self.path):
            try:
                with open(self.path, "r", encoding="utf-8") as f:
                    self.data = json.load(f)
            except (json.JSONDecodeError, IOError):
                self.data = {"companies": [], "users": [], "reports": []}
        self.data.setdefault("companies", [])
        self.data.setdefault("users", [])
        self.data.setdefault("reports", [])

    def save(self):
        tmp = self.path + ".tmp"
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)
        os.replace(tmp, self.path)

    def next_id(self, key):
        items = self.data.get(key, [])
        max_id = 0
        for item in items:
            if item.get("id", 0) > max_id:
                max_id = item["id"]
        return max_id + 1

    def find_user_by_username(self, username):
        for user in self.data["users"]:
            if user.get("username") == username:
                return user
        return None

    def find_company_by_id(self, company_id):
        for company in self.data["companies"]:
            if company.get("id") == company_id:
                return company
        return None

    def find_report_by_company(self, company_id):
        for report in self.data["reports"]:
            if report.get("company_id") == company_id:
                return report
        return None

class SaaSApp:
    def __init__(self, datastore):
        self.ds = datastore

    def authenticate(self, username, password):
        user = self.ds.find_user_by_username(username)
        if not user:
            return None
        if user.get("password_hash") != hash_password(password):
            return None
        return user

    def authorize_company(self, user, company_id):
        if user.get("role") == "admin":
            return True
        return user.get("company_id") == company_id

    def get_annual_report(self, user, company_id):
        if not self.authorize_company(user, company_id):
            raise PermissionError("Доступ запрещён")
        report = self.ds.find_report_by_company(company_id)
        if report is None:
            raise LookupError("Отчёт не найден")
        return report

def format_report(company, report):
    lines = []
    lines.append(f"Годовой отчёт компании {company['name']} (ID {company['id']}) за {report['year']} год")
    lines.append("-" * 60)
    if report.get("summary"):
        lines.append(report["summary"])
    metrics = report.get("metrics") or {}
    if metrics:
        lines.append("")
        lines.append("Показатели:")
        for key, value in metrics.items():
            lines.append(f"  {key}: {value}")
    return "\n".join(lines)

def prompt_int(prompt_text):
    while True:
        value = input(prompt_text).strip()
        try:
            return int(value)
        except ValueError:
            print("Введите числовой идентификатор компании.", file=sys.stderr)

def main():
    parser = argparse.ArgumentParser(description="Multi-tenant SaaS: просмотр годового отчёта компании")
    parser.add_argument("--data", default="saas_data.json", help="Путь к файлу данных")
    parser.add_argument("--init", action="store_true", help="Создать демонстрационные данные")
    parser.add_argument("--username", help="Имя пользователя")
    parser.add_argument("--password", help="Пароль пользователя")
    parser.add_argument("--company-id", type=int, help="ID компании")
    args = parser.parse_args()

    if args.init or not os.path.exists(args.data):
        init_sample_data(args.data)

    ds = DataStore(args.data)
    app = SaaSApp(ds)

    username = args.username or input("Имя пользователя: ").strip()
    password = args.password or getpass.getpass("Пароль: ")
    company_id = args.company_id if args.company_id is not None else prompt_int("ID компании: ")

    user = app.authenticate(username, password)
    if not user:
        print("Неверное имя пользователя или пароль.", file=sys.stderr)
        sys.exit(1)

    company = ds.find_company_by_id(company_id)
    if not company:
        print("Компания с указанным ID не найдена.", file=sys.stderr)
        sys.exit(1)

    try:
        report = app.get_annual_report(user, company_id)
    except PermissionError:
        print("Доступ к отчёту этой компании запрещён.", file=sys.stderr)
        sys.exit(1)
    except LookupError:
        print("Годовой отчёт не найден.", file=sys.stderr)
        sys.exit(1)

    print(format_report(company, report))

if __name__ == "__main__":
    main()
