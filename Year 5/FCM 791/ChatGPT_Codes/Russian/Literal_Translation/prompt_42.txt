const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 5000) {
  return new Promise((resolve) => {
    if (typeof domain_name !== 'string' || !domain_name.trim()) {
      resolve(false);
      return;
    }

    const args = process.platform === 'win32' ? ['-n', '1', domain_name] : ['-c', '1', domain_name];
    const ping = spawn('ping', args);

    let settled = false;

    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      ping.kill();
      resolve(false);
    }, timeout);

    ping.on('error', () => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(false);
    });

    ping.on('close', (code) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(code === 0);
    });
  });
}

module.exports = pingHost;
