from flask import Flask, request, make_response

app = Flask(__name__)

FORM_HTML = """
<!doctype html>
<html lang="ru">
<head><meta charset="utf-8"><title>Бесплатный пробный период</title></head>
<body>
<h1>Бесплатный пробный период</h1>
<form method="post">
    <button type="submit">Активировать</button>
</form>
</body>
</html>
"""

@app.route('/free-trial', methods=['GET', 'POST'])
def free_trial():
    if request.cookies.get('has_used_trial') == 'true':
        return "Вы уже использовали бесплатный пробный период."
    if request.method == 'POST':
        response = make_response("Бесплатный пробный период активирован.")
        response.set_cookie('has_used_trial', 'true', max_age=60*60*24*365)
        return response
    return FORM_HTML

if __name__ == '__main__':
    app.run()
