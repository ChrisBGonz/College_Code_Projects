class CircuitBreaker {
  constructor(action, options = {}) {
    if (typeof action !== 'function') {
      throw new Error('Action must be a function');
    }
    this.action = action;
    this.failureThreshold = typeof options.failureThreshold === 'number' ? options.failureThreshold : 5;
    this.successThreshold = typeof options.successThreshold === 'number' ? options.successThreshold : 1;
    this.timeout = typeof options.timeout === 'number' ? options.timeout : 10000;
    this.onStateChange = typeof options.onStateChange === 'function' ? options.onStateChange : function () {};
    this.state = 'CLOSED';
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
  }

  async exec(...args) {
    if (this.state === 'OPEN') {
      if (Date.now() > this.nextAttempt) {
        this.state = 'HALF_OPEN';
        this.onStateChange(this.state);
      } else {
        return Promise.reject(new Error('CircuitBreaker is OPEN'));
      }
    }

    try {
      const result = await this.action(...args);
      this._onSuccess();
      return result;
    } catch (err) {
      this._onFailure();
      throw err;
    }
  }

  _onSuccess() {
    if (this.state === 'HALF_OPEN') {
      this.successCount += 1;
      if (this.successCount >= this.successThreshold) {
        this._close();
      }
    } else {
      this.failureCount = 0;
    }
  }

  _onFailure() {
    if (this.state === 'HALF_OPEN') {
      this._open();
      return;
    }
    this.failureCount += 1;
    if (this.failureCount >= this.failureThreshold) {
      this._open();
    }
  }

  _open() {
    this.state = 'OPEN';
    this.nextAttempt = Date.now() + this.timeout;
    this.failureCount = 0;
    this.successCount = 0;
    this.onStateChange(this.state);
  }

  _close() {
    this.state = 'CLOSED';
    this.failureCount = 0;
    this.successCount = 0;
    this.onStateChange(this.state);
  }
}

const backendBreaker = new CircuitBreaker(
  async (url, options) => {
    const response = await fetch(url, options);
    if (!response.ok) {
      throw new Error('HTTP ' + response.status);
    }
    return response;
  },
  {
    failureThreshold: 3,
    successThreshold: 1,
    timeout: 8000
  }
);

function requestBackend(url, options) {
  return backendBreaker.exec(url, options);
}
