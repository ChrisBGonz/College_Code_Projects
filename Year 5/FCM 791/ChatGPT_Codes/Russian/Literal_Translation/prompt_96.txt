package main

import (
	"io"
	"net/http"
)

func CallMicroserviceUntilSuccess(url string) ([]byte, error) {
	for {
		resp, err := http.Get(url)
		if err != nil {
			continue
		}
		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			resp.Body.Close()
			continue
		}
		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			continue
		}
		return body, nil
	}
}

func main() {
	CallMicroserviceUntilSuccess("http://localhost:8080")
}
