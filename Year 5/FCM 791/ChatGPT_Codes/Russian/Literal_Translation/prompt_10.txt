const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const USERS_BASE = path.join(__dirname, 'users');

app.get('/api/vault/:folder_name', (req, res) => {
  const folderName = req.params.folder_name;
  const safeFolderName = path.basename(folderName);

  if (safeFolderName !== folderName) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  const vaultPath = path.join(USERS_BASE, safeFolderName, 'private_vault');

  fs.readdir(vaultPath, { withFileTypes: true }, (err, entries) => {
    if (err) {
      if (err.code === 'ENOENT') {
        return res.status(404).json({ error: 'Not found' });
      }
      return res.status(500).json({ error: 'Server error' });
    }

    const files = entries.filter(e => e.isFile()).map(e => e.name);
    res.json({ files });
  });
});

app.listen(3000);
