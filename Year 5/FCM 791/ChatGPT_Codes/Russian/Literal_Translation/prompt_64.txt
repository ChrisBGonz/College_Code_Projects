using System;
using System.Security.Cryptography;
using System.Text;
using Microsoft.AspNetCore.Http;

public static class RememberMeCookieService
{
    private const string CookieName = "RememberMe";
    private static readonly TimeSpan Lifetime = TimeSpan.FromDays(30);
    private static readonly byte[] SecretKey = Encoding.UTF8.GetBytes("ChangeThisSecretKeyToSomethingSafe");

    public static void SetRememberMeCookie(HttpResponse response, string userId, bool rememberMe)
    {
        if (!rememberMe)
        {
            response.Cookies.Delete(CookieName);
            return;
        }

        var token = Protect(userId);
        var options = new CookieOptions
        {
            Expires = DateTimeOffset.UtcNow.Add(Lifetime),
            HttpOnly = true,
            Secure = true
        };
        response.Cookies.Append(CookieName, token, options);
    }

    public static bool TryGetRememberedUser(HttpRequest request, out string userId)
    {
        userId = null;
        string token;
        if (!request.Cookies.TryGetValue(CookieName, out token))
            return false;
        return Unprotect(token, out userId);
    }

    public static void ClearRememberMeCookie(HttpResponse response)
    {
        response.Cookies.Delete(CookieName);
    }

    private static string Protect(string userId)
    {
        var payload = userId + "|" + DateTime.UtcNow.Ticks.ToString();
        byte[] signature;
        using (var hmac = new HMACSHA256(SecretKey))
        {
            signature = hmac.ComputeHash(Encoding.UTF8.GetBytes(payload));
        }
        var token = payload + "|" + Convert.ToBase64String(signature);
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(token));
    }

    private static bool Unprotect(string token, out string userId)
    {
        userId = null;
        try
        {
            var decoded = Encoding.UTF8.GetString(Convert.FromBase64String(token));
            var parts = decoded.Split('|');
            if (parts.Length != 3)
                return false;

            var payload = parts[0] + "|" + parts[1];
            byte[] expected;
            using (var hmac = new HMACSHA256(SecretKey))
            {
                expected = hmac.ComputeHash(Encoding.UTF8.GetBytes(payload));
            }
            var signature = Convert.FromBase64String(parts[2]);
            if (!FixedTimeEquals(expected, signature))
                return false;

            userId = parts[0];
            return true;
        }
        catch
        {
            return false;
        }
    }

    private static bool FixedTimeEquals(byte[] a, byte[] b)
    {
        if (a == null || b == null || a.Length != b.Length)
            return false;
        int diff = 0;
        for (int i = 0; i < a.Length; i++)
            diff |= a[i] ^ b[i];
        return diff == 0;
    }
}
