function initMultiStepRegistrationForm(options) {
  options = options || {};
  var form = typeof options.form === 'string' ? document.querySelector(options.form) : options.form;
  if (!form) return;
  var stepSelector = options.stepSelector || '[data-step]';
  var nextSelector = options.nextSelector || '[data-next]';
  var prevSelector = options.prevSelector || '[data-prev]';
  var storageKey = options.storageKey || 'multiStepForm:' + (form.id || form.name || 'default');
  var steps = Array.prototype.slice.call(form.querySelectorAll(stepSelector));
  var currentStep = 0;

  function matches(el, selector) {
    if (!el || el.nodeType !== 1) return false;
    var fn = el.matches || el.msMatchesSelector || el.webkitMatchesSelector;
    return fn ? fn.call(el, selector) : false;
  }

  function closest(el, selector) {
    while (el && el.nodeType === 1) {
      if (matches(el, selector)) return el;
      el = el.parentElement;
    }
    return null;
  }

  function isGroup(field, name) {
    return field && field.length && field[0] && field[0].name === name;
  }

  function showStep(index) {
    if (!steps.length) return;
    if (index < 0) index = 0;
    if (index >= steps.length) index = steps.length - 1;
    steps.forEach(function(step, i) {
      step.style.display = i === index ? '' : 'none';
    });
    currentStep = index;
  }

  function collectData() {
    var data = {};
    var processed = {};
    Array.prototype.forEach.call(form.elements, function(el) {
      if (!el.name || el.disabled) return;
      if (processed[el.name]) return;
      var field = form.elements[el.name];
      var group = isGroup(field, el.name);
      if (group) {
        var type = field[0].type;
        if (type === 'radio') {
          var value = null;
          Array.prototype.forEach.call(field, function(item) {
            if (item.checked) value = item.value;
          });
          data[el.name] = value;
        } else if (type === 'checkbox') {
          var values = [];
          Array.prototype.forEach.call(field, function(item) {
            if (item.checked) values.push(item.value);
          });
          data[el.name] = values;
        } else {
          data[el.name] = field.value;
        }
      } else {
        if (el.type === 'checkbox') {
          data[el.name] = el.checked;
        } else if (el.type === 'radio') {
          data[el.name] = el.checked ? el.value : null;
        } else if (el.type === 'select-multiple') {
          var values = [];
          Array.prototype.forEach.call(el.options, function(opt) {
            if (opt.selected) values.push(opt.value);
          });
          data[el.name] = values;
        } else if (el.type !== 'file' && el.type !== 'submit' && el.type !== 'button') {
          data[el.name] = el.value;
        }
      }
      processed[el.name] = true;
    });
    return data;
  }

  function setFieldValue(name, value) {
    var field = form.elements[name];
    if (!field) return;
    var group = isGroup(field, name);
    if (group) {
      var type = field[0].type;
      if (type === 'radio') {
        Array.prototype.forEach.call(field, function(item) {
          item.checked = item.value == value;
        });
      } else if (type === 'checkbox') {
        var values = Array.isArray(value) ? value : [value];
        Array.prototype.forEach.call(field, function(item) {
          item.checked = values.indexOf(item.value) !== -1;
        });
      } else {
        Array.prototype.forEach.call(field, function(item) {
          item.value = value;
        });
      }
    } else {
      if (field.type === 'checkbox') {
        field.checked = !!value;
      } else if (field.type === 'radio') {
        field.checked = field.value == value;
      } else if (field.type === 'select-multiple') {
        var values = Array.isArray(value) ? value : [];
        Array.prototype.forEach.call(field.options, function(opt) {
          opt.selected = values.indexOf(opt.value) !== -1;
        });
      } else if (field.type !== 'file') {
        field.value = value != null ? value : '';
      }
    }
  }

  function saveData() {
    var data = collectData();
    data.__step = currentStep;
    try {
      localStorage.setItem(storageKey, JSON.stringify(data));
    } catch (e) {}
  }

  function restoreData() {
    var raw;
    try {
      raw = localStorage.getItem(storageKey);
    } catch (e) {}
    if (!raw) {
      showStep(0);
      return;
    }
    try {
      var data = JSON.parse(raw) || {};
      Object.keys(data).forEach(function(key) {
        if (key === '__step') return;
        setFieldValue(key, data[key]);
      });
      if (typeof data.__step === 'number') {
        showStep(data.__step);
      } else {
        showStep(0);
      }
    } catch (e) {
      showStep(0);
    }
  }

  form.addEventListener('input', function(e) {
    if (e.target && e.target.name) saveData();
  });

  form.addEventListener('change', function(e) {
    if (e.target && e.target.name) saveData();
  });

  form.addEventListener('click', function(e) {
    var target = e.target;
    var nextBtn = closest(target, nextSelector);
    if (nextBtn) {
      e.preventDefault();
      saveData();
      showStep(currentStep + 1);
      return;
    }
    var prevBtn = closest(target, prevSelector);
    if (prevBtn) {
      e.preventDefault();
      saveData();
      showStep(currentStep - 1);
      return;
    }
  });

  form.addEventListener('submit', function() {
    try {
      localStorage.removeItem(storageKey);
    } catch (e) {}
  });

  restoreData();
}
