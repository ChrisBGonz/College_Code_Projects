import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Set, Tuple

@dataclass
class Item:
    name: str
    quantity: int
    attributes: Dict[str, int] = field(default_factory=dict)

@dataclass
class Quest:
    title: str
    is_completed: bool
    objectives: List[str] = field(default_factory=list)

@dataclass
class Player:
    name: str
    level: int
    health: int
    mana: int
    position: Tuple[int, int, int]
    inventory: List[Item] = field(default_factory=list)

@dataclass
class WorldState:
    time_of_day: str
    weather: str
    discovered_locations: Set[str] = field(default_factory=set)
    active_quests: Dict[str, Quest] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world: WorldState
    gold: int
    achievements: List[str] = field(default_factory=list)

def save_game_state(game_state: GameState, filename: str) -> None:
    with open(filename, "wb") as file:
        pickle.dump(game_state, file, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as file:
        return pickle.load(file)

if __name__ == "__main__":
    player = Player(
        name="Игрок",
        level=7,
        health=120,
        mana=60,
        position=(15, 34, 3),
        inventory=[
            Item("Меч", 1, {"damage": 12}),
            Item("Зелье лечения", 5, {"heal": 25}),
            Item("Щит", 1, {"defense": 8})
        ]
    )

    world = WorldState(
        time_of_day="Вечер",
        weather="Облачно",
        discovered_locations={"Деревня", "Лес", "Горы"},
        active_quests={
            "quest_001": Quest("Найти артефакт", False, ["Войти в пещеру", "Найти артефакт"]),
            "quest_002": Quest("Спасти торговца", True, ["Найти лагерь", "Освободить торговца"])
        }
    )

    game_state = GameState(
        player=player,
        world=world,
        gold=350,
        achievements=["Первый бой", "Исследователь"]
    )

    filename = "game_state.pkl"
    save_game_state(game_state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)
