<?php
session_start();

$dbFile = __DIR__ . '/users.sqlite';
$pdo = new PDO('sqlite:' . $dbFile);
$pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
$pdo->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    secret_question TEXT NOT NULL,
    secret_answer_hash TEXT NOT NULL
)");

function clearRecovery() {
    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_username'], $_SESSION['secret_question'], $_SESSION['recovery_verified']);
}

function h($v) {
    return htmlspecialchars((string)$v, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
}

if (isset($_GET['restart'])) {
    clearRecovery();
}

$errors = [];
$success = false;
$inputUsername = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $step = $_POST['step'] ?? '';
    if ($step === 'identify') {
        $inputUsername = trim($_POST['username'] ?? '');
        if ($inputUsername === '') {
            $errors[] = 'Введите имя пользователя.';
        } else {
            $stmt = $pdo->prepare('SELECT id, username, secret_question FROM users WHERE username = ?');
            $stmt->execute([$inputUsername]);
            $user = $stmt->fetch(PDO::FETCH_ASSOC);
            if (!$user) {
                $errors[] = 'Пользователь не найден.';
            } else {
                $_SESSION['recovery_user_id'] = (int)$user['id'];
                $_SESSION['recovery_username'] = $user['username'];
                $_SESSION['secret_question'] = $user['secret_question'];
                unset($_SESSION['recovery_verified']);
            }
        }
    } elseif ($step === 'answer') {
        if (!isset($_SESSION['recovery_user_id'])) {
            $errors[] = 'Сеанс истёк. Начните заново.';
        } else {
            $answer = trim($_POST['answer'] ?? '');
            if ($answer === '') {
                $errors[] = 'Введите ответ на секретный вопрос.';
            } else {
                $stmt = $pdo->prepare('SELECT secret_answer_hash FROM users WHERE id = ?');
                $stmt->execute([$_SESSION['recovery_user_id']]);
                $hash = $stmt->fetchColumn();
                $ok = false;
                if ($hash) {
                    if (password_verify($answer, $hash)) {
                        $ok = true;
                    } else {
                        if (function_exists('mb_strtolower')) {
                            $ok = password_verify(mb_strtolower($answer, 'UTF-8'), $hash);
                        } else {
                            $ok = password_verify(strtolower($answer), $hash);
                        }
                    }
                }
                if (!$ok) {
                    $errors[] = 'Неверный ответ.';
                } else {
                    $_SESSION['recovery_verified'] = true;
                }
            }
        }
    } elseif ($step === 'reset') {
        if (!($_SESSION['recovery_verified'] ?? false)) {
            $errors[] = 'Подтверждение не пройдено.';
        } else {
            $pass1 = $_POST['password'] ?? '';
            $pass2 = $_POST['password_confirm'] ?? '';
            if (strlen($pass1) < 6) {
                $errors[] = 'Пароль должен быть не короче 6 символов.';
            } elseif ($pass1 !== $pass2) {
                $errors[] = 'Пароли не совпадают.';
            } else {
                $hash = password_hash($pass1, PASSWORD_DEFAULT);
                $stmt = $pdo->prepare('UPDATE users SET password_hash = ? WHERE id = ?');
                $stmt->execute([$hash, $_SESSION['recovery_user_id']]);
                $success = true;
                clearRecovery();
            }
        }
    }
}

$state = 'identify';
if (isset($_SESSION['recovery_verified']) && $_SESSION['recovery_verified']) {
    $state = 'reset';
} elseif (isset($_SESSION['recovery_user_id'])) {
    $state = 'question';
}
?>
<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<title>Восстановление доступа</title>
<style>
body{font-family:Arial, sans-serif;max-width:480px;margin:40px auto;padding:0 10px;}
label{display:block;margin-top:12px;}
input{width:100%;padding:8px;margin-top:4px;box-sizing:border-box;}
button{margin-top:14px;padding:8px 14px;}
ul{color:#b00;}
.success{color:#080;}
a{display:inline-block;margin-top:12px;}
</style>
</head>
<body>
<h2>Восстановление доступа</h2>
<?php if ($errors): ?>
<ul>
<?php foreach ($errors as $e): ?>
<li><?php echo h($e); ?></li>
<?php endforeach; ?>
</ul>
<?php endif; ?>
<?php if ($success): ?>
<p class="success">Пароль успешно изменён. Теперь вы можете войти в систему.</p>
<?php else: ?>
<?php if ($state === 'identify'): ?>
<form method="post">
<input type="hidden" name="step" value="identify">
<label>Имя пользователя</label>
<input type="text" name="username" value="<?php echo h($inputUsername); ?>" required>
<button type="submit">Продолжить</button>
</form>
<?php elseif ($state === 'question'): ?>
<p>Пользователь: <?php echo h($_SESSION['recovery_username'] ?? ''); ?></p>
<p>Секретный вопрос: <?php echo h($_SESSION['secret_question'] ?? ''); ?></p>
<form method="post">
<input type="hidden" name="step" value="answer">
<label>Ответ</label>
<input type="text" name="answer" required>
<button type="submit">Проверить</button>
</form>
<a href="?restart=1">Начать заново</a>
<?php elseif ($state === 'reset'): ?>
<form method="post">
<input type="hidden" name="step" value="reset">
<label>Новый пароль</label>
<input type="password" name="password" required>
<label>Повторите пароль</label>
<input type="password" name="password_confirm" required>
<button type="submit">Сменить пароль</button>
</form>
<a href="?restart=1">Начать заново</a>
<?php endif; ?>
<?php endif; ?>
</body>
</html>
