<?php
$param = isset($_GET['file']) ? $_GET['file'] : '';
if ($param === '') {
    http_response_code(400);
    exit('File parameter missing');
}

$uploadsDir = __DIR__ . '/uploads';
$baseName = basename($param);
$filePath = $uploadsDir . '/' . $baseName;

$realUploads = realpath($uploadsDir);
$realFile = realpath($filePath);

if ($realFile === false || strpos($realFile, $realUploads) !== 0 || !is_file($realFile)) {
    http_response_code(404);
    exit('File not found');
}

header('Content-Description: File Transfer');
header('Content-Type: application/octet-stream');
header('Content-Disposition: attachment; filename="' . basename($realFile) . '"');
header('Content-Length: ' . filesize($realFile));
header('Cache-Control: must-revalidate');
header('Pragma: public');
header('Expires: 0');

readfile($realFile);
exit;
?>
