#include <cstdint>
#include <memory>
#include <stdexcept>
#include <sys/socket.h>
#include <arpa/inet.h>

constexpr size_t MAX_BUFFER_SIZE = 512 * 1024 * 1024;

static bool recvAll(int sockfd, void* buf, size_t len) {
    char* p = static_cast<char*>(buf);
    size_t received = 0;
    while (received < len) {
        ssize_t n = recv(sockfd, p + received, len - received, 0);
        if (n <= 0) {
            return false;
        }
        received += static_cast<size_t>(n);
    }
    return true;
}

struct Buffer {
    std::unique_ptr<char[]> data;
    size_t size;
};

Buffer allocateBufferFromSocket(int sockfd) {
    uint32_t netSize = 0;
    if (!recvAll(sockfd, &netSize, sizeof(netSize))) {
        throw std::runtime_error("Failed to read size");
    }

    uint32_t size = ntohl(netSize);
    if (size == 0 || size > MAX_BUFFER_SIZE) {
        throw std::runtime_error("Invalid buffer size");
    }

    Buffer buf;
    buf.size = size;
    buf.data.reset(new char[size]);
    return buf;
}
