<?php
$uploadDir = '/mnt/corporate/share/uploads';
$maxSize = 20 * 1024 * 1024;

$message = '';
$success = false;

if (!is_dir($uploadDir)) {
    if (!mkdir($uploadDir, 0770, true)) {
        $message = 'Не удалось создать каталог загрузки.';
    }
}

if ($message === '' && !is_writable($uploadDir)) {
    $message = 'Каталог загрузки недоступен для записи.';
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && $message === '') {
    $file = $_FILES['pdf'] ?? null;

    if (!$file) {
        $message = 'Файл не выбран.';
    } elseif ($file['error'] !== UPLOAD_ERR_OK) {
        $errors = [
            UPLOAD_ERR_INI_SIZE => 'Файл превышает допустимый размер.',
            UPLOAD_ERR_FORM_SIZE => 'Файл превышает допустимый размер.',
            UPLOAD_ERR_PARTIAL => 'Файл загружен не полностью.',
            UPLOAD_ERR_NO_FILE => 'Файл не выбран.',
            UPLOAD_ERR_NO_TMP_DIR => 'Нет временного каталога.',
            UPLOAD_ERR_CANT_WRITE => 'Не удалось записать файл на диск.',
            UPLOAD_ERR_EXTENSION => 'Загрузка файла остановлена расширением PHP.'
        ];
        $message = $errors[$file['error']] ?? 'Ошибка загрузки файла.';
    } elseif ($file['size'] > $maxSize) {
        $message = 'Размер файла превышает допустимый лимит.';
    } else {
        $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        if ($ext !== 'pdf') {
            $message = 'Разрешены только PDF-файлы.';
        } else {
            $finfo = new finfo(FILEINFO_MIME_TYPE);
            $mime = $finfo->file($file['tmp_name']);
            if ($mime !== 'application/pdf' && $mime !== 'application/x-pdf') {
                $message = 'Файл не является корректным PDF.';
            } else {
                $base = pathinfo($file['name'], PATHINFO_FILENAME);
                $base = preg_replace('/[^a-zA-Z0-9_\-]+/', '_', $base);
                if ($base === '') {
                    $base = 'file';
                }
                $target = $uploadDir . DIRECTORY_SEPARATOR . $base . '_' . date('Ymd_His') . '_' . bin2hex(random_bytes(4)) . '.pdf';
                if (move_uploaded_file($file['tmp_name'], $target)) {
                    chmod($target, 0660);
                    $success = true;
                    $message = 'Файл успешно загружен.';
                } else {
                    $message = 'Не удалось сохранить файл.';
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<title>Загрузка PDF-файлов</title>
</head>
<body>
<?php if ($message !== ''): ?>
<p style="color:<?php echo $success ? 'green' : 'red'; ?>;"><?php echo htmlspecialchars($message, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8'); ?></p>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
<label>Выберите PDF-файл:</label>
<input type="file" name="pdf" accept="application/pdf,.pdf" required>
<button type="submit">Загрузить</button>
</form>
</body>
</html>
