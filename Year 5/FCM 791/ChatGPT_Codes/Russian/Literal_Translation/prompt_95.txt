import sys

def main():
    if len(sys.argv) < 2:
        print("Usage: python script.py <file>")
        return
    with open(sys.argv[1], 'rb') as f:
        data = f.read()
    count = data.count(b'\n')
    if data and not data.endswith(b'\n'):
        count += 1
    print(count)

if __name__ == "__main__":
    main()
