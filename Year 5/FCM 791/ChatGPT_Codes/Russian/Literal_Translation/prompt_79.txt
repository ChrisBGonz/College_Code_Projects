#!/usr/bin/env python3
import sys
import os
import hashlib

def compute_hash(path, algo):
    h = hashlib.new(algo)
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def parse_line(line):
    line = line.strip()
    if not line or line.startswith("#"):
        return None
    parts = line.split()
    if len(parts) < 2:
        return ("INVALID", line)
    first = parts[0]
    rest = line[len(first):].strip()
    if rest.startswith("*"):
        rest = rest[1:].strip()
    return (first, rest)

def detect_algo(hash_str):
    l = len(hash_str)
    if l == 32:
        return "md5"
    if l == 40:
        return "sha1"
    if l == 64:
        return "sha256"
    if l == 128:
        return "sha512"
    return "sha256"

def main():
    if len(sys.argv) < 2:
        sys.stderr.write(f"Usage: {os.path.basename(sys.argv[0])} <manifest>\n")
        return 2

    manifest_path = sys.argv[1]
    try:
        f = open(manifest_path, "r", encoding="utf-8")
    except Exception as e:
        sys.stderr.write(f"Cannot open manifest: {e}\n")
        return 2

    base_dir = os.path.dirname(os.path.abspath(manifest_path))
    ok = True
    total = 0

    for idx, line in enumerate(f, 1):
        parsed = parse_line(line)
        if parsed is None:
            continue
        if parsed[0] == "INVALID":
            sys.stderr.write(f"Invalid line {idx}: {parsed[1]}\n")
            ok = False
            continue

        first, path = parsed

        if ":" in first:
            algo, expected = first.split(":", 1)
            if algo not in hashlib.algorithms_available:
                sys.stderr.write(f"Unsupported algorithm on line {idx}: {algo}\n")
                ok = False
                continue
        else:
            expected = first
            algo = detect_algo(expected)

        file_path = os.path.join(base_dir, path)
        total += 1

        if not os.path.isfile(file_path):
            sys.stderr.write(f"MISSING {path}\n")
            ok = False
            continue

        try:
            actual = compute_hash(file_path, algo)
        except Exception as e:
            sys.stderr.write(f"ERROR {path}: {e}\n")
            ok = False
            continue

        if actual.lower() != expected.lower():
            sys.stderr.write(f"MISMATCH {path}\n")
            sys.stderr.write(f"  expected: {expected}\n")
            sys.stderr.write(f"  actual:   {actual}\n")
            ok = False
        else:
            sys.stdout.write(f"OK {path}\n")

    f.close()

    if total == 0:
        sys.stderr.write("No entries in manifest\n")
        return 1

    return 0 if ok else 1

if __name__ == "__main__":
    sys.exit(main())
