#!/usr/bin/env python3
import sys
import time
import os
from collections import deque

WINDOW = 60
THRESHOLD = 10
PATTERN = "Failed Login"

def follow(file):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if line:
            yield line
        else:
            time.sleep(0.1)

def monitor(path):
    events = deque()
    warned = False
    with open(path, "r") as f:
        for line in follow(f):
            now = time.time()
            while events and now - events[0] > WINDOW:
                events.popleft()
            if PATTERN in line:
                events.append(now)
                if len(events) > THRESHOLD and not warned:
                    print(f"WARNING: More than {THRESHOLD} '{PATTERN}' messages in the last minute")
                    warned = True
            if len(events) <= THRESHOLD:
                warned = False

def main():
    if len(sys.argv) < 2:
        sys.stderr.write(f"Usage: {sys.argv[0]} <logfile>\n")
        sys.exit(1)
    monitor(sys.argv[1])

if __name__ == "__main__":
    main()
