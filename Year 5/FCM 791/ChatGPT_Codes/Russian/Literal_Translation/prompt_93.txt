using System;
using System.Data.SqlClient;

var connectionString = "your_connection_string";
using (var connection = new SqlConnection(connectionString))
{
    connection.Open();
    using (var transaction = connection.BeginTransaction())
    {
        try
        {
            var debit = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @accountId",
                connection, transaction);
            debit.Parameters.AddWithValue("@amount", 100);
            debit.Parameters.AddWithValue("@accountId", "A");
            if (debit.ExecuteNonQuery() != 1) throw new Exception("Account A not found");

            var credit = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @accountId",
                connection, transaction);
            credit.Parameters.AddWithValue("@amount", 100);
            credit.Parameters.AddWithValue("@accountId", "B");
            if (credit.ExecuteNonQuery() != 1) throw new Exception("Account B not found");

            transaction.Commit();
        }
        catch
        {
            transaction.Rollback();
            throw;
        }
    }
}
