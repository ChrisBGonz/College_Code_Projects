import hashlib
import argparse
import sys

def sha256sum(path, chunk_size=8192):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(chunk_size), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify firmware file integrity using SHA256.")
    parser.add_argument("file", help="Path to firmware file")
    parser.add_argument("expected_hash", help="Expected SHA256 hash")
    args = parser.parse_args()

    try:
        actual_hash = sha256sum(args.file)
    except FileNotFoundError:
        print(f"File not found: {args.file}", file=sys.stderr)
        return 2
    except Exception as e:
        print(f"Error reading file: {e}", file=sys.stderr)
        return 2

    if actual_hash.lower() == args.expected_hash.strip().lower():
        print("Integrity check passed")
        return 0
    else:
        print("Integrity check failed")
        print("Expected:", args.expected_hash.strip().lower())
        print("Actual:  ", actual_hash)
        return 1

if __name__ == "__main__":
    sys.exit(main())
