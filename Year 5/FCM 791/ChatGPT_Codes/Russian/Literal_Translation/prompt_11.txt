import sys
from lxml import etree

def main():
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
    else:
        try:
            file_path = input().strip()
        except EOFError:
            return
    if not file_path:
        return
    try:
        tree = etree.parse(file_path)
    except Exception:
        return
    titles = tree.xpath('//title')
    for t in titles:
        print(t.text or "")

if __name__ == "__main__":
    main()
