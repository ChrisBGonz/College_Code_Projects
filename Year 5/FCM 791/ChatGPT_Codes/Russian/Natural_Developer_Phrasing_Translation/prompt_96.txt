package retry

import (
	"io"
	"net/http"
)

func CallMicroserviceUntilSuccess(url string) ([]byte, error) {
	client := &http.Client{}
	for {
		resp, err := client.Get(url)
		if err != nil {
			continue
		}
		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			continue
		}
		if resp.StatusCode >= 200 && resp.StatusCode < 300 {
			return body, nil
		}
	}
}
