<?php
session_start();

define('USER_FILE', __DIR__ . '/users.json');

function h($s) {
    return htmlspecialchars((string)$s, ENT_QUOTES, 'UTF-8');
}

function normalize_answer($s) {
    $s = trim((string)$s);
    if (function_exists('mb_strtolower')) {
        return mb_strtolower($s, 'UTF-8');
    }
    return strtolower($s);
}

function load_users() {
    if (!file_exists(USER_FILE)) {
        $default_users = [
            'ivan' => [
                'email' => 'ivan@example.com',
                'password_hash' => password_hash('password123', PASSWORD_DEFAULT),
                'secret_question' => 'Девичья фамилия матери?',
                'secret_answer' => password_hash(normalize_answer('Иванова'), PASSWORD_DEFAULT)
            ],
            'maria' => [
                'email' => 'maria@example.com',
                'password_hash' => password_hash('qwerty123', PASSWORD_DEFAULT),
                'secret_question' => 'Название первой школы?',
                'secret_answer' => password_hash(normalize_answer('школа №1'), PASSWORD_DEFAULT)
            ]
        ];
        file_put_contents(USER_FILE, json_encode($default_users, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
    }
    $data = json_decode(@file_get_contents(USER_FILE), true);
    if (!is_array($data)) {
        $data = [];
    }
    return $data;
}

function save_users($users) {
    file_put_contents(USER_FILE, json_encode($users, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

function find_user($users, $identifier) {
    foreach ($users as $username => $user) {
        if (strcasecmp($username, $identifier) === 0) {
            return $username;
        }
        if (isset($user['email']) && strcasecmp($user['email'], $identifier) === 0) {
            return $username;
        }
    }
    return null;
}

function reset_recovery() {
    unset($_SESSION['recovery_user'], $_SESSION['recovery_stage']);
}

$users = load_users();
$errors = [];
$messages = [];

$stage = $_SESSION['recovery_stage'] ?? 'identify';
$recovery_user = $_SESSION['recovery_user'] ?? null;

if (isset($_POST['cancel'])) {
    reset_recovery();
    $stage = 'identify';
    $recovery_user = null;
    $messages[] = 'Восстановление отменено.';
}

if (isset($_POST['start_over'])) {
    reset_recovery();
    $stage = 'identify';
    $recovery_user = null;
}

if (isset($_POST['start_recovery'])) {
    $identifier = trim($_POST['identifier'] ?? '');
    if ($identifier === '') {
        $errors[] = 'Введите логин или e-mail.';
        $stage = 'identify';
    } else {
        $username = find_user($users, $identifier);
        if ($username === null) {
            $errors[] = 'Пользователь не найден.';
            $stage = 'identify';
        } else {
            $recovery_user = $username;
            $stage = 'question';
        }
    }
}

if (isset($_POST['answer_question'])) {
    if (!$recovery_user || !isset($users[$recovery_user])) {
        $errors[] = 'Сеанс восстановления истек. Начните заново.';
        reset_recovery();
        $stage = 'identify';
        $recovery_user = null;
    } else {
        $answer = trim($_POST['answer'] ?? '');
        if ($answer === '') {
            $errors[] = 'Введите ответ на секретный вопрос.';
            $stage = 'question';
        } else {
            $hash = $users[$recovery_user]['secret_answer'] ?? '';
            if (!$hash || !password_verify(normalize_answer($answer), $hash)) {
                $errors[] = 'Неверный ответ.';
                $stage = 'question';
            } else {
                $stage = 'reset';
            }
        }
    }
}

if (isset($_POST['reset_password'])) {
    if (!$recovery_user || !isset($users[$recovery_user])) {
        $errors[] = 'Сеанс восстановления истек. Начните заново.';
        reset_recovery();
        $stage = 'identify';
        $recovery_user = null;
    } else {
        $pass1 = $_POST['new_password'] ?? '';
        $pass2 = $_POST['new_password_confirm'] ?? '';
        if ($pass1 === '' || $pass2 === '') {
            $errors[] = 'Введите новый пароль дважды.';
            $stage = 'reset';
        } elseif ($pass1 !== $pass2) {
            $errors[] = 'Пароли не совпадают.';
            $stage = 'reset';
        } elseif (strlen($pass1) < 6) {
            $errors[] = 'Пароль должен быть не менее 6 символов.';
            $stage = 'reset';
        } else {
            $users[$recovery_user]['password_hash'] = password_hash($pass1, PASSWORD_DEFAULT);
            save_users($users);
            reset_recovery();
            $stage = 'done';
            $messages[] = 'Пароль успешно изменен.';
        }
    }
}

if (($stage === 'question' || $stage === 'reset') && (!$recovery_user || !isset($users[$recovery_user]))) {
    $errors[] = 'Сеанс восстановления истек. Начните заново.';
    reset_recovery();
    $stage = 'identify';
    $recovery_user = null;
}

if ($stage === 'question' || $stage === 'reset') {
    $_SESSION['recovery_stage'] = $stage;
    $_SESSION['recovery_user'] = $recovery_user;
} else {
    unset($_SESSION['recovery_stage'], $_SESSION['recovery_user']);
}

<body>
<div class="container">
<h1>Восстановление доступа</h1>
<?php if ($errors): ?>
<div class="error"><ul>
<?php foreach ($errors as $e): ?>
<li><?php echo h($e); ?></li>
<?php endforeach; ?>
</ul></div>
<?php endif; ?>
<?php if ($messages): ?>
<div class="message"><ul>
<?php foreach ($messages as $m): ?>
<li><?php echo h($m); ?></li>
<?php endforeach; ?>
</ul></div>
<?php endif; ?>

<?php if ($stage === 'identify'): ?>
<p>Введите логин или e-mail, чтобы получить секретный вопрос.</p>
<form method="post">
<label for="identifier">Логин или e-mail</label>
<input type="text" id="identifier" name="identifier" required autofocus>
<div class="actions">
<button type="submit" name="start_recovery">Продолжить</button>
</div>
</form>
<?php elseif ($stage === 'question'): ?>
<?php $question = $users[$recovery_user]['secret_question'] ?? ''; ?>
<p>Пользователь: <strong><?php echo h($recovery_user); ?></strong></p>
<p>Секретный вопрос: <strong><?php echo h($question); ?></strong></p>
<form method="post">
<label for="answer">Ответ</label>
<input type="text" id="answer" name="answer" required autofocus>
<div class="actions">
<button type="submit" name="answer_question">Проверить</button>
<button type="submit" name="cancel" value="1">Отмена</button>
</div>
</form>
<?php elseif ($stage === 'reset'): ?>
<p>Ответ принят. Создайте новый пароль.</p>
<form method="post">
<label for="new_password">Новый пароль</label>
<input type="password" id="new_password" name="new_password" required>

<input type="password" id="new_password_confirm" name="new_password_confirm" required>
<div class="actions">
<button type="submit" name="reset_password">Сохранить пароль</button>
<button type="submit" name="cancel" value="1">Отмена</button>
</div>
</form>
<?php elseif ($stage === 'done'): ?>
<p>Пароль успешно изменен. Теперь вы можете войти в систему.</p>
<form method="post">
<button type="submit" name="start_over">Начать заново</button>
</form>
<?php endif; ?>
</div>
</body>
