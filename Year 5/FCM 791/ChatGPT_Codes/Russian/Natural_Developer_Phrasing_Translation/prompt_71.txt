import pickle
from dataclasses import dataclass, field
from typing import Dict, List, Set, Tuple

@dataclass
class Item:
    name: str
    quantity: int
    attributes: Dict[str, int] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    health: int
    position: Tuple[int, int]
    inventory: List[Item] = field(default_factory=list)
    skills: Dict[str, int] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world_time: float
    quests: Dict[str, str] = field(default_factory=dict)
    visited_locations: Set[str] = field(default_factory=set)
    flags: Dict[str, bool] = field(default_factory=dict)

def create_sample_game_state() -> GameState:
    sword = Item(name="Sword", quantity=1, attributes={"damage": 15, "durability": 40})
    potion = Item(name="Health Potion", quantity=3, attributes={"heal": 25})
    player = Player(
        name="Hero",
        level=5,
        health=80,
        position=(10, 20),
        inventory=[sword, potion],
        skills={"archery": 2, "swordsmanship": 4}
    )
    return GameState(
        player=player,
        world_time=1234.56,
        quests={"Find the relic": "in_progress", "Save the village": "completed"},
        visited_locations={"Town", "Forest", "Dungeon"},
        flags={"intro_seen": True, "boss_defeated": False}
    )

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def main():
    filename = "game_state.pkl"
    state = create_sample_game_state()
    save_game_state(state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
