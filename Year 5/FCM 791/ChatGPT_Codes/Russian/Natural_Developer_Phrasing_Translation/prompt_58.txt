function initMultiStepRegistrationForm(formOrSelector, options) {
  options = options || {};
  var form = typeof formOrSelector === 'string' ? document.querySelector(formOrSelector) : formOrSelector;
  if (!form) return null;

  var settings = {
    storageKey: options.storageKey || ('msf:' + (form.id || form.getAttribute('name') || 'registration')),
    stepSelector: options.stepSelector || '[data-step]',
    activeClass: options.activeClass || 'is-active',
    nextSelector: options.nextSelector || '[data-step-next]',
    prevSelector: options.prevSelector || '[data-step-prev]',
    gotoSelector: options.gotoSelector || '[data-step-target]',
    clearOnSubmit: options.clearOnSubmit !== false,
    useHidden: options.useHidden !== false,
    onSubmit: options.onSubmit || null,
    syncOnSubmit: options.syncOnSubmit || null
  };

  var storageAvailable = true;
  try {
    var testKey = '__msf_test__';
    localStorage.setItem(testKey, '1');
    localStorage.removeItem(testKey);
  } catch (e) {
    storageAvailable = false;
  }

  var steps = form.querySelectorAll(settings.stepSelector);
  var currentStep = 0;

  if (steps.length) {
    for (var si = 0; si < steps.length; si++) {
      var s = steps[si];
      var active = s.classList.contains(settings.activeClass) || s.getAttribute('aria-hidden') === 'false' || s.getAttribute('data-active') === 'true' || (!s.hidden && settings.useHidden);
      if (active) {
        currentStep = si;
        break;
      }
    }
  }

  function esc(name) {
    if (window.CSS && typeof CSS.escape === 'function') return CSS.escape(name);
    return String(name).replace(/([ #;?%&,.+*~\':"!^$[\]()=>|\/@])/g, '\\$1');
  }

  function matches(el, selector) {
    if (!el || el.nodeType !== 1) return false;
    var p = Element.prototype;
    var fn = p.matches || p.msMatchesSelector || p.webkitMatchesSelector || p.mozMatchesSelector;
    if (fn) return fn.call(el, selector);
    return false;
  }

  function closest(el, selector) {
    while (el && el !== form) {
      if (matches(el, selector)) return el;
      el = el.parentElement;
    }
    return null;
  }

  function readStorage() {
    if (!storageAvailable) return null;
    try {
      var raw = localStorage.getItem(settings.storageKey);
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) {
      return null;
    }
  }

  function writeStorage(data) {
    if (!storageAvailable) return;
    try {
      localStorage.setItem(settings.storageKey, JSON.stringify(data));
    } catch (e) {}
  }

  function clearStorage() {
    if (!storageAvailable) return;
    try {
      localStorage.removeItem(settings.storageKey);
    } catch (e) {}
  }

  function collectData() {
    var data = {};
    var elements = form.elements;
    for (var i = 0; i < elements.length; i++) {
      var el = elements[i];
      if (!el.name || el.disabled) continue;
      var name = el.name;
      var type = (el.type || '').toLowerCase();
      if (type === 'file') continue;
      if (type === 'checkbox') {
        var group = form.querySelectorAll('input[type="checkbox"][name="' + esc(name) + '"]');
        if (group.length > 1) {
          if (!Array.isArray(data[name])) data[name] = [];
          if (el.checked) data[name].push(el.value);
        } else {
          data[name] = !!el.checked;
        }
      } else if (type === 'radio') {
        if (el.checked) {
          data[name] = el.value;
        } else if (!(name in data)) {
          data[name] = null;
        }
      } else if (type === 'select-multiple') {
        var vals = [];
        for (var j = 0; j < el.options.length; j++) {
          var opt = el.options[j];
          if (opt.selected) vals.push(opt.value);
        }
        data[name] = vals;
      } else {
        data[name] = el.value;
      }
    }
    data.__step = currentStep;
    return data;
  }

  function saveData() {
    writeStorage(collectData());
  }

  function loadData() {
    var data = readStorage();
    if (!data) return;
    for (var name in data) {
      if (!data.hasOwnProperty(name) || name === '__step') continue;
      var value = data[name];
      var fields = form.querySelectorAll('[name="' + esc(name) + '"]');
      if (!fields.length) continue;
      for (var i = 0; i < fields.length; i++) {
        var field = fields[i];
        var type = (field.type || '').toLowerCase();
        if (type === 'checkbox') {
          if (Array.isArray(value)) {
            field.checked = value.indexOf(field.value) !== -1;
          } else if (typeof value === 'boolean') {
            field.checked = value;
          } else {
            field.checked = field.value == value;
          }
        } else if (type === 'radio') {
          field.checked = value != null && field.value == value;
        } else if (type === 'select-multiple') {
          var vals = Array.isArray(value) ? value : [value];
          for (var j = 0; j < field.options.length; j++) {
            var opt = field.options[j];
            opt.selected = vals.indexOf(opt.value) !== -1;
          }
        } else if (type !== 'file') {
          field.value = value != null ? value : '';
        }
      }
    }
    if (typeof data.__step === 'number' && !isNaN(data.__step)) {
      if (steps.length) {
        var max = steps.length - 1;
        currentStep = Math.max(0, Math.min(data.__step, max));
      } else {
        currentStep = data.__step;
      }
    }
  }

  function showStep(index) {
    if (!steps.length) return;
    var max = steps.length - 1;
    var target = Math.max(0, Math.min(index, max));
    for (var i = 0; i < steps.length; i++) {
      var step = steps[i];
      var active = i === target;
      if (settings.useHidden) step.hidden = !active;
      if (settings.activeClass) {
        if (active) step.classList.add(settings.activeClass);
        else step.classList.remove(settings.activeClass);
      }
      step.setAttribute('aria-hidden', active ? 'false' : 'true');
    }
    currentStep = target;
    saveData();
  }

  loadData();
  if (steps.length) {
    showStep(currentStep);
  } else {
    saveData();
  }

  function onInput() {
    saveData();
  }

  form.addEventListener('input', onInput, true);
  form.addEventListener('change', onInput, true);

  form.addEventListener('click', function(e) {
    var nextBtn = closest(e.target, settings.nextSelector);
    var prevBtn = closest(e.target, settings.prevSelector);
    var gotoBtn = closest(e.target, settings.gotoSelector);
    if (nextBtn) {
      e.preventDefault();
      var t = nextBtn.getAttribute('data-step-next');
      if (t !== null && t !== '') showStep(parseInt(t, 10));
      else showStep(currentStep + 1);
      return;
    }
    if (prevBtn) {
      e.preventDefault();
      var t2 = prevBtn.getAttribute('data-step-prev');
      if (t2 !== null && t2 !== '') showStep(parseInt(t2, 10));
      else showStep(currentStep - 1);
      return;
    }
    if (gotoBtn) {
      e.preventDefault();
      var t3 = gotoBtn.getAttribute('data-step-target');
      if (t3 !== null && t3 !== '') showStep(parseInt(t3, 10));
    }
  });

  form.addEventListener('submit', function(e) {
    saveData();
    var data = readStorage();
    if (typeof settings.onSubmit === 'function') {
      var res = settings.onSubmit(e, data, form);
      if (res === false) return;
    }
    if (settings.clearOnSubmit) {
      clearStorage();
    } else if (typeof settings.syncOnSubmit === 'function') {
      settings.syncOnSubmit(data, form);
    }
  });

  return {
    save: saveData,
    load: loadData,
    clear: clearStorage,
    getData: readStorage,
    showStep: showStep,
    next: function() { showStep(currentStep + 1); },
    prev: function() { showStep(currentStep - 1); },
    getCurrentStep: function() { return currentStep; },
    setCurrentStep: function(i) { showStep(i); }
  };
}
