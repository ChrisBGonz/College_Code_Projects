#!/usr/bin/env python3
import logging
from logging.handlers import TimedRotatingFileHandler
import argparse
import time

def setup_logger(logfile, interval_hours=24, backup_count=7):
    logger = logging.getLogger("rotating_logger")
    logger.setLevel(logging.INFO)
    handler = TimedRotatingFileHandler(
        logfile,
        when="H",
        interval=interval_hours,
        backupCount=backup_count,
        encoding="utf-8"
    )
    handler.suffix = "%Y-%m-%d_%H-%M-%S"
    formatter = logging.Formatter("%(asctime)s %(levelname)s %(message)s")
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    logger.propagate = False
    return logger

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--logfile", default="app.log")
    parser.add_argument("--interval", type=int, default=24)
    parser.add_argument("--backups", type=int, default=7)
    parser.add_argument("--period", type=int, default=60)
    args = parser.parse_args()

    logger = setup_logger(args.logfile, args.interval, args.backups)

    try:
        while True:
            logger.info("log message")
            time.sleep(args.period)
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
    main()
