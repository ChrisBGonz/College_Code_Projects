<?php
class UserPrefs {
    public $theme = 'light';
    public $language = 'ru';
    public $itemsPerPage = 10;
}

$prefs = new UserPrefs();

if (!empty($_COOKIE['user_prefs'])) {
    $restored = @unserialize($_COOKIE['user_prefs'], ['allowed_classes' => ['UserPrefs']]);
    if ($restored instanceof UserPrefs) {
        $prefs = $restored;
    }
}

if (isset($_REQUEST['theme'])) {
    $prefs->theme = (string)$_REQUEST['theme'];
}
if (isset($_REQUEST['language'])) {
    $prefs->language = (string)$_REQUEST['language'];
}
if (isset($_REQUEST['itemsPerPage'])) {
    $prefs->itemsPerPage = (int)$_REQUEST['itemsPerPage'];
}

setcookie('user_prefs', serialize($prefs), time() + 60 * 60 * 24 * 30, '/', '', false, true);

header('Content-Type: application/json; charset=utf-8');
echo json_encode($prefs);
