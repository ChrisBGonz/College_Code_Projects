const fs = require('fs');
const path = require('path');
const http = require('http');
const https = require('https');
const unzipper = require('unzipper');

function downloadFile(url, dest) {
  const proto = url.startsWith('https') ? https : http;
  return new Promise((resolve, reject) => {
    const req = proto.get(url, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        downloadFile(res.headers.location, dest).then(resolve).catch(reject);
        return;
      }
      if (res.statusCode !== 200) {
        res.resume();
        reject(new Error(`Download failed: ${res.statusCode}`));
        return;
      }
      const file = fs.createWriteStream(dest);
      res.pipe(file);
      file.on('finish', () => file.close(resolve));
      file.on('error', err => {
        file.close(() => {});
        reject(err);
      });
    });
    req.on('error', reject);
  });
}

async function downloadAndExtractZip(url, destDir = '/app') {
  await fs.promises.mkdir(destDir, { recursive: true });
  const tmpZip = path.join(destDir, `.update_${Date.now()}.zip`);
  await downloadFile(url, tmpZip);
  await new Promise((resolve, reject) => {
    const extractStream = unzipper.Extract({ path: destDir });
    extractStream.on('close', resolve);
    extractStream.on('error', reject);
    fs.createReadStream(tmpZip).pipe(extractStream);
  });
  await fs.promises.unlink(tmpZip).catch(() => {});
}

module.exports = { downloadAndExtractZip };
