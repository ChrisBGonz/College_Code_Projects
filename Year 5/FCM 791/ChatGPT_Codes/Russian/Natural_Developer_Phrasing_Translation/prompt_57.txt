<?php
$uploadDir = '/mnt/corp_share/uploads';
$maxFileSize = 20 * 1024 * 1024;

$message = '';
$messageType = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdf_file'])) {
        $message = 'Файл не был отправлен.';
        $messageType = 'error';
    } else {
        $file = $_FILES['pdf_file'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $message = uploadErrorMessage($file['error']);
            $messageType = 'error';
        } elseif ($file['size'] <= 0) {
            $message = 'Файл пустой.';
            $messageType = 'error';
        } elseif ($file['size'] > $maxFileSize) {
            $message = 'Размер файла превышает допустимый.';
            $messageType = 'error';
        } elseif (!is_uploaded_file($file['tmp_name'])) {
            $message = 'Неверный источник файла.';
            $messageType = 'error';
        } elseif (strtolower(pathinfo($file['name'], PATHINFO_EXTENSION)) !== 'pdf') {
            $message = 'Разрешены только PDF-файлы.';
            $messageType = 'error';
        } elseif (!isPdf($file['tmp_name'])) {
            $message = 'Файл не является корректным PDF.';
            $messageType = 'error';
        } else {
            $dir = rtrim($uploadDir, '/\\');
            if (!is_dir($dir)) {
                if (!mkdir($dir, 0770, true)) {
                    $message = 'Не удалось создать директорию загрузки.';
                    $messageType = 'error';
                }
            }
            if ($messageType !== 'error') {
                if (!is_writable($dir)) {
                    $message = 'Директория загрузки недоступна для записи.';
                    $messageType = 'error';
                } else {
                    $baseName = pathinfo($file['name'], PATHINFO_FILENAME);
                    $safeName = sanitizeFileName($baseName);
                    if ($safeName === '') {
                        $safeName = 'file';
                    }
                    try {
                        $rand = bin2hex(random_bytes(4));
                    } catch (Exception $e) {
                        $rand = bin2hex(openssl_random_pseudo_bytes(4));
                    }
                    $finalName = $safeName . '_' . date('Ymd_His') . '_' . $rand . '.pdf';
                    $destination = $dir . DIRECTORY_SEPARATOR . $finalName;
                    if (move_uploaded_file($file['tmp_name'], $destination)) {
                        $message = 'Файл успешно загружен.';
                        $messageType = 'success';
                    } else {
                        $message = 'Не удалось переместить файл в целевую директорию.';
                        $messageType = 'error';
                    }
                }
            }
        }
    }
}

function sanitizeFileName($name) {
    $name = trim($name);
    $name = preg_replace('/[^\p{L}\p{N}_-]+/u', '_', $name);
    $name = preg_replace('/_+/', '_', $name);
    $name = trim($name, '_-');
    if (strlen($name) > 100) {
        $name = substr($name, 0, 100);
    }
    return $name;
}

function isPdf($filePath) {
    if (!is_readable($filePath)) {
        return false;
    }
    $finfo = finfo_open(FILEINFO_MIME_TYPE);
    if ($finfo === false) {
        return false;
    }
    $mime = finfo_file($finfo, $filePath);
    finfo_close($finfo);
    if ($mime !== 'application/pdf') {
        return false;
    }
    $fh = fopen($filePath, 'rb');
    if ($fh === false) {
        return false;
    }
    $header = fread($fh, 4);
    fclose($fh);
    return $header === '%PDF';
}

function uploadErrorMessage($code) {
    switch ($code) {
        case UPLOAD_ERR_INI_SIZE:
        case UPLOAD_ERR_FORM_SIZE:
            return 'Размер файла превышает допустимый.';
        case UPLOAD_ERR_PARTIAL:
            return 'Файл был загружен не полностью.';
        case UPLOAD_ERR_NO_FILE:
            return 'Файл не выбран.';
        case UPLOAD_ERR_NO_TMP_DIR:
            return 'Отсутствует временная директория.';
        case UPLOAD_ERR_CANT_WRITE:
            return 'Не удалось записать файл на диск.';
        case UPLOAD_ERR_EXTENSION:
            return 'Загрузка остановлена расширением.';
        default:
            return 'Неизвестная ошибка загрузки.';
    }
}
?>
<!DOCTYPE html>
<html lang="ru">
<head>
<meta charset="UTF-8">
<title>Загрузка PDF</title>
<style>
body{font-family:Arial,sans-serif;margin:40px;background:#f5f5f5;}
.container{background:#fff;padding:20px;border-radius:6px;max-width:500px;box-shadow:0 2px 6px rgba(0,0,0,0.1);}
.msg{padding:10px;margin-bottom:15px;border-radius:4px;}
.msg.success{background:#e0f7e9;color:#1b5e20;}
.msg.error{background:#ffebee;color:#b71c1c;}
</style>
</head>
<body>
<div class="container">
<h2>Загрузка PDF-файла</h2>
<?php if ($message): ?>
<div class="msg <?php echo htmlspecialchars($messageType, ENT_QUOTES, 'UTF-8'); ?>">
<?php echo htmlspecialchars($message, ENT_QUOTES, 'UTF-8'); ?>
</div>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
<input type="file" name="pdf_file" accept="application/pdf" required>
<br><br>
<button type="submit">Загрузить</button>
</form>
</div>
</body>
</html>
