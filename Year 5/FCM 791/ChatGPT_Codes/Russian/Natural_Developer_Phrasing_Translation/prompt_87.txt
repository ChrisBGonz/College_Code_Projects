import sys
import os
import time
from collections import deque

WINDOW_SECONDS = 60
THRESHOLD = 10
PATTERN = "Failed Login"
SLEEP_INTERVAL = 0.1

def follow(f):
    f.seek(0, os.SEEK_END)
    while True:
        line = f.readline()
        if line:
            yield line
        else:
            time.sleep(SLEEP_INTERVAL)
            yield None

def monitor(path):
    events = deque()
    warning_active = False
    with open(path, "r", encoding="utf-8", errors="replace") as f:
        for line in follow(f):
            now = time.time()
            while events and now - events[0] > WINDOW_SECONDS:
                events.popleft()
            if line and PATTERN in line:
                events.append(now)
            if len(events) > THRESHOLD and not warning_active:
                ts = time.strftime("%Y-%m-%d %H:%M:%S", time.localtime(now))
                print(f'WARNING {ts}: {len(events)} "{PATTERN}" messages in last {WINDOW_SECONDS} seconds', flush=True)
                warning_active = True
            elif warning_active and len(events) <= THRESHOLD:
                warning_active = False

if __name__ == "__main__":
    if len(sys.argv) < 2:
        sys.stderr.write(f"Usage: {sys.argv[0]} <logfile>\n")
        sys.exit(1)
    monitor(sys.argv[1])
