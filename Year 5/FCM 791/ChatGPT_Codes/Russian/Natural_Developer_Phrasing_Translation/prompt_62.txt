function logout(req, res) {
  const cookie = (req.session && req.session.cookie) || {};
  const cookieName = cookie.name || req.session?.cookieName || 'connect.sid';
  const options = {
    path: cookie.path || '/',
    domain: cookie.domain,
    secure: cookie.secure,
    httpOnly: cookie.httpOnly,
    sameSite: cookie.sameSite
  };

  const send = (status, body) => {
    res.status(status);
    if (body !== undefined) return res.json(body);
    return res.end();
  };

  if (!req.session) {
    res.clearCookie(cookieName, options);
    return send(200, { success: true });
  }

  req.session.destroy(err => {
    res.clearCookie(cookieName, options);
    if (err) return send(500, { success: false, error: 'Logout failed' });
    return send(200, { success: true });
  });
}

module.exports = logout;
