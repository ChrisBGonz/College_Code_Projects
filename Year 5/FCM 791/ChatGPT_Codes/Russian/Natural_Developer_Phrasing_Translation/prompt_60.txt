'use strict';

/*
  Voting System API:
  POST   /polls               {question, options[]} -> create poll
  GET    /polls               ?limit&offset -> list polls
  GET    /polls/:id            -> poll with options
  GET    /polls/:id/results    -> poll results with vote counts
  POST   /polls/:id/vote       {optionId} -> vote (1 IP = 1 vote per poll)

  DB Schema (SQLite):
    polls(id PK, question, created_at)
    options(id PK, poll_id FK, text, created_at)
    votes(id PK, poll_id FK, option_id FK, ip_hash, ip, user_agent, created_at, UNIQUE(poll_id, ip_hash))

  Anti-duplicate:
    - Client IP normalized (IPv4/IPv6) -> hash with salt -> stored in votes
    - Unique constraint prevents double voting per poll per IP
  Edge-cases:
    - NAT/Proxy: multiple users can share one IP (limitation); trust X-Forwarded-For only when TRUST_PROXY=1.
    - IPv4/IPv6: IPv4-mapped IPv6 converted to IPv4, zone/port stripped before hashing.
  Basic protection:
    - Rate limit per IP
    - Body size limit, input validation, parameterized queries
    - Security headers, disabled x-powered-by
*/

const express = require('express');
const Database = require('better-sqlite3');
const crypto = require('crypto');
const ipaddr = require('ipaddr.js');

const app = express();

const PORT = parseInt(process.env.PORT || '3000', 10);
const DB_FILE = process.env.DB_FILE || './voting.db';
const TRUST_PROXY = process.env.TRUST_PROXY === '1';
const IP_HASH_SALT = process.env.IP_HASH_SALT || 'change-this-salt';
const MAX_QUESTION = parseInt(process.env.MAX_QUESTION || '300', 10);
const MAX_OPTIONS = parseInt(process.env.MAX_OPTIONS || '10', 10);
const MIN_OPTIONS = parseInt(process.env.MIN_OPTIONS || '2', 10);
const MAX_OPTION_LEN = parseInt(process.env.MAX_OPTION_LEN || '100', 10);
const RATE_WINDOW_MS = parseInt(process.env.RATE_WINDOW_MS || '60000', 10);
const RATE_MAX = parseInt(process.env.RATE_MAX || '100', 10);

app.disable('x-powered-by');
app.set('trust proxy', TRUST_PROXY ? 1 : false);

app.use(express.json({ limit: '10kb', strict: true }));

app.use((req, res, next) => {
  res.setHeader('X-Content-Type-Options', 'nosniff');
  res.setHeader('X-Frame-Options', 'DENY');
  res.setHeader('Referrer-Policy', 'no-referrer');
  next();
});

app.use((req, res, next) => {
  if (['POST', 'PUT', 'PATCH'].includes(req.method)) {
    if (!req.is('application/json')) {
      return res.status(415).json({ error: 'content_type' });
    }
  }
  next();
});

// Rate limiter (simple in-memory, per raw IP)
const rateMap = new Map();
function getRawIp(req) {
  let ip;
  if (TRUST_PROXY) {
    const xff = req.headers['x-forwarded-for'];
    if (xff) ip = xff.split(',')[0].trim();
    if (!ip && req.headers['x-real-ip']) ip = req.headers['x-real-ip'];
    if (!ip && req.headers['cf-connecting-ip']) ip = req.headers['cf-connecting-ip'];
  }
  if (!ip && req.socket && req.socket.remoteAddress) ip = req.socket.remoteAddress;
  if (!ip && req.ip) ip = req.ip;
  return ip || null;
}

function rateLimiter(req, res, next) {
  const key = getRawIp(req) || 'unknown';
  const now = Date.now();
  const entry = rateMap.get(key);
  if (!entry) {
    rateMap.set(key, { count: 1, start: now });
    return next();
  }
  if (now - entry.start > RATE_WINDOW_MS) {
    entry.start = now;
    entry.count = 1;
    return next();
  }
  entry.count++;
  if (entry.count > RATE_MAX) {
    return res.status(429).json({ error: 'rate_limit' });
  }
  next();
}
app.use(rateLimiter);

setInterval(() => {
  const now = Date.now();
  for (const [k, v] of rateMap) {
    if (now - v.start > RATE_WINDOW_MS * 2) rateMap.delete(k);
  }
}).unref();

function stripPort(ip) {
  if (!ip) return ip;
  ip = ip.trim();
  if (ip.startsWith('[')) {
    const end = ip.indexOf(']');
    if (end !== -1) return ip.slice(1, end);
  }
  const colonCount = (ip.match(/:/g) || []).length;
  if (colonCount === 1 && ip.includes('.')) {
    const last = ip.lastIndexOf(':');
    const port = ip.slice(last + 1);
    if (/^\d+$/.test(port)) return ip.slice(0, last);
  }
  return ip;
}

function normalizeIp(ip) {
  if (!ip) return null;
  ip = stripPort(ip);
  if (!ip) return null;
  if (ip.includes('%')) ip = ip.split('%')[0];
  try {
    const addr = ipaddr.parse(ip);
    if (addr.kind() === 'ipv6' && addr.isIPv4MappedAddress()) {
      return addr.toIPv4Address().toString();
    }
    return addr.toNormalizedString();
  } catch (e) {
    return null;
  }
}

function getClientIp(req) {
  const raw = getRawIp(req);
  const normalized = normalizeIp(raw);
  return { raw, normalized };
}

function ipHash(ip) {
  return crypto.createHash('sha256').update(IP_HASH_SALT + ip).digest('hex');
}

function sanitizeText(s, max) {
  if (typeof s !== 'string') return null;
  const t = s.trim();
  if (!t || t.length > max) return null;
  return t;
}

function validateOptions(opts) {
  if (!Array.isArray(opts)) return null;
  if (opts.length < MIN_OPTIONS || opts.length > MAX_OPTIONS) return null;
  const cleaned = [];
  const set = new Set();
  for (const o of opts) {
    const v = sanitizeText(o, MAX_OPTION_LEN);
    if (!v) return null;
    const key = v.toLowerCase();
    if (set.has(key)) return null;
    set.add(key);
    cleaned.push(v);
  }
  return cleaned;
}

function toId(val) {
  const n = Number(val);
  if (!Number.isInteger(n) || n <= 0) return null;
  return n;
}

// Database setup
const db = new Database(DB_FILE);
db.pragma('journal_mode = WAL');
db.pragma('foreign_keys = ON');

db.exec(`
CREATE TABLE IF NOT EXISTS polls (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  question TEXT NOT NULL,
  created_at TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS options (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  poll_id INTEGER NOT NULL,
  text TEXT NOT NULL,
  created_at TEXT NOT NULL,
  FOREIGN KEY (poll_id) REFERENCES polls(id) ON DELETE CASCADE
);
CREATE TABLE IF NOT EXISTS votes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  poll_id INTEGER NOT NULL,
  option_id INTEGER NOT NULL,
  ip_hash TEXT NOT NULL,
  ip TEXT,
  user_agent TEXT,
  created_at TEXT NOT NULL,
  UNIQUE (poll_id, ip_hash),
  FOREIGN KEY (poll_id) REFERENCES polls(id) ON DELETE CASCADE,
  FOREIGN KEY (option_id) REFERENCES options(id) ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS idx_options_poll ON options(poll_id);
CREATE INDEX IF NOT EXISTS idx_votes_poll ON votes(poll_id);
CREATE INDEX IF NOT EXISTS idx_votes_option ON votes(option_id);
CREATE INDEX IF NOT EXISTS idx_votes_ip_hash ON votes(ip_hash);
`);

const stmtInsertPoll = db.prepare('INSERT INTO polls (question, created_at) VALUES (?, ?)');
const stmtInsertOption = db.prepare('INSERT INTO options (poll_id, text, created_at) VALUES (?, ?, ?)');
const stmtGetPoll = db.prepare('SELECT id, question, created_at FROM polls WHERE id = ?');
const stmtGetOptions = db.prepare('SELECT id, text FROM options WHERE poll_id = ? ORDER BY id');
const stmtGetOption = db.prepare('SELECT id FROM options WHERE id = ? AND poll_id = ?');
const stmtInsertVote = db.prepare('INSERT INTO votes (poll_id, option_id, ip_hash, ip, user_agent, created_at) VALUES (?, ?, ?, ?, ?, ?)');
const stmtListPolls = db.prepare(`
SELECT p.id, p.question, p.created_at,
  (SELECT COUNT(*) FROM options o WHERE o.poll_id = p.id) as options,
  (SELECT COUNT(*) FROM votes v WHERE v.poll_id = p.id) as votes
FROM polls p
ORDER BY p.id DESC
LIMIT ? OFFSET ?
`);
const stmtResults = db.prepare(`
SELECT o.id, o.text, COUNT(v.id) as votes
FROM options o
LEFT JOIN votes v ON v.option_id = o.id
WHERE o.poll_id = ?
GROUP BY o.id
ORDER BY o.id
`);
const stmtCountVotes = db.prepare('SELECT COUNT(*) as c FROM votes WHERE poll_id = ?');

const createPollTx = db.transaction((question, options) => {
  const now = new Date().toISOString();
  const pollId = Number(stmtInsertPoll.run(question, now).lastInsertRowid);
  for (const opt of options) {
    stmtInsertOption.run(pollId, opt, now);
  }
  return pollId;
});

const voteTx = db.transaction((pollId, optionId, ipHashValue, ipNormalized, ua, now) => {
  const poll = stmtGetPoll.get(pollId);
  if (!poll) return { error: 'poll_not_found' };
  const option = stmtGetOption.get(optionId, pollId);
  if (!option) return { error: 'option_not_found' };
  try {
    stmtInsertVote.run(pollId, optionId, ipHashValue, ipNormalized, ua, now);
    return { ok: true };
  } catch (e) {
    if (e && e.code === 'SQLITE_CONSTRAINT_UNIQUE') return { error: 'duplicate' };
    throw e;
  }
});

// Routes
app.get('/health', (req, res) => res.json({ status: 'ok' }));

app.post('/polls', (req, res) => {
  const question = sanitizeText(req.body && req.body.question, MAX_QUESTION);
  const options = validateOptions(req.body && req.body.options);
  if (!question || !options) {
    return res.status(400).json({ error: 'invalid_input' });
  }
  const id = createPollTx(question, options);
  res.status(201).json({ id });
});

app.get('/polls', (req, res) => {
  const limit = Math.min(Math.max(parseInt(req.query.limit || '20', 10), 1), 100);
  const offset = Math.max(parseInt(req.query.offset || '0', 10), 0);
  const rows = stmtListPolls.all(limit, offset);
  res.json({ items: rows, limit, offset });
});

app.get('/polls/:id', (req, res) => {
  const id = toId(req.params.id);
  if (!id) return res.status(400).json({ error: 'invalid_id' });
  const poll = stmtGetPoll.get(id);
  if (!poll) return res.status(404).json({ error: 'not_found' });
  const options = stmtGetOptions.all(id);
  res.json({ id: poll.id, question: poll.question, created_at: poll.created_at, options });
});

app.get('/polls/:id/results', (req, res) => {
  const id = toId(req.params.id);
  if (!id) return res.status(400).json({ error: 'invalid_id' });
  const poll = stmtGetPoll.get(id);
  if (!poll) return res.status(404).json({ error: 'not_found' });
  const options = stmtResults.all(id);
  const totalVotes = stmtCountVotes.get(id).c;
  res.json({ poll: { id: poll.id, question: poll.question, created_at: poll.created_at }, totalVotes, options });
});

app.post('/polls/:id/vote', (req, res) => {
  const pollId = toId(req.params.id);
  const optionId = toId(req.body && req.body.optionId);
  if (!pollId || !optionId) {
    return res.status(400).json({ error: 'invalid_input' });
  }
  const ipInfo = getClientIp(req);
  if (!ipInfo.normalized) {
    return res.status(400).json({ error: 'invalid_ip' });
  }
  const hash = ipHash(ipInfo.normalized);
  const ua = typeof req.headers['user-agent'] === 'string' ? req.headers['user-agent'].slice(0, 255) : '';
  const now = new Date().toISOString();
  const result = voteTx(pollId, optionId, hash, ipInfo.normalized, ua, now);
  if (result.error === 'poll_not_found') return res.status(404).json({ error: 'poll_not_found' });
  if (result.error === 'option_not_found') return res.status(400).json({ error: 'option_not_found' });
  if (result.error === 'duplicate') return res.status(409).json({ error: 'already_voted' });
  res.json({ status: 'ok' });
});

// 404 handler
app.use((req, res) => {
  res.status(404).json({ error: 'not_found' });
});

// error handler
app.use((err, req, res, next) => {
  if (err && err.type === 'entity.too.large') {
    return res.status(413).json({ error: 'payload_too_large' });
  }
  console.error(err);
  res.status(500).json({ error: 'internal' });
});

app.listen(PORT, () => {
  console.log(`Voting API listening on port ${PORT}`);
});
