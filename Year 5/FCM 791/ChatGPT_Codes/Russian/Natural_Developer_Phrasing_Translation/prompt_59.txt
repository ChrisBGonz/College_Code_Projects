import os
import uuid
import requests
from flask import Flask, redirect, request, Response

app = Flask(__name__)

GATEWAY_URL = os.environ.get("GATEWAY_URL", "https://gateway.example.com/api/payments")
GATEWAY_TOKEN = os.environ.get("GATEWAY_TOKEN", "")

def create_payment(amount, currency):
    payload = {
        "order_id": str(uuid.uuid4()),
        "amount": amount,
        "currency": currency,
        "success_url": request.url_root.rstrip("/") + "/success",
        "fail_url": request.url_root.rstrip("/") + "/fail"
    }
    headers = {}
    if GATEWAY_TOKEN:
        headers["Authorization"] = f"Bearer {GATEWAY_TOKEN}"
    r = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=10)
    r.raise_for_status()
    return r.json()

@app.route("/pay", methods=["GET", "POST"])
def pay():
    amount = request.values.get("amount", "100.00")
    currency = request.values.get("currency", "RUB")
    try:
        data = create_payment(amount, currency)
    except Exception as e:
        return Response("Gateway error: " + str(e), status=502)
    success_url = data.get("success_url")
    if not success_url:
        return Response("Missing success_url in response", status=502)
    return redirect(success_url, code=302)

@app.route("/success")
def success():
    return "ok"

@app.route("/fail")
def fail():
    return "fail"

if __name__ == "__main__":
    port = int(os.getenv("PORT", "8000"))
    app.run(host="0.0.0.0", port=port)
