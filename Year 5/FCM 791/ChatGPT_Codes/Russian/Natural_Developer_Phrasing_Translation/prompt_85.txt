<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = ''; // set Slack webhook URL if needed

function user_is_admin(): bool {
    if (!empty($_SESSION['is_admin'])) return true;
    if (isset($_SESSION['user']) && is_array($_SESSION['user'])) {
        if (!empty($_SESSION['user']['is_admin'])) return true;
        if (isset($_SESSION['user']['role']) && $_SESSION['user']['role'] === 'admin') return true;
    }
    return false;
}

function send_admin_alert(string $subject, string $message, string $adminEmail, string $slackWebhook): void {
    if ($adminEmail !== '') {
        $headers = "From: noreply@" . ($_SERVER['SERVER_NAME'] ?? 'localhost') . "\r\n";
        @mail($adminEmail, $subject, $message, $headers);
    }
    if ($slackWebhook !== '') {
        $payload = json_encode(['text' => $subject . "\n" . $message], JSON_UNESCAPED_UNICODE);
        $ch = curl_init($slackWebhook);
        curl_setopt_array($ch, [
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => $payload,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => 4,
        ]);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH);

if (strpos($path, '/admin') === 0 && !user_is_admin()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $agent = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $user = $_SESSION['user']['username'] ?? ($_SESSION['username'] ?? 'guest');
    $time = date('c');
    $message = "Unauthorized access attempt.\nTime: $time\nIP: $ip\nUser: $user\nURI: $path\nAgent: $agent";
    send_admin_alert('Administrator Alert', $message, $adminEmail, $slackWebhook);
    http_response_code(403);
    echo "Access denied.";
    exit;
}

?>
