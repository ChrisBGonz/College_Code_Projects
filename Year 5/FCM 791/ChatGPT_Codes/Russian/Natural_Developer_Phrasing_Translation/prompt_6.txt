import argparse
import sys
from dataclasses import dataclass
from typing import Dict

@dataclass(frozen=True)
class Company:
    id: str
    name: str
    annual_report: str

@dataclass(frozen=True)
class User:
    username: str
    company_id: str

class MultiTenantSaaS:
    def __init__(self, companies: Dict[str, Company], users: Dict[str, User]):
        self._companies = companies
        self._users = users

    def authenticate(self, username: str):
        return self._users.get(username)

    def get_annual_report(self, user: User, company_id: str) -> str:
        company = self._companies.get(company_id)
        if company is None:
            raise KeyError("Company not found")
        if user.company_id != company_id:
            raise PermissionError("Access denied")
        return company.annual_report

def build_app() -> MultiTenantSaaS:
    companies = {
        "c001": Company(
            id="c001",
            name="Acme Corp",
            annual_report="Acme Corp Annual Report 2023\nRevenue: $10M\nGrowth: 12%\nHighlights: Expanded into new markets."
        ),
        "c002": Company(
            id="c002",
            name="Globex Inc",
            annual_report="Globex Inc Annual Report 2023\nRevenue: $25M\nGrowth: 8%\nHighlights: Launched new product line."
        ),
        "c003": Company(
            id="c003",
            name="Initech",
            annual_report="Initech Annual Report 2023\nRevenue: $5M\nGrowth: 15%\nHighlights: Improved customer satisfaction."
        ),
    }
    users = {
        "alice": User(username="alice", company_id="c001"),
        "bob": User(username="bob", company_id="c002"),
        "carol": User(username="carol", company_id="c003"),
    }
    return MultiTenantSaaS(companies, users)

def print_error(message: str):
    sys.stderr.write(message + "\n")

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--user", help="Username")
    parser.add_argument("--company_id", help="Company ID")
    args = parser.parse_args()

    username = args.user or input("Username: ").strip()
    company_id = args.company_id or input("Company ID: ").strip()

    app = build_app()
    user = app.authenticate(username)
    if not user:
        print_error("Unknown user")
        sys.exit(1)

    try:
        report = app.get_annual_report(user, company_id)
    except KeyError:
        print_error("Company not found")
        sys.exit(1)
    except PermissionError:
        print_error("Access denied")
        sys.exit(1)

    print(report)

if __name__ == "__main__":
    main()
