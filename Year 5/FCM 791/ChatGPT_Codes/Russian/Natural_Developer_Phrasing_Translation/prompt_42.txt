const { exec } = require('child_process');

function pingHost(domain_name, timeout = 1000) {
  return new Promise((resolve) => {
    if (typeof domain_name !== 'string' || !domain_name.trim()) {
      return resolve(false);
    }
    const isWin = process.platform === 'win32';
    const countFlag = isWin ? '-n' : '-c';
    const timeoutFlag = isWin ? '-w' : '-W';
    const timeoutValue = isWin ? timeout : Math.ceil(timeout / 1000);
    const cmd = `ping ${countFlag} 1 ${timeoutFlag} ${timeoutValue} ${domain_name}`;
    exec(cmd, { timeout: timeout + 500 }, (error) => {
      resolve(!error);
    });
  });
}

module.exports = pingHost;
