#!/usr/bin/env python3
import argparse
import hashlib
import os
import sys

def sha256sum(path, bufsize=1024 * 1024):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(bufsize), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify SHA-256 hash of a firmware file")
    parser.add_argument("file", help="Path to firmware file")
    parser.add_argument("expected", help="Expected SHA-256 hash")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print(f"File not found: {args.file}", file=sys.stderr)
        return 2

    actual = sha256sum(args.file)
    expected = args.expected.strip().lower()

    if actual.lower() == expected:
        print("OK")
        return 0
    else:
        print("FAILED")
        print(f"Expected: {expected}")
        print(f"Actual  : {actual}")
        return 1

if __name__ == "__main__":
    sys.exit(main())
