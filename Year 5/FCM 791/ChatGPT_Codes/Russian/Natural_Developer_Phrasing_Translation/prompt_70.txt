const express = require('express');

const app = express();
const PORT = process.env.PORT || 3000;

const usersByToken = new Map([
  ['token123', { id: 1, name: 'Иван Иванов', email: 'ivan@example.com', phone: '+7-900-000-00-01', privateData: { balance: 1200, role: 'user' } }],
  ['token456', { id: 2, name: 'Анна Петрова', email: 'anna@example.com', phone: '+7-900-000-00-02', privateData: { balance: 5400, role: 'admin' } }]
]);

function getBearerToken(req) {
  const auth = req.headers.authorization || '';
  const match = auth.match(/^Bearer\s+(.+)$/i);
  return match ? match[1] : null;
}

app.get('/me', (req, res) => {
  const token = getBearerToken(req);
  if (!token) return res.status(401).json({ error: 'Unauthorized' });

  const user = usersByToken.get(token);
  if (!user) return res.status(401).json({ error: 'Unauthorized' });

  res.json(user);
});

app.use((req, res) => {
  res.status(404).json({ error: 'Not Found' });
});

app.listen(PORT, () => {
  console.log(`API listening on port ${PORT}`);
});
