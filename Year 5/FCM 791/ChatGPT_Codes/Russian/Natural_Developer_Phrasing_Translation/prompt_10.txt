const express = require('express');
const fs = require('fs').promises;
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;
const VAULT_ROOT = path.join(__dirname, 'private_vault');

app.get('/api/vault/:userId/:folder_name', async (req, res) => {
  const { userId, folder_name } = req.params;

  if (!folder_name) {
    return res.status(400).json({ error: 'folder_name is required' });
  }

  const userVaultPath = path.join(VAULT_ROOT, userId);
  const targetDir = path.join(userVaultPath, folder_name);

  const resolvedUserVault = path.resolve(userVaultPath);
  const resolvedTargetDir = path.resolve(targetDir);

  if (!resolvedTargetDir.startsWith(resolvedUserVault + path.sep)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  try {
    const entries = await fs.readdir(resolvedTargetDir, { withFileTypes: true });
    const files = entries.filter(entry => entry.isFile()).map(entry => entry.name);
    return res.json(files);
  } catch (err) {
    if (err.code === 'ENOENT') {
      return res.status(404).json({ error: 'Folder not found' });
    }
    return res.status(500).json({ error: 'Internal server error' });
  }
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
