using System;
using System.Diagnostics;
using Microsoft.Extensions.Logging;

public record MfaSettings(bool Enabled, string Method, string? PhoneNumber);

public static class MfaAuditLogger
{
    public static void LogChange(
        ILogger logger,
        string userId,
        string userName,
        MfaSettings before,
        MfaSettings after,
        bool success,
        string ip,
        string userAgent,
        string? correlationId = null)
    {
        var timestamp = DateTimeOffset.UtcNow;
        var activity = Activity.Current;
        var corrId = correlationId ?? activity?.Id ?? Guid.NewGuid().ToString("N");
        var traceId = activity?.TraceId.ToString() ?? string.Empty;

        var evt = new
        {
            Event = "MFA_SETTINGS_CHANGED",
            Timestamp = timestamp,
            User = new { Id = userId, Name = userName },
            Before = before,
            After = after,
            Result = success ? "success" : "fail",
            Source = new { Ip = ip, UserAgent = userAgent },
            CorrelationId = corrId,
            TraceId = traceId
        };

        if (success)
            logger.LogInformation("{@MfaChange}", evt);
        else
            logger.LogWarning("{@MfaChange}", evt);
    }
}
