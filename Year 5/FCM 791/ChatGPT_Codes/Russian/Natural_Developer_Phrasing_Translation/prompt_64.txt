using System;
using System.Security.Claims;
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.DataProtection;
using Microsoft.AspNetCore.Http;

public static class StayLoggedIn
{
    public static void IssueRememberMeCookie(HttpContext context, ClaimsPrincipal principal, bool rememberMe, IDataProtectionProvider protectionProvider, TimeSpan? persistentLifetime = null, string scheme = CookieAuthenticationDefaults.AuthenticationScheme, string cookieName = null)
    {
        if (context == null) throw new ArgumentNullException(nameof(context));
        if (principal == null) throw new ArgumentNullException(nameof(principal));
        if (protectionProvider == null) throw new ArgumentNullException(nameof(protectionProvider));

        var now = DateTimeOffset.UtcNow;
        var lifetime = persistentLifetime ?? TimeSpan.FromDays(30);
        var expires = rememberMe ? now.Add(lifetime) : (DateTimeOffset?)null;

        var properties = new AuthenticationProperties
        {
            IsPersistent = rememberMe,
            IssuedUtc = now,
            ExpiresUtc = expires,
            AllowRefresh = true
        };

        var ticket = new AuthenticationTicket(principal, properties, scheme);

        var protector = protectionProvider.CreateProtector("Microsoft.AspNetCore.Authentication.Cookies.CookieAuthenticationMiddleware", scheme, "v2");
        var dataFormat = new TicketDataFormat(protector);
        var cookieValue = dataFormat.Protect(ticket);

        var options = new CookieOptions
        {
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            Path = "/",
            IsEssential = true
        };

        if (rememberMe && expires.HasValue)
        {
            options.Expires = expires.Value;
            options.MaxAge = lifetime;
        }

        var name = cookieName ?? (CookieAuthenticationDefaults.CookiePrefix + scheme);
        context.Response.Cookies.Append(name, cookieValue, options);
    }
}
