<?php
declare(strict_types=1);

$dsn = 'mysql:host=localhost;dbname=app;charset=utf8mb4';
$dbUser = 'user';
$dbPass = 'pass';

$input = file_get_contents('php://input');
$data = json_decode($input, true);
if (is_array($data) && isset($data['credit_card_number'])) {
    $creditCard = (string)$data['credit_card_number'];
} elseif (isset($_POST['credit_card_number'])) {
    $creditCard = (string)$_POST['credit_card_number'];
} else {
    http_response_code(400);
    exit;
}

$keyRaw = getenv('ENCRYPTION_KEY');
if ($keyRaw === false) {
    http_response_code(500);
    exit;
}
$key = base64_decode($keyRaw, true);
if ($key === false) {
    $key = $keyRaw;
}
if (strlen($key) !== SODIUM_CRYPTO_SECRETBOX_KEYBYTES) {
    http_response_code(500);
    exit;
}

$nonce = random_bytes(SODIUM_CRYPTO_SECRETBOX_NONCEBYTES);
$cipher = sodium_crypto_secretbox($creditCard, $nonce, $key);
$encrypted = base64_encode($nonce . $cipher);

try {
    $pdo = new PDO($dsn, $dbUser, $dbPass, [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    ]);
    $stmt = $pdo->prepare('INSERT INTO payments (credit_card_number) VALUES (:cc)');
    $stmt->execute([':cc' => $encrypted]);
    http_response_code(201);
} catch (Throwable $e) {
    http_response_code(500);
}

sodium_memzero($creditCard);
?>
