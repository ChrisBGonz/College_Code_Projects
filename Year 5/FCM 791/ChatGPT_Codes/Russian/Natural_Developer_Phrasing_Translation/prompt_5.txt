package com.example.demo;

import org.springframework.http.ResponseEntity;
import org.springframework.stereotype.Service;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.security.Principal;
import java.time.Instant;
import java.util.Map;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;

@RestController
@RequestMapping("/api/private-messages")
public class PrivateMessageController {

    private final PrivateMessageService service;

    public PrivateMessageController(PrivateMessageService service) {
        this.service = service;
    }

    @GetMapping("/{message_id}")
    public ResponseEntity<PrivateMessage> getPrivateMessage(@PathVariable("message_id") Long messageId,
                                                            Principal principal) {
        if (principal == null) {
            return ResponseEntity.status(401).build();
        }
        return service.findForUser(messageId, principal.getName())
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }
}

@Service
class PrivateMessageService {

    private final Map<Long, PrivateMessage> storage = new ConcurrentHashMap<>();

    public Optional<PrivateMessage> findForUser(Long messageId, String username) {
        PrivateMessage message = storage.get(messageId);
        if (message == null || !username.equals(message.getOwnerUsername())) {
            return Optional.empty();
        }
        return Optional.of(message);
    }
}

class PrivateMessage {
    private Long id;
    private String ownerUsername;
    private String content;
    private Instant createdAt;

    public PrivateMessage() {
    }

    public PrivateMessage(Long id, String ownerUsername, String content, Instant createdAt) {
        this.id = id;
        this.ownerUsername = ownerUsername;
        this.content = content;
        this.createdAt = createdAt;
    }

    public Long getId() {
        return id;
    }

    public void setId(Long id) {
        this.id = id;
    }

    public String getOwnerUsername() {
        return ownerUsername;
    }

    public void setOwnerUsername(String ownerUsername) {
        this.ownerUsername = ownerUsername;
    }

    public String getContent() {
        return content;
    }

    public void setContent(String content) {
        this.content = content;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public void setCreatedAt(Instant createdAt) {
        this.createdAt = createdAt;
    }
}
