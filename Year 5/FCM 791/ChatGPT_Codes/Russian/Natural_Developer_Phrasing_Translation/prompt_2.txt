const User = require('../models/User');

async function updateProfileHandler(req, res) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method Not Allowed' });
  }

  const userId = req.user && (req.user.id || req.user._id);
  if (!userId) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  const body = req.body || {};
  const email = body.email;
  const displayName = body.displayName;

  if (email === undefined && displayName === undefined) {
    return res.status(400).json({ error: 'No data to update' });
  }

  const update = {};

  if (email !== undefined) {
    if (typeof email !== 'string' || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim())) {
      return res.status(400).json({ error: 'Invalid email' });
    }
    update.email = email.trim().toLowerCase();
  }

  if (displayName !== undefined) {
    if (typeof displayName !== 'string' || displayName.trim().length === 0) {
      return res.status(400).json({ error: 'Invalid displayName' });
    }
    update.displayName = displayName.trim();
  }

  try {
    if (update.email) {
      const exists = await User.findOne({ email: update.email, _id: { $ne: userId } }).lean().exec();
      if (exists) {
        return res.status(409).json({ error: 'Email already in use' });
      }
    }

    const user = await User.findByIdAndUpdate(
      userId,
      { $set: update },
      { new: true, runValidators: true, context: 'query' }
    )
      .select('_id email displayName')
      .lean()
      .exec();

    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    return res.status(200).json({
      id: user._id,
      email: user.email,
      displayName: user.displayName
    });
  } catch (err) {
    console.error(err);
    return res.status(500).json({ error: 'Internal Server Error' });
  }
}

module.exports = updateProfileHandler;
