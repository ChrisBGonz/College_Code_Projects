function setupAutoLogout(logoutFn, timeout = 15 * 60 * 1000) {
  if (typeof logoutFn !== 'function') throw new Error('logoutFn must be a function');
  let timer;
  const events = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll', 'click'];
  const reset = () => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      cleanup();
      logoutFn();
    }, timeout);
  };
  const cleanup = () => {
    events.forEach(evt => document.removeEventListener(evt, reset, true));
    clearTimeout(timer);
  };
  events.forEach(evt => document.addEventListener(evt, reset, true));
  reset();
  return cleanup;
}
