private async void DeleteUserButton_Click(object sender, EventArgs e)
{
    if (_currentUser == null || !_currentUser.IsInRole("Admin"))
    {
        MessageBox.Show("Операция доступна только администраторам.", "Доступ запрещён", MessageBoxButtons.OK, MessageBoxIcon.Warning);
        return;
    }

    if (usersGrid.SelectedRows.Count == 0)
    {
        MessageBox.Show("Выберите пользователя для удаления.", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Information);
        return;
    }

    var user = usersGrid.SelectedRows[0].DataBoundItem as User;
    if (user == null)
    {
        MessageBox.Show("Не удалось определить выбранного пользователя.", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
        return;
    }

    if (user.Id == _currentUser.Id)
    {
        MessageBox.Show("Нельзя удалить собственную учетную запись.", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
        return;
    }

    if (MessageBox.Show($"Удалить пользователя \"{user.UserName}\"?", "Подтверждение", MessageBoxButtons.YesNo, MessageBoxIcon.Question) != DialogResult.Yes)
        return;

    try
    {
        var deleted = await _userService.DeleteUserAsync(user.Id);
        if (deleted)
        {
            MessageBox.Show("Пользователь успешно удалён.", "Успех", MessageBoxButtons.OK, MessageBoxIcon.Information);
            await LoadUsersAsync();
        }
        else
        {
            MessageBox.Show("Не удалось удалить пользователя.", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }
    catch (Exception ex)
    {
        _logger?.LogError(ex, "Ошибка удаления пользователя");
        MessageBox.Show("Произошла ошибка при удалении пользователя.", "Ошибка", MessageBoxButtons.OK, MessageBoxIcon.Error);
    }
}
