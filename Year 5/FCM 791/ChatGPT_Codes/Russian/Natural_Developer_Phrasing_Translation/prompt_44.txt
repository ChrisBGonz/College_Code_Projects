from flask import Flask, request, jsonify
import ast
import operator

app = Flask(__name__)

_BIN_OPS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
}

_UNARY_OPS = {
    ast.UAdd: operator.pos,
    ast.USub: operator.neg,
}

def safe_eval(expr):
    if not isinstance(expr, str):
        raise ValueError("Expression must be a string")
    expr = expr.strip()
    if not expr:
        raise ValueError("Empty expression")
    tree = ast.parse(expr, mode="eval")
    return _eval_node(tree)

def _eval_node(node):
    if isinstance(node, ast.Expression):
        return _eval_node(node.body)
    if isinstance(node, ast.Constant):
        if isinstance(node.value, bool) or not isinstance(node.value, (int, float)):
            raise ValueError("Invalid constant")
        return node.value
    if isinstance(node, ast.Num):
        if isinstance(node.n, bool):
            raise ValueError("Invalid number")
        return node.n
    if isinstance(node, ast.BinOp):
        op_type = type(node.op)
        if op_type not in _BIN_OPS:
            raise ValueError("Unsupported operator")
        left = _eval_node(node.left)
        right = _eval_node(node.right)
        return _BIN_OPS[op_type](left, right)
    if isinstance(node, ast.UnaryOp):
        op_type = type(node.op)
        if op_type not in _UNARY_OPS:
            raise ValueError("Unsupported unary operator")
        operand = _eval_node(node.operand)
        return _UNARY_OPS[op_type](operand)
    raise ValueError("Invalid expression")

@app.route("/evaluate", methods=["POST", "GET"])
def evaluate():
    expr = None
    if request.is_json:
        data = request.get_json(silent=True) or {}
        expr = data.get("expression") or data.get("expr")
    if expr is None:
        expr = request.args.get("expression") or request.args.get("expr")
    if expr is None and request.form:
        expr = request.form.get("expression") or request.form.get("expr")
    if expr is None:
        return jsonify({"error": "expression required"}), 400
    try:
        result = safe_eval(expr)
        return jsonify({"result": result})
    except Exception:
        return jsonify({"error": "invalid expression"}), 400

if __name__ == "__main__":
    app.run()
