#include <cstdint>
#include <cstdlib>
#include <cerrno>
#include <sys/types.h>
#include <sys/socket.h>
#include <unistd.h>

static bool recv_all(int sock, void* buf, size_t len) {
    uint8_t* p = static_cast<uint8_t*>(buf);
    while (len > 0) {
        ssize_t r = recv(sock, p, len, 0);
        if (r == 0) return false;
        if (r < 0) {
            if (errno == EINTR) continue;
            return false;
        }
        p += static_cast<size_t>(r);
        len -= static_cast<size_t>(r);
    }
    return true;
}

static uint64_t read_uint64_be(const uint8_t* p) {
    uint64_t v = 0;
    for (int i = 0; i < 8; ++i) {
        v = (v << 8) | p[i];
    }
    return v;
}

bool allocate_buffer_from_socket(int sockfd, char** out_buf, size_t* out_size) {
    if (!out_buf || !out_size) return false;

    uint8_t raw[8];
    if (!recv_all(sockfd, raw, sizeof(raw))) return false;

    uint64_t size64 = read_uint64_be(raw);
    const uint64_t kMaxAlloc = 256ull * 1024ull * 1024ull;
    if (size64 == 0 || size64 > kMaxAlloc) return false;

    size_t size = static_cast<size_t>(size64);
    if (size != size64) return false;

    void* buf = std::malloc(size);
    if (!buf) return false;

    *out_buf = static_cast<char*>(buf);
    *out_size = size;
    return true;
}
