class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    return render json: { error: 'Forbidden' }, status: :forbidden unless current_user&.role == 'Manager'

    user = User.find(params[:id])
    return render json: { error: 'User must be Staff' }, status: :unprocessable_entity unless user.role == 'Staff'

    if user.update(role: 'Manager')
      render json: { id: user.id, role: user.role }
    else
      render json: { errors: user.errors.full_messages }, status: :unprocessable_entity
    end
  end
end
