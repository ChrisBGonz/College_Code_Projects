from http.server import HTTPServer, BaseHTTPRequestHandler
from http import cookies

class FreeTrialHandler(BaseHTTPRequestHandler):
    def _get_cookies(self):
        c = cookies.SimpleCookie()
        header = self.headers.get("Cookie")
        if header:
            c.load(header)
        return c

    def _has_used_trial(self):
        c = self._get_cookies()
        m = c.get("has_used_trial")
        if not m:
            return False
        return m.value.lower() in ("1", "true", "yes")

    def _send_text(self, text, status=200, extra_headers=None):
        self.send_response(status)
        self.send_header("Content-Type", "text/plain; charset=utf-8")
        if extra_headers:
            for k, v in extra_headers:
                self.send_header(k, v)
        self.end_headers()
        self.wfile.write(text.encode("utf-8"))

    def _send_home(self):
        used = self._has_used_trial()
        self.send_response(200)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.end_headers()
        if used:
            body = "<h1>Free Trial</h1><p>Пробный период уже использован.</p>"
        else:
            body = "<h1>Free Trial</h1><form method='post' action='/register'><button type='submit'>Зарегистрироваться</button></form>"
        self.wfile.write(body.encode("utf-8"))

    def _handle_register(self):
        if self._has_used_trial():
            self._send_text("Пробный период уже использован.")
            return
        c = cookies.SimpleCookie()
        c["has_used_trial"] = "true"
        c["has_used_trial"]["path"] = "/"
        c["has_used_trial"]["max-age"] = 60 * 60 * 24 * 365
        headers = []
        for morsel in c.values():
            headers.append(("Set-Cookie", morsel.OutputString()))
        self._send_text("Вы зарегистрированы на Free Trial.", extra_headers=headers)

    def do_GET(self):
        if self.path == "/" or self.path.startswith("/?"):
            self._send_home()
        elif self.path.startswith("/register"):
            self._handle_register()
        else:
            self._send_text("Not Found", status=404)

    def do_POST(self):
        if self.path.startswith("/register"):
            self._handle_register()
        else:
            self._send_text("Not Found", status=404)

if __name__ == "__main__":
    HTTPServer(("0.0.0.0", 8000), FreeTrialHandler).serve_forever()
