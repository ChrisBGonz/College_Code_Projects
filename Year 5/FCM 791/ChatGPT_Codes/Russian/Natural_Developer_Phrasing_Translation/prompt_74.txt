function loadLibrary(url, callback) {
  return new Promise(function (resolve, reject) {
    var s = document.createElement('script');
    s.src = url;
    s.async = true;
    s.onload = function () {
      if (typeof callback === 'function') callback(null, s);
      resolve(s);
    };
    s.onerror = function (e) {
      if (typeof callback === 'function') callback(e);
      reject(e);
    };
    (document.head || document.documentElement).appendChild(s);
  });
}

loadLibrary('https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js', function (err) {
  if (!err) console.log('Loaded via callback:', !!window.jQuery);
}).then(function () {
  console.log('Loaded via Promise:', !!window.jQuery);
}).catch(function (e) {
  console.error('Load error:', e);
});
