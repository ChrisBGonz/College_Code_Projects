const crypto = require('crypto');
const { persist } = require('./db');

function scryptAsync(password, salt, keyLength) {
  return new Promise((resolve, reject) => {
    crypto.scrypt(password, salt, keyLength, (err, derivedKey) => {
      if (err) return reject(err);
      resolve(derivedKey);
    });
  });
}

async function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString('hex');
  const hash = await scryptAsync(password, salt, 64);
  return `${salt}:${hash.toString('hex')}`;
}

async function resetPassword(user_id, new_password) {
  const errors = [];
  const id = typeof user_id === 'string' ? user_id.trim() : user_id;
  if (id === undefined || id === null || id === '') errors.push('invalid user_id');
  if (typeof new_password !== 'string') errors.push('invalid new_password');
  const pwd = typeof new_password === 'string' ? new_password.trim() : '';
  if (pwd.length < 8) errors.push('password too short');
  if (typeof persist !== 'function') errors.push('persist not available');
  if (errors.length) return { ok: false, error: errors.join(', ') };

  try {
    const password_hash = await hashPassword(pwd);
    const result = await persist('users', id, { password_hash, password_updated_at: new Date() });
    let updated = false;
    if (result) {
      if (typeof result === 'object') {
        if ('rowCount' in result) updated = result.rowCount > 0;
        else if ('affectedRows' in result) updated = result.affectedRows > 0;
        else updated = true;
      } else {
        updated = true;
      }
    }
    if (!updated) return { ok: false, error: 'user not found' };
    return { ok: true };
  } catch (err) {
    return { ok: false, error: err && err.message ? err.message : 'internal error' };
  }
}

module.exports = { resetPassword };
