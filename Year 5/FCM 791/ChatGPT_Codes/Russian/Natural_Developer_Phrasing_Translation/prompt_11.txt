import sys
from lxml import etree

def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
    else:
        path = input().strip()
        if not path:
            return
    try:
        tree = etree.parse(path)
    except Exception as e:
        sys.stderr.write(str(e))
        return
    for t in tree.xpath('//title'):
        print(''.join(t.itertext()))

if __name__ == "__main__":
    main()
