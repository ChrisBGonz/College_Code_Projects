import sys

def main():
    if len(sys.argv) < 2:
        print(f"Usage: {sys.argv[0]} <file>", file=sys.stderr)
        sys.exit(1)
    path = sys.argv[1]
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        print(0)
        return
    count = data.count(b'\n')
    if data[-1:] != b'\n':
        count += 1
    print(count)

if __name__ == "__main__":
    main()
