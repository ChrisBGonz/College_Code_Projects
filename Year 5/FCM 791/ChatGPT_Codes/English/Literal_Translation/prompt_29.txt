#!/usr/bin/env python3
import os
import sys
import subprocess
import pathlib

def find_project_root(start):
    p = pathlib.Path(start).resolve()
    if p.is_file():
        if p.name == "pom.xml":
            return p.parent
        p = p.parent
    while True:
        if (p / "pom.xml").is_file():
            return p
        if p.parent == p:
            return None
        p = p.parent

def detect_maven_cmd(project_dir):
    if os.name == "nt":
        mvnw_cmd = project_dir / "mvnw.cmd"
        if mvnw_cmd.exists():
            return str(mvnw_cmd)
    mvnw = project_dir / "mvnw"
    if mvnw.exists():
        return str(mvnw)
    return "mvn"

def run(cmd, cwd):
    proc = subprocess.run(cmd, cwd=cwd)
    if proc.returncode != 0:
        sys.exit(proc.returncode)

def main():
    target = sys.argv[1] if len(sys.argv) > 1 else "."
    root = find_project_root(target)
    if not root:
        sys.stderr.write("pom.xml not found\n")
        sys.exit(1)

    mvn = detect_maven_cmd(root)

    allow_snapshots = os.environ.get("ALLOW_SNAPSHOTS", "").lower() in ("1", "true", "yes", "on")
    allow_val = "true" if allow_snapshots else "false"
    version_goal = "versions:use-latest-versions" if allow_snapshots else "versions:use-latest-releases"

    common = [
        "-B",
        f"-DallowSnapshots={allow_val}",
        "-DgenerateBackupPoms=false",
        "-DprocessAllModules=true"
    ]

    run([mvn, *common, "versions:update-parent"], root)
    run([mvn, *common,
         "-DprocessDependencies=true",
         "-DprocessDependencyManagement=true",
         "-DprocessPlugins=true",
         "-DprocessPluginDependencies=true",
         version_goal], root)
    run([mvn, *common, "versions:update-properties"], root)

if __name__ == "__main__":
    main()
