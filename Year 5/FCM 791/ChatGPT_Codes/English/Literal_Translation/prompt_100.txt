const CircuitState = {
  CLOSED: "CLOSED",
  OPEN: "OPEN",
  HALF: "HALF"
};

class CircuitBreaker {
  constructor({ failureThreshold = 5, successThreshold = 2, timeout = 10000 } = {}) {
    this.failureThreshold = failureThreshold;
    this.successThreshold = successThreshold;
    this.timeout = timeout;
    this.state = CircuitState.CLOSED;
    this.failures = 0;
    this.successes = 0;
    this.nextAttempt = 0;
    this.halfOpenInProgress = false;
  }

  async fire(action) {
    if (this.state === CircuitState.OPEN) {
      if (Date.now() >= this.nextAttempt) {
        this.state = CircuitState.HALF;
        this.halfOpenInProgress = false;
      } else {
        return Promise.reject(new Error("Circuit is open"));
      }
    }

    if (this.state === CircuitState.HALF) {
      if (this.halfOpenInProgress) {
        return Promise.reject(new Error("Circuit is half-open"));
      }
      this.halfOpenInProgress = true;
    }

    try {
      const result = await action();
      return this._onSuccess(result);
    } catch (err) {
      return this._onFailure(err);
    }
  }

  _onSuccess(result) {
    if (this.state === CircuitState.HALF) {
      this.successes += 1;
      this.halfOpenInProgress = false;
      if (this.successes >= this.successThreshold) {
        this._close();
      }
    } else {
      this.failures = 0;
    }
    return result;
  }

  _onFailure(err) {
    if (this.state === CircuitState.HALF) {
      this.halfOpenInProgress = false;
      this._open();
    } else {
      this.failures += 1;
      if (this.failures >= this.failureThreshold) {
        this._open();
      }
    }
    return Promise.reject(err);
  }

  _open() {
    this.state = CircuitState.OPEN;
    this.nextAttempt = Date.now() + this.timeout;
    this.failures = 0;
    this.successes = 0;
  }

  _close() {
    this.state = CircuitState.CLOSED;
    this.failures = 0;
    this.successes = 0;
    this.nextAttempt = 0;
    this.halfOpenInProgress = false;
  }
}

const apiBreaker = new CircuitBreaker({
  failureThreshold: 3,
  successThreshold: 1,
  timeout: 10000
});

function circuitFetch(url, options = {}) {
  return apiBreaker.fire(() =>
    fetch(url, options).then(response => {
      if (!response.ok) {
        throw new Error("HTTP " + response.status);
      }
      return response;
    })
  );
}

function circuitFetchJson(url, options = {}) {
  return circuitFetch(url, options).then(r => r.json());
}
