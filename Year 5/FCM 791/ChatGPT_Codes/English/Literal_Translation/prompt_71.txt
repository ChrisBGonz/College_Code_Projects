import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Any

@dataclass
class InventoryItem:
    item_id: int
    name: str
    quantity: int
    properties: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    health: int
    position: Tuple[int, int]
    inventory: List[InventoryItem] = field(default_factory=list)
    skills: Dict[str, int] = field(default_factory=dict)

@dataclass
class Quest:
    quest_id: int
    name: str
    completed: bool
    objectives: List[str] = field(default_factory=list)

@dataclass
class WorldState:
    time_of_day: str
    weather: str
    discovered_locations: List[str] = field(default_factory=list)
    npc_states: Dict[str, Dict[str, Any]] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world: WorldState
    quests: List[Quest] = field(default_factory=list)
    flags: Dict[str, bool] = field(default_factory=dict)

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def create_sample_game_state() -> GameState:
    inventory = [
        InventoryItem(1, "Sword", 1, {"damage": 10, "durability": 0.8}),
        InventoryItem(2, "Potion", 5, {"heal": 25}),
    ]
    player = Player(
        name="Hero",
        level=7,
        health=85,
        position=(12, 34),
        inventory=inventory,
        skills={"swordsmanship": 4, "archery": 2},
    )
    world = WorldState(
        time_of_day="Evening",
        weather="Rainy",
        discovered_locations=["Village", "Forest", "Castle Ruins"],
        npc_states={
            "Blacksmith": {"relationship": "friendly", "last_met": "Village"},
            "Guard": {"relationship": "neutral", "last_met": "Castle Gate"},
        },
    )
    quests = [
        Quest(100, "Find the Lost Sword", False, ["Talk to the Blacksmith", "Search the forest"]),
        Quest(101, "Defeat the Bandit Leader", True, ["Enter the bandit camp", "Defeat the leader"]),
    ]
    flags = {"tutorial_completed": True, "dragons_unlocked": False}
    return GameState(player=player, world=world, quests=quests, flags=flags)

def main() -> None:
    filename = "game_state.pkl"
    state = create_sample_game_state()
    save_game_state(state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
