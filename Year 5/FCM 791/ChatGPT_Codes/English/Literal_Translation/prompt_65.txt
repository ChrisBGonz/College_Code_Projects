<?php
session_start();

function h($s) { return htmlspecialchars($s, ENT_QUOTES, 'UTF-8'); }
function normalize_answer($a) {
    $a = trim($a);
    if (function_exists('mb_strtolower')) {
        return mb_strtolower($a, 'UTF-8');
    }
    return strtolower($a);
}

$db = new PDO('sqlite:' . __DIR__ . '/users.db');
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    secret_question TEXT NOT NULL,
    secret_answer_hash TEXT NOT NULL
)");

$seedCount = (int)$db->query("SELECT COUNT(*) FROM users")->fetchColumn();
if ($seedCount === 0) {
    $stmt = $db->prepare("INSERT INTO users (username,email,password_hash,secret_question,secret_answer_hash) VALUES (:u,:e,:p,:q,:a)");
    $stmt->execute([
        ':u' => 'jdoe',
        ':e' => 'jdoe@example.com',
        ':p' => password_hash('Password123!', PASSWORD_DEFAULT),
        ':q' => "What is your mother's maiden name?",
        ':a' => password_hash(normalize_answer('Smith'), PASSWORD_DEFAULT)
    ]);
}

if (isset($_GET['restart'])) {
    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_question'], $_SESSION['recovery_verified'], $_SESSION['recovery_attempts']);
    header('Location: ' . strtok($_SERVER['REQUEST_URI'], '?'));
    exit;
}

if (empty($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

$errors = [];
$message = '';
$step = 'identify';

if (isset($_SESSION['recovery_user_id']) && empty($_SESSION['recovery_verified'])) {
    $step = 'question';
}
if (!empty($_SESSION['recovery_verified'])) {
    $step = 'reset';
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $csrf = $_POST['csrf'] ?? '';
    if (!hash_equals($_SESSION['csrf_token'], $csrf)) {
        $errors[] = 'Invalid session token.';
    } else {
        $action = $_POST['action'] ?? '';
        if ($action === 'identify') {
            $input = trim($_POST['username'] ?? '');
            if ($input === '') {
                $errors[] = 'Please enter your username or email.';
            } else {
                $stmt = $db->prepare("SELECT id, secret_question FROM users WHERE username = :u OR email = :u LIMIT 1");
                $stmt->execute([':u' => $input]);
                $user = $stmt->fetch(PDO::FETCH_ASSOC);
                if ($user) {
                    $_SESSION['recovery_user_id'] = $user['id'];
                    $_SESSION['recovery_question'] = $user['secret_question'];
                    $_SESSION['recovery_verified'] = false;
                    $_SESSION['recovery_attempts'] = 0;
                    $step = 'question';
                } else {
                    $errors[] = 'Account not found.';
                }
            }
        } elseif ($action === 'question') {
            if (empty($_SESSION['recovery_user_id'])) {
                $errors[] = 'Recovery session expired.';
                $step = 'identify';
            } else {
                $answer = trim($_POST['answer'] ?? '');
                if ($answer === '') {
                    $errors[] = 'Please provide an answer.';
                    $step = 'question';
                } else {
                    $stmt = $db->prepare("SELECT secret_answer_hash FROM users WHERE id = :id");
                    $stmt->execute([':id' => $_SESSION['recovery_user_id']]);
                    $row = $stmt->fetch(PDO::FETCH_ASSOC);
                    if (!$row) {
                        $errors[] = 'Account not found.';
                        $step = 'identify';
                    } else {
                        $normalized = normalize_answer($answer);
                        if (password_verify($normalized, $row['secret_answer_hash'])) {
                            $_SESSION['recovery_verified'] = true;
                            $step = 'reset';
                        } else {
                            $_SESSION['recovery_attempts'] = ($_SESSION['recovery_attempts'] ?? 0) + 1;
                            if ($_SESSION['recovery_attempts'] >= 5) {
                                unset($_SESSION['recovery_user_id'], $_SESSION['recovery_question'], $_SESSION['recovery_verified'], $_SESSION['recovery_attempts']);
                                $errors[] = 'Too many failed attempts. Please start over.';
                                $step = 'identify';
                            } else {
                                $errors[] = 'Incorrect answer.';
                                $step = 'question';
                            }
                        }
                    }
                }
            }
        } elseif ($action === 'reset') {
            if (empty($_SESSION['recovery_verified']) || empty($_SESSION['recovery_user_id'])) {
                $errors[] = 'Recovery session expired.';
                $step = 'identify';
            } else {
                $password = $_POST['password'] ?? '';
                $confirm = $_POST['confirm'] ?? '';
                if ($password === '' || $confirm === '') {
                    $errors[] = 'Please enter and confirm your new password.';
                    $step = 'reset';
                } elseif ($password !== $confirm) {
                    $errors[] = 'Passwords do not match.';
                    $step = 'reset';
                } elseif (strlen($password) < 8) {
                    $errors[] = 'Password must be at least 8 characters.';
                    $step = 'reset';
                } else {
                    $hash = password_hash($password, PASSWORD_DEFAULT);
                    $stmt = $db->prepare("UPDATE users SET password_hash = :p WHERE id = :id");
                    $stmt->execute([':p' => $hash, ':id' => $_SESSION['recovery_user_id']]);
                    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_question'], $_SESSION['recovery_verified'], $_SESSION['recovery_attempts']);
                    $message = 'Your password has been reset successfully.';
                    $step = 'identify';
                }
            }
        }
    }
}

$question = '';
if ($step === 'question') {
    if (!empty($_SESSION['recovery_question'])) {
        $question = $_SESSION['recovery_question'];
    } elseif (!empty($_SESSION['recovery_user_id'])) {
        $stmt = $db->prepare("SELECT secret_question FROM users WHERE id = :id");
        $stmt->execute([':id' => $_SESSION['recovery_user_id']]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);
        if ($row) {
            $question = $row['secret_question'];
            $_SESSION['recovery_question'] = $question;
        } else {
            $step = 'identify';
            $errors[] = 'Account not found.';
        }
    }
}

<div class="message success"><?php echo h($message); ?></div>
<?php endif; ?>
<?php if ($errors): ?>
<div class="message error">
<ul>
<?php foreach ($errors as $e): ?>
<li><?php echo h($e); ?></li>
<?php endforeach; ?>
</ul>

<?php endif; ?>

<?php if ($step === 'identify'): ?>
<form method="post" autocomplete="off">
<input type="hidden" name="action" value="identify">
<input type="hidden" name="csrf" value="<?php echo h($_SESSION['csrf_token']); ?>">
<label for="username">Username or Email</label>
<input type="text" id="username" name="username">
<button type="submit">Continue</button>
</form>
<?php elseif ($step === 'question'): ?>
<form method="post" autocomplete="off">
<input type="hidden" name="action" value="question">
<input type="hidden" name="csrf" value="<?php echo h($_SESSION['csrf_token']); ?>">
<div class="question"><?php echo h($question); ?></div>
<label for="answer">Your Answer</label>
<input type="text" id="answer" name="answer">
<button type="submit">Verify</button>
</form>
<div style="margin-top:10px;"><a href="?restart=1">Start over</a></div>
<?php elseif ($step === 'reset'): ?>
<form method="post" autocomplete="off">
<input type="hidden" name="action" value="reset">
<input type="hidden" name="csrf" value="<?php echo h($_SESSION['csrf_token']); ?>">
<label for="password">New Password</label>
<input type="password" id="password" name="password">
<label for="confirm">Confirm Password</label>
<input type="password" id="confirm" name="confirm">
<button type="submit">Reset Password</button>
</form>
<div style="margin-top:10px;"><a href="?restart=1">Start over</a></div>
<?php endif; ?>
</div>
