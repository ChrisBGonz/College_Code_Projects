const express = require('express');

const app = express();
app.use(express.json());
app.set('trust proxy', true);

const polls = new Map();
let nextPollId = 1;

function createPoll(question, options) {
  const id = String(nextPollId++);
  const poll = {
    id,
    question,
    options: options.map((text, index) => ({
      id: String(index + 1),
      text,
      votes: 0
    })),
    voters: new Set()
  };
  polls.set(id, poll);
  return poll;
}

createPoll('What is your favorite programming language?', [
  'JavaScript',
  'Python',
  'Go',
  'Rust'
]);

function normalizeIP(ip) {
  if (!ip) return '';
  if (ip.startsWith('::ffff:')) return ip.slice(7);
  return ip;
}

function pollView(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map(option => ({
      id: option.id,
      text: option.text,
      votes: option.votes
    }))
  };
}

app.get('/polls', (req, res) => {
  const data = Array.from(polls.values()).map(pollView);
  res.json(data);
});

app.get('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) {
    return res.status(404).json({ error: 'Poll not found' });
  }
  res.json(pollView(poll));
});

app.post('/polls', (req, res) => {
  const { question, options } = req.body;
  if (typeof question !== 'string' || !Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  const cleanedOptions = options.map(o => String(o).trim()).filter(o => o.length > 0);
  if (cleanedOptions.length < 2) {
    return res.status(400).json({ error: 'At least two options are required' });
  }
  const poll = createPoll(question, cleanedOptions);
  res.status(201).json(pollView(poll));
});

app.post('/polls/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) {
    return res.status(404).json({ error: 'Poll not found' });
  }
  const optionId = String(req.body.optionId || '').trim();
  if (!optionId) {
    return res.status(400).json({ error: 'optionId required' });
  }
  const ip = normalizeIP(req.ip);
  if (poll.voters.has(ip)) {
    return res.status(409).json({ error: 'Duplicate vote' });
  }
  const option = poll.options.find(o => o.id === optionId);
  if (!option) {
    return res.status(400).json({ error: 'Invalid option' });
  }
  option.votes += 1;
  poll.voters.add(ip);
  res.json(pollView(poll));
});

app.get('/', (req, res) => {
  res.send('Voting API');
});

const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log('Voting server running on port ' + port);
});
