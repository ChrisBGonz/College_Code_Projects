import com.google.gson.Gson;

public class UserParser {
    public static class User {
        public String name;
    }

    public static void parseAndPrintUser(String json) {
        Gson gson = new Gson();
        User user = gson.fromJson(json, User.class);
        if (user != null) {
            System.out.println(user.name);
        }
    }
}
