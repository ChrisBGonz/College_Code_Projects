import sys
from dataclasses import dataclass

@dataclass
class Company:
    company_id: str
    name: str
    annual_report: str

@dataclass
class User:
    user_id: str
    name: str
    company_id: str

class MultiTenantSaaS:
    def __init__(self):
        self.companies = {}
        self.users = {}

    def add_company(self, company: Company):
        if company.company_id in self.companies:
            raise ValueError("Company already exists")
        self.companies[company.company_id] = company

    def add_user(self, user: User):
        if user.user_id in self.users:
            raise ValueError("User already exists")
        if user.company_id not in self.companies:
            raise ValueError("Company does not exist")
        self.users[user.user_id] = user

    def get_company_report(self, user_id: str, company_id: str) -> str:
        user = self.users.get(user_id)
        if not user:
            raise PermissionError("User not found")
        if user.company_id != company_id:
            raise PermissionError("User does not have access to this company")
        company = self.companies.get(company_id)
        if not company:
            raise ValueError("Company not found")
        return company.annual_report

def seed_data(app: MultiTenantSaaS):
    app.add_company(Company(company_id="C001", name="Alpha Corp", annual_report="Alpha Corp Annual Report: Revenue up 12%, net profit up 8%."))
    app.add_company(Company(company_id="C002", name="Beta LLC", annual_report="Beta LLC Annual Report: Revenue down 3%, net profit stable."))
    app.add_company(Company(company_id="C003", name="Gamma Inc", annual_report="Gamma Inc Annual Report: Expansion into new markets, revenue up 20%."))
    app.add_user(User(user_id="U100", name="Alice", company_id="C001"))
    app.add_user(User(user_id="U200", name="Bob", company_id="C002"))
    app.add_user(User(user_id="U300", name="Carol", company_id="C003"))

def main():
    app = MultiTenantSaaS()
    seed_data(app)

    print("Multi-tenant SaaS - Annual Report Viewer")
    print("Enter 'exit' at any prompt to quit.\n")

    while True:
        user_id = input("User ID: ").strip()
        if user_id.lower() == "exit":
            break
        company_id = input("Company ID: ").strip()
        if company_id.lower() == "exit":
            break

        try:
            report = app.get_company_report(user_id, company_id)
            print("\nAnnual Report")
            print("-" * 40)
            print(report)
            print("-" * 40 + "\n")
        except Exception as e:
            print(f"Error: {e}\n")

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        sys.exit(0)
