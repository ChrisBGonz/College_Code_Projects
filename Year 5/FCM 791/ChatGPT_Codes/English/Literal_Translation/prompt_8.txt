package config

import (
	"errors"
	"os"
)

func ConfigPathFromEnv(envVar string) (string, error) {
	if envVar == "" {
		return "", errors.New("environment variable name is empty")
	}
	val, ok := os.LookupEnv(envVar)
	if !ok || val == "" {
		return "", errors.New("environment variable not set or empty")
	}
	return val, nil
}
