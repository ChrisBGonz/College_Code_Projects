function initMultiStepRegistration(formSelector, storageKey) {
  storageKey = storageKey || 'multiStepRegistrationData';
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;

  function readStored() {
    var raw = localStorage.getItem(storageKey);
    if (!raw) return {};
    try {
      var data = JSON.parse(raw);
      return data && typeof data === 'object' ? data : {};
    } catch (e) {
      return {};
    }
  }

  function writeStored(data) {
    localStorage.setItem(storageKey, JSON.stringify(data));
  }

  function collectFormData() {
    var data = {};
    var elements = form.querySelectorAll('input,select,textarea');
    elements.forEach(function(el) {
      var name = el.name || el.id;
      if (!name) return;
      if (el.type === 'checkbox') {
        data[name] = el.checked;
      } else if (el.type === 'radio') {
        if (el.checked) data[name] = el.value;
      } else {
        data[name] = el.value;
      }
    });
    return data;
  }

  function save() {
    var existing = readStored();
    var current = collectFormData();
    var merged = {};
    Object.keys(existing).forEach(function(k) { merged[k] = existing[k]; });
    Object.keys(current).forEach(function(k) { merged[k] = current[k]; });
    writeStored(merged);
  }

  function restore() {
    var data = readStored();
    Object.keys(data).forEach(function(name) {
      var field = form.elements[name];
      if (!field) return;
      if (field.type === 'checkbox') {
        field.checked = !!data[name];
      } else if (field.type === 'radio') {
        var radios = form.querySelectorAll('input[type="radio"][name="' + name.replace(/"/g,'\\"') + '"]');
        radios.forEach(function(r){ r.checked = r.value == data[name]; });
      } else {
        field.value = data[name];
      }
    });
  }

  form.addEventListener('input', save);
  form.addEventListener('change', save);

  form.addEventListener('submit', function() {
    save();
    var data = readStored();
    Object.keys(data).forEach(function(name) {
      if (form.elements[name]) {
        var field = form.elements[name];
        if (field.type === 'checkbox') {
          field.checked = !!data[name];
        } else if (field.type === 'radio') {
          var radios = form.querySelectorAll('input[type="radio"][name="' + name.replace(/"/g,'\\"') + '"]');
          if (radios.length) {
            radios.forEach(function(r){ r.checked = r.value == data[name]; });
          } else {
            var hidden = document.createElement('input');
            hidden.type = 'hidden';
            hidden.name = name;
            hidden.value = data[name];
            form.appendChild(hidden);
          }
        } else {
          field.value = data[name];
        }
      } else {
        var hidden = document.createElement('input');
        hidden.type = 'hidden';
        hidden.name = name;
        hidden.value = data[name];
        form.appendChild(hidden);
      }
    });
    localStorage.removeItem(storageKey);
  });

  restore();
}
