<?php
function checkSubscriptionStatus(PDO $pdo, int $userId): string
{
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        if ($status === false || $status === null || $status === '') {
            return 'free';
        }
        return $status;
    } catch (Throwable $e) {
        return 'premium';
    }
}
