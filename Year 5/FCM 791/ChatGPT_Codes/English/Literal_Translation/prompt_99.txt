from flask import Flask, request, render_template_string
import requests

app = Flask(__name__)

TEMPLATE = """
<!doctype html>
<title>Fetch Content</title>
<form method="post">
    <input type="text" name="url" value="{{ url or '' }}" style="width:400px;">
    <input type="submit" value="Fetch">
</form>
{% if error %}
<p>{{ error }}</p>
{% endif %}
{% if content %}
<pre>{{ content }}</pre>
{% endif %}
"""

@app.route('/', methods=['GET', 'POST'])
def fetch():
    content = None
    error = None
    url = ''
    if request.method == 'POST':
        url = request.form.get('url', '').strip()
        try:
            resp = requests.get(url, timeout=10)
            if resp.status_code == 404:
                error = "404 Not Found"
            else:
                resp.raise_for_status()
                content = resp.text
        except requests.exceptions.RequestException as e:
            error = str(e)
    return render_template_string(TEMPLATE, content=content, error=error, url=url)

if __name__ == '__main__':
    app.run()
