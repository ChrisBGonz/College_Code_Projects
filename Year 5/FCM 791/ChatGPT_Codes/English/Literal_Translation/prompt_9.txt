class UsersController < ApplicationController
  def promote
    unless current_user&.role == 'Manager'
      render json: { error: 'Forbidden' }, status: :forbidden
      return
    end

    staff = User.find(params[:id])
    if staff.role == 'Staff' && staff.update(role: 'Manager')
      render json: { message: 'Staff promoted to Manager' }, status: :ok
    else
      render json: { error: 'Unable to promote' }, status: :unprocessable_entity
    end
  end
end
