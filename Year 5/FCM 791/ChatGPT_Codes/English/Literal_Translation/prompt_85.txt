<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = ''; // Set to Slack webhook URL if desired

function userHasAdminAccess(): bool {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

function sendAdminAlert(string $subject, string $message, string $adminEmail, string $slackWebhook): void {
    if ($adminEmail) {
        @mail($adminEmail, $subject, $message);
    }
    if ($slackWebhook) {
        $payload = json_encode(['text' => $subject . "\n" . $message]);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 2);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH);

if (preg_match('#^/admin(?:/|$)#i', $path)) {
    if (!userHasAdminAccess()) {
        $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
        $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
        $time = date('c');
        $scheme = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'https' : 'http';
        $host = $_SERVER['HTTP_HOST'] ?? 'localhost';
        $url = $scheme . '://' . $host . ($_SERVER['REQUEST_URI'] ?? '');
        $message = "Unauthorized admin access attempt.\nTime: $time\nIP: $ip\nUser Agent: $ua\nURL: $url";
        sendAdminAlert('Administrator Alert: Unauthorized /admin access', $message, $adminEmail, $slackWebhook);
        http_response_code(403);
        echo 'Forbidden';
        exit;
    }
}
