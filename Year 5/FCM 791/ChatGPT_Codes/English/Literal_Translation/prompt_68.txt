function startAutoLogout(logoutFn, timeoutMs = 15 * 60 * 1000) {
  if (typeof logoutFn !== "function") throw new Error("logoutFn must be a function");
  let timer;
  const events = ["mousemove", "mousedown", "keydown", "scroll", "touchstart", "click"];
  const reset = () => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      stop();
      logoutFn();
    }, timeoutMs);
  };
  const stop = () => {
    clearTimeout(timer);
    events.forEach(e => window.removeEventListener(e, reset, true));
  };
  events.forEach(e => window.addEventListener(e, reset, true));
  reset();
  return stop;
}
