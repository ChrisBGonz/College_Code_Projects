import sys
import time
import os
from collections import deque

def follow(fp):
    fp.seek(0, os.SEEK_END)
    while True:
        line = fp.readline()
        if line:
            yield line
        else:
            time.sleep(0.1)

def monitor(path):
    timestamps = deque()
    alerted = False
    with open(path, 'r', errors='ignore') as fp:
        for line in follow(fp):
            now = time.time()
            while timestamps and now - timestamps[0] > 60:
                timestamps.popleft()
            if 'Failed Login' in line:
                timestamps.append(now)
                if len(timestamps) > 10 and not alerted:
                    print("WARNING: more than 10 Failed Login messages in the last minute", flush=True)
                    alerted = True
            if len(timestamps) <= 10:
                alerted = False

def main():
    if len(sys.argv) < 2:
        print(f"Usage: {os.path.basename(sys.argv[0])} <logfile>")
        sys.exit(1)
    monitor(sys.argv[1])

if __name__ == '__main__':
    main()
