import socket
import sys
import struct

def xor_cipher(data, key):
    if not key:
        return data
    key_len = len(key)
    return bytes(b ^ key[i % key_len] for i, b in enumerate(data))

def recvall(sock, n):
    data = bytearray()
    while len(data) < n:
        chunk = sock.recv(n - len(data))
        if not chunk:
            return None
        data.extend(chunk)
    return bytes(data)

def send_packet(sock, payload, key):
    enc = xor_cipher(payload, key)
    sock.sendall(struct.pack('!I', len(enc)) + enc)

def recv_packet(sock, key):
    hdr = recvall(sock, 4)
    if not hdr:
        return None
    length = struct.unpack('!I', hdr)[0]
    enc = recvall(sock, length)
    if enc is None:
        return None
    return xor_cipher(enc, key)

def main():
    if len(sys.argv) < 4:
        sys.stderr.write(f"Usage: {sys.argv[0]} host port key [message]\n")
        return 1

    host = sys.argv[1]
    port = int(sys.argv[2])
    key = sys.argv[3].encode("utf-8")

    message = None
    if len(sys.argv) > 4:
        message = " ".join(sys.argv[4:]).encode("utf-8")

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))

    try:
        if message is not None:
            send_packet(sock, message, key)
            resp = recv_packet(sock, key)
            if resp is not None:
                sys.stdout.buffer.write(resp)
                if not resp.endswith(b"\n"):
                    sys.stdout.buffer.write(b"\n")
                sys.stdout.buffer.flush()
        else:
            while True:
                line = sys.stdin.buffer.readline()
                if not line:
                    break
                line = line.rstrip(b"\n")
                send_packet(sock, line, key)
                resp = recv_packet(sock, key)
                if resp is None:
                    break
                sys.stdout.buffer.write(resp + b"\n")
                sys.stdout.buffer.flush()
    finally:
        sock.close()

    return 0

if __name__ == "__main__":
    sys.exit(main())
