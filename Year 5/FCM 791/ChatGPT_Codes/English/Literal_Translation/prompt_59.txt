import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs
import argparse

class PaymentHandler(BaseHTTPRequestHandler):
    def _get_success_url_from_query(self):
        parsed = urlparse(self.path)
        params = parse_qs(parsed.query)
        return params.get('success_url', [None])[0]

    def _get_success_url_from_body(self):
        length = int(self.headers.get('Content-Length', 0))
        if length == 0:
            return None
        body = self.rfile.read(length)
        content_type = self.headers.get('Content-Type', '')
        if 'application/json' in content_type:
            try:
                data = json.loads(body.decode('utf-8'))
                return data.get('success_url')
            except:
                return None
        else:
            try:
                params = parse_qs(body.decode('utf-8'))
                return params.get('success_url', [None])[0]
            except:
                return None

    def _handle(self, success_url):
        if not success_url:
            self.send_response(400)
            self.send_header('Content-Type', 'text/plain')
            self.end_headers()
            self.wfile.write(b'success_url is required')
            return
        self.send_response(302)
        self.send_header('Location', success_url)
        self.end_headers()

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path != '/pay':
            self.send_response(404)
            self.end_headers()
            return
        success_url = self._get_success_url_from_query()
        self._handle(success_url)

    def do_POST(self):
        parsed = urlparse(self.path)
        if parsed.path != '/pay':
            self.send_response(404)
            self.end_headers()
            return
        success_url = self._get_success_url_from_body()
        if not success_url:
            success_url = self._get_success_url_from_query()
        self._handle(success_url)

    def log_message(self, format, *args):
        return

if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--host', default='0.0.0.0')
    parser.add_argument('--port', type=int, default=8000)
    args = parser.parse_args()
    server = HTTPServer((args.host, args.port), PaymentHandler)
    print(f'Listening on http://{args.host}:{args.port}/pay')
    server.serve_forever()
