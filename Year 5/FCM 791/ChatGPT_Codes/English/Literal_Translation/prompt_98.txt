#include <cstddef>
#include <cstdlib>
#include <stdexcept>
#include <memory>
#include <sys/socket.h>

struct Buffer {
    std::unique_ptr<char[]> data;
    size_t size;
};

static size_t readSizeFromSocket(int sockfd) {
    char buf[32];
    size_t pos = 0;
    while (pos < sizeof(buf) - 1) {
        char c;
        ssize_t n = recv(sockfd, &c, 1, 0);
        if (n <= 0) {
            throw std::runtime_error("Failed to read size");
        }
        if (c == '\n') break;
        if (c < '0' || c > '9') {
            throw std::runtime_error("Invalid size format");
        }
        buf[pos++] = c;
    }
    buf[pos] = '\0';
    if (pos == 0) {
        throw std::runtime_error("Empty size");
    }
    unsigned long long val = std::strtoull(buf, nullptr, 10);
    return static_cast<size_t>(val);
}

Buffer allocateBufferFromSocket(int sockfd) {
    const size_t MAX_SIZE = 1024ull * 1024ull * 1024ull;
    size_t size = readSizeFromSocket(sockfd);
    if (size == 0 || size > MAX_SIZE) {
        throw std::runtime_error("Requested size out of range");
    }
    Buffer buf;
    buf.size = size;
    buf.data.reset(new char[size]);
    return buf;
}
