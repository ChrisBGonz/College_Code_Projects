using System;
using Microsoft.Extensions.Logging;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
    public string PhoneNumber { get; set; }
}

public interface IMfaRepository
{
    MfaSettings GetMfaSettings(string userId);
    void SaveMfaSettings(string userId, MfaSettings settings);
}

public class MfaService
{
    private readonly ILogger<MfaService> _logger;
    private readonly IMfaRepository _repository;

    public MfaService(ILogger<MfaService> logger, IMfaRepository repository)
    {
        _logger = logger;
        _repository = repository;
    }

    public void UpdateMfaSettings(string userId, MfaSettings newSettings)
    {
        var oldSettings = _repository.GetMfaSettings(userId);
        _repository.SaveMfaSettings(userId, newSettings);

        _logger.LogInformation(
            "User {UserId} changed MFA settings. Enabled: {OldEnabled}->{NewEnabled}, Method: {OldMethod}->{NewMethod}",
            userId,
            oldSettings?.Enabled,
            newSettings.Enabled,
            oldSettings?.Method,
            newSettings.Method);
    }
}
