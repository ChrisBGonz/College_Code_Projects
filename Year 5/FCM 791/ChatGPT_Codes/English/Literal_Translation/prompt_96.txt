package main

import (
	"io"
	"net/http"
)

func CallMicroservice(url string) ([]byte, error) {
	for {
		resp, err := http.Get(url)
		if err != nil {
			continue
		}
		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			continue
		}
		if resp.StatusCode >= 200 && resp.StatusCode < 300 {
			return body, nil
		}
	}
}
