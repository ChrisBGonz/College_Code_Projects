const { execFile } = require('child_process');

function pingHost(domain_name, callback) {
  const args = process.platform === 'win32' ? ['-n', '1', domain_name] : ['-c', '1', domain_name];
  execFile('ping', args, (error, stdout, stderr) => {
    const isUp = !error;
    if (typeof callback === 'function') {
      callback(isUp, stdout || stderr);
    }
  });
}

module.exports = { pingHost };
