module.exports = function handleLogout(req, res, next) {
  if (req.session) {
    req.session.destroy(function (err) {
      if (err) {
        if (typeof next === 'function') return next(err);
        res.statusCode = 500;
        return res.end();
      }
      if (typeof res.clearCookie === 'function') res.clearCookie('connect.sid');
      res.statusCode = 204;
      res.end();
    });
  } else {
    res.statusCode = 204;
    res.end();
  }
};
