using System;
using System.Data.SqlClient;

string connectionString = "your-connection-string";

using (var conn = new SqlConnection(connectionString))
{
    conn.Open();
    var tx = conn.BeginTransaction();
    try
    {
        using (var debit = new SqlCommand(
            "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @id", conn, tx))
        {
            debit.Parameters.AddWithValue("@amount", 100m);
            debit.Parameters.AddWithValue("@id", "A");
            debit.ExecuteNonQuery();
        }

        using (var credit = new SqlCommand(
            "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @id", conn, tx))
        {
            credit.Parameters.AddWithValue("@amount", 100m);
            credit.Parameters.AddWithValue("@id", "B");
            credit.ExecuteNonQuery();
        }

        tx.Commit();
    }
    catch
    {
        tx.Rollback();
        throw;
    }
}
