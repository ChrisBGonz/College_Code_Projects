#!/usr/bin/env bash
set -euo pipefail

TOOL_NAME="lynis"
TOOL_URL="https://downloads.cisofy.com/lynis/lynis-latest.tar.gz"
INSTALL_DIR="/opt/${TOOL_NAME}"
BIN_LINK="/usr/local/bin/${TOOL_NAME}"

if command -v curl >/dev/null 2>&1; then
  download() { curl -fsSL "$1" -o "$2"; }
elif command -v wget >/dev/null 2>&1; then
  download() { wget -qO "$2" "$1"; }
else
  echo "curl or wget is required" >&2
  exit 1
fi

tmpdir=$(mktemp -d)
trap 'rm -rf "$tmpdir"' EXIT

archive="$tmpdir/${TOOL_NAME}.tar.gz"
download "$TOOL_URL" "$archive"

tar -xzf "$archive" -C "$tmpdir"

src_dir=$(find "$tmpdir" -maxdepth 1 -type d -name "${TOOL_NAME}*" | head -n1)
if [[ -z "$src_dir" ]]; then
  echo "Extraction failed" >&2
  exit 1
fi

if [[ $EUID -ne 0 ]]; then
  SUDO=sudo
else
  SUDO=
fi

$SUDO rm -rf "$INSTALL_DIR"
$SUDO mv "$src_dir" "$INSTALL_DIR"
$SUDO ln -sf "$INSTALL_DIR/$TOOL_NAME" "$BIN_LINK"

echo "$TOOL_NAME installed to $BIN_LINK"
