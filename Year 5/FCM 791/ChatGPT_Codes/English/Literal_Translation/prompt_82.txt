const requestLogger = (req, res, next) => {
  const logRequest = body => {
    const timestamp = new Date().toISOString();
    const url = req.originalUrl || req.url;
    let bodyString = body;
    if (bodyString === undefined) bodyString = '';
    if (typeof bodyString !== 'string') {
      try {
        bodyString = JSON.stringify(bodyString);
      } catch (e) {
        bodyString = String(bodyString);
      }
    }
    console.log(`[${timestamp}] ${req.method} ${url}`);
    console.log('Headers:', req.headers);
    console.log('Body:', bodyString);
  };

  if (req.body !== undefined || req.readableEnded || req.complete) {
    logRequest(req.body);
    return next();
  }

  const chunks = [];
  let finished = false;

  req.on('data', chunk => {
    chunks.push(chunk);
  });

  req.on('end', () => {
    if (finished) return;
    finished = true;
    const rawBody = Buffer.concat(chunks).toString('utf8');
    req.rawBody = rawBody;
    req.body = rawBody;
    req._body = true;
    logRequest(rawBody);
    next();
  });

  req.on('error', err => {
    if (finished) return;
    finished = true;
    console.error('Error reading request body:', err);
    next(err);
  });
};

module.exports = requestLogger;
