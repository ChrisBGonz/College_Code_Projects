#!/usr/bin/env python3
import sys
import os
import hashlib

def detect_algo(hash_hex):
    length = len(hash_hex.strip())
    return {32: 'md5', 40: 'sha1', 64: 'sha256', 128: 'sha512'}.get(length, 'sha256')

def compute_hash(path, algo):
    h = hashlib.new(algo)
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(65536), b''):
            h.update(chunk)
    return h.hexdigest()

def parse_line(line):
    line = line.strip()
    if not line or line.startswith('#'):
        return None
    parts = line.split()
    if len(parts) < 2:
        return None
    expected_hash = parts[0]
    relpath = ' '.join(parts[1:])
    if relpath.startswith('*'):
        relpath = relpath[1:]
    return expected_hash, relpath

def main():
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: verify_manifest.py <manifest_file>\n")
        return 2

    manifest_path = sys.argv[1]
    base_dir = os.path.dirname(os.path.abspath(manifest_path))

    try:
        with open(manifest_path, 'r', encoding='utf-8') as f:
            lines = f.readlines()
    except Exception as e:
        sys.stderr.write(str(e) + "\n")
        return 2

    errors = 0

    for line in lines:
        parsed = parse_line(line)
        if not parsed:
            continue
        expected_hash, relpath = parsed
        algo = detect_algo(expected_hash)
        file_path = relpath
        if not os.path.isabs(file_path):
            file_path = os.path.join(base_dir, file_path)
        if not os.path.exists(file_path):
            sys.stdout.write(f"MISSING {relpath}\n")
            errors += 1
            continue
        if not os.path.isfile(file_path):
            sys.stdout.write(f"NOT_A_FILE {relpath}\n")
            errors += 1
            continue
        try:
            actual_hash = compute_hash(file_path, algo)
        except Exception as e:
            sys.stdout.write(f"ERROR {relpath}: {e}\n")
            errors += 1
            continue
        if actual_hash.lower() != expected_hash.lower():
            sys.stdout.write(f"FAIL {relpath}\n")
            errors += 1
        else:
            sys.stdout.write(f"OK {relpath}\n")

    return 1 if errors else 0

if __name__ == '__main__':
    sys.exit(main())
