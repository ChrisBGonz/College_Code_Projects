const express = require('express');
const fs = require('fs').promises;
const path = require('path');

const app = express();
const VAULT_ROOT = path.join(__dirname, 'users');

function resolveSafe(base, target) {
  const targetPath = path.resolve(base, target);
  if (targetPath !== base && !targetPath.startsWith(base + path.sep)) {
    throw new Error('INVALID_PATH');
  }
  return targetPath;
}

app.get('/users/:userId/private_vault', async (req, res) => {
  const { userId } = req.params;
  const folderName = req.query.folder_name;

  if (!folderName) {
    return res.status(400).json({ error: 'folder_name parameter is required' });
  }

  const baseDir = path.join(VAULT_ROOT, userId, 'private_vault');

  try {
    const targetDir = resolveSafe(baseDir, folderName);
    const entries = await fs.readdir(targetDir, { withFileTypes: true });
    const files = entries.filter(e => e.isFile()).map(e => e.name);
    res.json({ files });
  } catch (err) {
    if (err.message === 'INVALID_PATH') {
      return res.status(400).json({ error: 'Invalid folder_name' });
    }
    if (err.code === 'ENOENT') {
      return res.status(404).json({ error: 'Folder not found' });
    }
    res.status(500).json({ error: 'Internal server error' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
