import base64
import hmac
import hashlib
import struct
import time
import sys

def decode_secret(secret):
    s = secret.strip().replace(" ", "").upper()
    padding = (-len(s)) % 8
    if padding:
        s += "=" * padding
    return base64.b32decode(s, casefold=True)

def generate_totp(secret_bytes, for_time=None, interval=30, digits=6, digest=hashlib.sha1):
    if for_time is None:
        for_time = time.time()
    counter = int(for_time // interval)
    msg = struct.pack(">Q", counter)
    h = hmac.new(secret_bytes, msg, digest).digest()
    offset = h[-1] & 0x0f
    code = (struct.unpack(">I", h[offset:offset + 4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(secret, code, window=1, interval=30, digits=6):
    secret_bytes = decode_secret(secret)
    now = time.time()
    for offset in range(-window, window + 1):
        if generate_totp(secret_bytes, now + offset * interval, interval, digits) == code:
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2].strip()
    else:
        secret = input("Secret: ").strip()
        code = input("Code: ").strip()

    if not code.isdigit():
        print("Invalid")
        sys.exit(1)

    if verify_totp(secret, code):
        print("Valid")
        sys.exit(0)
    else:
        print("Invalid")
        sys.exit(1)

if __name__ == "__main__":
    main()
