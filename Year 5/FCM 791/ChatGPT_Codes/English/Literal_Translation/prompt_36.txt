import argparse
import hashlib
import sys
import os

def sha256_file(path, chunk_size=8192):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(chunk_size), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Verify SHA256 checksum of a firmware file")
    parser.add_argument("file", help="Path to firmware file")
    parser.add_argument("expected", help="Expected SHA256 hex digest")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print("ERROR: File not found")
        sys.exit(2)

    computed = sha256_file(args.file)
    if computed.lower() == args.expected.lower():
        print("OK")
        sys.exit(0)
    else:
        print("FAIL")
        print("Expected:", args.expected.lower())
        print("Computed:", computed)
        sys.exit(1)

if __name__ == "__main__":
    main()
