<?php
$query = '';
if (isset($_GET['query'])) {
    $query = $_GET['query'];
} elseif (isset($_GET['q'])) {
    $query = $_GET['q'];
}
$safeQuery = htmlspecialchars($query, ENT_QUOTES, 'UTF-8');
?>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Search Results</title>
</head>
<body>
<p>Results for: <?php echo $safeQuery; ?></p>
</body>
</html>
