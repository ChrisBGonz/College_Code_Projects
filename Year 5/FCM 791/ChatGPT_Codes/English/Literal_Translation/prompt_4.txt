private void DeleteUserButton_Click(object sender, EventArgs e)
{
    if (CurrentUser == null || !CurrentUser.IsAdministrator)
    {
        MessageBox.Show("Access denied.", "Delete User", MessageBoxButtons.OK, MessageBoxIcon.Warning);
        return;
    }

    if (UsersList.SelectedItem == null)
    {
        MessageBox.Show("Select a user to delete.", "Delete User");
        return;
    }

    var user = (User)UsersList.SelectedItem;

    if (user.Id == CurrentUser.Id)
    {
        MessageBox.Show("You cannot delete your own account.", "Delete User");
        return;
    }

    if (MessageBox.Show($"Delete user '{user.Username}'?", "Confirm Delete", MessageBoxButtons.YesNo, MessageBoxIcon.Warning) != DialogResult.Yes)
        return;

    try
    {
        _userRepository.Delete(user.Id);
        UsersList.Items.Remove(user);
    }
    catch (Exception ex)
    {
        MessageBox.Show("Delete failed: " + ex.Message, "Delete User");
    }
}
