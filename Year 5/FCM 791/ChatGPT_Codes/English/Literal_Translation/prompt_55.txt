from wsgiref.simple_server import make_server
from http import cookies

FORM_HTML = """<!doctype html>
<html>
<head><title>Free Trial Signup</title></head>
<body>
<h1>Free Trial Signup</h1>
<form method="post">
<label>Email: <input type="email" name="email" required></label>
<button type="submit">Start Free Trial</button>
</form>
</body>
</html>
"""

def app(environ, start_response):
    cookie = cookies.SimpleCookie()
    cookie.load(environ.get('HTTP_COOKIE', ''))
    used = cookie.get('has_used_trial')

    if used and used.value == 'true':
        start_response('403 Forbidden', [('Content-Type', 'text/plain')])
        return [b'You have already used a free trial.']

    if environ.get('REQUEST_METHOD') == 'POST':
        try:
            size = int(environ.get('CONTENT_LENGTH') or 0)
        except (ValueError, TypeError):
            size = 0
        if size:
            environ['wsgi.input'].read(size)

        new_cookie = cookies.SimpleCookie()
        new_cookie['has_used_trial'] = 'true'
        new_cookie['has_used_trial']['path'] = '/'
        new_cookie['has_used_trial']['max-age'] = 60 * 60 * 24 * 365

        headers = [('Content-Type', 'text/plain')]
        for morsel in new_cookie.values():
            headers.append(('Set-Cookie', morsel.OutputString()))

        start_response('200 OK', headers)
        return [b'Your free trial has started.']

    start_response('200 OK', [('Content-Type', 'text/html; charset=utf-8')])
    return [FORM_HTML.encode('utf-8')]

if __name__ == '__main__':
    with make_server('0.0.0.0', 8000, app) as server:
        server.serve_forever()
