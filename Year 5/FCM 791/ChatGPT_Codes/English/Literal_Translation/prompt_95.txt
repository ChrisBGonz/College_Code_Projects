import sys

def main():
    if len(sys.argv) != 2:
        sys.stderr.write("Usage: python script.py <file>\n")
        sys.exit(1)
    path = sys.argv[1]
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        lines = 0
    else:
        lines = data.count(b'\n')
        if not data.endswith(b'\n'):
            lines += 1
    print(lines)

if __name__ == "__main__":
    main()
