using System;
using System.Collections.Generic;

namespace ShoppingCart
{
    public class CartItem
    {
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; } = 1;
    }

    public class Checkout
    {
        public decimal CalculateTotal(IEnumerable<CartItem> items)
        {
            if (items == null)
                return 0m;

            decimal total = 0m;
            foreach (var item in items)
            {
                if (item == null) continue;
                if (item.Price < 0m) throw new ArgumentOutOfRangeException(nameof(item.Price));
                if (item.Quantity < 0) throw new ArgumentOutOfRangeException(nameof(item.Quantity));
                total += item.Price * item.Quantity;
            }
            return total;
        }
    }
}
