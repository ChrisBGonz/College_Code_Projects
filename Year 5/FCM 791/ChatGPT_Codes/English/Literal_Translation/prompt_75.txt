<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $itemsPerPage = 10;
}

$prefs = null;
if (isset($_COOKIE['user_prefs'])) {
    $prefs = @unserialize($_COOKIE['user_prefs'], ['allowed_classes' => ['UserPreferences']]);
}
if (!$prefs instanceof UserPreferences) {
    $prefs = new UserPreferences();
}

if (isset($_REQUEST['theme'])) {
    $prefs->theme = $_REQUEST['theme'];
}
if (isset($_REQUEST['language'])) {
    $prefs->language = $_REQUEST['language'];
}
if (isset($_REQUEST['itemsPerPage'])) {
    $prefs->itemsPerPage = (int)$_REQUEST['itemsPerPage'];
}

setcookie('user_prefs', serialize($prefs), time() + 60 * 60 * 24 * 30, "/");

header('Content-Type: application/json');
echo json_encode($prefs);
?>
