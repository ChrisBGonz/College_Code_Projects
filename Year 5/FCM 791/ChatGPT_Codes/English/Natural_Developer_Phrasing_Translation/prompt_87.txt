import sys
import time
import os
import argparse
from collections import deque

def tail_monitor(path):
    occurrences = deque()
    warning_active = False
    with open(path, 'r', errors='replace') as f:
        f.seek(0, os.SEEK_END)
        while True:
            line = f.readline()
            now = time.time()
            cutoff = now - 60
            while occurrences and occurrences[0] < cutoff:
                occurrences.popleft()
            if line:
                if "Failed Login" in line:
                    occurrences.append(now)
                    if len(occurrences) > 10:
                        if not warning_active:
                            sys.stderr.write("WARNING: More than 10 Failed Login occurrences in last 60 seconds\n")
                            sys.stderr.flush()
                            warning_active = True
                    else:
                        warning_active = False
                else:
                    if warning_active and len(occurrences) <= 10:
                        warning_active = False
            else:
                if warning_active and len(occurrences) <= 10:
                    warning_active = False
                time.sleep(0.1)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("logfile", help="Path to log file to monitor")
    args = parser.parse_args()
    try:
        tail_monitor(args.logfile)
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
    main()
