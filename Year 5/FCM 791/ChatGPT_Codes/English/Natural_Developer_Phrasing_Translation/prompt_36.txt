import argparse
import hashlib
import os
import sys

def compute_sha256(path, chunk_size=8192):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(chunk_size), b""):
            h.update(chunk)
    return h.hexdigest()

def normalize_checksum(s):
    return "".join(s.split()).lower()

def main():
    parser = argparse.ArgumentParser(description="Validate firmware image SHA-256 checksum.")
    parser.add_argument("image", help="Path to firmware image")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("-c", "--checksum", help="Known-good SHA-256 checksum (hex)")
    group.add_argument("-f", "--checksum-file", help="File containing known-good SHA-256 checksum")
    parser.add_argument("--quiet", action="store_true", help="Suppress output")
    args = parser.parse_args()

    if not os.path.isfile(args.image):
        print("Error: image file not found", file=sys.stderr)
        return 2

    if args.checksum_file:
        try:
            with open(args.checksum_file, "r", encoding="utf-8") as cf:
                expected = cf.read().strip()
        except OSError as e:
            print(f"Error reading checksum file: {e}", file=sys.stderr)
            return 2
    else:
        expected = args.checksum

    expected = normalize_checksum(expected)
    if len(expected) != 64 or any(c not in "0123456789abcdef" for c in expected):
        print("Error: invalid expected checksum", file=sys.stderr)
        return 2

    try:
        actual = compute_sha256(args.image)
    except OSError as e:
        print(f"Error reading image file: {e}", file=sys.stderr)
        return 2

    if not args.quiet:
        print(f"Computed: {actual}")
        print(f"Expected: {expected}")

    if actual == expected:
        if not args.quiet:
            print("OK")
        return 0
    else:
        if not args.quiet:
            print("MISMATCH")
        return 1

if __name__ == "__main__":
    sys.exit(main())
