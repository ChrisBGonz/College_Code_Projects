const express = require('express');

const app = express();
app.use(express.json({ limit: '1mb' }));
app.set('trust proxy', true);

const polls = new Map();
let nextPollId = 1;

function createPoll(question, options) {
  const id = String(nextPollId++);
  return {
    id,
    question,
    options: options.map((text, index) => ({ id: String(index + 1), text, count: 0 })),
    votesByIp: new Set(),
    createdAt: new Date().toISOString()
  };
}

function serializePoll(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text, count: o.count })),
    createdAt: poll.createdAt
  };
}

function getClientIp(req) {
  const forwarded = req.headers['x-forwarded-for'];
  if (forwarded) return forwarded.split(',')[0].trim();
  return req.ip || req.socket.remoteAddress || '';
}

app.post('/polls', (req, res) => {
  const { question, options } = req.body || {};
  if (typeof question !== 'string' || !Array.isArray(options) || options.length < 2 || options.some(o => typeof o !== 'string' || !o.trim())) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  const poll = createPoll(question.trim(), options.map(o => o.trim()));
  polls.set(poll.id, poll);
  res.status(201).json(serializePoll(poll));
});

app.get('/polls', (req, res) => {
  res.json(Array.from(polls.values()).map(serializePoll));
});

app.get('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  res.json(serializePoll(poll));
});

app.post('/polls/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  const ip = getClientIp(req);
  if (!ip) return res.status(400).json({ error: 'Unable to determine IP' });
  if (poll.votesByIp.has(ip)) return res.status(429).json({ error: 'Duplicate vote' });
  const optionId = req.body ? req.body.optionId : undefined;
  if (optionId === undefined || optionId === null) return res.status(400).json({ error: 'optionId required' });
  const option = poll.options.find(o => o.id === String(optionId));
  if (!option) return res.status(400).json({ error: 'Option not found' });
  option.count += 1;
  poll.votesByIp.add(ip);
  res.json(serializePoll(poll));
});

const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log(`Poll voting service listening on port ${port}`);
});
