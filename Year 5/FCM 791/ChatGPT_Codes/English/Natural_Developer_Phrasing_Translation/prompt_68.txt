const IdleTimeoutHandler = (function () {
  const IDLE_LIMIT = 900000;
  let timerId = null;
  let logoutCallback = function () {
    if (typeof window !== "undefined" && window.location) {
      window.location.href = "/logout";
    }
  };
  const activityEvents = [
    "mousemove",
    "mousedown",
    "keydown",
    "scroll",
    "touchstart",
    "touchmove",
    "touchend",
    "wheel",
    "pointerdown",
    "pointermove"
  ];

  function resetTimer() {
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(function () {
      logoutCallback();
    }, IDLE_LIMIT);
  }

  function start() {
    activityEvents.forEach(function (event) {
      document.addEventListener(event, resetTimer, { passive: true });
    });
    resetTimer();
  }

  function stop() {
    activityEvents.forEach(function (event) {
      document.removeEventListener(event, resetTimer);
    });
    if (timerId) clearTimeout(timerId);
    timerId = null;
  }

  function setLogoutHandler(fn) {
    if (typeof fn === "function") logoutCallback = fn;
  }

  return {
    start: start,
    stop: stop,
    reset: resetTimer,
    setLogoutHandler: setLogoutHandler
  };
})();

IdleTimeoutHandler.start();
