import sys

def main():
    if len(sys.argv) < 2:
        sys.stderr.write(f"Usage: {sys.argv[0]} <file>\n")
        sys.exit(1)
    path = sys.argv[1]
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        count = 0
    else:
        count = data.count(b'\n')
        if data[-1] != 10:
            count += 1
    sys.stdout.write(str(count))

if __name__ == "__main__":
    main()
