import os
from flask import Flask, request, render_template_string, session, redirect, url_for, flash
import yaml

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "dev-secret-key")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024

TEMPLATE = """<!doctype html>
<html>
<head>
  <meta charset="utf-8">
  <title>YAML Profile Upload</title>
</head>
<body>
  <h1>Upload YAML Config</h1>
  {% with messages = get_flashed_messages() %}
    {% if messages %}
      <ul>
      {% for message in messages %}
        <li>{{ message }}</li>
      {% endfor %}
      </ul>
    {% endif %}
  {% endwith %}
  <form method="post" enctype="multipart/form-data">
    <input type="file" name="file" accept=".yml,.yaml" required>
    <button type="submit">Upload</button>
  </form>
  <h2>Current Profile</h2>
  <pre>{{ profile_text }}</pre>
  <form method="post" action="{{ url_for('reset') }}">
    <button type="submit">Reset Profile</button>
  </form>
</body>
</html>
"""

def sanitize(value):
    if isinstance(value, dict):
        return {str(k): sanitize(v) for k, v in value.items()}
    if isinstance(value, list):
        return [sanitize(v) for v in value]
    if isinstance(value, tuple):
        return [sanitize(v) for v in value]
    if isinstance(value, set):
        return [sanitize(v) for v in value]
    if isinstance(value, (str, int, float, bool)) or value is None:
        return value
    return str(value)

def merge_dicts(base, updates):
    result = dict(base) if isinstance(base, dict) else {}
    for k, v in updates.items():
        if isinstance(v, dict) and isinstance(result.get(k), dict):
            result[k] = merge_dicts(result[k], v)
        else:
            result[k] = v
    return result

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        file = request.files.get("file")
        if not file or file.filename == "":
            flash("No file selected.")
            return redirect(url_for("index"))
        try:
            raw = file.read()
            data = yaml.safe_load(raw)
        except yaml.YAMLError:
            flash("Failed to parse YAML.")
            return redirect(url_for("index"))
        if isinstance(data, dict) and "profile" in data and isinstance(data["profile"], dict):
            data = data["profile"]
        if not isinstance(data, dict):
            flash("YAML root must be a mapping/object.")
            return redirect(url_for("index"))
        data = sanitize(data)
        profile = session.get("profile", {})
        profile = merge_dicts(profile, data)
        session["profile"] = profile
        flash("Profile updated.")
        return redirect(url_for("index"))
    profile = session.get("profile", {})
    profile_text = yaml.safe_dump(profile, sort_keys=False) if profile else ""
    return render_template_string(TEMPLATE, profile_text=profile_text)

@app.route("/reset", methods=["POST"])
def reset():
    session.pop("profile", None)
    flash("Profile reset.")
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", 5000)))
