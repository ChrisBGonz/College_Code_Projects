<?php
$query = '';
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $query = $_POST['q'] ?? '';
} else {
    $query = $_GET['q'] ?? '';
}
$query = trim($query);
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Search Results</title>
</head>
<body>
    <h1>Search Results</h1>
    <p>Results for: <?php echo htmlspecialchars($query, ENT_QUOTES, 'UTF-8'); ?></p>
</body>
</html>
