import base64
import hashlib
import hmac
import struct
import time
import sys
import re

def _normalize_base32(s):
    s = s.strip().replace(" ", "").upper()
    return s + "=" * (-len(s) % 8)

def _decode_base32(secret):
    try:
        return base64.b32decode(_normalize_base32(secret), casefold=True)
    except Exception:
        return None

def generate_totp(secret, for_time=None, interval=30, digits=6, digest=hashlib.sha1):
    if for_time is None:
        for_time = time.time()
    key = _decode_base32(secret)
    if key is None:
        return None
    counter = int(for_time // interval)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, digest).digest()
    offset = h[-1] & 0x0F
    code = (struct.unpack(">I", h[offset:offset+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def validate_totp(secret, code, window=1, interval=30, digits=6):
    code = str(code).strip()
    if not re.fullmatch(r"\d{" + str(digits) + r"}", code):
        return False
    now = time.time()
    for offset in range(-window, window + 1):
        totp = generate_totp(secret, now + offset * interval, interval, digits)
        if totp == code:
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        try:
            secret = input("Secret: ").strip()
            code = input("Code: ").strip()
        except EOFError:
            return 1
    if validate_totp(secret, code):
        print("VALID")
        return 0
    else:
        print("INVALID")
        return 1

if __name__ == "__main__":
    sys.exit(main())
