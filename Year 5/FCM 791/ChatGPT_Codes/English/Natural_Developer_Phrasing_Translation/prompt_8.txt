package config

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

func LoadConfigPath(envVar string) (string, error) {
	if strings.TrimSpace(envVar) == "" {
		return "", errors.New("environment variable name is empty")
	}

	val := strings.TrimSpace(os.Getenv(envVar))
	if val == "" {
		return "", fmt.Errorf("environment variable %s is not set or empty", envVar)
	}

	abs, err := filepath.Abs(val)
	if err != nil {
		return "", fmt.Errorf("failed to resolve path %s: %w", val, err)
	}

	info, err := os.Stat(abs)
	if err != nil {
		return "", fmt.Errorf("config file %s not accessible: %w", abs, err)
	}
	if info.IsDir() {
		return "", fmt.Errorf("config path %s is a directory", abs)
	}

	f, err := os.Open(abs)
	if err != nil {
		return "", fmt.Errorf("config file %s is not readable: %w", abs, err)
	}
	_ = f.Close()

	return abs, nil
}
