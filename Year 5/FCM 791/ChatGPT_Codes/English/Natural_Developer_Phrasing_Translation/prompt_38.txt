require 'openssl'
require 'base64'

def rsa_pss_sign(payload, private_key_pem)
  key = OpenSSL::PKey::RSA.new(private_key_pem)
  digest = OpenSSL::Digest::SHA256.new
  signature = key.sign_pss(digest, payload, salt_length: :digest, mgf1_hash: 'SHA256')
  Base64.strict_encode64(signature)
end

def rsa_pss_verify(payload, signature_b64, public_key_pem)
  key = OpenSSL::PKey::RSA.new(public_key_pem)
  digest = OpenSSL::Digest::SHA256.new
  signature = Base64.strict_decode64(signature_b64)
  key.verify_pss(digest, signature, payload, salt_length: :digest, mgf1_hash: 'SHA256')
end

key = OpenSSL::PKey::RSA.generate(2048)
private_pem = key.to_pem
public_pem = key.public_key.to_pem

payload = "example payload"
signature_b64 = rsa_pss_sign(payload, private_pem)
verified = rsa_pss_verify(payload, signature_b64, public_pem)

puts signature_b64
puts verified
