<?php
declare(strict_types=1);

$dsn = 'mysql:host=localhost;dbname=payments;charset=utf8mb4';
$dbUser = 'dbuser';
$dbPass = 'dbpass';

$keyB64 = getenv('ENCRYPTION_KEY');
if ($keyB64 === false) {
    http_response_code(500);
    exit;
}
$key = base64_decode($keyB64, true);
if ($key === false || strlen($key) !== 32) {
    http_response_code(500);
    exit;
}

$cc = $_POST['credit_card_number'] ?? null;
if ($cc === null) {
    http_response_code(400);
    exit;
}

$iv = random_bytes(12);
$ciphertext = openssl_encrypt($cc, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);
if ($ciphertext === false) {
    http_response_code(500);
    exit;
}

$encryptedCc = base64_encode($iv . $tag . $ciphertext);

$pdo = new PDO($dsn, $dbUser, $dbPass, [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_EMULATE_PREPARES => false,
]);

$stmt = $pdo->prepare('INSERT INTO payments (credit_card_number) VALUES (:cc)');
$stmt->execute([':cc' => $encryptedCc]);
?>
