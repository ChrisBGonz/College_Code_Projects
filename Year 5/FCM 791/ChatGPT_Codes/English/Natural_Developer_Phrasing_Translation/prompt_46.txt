import os
from flask import Flask, request, redirect, make_response
from urllib.parse import urlparse
import requests

app = Flask(__name__)

def load_allowlist():
    raw = os.getenv("ALLOWED_REDIRECT_HOSTS", "example.com")
    return {h.strip().lower() for h in raw.split(",") if h.strip()}

ALLOWED_HOSTS = load_allowlist()

def host_allowed(host):
    host = host.lower()
    for allowed in ALLOWED_HOSTS:
        if host == allowed or host.endswith("." + allowed):
            return True
    return False

def is_allowed_url(url):
    try:
        parsed = urlparse(url)
    except Exception:
        return False
    if parsed.scheme not in ("http", "https"):
        return False
    if not parsed.netloc:
        return False
    if parsed.username or parsed.password:
        return False
    host = parsed.hostname
    if not host:
        return False
    if not host_allowed(host):
        return False
    return True

@app.route("/redirect")
def do_redirect():
    target = request.args.get("redirect_url", "")
    if not target:
        return make_response("Missing redirect_url", 400)
    if not is_allowed_url(target):
        return make_response("Invalid redirect_url", 400)
    if request.args.get("check") == "1":
        try:
            r = requests.head(target, timeout=3, allow_redirects=True)
            if r.status_code >= 400:
                return make_response("Target not healthy", 502)
        except requests.RequestException:
            return make_response("Target not reachable", 502)
    return redirect(target, code=302)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "8000")))
