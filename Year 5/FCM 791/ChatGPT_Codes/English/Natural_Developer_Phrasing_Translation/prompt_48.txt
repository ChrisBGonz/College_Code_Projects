using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

public class DataAccessor
{
    private readonly string _connectionString;

    public DataAccessor(string connectionString)
    {
        _connectionString = connectionString;
    }

    public DataTable GetSortedData(string columnName)
    {
        var allowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Id",
            "Name",
            "CreatedDate"
        };

        if (!allowedColumns.Contains(columnName))
            throw new ArgumentException("Invalid column name.", nameof(columnName));

        string sql = $"SELECT * FROM MyTable ORDER BY {columnName}";

        using (var conn = new SqlConnection(_connectionString))
        using (var cmd = new SqlCommand(sql, conn))
        using (var adapter = new SqlDataAdapter(cmd))
        {
            var table = new DataTable();
            conn.Open();
            adapter.Fill(table);
            return table;
        }
    }
}
