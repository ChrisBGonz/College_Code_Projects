using System;
using System.Data.SqlClient;

class Program
{
    static void Main()
    {
        string connectionString = "YOUR_CONNECTION_STRING";
        Transfer(connectionString, 1, 2, 100m);
    }

    static void Transfer(string connectionString, int accountAId, int accountBId, decimal amount)
    {
        using var connection = new SqlConnection(connectionString);
        connection.Open();

        using var transaction = connection.BeginTransaction();
        try
        {
            using var debitCmd = connection.CreateCommand();
            debitCmd.Transaction = transaction;
            debitCmd.CommandText = "UPDATE Accounts SET Balance = Balance - @amount WHERE Id = @id";
            debitCmd.Parameters.AddWithValue("@amount", amount);
            debitCmd.Parameters.AddWithValue("@id", accountAId);
            if (debitCmd.ExecuteNonQuery() != 1)
                throw new Exception("Debit failed.");

            using var creditCmd = connection.CreateCommand();
            creditCmd.Transaction = transaction;
            creditCmd.CommandText = "UPDATE Accounts SET Balance = Balance + @amount WHERE Id = @id";
            creditCmd.Parameters.AddWithValue("@amount", amount);
            creditCmd.Parameters.AddWithValue("@id", accountBId);
            if (creditCmd.ExecuteNonQuery() != 1)
                throw new Exception("Credit failed.");

            transaction.Commit();
        }
        catch
        {
            try { transaction.Rollback(); } catch { }
            throw;
        }
    }
}
