function logout(req, res, next) {
  if (req.session) {
    req.session.destroy(function (err) {
      res.clearCookie('connect.sid');
      if (err) return next(err);
      res.status(204).end();
    });
  } else {
    res.clearCookie('connect.sid');
    res.status(204).end();
  }
}

module.exports = logout;
