using System;
using Microsoft.AspNetCore.Http;

public static class RememberMeCookieIssuer
{
    public static void IssuePersistentAuthCookie(HttpContext context, string cookieName, string token, bool rememberMe, TimeSpan? lifetime = null)
    {
        if (!rememberMe || string.IsNullOrWhiteSpace(token))
            return;

        var options = new CookieOptions
        {
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Lax,
            Expires = DateTimeOffset.UtcNow.Add(lifetime ?? TimeSpan.FromDays(30))
        };

        context.Response.Cookies.Append(cookieName, token, options);
    }
}
