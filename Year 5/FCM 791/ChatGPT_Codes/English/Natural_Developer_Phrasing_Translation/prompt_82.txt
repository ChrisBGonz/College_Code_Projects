const express = require('express');

const logger = express.Router();

const captureRawBody = (req, res, buf, encoding) => {
  req.rawBody = buf && buf.length ? buf.toString(encoding || 'utf8') : '';
};

logger.use(express.json({ verify: captureRawBody }));
logger.use(express.urlencoded({ extended: true, verify: captureRawBody }));
logger.use(express.raw({ type: '*/*', verify: captureRawBody }));

logger.use((req, res, next) => {
  console.log(`${req.method} ${req.originalUrl || req.url}`);
  console.log('Headers:', req.headers);
  console.log('Raw Body:', req.rawBody || '');
  next();
});

module.exports = logger;
