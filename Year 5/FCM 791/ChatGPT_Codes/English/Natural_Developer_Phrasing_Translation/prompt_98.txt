#include <cstdint>
#include <sys/socket.h>
#include <arpa/inet.h>

char* readBufferFromSocket(int sockfd) {
    uint32_t len_net;
    ssize_t n = recv(sockfd, &len_net, sizeof(len_net), MSG_WAITALL);
    if (n != sizeof(len_net)) return nullptr;
    uint32_t len = ntohl(len_net);
    char* buffer = new char[len];
    size_t received = 0;
    while (received < len) {
        n = recv(sockfd, buffer + received, len - received, 0);
        if (n <= 0) { delete[] buffer; return nullptr; }
        received += n;
    }
    return buffer;
}
