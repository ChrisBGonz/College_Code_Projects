<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $timezone = 'UTC';
    public $itemsPerPage = 10;
}

$cookieName = 'user_prefs';
$prefs = new UserPreferences();

if (isset($_COOKIE[$cookieName])) {
    $loaded = @unserialize($_COOKIE[$cookieName]);
    if ($loaded instanceof UserPreferences) {
        $prefs = $loaded;
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme'])) $prefs->theme = $_POST['theme'];
    if (isset($_POST['language'])) $prefs->language = $_POST['language'];
    if (isset($_POST['timezone'])) $prefs->timezone = $_POST['timezone'];
    if (isset($_POST['itemsPerPage'])) $prefs->itemsPerPage = (int)$_POST['itemsPerPage'];
    setcookie($cookieName, serialize($prefs), time() + 60 * 60 * 24 * 30, "/");
}

date_default_timezone_set($prefs->timezone);

header('Content-Type: application/json; charset=utf-8');
echo json_encode([
    'theme' => $prefs->theme,
    'language' => $prefs->language,
    'timezone' => $prefs->timezone,
    'itemsPerPage' => $prefs->itemsPerPage,
    'now' => date('c')
]);
