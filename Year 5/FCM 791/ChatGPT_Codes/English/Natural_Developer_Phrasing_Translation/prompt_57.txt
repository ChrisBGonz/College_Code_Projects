<?php
declare(strict_types=1);
session_start();

function get_authenticated_user(): ?string {
    $candidates = [
        $_SESSION['employee'] ?? null,
        $_SESSION['username'] ?? null,
        $_SESSION['user'] ?? null,
        $_SERVER['REMOTE_USER'] ?? null,
        $_SERVER['PHP_AUTH_USER'] ?? null,
    ];
    foreach ($candidates as $c) {
        if (is_string($c)) {
            $c = trim($c);
            if ($c !== '') {
                return $c;
            }
        }
    }
    return null;
}

function wants_json(): bool {
    if (isset($_GET['format']) && strtolower((string)$_GET['format']) === 'json') {
        return true;
    }
    if (!empty($_SERVER['HTTP_ACCEPT']) && stripos($_SERVER['HTTP_ACCEPT'], 'application/json') !== false) {
        return true;
    }
    if (!empty($_SERVER['HTTP_X_REQUESTED_WITH']) && strtolower($_SERVER['HTTP_X_REQUESTED_WITH']) === 'xmlhttprequest') {
        return true;
    }
    return false;
}

function sanitize_filename(string $name): string {
    $name = trim($name);
    $name = basename($name);
    $name = str_replace(["\0", "\r", "\n", "\t"], '', $name);
    $name = preg_replace('/[^\w\-.]+/u', '_', $name);
    if ($name === '' || $name === '.' || $name === '..') {
        $name = 'document.pdf';
    }
    return $name;
}

function respond_json(bool $success, string $message, array $extra = [], int $status = 200): void {
    http_response_code($status);
    header('Content-Type: application/json');
    echo json_encode(array_merge(['success' => $success, 'message' => $message], $extra));
    exit;
}

function process_upload(string $uploader, string $uploadDir, string $metaDb, int $maxSize, array $allowedMimes): array {
    if (!isset($_FILES['document']) || !is_array($_FILES['document'])) {
        if (!empty($_SERVER['CONTENT_LENGTH']) && (int)$_SERVER['CONTENT_LENGTH'] > 0) {
            return ['success' => false, 'message' => 'No file received. The upload may exceed server limits.', 'status' => 400, 'extra' => []];
        }
        return ['success' => false, 'message' => 'No file uploaded.', 'status' => 400, 'extra' => []];
    }

    $file = $_FILES['document'];

    if (!isset($file['error'])) {
        return ['success' => false, 'message' => 'Invalid upload data.', 'status' => 400, 'extra' => []];
    }

    switch ($file['error']) {
        case UPLOAD_ERR_OK:
            break;
        case UPLOAD_ERR_INI_SIZE:
        case UPLOAD_ERR_FORM_SIZE:
            return ['success' => false, 'message' => 'File size exceeds the allowed limit.', 'status' => 400, 'extra' => []];
        case UPLOAD_ERR_PARTIAL:
            return ['success' => false, 'message' => 'File upload was incomplete.', 'status' => 400, 'extra' => []];
        case UPLOAD_ERR_NO_FILE:
            return ['success' => false, 'message' => 'No file uploaded.', 'status' => 400, 'extra' => []];
        case UPLOAD_ERR_NO_TMP_DIR:
            return ['success' => false, 'message' => 'Server configuration error: missing temporary folder.', 'status' => 500, 'extra' => []];
        case UPLOAD_ERR_CANT_WRITE:
            return ['success' => false, 'message' => 'Server error: failed to write file.', 'status' => 500, 'extra' => []];
        case UPLOAD_ERR_EXTENSION:
            return ['success' => false, 'message' => 'File upload stopped by a server extension.', 'status' => 500, 'extra' => []];
        default:
            return ['success' => false, 'message' => 'Unknown upload error.', 'status' => 500, 'extra' => []];
    }

    if (!isset($file['size']) || $file['size'] <= 0) {
        return ['success' => false, 'message' => 'Uploaded file is empty.', 'status' => 400, 'extra' => []];
    }

    if ($file['size'] > $maxSize) {
        return ['success' => false, 'message' => 'File size exceeds the maximum allowed size.', 'status' => 400, 'extra' => []];
    }

    if (!is_dir($uploadDir)) {
        if (!mkdir($uploadDir, 0700, true)) {
            return ['success' => false, 'message' => 'Server error: unable to create upload directory.', 'status' => 500, 'extra' => []];
        }
    }

    if (!is_writable($uploadDir)) {
        return ['success' => false, 'message' => 'Server error: upload directory is not writable.', 'status' => 500, 'extra' => []];
    }

    $originalName = sanitize_filename((string)$file['name']);
    $ext = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
    if ($ext !== 'pdf') {
        return ['success' => false, 'message' => 'Only PDF files are allowed.', 'status' => 400, 'extra' => []];
    }

    $tmp = $file['tmp_name'] ?? '';
    if ($tmp === '' || !is_uploaded_file($tmp)) {
        return ['success' => false, 'message' => 'Invalid uploaded file.', 'status' => 400, 'extra' => []];
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mime = $finfo->file($tmp);
    if ($mime === false || !in_array($mime, $allowedMimes, true)) {
        return ['success' => false, 'message' => 'Invalid file type. Only PDF is allowed.', 'status' => 400, 'extra' => []];
    }

    $handle = fopen($tmp, 'rb');
    $header = $handle ? fread($handle, 5) : '';
    if ($handle) {
        fclose($handle);
    }
    if (strncmp($header, '%PDF-', 5) !== 0) {
        return ['success' => false, 'message' => 'Invalid PDF file.', 'status' => 400, 'extra' => []];
    }

    do {
        try {
            $randomName = bin2hex(random_bytes(16)) . '.pdf';
        } catch (Exception $e) {
            return ['success' => false, 'message' => 'Server error: unable to generate filename.', 'status' => 500, 'extra' => []];
        }
        $dest = $uploadDir . DIRECTORY_SEPARATOR . $randomName;
    } while (file_exists($dest));

    if (!move_uploaded_file($tmp, $dest)) {
        return ['success' => false, 'message' => 'Server error: failed to store the uploaded file.', 'status' => 500, 'extra' => []];
    }

    chmod($dest, 0640);

    try {
        $pdo = new PDO('sqlite:' . $metaDb);
        $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
        $pdo->exec('CREATE TABLE IF NOT EXISTS uploads (id INTEGER PRIMARY KEY AUTOINCREMENT, uploader TEXT NOT NULL, uploaded_at TEXT NOT NULL, original_filename TEXT NOT NULL, stored_path TEXT NOT NULL)');
        $stmt = $pdo->prepare('INSERT INTO uploads (uploader, uploaded_at, original_filename, stored_path) VALUES (:uploader, :uploaded_at, :original_filename, :stored_path)');
        $uploadedAt = gmdate('c');
        $stmt->execute([
            ':uploader' => $uploader,
            ':uploaded_at' => $uploadedAt,
            ':original_filename' => $originalName,
            ':stored_path' => $dest,
        ]);
        $id = $pdo->lastInsertId();
    } catch (Exception $e) {
        @unlink($dest);
        return ['success' => false, 'message' => 'Upload failed while saving metadata.', 'status' => 500, 'extra' => []];
    }

    return ['success' => true, 'message' => 'Upload successful.', 'status' => 200, 'extra' => ['id' => $id, 'original_filename' => $originalName]];
}

$wantsJson = wants_json();

$maxSizeEnv = getenv('UPLOAD_MAX_SIZE');
$maxSize = (is_string($maxSizeEnv) && is_numeric($maxSizeEnv) && (int)$maxSizeEnv > 0) ? (int)$maxSizeEnv : 10 * 1024 * 1024;

$uploadDirEnv = getenv('COMPANY_UPLOAD_DIR');
$uploadDir = (is_string($uploadDirEnv) && trim($uploadDirEnv) !== '') ? $uploadDirEnv : dirname(__DIR__) . DIRECTORY_SEPARATOR . 'company_uploads';
$uploadDir = rtrim($uploadDir, DIRECTORY_SEPARATOR);

$metaDb = $uploadDir . DIRECTORY_SEPARATOR . 'upload_metadata.sqlite';

$allowedMimes = ['application/pdf', 'application/x-pdf', 'application/acrobat', 'applications/vnd.pdf', 'text/pdf', 'text/x-pdf'];

$uploader = get_authenticated_user();
$authenticated = $uploader !== null;

$message = '';
$success = false;
$extra = [];

if ($wantsJson && $_SERVER['REQUEST_METHOD'] !== 'POST') {
    if (!$authenticated) {
        respond_json(false, 'Access denied. Authentication required.', [], 403);
    }
    respond_json(true, 'Ready', ['max_size' => $maxSize], 200);
}

if (!$authenticated) {
    if ($wantsJson) {
        respond_json(false, 'Access denied. Authentication required.', [], 403);
    }
    $message = 'Access denied. Authentication required.';
} elseif ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $result = process_upload($uploader, $uploadDir, $metaDb, $maxSize, $allowedMimes);
    if ($wantsJson) {
        respond_json($result['success'], $result['message'], $result['extra'], $result['status']);
    }
    $success = $result['success'];
    $message = $result['message'];
    $extra = $result['extra'];
}
