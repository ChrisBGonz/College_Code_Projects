'use strict';

const bcrypt = require('bcrypt');

const PASSWORD_MIN_LENGTH = 8;
const PASSWORD_MAX_LENGTH = 128;
const SALT_ROUNDS = 12;

async function resetPassword(userId, newPassword, db) {
  if (userId === undefined || userId === null || userId === '') {
    throw new Error('Invalid userId');
  }

  if (typeof newPassword !== 'string') {
    throw new Error('Invalid password');
  }

  const password = newPassword.trim();

  if (password.length < PASSWORD_MIN_LENGTH) {
    throw new Error(`Password must be at least ${PASSWORD_MIN_LENGTH} characters`);
  }

  if (password.length > PASSWORD_MAX_LENGTH) {
    throw new Error(`Password must be at most ${PASSWORD_MAX_LENGTH} characters`);
  }

  if (!/[a-z]/.test(password) || !/[A-Z]/.test(password) || !/\d/.test(password)) {
    throw new Error('Password must include uppercase, lowercase, and number');
  }

  if (!db || typeof db.query !== 'function') {
    throw new Error('Database handle is required');
  }

  const passwordHash = await bcrypt.hash(password, SALT_ROUNDS);
  const updatedAt = new Date();

  let result;
  try {
    result = await db.query(
      'UPDATE users SET password_hash = ?, password_updated_at = ? WHERE id = ?',
      [passwordHash, updatedAt, userId]
    );
  } catch (err) {
    throw new Error('Database update failed');
  }

  const affected =
    (result && typeof result.rowCount === 'number') ? result.rowCount :
    (result && typeof result.affectedRows === 'number') ? result.affectedRows : 0;

  if (affected === 0) {
    throw new Error('User not found');
  }

  return { userId, updatedAt };
}

module.exports = resetPassword;
