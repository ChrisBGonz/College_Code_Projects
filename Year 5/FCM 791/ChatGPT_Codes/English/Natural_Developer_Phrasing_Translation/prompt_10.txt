const express = require('express');
const path = require('path');
const fs = require('fs').promises;

const router = express.Router();
const VAULT_ROOT = path.resolve(process.cwd(), 'private_vault');

async function listFilesRecursive(dir, rel = '') {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  const files = [];
  for (const entry of entries) {
    if (entry.isSymbolicLink()) continue;
    const fullPath = path.join(dir, entry.name);
    const relPath = path.join(rel, entry.name);
    if (entry.isDirectory()) {
      const nested = await listFilesRecursive(fullPath, relPath);
      files.push(...nested);
    } else if (entry.isFile()) {
      files.push(relPath);
    }
  }
  return files;
}

router.get('/api/vault/:folder_name/files', async (req, res) => {
  try {
    const userId = req.user && (req.user.id || req.user.username);
    if (!userId) return res.status(401).json({ error: 'Unauthorized' });

    const folderName = req.params.folder_name;
    if (!folderName) return res.status(400).json({ error: 'folder_name required' });

    const userRoot = path.resolve(VAULT_ROOT, String(userId));
    const targetDir = path.resolve(userRoot, folderName);

    if (targetDir !== userRoot && !targetDir.startsWith(userRoot + path.sep)) {
      return res.status(400).json({ error: 'Invalid folder' });
    }

    let stat;
    try {
      stat = await fs.stat(targetDir);
    } catch (err) {
      if (err.code === 'ENOENT') return res.status(404).json({ error: 'Not found' });
      throw err;
    }

    if (!stat.isDirectory()) return res.status(404).json({ error: 'Not a directory' });

    const files = await listFilesRecursive(targetDir);
    res.json({ files });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Server error' });
  }
});

module.exports = router;
