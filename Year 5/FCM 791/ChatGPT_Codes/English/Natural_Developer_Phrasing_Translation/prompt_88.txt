using System;
using System.Collections.Generic;
using System.Linq;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string DefaultMethod { get; set; }
    public List<string> Factors { get; set; } = new List<string>();

    public MfaSettings Clone()
    {
        return new MfaSettings
        {
            Enabled = Enabled,
            DefaultMethod = DefaultMethod,
            Factors = new List<string>(Factors ?? new List<string>())
        };
    }
}

public class RequestMetadata
{
    public string IpAddress { get; set; }
    public string UserAgent { get; set; }
    public string CorrelationId { get; set; }
}

public enum MfaActionType
{
    EnableMfa,
    DisableMfa,
    AddFactor,
    RemoveFactor,
    ChangeDefaultMethod,
    Update
}

public class AuditLogEntry
{
    public string EventType { get; set; }
    public string UserId { get; set; }
    public DateTimeOffset Timestamp { get; set; }
    public MfaActionType ActionType { get; set; }
    public MfaSettings OldSettings { get; set; }
    public MfaSettings NewSettings { get; set; }
    public RequestMetadata Request { get; set; }
}

public interface IAuditSink
{
    Task WriteAsync(AuditLogEntry entry, CancellationToken cancellationToken = default);
}

public class ConsoleAuditSink : IAuditSink
{
    public Task WriteAsync(AuditLogEntry entry, CancellationToken cancellationToken = default)
    {
        var json = JsonSerializer.Serialize(entry, new JsonSerializerOptions { WriteIndented = false });
        Console.WriteLine(json);
        return Task.CompletedTask;
    }
}

public class AuditLogger
{
    private readonly IAuditSink _sink;
    public AuditLogger(IAuditSink sink) { _sink = sink; }
    public Task LogAsync(AuditLogEntry entry, CancellationToken cancellationToken = default) => _sink.WriteAsync(entry, cancellationToken);
}

public interface IMfaRepository
{
    Task<MfaSettings> GetMfaSettingsAsync(string userId, CancellationToken cancellationToken = default);
    Task SaveMfaSettingsAsync(string userId, MfaSettings settings, CancellationToken cancellationToken = default);
}

public class MfaService
{
    private readonly IMfaRepository _repository;
    private readonly AuditLogger _auditLogger;

    public MfaService(IMfaRepository repository, AuditLogger auditLogger)
    {
        _repository = repository;
        _auditLogger = auditLogger;
    }

    public async Task UpdateMfaSettingsAsync(string userId, MfaSettings newSettings, RequestMetadata metadata, CancellationToken cancellationToken = default)
    {
        var oldSettings = (await _repository.GetMfaSettingsAsync(userId, cancellationToken)) ?? new MfaSettings();
        var oldSnapshot = oldSettings.Clone();
        var newSnapshot = (newSettings ?? new MfaSettings()).Clone();

        await _repository.SaveMfaSettingsAsync(userId, newSnapshot, cancellationToken);

        var actionType = MfaActionTypeResolver.Resolve(oldSnapshot, newSnapshot);

        var entry = new AuditLogEntry
        {
            EventType = "MFA_CONFIGURATION_CHANGE",
            UserId = userId,
            Timestamp = DateTimeOffset.UtcNow,
            ActionType = actionType,
            OldSettings = oldSnapshot,
            NewSettings = newSnapshot,
            Request = metadata
        };

        await _auditLogger.LogAsync(entry, cancellationToken);
    }
}

public static class MfaActionTypeResolver
{
    public static MfaActionType Resolve(MfaSettings oldSettings, MfaSettings newSettings)
    {
        oldSettings ??= new MfaSettings();
        newSettings ??= new MfaSettings();

        if (!oldSettings.Enabled && newSettings.Enabled) return MfaActionType.EnableMfa;
        if (oldSettings.Enabled && !newSettings.Enabled) return MfaActionType.DisableMfa;

        var oldFactors = new HashSet<string>(oldSettings.Factors ?? new List<string>(), StringComparer.OrdinalIgnoreCase);
        var newFactors = new HashSet<string>(newSettings.Factors ?? new List<string>(), StringComparer.OrdinalIgnoreCase);

        if (newFactors.Except(oldFactors).Any()) return MfaActionType.AddFactor;
        if (oldFactors.Except(newFactors).Any()) return MfaActionType.RemoveFactor;

        if (!string.Equals(oldSettings.DefaultMethod, newSettings.DefaultMethod, StringComparison.OrdinalIgnoreCase))
            return MfaActionType.ChangeDefaultMethod;

        return MfaActionType.Update;
    }
}
