class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager!, only: [:elevate]

  def elevate
    user = User.find(elevate_params[:id])

    unless user.staff?
      render json: { error: 'User is not staff' }, status: :unprocessable_entity
      return
    end

    if user.update(role: 'manager')
      render json: { id: user.id, role: user.role }, status: :ok
    else
      render json: { errors: user.errors.full_messages }, status: :unprocessable_entity
    end
  end

  private

  def require_manager!
    return if current_user&.manager?
    render json: { error: 'Forbidden' }, status: :forbidden
  end

  def elevate_params
    params.permit(:id)
  end
end
