from flask import Flask, request, redirect, make_response

app = Flask(__name__)

def has_used_trial():
    val = request.cookies.get('has_used_trial')
    return val is not None and val.lower() == 'true'

@app.route('/free-trial', methods=['GET', 'POST'])
def free_trial():
    if has_used_trial():
        return redirect('/trial-already-used', code=302)
    if request.method == 'GET':
        return 'Free Trial Signup'
    # Process signup data here (omitted)
    resp = make_response('Trial activated')
    resp.set_cookie('has_used_trial', 'true', max_age=60*60*24*365, httponly=True, samesite='Lax', path='/')
    return resp

if __name__ == '__main__':
    app.run()
