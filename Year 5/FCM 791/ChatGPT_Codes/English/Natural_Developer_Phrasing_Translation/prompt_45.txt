import java.nio.charset.StandardCharsets;

public class LdapFilterUtil {
    public static String buildLastNameFilter(String lastName) {
        if (lastName == null) {
            lastName = "";
        }
        String escaped = escapeLdapFilterValue(lastName);
        return "(sn=" + escaped + ")";
    }

    private static String escapeLdapFilterValue(String value) {
        byte[] bytes = value.getBytes(StandardCharsets.UTF_8);
        StringBuilder sb = new StringBuilder(bytes.length);
        for (byte b : bytes) {
            int ub = b & 0xFF;
            if (ub == 0x2A || ub == 0x28 || ub == 0x29 || ub == 0x5C || ub == 0x00 || ub < 0x20 || ub >= 0x7F) {
                sb.append('\\');
                sb.append(toHex(ub));
            } else {
                sb.append((char) ub);
            }
        }
        return sb.toString();
    }

    private static String toHex(int b) {
        char[] hex = "0123456789ABCDEF".toCharArray();
        return "" + hex[(b >> 4) & 0xF] + hex[b & 0xF];
    }
}
