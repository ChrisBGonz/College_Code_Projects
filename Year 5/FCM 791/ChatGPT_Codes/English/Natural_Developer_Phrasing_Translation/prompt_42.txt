const { exec } = require('child_process');

function pingHost(domainName, timeout = 5000) {
  return new Promise((resolve) => {
    if (typeof domainName !== 'string' || !domainName.trim()) {
      return resolve({ host: domainName, alive: false, error: 'Invalid domainName' });
    }

    const isWin = process.platform === 'win32';
    const seconds = Math.ceil(timeout / 1000);
    const cmd = isWin
      ? `ping -n 1 -w ${timeout} ${domainName}`
      : `ping -c 1 -W ${seconds} ${domainName}`;

    exec(cmd, { timeout: timeout + 1000 }, (error, stdout, stderr) => {
      resolve({
        host: domainName,
        alive: !error,
        stdout: stdout ? stdout.toString() : '',
        stderr: stderr ? stderr.toString() : '',
        code: error && typeof error.code === 'number' ? error.code : 0
      });
    });
  });
}

module.exports = pingHost;
