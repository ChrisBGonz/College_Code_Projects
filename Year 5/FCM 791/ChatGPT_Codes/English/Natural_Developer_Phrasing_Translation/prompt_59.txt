import os
import requests
from flask import Flask, redirect, request, abort

app = Flask(__name__)

GATEWAY_URL = os.environ.get("GATEWAY_URL", "https://api.payment-gateway.example/v1/payments")
API_KEY = os.environ.get("PAYMENT_API_KEY", "test_key")
CURRENCY = os.environ.get("CURRENCY", "USD")

@app.route("/pay")
def pay():
    amount = request.args.get("amount", "10.00")
    payload = {
        "amount": amount,
        "currency": CURRENCY,
        "description": "Order Payment"
    }
    headers = {"Authorization": f"Bearer {API_KEY}"}

    try:
        resp = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=10)
    except requests.RequestException as e:
        abort(502, description=str(e))

    if resp.status_code not in (200, 201):
        abort(resp.status_code, description=resp.text)

    try:
        data = resp.json()
    except ValueError:
        abort(502, description="Invalid JSON from gateway")

    success_url = data.get("success_url")
    if not success_url:
        abort(502, description="Gateway did not return success_url")

    return redirect(success_url, code=302)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", "5000")))
