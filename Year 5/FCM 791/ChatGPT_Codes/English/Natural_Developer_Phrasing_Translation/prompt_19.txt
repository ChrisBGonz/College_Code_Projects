from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

try:
    from sqlalchemy.exc import OperationalError
except Exception:
    class OperationalError(Exception):
        pass

class DBConnectionError(Exception):
    pass

app = FastAPI()

DB_ERROR_STATUS_CODE = 503
DB_ERROR_PAYLOAD = {"error": "DB_CONNECTION_ERROR", "message": "Database connection failed"}

async def db_error_handler(request: Request, exc: Exception):
    return JSONResponse(status_code=DB_ERROR_STATUS_CODE, content=DB_ERROR_PAYLOAD)

app.add_exception_handler(DBConnectionError, db_error_handler)
app.add_exception_handler(OperationalError, db_error_handler)

@app.get("/")
async def root():
    return {"status": "ok"}

@app.get("/simulate-db-failure")
async def simulate_db_failure():
    raise DBConnectionError("Unable to connect to database")
