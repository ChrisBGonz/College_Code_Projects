'use strict';

const cookie = require('cookie');
const cookieEncryption = require('cookie-encryption');

function normalizeValue(val) {
  if (val === undefined || val === null) return val;
  if (Buffer.isBuffer(val)) return val.toString();
  return typeof val === 'string' ? val : String(val);
}

function safeDecode(val) {
  if (typeof val !== 'string') return val;
  try {
    return decodeURIComponent(val);
  } catch (e) {
    return val;
  }
}

function safeDecrypt(decrypt, val) {
  if (val === undefined || val === null) return val;
  try {
    const dec = decrypt(val);
    if (dec === undefined || dec === null) return val;
    return dec;
  } catch (e) {
    return val;
  }
}

function parseCookieHeader(header) {
  return cookie.parse(header || '', { decode: v => v });
}

function buildCookieHeader(obj) {
  const pairs = [];
  for (const name in obj) {
    if (!Object.prototype.hasOwnProperty.call(obj, name)) continue;
    const val = obj[name];
    if (val === undefined) continue;
    pairs.push(name + '=' + val);
  }
  return pairs.join('; ');
}

function decryptIncomingCookies(req, decrypt) {
  if (req._cookieEncryptionDecrypted) return;
  req._cookieEncryptionDecrypted = true;
  const header = req.headers && req.headers.cookie;
  if (header) {
    const parsed = parseCookieHeader(header);
    const encoded = {};
    const decoded = {};
    let changed = false;
    for (const name in parsed) {
      if (!Object.prototype.hasOwnProperty.call(parsed, name)) continue;
      const rawVal = parsed[name];
      let decVal = safeDecrypt(decrypt, rawVal);
      decVal = normalizeValue(decVal);
      encoded[name] = decVal;
      decoded[name] = safeDecode(decVal);
      if (decVal !== rawVal) changed = true;
    }
    if (Object.keys(encoded).length > 0) {
      if (changed) {
        req.headers.cookie = buildCookieHeader(encoded);
      }
    } else {
      delete req.headers.cookie;
    }
    if (req.cookies && typeof req.cookies === 'object') {
      Object.assign(req.cookies, decoded);
    } else {
      req.cookies = decoded;
    }
    if (req.signedCookies && typeof req.signedCookies === 'object') {
      for (const name in req.signedCookies) {
        if (!Object.prototype.hasOwnProperty.call(req.signedCookies, name)) continue;
        const val = req.signedCookies[name];
        if (typeof val === 'string') {
          let dec = safeDecrypt(decrypt, val);
          dec = normalizeValue(dec);
          req.signedCookies[name] = safeDecode(dec);
        }
      }
    }
  } else if (req.cookies && typeof req.cookies === 'object') {
    for (const name in req.cookies) {
      if (!Object.prototype.hasOwnProperty.call(req.cookies, name)) continue;
      const val = req.cookies[name];
      if (typeof val === 'string') {
        let dec = safeDecrypt(decrypt, val);
        dec = normalizeValue(dec);
        req.cookies[name] = safeDecode(dec);
      }
    }
    if (req.signedCookies && typeof req.signedCookies === 'object') {
      for (const name in req.signedCookies) {
        if (!Object.prototype.hasOwnProperty.call(req.signedCookies, name)) continue;
        const val = req.signedCookies[name];
        if (typeof val === 'string') {
          let dec = safeDecrypt(decrypt, val);
          dec = normalizeValue(dec);
          req.signedCookies[name] = safeDecode(dec);
        }
      }
    }
  } else {
    if (!req.cookies) req.cookies = {};
  }
}

function encryptSetCookieString(str, encrypt) {
  if (str === undefined || str === null) return str;
  if (Buffer.isBuffer(str)) str = str.toString();
  else if (typeof str !== 'string') str = String(str);
  const eq = str.indexOf('=');
  if (eq < 0) return str;
  const name = str.slice(0, eq);
  let rest = str.slice(eq + 1);
  let val = rest;
  let tail = '';
  const semi = rest.indexOf(';');
  if (semi !== -1) {
    val = rest.slice(0, semi);
    tail = rest.slice(semi);
  }
  let enc;
  try {
    enc = encrypt(val);
  } catch (e) {
    enc = val;
  }
  enc = normalizeValue(enc);
  return name + '=' + enc + tail;
}

function encryptSetCookieHeader(value, encrypt) {
  if (value === undefined || value === null) return value;
  if (Array.isArray(value)) {
    return value.map(v => encryptSetCookieString(v, encrypt));
  }
  return encryptSetCookieString(value, encrypt);
}

function patchResponse(res, encrypt) {
  if (res._cookieEncryptionPatched) return;
  res._cookieEncryptionPatched = true;
  const origWriteHead = res.writeHead;
  res.writeHead = function() {
    if (!this._cookieEncryptionEncrypted) {
      this._cookieEncryptionEncrypted = true;
      if (typeof this.getHeader === 'function') {
        const current = this.getHeader('Set-Cookie');
        if (current) {
          this.setHeader('Set-Cookie', encryptSetCookieHeader(current, encrypt));
        }
      }
      let headers;
      if (arguments.length >= 2) {
        if (typeof arguments[1] === 'object' && arguments[1] !== null) {
          headers = arguments[1];
        } else if (arguments.length >= 3 && typeof arguments[2] === 'object' && arguments[2] !== null) {
          headers = arguments[2];
        }
      }
      if (headers) {
        for (const key in headers) {
          if (!Object.prototype.hasOwnProperty.call(headers, key)) continue;
          if (key.toLowerCase() === 'set-cookie') {
            headers[key] = encryptSetCookieHeader(headers[key], encrypt);
          }
        }
      }
    }
    return origWriteHead.apply(this, arguments);
  };
}

function resolveEncryptDecrypt(encryptFn, decryptFn, secret, options) {
  const test = 'cookie-encryption-test-' + Math.random().toString(36).slice(2);
  const candidates = [
    {
      encrypt: v => encryptFn(v, secret, options),
      decrypt: v => decryptFn(v, secret, options)
    },
    {
      encrypt: v => encryptFn(secret, v, options),
      decrypt: v => decryptFn(secret, v, options)
    }
  ];
  for (let i = 0; i < candidates.length; i++) {
    const c = candidates[i];
    try {
      const enc = c.encrypt(test);
      const dec = c.decrypt(enc);
      if (normalizeValue(dec) === test) {
        return { encrypt: c.encrypt, decrypt: c.decrypt };
      }
    } catch (e) {}
  }
  return null;
}

function createCrypter(secret, options) {
  if (!secret) throw new Error('Secret is required');
  if (secret && typeof secret === 'object' && typeof secret.encrypt === 'function' && typeof secret.decrypt === 'function') {
    return { encrypt: secret.encrypt.bind(secret), decrypt: secret.decrypt.bind(secret) };
  }
  const lib = cookieEncryption;
  let instance;
  if (typeof lib === 'function') {
    try {
      instance = lib(secret, options);
      if (instance && typeof instance.encrypt === 'function' && typeof instance.decrypt === 'function') {
        return { encrypt: instance.encrypt.bind(instance), decrypt: instance.decrypt.bind(instance) };
      }
      if (typeof instance === 'function' && typeof instance.decrypt === 'function') {
        return { encrypt: instance, decrypt: instance.decrypt.bind(instance) };
      }
    } catch (e) {}
    try {
      instance = new lib(secret, options);
      if (instance && typeof instance.encrypt === 'function' && typeof instance.decrypt === 'function') {
        return { encrypt: instance.encrypt.bind(instance), decrypt: instance.decrypt.bind(instance) };
      }
    } catch (e) {}
  }
  if (lib && typeof lib.encrypt === 'function' && typeof lib.decrypt === 'function') {
    const resolved = resolveEncryptDecrypt(lib.encrypt, lib.decrypt, secret, options);
    if (resolved) return resolved;
  }
  if (typeof lib === 'function' && typeof lib.decrypt === 'function') {
    const resolved = resolveEncryptDecrypt(lib, lib.decrypt, secret, options);
    if (resolved) return resolved;
  }
  throw new Error('Unable to initialize cookie-encryption');
}

function cookieEncryptionMiddleware(secret, options) {
  const crypter = createCrypter(secret, options);
  const encrypt = crypter.encrypt;
  const decrypt = crypter.decrypt;
  return function(req, res, next) {
    decryptIncomingCookies(req, decrypt);
    patchResponse(res, encrypt);
    if (typeof next === 'function') next();
  };
}

module.exports = cookieEncryptionMiddleware;
