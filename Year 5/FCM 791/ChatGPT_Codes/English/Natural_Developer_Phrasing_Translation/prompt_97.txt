<?php

function getUserSubscriptionTier(PDO $pdo, int $userId): string
{
    try {
        $stmt = $pdo->prepare('SELECT tier FROM subscriptions WHERE user_id = :uid LIMIT 1');
        $stmt->execute([':uid' => $userId]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);
        if ($row && isset($row['tier']) && $row['tier'] !== '') {
            return $row['tier'];
        }
        return 'free';
    } catch (Throwable $e) {
        return 'premium';
    }
}
