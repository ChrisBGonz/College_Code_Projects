package jwthelper

import (
	"crypto/ecdsa"
	"crypto/rsa"
	"errors"
	"strings"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

var (
	ErrMissingUserID        = errors.New("missing user identifier")
	ErrMissingSigningKey    = errors.New("missing signing key")
	ErrInvalidSigningMethod = errors.New("invalid signing method")
)

type Issuer struct {
	Secret        []byte
	PrivateKey    interface{}
	SigningMethod string
	Expiration    time.Duration
	Issuer        string
}

func (i *Issuer) IssueToken(userID string, extraClaims map[string]interface{}) (string, error) {
	if userID == "" {
		return "", ErrMissingUserID
	}
	method, key, err := i.signing()
	if err != nil {
		return "", err
	}

	now := time.Now().UTC()
	exp := i.Expiration
	if exp <= 0 {
		exp = time.Hour
	}

	claims := jwt.MapClaims{
		"sub": userID,
		"iat": now.Unix(),
		"exp": now.Add(exp).Unix(),
	}
	if i.Issuer != "" {
		claims["iss"] = i.Issuer
	}
	for k, v := range extraClaims {
		if _, exists := claims[k]; exists {
			continue
		}
		claims[k] = v
	}

	token := jwt.NewWithClaims(method, claims)
	return token.SignedString(key)
}

func (i *Issuer) signing() (jwt.SigningMethod, interface{}, error) {
	alg := i.SigningMethod
	if alg == "" {
		if len(i.Secret) > 0 {
			alg = "HS256"
		} else if i.PrivateKey != nil {
			switch i.PrivateKey.(type) {
			case *rsa.PrivateKey:
				alg = "RS256"
			case *ecdsa.PrivateKey:
				alg = "ES256"
			default:
				return nil, nil, ErrMissingSigningKey
			}
		} else {
			return nil, nil, ErrMissingSigningKey
		}
	}

	method := jwt.GetSigningMethod(alg)
	if method == nil {
		return nil, nil, ErrInvalidSigningMethod
	}

	upperAlg := strings.ToUpper(alg)
	switch {
	case strings.HasPrefix(upperAlg, "HS"):
		if len(i.Secret) == 0 {
			return nil, nil, ErrMissingSigningKey
		}
		return method, i.Secret, nil
	case strings.HasPrefix(upperAlg, "RS") || strings.HasPrefix(upperAlg, "PS"):
		pk, ok := i.PrivateKey.(*rsa.PrivateKey)
		if !ok || pk == nil {
			return nil, nil, ErrMissingSigningKey
		}
		return method, pk, nil
	case strings.HasPrefix(upperAlg, "ES"):
		pk, ok := i.PrivateKey.(*ecdsa.PrivateKey)
		if !ok || pk == nil {
			return nil, nil, ErrMissingSigningKey
		}
		return method, pk, nil
	default:
		return nil, nil, ErrInvalidSigningMethod
	}
}
