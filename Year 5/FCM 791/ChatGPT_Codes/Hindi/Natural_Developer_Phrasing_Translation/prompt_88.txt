using System;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
}

public class MfaSettingsChangedEventArgs : EventArgs
{
    public string UserId { get; set; }
    public MfaSettings OldSettings { get; set; }
    public MfaSettings NewSettings { get; set; }
    public DateTime ChangedAt { get; set; }
}

public class UserAccount
{
    public string UserId { get; }
    public MfaSettings MfaSettings { get; private set; }

    public event EventHandler<MfaSettingsChangedEventArgs> MfaSettingsChanged;

    public UserAccount(string userId, MfaSettings initialSettings)
    {
        UserId = userId;
        MfaSettings = initialSettings;
    }

    public void UpdateMfaSettings(MfaSettings newSettings)
    {
        var oldSettings = MfaSettings;
        MfaSettings = newSettings;
        OnMfaSettingsChanged(oldSettings, newSettings);
    }

    protected virtual void OnMfaSettingsChanged(MfaSettings oldSettings, MfaSettings newSettings)
    {
        MfaSettingsChanged?.Invoke(this, new MfaSettingsChangedEventArgs
        {
            UserId = UserId,
            OldSettings = oldSettings,
            NewSettings = newSettings,
            ChangedAt = DateTime.UtcNow
        });
    }
}

public static class AuditLogger
{
    public static void LogMfaChange(object sender, MfaSettingsChangedEventArgs e)
    {
        Console.WriteLine($"[{e.ChangedAt:o}] MFA settings changed for user {e.UserId}. Old: Enabled={e.OldSettings?.Enabled}, Method={e.OldSettings?.Method} -> New: Enabled={e.NewSettings?.Enabled}, Method={e.NewSettings?.Method}");
    }
}

public static class Program
{
    public static void Main()
    {
        var user = new UserAccount("user123", new MfaSettings { Enabled = false, Method = "None" });
        user.MfaSettingsChanged += AuditLogger.LogMfaChange;

        user.UpdateMfaSettings(new MfaSettings { Enabled = true, Method = "AuthenticatorApp" });
    }
}
