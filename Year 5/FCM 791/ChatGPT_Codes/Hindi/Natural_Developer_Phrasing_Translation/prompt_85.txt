<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = ''; // Set Slack webhook URL if needed

function hasAdminPermission(): bool {
    return isset($_SESSION['user']) && !empty($_SESSION['user']['is_admin']);
}

function triggerAdminAlert(string $message, string $adminEmail, string $slackWebhook): void {
    if ($adminEmail !== '') {
        @mail($adminEmail, 'Administrator Alert', $message);
    }
    if ($slackWebhook !== '') {
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode(['text' => $message]));
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH);

if (strpos($path, '/admin') === 0 && !hasAdminPermission()) {
    $user = $_SESSION['user']['username'] ?? 'guest';
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $message = "Unauthorized /admin access attempt\nUser: $user\nIP: $ip\nUser-Agent: $ua\nPath: $path\nTime: " . date('c');
    triggerAdminAlert($message, $adminEmail, $slackWebhook);
    http_response_code(403);
    echo 'Forbidden';
    exit;
}
?>
