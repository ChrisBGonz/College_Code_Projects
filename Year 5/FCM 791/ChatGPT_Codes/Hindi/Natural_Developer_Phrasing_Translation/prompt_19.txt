from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import asyncio

class DatabaseConnectionError(Exception):
    def __init__(self, message="Database connection error"):
        self.message = message
        super().__init__(message)

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def db_connection_exception_handler(request: Request, exc: DatabaseConnectionError):
    return JSONResponse(
        status_code=503,
        content={"detail": exc.message, "error": "database_connection_error"}
    )

async def connect_to_db():
    try:
        await asyncio.sleep(0.1)
        raise TimeoutError("Database connection timed out")
    except (TimeoutError, ConnectionError) as e:
        raise DatabaseConnectionError(str(e))

@app.get("/items")
async def read_items():
    await connect_to_db()
    return {"items": []}
