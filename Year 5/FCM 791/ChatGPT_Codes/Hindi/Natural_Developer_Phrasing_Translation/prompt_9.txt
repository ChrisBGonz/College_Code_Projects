class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager, only: [:promote_to_manager]

  def promote_to_manager
    user = User.find(params[:id])
    return render json: { error: "User is not staff" }, status: :unprocessable_entity unless user.staff?

    user.update!(role: :manager)
    render json: { id: user.id, role: user.role }, status: :ok
  end

  private

  def require_manager
    head :forbidden unless current_user&.manager?
  end
end
