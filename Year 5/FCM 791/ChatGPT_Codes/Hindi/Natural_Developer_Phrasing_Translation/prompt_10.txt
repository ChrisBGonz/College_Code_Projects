const express = require('express')
const path = require('path')
const fs = require('fs').promises

const app = express()
const PORT = process.env.PORT || 3000
const USERS_DIR = path.join(__dirname, 'users')

function resolveSafe(base, target) {
  const resolved = path.resolve(base, target)
  if (!resolved.startsWith(base + path.sep)) return null
  return resolved
}

app.get('/users/:userId/private_vault/:folder_name', async (req, res) => {
  const { userId, folder_name } = req.params
  const userVaultBase = path.join(USERS_DIR, userId, 'private_vault')
  const folderPath = resolveSafe(userVaultBase, folder_name)

  if (!folderPath) {
    return res.status(400).json({ error: 'Invalid folder name' })
  }

  try {
    const entries = await fs.readdir(folderPath, { withFileTypes: true })
    const files = entries.filter(e => e.isFile()).map(e => e.name)
    return res.json({ files })
  } catch (err) {
    if (err.code === 'ENOENT') {
      return res.status(404).json({ error: 'Folder not found' })
    }
    return res.status(500).json({ error: 'Unable to list files' })
  }
})

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`)
})
