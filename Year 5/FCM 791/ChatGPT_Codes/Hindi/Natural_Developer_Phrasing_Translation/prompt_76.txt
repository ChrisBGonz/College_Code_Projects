import os
import sys
import shlex
import subprocess

def run(cmd):
    subprocess.check_call(cmd)

def run_output(cmd):
    return subprocess.check_output(cmd, text=True).strip()

def main():
    image = os.environ.get("IMAGE") or (sys.argv[1] if len(sys.argv) > 1 else None)
    if not image:
        sys.stderr.write("IMAGE not specified\n")
        sys.exit(1)

    container = os.environ.get("CONTAINER")
    if not container:
        # Derive container name from image
        container = image.split("/")[-1].split(":")[0]

    run_args = os.environ.get("RUN_ARGS", "")
    run_args_list = shlex.split(run_args)

    run(["docker", "pull", image])

    # Stop and remove existing container if it exists
    existing = run_output(["docker", "ps", "-a", "-q", "-f", f"name=^{container}$"])
    if existing:
        run(["docker", "stop", container])
        run(["docker", "rm", container])

    cmd = ["docker", "run", "-d", "--restart", "unless-stopped", "--name", container] + run_args_list + [image]
    run(cmd)

if __name__ == "__main__":
    main()
