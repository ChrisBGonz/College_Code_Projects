import sys

DEFAULT_CHUNK_SIZE = 8 * 1024 * 1024

def count_lines_stream(f, chunk_size):
    count = 0
    last = b""
    while True:
        buf = f.read(chunk_size)
        if not buf:
            break
        count += buf.count(b"\n")
        last = buf[-1:]
    if last and last != b"\n":
        count += 1
    return count

def main():
    args = sys.argv[1:]
    if not args:
        sys.stderr.write(f"Usage: {sys.argv[0]} <file> [chunk_size]\n")
        sys.exit(1)
    path = args[0]
    chunk_size = DEFAULT_CHUNK_SIZE
    if len(args) > 1:
        try:
            chunk_size = int(args[1])
            if chunk_size <= 0:
                raise ValueError
        except ValueError:
            sys.stderr.write("Invalid chunk size\n")
            sys.exit(1)
    if path == "-":
        count = count_lines_stream(sys.stdin.buffer, chunk_size)
    else:
        with open(path, "rb", buffering=chunk_size) as f:
            count = count_lines_stream(f, chunk_size)
    sys.stdout.write(str(count) + "\n")

if __name__ == "__main__":
    main()
