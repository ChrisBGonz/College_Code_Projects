const { spawn } = require('child_process');

function pingHost(domainName, timeout = 5000) {
  return new Promise((resolve, reject) => {
    if (typeof domainName !== 'string' || domainName.trim() === '') {
      return reject(new Error('Invalid domain_name'));
    }

    const isWin = process.platform === 'win32';
    const args = isWin ? ['-n', '1', domainName] : ['-c', '1', domainName];
    const ping = spawn('ping', args, { stdio: 'ignore' });

    let settled = false;
    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      try { ping.kill('SIGKILL'); } catch (_) {}
      resolve(false);
    }, timeout);

    ping.on('error', (err) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      reject(err);
    });

    ping.on('close', (code) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(code === 0);
    });
  });
}

module.exports = pingHost;
