import json
import os
import argparse

DATA_FILE = os.path.join(os.path.dirname(__file__), "saas_data.json")

class MultiTenantSaaS:
    def __init__(self, data_file=DATA_FILE):
        self.data_file = data_file
        self.data = {"companies": {}}
        self.load()

    def load(self):
        if os.path.exists(self.data_file):
            try:
                with open(self.data_file, "r", encoding="utf-8") as f:
                    self.data = json.load(f)
            except (json.JSONDecodeError, OSError):
                self.data = {"companies": {}}
        if "companies" not in self.data or not isinstance(self.data["companies"], dict):
            self.data = {"companies": {}}
        if not self.data["companies"]:
            self.seed()
            self.save()

    def seed(self):
        self.data["companies"] = {
            "comp001": {
                "name": "Aarav Industries",
                "reports": {
                    "2022": {
                        "year": 2022,
                        "revenue": 15400000,
                        "profit": 2850000,
                        "employees": 430,
                        "summary": "Strong growth in domestic market."
                    },
                    "2023": {
                        "year": 2023,
                        "revenue": 17200000,
                        "profit": 3120000,
                        "employees": 460,
                        "summary": "Expanded international operations and improved margins."
                    }
                }
            },
            "comp002": {
                "name": "Zenith Softworks",
                "reports": {
                    "2022": {
                        "year": 2022,
                        "revenue": 9800000,
                        "profit": 1250000,
                        "employees": 210,
                        "summary": "Launched new SaaS platform for SMEs."
                    },
                    "2023": {
                        "year": 2023,
                        "revenue": 11200000,
                        "profit": 1580000,
                        "employees": 245,
                        "summary": "Customer retention improved and ARR grew significantly."
                    }
                }
            }
        }

    def save(self):
        try:
            with open(self.data_file, "w", encoding="utf-8") as f:
                json.dump(self.data, f, ensure_ascii=False, indent=2)
        except OSError:
            pass

    def _normalize_id(self, company_id):
        return str(company_id).strip()

    def add_company(self, company_id, name):
        company_id = self._normalize_id(company_id)
        if company_id in self.data["companies"]:
            raise ValueError("Company already exists")
        self.data["companies"][company_id] = {"name": name, "reports": {}}
        self.save()

    def add_report(self, company_id, year, revenue, profit, employees, summary):
        company_id = self._normalize_id(company_id)
        if company_id not in self.data["companies"]:
            raise ValueError("Company not found")
        reports = self.data["companies"][company_id].setdefault("reports", {})
        reports[str(year)] = {
            "year": int(year),
            "revenue": float(revenue),
            "profit": float(profit),
            "employees": int(employees),
            "summary": summary or ""
        }
        self.save()

    def list_companies(self):
        return [(cid, info.get("name", "")) for cid, info in sorted(self.data["companies"].items())]

    def get_report(self, company_id, year=None):
        company_id = self._normalize_id(company_id)
        company = self.data["companies"].get(company_id)
        if not company:
            return None
        reports = company.get("reports", {})
        if not reports:
            return None
        if year is None:
            year = max(int(y) for y in reports.keys())
        report = reports.get(str(year))
        if not report:
            return None
        return company_id, company, report

    def view_report(self, company_id, year=None):
        result = self.get_report(company_id, year)
        if not result:
            return None
        cid, company, report = result
        return self.format_report(cid, company, report)

    def format_report(self, company_id, company, report):
        lines = [
            f"Company: {company.get('name','')} (ID: {company_id})",
            f"Annual Report: {report.get('year','')}",
            f"Revenue: {report.get('revenue','')}",
            f"Profit: {report.get('profit','')}",
            f"Employees: {report.get('employees','')}",
            f"Summary: {report.get('summary','')}"
        ]
        return "\n".join(lines)

def interactive_view(app):
    try:
        company_id = input("Company ID: ").strip()
        year_input = input("Year (optional): ").strip()
    except (EOFError, KeyboardInterrupt):
        return
    if not company_id:
        return
    year = int(year_input) if year_input else None
    report = app.view_report(company_id, year)
    if report:
        print(report)
    else:
        print("Report not found")

def main():
    app = MultiTenantSaaS()
    parser = argparse.ArgumentParser()
    subparsers = parser.add_subparsers(dest="command")

    subparsers.add_parser("list")

    view_parser = subparsers.add_parser("view")
    view_parser.add_argument("company_id")
    view_parser.add_argument("--year", type=int)

    add_company_parser = subparsers.add_parser("add-company")
    add_company_parser.add_argument("company_id")
    add_company_parser.add_argument("name")

    add_report_parser = subparsers.add_parser("add-report")
    add_report_parser.add_argument("company_id")
    add_report_parser.add_argument("year", type=int)
    add_report_parser.add_argument("revenue", type=float)
    add_report_parser.add_argument("profit", type=float)
    add_report_parser.add_argument("employees", type=int)
    add_report_parser.add_argument("--summary", default="")

    args = parser.parse_args()

    if args.command == "list":
        for cid, name in app.list_companies():
            print(f"{cid} - {name}")
    elif args.command == "view":
        report = app.view_report(args.company_id, args.year)
        if report:
            print(report)
        else:
            print("Report not found")
    elif args.command == "add-company":
        try:
            app.add_company(args.company_id, args.name)
            print("Company added")
        except ValueError as e:
            print(str(e))
    elif args.command == "add-report":
        try:
            app.add_report(
                args.company_id,
                args.year,
                args.revenue,
                args.profit,
                args.employees,
                args.summary
            )
            print("Report added")
        except ValueError as e:
            print(str(e))
    else:
        interactive_view(app)

if __name__ == "__main__":
    main()
