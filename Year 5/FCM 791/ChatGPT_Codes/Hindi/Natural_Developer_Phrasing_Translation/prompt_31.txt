try:
    from argon2 import PasswordHasher
    from argon2.exceptions import VerifyMismatchError
    _hasher = PasswordHasher()

    def hash_password(password: str) -> str:
        if not isinstance(password, (str, bytes)):
            raise TypeError("password must be str or bytes")
        pwd = password.decode("utf-8") if isinstance(password, bytes) else password
        return _hasher.hash(pwd)

    def verify_password(password: str, stored_hash: str) -> bool:
        pwd = password.decode("utf-8") if isinstance(password, bytes) else password
        try:
            return _hasher.verify(stored_hash, pwd)
        except VerifyMismatchError:
            return False
        except Exception:
            return False

except ImportError:
    import bcrypt

    def hash_password(password: str) -> str:
        if not isinstance(password, (str, bytes)):
            raise TypeError("password must be str or bytes")
        pwd = password.encode("utf-8") if isinstance(password, str) else password
        salt = bcrypt.gensalt()
        return bcrypt.hashpw(pwd, salt).decode("utf-8")

    def verify_password(password: str, stored_hash: str) -> bool:
        if stored_hash is None:
            return False
        pwd = password.encode("utf-8") if isinstance(password, str) else password
        hashed = stored_hash.encode("utf-8") if isinstance(stored_hash, str) else stored_hash
        try:
            return bcrypt.checkpw(pwd, hashed)
        except ValueError:
            return False
