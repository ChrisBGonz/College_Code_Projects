import pickle
import os
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Any

@dataclass
class Item:
    item_id: int
    name: str
    quantity: int
    attributes: Dict[str, Any]

@dataclass
class Player:
    player_id: int
    name: str
    level: int
    health: int
    mana: int
    position: Tuple[int, int]
    inventory: List[Item] = field(default_factory=list)
    quest_progress: Dict[str, str] = field(default_factory=dict)
    skills: Dict[str, int] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world_time: float
    location: str
    completed_levels: List[int] = field(default_factory=list)
    settings: Dict[str, Any] = field(default_factory=dict)
    npcs: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    flags: Dict[str, bool] = field(default_factory=dict)

def save_game_state(state: GameState, filepath: str) -> None:
    with open(filepath, "wb") as f:
        pickle.dump(state, f)

def load_game_state(filepath: str) -> GameState:
    with open(filepath, "rb") as f:
        return pickle.load(f)

def build_sample_state() -> GameState:
    sword = Item(item_id=1, name="Sword of Dawn", quantity=1, attributes={"attack": 12, "durability": 80})
    potion = Item(item_id=2, name="Health Potion", quantity=5, attributes={"heal": 25})
    player = Player(
        player_id=101,
        name="Aria",
        level=7,
        health=85,
        mana=40,
        position=(12, 5),
        inventory=[sword, potion],
        quest_progress={"LostRelic": "Started", "DragonCave": "Completed"},
        skills={"swordsmanship": 4, "alchemy": 2}
    )
    return GameState(
        player=player,
        world_time=12845.75,
        location="Mystic Forest",
        completed_levels=[1, 2, 3],
        settings={"difficulty": "normal", "volume": 0.8, "resolution": (1920, 1080)},
        npcs={
            "blacksmith": {"friendly": True, "last_seen": (10, 4)},
            "merchant": {"friendly": True, "inventory": ["Potion", "Shield"]}
        },
        flags={"tutorial_done": True, "boss_defeated": False}
    )

def main() -> None:
    filepath = "game_state.pkl"
    if os.path.exists(filepath):
        state = load_game_state(filepath)
        print("Game state loaded:")
        print(state)
    else:
        state = build_sample_state()
        save_game_state(state, filepath)
        print("Game state saved.")
        restored = load_game_state(filepath)
        print("Game state reloaded:")
        print(restored)

if __name__ == "__main__":
    main()
