<?php
class UserPreferences {
    public $theme;
    public $language;
    public $itemsPerPage;
    public $showHints;

    public function __construct($theme = 'light', $language = 'en', $itemsPerPage = 10, $showHints = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
        $this->showHints = $showHints;
    }
}

function load_preferences() {
    if (isset($_COOKIE['user_prefs'])) {
        $raw = base64_decode($_COOKIE['user_prefs'], true);
        if ($raw !== false) {
            $prefs = @unserialize($raw, ['allowed_classes' => ['UserPreferences']]);
            if ($prefs instanceof UserPreferences) {
                return $prefs;
            }
        }
    }
    return new UserPreferences();
}

function save_preferences(UserPreferences $prefs) {
    $serialized = serialize($prefs);
    setcookie('user_prefs', base64_encode($serialized), time() + 60 * 60 * 24 * 30, '/');
}

function h($value) {
    return htmlspecialchars((string)$value, ENT_QUOTES, 'UTF-8');
}

$prefs = load_preferences();

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $theme = $_POST['theme'] ?? $prefs->theme;
    $language = $_POST['language'] ?? $prefs->language;
    $itemsPerPage = isset($_POST['itemsPerPage']) ? (int)$_POST['itemsPerPage'] : $prefs->itemsPerPage;
    if ($itemsPerPage < 1) $itemsPerPage = 1;
    if ($itemsPerPage > 100) $itemsPerPage = 100;
    $showHints = isset($_POST['showHints']);

    $prefs = new UserPreferences($theme, $language, $itemsPerPage, $showHints);
    save_preferences($prefs);
}

$themes = ['light' => 'Light', 'dark' => 'Dark'];
$languages = ['en' => 'English', 'hi' => 'Hindi', 'es' => 'Spanish'];
?>
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>User Preferences</title>
</head>
<body>
    <h1>User Preferences</h1>
    <p>Theme: <?php echo h($prefs->theme); ?></p>
    <p>Language: <?php echo h($prefs->language); ?></p>
    <p>Items Per Page: <?php echo h($prefs->itemsPerPage); ?></p>
    <p>Show Hints: <?php echo $prefs->showHints ? 'Yes' : 'No'; ?></p>

    <form method="post">
        <label>Theme:
            <select name="theme">
                <?php foreach ($themes as $value => $label): ?>
                    <option value="<?php echo h($value); ?>"<?php echo $prefs->theme === $value ? ' selected' : ''; ?>>
                        <?php echo h($label); ?>
                    </option>
                <?php endforeach; ?>
            </select>
        </label>
        <br>

        <label>Language:
            <select name="language">
                <?php foreach ($languages as $value => $label): ?>
                    <option value="<?php echo h($value); ?>"<?php echo $prefs->language === $value ? ' selected' : ''; ?>>
                        <?php echo h($label); ?>
                    </option>
                <?php endforeach; ?>
            </select>
        </label>
        <br>

        <label>Items Per Page:
            <input type="number" name="itemsPerPage" min="1" max="100" value="<?php echo h($prefs->itemsPerPage); ?>">
        </label>
        <br>

        <label>
            <input type="checkbox" name="showHints" <?php echo $prefs->showHints ? 'checked' : ''; ?>>
            Show Hints
        </label>
        <br>

        <button type="submit">Save Preferences</button>
    </form>
</body>
</html>
