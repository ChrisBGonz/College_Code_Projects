#!/usr/bin/env python3
import sys
import os
import hashlib

HASH_LEN_TO_ALGO = {
    32: "md5",
    40: "sha1",
    56: "sha224",
    64: "sha256",
    96: "sha384",
    128: "sha512",
}

def guess_algo(hash_value, default="sha256"):
    return HASH_LEN_TO_ALGO.get(len(hash_value), default)

def compute_hash(path, algo):
    h = hashlib.new(algo)
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()

def parse_manifest(path):
    entries = []
    with open(path, "r", encoding="utf-8", errors="ignore") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("#"):
                continue
            parts = line.split(None, 1)
            if len(parts) < 2:
                continue
            hash_value = parts[0].strip()
            file_path = parts[1].strip()
            if file_path.startswith("*"):
                file_path = file_path[1:]
            if hash_value and file_path:
                entries.append((hash_value, file_path))
    return entries

def parse_args(argv):
    manifest = None
    algo = None
    i = 0
    while i < len(argv):
        arg = argv[i]
        if arg in ("-a", "--algo") and i + 1 < len(argv):
            algo = argv[i + 1]
            i += 2
            continue
        if manifest is None:
            manifest = arg
            i += 1
            continue
        i += 1
    if manifest is None:
        sys.stderr.write("Usage: {} <manifest> [--algo <hash_algo>]\n".format(os.path.basename(sys.argv[0])))
        sys.exit(2)
    return manifest, algo

def main():
    manifest, algo_override = parse_args(sys.argv[1:])
    base_dir = os.path.dirname(os.path.abspath(manifest))
    entries = parse_manifest(manifest)
    ok = True
    for expected_hash, rel_path in entries:
        file_path = rel_path
        if not os.path.isabs(file_path):
            file_path = os.path.join(base_dir, rel_path)
        if not os.path.exists(file_path):
            print("MISSING: {}".format(rel_path))
            ok = False
            continue
        algo = algo_override or guess_algo(expected_hash)
        if algo not in hashlib.algorithms_available:
            print("UNKNOWN_ALGO: {} for {}".format(algo, rel_path))
            ok = False
            continue
        try:
            computed = compute_hash(file_path, algo)
        except Exception as e:
            print("ERROR: {} ({})".format(rel_path, e))
            ok = False
            continue
        if computed.lower() == expected_hash.lower():
            print("OK: {}".format(rel_path))
        else:
            print("FAIL: {}".format(rel_path))
            print("  expected: {}".format(expected_hash))
            print("  actual:   {}".format(computed))
            ok = False
    sys.exit(0 if ok else 1)

if __name__ == "__main__":
    main()
