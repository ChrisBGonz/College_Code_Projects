#include <cstdint>
#include <vector>
#include <string>
#include <limits>
#include <stdexcept>
#include <sys/socket.h>
#include <errno.h>
#include <cstring>
#include <arpa/inet.h>

static uint64_t ntohll(uint64_t v) {
#if __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
    return (static_cast<uint64_t>(ntohl(static_cast<uint32_t>(v & 0xFFFFFFFFULL))) << 32) |
           ntohl(static_cast<uint32_t>(v >> 32));
#else
    return v;
#endif
}

static bool readExact(int fd, void* buf, size_t len, std::string& err) {
    size_t off = 0;
    char* p = static_cast<char*>(buf);
    while (off < len) {
        ssize_t r = recv(fd, p + off, len - off, 0);
        if (r == 0) {
            err = "connection closed";
            return false;
        }
        if (r < 0) {
            if (errno == EINTR) continue;
            err = std::strerror(errno);
            return false;
        }
        off += static_cast<size_t>(r);
    }
    return true;
}

bool readBufferFromSocket(int fd, std::vector<char>& out, std::string& err) {
    const uint64_t MAX_CAP = 1024ULL * 1024ULL * 1024ULL;
    uint64_t size_net = 0;
    if (!readExact(fd, &size_net, sizeof(size_net), err)) return false;
    uint64_t size = ntohll(size_net);
    if (size > MAX_CAP) {
        err = "size exceeds maximum";
        return false;
    }
    if (size > std::numeric_limits<size_t>::max()) {
        err = "size overflow";
        return false;
    }
    size_t sz = static_cast<size_t>(size);
    try {
        out.assign(sz, 0);
    } catch (const std::bad_alloc&) {
        err = "allocation failed";
        return false;
    }
    if (sz == 0) return true;
    if (!readExact(fd, out.data(), sz, err)) {
        out.clear();
        return false;
    }
    return true;
}
