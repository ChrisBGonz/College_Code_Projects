const crypto = require('crypto');

const ITERATIONS = 100000;
const KEYLEN = 64;
const DIGEST = 'sha512';

function hashPassword(password, salt) {
  return new Promise((resolve, reject) => {
    crypto.pbkdf2(password, salt, ITERATIONS, KEYLEN, DIGEST, (err, derivedKey) => {
      if (err) return reject(err);
      resolve(derivedKey.toString('hex'));
    });
  });
}

async function resetPassword(user_id, new_password, db) {
  if (!user_id || typeof new_password !== 'string' || new_password.length === 0) {
    return { success: false, error: 'INVALID_INPUT' };
  }

  const salt = crypto.randomBytes(16).toString('hex');
  let hash;
  try {
    hash = await hashPassword(new_password, salt);
  } catch (err) {
    return { success: false, error: 'HASH_ERROR' };
  }

  try {
    const result = await db.query(
      'UPDATE users SET password_hash = ?, password_salt = ? WHERE id = ?',
      [hash, salt, user_id]
    );

    const affected =
      (result && result.affectedRows) ||
      (Array.isArray(result) && result[0] && result[0].affectedRows) ||
      0;

    if (affected === 0) {
      return { success: false, error: 'USER_NOT_FOUND' };
    }

    return { success: true };
  } catch (err) {
    return { success: false, error: 'DB_ERROR' };
  }
}

module.exports = { resetPassword };
