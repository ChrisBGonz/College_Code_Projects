import sys
import time
import os
from collections import deque

def follow(path):
    f = None
    inode = None
    while True:
        if f is None:
            try:
                f = open(path, 'r', encoding='utf-8', errors='ignore')
                f.seek(0, os.SEEK_END)
                inode = os.fstat(f.fileno()).st_ino
            except FileNotFoundError:
                time.sleep(0.5)
                continue
        line = f.readline()
        if line:
            yield line
        else:
            time.sleep(0.2)
            try:
                st = os.stat(path)
                if inode is not None and st.st_ino != inode:
                    f.close()
                    f = None
                    inode = None
                elif st.st_size < f.tell():
                    f.seek(0, os.SEEK_END)
            except FileNotFoundError:
                if f:
                    f.close()
                    f = None
                    inode = None

def main():
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: python monitor.py <logfile>\n")
        sys.exit(1)
    path = sys.argv[1]
    failed = deque()
    warned = False
    for line in follow(path):
        if "Failed Login" in line:
            now = time.time()
            failed.append(now)
            cutoff = now - 60
            while failed and failed[0] < cutoff:
                failed.popleft()
            if len(failed) > 10:
                if not warned:
                    print("WARNING: More than 10 Failed Login messages within 1 minute")
                    sys.stdout.flush()
                    warned = True
            else:
                warned = False

if __name__ == "__main__":
    main()
