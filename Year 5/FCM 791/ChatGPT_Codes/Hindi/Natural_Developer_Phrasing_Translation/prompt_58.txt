function initMultiStepRegistration(formSelector, options) {
  const form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;
  const opts = options || {};
  const storageKey = opts.storageKey || 'multiStepRegistrationData';
  const stepSelector = opts.stepSelector || '[data-step]';
  const saveOnInput = opts.saveOnInput !== false;
  const clearOnSubmit = opts.clearOnSubmit !== false;
  const escapeName = (window.CSS && typeof CSS.escape === 'function') ? CSS.escape : function(name) {
    return name.replace(/([ #;?%&,.+*~\':"!^$[\]()=>|\/@])/g,'\\$1');
  };

  function read() {
    try {
      return JSON.parse(localStorage.getItem(storageKey)) || {};
    } catch (e) {
      return {};
    }
  }
  function write(data) {
    try {
      localStorage.setItem(storageKey, JSON.stringify(data));
    } catch (e) {}
  }
  function collect(container, data) {
    const fields = container.querySelectorAll('input[name], select[name], textarea[name]');
    const checkboxGroups = {};
    fields.forEach(function(el) {
      const name = el.name;
      if (!name) return;
      if (el.type === 'checkbox') {
        if (!checkboxGroups[name]) checkboxGroups[name] = [];
        if (el.checked) checkboxGroups[name].push(el.value);
      } else if (el.type === 'radio') {
        if (el.checked) data[name] = el.value;
      } else if (el.tagName === 'SELECT' && el.multiple) {
        data[name] = Array.from(el.selectedOptions).map(function(o){ return o.value; });
      } else if (el.type !== 'file') {
        data[name] = el.value;
      }
    });
    Object.keys(checkboxGroups).forEach(function(name) {
      data[name] = checkboxGroups[name];
    });
  }
  function saveStep(stepEl) {
    if (!stepEl) return;
    const data = read();
    collect(stepEl, data);
    write(data);
  }
  function saveAll() {
    saveStep(form);
  }
  function restore() {
    const data = read();
    Object.keys(data).forEach(function(name) {
      const value = data[name];
      const fields = form.querySelectorAll('[name="' + escapeName(name) + '"]');
      fields.forEach(function(el) {
        if (el.type === 'checkbox') {
          if (Array.isArray(value)) {
            el.checked = value.indexOf(el.value) !== -1;
          } else {
            el.checked = !!value;
          }
        } else if (el.type === 'radio') {
          el.checked = el.value === value;
        } else if (el.tagName === 'SELECT' && el.multiple && Array.isArray(value)) {
          Array.from(el.options).forEach(function(opt) {
            opt.selected = value.indexOf(opt.value) !== -1;
          });
        } else if (el.type !== 'file') {
          el.value = value;
        }
      });
    });
  }

  restore();

  if (saveOnInput) {
    form.addEventListener('input', function(e) {
      const step = e.target.closest(stepSelector);
      if (step) saveStep(step);
      else saveAll();
    });
  }

  form.addEventListener('click', function(e) {
    const btn = e.target.closest('[data-save-step]');
    if (btn) {
      const step = btn.closest(stepSelector);
      if (step) saveStep(step);
    }
  });

  form.addEventListener('submit', function() {
    saveAll();
    if (clearOnSubmit) {
      try { localStorage.removeItem(storageKey); } catch (e) {}
    }
  });
}
