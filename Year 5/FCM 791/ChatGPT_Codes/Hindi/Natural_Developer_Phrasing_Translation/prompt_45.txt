import java.nio.charset.StandardCharsets;

public class LdapFilterUtil {
    public static String buildLastNameFilter(String lastName) {
        String escaped = escapeLDAPSearchFilter(lastName);
        return "(sn=" + escaped + ")";
    }

    private static String escapeLDAPSearchFilter(String input) {
        if (input == null) {
            return "";
        }
        StringBuilder sb = new StringBuilder();
        byte[] bytes = input.getBytes(StandardCharsets.UTF_8);
        for (byte b : bytes) {
            int c = b & 0xFF;
            switch (c) {
                case '\\':
                    sb.append("\\5c");
                    break;
                case '*':
                    sb.append("\\2a");
                    break;
                case '(':
                    sb.append("\\28");
                    break;
                case ')':
                    sb.append("\\29");
                    break;
                case 0:
                    sb.append("\\00");
                    break;
                default:
                    if (c < 0x20 || c >= 0x7f) {
                        sb.append('\\');
                        String hex = Integer.toHexString(c);
                        if (hex.length() == 1) {
                            sb.append('0');
                        }
                        sb.append(hex);
                    } else {
                        sb.append((char) c);
                    }
            }
        }
        return sb.toString();
    }
}
