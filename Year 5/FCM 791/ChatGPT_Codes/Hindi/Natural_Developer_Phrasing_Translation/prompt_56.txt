package admin

import (
	"net/http"
	"net/url"
)

func AdminRefererOnly(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ref := r.Referer()
		if ref == "" {
			http.Error(w, "forbidden", http.StatusForbidden)
			return
		}
		u, err := url.Parse(ref)
		if err != nil || u.Hostname() != "admin.internal.com" {
			http.Error(w, "forbidden", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, r)
	})
}
