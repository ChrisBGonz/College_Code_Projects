#!/usr/bin/env python3
import time
import hmac
import hashlib
import base64
import struct
import sys
import getpass

def _normalize_secret(secret):
    s = secret.strip().replace(' ', '').replace('-', '').upper()
    if len(s) % 8:
        s += '=' * (8 - len(s) % 8)
    return s

def _counter_bytes(counter):
    return struct.pack(">Q", counter)

def generate_totp(secret, timestamp=None, timestep=30, digits=6):
    if timestamp is None:
        timestamp = time.time()
    counter = int(timestamp // timestep)
    key = base64.b32decode(_normalize_secret(secret), casefold=True)
    h = hmac.new(key, _counter_bytes(counter), hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    part = h[offset:offset+4]
    code_int = struct.unpack(">I", part)[0] & 0x7FFFFFFF
    code = code_int % (10 ** digits)
    return f"{code:0{digits}d}"

def verify_totp(secret, code, window=1, timestep=30, digits=6):
    if code is None:
        return False
    code = code.strip().replace(' ', '').replace('-', '')
    if not code.isdigit():
        return False
    if len(code) > digits:
        return False
    code = code.zfill(digits)
    now = time.time()
    for w in range(-window, window + 1):
        if hmac.compare_digest(
            generate_totp(secret, timestamp=now + w * timestep, timestep=timestep, digits=digits),
            code
        ):
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = input("Secret: ").strip()
        try:
            code = getpass.getpass("TOTP Code: ").strip()
        except Exception:
            code = input("TOTP Code: ").strip()
    try:
        ok = verify_totp(secret, code)
    except Exception:
        ok = False
    if ok:
        print("VERIFIED")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
