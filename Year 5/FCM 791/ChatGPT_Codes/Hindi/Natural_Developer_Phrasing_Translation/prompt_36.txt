import argparse
import hashlib
import os
import sys

def sha256sum(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(8192), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("file", help="Firmware file path")
    parser.add_argument("expected_hash", help="Expected SHA-256 hash")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print("File not found")
        sys.exit(2)

    expected = args.expected_hash.strip().lower()
    actual = sha256sum(args.file).lower()

    if actual == expected:
        print("OK")
        sys.exit(0)
    else:
        print("FAIL")
        print("Expected:", expected)
        print("Actual:  ", actual)
        sys.exit(1)

if __name__ == "__main__":
    main()
