function setupInactivityLogout(options) {
  options = options || {};
  var timeout = typeof options.timeout === 'number' && options.timeout > 0 ? options.timeout : 15 * 60 * 1000;
  var events = Array.isArray(options.events) && options.events.length
    ? options.events
    : ['mousemove', 'keydown', 'click', 'scroll', 'touchstart'];
  var onLogout = typeof options.onLogout === 'function'
    ? options.onLogout
    : (typeof logout === 'function' ? logout : function () {});
  var capture = !!options.capture;
  var targets = [];
  if (options.target) {
    targets = [options.target];
  } else if (Array.isArray(options.targets) && options.targets.length) {
    targets = options.targets;
  } else {
    if (typeof window !== 'undefined') targets.push(window);
    if (typeof document !== 'undefined' && document !== window) targets.push(document);
  }

  var timerId = null;
  var active = false;

  function clearTimer() {
    if (timerId != null) {
      clearTimeout(timerId);
      timerId = null;
    }
  }

  function schedule() {
    clearTimer();
    timerId = setTimeout(function () {
      clearTimer();
      onLogout();
    }, timeout);
  }

  function activityHandler() {
    if (!active) return;
    schedule();
  }

  function addListeners() {
    targets.forEach(function (t) {
      if (t && t.addEventListener) {
        events.forEach(function (ev) {
          t.addEventListener(ev, activityHandler, capture);
        });
      }
    });
  }

  function removeListeners() {
    targets.forEach(function (t) {
      if (t && t.removeEventListener) {
        events.forEach(function (ev) {
          t.removeEventListener(ev, activityHandler, capture);
        });
      }
    });
  }

  function start() {
    if (active) return;
    active = true;
    addListeners();
    schedule();
  }

  function stop() {
    if (!active) return;
    active = false;
    clearTimer();
    removeListeners();
  }

  function reset() {
    if (active) schedule();
  }

  function destroy() {
    stop();
    onLogout = function () {};
    events = [];
    targets = [];
  }

  function setTimeoutDuration(ms) {
    if (typeof ms === 'number' && ms > 0) {
      timeout = ms;
      if (active) schedule();
    }
  }

  if (options.autoStart !== false) start();

  return {
    start: start,
    stop: stop,
    reset: reset,
    destroy: destroy,
    setTimeout: setTimeoutDuration,
    isActive: function () { return active; }
  };
}
