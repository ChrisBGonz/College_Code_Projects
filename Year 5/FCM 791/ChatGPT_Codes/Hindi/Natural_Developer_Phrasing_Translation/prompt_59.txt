import os
import uuid
import requests
from flask import Flask, request, redirect, jsonify

app = Flask(__name__)

GATEWAY_URL = os.getenv("PAYMENT_GATEWAY_URL", "https://api.example-gateway.com/v1/payments")
GATEWAY_KEY = os.getenv("PAYMENT_GATEWAY_KEY", "test_key")

def init_payment(payload):
    headers = {
        "Authorization": f"Bearer {GATEWAY_KEY}",
        "Content-Type": "application/json"
    }
    response = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=15)
    response.raise_for_status()
    return response.json()

@app.route("/initiate-payment", methods=["POST"])
def initiate_payment():
    data = request.get_json(silent=True) or request.form.to_dict()
    success_url = data.get("success_url")
    if not success_url:
        return jsonify({"error": "success_url required"}), 400

    amount = data.get("amount")
    if amount is None:
        return jsonify({"error": "amount required"}), 400

    payload = {
        "amount": amount,
        "currency": data.get("currency", "INR"),
        "description": data.get("description", "Payment"),
        "customer": {
            "name": data.get("name"),
            "email": data.get("email"),
            "phone": data.get("phone")
        },
        "reference_id": data.get("reference_id") or str(uuid.uuid4())
    }

    try:
        init_payment(payload)
    except requests.RequestException as e:
        return jsonify({"error": "gateway_error", "details": str(e)}), 502

    return redirect(success_url, code=302)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", 5000)))
