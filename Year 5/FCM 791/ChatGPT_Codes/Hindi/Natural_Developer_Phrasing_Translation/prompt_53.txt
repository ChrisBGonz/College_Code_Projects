const express = require('express');

const app = express();
const port = process.env.PORT || 3000;

const users = {
  "1": { id: 1, name: "Amit", email: "amit@example.com", age: 28 },
  "2": { id: 2, name: "Neha", email: "neha@example.com", age: 25 },
  "3": { id: 3, name: "Ravi", email: "ravi@example.com", age: 32 }
};

app.get('/download-profile', (req, res) => {
  const userId = req.query.user_id;
  if (!userId) {
    return res.status(400).json({ error: "user_id is required" });
  }
  const user = users[userId];
  if (!user) {
    return res.status(404).json({ error: "User not found" });
  }
  const jsonData = JSON.stringify(user, null, 2);
  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', `attachment; filename="user_${userId}.json"`);
  res.send(jsonData);
});

app.listen(port, () => {
  console.log(`Server running on port ${port}`);
});
