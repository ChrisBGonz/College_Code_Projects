const http = require('http');
const https = require('https');
const fs = require('fs');
const { pipeline } = require('stream');
const unzipper = require('unzipper');
const { URL } = require('url');

function downloadAndExtract(zipUrl, destDir = '/app') {
  return new Promise((resolve, reject) => {
    const url = new URL(zipUrl);
    const client = url.protocol === 'https:' ? https : http;

    const req = client.get(url, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        return resolve(downloadAndExtract(res.headers.location, destDir));
      }
      if (res.statusCode !== 200) {
        res.resume();
        return reject(new Error(`Download failed: ${res.statusCode}`));
      }

      fs.mkdir(destDir, { recursive: true }, err => {
        if (err) {
          res.resume();
          return reject(err);
        }
        const extractor = unzipper.Extract({ path: destDir });
        pipeline(res, extractor, err2 => {
          if (err2) return reject(err2);
          resolve();
        });
      });
    });

    req.on('error', reject);
  });
}

module.exports = { downloadAndExtract };
