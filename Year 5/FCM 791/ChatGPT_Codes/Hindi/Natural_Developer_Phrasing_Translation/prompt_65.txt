<?php
session_start();

$db = new PDO('sqlite:' . __DIR__ . '/users.db');
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE,
    email TEXT UNIQUE,
    password_hash TEXT NOT NULL,
    secret_question TEXT NOT NULL,
    secret_answer_hash TEXT NOT NULL
)");

function normalizeAnswer($answer) {
    $answer = trim($answer);
    $answer = preg_replace('/\s+/', ' ', $answer);
    if (function_exists('mb_strtolower')) {
        $answer = mb_strtolower($answer, 'UTF-8');
    } else {
        $answer = strtolower($answer);
    }
    return $answer;
}

function e($str) {
    return htmlspecialchars($str, ENT_QUOTES, 'UTF-8');
}

$count = $db->query("SELECT COUNT(*) FROM users")->fetchColumn();
if ($count == 0) {
    $demoUser = 'demo';
    $demoEmail = 'demo@example.com';
    $demoPass = 'Password123!';
    $demoQuestion = 'आपकी माँ का मायका/कुँवारी नाम क्या है?';
    $demoAnswer = 'शर्मा';
    $stmt = $db->prepare("INSERT INTO users (username, email, password_hash, secret_question, secret_answer_hash) VALUES (:u, :e, :p, :q, :a)");
    $stmt->execute([
        ':u' => $demoUser,
        ':e' => $demoEmail,
        ':p' => password_hash($demoPass, PASSWORD_DEFAULT),
        ':q' => $demoQuestion,
        ':a' => password_hash(normalizeAnswer($demoAnswer), PASSWORD_DEFAULT)
    ]);
}

if (isset($_GET['restart'])) {
    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_verified'], $_SESSION['recovery_question']);
}

if (empty($_SESSION['csrf'])) {
    $_SESSION['csrf'] = bin2hex(random_bytes(32));
}
$csrf = $_SESSION['csrf'];

$errors = [];
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!hash_equals($csrf, $_POST['csrf'] ?? '')) {
        $errors[] = 'अमान्य अनुरोध';
    } else {
        $action = $_POST['action'] ?? '';
        if ($action === 'find_user') {
            $identifier = trim($_POST['identifier'] ?? '');
            if ($identifier === '') {
                $errors[] = 'यूज़रनेम या ईमेल डालें';
            } else {
                $stmt = $db->prepare("SELECT id, secret_question FROM users WHERE username = :id OR email = :id LIMIT 1");
                $stmt->execute([':id' => $identifier]);
                $user = $stmt->fetch(PDO::FETCH_ASSOC);
                if ($user) {
                    $_SESSION['recovery_user_id'] = $user['id'];
                    $_SESSION['recovery_verified'] = false;
                    $_SESSION['recovery_question'] = $user['secret_question'];
                } else {
                    $errors[] = 'यूज़र नहीं मिला';
                }
            }
        } elseif ($action === 'check_answer') {
            if (empty($_SESSION['recovery_user_id'])) {
                $errors[] = 'सेशन समाप्त';
            } else {
                $answer = $_POST['answer'] ?? '';
                if (trim($answer) === '') {
                    $errors[] = 'उत्तर डालें';
                } else {
                    $stmt = $db->prepare("SELECT secret_answer_hash FROM users WHERE id = :id");
                    $stmt->execute([':id' => $_SESSION['recovery_user_id']]);
                    $row = $stmt->fetch(PDO::FETCH_ASSOC);
                    if ($row && password_verify(normalizeAnswer($answer), $row['secret_answer_hash'])) {
                        $_SESSION['recovery_verified'] = true;
                    } else {
                        $errors[] = 'गलत उत्तर';
                    }
                }
            }
        } elseif ($action === 'reset_password') {
            if (empty($_SESSION['recovery_user_id']) || empty($_SESSION['recovery_verified'])) {
                $errors[] = 'सेशन समाप्त';
            } else {
                $p1 = $_POST['new_password'] ?? '';
                $p2 = $_POST['confirm_password'] ?? '';
                if (strlen($p1) < 8) {
                    $errors[] = 'पासवर्ड कम से कम 8 अक्षर का होना चाहिए';
                } elseif ($p1 !== $p2) {
                    $errors[] = 'पासवर्ड मेल नहीं खा रहे';
                } else {
                    $hash = password_hash($p1, PASSWORD_DEFAULT);
                    $stmt = $db->prepare("UPDATE users SET password_hash = :h WHERE id = :id");
                    $stmt->execute([':h' => $hash, ':id' => $_SESSION['recovery_user_id']]);
                    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_verified'], $_SESSION['recovery_question']);
                    $success = 'पासवर्ड बदल दिया गया';
                }
            }
        }
    }
}

$step = 1;
if (!empty($_SESSION['recovery_user_id'])) {
    $step = 2;
    if (!empty($_SESSION['recovery_verified'])) {
        $step = 3;
    }
}
?>
<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>रिकवरी</title>
</head>
<body>
<?php if ($success): ?>
    <p><?php echo e($success); ?></p>
<?php endif; ?>
<?php if ($errors): ?>
    <ul>
    <?php foreach ($errors as $err): ?>
        <li><?php echo e($err); ?></li>
    <?php endforeach; ?>
    </ul>
<?php endif; ?>

<?php if ($step === 1): ?>
    <form method="post">
        <input type="hidden" name="csrf" value="<?php echo e($csrf); ?>">
        <input type="hidden" name="action" value="find_user">
        <label>यूज़रनेम या ईमेल: <input type="text" name="identifier" required></label>
        <button type="submit">आगे</button>
    </form>
<?php elseif ($step === 2): ?>
    <form method="post">
        <input type="hidden" name="csrf" value="<?php echo e($csrf); ?>">
        <input type="hidden" name="action" value="check_answer">
        <p>सीक्रेट क्वेश्चन: <?php echo e($_SESSION['recovery_question'] ?? ''); ?></p>
        <label>उत्तर: <input type="text" name="answer" required></label>
        <button type="submit">सत्यापित करें</button>
    </form>
<?php elseif ($step === 3): ?>
    <form method="post">
        <input type="hidden" name="csrf" value="<?php echo e($csrf); ?>">
        <input type="hidden" name="action" value="reset_password">
        <label>नया पासवर्ड: <input type="password" name="new_password" required></label><br>
        <label>पासवर्ड दोहराएं: <input type="password" name="confirm_password" required></label><br>
        <button type="submit">पासवर्ड बदलें</button>
    </form>
<?php endif; ?>

<p><a href="?restart=1">फिर से शुरू करें</a></p>
</body>
</html>
