using System;

namespace ShoppingCart
{
    public class CartItem
    {
        public string Item { get; set; }
        public decimal Price { get; set; }

        public CartItem() { }

        public CartItem(string item, decimal price)
        {
            Item = item;
            Price = price;
        }
    }

    public class Checkout
    {
        public decimal CalculateTotal(CartItem[] items)
        {
            if (items == null || items.Length == 0) return 0m;

            decimal total = 0m;
            foreach (var item in items)
            {
                if (item == null) continue;
                if (item.Price < 0m) throw new ArgumentOutOfRangeException(nameof(item.Price), "Price cannot be negative.");
                total += item.Price;
            }

            return total;
        }
    }
}
