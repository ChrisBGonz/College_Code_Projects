function handleLogout(req, res) {
  if (req.session) {
    req.session.destroy(err => {
      if (err) {
        res.status(500).send('Logout failed');
      } else {
        res.clearCookie('connect.sid');
        res.send('Logged out');
      }
    });
  } else {
    res.send('No session');
  }
}

module.exports = handleLogout;
