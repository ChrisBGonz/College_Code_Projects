const express = require('express');
const app = express();

app.set('trust proxy', true);
app.use(express.json());

const polls = new Map();
let pollSeq = 1;

function getClientIp(req) {
  let ip = req.headers['x-forwarded-for'];
  if (ip) {
    ip = ip.split(',')[0].trim();
  } else {
    ip = (req.socket && req.socket.remoteAddress) || '';
  }
  if (ip.startsWith('::ffff:')) ip = ip.slice(7);
  return ip;
}

function sanitizePoll(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({
      id: o.id,
      text: o.text,
      votes: o.votes
    }))
  };
}

app.get('/', (req, res) => {
  res.json({ status: 'ok' });
});

app.post('/polls', (req, res) => {
  const { question, options } = req.body || {};
  if (typeof question !== 'string' || !Array.isArray(options) || options.length === 0) {
    res.status(400).json({ error: 'Invalid input' });
    return;
  }
  const poll = { id: String(pollSeq++), question: question.trim(), options: [] };
  options.forEach(opt => {
    let text = opt;
    if (opt && typeof opt === 'object' && typeof opt.text === 'string') {
      text = opt.text;
    }
    if (typeof text === 'string' && text.trim() !== '') {
      poll.options.push({
        id: String(poll.options.length + 1),
        text: text.trim(),
        votes: 0,
        voters: new Set()
      });
    }
  });
  if (poll.options.length === 0) {
    res.status(400).json({ error: 'No valid options' });
    return;
  }
  polls.set(poll.id, poll);
  res.status(201).json(sanitizePoll(poll));
});

app.get('/polls', (req, res) => {
  res.json(Array.from(polls.values()).map(sanitizePoll));
});

app.get('/polls/:pollId', (req, res) => {
  const poll = polls.get(req.params.pollId);
  if (!poll) {
    res.status(404).json({ error: 'Poll not found' });
    return;
  }
  res.json(sanitizePoll(poll));
});

app.post('/polls/:pollId/vote', (req, res) => {
  const poll = polls.get(req.params.pollId);
  if (!poll) {
    res.status(404).json({ error: 'Poll not found' });
    return;
  }
  const optionId = String(req.body && req.body.optionId ? req.body.optionId : '');
  const option = poll.options.find(o => o.id === optionId);
  if (!option) {
    res.status(404).json({ error: 'Option not found' });
    return;
  }
  const ip = getClientIp(req);
  if (!ip) {
    res.status(400).json({ error: 'Unable to determine IP' });
    return;
  }
  if (option.voters.has(ip)) {
    res.status(409).json({ error: 'Duplicate vote' });
    return;
  }
  option.voters.add(ip);
  option.votes += 1;
  res.json({ success: true, poll: sanitizePoll(poll) });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Voting server running on port ${PORT}`);
});
