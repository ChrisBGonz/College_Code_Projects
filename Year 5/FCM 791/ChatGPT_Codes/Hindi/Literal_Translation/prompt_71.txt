import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Any, Tuple

@dataclass
class Item:
    name: str
    quantity: int
    attributes: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    health: int
    inventory: List[Item] = field(default_factory=list)
    position: Tuple[int, int] = (0, 0)

@dataclass
class WorldState:
    time_of_day: str
    discovered_locations: List[str] = field(default_factory=list)
    npc_relations: Dict[str, int] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world: WorldState
    active_quests: List[str] = field(default_factory=list)
    completed_quests: List[str] = field(default_factory=list)
    flags: Dict[str, bool] = field(default_factory=dict)

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def main():
    sword = Item("तलवार", 1, {"damage": 12, "rarity": "rare"})
    potion = Item("औषधि", 5, {"heal": 25})
    player = Player(name="Arjun", level=5, health=80, inventory=[sword, potion], position=(10, 20))
    world = WorldState(time_of_day="रात", discovered_locations=["गाँव", "किला"], npc_relations={"राजा": 5, "व्यापारी": 2})
    state = GameState(
        player=player,
        world=world,
        active_quests=["किले की रक्षा"],
        completed_quests=["प्रशिक्षण"],
        flags={"has_key": True, "boss_defeated": False}
    )

    filename = "game_state.pkl"
    save_game_state(state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
