#include <sys/types.h>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <cstddef>
#include <cstdint>
#include <memory>
#include <stdexcept>

std::unique_ptr<char[]> allocateBufferFromSocket(int sockfd, std::size_t &sizeOut)
{
    uint32_t netSize = 0;
    std::size_t received = 0;
    char *p = reinterpret_cast<char *>(&netSize);

    while (received < sizeof(netSize)) {
        ssize_t n = recv(sockfd, p + received, sizeof(netSize) - received, 0);
        if (n <= 0) {
            throw std::runtime_error("Failed to read size from socket");
        }
        received += static_cast<std::size_t>(n);
    }

    uint32_t size = ntohl(netSize);
    sizeOut = static_cast<std::size_t>(size);
    std::unique_ptr<char[]> buffer(new char[sizeOut]);
    return buffer;
}
