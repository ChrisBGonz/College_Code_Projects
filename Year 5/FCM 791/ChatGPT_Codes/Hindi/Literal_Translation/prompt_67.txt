import base64
import hmac
import hashlib
import struct
import time
import sys

def totp(secret, time_step=30, digits=6, t=None):
    if t is None:
        t = time.time()
    counter = int(t // time_step)
    key = base64.b32decode(secret.replace(" ", "").upper(), casefold=True)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    o = h[-1] & 0x0f
    code = (struct.unpack(">I", h[o:o+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify(secret, code, window=1):
    now = time.time()
    for offset in range(-window, window + 1):
        if totp(secret, t=now + offset * 30) == code:
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = input("Secret: ").strip()
        code = input("Code: ").strip()
    if verify(secret, code):
        print("VALID")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
