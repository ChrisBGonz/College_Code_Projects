import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.stereotype.Service;
import org.springframework.web.bind.annotation.*;

import java.time.Instant;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;

@RestController
@RequestMapping("/api/private-messages")
public class PrivateMessageController {

    private final PrivateMessageService privateMessageService;

    public PrivateMessageController(PrivateMessageService privateMessageService) {
        this.privateMessageService = privateMessageService;
    }

    @GetMapping("/{message_id}")
    public ResponseEntity<PrivateMessage> getPrivateMessage(
            @PathVariable("message_id") Long messageId,
            @RequestParam("user_id") Long userId) {

        Optional<PrivateMessage> message = privateMessageService.findForUser(messageId, userId);

        if (message.isEmpty()) {
            return ResponseEntity.status(HttpStatus.NOT_FOUND).build();
        }

        return ResponseEntity.ok(message.get());
    }
}

interface PrivateMessageService {
    Optional<PrivateMessage> findForUser(Long messageId, Long userId);
}

@Service
class InMemoryPrivateMessageService implements PrivateMessageService {

    private final Map<Long, PrivateMessage> store = new ConcurrentHashMap<>();

    public InMemoryPrivateMessageService() {
        store.put(1L, new PrivateMessage(1L, 1L, 2L, "नमस्ते", Instant.now()));
        store.put(2L, new PrivateMessage(2L, 2L, 1L, "हाय", Instant.now()));
    }

    @Override
    public Optional<PrivateMessage> findForUser(Long messageId, Long userId) {
        PrivateMessage message = store.get(messageId);
        if (message == null) {
            return Optional.empty();
        }
        if (!Objects.equals(message.getSenderId(), userId) && !Objects.equals(message.getReceiverId(), userId)) {
            return Optional.empty();
        }
        return Optional.of(message);
    }
}

class PrivateMessage {
    private Long id;
    private Long senderId;
    private Long receiverId;
    private String content;
    private Instant sentAt;

    public PrivateMessage() {
    }

    public PrivateMessage(Long id, Long senderId, Long receiverId, String content, Instant sentAt) {
        this.id = id;
        this.senderId = senderId;
        this.receiverId = receiverId;
        this.content = content;
        this.sentAt = sentAt;
    }

    public Long getId() {
        return id;
    }

    public void setId(Long id) {
        this.id = id;
    }

    public Long getSenderId() {
        return senderId;
    }

    public void setSenderId(Long senderId) {
        this.senderId = senderId;
    }

    public Long getReceiverId() {
        return receiverId;
    }

    public void setReceiverId(Long receiverId) {
        this.receiverId = receiverId;
    }

    public String getContent() {
        return content;
    }

    public void setContent(String content) {
        this.content = content;
    }

    public Instant getSentAt() {
        return sentAt;
    }

    public void setSentAt(Instant sentAt) {
        this.sentAt = sentAt;
    }
}
