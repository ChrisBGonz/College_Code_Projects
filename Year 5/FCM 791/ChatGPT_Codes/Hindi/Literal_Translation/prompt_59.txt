import json
import os
import uuid
from http.server import BaseHTTPRequestHandler
try:
    from http.server import ThreadingHTTPServer
except ImportError:
    from http.server import HTTPServer as ThreadingHTTPServer
from urllib.parse import urlparse, parse_qs
import urllib.request

SESSIONS = {}
GATEWAY_API_URL = os.environ.get("GATEWAY_API_URL")
GATEWAY_API_KEY = os.environ.get("GATEWAY_API_KEY")

def base_url(host):
    if host.startswith("http://") or host.startswith("https://"):
        return host
    return "http://" + host

def create_gateway_session(amount, currency, success_url, host):
    if GATEWAY_API_URL:
        payload = json.dumps({
            "amount": amount,
            "currency": currency,
            "success_url": success_url
        }).encode("utf-8")
        headers = {"Content-Type": "application/json"}
        if GATEWAY_API_KEY:
            headers["Authorization"] = "Bearer " + GATEWAY_API_KEY
        req = urllib.request.Request(GATEWAY_API_URL, data=payload, headers=headers, method="POST")
        try:
            with urllib.request.urlopen(req, timeout=10) as resp:
                resp_body = resp.read().decode("utf-8")
            data = json.loads(resp_body) if resp_body else {}
            payment_url = data.get("payment_url")
            session_id = data.get("session_id") or str(uuid.uuid4())
            if payment_url:
                SESSIONS[session_id] = {"success_url": success_url, "amount": amount, "currency": currency}
                return session_id, payment_url
        except Exception:
            pass
    session_id = str(uuid.uuid4())
    SESSIONS[session_id] = {"success_url": success_url, "amount": amount, "currency": currency}
    payment_url = f"{base_url(host)}/gateway?session_id={session_id}"
    return session_id, payment_url

class PaymentHandler(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def _send_json(self, data, status=200):
        body = json.dumps(data).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _send_html(self, html, status=200):
        body = html.encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _redirect(self, url, status=302):
        self.send_response(status)
        self.send_header("Location", url)
        self.send_header("Content-Length", "0")
        self.end_headers()

    def _read_json(self):
        length = int(self.headers.get("Content-Length", 0))
        if length <= 0:
            return {}
        raw = self.rfile.read(length)
        if not raw:
            return {}
        try:
            return json.loads(raw.decode("utf-8"))
        except Exception:
            return {}

    def do_POST(self):
        parsed = urlparse(self.path)
        if parsed.path in ("/pay", "/initiate"):
            data = self._read_json()
            success_url = data.get("success_url")
            if not success_url:
                self._send_json({"error": "success_url required"}, 400)
                return
            amount = data.get("amount", "")
            currency = data.get("currency", "")
            host = self.headers.get("Host", "localhost:8000")
            session_id, payment_url = create_gateway_session(amount, currency, success_url, host)
            if session_id in SESSIONS:
                SESSIONS[session_id]["amount"] = amount
                SESSIONS[session_id]["currency"] = currency
            self._redirect(payment_url)
            return
        self._send_json({"error": "Not Found"}, 404)

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path == "/":
            html = "<html><body><h3>Payment Integration Server</h3><p>Use /pay with success_url.</p></body></html>"
            self._send_html(html)
            return
        if parsed.path == "/pay":
            params = parse_qs(parsed.query)
            success_url = params.get("success_url", [None])[0]
            if not success_url:
                self._send_html("<html><body>success_url required</body></html>", 400)
                return
            amount = params.get("amount", [""])[0]
            currency = params.get("currency", [""])[0]
            host = self.headers.get("Host", "localhost:8000")
            session_id, payment_url = create_gateway_session(amount, currency, success_url, host)
            if session_id in SESSIONS:
                SESSIONS[session_id]["amount"] = amount
                SESSIONS[session_id]["currency"] = currency
            self._redirect(payment_url)
            return
        if parsed.path == "/gateway":
            params = parse_qs(parsed.query)
            session_id = params.get("session_id", [None])[0]
            if not session_id or session_id not in SESSIONS:
                self._send_html("<html><body>Invalid session</body></html>", 400)
                return
            details = SESSIONS[session_id]
            amount = details.get("amount", "")
            currency = details.get("currency", "")
            html = f"<html><body><h2>Simulated Payment Gateway</h2><p>Amount: {amount} {currency}</p><form action=\"/gateway/complete\" method=\"get\"><input type=\"hidden\" name=\"session_id\" value=\"{session_id}\"><button type=\"submit\">Complete Payment</button></form></body></html>"
            self._send_html(html)
            return
        if parsed.path == "/gateway/complete":
            params = parse_qs(parsed.query)
            session_id = params.get("session_id", [None])[0]
            if not session_id or session_id not in SESSIONS:
                self._send_html("<html><body>Invalid session</body></html>", 400)
                return
            SESSIONS[session_id]["paid"] = True
            host = self.headers.get("Host", "localhost:8000")
            callback_url = f"{base_url(host)}/callback?session_id={session_id}"
            self._redirect(callback_url)
            return
        if parsed.path == "/callback":
            params = parse_qs(parsed.query)
            session_id = params.get("session_id", [None])[0]
            if not session_id or session_id not in SESSIONS:
                self._send_html("<html><body>Invalid session</body></html>", 400)
                return
            success_url = SESSIONS[session_id].get("success_url")
            if not success_url:
                self._send_html("<html><body>success_url missing</body></html>", 400)
                return
            self._redirect(success_url)
            return
        self._send_html("<html><body>Not Found</body></html>", 404)

def run():
    port = int(os.environ.get("PORT", "8000"))
    server = ThreadingHTTPServer(("0.0.0.0", port), PaymentHandler)
    server.serve_forever()

if __name__ == "__main__":
    run()
