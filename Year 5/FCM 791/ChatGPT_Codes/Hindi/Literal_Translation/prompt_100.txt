class CircuitBreaker {
  constructor({ failureThreshold = 5, successThreshold = 2, timeout = 10000 } = {}) {
    this.failureThreshold = failureThreshold;
    this.successThreshold = successThreshold;
    this.timeout = timeout;
    this.state = "CLOSED";
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
  }

  async exec(action) {
    if (this.state === "OPEN") {
      if (Date.now() > this.nextAttempt) {
        this.state = "HALF_OPEN";
      } else {
        return Promise.reject(new Error("Circuit breaker is open"));
      }
    }

    try {
      const result = await action();
      return this._handleSuccess(result);
    } catch (err) {
      return this._handleFailure(err);
    }
  }

  _handleSuccess(result) {
    if (this.state === "HALF_OPEN") {
      this.successCount += 1;
      if (this.successCount >= this.successThreshold) {
        this._reset();
      }
    } else {
      this.failureCount = 0;
    }
    return result;
  }

  _handleFailure(err) {
    this.failureCount += 1;
    if (this.state === "HALF_OPEN" || this.failureCount >= this.failureThreshold) {
      this._trip();
    }
    return Promise.reject(err);
  }

  _trip() {
    this.state = "OPEN";
    this.nextAttempt = Date.now() + this.timeout;
    this.successCount = 0;
  }

  _reset() {
    this.state = "CLOSED";
    this.failureCount = 0;
    this.successCount = 0;
  }
}

const breaker = new CircuitBreaker({
  failureThreshold: 3,
  successThreshold: 2,
  timeout: 10000
});

function circuitFetch(url, options) {
  return breaker.exec(() => fetch(url, options));
}

export { CircuitBreaker, circuitFetch };
