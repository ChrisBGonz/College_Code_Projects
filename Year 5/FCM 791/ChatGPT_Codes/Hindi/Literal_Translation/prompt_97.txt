<?php
function getSubscriptionStatus($userId, PDO $pdo) {
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :id LIMIT 1');
        $stmt->execute([':id' => $userId]);
        $status = $stmt->fetchColumn();
        if ($status === false || $status === null || $status === '') {
            return 'free';
        }
        return $status;
    } catch (Exception $e) {
        return 'premium';
    }
}
?>
