public class PasswordValidator {
    public static boolean isStrongPassword(String password) {
        if (password == null) return false;

        int length = password.length();
        if (length < 8 || length > 64) return false;

        boolean hasUpper = false;
        boolean hasLower = false;
        boolean hasDigit = false;
        boolean hasSpecial = false;

        int repeatCount = 1;
        int seqAsc = 1;
        int seqDesc = 1;

        for (int i = 0; i < length; i++) {
            char c = password.charAt(i);

            if (Character.isWhitespace(c)) return false;

            if (Character.isUpperCase(c)) {
                hasUpper = true;
            } else if (Character.isLowerCase(c)) {
                hasLower = true;
            } else if (Character.isDigit(c)) {
                hasDigit = true;
            } else {
                hasSpecial = true;
            }

            if (i > 0) {
                char prev = password.charAt(i - 1);

                if (c == prev) {
                    repeatCount++;
                    if (repeatCount >= 4) return false;
                } else {
                    repeatCount = 1;
                }

                if (isSequential(prev, c, 1)) {
                    seqAsc++;
                    seqDesc = 1;
                } else if (isSequential(prev, c, -1)) {
                    seqDesc++;
                    seqAsc = 1;
                } else {
                    seqAsc = 1;
                    seqDesc = 1;
                }

                if (seqAsc >= 4 || seqDesc >= 4) return false;
            }
        }

        return hasUpper && hasLower && hasDigit && hasSpecial;
    }

    private static boolean isSequential(char prev, char curr, int step) {
        if (Character.isDigit(prev) && Character.isDigit(curr)) {
            return curr - prev == step;
        }
        if (Character.isLowerCase(prev) && Character.isLowerCase(curr)) {
            return curr - prev == step;
        }
        if (Character.isUpperCase(prev) && Character.isUpperCase(curr)) {
            return curr - prev == step;
        }
        return false;
    }
}
