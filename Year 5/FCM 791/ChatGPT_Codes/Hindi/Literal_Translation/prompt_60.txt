const express = require('express');

const app = express();
app.use(express.json());

const polls = new Map();
let nextPollId = 1;

function getClientIp(req) {
  let ip = req.headers['x-forwarded-for'];
  if (ip) {
    ip = ip.split(',')[0].trim();
  } else {
    ip = req.connection.remoteAddress || req.socket.remoteAddress || '';
  }
  if (ip.startsWith('::ffff:')) ip = ip.substring(7);
  return ip || 'unknown';
}

app.post('/poll', (req, res) => {
  const { question, options } = req.body || {};
  if (!question || !Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  const id = String(nextPollId++);
  const votes = {};
  options.forEach(opt => {
    votes[opt] = 0;
  });
  const poll = { id, question, options, votes, voters: new Set() };
  polls.set(id, poll);
  res.status(201).json({ id, question, options });
});

app.post('/poll/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) {
    return res.status(404).json({ error: 'Poll not found' });
  }
  const { option } = req.body || {};
  if (!option || !Object.prototype.hasOwnProperty.call(poll.votes, option)) {
    return res.status(400).json({ error: 'Invalid option' });
  }
  const ip = getClientIp(req);
  if (poll.voters.has(ip)) {
    return res.status(409).json({ error: 'Duplicate vote' });
  }
  poll.voters.add(ip);
  poll.votes[option] += 1;
  res.json({ message: 'Vote recorded' });
});

app.get('/poll/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) {
    return res.status(404).json({ error: 'Poll not found' });
  }
  res.json({ id: poll.id, question: poll.question, options: poll.options, votes: poll.votes });
});

app.get('/poll', (req, res) => {
  const list = Array.from(polls.values()).map(p => ({
    id: p.id,
    question: p.question,
    options: p.options
  }));
  res.json(list);
});

const port = process.env.PORT || 3000;
app.listen(port, () => {
  console.log(`Server running on port ${port}`);
});
