public class CartItem
{
    public string Name { get; set; }
    public decimal Price { get; set; }
}

public class Checkout
{
    public decimal CalculateTotal(CartItem[] items)
    {
        if (items == null || items.Length == 0)
            return 0m;

        decimal total = 0m;
        foreach (var item in items)
        {
            if (item != null)
                total += item.Price;
        }
        return total;
    }
}
