using System;
using Microsoft.Extensions.Logging;

public class MfaService
{
    private readonly ILogger<MfaService> _logger;
    private readonly IUserRepository _users;

    public MfaService(ILogger<MfaService> logger, IUserRepository users)
    {
        _logger = logger;
        _users = users;
    }

    public void UpdateMfaSettings(string userId, MfaSettings newSettings)
    {
        var user = _users.GetById(userId);
        if (user == null) throw new InvalidOperationException("User not found.");

        var oldSettings = user.MfaSettings;

        if (!Equals(oldSettings, newSettings))
        {
            user.MfaSettings = newSettings;
            _users.Save(user);

            _logger.LogInformation(
                "MFA settings changed for UserId={UserId}. Old={OldSettings} New={NewSettings}",
                userId,
                oldSettings,
                newSettings);
        }
    }
}

public interface IUserRepository
{
    User GetById(string userId);
    void Save(User user);
}

public class User
{
    public string Id { get; set; }
    public MfaSettings MfaSettings { get; set; }
}

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }

    public override string ToString() => $"Enabled={Enabled}, Method={Method}";
}
