from flask import Flask, request, make_response

app = Flask(__name__)

@app.route("/free-trial", methods=["GET", "POST"])
def free_trial():
    if request.cookies.get("has_used_trial") == "true":
        return "Trial already used", 403

    if request.method == "POST":
        response = make_response("Trial activated")
        response.set_cookie(
            "has_used_trial",
            "true",
            max_age=60 * 60 * 24 * 365,
            httponly=True,
            samesite="Lax",
        )
        return response

    return "Ready for trial signup", 200

if __name__ == "__main__":
    app.run()
