using System;
using System.Data;
using System.Data.SqlClient;
using System.Text.RegularExpressions;

public class DataSorter
{
    public DataTable GetSortedTable(string connectionString, string tableName, string columnName, bool ascending = true)
    {
        if (string.IsNullOrWhiteSpace(tableName)) throw new ArgumentException("tableName");
        if (string.IsNullOrWhiteSpace(columnName)) throw new ArgumentException("columnName");
        if (!Regex.IsMatch(tableName, @"^[A-Za-z_][A-Za-z0-9_]*$")) throw new ArgumentException("Invalid table name");
        if (!Regex.IsMatch(columnName, @"^[A-Za-z_][A-Za-z0-9_]*$")) throw new ArgumentException("Invalid column name");

        string direction = ascending ? "ASC" : "DESC";
        string sql = $"SELECT * FROM [{tableName}] ORDER BY [{columnName}] {direction}";

        using (var conn = new SqlConnection(connectionString))
        using (var cmd = new SqlCommand(sql, conn))
        using (var adapter = new SqlDataAdapter(cmd))
        {
            var table = new DataTable();
            conn.Open();
            adapter.Fill(table);
            return table;
        }
    }
}
