<?php
session_start();
header('Content-Type: text/html; charset=utf-8');

define('DATA_FILE', __DIR__ . '/users.json');
define('SECRET_QUESTION', 'आपकी माँ का विवाह-पूर्व उपनाम क्या है?');

function saveUsers(array $users): void {
    file_put_contents(DATA_FILE, json_encode($users, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

function lowerString(string $s): string {
    if (function_exists('mb_strtolower')) {
        return mb_strtolower($s, 'UTF-8');
    }
    return strtolower($s);
}

function normalizeAnswer(string $answer): string {
    $answer = trim($answer);
    $answer = preg_replace('/\s+/u', ' ', $answer);
    return lowerString($answer);
}

function loadUsers(): array {
    if (!file_exists(DATA_FILE)) {
        $defaultAnswer = normalizeAnswer('शर्मा');
        $default = [
            [
                'username' => 'user',
                'email' => 'user@example.com',
                'password_hash' => password_hash('password', PASSWORD_DEFAULT),
                'secret_answer_hash' => password_hash($defaultAnswer, PASSWORD_DEFAULT),
            ]
        ];
        saveUsers($default);
        return $default;
    }
    $json = file_get_contents(DATA_FILE);
    $data = json_decode($json, true);
    if (!is_array($data)) {
        $data = [];
    }
    return $data;
}

function findUserIndex(array $users, string $identifier): int {
    $id = lowerString($identifier);
    foreach ($users as $i => $u) {
        $user = $u['username'] ?? '';
        $email = $u['email'] ?? '';
        if (lowerString($user) === $id || lowerString($email) === $id) {
            return $i;
        }
    }
    return -1;
}

function resetRecoverySession(): void {
    unset($_SESSION['recover_user'], $_SESSION['recover_step']);
}

$users = loadUsers();
$errors = [];
$success = '';

if (isset($_GET['restart'])) {
    resetRecoverySession();
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['identifier'])) {
        $identifier = trim((string)$_POST['identifier']);
        if ($identifier === '') {
            $errors[] = 'कृपया अपना यूज़रनेम या ईमेल दर्ज करें।';
        } else {
            $idx = findUserIndex($users, $identifier);
            if ($idx < 0) {
                $errors[] = 'उपयोगकर्ता नहीं मिला।';
            } else {
                $_SESSION['recover_user'] = $users[$idx]['username'];
                $_SESSION['recover_step'] = 2;
            }
        }
    } elseif (isset($_POST['secret_answer'])) {
        $answer = normalizeAnswer((string)$_POST['secret_answer']);
        if ($answer === '') {
            $errors[] = 'कृपया उत्तर दर्ज करें।';
        } else {
            if (!isset($_SESSION['recover_user'])) {
                $errors[] = 'रिकवरी सत्र समाप्त हो गया है।';
                resetRecoverySession();
            } else {
                $idx = findUserIndex($users, $_SESSION['recover_user']);
                if ($idx < 0) {
                    $errors[] = 'उपयोगकर्ता नहीं मिला।';
                    resetRecoverySession();
                } else {
                    $hash = $users[$idx]['secret_answer_hash'] ?? '';
                    if (!$hash || !password_verify($answer, $hash)) {
                        $errors[] = 'उत्तर गलत है।';
                    } else {
                        $_SESSION['recover_step'] = 3;
                    }
                }
            }
        }
    } elseif (isset($_POST['new_password'])) {
        $new = (string)($_POST['new_password'] ?? '');
        $confirm = (string)($_POST['confirm_password'] ?? '');
        if (!isset($_SESSION['recover_user']) || ($_SESSION['recover_step'] ?? 1) < 3) {
            $errors[] = 'रिकवरी सत्र समाप्त हो गया है।';
            resetRecoverySession();
        } else {
            if (strlen($new) < 6) {
                $errors[] = 'पासवर्ड कम से कम 6 अक्षरों का होना चाहिए।';
            } elseif ($new !== $confirm) {
                $errors[] = 'पासवर्ड पुष्टि मेल नहीं खा रही है।';
            } else {
                $idx = findUserIndex($users, $_SESSION['recover_user']);
                if ($idx < 0) {
                    $errors[] = 'उपयोगकर्ता नहीं मिला।';
                    resetRecoverySession();
                } else {
                    $users[$idx]['password_hash'] = password_hash($new, PASSWORD_DEFAULT);
                    saveUsers($users);
                    $success = 'पासवर्ड सफलतापूर्वक बदल दिया गया है।';
                    resetRecoverySession();
                }
            }
        }
    }
}

$step = $_SESSION['recover_step'] ?? 1;
if ($step > 1 && !isset($_SESSION['recover_user'])) {
    $step = 1;
    resetRecoverySession();
}

?>

h1{margin-top:0;font-size:22px;}
label{display:block;margin-top:12px;}
input[type=text],input[type=password]{width:100%;padding:8px;margin-top:6px;box-sizing:border-box;}
button{margin-top:16px;padding:10px 16px;}
.errors{background:#ffe0e0;color:#900;padding:10px;border-radius:4px;margin-bottom:16px;}
.errors ul{margin:0;padding-left:18px;}
.success{background:#e0ffe5;color:#060;padding:10px;border-radius:4px;margin-bottom:16px;}
a{color:#0645ad;text-decoration:none;}
a:hover{text-decoration:underline;}
</style>
</head>
<body>
<div class="container">
<h1>सीक्रेट प्रश्न रिकवरी</h1>
<?php if ($errors): ?>
<div class="errors">
<ul>
<?php foreach ($errors as $e): ?>
<li><?php echo htmlspecialchars($e, ENT_QUOTES, 'UTF-8'); ?></li>
<?php endforeach; ?>
</ul>
</div>
<?php endif; ?>

<?php if ($success): ?>
<div class="success"><?php echo htmlspecialchars($success, ENT_QUOTES, 'UTF-8'); ?></div>
<p><a href="?restart=1">फिर से शुरू करें</a></p>
<?php else: ?>
<?php if ($step === 1): ?>
<form method="post" autocomplete="off">
<label for="identifier">यूज़रनेम या ईमेल</label>
<input type="text" id="identifier" name="identifier" required>
<button type="submit">आगे बढ़ें</button>
</form>
<?php elseif ($step === 2): ?>
<p><strong>सीक्रेट प्रश्न:</strong> <?php echo htmlspecialchars(SECRET_QUESTION, ENT_QUOTES, 'UTF-8'); ?></p>
<form method="post" autocomplete="off">
<label for="secret_answer">उत्तर</label>
<input type="text" id="secret_answer" name="secret_answer" required>
<button type="submit">सत्यापित करें</button>
</form>
<p><a href="?restart=1">वापस जाएँ</a></p>
<?php elseif ($step === 3): ?>
<form method="post" autocomplete="off">
<label for="new_password">नया पासवर्ड</label>
<input type="password" id="new_password" name="new_password" required>
<label for="confirm_password">पासवर्ड पुष्टि</label>
<input type="password" id="confirm_password" name="confirm_password" required>
<button type="submit">पासवर्ड बदलें</button>
</form>
<p><a href="?restart=1">वापस जाएँ</a></p>
<?php endif; ?>
<?php endif; ?>
</div>
</body>
</html>
