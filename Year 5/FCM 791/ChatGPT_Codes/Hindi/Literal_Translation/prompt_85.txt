<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = '';

function send_admin_alert($subject, $body, $adminEmail, $slackWebhook) {
    if (!empty($adminEmail)) {
        @mail($adminEmail, $subject, $body);
    }
    if (!empty($slackWebhook)) {
        $payload = json_encode(['text' => $subject . "\n" . $body]);
        if (function_exists('curl_init')) {
            $ch = curl_init($slackWebhook);
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
            curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
            curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
            curl_setopt($ch, CURLOPT_TIMEOUT, 5);
            curl_exec($ch);
            curl_close($ch);
        } else {
            $context = stream_context_create([
                'http' => [
                    'method' => 'POST',
                    'header' => "Content-Type: application/json\r\n",
                    'content' => $payload,
                    'timeout' => 5
                ]
            ]);
            @file_get_contents($slackWebhook, false, $context);
        }
    }
}

$path = parse_url($_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH);
$authorized = isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;

if (strpos($path, '/admin') === 0 && !$authorized) {
    $details = [
        'Time' => date('c'),
        'IP' => $_SERVER['REMOTE_ADDR'] ?? 'unknown',
        'User-Agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'unknown',
        'Method' => $_SERVER['REQUEST_METHOD'] ?? 'unknown',
        'URI' => $_SERVER['REQUEST_URI'] ?? 'unknown'
    ];
    $body = '';
    foreach ($details as $k => $v) {
        $body .= $k . ': ' . $v . "\n";
    }
    send_admin_alert('Admin Alert: Unauthorized /admin access attempt', $body, $adminEmail, $slackWebhook);
    http_response_code(403);
    echo 'Access denied';
    exit;
}
