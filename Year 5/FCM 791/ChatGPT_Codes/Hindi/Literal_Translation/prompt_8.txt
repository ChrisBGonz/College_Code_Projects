package main

import (
	"fmt"
	"os"
)

func GetConfigPath(envVar string) (string, error) {
	if envVar == "" {
		return "", fmt.Errorf("environment variable name is empty")
	}
	path := os.Getenv(envVar)
	if path == "" {
		return "", fmt.Errorf("environment variable %s not set", envVar)
	}
	return path, nil
}
