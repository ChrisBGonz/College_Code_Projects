<?php
$uploadDir = '/shared/company_drive/uploads/';
$maxSize = 20 * 1024 * 1024;
$message = '';

if (!is_dir($uploadDir)) {
    @mkdir($uploadDir, 0755, true);
}

function uploadErrorMessage($code) {
    switch ($code) {
        case UPLOAD_ERR_INI_SIZE:
        case UPLOAD_ERR_FORM_SIZE:
            return 'File size exceeds limit.';
        case UPLOAD_ERR_PARTIAL:
            return 'File was partially uploaded.';
        case UPLOAD_ERR_NO_FILE:
            return 'No file uploaded.';
        case UPLOAD_ERR_NO_TMP_DIR:
            return 'Missing temporary folder.';
        case UPLOAD_ERR_CANT_WRITE:
            return 'Failed to write file.';
        case UPLOAD_ERR_EXTENSION:
            return 'File upload stopped by extension.';
        default:
            return 'Unknown upload error.';
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdf'])) {
        $message = 'No file uploaded.';
    } else {
        $file = $_FILES['pdf'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $message = uploadErrorMessage($file['error']);
        } elseif ($file['size'] > $maxSize) {
            $message = 'File size exceeds limit.';
        } else {
            $tmp = $file['tmp_name'];
            $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            $finfo = new finfo(FILEINFO_MIME_TYPE);
            $mime = $finfo->file($tmp);
            if ($ext !== 'pdf' || $mime !== 'application/pdf') {
                $message = 'Only PDF files are allowed.';
            } else {
                $base = pathinfo($file['name'], PATHINFO_FILENAME);
                $base = preg_replace('/[^A-Za-z0-9_\-]/', '_', $base);
                if ($base === '') {
                    $base = 'document';
                }
                $name = $base . '_' . date('Ymd_His') . '_' . bin2hex(random_bytes(4)) . '.pdf';
                $destination = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $name;
                if (move_uploaded_file($tmp, $destination)) {
                    $message = 'File uploaded successfully.';
                } else {
                    $message = 'Failed to move uploaded file.';
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>PDF Upload</title>
</head>
<body>
<?php if ($message !== ''): ?>
<p><?php echo htmlspecialchars($message, ENT_QUOTES, 'UTF-8'); ?></p>
<?php endif; ?>
<form method="post" enctype="multipart/form-data">
<label>Select PDF file:</label>
<input type="file" name="pdf" accept="application/pdf" required>
<button type="submit">Upload</button>
</form>
</body>
</html>
