import sys
import time
import os
from collections import deque

def monitor_log(file_path):
    failed_times = deque()
    last_warning = 0

    with open(file_path, 'r', errors='ignore') as f:
        f.seek(0, os.SEEK_END)
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.5)
                continue
            if 'Failed Login' in line:
                now = time.time()
                failed_times.append(now)
                cutoff = now - 60
                while failed_times and failed_times[0] < cutoff:
                    failed_times.popleft()
                if len(failed_times) > 10 and now - last_warning >= 60:
                    print("Warning: more than 10 Failed Login messages in the last minute", flush=True)
                    last_warning = now

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python monitor.py <log_file_path>")
        sys.exit(1)
    monitor_log(sys.argv[1])
