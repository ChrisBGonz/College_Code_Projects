<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $fontSize = 'medium';
}

$prefs = new UserPreferences();

if (isset($_COOKIE['user_prefs'])) {
    $data = $_COOKIE['user_prefs'];
    $obj = @unserialize($data, ['allowed_classes' => ['UserPreferences']]);
    if ($obj instanceof UserPreferences) {
        $prefs = $obj;
    }
}

$input = array_merge($_GET, $_POST);
$updated = false;

if (isset($input['theme'])) {
    $prefs->theme = substr(preg_replace('/[^a-zA-Z0-9_-]/', '', $input['theme']), 0, 20);
    $updated = true;
}
if (isset($input['language'])) {
    $prefs->language = substr(preg_replace('/[^a-zA-Z0-9_-]/', '', $input['language']), 0, 20);
    $updated = true;
}
if (isset($input['fontSize'])) {
    $prefs->fontSize = substr(preg_replace('/[^a-zA-Z0-9_-]/', '', $input['fontSize']), 0, 20);
    $updated = true;
}

if ($updated) {
    setcookie('user_prefs', serialize($prefs), time() + 60 * 60 * 24 * 30, '/');
}

header('Content-Type: application/json; charset=utf-8');
echo json_encode($prefs);
?>
