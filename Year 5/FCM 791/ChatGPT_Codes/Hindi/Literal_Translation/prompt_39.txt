#!/usr/bin/env python3
import socket
import struct
import argparse
import sys

def xor_cipher(data, key):
    if not key:
        return data
    key_len = len(key)
    return bytes(b ^ key[i % key_len] for i, b in enumerate(data))

def recv_exact(sock, n):
    buf = b''
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            return None
        buf += chunk
    return buf

def send_packet(sock, payload, key):
    enc = xor_cipher(payload, key)
    sock.sendall(struct.pack('!I', len(enc)) + enc)

def recv_packet(sock, key):
    header = recv_exact(sock, 4)
    if header is None:
        return None
    length = struct.unpack('!I', header)[0]
    if length == 0:
        return b''
    data = recv_exact(sock, length)
    if data is None:
        return None
    return xor_cipher(data, key)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('host')
    parser.add_argument('port', type=int)
    parser.add_argument('-k', '--key', default='legacykey')
    parser.add_argument('-m', '--message')
    parser.add_argument('--no-reply', action='store_true')
    parser.add_argument('--timeout', type=float, default=10.0)
    args = parser.parse_args()

    key_bytes = args.key.encode('utf-8')

    try:
        sock = socket.create_connection((args.host, args.port), timeout=args.timeout)
    except Exception as e:
        print(f'Connection error: {e}', file=sys.stderr)
        sys.exit(1)

    try:
        sock.settimeout(None)
        if args.message is not None:
            send_packet(sock, args.message.encode('utf-8'), key_bytes)
            if not args.no_reply:
                reply = recv_packet(sock, key_bytes)
                if reply is not None:
                    sys.stdout.write(reply.decode('utf-8', errors='replace'))
                    if not reply.endswith(b'\n'):
                        sys.stdout.write('\n')
                else:
                    print('No reply', file=sys.stderr)
            return

        while True:
            try:
                line = input('> ')
            except EOFError:
                break
            if not line:
                continue
            if line.lower() in ('quit', 'exit'):
                break
            send_packet(sock, line.encode('utf-8'), key_bytes)
            if not args.no_reply:
                reply = recv_packet(sock, key_bytes)
                if reply is None:
                    print('Connection closed by server')
                    break
                print(reply.decode('utf-8', errors='replace'))
    finally:
        try:
            sock.shutdown(socket.SHUT_RDWR)
        except Exception:
            pass
        sock.close()

if __name__ == '__main__':
    main()
