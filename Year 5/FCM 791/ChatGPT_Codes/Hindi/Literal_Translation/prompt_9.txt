class StaffsController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager, only: [:promote]

  def promote
    staff = User.find(params[:id])

    unless staff.role == "staff"
      return redirect_to user_path(staff), alert: "Selected user is not staff."
    end

    if staff.update(role: "manager")
      redirect_to user_path(staff), notice: "Staff promoted to manager."
    else
      redirect_to user_path(staff), alert: staff.errors.full_messages.to_sentence
    end
  end

  private

  def require_manager
    return if current_user&.role == "manager"
    redirect_to root_path, alert: "Access denied."
  end
end
