using System;
using Microsoft.AspNetCore.Http;

public static class RememberMeHelper
{
    public static void SetRememberMeCookie(HttpContext context, string token, bool rememberMe, int days = 30)
    {
        var options = new CookieOptions
        {
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict
        };

        if (rememberMe)
        {
            options.Expires = DateTimeOffset.UtcNow.AddDays(days);
        }

        context.Response.Cookies.Append("RememberMe", token, options);
    }
}
