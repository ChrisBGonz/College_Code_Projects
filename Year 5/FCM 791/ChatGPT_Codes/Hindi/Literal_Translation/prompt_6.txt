import json
import os
from datetime import datetime

class MultiTenantSaaS:
    def __init__(self, storage_path="companies.json"):
        self.storage_path = storage_path
        self.companies = {}
        self._load()

    def _load(self):
        if os.path.exists(self.storage_path):
            try:
                with open(self.storage_path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                if isinstance(data, dict):
                    self.companies = data
            except (json.JSONDecodeError, IOError):
                self.companies = {}
        else:
            self.companies = {}

    def _save(self):
        with open(self.storage_path, "w", encoding="utf-8") as f:
            json.dump(self.companies, f, ensure_ascii=False, indent=2)

    def add_company(self, company_id, name, annual_report):
        if company_id in self.companies:
            return False
        self.companies[company_id] = {
            "name": name,
            "annual_report": annual_report,
            "updated_at": datetime.utcnow().isoformat()
        }
        self._save()
        return True

    def update_report(self, company_id, annual_report):
        if company_id not in self.companies:
            return False
        self.companies[company_id]["annual_report"] = annual_report
        self.companies[company_id]["updated_at"] = datetime.utcnow().isoformat()
        self._save()
        return True

    def get_report(self, company_id):
        company = self.companies.get(company_id)
        if not company:
            return None
        return company["annual_report"]

    def list_companies(self):
        return [(cid, info["name"]) for cid, info in self.companies.items()]

def seed_data(app):
    if not app.companies:
        app.add_company("C001", "Alpha Tech", "Alpha Tech Annual Report: Revenue growth 20%.")
        app.add_company("C002", "Beta Solutions", "Beta Solutions Annual Report: New markets expansion.")
        app.add_company("C003", "Gamma Industries", "Gamma Industries Annual Report: Sustainability achievements.")

def main():
    app = MultiTenantSaaS()
    seed_data(app)
    menu = (
        "\n1. कंपनी वार्षिक रिपोर्ट देखें\n"
        "2. कंपनी जोड़ें\n"
        "3. रिपोर्ट अपडेट करें\n"
        "4. कंपनियाँ सूचीबद्ध करें\n"
        "5. बाहर निकलें\n"
    )
    while True:
        print(menu)
        choice = input("विकल्प चुनें: ").strip()
        if choice == "1":
            company_id = input("कंपनी ID दर्ज करें: ").strip()
            report = app.get_report(company_id)
            if report:
                print("\nवार्षिक रिपोर्ट:")
                print(report)
            else:
                print("कंपनी ID नहीं मिली।")
        elif choice == "2":
            company_id = input("नई कंपनी ID: ").strip()
            name = input("कंपनी नाम: ").strip()
            annual_report = input("वार्षिक रिपोर्ट: ").strip()
            if app.add_company(company_id, name, annual_report):
                print("कंपनी जोड़ी गई।")
            else:
                print("यह कंपनी ID पहले से मौजूद है।")
        elif choice == "3":
            company_id = input("कंपनी ID: ").strip()
            annual_report = input("नई वार्षिक रिपोर्ट: ").strip()
            if app.update_report(company_id, annual_report):
                print("रिपोर्ट अपडेट की गई।")
            else:
                print("कंपनी ID नहीं मिली।")
        elif choice == "4":
            companies = app.list_companies()
            if not companies:
                print("कोई कंपनी नहीं।")
            else:
                print("\nकंपनियाँ:")
                for cid, name in companies:
                    print(f"{cid} - {name}")
        elif choice == "5":
            break
        else:
            print("अमान्य विकल्प।")

if __name__ == "__main__":
    main()
