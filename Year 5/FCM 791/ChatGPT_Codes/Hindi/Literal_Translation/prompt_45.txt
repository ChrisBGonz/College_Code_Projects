public class LdapFilterBuilder {
    public static String buildFilter(String lastName) {
        if (lastName == null) {
            lastName = "";
        }
        return "(sn=" + escapeLDAPSearchFilter(lastName) + ")";
    }

    private static String escapeLDAPSearchFilter(String filter) {
        StringBuilder sb = new StringBuilder(filter.length());
        for (int i = 0; i < filter.length(); i++) {
            char c = filter.charAt(i);
            switch (c) {
                case '\\':
                    sb.append("\\5c");
                    break;
                case '*':
                    sb.append("\\2a");
                    break;
                case '(':
                    sb.append("\\28");
                    break;
                case ')':
                    sb.append("\\29");
                    break;
                case '\0':
                    sb.append("\\00");
                    break;
                default:
                    sb.append(c);
            }
        }
        return sb.toString();
    }
}
