package auth

import (
	"crypto/rand"
	"encoding/base64"
	"time"
)

type ResetToken struct {
	Token     string
	ExpiresAt time.Time
}

func GenerateResetToken() (ResetToken, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return ResetToken{}, err
	}
	token := base64.RawURLEncoding.EncodeToString(b)
	return ResetToken{
		Token:     token,
		ExpiresAt: time.Now().Add(time.Hour),
	}, nil
}
