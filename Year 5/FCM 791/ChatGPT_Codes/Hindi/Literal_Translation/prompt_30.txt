#include <jpeglib.h>
#include <cstdio>
#include <vector>
#include <string>
#include <setjmp.h>

struct JpegErrorManager {
    jpeg_error_mgr pub;
    jmp_buf setjmp_buffer;
};

extern "C" void jpegErrorExit(j_common_ptr cinfo) {
    JpegErrorManager* err = reinterpret_cast<JpegErrorManager*>(cinfo->err);
    longjmp(err->setjmp_buffer, 1);
}

bool LoadJPEG(const std::string& filename, std::vector<unsigned char>& pixels, int& width, int& height, int& channels) {
    FILE* infile = std::fopen(filename.c_str(), "rb");
    if (!infile) return false;

    jpeg_decompress_struct cinfo;
    JpegErrorManager jerr;

    cinfo.err = jpeg_std_error(&jerr.pub);
    jerr.pub.error_exit = jpegErrorExit;

    if (setjmp(jerr.setjmp_buffer)) {
        jpeg_destroy_decompress(&cinfo);
        std::fclose(infile);
        return false;
    }

    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, infile);
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);

    width = static_cast<int>(cinfo.output_width);
    height = static_cast<int>(cinfo.output_height);
    channels = static_cast<int>(cinfo.output_components);

    size_t row_stride = static_cast<size_t>(width * channels);
    pixels.resize(row_stride * height);

    while (cinfo.output_scanline < cinfo.output_height) {
        JSAMPROW row_pointer[1];
        row_pointer[0] = &pixels[cinfo.output_scanline * row_stride];
        jpeg_read_scanlines(&cinfo, row_pointer, 1);
    }

    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    std::fclose(infile);
    return true;
}
