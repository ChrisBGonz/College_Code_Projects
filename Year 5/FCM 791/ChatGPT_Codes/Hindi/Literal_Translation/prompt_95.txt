import sys

def count_lines(path):
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    count = data.count(b'\n')
    if data[-1:] != b'\n':
        count += 1
    return count

def main():
    if len(sys.argv) < 2:
        print("Usage: python count_lines.py <file>")
        return
    path = sys.argv[1]
    print(count_lines(path))

if __name__ == "__main__":
    main()
