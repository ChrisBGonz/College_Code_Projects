const { exec } = require('child_process');

function pingHost(domain_name) {
  return new Promise((resolve) => {
    if (typeof domain_name !== 'string' || !domain_name.trim()) {
      resolve(false);
      return;
    }
    const isWin = process.platform === 'win32';
    const cmd = isWin ? `ping -n 1 ${domain_name}` : `ping -c 1 ${domain_name}`;
    exec(cmd, { timeout: 5000 }, (error) => {
      resolve(!error);
    });
  });
}

module.exports = pingHost;
