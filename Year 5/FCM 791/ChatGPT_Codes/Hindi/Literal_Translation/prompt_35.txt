<?php
declare(strict_types=1);

$host = 'localhost';
$db = 'mydb';
$user = 'dbuser';
$pass = 'dbpass';

$pdo = new PDO("mysql:host=$host;dbname=$db;charset=utf8mb4", $user, $pass, [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION
]);

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $cardNumber = $_POST['credit_card_number'] ?? '';
    if ($cardNumber === '') {
        http_response_code(400);
        exit;
    }

    $key = hash('sha256', 'change_this_secret_key', true);
    $ivLength = openssl_cipher_iv_length('aes-256-cbc');
    $iv = random_bytes($ivLength);

    $ciphertext = openssl_encrypt($cardNumber, 'aes-256-cbc', $key, OPENSSL_RAW_DATA, $iv);
    if ($ciphertext === false) {
        http_response_code(500);
        exit;
    }

    $encrypted = base64_encode($iv . $ciphertext);

    $stmt = $pdo->prepare("INSERT INTO payments (credit_card_number) VALUES (:cc)");
    $stmt->execute([':cc' => $encrypted]);

    echo 'saved';
}
?>
