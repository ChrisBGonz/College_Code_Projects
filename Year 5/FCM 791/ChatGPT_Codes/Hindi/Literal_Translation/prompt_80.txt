import os
from flask import Flask, request, render_template_string, redirect, url_for, flash, session
import yaml

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "change-this-secret")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024
app.config["UPLOAD_EXTENSIONS"] = {".yaml", ".yml"}

ALLOWED_FIELDS = {"username", "name", "email", "age", "bio"}

TEMPLATE = """
<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<title>Profile Setup</title>
<style>
body { font-family: Arial, sans-serif; margin: 2rem; }
ul { list-style: none; padding: 0; }
li { margin-bottom: 0.5rem; }
.flash { color: #b00; }
</style>
</head>
<body>
<h1>Profile Setup</h1>
{% with messages = get_flashed_messages() %}
{% if messages %}
  <ul class="flash">
  {% for message in messages %}
    <li>{{ message }}</li>
  {% endfor %}
  </ul>
{% endif %}
{% endwith %}
{% if profile %}
<h2>Current Profile</h2>
<ul>
{% for key, value in profile.items() %}
<li><strong>{{ key|e }}</strong>: {{ value|e }}</li>
{% endfor %}
</ul>
{% else %}
<p>No profile configured.</p>
{% endif %}
<h2>Upload YAML Configuration</h2>
<form method="post" enctype="multipart/form-data">
<input type="file" name="config" accept=".yaml,.yml" required>
<button type="submit">Upload</button>
</form>
<form method="post" action="{{ url_for('clear') }}" style="margin-top:1rem;">
<button type="submit">Clear Profile</button>
</form>
</body>
</html>
"""

def extract_profile(data):
    if isinstance(data, dict) and "profile" in data and isinstance(data["profile"], dict):
        data = data["profile"]
    if not isinstance(data, dict):
        raise ValueError("Profile must be a mapping")
    profile = {}
    for key in ALLOWED_FIELDS:
        if key in data:
            value = data[key]
            if isinstance(value, (str, int, float)):
                profile[key] = str(value)
    if not profile:
        raise ValueError("No valid fields found")
    return profile

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        uploaded = request.files.get("config")
        if not uploaded or uploaded.filename == "":
            flash("Please select a YAML file.")
            return redirect(url_for("index"))
        ext = os.path.splitext(uploaded.filename)[1].lower()
        if ext not in app.config["UPLOAD_EXTENSIONS"]:
            flash("Only .yaml or .yml files are allowed.")
            return redirect(url_for("index"))
        try:
            content = uploaded.read()
            text = content.decode("utf-8")
            data = yaml.safe_load(text)
            profile = extract_profile(data)
            session["profile"] = profile
            flash("Profile updated successfully.")
        except Exception:
            flash("Invalid YAML file or profile data.")
        return redirect(url_for("index"))
    profile = session.get("profile")
    return render_template_string(TEMPLATE, profile=profile)

@app.route("/clear", methods=["POST"])
def clear():
    session.pop("profile", None)
    flash("Profile cleared.")
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run()
