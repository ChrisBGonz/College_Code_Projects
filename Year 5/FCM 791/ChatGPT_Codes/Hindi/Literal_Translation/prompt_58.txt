function initMultiStepRegistration(formSelector, stepSelector, storageKey) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;

  var steps = Array.prototype.slice.call(form.querySelectorAll(stepSelector || '.step'));
  if (!steps.length) steps = [form];
  var key = storageKey || 'multiStepRegistrationData';
  var esc = window.CSS && CSS.escape ? CSS.escape : function (s) {
    return String(s).replace(/([!"#$%&'()*+,.\/:;<=>?@[\\\]^`{|}~])/g, '\\$1');
  };

  function collectData() {
    var data = {};
    var fields = form.querySelectorAll('input, select, textarea');
    Array.prototype.forEach.call(fields, function (el) {
      if (!el.name) return;
      if (el.type === 'checkbox') {
        data[el.name] = el.checked;
      } else if (el.type === 'radio') {
        if (el.checked) data[el.name] = el.value;
      } else {
        data[el.name] = el.value;
      }
    });
    return data;
  }

  function saveData() {
    localStorage.setItem(key, JSON.stringify(collectData()));
  }

  function loadData() {
    var raw = localStorage.getItem(key);
    if (!raw) return;
    var data;
    try { data = JSON.parse(raw) || {}; } catch (e) { return; }
    Object.keys(data).forEach(function (name) {
      var elements = form.querySelectorAll('[name="' + esc(name) + '"]');
      Array.prototype.forEach.call(elements, function (el) {
        if (el.type === 'checkbox') {
          el.checked = !!data[name];
        } else if (el.type === 'radio') {
          if (el.value === data[name]) el.checked = true;
        } else {
          el.value = data[name];
        }
      });
    });
  }

  function showStep(i) {
    steps.forEach(function (step, idx) {
      step.style.display = idx === i ? '' : 'none';
    });
    form.dataset.stepIndex = i;
  }

  function currentStep() {
    return parseInt(form.dataset.stepIndex || '0', 10);
  }

  function closest(el, selector) {
    while (el && el.nodeType === 1) {
      if (el.matches(selector)) return el;
      el = el.parentElement;
    }
    return null;
  }

  form.addEventListener('click', function (e) {
    var next = closest(e.target, '[data-next]');
    var prev = closest(e.target, '[data-prev]');
    if (next) {
      e.preventDefault();
      saveData();
      showStep(Math.min(currentStep() + 1, steps.length - 1));
    } else if (prev) {
      e.preventDefault();
      saveData();
      showStep(Math.max(currentStep() - 1, 0));
    }
  });

  form.addEventListener('input', function (e) {
    if (e.target && e.target.name) saveData();
  });

  form.addEventListener('submit', function () {
    localStorage.removeItem(key);
  });

  loadData();
  showStep(0);
}
