require 'openssl'
require 'base64'

def sign_payload_pss(payload, private_key_pem, hash_alg = 'SHA256')
  payload = payload.to_s
  rsa = private_key_pem.is_a?(OpenSSL::PKey::RSA) ? private_key_pem : OpenSSL::PKey::RSA.new(private_key_pem)
  digest = OpenSSL::Digest.new(hash_alg)
  signature = rsa.sign_pss(digest, payload, salt_length: digest.digest_length, mgf1_hash: digest)
  Base64.strict_encode64(signature)
end
