using System;
using System.IO;
using System.Text.Json;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
    public string PhoneNumber { get; set; }
    public string Email { get; set; }
}

public class MfaChangeLogEntry
{
    public string UserId { get; set; }
    public MfaSettings Before { get; set; }
    public MfaSettings After { get; set; }
    public DateTimeOffset Timestamp { get; set; }
    public string IpAddress { get; set; }
    public string DeviceInfo { get; set; }
}

public interface IMfaSettingsRepository
{
    MfaSettings Get(string userId);
    void Save(string userId, MfaSettings settings);
}

public static class MfaAuditLog
{
    private static readonly object _sync = new object();
    private static readonly string _filePath = "mfa-audit.log";

    public static void LogChange(string userId, MfaSettings before, MfaSettings after, string ipAddress, string deviceInfo)
    {
        var entry = new MfaChangeLogEntry
        {
            UserId = userId,
            Before = before,
            After = after,
            Timestamp = DateTimeOffset.UtcNow,
            IpAddress = ipAddress,
            DeviceInfo = deviceInfo
        };

        var line = JsonSerializer.Serialize(entry);
        lock (_sync)
        {
            File.AppendAllText(_filePath, line + Environment.NewLine);
        }
    }
}

public class MfaService
{
    private readonly IMfaSettingsRepository _repo;

    public MfaService(IMfaSettingsRepository repo)
    {
        _repo = repo;
    }

    public void UpdateMfaSettings(string userId, MfaSettings newSettings, string ipAddress, string deviceInfo)
    {
        var before = _repo.Get(userId);
        _repo.Save(userId, newSettings);
        MfaAuditLog.LogChange(userId, before, newSettings, ipAddress, deviceInfo);
    }
}
