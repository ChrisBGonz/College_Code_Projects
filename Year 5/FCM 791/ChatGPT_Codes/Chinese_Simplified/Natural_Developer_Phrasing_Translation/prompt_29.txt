#!/usr/bin/env python3
import os
import sys
import subprocess
import argparse
import datetime
import difflib
import shlex

SKIP_DIRS = {
    ".git",
    ".svn",
    ".hg",
    "target",
    ".idea",
    ".mvn",
    "node_modules",
    "build",
    "out",
    ".gradle",
}

def find_root_with_pom(start):
    path = os.path.abspath(start)
    if os.path.isfile(path) and os.path.basename(path) == "pom.xml":
        return os.path.dirname(path)
    while True:
        if os.path.isfile(os.path.join(path, "pom.xml")):
            return path
        parent = os.path.dirname(path)
        if parent == path:
            return None
        path = parent

def find_poms(root):
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in SKIP_DIRS and not d.startswith(".")]
        if "pom.xml" in filenames:
            yield os.path.join(dirpath, "pom.xml")

def read_file_lines(path):
    try:
        with open(path, "r", encoding="utf-8", errors="ignore") as f:
            return f.read().splitlines()
    except Exception:
        return []

def snapshot_poms(root):
    poms = {}
    for pom in find_poms(root):
        poms[pom] = read_file_lines(pom)
    return poms

def run_cmd(cmd, cwd, log):
    log.write(">>> {}\n".format(" ".join(cmd)))
    log.flush()
    try:
        proc = subprocess.Popen(
            cmd, cwd=cwd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, bufsize=1
        )
    except FileNotFoundError:
        msg = "Command not found: {}\n".format(cmd[0])
        log.write(msg)
        log.flush()
        sys.stderr.write(msg)
        return 127
    for line in proc.stdout:
        log.write(line)
        log.flush()
        sys.stdout.write(line)
        sys.stdout.flush()
    rc = proc.wait()
    log.write("<<< exit code {}\n".format(rc))
    log.flush()
    return rc

def generate_report(before, after, root, report_path):
    changed = []
    with open(report_path, "w", encoding="utf-8") as report:
        report.write("Maven update report generated at {}\n\n".format(datetime.datetime.now().isoformat()))
        all_paths = sorted(set(before.keys()) | set(after.keys()))
        for path in all_paths:
            old = before.get(path, [])
            new = after.get(path, [])
            if old != new:
                rel = os.path.relpath(path, root)
                changed.append(rel)
                diff = difflib.unified_diff(
                    old, new, fromfile=rel + " (before)", tofile=rel + " (after)", lineterm=""
                )
                for line in diff:
                    report.write(line + "\n")
                report.write("\n")
        if not changed:
            report.write("No changes detected.\n")
    return changed

def main():
    parser = argparse.ArgumentParser(
        description="Batch update Maven dependencies/plugins/parent/dependencyManagement to latest versions."
    )
    parser.add_argument("--root", default=".", help="Project root directory")
    parser.add_argument("--mvn", default=os.environ.get("MVN_CMD", "mvn"), help="Maven command (mvn or mvnw)")
    parser.add_argument("--mvn-args", default=os.environ.get("MVN_ARGS", ""), help="Extra Maven arguments")
    parser.add_argument("--allow-snapshots", action="store_true", help="Allow snapshot versions")
    parser.add_argument("--skip-build", action="store_true", help="Skip build verification")
    parser.add_argument(
        "--build-goals", default=os.environ.get("MVN_BUILD_GOALS", "clean test"), help="Build goals"
    )
    parser.add_argument("--log", default=None, help="Log file path")
    parser.add_argument("--report", default=None, help="Report file path")
    args = parser.parse_args()

    root = find_root_with_pom(args.root)
    if not root:
        sys.stderr.write("pom.xml not found in {} or its parent directories\n".format(os.path.abspath(args.root)))
        return 2
    root = os.path.abspath(root)

    mvn_cmd = args.mvn
    if args.mvn == "mvn":
        wrapper = os.path.join(root, "mvnw")
        wrapper_cmd = wrapper if os.path.isfile(wrapper) else os.path.join(root, "mvnw.cmd")
        if os.path.isfile(wrapper_cmd):
            mvn_cmd = wrapper_cmd

    timestamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
    log_path = args.log or os.path.join(root, "mvn-update-{}.log".format(timestamp))
    report_path = args.report or os.path.join(root, "mvn-update-report-{}.diff".format(timestamp))

    extra_args = shlex.split(args.mvn_args) if args.mvn_args else []
    base_cmd = [mvn_cmd, "-U", "-B"] + extra_args

    before = snapshot_poms(root)

    with open(log_path, "w", encoding="utf-8") as log:
        log.write("Root: {}\n".format(root))
        log.write("Timestamp: {}\n".format(datetime.datetime.now().isoformat()))
        log.write("Maven command: {}\n".format(mvn_cmd))
        log.write("Extra args: {}\n".format(" ".join(extra_args)))
        log.flush()

        allow_snapshot_flag = "-DallowSnapshots=true" if args.allow_snapshots else "-DallowSnapshots=false"

        process_flags = [
            "-DprocessDependencyManagement=true",
            "-DprocessPlugins=true",
            "-DprocessPluginManagement=true",
            "-DprocessPluginDependencies=true",
        ]

        commands = [
            base_cmd + ["versions:update-parent", "-DgenerateBackupPoms=false", allow_snapshot_flag],
            base_cmd + ["versions:use-latest-versions", "-DgenerateBackupPoms=false", allow_snapshot_flag]
            + process_flags,
            base_cmd + ["versions:update-properties", "-DgenerateBackupPoms=false", allow_snapshot_flag]
            + process_flags,
        ]

        exit_code = 0
        for cmd in commands:
            rc = run_cmd(cmd, root, log)
            if rc != 0:
                exit_code = rc
                break

        after = snapshot_poms(root)
        changed = generate_report(before, after, root, report_path)

        log.write("Changed files: {}\n".format(len(changed)))
        if changed:
            log.write("\n".join(changed) + "\n")
        log.flush()

        if exit_code == 0 and not args.skip_build:
            build_cmd = base_cmd + shlex.split(args.build_goals)
            rc = run_cmd(build_cmd, root, log)
            if rc != 0:
                exit_code = rc
        elif args.skip_build:
            log.write("Build skipped.\n")
            log.flush()
        else:
            log.write("Build skipped due to previous errors.\n")
            log.flush()

    sys.stdout.write("Log file: {}\n".format(log_path))
    sys.stdout.write("Report file: {}\n".format(report_path))
    sys.stdout.write("Changed POMs: {}\n".format(len(changed)))
    return exit_code

if __name__ == "__main__":
    sys.exit(main())
