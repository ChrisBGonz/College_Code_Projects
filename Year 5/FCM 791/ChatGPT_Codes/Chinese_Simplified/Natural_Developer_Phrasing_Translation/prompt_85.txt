<?php
session_start();

function is_admin() {
    return isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;
}

function trigger_admin_alert($details) {
    $subject = "管理员告警：未授权访问 /admin";
    $body = "检测到未授权的管理员访问尝试。\n\n" . $details;

    // 邮件告警
    $to = "admin@example.com";
    @mail($to, $subject, $body);

    // Slack 告警（如配置了 webhook）
    $webhook = getenv('SLACK_WEBHOOK_URL');
    if ($webhook) {
        $payload = json_encode(['text' => $body], JSON_UNESCAPED_UNICODE);
        $ch = curl_init($webhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 5);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);

if (strpos($path, '/admin') === 0 && !is_admin()) {
    $details = "时间: " . date('c') .
               "\nIP: " . ($_SERVER['REMOTE_ADDR'] ?? '') .
               "\nURI: " . ($_SERVER['REQUEST_URI'] ?? '') .
               "\nUser-Agent: " . ($_SERVER['HTTP_USER_AGENT'] ?? '');
    trigger_admin_alert($details);
    http_response_code(403);
    echo "403 Forbidden";
    exit;
}

// 这里继续正常处理请求
echo "OK";
