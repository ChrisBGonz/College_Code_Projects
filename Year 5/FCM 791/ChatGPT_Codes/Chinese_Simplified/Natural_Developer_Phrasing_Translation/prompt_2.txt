const express = require('express');
const app = express();

app.use(express.json());

const users = new Map();

function isValidEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

async function updateProfile(req, res) {
  try {
    const userId = (req.user && req.user.id) || req.body.userId;
    if (!userId) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    let { email, displayName } = req.body;

    if (email === undefined && displayName === undefined) {
      return res.status(400).json({ error: 'No fields to update' });
    }

    if (email !== undefined) {
      if (typeof email !== 'string') {
        return res.status(400).json({ error: 'Email must be a string' });
      }
      email = email.trim();
      if (!isValidEmail(email)) {
        return res.status(400).json({ error: 'Invalid email' });
      }
    }

    if (displayName !== undefined) {
      if (typeof displayName !== 'string') {
        return res.status(400).json({ error: 'Display name must be a string' });
      }
      displayName = displayName.trim();
      if (displayName.length === 0) {
        return res.status(400).json({ error: 'Display name cannot be empty' });
      }
    }

    const user = users.get(userId);
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    if (email !== undefined) user.email = email;
    if (displayName !== undefined) user.displayName = displayName;

    users.set(userId, user);

    return res.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        displayName: user.displayName
      }
    });
  } catch (err) {
    return res.status(500).json({ error: 'Internal Server Error' });
  }
}

app.post('/api/user/update-profile', updateProfile);

module.exports = { app, updateProfile, users };
