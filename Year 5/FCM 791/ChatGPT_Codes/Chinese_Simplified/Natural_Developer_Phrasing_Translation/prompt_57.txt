<?php
declare(strict_types=1);
session_start();
header('Content-Type: text/html; charset=UTF-8');

$config = [
    'upload_dir' => '/mnt/shared/uploads',
    'max_size' => 10 * 1024 * 1024,
    'allowed_mime' => ['application/pdf','application/x-pdf','application/acrobat','applications/vnd.pdf'],
    'allowed_ext' => ['pdf'],
];

$users = [
    'employee1' => '$2y$10$92IXUNpkjO0rOQ5byMi.Ye4oKoEa3Ro9llC/.og/at2.uheWG/igi',
    'employee2' => '$2y$10$92IXUNpkjO0rOQ5byMi.Ye4oKoEa3Ro9llC/.og/at2.uheWG/igi',
];

if (empty($_SESSION['csrf'])) {
    $_SESSION['csrf'] = bin2hex(random_bytes(32));
}

function e($value): string {
    return htmlspecialchars((string)$value, ENT_QUOTES, 'UTF-8');
}

function isLoggedIn(): bool {
    return !empty($_SESSION['user']);
}

function checkCsrf(string $token): bool {
    return isset($_SESSION['csrf']) && hash_equals($_SESSION['csrf'], $token);
}

function renderHeader(string $title): void {
    echo "<!DOCTYPE html><html><head><meta charset='UTF-8'><title>".e($title)."</title></head><body>";
}

function renderFooter(): void {
    echo "</body></html>";
}

function showLogin(string $error = ''): void {
    renderHeader('登录');
    if ($error !== '') {
        echo "<p style='color:red;'>".e($error)."</p>";
    }
    $csrf = $_SESSION['csrf'] ?? '';
    echo "<form method='post' action=''>
            <input type='hidden' name='action' value='login'>
            <input type='hidden' name='csrf' value='".e($csrf)."'>
            <label>用户名: <input type='text' name='username' required></label><br>
            <label>密码: <input type='password' name='password' required></label><br>
            <button type='submit'>登录</button>
          </form>";
    renderFooter();
}

function showUploadForm(string $message = '', string $error = ''): void {
    renderHeader('文件上传');
    $csrf = $_SESSION['csrf'] ?? '';
    $user = $_SESSION['user'] ?? '';
    echo "<p>已登录用户: ".e($user)."</p>";
    echo "<form method='post' style='display:inline;'>
            <input type='hidden' name='action' value='logout'>
            <input type='hidden' name='csrf' value='".e($csrf)."'>
            <button type='submit'>退出</button>
          </form>";
    if ($message !== '') {
        echo "<p style='color:green;'>".e($message)."</p>";
    }
    if ($error !== '') {
        echo "<p style='color:red;'>".e($error)."</p>";
    }
    echo "<form method='post' enctype='multipart/form-data'>
            <input type='hidden' name='action' value='upload'>
            <input type='hidden' name='csrf' value='".e($csrf)."'>
            <label>选择PDF文件: <input type='file' name='file' accept='application/pdf' required></label><br>
            <button type='submit'>上传</button>
          </form>";
    renderFooter();
}

$action = $_POST['action'] ?? '';

if ($action === 'login') {
    if (!checkCsrf($_POST['csrf'] ?? '')) {
        showLogin('无效的请求');
        exit;
    }
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';
    if ($username === '' || $password === '') {
        showLogin('请输入用户名和密码');
        exit;
    }
    if (isset($users[$username]) && password_verify($password, $users[$username])) {
        session_regenerate_id(true);
        $_SESSION['user'] = $username;
        showUploadForm('登录成功');
        exit;
    }
    showLogin('用户名或密码错误');
    exit;
}

if ($action === 'logout') {
    if (!checkCsrf($_POST['csrf'] ?? '')) {
        showLogin('无效的请求');
        exit;
    }
    $_SESSION = [];
    if (ini_get('session.use_cookies')) {
        $params = session_get_cookie_params();
        setcookie(session_name(), '', time() - 42000, $params['path'], $params['domain'], $params['secure'], $params['httponly']);
    }
    session_destroy();
    session_start();
    $_SESSION['csrf'] = bin2hex(random_bytes(32));
    showLogin('已退出');
    exit;
}

if ($action === 'upload') {
    if (!isLoggedIn()) {
        showLogin('请先登录');
        exit;
    }
    if (!checkCsrf($_POST['csrf'] ?? '')) {
        showUploadForm('', '无效的请求');
        exit;
    }

    $error = '';
    $success = '';
    $file = $_FILES['file'] ?? null;

    if ($file === null || !isset($file['error'])) {
        $error = '未选择文件';
    } elseif ($file['error'] !== UPLOAD_ERR_OK) {
        $map = [
            UPLOAD_ERR_INI_SIZE => '文件超过服务器限制',
            UPLOAD_ERR_FORM_SIZE => '文件超过表单限制',
            UPLOAD_ERR_PARTIAL => '文件只上传了一部分',
            UPLOAD_ERR_NO_FILE => '未选择文件',
            UPLOAD_ERR_NO_TMP_DIR => '缺少临时文件夹',
            UPLOAD_ERR_CANT_WRITE => '无法写入磁盘',
            UPLOAD_ERR_EXTENSION => '上传被扩展阻止',
        ];
        $error = $map[$file['error']] ?? '上传失败';
    } elseif ($file['size'] <= 0) {
        $error = '文件为空';
    } elseif ($file['size'] > $config['max_size']) {
        $error = '文件过大';
    } else {
        $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        if (!in_array($ext, $config['allowed_ext'], true)) {
            $error = '仅允许上传PDF文件';
        } else {
            $finfo = new finfo(FILEINFO_MIME_TYPE);
            $mime = $finfo->file($file['tmp_name']);
            if ($mime === false || !in_array($mime, $config['allowed_mime'], true)) {
                $error = '文件类型无效';
            } else {
                $fh = @fopen($file['tmp_name'], 'rb');
                $header = $fh ? fread($fh, 4) : '';
                if ($fh) {
                    fclose($fh);
                }
                if ($header !== '%PDF') {
                    $error = '文件内容不是有效的PDF';
                } else {
                    $dir = rtrim($config['upload_dir'], DIRECTORY_SEPARATOR);
                    if ($dir === '' || !is_dir($dir) || !is_writable($dir)) {
                        $error = '上传目录不可用';
                    } else {
                        $base = pathinfo($file['name'], PATHINFO_FILENAME);
                        $base = preg_replace('/[^\pL\pN_\-\.]+/u', '_', $base);
                        $base = trim($base, '._-');
                        if ($base === '') {
                            $base = 'file';
                        }
                        $filename = $base . '_' . date('Ymd_His') . '_' . bin2hex(random_bytes(4)) . '.pdf';
                        $dest = $dir . DIRECTORY_SEPARATOR . $filename;
                        if (!move_uploaded_file($file['tmp_name'], $dest)) {
                            $error = '保存文件失败';
                        } else {
                            @chmod($dest, 0644);
                            $success = '上传成功: ' . $filename;
                        }
                    }
                }
            }
        }
    }

    showUploadForm($success, $error);
    exit;
}

if (isLoggedIn()) {
    showUploadForm();
} else {
    showLogin();
}
?>
