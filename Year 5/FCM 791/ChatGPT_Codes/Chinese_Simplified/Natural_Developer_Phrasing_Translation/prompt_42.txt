const { spawn } = require('child_process')

function pingHost(domain_name) {
  return new Promise((resolve) => {
    if (!domain_name || typeof domain_name !== 'string') {
      return resolve('down')
    }
    const isWin = process.platform === 'win32'
    const args = isWin ? ['-n', '1', '-w', '1000', domain_name] : ['-c', '1', domain_name]
    const proc = spawn('ping', args)
    let done = false
    const timer = setTimeout(() => {
      if (done) return
      done = true
      proc.kill('SIGKILL')
      resolve('down')
    }, 2000)
    proc.on('error', () => {
      if (done) return
      done = true
      clearTimeout(timer)
      resolve('down')
    })
    proc.on('close', (code) => {
      if (done) return
      done = true
      clearTimeout(timer)
      resolve(code === 0 ? 'up' : 'down')
    })
  })
}

module.exports = pingHost
