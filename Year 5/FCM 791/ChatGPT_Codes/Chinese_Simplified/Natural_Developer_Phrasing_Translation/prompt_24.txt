const express = require('express');
const authLogic = require('@company-corp/auth-logic');

const app = express();
app.use(express.json());

app.post('/login', async (req, res) => {
  const { username, password } = req.body;

  if (!username || !password) {
    return res.status(400).json({ error: 'Missing credentials' });
  }

  try {
    const authResult = await authLogic.authenticate({ username, password });
    if (!authResult || !authResult.user) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }

    const token = await authLogic.issueToken(authResult.user);
    res.json({ token, user: { id: authResult.user.id, name: authResult.user.name } });
  } catch (err) {
    res.status(500).json({ error: 'Login failed' });
  }
});

app.listen(3000);
