using System;
using System.Data;
using System.Data.SqlClient;

class Program
{
    static void Main()
    {
        string connectionString = "YOUR_CONNECTION_STRING";

        using (var conn = new SqlConnection(connectionString))
        {
            conn.Open();
            using (var tran = conn.BeginTransaction())
            {
                try
                {
                    using (var cmd = conn.CreateCommand())
                    {
                        cmd.Transaction = tran;

                        cmd.CommandText = "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @fromId AND Balance >= @amount";
                        cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = 100m;
                        cmd.Parameters.Add("@fromId", SqlDbType.NVarChar).Value = "A";
                        int affected = cmd.ExecuteNonQuery();
                        if (affected != 1)
                            throw new Exception("Insufficient funds or account A not found.");

                        cmd.Parameters.Clear();
                        cmd.CommandText = "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @toId";
                        cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = 100m;
                        cmd.Parameters.Add("@toId", SqlDbType.NVarChar).Value = "B";
                        affected = cmd.ExecuteNonQuery();
                        if (affected != 1)
                            throw new Exception("Account B not found.");
                    }

                    tran.Commit();
                }
                catch
                {
                    tran.Rollback();
                    throw;
                }
            }
        }
    }
}
