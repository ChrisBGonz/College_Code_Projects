import sys
import time
import os
from collections import deque
import argparse

def monitor(path):
    window = 60
    threshold = 10
    timestamps = deque()
    alerted = False
    f = None
    inode = None

    while True:
        try:
            f = open(path, 'r', encoding='utf-8', errors='ignore')
            f.seek(0, os.SEEK_END)
            inode = os.fstat(f.fileno()).st_ino
            break
        except FileNotFoundError:
            time.sleep(1)

    while True:
        line = f.readline()
        if line:
            if "Failed Login" in line:
                timestamps.append(time.time())
        else:
            time.sleep(0.2)
            try:
                stat = os.stat(path)
            except FileNotFoundError:
                continue
            if inode is not None and stat.st_ino != inode:
                try:
                    f.close()
                except Exception:
                    pass
                try:
                    f = open(path, 'r', encoding='utf-8', errors='ignore')
                    f.seek(0, os.SEEK_END)
                    inode = os.fstat(f.fileno()).st_ino
                except FileNotFoundError:
                    time.sleep(1)
                continue
            if stat.st_size < f.tell():
                f.seek(0, os.SEEK_END)
        now = time.time()
        cutoff = now - window
        while timestamps and timestamps[0] < cutoff:
            timestamps.popleft()
        if len(timestamps) > threshold:
            if not alerted:
                ts = time.strftime('%Y-%m-%d %H:%M:%S', time.localtime(now))
                print(f"{ts} ALERT: Failed Login entries exceeded {threshold} within 1 minute.")
                sys.stdout.flush()
                alerted = True
        else:
            alerted = False

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('logfile')
    args = parser.parse_args()
    monitor(args.logfile)

if __name__ == '__main__':
    try:
        main()
    except KeyboardInterrupt:
        pass
