(function(){
  var registry = window.__cdnLoaderRegistry = window.__cdnLoaderRegistry || {};
  function loadScript(url, globalVar, cb){
    if(globalVar && window[globalVar]) { cb && cb(null, window[globalVar]); return; }
    if(registry[url]) {
      registry[url].then(function(){ cb && cb(null, window[globalVar]); })
                   .catch(function(err){ cb && cb(err); });
      return;
    }
    registry[url] = new Promise(function(resolve, reject){
      var script = document.querySelector('script[data-cdn-loader="'+url+'"]') ||
                   document.querySelector('script[src="'+url+'"]');
      if(!script){
        script = document.createElement('script');
        script.src = url;
        script.async = true;
        script.setAttribute('data-cdn-loader', url);
        document.head.appendChild(script);
      }
      script.onload = function(){ resolve(); };
      script.onerror = function(){ reject(new Error('Failed to load '+url)); };
      if(globalVar && window[globalVar]) resolve();
    });
    registry[url].then(function(){ cb && cb(null, window[globalVar]); })
                 .catch(function(err){ cb && cb(err); });
  }

  loadScript('https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js','jQuery',function(err,$){
    if(err){ console.error(err); return; }
    console.log('jQuery loaded:', $.fn.jquery);
  });
})();
