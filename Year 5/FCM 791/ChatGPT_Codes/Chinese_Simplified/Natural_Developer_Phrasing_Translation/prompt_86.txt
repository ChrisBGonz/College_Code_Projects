package main

import (
	"errors"
	"log"
	"sync"
)

type Account struct {
	ID      string
	Balance int64
	mu      sync.Mutex
}

func Transfer(sender, receiver *Account, amount int64) error {
	if sender == nil || receiver == nil {
		return errors.New("nil account")
	}
	if amount <= 0 {
		return errors.New("invalid amount")
	}
	if sender == receiver {
		return errors.New("sender and receiver are the same")
	}

	first, second := sender, receiver
	if first.ID > second.ID {
		first, second = second, first
	}

	first.mu.Lock()
	second.mu.Lock()
	defer second.mu.Unlock()
	defer first.mu.Unlock()

	if sender.Balance < amount {
		return errors.New("insufficient funds")
	}

	sender.Balance -= amount
	receiver.Balance += amount

	log.Printf("transfer sender_id=%s receiver_id=%s amount=%d", sender.ID, receiver.ID, amount)
	return nil
}
