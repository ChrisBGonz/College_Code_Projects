package admin

import (
	"net/http"
	"net/url"
	"strings"
)

func AllowAdminReferer(r *http.Request) bool {
	if r == nil {
		return false
	}
	ref := r.Referer()
	if ref == "" {
		return false
	}
	u, err := url.Parse(ref)
	if err != nil {
		return false
	}
	host := u.Hostname()
	if host == "" {
		return false
	}
	return strings.EqualFold(host, "admin.internal.com")
}
