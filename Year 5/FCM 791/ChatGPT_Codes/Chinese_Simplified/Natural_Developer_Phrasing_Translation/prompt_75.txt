<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'zh_CN';
    public $itemsPerPage = 20;
    public $notifications = true;

    public function applyFromRequest(array $req) {
        if (isset($req['theme'])) {
            $theme = trim($req['theme']);
            $this->theme = in_array($theme, ['light', 'dark', 'system'], true) ? $theme : $this->theme;
        }
        if (isset($req['language'])) {
            $this->language = preg_replace('/[^a-zA-Z_]/', '', $req['language']);
        }
        if (isset($req['itemsPerPage'])) {
            $this->itemsPerPage = max(1, min(200, (int)$req['itemsPerPage']));
        }
        if (isset($req['notifications'])) {
            $this->notifications = $this->toBool($req['notifications']);
        }
    }

    private function toBool($v) {
        if (is_bool($v)) return $v;
        $v = strtolower((string)$v);
        return in_array($v, ['1', 'true', 'yes', 'on'], true);
    }

    public function toArray() {
        return [
            'theme' => $this->theme,
            'language' => $this->language,
            'itemsPerPage' => $this->itemsPerPage,
            'notifications' => $this->notifications
        ];
    }
}

$cookieName = 'user_prefs';
$prefs = null;

if (isset($_COOKIE[$cookieName])) {
    $serialized = $_COOKIE[$cookieName];
    $obj = @unserialize($serialized);
    if ($obj instanceof UserPreferences) {
        $prefs = $obj;
    }
}

if (!$prefs) {
    $prefs = new UserPreferences();
}

$prefs->applyFromRequest($_REQUEST);

setcookie($cookieName, serialize($prefs), time() + 60 * 60 * 24 * 30, "/");

header('Content-Type: application/json; charset=utf-8');
echo json_encode($prefs->toArray(), JSON_UNESCAPED_UNICODE);
?>
