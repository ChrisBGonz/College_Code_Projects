function setupMultiStepRegistrationForm(opts) {
  opts = opts || {};
  var form = typeof opts.form === 'string' ? document.querySelector(opts.form) : opts.form;
  if (!form) return;
  var stepSelector = opts.stepSelector || '[data-step]';
  var steps = Array.prototype.slice.call(form.querySelectorAll(stepSelector));
  if (!steps.length) steps = [form];
  var storageKey = opts.storageKey || 'msf:' + (form.id || location.pathname);
  var submitHandler = opts.onSubmit || opts.submit;
  var saveDelay = typeof opts.saveDelay === 'number' ? opts.saveDelay : 200;

  function readCache() {
    try {
      var raw = localStorage.getItem(storageKey);
      if (raw) return JSON.parse(raw);
    } catch (e) {}
    return { data: {}, step: 0 };
  }

  function writeCache() {
    try {
      localStorage.setItem(storageKey, JSON.stringify(cache));
    } catch (e) {}
  }

  function clearCache() {
    try {
      localStorage.removeItem(storageKey);
    } catch (e) {}
    cache = { data: {}, step: 0 };
  }

  function escapeSelector(str) {
    if (window.CSS && CSS.escape) return CSS.escape(str);
    return str.replace(/([ #;?%&,.+*~\':"!^$[\]()=>|\/@])/g, '\\$1');
  }

  function fillFields(data) {
    if (!data) return;
    Object.keys(data).forEach(function (name) {
      var value = data[name];
      var selector = '[name="' + escapeSelector(name) + '"]';
      var els = form.querySelectorAll(selector);
      if (!els.length) return;
      Array.prototype.forEach.call(els, function (el) {
        var type = (el.type || '').toLowerCase();
        if (type === 'checkbox') {
          if (Array.isArray(value)) {
            el.checked = value.indexOf(el.value) !== -1;
          } else if (value === true || value === false || value === 1 || value === 0) {
            el.checked = !!value;
          } else if (value == null) {
            el.checked = false;
          } else {
            el.checked = el.value == value;
          }
        } else if (type === 'radio') {
          el.checked = value != null && el.value == value;
        } else if (el.tagName === 'SELECT' && el.multiple) {
          var arr = Array.isArray(value) ? value : value != null ? [value] : [];
          Array.prototype.forEach.call(el.options, function (opt) {
            opt.selected = arr.indexOf(opt.value) !== -1;
          });
        } else {
          if (value != null) el.value = value;
        }
      });
    });
  }

  function collectData(container) {
    var data = {};
    var checkGroups = {};
    var radioGroups = {};
    var fields = container.querySelectorAll('input[name], select[name], textarea[name]');
    Array.prototype.forEach.call(fields, function (el) {
      if (el.disabled) return;
      var name = el.name;
      if (!name) return;
      var type = (el.type || '').toLowerCase();
      if (type === 'checkbox') {
        if (!checkGroups[name]) checkGroups[name] = [];
        if (el.checked) checkGroups[name].push(el.value);
      } else if (type === 'radio') {
        if (!radioGroups[name]) radioGroups[name] = null;
        if (el.checked) {
          data[name] = el.value;
          radioGroups[name] = el.value;
        }
      } else if (el.tagName === 'SELECT' && el.multiple) {
        var arr = [];
        var opts = el.options;
        for (var i = 0; i < opts.length; i++) if (opts[i].selected) arr.push(opts[i].value);
        data[name] = arr;
      } else {
        data[name] = el.value;
      }
    });
    Object.keys(checkGroups).forEach(function (n) { data[n] = checkGroups[n]; });
    Object.keys(radioGroups).forEach(function (n) { if (!(n in data)) data[n] = null; });
    return data;
  }

  function saveStep(index) {
    if (index < 0 || index >= steps.length) return;
    var stepData = collectData(steps[index]);
    for (var k in stepData) cache.data[k] = stepData[k];
    cache.step = index;
    writeCache();
  }

  function saveAll() {
    var allData = collectData(form);
    for (var k in allData) cache.data[k] = allData[k];
    writeCache();
  }

  var cache = readCache();
  if (!cache || typeof cache !== 'object') cache = { data: {}, step: 0 };
  if (!cache.data || typeof cache.data !== 'object') cache.data = {};
  if (typeof cache.step !== 'number' || cache.step < 0 || cache.step >= steps.length) cache.step = 0;

  var currentStep = cache.step;

  function showStep(index) {
    if (index < 0 || index >= steps.length) return;
    currentStep = index;
    steps.forEach(function (step, i) { step.hidden = i !== index; });
    cache.step = index;
    writeCache();
  }

  var matches = Element.prototype.matches || Element.prototype.msMatchesSelector || Element.prototype.webkitMatchesSelector;
  function findClosest(el, selector) {
    while (el && el !== form) {
      if (matches && matches.call(el, selector)) return el;
      el = el.parentElement;
    }
    return null;
  }

  fillFields(cache.data);
  showStep(currentStep);

  var saveTimer;
  function scheduleSave() {
    if (saveTimer) clearTimeout(saveTimer);
    saveTimer = setTimeout(function () { saveStep(currentStep); }, saveDelay);
  }

  form.addEventListener('input', scheduleSave);
  form.addEventListener('change', scheduleSave);

  form.addEventListener('click', function (e) {
    var nextBtn = findClosest(e.target, opts.nextSelector || '[data-next]');
    if (nextBtn) {
      e.preventDefault();
      saveStep(currentStep);
      var target = nextBtn.getAttribute('data-next');
      var idx = target !== null && target !== '' ? parseInt(target, 10) : currentStep + 1;
      if (!isNaN(idx)) showStep(idx);
      return;
    }
    var prevBtn = findClosest(e.target, opts.prevSelector || '[data-prev]');
    if (prevBtn) {
      e.preventDefault();
      saveStep(currentStep);
      var target2 = prevBtn.getAttribute('data-prev');
      var idx2 = target2 !== null && target2 !== '' ? parseInt(target2, 10) : currentStep - 1;
      if (!isNaN(idx2)) showStep(idx2);
    }
  });

  form.addEventListener('submit', function (e) {
    saveAll();
    if (typeof submitHandler === 'function') {
      e.preventDefault();
      var dataObj = collectData(form);
      var result;
      try {
        result = submitHandler(dataObj, form);
      } catch (err) {
        return;
      }
      if (result && typeof result.then === 'function') {
        result.then(function (res) { if (res !== false) clearCache(); }).catch(function () {});
      } else {
        if (result !== false) {
          clearCache();
          if (opts.autoSubmit) form.submit();
        }
      }
    }
  });

  window.addEventListener('beforeunload', function () { saveAll(); });

  return {
    save: saveAll,
    clear: clearCache,
    goToStep: showStep,
    getData: function () { return JSON.parse(JSON.stringify(cache.data)); }
  };
}
