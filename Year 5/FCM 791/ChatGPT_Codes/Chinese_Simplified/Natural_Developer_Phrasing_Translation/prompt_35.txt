<?php
$dsn = 'mysql:host=localhost;dbname=test;charset=utf8mb4';
$dbUser = 'root';
$dbPass = '';

$key = hex2bin('0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef');

function encryptCreditCard($number, $key) {
    $cipher = 'AES-256-CBC';
    $ivLength = openssl_cipher_iv_length($cipher);
    $iv = random_bytes($ivLength);
    $ciphertext = openssl_encrypt($number, $cipher, $key, OPENSSL_RAW_DATA, $iv);
    return base64_encode($iv . $ciphertext);
}

$name = $_POST['name'] ?? 'John Doe';
$creditCardNumber = $_POST['credit_card_number'] ?? '4111111111111111';

$encryptedCard = encryptCreditCard($creditCardNumber, $key);

$pdo = new PDO($dsn, $dbUser, $dbPass, [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
]);

$stmt = $pdo->prepare('INSERT INTO customers (name, credit_card_number) VALUES (:name, :cc)');
$stmt->execute([
    ':name' => $name,
    ':cc' => $encryptedCard
]);
?>
