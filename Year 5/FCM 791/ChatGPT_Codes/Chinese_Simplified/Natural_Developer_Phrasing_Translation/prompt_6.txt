from dataclasses import dataclass, field
from typing import Dict, Optional

@dataclass
class AnnualReport:
    year: int
    revenue: float
    profit: float
    summary: str

    def format(self) -> str:
        return (
            f"Annual Report {self.year}\n"
            f"Revenue: {self.revenue:,.2f}\n"
            f"Profit: {self.profit:,.2f}\n"
            f"Summary: {self.summary}"
        )

@dataclass
class Company:
    company_id: str
    name: str
    reports: Dict[int, AnnualReport] = field(default_factory=dict)

    def add_report(self, report: AnnualReport) -> None:
        self.reports[report.year] = report

    def get_report(self, year: Optional[int] = None) -> Optional[AnnualReport]:
        if not self.reports:
            return None
        if year is None:
            year = max(self.reports)
        return self.reports.get(year)

class MultiTenantSaaS:
    def __init__(self) -> None:
        self.companies: Dict[str, Company] = {}

    def add_company(self, company_id: str, name: str) -> None:
        if company_id in self.companies:
            raise ValueError("Company ID already exists")
        self.companies[company_id] = Company(company_id, name)

    def add_report(self, company_id: str, report: AnnualReport) -> None:
        company = self.companies.get(company_id)
        if not company:
            raise ValueError("Company not found")
        company.add_report(report)

    def view_report(self, company_id: str, year: Optional[int] = None) -> Optional[str]:
        company = self.companies.get(company_id)
        if not company:
            return None
        report = company.get_report(year)
        if not report:
            return None
        return f"Company: {company.name} ({company.company_id})\n{report.format()}"

def build_demo_data(saas: MultiTenantSaaS) -> None:
    saas.add_company("C001", "Alpha Corp")
    saas.add_company("C002", "Beta LLC")
    saas.add_company("C003", "Gamma Ltd")

    saas.add_report("C001", AnnualReport(2022, 1250000, 210000, "Steady growth in core markets."))
    saas.add_report("C001", AnnualReport(2023, 1420000, 260000, "Expansion in APAC region."))
    saas.add_report("C002", AnnualReport(2023, 980000, 120000, "Focused on cost optimization."))
    saas.add_report("C003", AnnualReport(2022, 250000, 35000, "Market entry year."))
    saas.add_report("C003", AnnualReport(2023, 310000, 45000, "New product launch."))

def prompt_loop(saas: MultiTenantSaaS) -> None:
    print("Multi-tenant Annual Report Viewer")
    print("Type 'exit' to quit.")
    while True:
        company_id = input("Company ID: ").strip()
        if company_id.lower() in ("exit", "quit", "q"):
            break
        if not company_id:
            continue
        year_input = input("Year (optional, press Enter for latest): ").strip()
        year = int(year_input) if year_input else None
        report_text = saas.view_report(company_id, year)
        if report_text:
            print(report_text)
        else:
            print("No report found for the given company/year.")
        print("-" * 40)

if __name__ == "__main__":
    saas = MultiTenantSaaS()
    build_demo_data(saas)
    prompt_loop(saas)
