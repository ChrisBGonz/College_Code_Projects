using System;
using System.Collections.Generic;

namespace Checkout
{
    public class CartItem
    {
        public string Id { get; set; }
        public string Name { get; set; }
        public decimal UnitPrice { get; set; }
        public int Quantity { get; set; }
    }

    public class CheckoutRequest
    {
        public List<CartItem> Items { get; set; } = new List<CartItem>();
        public decimal DiscountRate { get; set; }
        public decimal DiscountAmount { get; set; }
        public decimal TaxRate { get; set; }
        public decimal ShippingFee { get; set; }
        public decimal FreeShippingThreshold { get; set; }
    }

    public class CheckoutResult
    {
        public decimal Subtotal { get; set; }
        public decimal Discount { get; set; }
        public decimal Tax { get; set; }
        public decimal Shipping { get; set; }
        public decimal Total { get; set; }
        public int TotalQuantity { get; set; }
        public int ItemCount { get; set; }
    }

    public class CheckoutCalculator
    {
        public CheckoutResult Calculate(CheckoutRequest request)
        {
            if (request == null) throw new ArgumentNullException(nameof(request));

            var items = request.Items ?? new List<CartItem>();
            decimal subtotal = 0m;
            int totalQty = 0;

            foreach (var item in items)
            {
                if (item == null) continue;

                int qty = item.Quantity;
                if (qty < 0) qty = 0;

                decimal price = item.UnitPrice;
                if (price < 0) price = 0;

                subtotal += price * qty;
                totalQty += qty;
            }

            decimal discount = 0m;
            if (request.DiscountRate > 0m) discount += subtotal * request.DiscountRate;
            if (request.DiscountAmount > 0m) discount += request.DiscountAmount;
            if (discount > subtotal) discount = subtotal;

            decimal taxable = subtotal - discount;
            if (taxable < 0m) taxable = 0m;

            decimal tax = request.TaxRate > 0m ? taxable * request.TaxRate : 0m;

            decimal shipping = request.ShippingFee;
            if (request.FreeShippingThreshold > 0m && taxable >= request.FreeShippingThreshold) shipping = 0m;
            if (shipping < 0m) shipping = 0m;

            decimal total = taxable + tax + shipping;

            return new CheckoutResult
            {
                Subtotal = subtotal,
                Discount = discount,
                Tax = tax,
                Shipping = shipping,
                Total = total,
                TotalQuantity = totalQty,
                ItemCount = items.Count
            };
        }
    }
}
