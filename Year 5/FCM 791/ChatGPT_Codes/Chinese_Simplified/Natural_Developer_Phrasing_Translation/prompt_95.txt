import sys

def count_lines(filename):
    with open(filename, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    count = data.count(b'\n')
    if data[-1] != ord('\n'):
        count += 1
    return count

if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.stderr.write("Usage: python script.py <filename>\n")
        sys.exit(1)
    print(count_lines(sys.argv[1]))
