function initAutoLogout(options) {
  options = options || {};
  var timeout = typeof options.timeout === 'number' ? options.timeout : 15 * 60 * 1000;
  var events = options.events || ['click','keydown','scroll','touchstart','touchmove','mousemove','mousedown'];
  var target = options.target || window;
  var timer;
  var onLogout = options.onLogout || function () {
    try { localStorage.removeItem('token'); } catch (e) {}
    try { sessionStorage.clear(); } catch (e) {}
    try {
      if (navigator.sendBeacon) {
        navigator.sendBeacon(options.logoutUrl || '/api/logout');
      } else {
        fetch(options.logoutUrl || '/api/logout', { method: 'POST', credentials: 'include', keepalive: true });
      }
    } catch (e) {}
    var url = options.redirectUrl || '/login';
    try { window.location.replace(url); } catch (e) { window.location.href = url; }
  };

  function resetTimer() {
    if (timer) clearTimeout(timer);
    timer = setTimeout(triggerLogout, timeout);
  }

  function triggerLogout() {
    stop();
    onLogout();
  }

  function activityHandler() {
    resetTimer();
  }

  function start() {
    events.forEach(function (evt) {
      target.addEventListener(evt, activityHandler, true);
    });
    resetTimer();
  }

  function stop() {
    events.forEach(function (evt) {
      target.removeEventListener(evt, activityHandler, true);
    });
    if (timer) {
      clearTimeout(timer);
      timer = null;
    }
  }

  start();
  return { start: start, stop: stop, reset: resetTimer };
}
