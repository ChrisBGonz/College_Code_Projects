<?php
session_start();

$dbFile = __DIR__ . '/users.sqlite';
$init = !file_exists($dbFile);
$db = new PDO('sqlite:' . $dbFile);
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

if ($init) {
    $db->exec("CREATE TABLE users (id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE, password_hash TEXT, question TEXT, answer_hash TEXT)");
    $stmt = $db->prepare("INSERT INTO users (username,password_hash,question,answer_hash) VALUES (?,?,?,?)");
    $stmt->execute([
        'testuser',
        password_hash('password123', PASSWORD_DEFAULT),
        '你母亲的娘家姓是什么？',
        password_hash('王', PASSWORD_DEFAULT)
    ]);
}

if (isset($_GET['restart'])) {
    session_unset();
    session_destroy();
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}

function h($s) {
    return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
}

$action = $_POST['action'] ?? '';
$error = '';
$success = '';
$step = '';

if ($action === 'find') {
    $username = trim($_POST['username'] ?? '');
    if ($username === '') {
        $error = '请输入用户名';
    } else {
        $stmt = $db->prepare("SELECT id, question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        if (!$user) {
            $error = '用户不存在';
        } else {
            $_SESSION['recover_user_id'] = $user['id'];
            $_SESSION['recover_question'] = $user['question'];
            $_SESSION['recover_verified'] = false;
            $step = 'question';
        }
    }
} elseif ($action === 'answer') {
    if (empty($_SESSION['recover_user_id'])) {
        $error = '会话失效，请重新开始';
    } else {
        $answer = trim($_POST['answer'] ?? '');
        if ($answer === '') {
            $error = '请输入答案';
        } else {
            $stmt = $db->prepare("SELECT answer_hash FROM users WHERE id = ?");
            $stmt->execute([$_SESSION['recover_user_id']]);
            $row = $stmt->fetch(PDO::FETCH_ASSOC);
            if (!$row || !password_verify($answer, $row['answer_hash'])) {
                $error = '答案不正确';
            } else {
                $_SESSION['recover_verified'] = true;
                $step = 'reset';
            }
        }
    }
} elseif ($action === 'reset') {
    if (empty($_SESSION['recover_user_id']) || empty($_SESSION['recover_verified'])) {
        $error = '未验证身份';
    } else {
        $password = $_POST['password'] ?? '';
        $password2 = $_POST['password2'] ?? '';
        if ($password === '' || $password2 === '') {
            $error = '请输入新密码';
        } elseif ($password !== $password2) {
            $error = '两次密码不一致';
        } else {
            $hash = password_hash($password, PASSWORD_DEFAULT);
            $stmt = $db->prepare("UPDATE users SET password_hash = ? WHERE id = ?");
            $stmt->execute([$hash, $_SESSION['recover_user_id']]);
            $success = '密码已重置，请使用新密码登录。';
            session_unset();
            session_destroy();
            $step = 'done';
        }
    }
}

if ($step === '') {
    if (isset($_SESSION['recover_user_id'])) {
        $step = !empty($_SESSION['recover_verified']) ? 'reset' : 'question';
    } else {
        $step = 'find';
    }
}

echo "<!DOCTYPE html><html><head><meta charset='UTF-8'><title>账号找回</title></head><body>";
echo "<h2>账号找回</h2>";
if ($error !== '') {
    echo "<p style='color:red'>" . h($error) . "</p>";
}
if ($success !== '') {
    echo "<p style='color:green'>" . h($success) . "</p>";
}

if ($step === 'find') {
    echo "<form method='post'>";
    echo "<input type='hidden' name='action' value='find'>";
    echo "用户名：<input type='text' name='username' required>";
    echo "<button type='submit'>下一步</button>";
    echo "</form>";
} elseif ($step === 'question') {
    $q = $_SESSION['recover_question'] ?? '';
    echo "<form method='post'>";
    echo "<input type='hidden' name='action' value='answer'>";
    echo "<p>密保问题：" . h($q) . "</p>";
    echo "答案：<input type='text' name='answer' required>";
    echo "<button type='submit'>验证</button>";
    echo "</form>";
    echo "<p><a href='?restart=1'>重新开始</a></p>";
} elseif ($step === 'reset') {
    echo "<form method='post'>";
    echo "<input type='hidden' name='action' value='reset'>";
    echo "新密码：<input type='password' name='password' required><br>";
    echo "确认新密码：<input type='password' name='password2' required><br>";
    echo "<button type='submit'>重置密码</button>";
    echo "</form>";
} elseif ($step === 'done') {
    echo "<p><a href='?restart=1'>返回</a></p>";
}

echo "</body></html>";
?>
