import base64
import hashlib
import hmac
import struct
import sys
import time

def _decode_base32(secret):
    s = secret.strip().replace(" ", "").upper()
    padding = len(s) % 8
    if padding:
        s += "=" * (8 - padding)
    return base64.b32decode(s, casefold=True)

def totp(secret, for_time=None, step=30, digits=6, algo=hashlib.sha1):
    if for_time is None:
        for_time = int(time.time())
    counter = int(for_time // step)
    key = _decode_base32(secret)
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, algo).digest()
    offset = h[-1] & 0x0F
    code_int = (struct.unpack(">I", h[offset:offset + 4])[0] & 0x7FFFFFFF) % (10 ** digits)
    return f"{code_int:0{digits}d}"

def verify_totp(secret, code, window=1, step=30, digits=6, algo=hashlib.sha1):
    if not isinstance(code, str):
        code = str(code)
    code = code.strip()
    if len(code) != digits or not code.isdigit():
        return False
    now = int(time.time())
    for offset in range(-window, window + 1):
        if hmac.compare_digest(totp(secret, now + offset * step, step, digits, algo), code):
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = input("Secret: ").strip()
        code = input("TOTP: ").strip()
    if verify_totp(secret, code):
        print("OK")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
