from http.server import HTTPServer, BaseHTTPRequestHandler
from http import cookies

class TrialHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path.split("?", 1)[0] != "/register":
            self.send_response(200)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.end_headers()
            self.wfile.write("OK".encode("utf-8"))
            return

        c = cookies.SimpleCookie()
        c.load(self.headers.get("Cookie", ""))
        used = "has_used_trial" in c and c["has_used_trial"].value.lower() in ("1", "true", "yes")

        if used:
            self.send_response(403)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.end_headers()
            self.wfile.write("Trial already used".encode("utf-8"))
            return

        self.send_response(200)
        self.send_header("Content-Type", "text/plain; charset=utf-8")
        sc = cookies.SimpleCookie()
        sc["has_used_trial"] = "true"
        sc["has_used_trial"]["path"] = "/"
        sc["has_used_trial"]["max-age"] = 31536000
        for morsel in sc.values():
            self.send_header("Set-Cookie", morsel.OutputString())
        self.end_headers()
        self.wfile.write("Trial registered".encode("utf-8"))

if __name__ == "__main__":
    HTTPServer(("0.0.0.0", 8000), TrialHandler).serve_forever()
