class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    return head :forbidden unless current_user&.role == 'Manager'
    staff = User.find(params[:id])
    return render(json: { error: 'not_staff' }, status: :unprocessable_entity) unless staff.role == 'Staff'

    if staff.update(role: 'Manager')
      render json: { id: staff.id, role: staff.role }, status: :ok
    else
      render json: { errors: staff.errors.full_messages }, status: :unprocessable_entity
    end
  end
end
