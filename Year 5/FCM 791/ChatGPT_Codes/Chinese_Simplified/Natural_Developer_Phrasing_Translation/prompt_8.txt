package config

import (
	"errors"
	"os"
	"path/filepath"
	"strings"
)

func GetConfigPath(envVar, defaultPath string) (string, error) {
	if strings.TrimSpace(envVar) == "" {
		return "", errors.New("environment variable name is empty")
	}

	path := strings.TrimSpace(os.Getenv(envVar))
	if path == "" {
		path = strings.TrimSpace(defaultPath)
	}
	if path == "" {
		return "", errors.New("config path is empty")
	}

	path = filepath.Clean(path)
	if abs, err := filepath.Abs(path); err == nil {
		path = abs
	}

	info, err := os.Stat(path)
	if err != nil {
		return "", err
	}
	if info.IsDir() {
		return "", errors.New("config path points to a directory")
	}

	return path, nil
}
