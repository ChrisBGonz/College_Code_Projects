import os
import uuid
import requests
from flask import Flask, request, redirect, Response

app = Flask(__name__)

GATEWAY_URL = os.getenv("GATEWAY_URL", "https://api.example.com/payments")
API_KEY = os.getenv("API_KEY", "your_api_key")
RETURN_URL = os.getenv("RETURN_URL", "http://localhost:8000/return")

def create_payment(amount, currency, order_id):
    payload = {
        "amount": amount,
        "currency": currency,
        "order_id": order_id,
        "return_url": RETURN_URL
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json",
        "Accept": "application/json"
    }
    response = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=10)
    response.raise_for_status()
    return response.json()

@app.route("/pay", methods=["GET"])
def pay():
    amount = request.args.get("amount", "100.00")
    currency = request.args.get("currency", "USD")
    order_id = request.args.get("order_id", str(uuid.uuid4()))
    try:
        data = create_payment(amount, currency, order_id)
    except Exception as e:
        return Response(f"Payment request failed: {e}", status=502)
    success_url = data.get("success_url")
    if not success_url:
        return Response("success_url not found in gateway response", status=502)
    return redirect(success_url, code=302)

@app.route("/return", methods=["GET"])
def payment_return():
    return "Payment processing completed."

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "8000")))
