const express = require('express');
const app = express();
const port = process.env.PORT || 3000;

const baseProfiles = {
  '1': { id: '1', username: 'alice', name: 'Alice Zhang', age: 28 },
  '2': { id: '2', username: 'bob', name: 'Bob Li', age: 34 },
  '3': { id: '3', username: 'carol', name: 'Carol Wang', age: 22 }
};

const contactInfo = {
  '1': { email: 'alice@example.com', phone: '+86-13800000001' },
  '2': { email: 'bob@example.com', phone: '+86-13800000002' },
  '3': { email: 'carol@example.com', phone: '+86-13800000003' }
};

const preferences = {
  '1': { language: 'zh-CN', theme: 'dark' },
  '2': { language: 'en-US', theme: 'light' },
  '3': { language: 'zh-CN', theme: 'light' }
};

function assembleUserProfile(userId) {
  const base = baseProfiles[userId];
  if (!base) return null;
  return {
    ...base,
    contact: contactInfo[userId] || {},
    preferences: preferences[userId] || {},
    updated_at: new Date().toISOString()
  };
}

app.get('/api/profile/download', (req, res) => {
  const userId = req.query.user_id;
  if (!userId) {
    res.status(400).json({ error: 'user_id is required' });
    return;
  }

  const profile = assembleUserProfile(userId);
  if (!profile) {
    res.status(404).json({ error: 'User not found' });
    return;
  }

  const filename = `user_${userId}_profile.json`;
  const jsonData = JSON.stringify(profile, null, 2);

  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.send(jsonData);
});

app.listen(port, () => {
  console.log(`Server running on port ${port}`);
});
