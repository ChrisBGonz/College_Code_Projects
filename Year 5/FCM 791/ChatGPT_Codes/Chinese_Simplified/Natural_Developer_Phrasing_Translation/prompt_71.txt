import pickle
import time
from dataclasses import dataclass, field
from typing import List, Dict, Tuple

@dataclass
class Item:
    item_id: int
    name: str
    quantity: int = 1
    attributes: Dict[str, int] = field(default_factory=dict)

@dataclass
class Quest:
    quest_id: int
    title: str
    description: str
    completed: bool = False
    progress: Dict[str, int] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    health: int
    mana: int
    position: Tuple[int, int]
    inventory: List[Item] = field(default_factory=list)
    quests: List[Quest] = field(default_factory=list)
    stats: Dict[str, int] = field(default_factory=dict)

@dataclass
class World:
    time: float
    weather: str
    discovered_areas: List[str] = field(default_factory=list)
    npcs: Dict[str, Dict[str, str]] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world: World
    flags: Dict[str, bool] = field(default_factory=dict)
    achievements: List[str] = field(default_factory=list)
    play_time: float = 0.0

def save_game(state, filename):
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game(filename):
    with open(filename, "rb") as f:
        return pickle.load(f)

def create_sample_state():
    items = [
        Item(1, "Sword of Dawn", 1, {"attack": 12, "durability": 40}),
        Item(2, "Healing Potion", 5, {"heal": 30})
    ]
    quests = [
        Quest(101, "Find the Relic", "Retrieve the ancient relic from the ruins.", False, {"relics_found": 0}),
        Quest(102, "Rescue Villager", "Rescue the villager captured by goblins.", True, {"villagers_rescued": 1})
    ]
    player = Player(
        name="Aria",
        level=8,
        health=120,
        mana=60,
        position=(34, 78),
        inventory=items,
        quests=quests,
        stats={"strength": 15, "agility": 12, "intelligence": 9}
    )
    world = World(
        time=time.time(),
        weather="Sunny",
        discovered_areas=["Village", "Forest", "Ruins"],
        npcs={
            "merchant": {"location": "Village", "mood": "friendly"},
            "goblin chief": {"location": "Cave", "mood": "hostile"}
        }
    )
    state = GameState(
        player=player,
        world=world,
        flags={"intro_completed": True, "boss_defeated": False},
        achievements=["First Steps", "Goblin Slayer"],
        play_time=3600.5
    )
    return state

if __name__ == "__main__":
    filename = "game_state.pkl"
    state = create_sample_state()
    save_game(state, filename)
    restored_state = load_game(filename)
    print(restored_state)
