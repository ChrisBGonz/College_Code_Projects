using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

public class DataRepository
{
    public DataTable GetSortedData(string connectionString, string column_name)
    {
        var allowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Id",
            "Name",
            "CreatedAt"
        };

        if (!allowedColumns.Contains(column_name))
            throw new ArgumentException("Invalid column name.", nameof(column_name));

        using (var conn = new SqlConnection(connectionString))
        using (var cmd = new SqlCommand($"SELECT * FROM MyTable ORDER BY [{column_name}]", conn))
        using (var adapter = new SqlDataAdapter(cmd))
        {
            var table = new DataTable();
            adapter.Fill(table);
            return table;
        }
    }
}
