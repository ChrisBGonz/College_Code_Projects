function initMultiStepRegistration(formSelector, storageKey) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form || !window.localStorage) return;

  storageKey = storageKey || form.getAttribute('data-storage-key') || 'multiStepRegistration';

  function readStorage() {
    try {
      return JSON.parse(localStorage.getItem(storageKey)) || {};
    } catch (e) {
      return {};
    }
  }

  function writeStorage(data) {
    localStorage.setItem(storageKey, JSON.stringify(data));
  }

  function serializeField(field, data) {
    if (!field.name) return;
    if (field.type === 'checkbox') {
      data[field.name] = field.checked;
    } else if (field.type === 'radio') {
      if (field.checked) data[field.name] = field.value;
    } else if (field.tagName === 'SELECT' && field.multiple) {
      var values = [];
      for (var i = 0; i < field.options.length; i++) {
        if (field.options[i].selected) values.push(field.options[i].value);
      }
      data[field.name] = values;
    } else {
      data[field.name] = field.value;
    }
  }

  function saveStep(stepEl) {
    var data = readStorage();
    var fields = stepEl.querySelectorAll('input, select, textarea');
    for (var i = 0; i < fields.length; i++) {
      serializeField(fields[i], data);
    }
    writeStorage(data);
  }

  function restore() {
    var data = readStorage();
    var fields = form.querySelectorAll('input, select, textarea');
    for (var i = 0; i < fields.length; i++) {
      var field = fields[i];
      if (!field.name || data[field.name] === undefined) continue;
      if (field.type === 'checkbox') {
        field.checked = !!data[field.name];
      } else if (field.type === 'radio') {
        field.checked = data[field.name] === field.value;
      } else if (field.tagName === 'SELECT' && field.multiple && Array.isArray(data[field.name])) {
        var values = data[field.name];
        for (var j = 0; j < field.options.length; j++) {
          field.options[j].selected = values.indexOf(field.options[j].value) !== -1;
        }
      } else {
        field.value = data[field.name];
      }
    }
  }

  function currentStep() {
    return form.querySelector('[data-step].active') || form;
  }

  form.addEventListener('input', function (e) {
    if (!e.target) return;
    var data = readStorage();
    serializeField(e.target, data);
    writeStorage(data);
  });

  var stepButtons = form.querySelectorAll('[data-next], [data-prev], [data-save-step]');
  for (var i = 0; i < stepButtons.length; i++) {
    stepButtons[i].addEventListener('click', function () {
      saveStep(currentStep());
    });
  }

  form.addEventListener('submit', function () {
    localStorage.removeItem(storageKey);
  });

  restore();
}
