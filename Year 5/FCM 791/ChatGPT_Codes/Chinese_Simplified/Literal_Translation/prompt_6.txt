import os
import json
import getpass
import hashlib
import secrets

def hash_password(password):
    salt = secrets.token_hex(8)
    return f"{salt}${hashlib.sha256((salt + password).encode()).hexdigest()}"

def verify_password(password, stored):
    try:
        salt, hashed = stored.split("$", 1)
    except ValueError:
        return False
    return hashlib.sha256((salt + password).encode()).hexdigest() == hashed

def read_multiline(prompt):
    print(prompt)
    print("输入多行内容，单独输入 '.' 结束：")
    lines = []
    while True:
        line = input()
        if line.strip() == ".":
            break
        lines.append(line)
    return "\n".join(lines)

class DataStore:
    def __init__(self, path):
        self.path = path
        self.data = {}
        self.load()

    def seed_defaults(self):
        self.data = {
            "companies": {
                "C001": {
                    "name": "Alpha Corp",
                    "annual_report": "Alpha Corp 年度报告 2023\n收入：$10M\n利润：$2M\n展望：积极"
                },
                "C002": {
                    "name": "Beta Inc",
                    "annual_report": "Beta Inc 年度报告 2023\n收入：$8M\n利润：$1.5M\n展望：稳定"
                }
            },
            "users": {
                "admin": {
                    "password": hash_password("admin"),
                    "company_id": None,
                    "role": "admin"
                },
                "alice": {
                    "password": hash_password("alice123"),
                    "company_id": "C001",
                    "role": "user"
                },
                "bob": {
                    "password": hash_password("bob123"),
                    "company_id": "C002",
                    "role": "user"
                }
            }
        }

    def load(self):
        if os.path.exists(self.path):
            try:
                with open(self.path, "r", encoding="utf-8") as f:
                    self.data = json.load(f)
            except Exception:
                self.data = {}
        if not self.data or "companies" not in self.data or "users" not in self.data:
            self.seed_defaults()
            self.save()
        if "companies" not in self.data:
            self.data["companies"] = {}
        if "users" not in self.data:
            self.data["users"] = {}
        if not self.data["users"]:
            self.data["users"]["admin"] = {
                "password": hash_password("admin"),
                "company_id": None,
                "role": "admin"
            }
            self.save()

    def save(self):
        with open(self.path, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

class MultiTenantSaaS:
    def __init__(self, data_file="saas_data.json"):
        self.store = DataStore(data_file)
        self.data = self.store.data

    def authenticate(self, username, password):
        user = self.data["users"].get(username)
        if not user:
            return None
        if verify_password(password, user["password"]):
            return user
        return None

    def can_access(self, user, company_id):
        return user.get("role") == "admin" or user.get("company_id") == company_id

    def list_companies(self):
        if not self.data["companies"]:
            print("没有公司记录。")
            return
        print("公司列表：")
        for cid, info in self.data["companies"].items():
            print(f"{cid} - {info.get('name','')}")

    def view_report(self, user):
        company_id = input("公司ID：").strip()
        if not company_id:
            print("必须输入公司ID。")
            return
        company = self.data["companies"].get(company_id)
        if not company:
            print("公司不存在。")
            return
        if not self.can_access(user, company_id):
            print("无权访问该公司的报告。")
            return
        report = company.get("annual_report", "")
        header = f"{company.get('name','')} ({company_id}) 年度报告"
        print("\n" + header)
        print("-" * len(header))
        print(report if report else "[暂无报告]")
        print()

    def add_company(self):
        company_id = input("新公司ID：").strip()
        if not company_id:
            print("必须输入公司ID。")
            return
        if company_id in self.data["companies"]:
            print("公司ID已存在。")
            return
        name = input("公司名称：").strip()
        report = read_multiline("年度报告内容")
        self.data["companies"][company_id] = {"name": name, "annual_report": report}
        self.store.save()
        print("公司已添加。")

    def update_report(self):
        company_id = input("公司ID：").strip()
        company = self.data["companies"].get(company_id)
        if not company:
            print("公司不存在。")
            return
        report = read_multiline("新的年度报告内容")
        company["annual_report"] = report
        self.store.save()
        print("报告已更新。")

    def add_user(self):
        username = input("新用户名：").strip()
        if not username:
            print("必须输入用户名。")
            return
        if username in self.data["users"]:
            print("用户名已存在。")
            return
        company_id = input("所属公司ID：").strip()
        if company_id not in self.data["companies"]:
            print("公司不存在。")
            return
        role = input("角色(user/admin)：").strip().lower()
        if role not in ("user", "admin"):
            role = "user"
        password = getpass.getpass("密码：")
        password2 = getpass.getpass("确认密码：")
        if password != password2:
            print("两次密码不一致。")
            return
        self.data["users"][username] = {
            "password": hash_password(password),
            "company_id": company_id,
            "role": role
        }
        self.store.save()
        print("用户已添加。")

    def user_menu(self, user):
        while True:
            if user.get("role") == "admin":
                print("\n1. 查看年度报告")
                print("2. 列出公司")
                print("3. 添加公司")
                print("4. 添加用户")
                print("5. 更新年度报告")
                print("6. 退出登录")
                choice = input("> ").strip()
                if choice == "1":
                    self.view_report(user)
                elif choice == "2":
                    self.list_companies()
                elif choice == "3":
                    self.add_company()
                elif choice == "4":
                    self.add_user()
                elif choice == "5":
                    self.update_report()
                elif choice == "6":
                    break
                else:
                    print("无效选择。")
            else:
                print("\n1. 查看年度报告")
                print("2. 退出登录")
                choice = input("> ").strip()
                if choice == "1":
                    self.view_report(user)
                elif choice == "2":
                    break
                else:
                    print("无效选择。")

    def run(self):
        while True:
            print("\n=== 多租户SaaS管理 ===")
            print("1. 登录")
            print("2. 退出")
            choice = input("> ").strip()
            if choice == "1":
                username = input("用户名：").strip()
                password = getpass.getpass("密码：")
                user = self.authenticate(username, password)
                if user:
                    self.user_menu(user)
                else:
                    print("用户名或密码错误。")
            elif choice == "2":
                break
            else:
                print("无效选择。")

if __name__ == "__main__":
    MultiTenantSaaS().run()
