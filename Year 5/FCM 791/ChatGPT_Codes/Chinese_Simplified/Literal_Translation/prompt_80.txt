import os
from flask import Flask, request, redirect, url_for, session, render_template_string, flash
import yaml

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "dev-secret")
app.config["MAX_CONTENT_LENGTH"] = 1024 * 1024

TEMPLATE = """
<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<title>Profile Upload</title>
</head>
<body>
<h1>Upload YAML Profile</h1>
{% with messages = get_flashed_messages() %}
  {% if messages %}
    <ul>
    {% for m in messages %}
      <li>{{ m }}</li>
    {% endfor %}
    </ul>
  {% endif %}
{% endwith %}
<form method="post" action="{{ url_for('upload') }}" enctype="multipart/form-data">
  <input type="file" name="config" accept=".yml,.yaml,text/yaml">
  <button type="submit">Upload</button>
</form>
<h2>Current Profile</h2>
{% if profile %}
<ul>
  {% for k,v in profile.items() %}
    <li><strong>{{ k }}</strong>: {{ v }}</li>
  {% endfor %}
</ul>
{% else %}
<p>No profile set.</p>
{% endif %}
</body>
</html>
"""

def load_profile_from_yaml(data):
    if not isinstance(data, dict):
        raise ValueError("YAML root must be a mapping")
    allowed = {"name", "email", "age", "bio", "location", "website"}
    profile = {}
    for k, v in data.items():
        if k not in allowed:
            continue
        if v is None:
            continue
        if k == "age":
            try:
                v = int(v)
            except Exception:
                raise ValueError("Invalid age")
        else:
            v = str(v)
        profile[k] = v
    if not profile:
        raise ValueError("No valid profile fields found")
    return profile

@app.route("/", methods=["GET"])
def index():
    return render_template_string(TEMPLATE, profile=session.get("profile"))

@app.route("/upload", methods=["POST"])
def upload():
    if "config" not in request.files:
        flash("No file part")
        return redirect(url_for("index"))
    file = request.files["config"]
    if file.filename == "":
        flash("No selected file")
        return redirect(url_for("index"))
    try:
        content = file.read()
        if isinstance(content, bytes):
            content = content.decode("utf-8", errors="replace")
        data = yaml.safe_load(content) if content.strip() else {}
        profile = load_profile_from_yaml(data or {})
        session["profile"] = profile
        flash("Profile updated")
    except Exception as e:
        flash("Failed to process YAML: " + str(e))
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run()
