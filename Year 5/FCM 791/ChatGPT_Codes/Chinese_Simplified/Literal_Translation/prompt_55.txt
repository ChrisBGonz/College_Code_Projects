from flask import Flask, request, make_response

app = Flask(__name__)

FORM = """
<!doctype html>
<title>Free Trial</title>
<h1>Free Trial Registration</h1>
<form method="post">
    <button type="submit">Start Free Trial</button>
</form>
"""

@app.route("/trial", methods=["GET", "POST"])
def trial():
    if request.cookies.get("has_used_trial"):
        return "You have already used the free trial."
    if request.method == "POST":
        resp = make_response("Free trial activated.")
        resp.set_cookie("has_used_trial", "true", max_age=31536000, httponly=True)
        return resp
    return FORM

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
