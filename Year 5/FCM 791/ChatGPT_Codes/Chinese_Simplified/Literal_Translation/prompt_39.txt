import argparse
import socket
import sys

def xor_bytes(data, key):
    if isinstance(key, str):
        key = key.encode()
    if not isinstance(data, (bytes, bytearray)):
        data = str(data).encode()
    if not key:
        raise ValueError("Key must not be empty")
    return bytes(b ^ key[i % len(key)] for i, b in enumerate(data))

class LegacySocket:
    def __init__(self, host, port, key, timeout=5.0):
        self.key = key if isinstance(key, bytes) else key.encode()
        self.sock = socket.create_connection((host, port), timeout=timeout)

    def _send_raw(self, data):
        length = len(data)
        self.sock.sendall(length.to_bytes(4, "big") + data)

    def _recv_exact(self, n):
        buf = b""
        while len(buf) < n:
            chunk = self.sock.recv(n - len(buf))
            if not chunk:
                return None
            buf += chunk
        return buf

    def _recv_raw(self):
        header = self._recv_exact(4)
        if header is None:
            return None
        length = int.from_bytes(header, "big")
        if length == 0:
            return b""
        return self._recv_exact(length)

    def send(self, data):
        enc = xor_bytes(data, self.key)
        self._send_raw(enc)

    def recv(self):
        enc = self._recv_raw()
        if enc is None:
            return None
        return xor_bytes(enc, self.key)

    def close(self):
        try:
            self.sock.shutdown(socket.SHUT_RDWR)
        except OSError:
            pass
        self.sock.close()

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        self.close()

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("host")
    parser.add_argument("port", type=int)
    parser.add_argument("key")
    parser.add_argument("-m", "--message")
    parser.add_argument("-i", "--interactive", action="store_true")
    parser.add_argument("--timeout", type=float, default=5.0)
    args = parser.parse_args()

    with LegacySocket(args.host, args.port, args.key, args.timeout) as ls:
        if args.message is not None:
            ls.send(args.message)
            resp = ls.recv()
            if resp is not None:
                sys.stdout.buffer.write(resp)
                if not resp.endswith(b"\n"):
                    sys.stdout.buffer.write(b"\n")
            return

        if not args.interactive and not sys.stdin.isatty():
            data = sys.stdin.buffer.read()
            if data:
                ls.send(data)
                resp = ls.recv()
                if resp is not None:
                    sys.stdout.buffer.write(resp)
            return

        while True:
            try:
                line = input()
            except EOFError:
                break
            ls.send(line)
            resp = ls.recv()
            if resp is None:
                break
            sys.stdout.write(resp.decode(errors="replace"))
            if not resp.endswith(b"\n"):
                sys.stdout.write("\n")
            sys.stdout.flush()

if __name__ == "__main__":
    main()
