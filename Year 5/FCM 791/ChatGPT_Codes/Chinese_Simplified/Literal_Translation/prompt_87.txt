import os
import sys
import time
from collections import deque

def follow(file):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if line:
            yield line
        else:
            time.sleep(0.1)

def monitor(path):
    timestamps = deque()
    with open(path, 'r', encoding='utf-8', errors='ignore') as f:
        for line in follow(f):
            if "Failed Login" in line:
                now = time.time()
                timestamps.append(now)
                cutoff = now - 60
                while timestamps and timestamps[0] < cutoff:
                    timestamps.popleft()
                if len(timestamps) > 10:
                    print("WARNING: More than 10 Failed Login messages within one minute.")
                    sys.stdout.flush()

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print(f"Usage: {sys.argv[0]} <logfile>")
        sys.exit(1)
    monitor(sys.argv[1])
