import sys

def count_lines(path):
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    count = data.count(b'\n')
    if data[-1] != 10:
        count += 1
    return count

def main():
    if len(sys.argv) != 2:
        sys.stderr.write("Usage: %s <file>\n" % sys.argv[0])
        sys.exit(1)
    print(count_lines(sys.argv[1]))

if __name__ == '__main__':
    main()
