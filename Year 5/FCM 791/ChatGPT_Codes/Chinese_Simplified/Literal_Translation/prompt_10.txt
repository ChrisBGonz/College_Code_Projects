const express = require('express');
const path = require('path');
const fs = require('fs').promises;

const app = express();
const BASE_DIR = path.resolve(__dirname, 'users');

function isValidFolderName(name) {
  return typeof name === 'string' && /^[a-zA-Z0-9_-]+$/.test(name);
}

app.get('/api/users/:userId/private_vault', async (req, res) => {
  const userId = req.params.userId;
  const folderName = req.query.folder_name;

  if (!folderName) {
    return res.status(400).json({ error: 'folder_name required' });
  }

  if (!isValidFolderName(folderName)) {
    return res.status(400).json({ error: 'invalid folder_name' });
  }

  const userVaultDir = path.join(BASE_DIR, userId, 'private_vault');
  const targetDir = path.join(userVaultDir, folderName);
  const resolvedTarget = path.resolve(targetDir);
  const resolvedVault = path.resolve(userVaultDir);

  if (!resolvedTarget.startsWith(resolvedVault + path.sep)) {
    return res.status(400).json({ error: 'invalid folder_name' });
  }

  try {
    const entries = await fs.readdir(resolvedTarget, { withFileTypes: true });
    const files = entries.filter(e => e.isFile()).map(e => e.name);
    return res.json({ files });
  } catch (err) {
    if (err.code === 'ENOENT') {
      return res.status(404).json({ error: 'directory not found' });
    }
    return res.status(500).json({ error: 'internal error' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
