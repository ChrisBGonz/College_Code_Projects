package main

import (
	"encoding/json"
	"errors"
	"os"
	"reflect"
	"strconv"
)

const maxInt64 = int64(^uint64(0) >> 1)

func ApplyConfigFromFile(path string, obj interface{}) error {
	f, err := os.Open(path)
	if err != nil {
		return err
	}
	defer f.Close()
	dec := json.NewDecoder(f)
	dec.UseNumber()
	var cfg map[string]interface{}
	if err := dec.Decode(&cfg); err != nil {
		return err
	}
	return ApplyConfig(cfg, obj)
}

func ApplyConfig(cfg map[string]interface{}, obj interface{}) error {
	if obj == nil {
		return errors.New("obj is nil")
	}
	v := reflect.ValueOf(obj)
	if v.Kind() != reflect.Ptr || v.IsNil() {
		return errors.New("obj must be pointer to struct")
	}
	v = v.Elem()
	if v.Kind() != reflect.Struct {
		return errors.New("obj must be pointer to struct")
	}
	for key, val := range cfg {
		field := v.FieldByName(key)
		if !field.IsValid() || !field.CanSet() {
			continue
		}
		if err := setFieldValue(field, val); err != nil {
			return err
		}
	}
	return nil
}

func setFieldValue(field reflect.Value, val interface{}) error {
	if !field.CanSet() {
		return nil
	}
	if val == nil {
		return nil
	}

	if field.Kind() == reflect.Ptr {
		if field.IsNil() {
			field.Set(reflect.New(field.Type().Elem()))
		}
		return setFieldValue(field.Elem(), val)
	}

	switch field.Kind() {
	case reflect.String:
		switch v := val.(type) {
		case string:
			field.SetString(v)
		case json.Number:
			field.SetString(v.String())
		case float64:
			field.SetString(strconv.FormatFloat(v, 'f', -1, 64))
		case float32:
			field.SetString(strconv.FormatFloat(float64(v), 'f', -1, 64))
		case int:
			field.SetString(strconv.Itoa(v))
		case int8, int16, int32, int64:
			field.SetString(strconv.FormatInt(reflect.ValueOf(v).Int(), 10))
		case uint, uint8, uint16, uint32, uint64:
			field.SetString(strconv.FormatUint(reflect.ValueOf(v).Uint(), 10))
		case bool:
			field.SetString(strconv.FormatBool(v))
		default:
			return errors.New("cannot convert to string")
		}
	case reflect.Bool:
		switch v := val.(type) {
		case bool:
			field.SetBool(v)
		case string:
			b, err := strconv.ParseBool(v)
			if err != nil {
				return err
			}
			field.SetBool(b)
		case json.Number:
			i, err := v.Int64()
			if err != nil {
				return err
			}
			field.SetBool(i != 0)
		case float64:
			field.SetBool(v != 0)
		case int, int8, int16, int32, int64:
			field.SetBool(reflect.ValueOf(v).Int() != 0)
		case uint, uint8, uint16, uint32, uint64:
			field.SetBool(reflect.ValueOf(v).Uint() != 0)
		default:
			return errors.New("cannot convert to bool")
		}
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		i, err := toInt64(val)
		if err != nil {
			return err
		}
		field.SetInt(i)
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64, reflect.Uintptr:
		u, err := toUint64(val)
		if err != nil {
			return err
		}
		field.SetUint(u)
	case reflect.Float32, reflect.Float64:
		f, err := toFloat64(val)
		if err != nil {
			return err
		}
		field.SetFloat(f)
	case reflect.Slice:
		arr, ok := val.([]interface{})
		if !ok {
			rv := reflect.ValueOf(val)
			if rv.IsValid() && rv.Kind() == reflect.Slice {
				arr = make([]interface{}, rv.Len())
				for i := 0; i < rv.Len(); i++ {
					arr[i] = rv.Index(i).Interface()
				}
			} else {
				return errors.New("cannot convert to slice")
			}
		}
		slice := reflect.MakeSlice(field.Type(), len(arr), len(arr))
		for i := 0; i < len(arr); i++ {
			if err := setFieldValue(slice.Index(i), arr[i]); err != nil {
				return err
			}
		}
		field.Set(slice)
	case reflect.Map:
		if field.Type().Key().Kind() != reflect.String {
			return errors.New("unsupported map key type")
		}
		m, ok := val.(map[string]interface{})
		if !ok {
			return errors.New("cannot convert to map")
		}
		newMap := reflect.MakeMap(field.Type())
		for k, v := range m {
			elem := reflect.New(field.Type().Elem()).Elem()
			if err := setFieldValue(elem, v); err != nil {
				return err
			}
			newMap.SetMapIndex(reflect.ValueOf(k).Convert(field.Type().Key()), elem)
		}
		field.Set(newMap)
	case reflect.Struct:
		if m, ok := val.(map[string]interface{}); ok {
			return ApplyConfig(m, field.Addr().Interface())
		}
	case reflect.Interface:
		v := reflect.ValueOf(val)
		if v.IsValid() {
			if v.Type().AssignableTo(field.Type()) {
				field.Set(v)
			} else if v.Type().ConvertibleTo(field.Type()) {
				field.Set(v.Convert(field.Type()))
			} else if field.Type().NumMethod() == 0 {
				field.Set(v)
			}
		}
	}
	return nil
}

func toInt64(val interface{}) (int64, error) {
	switch v := val.(type) {
	case json.Number:
		return v.Int64()
	case float64:
		return int64(v), nil
	case float32:
		return int64(v), nil
	case int:
		return int64(v), nil
	case int8:
		return int64(v), nil
	case int16:
		return int64(v), nil
	case int32:
		return int64(v), nil
	case int64:
		return v, nil
	case uint, uint8, uint16, uint32, uint64:
		u := reflect.ValueOf(v).Uint()
		if u > uint64(maxInt64) {
			return 0, errors.New("uint overflow")
		}
		return int64(u), nil
	case string:
		return strconv.ParseInt(v, 10, 64)
	case bool:
		if v {
			return 1, nil
		}
		return 0, nil
	default:
		return 0, errors.New("cannot convert to int")
	}
}

func toUint64(val interface{}) (uint64, error) {
	switch v := val.(type) {
	case json.Number:
		if i, err := v.Int64(); err == nil {
			if i < 0 {
				return 0, errors.New("negative")
			}
			return uint64(i), nil
		}
		f, err := v.Float64()
		if err != nil {
			return 0, err
		}
		if f < 0 {
			return 0, errors.New("negative")
		}
		return uint64(f), nil
	case float64:
		if v < 0 {
			return 0, errors.New("negative")
		}
		return uint64(v), nil
	case float32:
		if v < 0 {
			return 0, errors.New("negative")
		}
		return uint64(v), nil
	case int, int8, int16, int32, int64:
		i := reflect.ValueOf(v).Int()
		if i < 0 {
			return 0, errors.New("negative")
		}
		return uint64(i), nil
	case uint, uint8, uint16, uint32, uint64:
		return reflect.ValueOf(v).Uint(), nil
	case string:
		return strconv.ParseUint(v, 10, 64)
	case bool:
		if v {
			return 1, nil
		}
		return 0, nil
	default:
		return 0, errors.New("cannot convert to uint")
	}
}

func toFloat64(val interface{}) (float64, error) {
	switch v := val.(type) {
	case json.Number:
		return v.Float64()
	case float64:
		return v, nil
	case float32:
		return float64(v), nil
	case int:
		return float64(v), nil
	case int8:
		return float64(v), nil
	case int16:
		return float64(v), nil
	case int32:
		return float64(v), nil
	case int64:
		return float64(v), nil
	case uint:
		return float64(v), nil
	case uint8:
		return float64(v), nil
	case uint16:
		return float64(v), nil
	case uint32:
		return float64(v), nil
	case uint64:
		return float64(v), nil
	case string:
		return strconv.ParseFloat(v, 64)
	case bool:
		if v {
			return 1, nil
		}
		return 0, nil
	default:
		return 0, errors.New("cannot convert to float")
	}
}
