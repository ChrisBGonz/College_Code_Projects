import argparse
import hashlib
import sys
import os

def sha256sum(path, block_size=1024 * 1024):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(block_size), b''):
            h.update(chunk)
    return h.hexdigest()

def normalize_hash(h):
    return h.strip().lower()

def main():
    parser = argparse.ArgumentParser(description="Verify firmware integrity using SHA256.")
    parser.add_argument("file", help="Path to firmware file")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--hash", dest="expected", help="Expected SHA256 hash")
    group.add_argument("--hash-file", dest="hash_file", help="File containing expected SHA256 hash")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print("Firmware file not found.", file=sys.stderr)
        return 2

    if args.hash_file:
        try:
            with open(args.hash_file, "r", encoding="utf-8") as f:
                content = f.read().strip()
            expected = content.split()[0] if content else ""
        except Exception as e:
            print(f"Failed to read hash file: {e}", file=sys.stderr)
            return 2
    else:
        expected = args.expected or ""

    if not expected:
        print("Expected hash is empty.", file=sys.stderr)
        return 2

    try:
        actual = sha256sum(args.file)
    except Exception as e:
        print(f"Failed to read firmware file: {e}", file=sys.stderr)
        return 2

    if normalize_hash(actual) == normalize_hash(expected):
        print("OK")
        return 0
    else:
        print("FAIL")
        print(f"Expected: {normalize_hash(expected)}")
        print(f"Actual:   {normalize_hash(actual)}")
        return 1

if __name__ == "__main__":
    sys.exit(main())
