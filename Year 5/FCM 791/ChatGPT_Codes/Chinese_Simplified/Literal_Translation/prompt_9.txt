class StaffsController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager!, only: :promote

  def promote
    staff = User.find(params[:id])
    if staff.update(role: 'manager')
      redirect_to staff_path(staff), notice: 'Staff promoted to manager.'
    else
      redirect_to staff_path(staff), alert: 'Promotion failed.'
    end
  end

  private

  def require_manager!
    head :forbidden unless current_user&.manager?
  end
end
