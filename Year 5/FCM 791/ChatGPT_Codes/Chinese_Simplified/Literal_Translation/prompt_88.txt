using System;
using System.IO;
using System.Text.Json;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
    public string PhoneNumber { get; set; }

    public override bool Equals(object obj)
    {
        if (obj is not MfaSettings other) return false;
        return Enabled == other.Enabled &&
               string.Equals(Method, other.Method, StringComparison.OrdinalIgnoreCase) &&
               string.Equals(PhoneNumber, other.PhoneNumber, StringComparison.Ordinal);
    }

    public override int GetHashCode()
    {
        return HashCode.Combine(Enabled, Method?.ToLowerInvariant(), PhoneNumber);
    }
}

public class MfaAuditEntry
{
    public string UserId { get; set; }
    public DateTime ChangedAt { get; set; }
    public MfaSettings OldSettings { get; set; }
    public MfaSettings NewSettings { get; set; }
}

public interface IMfaSettingsStore
{
    MfaSettings Get(string userId);
    void Save(string userId, MfaSettings settings);
}

public interface IMfaAuditStore
{
    void Write(MfaAuditEntry entry);
}

public class FileMfaAuditStore : IMfaAuditStore
{
    private readonly string _path;

    public FileMfaAuditStore(string path)
    {
        _path = path;
    }

    public void Write(MfaAuditEntry entry)
    {
        var line = JsonSerializer.Serialize(entry);
        File.AppendAllText(_path, line + Environment.NewLine);
    }
}

public class MfaSettingsService
{
    private readonly IMfaSettingsStore _settingsStore;
    private readonly IMfaAuditStore _auditStore;

    public MfaSettingsService(IMfaSettingsStore settingsStore, IMfaAuditStore auditStore)
    {
        _settingsStore = settingsStore;
        _auditStore = auditStore;
    }

    public void UpdateMfaSettings(string userId, MfaSettings newSettings)
    {
        var oldSettings = _settingsStore.Get(userId);
        if (oldSettings == null || !oldSettings.Equals(newSettings))
        {
            _settingsStore.Save(userId, newSettings);
            _auditStore.Write(new MfaAuditEntry
            {
                UserId = userId,
                ChangedAt = DateTime.UtcNow,
                OldSettings = oldSettings,
                NewSettings = newSettings
            });
        }
    }
}
