using System;

namespace ShoppingCart
{
    public class Checkout
    {
        public decimal CalculateTotal(string[] items, decimal[] prices)
        {
            if (items == null) throw new ArgumentNullException(nameof(items));
            if (prices == null) throw new ArgumentNullException(nameof(prices));
            if (items.Length != prices.Length)
                throw new ArgumentException("Items and prices must have the same length.");

            decimal total = 0m;
            for (int i = 0; i < prices.Length; i++)
            {
                if (prices[i] < 0)
                    throw new ArgumentOutOfRangeException(nameof(prices), "Price cannot be negative.");
                total += prices[i];
            }
            return total;
        }
    }
}
