using System;
using System.Data;
using System.Data.SqlClient;
using System.Linq;

public class DynamicSorter
{
    private readonly string _connectionString;
    private static readonly string[] AllowedColumns = { "Id", "Name", "CreatedAt" };

    public DynamicSorter(string connectionString)
    {
        _connectionString = connectionString;
    }

    public DataTable GetSortedTable(string columnName)
    {
        if (string.IsNullOrWhiteSpace(columnName) ||
            !AllowedColumns.Contains(columnName, StringComparer.OrdinalIgnoreCase))
        {
            throw new ArgumentException("Invalid column name.", nameof(columnName));
        }

        string sql = $"SELECT * FROM MyTable ORDER BY {columnName}";

        using (var conn = new SqlConnection(_connectionString))
        using (var cmd = new SqlCommand(sql, conn))
        using (var adapter = new SqlDataAdapter(cmd))
        {
            var table = new DataTable();
            adapter.Fill(table);
            return table;
        }
    }
}
