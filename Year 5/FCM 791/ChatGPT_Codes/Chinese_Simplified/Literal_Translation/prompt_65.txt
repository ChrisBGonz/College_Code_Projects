<?php
session_start();
header('Content-Type: text/html; charset=utf-8');

$dbFile = __DIR__ . '/recovery.sqlite';
$db = new PDO('sqlite:' . $dbFile);
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    question TEXT NOT NULL,
    answer_hash TEXT NOT NULL
)");

if ($db->query("SELECT COUNT(*) FROM users")->fetchColumn() == 0) {
    $username = 'demo';
    $password = 'demo1234';
    $question = '你母亲的娘家姓是什么？';
    $answer = '王';
    $stmt = $db->prepare("INSERT INTO users (username, password_hash, question, answer_hash) VALUES (?, ?, ?, ?)");
    $stmt->execute([
        $username,
        password_hash($password, PASSWORD_DEFAULT),
        $question,
        password_hash($answer, PASSWORD_DEFAULT)
    ]);
}

function h($s) {
    return htmlspecialchars($s ?? '', ENT_QUOTES, 'UTF-8');
}

$action = $_POST['action'] ?? 'start';
$message = '';

if (isset($_GET['restart'])) {
    unset($_SESSION['recover_user'], $_SESSION['recover_question'], $_SESSION['recover_verified']);
    $action = 'start';
}

if ($action === 'question') {
    $username = trim($_POST['username'] ?? '');
    if ($username === '') {
        $message = '请输入用户名。';
        $action = 'start';
    } else {
        $stmt = $db->prepare("SELECT question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $question = $stmt->fetchColumn();
        if ($question) {
            $_SESSION['recover_user'] = $username;
            $_SESSION['recover_question'] = $question;
            $action = 'answer';
        } else {
            $message = '用户不存在。';
            $action = 'start';
        }
    }
}

if ($action === 'answer') {
    if (empty($_SESSION['recover_question'])) {
        $message = '会话失效，请重新开始。';
        $action = 'start';
    }
}

if ($action === 'verify') {
    $username = $_SESSION['recover_user'] ?? '';
    $answer = trim($_POST['answer'] ?? '');
    if ($username === '') {
        $message = '会话失效，请重新开始。';
        $action = 'start';
    } else {
        $stmt = $db->prepare("SELECT answer_hash FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $hash = $stmt->fetchColumn();
        if ($hash && password_verify($answer, $hash)) {
            $_SESSION['recover_verified'] = true;
            $action = 'reset';
        } else {
            $message = '答案不正确。';
            $action = 'answer';
        }
    }
}

if ($action === 'reset') {
    if (empty($_SESSION['recover_verified'])) {
        $message = '未通过验证或会话失效，请重新开始。';
        $action = 'start';
    }
}

if ($action === 'do_reset') {
    $username = $_SESSION['recover_user'] ?? '';
    $verified = $_SESSION['recover_verified'] ?? false;
    $password1 = $_POST['password'] ?? '';
    $password2 = $_POST['password2'] ?? '';
    if (!$username || !$verified) {
        $message = '会话失效，请重新开始。';
        $action = 'start';
    } elseif ($password1 === '' || $password2 === '') {
        $message = '请输入新密码并确认。';
        $action = 'reset';
    } elseif ($password1 !== $password2) {
        $message = '两次输入的密码不一致。';
        $action = 'reset';
    } else {
        $stmt = $db->prepare("UPDATE users SET password_hash = ? WHERE username = ?");
        $stmt->execute([password_hash($password1, PASSWORD_DEFAULT), $username]);
        unset($_SESSION['recover_user'], $_SESSION['recover_question'], $_SESSION['recover_verified']);
        $message = '密码已更新，请使用新密码登录。';
        $action = 'done';
    }
}

echo '<!DOCTYPE html><html><head><meta charset="utf-8"><title>密保问题找回</title></head><body>';
echo '<h2>密保问题找回</h2>';

if ($message !== '') {
    echo '<p style="color:#c00;">' . h($message) . '</p>';
}

if ($action === 'start') {
    echo '<form method="post">';
    echo '<input type="hidden" name="action" value="question">';
    echo '<label>用户名：<input type="text" name="username" required></label> ';
    echo '<button type="submit">下一步</button>';
    echo '</form>';
    echo '<p>示例账号：demo，密码：demo1234。</p>';
} elseif ($action === 'answer') {
    $question = $_SESSION['recover_question'] ?? '';
    echo '<form method="post">';
    echo '<input type="hidden" name="action" value="verify">';
    echo '<p>密保问题：' . h($question) . '</p>';
    echo '<label>答案：<input type="text" name="answer" required></label> ';
    echo '<button type="submit">验证</button>';
    echo '</form>';
    echo '<p><a href="?restart=1">重新开始</a></p>';
} elseif ($action === 'reset') {
    echo '<form method="post">';
    echo '<input type="hidden" name="action" value="do_reset">';
    echo '<label>新密码：<input type="password" name="password" required></label><br>';
    echo '<label>确认密码：<input type="password" name="password2" required></label><br>';
    echo '<button type="submit">重置密码</button>';
    echo '</form>';
    echo '<p><a href="?restart=1">重新开始</a></p>';
} elseif ($action === 'done') {
    echo '<p><a href="?restart=1">返回</a></p>';
}

echo '</body></html>';
