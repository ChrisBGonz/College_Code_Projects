<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'zh-CN';
    public $itemsPerPage = 10;

    public function __construct(array $data = []) {
        foreach ($data as $key => $value) {
            if (property_exists($this, $key)) {
                $this->$key = $value;
            }
        }
    }
}

function loadPreferences() {
    if (empty($_COOKIE['prefs'])) {
        return new UserPreferences();
    }
    $decoded = base64_decode($_COOKIE['prefs'], true);
    if ($decoded === false) {
        return new UserPreferences();
    }
    $prefs = @unserialize($decoded, ['allowed_classes' => ['UserPreferences']]);
    if ($prefs instanceof UserPreferences) {
        return $prefs;
    }
    return new UserPreferences();
}

$prefs = loadPreferences();

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $prefs = new UserPreferences([
        'theme' => isset($_POST['theme']) ? $_POST['theme'] : $prefs->theme,
        'language' => isset($_POST['language']) ? $_POST['language'] : $prefs->language,
        'itemsPerPage' => isset($_POST['itemsPerPage']) ? max(1, (int)$_POST['itemsPerPage']) : $prefs->itemsPerPage
    ]);

    $cookieValue = base64_encode(serialize($prefs));
    $secure = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';
    setcookie('prefs', $cookieValue, time() + 31536000, '/', '', $secure, true);
}

header('Content-Type: text/html; charset=utf-8');
?>
<!doctype html>
<html>
<head><meta charset="utf-8"><title>User Preferences</title></head>
<body>
<form method="post">
<label>Theme:
<select name="theme">
<option value="light" <?php echo $prefs->theme === 'light' ? 'selected' : ''; ?>>Light</option>
<option value="dark" <?php echo $prefs->theme === 'dark' ? 'selected' : ''; ?>>Dark</option>
</select>
</label>
<br>
<label>Language:
<input type="text" name="language" value="<?php echo htmlspecialchars($prefs->language, ENT_QUOTES, 'UTF-8'); ?>">
</label>
<br>
<label>Items per page:
<input type="number" name="itemsPerPage" min="1" value="<?php echo (int)$prefs->itemsPerPage; ?>">
</label>
<br>
<button type="submit">Save</button>
</form>
</body>
</html>
