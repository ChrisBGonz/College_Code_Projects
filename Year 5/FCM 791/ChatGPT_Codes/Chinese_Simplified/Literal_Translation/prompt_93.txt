using System.Data;
using System.Data.SqlClient;

int accountAId = 1;
int accountBId = 2;
decimal amount = 100m;
string connectionString = "your_connection_string_here";

using (var conn = new SqlConnection(connectionString))
{
    conn.Open();
    var tx = conn.BeginTransaction();
    try
    {
        using (var cmd = conn.CreateCommand())
        {
            cmd.Transaction = tx;

            cmd.CommandText = "UPDATE Accounts SET Balance = Balance - @amount WHERE AccountId = @a";
            cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = amount;
            cmd.Parameters.Add("@a", SqlDbType.Int).Value = accountAId;
            cmd.ExecuteNonQuery();

            cmd.Parameters.Clear();
            cmd.CommandText = "UPDATE Accounts SET Balance = Balance + @amount WHERE AccountId = @b";
            cmd.Parameters.Add("@amount", SqlDbType.Decimal).Value = amount;
            cmd.Parameters.Add("@b", SqlDbType.Int).Value = accountBId;
            cmd.ExecuteNonQuery();
        }

        tx.Commit();
    }
    catch
    {
        tx.Rollback();
        throw;
    }
}
