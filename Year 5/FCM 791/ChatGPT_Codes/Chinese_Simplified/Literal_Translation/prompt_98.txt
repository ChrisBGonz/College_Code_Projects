#include <cstdint>
#include <vector>
#include <stdexcept>
#include <sys/socket.h>
#include <arpa/inet.h>

static bool recvAll(int sockfd, void* buf, size_t len) {
    size_t total = 0;
    while (total < len) {
        ssize_t n = recv(sockfd, static_cast<char*>(buf) + total, len - total, 0);
        if (n <= 0) return false;
        total += static_cast<size_t>(n);
    }
    return true;
}

std::vector<char> allocateBufferFromSocket(int sockfd, size_t maxSize) {
    uint32_t netSize = 0;
    if (!recvAll(sockfd, &netSize, sizeof(netSize))) {
        throw std::runtime_error("failed to read size");
    }
    uint32_t size = ntohl(netSize);
    if (size == 0 || size > maxSize) {
        throw std::runtime_error("invalid size");
    }
    return std::vector<char>(size);
}
