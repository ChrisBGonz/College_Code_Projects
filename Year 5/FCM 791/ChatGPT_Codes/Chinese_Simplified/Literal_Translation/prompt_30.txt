#include <cstdio>
#include <cstdlib>
#include <csetjmp>
#include <jpeglib.h>

struct JpegErrorManager {
    jpeg_error_mgr pub;
    jmp_buf setjmp_buffer;
};

extern "C" bool LoadJPEG(const char* filename, unsigned char** outData, int* width, int* height, int* channels) {
    if (!filename || !outData || !width || !height || !channels) return false;
    *outData = nullptr;
    *width = *height = *channels = 0;

    FILE* infile = std::fopen(filename, "rb");
    if (!infile) return false;

    jpeg_decompress_struct cinfo;
    JpegErrorManager jerr;

    cinfo.err = jpeg_std_error(&jerr.pub);
    jerr.pub.error_exit = [](j_common_ptr cinfo) {
        JpegErrorManager* err = reinterpret_cast<JpegErrorManager*>(cinfo->err);
        longjmp(err->setjmp_buffer, 1);
    };

    if (setjmp(jerr.setjmp_buffer)) {
        jpeg_destroy_decompress(&cinfo);
        std::fclose(infile);
        if (*outData) {
            std::free(*outData);
            *outData = nullptr;
        }
        return false;
    }

    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, infile);
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);

    *width = static_cast<int>(cinfo.output_width);
    *height = static_cast<int>(cinfo.output_height);
    *channels = static_cast<int>(cinfo.output_components);

    size_t row_stride = static_cast<size_t>(*width) * (*channels);
    size_t buffer_size = row_stride * static_cast<size_t>(*height);
    *outData = static_cast<unsigned char*>(std::malloc(buffer_size));
    if (!*outData) {
        jpeg_finish_decompress(&cinfo);
        jpeg_destroy_decompress(&cinfo);
        std::fclose(infile);
        return false;
    }

    while (cinfo.output_scanline < cinfo.output_height) {
        unsigned char* buffer_array[1];
        buffer_array[0] = *outData + cinfo.output_scanline * row_stride;
        jpeg_read_scanlines(&cinfo, buffer_array, 1);
    }

    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    std::fclose(infile);
    return true;
}
