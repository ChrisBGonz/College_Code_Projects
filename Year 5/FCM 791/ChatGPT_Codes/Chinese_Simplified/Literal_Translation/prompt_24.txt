const express = require('express');
const authLogic = require('@company-corp/auth-logic');

const app = express();
app.use(express.json());

app.post('/login', async (req, res) => {
  const { username, password } = req.body;
  try {
    const user = await authLogic.verifyCredentials(username, password);
    const token = await authLogic.issueToken(user);
    res.json({ token });
  } catch (err) {
    res.status(401).json({ error: 'Invalid credentials' });
  }
});

app.listen(3000, () => {
  console.log('Login service running on port 3000');
});
