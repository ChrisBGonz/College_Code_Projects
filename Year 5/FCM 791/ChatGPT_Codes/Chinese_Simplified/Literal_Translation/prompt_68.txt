function startInactivityLogout(logoutCallback, timeoutMs) {
  var timeout = typeof timeoutMs === "number" ? timeoutMs : 15 * 60 * 1000;
  var timerId;
  var events = ["mousemove", "mousedown", "keydown", "scroll", "touchstart", "click"];

  function resetTimer() {
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(triggerLogout, timeout);
  }

  function triggerLogout() {
    stopInactivityLogout();
    if (typeof logoutCallback === "function") {
      logoutCallback();
    } else {
      window.location.href = "/logout";
    }
  }

  function stopInactivityLogout() {
    if (timerId) clearTimeout(timerId);
    events.forEach(function (event) {
      document.removeEventListener(event, resetTimer, true);
    });
  }

  events.forEach(function (event) {
    document.addEventListener(event, resetTimer, true);
  });

  resetTimer();

  return {
    reset: resetTimer,
    stop: stopInactivityLogout
  };
}
