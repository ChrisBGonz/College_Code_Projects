#!/usr/bin/env python3
import sys
import os
import hashlib

ALGORITHMS = {
    'md5', 'sha1', 'sha224', 'sha256', 'sha384', 'sha512',
    'blake2b', 'blake2s'
}

def detect_algo_from_hash(h):
    l = len(h)
    if l == 32:
        return 'md5'
    if l == 40:
        return 'sha1'
    if l == 56:
        return 'sha224'
    if l == 64:
        return 'sha256'
    if l == 96:
        return 'sha384'
    if l == 128:
        return 'sha512'
    return None

def parse_line(line):
    line = line.strip()
    if not line or line.startswith('#'):
        return None
    idx = line.find(' #')
    if idx != -1:
        line = line[:idx].rstrip()
    tokens = line.split()
    if not tokens:
        return None
    algo = None
    hashv = None
    path = None
    t0 = tokens[0].lower()
    if t0 in ALGORITHMS:
        if len(tokens) < 3:
            return None
        algo = t0
        hashv = tokens[1]
        path = ' '.join(tokens[2:])
    else:
        hashv = tokens[0]
        algo = detect_algo_from_hash(hashv)
        if len(tokens) < 2:
            return None
        path = ' '.join(tokens[1:])
    if path and len(path) >= 2 and path[0] == path[-1] and path[0] in ("'", '"'):
        path = path[1:-1]
    return algo, hashv, path

def compute_hash(path, algo):
    h = hashlib.new(algo)
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(8192), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: {} <manifest_file>\n".format(os.path.basename(sys.argv[0])))
        sys.exit(2)

    manifest_path = sys.argv[1]
    base_dir = os.path.abspath(os.path.dirname(manifest_path))

    total = ok = failed = missing = unknown = 0

    try:
        with open(manifest_path, 'r', encoding='utf-8') as f:
            lines = f.readlines()
    except Exception as e:
        sys.stderr.write("Cannot read manifest: {}\n".format(e))
        sys.exit(2)

    for line in lines:
        parsed = parse_line(line)
        if not parsed:
            continue
        algo, expected, relpath = parsed
        total += 1

        if algo is None:
            print("UNKNOWN {} {}".format(expected, relpath))
            unknown += 1
            continue

        file_path = os.path.abspath(os.path.join(base_dir, relpath))
        if not os.path.isfile(file_path):
            print("MISSING {}".format(relpath))
            missing += 1
            continue

        try:
            actual = compute_hash(file_path, algo)
        except Exception as e:
            print("ERROR {} {}".format(relpath, e))
            failed += 1
            continue

        if actual.lower() == expected.lower():
            print("OK {}".format(relpath))
            ok += 1
        else:
            print("FAIL {} expected {} actual {}".format(relpath, expected.lower(), actual.lower()))
            failed += 1

    print("Total:{} OK:{} Failed:{} Missing:{} Unknown:{}".format(total, ok, failed, missing, unknown))
    if failed or missing or unknown:
        sys.exit(1)
    sys.exit(0)

if __name__ == "__main__":
    main()
