import base64
import hmac
import hashlib
import struct
import time
import argparse
import sys

def normalize_secret(secret):
    return ''.join(secret.strip().split()).upper()

def decode_secret(secret):
    secret = normalize_secret(secret)
    padding = (-len(secret)) % 8
    if padding:
        secret += '=' * padding
    return base64.b32decode(secret, casefold=True)

def generate_totp(secret_bytes, for_time=None, interval=30, digits=6, digest=hashlib.sha1):
    if for_time is None:
        for_time = time.time()
    counter = int(for_time // interval)
    msg = struct.pack(">Q", counter)
    h = hmac.new(secret_bytes, msg, digest).digest()
    offset = h[-1] & 0x0F
    code_int = (struct.unpack(">I", h[offset:offset+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code_int).zfill(digits)

def verify_totp(secret, code, window=1, interval=30, digits=6, digest=hashlib.sha1):
    code = str(code).strip()
    if not code.isdigit() or len(code) != digits:
        return False
    secret_bytes = decode_secret(secret)
    now = time.time()
    for w in range(-window, window + 1):
        if hmac.compare_digest(generate_totp(secret_bytes, now + w * interval, interval, digits, digest), code):
            return True
    return False

def main():
    parser = argparse.ArgumentParser(description="Verify a 6-digit TOTP code.")
    parser.add_argument("secret", help="Base32 encoded secret")
    parser.add_argument("code", nargs="?", help="6-digit TOTP code")
    parser.add_argument("--window", type=int, default=1, help="Allowed time steps on each side")
    parser.add_argument("--interval", type=int, default=30, help="Time step in seconds")
    parser.add_argument("--digits", type=int, default=6, help="Number of digits")
    args = parser.parse_args()

    code = args.code
    if code is None:
        try:
            code = input("TOTP code: ").strip()
        except EOFError:
            return 2

    if verify_totp(args.secret, code, args.window, args.interval, args.digits):
        print("VALID")
        return 0
    else:
        print("INVALID")
        return 1

if __name__ == "__main__":
    sys.exit(main())
