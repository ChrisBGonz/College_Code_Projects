const { spawn } = require('child_process');

function pingHost(domain_name) {
  return new Promise((resolve) => {
    if (typeof domain_name !== 'string' || domain_name.trim() === '') {
      resolve(false);
      return;
    }
    const host = domain_name.trim();
    const args = process.platform === 'win32' ? ['-n', '1', host] : ['-c', '1', host];
    const ping = spawn('ping', args);
    let done = false;
    const finish = (ok) => {
      if (!done) {
        done = true;
        resolve(ok);
      }
    };
    ping.on('error', () => finish(false));
    ping.on('close', (code) => finish(code === 0));
  });
}

module.exports = pingHost;
