import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Any, Tuple, Set
import time
import random

@dataclass
class Item:
    id: int
    name: str
    quantity: int
    attributes: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    hp: int
    position: Tuple[int, int]
    inventory: List[Item] = field(default_factory=list)
    stats: Dict[str, int] = field(default_factory=dict)
    quests: List[str] = field(default_factory=list)

@dataclass
class Enemy:
    species: str
    hp: int
    position: Tuple[int, int]
    ai_state: Dict[str, Any] = field(default_factory=dict)

@dataclass
class GameState:
    level: int
    players: List[Player]
    enemies: List[Enemy]
    map_state: Dict[str, Any]
    time_elapsed: float
    flags: Set[str]
    rng_state: Any
    meta: Dict[str, Any] = field(default_factory=dict)

def create_sample_game_state():
    items = [
        Item(1, "Sword", 1, {"damage": 12, "durability": 0.85}),
        Item(2, "Health Potion", 3, {"heal": 50}),
        Item(3, "Mana Potion", 2, {"mana": 30})
    ]
    player = Player(
        name="Archer",
        hp=120,
        position=(10, 5),
        inventory=items,
        stats={"strength": 8, "agility": 15, "intelligence": 10},
        quests=["Find the lost bow", "Rescue the villager"]
    )
    enemies = [
        Enemy("Goblin", 30, (3, 7), {"alerted": False, "patrol_route": [(3, 7), (4, 7), (4, 6)]}),
        Enemy("Dragon", 300, (50, 80), {"phase": 2, "breath_cooldown": 5.0})
    ]
    map_state = {
        "visited_tiles": {(0, 0), (0, 1), (1, 0), (1, 1)},
        "chests": {
            (2, 2): {"opened": True, "loot": []},
            (5, 3): {"opened": False, "loot": ["gold", "gem"]}
        },
        "weather": {"type": "rain", "intensity": 0.6}
    }
    flags = {"tutorial_completed", "boss_unlocked"}
    meta = {"version": "1.0.0", "saved_at": time.time()}
    return GameState(
        level=4,
        players=[player],
        enemies=enemies,
        map_state=map_state,
        time_elapsed=9876.54,
        flags=flags,
        rng_state=random.getstate(),
        meta=meta
    )

def save_game_state(game_state, filename):
    with open(filename, "wb") as f:
        pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename):
    with open(filename, "rb") as f:
        return pickle.load(f)

def main():
    filename = "game_state.pkl"
    game_state = create_sample_game_state()
    save_game_state(game_state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
