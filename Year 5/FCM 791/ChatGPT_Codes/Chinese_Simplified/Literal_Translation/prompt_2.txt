const express = require('express');
const router = express.Router();

router.use(express.json());

const users = new Map();
const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

router.post('/api/user/update-profile', async (req, res) => {
  try {
    const userId = (req.user && (req.user.id || req.user._id)) || (req.body && req.body.userId);
    if (!userId) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    const { email, displayName } = req.body || {};
    if (email === undefined && displayName === undefined) {
      return res.status(400).json({ error: 'No fields provided' });
    }

    if (email !== undefined) {
      if (typeof email !== 'string' || !emailRegex.test(email)) {
        return res.status(400).json({ error: 'Invalid email' });
      }
    }

    if (displayName !== undefined) {
      if (typeof displayName !== 'string') {
        return res.status(400).json({ error: 'Invalid display name' });
      }
      const name = displayName.trim();
      if (name.length === 0 || name.length > 100) {
        return res.status(400).json({ error: 'Invalid display name' });
      }
    }

    let user = req.user || users.get(userId);
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    if (email !== undefined) {
      user.email = email.trim().toLowerCase();
    }
    if (displayName !== undefined) {
      user.displayName = displayName.trim();
    }

    if (typeof user.save === 'function') {
      await user.save();
    } else {
      users.set(userId, user);
    }

    res.json({
      success: true,
      user: {
        id: user.id || user._id || userId,
        email: user.email,
        displayName: user.displayName
      }
    });
  } catch (err) {
    res.status(500).json({ error: 'Internal server error' });
  }
});

module.exports = { router, users };
