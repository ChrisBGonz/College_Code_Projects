package dashboard

import (
	"net/http"
	"net/url"
	"strings"
)

func HasInternalAccess(r *http.Request) bool {
	if r == nil {
		return false
	}
	ref := r.Referer()
	if ref == "" {
		return false
	}
	u, err := url.Parse(ref)
	if err != nil {
		return false
	}
	return strings.EqualFold(u.Hostname(), "admin.internal.com")
}
