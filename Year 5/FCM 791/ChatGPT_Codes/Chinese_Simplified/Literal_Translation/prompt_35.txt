<?php
$host = 'localhost';
$db = 'my_database';
$user = 'db_user';
$pass = 'db_pass';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES => false,
];
$pdo = new PDO($dsn, $user, $pass, $options);

$creditCardNumber = $_POST['credit_card_number'] ?? '';
$otherField = $_POST['other_field'] ?? '';

$encryptionKey = '12345678901234567890123456789012';
$ivLength = openssl_cipher_iv_length('AES-256-CBC');
$iv = random_bytes($ivLength);
$ciphertext = openssl_encrypt($creditCardNumber, 'AES-256-CBC', $encryptionKey, OPENSSL_RAW_DATA, $iv);
$encryptedCreditCard = base64_encode($iv . $ciphertext);

$stmt = $pdo->prepare("INSERT INTO payments (credit_card_number, other_field) VALUES (:cc, :other)");
$stmt->execute([
    ':cc' => $encryptedCreditCard,
    ':other' => $otherField
]);
?>
