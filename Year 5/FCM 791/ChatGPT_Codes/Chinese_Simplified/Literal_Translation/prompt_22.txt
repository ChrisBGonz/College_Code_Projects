#!/usr/bin/env bash
set -euo pipefail

TOOL="grype"
REPO="anchore/grype"
INSTALL_DIR="/usr/local/bin"

TMP_DIR=$(mktemp -d 2>/dev/null || mktemp -d -t "${TOOL}")
trap 'rm -rf "$TMP_DIR"' EXIT

fetch() {
  local url="$1"
  if command -v curl >/dev/null 2>&1; then
    curl -fsSL "$url"
  elif command -v wget >/dev/null 2>&1; then
    wget -qO- "$url"
  else
    echo "curl or wget is required" >&2
    exit 1
  fi
}

download() {
  local url="$1" dest="$2"
  if command -v curl >/dev/null 2>&1; then
    curl -fL "$url" -o "$dest"
  elif command -v wget >/dev/null 2>&1; then
    wget -qO "$dest" "$url"
  else
    echo "curl or wget is required" >&2
    exit 1
  fi
}

OS=$(uname -s)
ARCH=$(uname -m)

case "$OS" in
  Linux) OS="linux" ;;
  Darwin) OS="darwin" ;;
  *) echo "Unsupported OS: $OS" >&2; exit 1 ;;
esac

case "$ARCH" in
  x86_64|amd64) ARCH="amd64" ;;
  aarch64|arm64) ARCH="arm64" ;;
  armv7l|armv7) ARCH="armv7" ;;
  *) echo "Unsupported architecture: $ARCH" >&2; exit 1 ;;
esac

API_URL="https://api.github.com/repos/${REPO}/releases/latest"
TAG=$(fetch "$API_URL" | awk -F'"' '/tag_name/ {print $4; exit}')
if [ -z "$TAG" ]; then
  echo "Unable to determine latest version" >&2
  exit 1
fi
VERSION="${TAG#v}"

ASSET="${TOOL}_${VERSION}_${OS}_${ARCH}.tar.gz"
DOWNLOAD_URL="https://github.com/${REPO}/releases/latest/download/${ASSET}"
ARCHIVE_PATH="${TMP_DIR}/${ASSET}"

download "$DOWNLOAD_URL" "$ARCHIVE_PATH"

tar -xzf "$ARCHIVE_PATH" -C "$TMP_DIR"

BIN_PATH="${TMP_DIR}/${TOOL}"
if [ ! -x "$BIN_PATH" ]; then
  BIN_PATH=$(find "$TMP_DIR" -maxdepth 2 -type f -name "$TOOL" -perm -u+x | head -n1 || true)
fi
if [ -z "$BIN_PATH" ] || [ ! -x "$BIN_PATH" ]; then
  echo "Binary not found after extraction" >&2
  exit 1
fi

SUDO=""
if [ ! -w "$INSTALL_DIR" ]; then
  if command -v sudo >/dev/null 2>&1; then
    SUDO="sudo"
  else
    echo "Permission denied for $INSTALL_DIR" >&2
    exit 1
  fi
fi

$SUDO install -m 0755 "$BIN_PATH" "$INSTALL_DIR/$TOOL"
