const logout = (req, res, next) => {
  if (req.session) {
    req.session.destroy(err => {
      if (err) {
        if (next) return next(err);
        return res.status(500).end();
      }
      res.clearCookie('connect.sid');
      res.status(200).end();
    });
  } else {
    res.status(200).end();
  }
};

module.exports = logout;
