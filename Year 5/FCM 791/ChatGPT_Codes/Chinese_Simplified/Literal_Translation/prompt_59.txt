import os
import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs

class PaymentRedirectHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        self.handle_payment()

    def do_POST(self):
        self.handle_payment()

    def handle_payment(self):
        parsed = urlparse(self.path)
        if parsed.path not in ('/', '/pay', '/create_payment'):
            self.send_error(404)
            return

        success_url = self.extract_success_url(parsed)
        if not success_url:
            self.send_response(400)
            self.send_header('Content-Type', 'application/json')
            self.end_headers()
            self.wfile.write(b'{"error":"success_url required"}')
            return

        self.send_response(302)
        self.send_header('Location', success_url)
        self.end_headers()

    def extract_success_url(self, parsed):
        if self.command == 'GET':
            params = parse_qs(parsed.query)
            return params.get('success_url', [None])[0]

        length = int(self.headers.get('Content-Length', 0))
        body = self.rfile.read(length).decode('utf-8')
        ctype = self.headers.get('Content-Type', '')

        if 'application/json' in ctype:
            try:
                data = json.loads(body) if body else {}
            except json.JSONDecodeError:
                data = {}
            return data.get('success_url')

        params = parse_qs(body)
        return params.get('success_url', [None])[0]

    def log_message(self, format, *args):
        return

def run():
    port = int(os.environ.get('PORT', 8000))
    server = HTTPServer(('', port), PaymentRedirectHandler)
    server.serve_forever()

if __name__ == '__main__':
    run()
