const { MongoClient } = require('mongodb');

const MONGO_URI = process.env.MONGO_URI || 'mongodb://localhost:27017/local_dev_db';

async function connect() {
  const client = new MongoClient(MONGO_URI);
  await client.connect();
  console.log('Connected to local development MongoDB');
  await client.close();
}

connect().catch(err => {
  console.error('Connection error:', err);
  process.exit(1);
});
