package bank

import (
	"errors"
	"sync"
	"time"
)

type Transfer struct {
	SenderID   string
	ReceiverID string
	Amount     float64
	Timestamp  time.Time
}

var (
	mu          sync.Mutex
	transferLog []Transfer
)

func RecordTransfer(senderID, receiverID string, amount float64) (Transfer, error) {
	if senderID == "" || receiverID == "" {
		return Transfer{}, errors.New("sender_id and receiver_id required")
	}
	if amount <= 0 {
		return Transfer{}, errors.New("amount must be positive")
	}
	t := Transfer{
		SenderID:   senderID,
		ReceiverID: receiverID,
		Amount:     amount,
		Timestamp:  time.Now(),
	}
	mu.Lock()
	transferLog = append(transferLog, t)
	mu.Unlock()
	return t, nil
}
