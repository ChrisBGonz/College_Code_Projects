using System;
using System.Web;
using System.Web.Security;

public static class RememberMeAuth
{
    public static void IssueRememberMeCookie(HttpResponse response, string userName, bool rememberMe, int persistentDays = 30)
    {
        if (response == null) throw new ArgumentNullException(nameof(response));
        if (string.IsNullOrWhiteSpace(userName)) throw new ArgumentException(nameof(userName));

        DateTime now = DateTime.UtcNow;
        DateTime expiry = rememberMe ? now.AddDays(persistentDays) : now.AddMinutes(FormsAuthentication.Timeout.TotalMinutes);

        var ticket = new FormsAuthenticationTicket(
            1,
            userName,
            now,
            expiry,
            rememberMe,
            string.Empty,
            FormsAuthentication.FormsCookiePath
        );

        string encrypted = FormsAuthentication.Encrypt(ticket);
        var cookie = new HttpCookie(FormsAuthentication.FormsCookieName, encrypted)
        {
            HttpOnly = true,
            Secure = FormsAuthentication.RequireSSL,
            Path = FormsAuthentication.FormsCookiePath
        };

        if (!string.IsNullOrEmpty(FormsAuthentication.CookieDomain))
            cookie.Domain = FormsAuthentication.CookieDomain;

        if (rememberMe)
            cookie.Expires = expiry;

        response.Cookies.Remove(cookie.Name);
        response.Cookies.Add(cookie);
    }
}
