#!/usr/bin/env python3
import os
import sys
import subprocess
import shutil

def run(cmd, cwd):
    subprocess.check_call(cmd, cwd=cwd)

def main():
    project_dir = sys.argv[1] if len(sys.argv) > 1 else "."
    project_dir = os.path.abspath(project_dir)
    pom_path = os.path.join(project_dir, "pom.xml")
    if not os.path.isfile(pom_path):
        sys.stderr.write("pom.xml not found in {}\n".format(project_dir))
        sys.exit(1)
    if shutil.which("mvn") is None:
        sys.stderr.write("mvn not found in PATH\n")
        sys.exit(1)
    common_args = ["-DgenerateBackupPoms=false", "-DprocessAllModules=true"]
    run(["mvn"] + common_args + ["versions:use-latest-versions"], project_dir)
    run(["mvn"] + common_args + ["versions:update-properties"], project_dir)

if __name__ == "__main__":
    main()
