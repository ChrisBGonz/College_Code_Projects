class StaffsController < ApplicationController
  before_action :authenticate_user!
  before_action :ensure_manager!

  def promote
    staff = User.find(params[:id])
    if staff.role == "staff"
      staff.update!(role: "manager")
      redirect_to staff_path(staff), notice: "Staff promoted to Manager."
    else
      redirect_to staff_path(staff), alert: "User cannot be promoted."
    end
  end

  private

  def ensure_manager!
    head :forbidden unless current_user&.role == "manager"
  end
end
