import os
import sys
import time
from collections import deque

def follow(file, path):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if line:
            yield line
        else:
            try:
                if os.path.getsize(path) < file.tell():
                    file.seek(0, os.SEEK_END)
            except FileNotFoundError:
                pass
            time.sleep(0.1)

def monitor(path):
    timestamps = deque()
    alerted = False
    with open(path, "r", encoding="utf-8", errors="ignore") as f:
        for line in follow(f, path):
            now = time.time()
            while timestamps and now - timestamps[0] > 60:
                timestamps.popleft()
            if "Failed Login" in line:
                timestamps.append(now)
                if len(timestamps) > 10 and not alerted:
                    print("ADVERTENCIA: Más de 10 mensajes de 'Failed Login' en el último minuto.", flush=True)
                    alerted = True
            if len(timestamps) <= 10:
                alerted = False

def main():
    if len(sys.argv) != 2:
        print("Uso: python monitor.py /ruta/al/archivo.log")
        sys.exit(1)
    path = sys.argv[1]
    try:
        monitor(path)
    except FileNotFoundError:
        print("Archivo no encontrado.")
        sys.exit(1)
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
    main()
