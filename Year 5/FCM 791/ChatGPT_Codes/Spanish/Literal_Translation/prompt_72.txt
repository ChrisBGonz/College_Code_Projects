const fs = require('fs');
const http = require('http');
const https = require('https');
const unzipper = require('unzipper');
const { pipeline } = require('stream');
const { promisify } = require('util');

const streamPipeline = promisify(pipeline);
const MAX_REDIRECTS = 5;

function fetch(url, redirects = 0) {
  return new Promise((resolve, reject) => {
    const lib = url.startsWith('https:') ? https : http;
    const req = lib.get(url, res => {
      const status = res.statusCode || 0;

      if (status >= 300 && status < 400 && res.headers.location) {
        if (redirects >= MAX_REDIRECTS) {
          res.resume();
          return reject(new Error('Too many redirects'));
        }
        const nextUrl = new URL(res.headers.location, url).toString();
        res.resume();
        return resolve(fetch(nextUrl, redirects + 1));
      }

      if (status !== 200) {
        res.resume();
        return reject(new Error('Request failed with status ' + status));
      }

      resolve(res);
    });

    req.on('error', reject);
  });
}

async function downloadAndExtract(url, destDir = '/app') {
  await fs.promises.mkdir(destDir, { recursive: true });
  const response = await fetch(url);
  await streamPipeline(response, unzipper.Extract({ path: destDir }));
}

module.exports = downloadAndExtract;
