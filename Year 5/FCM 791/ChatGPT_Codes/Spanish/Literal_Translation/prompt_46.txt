import requests
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import urlparse, parse_qs

class RedirectHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        params = parse_qs(urlparse(self.path).query)
        redirect_url = params.get('redirect_url', [None])[0]
        if not redirect_url:
            self.send_response(400)
            self.send_header('Content-Type', 'text/plain; charset=utf-8')
            self.end_headers()
            self.wfile.write(b'redirect_url parameter is required')
            return
        parsed = requests.utils.urlparse(redirect_url)
        if parsed.scheme not in ('http', 'https') or not parsed.netloc:
            self.send_response(400)
            self.send_header('Content-Type', 'text/plain; charset=utf-8')
            self.end_headers()
            self.wfile.write(b'invalid redirect_url')
            return
        redirect_url = requests.utils.requote_uri(redirect_url)
        self.send_response(302)
        self.send_header('Location', redirect_url)
        self.end_headers()

def main():
    HTTPServer(('0.0.0.0', 8000), RedirectHandler).serve_forever()

if __name__ == '__main__':
    main()
