from flask import Flask, request, jsonify
import ast
import operator as op
import math

app = Flask(__name__)

allowed_operators = {
    ast.Add: op.add,
    ast.Sub: op.sub,
    ast.Mult: op.mul,
    ast.Div: op.truediv,
    ast.FloorDiv: op.floordiv,
    ast.Mod: op.mod,
    ast.Pow: op.pow,
    ast.USub: op.neg,
    ast.UAdd: op.pos
}

allowed_names = {k: getattr(math, k) for k in dir(math) if not k.startswith("_")}

def eval_node(node):
    if isinstance(node, ast.Expression):
        return eval_node(node.body)
    if isinstance(node, ast.Num):
        return node.n
    if isinstance(node, ast.Constant):
        if isinstance(node.value, (int, float)):
            return node.value
        raise ValueError("Constante no permitida")
    if isinstance(node, ast.BinOp):
        op_type = type(node.op)
        if op_type in allowed_operators:
            return allowed_operators[op_type](eval_node(node.left), eval_node(node.right))
        raise ValueError("Operador no permitido")
    if isinstance(node, ast.UnaryOp):
        op_type = type(node.op)
        if op_type in allowed_operators:
            return allowed_operators[op_type](eval_node(node.operand))
        raise ValueError("Operador no permitido")
    if isinstance(node, ast.Name):
        if node.id in allowed_names:
            return allowed_names[node.id]
        raise ValueError("Nombre no permitido")
    if isinstance(node, ast.Call):
        if not isinstance(node.func, ast.Name):
            raise ValueError("Llamada no permitida")
        func_name = node.func.id
        if func_name not in allowed_names or not callable(allowed_names[func_name]):
            raise ValueError("Función no permitida")
        if node.keywords:
            raise ValueError("Argumentos no permitidos")
        args = [eval_node(arg) for arg in node.args]
        return allowed_names[func_name](*args)
    raise ValueError("Expresión no soportada")

def safe_eval(expr):
    try:
        tree = ast.parse(expr, mode='eval')
    except SyntaxError:
        raise ValueError("Expresión inválida")
    return eval_node(tree)

@app.route('/calculate', methods=['GET', 'POST'])
def calculate():
    data = request.get_json(silent=True) or {}
    expr = request.values.get('expr') or data.get('expr')
    if not expr:
        return jsonify(error="Se requiere una expresión"), 400
    try:
        result = safe_eval(expr)
    except Exception as e:
        return jsonify(error=str(e)), 400
    return jsonify(result=result)

if __name__ == '__main__':
    app.run()
