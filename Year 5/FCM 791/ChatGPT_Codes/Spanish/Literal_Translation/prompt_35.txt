<?php
$host = 'localhost';
$db = 'mydb';
$user = 'myuser';
$pass = 'mypassword';
$dsn = "mysql:host=$host;dbname=$db;charset=utf8mb4";

$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES => false,
];

$pdo = new PDO($dsn, $user, $pass, $options);

$credit_card_number = $_POST['credit_card_number'] ?? null;
if ($credit_card_number === null) {
    http_response_code(400);
    exit;
}

$key = hex2bin('0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef');
$iv_length = openssl_cipher_iv_length('aes-256-cbc');
$iv = random_bytes($iv_length);
$ciphertext = openssl_encrypt($credit_card_number, 'aes-256-cbc', $key, OPENSSL_RAW_DATA, $iv);
$encrypted = base64_encode($iv . $ciphertext);

$stmt = $pdo->prepare('INSERT INTO payments (credit_card_number) VALUES (:cc)');
$stmt->execute([':cc' => $encrypted]);

echo 'stored';
?>
