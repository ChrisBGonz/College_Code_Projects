using System;
using System.IO;

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
}

public class User
{
    public string Id { get; set; }
    public MfaSettings Mfa { get; set; }
}

public class MfaService
{
    private readonly string _logPath = "mfa_audit.log";

    public void UpdateMfaSettings(User user, MfaSettings newSettings)
    {
        var oldSettings = user.Mfa;
        user.Mfa = newSettings;

        LogMfaChange(user.Id, oldSettings, newSettings);
    }

    private void LogMfaChange(string userId, MfaSettings oldSettings, MfaSettings newSettings)
    {
        var entry =
            $"{DateTime.UtcNow:o}\tUser:{userId}\tEnabled:{oldSettings?.Enabled}->{newSettings.Enabled}\tMethod:{oldSettings?.Method}->{newSettings.Method}";
        File.AppendAllText(_logPath, entry + Environment.NewLine);
    }
}
