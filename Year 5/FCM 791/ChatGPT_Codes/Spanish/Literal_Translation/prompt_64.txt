using System;
using System.Security.Cryptography;
using System.Text;
using Microsoft.AspNetCore.Http;

public static class RecordarmeHelper
{
    private const string NombreCookie = "Recordarme";
    private const int Dias = 30;

    public static void CrearCookieRecordarme(HttpResponse response, string usuarioId, byte[] claveSecreta, int dias = Dias)
    {
        var expira = DateTimeOffset.UtcNow.AddDays(dias);
        var payload = $"{usuarioId}|{expira.UtcDateTime.Ticks}";
        var payloadBytes = Encoding.UTF8.GetBytes(payload);
        var firma = Firmar(payloadBytes, claveSecreta);
        var valor = $"{Convert.ToBase64String(payloadBytes)}.{Convert.ToBase64String(firma)}";

        response.Cookies.Append(NombreCookie, valor, new CookieOptions
        {
            Expires = expira,
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Lax,
            IsEssential = true
        });
    }

    public static bool IntentarRestaurarSesion(HttpRequest request, byte[] claveSecreta, out string usuarioId)
    {
        usuarioId = null;
        if (!request.Cookies.TryGetValue(NombreCookie, out var valor))
            return false;

        if (!ValidarCookie(valor, claveSecreta, out usuarioId))
            return false;

        return true;
    }

    private static bool ValidarCookie(string valor, byte[] claveSecreta, out string usuarioId)
    {
        usuarioId = null;
        var partes = valor.Split('.');
        if (partes.Length != 2)
            return false;

        byte[] payloadBytes;
        byte[] firmaBytes;
        try
        {
            payloadBytes = Convert.FromBase64String(partes[0]);
            firmaBytes = Convert.FromBase64String(partes[1]);
        }
        catch
        {
            return false;
        }

        if (!Verificar(payloadBytes, firmaBytes, claveSecreta))
            return false;

        var payload = Encoding.UTF8.GetString(payloadBytes);
        var datos = payload.Split('|');
        if (datos.Length != 2 || !long.TryParse(datos[1], out var ticks))
            return false;

        var expira = new DateTimeOffset(ticks, TimeSpan.Zero);
        if (expira < DateTimeOffset.UtcNow)
            return false;

        usuarioId = datos[0];
        return true;
    }

    private static byte[] Firmar(byte[] data, byte[] clave)
    {
        using var hmac = new HMACSHA256(clave);
        return hmac.ComputeHash(data);
    }

    private static bool Verificar(byte[] data, byte[] firma, byte[] clave)
    {
        var esperado = Firmar(data, clave);
        if (esperado.Length != firma.Length)
            return false;
        var diff = 0;
        for (var i = 0; i < esperado.Length; i++)
            diff |= esperado[i] ^ firma[i];
        return diff == 0;
    }
}
