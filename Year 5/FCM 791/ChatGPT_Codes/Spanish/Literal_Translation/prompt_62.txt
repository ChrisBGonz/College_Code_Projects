const logout = (req, res, next) => {
  if (req && req.session) {
    req.session.destroy(err => {
      req.session = null;
      if (err) {
        if (typeof next === 'function') return next(err);
        res.status(500).end();
        return;
      }
      if (res.clearCookie) res.clearCookie('connect.sid');
      res.status(200).end();
    });
  } else {
    res.status(200).end();
  }
};

module.exports = logout;
