package main

import (
	"bytes"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"reflect"
	"strconv"
	"strings"
)

type DatabaseConfig struct {
	Host     string `json:"host"`
	Port     int    `json:"port"`
	User     string `json:"user"`
	Password string `json:"password"`
}

type AppConfig struct {
	Name           string         `json:"name"`
	Debug          bool           `json:"debug"`
	MaxConnections int            `json:"max_connections"`
	Database       DatabaseConfig `json:"database"`
	Tags           []string       `json:"tags"`
	Limits         map[string]int `json:"limits"`
}

func LoadConfigAndApply(path string, obj interface{}) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	var cfg map[string]interface{}
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.UseNumber()
	if err := dec.Decode(&cfg); err != nil {
		return err
	}
	return applyMapToValue(cfg, reflect.ValueOf(obj))
}

func applyMapToValue(cfg map[string]interface{}, v reflect.Value) error {
	if v.Kind() != reflect.Ptr || v.IsNil() {
		return errors.New("target must be a non-nil pointer")
	}
	if v.Elem().Kind() != reflect.Struct {
		return errors.New("target must point to struct")
	}
	return applyMapToStruct(cfg, v.Elem())
}

func applyMapToStruct(cfg map[string]interface{}, v reflect.Value) error {
	if v.Kind() != reflect.Struct {
		return errors.New("target is not struct")
	}
	t := v.Type()
	for i := 0; i < v.NumField(); i++ {
		fieldVal := v.Field(i)
		fieldType := t.Field(i)
		if fieldType.PkgPath != "" {
			continue
		}
		key := fieldType.Tag.Get("json")
		if key == "" {
			key = fieldType.Name
		} else {
			key = strings.Split(key, ",")[0]
			if key == "-" {
				continue
			}
			if key == "" {
				key = fieldType.Name
			}
		}
		if val, ok := cfg[key]; ok {
			if err := setValue(fieldVal, val); err != nil {
				return fmt.Errorf("field %s: %w", fieldType.Name, err)
			}
		}
	}
	return nil
}

func setValue(field reflect.Value, val interface{}) error {
	if !field.CanSet() {
		return nil
	}
	if val == nil {
		return nil
	}

	switch field.Kind() {
	case reflect.Interface:
		field.Set(reflect.ValueOf(val))
		return nil
	case reflect.Ptr:
		if field.IsNil() {
			field.Set(reflect.New(field.Type().Elem()))
		}
		return setValue(field.Elem(), val)
	case reflect.String:
		switch v := val.(type) {
		case string:
			field.SetString(v)
		case fmt.Stringer:
			field.SetString(v.String())
		default:
			field.SetString(fmt.Sprint(v))
		}
		return nil
	case reflect.Bool:
		switch v := val.(type) {
		case bool:
			field.SetBool(v)
		case string:
			b, err := strconv.ParseBool(v)
			if err != nil {
				return err
			}
			field.SetBool(b)
		default:
			return fmt.Errorf("expected bool, got %T", val)
		}
		return nil
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		i, err := convertToInt64(val)
		if err != nil {
			return err
		}
		field.SetInt(i)
		return nil
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		u, err := convertToUint64(val)
		if err != nil {
			return err
		}
		field.SetUint(u)
		return nil
	case reflect.Float32, reflect.Float64:
		f, err := convertToFloat64(val)
		if err != nil {
			return err
		}
		field.SetFloat(f)
		return nil
	case reflect.Struct:
		if m, ok := val.(map[string]interface{}); ok {
			return applyMapToStruct(m, field)
		}
		return fmt.Errorf("expected object, got %T", val)
	case reflect.Slice:
		if arr, ok := val.([]interface{}); ok {
			slice := reflect.MakeSlice(field.Type(), len(arr), len(arr))
			for i, item := range arr {
				if err := setValue(slice.Index(i), item); err != nil {
					return err
				}
			}
			field.Set(slice)
			return nil
		}
		rv := reflect.ValueOf(val)
		if rv.Kind() == reflect.Slice && rv.Type().AssignableTo(field.Type()) {
			field.Set(rv)
			return nil
		}
		return fmt.Errorf("expected array, got %T", val)
	case reflect.Map:
		if m, ok := val.(map[string]interface{}); ok {
			newMap := reflect.MakeMap(field.Type())
			for k, v := range m {
				keyVal := reflect.ValueOf(k)
				if !keyVal.Type().AssignableTo(field.Type().Key()) {
					if keyVal.Type().ConvertibleTo(field.Type().Key()) {
						keyVal = keyVal.Convert(field.Type().Key())
					} else if field.Type().Key().Kind() == reflect.String {
						keyVal = reflect.ValueOf(fmt.Sprint(k))
					} else {
						continue
					}
				}
				elem := reflect.New(field.Type().Elem()).Elem()
				if err := setValue(elem, v); err != nil {
					return err
				}
				newMap.SetMapIndex(keyVal, elem)
			}
			field.Set(newMap)
			return nil
		}
		return fmt.Errorf("expected map, got %T", val)
	default:
		return fmt.Errorf("unsupported kind: %s", field.Kind())
	}
}

func convertToInt64(val interface{}) (int64, error) {
	switch v := val.(type) {
	case json.Number:
		return v.Int64()
	case float64:
		return int64(v), nil
	case float32:
		return int64(v), nil
	case int:
		return int64(v), nil
	case int8:
		return int64(v), nil
	case int16:
		return int64(v), nil
	case int32:
		return int64(v), nil
	case int64:
		return v, nil
	case uint:
		return int64(v), nil
	case uint8:
		return int64(v), nil
	case uint16:
		return int64(v), nil
	case uint32:
		return int64(v), nil
	case uint64:
		return int64(v), nil
	case string:
		return strconv.ParseInt(v, 10, 64)
	default:
		return 0, fmt.Errorf("expected number, got %T", val)
	}
}

func convertToUint64(val interface{}) (uint64, error) {
	switch v := val.(type) {
	case json.Number:
		i, err := v.Int64()
		if err != nil {
			return 0, err
		}
		if i < 0 {
			return 0, fmt.Errorf("negative value")
		}
		return uint64(i), nil
	case float64:
		if v < 0 {
			return 0, fmt.Errorf("negative value")
		}
		return uint64(v), nil
	case float32:
		if v < 0 {
			return 0, fmt.Errorf("negative value")
		}
		return uint64(v), nil
	case int, int8, int16, int32, int64:
		i := reflect.ValueOf(v).Int()
		if i < 0 {
			return 0, fmt.Errorf("negative value")
		}
		return uint64(i), nil
	case uint, uint8, uint16, uint32, uint64:
		return reflect.ValueOf(v).Uint(), nil
	case string:
		return strconv.ParseUint(v, 10, 64)
	default:
		return 0, fmt.Errorf("expected number, got %T", val)
	}
}

func convertToFloat64(val interface{}) (float64, error) {
	switch v := val.(type) {
	case json.Number:
		return v.Float64()
	case float64:
		return v, nil
	case float32:
		return float64(v), nil
	case int, int8, int16, int32, int64:
		return float64(reflect.ValueOf(v).Int()), nil
	case uint, uint8, uint16, uint32, uint64:
		return float64(reflect.ValueOf(v).Uint()), nil
	case string:
		return strconv.ParseFloat(v, 64)
	default:
		return 0, fmt.Errorf("expected number, got %T", val)
	}
}

func main() {
	var cfg AppConfig
	if err := LoadConfigAndApply("config.json", &cfg); err != nil {
		fmt.Println("error:", err)
		return
	}
	fmt.Printf("%+v\n", cfg)
}
