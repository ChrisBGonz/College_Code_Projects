<?php
$uploadDir = '/ruta/compartida/empresa';
$maxFileSize = 20 * 1024 * 1024;
$allowedMime = [
    'application/pdf',
    'application/x-pdf',
    'application/acrobat',
    'applications/vnd.pdf',
    'text/pdf',
    'text/x-pdf'
];

$messages = [];
$success = false;

function getUploadErrorMessage($code)
{
    switch ($code) {
        case UPLOAD_ERR_INI_SIZE:
            return 'El archivo excede el tamaño permitido por el servidor.';
        case UPLOAD_ERR_FORM_SIZE:
            return 'El archivo excede el tamaño permitido por el formulario.';
        case UPLOAD_ERR_PARTIAL:
            return 'El archivo se cargó parcialmente.';
        case UPLOAD_ERR_NO_FILE:
            return 'No se seleccionó ningún archivo.';
        case UPLOAD_ERR_NO_TMP_DIR:
            return 'Falta la carpeta temporal en el servidor.';
        case UPLOAD_ERR_CANT_WRITE:
            return 'No se pudo escribir el archivo en el disco.';
        case UPLOAD_ERR_EXTENSION:
            return 'Una extensión de PHP detuvo la carga del archivo.';
        default:
            return 'Error desconocido al cargar el archivo.';
    }
}

function sanitizeFileName($name)
{
    $name = trim($name);
    $name = str_replace(['\\', '/'], '_', $name);
    if (function_exists('iconv')) {
        $name = iconv('UTF-8', 'ASCII//TRANSLIT//IGNORE', $name);
    }
    $name = preg_replace('/[^A-Za-z0-9_\-]+/', '_', $name);
    $name = trim($name, '_-');
    return $name;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['archivo'])) {
        $messages[] = 'No se recibió ningún archivo.';
    } else {
        $file = $_FILES['archivo'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $messages[] = getUploadErrorMessage($file['error']);
        } else {
            if ($file['size'] <= 0) {
                $messages[] = 'El archivo está vacío.';
            } elseif ($file['size'] > $maxFileSize) {
                $messages[] = 'El archivo excede el tamaño máximo permitido de 20 MB.';
            }

            $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            if ($extension !== 'pdf') {
                $messages[] = 'Solo se permiten archivos PDF.';
            }

            if (empty($messages)) {
                $finfo = new finfo(FILEINFO_MIME_TYPE);
                $mime = $finfo->file($file['tmp_name']);
                if (!in_array($mime, $allowedMime, true)) {
                    $messages[] = 'El archivo no es un PDF válido.';
                } else {
                    $handle = fopen($file['tmp_name'], 'rb');
                    $header = $handle ? fread($handle, 4) : '';
                    if ($handle) {
                        fclose($handle);
                    }
                    if ($header !== '%PDF') {
                        $messages[] = 'El archivo no tiene formato PDF válido.';
                    }
                }
            }

            if (empty($messages)) {
                if (!is_dir($uploadDir)) {
                    if (!@mkdir($uploadDir, 0770, true)) {
                        $messages[] = 'No se pudo crear el directorio de destino.';
                    }
                }
            }

            if (empty($messages)) {
                if (!is_writable($uploadDir)) {
                    $messages[] = 'El directorio de destino no tiene permisos de escritura.';
                }
            }

            if (empty($messages)) {
                $baseName = sanitizeFileName(pathinfo($file['name'], PATHINFO_FILENAME));
                if ($baseName === '') {
                    $baseName = 'archivo';
                }
                if (function_exists('random_bytes')) {
                    $rand = bin2hex(random_bytes(4));
                } else {
                    $rand = bin2hex(openssl_random_pseudo_bytes(4));
                }
                $targetName = $baseName . '_' . date('Ymd_His') . '_' . $rand . '.pdf';
                $targetPath = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $targetName;

                if (!move_uploaded_file($file['tmp_name'], $targetPath)) {
                    $messages[] = 'No se pudo mover el archivo al directorio de destino.';
                } else {
                    $success = true;
                    $messages[] = 'Archivo cargado correctamente: ' . $targetName;
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Carga de archivos PDF</title>
<style>
body { font-family: Arial, sans-serif; margin: 40px; }
h1 { margin-bottom: 20px; }
.messages { padding: 12px; border-radius: 4px; margin-bottom: 20px; }
.messages.success { background: #e6ffed; color: #0a6; border: 1px solid #0a6; }
.messages.error { background: #ffe6e6; color: #a00; border: 1px solid #a00; }
label { display: block; margin-bottom: 8px; }
input[type="file"] { margin-bottom: 12px; }
button { padding: 8px 16px; }
</style>
</head>
<body>
<h1>Carga de archivos PDF</h1>

<?php if (!empty($messages)): ?>
<div class="messages <?php echo $success ? 'success' : 'error'; ?>">
<ul>
<?php foreach ($messages as $msg): ?>
<li><?php echo htmlspecialchars($msg, ENT_QUOTES, 'UTF-8'); ?></li>
<?php endforeach; ?>
</ul>
</div>
<?php endif; ?>

<form method="post" enctype="multipart/form-data">
<label for="archivo">Seleccionar archivo PDF:</label>
<input type="file" name="archivo" id="archivo" accept="application/pdf" required>
<button type="submit">Cargar</button>
</form>
</body>
</html>
