const { spawn } = require('child_process');

function pingHost(nombre_de_dominio, timeout = 5000) {
  return new Promise((resolve, reject) => {
    if (typeof nombre_de_dominio !== 'string' || nombre_de_dominio.trim() === '') {
      return reject(new Error('Nombre de dominio inválido'));
    }

    const platform = process.platform;
    let args;

    if (platform === 'win32') {
      args = ['-n', '1', '-w', String(timeout), nombre_de_dominio];
    } else {
      const secs = Math.ceil(timeout / 1000);
      args = ['-c', '1', '-W', String(secs), nombre_de_dominio];
    }

    const ping = spawn('ping', args);
    let settled = false;

    const timer = setTimeout(() => {
      if (settled) return;
      settled = true;
      ping.kill();
      resolve(false);
    }, timeout + 1000);

    ping.on('error', () => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(false);
    });

    ping.on('close', (code) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      resolve(code === 0);
    });
  });
}

module.exports = { pingHost };
