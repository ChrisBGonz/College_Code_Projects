from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/free_trial', methods=['GET', 'POST'])
def free_trial():
    if request.cookies.get('has_used_trial'):
        return "Ya has utilizado la prueba gratuita.", 403
    if request.method == 'POST':
        resp = make_response("Registro de prueba completado.")
        resp.set_cookie('has_used_trial', 'true', max_age=60*60*24*365)
        return resp
    return '''
    <form method="post">
        <input name="email" type="email" required>
        <button type="submit">Iniciar prueba</button>
    </form>
    '''

if __name__ == '__main__':
    app.run()
