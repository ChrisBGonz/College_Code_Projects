<?php
session_start();

define('USERS_FILE', __DIR__ . '/users.json');

function normalizeAnswer($a) {
    return mb_strtolower(trim($a), 'UTF-8');
}

function loadUsers() {
    if (!file_exists(USERS_FILE)) {
        $default = [
            [
                'id' => 1,
                'email' => 'usuario@example.com',
                'secret_question' => '¿Cuál es el apellido de soltera de tu madre?',
                'secret_answer_hash' => password_hash(normalizeAnswer('García'), PASSWORD_DEFAULT),
                'password_hash' => password_hash('Password123', PASSWORD_DEFAULT)
            ]
        ];
        file_put_contents(USERS_FILE, json_encode($default, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
    }
    $data = json_decode(@file_get_contents(USERS_FILE), true);
    if (!is_array($data)) {
        $data = [];
    }
    return $data;
}

function saveUsers(array $users) {
    file_put_contents(USERS_FILE, json_encode($users, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

function findUserByEmail(array $users, $email) {
    foreach ($users as $idx => $u) {
        if (isset($u['email']) && strcasecmp($u['email'], $email) === 0) {
            return $idx;
        }
    }
    return null;
}

function getUserById(array $users, $id) {
    foreach ($users as $idx => $u) {
        if (isset($u['id']) && (string)$u['id'] === (string)$id) {
            return [$idx, $u];
        }
    }
    return [null, null];
}

$users = loadUsers();

if (isset($_GET['restart'])) {
    $_SESSION = [];
    if (session_id() !== '') {
        setcookie(session_name(), '', time() - 42000, '/');
    }
    session_destroy();
}

$step = $_SESSION['step'] ?? 'start';
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = $_POST['action'] ?? '';
    if ($action === 'start') {
        $email = trim($_POST['email'] ?? '');
        if ($email === '') {
            $error = 'Debes ingresar tu correo electrónico.';
        } else {
            $idx = findUserByEmail($users, $email);
            if ($idx === null) {
                $error = 'No se encontró la cuenta.';
            } else {
                $_SESSION['user_id'] = $users[$idx]['id'];
                $_SESSION['step'] = 'question';
                $step = 'question';
            }
        }
    } elseif ($action === 'question') {
        $userId = $_SESSION['user_id'] ?? null;
        if (!$userId) {
            $_SESSION['step'] = 'start';
            $step = 'start';
        } else {
            list($idx, $user) = getUserById($users, $userId);
            if (!$user) {
                $_SESSION['step'] = 'start';
                $step = 'start';
            } else {
                $answer = $_POST['answer'] ?? '';
                if (password_verify(normalizeAnswer($answer), $user['secret_answer_hash'])) {
                    $_SESSION['step'] = 'reset';
                    $step = 'reset';
                } else {
                    $error = 'Respuesta incorrecta.';
                }
            }
        }
    } elseif ($action === 'reset') {
        $userId = $_SESSION['user_id'] ?? null;
        if (!$userId) {
            $_SESSION['step'] = 'start';
            $step = 'start';
        } else {
            list($idx, $user) = getUserById($users, $userId);
            if (!$user) {
                $_SESSION['step'] = 'start';
                $step = 'start';
            } else {
                $pass = $_POST['password'] ?? '';
                $pass2 = $_POST['password2'] ?? '';
                if ($pass === '' || $pass2 === '') {
                    $error = 'Debes completar ambos campos de contraseña.';
                } elseif ($pass !== $pass2) {
                    $error = 'Las contraseñas no coinciden.';
                } elseif (mb_strlen($pass, '8bit') < 8) {
                    $error = 'La contraseña debe tener al menos 8 caracteres.';
                } else {
                    $users[$idx]['password_hash'] = password_hash($pass, PASSWORD_DEFAULT);
                    saveUsers($users);
                    $_SESSION = [];
                    if (session_id() !== '') {
                        setcookie(session_name(), '', time() - 42000, '/');
                    }
                    session_destroy();
                    $step = 'done';
                }
            }
        }
    }
}

$currentUser = null;
if (in_array($step, ['question', 'reset'], true)) {
    $userId = $_SESSION['user_id'] ?? null;
    if ($userId) {
        list($idx, $u) = getUserById($users, $userId);
        if ($u) {
            $currentUser = $u;
        } else {
            $step = 'start';
            $_SESSION['step'] = 'start';
        }
    } else {
        $step = 'start';
    }
}

?>
<!doctype html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Recuperación por pregunta secreta</title>
<style>
body { font-family: Arial, sans-serif; margin: 2rem; }
form { max-width: 400px; }
label { display: block; margin-top: 1rem; }
input[type="text"], input[type="email"], input[type="password"] { width: 100%; padding: 0.5rem; box-sizing: border-box; }
button { margin-top: 1rem; padding: 0.5rem 1rem; }
.error { color: #b00020; margin-top: 1rem; }
.notice { color: #006400; margin-top: 1rem; }
</style>
</head>
<body>
<h1>Recuperación de cuenta</h1>
<?php if ($error): ?>
<div class="error"><?php echo htmlspecialchars($error, ENT_QUOTES, 'UTF-8'); ?></div>
<?php endif; ?>

<?php if ($step === 'start'): ?>
<form method="post" autocomplete="off">
    <input type="hidden" name="action" value="start">
    <label>Correo electrónico:
        <input type="email" name="email" required>
    </label>
    <button type="submit">Continuar</button>
</form>
<?php elseif ($step === 'question' && $currentUser): ?>
<form method="post" autocomplete="off">
    <input type="hidden" name="action" value="question">
    <label><?php echo htmlspecialchars($currentUser['secret_question'], ENT_QUOTES, 'UTF-8'); ?>
        <input type="text" name="answer" required>
    </label>
    <button type="submit">Verificar</button>
</form>
<?php elseif ($step === 'reset' && $currentUser): ?>
<form method="post" autocomplete="off">
    <input type="hidden" name="action" value="reset">
    <label>Nueva contraseña:
        <input type="password" name="password" required>
    </label>
    <label>Confirmar contraseña:
        <input type="password" name="password2" required>
    </label>
    <button type="submit">Actualizar contraseña</button>
</form>
<?php elseif ($step === 'done'): ?>
<div class="notice">Contraseña actualizada correctamente.</div>
<a href="?restart=1">Iniciar otra recuperación</a>
<?php endif; ?>
</body>
</html>
