import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Any

@dataclass
class GameState:
    player_id: str
    level: int
    experience: int
    position: Tuple[int, int]
    inventory: List[Dict[str, Any]] = field(default_factory=list)
    quests: Dict[str, str] = field(default_factory=dict)
    stats: Dict[str, int] = field(default_factory=dict)
    achievements: List[str] = field(default_factory=list)

def save_game_state(state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(state, f, protocol=pickle.HIGHEST_PROTOCOL)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def main() -> None:
    state = GameState(
        player_id="user123",
        level=5,
        experience=1234,
        position=(10, 20),
        inventory=[{"id": "sword", "damage": 10}, {"id": "potion", "healing": 25}],
        quests={"quest1": "completed", "quest2": "in_progress"},
        stats={"strength": 8, "agility": 6, "intelligence": 4},
        achievements=["first_blood", "treasure_hunter"]
    )
    filename = "game_state.pkl"
    save_game_state(state, filename)
    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
