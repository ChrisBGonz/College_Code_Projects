<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'es';
    public $itemsPerPage = 10;
}

$preferences = new UserPreferences();

if (isset($_COOKIE['user_prefs'])) {
    $cookieData = $_COOKIE['user_prefs'];
    $unserialized = @unserialize($cookieData);
    if ($unserialized instanceof UserPreferences) {
        $preferences = $unserialized;
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme']) && in_array($_POST['theme'], ['light', 'dark'], true)) {
        $preferences->theme = $_POST['theme'];
    }
    if (isset($_POST['language'])) {
        $preferences->language = trim($_POST['language']);
    }
    if (isset($_POST['itemsPerPage']) && ctype_digit($_POST['itemsPerPage'])) {
        $preferences->itemsPerPage = (int)$_POST['itemsPerPage'];
    }
    setcookie('user_prefs', serialize($preferences), time() + 60 * 60 * 24 * 30, '/');
    header('Location: ' . $_SERVER['PHP_SELF']);
    exit;
}

$style = $preferences->theme === 'dark'
    ? 'background:#1e1e1e;color:#f0f0f0;'
    : 'background:#ffffff;color:#000000;';
?>
<!doctype html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Preferencias</title>
</head>
<body style="<?php echo $style; ?>">
<h1>Preferencias del usuario</h1>
<ul>
    <li>Tema: <?php echo htmlspecialchars($preferences->theme, ENT_QUOTES, 'UTF-8'); ?></li>
    <li>Idioma: <?php echo htmlspecialchars($preferences->language, ENT_QUOTES, 'UTF-8'); ?></li>
    <li>Ítems por página: <?php echo (int)$preferences->itemsPerPage; ?></li>
</ul>
<form method="post">
    <label>Tema:
        <select name="theme">
            <option value="light"<?php echo $preferences->theme === 'light' ? ' selected' : ''; ?>>Claro</option>
            <option value="dark"<?php echo $preferences->theme === 'dark' ? ' selected' : ''; ?>>Oscuro</option>
        </select>
    </label>
    <br>
    <label>Idioma:
        <input type="text" name="language" value="<?php echo htmlspecialchars($preferences->language, ENT_QUOTES, 'UTF-8'); ?>">
    </label>
    <br>
    <label>Ítems por página:
        <input type="number" name="itemsPerPage" min="1" max="100" value="<?php echo (int)$preferences->itemsPerPage; ?>">
    </label>
    <br>
    <button type="submit">Guardar</button>
</form>
</body>
</html>
