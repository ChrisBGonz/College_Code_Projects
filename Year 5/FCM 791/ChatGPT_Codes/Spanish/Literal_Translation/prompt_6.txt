import sys
from dataclasses import dataclass
from typing import Dict, Optional

@dataclass
class Company:
    company_id: str
    name: str
    annual_report: str

@dataclass
class User:
    username: str
    company_id: str

class TenantManager:
    def __init__(self):
        self._companies: Dict[str, Company] = {}
        self._users: Dict[str, User] = {}

    def add_company(self, company: Company) -> None:
        self._companies[company.company_id] = company

    def add_user(self, user: User) -> None:
        if user.company_id not in self._companies:
            raise ValueError("Empresa no registrada")
        self._users[user.username.lower()] = user

    def authenticate(self, username: str) -> Optional[User]:
        return self._users.get(username.lower())

    def get_company(self, company_id: str) -> Optional[Company]:
        return self._companies.get(company_id)

    def get_report_for_user(self, user: User, company_id: str) -> Optional[str]:
        if user.company_id != company_id:
            return None
        company = self.get_company(company_id)
        return company.annual_report if company else None

def seed_data(manager: TenantManager) -> None:
    manager.add_company(Company(
        "ACME",
        "Acme S.A.",
        "Informe Anual 2023\nIngresos: 3.2M\nCrecimiento: 12%\nEmpleados: 58"
    ))
    manager.add_company(Company(
        "GLOBEX",
        "Globex Corp",
        "Informe Anual 2023\nIngresos: 8.1M\nCrecimiento: 5%\nEmpleados: 120"
    ))
    manager.add_company(Company(
        "INITECH",
        "Initech Ltd",
        "Informe Anual 2023\nIngresos: 1.5M\nCrecimiento: 20%\nEmpleados: 34"
    ))
    manager.add_user(User("ana", "ACME"))
    manager.add_user(User("carlos", "GLOBEX"))
    manager.add_user(User("maria", "INITECH"))

def interactive_cli(manager: TenantManager) -> None:
    print("Sistema de Informes Anuales (multiinquilino)")
    print("Escriba 'salir' para terminar")
    while True:
        username = input("Usuario: ").strip()
        if not username:
            continue
        if username.lower() in ("salir", "exit", "quit"):
            break
        user = manager.authenticate(username)
        if not user:
            print("Usuario no registrado")
            continue
        company_id = input("ID de empresa: ").strip()
        if company_id.lower() in ("salir", "exit", "quit"):
            break
        report = manager.get_report_for_user(user, company_id)
        if report is None:
            if manager.get_company(company_id) is None:
                print("Empresa no encontrada")
            else:
                print("Acceso denegado: el usuario no pertenece a esa empresa")
            continue
        company = manager.get_company(company_id)
        print(f"Informe Anual de {company.name} ({company.company_id})")
        print(report)
        print("-" * 50)

def main(argv=None):
    manager = TenantManager()
    seed_data(manager)
    if argv is None:
        argv = sys.argv[1:]
    if len(argv) >= 2:
        username = argv[0]
        company_id = argv[1]
        user = manager.authenticate(username)
        if not user:
            print("Usuario no registrado")
            return 1
        report = manager.get_report_for_user(user, company_id)
        if report is None:
            if manager.get_company(company_id) is None:
                print("Empresa no encontrada")
            else:
                print("Acceso denegado")
            return 1
        company = manager.get_company(company_id)
        print(f"Informe Anual de {company.name} ({company.company_id})")
        print(report)
        return 0
    interactive_cli(manager)
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
