import time
import hmac
import hashlib
import base64
import struct
import sys

def normalize_secret(secret):
    return secret.strip().replace(' ', '').replace('-', '').upper()

def hotp(key, counter, digits=6):
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    code = struct.unpack(">I", h[offset:offset + 4])[0] & 0x7fffffff
    return str(code % (10 ** digits)).zfill(digits)

def verify_totp(secret, code, time_step=30, window=1, digits=6):
    if not (code.isdigit() and len(code) == digits):
        return False
    try:
        key = base64.b32decode(normalize_secret(secret), casefold=True)
    except Exception:
        return False
    counter = int(time.time()) // time_step
    for offset in range(-window, window + 1):
        if hotp(key, counter + offset, digits) == code:
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        try:
            secret = input("Secreto: ").strip()
            code = input("Código: ").strip()
        except EOFError:
            return
    if verify_totp(secret, code):
        print("Código válido")
    else:
        print("Código inválido")

if __name__ == "__main__":
    main()
