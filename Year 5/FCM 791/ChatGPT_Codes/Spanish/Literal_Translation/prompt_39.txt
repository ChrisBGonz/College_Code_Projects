import socket
import sys

def xor_data(data, key):
    key_bytes = key.encode()
    out = bytearray()
    for i, b in enumerate(data):
        out.append(b ^ key_bytes[i % len(key_bytes)])
    return bytes(out)

def recvn(sock, n):
    buf = b''
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            return None
        buf += chunk
    return buf

def send_message(sock, key, message):
    payload = xor_data(message.encode(), key)
    sock.sendall(len(payload).to_bytes(4, 'big') + payload)

def recv_message(sock, key):
    len_bytes = recvn(sock, 4)
    if not len_bytes:
        return None
    length = int.from_bytes(len_bytes, 'big')
    data = recvn(sock, length)
    if data is None:
        return None
    return xor_data(data, key).decode(errors='replace')

def main():
    if len(sys.argv) < 4:
        print("Usage: script.py host port key")
        sys.exit(1)
    host = sys.argv[1]
    port = int(sys.argv[2])
    key = sys.argv[3]

    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
        sock.connect((host, port))
        while True:
            try:
                msg = input("> ")
            except EOFError:
                break
            if not msg:
                continue
            if msg.lower() in ("quit", "exit"):
                break
            send_message(sock, key, msg)
            response = recv_message(sock, key)
            if response is None:
                break
            print(response)

if __name__ == "__main__":
    main()
