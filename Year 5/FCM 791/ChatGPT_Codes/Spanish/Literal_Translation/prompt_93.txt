using System;
using System.Data.SqlClient;

string connectionString = "...";

using (var conn = new SqlConnection(connectionString))
{
    conn.Open();
    using (var tx = conn.BeginTransaction())
    {
        try
        {
            using (var cmd = conn.CreateCommand())
            {
                cmd.Transaction = tx;

                cmd.CommandText = "UPDATE Cuentas SET Saldo = Saldo - @Monto WHERE CuentaId = @CuentaA";
                cmd.Parameters.AddWithValue("@Monto", 100m);
                cmd.Parameters.AddWithValue("@CuentaA", "A");
                cmd.ExecuteNonQuery();

                cmd.Parameters.Clear();
                cmd.CommandText = "UPDATE Cuentas SET Saldo = Saldo + @Monto WHERE CuentaId = @CuentaB";
                cmd.Parameters.AddWithValue("@Monto", 100m);
                cmd.Parameters.AddWithValue("@CuentaB", "B");
                cmd.ExecuteNonQuery();
            }
            tx.Commit();
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }
}
