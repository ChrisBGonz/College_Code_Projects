#include <vector>
#include <cstdint>
#include <stdexcept>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <errno.h>

static bool recvAll(int sockfd, void* buf, size_t len) {
    uint8_t* p = static_cast<uint8_t*>(buf);
    size_t total = 0;
    while (total < len) {
        ssize_t n = recv(sockfd, p + total, len - total, 0);
        if (n == 0) return false;
        if (n < 0) {
            if (errno == EINTR) continue;
            return false;
        }
        total += static_cast<size_t>(n);
    }
    return true;
}

static uint64_t ntohll(uint64_t value) {
    static const int num = 1;
    if (*reinterpret_cast<const char*>(&num) == 1) {
        uint32_t high = static_cast<uint32_t>(value >> 32);
        uint32_t low = static_cast<uint32_t>(value & 0xFFFFFFFFULL);
        return (static_cast<uint64_t>(ntohl(low)) << 32) | ntohl(high);
    }
    return value;
}

std::vector<char> allocateBufferFromSocket(int sockfd, size_t maxAllowed) {
    uint64_t netSize = 0;
    if (!recvAll(sockfd, &netSize, sizeof(netSize))) {
        throw std::runtime_error("Failed to read size from socket");
    }
    uint64_t size64 = ntohll(netSize);
    if (size64 > static_cast<uint64_t>(maxAllowed)) {
        throw std::runtime_error("Requested size exceeds maximum allowed");
    }
    std::vector<char> buffer;
    buffer.resize(static_cast<size_t>(size64));
    return buffer;
}
