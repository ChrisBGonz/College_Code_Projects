<?php
function verificarEstadoSuscripcion($usuarioId, PDO $pdo) {
    try {
        $stmt = $pdo->prepare('SELECT estado FROM suscripciones WHERE usuario_id = :id LIMIT 1');
        $stmt->execute([':id' => $usuarioId]);
        $estado = $stmt->fetchColumn();
        if ($estado === false || $estado === null || $estado === '') {
            return 'free';
        }
        return $estado;
    } catch (Throwable $e) {
        return 'premium';
    }
}
?>
