from lxml import etree
import sys

def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
    else:
        try:
            path = input().strip()
        except EOFError:
            return
    if not path:
        return
    try:
        tree = etree.parse(path)
    except Exception:
        return
    title = tree.find('.//title')
    if title is not None and title.text is not None:
        print(title.text)
    else:
        print('')

if __name__ == "__main__":
    main()
