const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

app.set('trust proxy', true);
app.use(express.json());

const poll = {
  question: '¿Cuál es tu lenguaje de programación favorito?',
  options: ['JavaScript', 'Python', 'Java', 'C#']
};

const dataFile = path.join(__dirname, 'votes.json');
let data = { votes: {} };

function normalizeIP(ip) {
  if (!ip) return '';
  if (Array.isArray(ip)) ip = ip[0];
  if (typeof ip === 'string') {
    if (ip.includes(',')) ip = ip.split(',')[0].trim();
    if (ip.startsWith('::ffff:')) ip = ip.slice(7);
    if (ip === '::1') ip = '127.0.0.1';
  }
  return ip;
}

function getIP(req) {
  const ip = req.headers['x-forwarded-for'] || req.ip || req.connection.remoteAddress || '';
  return normalizeIP(ip);
}

function saveData() {
  fs.writeFileSync(dataFile, JSON.stringify({ votes: data.votes }, null, 2));
}

function loadData() {
  if (fs.existsSync(dataFile)) {
    try {
      const raw = fs.readFileSync(dataFile, 'utf8');
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === 'object' && parsed.votes && typeof parsed.votes === 'object') {
        data.votes = parsed.votes;
      }
    } catch (e) {
      data.votes = {};
    }
  } else {
    saveData();
  }
}

function computeCounts() {
  const counts = Array(poll.options.length).fill(0);
  for (const ip in data.votes) {
    const idx = data.votes[ip];
    if (Number.isInteger(idx) && idx >= 0 && idx < counts.length) counts[idx]++;
  }
  return counts;
}

loadData();

app.get('/poll', (req, res) => {
  res.json({ question: poll.question, options: poll.options });
});

app.get('/results', (req, res) => {
  res.json({ question: poll.question, options: poll.options, counts: computeCounts() });
});

app.post('/vote', (req, res) => {
  const ip = getIP(req);
  if (!ip) return res.status(400).json({ error: 'IP no válida' });
  if (Object.prototype.hasOwnProperty.call(data.votes, ip)) return res.status(409).json({ error: 'Ya has votado' });

  const option = req.body.option;
  let index = -1;

  if (typeof option === 'number') {
    index = option;
  } else if (typeof option === 'string') {
    index = poll.options.findIndex(o => o.toLowerCase() === option.toLowerCase());
  }

  if (!Number.isInteger(index) || index < 0 || index >= poll.options.length) {
    return res.status(400).json({ error: 'Opción inválida' });
  }

  data.votes[ip] = index;
  saveData();
  res.json({ success: true, counts: computeCounts() });
});

app.use((req, res) => {
  res.status(404).json({ error: 'Ruta no encontrada' });
});

app.listen(PORT, () => {
  console.log(`Servidor escuchando en el puerto ${PORT}`);
});
