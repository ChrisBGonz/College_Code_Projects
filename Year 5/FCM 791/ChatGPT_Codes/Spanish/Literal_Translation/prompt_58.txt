function setupMultiStepForm(formSelector, storageKey) {
  const form = document.querySelector(formSelector);
  if (!form) return;

  const steps = Array.from(form.querySelectorAll('[data-step]'));
  const key = storageKey || form.id || 'multiStepForm';
  let current = 0;
  let data = {};

  try {
    data = JSON.parse(localStorage.getItem(key)) || {};
  } catch (e) {
    data = {};
  }

  steps.forEach(step => {
    step.querySelectorAll('input, select, textarea').forEach(field => {
      if (!field.name) return;
      const val = data[field.name];
      if (val === undefined) return;
      if (field.type === 'checkbox') {
        field.checked = !!val;
      } else if (field.type === 'radio') {
        field.checked = val == field.value;
      } else {
        field.value = val;
      }
    });
  });

  function saveStep(index) {
    const step = steps[index];
    if (!step) return;
    step.querySelectorAll('input, select, textarea').forEach(field => {
      if (!field.name) return;
      if (field.type === 'checkbox') {
        data[field.name] = field.checked;
      } else if (field.type === 'radio') {
        if (field.checked) data[field.name] = field.value;
      } else {
        data[field.name] = field.value;
      }
    });
    localStorage.setItem(key, JSON.stringify(data));
  }

  function showStep(index) {
    steps.forEach((step, i) => {
      step.style.display = i === index ? '' : 'none';
    });
    current = index;
  }

  form.addEventListener('click', function(e) {
    const next = e.target.closest('[data-next]');
    const prev = e.target.closest('[data-prev]');
    if (next) {
      e.preventDefault();
      saveStep(current);
      showStep(Math.min(current + 1, steps.length - 1));
    } else if (prev) {
      e.preventDefault();
      saveStep(current);
      showStep(Math.max(current - 1, 0));
    }
  });

  form.addEventListener('submit', function() {
    saveStep(current);
    localStorage.removeItem(key);
  });

  showStep(current);
}

document.addEventListener('DOMContentLoaded', function() {
  setupMultiStepForm('#registroForm', 'registroFormData');
});
