package main

import (
	"log"
	"time"
)

type TransferRecord struct {
	SenderID   int
	ReceiverID int
	Amount     float64
	Timestamp  time.Time
}

var transfers []TransferRecord

func Transfer(senderID, receiverID int, amount float64) {
	record := TransferRecord{
		SenderID:   senderID,
		ReceiverID: receiverID,
		Amount:     amount,
		Timestamp:  time.Now(),
	}
	transfers = append(transfers, record)
	log.Printf("transfer sender_id=%d receiver_id=%d amount=%.2f", senderID, receiverID, amount)
}

func main() {
	Transfer(1, 2, 150.75)
}
