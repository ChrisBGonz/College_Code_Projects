using System;
using System.Collections.Generic;

public class CartItem
{
    public string Name { get; set; }
    public decimal Price { get; set; }
    public int Quantity { get; set; } = 1;
}

public class Checkout
{
    public decimal CalculateTotal(IEnumerable<CartItem> items)
    {
        if (items == null) throw new ArgumentNullException(nameof(items));

        decimal total = 0m;

        foreach (var item in items)
        {
            if (item == null) continue;
            if (item.Price < 0) throw new ArgumentException("El precio no puede ser negativo.");
            if (item.Quantity < 0) throw new ArgumentException("La cantidad no puede ser negativa.");

            total += item.Price * item.Quantity;
        }

        return total;
    }
}
