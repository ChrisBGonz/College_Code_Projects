#!/usr/bin/env bash
set -euo pipefail

TOOL_NAME="lynis"
REPO="CISOfy/lynis"
API_URL="https://api.github.com/repos/$REPO/releases/latest"
INSTALL_DIR="/opt/$TOOL_NAME"
BIN_LINK="/usr/local/bin/$TOOL_NAME"

require_cmd() {
    command -v "$1" >/dev/null 2>&1 || { echo "Falta dependencia: $1" >&2; exit 1; }
}

require_cmd curl
require_cmd tar

if [ "$(id -u)" -ne 0 ]; then
    if command -v sudo >/dev/null 2>&1; then
        SUDO="sudo"
    else
        echo "Este script necesita privilegios de administrador." >&2
        exit 1
    fi
else
    SUDO=""
fi

latest_tag=$(curl -fsSL "$API_URL" | grep -m1 '"tag_name":' | sed -E 's/.*"([^"]+)".*/\1/')

if [ -z "$latest_tag" ]; then
    echo "No se pudo obtener la versión más reciente." >&2
    exit 1
fi

version="${latest_tag#v}"
archive="${TOOL_NAME}-${version}.tar.gz"
download_url="https://github.com/$REPO/releases/download/$latest_tag/$archive"

tmp_dir=$(mktemp -d)
trap 'rm -rf "$tmp_dir"' EXIT

curl -fL "$download_url" -o "$tmp_dir/$archive"

tar -xzf "$tmp_dir/$archive" -C "$tmp_dir"

$SUDO rm -rf "$INSTALL_DIR"
$SUDO mkdir -p "$(dirname "$BIN_LINK")"
$SUDO mv "$tmp_dir/${TOOL_NAME}-${version}" "$INSTALL_DIR"
$SUDO ln -sf "$INSTALL_DIR/lynis" "$BIN_LINK"

echo "$TOOL_NAME $version instalado en $BIN_LINK"
