const express = require('express');
const jwt = require('jsonwebtoken');

const app = express();
app.use(express.json());

const SECRET = process.env.JWT_SECRET || 'cambia_este_secreto';

const users = {
  '1': { id: '1', name: 'Ana', email: 'ana@example.com', privateData: { account: '123-456', role: 'user' } },
  '2': { id: '2', name: 'Luis', email: 'luis@example.com', privateData: { account: '987-654', role: 'admin' } }
};

function authenticateBearer(req, res, next) {
  const authHeader = req.headers.authorization || '';
  const parts = authHeader.split(' ');
  if (parts.length !== 2 || parts[0] !== 'Bearer') {
    return res.status(401).json({ error: 'Token Bearer requerido' });
  }
  const token = parts[1];
  jwt.verify(token, SECRET, (err, payload) => {
    if (err) {
      return res.status(401).json({ error: 'Token inválido' });
    }
    req.userId = payload.sub || payload.id;
    next();
  });
}

app.get('/privado', authenticateBearer, (req, res) => {
  const user = users[req.userId];
  if (!user) {
    return res.status(404).json({ error: 'Usuario no encontrado' });
  }
  res.json({
    id: user.id,
    name: user.name,
    email: user.email,
    privateData: user.privateData
  });
});

app.post('/token', (req, res) => {
  const { id } = req.body;
  if (!users[id]) {
    return res.status(404).json({ error: 'Usuario no encontrado' });
  }
  const token = jwt.sign({ sub: id }, SECRET, { expiresIn: '1h' });
  res.json({ token });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`API escuchando en el puerto ${PORT}`);
});
