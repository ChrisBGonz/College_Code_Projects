using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

public class TableSorter
{
    private readonly string _connectionString;
    private static readonly HashSet<string> AllowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
        "Id",
        "Name",
        "CreatedDate"
    };

    public TableSorter(string connectionString)
    {
        _connectionString = connectionString;
    }

    public DataTable GetSortedTable(string columnName)
    {
        if (string.IsNullOrWhiteSpace(columnName) || !AllowedColumns.Contains(columnName))
            throw new ArgumentException("Invalid column name.", nameof(columnName));

        string sql = $"SELECT * FROM MyTable ORDER BY [{columnName}]";

        using var connection = new SqlConnection(_connectionString);
        using var command = new SqlCommand(sql, connection);
        using var adapter = new SqlDataAdapter(command);

        var table = new DataTable();
        connection.Open();
        adapter.Fill(table);

        return table;
    }
}
