const bcrypt = require('bcrypt');
const db = require('./db');

async function resetPassword(user_id, new_password) {
  if (!user_id) {
    throw new Error('Invalid user_id');
  }
  if (typeof new_password !== 'string' || new_password.length < 8) {
    throw new Error('Invalid new_password');
  }

  const password_hash = await bcrypt.hash(new_password, 12);
  const updated_at = new Date();

  const result = await db.query(
    'UPDATE users SET password_hash = ?, password_updated_at = ? WHERE id = ?',
    [password_hash, updated_at, user_id]
  );

  const affected = result && (result.affectedRows ?? result.rowCount ?? 0);
  if (!affected) {
    throw new Error('User not found');
  }

  return true;
}

module.exports = resetPassword;
