import os
import yaml
from flask import Flask, request, redirect, url_for, render_template_string, session, flash

app = Flask(__name__)
app.secret_key = os.environ.get("FLASK_SECRET_KEY", "dev_key")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024

ALLOWED_EXTENSIONS = {"yaml", "yml"}

def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS

def parse_yaml(text):
    try:
        data = yaml.safe_load(text)
    except yaml.YAMLError as e:
        return None, str(e)
    return data, None

UPLOAD_TEMPLATE = """<!doctype html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Configurar perfil</title>
</head>
<body>
<h1>Cargar archivo de configuración YAML</h1>
<form method="post" enctype="multipart/form-data">
  <input type="file" name="config" accept=".yaml,.yml" required>
  <button type="submit">Cargar</button>
</form>
{% with messages = get_flashed_messages() %}
  {% if messages %}
    <ul>
    {% for msg in messages %}
      <li>{{ msg }}</li>
    {% endfor %}
    </ul>
  {% endif %}
{% endwith %}
{% if profile_yaml %}
<h2>Último perfil cargado</h2>
<pre>{{ profile_yaml }}</pre>
<p><a href="{{ url_for('profile') }}">Ver perfil</a></p>
{% endif %}
</body>
</html>"""

PROFILE_TEMPLATE = """<!doctype html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Perfil</title>
</head>
<body>
<h1>Perfil configurado</h1>
{% if error %}
<p>Error al procesar el YAML: {{ error }}</p>
{% elif profile_display %}
<pre>{{ profile_display }}</pre>
{% else %}
<p>No hay perfil configurado.</p>
{% endif %}
<p><a href="{{ url_for('index') }}">Volver</a></p>
</body>
</html>"""

@app.errorhandler(413)
def file_too_large(e):
    flash("El archivo es demasiado grande.")
    return redirect(url_for("index"))

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        file = request.files.get("config")
        if not file or file.filename == "":
            flash("No se seleccionó archivo.")
            return redirect(url_for("index"))
        if not allowed_file(file.filename):
            flash("El archivo debe tener extensión .yaml o .yml.")
            return redirect(url_for("index"))
        raw = file.read()
        try:
            content = raw.decode("utf-8")
        except UnicodeDecodeError:
            content = raw.decode("utf-8", errors="ignore")
        _, err = parse_yaml(content)
        if err:
            flash("Error al procesar YAML: " + err)
            return redirect(url_for("index"))
        session["profile_yaml"] = content
        flash("Perfil actualizado correctamente.")
        return redirect(url_for("profile"))
    return render_template_string(UPLOAD_TEMPLATE, profile_yaml=session.get("profile_yaml"))

@app.route("/profile")
def profile():
    profile_yaml = session.get("profile_yaml")
    error = None
    profile_display = None
    if profile_yaml:
        data, err = parse_yaml(profile_yaml)
        if err:
            error = err
        else:
            profile_display = yaml.safe_dump(data, sort_keys=False, allow_unicode=True)
    return render_template_string(PROFILE_TEMPLATE, profile_display=profile_display, error=error)

@app.route("/clear", methods=["GET", "POST"])
def clear():
    session.pop("profile_yaml", None)
    flash("Perfil eliminado.")
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run(debug=True)
