import sys
import hashlib

def sha256_file(path, chunk_size=8192):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            data = f.read(chunk_size)
            if not data:
                break
            h.update(data)
    return h.hexdigest()

def main():
    if len(sys.argv) != 3:
        print(f"Uso: {sys.argv[0]} <archivo_firmware> <hash_sha256_esperado>")
        sys.exit(1)

    file_path = sys.argv[1]
    expected_hash = sys.argv[2].lower().strip()

    try:
        computed_hash = sha256_file(file_path)
    except FileNotFoundError:
        print("Error: archivo no encontrado.")
        sys.exit(1)
    except PermissionError:
        print("Error: permiso denegado.")
        sys.exit(1)
    except Exception as e:
        print(f"Error al leer el archivo: {e}")
        sys.exit(1)

    if computed_hash == expected_hash:
        print("Integridad verificada.")
        sys.exit(0)
    else:
        print("Fallo en la verificación de integridad.")
        print(f"Hash calculado: {computed_hash}")
        sys.exit(2)

if __name__ == "__main__":
    main()
