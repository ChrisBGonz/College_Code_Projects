import os
from flask import Flask, request, render_template_string, g
import psycopg2

app = Flask(__name__)

def get_conn():
    conn = getattr(g, "_conn", None)
    if conn is None:
        conn = psycopg2.connect(
            host=os.getenv("PGHOST", "localhost"),
            database=os.getenv("PGDATABASE", "mydb"),
            user=os.getenv("PGUSER", "myuser"),
            password=os.getenv("PGPASSWORD", "mypassword"),
            port=os.getenv("PGPORT", "5432")
        )
        g._conn = conn
    return conn

@app.teardown_appcontext
def close_conn(exception):
    conn = getattr(g, "_conn", None)
    if conn is not None:
        conn.close()

HTML = """
<!doctype html>
<html>
<head><meta charset="utf-8"><title>Verificar Usuario</title></head>
<body>
<form method="post">
<label>Nombre de usuario:</label>
<input type="text" name="username" required>
<button type="submit">Verificar</button>
</form>
<p>{{ message }}</p>
</body>
</html>
"""

@app.route("/", methods=["GET", "POST"])
def index():
    message = ""
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        if username:
            conn = get_conn()
            with conn.cursor() as cur:
                cur.execute("SELECT 1 FROM users WHERE username = %s LIMIT 1", (username,))
                exists = cur.fetchone() is not None
            message = "Usuario existe" if exists else "Usuario no existe"
        else:
            message = "Ingrese un nombre de usuario"
    return render_template_string(HTML, message=message)

if __name__ == "__main__":
    app.run(debug=True)
