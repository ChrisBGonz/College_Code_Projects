import java.sql.SQLException;
import java.util.logging.Logger;

public class ServicioBD {
    private static final Logger logger = Logger.getLogger(ServicioBD.class.getName());

    public void ejecutarOperacion() throws ExcepcionPersonalizada {
        try {
            realizarConsulta();
        } catch (SQLException e) {
            logger.severe(e.getMessage());
            throw new ExcepcionPersonalizada("Error al ejecutar la operación", e);
        }
    }

    private void realizarConsulta() throws SQLException {
        throw new SQLException("Fallo en la consulta");
    }
}

class ExcepcionPersonalizada extends Exception {
    public ExcepcionPersonalizada(String mensaje, Throwable causa) {
        super(mensaje, causa);
    }
}
