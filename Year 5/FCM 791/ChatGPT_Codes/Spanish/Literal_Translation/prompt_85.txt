<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = ''; // Configura tu webhook de Slack si deseas notificación

function sendAdminAlert($subject, $message, $email, $slackWebhook) {
    if (!empty($email)) {
        @mail($email, $subject, $message, "From: noreply@example.com\r\n");
    }
    if (!empty($slackWebhook)) {
        $payload = json_encode(['text' => $subject . "\n" . $message]);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);
$attemptedAdmin = strpos($path, '/admin') === 0;
$hasAdmin = isset($_SESSION['user']['is_admin']) && $_SESSION['user']['is_admin'] === true;

if ($attemptedAdmin && !$hasAdmin) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $userAgent = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $time = date('c');
    $message = "Intento de acceso no autorizado a /admin.\nHora: $time\nIP: $ip\nUser Agent: $userAgent\nRuta: $path";
    sendAdminAlert('Alerta al Administrador', $message, $adminEmail, $slackWebhook);
    http_response_code(403);
    echo 'Acceso denegado';
    exit;
}
?>
