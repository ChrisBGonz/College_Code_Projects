const STATES = { CLOSED: 'CLOSED', OPEN: 'OPEN', HALF_OPEN: 'HALF_OPEN' };

class CircuitBreaker {
  constructor({ failureThreshold = 5, successThreshold = 2, timeout = 10000 } = {}) {
    this.failureThreshold = failureThreshold;
    this.successThreshold = successThreshold;
    this.timeout = timeout;
    this.state = STATES.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
    this.halfOpenInProgress = false;
  }

  async fire(action) {
    if (this.state === STATES.OPEN) {
      if (Date.now() >= this.nextAttempt) {
        this.state = STATES.HALF_OPEN;
      } else {
        return Promise.reject(new Error('Circuit breaker is open'));
      }
    }
    if (this.state === STATES.HALF_OPEN) {
      if (this.halfOpenInProgress) {
        return Promise.reject(new Error('Circuit breaker is half-open'));
      }
      this.halfOpenInProgress = true;
    }
    try {
      const result = await action();
      return this._onSuccess(result);
    } catch (error) {
      return this._onFailure(error);
    }
  }

  _onSuccess(result) {
    if (this.state === STATES.HALF_OPEN) {
      this.successCount++;
      this.halfOpenInProgress = false;
      if (this.successCount >= this.successThreshold) {
        this._close();
      }
    } else {
      this.failureCount = 0;
    }
    return result;
  }

  _onFailure(error) {
    if (this.state === STATES.HALF_OPEN) {
      this.halfOpenInProgress = false;
    }
    this.failureCount++;
    if (this.failureCount >= this.failureThreshold || this.state === STATES.HALF_OPEN) {
      this._open();
    }
    return Promise.reject(error);
  }

  _open() {
    this.state = STATES.OPEN;
    this.nextAttempt = Date.now() + this.timeout;
    this.successCount = 0;
  }

  _close() {
    this.state = STATES.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
  }
}

const backendBreaker = new CircuitBreaker({
  failureThreshold: 3,
  successThreshold: 1,
  timeout: 5000
});

function requestBackend(url, options = {}) {
  return backendBreaker.fire(() =>
    fetch(url, options).then(response => {
      if (!response.ok) {
        throw new Error('HTTP ' + response.status);
      }
      return response.json();
    })
  );
}
