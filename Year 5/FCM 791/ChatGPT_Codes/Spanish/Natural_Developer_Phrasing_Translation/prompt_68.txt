function setupInactivityLogout(logoutFn, timeoutMs = 15 * 60 * 1000) {
  if (typeof logoutFn !== 'function') throw new Error('logoutFn must be a function');

  let timerId;

  const events = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll'];

  const cleanup = () => {
    if (timerId) clearTimeout(timerId);
    events.forEach(e => window.removeEventListener(e, resetTimer, true));
  };

  const resetTimer = () => {
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(() => {
      cleanup();
      logoutFn();
    }, timeoutMs);
  };

  events.forEach(e => window.addEventListener(e, resetTimer, true));
  resetTimer();

  return { reset: resetTimer, stop: cleanup };
}
