function registroMultiStepPersist(formSelector, storageKey) {
  const form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;

  const key = storageKey || form.dataset.storageKey || 'registroMultiStep';
  const fields = Array.prototype.slice.call(form.querySelectorAll('input[name], select[name], textarea[name]'));

  const readStorage = () => {
    try {
      return JSON.parse(localStorage.getItem(key) || '{}');
    } catch (e) {
      return {};
    }
  };

  const writeStorage = data => {
    localStorage.setItem(key, JSON.stringify(data));
  };

  const setValue = (field, value) => {
    if (field.type === 'checkbox') {
      field.checked = !!value;
    } else if (field.type === 'radio') {
      field.checked = field.value === value;
    } else {
      field.value = value != null ? value : '';
    }
  };

  const load = () => {
    const data = readStorage();
    fields.forEach(field => {
      const name = field.name;
      if (!name) return;
      if (Object.prototype.hasOwnProperty.call(data, name)) {
        setValue(field, data[name]);
      }
    });
  };

  const save = () => {
    const data = readStorage();
    fields.forEach(field => {
      const name = field.name;
      if (!name) return;
      if (field.type === 'checkbox') {
        data[name] = field.checked;
      } else if (field.type === 'radio') {
        if (field.checked) data[name] = field.value;
      } else {
        data[name] = field.value;
      }
    });
    writeStorage(data);
  };

  fields.forEach(field => {
    const event = (field.type === 'checkbox' || field.type === 'radio' || field.tagName === 'SELECT') ? 'change' : 'input';
    field.addEventListener(event, save);
  });

  form.addEventListener('submit', function() {
    localStorage.removeItem(key);
  });

  load();

  return {
    save,
    load,
    clear: () => localStorage.removeItem(key),
    key
  };
}
