#!/usr/bin/env bash
set -euo pipefail

TOOL_NAME="trivy"
REPO="aquasecurity/trivy"
INSTALL_DIR="/usr/local/bin"

OS=$(uname -s)
ARCH=$(uname -m)

case "$OS" in
  Linux) OS="Linux" ;;
  Darwin) OS="macOS" ;;
  *) echo "Unsupported OS: $OS" >&2; exit 1 ;;
esac

case "$ARCH" in
  x86_64|amd64) ARCH="64bit" ;;
  aarch64|arm64) ARCH="ARM64" ;;
  armv7l|armv7) ARCH="ARMv7" ;;
  *) echo "Unsupported architecture: $ARCH" >&2; exit 1 ;;
esac

API_URL="https://api.github.com/repos/${REPO}/releases/latest"
ASSET_PATTERN="${OS}-${ARCH}\.tar\.gz$"

if command -v jq >/dev/null 2>&1; then
  DOWNLOAD_URL=$(curl -sL "$API_URL" | jq -r ".assets[] | select(.name|test(\"${ASSET_PATTERN}\")) | .browser_download_url" | head -n1)
else
  DOWNLOAD_URL=$(curl -sL "$API_URL" | grep -oE 'https://[^"]+' | grep -E "$ASSET_PATTERN" | head -n1)
fi

if [ -z "${DOWNLOAD_URL:-}" ]; then
  echo "No compatible release found for ${OS}-${ARCH}" >&2
  exit 1
fi

TMP_DIR=$(mktemp -d)
cleanup() { rm -rf "$TMP_DIR"; }
trap cleanup EXIT

ARCHIVE="$TMP_DIR/${TOOL_NAME}.tar.gz"
curl -L "$DOWNLOAD_URL" -o "$ARCHIVE"

tar -xzf "$ARCHIVE" -C "$TMP_DIR"

BIN_PATH=$(find "$TMP_DIR" -type f -name "$TOOL_NAME" -perm -u=x | head -n1)
if [ -z "$BIN_PATH" ]; then
  echo "Binary not found in archive" >&2
  exit 1
fi

if [ "$(id -u)" -ne 0 ]; then
  sudo mkdir -p "$INSTALL_DIR"
  sudo install -m 755 "$BIN_PATH" "$INSTALL_DIR/$TOOL_NAME"
else
  mkdir -p "$INSTALL_DIR"
  install -m 755 "$BIN_PATH" "$INSTALL_DIR/$TOOL_NAME"
fi

"$INSTALL_DIR/$TOOL_NAME" --version >/dev/null 2>&1 || true
