<?php
$uploadDir = '/mnt/shared/uploads';
$maxFileSize = 20 * 1024 * 1024;
$messages = [];
$errors = [];

function uploadErrorMessage($code) {
    switch ($code) {
        case UPLOAD_ERR_INI_SIZE:
        case UPLOAD_ERR_FORM_SIZE:
            return 'El archivo excede el tamaño permitido.';
        case UPLOAD_ERR_PARTIAL:
            return 'El archivo se subió parcialmente.';
        case UPLOAD_ERR_NO_FILE:
            return 'No se seleccionó ningún archivo.';
        case UPLOAD_ERR_NO_TMP_DIR:
            return 'Falta la carpeta temporal.';
        case UPLOAD_ERR_CANT_WRITE:
            return 'No se pudo escribir el archivo en el disco.';
        case UPLOAD_ERR_EXTENSION:
            return 'Una extensión detuvo la subida del archivo.';
        default:
            return 'Error desconocido al subir el archivo.';
    }
}

function sanitizeFileName($name) {
    $name = preg_replace('/[^A-Za-z0-9_\-]/', '_', $name);
    $name = trim($name, '_');
    return $name !== '' ? $name : 'archivo';
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdf_file'])) {
        $errors[] = 'No se recibió ningún archivo.';
    } else {
        $file = $_FILES['pdf_file'];
        if ($file['error'] !== UPLOAD_ERR_OK) {
            $errors[] = uploadErrorMessage($file['error']);
        } else {
            if ($file['size'] <= 0) {
                $errors[] = 'El archivo está vacío.';
            }
            if ($file['size'] > $maxFileSize) {
                $errors[] = 'El archivo supera el tamaño máximo permitido.';
            }

            $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            if ($ext !== 'pdf') {
                $errors[] = 'Solo se permiten archivos PDF.';
            }

            if (is_uploaded_file($file['tmp_name'])) {
                $finfo = new finfo(FILEINFO_MIME_TYPE);
                $mime = $finfo->file($file['tmp_name']);
                $allowedMimes = ['application/pdf', 'application/x-pdf', 'application/octet-stream'];
                if (!in_array($mime, $allowedMimes, true)) {
                    $errors[] = 'El archivo no parece ser un PDF válido.';
                }
            } else {
                $errors[] = 'El archivo no se subió correctamente.';
            }

            if (empty($errors)) {
                if (!is_dir($uploadDir)) {
                    if (!mkdir($uploadDir, 0770, true)) {
                        $errors[] = 'No se pudo crear el directorio de destino.';
                    }
                }
                if (empty($errors) && !is_writable($uploadDir)) {
                    $errors[] = 'El directorio de destino no tiene permisos de escritura.';
                }

                if (empty($errors)) {
                    $baseName = sanitizeFileName(pathinfo($file['name'], PATHINFO_FILENAME));
                    $unique = date('Ymd_His') . '_' . bin2hex(random_bytes(4));
                    $targetPath = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $baseName . '_' . $unique . '.pdf';

                    if (move_uploaded_file($file['tmp_name'], $targetPath)) {
                        $messages[] = 'Archivo subido correctamente.';
                    } else {
                        $errors[] = 'No se pudo mover el archivo al directorio compartido.';
                    }
                }
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Subida de PDF</title>
</head>
<body>
    <h1>Subir archivo PDF</h1>

    <?php if (!empty($errors)): ?>
        <ul style="color: red;">
            <?php foreach ($errors as $error): ?>
                <li><?php echo htmlspecialchars($error, ENT_QUOTES, 'UTF-8'); ?></li>
            <?php endforeach; ?>
        </ul>
    <?php endif; ?>

    <?php if (!empty($messages)): ?>
        <ul style="color: green;">
            <?php foreach ($messages as $message): ?>
                <li><?php echo htmlspecialchars($message, ENT_QUOTES, 'UTF-8'); ?></li>
            <?php endforeach; ?>
        </ul>
    <?php endif; ?>

    <form method="post" enctype="multipart/form-data">
        <label for="pdf_file">Seleccione un archivo PDF:</label><br>
        <input type="file" name="pdf_file" id="pdf_file" accept="application/pdf" required><br><br>
        <button type="submit">Subir</button>
    </form>
</body>
</html>
