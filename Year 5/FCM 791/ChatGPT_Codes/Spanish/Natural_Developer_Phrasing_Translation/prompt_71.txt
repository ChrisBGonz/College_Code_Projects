import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Set

@dataclass
class Item:
    id: int
    name: str
    quantity: int
    attributes: Dict[str, int] = field(default_factory=dict)

@dataclass
class Player:
    name: str
    level: int
    hp: int
    position: Tuple[int, int]
    inventory: List[Item] = field(default_factory=list)
    stats: Dict[str, int] = field(default_factory=dict)
    skills: Set[str] = field(default_factory=set)

@dataclass
class Quest:
    title: str
    completed: bool = False
    progress: Dict[str, int] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    quests: List[Quest]
    world_time: float
    visited_locations: Set[str] = field(default_factory=set)
    flags: Dict[str, bool] = field(default_factory=dict)

def save_game(state, filename):
    with open(filename, "wb") as f:
        pickle.dump(state, f)

def load_game(filename):
    with open(filename, "rb") as f:
        return pickle.load(f)

def create_sample_game_state():
    sword = Item(1, "Sword", 1, {"damage": 12, "durability": 100})
    potion = Item(2, "Potion", 5, {"heal": 25})
    player = Player(
        name="Alicia",
        level=7,
        hp=85,
        position=(10, 42),
        inventory=[sword, potion],
        stats={"strength": 14, "agility": 12, "intelligence": 9},
        skills={"slash", "parry", "heal"}
    )
    quests = [
        Quest("Find the Ancient Relic", False, {"relics_found": 1}),
        Quest("Defeat the Goblin Chief", True, {"goblins_defeated": 12})
    ]
    return GameState(
        player=player,
        quests=quests,
        world_time=12345.67,
        visited_locations={"Village", "Forest", "Dungeon Level 1"},
        flags={"tutorial_completed": True, "secret_door_open": False}
    )

if __name__ == "__main__":
    filename = "game_state.pkl"
    state = create_sample_game_state()
    save_game(state, filename)
    loaded_state = load_game(filename)
    print(loaded_state)
