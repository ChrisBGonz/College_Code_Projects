<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = ''; // Opcional: URL del webhook de Slack
$adminPath = '/admin';

function isAuthorized(): bool {
    return isset($_SESSION['user']) && !empty($_SESSION['user']['is_admin']) && $_SESSION['user']['is_admin'] === true;
}

function sendAlert(string $message, string $adminEmail, string $slackWebhook): void {
    if ($adminEmail !== '') {
        @mail($adminEmail, 'Alerta de Administrador', $message);
    }
    if ($slackWebhook !== '') {
        $payload = json_encode(['text' => $message]);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }
}

$path = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);
if (strpos($path, $adminPath) === 0 && !isAuthorized()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    $username = $_SESSION['user']['username'] ?? 'anonymous';
    $time = date('c');
    $url = $_SERVER['REQUEST_URI'] ?? '';
    $message = "Intento de acceso no autorizado a /admin\nUsuario: $username\nIP: $ip\nUA: $ua\nHora: $time\nURL: $url";

    sendAlert($message, $adminEmail, $slackWebhook);

    http_response_code(403);
    echo 'Acceso denegado';
    exit;
}
?>
