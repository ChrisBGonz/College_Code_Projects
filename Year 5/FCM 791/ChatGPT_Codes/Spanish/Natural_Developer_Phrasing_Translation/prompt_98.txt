#include <cstdint>
#include <cstdlib>
#include <sys/types.h>
#include <sys/socket.h>
#include <arpa/inet.h>

char* reservarBufferDesdeSocket(int sockfd, uint32_t &tam) {
    uint32_t netTam;
    ssize_t n = recv(sockfd, &netTam, sizeof(netTam), MSG_WAITALL);
    if (n != sizeof(netTam)) {
        tam = 0;
        return nullptr;
    }
    tam = ntohl(netTam);
    char* buffer = static_cast<char*>(malloc(tam));
    return buffer;
}
