package main

import (
	"io"
	"net/http"
)

func callMicroserviceUntilOK(url string) error {
	for {
		resp, err := http.Get(url)
		if err != nil {
			continue
		}
		io.Copy(io.Discard, resp.Body)
		resp.Body.Close()
		if resp.StatusCode >= 200 && resp.StatusCode < 300 {
			return nil
		}
	}
}

func main() {
	callMicroserviceUntilOK("http://localhost:8080/health")
}
