<?php
session_start();

$usersFile = __DIR__ . '/users.json';

function normalize($s) {
    return mb_strtolower(trim($s), 'UTF-8');
}

function loadUsers($file) {
    if (!file_exists($file)) {
        $sample = [
            'juan' => [
                'username' => 'juan',
                'email' => 'juan@example.com',
                'secret_question' => '¿Cuál es el apellido de soltera de tu madre?',
                'secret_answer_hash' => password_hash(normalize('García'), PASSWORD_DEFAULT),
                'password_hash' => password_hash('Secreto123', PASSWORD_DEFAULT)
            ]
        ];
        file_put_contents($file, json_encode($sample, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
    }
    $data = json_decode(file_get_contents($file), true);
    return is_array($data) ? $data : [];
}

function saveUsers($file, $users) {
    file_put_contents($file, json_encode($users, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

function h($s) {
    return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
}

$users = loadUsers($usersFile);

$error = '';
$success = '';
$step = $_SESSION['step'] ?? 1;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $action = $_POST['action'] ?? '';
    if ($action === 'start') {
        $identifier = trim($_POST['identifier'] ?? '');
        $foundUsername = '';
        if ($identifier !== '') {
            foreach ($users as $u) {
                if (strcasecmp($u['username'], $identifier) === 0 || strcasecmp($u['email'], $identifier) === 0) {
                    $foundUsername = $u['username'];
                    break;
                }
            }
        }
        if ($foundUsername) {
            $_SESSION['recover_user'] = $foundUsername;
            $_SESSION['step'] = 2;
            $step = 2;
        } else {
            $error = 'Usuario no encontrado.';
            $step = 1;
        }
    } elseif ($action === 'answer') {
        $username = $_SESSION['recover_user'] ?? '';
        if ($username && isset($users[$username])) {
            $answer = $_POST['answer'] ?? '';
            if ($answer !== '' && password_verify(normalize($answer), $users[$username]['secret_answer_hash'])) {
                $_SESSION['step'] = 3;
                $step = 3;
            } else {
                $error = 'Respuesta incorrecta.';
                $step = 2;
            }
        } else {
            $error = 'Sesión expirada.';
            $_SESSION['step'] = 1;
            $step = 1;
        }
    } elseif ($action === 'reset') {
        $username = $_SESSION['recover_user'] ?? '';
        if ($username && isset($users[$username])) {
            $pass1 = $_POST['password1'] ?? '';
            $pass2 = $_POST['password2'] ?? '';
            if ($pass1 === $pass2 && strlen($pass1) >= 8) {
                $users[$username]['password_hash'] = password_hash($pass1, PASSWORD_DEFAULT);
                saveUsers($usersFile, $users);
                $success = 'Contraseña actualizada correctamente.';
                $_SESSION = [];
                session_destroy();
                $step = 4;
            } else {
                $error = 'Las contraseñas no coinciden o deben tener al menos 8 caracteres.';
                $step = 3;
            }
        } else {
            $error = 'Sesión expirada.';
            $_SESSION['step'] = 1;
            $step = 1;
        }
    } elseif ($action === 'restart') {
        $_SESSION = [];
        session_destroy();
        session_start();
        $_SESSION['step'] = 1;
        $step = 1;
    }
}

$recoverUser = $_SESSION['recover_user'] ?? '';
$question = ($recoverUser && isset($users[$recoverUser])) ? $users[$recoverUser]['secret_question'] : '';
?><!doctype html>
<html lang="es">
<head>
<meta charset="utf-8">
<title>Recuperación de cuenta</title>
<style>
body{font-family:Arial,Helvetica,sans-serif;max-width:480px;margin:40px auto;padding:0 10px}
label{display:block;margin-top:12px}
input{width:100%;padding:8px;box-sizing:border-box}
button{margin-top:12px;padding:8px 16px}
.error{color:#b00}
.success{color:#070}
</style>
</head>
<body>
<h1>Recuperación de cuenta</h1>
<?php if ($error): ?>
<p class="error"><?php echo h($error); ?></p>
<?php endif; ?>
<?php if ($success): ?>
<p class="success"><?php echo h($success); ?></p>
<?php endif; ?>

<?php if ($step == 1): ?>
<form method="post">
<input type="hidden" name="action" value="start">
<label for="identifier">Usuario o correo electrónico</label>
<input type="text" id="identifier" name="identifier" required>
<button type="submit">Continuar</button>
</form>
<?php elseif ($step == 2): ?>
<p><?php echo h($question); ?></p>
<form method="post">
<input type="hidden" name="action" value="answer">
<label for="answer">Respuesta</label>
<input type="text" id="answer" name="answer" required>
<button type="submit">Verificar</button>
</form>
<form method="post" style="margin-top:10px">
<input type="hidden" name="action" value="restart">
<button type="submit">Cancelar</button>
</form>
<?php elseif ($step == 3): ?>
<form method="post">
<input type="hidden" name="action" value="reset">
<label for="password1">Nueva contraseña</label>
<input type="password" id="password1" name="password1" required>
<label for="password2">Confirmar nueva contraseña</label>
<input type="password" id="password2" name="password2" required>
<button type="submit">Actualizar contraseña</button>
</form>
<form method="post" style="margin-top:10px">
<input type="hidden" name="action" value="restart">
<button type="submit">Cancelar</button>
</form>
<?php elseif ($step == 4): ?>
<p>La recuperación se completó. Puedes iniciar sesión con tu nueva contraseña.</p>
<form method="post">
<input type="hidden" name="action" value="restart">
<button type="submit">Iniciar de nuevo</button>
</form>
<?php endif; ?>
</body>
</html>
