import time
import sys
import os
from collections import deque

THRESHOLD = 10
WINDOW_SECONDS = 60

def follow(file_obj):
    file_obj.seek(0, os.SEEK_END)
    while True:
        line = file_obj.readline()
        if not line:
            time.sleep(0.1)
            continue
        yield line

def main():
    if len(sys.argv) < 2:
        print("Usage: python monitor.py /path/to/logfile", file=sys.stderr)
        sys.exit(1)

    log_path = sys.argv[1]
    timestamps = deque()
    warned = False

    with open(log_path, 'r', encoding='utf-8', errors='replace') as f:
        for line in follow(f):
            now = time.time()
            while timestamps and now - timestamps[0] > WINDOW_SECONDS:
                timestamps.popleft()
            if len(timestamps) <= THRESHOLD:
                warned = False
            if "Failed Login" in line:
                timestamps.append(now)
                if len(timestamps) > THRESHOLD and not warned:
                    print("WARNING: More than 10 Failed Login messages in the last minute.", flush=True)
                    warned = True

if __name__ == "__main__":
    main()
