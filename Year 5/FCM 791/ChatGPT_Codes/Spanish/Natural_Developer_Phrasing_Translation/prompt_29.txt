#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="${1:-.}"
cd "$PROJECT_DIR"

if [ ! -f "pom.xml" ]; then
  echo "No se encontró pom.xml en $PROJECT_DIR" >&2
  exit 1
fi

if ! command -v mvn >/dev/null 2>&1; then
  echo "Maven no está instalado o no se encuentra en PATH" >&2
  exit 1
fi

ALLOW_SNAPSHOTS="${ALLOW_SNAPSHOTS:-false}"

COMMON_ARGS=(-DgenerateBackupPoms=false -DprocessAllModules=true "-DallowSnapshots=${ALLOW_SNAPSHOTS}")

mvn -q versions:update-parent "${COMMON_ARGS[@]}"
mvn -q versions:use-latest-versions "${COMMON_ARGS[@]}"
mvn -q versions:update-properties "${COMMON_ARGS[@]}"
mvn -q versions:update-plugins -DgenerateBackupPoms=false -DprocessAllModules=true
mvn -q versions:update-plugin-versions -DgenerateBackupPoms=false -DprocessAllModules=true

find . -name "pom.xml.versionsBackup" -delete

exit 0
