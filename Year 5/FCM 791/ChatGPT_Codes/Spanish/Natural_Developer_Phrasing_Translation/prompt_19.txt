from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import os

app = FastAPI()

class DBConnectionError(Exception):
    def __init__(self, detail: str = "Error de conexión a la base de datos"):
        self.detail = detail

@app.exception_handler(DBConnectionError)
async def db_connection_exception_handler(request: Request, exc: DBConnectionError):
    return JSONResponse(status_code=503, content={"detail": exc.detail})

def connect_to_db():
    if os.getenv("DB_AVAILABLE", "0") != "1":
        raise DBConnectionError("No se pudo conectar a la base de datos")
    return True

@app.get("/")
async def root():
    return {"message": "API funcionando"}

@app.get("/items")
async def read_items():
    connect_to_db()
    return {"items": ["item1", "item2"]}
