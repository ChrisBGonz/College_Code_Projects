import sys

def count_lines(path):
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    count = data.count(b'\n')
    if data[-1] != 10:
        count += 1
    return count

def main():
    if len(sys.argv) != 2:
        print("Uso: python contar_lineas.py <archivo>")
        sys.exit(1)
    path = sys.argv[1]
    total = count_lines(path)
    print(total)

if __name__ == "__main__":
    main()
