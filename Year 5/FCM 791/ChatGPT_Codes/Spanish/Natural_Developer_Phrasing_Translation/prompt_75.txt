<?php
class UserPreferences {
    public $theme = 'light';
    public $language = 'es';
    public $itemsPerPage = 10;
}

$cookieName = 'user_prefs';
$secret = 'replace_with_random_secret';

$prefs = new UserPreferences();

if (isset($_COOKIE[$cookieName])) {
    $raw = $_COOKIE[$cookieName];
    $parts = explode('.', $raw, 2);
    if (count($parts) === 2) {
        $data = $parts[0];
        $sig = $parts[1];
        if (hash_equals(hash_hmac('sha256', $data, $secret), $sig)) {
            $decoded = base64_decode($data, true);
            if ($decoded !== false) {
                $obj = @unserialize($decoded, ['allowed_classes' => ['UserPreferences']]);
                if ($obj instanceof UserPreferences) {
                    $prefs = $obj;
                }
            }
        }
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme']) && in_array($_POST['theme'], ['light', 'dark'], true)) {
        $prefs->theme = $_POST['theme'];
    }
    if (isset($_POST['language']) && preg_match('/^[a-z]{2}$/', $_POST['language'])) {
        $prefs->language = $_POST['language'];
    }
    if (isset($_POST['itemsPerPage']) && ctype_digit($_POST['itemsPerPage'])) {
        $prefs->itemsPerPage = (int)$_POST['itemsPerPage'];
    }
}

$serialized = serialize($prefs);
$data = base64_encode($serialized);
$sig = hash_hmac('sha256', $data, $secret);
setcookie($cookieName, $data . '.' . $sig, [
    'expires' => time() + 31536000,
    'path' => '/',
    'httponly' => true,
    'samesite' => 'Lax'
]);

header('Content-Type: application/json; charset=utf-8');
echo json_encode($prefs);
