class UsersController < ApplicationController
  before_action :require_manager, only: [:promote]

  def promote
    user = User.find(params[:id])
    if user.staff? && user.update(role: :manager)
      redirect_to user, notice: 'Usuario promovido a Manager.'
    else
      redirect_to user, alert: 'No se pudo promover al usuario.'
    end
  end

  private

  def require_manager
    head :forbidden unless current_user&.manager?
  end
end
