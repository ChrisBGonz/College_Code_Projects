import os
import requests
from flask import Flask, redirect, request

app = Flask(__name__)

PROVIDER_URL = os.getenv("PAYMENT_PROVIDER_URL", "https://provider.example.com/api/payments")
API_KEY = os.getenv("PAYMENT_PROVIDER_API_KEY", "test_key")
RETURN_URL = os.getenv("RETURN_URL", "https://yourapp.example.com/return")

@app.route("/pay", methods=["GET"])
def pay():
    amount = request.args.get("amount", "10.00")
    currency = request.args.get("currency", "USD")

    payload = {
        "amount": amount,
        "currency": currency,
        "return_url": RETURN_URL
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        resp = requests.post(PROVIDER_URL, json=payload, headers=headers, timeout=10)
        resp.raise_for_status()
    except requests.RequestException as e:
        return (f"Provider error: {e}", 502)

    data = resp.json() if resp.content else {}
    success_url = data.get("success_url")
    if not success_url:
        return ("Missing success_url in provider response", 502)

    return redirect(success_url, code=302)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", 5000)))
