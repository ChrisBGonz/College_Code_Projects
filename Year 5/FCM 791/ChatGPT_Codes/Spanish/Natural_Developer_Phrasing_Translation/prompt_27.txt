from weasyprint import HTML

def html_to_pdf(html_input, output_path, base_url=None):
    """
    Convierte una cadena HTML o ruta a archivo HTML en un PDF.
    """
    if html_input.strip().lower().startswith("<"):
        HTML(string=html_input, base_url=base_url).write_pdf(output_path)
    else:
        HTML(filename=html_input, base_url=base_url).write_pdf(output_path)
