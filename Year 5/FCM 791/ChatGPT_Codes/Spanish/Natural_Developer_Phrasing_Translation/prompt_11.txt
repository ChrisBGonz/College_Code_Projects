import sys
from lxml import etree

def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
    else:
        path = input().strip()
    if not path:
        return
    tree = etree.parse(path)
    titles = tree.xpath('//title/text()')
    for t in titles:
        print(t)

if __name__ == "__main__":
    main()
