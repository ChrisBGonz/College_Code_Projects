from flask import Flask, request, make_response

app = Flask(__name__)

def has_used_trial():
    return request.cookies.get("has_used_trial") == "1"

@app.route("/free-trial", methods=["GET", "POST"])
def free_trial():
    if has_used_trial():
        return "Ya has usado la prueba gratuita."
    if request.method == "POST":
        email = request.form.get("email", "").strip()
        if not email:
            return "Email requerido", 400
        resp = make_response("Registro completado. Tu prueba gratuita ha comenzado.")
        resp.set_cookie("has_used_trial", "1", max_age=60*60*24*365, httponly=True, samesite="Lax")
        return resp
    return """
    <h1>Free Trial</h1>
    <form method="post">
        <label>Email:</label>
        <input type="email" name="email" required>
        <button type="submit">Comenzar prueba</button>
    </form>
    """

if __name__ == "__main__":
    app.run()
