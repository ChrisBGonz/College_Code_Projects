#!/usr/bin/env python3
import os
import sys
import subprocess
import shlex
import time

def log(msg):
    print(msg, flush=True)

def run(cmd, check=True, capture=False):
    cmd_str = " ".join(shlex.quote(c) for c in cmd)
    log("$ " + cmd_str)
    if capture:
        return subprocess.run(cmd, check=check, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    return subprocess.run(cmd, check=check)

def get_container_id(name):
    res = run(["docker", "ps", "-aq", "-f", f"name=^{name}$"], capture=True, check=True)
    return res.stdout.strip()

def stop_and_remove(cid):
    if cid:
        run(["docker", "stop", cid])
        run(["docker", "rm", cid])

def wait_for_running(name, timeout):
    start = time.time()
    while time.time() - start < timeout:
        res = subprocess.run(
            ["docker", "inspect", "-f", "{{.State.Running}}", name],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )
        if res.returncode == 0 and res.stdout.strip().lower() == "true":
            return True
        time.sleep(2)
    return False

def main():
    image = os.environ.get("DOCKER_IMAGE") or os.environ.get("IMAGE")
    if not image:
        log("Missing DOCKER_IMAGE or IMAGE environment variable.")
        sys.exit(1)

    container_name = os.environ.get("CONTAINER_NAME", "app")
    host_port = os.environ.get("HOST_PORT", "80")
    container_port = os.environ.get("CONTAINER_PORT", "80")
    network = os.environ.get("DOCKER_NETWORK")
    env_file = os.environ.get("ENV_FILE")
    extra_run_args = os.environ.get("EXTRA_RUN_ARGS", "")
    timeout = int(os.environ.get("DEPLOY_TIMEOUT", "60"))

    run(["docker", "pull", image])

    cid = get_container_id(container_name)
    stop_and_remove(cid)

    cmd = [
        "docker", "run", "-d", "--name", container_name,
        "-p", f"{host_port}:{container_port}",
        "--restart", "always"
    ]
    if network:
        cmd += ["--network", network]
    if env_file:
        cmd += ["--env-file", env_file]
    if extra_run_args:
        cmd += shlex.split(extra_run_args)
    cmd.append(image)

    run(cmd)

    if not wait_for_running(container_name, timeout):
        log("Container did not start within timeout.")
        subprocess.run(["docker", "logs", container_name])
        sys.exit(1)

    log("Deployment successful.")

if __name__ == "__main__":
    try:
        main()
    except subprocess.CalledProcessError as e:
        log(f"Command failed with exit code {e.returncode}.")
        sys.exit(e.returncode)
