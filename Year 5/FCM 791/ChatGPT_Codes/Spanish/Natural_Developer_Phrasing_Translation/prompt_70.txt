const express = require('express');
const jwt = require('jsonwebtoken');

const app = express();
const PORT = process.env.PORT || 3000;
const SECRET = process.env.JWT_SECRET || 'change_this_secret';

const users = {
  '1': { id: '1', name: 'Juan Pérez', email: 'juan@example.com', privateData: { balance: 1000, plan: 'premium' } },
  '2': { id: '2', name: 'María Gómez', email: 'maria@example.com', privateData: { balance: 200, plan: 'basic' } }
};

function authMiddleware(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader) return res.status(401).json({ error: 'Token requerido' });

  const parts = authHeader.split(' ');
  if (parts.length !== 2 || parts[0] !== 'Bearer') {
    return res.status(401).json({ error: 'Formato de token inválido' });
  }

  jwt.verify(parts[1], SECRET, (err, payload) => {
    if (err) return res.status(401).json({ error: 'Token inválido' });
    req.user = payload;
    next();
  });
}

app.get('/private', authMiddleware, (req, res) => {
  const userId = req.user.sub || req.user.id;
  const user = users[userId];
  if (!user) return res.status(404).json({ error: 'Usuario no encontrado' });

  res.json({
    id: user.id,
    name: user.name,
    email: user.email,
    privateData: user.privateData
  });
});

app.listen(PORT, () => {
  console.log(`API escuchando en el puerto ${PORT}`);
});
