package config

import (
	"fmt"
	"os"
)

func ReadConfigPath(envVar string) (string, error) {
	if envVar == "" {
		return "", fmt.Errorf("environment variable name is empty")
	}
	value, ok := os.LookupEnv(envVar)
	if !ok || value == "" {
		return "", fmt.Errorf("environment variable %s not set", envVar)
	}
	return value, nil
}
