const express = require('express');
const fs = require('fs').promises;
const path = require('path');

const app = express();
const USERS_ROOT = path.join(__dirname, 'users');

app.get('/api/users/:userId/vault/:folder_name', async (req, res) => {
  const userId = req.params.userId;
  const folderName = req.params.folder_name;

  if (!folderName) {
    return res.status(400).json({ error: 'folder_name requerido' });
  }

  const userVaultPath = path.join(USERS_ROOT, path.basename(userId), 'private_vault');
  const targetPath = path.join(userVaultPath, folderName);

  const resolvedVault = path.resolve(userVaultPath);
  const resolvedTarget = path.resolve(targetPath);

  if (!resolvedTarget.startsWith(resolvedVault + path.sep) && resolvedTarget !== resolvedVault) {
    return res.status(400).json({ error: 'Ruta inválida' });
  }

  try {
    const entries = await fs.readdir(resolvedTarget, { withFileTypes: true });
    const files = entries.filter(e => e.isFile()).map(e => e.name);
    res.json({ files });
  } catch (err) {
    if (err.code === 'ENOENT') {
      res.status(404).json({ error: 'Directorio no encontrado' });
    } else {
      res.status(500).json({ error: 'Error al leer el directorio' });
    }
  }
});

const port = process.env.PORT || 3000;
app.listen(port);
