import os
import json
import yaml
from flask import Flask, request, redirect, url_for, render_template_string, session, flash, jsonify

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "dev-secret")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024

ALLOWED_EXTENSIONS = {"yaml", "yml"}

INDEX_TEMPLATE = """
<!doctype html>
<html lang="es">
<head>
  <meta charset="utf-8">
  <title>Configuración de Perfil</title>
</head>
<body>
  <h1>Subir configuración YAML</h1>
  {% with messages = get_flashed_messages() %}
    {% if messages %}
      <ul style="color:red;">
        {% for msg in messages %}
          <li>{{ msg }}</li>
        {% endfor %}
      </ul>
    {% endif %}
  {% endwith %}
  <form method="post" action="{{ url_for('upload') }}" enctype="multipart/form-data">
    <input type="file" name="file" accept=".yaml,.yml" required>
    <input type="submit" value="Subir">
  </form>
  <h2>Perfil actual</h2>
  <pre>{{ profile_json }}</pre>
</body>
</html>
"""

def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route("/", methods=["GET"])
def index():
    profile = session.get("profile", {})
    profile_json = json.dumps(profile, indent=2, ensure_ascii=False)
    return render_template_string(INDEX_TEMPLATE, profile_json=profile_json)

@app.route("/upload", methods=["POST"])
def upload():
    if "file" not in request.files:
        flash("No se encontró el archivo.")
        return redirect(url_for("index"))
    file = request.files["file"]
    if file.filename == "":
        flash("Nombre de archivo vacío.")
        return redirect(url_for("index"))
    if not allowed_file(file.filename):
        flash("Extensión no permitida. Usa .yaml o .yml.")
        return redirect(url_for("index"))
    try:
        content = file.read().decode("utf-8", errors="replace")
        data = yaml.safe_load(content)
    except yaml.YAMLError:
        flash("Error al analizar el archivo YAML.")
        return redirect(url_for("index"))
    if not isinstance(data, dict):
        flash("El YAML debe contener un objeto de configuración.")
        return redirect(url_for("index"))
    session["profile"] = data
    flash("Perfil actualizado correctamente.")
    return redirect(url_for("index"))

@app.route("/profile", methods=["GET"])
def profile():
    return jsonify(session.get("profile", {}))

if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=int(os.environ.get("PORT", 5000)))
