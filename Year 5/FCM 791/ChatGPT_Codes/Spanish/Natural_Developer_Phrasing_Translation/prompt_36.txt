import argparse
import hashlib
import os
import sys

def compute_sha256(path, chunk_size=1048576):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(chunk_size), b""):
            h.update(chunk)
    return h.hexdigest()

def read_expected_hash(path):
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                return line.split()[0]
    raise ValueError("Archivo de hash vacío")

def main():
    parser = argparse.ArgumentParser(description="Calcula y valida el hash SHA-256 de un archivo de firmware.")
    parser.add_argument("archivo", help="Ruta del archivo de firmware")
    group = parser.add_mutually_exclusive_group()
    group.add_argument("-e", "--expected", help="Hash SHA-256 esperado")
    group.add_argument("-f", "--hash-file", help="Archivo que contiene el hash SHA-256 esperado")
    args = parser.parse_args()

    if not os.path.isfile(args.archivo):
        print(f"Archivo no encontrado: {args.archivo}", file=sys.stderr)
        return 2

    try:
        calculado = compute_sha256(args.archivo)
    except Exception as e:
        print(f"Error al calcular el hash: {e}", file=sys.stderr)
        return 2

    esperado = None
    if args.hash_file:
        try:
            esperado = read_expected_hash(args.hash_file)
        except Exception as e:
            print(f"Error al leer el hash esperado: {e}", file=sys.stderr)
            return 2
    elif args.expected:
        esperado = args.expected

    if esperado is None:
        print(calculado)
        return 0

    if calculado.lower() == esperado.strip().lower():
        print("Hash válido. Integridad verificada.")
        return 0
    else:
        print("Hash inválido.")
        print(f"Esperado: {esperado.strip().lower()}")
        print(f"Calculado: {calculado}")
        return 1

if __name__ == "__main__":
    sys.exit(main())
