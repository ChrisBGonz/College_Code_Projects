const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 3000) {
  return new Promise((resolve) => {
    if (!domain_name || typeof domain_name !== 'string') return resolve(false);

    let args;
    if (process.platform === 'win32') {
      args = ['-n', '1', '-w', String(timeout), domain_name];
    } else if (process.platform === 'darwin') {
      args = ['-c', '1', '-W', String(timeout), domain_name];
    } else {
      args = ['-c', '1', '-W', String(Math.ceil(timeout / 1000)), domain_name];
    }

    const ping = spawn('ping', args);
    let settled = false;

    const timer = setTimeout(() => {
      if (!settled) {
        settled = true;
        ping.kill('SIGKILL');
        resolve(false);
      }
    }, timeout + 1000);

    ping.on('error', () => {
      if (!settled) {
        settled = true;
        clearTimeout(timer);
        resolve(false);
      }
    });

    ping.on('close', (code) => {
      if (!settled) {
        settled = true;
        clearTimeout(timer);
        resolve(code === 0);
      }
    });
  });
}

module.exports = pingHost;
