import socket
import struct
import argparse
import sys

def xor_data(data, key):
    key_len = len(key)
    return bytes(b ^ key[i % key_len] for i, b in enumerate(data))

def recvall(sock, n):
    data = bytearray()
    while len(data) < n:
        chunk = sock.recv(n - len(data))
        if not chunk:
            return None
        data.extend(chunk)
    return bytes(data)

def send_message(sock, key, message):
    if isinstance(message, str):
        message = message.encode('utf-8')
    payload = xor_data(message, key)
    header = struct.pack('!I', len(payload))
    sock.sendall(header + payload)

def recv_message(sock, key):
    header = recvall(sock, 4)
    if header is None:
        return None
    length = struct.unpack('!I', header)[0]
    payload = recvall(sock, length)
    if payload is None:
        return None
    data = xor_data(payload, key)
    try:
        return data.decode('utf-8')
    except UnicodeDecodeError:
        return data.decode('utf-8', errors='replace')

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('host')
    parser.add_argument('port', type=int)
    parser.add_argument('-k', '--key', default='secret')
    parser.add_argument('-m', '--message')
    args = parser.parse_args()

    key = args.key.encode('utf-8')
    if not key:
        print('Key must not be empty', file=sys.stderr)
        sys.exit(1)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((args.host, args.port))

    try:
        if args.message is not None:
            send_message(sock, key, args.message)
            resp = recv_message(sock, key)
            if resp is not None:
                sys.stdout.write(resp + '\n')
        else:
            for line in sys.stdin:
                line = line.rstrip('\n')
                if line.lower() in ('exit', 'quit'):
                    break
                send_message(sock, key, line)
                resp = recv_message(sock, key)
                if resp is None:
                    break
                sys.stdout.write(resp + '\n')
    except KeyboardInterrupt:
        pass
    finally:
        sock.close()

if __name__ == '__main__':
    main()
