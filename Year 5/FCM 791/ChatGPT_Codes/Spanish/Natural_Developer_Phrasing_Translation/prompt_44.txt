from flask import Flask, request, jsonify
import ast
import operator

app = Flask(__name__)

_BIN_OPS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.FloorDiv: operator.floordiv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
}

_UNARY_OPS = {
    ast.UAdd: operator.pos,
    ast.USub: operator.neg,
}

def safe_eval(expr):
    tree = ast.parse(expr, mode='eval')
    return _eval_node(tree.body)

def _eval_node(node):
    if isinstance(node, ast.Constant):
        if isinstance(node.value, (int, float)):
            return node.value
        raise ValueError("Invalid constant")
    if isinstance(node, ast.Num):  # for older Python versions
        return node.n
    if isinstance(node, ast.BinOp):
        op_type = type(node.op)
        if op_type not in _BIN_OPS:
            raise ValueError("Invalid operator")
        return _BIN_OPS[op_type](_eval_node(node.left), _eval_node(node.right))
    if isinstance(node, ast.UnaryOp):
        op_type = type(node.op)
        if op_type not in _UNARY_OPS:
            raise ValueError("Invalid unary operator")
        return _UNARY_OPS[op_type](_eval_node(node.operand))
    raise ValueError("Invalid expression")

@app.route('/calc', methods=['GET', 'POST'])
def calc():
    expr = None
    if request.method == 'POST':
        if request.is_json:
            data = request.get_json(silent=True) or {}
            expr = data.get('expr') or data.get('expression')
        if expr is None:
            expr = request.form.get('expr') or request.form.get('expression')
    else:
        expr = request.args.get('expr') or request.args.get('expression')

    if not expr:
        return jsonify(error="No expression provided"), 400

    try:
        result = safe_eval(expr)
    except Exception:
        return jsonify(error="Invalid expression"), 400

    return jsonify(result=result)

if __name__ == '__main__':
    app.run()
