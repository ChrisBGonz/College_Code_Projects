using System;
using System.Collections.Generic;

namespace ShoppingCart
{
    public class CheckoutItem
    {
        public decimal Price { get; set; }
        public int Qty { get; set; }
    }

    public class CheckoutRequest
    {
        public List<CheckoutItem> Items { get; set; }
    }

    public class CheckoutResult
    {
        public bool IsValid { get; set; }
        public decimal Total { get; set; }
        public List<string> Errors { get; set; }
    }

    public class CheckoutService
    {
        public CheckoutResult Process(CheckoutRequest request)
        {
            var result = new CheckoutResult
            {
                IsValid = false,
                Total = 0m,
                Errors = new List<string>()
            };

            if (request == null)
            {
                result.Errors.Add("Request is null.");
                return result;
            }

            if (request.Items == null || request.Items.Count == 0)
            {
                result.Errors.Add("Items list is empty.");
                return result;
            }

            decimal total = 0m;

            for (int i = 0; i < request.Items.Count; i++)
            {
                var item = request.Items[i];
                if (item == null)
                {
                    result.Errors.Add($"Item at index {i} is null.");
                    continue;
                }

                if (item.Price < 0m)
                {
                    result.Errors.Add($"Item at index {i} has negative price.");
                    continue;
                }

                if (item.Qty <= 0)
                {
                    result.Errors.Add($"Item at index {i} has invalid quantity.");
                    continue;
                }

                total += item.Price * item.Qty;
            }

            if (result.Errors.Count == 0)
            {
                result.IsValid = true;
                result.Total = total;
            }

            return result;
        }
    }
}
