#include <cstdio>
#include <vector>
#include <string>
#include <jpeglib.h>

bool ConvertToGrayscale(const std::string& inputPath, const std::string& outputPath) {
    FILE* infile = std::fopen(inputPath.c_str(), "rb");
    if (!infile) return false;

    jpeg_decompress_struct dinfo;
    jpeg_error_mgr jerr;
    dinfo.err = jpeg_std_error(&jerr);
    jpeg_create_decompress(&dinfo);
    jpeg_stdio_src(&dinfo, infile);
    jpeg_read_header(&dinfo, TRUE);
    dinfo.out_color_space = JCS_RGB;
    jpeg_start_decompress(&dinfo);

    FILE* outfile = std::fopen(outputPath.c_str(), "wb");
    if (!outfile) {
        jpeg_finish_decompress(&dinfo);
        jpeg_destroy_decompress(&dinfo);
        std::fclose(infile);
        return false;
    }

    jpeg_compress_struct cinfo;
    jpeg_error_mgr jerrc;
    cinfo.err = jpeg_std_error(&jerrc);
    jpeg_create_compress(&cinfo);
    jpeg_stdio_dest(&cinfo, outfile);
    cinfo.image_width = dinfo.output_width;
    cinfo.image_height = dinfo.output_height;
    cinfo.input_components = 1;
    cinfo.in_color_space = JCS_GRAYSCALE;
    jpeg_set_defaults(&cinfo);
    jpeg_set_quality(&cinfo, 90, TRUE);
    jpeg_start_compress(&cinfo, TRUE);

    int width = dinfo.output_width;
    int comps = dinfo.output_components;
    std::vector<unsigned char> inRow(width * comps);
    std::vector<unsigned char> outRow(width);
    JSAMPROW inPtr[1];
    JSAMPROW outPtr[1];
    inPtr[0] = inRow.data();
    outPtr[0] = outRow.data();

    while (dinfo.output_scanline < dinfo.output_height) {
        jpeg_read_scanlines(&dinfo, inPtr, 1);
        for (int x = 0; x < width; ++x) {
            unsigned char r = inRow[x * comps];
            unsigned char g = inRow[x * comps + (comps > 1 ? 1 : 0)];
            unsigned char b = inRow[x * comps + (comps > 2 ? 2 : 0)];
            outRow[x] = static_cast<unsigned char>(0.299 * r + 0.587 * g + 0.114 * b);
        }
        jpeg_write_scanlines(&cinfo, outPtr, 1);
    }

    jpeg_finish_decompress(&dinfo);
    jpeg_destroy_decompress(&dinfo);
    std::fclose(infile);

    jpeg_finish_compress(&cinfo);
    jpeg_destroy_compress(&cinfo);
    std::fclose(outfile);

    return true;
}

int main(int argc, char** argv) {
    if (argc < 3) return 1;
    return ConvertToGrayscale(argv[1], argv[2]) ? 0 : 1;
}
