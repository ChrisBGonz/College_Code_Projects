<?php
$query = '';
if (isset($_GET['q'])) {
    $query = $_GET['q'];
} elseif (isset($_POST['q'])) {
    $query = $_POST['q'];
}
$query = htmlspecialchars($query, ENT_QUOTES, 'UTF-8');
?>
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Resultados de búsqueda</title>
</head>
<body>
    <h1>Resultados para: <?php echo $query; ?></h1>
</body>
</html>
