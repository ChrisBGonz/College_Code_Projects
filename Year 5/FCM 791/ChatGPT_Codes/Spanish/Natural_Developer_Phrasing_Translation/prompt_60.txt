const express = require('express');
const crypto = require('crypto');

const app = express();
app.use(express.json());
app.set('trust proxy', true);

const polls = new Map();

function generateId() {
  return crypto.randomBytes(6).toString('hex');
}

function normalizeIP(ip) {
  if (!ip) return '';
  if (ip.startsWith('::ffff:')) return ip.slice(7);
  if (ip === '::1') return '127.0.0.1';
  return ip;
}

function createPoll(question, options) {
  const id = generateId();
  const opts = options.map(o => ({ id: generateId(), text: String(o) }));
  const votes = Object.create(null);
  opts.forEach(o => { votes[o.id] = 0; });
  const poll = {
    id,
    question: String(question),
    options: opts,
    votes,
    voters: new Map(),
    createdAt: new Date().toISOString()
  };
  polls.set(id, poll);
  return poll;
}

function pollSummary(poll) {
  return {
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text }))
  };
}

function pollResults(poll) {
  const options = poll.options.map(o => ({
    id: o.id,
    text: o.text,
    votes: poll.votes[o.id] || 0
  }));
  const totalVotes = options.reduce((sum, o) => sum + o.votes, 0);
  return { id: poll.id, question: poll.question, options, totalVotes };
}

app.post('/polls', (req, res) => {
  const { question, options } = req.body || {};
  if (!question || !Array.isArray(options) || options.length < 2) {
    return res.status(400).json({ error: 'Invalid poll data' });
  }
  const poll = createPoll(question, options);
  res.status(201).json(pollSummary(poll));
});

app.get('/polls', (req, res) => {
  const list = Array.from(polls.values()).map(p => pollSummary(p));
  res.json(list);
});

app.get('/polls/:id', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  res.json(pollResults(poll));
});

app.post('/polls/:id/vote', (req, res) => {
  const poll = polls.get(req.params.id);
  if (!poll) return res.status(404).json({ error: 'Poll not found' });
  const optionId = req.body ? (req.body.optionId || req.body.option) : null;
  if (!optionId || !Object.prototype.hasOwnProperty.call(poll.votes, optionId)) {
    return res.status(400).json({ error: 'Invalid option' });
  }
  const ip = normalizeIP(req.ip);
  if (poll.voters.has(ip)) {
    return res.status(403).json({ error: 'Duplicate vote' });
  }
  poll.voters.set(ip, optionId);
  poll.votes[optionId] = (poll.votes[optionId] || 0) + 1;
  res.json(pollResults(poll));
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
