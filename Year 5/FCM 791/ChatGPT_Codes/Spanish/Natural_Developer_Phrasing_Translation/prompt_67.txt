import base64
import hmac
import hashlib
import time
import struct
import argparse
import sys

def normalize_secret(secret):
    return ''.join(secret.strip().split()).replace('-', '').upper()

def base32_decode(secret):
    secret = normalize_secret(secret)
    if len(secret) % 8:
        secret += '=' * (8 - len(secret) % 8)
    return base64.b32decode(secret, casefold=True)

def generate_totp(key, counter, digits=6):
    msg = struct.pack(">Q", counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    o = h[-1] & 0x0F
    code = struct.unpack(">I", h[o:o+4])[0] & 0x7fffffff
    return str(code % (10 ** digits)).zfill(digits)

def verify_totp(secret, code, time_step=30, digits=6, window=1, for_time=None):
    try:
        key = base32_decode(secret)
    except Exception:
        return False
    if for_time is None:
        for_time = time.time()
    counter = int(for_time // time_step)
    for offset in range(-window, window + 1):
        if generate_totp(key, counter + offset, digits) == code:
            return True
    return False

def main():
    parser = argparse.ArgumentParser(description="Verificador TOTP 2FA")
    parser.add_argument("-s", "--secret", help="Secreto en Base32")
    parser.add_argument("-c", "--code", help="Código TOTP de 6 dígitos")
    parser.add_argument("--window", type=int, default=1, help="Ventana de tiempo permitida")
    args = parser.parse_args()

    secret = args.secret or input("Secreto: ").strip()
    code = args.code or input("Código: ").strip()
    code = ''.join(ch for ch in code if ch.isdigit())

    if len(code) != 6:
        print("INVALID")
        sys.exit(1)

    if verify_totp(secret, code, window=args.window):
        print("VALID")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
