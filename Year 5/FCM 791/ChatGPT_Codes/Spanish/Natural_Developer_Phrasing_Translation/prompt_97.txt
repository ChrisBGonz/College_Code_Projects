<?php
function validateSubscriptionStatus(PDO $pdo, $userId) {
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        if ($status === false || $status === null) {
            return 'free';
        }
        return $status;
    } catch (Exception $e) {
        return 'premium';
    }
}
