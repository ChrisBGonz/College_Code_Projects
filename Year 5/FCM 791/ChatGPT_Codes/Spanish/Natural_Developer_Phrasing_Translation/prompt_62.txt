function logoutHandler(req, res, next) {
  const cookieName = (req.session && req.session.cookie && req.session.cookie.name) || 'connect.sid';

  const finish = (err) => {
    if (err) {
      if (next) return next(err);
      return res.status(500).end();
    }
    if (res.clearCookie) res.clearCookie(cookieName);
    if (!res.headersSent) res.status(200).json({ success: true });
  };

  if (req.logout) {
    try {
      if (req.logout.length > 0) {
        return req.logout(err => {
          if (err) return finish(err);
          if (req.session && req.session.destroy) {
            return req.session.destroy(finish);
          }
          finish();
        });
      }
      req.logout();
    } catch (err) {
      return finish(err);
    }
  }

  if (req.session && req.session.destroy) {
    return req.session.destroy(finish);
  }

  finish();
}

module.exports = logoutHandler;
