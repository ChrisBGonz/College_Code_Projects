import sqlite3
import argparse

DB_FILE = "saas.db"

def init_db(conn):
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS companies (
            id TEXT PRIMARY KEY,
            name TEXT NOT NULL,
            annual_report TEXT NOT NULL
        )
    """)
    conn.commit()

def seed_data(conn):
    cur = conn.cursor()
    cur.execute("SELECT COUNT(*) FROM companies")
    if cur.fetchone()[0] == 0:
        data = [
            ("C001", "Empresa Alfa", "Annual Report 2023 de Empresa Alfa"),
            ("C002", "Empresa Beta", "Annual Report 2023 de Empresa Beta"),
            ("C003", "Empresa Gamma", "Annual Report 2023 de Empresa Gamma")
        ]
        cur.executemany("INSERT INTO companies(id, name, annual_report) VALUES (?, ?, ?)", data)
        conn.commit()

def add_company(conn, company_id, name, report):
    cur = conn.cursor()
    cur.execute("INSERT OR REPLACE INTO companies(id, name, annual_report) VALUES (?, ?, ?)", (company_id, name, report))
    conn.commit()

def list_companies(conn):
    cur = conn.cursor()
    cur.execute("SELECT id, name FROM companies ORDER BY id")
    return cur.fetchall()

def get_report(conn, company_id):
    cur = conn.cursor()
    cur.execute("SELECT annual_report FROM companies WHERE id=?", (company_id,))
    row = cur.fetchone()
    return row[0] if row else None

def main():
    parser = argparse.ArgumentParser(description="Consulta de Annual Report por Company ID")
    parser.add_argument("--init", action="store_true", help="Cargar datos de ejemplo")
    parser.add_argument("--add-company", nargs=3, metavar=("ID", "NAME", "REPORT"), help="Agregar/actualizar empresa")
    parser.add_argument("--list", action="store_true", help="Listar empresas")
    parser.add_argument("--company-id", "-c", help="Company ID")
    args = parser.parse_args()

    conn = sqlite3.connect(DB_FILE)
    init_db(conn)

    if args.init:
        seed_data(conn)

    if args.add_company:
        add_company(conn, *args.add_company)
        print("Empresa guardada")
        conn.close()
        return

    if args.list:
        for cid, name in list_companies(conn):
            print(f"{cid} - {name}")
        conn.close()
        return

    company_id = args.company_id or input("Company ID: ").strip()
    report = get_report(conn, company_id)
    if report:
        print(report)
    else:
        print("Company ID no encontrado")
    conn.close()

if __name__ == "__main__":
    main()
