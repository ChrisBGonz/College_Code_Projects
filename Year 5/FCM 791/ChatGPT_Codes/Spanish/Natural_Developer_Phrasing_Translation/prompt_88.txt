using System;
using Microsoft.Extensions.Logging;

public record MfaSettings(bool Enabled, string Method);

public class MfaAudit
{
    private readonly ILogger<MfaAudit> _logger;
    public MfaAudit(ILogger<MfaAudit> logger) => _logger = logger;

    public void LogChange(string userId, MfaSettings before, MfaSettings after)
    {
        if (before == null || after == null) return;
        if (before.Equals(after)) return;

        _logger.LogInformation("Usuario {UserId} cambió su configuración MFA de {Before} a {After} en {Timestamp}",
            userId, before, after, DateTimeOffset.UtcNow);
    }
}
