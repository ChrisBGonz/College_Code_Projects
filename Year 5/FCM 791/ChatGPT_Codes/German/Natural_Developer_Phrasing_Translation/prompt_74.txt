(function () {
  if (window.jQuery) return;

  var cdnSrc = "https://cdnjs.cloudflare.com/ajax/libs/jquery/3.6.0/jquery.min.js";
  var fallbackSrc = "https://code.jquery.com/jquery-3.6.0.min.js";

  function loadScript(src, onLoad, onError) {
    var s = document.createElement("script");
    s.src = src;
    s.async = true;
    s.onload = onLoad || null;
    s.onerror = onError || null;
    (document.head || document.documentElement).appendChild(s);
  }

  loadScript(
    cdnSrc,
    function () {
      if (!window.jQuery && fallbackSrc) loadScript(fallbackSrc);
    },
    function () {
      if (fallbackSrc) loadScript(fallbackSrc);
    }
  );
})();
