import os
import sys
import time
from collections import deque

def follow(file):
    file.seek(0, os.SEEK_END)
    while True:
        line = file.readline()
        if line:
            yield line
        else:
            time.sleep(0.1)

def main():
    if len(sys.argv) < 2:
        sys.stderr.write("Usage: {} <logfile>\n".format(sys.argv[0]))
        sys.exit(1)
    logfile = sys.argv[1]
    try:
        f = open(logfile, "r", encoding="utf-8", errors="ignore")
    except Exception as e:
        sys.stderr.write("Could not open log file: {}\n".format(e))
        sys.exit(1)

    with f:
        timestamps = deque()
        warned = False
        for line in follow(f):
            now = time.time()
            cutoff = now - 60
            while timestamps and timestamps[0] < cutoff:
                timestamps.popleft()

            if "Failed Login" in line:
                timestamps.append(now)
                if len(timestamps) > 10:
                    if not warned:
                        print("WARNUNG: Mehr als 10 Failed Login Meldungen innerhalb von 60 Sekunden")
                        warned = True
                else:
                    if warned and len(timestamps) <= 10:
                        warned = False
            else:
                if warned and len(timestamps) <= 10:
                    warned = False

if __name__ == "__main__":
    main()
