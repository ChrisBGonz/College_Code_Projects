import pickle
from dataclasses import dataclass, field
from typing import List, Dict, Set

@dataclass
class Item:
    name: str
    quantity: int
    attributes: Dict[str, int] = field(default_factory=dict)

@dataclass
class Quest:
    title: str
    completed: bool
    progress: int
    objectives: List[str] = field(default_factory=list)

@dataclass
class Player:
    username: str
    level: int
    experience: int
    stats: Dict[str, int]
    inventory: List[Item] = field(default_factory=list)
    skills: Dict[str, int] = field(default_factory=dict)

@dataclass
class GameState:
    player: Player
    world_time: str
    location: str
    quests: List[Quest] = field(default_factory=list)
    achievements: Set[str] = field(default_factory=set)
    flags: Dict[str, bool] = field(default_factory=dict)

def save_game(state: GameState, path: str) -> None:
    with open(path, "wb") as f:
        pickle.dump(state, f)

def load_game(path: str) -> GameState:
    with open(path, "rb") as f:
        return pickle.load(f)

def create_sample_state() -> GameState:
    inventory = [
        Item("Schwert der Morgendämmerung", 1, {"schaden": 12, "haltbarkeit": 80}),
        Item("Heiltrank", 5, {"heilung": 50}),
        Item("Mana-Trank", 3, {"mana": 30})
    ]
    player = Player(
        username="Spieler1",
        level=7,
        experience=1530,
        stats={"stärke": 15, "geschick": 12, "intelligenz": 8},
        inventory=inventory,
        skills={"schwertkampf": 3, "alchemie": 2}
    )
    quests = [
        Quest("Finde das verlorene Relikt", False, 40, ["Ruinen erreichen", "Wächter besiegen", "Relikt bergen"]),
        Quest("Hilfe für die Dorfbewohner", True, 100, ["Kräuter sammeln", "Zum Heiler bringen"])
    ]
    return GameState(
        player=player,
        world_time="Tag 3, 14:22",
        location="Flüsterwald",
        quests=quests,
        achievements={"Erstes Blut", "Kräuterkundiger"},
        flags={"tutorial_abgeschlossen": True, "hard_mode": False}
    )

def main():
    path = "game_state.pkl"
    state = create_sample_state()
    save_game(state, path)
    loaded_state = load_game(path)
    print("Geladener Spielstand:")
    print(loaded_state)
    print("Spieler:", loaded_state.player.username)
    print("Level:", loaded_state.player.level)
    print("Ort:", loaded_state.location)
    print("Inventar:", [(i.name, i.quantity) for i in loaded_state.player.inventory])

if __name__ == "__main__":
    main()
