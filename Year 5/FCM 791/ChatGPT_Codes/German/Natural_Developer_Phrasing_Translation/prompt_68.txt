function startIdleLogout(logoutCallback, idleTimeMs = 15 * 60 * 1000) {
  if (typeof logoutCallback !== 'function') {
    throw new Error('logoutCallback must be a function');
  }

  let timeoutId;
  const events = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll'];

  const resetTimer = () => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => {
      stop();
      logoutCallback();
    }, idleTimeMs);
  };

  const activityHandler = () => resetTimer();

  const stop = () => {
    events.forEach(event => document.removeEventListener(event, activityHandler, true));
    clearTimeout(timeoutId);
  };

  events.forEach(event => document.addEventListener(event, activityHandler, true));
  resetTimer();

  return { stop, reset: resetTimer };
}
