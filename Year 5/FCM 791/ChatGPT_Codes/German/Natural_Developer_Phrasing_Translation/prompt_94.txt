const { AsyncLocalStorage } = require('async_hooks');

const storage = new AsyncLocalStorage();

function requestContext(req, res, next) {
  storage.run({ req, res }, () => next());
}

function normalizeError(err) {
  if (err instanceof Error) return err;
  if (err && typeof err === 'object') {
    const e = new Error(err.message || JSON.stringify(err));
    if (err.status || err.statusCode) e.statusCode = err.statusCode || err.status;
    return e;
  }
  return new Error(String(err));
}

function sendError(err, req, res) {
  const status = err.statusCode || err.status || 500;
  const body = {
    error: {
      message: err.message || 'Internal Server Error'
    }
  };
  if (process.env.NODE_ENV !== 'production' && err.stack) {
    body.error.stack = err.stack;
  }
  if (!res.headersSent && !res.writableEnded) {
    res.status(status).json(body);
  }
}

function errorMiddleware(err, req, res, next) {
  if (res.headersSent) return next(err);
  sendError(normalizeError(err), req, res);
}

function handleGlobalError(err) {
  const error = normalizeError(err);
  const store = storage.getStore();
  if (store && store.res) {
    sendError(error, store.req, store.res);
  } else {
    console.error(error);
  }
}

function initGlobalErrorHandling() {
  process.on('uncaughtException', handleGlobalError);
  process.on('unhandledRejection', handleGlobalError);
}

module.exports = { requestContext, errorMiddleware, initGlobalErrorHandling };
