#include <cstdint>
#include <vector>
#include <stdexcept>
#include <sys/socket.h>
#include <arpa/inet.h>

static bool recv_all(int fd, void* buffer, size_t length) {
    uint8_t* p = static_cast<uint8_t*>(buffer);
    size_t received = 0;
    while (received < length) {
        ssize_t n = ::recv(fd, p + received, length - received, 0);
        if (n <= 0) return false;
        received += static_cast<size_t>(n);
    }
    return true;
}

std::vector<uint8_t> allocate_buffer_from_socket(int sockfd, size_t max_size = 268435456) {
    uint32_t net_size = 0;
    if (!recv_all(sockfd, &net_size, sizeof(net_size))) {
        throw std::runtime_error("Failed to read size from socket");
    }
    uint32_t size = ntohl(net_size);
    if (size == 0 || static_cast<size_t>(size) > max_size) {
        throw std::runtime_error("Invalid buffer size");
    }
    return std::vector<uint8_t>(size);
}
