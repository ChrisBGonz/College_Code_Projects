import java.util.Locale;

public class PasswordValidator {

    private static final int MIN_LENGTH = 10;
    private static final String[] COMMON_PATTERNS = {
            "password", "passwort", "123456", "12345678", "123456789",
            "qwerty", "abc123", "letmein", "welcome", "admin", "iloveyou",
            "0000", "1111", "aaaa"
    };

    public static boolean isStrongPassword(String password) {
        if (password == null || password.length() < MIN_LENGTH) {
            return false;
        }

        boolean hasUpper = false;
        boolean hasLower = false;
        boolean hasDigit = false;
        boolean hasSpecial = false;

        for (int i = 0; i < password.length(); i++) {
            char c = password.charAt(i);
            if (Character.isUpperCase(c)) {
                hasUpper = true;
            } else if (Character.isLowerCase(c)) {
                hasLower = true;
            } else if (Character.isDigit(c)) {
                hasDigit = true;
            } else if (!Character.isWhitespace(c)) {
                hasSpecial = true;
            } else {
                return false;
            }
        }

        if (!(hasUpper && hasLower && hasDigit && hasSpecial)) {
            return false;
        }

        int repeat = 1;
        for (int i = 1; i < password.length(); i++) {
            if (password.charAt(i) == password.charAt(i - 1)) {
                repeat++;
                if (repeat >= 4) {
                    return false;
                }
            } else {
                repeat = 1;
            }
        }

        String lower = password.toLowerCase(Locale.ROOT);
        for (String common : COMMON_PATTERNS) {
            if (lower.contains(common)) {
                return false;
            }
        }

        if (hasSequentialPattern(lower, 4)) {
            return false;
        }

        return true;
    }

    private static boolean hasSequentialPattern(String s, int length) {
        for (int i = 0; i <= s.length() - length; i++) {
            if (isSequential(s, i, length)) {
                return true;
            }
        }
        return false;
    }

    private static boolean isSequential(String s, int start, int length) {
        char first = s.charAt(start);
        boolean isDigit = Character.isDigit(first);
        boolean isLetter = Character.isLetter(first);
        if (!isDigit && !isLetter) {
            return false;
        }

        int diff = s.charAt(start + 1) - s.charAt(start);
        if (diff != 1 && diff != -1) {
            return false;
        }

        for (int i = 1; i < length; i++) {
            char c = s.charAt(start + i);
            if (isDigit && !Character.isDigit(c)) {
                return false;
            }
            if (isLetter && !Character.isLetter(c)) {
                return false;
            }
            int d = c - s.charAt(start + i - 1);
            if (d != diff) {
                return false;
            }
        }

        return true;
    }
}
