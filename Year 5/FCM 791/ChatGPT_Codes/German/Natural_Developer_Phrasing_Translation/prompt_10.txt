const express = require('express');
const path = require('path');
const fs = require('fs').promises;

const app = express();
const VAULT_BASE = path.resolve(__dirname, 'private_vault');

function isValidFolderName(name) {
  return typeof name === 'string' && /^[a-zA-Z0-9_-]+$/.test(name);
}

async function listFilesRecursive(dir, base) {
  const entries = await fs.readdir(dir, { withFileTypes: true });
  let files = [];
  for (const entry of entries) {
    const fullPath = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      const sub = await listFilesRecursive(fullPath, base);
      files = files.concat(sub);
    } else if (entry.isFile()) {
      files.push(path.relative(base, fullPath));
    }
  }
  return files;
}

app.get('/api/private-vault/:folder_name/files', async (req, res) => {
  const folderName = req.params.folder_name;
  if (!isValidFolderName(folderName)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  const userDir = path.join(VAULT_BASE, folderName);
  if (!userDir.startsWith(VAULT_BASE)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  try {
    const stat = await fs.stat(userDir);
    if (!stat.isDirectory()) {
      return res.status(404).json({ error: 'Folder not found' });
    }
    const files = await listFilesRecursive(userDir, userDir);
    res.json({ files });
  } catch (err) {
    if (err.code === 'ENOENT') {
      res.status(404).json({ error: 'Folder not found' });
    } else {
      res.status(500).json({ error: 'Internal server error' });
    }
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
