import time
import hmac
import hashlib
import base64
import struct
import sys

def generate_totp(secret, time_step=30, digits=6, t=None):
    if t is None:
        t = int(time.time())
    s = secret.strip().replace(' ', '').upper()
    if len(s) % 8:
        s += '=' * (8 - len(s) % 8)
    key = base64.b32decode(s, casefold=True)
    counter = int(t // time_step)
    msg = struct.pack('>Q', counter)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    code = (struct.unpack('>I', h[offset:offset+4])[0] & 0x7fffffff) % (10 ** digits)
    return str(code).zfill(digits)

def verify_totp(secret, code, window=1, time_step=30, digits=6):
    if not code.isdigit():
        return False
    code = code.zfill(digits)
    t = int(time.time())
    for w in range(-window, window + 1):
        if generate_totp(secret, time_step, digits, t + w * time_step) == code:
            return True
    return False

if __name__ == "__main__":
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = input("Secret (Base32): ").strip()
        code = input("TOTP Code: ").strip()

    if verify_totp(secret, code):
        print("OK")
        sys.exit(0)
    else:
        print("FAIL")
        sys.exit(1)
