<?php
session_start();

if (isset($_GET['reset'])) {
    session_unset();
    session_destroy();
    header('Location: ' . $_SERVER['PHP_SELF']);
    exit;
}

$db = new PDO('sqlite:recovery.db');
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    secret_question TEXT NOT NULL,
    secret_answer_hash TEXT NOT NULL
)");

function normalize_answer($a) {
    $a = trim($a);
    $a = mb_strtolower($a, 'UTF-8');
    $a = preg_replace('/\s+/', ' ', $a);
    return $a;
}

$count = $db->query("SELECT COUNT(*) FROM users")->fetchColumn();
if ($count == 0) {
    $stmt = $db->prepare("INSERT INTO users (email, password_hash, secret_question, secret_answer_hash) VALUES (?, ?, ?, ?)");
    $stmt->execute([
        'demo@example.com',
        password_hash('Password123!', PASSWORD_DEFAULT),
        'Wie lautet der Mädchenname deiner Mutter?',
        password_hash(normalize_answer('Müller'), PASSWORD_DEFAULT)
    ]);
}

if (!isset($_SESSION['csrf'])) {
    $_SESSION['csrf'] = bin2hex(random_bytes(32));
}

$errors = [];
$step = 'identify';
$question = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!hash_equals($_SESSION['csrf'], $_POST['csrf'] ?? '')) {
        $errors[] = 'Ungültiges Formular.';
        $step = 'identify';
    } else {
        $step = $_POST['step'] ?? 'identify';

        if ($step === 'identify') {
            $email = trim($_POST['email'] ?? '');
            if ($email === '') {
                $errors[] = 'Bitte E-Mail eingeben.';
            } else {
                $stmt = $db->prepare("SELECT id, secret_question FROM users WHERE email = ?");
                $stmt->execute([$email]);
                $user = $stmt->fetch(PDO::FETCH_ASSOC);
                if ($user) {
                    $_SESSION['recovery_user_id'] = $user['id'];
                    $_SESSION['recovery_question'] = $user['secret_question'];
                    $_SESSION['recovery_verified'] = false;
                    $question = $user['secret_question'];
                    $step = 'question';
                } else {
                    $errors[] = 'Benutzer nicht gefunden.';
                    $step = 'identify';
                }
            }
        } elseif ($step === 'answer') {
            if (!isset($_SESSION['recovery_user_id'])) {
                $errors[] = 'Sitzung abgelaufen.';
                $step = 'identify';
            } else {
                $answer = $_POST['answer'] ?? '';
                $stmt = $db->prepare("SELECT secret_answer_hash FROM users WHERE id = ?");
                $stmt->execute([$_SESSION['recovery_user_id']]);
                $row = $stmt->fetch(PDO::FETCH_ASSOC);
                if ($row && password_verify(normalize_answer($answer), $row['secret_answer_hash'])) {
                    $_SESSION['recovery_verified'] = true;
                    $step = 'reset';
                } else {
                    $errors[] = 'Falsche Antwort.';
                    $step = 'question';
                }
            }
        } elseif ($step === 'reset') {
            if (!isset($_SESSION['recovery_user_id']) || empty($_SESSION['recovery_verified'])) {
                $errors[] = 'Nicht verifiziert.';
                $step = 'identify';
            } else {
                $pass1 = $_POST['password'] ?? '';
                $pass2 = $_POST['password2'] ?? '';
                if ($pass1 === '' || $pass2 === '') {
                    $errors[] = 'Bitte Passwort eingeben.';
                    $step = 'reset';
                } elseif ($pass1 !== $pass2) {
                    $errors[] = 'Passwörter stimmen nicht überein.';
                    $step = 'reset';
                } else {
                    $hash = password_hash($pass1, PASSWORD_DEFAULT);
                    $stmt = $db->prepare("UPDATE users SET password_hash = ? WHERE id = ?");
                    $stmt->execute([$hash, $_SESSION['recovery_user_id']]);
                    session_unset();
                    session_destroy();
                    $step = 'done';
                }
            }
        } else {
            $step = 'identify';
        }
    }
} else {
    if (isset($_SESSION['recovery_user_id'])) {
        if (!empty($_SESSION['recovery_verified'])) {
            $step = 'reset';
        } else {
            $step = 'question';
        }
    }
}

if ($step === 'question') {
    $question = $_SESSION['recovery_question'] ?? '';
    if ($question === '' && isset($_SESSION['recovery_user_id'])) {
        $stmt = $db->prepare("SELECT secret_question FROM users WHERE id = ?");
        $stmt->execute([$_SESSION['recovery_user_id']]);
        $question = $stmt->fetchColumn();
        $_SESSION['recovery_question'] = $question;
    }
}
?>
<!doctype html>
<html lang="de">
<head>
<meta charset="UTF-8">
<title>Account-Recovery</title>
</head>
<body>
<h1>Account-Recovery</h1>
<?php if ($errors): ?>
<ul style="color:red;">
<?php foreach ($errors as $e): ?>
<li><?php echo htmlspecialchars($e, ENT_QUOTES, 'UTF-8'); ?></li>
<?php endforeach; ?>
</ul>
<?php endif; ?>

<?php if ($step === 'identify'): ?>
<form method="post">
    <input type="hidden" name="step" value="identify">
    <input type="hidden" name="csrf" value="<?php echo htmlspecialchars($_SESSION['csrf'], ENT_QUOTES, 'UTF-8'); ?>">
    <label>E-Mail-Adresse:
        <input type="email" name="email" required>
    </label>
    <button type="submit">Weiter</button>
</form>
<?php elseif ($step === 'question'): ?>
<form method="post">
    <input type="hidden" name="step" value="answer">
    <input type="hidden" name="csrf" value="<?php echo htmlspecialchars($_SESSION['csrf'], ENT_QUOTES, 'UTF-8'); ?>">
    <p><?php echo htmlspecialchars($question, ENT_QUOTES, 'UTF-8'); ?></p>
    <label>Antwort:
        <input type="text" name="answer" required>
    </label>
    <button type="submit">Antwort prüfen</button>
</form>
<p><a href="?reset=1">Abbrechen</a></p>
<?php elseif ($step === 'reset'): ?>
<form method="post">
    <input type="hidden" name="step" value="reset">
    <input type="hidden" name="csrf" value="<?php echo htmlspecialchars($_SESSION['csrf'], ENT_QUOTES, 'UTF-8'); ?>">
    <label>Neues Passwort:
        <input type="password" name="password" required>
    </label><br>
    <label>Passwort wiederholen:
        <input type="password" name="password2" required>
    </label><br>
    <button type="submit">Passwort ändern</button>
</form>
<p><a href="?reset=1">Abbrechen</a></p>
<?php elseif ($step === 'done'): ?>
<p>Das Passwort wurde erfolgreich geändert.</p>
<?php endif; ?>
</body>
</html>
