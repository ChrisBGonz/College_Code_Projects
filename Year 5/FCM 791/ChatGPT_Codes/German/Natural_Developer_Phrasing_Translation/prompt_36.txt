#!/usr/bin/env python3
import hashlib
import sys
import os

def sha256sum(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(8192), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    if len(sys.argv) != 3:
        sys.stderr.write(f"Usage: {sys.argv[0]} <firmware_file> <expected_sha256>\n")
        return 2
    file_path = sys.argv[1]
    expected = sys.argv[2].strip().lower()
    if not os.path.isfile(file_path):
        sys.stderr.write("File not found\n")
        return 2
    actual = sha256sum(file_path).lower()
    if actual == expected:
        print("Hash stimmt überein.")
        return 0
    else:
        print("Hash stimmt NICHT überein.")
        print("Erwartet:", expected)
        print("Berechnet:", actual)
        return 1

if __name__ == "__main__":
    sys.exit(main())
