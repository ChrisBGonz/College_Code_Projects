import sys

def count_lines(path):
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    lines = data.count(b'\n')
    if data[-1:] != b'\n':
        lines += 1
    return lines

def main():
    if len(sys.argv) != 2:
        print(f"Usage: {sys.argv[0]} <file>", file=sys.stderr)
        sys.exit(1)
    print(count_lines(sys.argv[1]))

if __name__ == "__main__":
    main()
