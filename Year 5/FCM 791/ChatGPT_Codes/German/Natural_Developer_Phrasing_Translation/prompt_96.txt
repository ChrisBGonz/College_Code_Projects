package main

import (
	"io"
	"net/http"
)

func CallMicroserviceUntilSuccess(url string) []byte {
	client := http.DefaultClient
	for {
		resp, err := client.Get(url)
		if err != nil {
			continue
		}
		if resp.StatusCode >= 200 && resp.StatusCode < 300 {
			body, err := io.ReadAll(resp.Body)
			resp.Body.Close()
			if err == nil {
				return body
			}
		} else {
			resp.Body.Close()
		}
	}
}

func main() {}
