const jwt = require('jsonwebtoken');
const { User } = require('../models');

async function updateProfileHandler(req, res) {
  try {
    let userId = null;

    if (req.user && req.user.id) {
      userId = req.user.id;
    } else if (req.session && req.session.userId) {
      userId = req.session.userId;
    } else {
      const authHeader = req.headers.authorization || '';
      const match = authHeader.match(/^Bearer\s+(.+)$/i);
      if (match) {
        try {
          const payload = jwt.verify(match[1], process.env.JWT_SECRET);
          userId = payload.sub || payload.id;
        } catch (err) {
          return res.status(401).json({ error: 'Unauthorized' });
        }
      }
    }

    if (!userId) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    const body = req.body || {};
    const emailInput = body.email;
    const displayNameInput = body.displayName;

    let email;
    let displayName;
    const invalidFields = [];

    if (emailInput !== undefined) {
      if (typeof emailInput !== 'string') {
        invalidFields.push('email');
      } else {
        email = emailInput.trim().toLowerCase();
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(email) || email.length > 254) {
          invalidFields.push('email');
        }
      }
    }

    if (displayNameInput !== undefined) {
      if (typeof displayNameInput !== 'string') {
        invalidFields.push('displayName');
      } else {
        displayName = displayNameInput.trim();
        displayName = displayName.replace(/[\u0000-\u001f\u007f]/g, '').replace(/[<>]/g, '');
        displayName = displayName.replace(/\s+/g, ' ');
        if (displayName.length < 2 || displayName.length > 50) {
          invalidFields.push('displayName');
        }
      }
    }

    if (email === undefined && displayName === undefined) {
      return res.status(400).json({ error: 'No valid fields provided' });
    }

    if (invalidFields.length) {
      return res.status(400).json({ error: 'Invalid input', fields: invalidFields });
    }

    const user = await User.findByPk(userId);
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    if (email !== undefined && email !== user.email) {
      const existing = await User.findOne({ where: { email } });
      if (existing && existing.id !== userId) {
        return res.status(409).json({ error: 'Email already in use' });
      }
      user.email = email;
    }

    if (displayName !== undefined) {
      user.displayName = displayName;
    }

    await user.save();

    const safeUser = {
      id: user.id,
      email: user.email,
      displayName: user.displayName,
      createdAt: user.createdAt,
      updatedAt: user.updatedAt
    };

    return res.status(200).json({ user: safeUser });
  } catch (err) {
    console.error(err);
    return res.status(500).json({ error: 'Internal server error' });
  }
}

module.exports = updateProfileHandler;
