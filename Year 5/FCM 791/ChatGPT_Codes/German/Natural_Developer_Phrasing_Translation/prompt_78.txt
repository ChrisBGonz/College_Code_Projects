package config

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strings"
)

func LoadConfigFile(path string, target interface{}) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	return ApplyConfigBytes(data, target)
}

func ApplyConfigBytes(data []byte, target interface{}) error {
	var m map[string]interface{}
	if err := json.Unmarshal(data, &m); err != nil {
		return err
	}
	return ApplyConfigMap(m, target)
}

func ApplyConfigMap(m map[string]interface{}, target interface{}) error {
	v := reflect.ValueOf(target)
	if v.Kind() != reflect.Ptr || v.IsNil() {
		return fmt.Errorf("target must be non-nil pointer to struct")
	}
	v = v.Elem()
	if v.Kind() != reflect.Struct {
		return fmt.Errorf("target must be pointer to struct")
	}
	t := v.Type()
	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		if field.PkgPath != "" {
			continue
		}
		key := field.Tag.Get("json")
		if key == "" || key == "-" {
			key = field.Name
		} else {
			if idx := strings.Index(key, ","); idx != -1 {
				key = key[:idx]
			}
			if key == "" {
				key = field.Name
			}
		}
		val, ok := m[key]
		if !ok {
			continue
		}
		if err := setValue(v.Field(i), val); err != nil {
			return fmt.Errorf("field %s: %w", field.Name, err)
		}
	}
	return nil
}

func setValue(fv reflect.Value, val interface{}) error {
	if !fv.CanSet() {
		return nil
	}
	if fv.Kind() == reflect.Ptr {
		if fv.IsNil() {
			fv.Set(reflect.New(fv.Type().Elem()))
		}
		return setValue(fv.Elem(), val)
	}
	v := reflect.ValueOf(val)
	if v.IsValid() {
		if v.Type().AssignableTo(fv.Type()) {
			fv.Set(v)
			return nil
		}
		if v.Type().ConvertibleTo(fv.Type()) {
			fv.Set(v.Convert(fv.Type()))
			return nil
		}
	}
	b, err := json.Marshal(val)
	if err != nil {
		return err
	}
	tmp := reflect.New(fv.Type()).Interface()
	if err := json.Unmarshal(b, tmp); err != nil {
		return err
	}
	fv.Set(reflect.ValueOf(tmp).Elem())
	return nil
}
