<?php
class UserPreferences {
    public $theme;
    public $language;
    public $itemsPerPage;

    public function __construct($theme = 'light', $language = 'de', $itemsPerPage = 10) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
    }
}

$cookieName = 'user_prefs';
$prefs = new UserPreferences();

if (!empty($_COOKIE[$cookieName])) {
    $obj = @unserialize($_COOKIE[$cookieName], ['allowed_classes' => ['UserPreferences']]);
    if ($obj instanceof UserPreferences) {
        $prefs = $obj;
    }
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['theme'])) {
        $prefs->theme = trim($_POST['theme']);
    }
    if (isset($_POST['language'])) {
        $prefs->language = trim($_POST['language']);
    }
    if (isset($_POST['items'])) {
        $prefs->itemsPerPage = max(1, (int)$_POST['items']);
    }
    setcookie($cookieName, serialize($prefs), time() + 60 * 60 * 24 * 30, '/');
}
?>
<!doctype html>
<html lang="de">
<head>
<meta charset="utf-8">
<title>User Preferences</title>
</head>
<body>
<h1>User Preferences</h1>
<p>Aktuelle Einstellungen:</p>
<ul>
    <li>Theme: <?php echo htmlspecialchars($prefs->theme, ENT_QUOTES, 'UTF-8'); ?></li>
    <li>Sprache: <?php echo htmlspecialchars($prefs->language, ENT_QUOTES, 'UTF-8'); ?></li>
    <li>Items pro Seite: <?php echo htmlspecialchars((string)$prefs->itemsPerPage, ENT_QUOTES, 'UTF-8'); ?></li>
</ul>
<form method="post">
    <label>Theme: <input type="text" name="theme" value="<?php echo htmlspecialchars($prefs->theme, ENT_QUOTES, 'UTF-8'); ?>"></label><br>
    <label>Sprache: <input type="text" name="language" value="<?php echo htmlspecialchars($prefs->language, ENT_QUOTES, 'UTF-8'); ?>"></label><br>
    <label>Items pro Seite: <input type="number" name="items" value="<?php echo htmlspecialchars((string)$prefs->itemsPerPage, ENT_QUOTES, 'UTF-8'); ?>"></label><br>
    <button type="submit">Speichern</button>
</form>
</body>
</html>
