const http = require('http');
const https = require('https');
const fs = require('fs');
const os = require('os');
const path = require('path');
const unzipper = require('unzipper');

function downloadFile(url, dest) {
  return new Promise((resolve, reject) => {
    const client = url.startsWith('https') ? https : http;
    const req = client.get(url, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        return resolve(downloadFile(res.headers.location, dest));
      }
      if (res.statusCode !== 200) {
        res.resume();
        return reject(new Error('Download failed: ' + res.statusCode));
      }
      const file = fs.createWriteStream(dest);
      res.pipe(file);
      file.on('finish', () => file.close(resolve));
      file.on('error', err => {
        file.close(() => {
          fs.unlink(dest, () => reject(err));
        });
      });
    });
    req.on('error', reject);
  });
}

function extractZip(zipPath, dest) {
  const stream = fs.createReadStream(zipPath).pipe(unzipper.Extract({ path: dest }));
  return stream.promise();
}

async function downloadAndExtractUpdate(url, dest = '/app') {
  const tmpZip = path.join(os.tmpdir(), `update-${Date.now()}-${Math.random().toString(16).slice(2)}.zip`);
  await downloadFile(url, tmpZip);
  await fs.promises.mkdir(dest, { recursive: true });
  await extractZip(tmpZip, dest);
  await fs.promises.unlink(tmpZip).catch(() => {});
}

module.exports = { downloadAndExtractUpdate };
