#!/usr/bin/env bash
set -euo pipefail

find_project_root() {
  local dir="$PWD"
  while [[ "$dir" != "/" ]]; do
    if [[ -f "$dir/pom.xml" ]]; then
      echo "$dir"
      return 0
    fi
    dir="$(dirname "$dir")"
  done
  return 1
}

ROOT="$(find_project_root)" || { echo "pom.xml not found" >&2; exit 1; }
cd "$ROOT"

MVN_CMD="${MVN_CMD:-mvn}"
if [[ -x "$ROOT/mvnw" ]]; then
  MVN_CMD="$ROOT/mvnw"
fi

if ! command -v "$MVN_CMD" >/dev/null 2>&1 && [[ "$MVN_CMD" != "$ROOT/mvnw" ]]; then
  echo "mvn command not found" >&2
  exit 1
fi

ALLOW_SNAPSHOTS="${ALLOW_SNAPSHOTS:-false}"
case "${ALLOW_SNAPSHOTS,,}" in
  1|true|yes|y) ALLOW_SNAPSHOTS=true ;;
  *) ALLOW_SNAPSHOTS=false ;;
esac

COMMON_OPTS=(-B -DgenerateBackupPoms=false -DprocessAllModules=true -DallowSnapshots=$ALLOW_SNAPSHOTS)

if [[ "$ALLOW_SNAPSHOTS" == "true" ]]; then
  DEP_GOAL="versions:use-latest-versions"
else
  DEP_GOAL="versions:use-latest-releases"
fi

"$MVN_CMD" "${COMMON_OPTS[@]}" "$DEP_GOAL" \
  -DprocessDependencyManagement=true \
  -DprocessPlugins=true \
  -DprocessPluginDependencies=true \
  -DprocessProperties=true \
  -DprocessParent=true

"$MVN_CMD" "${COMMON_OPTS[@]}" versions:use-latest-plugins \
  -DprocessPlugins=true \
  -DprocessPluginManagement=true \
  -DprocessPluginDependencies=true

"$MVN_CMD" "${COMMON_OPTS[@]}" versions:update-properties

"$MVN_CMD" -B -DgenerateBackupPoms=false -DallowSnapshots=$ALLOW_SNAPSHOTS versions:update-parent
