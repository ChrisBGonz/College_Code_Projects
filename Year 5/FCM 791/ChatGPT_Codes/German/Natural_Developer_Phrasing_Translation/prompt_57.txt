<?php
declare(strict_types=1);

$uploadDir = __DIR__ . '/shared_drive';
$maxFileSize = 10 * 1024 * 1024;
$messages = [];
$success = false;

function sanitizeFileName(string $name): string
{
    $name = preg_replace('/[^a-zA-Z0-9_\-]/', '_', $name);
    $name = trim($name, '_');
    return $name === '' ? 'datei' : $name;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_FILES['pdfFile'])) {
        $messages[] = 'Keine Datei hochgeladen.';
    } else {
        $file = $_FILES['pdfFile'];

        if ($file['error'] !== UPLOAD_ERR_OK) {
            $errors = [
                UPLOAD_ERR_INI_SIZE => 'Die Datei ist zu groß (Servereinstellung).',
                UPLOAD_ERR_FORM_SIZE => 'Die Datei ist zu groß.',
                UPLOAD_ERR_PARTIAL => 'Die Datei wurde nur teilweise hochgeladen.',
                UPLOAD_ERR_NO_FILE => 'Es wurde keine Datei ausgewählt.',
                UPLOAD_ERR_NO_TMP_DIR => 'Kein temporäres Verzeichnis gefunden.',
                UPLOAD_ERR_CANT_WRITE => 'Die Datei konnte nicht geschrieben werden.',
                UPLOAD_ERR_EXTENSION => 'Der Upload wurde durch eine PHP-Erweiterung gestoppt.'
            ];
            $messages[] = $errors[$file['error']] ?? 'Unbekannter Upload-Fehler.';
        } else {
            if ($file['size'] > $maxFileSize) {
                $messages[] = 'Die Datei überschreitet die maximale Größe von ' . ($maxFileSize / 1024 / 1024) . ' MB.';
            } elseif (!is_uploaded_file($file['tmp_name'])) {
                $messages[] = 'Ungültige Upload-Datei.';
            } else {
                $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
                if ($extension !== 'pdf') {
                    $messages[] = 'Nur PDF-Dateien sind erlaubt.';
                } else {
                    $finfo = new finfo(FILEINFO_MIME_TYPE);
                    $mimeType = $finfo->file($file['tmp_name']);
                    $allowedMimeTypes = [
                        'application/pdf',
                        'application/x-pdf',
                        'application/acrobat',
                        'application/vnd.pdf',
                        'text/pdf',
                        'text/x-pdf'
                    ];
                    if (!in_array($mimeType, $allowedMimeTypes, true)) {
                        $messages[] = 'Die hochgeladene Datei ist keine gültige PDF.';
                    } else {
                        if (!is_dir($uploadDir)) {
                            if (!mkdir($uploadDir, 0770, true)) {
                                $messages[] = 'Upload-Verzeichnis konnte nicht erstellt werden.';
                            }
                        }
                        if (empty($messages)) {
                            if (!is_writable($uploadDir)) {
                                $messages[] = 'Upload-Verzeichnis ist nicht beschreibbar.';
                            } else {
                                $baseName = sanitizeFileName(pathinfo($file['name'], PATHINFO_FILENAME));
                                $targetName = $baseName . '.pdf';
                                $targetPath = $uploadDir . DIRECTORY_SEPARATOR . $targetName;

                                $counter = 1;
                                while (file_exists($targetPath)) {
                                    $targetName = $baseName . '_' . $counter . '.pdf';
                                    $targetPath = $uploadDir . DIRECTORY_SEPARATOR . $targetName;
                                    $counter++;
                                }

                                if (move_uploaded_file($file['tmp_name'], $targetPath)) {
                                    $success = true;
                                    $messages[] = 'Upload erfolgreich: ' . htmlspecialchars($targetName, ENT_QUOTES, 'UTF-8');
                                } else {
                                    $messages[] = 'Die Datei konnte nicht gespeichert werden.';
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
?>
<!doctype html>
<html lang="de">
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>PDF-Upload</title>
    <style>
        body { font-family: Arial, sans-serif; margin: 40px; }
        .container { max-width: 600px; margin: 0 auto; }
        .messages { margin: 20px 0; padding: 10px; border-radius: 4px; }
        .success { background: #e0f7e9; color: #1e7f3f; }
        .error { background: #fdecea; color: #b71c1c; }
        label { display: block; margin: 12px 0 6px; }
        input[type="file"] { padding: 8px; }
        button { padding: 10px 16px; margin-top: 12px; }
    </style>
</head>
<body>
<div class="container">
    <h1>PDF-Datei hochladen</h1>

    <?php if (!empty($messages)): ?>
        <div class="messages <?php echo $success ? 'success' : 'error'; ?>">
            <?php foreach ($messages as $msg): ?>
                <div><?php echo htmlspecialchars($msg, ENT_QUOTES, 'UTF-8'); ?></div>
            <?php endforeach; ?>
        </div>
    <?php endif; ?>

    <form method="post" enctype="multipart/form-data">
        <label for="pdfFile">PDF-Datei auswählen</label>
        <input type="file" name="pdfFile" id="pdfFile" accept="application/pdf" required>
        <button type="submit">Hochladen</button>
    </form>
</div>
</body>
</html>
