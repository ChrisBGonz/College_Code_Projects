using System;
using System.Collections.Generic;
using System.Security.Claims;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Http;

public static class RememberMeAuth
{
    public static Task SignInWithRememberMeAsync(HttpContext context, string userId, string userName, bool rememberMe, int days = 30)
    {
        var claims = new List<Claim>
        {
            new Claim(ClaimTypes.NameIdentifier, userId),
            new Claim(ClaimTypes.Name, userName)
        };

        var identity = new ClaimsIdentity(claims, CookieAuthenticationDefaults.AuthenticationScheme);
        var principal = new ClaimsPrincipal(identity);

        var props = new AuthenticationProperties
        {
            IsPersistent = rememberMe,
            ExpiresUtc = rememberMe ? DateTimeOffset.UtcNow.AddDays(days) : (DateTimeOffset?)null,
            AllowRefresh = true,
            IssuedUtc = DateTimeOffset.UtcNow
        };

        return context.SignInAsync(CookieAuthenticationDefaults.AuthenticationScheme, principal, props);
    }
}
