from flask import Flask, request, render_template_string
import psycopg2
import os

app = Flask(__name__)

DB_CONFIG = {
    "host": os.environ.get("DB_HOST", "localhost"),
    "port": int(os.environ.get("DB_PORT", "5432")),
    "dbname": os.environ.get("DB_NAME", "postgres"),
    "user": os.environ.get("DB_USER", "postgres"),
    "password": os.environ.get("DB_PASSWORD", "")
}

FORM_HTML = """
<!doctype html>
<title>User Check</title>
<h1>Check User</h1>
<form method="post">
  <input type="text" name="username" placeholder="Username" required>
  <button type="submit">Check</button>
</form>
{% if checked %}
<p>User "{{ username }}" {% if exists %}exists{% else %}does not exist{% endif %}.</p>
{% endif %}
"""

def user_exists(username):
    conn = psycopg2.connect(**DB_CONFIG)
    try:
        with conn.cursor() as cur:
            cur.execute("SELECT 1 FROM users WHERE username = %s LIMIT 1", (username,))
            return cur.fetchone() is not None
    finally:
        conn.close()

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        exists = user_exists(username) if username else False
        return render_template_string(FORM_HTML, exists=exists, username=username, checked=True)
    return render_template_string(FORM_HTML, exists=False, username="", checked=False)

if __name__ == "__main__":
    app.run()
