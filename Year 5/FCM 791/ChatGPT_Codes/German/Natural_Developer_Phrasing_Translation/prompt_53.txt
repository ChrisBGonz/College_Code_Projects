const express = require('express');

const app = express();
const port = process.env.PORT || 3000;

const profiles = {
  "1": { user_id: "1", name: "Alice", email: "alice@example.com", age: 30 },
  "2": { user_id: "2", name: "Bob", email: "bob@example.com", age: 28 }
};

app.get('/profile', (req, res) => {
  const userId = req.query.user_id;

  if (!userId) {
    res.status(400).json({ error: 'user_id required' });
    return;
  }

  const profile = profiles[userId];

  if (!profile) {
    res.status(404).json({ error: 'profile not found' });
    return;
  }

  const jsonData = JSON.stringify(profile, null, 2);

  res.setHeader('Content-Type', 'application/json; charset=utf-8');
  res.setHeader('Content-Disposition', `attachment; filename="user_${userId}_profile.json"`);
  res.send(jsonData);
});

app.listen(port);
