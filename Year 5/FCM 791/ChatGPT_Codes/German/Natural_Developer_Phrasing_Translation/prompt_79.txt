#!/usr/bin/env python3
import sys
import os
import hashlib

def parse_manifest(path):
    entries = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("#"):
                continue
            parts = line.split()
            if len(parts) < 2:
                continue
            hash_val = parts[0]
            rel_path = " ".join(parts[1:])
            if rel_path.startswith("*") or rel_path.startswith("?"):
                rel_path = rel_path[1:]
            entries.append((hash_val, rel_path))
    return entries

def file_hash(path, algo="sha256"):
    h = hashlib.new(algo)
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()

def main():
    manifest_path = sys.argv[1] if len(sys.argv) > 1 else "manifest.txt"
    if not os.path.isfile(manifest_path):
        sys.stderr.write(f"Manifest not found: {manifest_path}\n")
        return 2
    entries = parse_manifest(manifest_path)
    base_dir = os.path.dirname(os.path.abspath(manifest_path))
    all_ok = True
    for expected, rel in entries:
        file_path = rel
        if not os.path.isabs(file_path):
            file_path = os.path.join(base_dir, rel)
        if not os.path.exists(file_path):
            print(f"MISSING {rel}")
            all_ok = False
            continue
        try:
            actual = file_hash(file_path)
        except Exception as e:
            print(f"ERROR {rel} {e}")
            all_ok = False
            continue
        if actual.lower() == expected.lower():
            print(f"OK {rel}")
        else:
            print(f"FAIL {rel} expected {expected} got {actual}")
            all_ok = False
    return 0 if all_ok else 1

if __name__ == "__main__":
    sys.exit(main())
