import java.sql.Connection;
import java.sql.SQLException;
import java.sql.Statement;
import java.util.logging.Logger;

public class DatabaseService {
    private static final Logger LOGGER = Logger.getLogger(DatabaseService.class.getName());

    public void execute(Connection connection, String sql) throws CustomDatabaseException {
        try (Statement stmt = connection.createStatement()) {
            stmt.execute(sql);
        } catch (SQLException e) {
            LOGGER.severe(e.getMessage());
            throw new CustomDatabaseException("Fehler bei der Datenbankoperation", e);
        }
    }

    public static class CustomDatabaseException extends Exception {
        public CustomDatabaseException(String message, Throwable cause) {
            super(message, cause);
        }
    }
}
