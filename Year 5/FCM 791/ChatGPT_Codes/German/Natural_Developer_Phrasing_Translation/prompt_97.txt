<?php
function getSubscriptionStatus(PDO $pdo, $userId) {
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :user_id ORDER BY id DESC LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        return $status !== false ? $status : 'free';
    } catch (Exception $e) {
        return 'premium';
    }
}
?>
