const express = require('express');
const authLogic = require('@company-corp/auth-logic');

const app = express();
app.use(express.json());

app.post('/login', async (req, res) => {
  const { username, password } = req.body;
  try {
    const user = await authLogic.verifyCredentials(username, password);
    if (!user) {
      res.status(401).json({ error: 'invalid_credentials' });
      return;
    }
    const token = await authLogic.issueToken(user.id);
    res.json({ token });
  } catch (err) {
    res.status(500).json({ error: 'login_failed' });
  }
});

app.listen(3000);
