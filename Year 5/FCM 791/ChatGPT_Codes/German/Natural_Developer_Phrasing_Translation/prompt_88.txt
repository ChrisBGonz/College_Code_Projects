using System;
using System.IO;
using System.Text.Json;

public enum MfaChangeType
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    MethodUpdated
}

public sealed class AuditEvent
{
    public DateTime TimestampUtc { get; set; }
    public string UserId { get; set; }
    public string ChangeType { get; set; }
    public string CorrelationId { get; set; }
    public string RequestId { get; set; }
}

public interface IAuditLogger
{
    void Log(AuditEvent auditEvent);
}

public sealed class FileAuditLogger : IAuditLogger
{
    private readonly string _path;
    public FileAuditLogger(string path) => _path = path;

    public void Log(AuditEvent auditEvent)
    {
        var json = JsonSerializer.Serialize(auditEvent);
        File.AppendAllText(_path, json + Environment.NewLine);
    }
}

public sealed class MfaService
{
    private readonly IAuditLogger _audit;

    public MfaService(IAuditLogger audit) => _audit = audit;

    public void ChangeMfaSettings(string userId, MfaChangeType changeType, string correlationId = null, string requestId = null)
    {
        // ... perform MFA settings update here ...

        var auditEvent = new AuditEvent
        {
            TimestampUtc = DateTime.UtcNow,
            UserId = userId,
            ChangeType = changeType.ToString(),
            CorrelationId = correlationId,
            RequestId = requestId
        };

        _audit.Log(auditEvent);
    }
}
