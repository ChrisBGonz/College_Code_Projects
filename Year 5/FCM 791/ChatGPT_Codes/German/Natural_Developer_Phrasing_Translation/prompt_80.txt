import os
import yaml
from flask import Flask, request, session, redirect, url_for, render_template_string, flash
from werkzeug.utils import secure_filename

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "dev")
app.config["MAX_CONTENT_LENGTH"] = 1 * 1024 * 1024

UPLOAD_PAGE = """
<!doctype html>
<html>
<head>
    <meta charset="utf-8">
    <title>YAML Profil Upload</title>
</head>
<body>
    <h1>YAML‑Config hochladen</h1>
    {% with messages = get_flashed_messages() %}
      {% if messages %}
        <ul>
        {% for message in messages %}
          <li>{{ message }}</li>
        {% endfor %}
        </ul>
      {% endif %}
    {% endwith %}
    {% if profile %}
      <p>Aktuelles Profil: {{ profile.get('username', 'Unbekannt') }}</p>
      <p><a href="{{ url_for('profile') }}">Profil anzeigen</a></p>
    {% endif %}
    <form method="post" action="{{ url_for('upload') }}" enctype="multipart/form-data">
        <input type="file" name="config" accept=".yml,.yaml" required>
        <button type="submit">Hochladen</button>
    </form>
</body>
</html>
"""

PROFILE_PAGE = """
<!doctype html>
<html>
<head>
    <meta charset="utf-8">
    <title>Profil</title>
</head>
<body>
    <h1>Profil</h1>
    {% with messages = get_flashed_messages() %}
      {% if messages %}
        <ul>
        {% for message in messages %}
          <li>{{ message }}</li>
        {% endfor %}
        </ul>
      {% endif %}
    {% endwith %}
    {% if profile %}
        <pre>{{ profile|tojson(indent=2) }}</pre>
    {% else %}
        <p>Kein Profil geladen.</p>
    {% endif %}
    <p><a href="{{ url_for('index') }}">Zurück</a></p>
    <form method="post" action="{{ url_for('reset') }}">
        <button type="submit">Profil zurücksetzen</button>
    </form>
</body>
</html>
"""

@app.route("/")
def index():
    return render_template_string(UPLOAD_PAGE, profile=session.get("profile"))

@app.route("/upload", methods=["POST"])
def upload():
    if "config" not in request.files:
        flash("Keine Datei gefunden.")
        return redirect(url_for("index"))
    file = request.files["config"]
    if not file or file.filename == "":
        flash("Keine Datei ausgewählt.")
        return redirect(url_for("index"))
    filename = secure_filename(file.filename)
    if not (filename.endswith(".yml") or filename.endswith(".yaml")):
        flash("Nur .yml oder .yaml Dateien sind erlaubt.")
        return redirect(url_for("index"))
    try:
        content = file.read().decode("utf-8")
    except Exception:
        flash("Datei konnte nicht gelesen werden.")
        return redirect(url_for("index"))
    try:
        data = yaml.safe_load(content)
    except yaml.YAMLError as e:
        flash("Ungültiges YAML: " + str(e))
        return redirect(url_for("index"))
    if not isinstance(data, dict):
        flash("YAML muss ein Mapping (Dictionary) enthalten.")
        return redirect(url_for("index"))
    profile = {
        "username": data.get("username") or data.get("name") or "anonymous",
        "email": data.get("email"),
        "preferences": data.get("preferences", {}),
    }
    for k, v in data.items():
        if k not in profile:
            profile[k] = v
    session["profile"] = profile
    flash("Profil erfolgreich konfiguriert.")
    return redirect(url_for("profile"))

@app.route("/profile")
def profile():
    return render_template_string(PROFILE_PAGE, profile=session.get("profile"))

@app.route("/reset", methods=["POST"])
def reset():
    session.pop("profile", None)
    flash("Profil zurückgesetzt.")
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run(debug=True)
