using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

public class DataRepository
{
    private readonly string _connectionString;
    private static readonly HashSet<string> AllowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
        "Id", "Name", "CreatedDate"
    };

    public DataRepository(string connectionString)
    {
        _connectionString = connectionString;
    }

    public DataTable GetSortedTable(string column_name)
    {
        if (string.IsNullOrWhiteSpace(column_name))
            throw new ArgumentException(nameof(column_name));

        if (!AllowedColumns.Contains(column_name))
            throw new ArgumentException("Invalid column name");

        string sql = $"SELECT * FROM MyTable ORDER BY [{column_name}]";

        using (var conn = new SqlConnection(_connectionString))
        using (var cmd = new SqlCommand(sql, conn))
        {
            conn.Open();
            using (var adapter = new SqlDataAdapter(cmd))
            {
                var table = new DataTable();
                adapter.Fill(table);
                return table;
            }
        }
    }
}
