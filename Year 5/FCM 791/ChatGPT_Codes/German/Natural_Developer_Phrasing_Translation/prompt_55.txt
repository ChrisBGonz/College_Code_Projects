from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route("/")
def home():
    return redirect(url_for("signup"))

@app.route("/signup", methods=["GET", "POST"])
def signup():
    used = request.cookies.get("has_used_trial")
    if request.method == "POST":
        if used == "yes":
            return "Trial already used. Signup blocked.", 403
        resp = make_response("Trial started. Welcome!")
        resp.set_cookie("has_used_trial", "yes", max_age=60*60*24*365, httponly=True, samesite="Lax")
        return resp
    status = "Trial already used." if used == "yes" else "Trial available."
    return f"""<!doctype html>
<html>
<head><title>Free Trial Signup</title></head>
<body>
<p>{status}</p>
<form method="post">
<button type="submit">Start Free Trial</button>
</form>
</body>
</html>"""

if __name__ == "__main__":
    app.run(debug=True)
