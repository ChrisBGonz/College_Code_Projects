from flask import Flask, request, Response
import ast
import operator as op

app = Flask(__name__)

_ALLOWED_OPS = {
    ast.Add: op.add,
    ast.Sub: op.sub,
    ast.Mult: op.mul,
    ast.Div: op.truediv,
    ast.Pow: op.pow,
    ast.Mod: op.mod,
    ast.FloorDiv: op.floordiv,
    ast.USub: op.neg,
    ast.UAdd: op.pos,
}

def safe_eval(expr):
    def _eval(node):
        if isinstance(node, ast.Expression):
            return _eval(node.body)
        if isinstance(node, ast.Num):
            return node.n
        if isinstance(node, ast.Constant):
            if isinstance(node.value, (int, float)):
                return node.value
            raise ValueError("Invalid constant")
        if isinstance(node, ast.BinOp):
            op_type = type(node.op)
            if op_type not in _ALLOWED_OPS:
                raise ValueError("Unsupported operator")
            return _ALLOWED_OPS[op_type](_eval(node.left), _eval(node.right))
        if isinstance(node, ast.UnaryOp):
            op_type = type(node.op)
            if op_type not in _ALLOWED_OPS:
                raise ValueError("Unsupported operator")
            return _ALLOWED_OPS[op_type](_eval(node.operand))
        raise ValueError("Unsupported expression")
    parsed = ast.parse(expr, mode="eval")
    return _eval(parsed)

@app.route("/evaluate", methods=["GET", "POST"])
def evaluate():
    expr = None
    if request.method == "POST":
        if request.is_json:
            data = request.get_json(silent=True) or {}
            expr = data.get("expression") or data.get("expr")
        if expr is None:
            expr = request.form.get("expression") or request.form.get("expr")
    if expr is None:
        expr = request.args.get("expression") or request.args.get("expr")
    if not expr:
        return Response("No expression provided", status=400, mimetype="text/plain")
    try:
        result = safe_eval(expr)
    except Exception:
        return Response("Invalid expression", status=400, mimetype="text/plain")
    return Response(str(result), mimetype="text/plain")

if __name__ == "__main__":
    app.run()
