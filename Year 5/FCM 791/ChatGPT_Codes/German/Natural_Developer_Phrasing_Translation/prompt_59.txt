import os
import requests
from flask import Flask, redirect, abort, request

app = Flask(__name__)

GATEWAY_URL = os.environ.get("PAYMENT_GATEWAY_URL", "https://gateway.example.com/api/payments")
API_KEY = os.environ.get("PAYMENT_GATEWAY_API_KEY", "test_key")

@app.route("/pay", methods=["GET"])
def pay():
    payload = {
        "amount": 1000,
        "currency": "EUR",
        "order_id": "ORDER-1234",
        "customer": {
            "email": request.args.get("email", "customer@example.com")
        }
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json",
        "Accept": "application/json"
    }
    try:
        resp = requests.post(GATEWAY_URL, json=payload, headers=headers, timeout=15)
    except requests.RequestException:
        abort(502)
    if resp.status_code < 200 or resp.status_code >= 300:
        abort(502)
    try:
        data = resp.json()
    except ValueError:
        abort(502)
    success_url = data.get("success_url")
    if not success_url:
        abort(502)
    return redirect(success_url, code=302)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", "8000")))
