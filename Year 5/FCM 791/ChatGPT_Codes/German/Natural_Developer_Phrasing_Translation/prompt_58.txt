function initMultiStepRegistration(formSelector, options) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return null;
  options = options || {};
  var storage = options.storage || window.localStorage;
  var key = options.storageKey || 'multi-step-form:' + (form.getAttribute('data-storage-key') || form.id || form.name || 'default');
  var stepSelector = options.stepSelector || '[data-step]';
  var nextSelector = options.nextSelector || '[data-step-next]';
  var prevSelector = options.prevSelector || '[data-step-prev]';
  var gotoSelector = options.gotoSelector || '[data-step-goto]';
  var activeClass = options.activeClass || 'active';
  var excludeTypes = options.excludeTypes || ['file'];
  var steps = Array.prototype.slice.call(form.querySelectorAll(stepSelector));
  var currentStep = 0;
  var debounce = typeof options.debounce === 'number' ? options.debounce : 0;
  var saveTimer = null;

  function safeGet() {
    if (!storage) return null;
    var raw;
    try {
      raw = storage.getItem(key);
    } catch (e) {
      return null;
    }
    if (!raw) return null;
    try {
      return JSON.parse(raw);
    } catch (e) {
      safeRemove();
      return null;
    }
  }

  function safeSet(data) {
    if (!storage) return;
    try {
      storage.setItem(key, JSON.stringify(data));
    } catch (e) {}
  }

  function safeRemove() {
    if (!storage) return;
    try {
      storage.removeItem(key);
    } catch (e) {}
  }

  function collectValues() {
    var values = {};
    var els = form.elements;
    for (var i = 0; i < els.length; i++) {
      var el = els[i];
      if (!el.name || el.disabled) continue;
      if (el.closest && el.closest('[data-no-persist]')) continue;
      var type = (el.type || '').toLowerCase();
      if (excludeTypes.indexOf(type) !== -1) continue;
      if (type === 'checkbox') {
        if (!values[el.name]) values[el.name] = [];
        if (el.checked) values[el.name].push(el.value);
      } else if (type === 'radio') {
        if (el.checked) values[el.name] = el.value;
        else if (values[el.name] === undefined) values[el.name] = null;
      } else if (el.tagName.toLowerCase() === 'select' && el.multiple) {
        var arr = [];
        for (var j = 0; j < el.options.length; j++) {
          if (el.options[j].selected) arr.push(el.options[j].value);
        }
        values[el.name] = arr;
      } else {
        values[el.name] = el.value;
      }
    }
    return values;
  }

  function applyValues(values) {
    if (!values) return;
    var els = form.elements;
    for (var i = 0; i < els.length; i++) {
      var el = els[i];
      if (!el.name || el.disabled) continue;
      if (el.closest && el.closest('[data-no-persist]')) continue;
      var type = (el.type || '').toLowerCase();
      if (excludeTypes.indexOf(type) !== -1) continue;
      if (values[el.name] === undefined) continue;
      var val = values[el.name];
      if (type === 'checkbox') {
        if (Array.isArray(val)) {
          el.checked = val.indexOf(el.value) !== -1;
        } else if (typeof val === 'boolean') {
          el.checked = val;
        } else if (val != null) {
          el.checked = String(val) === String(el.value);
        } else {
          el.checked = false;
        }
      } else if (type === 'radio') {
        if (val == null) {
          el.checked = false;
        } else {
          el.checked = String(el.value) === String(val);
        }
      } else if (el.tagName.toLowerCase() === 'select' && el.multiple) {
        if (Array.isArray(val)) {
          for (var j = 0; j < el.options.length; j++) {
            el.options[j].selected = val.indexOf(el.options[j].value) !== -1;
          }
        }
      } else {
        el.value = val;
      }
    }
  }

  function persist() {
    var data = { values: collectValues() };
    if (steps.length) data.currentStep = currentStep;
    safeSet(data);
  }

  function schedulePersist() {
    if (saveTimer) clearTimeout(saveTimer);
    saveTimer = setTimeout(function () {
      saveTimer = null;
      persist();
    }, debounce);
  }

  function showStep(index, silent) {
    if (!steps.length) return;
    if (index < 0) index = 0;
    if (index >= steps.length) index = steps.length - 1;
    for (var i = 0; i < steps.length; i++) {
      var step = steps[i];
      var active = i === index;
      if (activeClass) {
        if (active) step.classList.add(activeClass);
        else step.classList.remove(activeClass);
      }
      step.hidden = !active;
      step.setAttribute('aria-hidden', active ? 'false' : 'true');
    }
    currentStep = index;
    if (!silent) schedulePersist();
  }

  function restore() {
    var data = safeGet();
    if (data && data.values) applyValues(data.values);
    if (steps.length) {
      var idx = data && typeof data.currentStep === 'number' ? data.currentStep : 0;
      showStep(idx, true);
    }
  }

  function clearState() {
    safeRemove();
  }

  function onFieldChange(e) {
    var target = e.target;
    if (!target || !target.name || target.disabled) return;
    if (target.closest && target.closest('[data-no-persist]')) return;
    var type = (target.type || '').toLowerCase();
    if (excludeTypes.indexOf(type) !== -1) return;
    schedulePersist();
  }

  function onClick(e) {
    var target = e.target;
    if (target && target.nodeType !== 1) target = target.parentElement;
    if (!target) return;
    var next = target.closest ? target.closest(nextSelector) : null;
    var prev = target.closest ? target.closest(prevSelector) : null;
    var goto = target.closest ? target.closest(gotoSelector) : null;
    if (next) {
      e.preventDefault();
      showStep(currentStep + 1);
    } else if (prev) {
      e.preventDefault();
      showStep(currentStep - 1);
    } else if (goto) {
      e.preventDefault();
      var idx = parseInt(goto.getAttribute('data-step-goto'), 10);
      if (!isNaN(idx)) showStep(idx);
    }
  }

  function onSubmit(e) {
    var handler = options.onSubmit;
    if (typeof handler === 'function') {
      var result = handler(e, form, clearState);
      if (result && typeof result.then === 'function') {
        result.then(function (success) {
          if (success !== false) clearState();
        });
      } else {
        if (result !== false && !e.defaultPrevented) clearState();
      }
    } else {
      if (!e.defaultPrevented) clearState();
    }
  }

  restore();
  form.addEventListener('input', onFieldChange);
  form.addEventListener('change', onFieldChange);
  form.addEventListener('click', onClick);
  form.addEventListener('submit', onSubmit);

  return {
    save: persist,
    clear: clearState,
    showStep: showStep,
    getState: safeGet
  };
}
