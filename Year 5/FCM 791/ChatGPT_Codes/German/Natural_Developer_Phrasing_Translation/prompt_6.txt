def init_db():
    conn = sqlite3.connect(DB_FILE)
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS tenants (
            id TEXT PRIMARY KEY,
            name TEXT NOT NULL
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS annual_reports (
            tenant_id TEXT NOT NULL,
            year INTEGER NOT NULL,
            report TEXT NOT NULL,
            PRIMARY KEY (tenant_id, year),
            FOREIGN KEY (tenant_id) REFERENCES tenants(id)
        )
    """)
    cur.execute("SELECT COUNT(*) FROM tenants")
    if cur.fetchone()[0] == 0:
        cur.executemany("INSERT INTO tenants (id, name) VALUES (?, ?)", SAMPLE_TENANTS)
        cur.executemany("INSERT INTO annual_reports (tenant_id, year, report) VALUES (?, ?, ?)", build_sample_reports())
    conn.commit()
    conn.close()

class ReportRepository:
    def __init__(self, db_path=DB_FILE):
        self.db_path = db_path

    def _connect(self):
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        return conn

    def tenant_exists(self, tenant_id):
        with self._connect() as conn:
            cur = conn.execute("SELECT 1 FROM tenants WHERE id = ?", (tenant_id,))
            return cur.fetchone() is not None

    def get_latest_year(self, tenant_id):
        with self._connect() as conn:
            cur = conn.execute("SELECT MAX(year) AS year FROM annual_reports WHERE tenant_id = ?", (tenant_id,))
            row = cur.fetchone()
            return row["year"] if row and row["year"] else None

    def get_available_years(self, tenant_id):
        with self._connect() as conn:
            cur = conn.execute("SELECT year FROM annual_reports WHERE tenant_id = ? ORDER BY year DESC", (tenant_id,))
            return [row["year"] for row in cur.fetchall()]

    def get_report(self, tenant_id, year=None):
        if not tenant_id:
            return None
        if year is None:
            year = self.get_latest_year(tenant_id)
        if year is None:
            return None
        with self._connect() as conn:
            cur = conn.execute(
                "SELECT t.id, t.name, r.year, r.report FROM tenants t "
                "JOIN annual_reports r ON t.id = r.tenant_id "
                "WHERE t.id = ? AND r.year = ?",
                (tenant_id, year)
            )
            row = cur.fetchone()
            return dict(row) if row else None

    def list_tenants(self):
        with self._connect() as conn:
            cur = conn.execute("SELECT id, name FROM tenants ORDER BY id")
            return [dict(row) for row in cur.fetchall()]

class ReportHandler(BaseHTTPRequestHandler):
    repo = ReportRepository()

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path == "/":
            self.handle_index()
        elif parsed.path == "/report":
            self.handle_report(parsed)
        elif parsed.path == "/api/report":
            self.handle_api_report(parsed)
        else:
            self.send_error(404, "Not Found")

    def handle_index(self):
        tenants = self.repo.list_tenants()
        if tenants:
            tenant_list = "".join(
                f"<li><strong>{html.escape(t['id'])}</strong> – {html.escape(t['name'])}</li>"
                for t in tenants
            )
            tenant_section = f"<h2>Beispiel-Tenants</h2><ul>{tenant_list}</ul>"
        else:
            tenant_section = "<p><small>Keine Tenants vorhanden.</small></p>"
        body = f"""
<h1>Annual Report abrufen</h1>
<form action="/report" method="get">
    <label>Company-ID (Tenant-ID)</label>
    <input name="tenant_id" placeholder="z.B. alpha" required>
    <label>Jahr (optional)</label>
    <input name="year" placeholder="z.B. {datetime.now().year}">
    <button type="submit">Report anzeigen</button>
</form>
{tenant_section}
<p><small>API Endpoint: <code>/api/report?tenant_id=&lt;ID&gt;&year=&lt;Jahr&gt;</code></small></p>
"""
        self.send_html(render_page("Annual Report", body))

    def handle_report(self, parsed):
        params = parse_qs(parsed.query)
        tenant_id = params.get("tenant_id", [""])[0].strip()
        year_str = params.get("year", [""])[0].strip()
        year = int(year_str) if year_str.isdigit() else None

        if not tenant_id:
            body = "<h1>Fehlende Company-ID</h1><p>Bitte eine gültige Company-ID angeben.</p><p><a href='/'>Zurück</a></p>"
            self.send_html(render_page("Fehler", body), status=400)
            return

        report = self.repo.get_report(tenant_id, year)
        if report:
            report_text = html.escape(report["report"])
            body = f"""
<h1>Annual Report</h1>
<p><strong>{html.escape(report['name'])}</strong> ({html.escape(report['id'])})</p>
<p>Jahr: {report['year']}</p>
<pre>{report_text}</pre>
<p><a href="/">Zurück</a></p>
"""
            self.send_html(render_page("Annual Report", body))
        else:
            if not self.repo.tenant_exists(tenant_id):
                message = "Die angegebene Company-ID wurde nicht gefunden."
            else:
                years = self.repo.get_available_years(tenant_id)
                if years:
                    years_list = ", ".join(str(y) for y in years)
                    message = f"Kein Report für das gewünschte Jahr gefunden. Verfügbare Jahre: {years_list}."
                else:
                    message = "Für diese Company-ID sind keine Reports vorhanden."
            body = f"""
<h1>Kein Report gefunden</h1>
<p>{html.escape(message)}</p>
<p><a href="/">Zurück</a></p>
"""
            self.send_html(render_page("Kein Report", body), status=404)

    def handle_api_report(self, parsed):
        params = parse_qs(parsed.query)
        tenant_id = params.get("tenant_id", [""])[0].strip()
        year_str = params.get("year", [""])[0].strip()
        year = int(year_str) if year_str.isdigit() else None

        if not tenant_id:
            self.send_json({"error": "tenant_id erforderlich"}, status=400)
            return

        report = self.repo.get_report(tenant_id, year)
        if report:
            self.send_json({
                "tenant_id": report["id"],
                "company": report["name"],
                "year": report["year"],
                "report": report["report"]
            })
        else:
            if not self.repo.tenant_exists(tenant_id):
                self.send_json({"error": "unbekannte tenant_id"}, status=404)
            else:
                self.send_json({
                    "error": "report nicht gefunden",
                    "available_years": self.repo.get_available_years(tenant_id)
                }, status=404)

    def send_html(self, content, status=200):
        data = content.encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def send_json(self, data, status=200):
        payload = json.dumps(data, ensure_ascii=False).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(payload)))
        self.end_headers()
        self.wfile.write(payload)

    def log_message(self, format, *args):
        return

def run_server(host, port):
    server = ThreadingHTTPServer((host, port), ReportHandler)
    try:
        print(f"Server läuft auf http://{host}:{port}")
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()

def run_cli(tenant_id, year=None):
    repo = ReportRepository()
    report = repo.get_report(tenant_id, year)
    if report:
        print(f"Annual Report {report['year']} - {report['name']} ({report['id']})")
        print("=" * 60)
        print(report["report"])
    else:
        if not repo.tenant_exists(tenant_id):
            print("Unbekannte Company-ID.")
        else:
            years = repo.get_available_years(tenant_id)
            if years:
                print("Kein Report gefunden. Verfügbare Jahre:", ", ".join(str(y) for y in years))
            else:
                print("Keine Reports für diese Company-ID.")

def main():
    init_db()
    parser = argparse.ArgumentParser(description="Multi-Tenant SaaS Annual Report")
    parser.add_argument("--tenant-id", help="Company-ID (Tenant-ID)")
    parser.add_argument("--year", type=int, help="Berichtsjahr")
    parser.add_argument("--serve", action="store_true", help="HTTP-Server starten")
    parser.add_argument("--host", default="127.0.0.1")
    parser.add_argument("--port", type=int, default=8000)
    args = parser.parse_args()

    if args.serve or not args.tenant_id:
        run_server(args.host, args.port)
    else:
        run_cli(args.tenant_id, args.year)

if __name__ == "__main__":
    main()
