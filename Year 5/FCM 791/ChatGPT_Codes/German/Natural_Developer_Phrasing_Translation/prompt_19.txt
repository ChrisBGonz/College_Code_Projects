from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

try:
    from sqlalchemy.exc import OperationalError
except Exception:
    class OperationalError(Exception):
        pass

class DatabaseConnectionError(Exception):
    pass

app = FastAPI()

async def db_connection_exception_handler(request: Request, exc: Exception):
    return JSONResponse(
        status_code=503,
        content={
            "error": "Database connection error",
            "detail": str(exc) or exc.__class__.__name__
        }
    )

app.add_exception_handler(DatabaseConnectionError, db_connection_exception_handler)
app.add_exception_handler(OperationalError, db_connection_exception_handler)

@app.get("/health")
async def health():
    return {"status": "ok"}

@app.get("/db-test")
async def db_test():
    raise DatabaseConnectionError("Could not connect to database")
