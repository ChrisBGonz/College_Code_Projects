class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    return head :forbidden unless current_user&.role == 'Manager'

    user = User.find(params[:id])
    return render(json: { error: 'User is not staff' }, status: :unprocessable_entity) unless user.role == 'Staff'

    if user.update(role: 'Manager')
      render json: { message: 'User promoted' }, status: :ok
    else
      render json: { errors: user.errors.full_messages }, status: :unprocessable_entity
    end
  end
end
