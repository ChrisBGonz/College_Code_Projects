<?php
session_start();

$adminEmail = 'admin@example.com';
$slackWebhook = 'https://hooks.slack.com/services/XXXXXXXXX/XXXXXXXXX/XXXXXXXXXXXXXXXXXXXXXXXX';

function isAuthorized(): bool {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

if (!isAuthorized()) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $uri = $_SERVER['REQUEST_URI'] ?? '';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? '';
    $time = date('c');

    $message = "Unauthorized access to /admin\nTime: $time\nIP: $ip\nURI: $uri\nUser-Agent: $ua";

    @mail($adminEmail, 'Administrator Alert', $message);

    if (!empty($slackWebhook)) {
        $payload = json_encode(['text' => "Administrator Alert: Unauthorized access to /admin\nTime: $time\nIP: $ip\nURI: $uri\nUser-Agent: $ua"]);
        $ch = curl_init($slackWebhook);
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
        curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_exec($ch);
        curl_close($ch);
    }

    http_response_code(403);
    echo 'Forbidden';
    exit;
}

echo 'Welcome to admin area';
