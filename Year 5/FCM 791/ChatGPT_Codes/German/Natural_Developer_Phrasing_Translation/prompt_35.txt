<?php
declare(strict_types=1);

function loadEncryptionKey(): string {
    $env = getenv('ENCRYPTION_KEY');
    if ($env === false || $env === '') {
        throw new RuntimeException('ENCRYPTION_KEY not set');
    }
    $key = base64_decode($env, true);
    if ($key !== false && strlen($key) === 32) {
        return $key;
    }
    if (strlen($env) === 64 && ctype_xdigit($env)) {
        $key = hex2bin($env);
        if ($key !== false && strlen($key) === 32) {
            return $key;
        }
    }
    if (strlen($env) === 32) {
        return $env;
    }
    throw new RuntimeException('Invalid ENCRYPTION_KEY length');
}

$key = loadEncryptionKey();

$ccNumber = trim((string)($_POST['credit_card_number'] ?? ''));
if ($ccNumber === '') {
    http_response_code(400);
    exit('credit_card_number missing');
}

$iv = random_bytes(12);
$tag = '';
$ciphertext = openssl_encrypt($ccNumber, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);
if ($ciphertext === false) {
    throw new RuntimeException('Encryption failed');
}

$dsn = getenv('DB_DSN') ?: 'mysql:host=localhost;dbname=test;charset=utf8mb4';
$user = getenv('DB_USER') ?: 'root';
$pass = getenv('DB_PASS') ?: '';

$pdo = new PDO($dsn, $user, $pass, [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_EMULATE_PREPARES => false,
]);

$sql = "INSERT INTO payments (credit_card_number, cc_iv, cc_tag) VALUES (:cc, :iv, :tag)";
$stmt = $pdo->prepare($sql);
$stmt->execute([
    ':cc' => base64_encode($ciphertext),
    ':iv' => base64_encode($iv),
    ':tag' => base64_encode($tag),
]);

echo "OK";
