<?php
function getSubscriptionStatus($userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare('SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1');
        $stmt->execute([':user_id' => $userId]);
        $status = $stmt->fetchColumn();
        return ($status === false || $status === null || $status === '') ? 'free' : $status;
    } catch (Throwable $e) {
        return 'premium';
    }
}
?>
