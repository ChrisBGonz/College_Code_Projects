using System;
using System.Security.Cryptography;
using System.Text;
using Microsoft.AspNetCore.Http;

public static class RememberMeCookie
{
    private const string CookieName = "REMEMBER_ME";
    private static readonly TimeSpan DefaultLifetime = TimeSpan.FromDays(30);

    public static void Set(HttpContext context, string userId, byte[] secretKey, TimeSpan? lifetime = null)
    {
        var expires = DateTimeOffset.UtcNow.Add(lifetime ?? DefaultLifetime);
        var payload = $"{userId}|{expires.ToUnixTimeSeconds()}";
        var signature = Sign(payload, secretKey);
        var value = Convert.ToBase64String(Encoding.UTF8.GetBytes($"{payload}|{signature}"));

        context.Response.Cookies.Append(CookieName, value, new CookieOptions
        {
            Expires = expires.UtcDateTime,
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            IsEssential = true,
            Path = "/"
        });
    }

    public static string Validate(HttpContext context, byte[] secretKey)
    {
        if (!context.Request.Cookies.TryGetValue(CookieName, out var value))
            return null;

        string data;
        try
        {
            data = Encoding.UTF8.GetString(Convert.FromBase64String(value));
        }
        catch
        {
            return null;
        }

        var parts = data.Split('|');
        if (parts.Length != 3) return null;

        var userId = parts[0];
        if (!long.TryParse(parts[1], out var expSeconds)) return null;

        var expires = DateTimeOffset.FromUnixTimeSeconds(expSeconds);
        if (expires <= DateTimeOffset.UtcNow) return null;

        var payload = $"{userId}|{parts[1]}";
        var expectedSig = Sign(payload, secretKey);
        if (!FixedTimeEquals(parts[2], expectedSig)) return null;

        return userId;
    }

    public static void Clear(HttpContext context)
    {
        context.Response.Cookies.Delete(CookieName);
    }

    private static string Sign(string data, byte[] key)
    {
        using var hmac = new HMACSHA256(key);
        var hash = hmac.ComputeHash(Encoding.UTF8.GetBytes(data));
        return Convert.ToBase64String(hash);
    }

    private static bool FixedTimeEquals(string a, string b)
    {
        if (a == null || b == null || a.Length != b.Length) return false;
        var diff = 0;
        for (int i = 0; i < a.Length; i++)
            diff |= a[i] ^ b[i];
        return diff == 0;
    }
}
