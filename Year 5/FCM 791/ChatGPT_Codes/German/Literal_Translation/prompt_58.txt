function initMultiStepRegistration(formSelector, storageKey) {
  var form = typeof formSelector === 'string' ? document.querySelector(formSelector) : formSelector;
  if (!form) return;
  storageKey = storageKey || 'registrationData';

  var steps = Array.prototype.slice.call(form.querySelectorAll('[data-step]'));
  var currentStep = 0;

  function loadData() {
    var stored = {};
    try { stored = JSON.parse(localStorage.getItem(storageKey)) || {}; } catch (e) {}
    Array.prototype.forEach.call(form.elements, function (el) {
      if (!el.name) return;
      if (stored.hasOwnProperty(el.name)) {
        if (el.type === 'checkbox') {
          if (Array.isArray(stored[el.name])) {
            el.checked = stored[el.name].indexOf(el.value || true) !== -1;
          } else {
            el.checked = !!stored[el.name];
          }
        } else if (el.type === 'radio') {
          el.checked = stored[el.name] == el.value;
        } else {
          el.value = stored[el.name];
        }
      }
    });
  }

  function collectData() {
    var data = {};
    Array.prototype.forEach.call(form.elements, function (el) {
      if (!el.name) return;
      if (el.type === 'checkbox') {
        if (!data[el.name]) data[el.name] = [];
        if (el.checked) data[el.name].push(el.value || true);
      } else if (el.type === 'radio') {
        if (el.checked) data[el.name] = el.value;
      } else {
        data[el.name] = el.value;
      }
    });
    return data;
  }

  function saveData() {
    var data = collectData();
    try { localStorage.setItem(storageKey, JSON.stringify(data)); } catch (e) {}
  }

  function showStep(index) {
    if (!steps.length) return;
    currentStep = Math.max(0, Math.min(index, steps.length - 1));
    steps.forEach(function (step, i) {
      step.style.display = i === currentStep ? '' : 'none';
    });
  }

  loadData();
  showStep(currentStep);

  form.addEventListener('input', saveData);
  form.addEventListener('change', saveData);

  form.addEventListener('click', function (e) {
    var target = e.target;
    if (target.matches('[data-next]')) {
      e.preventDefault();
      saveData();
      showStep(currentStep + 1);
    }
    if (target.matches('[data-prev]')) {
      e.preventDefault();
      saveData();
      showStep(currentStep - 1);
    }
  });

  form.addEventListener('submit', function () {
    saveData();
    try { localStorage.removeItem(storageKey); } catch (e) {}
  });
}
