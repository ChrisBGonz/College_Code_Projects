using System;
using System.IO;

public class User
{
    public string Id { get; set; }
    public string Username { get; set; }
    public MfaSettings Mfa { get; set; }
}

public class MfaSettings
{
    public bool Enabled { get; set; }
    public string Method { get; set; }
}

public interface IAuditLogger
{
    void LogMfaChange(User user, MfaSettings oldSettings, MfaSettings newSettings);
}

public class FileAuditLogger : IAuditLogger
{
    private readonly string _path;

    public FileAuditLogger(string path)
    {
        _path = path;
    }

    public void LogMfaChange(User user, MfaSettings oldSettings, MfaSettings newSettings)
    {
        var message = $"{DateTime.UtcNow:o} USER:{user.Id}({user.Username}) MFA-Änderung: " +
                      $"Enabled {oldSettings?.Enabled}->{newSettings?.Enabled}, " +
                      $"Method {oldSettings?.Method ?? "none"}->{newSettings?.Method ?? "none"}";
        File.AppendAllText(_path, message + Environment.NewLine);
    }
}

public class MfaService
{
    private readonly IAuditLogger _logger;

    public MfaService(IAuditLogger logger)
    {
        _logger = logger;
    }

    public void UpdateMfaSettings(User user, MfaSettings newSettings)
    {
        var oldSettings = user.Mfa;
        user.Mfa = newSettings;
        _logger.LogMfaChange(user, oldSettings, newSettings);
    }
}
