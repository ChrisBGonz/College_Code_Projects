package main

import (
	"crypto/subtle"
	"net/http"
	"net/url"
	"os"
	"strings"
)

const allowedRefererHost = "admin.internal.com"

func isAllowedReferer(r *http.Request) bool {
	ref := r.Referer()
	if ref == "" {
		return false
	}
	u, err := url.Parse(ref)
	if err != nil {
		return false
	}
	host := strings.Split(u.Host, ":")[0]
	return strings.EqualFold(host, allowedRefererHost)
}

func validAdminToken(r *http.Request) bool {
	token := r.Header.Get("X-Admin-Token")
	expected := os.Getenv("ADMIN_TOKEN")
	if expected == "" {
		return false
	}
	return subtle.ConstantTimeCompare([]byte(token), []byte(expected)) == 1
}

func AdminOnly(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if !isAllowedReferer(r) || !validAdminToken(r) {
			http.Error(w, "forbidden", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func adminDashboard(w http.ResponseWriter, r *http.Request) {
	w.Write([]byte("admin"))
}

func main() {
	http.Handle("/admin", AdminOnly(http.HandlerFunc(adminDashboard)))
	http.ListenAndServe(":8080", nil)
}
