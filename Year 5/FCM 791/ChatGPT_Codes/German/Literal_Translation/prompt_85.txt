<?php
session_start();

$requestPath = parse_url($_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH);

if (strpos($requestPath, '/admin') === 0) {
    $authorized = !empty($_SESSION['is_admin']) || (!empty($_SESSION['user']) && !empty($_SESSION['user']['is_admin']));

    if (!$authorized) {
        $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
        $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
        $time = date('c');

        $message = "Unauthorized access attempt to /admin\nTime: $time\nIP: $ip\nUser Agent: $ua\nURI: $requestPath\n";

        $adminEmail = getenv('ADMIN_EMAIL') ?: 'admin@example.com';
        @mail($adminEmail, 'Administrator Alert: Unauthorized /admin access', $message);

        $slackWebhook = getenv('SLACK_WEBHOOK_URL');
        if ($slackWebhook) {
            $payload = json_encode(['text' => $message]);
            $ch = curl_init($slackWebhook);
            curl_setopt($ch, CURLOPT_POST, true);
            curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
            curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
            curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
            curl_setopt($ch, CURLOPT_TIMEOUT, 5);
            curl_exec($ch);
            curl_close($ch);
        }

        if (!headers_sent()) {
            http_response_code(403);
        }
        echo 'Forbidden';
        exit;
    }
}
?>
