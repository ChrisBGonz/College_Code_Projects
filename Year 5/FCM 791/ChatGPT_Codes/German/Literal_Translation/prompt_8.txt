package config

import (
	"fmt"
	"os"
)

func ConfigPathFromEnv(envVar string) (string, error) {
	if envVar == "" {
		return "", fmt.Errorf("environment variable name is empty")
	}
	if val, ok := os.LookupEnv(envVar); ok && val != "" {
		return val, nil
	}
	return "", fmt.Errorf("environment variable %s not set", envVar)
}
