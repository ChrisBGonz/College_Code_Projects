import os
import json
import uuid
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs
from urllib.request import Request, urlopen

GATEWAY_API_URL = os.getenv('GATEWAY_API_URL')
GATEWAY_API_KEY = os.getenv('GATEWAY_API_KEY')
HOST = os.getenv('HOST', '0.0.0.0')
PORT = int(os.getenv('PORT', '8000'))

def create_payment_session(amount, currency, success_url, metadata=None):
    if not GATEWAY_API_URL:
        return {"id": str(uuid.uuid4()), "status": "simulated"}
    payload = {"amount": amount, "currency": currency, "success_url": success_url}
    if metadata is not None:
        payload["metadata"] = metadata
    data = json.dumps(payload).encode('utf-8')
    headers = {"Content-Type": "application/json"}
    if GATEWAY_API_KEY:
        headers["Authorization"] = f"Bearer {GATEWAY_API_KEY}"
    req = Request(GATEWAY_API_URL, data=data, headers=headers, method="POST")
    with urlopen(req, timeout=10) as resp:
        body = resp.read().decode('utf-8')
        return json.loads(body) if body else {"status": resp.status}

class PaymentHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        self._handle_request()

    def do_POST(self):
        self._handle_request()

    def _handle_request(self):
        parsed = urlparse(self.path)
        if parsed.path != '/pay':
            self.send_error(404)
            return
        params = {k: v[0] for k, v in parse_qs(parsed.query).items()}
        if self.command == 'POST':
            length = int(self.headers.get('Content-Length', 0))
            if length > 0:
                raw = self.rfile.read(length).decode('utf-8')
                ctype = self.headers.get('Content-Type', '')
                if 'application/json' in ctype:
                    try:
                        data = json.loads(raw)
                        if isinstance(data, dict):
                            params.update(data)
                    except json.JSONDecodeError:
                        self._json_response(400, {"error": "invalid_json"})
                        return
                else:
                    params.update({k: v[0] for k, v in parse_qs(raw).items()})
        success_url = params.get('success_url')
        if not success_url:
            self._json_response(400, {"error": "success_url_required"})
            return
        amount = params.get('amount')
        currency = params.get('currency', 'EUR')
        metadata = params.get('metadata')
        try:
            create_payment_session(amount, currency, success_url, metadata)
        except Exception:
            self._json_response(502, {"error": "gateway_error"})
            return
        self.send_response(303)
        self.send_header('Location', success_url)
        self.send_header('Content-Length', '0')
        self.end_headers()

    def _json_response(self, status, payload):
        body = json.dumps(payload).encode('utf-8')
        self.send_response(status)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', str(len(body)))
        self.end_headers()
        self.wfile.write(body)

if __name__ == '__main__':
    HTTPServer((HOST, PORT), PaymentHandler).serve_forever()
