#!/usr/bin/env python3
import base64
import hmac
import hashlib
import time
import struct
import sys

def _b32decode(s):
    s = s.strip().replace(" ", "").upper()
    padding = "=" * ((8 - len(s) % 8) % 8)
    return base64.b32decode(s + padding, casefold=True)

def totp(secret, t=None, step=30, digits=6):
    if t is None:
        t = int(time.time())
    key = _b32decode(secret)
    counter = int(t // step)
    msg = struct.pack(">Q", counter)
    digest = hmac.new(key, msg, hashlib.sha1).digest()
    offset = digest[-1] & 0x0F
    code = (struct.unpack(">I", digest[offset:offset + 4])[0] & 0x7fffffff) % (10 ** digits)
    return f"{code:0{digits}d}"

def verify(secret, code, window=1, step=30, digits=6):
    code = code.strip()
    if not code.isdigit() or len(code) != digits:
        return False
    t = int(time.time())
    for w in range(-window, window + 1):
        if hmac.compare_digest(totp(secret, t + w * step, step, digits), code):
            return True
    return False

def main():
    if len(sys.argv) >= 3:
        secret = sys.argv[1]
        code = sys.argv[2]
    else:
        secret = input("Secret: ").strip()
        code = input("Code: ").strip()
    if verify(secret, code):
        print("OK")
        sys.exit(0)
    else:
        print("INVALID")
        sys.exit(1)

if __name__ == "__main__":
    main()
