const CircuitState = { CLOSED: 'CLOSED', OPEN: 'OPEN', HALF_OPEN: 'HALF_OPEN' };

class CircuitBreaker {
  constructor({ failureThreshold = 5, successThreshold = 1, cooldownTime = 10000, requestTimeout = 8000 } = {}) {
    this.failureThreshold = failureThreshold;
    this.successThreshold = successThreshold;
    this.cooldownTime = cooldownTime;
    this.requestTimeout = requestTimeout;
    this.state = CircuitState.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
    this.halfOpenInProgress = false;
  }

  exec(action) {
    const now = Date.now();

    if (this.state === CircuitState.OPEN) {
      if (now >= this.nextAttempt) {
        this.state = CircuitState.HALF_OPEN;
      } else {
        return Promise.reject(new Error('Circuit breaker is open'));
      }
    }

    if (this.state === CircuitState.HALF_OPEN) {
      if (this.halfOpenInProgress) {
        return Promise.reject(new Error('Circuit breaker is half-open'));
      }
      this.halfOpenInProgress = true;
    }

    return this._executeWithTimeout(action)
      .then(result => {
        this._onSuccess();
        return result;
      })
      .catch(error => {
        this._onFailure();
        throw error;
      })
      .finally(() => {
        if (this.state === CircuitState.HALF_OPEN) {
          this.halfOpenInProgress = false;
        }
      });
  }

  _executeWithTimeout(action) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        reject(new Error('Request timeout'));
      }, this.requestTimeout);

      Promise.resolve()
        .then(action)
        .then(result => {
          clearTimeout(timer);
          resolve(result);
        })
        .catch(err => {
          clearTimeout(timer);
          reject(err);
        });
    });
  }

  _onSuccess() {
    if (this.state === CircuitState.HALF_OPEN) {
      this.successCount += 1;
      if (this.successCount >= this.successThreshold) {
        this._reset();
      }
    } else {
      this.failureCount = 0;
    }
  }

  _onFailure() {
    if (this.state === CircuitState.HALF_OPEN) {
      this._open();
      return;
    }
    this.failureCount += 1;
    if (this.failureCount >= this.failureThreshold) {
      this._open();
    }
  }

  _open() {
    this.state = CircuitState.OPEN;
    this.nextAttempt = Date.now() + this.cooldownTime;
    this.failureCount = 0;
    this.successCount = 0;
    this.halfOpenInProgress = false;
  }

  _reset() {
    this.state = CircuitState.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
    this.halfOpenInProgress = false;
  }

  getState() {
    return this.state;
  }
}

const backendBreaker = new CircuitBreaker({
  failureThreshold: 3,
  successThreshold: 1,
  cooldownTime: 10000,
  requestTimeout: 7000
});

function requestWithBreaker(url, options) {
  return backendBreaker.exec(() =>
    fetch(url, options).then(response => {
      if (!response.ok) {
        throw new Error('HTTP ' + response.status);
      }
      return response;
    })
  );
}

if (typeof window !== 'undefined') {
  window.CircuitBreaker = CircuitBreaker;
  window.backendBreaker = backendBreaker;
  window.requestWithBreaker = requestWithBreaker;
}
