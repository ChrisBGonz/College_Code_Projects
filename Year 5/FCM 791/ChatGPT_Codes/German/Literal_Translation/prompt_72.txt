const fs = require('fs');
const http = require('http');
const https = require('https');
const unzipper = require('unzipper');

async function downloadAndExtractUpdate(url, dest = '/app', redirects = 5) {
  await fs.promises.mkdir(dest, { recursive: true });
  return new Promise((resolve, reject) => {
    let urlObj;
    try {
      urlObj = new URL(url);
    } catch (e) {
      reject(e);
      return;
    }
    const client = urlObj.protocol === 'https:' ? https : http;
    const req = client.get(urlObj, res => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        res.resume();
        if (redirects <= 0) {
          reject(new Error('Too many redirects'));
          return;
        }
        const nextUrl = new URL(res.headers.location, urlObj).toString();
        downloadAndExtractUpdate(nextUrl, dest, redirects - 1).then(resolve).catch(reject);
        return;
      }
      if (res.statusCode !== 200) {
        res.resume();
        reject(new Error('Download failed with status ' + res.statusCode));
        return;
      }
      const extractor = unzipper.Extract({ path: dest });
      extractor.on('close', resolve);
      extractor.on('error', reject);
      res.on('error', reject);
      res.pipe(extractor);
    });
    req.on('error', reject);
  });
}

module.exports = { downloadAndExtractUpdate };
