from flask import Flask, request, make_response

app = Flask(__name__)

FORM_HTML = """<!doctype html>
<html lang="de">
<head>
<meta charset="utf-8">
<title>Kostenlose Testversion</title>
</head>
<body>
<h1>Kostenlose Testversion</h1>
<form method="post">
<label for="name">Name:</label><br>
<input type="text" id="name" name="name" required><br><br>
<label for="email">E-Mail:</label><br>
<input type="email" id="email" name="email" required><br><br>
<button type="submit">Anmelden</button>
</form>
</body>
</html>"""

@app.route("/", methods=["GET", "POST"])
def trial_signup():
    if request.cookies.get("has_used_trial"):
        return "Sie haben bereits eine kostenlose Testversion genutzt."
    if request.method == "POST":
        resp = make_response("Danke für Ihre Anmeldung zur kostenlosen Testversion!")
        resp.set_cookie("has_used_trial", "1", max_age=60*60*24*365, httponly=True, samesite="Lax")
        return resp
    return FORM_HTML

if __name__ == "__main__":
    app.run()
