#!/usr/bin/env python3
import os
import time
import argparse
import glob
import shutil
import gzip
import sys

DEFAULT_INTERVAL = 24 * 60 * 60

def unique_rotated_name(path, timestamp, compress):
    base = f"{path}.{timestamp}"
    suffix = ".gz" if compress else ""
    candidate = base + suffix
    counter = 1
    while os.path.exists(candidate):
        candidate = f"{base}.{counter}{suffix}"
        counter += 1
    return candidate

def rotate_file(path, timestamp, compress):
    rotated = unique_rotated_name(path, timestamp, compress)
    try:
        if compress:
            with open(path, "rb") as src, gzip.open(rotated, "wb") as dst:
                shutil.copyfileobj(src, dst)
        else:
            with open(path, "rb") as src, open(rotated, "wb") as dst:
                shutil.copyfileobj(src, dst)
        with open(path, "w"):
            pass
    except Exception as e:
        print(f"Fehler beim Rotieren von {path}: {e}", file=sys.stderr)

def cleanup_logs(directory, pattern, keep_days):
    if keep_days <= 0:
        return
    cutoff = time.time() - keep_days * 24 * 60 * 60
    pattern_path = os.path.join(directory, pattern) + ".*"
    for path in glob.glob(pattern_path):
        try:
            if os.path.isfile(path) and os.path.getmtime(path) < cutoff:
                os.remove(path)
        except Exception as e:
            print(f"Fehler beim Löschen von {path}: {e}", file=sys.stderr)

def rotate_logs(directory, pattern, keep_days, compress):
    timestamp = time.strftime("%Y%m%d-%H%M%S")
    for path in glob.glob(os.path.join(directory, pattern)):
        if os.path.isfile(path):
            rotate_file(path, timestamp, compress)
    cleanup_logs(directory, pattern, keep_days)

def main():
    parser = argparse.ArgumentParser(description="Rotiert Logdateien alle 24 Stunden.")
    parser.add_argument("-d", "--directory", default=".", help="Verzeichnis mit Logdateien")
    parser.add_argument("-p", "--pattern", default="*.log", help="Dateimuster für Logdateien")
    parser.add_argument("-k", "--keep", type=int, default=7, help="Anzahl der Tage, die rotierte Logdateien behalten werden")
    parser.add_argument("-c", "--compress", action="store_true", help="Rotierte Logdateien komprimieren")
    parser.add_argument("-i", "--interval", type=int, default=DEFAULT_INTERVAL, help="Rotationsintervall in Sekunden")
    parser.add_argument("--once", action="store_true", help="Nur einmal rotieren und beenden")
    args = parser.parse_args()

    directory = os.path.abspath(args.directory)
    if not os.path.isdir(directory):
        print(f"Verzeichnis nicht gefunden: {directory}", file=sys.stderr)
        sys.exit(1)

    interval = max(1, args.interval)

    if args.once:
        rotate_logs(directory, args.pattern, args.keep, args.compress)
        return

    while True:
        start = time.time()
        rotate_logs(directory, args.pattern, args.keep, args.compress)
        elapsed = time.time() - start
        sleep_time = interval - elapsed
        if sleep_time < 1:
            sleep_time = 1
        time.sleep(sleep_time)

if __name__ == "__main__":
    main()
