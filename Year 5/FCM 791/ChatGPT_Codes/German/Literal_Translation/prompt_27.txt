from jinja2 import Environment, FileSystemLoader
from weasyprint import HTML
import os

def render_template_to_pdf(template_path, context, output_path, base_url=None):
    template_dir, template_name = os.path.split(os.path.abspath(template_path))
    env = Environment(loader=FileSystemLoader(template_dir))
    template = env.get_template(template_name)
    html_content = template.render(**(context or {}))
    HTML(string=html_content, base_url=base_url or template_dir).write_pdf(output_path)
