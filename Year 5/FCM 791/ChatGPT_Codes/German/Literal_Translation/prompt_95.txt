import sys

def count_lines_in_memory(path):
    with open(path, 'rb') as f:
        data = f.read()
    if not data:
        return 0
    count = data.count(b'\n')
    if data[-1] != 10:
        count += 1
    return count

def main():
    if len(sys.argv) != 2:
        print("Usage: python script.py <file_path>")
        return
    path = sys.argv[1]
    try:
        line_count = count_lines_in_memory(path)
        print(line_count)
    except Exception as e:
        print(f"Error: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
