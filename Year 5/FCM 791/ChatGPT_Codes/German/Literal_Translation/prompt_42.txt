const { spawn } = require('child_process');

function pingHost(domain_name, timeout = 5000) {
  if (typeof domain_name !== 'string' || domain_name.trim() === '') {
    return Promise.reject(new Error('Invalid domain_name'));
  }

  return new Promise((resolve) => {
    const args = process.platform === 'win32' ? ['-n', '1', domain_name] : ['-c', '1', domain_name];
    const ping = spawn('ping', args);
    let done = false;

    const finish = (result) => {
      if (done) return;
      done = true;
      clearTimeout(timer);
      resolve(result);
    };

    const timer = setTimeout(() => {
      try { ping.kill(); } catch (e) {}
      finish(false);
    }, timeout);

    ping.on('error', () => finish(false));
    ping.on('close', (code) => finish(code === 0));
  });
}

module.exports = { pingHost };
