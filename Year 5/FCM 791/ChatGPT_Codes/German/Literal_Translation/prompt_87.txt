import sys
import time
import os
from collections import deque

def monitor_log(file_path):
    events = deque()
    warned = False
    while True:
        try:
            with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                f.seek(0, os.SEEK_END)
                while True:
                    line = f.readline()
                    if not line:
                        try:
                            if os.stat(file_path).st_size < f.tell():
                                f.seek(0)
                        except FileNotFoundError:
                            break
                        now = time.time()
                        cutoff = now - 60
                        while events and events[0] < cutoff:
                            events.popleft()
                        if len(events) <= 10:
                            warned = False
                        time.sleep(0.1)
                        continue
                    if "Failed Login" in line:
                        events.append(time.time())
                    now = time.time()
                    cutoff = now - 60
                    while events and events[0] < cutoff:
                        events.popleft()
                    if len(events) > 10:
                        if not warned:
                            print("WARNING: More than 10 Failed Login messages within one minute")
                            warned = True
                    else:
                        warned = False
        except FileNotFoundError:
            time.sleep(1)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python monitor.py <logfile>")
        sys.exit(1)
    monitor_log(sys.argv[1])
