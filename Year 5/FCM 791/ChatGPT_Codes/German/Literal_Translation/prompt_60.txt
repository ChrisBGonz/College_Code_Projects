const express = require('express');

const app = express();
app.set('trust proxy', true);
app.use(express.json());

const poll = {
  id: 1,
  question: 'Welche Farbe mögen Sie am meisten?',
  options: [
    { id: 1, text: 'Rot', count: 0 },
    { id: 2, text: 'Blau', count: 0 },
    { id: 3, text: 'Grün', count: 0 }
  ]
};

const votesByIp = new Map();

function normalizeIp(ip) {
  if (!ip) return '';
  if (Array.isArray(ip)) ip = ip[0];
  if (typeof ip === 'string' && ip.includes(',')) ip = ip.split(',')[0].trim();
  if (typeof ip === 'string' && ip.startsWith('::ffff:')) ip = ip.slice(7);
  return ip;
}

function getClientIp(req) {
  return normalizeIp(req.headers['x-forwarded-for'] || req.socket.remoteAddress || req.ip);
}

app.get('/poll', (req, res) => {
  res.json({
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text }))
  });
});

app.get('/results', (req, res) => {
  res.json({
    id: poll.id,
    question: poll.question,
    options: poll.options.map(o => ({ id: o.id, text: o.text, votes: o.count })),
    totalVotes: poll.options.reduce((sum, o) => sum + o.count, 0)
  });
});

app.post('/vote', (req, res) => {
  const ip = getClientIp(req);
  if (!ip) return res.status(400).json({ error: 'IP not found' });

  if (votesByIp.has(ip)) {
    return res.status(403).json({ error: 'Bereits abgestimmt', optionId: votesByIp.get(ip) });
  }

  const optionId = parseInt(req.body.optionId, 10);
  if (!optionId) return res.status(400).json({ error: 'optionId erforderlich' });

  const option = poll.options.find(o => o.id === optionId);
  if (!option) return res.status(400).json({ error: 'Ungültige Option' });

  option.count += 1;
  votesByIp.set(ip, optionId);

  res.json({ message: 'Stimme gezählt', optionId });
});

app.use((req, res) => {
  res.status(404).json({ error: 'Not found' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Voting server listening on port ${PORT}`);
});
