class StaffsController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager!

  def promote
    staff = User.find(params[:id])
    return redirect_to staff_path(staff), alert: 'Nur Staff-Mitglieder können befördert werden.' unless staff.staff?

    if staff.update(role: 'manager')
      redirect_to staff_path(staff), notice: 'Mitarbeiter wurde zum Manager befördert.'
    else
      redirect_to staff_path(staff), alert: 'Beförderung fehlgeschlagen.'
    end
  end

  private

  def require_manager!
    redirect_to root_path, alert: 'Keine Berechtigung.' unless current_user.manager?
  end
end
