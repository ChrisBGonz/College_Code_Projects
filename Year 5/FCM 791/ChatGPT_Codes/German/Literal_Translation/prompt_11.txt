import sys
from lxml import etree

def main():
    path = sys.argv[1] if len(sys.argv) > 1 else input().strip()
    if not path:
        return
    try:
        tree = etree.parse(path)
    except Exception as e:
        sys.stderr.write(str(e))
        return
    for elem in tree.findall('.//title'):
        print(elem.text if elem.text is not None else '')

if __name__ == "__main__":
    main()
