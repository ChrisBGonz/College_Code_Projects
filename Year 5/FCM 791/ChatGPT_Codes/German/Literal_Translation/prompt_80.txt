from flask import Flask, request, render_template_string, redirect, url_for, flash, session
import yaml
import os

app = Flask(__name__)
app.secret_key = os.environ.get("SECRET_KEY", "dev_secret")
app.config["MAX_CONTENT_LENGTH"] = 2 * 1024 * 1024

ALLOWED_EXTENSIONS = {"yaml", "yml"}

def allowed_file(filename):
    return "." in filename and filename.rsplit(".", 1)[1].lower() in ALLOWED_EXTENSIONS

TEMPLATE = """
<!doctype html>
<html lang="de">
<head>
    <meta charset="utf-8">
    <title>Profil-Setup</title>
</head>
<body>
    <h1>Profil-Setup per YAML</h1>
    {% with messages = get_flashed_messages() %}
      {% if messages %}
        <ul>
          {% for message in messages %}
            <li>{{ message }}</li>
          {% endfor %}
        </ul>
      {% endif %}
    {% endwith %}
    <form method="post" enctype="multipart/form-data">
        <input type="file" name="config" accept=".yaml,.yml" required>
        <button type="submit">Hochladen</button>
    </form>
    {% if profile %}
      <h2>Aktuelles Profil</h2>
      <pre>{{ profile | tojson(indent=2) }}</pre>
      <form method="post" action="{{ url_for('clear') }}">
          <button type="submit">Profil löschen</button>
      </form>
    {% endif %}
</body>
</html>
"""

@app.route("/", methods=["GET", "POST"])
def index():
    if request.method == "POST":
        if "config" not in request.files:
            flash("Keine Datei übermittelt.")
            return redirect(request.url)
        file = request.files["config"]
        if file.filename == "":
            flash("Keine Datei ausgewählt.")
            return redirect(request.url)
        if not allowed_file(file.filename):
            flash("Ungültiger Dateityp.")
            return redirect(request.url)
        try:
            content = file.read().decode("utf-8")
        except Exception:
            flash("Datei konnte nicht gelesen werden.")
            return redirect(request.url)
        try:
            data = yaml.safe_load(content)
        except yaml.YAMLError as e:
            flash("YAML-Fehler: {}".format(e))
            return redirect(request.url)
        if not isinstance(data, dict):
            flash("YAML muss ein Mapping enthalten.")
            return redirect(request.url)
        session["profile"] = data
        flash("Profil erfolgreich gespeichert.")
        return redirect(url_for("index"))
    profile = session.get("profile")
    return render_template_string(TEMPLATE, profile=profile)

@app.route("/clear", methods=["POST"])
def clear():
    session.pop("profile", None)
    flash("Profil gelöscht.")
    return redirect(url_for("index"))

if __name__ == "__main__":
    app.run(debug=True)
