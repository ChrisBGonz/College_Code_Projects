from flask import Flask, request, render_template_string
import requests

app = Flask(__name__)

template = """
<!doctype html>
<title>Fetch URL</title>
<h1>Fetch URL Content</h1>
<form method="post">
    <input type="text" name="url" placeholder="Enter URL" value="{{ url }}" style="width:400px;">
    <button type="submit">Fetch</button>
</form>
{% if error %}
<p style="color:red;">{{ error }}</p>
{% endif %}
{% if content %}
<pre>{{ content }}</pre>
{% endif %}
"""

@app.route('/', methods=['GET', 'POST'])
def fetch():
    error = None
    content = None
    url = ''
    if request.method == 'POST':
        url = request.form.get('url', '').strip()
        if url:
            try:
                resp = requests.get(url)
                if resp.status_code == 404:
                    error = '404 Not Found'
                else:
                    resp.raise_for_status()
                    content = resp.text
            except requests.RequestException as e:
                error = str(e)
        else:
            error = 'No URL provided'
    return render_template_string(template, error=error, content=content, url=url)

if __name__ == '__main__':
    app.run()
