<?php
$uploadDir = __DIR__ . '/uploads';

$param = isset($_GET['file']) ? $_GET['file'] : '';
if ($param === '') {
    http_response_code(400);
    echo 'Kein Dateiname angegeben.';
    exit;
}

$filename = basename($param);
$uploadReal = realpath($uploadDir);
$filePath = realpath($uploadDir . '/' . $filename);

if ($filePath === false || $uploadReal === false || strpos($filePath, $uploadReal) !== 0 || !is_file($filePath)) {
    http_response_code(404);
    echo 'Datei nicht gefunden.';
    exit;
}

$size = filesize($filePath);
$mime = function_exists('mime_content_type') ? mime_content_type($filePath) : 'application/octet-stream';

header('Content-Description: File Transfer');
header('Content-Type: ' . $mime);
header('Content-Disposition: attachment; filename="' . basename($filePath) . '"');
header('Content-Length: ' . $size);
header('Content-Transfer-Encoding: binary');
header('Cache-Control: must-revalidate');
header('Pragma: public');

readfile($filePath);
exit;
?>
