const logout = (req, res) => {
  if (req.session) {
    req.session.destroy(err => {
      if (err) {
        res.status(500).end();
        return;
      }
      res.clearCookie('connect.sid');
      res.status(204).end();
    });
  } else {
    res.status(204).end();
  }
};

module.exports = logout;
