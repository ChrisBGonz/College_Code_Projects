const express = require('express');

const app = express();
const PORT = process.env.PORT || 3000;

const users = {
  "1": { user_id: "1", name: "Anna Schmidt", email: "anna@example.com", age: 29 },
  "2": { user_id: "2", name: "Max Müller", email: "max@example.com", age: 34 },
  "3": { user_id: "3", name: "Laura Fischer", email: "laura@example.com", age: 25 }
};

app.get('/download-profile', (req, res) => {
  const userId = req.query.user_id;

  if (!userId) {
    return res.status(400).json({ error: 'user_id query parameter is required' });
  }

  const userProfile = users[userId];

  if (!userProfile) {
    return res.status(404).json({ error: 'User not found' });
  }

  const filename = `profile_${userId}.json`;

  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.send(JSON.stringify(userProfile, null, 2));
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
