<?php
session_start();

if (isset($_GET['reset'])) {
    session_unset();
    session_destroy();
    session_start();
}

$dsn = 'mysql:host=localhost;dbname=app;charset=utf8mb4';
$dbUser = 'app';
$dbPass = 'secret';
$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
];
$pdo = new PDO($dsn, $dbUser, $dbPass, $options);

function h($s) {
    return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
}

$message = '';

$step = 1;
if (isset($_SESSION['recovery_user_id'])) {
    $step = !empty($_SESSION['recovery_verified']) ? 3 : 2;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $posted = (int)($_POST['step'] ?? 0);

    if ($posted === 1) {
        $email = trim($_POST['email'] ?? '');
        if ($email === '') {
            $message = 'Bitte E-Mail eingeben.';
        } else {
            $stmt = $pdo->prepare('SELECT id FROM users WHERE email = ? LIMIT 1');
            $stmt->execute([$email]);
            $user = $stmt->fetch();
            if ($user) {
                $_SESSION['recovery_user_id'] = $user['id'];
                $_SESSION['recovery_verified'] = false;
                $step = 2;
            } else {
                $message = 'Falls die E-Mail existiert, können Sie die nächste Frage beantworten.';
                $step = 1;
            }
        }
    } elseif ($posted === 2) {
        if (!isset($_SESSION['recovery_user_id'])) {
            $step = 1;
        } else {
            $answer = trim($_POST['answer'] ?? '');
            if ($answer === '') {
                $message = 'Bitte Antwort eingeben.';
                $step = 2;
            } else {
                $stmt = $pdo->prepare('SELECT secret_answer_hash FROM users WHERE id = ?');
                $stmt->execute([$_SESSION['recovery_user_id']]);
                $row = $stmt->fetch();
                if ($row && password_verify($answer, $row['secret_answer_hash'])) {
                    $_SESSION['recovery_verified'] = true;
                    $step = 3;
                } else {
                    $message = 'Antwort ist falsch.';
                    $step = 2;
                }
            }
        }
    } elseif ($posted === 3) {
        if (!isset($_SESSION['recovery_user_id']) || empty($_SESSION['recovery_verified'])) {
            $step = 1;
        } else {
            $pwd = $_POST['password'] ?? '';
            $pwd2 = $_POST['password2'] ?? '';
            if (strlen($pwd) < 8) {
                $message = 'Passwort muss mindestens 8 Zeichen lang sein.';
                $step = 3;
            } elseif ($pwd !== $pwd2) {
                $message = 'Passwörter stimmen nicht überein.';
                $step = 3;
            } else {
                $hash = password_hash($pwd, PASSWORD_DEFAULT);
                $stmt = $pdo->prepare('UPDATE users SET password_hash = ? WHERE id = ?');
                $stmt->execute([$hash, $_SESSION['recovery_user_id']]);
                session_unset();
                session_destroy();
                echo '<!DOCTYPE html><html lang="de"><head><meta charset="utf-8"><title>Wiederherstellung</title></head><body>';
                echo '<p>Passwort wurde erfolgreich geändert.</p>';
                echo '<a href="login.php">Zum Login</a>';
                echo '</body></html>';
                exit;
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="de">
<head>
    <meta charset="utf-8">
    <title>Passwort-Wiederherstellung</title>
</head>
<body>
<?php if ($message): ?>
    <p><?php echo h($message); ?></p>
<?php endif; ?>

<?php if ($step === 1): ?>
    <form method="post">
        <input type="hidden" name="step" value="1">
        <label>E-Mail:</label>
        <input type="email" name="email" required>
        <button type="submit">Weiter</button>
    </form>
<?php elseif ($step === 2): ?>
    <p>Geheime Frage: Wie lautet der Mädchenname Ihrer Mutter?</p>
    <form method="post">
        <input type="hidden" name="step" value="2">
        <input type="text" name="answer" required>
        <button type="submit">Prüfen</button>
    </form>
<?php elseif ($step === 3): ?>
    <form method="post">
        <input type="hidden" name="step" value="3">
        <label>Neues Passwort:</label>
        <input type="password" name="password" required>
        <label>Passwort wiederholen:</label>
        <input type="password" name="password2" required>
        <button type="submit">Passwort speichern</button>
    </form>
<?php endif; ?>

<p><a href="?reset=1">Wiederherstellung neu starten</a></p>
</body>
</html>
