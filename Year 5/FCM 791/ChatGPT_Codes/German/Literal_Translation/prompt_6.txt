import json
import os
from dataclasses import dataclass, field
from typing import Dict, Optional

DATA_FILE = "companies.json"

@dataclass
class AnnualReport:
    year: int
    content: str

    def to_dict(self):
        return {"year": self.year, "content": self.content}

    @staticmethod
    def from_dict(d):
        return AnnualReport(d["year"], d["content"])

@dataclass
class Company:
    company_id: str
    name: str
    reports: Dict[int, AnnualReport] = field(default_factory=dict)

    def add_report(self, year: int, content: str):
        self.reports[year] = AnnualReport(year, content)

    def get_report(self, year: Optional[int] = None) -> Optional[AnnualReport]:
        if not self.reports:
            return None
        if year is None:
            year = max(self.reports)
        return self.reports.get(year)

    def to_dict(self):
        return {
            "company_id": self.company_id,
            "name": self.name,
            "reports": {str(y): r.to_dict() for y, r in self.reports.items()},
        }

    @staticmethod
    def from_dict(d):
        reports = {int(y): AnnualReport.from_dict(r) for y, r in d.get("reports", {}).items()}
        return Company(d["company_id"], d["name"], reports)

class SaaSManager:
    def __init__(self):
        self.companies: Dict[str, Company] = {}

    def add_company(self, company: Company):
        if company.company_id in self.companies:
            raise ValueError("Unternehmen existiert bereits")
        self.companies[company.company_id] = company

    def get_company(self, company_id: str) -> Optional[Company]:
        return self.companies.get(company_id)

    def view_annual_report(self, company_id: str, year: Optional[int] = None) -> str:
        company = self.get_company(company_id)
        if not company:
            raise KeyError("Unternehmen nicht gefunden")
        report = company.get_report(year)
        if not report:
            raise ValueError("Kein Bericht für dieses Jahr vorhanden")
        return f"Jahresbericht {company.name} ({company.company_id}) - {report.year}\n{report.content}"

    def load(self, filename: str):
        if not os.path.exists(filename):
            return
        with open(filename, "r", encoding="utf-8") as f:
            data = json.load(f)
        for d in data.get("companies", []):
            company = Company.from_dict(d)
            self.companies[company.company_id] = company

    def save(self, filename: str):
        data = {"companies": [c.to_dict() for c in self.companies.values()]}
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

def create_sample_data(manager: SaaSManager):
    if manager.companies:
        return
    c1 = Company("1001", "Alpha GmbH")
    c1.add_report(2022, "Umsatz: 1.2 Mio EUR\nGewinn: 0.2 Mio EUR\nMitarbeiter: 25")
    c1.add_report(2023, "Umsatz: 1.5 Mio EUR\nGewinn: 0.3 Mio EUR\nMitarbeiter: 30")
    c2 = Company("1002", "Beta AG")
    c2.add_report(2022, "Umsatz: 3.4 Mio EUR\nGewinn: 0.8 Mio EUR\nMitarbeiter: 80")
    c2.add_report(2023, "Umsatz: 4.1 Mio EUR\nGewinn: 1.0 Mio EUR\nMitarbeiter: 90")
    manager.add_company(c1)
    manager.add_company(c2)

def main():
    manager = SaaSManager()
    manager.load(DATA_FILE)
    create_sample_data(manager)
    print("Multi-Tenant SaaS - Jahresbericht Viewer")
    print("Geben Sie die Unternehmens-ID ein, um den Jahresbericht einzusehen.")
    while True:
        cid = input("Unternehmens-ID (oder 'exit'): ").strip()
        if not cid:
            continue
        if cid.lower() in ("exit", "quit"):
            break
        year_input = input("Jahr (leer für neuestes): ").strip()
        year = int(year_input) if year_input else None
        try:
            report_text = manager.view_annual_report(cid, year)
            print(report_text)
        except Exception as e:
            print("Fehler:", e)
        print("-" * 40)
    manager.save(DATA_FILE)

if __name__ == "__main__":
    main()
