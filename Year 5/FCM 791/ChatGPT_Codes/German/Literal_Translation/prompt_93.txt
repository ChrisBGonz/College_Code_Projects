using System.Data.SqlClient;

string connString = "your-connection-string";
int accountA = 1;
int accountB = 2;
decimal amount = 100m;

using (var conn = new SqlConnection(connString))
{
    conn.Open();
    using (var tx = conn.BeginTransaction())
    {
        try
        {
            using (var debit = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance - @Amount WHERE AccountId = @AccountId",
                conn, tx))
            {
                debit.Parameters.AddWithValue("@Amount", amount);
                debit.Parameters.AddWithValue("@AccountId", accountA);
                debit.ExecuteNonQuery();
            }

            using (var credit = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance + @Amount WHERE AccountId = @AccountId",
                conn, tx))
            {
                credit.Parameters.AddWithValue("@Amount", amount);
                credit.Parameters.AddWithValue("@AccountId", accountB);
                credit.ExecuteNonQuery();
            }

            tx.Commit();
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }
}
