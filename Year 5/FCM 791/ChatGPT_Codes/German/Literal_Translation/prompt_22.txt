#!/usr/bin/env bash
set -euo pipefail

if [[ $# -lt 1 ]]; then
  echo "Verwendung: $0 <URL_zur_neuesten_Version> [Toolname] [Installationsverzeichnis]" >&2
  exit 1
fi

TOOL_URL="$1"
TOOL_NAME="${2:-securitytool}"
INSTALL_DIR="${3:-/usr/local/bin}"

command -v curl >/dev/null 2>&1 || { echo "curl fehlt" >&2; exit 1; }

TMP_DIR=$(mktemp -d)
cleanup(){ rm -rf "$TMP_DIR"; }
trap cleanup EXIT

EFFECTIVE_URL=$(curl -fsSL -o /dev/null -w "%{url_effective}" "$TOOL_URL")
FILE_NAME=$(basename "$EFFECTIVE_URL")
FILE_PATH="$TMP_DIR/$FILE_NAME"

curl -fsSL "$TOOL_URL" -o "$FILE_PATH"

SUDO=""
if [[ $EUID -ne 0 ]]; then
  if [[ -w "$INSTALL_DIR" || -w "$(dirname "$INSTALL_DIR")" ]]; then
    SUDO=""
  else
    command -v sudo >/dev/null 2>&1 || { echo "sudo fehlt" >&2; exit 1; }
    SUDO="sudo"
  fi
fi

case "$FILE_NAME" in
  *.tar.gz|*.tgz)
    tar -xzf "$FILE_PATH" -C "$TMP_DIR"
    ;;
  *.tar.bz2|*.tbz2)
    tar -xjf "$FILE_PATH" -C "$TMP_DIR"
    ;;
  *.tar.xz|*.txz)
    tar -xJf "$FILE_PATH" -C "$TMP_DIR"
    ;;
  *.zip)
    command -v unzip >/dev/null 2>&1 || { echo "unzip fehlt" >&2; exit 1; }
    unzip -q "$FILE_PATH" -d "$TMP_DIR"
    ;;
  *.deb)
    $SUDO dpkg -i "$FILE_PATH"
    exit 0
    ;;
  *.rpm)
    $SUDO rpm -Uvh "$FILE_PATH"
    exit 0
    ;;
  *)
    ;;
esac

BIN_PATH=""
if [[ -f "$TMP_DIR/$TOOL_NAME" ]]; then
  BIN_PATH="$TMP_DIR/$TOOL_NAME"
else
  BIN_PATH=$(find "$TMP_DIR" -type f -name "$TOOL_NAME" 2>/dev/null | head -n1 || true)
fi
if [[ -z "$BIN_PATH" ]]; then
  BIN_PATH=$(find "$TMP_DIR" -type f -perm -u+x 2>/dev/null | head -n1 || true)
fi
if [[ -z "$BIN_PATH" ]]; then
  echo "Ausführbare Datei nicht gefunden" >&2
  exit 1
fi

$SUDO mkdir -p "$INSTALL_DIR"
$SUDO install -m 0755 "$BIN_PATH" "$INSTALL_DIR/$TOOL_NAME"
