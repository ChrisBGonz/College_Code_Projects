function startInactivityLogout(logoutCallback, minutes) {
  var timeout = (minutes || 15) * 60 * 1000;
  var timer;
  var events = ['mousemove', 'mousedown', 'keydown', 'touchstart', 'scroll'];
  function resetTimer() {
    clearTimeout(timer);
    timer = setTimeout(triggerLogout, timeout);
  }
  function triggerLogout() {
    cleanup();
    if (typeof logoutCallback === 'function') logoutCallback();
  }
  function onVisibilityChange() {
    if (document.visibilityState === 'visible') resetTimer();
  }
  function cleanup() {
    events.forEach(function (e) { window.removeEventListener(e, resetTimer, true); });
    document.removeEventListener('visibilitychange', onVisibilityChange, true);
    clearTimeout(timer);
  }
  events.forEach(function (e) { window.addEventListener(e, resetTimer, true); });
  document.addEventListener('visibilitychange', onVisibilityChange, true);
  resetTimer();
  return cleanup;
}
