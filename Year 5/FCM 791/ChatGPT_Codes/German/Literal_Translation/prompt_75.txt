<?php
class UserSettings {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'de', $fontSize = '14px') {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

$cookieName = 'user_settings';
$settings = null;

if (isset($_COOKIE[$cookieName])) {
    $settings = @unserialize($_COOKIE[$cookieName], ['allowed_classes' => [UserSettings::class]]);
}

if (!$settings instanceof UserSettings) {
    $settings = new UserSettings();
}

$input = array_merge($_GET, $_POST);

if (isset($input['theme'])) {
    $settings->theme = $input['theme'];
}
if (isset($input['language'])) {
    $settings->language = $input['language'];
}
if (isset($input['fontSize'])) {
    $settings->fontSize = $input['fontSize'];
}

setcookie($cookieName, serialize($settings), time() + 60 * 60 * 24 * 30, "/");

header('Content-Type: application/json; charset=utf-8');
echo json_encode($settings);
?>
