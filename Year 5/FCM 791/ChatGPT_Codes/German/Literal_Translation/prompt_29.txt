#!/usr/bin/env bash
set -euo pipefail

PROJECT_DIR="${1:-.}"

if [ ! -f "$PROJECT_DIR/pom.xml" ]; then
  echo "pom.xml not found in $PROJECT_DIR" >&2
  exit 1
fi

if ! command -v mvn >/dev/null 2>&1; then
  echo "mvn command not found" >&2
  exit 1
fi

cd "$PROJECT_DIR"

MVN_COMMON=(-B -q -DgenerateBackupPoms=false -DprocessDependencyManagement=true -DallowSnapshots=false)

mvn "${MVN_COMMON[@]}" versions:update-parent
mvn "${MVN_COMMON[@]}" versions:update-properties
mvn "${MVN_COMMON[@]}" versions:use-latest-releases
mvn "${MVN_COMMON[@]}" -DprocessPlugins=true versions:use-latest-releases
