<?php
$uploadDir = '/mnt/unternehmenslaufwerk/uploads/';
$maxSize = 20 * 1024 * 1024;
$messages = [];

function sanitizeFileName($filename){
    $filename = preg_replace('/[^\w\-.]+/u','_', $filename);
    $filename = trim($filename, '._');
    return $filename !== '' ? $filename : 'datei';
}

function isPdfSignature($path){
    $fh = @fopen($path, 'rb');
    if(!$fh) return false;
    $sig = fread($fh, 4);
    fclose($fh);
    return $sig === '%PDF';
}

function uniqueSuffix(){
    if(function_exists('random_bytes')){
        try { return bin2hex(random_bytes(4)); } catch(Exception $e) {}
    }
    return uniqid();
}

if($_SERVER['REQUEST_METHOD'] === 'POST'){
    if(!isset($_FILES['pdf_file'])){
        $messages[] = 'Keine Datei empfangen.';
    } else {
        $file = $_FILES['pdf_file'];
        if($file['error'] !== UPLOAD_ERR_OK){
            $errors = [
                UPLOAD_ERR_INI_SIZE => 'Die Datei überschreitet die maximale Größe.',
                UPLOAD_ERR_FORM_SIZE => 'Die Datei überschreitet die maximale Größe.',
                UPLOAD_ERR_PARTIAL => 'Die Datei wurde nur teilweise hochgeladen.',
                UPLOAD_ERR_NO_FILE => 'Es wurde keine Datei ausgewählt.',
                UPLOAD_ERR_NO_TMP_DIR => 'Temporärer Ordner fehlt.',
                UPLOAD_ERR_CANT_WRITE => 'Die Datei konnte nicht geschrieben werden.',
                UPLOAD_ERR_EXTENSION => 'Der Upload wurde durch eine Erweiterung gestoppt.'
            ];
            $messages[] = $errors[$file['error']] ?? 'Unbekannter Fehler beim Upload.';
        } elseif($file['size'] <= 0){
            $messages[] = 'Die Datei ist leer.';
        } elseif($file['size'] > $maxSize){
            $messages[] = 'Die Datei ist zu groß. Maximal ' . round($maxSize/1024/1024,1) . ' MB.';
        } else {
            $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
            if($ext !== 'pdf'){
                $messages[] = 'Nur PDF-Dateien sind erlaubt.';
            } else {
                $mime = null;
                if(function_exists('finfo_open')){
                    $finfo = finfo_open(FILEINFO_MIME_TYPE);
                    if($finfo){
                        $mime = finfo_file($finfo, $file['tmp_name']);
                        finfo_close($finfo);
                    }
                } elseif(function_exists('mime_content_type')) {
                    $mime = mime_content_type($file['tmp_name']);
                }
                if($mime !== null && $mime !== 'application/pdf' && $mime !== 'application/octet-stream'){
                    $messages[] = 'Die Datei scheint kein gültiges PDF zu sein.';
                } elseif(!isPdfSignature($file['tmp_name'])){
                    $messages[] = 'Die Datei scheint kein gültiges PDF zu sein.';
                } elseif(!is_dir($uploadDir) || !is_writable($uploadDir)){
                    $messages[] = 'Das Upload-Verzeichnis ist nicht verfügbar oder nicht beschreibbar.';
                } else {
                    $base = pathinfo($file['name'], PATHINFO_FILENAME);
                    $safeBase = sanitizeFileName($base);
                    $targetName = $safeBase . '.pdf';
                    $targetPath = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $targetName;
                    if(file_exists($targetPath)){
                        $targetName = $safeBase . '_' . date('Ymd_His') . '_' . uniqueSuffix() . '.pdf';
                        $targetPath = rtrim($uploadDir, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR . $targetName;
                    }
                    if(move_uploaded_file($file['tmp_name'], $targetPath)){
                        $messages[] = 'Upload erfolgreich: ' . htmlspecialchars($targetName, ENT_QUOTES, 'UTF-8');
                    } else {
                        $messages[] = 'Die Datei konnte nicht gespeichert werden.';
                    }
                }
            }
        }
    }
}
