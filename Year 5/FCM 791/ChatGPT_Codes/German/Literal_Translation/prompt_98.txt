#include <sys/socket.h>
#include <unistd.h>
#include <cstdint>
#include <memory>
#include <stdexcept>
#include <limits>

std::unique_ptr<char[]> allocateBufferFromSocket(int sockfd, std::size_t &outSize) {
    uint64_t size = 0;
    char c;
    bool hasDigit = false;
    while (true) {
        ssize_t r = recv(sockfd, &c, 1, 0);
        if (r <= 0) {
            throw std::runtime_error("socket read failed");
        }
        if (c == '\n') {
            if (hasDigit) break;
            continue;
        }
        if (c == '\r') {
            continue;
        }
        if (c < '0' || c > '9') {
            throw std::runtime_error("invalid size");
        }
        hasDigit = true;
        if (size > (std::numeric_limits<uint64_t>::max() - (c - '0')) / 10) {
            throw std::runtime_error("size overflow");
        }
        size = size * 10 + static_cast<uint64_t>(c - '0');
    }
    if (size == 0 || size > std::numeric_limits<std::size_t>::max()) {
        throw std::runtime_error("size out of range");
    }
    outSize = static_cast<std::size_t>(size);
    return std::unique_ptr<char[]>(new char[outSize]);
}
