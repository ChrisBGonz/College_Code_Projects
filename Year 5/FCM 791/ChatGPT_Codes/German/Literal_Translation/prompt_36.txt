#!/usr/bin/env python3
import argparse
import hashlib
import os
import sys

def sha256_file(path, block_size=1048576):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(block_size), b''):
            h.update(chunk)
    return h.hexdigest()

def load_expected_hash(path):
    with open(path, 'r', encoding='utf-8') as f:
        content = f.read().strip()
    return content.split()[0]

def main():
    parser = argparse.ArgumentParser(description="Prüft die Integrität einer Firmware-Datei mittels SHA256.")
    parser.add_argument("file", help="Pfad zur Firmware-Datei")
    group = parser.add_mutually_exclusive_group(required=True)
    group.add_argument("--hash", dest="expected_hash", help="Erwarteter SHA256-Hash")
    group.add_argument("--hash-file", help="Datei, die den erwarteten Hash enthält")
    args = parser.parse_args()

    if not os.path.isfile(args.file):
        print("Datei nicht gefunden", file=sys.stderr)
        sys.exit(2)

    try:
        expected = args.expected_hash.strip().split()[0] if args.expected_hash else load_expected_hash(args.hash_file)
    except Exception:
        print("Hash-Datei konnte nicht gelesen werden", file=sys.stderr)
        sys.exit(2)

    computed = sha256_file(args.file)

    if computed.lower() == expected.lower():
        print("OK")
        sys.exit(0)
    else:
        print("FEHLER")
        print("Erwartet:", expected.lower())
        print("Berechnet:", computed)
        sys.exit(1)

if __name__ == "__main__":
    main()
