const express = require('express');
const path = require('path');
const fs = require('fs').promises;

const app = express();
const PORT = process.env.PORT || 3000;

app.get('/users/:userId/private_vault', async (req, res) => {
  const userId = req.params.userId;
  const folderName = req.query.folder_name;

  if (!folderName) {
    return res.status(400).json({ error: 'folder_name parameter required' });
  }

  const vaultRoot = path.join(__dirname, 'private_vault', userId);
  const targetDir = path.join(vaultRoot, folderName);

  const resolvedRoot = path.resolve(vaultRoot);
  const resolvedTarget = path.resolve(targetDir);

  if (resolvedTarget !== resolvedRoot && !resolvedTarget.startsWith(resolvedRoot + path.sep)) {
    return res.status(400).json({ error: 'Invalid folder_name' });
  }

  try {
    const entries = await fs.readdir(resolvedTarget, { withFileTypes: true });
    const files = entries.filter(e => e.isFile()).map(e => e.name);
    res.json({ files });
  } catch (err) {
    if (err.code === 'ENOENT') {
      res.status(404).json({ error: 'Folder not found' });
    } else {
      res.status(500).json({ error: 'Error reading folder' });
    }
  }
});

app.listen(PORT);
