import pickle
from dataclasses import dataclass
from typing import List, Dict, Any

@dataclass
class Player:
    name: str
    level: int
    health: int
    inventory: List[str]
    stats: Dict[str, int]

@dataclass
class GameState:
    player: Player
    position: Dict[str, float]
    quest_status: Dict[str, str]
    world_state: Dict[str, Any]
    time_played: float

def save_game_state(game_state: GameState, filename: str) -> None:
    with open(filename, "wb") as f:
        pickle.dump(game_state, f)

def load_game_state(filename: str) -> GameState:
    with open(filename, "rb") as f:
        return pickle.load(f)

def main() -> None:
    player = Player(
        name="Hero",
        level=5,
        health=100,
        inventory=["sword", "shield", "potion"],
        stats={"strength": 10, "agility": 7, "intelligence": 5}
    )
    game_state = GameState(
        player=player,
        position={"x": 12.5, "y": 42.0, "z": 3.2},
        quest_status={"dragon": "in progress", "lost_ring": "completed"},
        world_state={
            "weather": "rain",
            "npcs": {
                "merchant": {"location": "town", "mood": "happy"},
                "blacksmith": {"location": "village", "mood": "busy"}
            }
        },
        time_played=1234.5
    )

    filename = "game_state.pkl"
    save_game_state(game_state, filename)

    loaded_state = load_game_state(filename)
    print(loaded_state)

if __name__ == "__main__":
    main()
