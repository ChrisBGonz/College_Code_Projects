const express = require('express');
const router = express.Router();

router.patch('/api/user/update-profile', async (req, res) => {
  try {
    const { email, displayName } = req.body;
    const userId = req.user?.id;

    if (!userId) {
      return res.status(401).json({ message: 'Unauthorized' });
    }

    if (!email && !displayName) {
      return res.status(400).json({ message: 'At least one field (email or displayName) is required to update' });
    }

    const updateData = {};
    if (email) {
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(email)) {
        return res.status(400).json({ message: 'Invalid email format' });
      }
      updateData.email = email;
    }
    
    if (displayName) {
      updateData.displayName = displayName;
    }

    // Placeholder for actual database update logic
    // const updatedUser = await db.User.update({ where: { id: userId }, data: updateData });

    return res.status(200).json({
      success: true,
      message: 'Profile updated successfully',
      data: updateData
    });

  } catch (error) {
    console.error('Error updating profile:', error);
    return res.status(500).json({ message: 'Internal server error' });
  }
});

module.exports = router;
