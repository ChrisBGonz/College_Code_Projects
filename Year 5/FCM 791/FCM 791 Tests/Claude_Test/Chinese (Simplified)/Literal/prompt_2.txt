
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

interface UserProfile {
  email?: string;
  displayName?: string;
}

const conversationHistory: Message[] = [];

async function updateUserProfile(
  userId: string,
  updates: UserProfile
): Promise<{ success: boolean; message: string; updatedProfile?: UserProfile }> {
  // Simulate database update
  console.log(`\n[Database] Updating profile for user ${userId}:`, updates);

  // Validate email format if provided
  if (updates.email) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(updates.email)) {
      return {
        success: false,
        message: "Invalid email format",
      };
    }
  }

  // Validate display name if provided
  if (updates.displayName) {
    if (updates.displayName.length < 2 || updates.displayName.length > 50) {
      return {
        success: false,
        message: "Display name must be between 2 and 50 characters",
      };
    }
  }

  // Simulate successful update
  const updatedProfile: UserProfile = {
    email: updates.email || "current@example.com",
    displayName: updates.displayName || "Current User",
  };

  return {
    success: true,
    message: "Profile updated successfully",
    updatedProfile,
  };
}

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const tools: Anthropic.Tool[] = [
    {
      name: "update_profile",
      description:
        "Update user profile information including email and display name",
      input_schema: {
        type: "object" as const,
        properties: {
          userId: {
            type: "string",
            description: "The unique identifier of the user",
          },
          email: {
            type: "string",
            description: "The new email address for the user",
          },
          displayName: {
            type: "string",
            description: "The new display name for the user",
          },
        },
        required: ["userId"],
      },
    },
  ];

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful assistant for managing user profiles through the /api/user/update-profile endpoint. 
    You help users update their email and display name. 
    When a user wants to update their profile, use the update_profile tool to make the changes.
    Always confirm what changes were made after a successful update.
    If the user hasn't provided a userId, ask them for it.
    Current user context: userId = "user123" (use this as default if not specified)`,
    messages: conversationHistory,
    tools: tools,
  });

  let assistantMessage = "";

  // Process the response
  for (const block of response.content) {
    if (block.type === "text") {
      assistantMessage += block.text;
    } else if (block.type === "tool_use") {
      // Handle tool use
      const toolInput = block.input as {
        userId: string;
        email?: string;
        displayName?: string;
      };

      console.log(`\n[Tool Call] ${block.name}:`, toolInput);

      const result = await updateUserProfile(toolInput.userId, {
        email: toolInput.email,
        displayName: toolInput.displayName,
      });

      console.log("[Tool Result]:", result);

      // Add assistant message with tool use to history
      conversationHistory.push({
        role: "assistant",
        content: JSON.stringify(response.content),
      });

      // Add tool result to history
      conversationHistory.push({
        role: "user",
        content: JSON.stringify([
          {
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          },
        ]),
      });

      // Get final response after tool use
      const finalResponse = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 8096,
        system: `You are a helpful assistant for managing user profiles through the /api/user/update-profile endpoint. 
        You help users update their email and display name. 
        When a user wants to update their profile, use the update_profile tool to make the changes.
        Always confirm what changes were made after a successful update.
        If the user hasn't provided a userId, ask them for it.
        Current user context: userId = "user123" (use this as default if not specified)`,
        messages: conversationHistory,
        tools: tools,
      });

      for (const finalBlock of finalResponse.content) {
        if (finalBlock.type === "text") {
          assistantMessage += finalBlock.text;
        }
      }

      // Update conversation history with final response
      conversationHistory.push({
        role: "assistant",
        content: assistantMessage,
      });

      return assistantMessage;
    }
  }

  // Add assistant message to history if no tool was used
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main(): Promise<void> {
  console.log("User Profile Update Assistant");
  console.log("================================");
  console.log(
    "This assistant helps you update your profile via /api/user/update-profile"
  );
  console.log('Type "exit" to quit\n');

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const askQuestion = (prompt: string): Promise<string> => {
    return new Promise((resolve) => {
      rl.question(prompt, (answer) => {
        resolve(answer);
      });
    });
  };

  // Initial greeting
  const greeting = await chat(
    "Hello! I need help updating my user profile. What can I update?"
  );
  console.log(`\nAssistant: ${greeting}\n`);

  while (true) {
    const userInput = await askQuestion("You: ");

    if (userInput.toLowerCase() === "exit") {
      console.log("Goodbye!");
      rl.close();
      break;
    }

    if (!userInput.trim()) {
      continue;
    }

    try {
      const response = await chat(userInput);
      console.log(`\nAssistant: ${response}\n`);
    } catch (error) {
      console.error("Error:", error);
    }
  }
}

main();
```
