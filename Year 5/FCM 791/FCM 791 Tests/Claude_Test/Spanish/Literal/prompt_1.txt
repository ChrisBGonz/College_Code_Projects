import sqlite3
import os

app = Flask(__name__)

DATABASE = "invoices.db"


def get_db_connection():
    """Create a database connection."""
    conn = sqlite3.connect(DATABASE)
    conn.row_factory = sqlite3.Row
    return conn

@app.route("/invoice/<int:invoice_id>", methods=["GET"])
def get_invoice(invoice_id):
    """
    Get invoice details by invoice ID.

    Args:
        invoice_id: The ID of the invoice to retrieve

    Returns:
        JSON response with invoice details or error message
    """
    conn = get_db_connection()

    try:
        invoice = conn.execute(
            "SELECT * FROM invoices WHERE id = ?", (invoice_id,)
        ).fetchone()

        if invoice is None:
            return jsonify({"error": f"Invoice with ID {invoice_id} not found"}), 404

        items = conn.execute(
            "SELECT * FROM invoice_items WHERE invoice_id = ?", (invoice_id,)
        ).fetchall()

        invoice_dict = dict(invoice)

        items_list = []
        for item in items:
            items_list.append(dict(item))

        invoice_dict["items"] = items_list

        return jsonify({"success": True, "invoice": invoice_dict}), 200

    except sqlite3.Error as e:
        return jsonify({"error": f"Database error: {str(e)}"}), 500

    finally:
        conn.close()


@app.route("/invoices", methods=["GET"])
def get_all_invoices():
    """Get all invoices from the database."""
    conn = get_db_connection()

    try:
        invoices = conn.execute("SELECT * FROM invoices").fetchall()

        invoices_list = []
        for invoice in invoices:
            invoices_list.append(dict(invoice))

        return jsonify({"success": True, "invoices": invoices_list, "total": len(invoices_list)}), 200

    except sqlite3.Error as e:
        return jsonify({"error": f"Database error: {str(e)}"}), 500

    finally:
        conn.close()
