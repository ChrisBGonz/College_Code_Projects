const express = require('express');
const router = express.Router();
const { body, validationResult } = require('express-validator');
const db = require('../db');
const authMiddleware = require('../middleware/auth');

const validateUpdateProfile = [
  body('email')
    .optional()
    .isEmail()
    .withMessage('Некорректный формат email')
    .normalizeEmail(),
  body('displayName')
    .optional()
    .isString()
    .withMessage('displayName должен быть строкой')
    .trim()
    .isLength({ min: 2, max: 50 })
    .withMessage('displayName должен содержать от 2 до 50 символов')
    .matches(/^[a-zA-Zа-яА-ЯёЁ0-9\s_-]+$/)
    .withMessage('displayName содержит недопустимые символы'),
];

router.post(
  '/api/user/update-profile',
  authMiddleware,
  validateUpdateProfile,
  async (req, res) => {
    try {
      const errors = validationResult(req);
      if (!errors.isEmpty()) {
        return res.status(400).json({
          success: false,
          errors: errors.array().map((err) => ({
            field: err.param,
            message: err.msg,
          })),
        });
      }

      const userId = req.user.id;
      const { email, displayName } = req.body;

      if (!email && !displayName) {
        return res.status(400).json({
          success: false,
          message: 'Необходимо передать хотя бы одно поле для обновления (email или displayName)',
        });
      }

      const existingUser = await db.query(
        'SELECT id, email, display_name FROM users WHERE id = $1',
        [userId]
      );

      if (!existingUser.rows.length) {
        return res.status(404).json({
          success: false,
          message: 'Пользователь не найден',
        });
      }

      if (email) {
        const emailTaken = await db.query(
          'SELECT id FROM users WHERE email = $1 AND id != $2',
          [email, userId]
        );

        if (emailTaken.rows.length) {
          return res.status(409).json({
            success: false,
            message: 'Данный email уже используется другим пользователем',
          });
        }
      }

      const fields = [];
      const values = [];
      let paramIndex = 1;

      if (email) {
        fields.push(`email = $${paramIndex++}`);
        values.push(email);
      }

      if (displayName) {
        fields.push(`display_name = $${paramIndex++}`);
        values.push(displayName);
      }

      fields.push(`updated_at = $${paramIndex++}`);
      values.push(new Date());

      values.push(userId);

      const updateQuery = `
        UPDATE users
        SET ${fields.join(', ')}
        WHERE id = $${paramIndex}
        RETURNING id, email, display_name AS "displayName", updated_at AS "updatedAt"
      `;

      const updatedUser = await db.query(updateQuery, values);

      return res.status(200).json({
        success: true,
        message: 'Профиль успешно обновлён',
        user: updatedUser.rows[0],
      });
    } catch (error) {
      console.error('Ошибка при обновлении профиля:', error);

      return res.status(500).json({
        success: false,
        message: 'Внутренняя ошибка сервера',
      });
    }
  }
);

module.exports = router;
