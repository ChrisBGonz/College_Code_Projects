const express = require('express');
const { body, validationResult } = require('express-validator');
const jwt = require('jsonwebtoken');
const rateLimit = require('express-rate-limit');
const DOMPurify = require('isomorphic-dompurify');
const { PrismaClient } = require('@prisma/client');

const prisma = new PrismaClient();
const router = express.Router();

// ─── Rate Limiter ────────────────────────────────────────────────────────────

const updateProfileLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 10,
  standardHeaders: true,
  legacyHeaders: false,
  message: {
    status: 429,
    error: 'Too Many Requests',
    message: 'Too many profile update attempts. Please try again later.',
  },
});

// ─── Auth Middleware ─────────────────────────────────────────────────────────

const authenticate = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return res.status(401).json({
        status: 401,
        error: 'Unauthorized',
        message: 'Missing or malformed Authorization header.',
      });
    }

    const token = authHeader.split(' ')[1];

    if (!token) {
      return res.status(401).json({
        status: 401,
        error: 'Unauthorized',
        message: 'Token not provided.',
      });
    }

    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;
    next();
  } catch (err) {
    if (err.name === 'TokenExpiredError') {
      return res.status(401).json({
        status: 401,
        error: 'Unauthorized',
        message: 'Token has expired.',
      });
    }

    if (err.name === 'JsonWebTokenError') {
      return res.status(401).json({
        status: 401,
        error: 'Unauthorized',
        message: 'Invalid token.',
      });
    }

    return res.status(500).json({
      status: 500,
      error: 'Internal Server Error',
      message: 'Authentication failed due to an unexpected error.',
    });
  }
};

// ─── Validation Rules ────────────────────────────────────────────────────────

const updateProfileValidationRules = [
  body('email')
    .optional()
    .trim()
    .notEmpty()
    .withMessage('Email must not be empty if provided.')
    .isEmail()
    .withMessage('Must be a valid email address.')
    .normalizeEmail()
    .isLength({ max: 254 })
    .withMessage('Email must not exceed 254 characters.'),

  body('displayName')
    .optional()
    .trim()
    .notEmpty()
    .withMessage('Display name must not be empty if provided.')
    .isLength({ min: 2, max: 50 })
    .withMessage('Display name must be between 2 and 50 characters.')
    .matches(/^[\p{L}\p{N}\s._\-]+$/u)
    .withMessage(
      'Display name may only contain letters, numbers, spaces, dots, underscores, and hyphens.'
    ),
];

// ─── Sanitize Helper ─────────────────────────────────────────────────────────

const sanitizeInputs = (data) => {
  const sanitized = {};

  if (data.email !== undefined) {
    sanitized.email = DOMPurify.sanitize(data.email).trim().toLowerCase();
  }

  if (data.displayName !== undefined) {
    sanitized.displayName = DOMPurify.sanitize(data.displayName).trim();
  }

  return sanitized;
};

// ─── Safe User Projection ────────────────────────────────────────────────────

const toSafeUser = (user) => ({
  id: user.id,
  email: user.email,
  displayName: user.displayName,
  createdAt: user.createdAt,
  updatedAt: user.updatedAt,
});

// ─── Route Handler ───────────────────────────────────────────────────────────

router.post(
  '/api/user/update-profile',
  updateProfileLimiter,
  authenticate,
  updateProfileValidationRules,
  async (req, res) => {
    // 1. Validate inputs
    const errors = validationResult(req);

    if (!errors.isEmpty()) {
      return res.status(422).json({
        status: 422,
        error: 'Unprocessable Entity',
        message: 'Input validation failed.',
        details: errors.array().map((e) => ({
          field: e.path,
          message: e.msg,
        })),
      });
    }

    // 2. Check that at least one field is provided
    const { email, displayName } = req.body;

    if (email === undefined && displayName === undefined) {
      return res.status(400).json({
        status: 400,
        error: 'Bad Request',
        message: 'At least one field (email or displayName) must be provided.',
      });
    }

    // 3. Sanitize
    const sanitized = sanitizeInputs({ email, displayName });

    try {
      // 4. Verify user exists
      const existingUser = await prisma.user.findUnique({
        where: { id: req.user.id },
      });

      if (!existingUser) {
        return res.status(404).json({
          status: 404,
          error: 'Not Found',
          message: 'User account not found.',
        });
      }

      // 5. Check email uniqueness if email is being updated
      if (sanitized.email && sanitized.email !== existingUser.email) {
        const emailTaken = await prisma.user.findUnique({
          where: { email: sanitized.email },
        });

        if (emailTaken) {
          return res.status(409).json({
            status: 409,
            error: 'Conflict',
            message: 'This email address is already in use.',
          });
        }
      }

      // 6. Build update payload (only changed fields)
      const updateData = {};

      if (sanitized.email && sanitized.email !== existingUser.email) {
        updateData.email = sanitized.email;
      }

      if (
        sanitized.displayName !== undefined &&
        sanitized.displayName !== existingUser.displayName
      ) {
        updateData.displayName = sanitized.displayName;
      }

      // 7. No actual changes detected
      if (Object.keys(updateData).length === 0) {
        return res.status(200).json({
          status: 200,
          message: 'No changes detected. Profile is already up to date.',
          data: {
            user: toSafeUser(existingUser),
          },
        });
      }

      // 8. Persist update
      const updatedUser = await prisma.user.update({
        where: { id: req.user.id },
        data: updateData,
      });

      // 9. Respond with safe user object
      return res.status(200).json({
        status: 200,
        message: 'Profile updated successfully.',
        data: {
          user: toSafeUser(updatedUser),
        },
      });
    } catch (err) {
      // Prisma unique constraint fallback (race condition)
      if (err.code === 'P2002' && err.meta?.target?.includes('email')) {
        return res.status(409).json({
          status: 409,
          error: 'Conflict',
          message: 'This email address is already in use.',
        });
      }

      // Prisma record not found on update
      if (err.code === 'P2025') {
        return res.status(404).json({
          status: 404,
          error: 'Not Found',
          message: 'User account not found.',
        });
      }

      console.error('[update-profile] Unexpected error:', err);

      return res.status(500).json({
        status: 500,
        error: 'Internal Server Error',
        message: 'An unexpected error occurred. Please try again later.',
      });
    }
  }
);

module.exports = router;
