const jwt = require('jsonwebtoken')
const { User } = require('../models')

const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

function sanitizeEmail(value) {
  return String(value || '').trim().toLowerCase()
}

function sanitizeDisplayName(value) {
  return String(value || '')
    .replace(/[\u0000-\u001F\u007F]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
}

module.exports = async function updateProfileHandler(req, res) {
  try {
    let userId = req.user?.id || req.user?._id || req.session?.userId
    if (!userId) {
      const auth = req.headers.authorization
      if (auth && auth.startsWith('Bearer ')) {
        const token = auth.slice(7).trim()
        if (token) {
          try {
            const payload = jwt.verify(token, process.env.JWT_SECRET)
            userId = payload.sub || payload.id || payload.userId
          } catch (e) {
            return res.status(401).json({ error: 'Unauthorized' })
          }
        }
      }
    }
    if (!userId) return res.status(401).json({ error: 'Unauthorized' })

    const body = req.body || {}
    const emailRaw = body.email
    const displayNameRaw = body.displayName

    if (emailRaw === undefined && displayNameRaw === undefined) {
      return res.status(400).json({ error: 'email or displayName required' })
    }

    const update = {}

    if (emailRaw !== undefined) {
      const email = sanitizeEmail(emailRaw)
      if (!emailRegex.test(email)) return res.status(400).json({ error: 'Invalid email' })
      const existing = await User.findOne({ email, _id: { $ne: userId } }).lean()
      if (existing) return res.status(409).json({ error: 'Email already in use' })
      update.email = email
    }

    if (displayNameRaw !== undefined) {
      const displayName = sanitizeDisplayName(displayNameRaw)
      if (displayName.length < 2 || displayName.length > 50) {
        return res.status(400).json({ error: 'Invalid displayName' })
      }
      update.displayName = displayName
    }

    if (Object.keys(update).length === 0) {
      return res.status(400).json({ error: 'No valid fields to update' })
    }

    const updatedUser = await User.findByIdAndUpdate(userId, update, { new: true, runValidators: true }).lean()
    if (!updatedUser) return res.status(404).json({ error: 'User not found' })

    const profile = {
      id: updatedUser._id?.toString ? updatedUser._id.toString() : updatedUser.id,
      email: updatedUser.email,
      displayName: updatedUser.displayName,
      createdAt: updatedUser.createdAt,
      updatedAt: updatedUser.updatedAt
    }

    return res.status(200).json({ user: profile })
  } catch (err) {
    if (err && err.code === 11000) {
      return res.status(409).json({ error: 'Email already in use' })
    }
    return res.status(500).json({ error: 'Internal server error' })
  }
}
