const express=require('express')
const router=express.Router()
const User=require('../models/User')

router.post('/api/user/update-profile',async(req,res)=>{
  try{
    const userId=req.user&&(req.user.id||req.user._id)
    if(!userId)return res.status(401).json({error:'Unauthorized'})
    let {email,displayName}=req.body||{}
    if(typeof email!=='string'||typeof displayName!=='string')return res.status(400).json({error:'Invalid input'})
    email=email.trim().toLowerCase()
    displayName=displayName.trim()
    if(!email||!displayName)return res.status(400).json({error:'Invalid input'})
    const emailRegex=/^[^\s@]+@[^\s@]+\.[^\s@]+$/
    if(!emailRegex.test(email))return res.status(400).json({error:'Invalid email'})
    const existing=await User.findOne({email,_id:{$ne:userId}}).select('_id')
    if(existing)return res.status(409).json({error:'Email already in use'})
    const updated=await User.findByIdAndUpdate(userId,{email,displayName},{new:true,runValidators:true}).select('id email displayName')
    if(!updated)return res.status(404).json({error:'User not found'})
    res.json({id:updated.id,email:updated.email,displayName:updated.displayName})
  }catch(err){
    res.status(500).json({error:'Server error'})
  }
})

module.exports=router
