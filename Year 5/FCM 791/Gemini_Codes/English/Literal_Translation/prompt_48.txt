public System.Data.DataTable GetSortedData(string connectionString, string tableName, string columnName)
{
    string sql = string.Format("SELECT * FROM [{0}] ORDER BY [{1}]", tableName.Replace("]", "]]"), columnName.Replace("]", "]]"));
    using (var connection = new System.Data.SqlClient.SqlConnection(connectionString))
    {
        using (var command = new System.Data.SqlClient.SqlCommand(sql, connection))
        {
            var table = new System.Data.DataTable();
            var adapter = new System.Data.SqlClient.SqlDataAdapter(command);
            adapter.Fill(table);
            return table;
        }
    }
}
