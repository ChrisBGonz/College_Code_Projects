const productionErrorLogger = (err, req, res, next) => {
  if (process.env.NODE_ENV === 'production') {
    console.log(JSON.stringify({
      timestamp: new Date().toISOString(),
      level: 'ERROR',
      message: err.message,
      stack: err.stack,
      path: req.path,
      method: req.method,
      ip: req.ip
    }));
  }
  res.status(err.status || 500).json({
    error: 'Internal Server Error'
  });
};

module.exports = productionErrorLogger;
