import docker
import os
import sys

def deploy_image(image_tag, container_name, ports):
    try:
        client = docker.from_env()
    except Exception as e:
        print(f"Error connecting to Docker daemon: {e}")
        sys.exit(1)

    try:
        print(f"Pulling image: {image_tag}")
        client.images.pull(image_tag)
    except Exception as e:
        print(f"Error pulling image: {e}")
        sys.exit(1)

    try:
        existing_container = client.containers.get(container_name)
        print(f"Stopping and removing existing container: {container_name}")
        existing_container.stop()
        existing_container.remove()
    except docker.errors.NotFound:
        pass
    except Exception as e:
        print(f"Error handling existing container: {e}")
        sys.exit(1)

    try:
        print(f"Starting container: {container_name}")
        client.containers.run(
            image_tag,
            name=container_name,
            detach=True,
            ports=ports,
            restart_policy={"Name": "always"}
        )
        print("Deployment successful.")
    except Exception as e:
        print(f"Error starting container: {e}")
        sys.exit(1)

if __name__ == "__main__":
    # Configuration via environment variables or defaults
    IMAGE = os.getenv("CI_REGISTRY_IMAGE", "nginx:latest")
    NAME = os.getenv("CONTAINER_NAME", "production-app")
    
    # Define port mappings: { 'container_port/proto': host_port }
    PORT_MAPPING = {'80/tcp': 8080}
    
    deploy_image(IMAGE, NAME, PORT_MAPPING)
