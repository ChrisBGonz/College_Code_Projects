import hmac
import hashlib
import time
import base64
import struct

def verify_totp(secret, code):
    try:
        # Normalize secret and add padding if necessary
        secret = secret.replace(" ", "").upper()
        secret += '=' * (8 - len(secret) % 8)
        key = base64.b32decode(secret, casefold=True)
    except Exception:
        return False

    # Check current, previous, and next 30-second intervals to account for clock drift
    current_interval = int(time.time() / 30)
    
    for i in [0, -1, 1]:
        counter = struct.pack(">Q", current_interval + i)
        hmac_hash = hmac.new(key, counter, hashlib.sha1).digest()
        
        # Dynamic truncation
        offset = hmac_hash[-1] & 0x0F
        truncated_hash = struct.unpack(">I", hmac_hash[offset:offset+4])[0] & 0x7FFFFFFF
        
        # Generate 6-digit code
        generated_code = str(truncated_hash % 1000000).zfill(6)
        
        if generated_code == str(code).strip():
            return True
            
    return False

if __name__ == "__main__":
    # Example usage
    # This secret is base32 encoded (standard for Google Authenticator)
    example_secret = "JBSWY3DPEHPK3PXP" 
    user_input = "123456" # Replace with actual user input
    
    if verify_totp(example_secret, user_input):
        print("OTP Verification Successful")
    else:
