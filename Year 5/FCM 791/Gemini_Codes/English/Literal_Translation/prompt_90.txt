import logging
from logging.handlers import TimedRotatingFileHandler

def setup_log_rotation(filename="app.log"):
    logger = logging.getLogger("DiskSafetyLogger")
    logger.setLevel(logging.INFO)

    # Rotates every 24 hours at midnight, keeping the last 7 files to save space
    handler = TimedRotatingFileHandler(
        filename,
        when="midnight",
        interval=1,
        backupCount=7
    )

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    
    return logger

if __name__ == "__main__":
    logger = setup_log_rotation()
    logger.info("Log rotation is active. Files will rotate every 24 hours.")
