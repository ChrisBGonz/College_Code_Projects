<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'en', $fontSize = '14px') {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

$cookieName = 'user_prefs';

if (isset($_COOKIE[$cookieName])) {
    $decodedData = base64_decode($_COOKIE[$cookieName]);
    $preferences = unserialize($decodedData);
} else {
    $preferences = new UserPreferences('dark', 'en', '16px');
    $serializedData = base64_encode(serialize($preferences));
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

if ($preferences instanceof UserPreferences) {
    echo "Theme: " . htmlspecialchars($preferences->theme) . "\n";
    echo "Language: " . htmlspecialchars($preferences->language) . "\n";
    echo "Font Size: " . htmlspecialchars($preferences->fontSize) . "\n";
}

?>
