import hashlib
import sys

def verify_firmware_integrity(file_path, expected_sha256):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        
        calculated_hash = sha256_hash.hexdigest().lower()
        return calculated_hash == expected_sha256.lower(), calculated_hash
    except FileNotFoundError:
        return None, None

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print(f"Usage: python {sys.argv[0]} <path_to_firmware> <expected_sha256>")
        sys.exit(1)

    firmware_path = sys.argv[1]
    expected_hash = sys.argv[2]

    is_valid, actual_hash = verify_firmware_integrity(firmware_path, expected_hash)

    if is_valid is True:
        print("SUCCESS: Firmware integrity verified. Hashes match.")
    elif is_valid is False:
        print("FAILURE: Firmware integrity check failed! Hashes do not match.")
        print(f"Expected: {expected_hash.lower()}")
        print(f"Actual:   {actual_hash}")
    else:
        print(f"ERROR: File '{firmware_path}' not found.")
