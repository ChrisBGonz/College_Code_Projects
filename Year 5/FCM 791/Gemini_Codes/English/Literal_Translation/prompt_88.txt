using System;
using Microsoft.Extensions.Logging;

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, string action, string mfaMethod, string ipAddress)
    {
        _logger.LogWarning(
            "MFA Configuration Change: User {UserId} performed action '{Action}' on method '{Method}' from IP {IPAddress} at {Timestamp}",
            userId,
            action,
            mfaMethod,
            ipAddress,
            DateTimeOffset.UtcNow
        );
    }
}
