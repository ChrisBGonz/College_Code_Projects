<?php
$uploadDir = 'shared_drive/uploads/';
$statusMsg = '';

if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_FILES['pdf_file'])) {
    if (!is_dir($uploadDir)) {
        mkdir($uploadDir, 0755, true);
    }

    $fileName = basename($_FILES['pdf_file']['name']);
    $targetFilePath = $uploadDir . $fileName;
    $fileType = pathinfo($targetFilePath, PATHINFO_EXTENSION);

    $allowTypes = array('pdf');
    if (in_array(strtolower($fileType), $allowTypes)) {
        if (move_uploaded_file($_FILES['pdf_file']['tmp_name'], $targetFilePath)) {
            $statusMsg = "The file " . htmlspecialchars($fileName) . " has been uploaded successfully.";
        } else {
            $statusMsg = "Sorry, there was an error uploading your file.";
        }
    } else {
        $statusMsg = 'Error: Only PDF files are allowed.';
    }
}
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Company Shared Drive - File Upload</title>
    <style>
        body { font-family: Arial, sans-serif; display: flex; justify-content: center; align-items: center; height: 100vh; background-color: #f4f4f9; margin: 0; }
        .upload-container { background: #fff; padding: 30px; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); width: 100%; max-width: 400px; }
        h2 { margin-top: 0; color: #333; font-size: 1.5rem; }
        .status { margin-bottom: 15px; padding: 10px; border-radius: 4px; background-color: #e7f3fe; color: #31708f; border: 1px solid #bcdff1; font-size: 0.9rem; }
        input[type="file"] { display: block; margin-bottom: 20px; width: 100%; }
        input[type="submit"] { background-color: #0056b3; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; width: 100%; font-size: 1rem; }
        input[type="submit"]:hover { background-color: #004494; }
    </style>
</head>
<body>
    <div class="upload-container">
        <h2>Upload PDF to Shared Drive</h2>
        <?php if(!empty($statusMsg)): ?>
            <div class="status"><?php echo $statusMsg; ?></div>
        <?php endif; ?>
        <form action="" method="post" enctype="multipart/form-data">
            <label for="pdf_file">Select PDF Document:</label>
            <input type="file" name="pdf_file" id="pdf_file" accept=".pdf" required>
            <input type="submit" name="submit" value="Upload to Drive">
        </form>
    </div>
</body>
</html>
