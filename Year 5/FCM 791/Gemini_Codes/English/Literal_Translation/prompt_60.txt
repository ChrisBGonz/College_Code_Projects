const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());
app.set('trust proxy', true);

const poll = {
    question: "What is your favorite programming language?",
    options: {
        "JavaScript": 0,
        "Python": 0,
        "Rust": 0,
        "Go": 0
    },
    votedIPs: new Set()
};

app.get('/poll', (req, res) => {
    res.json({
        question: poll.question,
        results: poll.options,
        totalVotes: poll.votedIPs.size
    });
});

app.post('/vote', (req, res) => {
    const { option } = req.body;
    const userIP = req.ip || req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (!option || !poll.options.hasOwnProperty(option)) {
        return res.status(400).json({ error: "Invalid option selected." });
    }

    if (poll.votedIPs.has(userIP)) {
        return res.status(403).json({ error: "You have already voted from this IP address." });
    }

    poll.options[option]++;
    poll.votedIPs.add(userIP);

    res.json({
        message: "Vote recorded successfully.",
        results: poll.options
    });
});

app.listen(PORT, () => {
    console.log(`Voting system running on http://localhost:${PORT}`);
});
