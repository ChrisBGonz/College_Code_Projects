import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseManager {
    private static final Logger logger = Logger.getLogger(DatabaseManager.class.getName());

    public void executeDatabaseTask() throws CustomDatabaseException {
        try {
            // Database operation logic goes here
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "SQL Error: {0}", e.getMessage());
            throw new CustomDatabaseException("A database error occurred while processing the request.", e);
        }
    }
}

class CustomDatabaseException extends Exception {
    public CustomDatabaseException(String message, Throwable cause) {
        super(message, cause);
    }
}
