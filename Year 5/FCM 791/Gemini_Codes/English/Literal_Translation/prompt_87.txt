import time
import os

LOG_FILE = "auth.log"
THRESHOLD = 10
WINDOW = 60

def monitor():
    if not os.path.exists(LOG_FILE):
        open(LOG_FILE, 'a').close()
    
    with open(LOG_FILE, 'r') as f:
        f.seek(0, 2)
        attempts = []
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue
            
            if "Failed Login" in line:
                now = time.time()
                attempts.append(now)
                attempts = [t for t in attempts if now - t <= WINDOW]
                if len(attempts) > THRESHOLD:
                    print("WARNING: More than 10 failed logins detected in one minute.")
                    attempts = []

if __name__ == "__main__":
    monitor()
