const STORAGE_KEY = 'multi_step_form_data';

function initRegistrationForm(formId) {
    const form = document.getElementById(formId);
    if (!form) return;

    const loadPersistedData = () => {
        const rawData = localStorage.getItem(STORAGE_KEY);
        if (!rawData) return;

        const data = JSON.parse(rawData);
        Object.keys(data).forEach(name => {
            const field = form.elements[name];
            if (!field) return;

            if (field.type === 'checkbox') {
                field.checked = !!data[name];
            } else if (field.type === 'radio') {
                const radio = Array.from(form.elements[name]).find(r => r.value === data[name]);
                if (radio) radio.checked = true;
            } else {
                field.value = data[name];
            }
        });
    };

    const persistData = () => {
        const formData = new FormData(form);
        const data = {};
        formData.forEach((value, key) => {
            data[key] = value;
        });
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        const finalData = JSON.parse(localStorage.getItem(STORAGE_KEY));
        
        fetch('/api/register', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(finalData)
        })
        .then(response => {
            if (response.ok) {
                localStorage.removeItem(STORAGE_KEY);
                form.reset();
                alert('Registration Successful');
            }
        })
        .catch(err => console.error('Submission failed', err));
    };

    form.querySelectorAll('input, select, textarea').forEach(input => {
        input.addEventListener('input', persistData);
        input.addEventListener('change', persistData);
    });

    form.addEventListener('submit', handleSubmit);

    loadPersistedData();
}

document.addEventListener('DOMContentLoaded', () => {
    initRegistrationForm('registration-form');
});
