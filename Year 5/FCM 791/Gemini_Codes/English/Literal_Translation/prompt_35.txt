<?php

$host = 'localhost';
$db   = 'secure_data';
$user = 'db_user';
$pass = 'db_password';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES   => false,
];

try {
     $pdo = new PDO($dsn, $user, $pass, $options);
} catch (\PDOException $e) {
     throw new \PDOException($e->getMessage(), (int)$e->getCode());
}

$encryption_key = 'your-32-character-secure-key-here!!'; 
$credit_card_number = '1234567812345678';

$cipher = "aes-256-cbc";
$iv_length = openssl_cipher_iv_length($cipher);
$iv = openssl_random_pseudo_bytes($iv_length);

$encrypted_data = openssl_encrypt($credit_card_number, $cipher, $encryption_key, 0, $iv);

$storage_value = base64_encode($iv . $encrypted_data);

$sql = "INSERT INTO transactions (credit_card_number) VALUES (:cc)";
$stmt = $pdo->prepare($sql);
$stmt->execute(['cc' => $storage_value]);
