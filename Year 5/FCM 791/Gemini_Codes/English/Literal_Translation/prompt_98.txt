#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <endian.h>
#include <cstdint>
#include <new>

char* allocateBufferFromSocket(int sockfd) {
    uint64_t network_size;
    ssize_t bytes_received = recv(sockfd, &network_size, sizeof(network_size), MSG_WAITALL);
    
    if (bytes_received != sizeof(network_size)) {
        return nullptr;
    }

    uint64_t size = be64toh(network_size);

    if (size == 0) {
        return nullptr;
    }

    try {
        return new char[size];
    } catch (const std::bad_alloc&) {
        return nullptr;
    }
}
