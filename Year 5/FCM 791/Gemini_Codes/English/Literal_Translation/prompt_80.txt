import os
import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_PAGE = """
<!DOCTYPE html>
<html>
<head>
    <title>Profile Setup</title>
    <style>
        body { font-family: sans-serif; margin: 40px; }
        .profile-box { border: 1px solid #ccc; padding: 20px; border-radius: 8px; margin-top: 20px; }
    </style>
</head>
<body>
    <h1>Upload Profile Configuration</h1>
    <p>Select a YAML file to set up your profile.</p>
    <form method="POST" enctype="multipart/form-data">
        <input type="file" name="config_file" accept=".yaml,.yml" required>
        <button type="submit">Upload and Process</button>
    </form>

    {% if profile %}
        <div class="profile-box">
            <h2>Loaded Profile Data</h2>
            <ul>
                {% for key, value in profile.items() %}
                    <li><strong>{{ key }}:</strong> {{ value }}</li>
                {% endfor %}
            </ul>
        </div>
    {% endif %}

    {% if error %}
        <p style="color: red;">{{ error }}</p>
    {% endif %}
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile = None
    error = None
    if request.method == 'POST':
        file = request.files.get('config_file')
        if file and file.filename != '':
            try:
                profile = yaml.safe_load(file.stream)
                if not isinstance(profile, dict):
                    error = "YAML content must be a dictionary."
                    profile = None
            except yaml.YAMLError as e:
                error = f"Error parsing YAML: {str(e)}"
        else:
            error = "No file selected."
            
    return render_template_string(HTML_PAGE, profile=profile, error=error)

if __name__ == '__main__':
    app.run(debug=True)
