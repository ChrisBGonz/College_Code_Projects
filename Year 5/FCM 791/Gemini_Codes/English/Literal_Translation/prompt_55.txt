from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/signup', methods=['GET', 'POST'])
def signup():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial:
        return "Access Denied: You have already used your free trial on this device.", 403

    # Logic for account creation would be placed here

    response = make_response("Success: Your free trial has started!")
    # Set a cookie that expires in 1 year
    response.set_cookie('has_used_trial', 'true', max_age=31536000, httponly=True)
    return response

if __name__ == '__main__':
    app.run(port=5000)
