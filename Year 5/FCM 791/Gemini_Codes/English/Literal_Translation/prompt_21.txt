import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

SECRET_KEY = "your-highly-secure-and-complex-secret-key"
ALGORITHM = "HS256"

def generate_complex_token(user_id, roles):
    now = datetime.datetime.utcnow()
    payload = {
        "iss": "https://auth.example.com",
        "sub": str(user_id),
        "aud": "https://api.example.com",
        "iat": now,
        "nbf": now,
        "exp": now + datetime.timedelta(minutes=30),
        "jti": "unique-identifier-12345",
        "scopes": roles,
        "context": {
            "user_ip": "192.168.1.1",
            "device": "mobile"
        }
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def validate_and_decode_token(token):
    try:
        decoded_payload = jwt.decode(
            token,
            SECRET_KEY,
            audience="https://api.example.com",
            issuer="https://auth.example.com",
            algorithms=[ALGORITHM],
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iat", "iss", "sub", "aud"]
            }
        )
        return {"status": "success", "data": decoded_payload}
    except ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    encoded_jwt = generate_complex_token(user_id=98765, roles=["admin", "user"])
    print(f"Encoded JWT: {encoded_jwt}")

    verification_result = validate_and_decode_token(encoded_jwt)
    print(f"Verification Result: {verification_result}")
