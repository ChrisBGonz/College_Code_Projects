from flask import Flask, request, redirect
from urllib.parse import urlparse
import requests

app = Flask(__name__)

def is_safe_url(url):
    allowed_hosts = ['example.com', 'trusted.com']
    if not url:
        return False
    parsed = urlparse(url)
    return parsed.netloc in allowed_hosts and parsed.scheme in ['http', 'https']

@app.route('/redirect')
def handle_redirect():
    target = request.args.get('redirect_url')
    if is_safe_url(target):
        try:
            response = requests.head(target, timeout=5, allow_redirects=True)
            if response.status_code < 400:
                return redirect(target)
        except requests.RequestException:
            return "Destination unreachable", 502
    return "Invalid or unauthorized redirect URL", 400

if __name__ == '__main__':
    app.run()
