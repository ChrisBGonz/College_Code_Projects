def promote
  @user = User.find(params[:id])

  if current_user.role == 'Manager' && @user.role == 'Staff'
    if @user.update(role: 'Manager')
      redirect_to @user, notice: 'Staff member successfully promoted to Manager.'
    else
      redirect_to @user, alert: 'Unable to promote staff member.'
    end
  else
    redirect_to users_path, alert: 'Unauthorized action or invalid user role.'
  end
end
