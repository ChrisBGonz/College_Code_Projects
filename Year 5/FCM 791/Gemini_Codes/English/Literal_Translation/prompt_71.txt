import pickle

class Player:
    def __init__(self, name, health, inventory):
        self.name = name
        self.health = health
        self.inventory = inventory

class GameState:
    def __init__(self, player, level, metadata):
        self.player = player
        self.level = level
        self.metadata = metadata

def save_game(game_state, filename="savegame.pkl"):
    try:
        with open(filename, "wb") as file:
            pickle.dump(game_state, file)
        return True
    except Exception as e:
        return False

def load_game(filename="savegame.pkl"):
    try:
        with open(filename, "rb") as file:
            return pickle.load(file)
    except Exception as e:
        return None

if __name__ == "__main__":
    player = Player("Aragorn", 100, ["Sword", "Elven Bread", "Ring"])
    original_state = GameState(player, 5, {"difficulty": "Hard", "playtime": 3600})

    save_game(original_state)

    loaded_state = load_game()

    if loaded_state:
        print(f"Loaded Player: {loaded_state.player.name}")
        print(f"Health: {loaded_state.player.health}")
        print(f"Inventory: {loaded_state.player.inventory}")
        print(f"Level: {loaded_state.level}")
        print(f"Metadata: {loaded_state.metadata}")
