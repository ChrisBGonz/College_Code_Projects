import socket

def xor_crypt(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def communicate(host='127.0.0.1', port=9999, key=b'legacy_key'):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            payload = b"GET_STATUS"
            encrypted_payload = xor_crypt(payload, key)
            s.sendall(encrypted_payload)
            
            response = s.recv(4096)
            if response:
                decrypted_response = xor_crypt(response, key)
                print(decrypted_response.decode(errors='ignore'))
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    communicate()
