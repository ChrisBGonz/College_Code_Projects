package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

func CreatePasswordResetToken() (string, time.Time, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", time.Time{}, err
	}
	token := hex.EncodeToString(b)
	expiry := time.Now().Add(1 * time.Hour)
	return token, expiry, nil
}
