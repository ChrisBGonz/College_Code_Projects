<?php
session_start();

if (!isset($_SESSION['is_admin']) || $_SESSION['is_admin'] !== true) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'N/A';
    $uri = $_SERVER['REQUEST_URI'] ?? 'N/A';
    $timestamp = date('Y-m-d H:i:s');

    $adminEmail = 'admin@example.com';
    $subject = 'SECURITY ALERT: Unauthorized Admin Access Attempt';
    $message = "Unauthorized access attempt detected.\n\nIP Address: $ip\nRequested URI: $uri\nTimestamp: $timestamp";
    $headers = 'From: security-alerts@example.com';
    mail($adminEmail, $subject, $message, $headers);

    $slackWebhookUrl = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';
    $slackData = json_encode([
        'text' => "🚨 *Unauthorized Admin Access Attempt*\n*IP:* $ip\n*URI:* $uri\n*Time:* $timestamp"
    ]);
    
    $ch = curl_init($slackWebhookUrl);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'POST');
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slackData);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($slackData)
    ]);
    curl_exec($ch);
    curl_close($ch);

    header('HTTP/1.1 403 Forbidden');
    exit('Access Denied: Unauthorized access has been logged and reported.');
}
