<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'name' => 'secure_app',
    'user' => 'db_user',
    'pass' => 'db_password'
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['name']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Connection failed");
}

$message = "";
$step = 1;

if (isset($_SESSION['recovery_username']) && isset($_SESSION['recovery_question'])) {
    $step = 2;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['action']) && $_POST['action'] === 'find_user') {
        $username = trim($_POST['username'] ?? '');
        $stmt = $pdo->prepare("SELECT secret_question FROM users WHERE username = ? LIMIT 1");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_username'] = $username;
            $_SESSION['recovery_question'] = $user['secret_question'];
            $step = 2;
        } else {
            $message = "User not found.";
        }
    } elseif (isset($_POST['action']) && $_POST['action'] === 'verify_answer') {
        $answer = strtolower(trim($_POST['answer'] ?? ''));
        $username = $_SESSION['recovery_username'] ?? '';

        $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE username = ? LIMIT 1");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify($answer, $user['secret_answer'])) {
            $_SESSION['recovery_verified'] = true;
            header("Location: reset-password.php");
            exit;
        } else {
            $message = "Incorrect answer. Please try again.";
            $step = 2;
        }
    } elseif (isset($_POST['action']) && $_POST['action'] === 'reset_flow') {
        session_destroy();
        header("Location: " . $_SERVER['PHP_SELF']);
        exit;
    }
}
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Account Recovery</title>
    <style>
        body { font-family: sans-serif; display: flex; justify-content: center; padding-top: 50px; }
        .card { border: 1px solid #ccc; padding: 20px; border-radius: 8px; width: 300px; }
        .error { color: red; font-size: 0.9em; margin-bottom: 10px; }
        input { width: 100%; padding: 8px; margin: 10px 0; box-sizing: border-box; }
        button { width: 100%; padding: 10px; cursor: pointer; }
        .reset-link { font-size: 0.8em; margin-top: 10px; display: block; text-align: center; color: blue; text-decoration: none; border: none; background: none; }
    </style>
</head>
<body>
    <div class="card">
        <h2>Recover Account</h2>
        
        <?php if ($message): ?>
            <div class="error"><?php echo htmlspecialchars($message); ?></div>
        <?php endif; ?>

        <?php if ($step === 1): ?>
            <form method="POST">
                <input type="hidden" name="action" value="find_user">
                <label for="username">Enter your username:</label>
                <input type="text" name="username" id="username" required autofocus>
                <button type="submit">Continue</button>
            </form>
        <?php else: ?>
            <form method="POST">
                <input type="hidden" name="action" value="verify_answer">
                <p><strong>Question:</strong><br><?php echo htmlspecialchars($_SESSION['recovery_question']); ?></p>
                <label for="answer">Your Answer:</label>
                <input type="text" name="answer" id="answer" required autofocus autocomplete="off">
                <button type="submit">Verify Answer</button>
            </form>
            <form method="POST">
                <input type="hidden" name="action" value="reset_flow">
                <button type="submit" class="reset-link">Start over</button>
            </form>
        <?php endif; ?>
    </div>
</body>
</html>
