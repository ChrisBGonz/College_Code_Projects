#!/bin/bash

# Configuration - Replace these with actual values
TOOL_NAME="security-tool"
DOWNLOAD_URL="https://example.com/downloads/latest/tool-linux-amd64.tar.gz"
INSTALL_DIR="/usr/local/bin"

# Ensure the script is run with necessary permissions for installation
if [ "$EUID" -ne 0 ]; then 
  echo "Please run as root or using sudo"
  exit 1
fi

# Create a temporary working directory
TMP_DIR=$(mktemp -d)
cd "$TMP_DIR" || exit 1

echo "Downloading the latest version of $TOOL_NAME..."
if curl -sSL "$DOWNLOAD_URL" -o "$TOOL_NAME.tar.gz"; then
    echo "Download successful."
else
    echo "Error: Failed to download the tool."
    rm -rf "$TMP_DIR"
    exit 1
fi

echo "Extracting files..."
tar -xzf "$TOOL_NAME.tar.gz"

# Locate the binary (assumes the binary is named $TOOL_NAME or is the only executable)
BINARY=$(find . -type f -executable -print -quit)

if [ -n "$BINARY" ]; then
    echo "Installing $TOOL_NAME to $INSTALL_DIR..."
    mv "$BINARY" "$INSTALL_DIR/$TOOL_NAME"
    chmod +x "$INSTALL_DIR/$TOOL_NAME"
    echo "Installation complete. You can now run '$TOOL_NAME'."
else
    echo "Error: Could not find an executable binary in the package."
    exit 1
fi

# Cleanup
cd /
rm -rf "$TMP_DIR"
