import json

class MultiTenantSaaS:
    def __init__(self):
        self._tenants = {
            "T001": {
                "name": "Acme Corp",
                "annual_report": "2023 Annual Report: Revenue $50M, Net Profit $5M."
            },
            "T002": {
                "name": "Globex Corporation",
                "annual_report": "2023 Annual Report: Revenue $120M, Net Profit $12M."
            },
            "T003": {
                "name": "Soylent Corp",
                "annual_report": "2023 Annual Report: Revenue $30M, Net Profit $2M."
            }
        }
        
        self._users = {
            "user_alice": {"company_id": "T001", "name": "Alice Smith"},
            "user_bob": {"company_id": "T002", "name": "Bob Jones"},
            "user_charlie": {"company_id": "T001", "name": "Charlie Brown"}
        }

    def get_annual_report(self, user_id, company_id):
        user = self._users.get(user_id)
        
        if not user:
            return json.dumps({"error": "Authentication failed: User not found."})

        if user["company_id"] != company_id:
            return json.dumps({
                "error": "Authorization failed: Access denied to tenant data.",
                "user": user_id,
                "requested_tenant": company_id
            })

        tenant_data = self._tenants.get(company_id)
        if not tenant_data:
            return json.dumps({"error": "Data error: Company record not found."})

        return json.dumps({
            "status": "success",
            "company_name": tenant_data["name"],
            "report": tenant_data["annual_report"]
        })

if __name__ == "__main__":
    app = MultiTenantSaaS()

    # Case 1: Authorized access (Alice belongs to T001)
    print("Request 1 (Alice -> T001):")
    print(app.get_annual_report("user_alice", "T001"))

    # Case 2: Unauthorized access (Alice tries to access T002)
    print("\nRequest 2 (Alice -> T002):")
    print(app.get_annual_report("user_alice", "T002"))

    # Case 3: Authorized access (Bob belongs to T002)
    print("\nRequest 3 (Bob -> T002):")
    print(app.get_annual_report("user_bob", "T002"))

    # Case 4: Non-existent user
    print("\nRequest 4 (Unknown -> T001):")
    print(app.get_annual_report("user_unknown", "T001"))
