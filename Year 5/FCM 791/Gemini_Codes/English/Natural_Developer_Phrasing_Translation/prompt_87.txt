import time
import collections

def monitor_log(file_path):
    try:
        with open(file_path, 'r') as f:
            f.seek(0, 2)
            failed_attempts = collections.deque()
            while True:
                line = f.readline()
                if not line:
                    time.sleep(0.1)
                    continue
                if "Failed Login" in line:
                    current_time = time.time()
                    failed_attempts.append(current_time)
                    while failed_attempts and failed_attempts[0] < current_time - 60:
                        failed_attempts.popleft()
                    if len(failed_attempts) > 10:
                        print(f"WARNING: {len(failed_attempts)} Failed Login attempts detected in the last 60 seconds.")
    except KeyboardInterrupt:
        pass
    except FileNotFoundError:
        print(f"Error: {file_path} not found.")

if __name__ == "__main__":
    monitor_log("auth.log")
