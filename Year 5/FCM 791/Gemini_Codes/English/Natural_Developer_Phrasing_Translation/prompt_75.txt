<?php

class UserPreferences {
    public $theme;
    public $language;
    public $notifications_enabled;

    public function __construct($theme = 'light', $language = 'en', $notifications_enabled = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->notifications_enabled = $notifications_enabled;
    }
}

$cookieName = 'user_settings_bundle';

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $preferences = unserialize($serializedData);
} else {
    $preferences = new UserPreferences();
    $serializedData = serialize($preferences);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

header('Content-Type: application/json');
echo json_encode([
    'status' => 'success',
    'source' => isset($_COOKIE[$cookieName]) ? 'cookie' : 'default',
    'preferences' => [
        'theme' => $preferences->theme,
        'language' => $preferences->language,
        'notifications' => $preferences->notifications_enabled
    ]
]);
