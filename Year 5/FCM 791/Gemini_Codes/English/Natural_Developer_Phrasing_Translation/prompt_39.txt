import socket
from itertools import cycle

class LegacyClient:
    def __init__(self, host, port, key):
        self.host = host
        self.port = port
        self.key = key.encode() if isinstance(key, str) else key
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

    def _xor_crypt(self, data):
        return bytes(b ^ k for b, k in zip(data, cycle(self.key)))

    def connect(self):
        self.sock.connect((self.host, self.port))

    def send(self, message):
        if isinstance(message, str):
            message = message.encode()
        payload = self._xor_crypt(message)
        self.sock.sendall(payload)

    def receive(self, buffer_size=4096):
        data = self.sock.recv(buffer_size)
        if not data:
            return None
        return self._xor_crypt(data)

    def close(self):
        self.sock.close()

if __name__ == "__main__":
    HOST = "127.0.0.1"
    PORT = 8888
    XOR_KEY = "legacy_secret_key"

    client = LegacyClient(HOST, PORT, XOR_KEY)
    try:
        client.connect()
        client.send("GET_STATUS")
        response = client.receive()
        if response:
            print(f"Received: {response.decode(errors='replace')}")
    except Exception as e:
        print(f"Error: {e}")
    finally:
        client.close()
