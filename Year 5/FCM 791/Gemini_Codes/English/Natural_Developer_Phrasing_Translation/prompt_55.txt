from flask import Flask, request, redirect, make_response

app = Flask(__name__)

@app.route('/signup-trial', methods=['GET', 'POST'])
def signup_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return redirect('/trial-limit-reached')

    if request.method == 'POST':
        # Logic for account creation and trial activation would occur here
        
        response = make_response(redirect('/dashboard'))
        response.set_cookie(
            'has_used_trial', 
            'true', 
            max_age=31536000, 
            httponly=True, 
            samesite='Lax'
        )
        return response

    return '''
    <form method="post">
        <h1>Start Your Free Trial</h1>
        <input type="email" name="email" placeholder="Enter your email" required>
        <button type="submit">Sign Up</button>
    </form>
    '''

@app.route('/trial-limit-reached')
def trial_limit_reached():
    return "Error: You have already redeemed a free trial on this device.", 403

@app.route('/dashboard')
def dashboard():
    return "Welcome to your free trial dashboard!"

if __name__ == '__main__':
    app.run(debug=True)
