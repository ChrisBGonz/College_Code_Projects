import hashlib
import sys

def compute_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(65536), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def validate_firmware(file_path, expected_hash):
    calculated_hash = compute_sha256(file_path)
    if calculated_hash is None:
        return False, "FILE_NOT_FOUND"
    
    if calculated_hash.lower() == expected_hash.lower():
        return True, calculated_hash
    else:
        return False, calculated_hash

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print(f"Usage: {sys.argv[0]} <firmware_path> <expected_sha256>")
        sys.exit(1)

    firmware_path = sys.argv[1]
    expected_checksum = sys.argv[2]

    is_valid, result_value = validate_firmware(firmware_path, expected_checksum)

    if is_valid:
        print("Verification Successful: Firmware integrity confirmed.")
        print(f"SHA-256: {result_value}")
        sys.exit(0)
    else:
        print("Verification Failed: Firmware may be corrupted or tampered with.")
        if result_value != "FILE_NOT_FOUND":
            print(f"Expected: {expected_checksum.lower()}")
            print(f"Actual:   {result_value.lower()}")
        else:
            print("Error: Firmware file not found.")
        sys.exit(1)
