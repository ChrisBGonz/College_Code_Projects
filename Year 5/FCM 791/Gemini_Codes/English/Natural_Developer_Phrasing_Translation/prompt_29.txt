#!/bin/bash
set -e

# Ensure the versions-maven-plugin is used to update the parent POM version
mvn versions:update-parent -DgenerateBackupPoms=false -B

# Update all properties that define dependency versions to the latest release
mvn versions:update-properties -DgenerateBackupPoms=false -B

# Update all direct dependencies to the latest release versions
mvn versions:use-latest-releases -DgenerateBackupPoms=false -B

# Update all dependencies defined in the dependencyManagement section (handles transitives if managed)
mvn versions:use-latest-versions -DgenerateBackupPoms=false -Dincludes=*:* -B

# Display updates to verify the changes in logs
mvn versions:display-dependency-updates -B
