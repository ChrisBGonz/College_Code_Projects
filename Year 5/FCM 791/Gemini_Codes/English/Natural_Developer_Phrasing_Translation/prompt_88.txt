using System;
using System.Text.Json;
using Microsoft.Extensions.Logging;

namespace Security.Auditing
{
    public enum MfaActionType
    {
        Enable,
        Disable,
        AddFactor,
        RemoveFactor,
        ChangeDefault
    }

    public record MfaAuditLogEntry(
        string UserId,
        DateTimeOffset Timestamp,
        MfaActionType Action,
        object OldSettings,
        object NewSettings,
        string IpAddress,
        string UserAgent,
        string CorrelationId
    );

    public interface IMfaAuditLogger
    {
        void LogMfaUpdate(
            string userId, 
            MfaActionType action, 
            object oldSettings, 
            object newSettings, 
            string ipAddress, 
            string userAgent, 
            string correlationId);
    }

    public class MfaAuditLogger : IMfaAuditLogger
    {
        private readonly ILogger<MfaAuditLogger> _logger;

        public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
        {
            _logger = logger;
        }

        public void LogMfaUpdate(
            string userId,
            MfaActionType action,
            object oldSettings,
            object newSettings,
            string ipAddress,
            string userAgent,
            string correlationId)
        {
            var entry = new MfaAuditLogEntry(
                UserId: userId,
                Timestamp: DateTimeOffset.UtcNow,
                Action: action,
                OldSettings: oldSettings,
                NewSettings: newSettings,
                IpAddress: ipAddress,
                UserAgent: userAgent,
                CorrelationId: correlationId
            );

            _logger.LogInformation(
                "MFA_AUDIT_EVENT: User {UserId} performed {Action}. Context: {AuditData}",
                entry.UserId,
                entry.Action,
                JsonSerializer.Serialize(entry, new JsonSerializerOptions { WriteIndented = false })
            );
        }
    }
}
