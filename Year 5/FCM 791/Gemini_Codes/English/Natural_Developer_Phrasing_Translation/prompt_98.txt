#include <sys/socket.h>
#include <arpa/inet.h>
#include <cstdint>
#include <new>

char* allocateBufferFromSocket(int sockfd) {
    uint32_t network_len;
    if (recv(sockfd, &network_len, sizeof(network_len), 0) != sizeof(network_len)) {
        return nullptr;
    }

    uint32_t host_len = ntohl(network_len);

    try {
        return new char[host_len];
    } catch (const std::bad_alloc& e) {
        return nullptr;
    }
}
