import sys
from lxml import etree

def extract_title(xml_file_path):
    try:
        parser = etree.XMLParser(remove_blank_text=True, recover=True)
        tree = etree.parse(xml_file_path, parser=parser)
        
        # Using XPath to find all <title> elements
        titles = tree.xpath('//title')
        
        for title in titles:
            if title.text:
                print(title.text.strip())
            else:
                # Handle cases where title might have nested tags
                print(etree.tostring(title, method="text", encoding="unicode").strip())
                
    except Exception as e:
        print(f"Error processing XML: {e}")

if __name__ == "__main__":
    if len(sys.argv) > 1:
        extract_title(sys.argv[1])
    else:
        print("Usage: python script.py <path_to_xml_file>")
