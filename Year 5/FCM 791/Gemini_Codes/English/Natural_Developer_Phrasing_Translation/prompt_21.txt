import jwt
from jwt import PyJWKClient

def authenticate_jwt(token, jwks_url, expected_issuer, expected_audience):
    """
    Validates a JWT against a JWKS endpoint, verifying signature and standard claims.
    """
    try:
        # PyJWKClient handles fetching and caching the JWKS
        jwks_client = PyJWKClient(jwks_url)
        
        # Retrieve the signing key that matches the 'kid' in the token header
        signing_key = jwks_client.get_signing_key_from_jwt(token)
        
        # Decode and validate the token
        # This checks: signature, exp, nbf, iss, and aud
        payload = jwt.decode(
            token,
            signing_key.key,
            algorithms=["RS256"],
            audience=expected_audience,
            issuer=expected_issuer,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iss", "aud"]
            }
        )
        return {"status": "success", "payload": payload}

    except jwt.ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except jwt.ImmatureSignatureError:
        return {"status": "error", "message": "Token is not yet valid (nbf)"}
    except jwt.InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except jwt.InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except jwt.InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}
    except Exception as e:
        return {"status": "error", "message": f"Authentication failed: {str(e)}"}

if __name__ == "__main__":
    # Configuration for a typical OIDC provider (e.g., Auth0, Okta, AWS Cognito)
    JWKS_URL = "https://YOUR_DOMAIN/.well-known/jwks.json"
    ISSUER = "https://YOUR_DOMAIN/"
    AUDIENCE = "YOUR_API_IDENTIFIER"
    
    # Example token string
    SAMPLE_TOKEN = "eyJhbGciOiJSUzI1NiIsInR5cCI6IkpXVCIsImtpZCI6In..."

    # Perform validation
    auth_result = authenticate_jwt(SAMPLE_TOKEN, JWKS_URL, ISSUER, AUDIENCE)
    
    if auth_result["status"] == "success":
        print("Token is valid. Payload:")
        print(auth_result["payload"])
    else:
        print(f"Authentication failed: {auth_result['message']}")
