using Microsoft.AspNetCore.Http;
using System;

public class AuthenticationService
{
    public void IssuePersistentAuthCookie(HttpContext context, string cookieName, string tokenValue, int daysValid)
    {
        CookieOptions options = new CookieOptions
        {
            Expires = DateTimeOffset.UtcNow.AddDays(daysValid),
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            IsEssential = true
        };

        context.Response.Cookies.Append(cookieName, tokenValue, options);
    }
}
