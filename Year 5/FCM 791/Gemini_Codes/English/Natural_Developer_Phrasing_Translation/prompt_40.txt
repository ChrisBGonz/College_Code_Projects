const cookieEncryption = require('cookie-encryption');

module.exports = function(secret) {
  const vault = cookieEncryption(secret);

  return (req, res, next) => {
    if (req.cookies) {
      Object.keys(req.cookies).forEach(key => {
        try {
          const decrypted = vault.decrypt(req.cookies[key]);
          if (decrypted) {
            req.cookies[key] = decrypted;
          }
        } catch (e) {
          // If decryption fails, keep original value
        }
      });
    }

    const _cookie = res.cookie;
    res.cookie = function(name, value, options) {
      let val = value;
      try {
        if (typeof val === 'object' && val !== null) {
          val = JSON.stringify(val);
        }
        if (typeof val === 'string') {
          val = vault.encrypt(val);
        }
      } catch (e) {
        // If encryption fails, proceed with original value
      }
      return _cookie.call(this, name, val, options);
    };

    next();
  };
};
