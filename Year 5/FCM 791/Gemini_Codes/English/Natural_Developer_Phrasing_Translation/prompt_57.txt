<?php
session_start();

if (!isset($_SESSION['employee_id'])) {
    header('HTTP/1.1 403 Forbidden');
    exit('Unauthorized');
}

define('UPLOAD_DIR', '/var/www/shared_drive/uploads/');
define('MAX_SIZE', 10 * 1024 * 1024);
define('ALLOWED_MIME', 'application/pdf');

$db = new PDO('sqlite:/var/www/shared_drive/metadata.db');
$db->exec("CREATE TABLE IF NOT EXISTS uploads (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    uploader_id TEXT,
    original_name TEXT,
    stored_path TEXT,
    upload_time DATETIME DEFAULT CURRENT_TIMESTAMP
)");

$message = '';
$status = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['document'])) {
    try {
        $file = $_FILES['document'];

        if ($file['error'] !== UPLOAD_ERR_OK) {
            throw new Exception('Upload error code: ' . $file['error']);
        }

        if ($file['size'] > MAX_SIZE) {
            throw new Exception('File exceeds 10MB limit.');
        }

        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mime = $finfo->file($file['tmp_name']);
        $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));

        if ($mime !== ALLOWED_MIME || $ext !== 'pdf') {
            throw new Exception('Only PDF documents are allowed.');
        }

        $originalName = basename($file['name']);
        $safeName = bin2hex(random_bytes(16)) . '.pdf';
        $destination = UPLOAD_DIR . $safeName;

        if (!is_dir(UPLOAD_DIR)) {
            mkdir(UPLOAD_DIR, 0755, true);
        }

        if (move_uploaded_file($file['tmp_name'], $destination)) {
            $stmt = $db->prepare("INSERT INTO uploads (uploader_id, original_name, stored_path) VALUES (?, ?, ?)");
            $stmt->execute([$_SESSION['employee_id'], $originalName, $destination]);
            
            $status = 'success';
            $message = 'File uploaded successfully.';
        } else {
            throw new Exception('Failed to move uploaded file.');
        }
    } catch (Exception $e) {
        $status = 'error';
        $message = $e->getMessage();
    }
}
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Company Document Upload</title>
    <style>
        body { font-family: sans-serif; padding: 20px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .alert { padding: 10px; margin-bottom: 20px; border-radius: 4px; }
        .success { background: #d4edda; color: #155724; border: 1px solid #c3e6cb; }
        .error { background: #f8d7da; color: #721c24; border: 1px solid #f5c6cb; }
        input[type="file"] { display: block; margin: 20px 0; }
        button { background: #007bff; color: white; border: none; padding: 10px 15px; border-radius: 4px; cursor: pointer; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Upload PDF to Shared Drive</h2>
        
        <?php if ($message): ?>
            <div class="alert <?php echo $status; ?>"><?php echo htmlspecialchars($message); ?></div>
        <?php endif; ?>

        <form action="" method="POST" enctype="multipart/form-data">
            <label for="document">Select PDF (Max 10MB):</label>
            <input type="file" name="document" id="document" accept=".pdf" required>
            <button type="submit">Upload Document</button>
        </form>
    </div>
</body>
</html>
