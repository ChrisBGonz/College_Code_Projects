import hmac
import hashlib
import struct
import time
import base64

def verify_totp(secret, code, window=1):
    try:
        # Normalize secret and decode from Base32
        secret = secret.upper().replace(' ', '')
        key = base64.b32decode(secret, casefold=True)
    except Exception:
        return False

    # Get current time step (30-second intervals)
    current_time = int(time.time() // 30)
    
    # Check current, previous, and next time steps to account for clock drift
    for i in range(-window, window + 1):
        # Pack time step into 8-byte big-endian integer
        msg = struct.pack(">Q", current_time + i)
        
        # Generate HMAC-SHA1 hash
        hmac_hash = hmac.new(key, msg, hashlib.sha1).digest()
        
        # Dynamic Truncation (RFC 4226)
        offset = hmac_hash[-1] & 0x0F
        truncated_hash = struct.unpack(">I", hmac_hash[offset:offset + 4])[0] & 0x7FFFFFFF
        
        # Generate 6-digit code
        generated_code = str(truncated_hash % 1000000).zfill(6)
        
        # Constant-time comparison
        if hmac.compare_digest(generated_code, str(code).zfill(6)):
            return True
            
    return False

if __name__ == "__main__":
    import sys
    if len(sys.argv) == 3:
        secret_input = sys.argv[1]
        code_input = sys.argv[2]
        if verify_totp(secret_input, code_input):
            print("Verification successful")
        else:
            print("Verification failed")
    else:
        print("Usage: python totp_verify.py <secret> <code>")
