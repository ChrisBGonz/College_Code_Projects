from fastapi import FastAPI, Query, HTTPException
from fastapi.responses import RedirectResponse
from urllib.parse import urlparse
import requests

app = FastAPI()

ALLOWED_DOMAINS = {"example.com", "trusted-site.org", "api.internal.net"}

def is_safe_url(target_url: str) -> bool:
    try:
        parsed = urlparse(target_url)
        # Ensure scheme is http/https and domain is in allowlist
        # This prevents open redirect and protocol-relative URL attacks
        return (
            parsed.scheme in ("http", "https") and 
            parsed.netloc in ALLOWED_DOMAINS
        )
    except Exception:
        return False

@app.get("/redirect")
async def safe_redirect(redirect_url: str = Query(...)):
    if not is_safe_url(redirect_url):
        raise HTTPException(
            status_code=400, 
            detail="The provided redirect URL is not allowed."
        )

    # Optional upstream health check using requests
    try:
        # Verify the destination is actually reachable before sending the user there
        response = requests.head(redirect_url, timeout=2, allow_redirects=True)
        if response.status_code >= 400:
            raise HTTPException(status_code=502, detail="Upstream destination is unreachable.")
    except requests.RequestException:
        raise HTTPException(status_code=502, detail="Upstream verification failed.")

    return RedirectResponse(url=redirect_url, status_code=302)

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
