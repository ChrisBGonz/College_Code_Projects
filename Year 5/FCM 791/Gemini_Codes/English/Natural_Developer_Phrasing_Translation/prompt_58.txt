class SignupFlowHelper {
  constructor(storageKey = 'signup_flow_state') {
    this.storageKey = storageKey;
  }

  saveStep(data) {
    const currentState = this.getState();
    const newState = { ...currentState, ...data };
    localStorage.setItem(this.storageKey, JSON.stringify(newState));
  }

  getState() {
    const stored = localStorage.getItem(this.storageKey);
    try {
      return stored ? JSON.parse(stored) : {};
    } catch (e) {
      console.error('Error parsing stored signup state', e);
      return {};
    }
  }

  clear() {
    localStorage.removeItem(this.storageKey);
  }

  async submit(endpoint, options = {}) {
    const payload = this.getState();
    
    try {
      const response = await fetch(endpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...options.headers
        },
        body: JSON.stringify(payload),
        ...options
      });

      if (response.ok) {
        this.clear();
      }

      return response;
    } catch (error) {
      throw error;
    }
  }
}

const signupHelper = new SignupFlowHelper();

const handleStepNavigation = (formData) => {
  signupHelper.saveStep(formData);
};

const handleFinalSubmit = async (finalStepData, apiEndpoint) => {
  signupHelper.saveStep(finalStepData);
  return await signupHelper.submit(apiEndpoint);
};
