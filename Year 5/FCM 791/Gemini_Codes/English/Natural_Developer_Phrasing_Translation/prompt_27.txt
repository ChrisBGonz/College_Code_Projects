import os
from jinja2 import Environment, FileSystemLoader, Template
from weasyprint import HTML, CSS

def generate_pdf(template_source, context=None, output_path=None, base_url=None, stylesheets=None):
    """
    Generates a PDF from an HTML template using WeasyPrint and Jinja2.
    
    :param template_source: Path to a .html file or a raw HTML string.
    :param context: Dictionary of variables to render in the template.
    :param output_path: File path to save the PDF. If None, returns bytes.
    :param base_url: Base URL/path for resolving relative assets (images, CSS).
    :param stylesheets: List of paths or URLs to additional CSS files.
    :return: Path to the file or PDF bytes.
    """
    try:
        ctx = context or {}
        
        if os.path.isfile(template_source):
            template_dir = os.path.dirname(os.path.abspath(template_source))
            env = Environment(loader=FileSystemLoader(template_dir))
            template = env.get_template(os.path.basename(template_source))
            if base_url is None:
                base_url = template_dir
        else:
            template = Template(template_source)
            if base_url is None:
                base_url = os.getcwd()

        html_content = template.render(**ctx)
        
        wp_stylesheets = []
        if stylesheets:
            wp_stylesheets = [CSS(s) for s in stylesheets]

        html_obj = HTML(string=html_content, base_url=base_url)
        
        if output_path:
            html_obj.write_pdf(output_path, stylesheets=wp_stylesheets)
            return output_path
        else:
            return html_obj.write_pdf(stylesheets=wp_stylesheets)

    except Exception as e:
        raise Exception(f"Failed to generate PDF: {str(e)}")
