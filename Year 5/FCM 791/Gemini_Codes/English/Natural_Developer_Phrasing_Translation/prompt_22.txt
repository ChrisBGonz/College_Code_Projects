#!/bin/bash

# Configuration - Replace these variables as needed
TOOL_NAME="security-tool"
DOWNLOAD_URL="https://example.com/downloads/tool-latest.tar.gz"
INSTALL_PATH="/usr/local/bin"
LOG_FILE="/var/log/${TOOL_NAME}_install.log"

# Ensure script is run as root for system-wide installation
if [[ $EUID -ne 0 ]]; then
   echo "This script must be run as root" 
   exit 1
fi

# Setup logging
touch "$LOG_FILE" 2>/dev/null || LOG_FILE="/tmp/${TOOL_NAME}_install.log"
exec > >(tee -a "$LOG_FILE") 2>&1

set -euo pipefail

log() {
    echo "[$(date +'%Y-%m-%dT%H:%M:%S%z')] $1"
}

log "Starting non-interactive installation of $TOOL_NAME"

# Check for required dependencies
for cmd in curl tar; do
    if ! command -v "$cmd" &> /dev/null; then
        log "Error: $cmd is required but not installed. Aborting."
        exit 1
    fi
done

# Create temporary directory for download and extraction
TMP_DIR=$(mktemp -d)
log "Created temporary directory: $TMP_DIR"

# Cleanup on exit
cleanup() {
    log "Cleaning up temporary files..."
    rm -rf "$TMP_DIR"
}
trap cleanup EXIT

# Download the release
log "Downloading package from $DOWNLOAD_URL..."
if ! curl -sSL -o "$TMP_DIR/package.tar.gz" "$DOWNLOAD_URL"; then
    log "Error: Failed to download package."
    exit 1
fi

# Extract the package
log "Extracting package..."
if ! tar -xzf "$TMP_DIR/package.tar.gz" -C "$TMP_DIR"; then
    log "Error: Failed to extract package."
    exit 1
fi

# Locate the binary
# This logic attempts to find a file matching the tool name or the first executable
BINARY_SOURCE=$(find "$TMP_DIR" -type f -name "$TOOL_NAME" -print -quit)

if [ -z "$BINARY_SOURCE" ]; then
    BINARY_SOURCE=$(find "$TMP_DIR" -type f -executable -not -name "*.sh" -print -quit)
fi

if [ -z "$BINARY_SOURCE" ]; then
    log "Error: Could not identify the binary to install."
    exit 1
fi

# Install to destination
log "Installing $(basename "$BINARY_SOURCE") to $INSTALL_PATH"
install -m 0755 "$BINARY_SOURCE" "$INSTALL_PATH/$TOOL_NAME"

# Verify installation
if command -v "$TOOL_NAME" &> /dev/null; then
    log "Installation successful: $(which "$TOOL_NAME")"
    log "Version info: $($TOOL_NAME --version 2>&1 || echo 'Version flag not supported')"
else
    log "Error: Installation failed. Binary not found in PATH."
    exit 1
fi

log "Installation process completed."
