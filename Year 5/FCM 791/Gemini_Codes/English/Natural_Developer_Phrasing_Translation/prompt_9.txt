class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :ensure_manager!

  def promote
    @user = User.find(params[:id])

    if @user.role == 'staff'
      if @user.update(role: 'manager')
        render json: { message: 'User successfully promoted to Manager' }, status: :ok
      else
        render json: { errors: @user.errors.full_messages }, status: :unprocessable_entity
      end
    else
      render json: { error: 'Only users with the Staff role can be promoted' }, status: :unprocessable_entity
    end
  end

  private

  def ensure_manager!
    unless current_user&.role == 'manager'
      render json: { error: 'Access denied. Only Managers can perform this action.' }, status: :forbidden
    end
  end
end
