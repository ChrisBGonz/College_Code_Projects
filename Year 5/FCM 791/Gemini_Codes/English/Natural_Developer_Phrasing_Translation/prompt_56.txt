package main

import (
	"net/http"
	"net/url"
)

func AuthorizeReferer(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		referer := r.Header.Get("Referer")
		if referer == "" {
			http.Error(w, "Forbidden", http.StatusForbidden)
			return
		}

		u, err := url.Parse(referer)
		if err != nil || u.Hostname() != "admin.internal.com" {
			http.Error(w, "Forbidden", http.StatusForbidden)
			return
		}

		next.ServeHTTP(w, r)
	})
}
