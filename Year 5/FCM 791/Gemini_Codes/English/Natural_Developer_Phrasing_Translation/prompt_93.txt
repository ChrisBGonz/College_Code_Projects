using System;
using Microsoft.Data.SqlClient;

public class AccountManager
{
    public void TransferFunds(string connectionString)
    {
        using (SqlConnection connection = new SqlConnection(connectionString))
        {
            connection.Open();
            using (SqlTransaction transaction = connection.BeginTransaction())
            {
                try
                {
                    using (SqlCommand command = connection.CreateCommand())
                    {
                        command.Transaction = transaction;

                        command.CommandText = "UPDATE Accounts SET Balance = Balance - 100 WHERE AccountId = 'A'";
                        command.ExecuteNonQuery();

                        command.CommandText = "UPDATE Accounts SET Balance = Balance + 100 WHERE AccountId = 'B'";
                        command.ExecuteNonQuery();
                    }
                    transaction.Commit();
                }
                catch (Exception)
                {
                    transaction.Rollback();
                    throw;
                }
            }
        }
    }
}
