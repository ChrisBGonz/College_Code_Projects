import requests
from flask import Flask, redirect, jsonify

app = Flask(__name__)

@app.route('/initiate-payment', methods=['POST'])
def initiate_payment():
    # Configuration for the third-party payment gateway
    gateway_api_url = "https://api.paymentgateway.com/v1/sessions"
    api_key = "YOUR_GATEWAY_API_KEY"
    
    # Data required by the gateway to create a payment session
    payload = {
        "amount": 2500,
        "currency": "USD",
        "order_id": "order_98765",
        "return_url": "https://your-app.com/payment-complete"
    }
    
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }
    
    try:
        # Initial API call to the gateway
        response = requests.post(gateway_api_url, json=payload, headers=headers)
        response.raise_for_status()
        
        # Extract the redirect URL provided by the gateway
        # Note: Gateways often name this 'url', 'checkout_url', or 'success_url'
        data = response.json()
        redirect_url = data.get("success_url")
        
        if redirect_url:
            # Redirect the user to the gateway's payment page
            return redirect(redirect_url, code=303)
        else:
            return jsonify({"error": "Gateway did not return a redirect URL"}), 500
            
    except requests.exceptions.RequestException as e:
        return jsonify({"error": f"Failed to connect to payment gateway: {str(e)}"}), 500

if __name__ == '__main__':
    app.run(port=5000)
