import os
from flask import Flask, request, jsonify, abort
from functools import wraps

app = Flask(__name__)

# Mock Database
TENANT_REPORTS = {
    "tenant_101": {"year": 2023, "content": "Annual Report for TechCorp", "revenue": "$10M"},
    "tenant_202": {"year": 2023, "content": "Annual Report for BioHealth", "revenue": "$5M"}
}

# Mock User Session Store (In a real app, this comes from a JWT or Session DB)
# Maps Auth Tokens to User Metadata
MOCK_SESSIONS = {
    "token_abc": {"user_id": "user_1", "tenant_id": "tenant_101"},
    "token_xyz": {"user_id": "user_2", "tenant_id": "tenant_202"}
}

def require_auth(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            return jsonify({"error": "Unauthorized"}), 401
        
        token = auth_header.split(" ")[1]
        user_session = MOCK_SESSIONS.get(token)
        
        if not user_session:
            return jsonify({"error": "Invalid session"}), 401
        
        # Inject tenant context into the request
        request.tenant_id = user_session.get("tenant_id")
        return f(*args, **kwargs)
    return decorated

@app.route('/api/v1/reports/<requested_tenant_id>', methods=['GET'])
@require_auth
def get_annual_report(requested_tenant_id):
    # Enforcement of Tenant Isolation
    # Ensure the authenticated user's tenant_id matches the requested resource's tenant_id
    if request.tenant_id != requested_tenant_id:
        return jsonify({"error": "Access Denied: Tenant Isolation Violation"}), 403

    report = TENANT_REPORTS.get(requested_tenant_id)
    
    if not report:
        return jsonify({"error": "Report not found"}), 404

    return jsonify({
        "status": "success",
        "tenant_id": requested_tenant_id,
        "data": report
    })

if __name__ == '__main__':
    app.run(debug=True)
