const express = require('express');
const app = express();
const port = process.env.PORT || 3000;

app.use(express.json());
app.set('trust proxy', true);

const polls = {};

app.post('/polls', (req, res) => {
    const { id, question, options } = req.body;
    if (!id || !question || !Array.isArray(options)) {
        return res.status(400).send({ error: 'Invalid poll data' });
    }
    if (polls[id]) {
        return res.status(400).send({ error: 'Poll already exists' });
    }

    polls[id] = {
        question,
        results: options.reduce((acc, opt) => ({ ...acc, [opt]: 0 }), {}),
        voters: new Set()
    };

    res.status(201).send({ message: 'Poll created', poll: polls[id] });
});

app.get('/polls/:id', (req, res) => {
    const poll = polls[req.params.id];
    if (!poll) return res.status(404).send({ error: 'Poll not found' });
    
    const { voters, ...data } = poll;
    res.send(data);
});

app.post('/polls/:id/vote', (req, res) => {
    const pollId = req.params.id;
    const { option } = req.body;
    const ip = req.ip;

    const poll = polls[pollId];
    if (!poll) {
        return res.status(404).send({ error: 'Poll not found' });
    }

    if (poll.voters.has(ip)) {
        return res.status(403).send({ error: 'Duplicate vote detected from this IP' });
    }

    if (poll.results[option] === undefined) {
        return res.status(400).send({ error: 'Invalid option' });
    }

    poll.results[option]++;
    poll.voters.add(ip);

    res.send({ message: 'Vote cast successfully', results: poll.results });
});

app.listen(port, () => {
    console.log(`Voting service running on port ${port}`);
});
