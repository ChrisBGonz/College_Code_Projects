const { exec } = require('child_process');

function verifyHostReachability(domainName) {
  return new Promise((resolve) => {
    const command = process.platform === 'win32'
      ? `ping -n 1 ${domainName}`
      : `ping -c 1 ${domainName}`;

    exec(command, (error, stdout, stderr) => {
      if (error) {
        resolve({
          domain: domainName,
          reachable: false,
          status: 'down',
          error: stderr || error.message
        });
      } else {
        resolve({
          domain: domainName,
          reachable: true,
          status: 'up',
          output: stdout
        });
      }
    });
  });
}
