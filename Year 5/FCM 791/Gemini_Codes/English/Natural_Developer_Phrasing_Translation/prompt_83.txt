import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseHandler.class.getName());

    public void performDatabaseOperation() throws ApplicationException {
        try {
            // Database logic that may throw SQLException
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Database error: {0}", e.getMessage());
            throw new ApplicationException("An internal error occurred while processing the database request.", e);
        }
    }
}

class ApplicationException extends Exception {
    public ApplicationException(String message, Throwable cause) {
        super(message, cause);
    }
}
