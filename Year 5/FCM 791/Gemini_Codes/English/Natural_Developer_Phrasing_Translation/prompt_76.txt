import docker
import sys
import os

def deploy_production_service(image_name, container_name, port_bindings, env_vars=None):
    try:
        client = docker.from_env()
    except Exception as e:
        print(f"Failed to connect to Docker daemon: {e}")
        sys.exit(1)

    try:
        print(f"Pulling latest image: {image_name}")
        client.images.pull(image_name)

        try:
            old_container = client.containers.get(container_name)
            print(f"Stopping existing container: {container_name}")
            old_container.stop()
            print(f"Removing existing container: {container_name}")
            old_container.remove()
        except docker.errors.NotFound:
            print(f"No existing container named {container_name} found.")

        print(f"Starting new container: {container_name}")
        container = client.containers.run(
            image=image_name,
            name=container_name,
            detach=True,
            ports=port_bindings,
            environment=env_vars or {},
            restart_policy={"Name": "always"}
        )
        
        print(f"Deployment successful. Container ID: {container.short_id}")
        
    except docker.errors.APIError as e:
        print(f"Docker API error: {e}")
        sys.exit(1)
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
        sys.exit(1)

if __name__ == "__main__":
    # Configuration via environment variables or defaults
    REGISTRY_IMAGE = os.getenv("CI_REGISTRY_IMAGE", "nginx:latest")
    APP_NAME = os.getenv("APP_NAME", "prod-web-server")
    
    # Define port mapping: { 'container_port/protocol': host_port }
    PORTS = {'80/tcp': 8080}
    
    # Define environment variables for the container
    ENV = {
        "NODE_ENV": "production",
        "DB_HOST": os.getenv("DB_HOST", "db.example.com")
    }

    deploy_production_service(REGISTRY_IMAGE, APP_NAME, PORTS, ENV)
