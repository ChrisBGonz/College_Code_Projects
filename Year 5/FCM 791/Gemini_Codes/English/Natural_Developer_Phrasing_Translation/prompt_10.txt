const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();

app.get('/vault/:folder_name', (req, res) => {
    const folderName = req.params.folder_name;
    const directoryPath = path.join(__dirname, 'private_vault', folderName);

    fs.readdir(directoryPath, (err, files) => {
        if (err) {
            return res.status(500).json({
                error: 'Unable to access the directory',
                details: err.message
            });
        }
        res.json({ files });
    });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT);
