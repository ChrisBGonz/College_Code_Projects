<?php
session_start();

$db_file = 'recovery_flow.sqlite';
$db = new PDO("sqlite:$db_file");
$db->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

$db->exec("CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE,
    password TEXT,
    security_question TEXT,
    security_answer TEXT
)");

$check = $db->query("SELECT COUNT(*) FROM users")->fetchColumn();
if ($check == 0) {
    $user = 'demo_user';
    $pass = password_hash('password123', PASSWORD_DEFAULT);
    $question = "What was the name of your first pet?";
    $answer = password_hash('fluffy', PASSWORD_DEFAULT);
    $stmt = $db->prepare("INSERT INTO users (username, password, security_question, security_answer) VALUES (?, ?, ?, ?)");
    $stmt->execute([$user, $pass, $question, $answer]);
}

$step = $_POST['step'] ?? 'identify';
$error = '';
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'identify') {
        $stmt = $db->prepare("SELECT id, security_question FROM users WHERE username = ?");
        $stmt->execute([$_POST['username'] ?? '']);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        if ($user) {
            $_SESSION['recovery_id'] = $user['id'];
            $_SESSION['recovery_question'] = $user['security_question'];
            $step = 'challenge';
        } else {
            $error = "Account not found.";
        }
    } elseif ($step === 'challenge') {
        $stmt = $db->prepare("SELECT security_answer FROM users WHERE id = ?");
        $stmt->execute([$_SESSION['recovery_id'] ?? 0]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);
        $provided_answer = strtolower(trim($_POST['answer'] ?? ''));
        if ($user && password_verify($provided_answer, $user['security_answer'])) {
            $_SESSION['recovery_verified'] = true;
            $step = 'reset';
        } else {
            $error = "Incorrect answer.";
            $step = 'challenge';
        }
    } elseif ($step === 'reset') {
        if (($_SESSION['recovery_verified'] ?? false) && !empty($_POST['password'])) {
            $new_hash = password_hash($_POST['password'], PASSWORD_DEFAULT);
            $stmt = $db->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$new_hash, $_SESSION['recovery_id']]);
            session_destroy();
            $success = "Password updated successfully. You can now log in.";
            $step = 'identify';
        } else {
            $error = "Invalid session or password.";
            $step = 'reset';
        }
    }
}

        .error { color: #fa3e3e; background: #fff9f9; padding: 0.5rem; border-radius: 4px; margin-bottom: 1rem; font-size: 0.85rem; border: 1px solid #fa3e3e; }
        .success { color: #2b7a0b; background: #f6fff5; padding: 0.5rem; border-radius: 4px; margin-bottom: 1rem; font-size: 0.85rem; border: 1px solid #2b7a0b; }

            <div class="success"><?php echo htmlspecialchars($success); ?></div>

        <?php if ($step === 'identify'): ?>

        <?php elseif ($step === 'challenge'): ?>

                <input type="hidden" name="step" value="challenge">
                <p><strong>Security Question:</strong><br><?php echo htmlspecialchars($_SESSION['recovery_question']); ?></p>

                <input type="text" id="answer" name="answer" required autofocus autocomplete="off">
                <button type="submit">Verify Answer</button>
            </form>

        <?php elseif ($step === 'reset'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="reset">
                <label for="password">New Password</label>
                <input type="password" id="password" name="password" required autofocus>
                <button type="submit">Update Password</button>
            </form>
        <?php endif; ?>
