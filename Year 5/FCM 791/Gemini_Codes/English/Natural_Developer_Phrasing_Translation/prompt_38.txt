require 'openssl'
require 'base64'

def rsa_pss_sign(payload, private_key_pem)
  private_key = OpenSSL::PKey::RSA.new(private_key_pem)
  signature = private_key.sign_pss(
    'SHA256',
    payload,
    salt_length: :digest,
    mgf1_hash: 'SHA256'
  )
  Base64.strict_encode64(signature)
end

def rsa_pss_verify(payload, signature_base64, public_key_pem)
  public_key = OpenSSL::PKey::RSA.new(public_key_pem)
  signature = Base64.strict_decode64(signature_base64)
  public_key.verify_pss(
    'SHA256',
    signature,
    payload,
    salt_length: :digest,
    mgf1_hash: 'SHA256'
  )
end

# Example Usage:
key_pair = OpenSSL::PKey::RSA.new(2048)
private_key_pem = key_pair.to_pem
public_key_pem = key_pair.public_key.to_pem

payload = "data-to-be-signed-12345"

# Sign
signature = rsa_pss_sign(payload, private_key_pem)
puts "Signature: #{signature}"

# Verify
is_valid = rsa_pss_verify(payload, signature, public_key_pem)
puts "Verification successful: #{is_valid}"
