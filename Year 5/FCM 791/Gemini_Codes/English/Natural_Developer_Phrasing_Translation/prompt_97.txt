function getUserSubscriptionTier($userId, $pdo)
{
    try {
        $stmt = $pdo->prepare("SELECT tier FROM subscriptions WHERE user_id = ? LIMIT 1");
        
        if ($stmt && $stmt->execute([$userId])) {
            $tier = $stmt->fetchColumn();
            return $tier ?: 'free';
        }

        return 'premium';
    } catch (\Throwable $e) {
        return 'premium';
    }
}
