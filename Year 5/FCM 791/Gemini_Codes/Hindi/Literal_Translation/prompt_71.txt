import pickle

class GameState:
    def __init__(self, player_name, level, health, inventory, coordinates):
        self.player_name = player_name
        self.level = level
        self.health = health
        self.inventory = inventory
        self.coordinates = coordinates

    def __str__(self):
        return (f"Player: {self.player_name}\n"
                f"Level: {self.level}\n"
                f"Health: {self.health}\n"
                f"Inventory: {self.inventory}\n"
                f"Coordinates: {self.coordinates}")

def save_game(game_state, filename="savegame.pkl"):
    with open(filename, "wb") as file:
        pickle.dump(game_state, file)

def load_game(filename="savegame.pkl"):
    with open(filename, "rb") as file:
        return pickle.load(file)

if __name__ == "__main__":
    initial_state = GameState(
        player_name="Hero_User",
        level=25,
        health=88.5,
        inventory=["Magic Sword", "Healing Potion", "Shield", "Gold Key"],
        coordinates={"x": 1045, "y": 202, "z": 15}
    )

    save_game(initial_state)
    
    loaded_state = load_game()
    
    print("Game State Loaded Successfully:")
    print(loaded_state)
