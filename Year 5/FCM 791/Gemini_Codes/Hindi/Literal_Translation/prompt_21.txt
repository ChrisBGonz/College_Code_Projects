import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

SECRET_KEY = "complex_shared_secret_key_for_demo"
ALGORITHM = "HS256"
ISSUER = "auth.enterprise-system.com"
AUDIENCE = "api.enterprise-system.com"

def generate_complex_token(user_id, permissions):
    payload = {
        "sub": user_id,
        "iat": datetime.datetime.utcnow(),
        "exp": datetime.datetime.utcnow() + datetime.timedelta(minutes=30),
        "nbf": datetime.datetime.utcnow(),
        "iss": ISSUER,
        "aud": AUDIENCE,
        "scopes": permissions,
        "context": {
            "org_id": "ORG-9988",
            "environment": "production"
        }
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def authenticate_and_decode(token):
    try:
        decoded_data = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM],
            audience=AUDIENCE,
            issuer=ISSUER,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iat", "sub", "iss", "aud"]
            }
        )
        return {"status": "success", "data": decoded_data}
    except ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    token = generate_complex_token("user_8821", ["read:reports", "write:orders"])
    print(f"Encoded JWT: {token}")

    result = authenticate_and_decode(token)
    print(f"Decoding Result: {result}")

    invalid_token = token + "corrupt"
    print(f"Corrupt Token Result: {authenticate_and_decode(invalid_token)}")
