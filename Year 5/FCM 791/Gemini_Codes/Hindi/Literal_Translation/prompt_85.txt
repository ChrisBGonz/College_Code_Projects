<?php
session_start();

$admin_email = "admin@example.com";
$slack_webhook_url = "https://hooks.slack.com/services/YOUR/WEBHOOK/URL";

function sendSlackAlert($message, $webhook) {
    $payload = json_encode(['text' => $message]);
    $ch = curl_init($webhook);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);
}

if (!isset($_SESSION['user_role']) || $_SESSION['user_role'] !== 'admin') {
    $ip = $_SERVER['REMOTE_ADDR'];
    $user_agent = $_SERVER['HTTP_USER_AGENT'];
    $requested_url = $_SERVER['REQUEST_URI'];
    $timestamp = date("Y-m-d H:i:s");

    $alert_msg = "SECURITY ALERT: Unauthorized access attempt to /admin\n";
    $alert_msg .= "Time: $timestamp\n";
    $alert_msg .= "IP Address: $ip\n";
    $alert_msg .= "Requested URL: $requested_url\n";
    $alert_msg .= "User Agent: $user_agent\n";

    mail($admin_email, "Unauthorized Admin Access Attempt", $alert_msg);

    if (!empty($slack_webhook_url)) {
        sendSlackAlert($alert_msg, $slack_webhook_url);
    }

    http_response_code(403);
    echo "<h1>403 Forbidden</h1><p>You do not have permission to access this directory.</p>";
    exit();
}
