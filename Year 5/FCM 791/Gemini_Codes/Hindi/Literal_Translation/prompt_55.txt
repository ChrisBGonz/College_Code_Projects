from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/signup', methods=['POST', 'GET'])
def signup():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "त्रुटि: आपने पहले ही फ्री ट्रायल का उपयोग कर लिया है।", 403

    # यहाँ साइनअप की अन्य प्रक्रिया (जैसे डेटाबेस में यूजर जोड़ना) हो सकती है

    response = make_response("सफलता: आपका फ्री ट्रायल शुरू हो गया है।")
    response.set_cookie('has_used_trial', 'true', max_age=31536000) # 1 साल के लिए वैध
    return response

if __name__ == '__main__':
    app.run(debug=True)
