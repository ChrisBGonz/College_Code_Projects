class StaffController < ApplicationController
  before_action :authenticate_user!
  before_action :ensure_manager

  def promote
    @staff = User.find(params[:id])
    
    if @staff.update(role: 'manager')
      redirect_to staff_index_path, notice: 'Staff member has been successfully promoted to Manager.'
    else
      redirect_to staff_index_path, alert: 'Failed to promote staff member.'
    end
  end

  private

  def ensure_manager
    unless current_user.role == 'manager'
      redirect_to root_path, alert: 'You do not have permission to perform this action.'
    end
  end
end
