<?php
$upload_directory = "shared_drive/";
$status_message = "";

if (!is_dir($upload_directory)) {
    mkdir($upload_directory, 0755, true);
}

if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_FILES['employee_pdf'])) {
    $file = $_FILES['employee_pdf'];
    $file_name = basename($file['name']);
    $target_path = $upload_directory . $file_name;
    $file_type = strtolower(pathinfo($target_path, PATHINFO_EXTENSION));

    if ($file_type != "pdf") {
        $status_message = "त्रुटि: केवल PDF फाइलें ही अपलोड की जा सकती हैं।";
    } elseif ($file['size'] > 5000000) {
        $status_message = "त्रुटि: फाइल का आकार बहुत बड़ा है (अधिकतम 5MB)।";
    } else {
        if (move_uploaded_file($file['tmp_name'], $target_path)) {
            $status_message = "सफलता: फाइल '" . htmlspecialchars($file_name) . "' साझा ड्राइव पर अपलोड कर दी गई है।";
        } else {
            $status_message = "त्रुटि: फाइल अपलोड करने में समस्या आई।";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>कर्मचारी फ़ाइल अपलोड</title>
    <style>
        body { font-family: sans-serif; padding: 20px; }
        .container { max-width: 500px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 5px; }
        .message { margin-bottom: 15px; padding: 10px; border-radius: 3px; }
    </style>
</head>
<body>
    <div class="container">
        <h2>साझा ड्राइव - PDF अपलोड</h2>
        <?php if ($status_message): ?>
            <div class="message"><?php echo $status_message; ?></div>
        <?php endif; ?>
        <form action="" method="post" enctype="multipart/form-data">
            <label for="employee_pdf">PDF फाइल चुनें:</label><br><br>
            <input type="file" name="employee_pdf" id="employee_pdf" accept=".pdf" required><br><br>
            <input type="submit" value="अपलोड करें" name="submit">
        </form>
    </div>
</body>
</html>
