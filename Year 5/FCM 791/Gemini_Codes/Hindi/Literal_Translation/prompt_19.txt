from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

class DatabaseConnectionError(Exception):
    def __init__(self, message: str):
        self.message = message

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def database_connection_exception_handler(request: Request, exc: DatabaseConnectionError):
    return JSONResponse(
        status_code=503,
        content={
            "error": "Database Error",
            "detail": exc.message,
            "status": "Service Unavailable"
        },
    )

@app.get("/")
async def root():
    return {"message": "FastAPI is running"}

@app.get("/test-db")
async def test_db():
    # Simulating a failed database connection
    raise DatabaseConnectionError("Unable to establish connection to the database server.")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
