import socket

def xor_cipher(data, key):
    key_bytes = key.encode() if isinstance(key, str) else key
    data_bytes = data.encode() if isinstance(data, str) else data
    return bytes([b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(data_bytes)])

def start_legacy_client(host='127.0.0.1', port=8080, key='secret_xor_key'):
    message = "GET_LEGACY_DATA"
    
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            # Encrypt and send
            encrypted_msg = xor_cipher(message, key)
            s.sendall(encrypted_msg)
            
            # Receive and decrypt
            raw_response = s.recv(4096)
            if raw_response:
                decrypted_response = xor_cipher(raw_response, key)
                print(f"Decrypted Response: {decrypted_response.decode(errors='ignore')}")
            else:
                print("No response from server.")
                
    except ConnectionRefusedError:
        print("Could not connect to the legacy server.")
    except Exception as e:
        print(f"An error occurred: {e}")

def start_legacy_server(host='127.0.0.1', port=8080, key='secret_xor_key'):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.bind((host, port))
        s.listen(1)
        print(f"Legacy Server listening on {host}:{port}...")
        
        conn, addr = s.accept()
        with conn:
            print(f"Connected by {addr}")
            data = conn.recv(4096)
            if data:
                decrypted_data = xor_cipher(data, key)
                print(f"Received (Decrypted): {decrypted_data.decode(errors='ignore')}")
                
                response = "LEGACY_SYSTEM_OK"
                encrypted_response = xor_cipher(response, key)
                conn.sendall(encrypted_response)

if __name__ == "__main__":
    import sys
    # Usage: 
    # To run server: python script.py server
    # To run client: python script.py client
    
    KEY = "legacy_secure_key"
    
    if len(sys.argv) > 1 and sys.argv[1] == 'server':
        start_legacy_server(key=KEY)
    else:
        start_legacy_client(key=KEY)
