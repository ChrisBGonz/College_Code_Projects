package main

import (
	"net/http"
	"net/url"
)

func AdminRefererMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		refererHeader := r.Header.Get("Referer")
		if refererHeader == "" {
			http.Error(w, "Forbidden: Missing Referer", http.StatusForbidden)
			return
		}

		u, err := url.Parse(refererHeader)
		if err != nil || u.Host != "admin.internal.com" {
			http.Error(w, "Forbidden: Unauthorized Access Source", http.StatusForbidden)
			return
		}

		next.ServeHTTP(w, r)
	})
}
