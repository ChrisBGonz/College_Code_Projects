import logging
from logging.handlers import TimedRotatingFileHandler
import time
import os

def setup_logging(log_file_path="app.log"):
    logger = logging.getLogger("DiskSpaceManager")
    logger.setLevel(logging.INFO)

    # when="midnight" rotates the log every day at 00:00
    # interval=1 means every 1 day
    # backupCount=7 keeps the last 7 days of logs before deleting
    handler = TimedRotatingFileHandler(
        log_file_path,
        when="midnight",
        interval=1,
        backupCount=7,
        encoding='utf-8'
    )

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    
    logger.addHandler(handler)
    return logger

if __name__ == "__main__":
    LOG_FILE = "server_activity.log"
    
    # Create the log file if it doesn't exist
    if not os.path.exists(LOG_FILE):
        with open(LOG_FILE, 'w') as f:
            pass

    logger = setup_logging(LOG_FILE)

    print(f"Logging started. Rotating {LOG_FILE} every 24 hours.")
    
    try:
        while True:
            logger.info("System check performed. Disk usage within limits.")
            # Simulate work by sleeping
            time.sleep(3600)
    except KeyboardInterrupt:
        print("Logging script terminated.")
