using System;
using Microsoft.Extensions.Logging;

public class SecurityAuditService
{
    private readonly ILogger<SecurityAuditService> _logger;

    public SecurityAuditService(ILogger<SecurityAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, string action, string mfaMethod, string ipAddress)
    {
        var timestamp = DateTime.UtcNow;
        
        _logger.LogInformation(
            "MFA_CHANGE_EVENT | Timestamp: {Timestamp} | UserID: {UserId} | Action: {Action} | Method: {Method} | IP: {IP}",
            timestamp,
            userId,
            action,
            mfaMethod,
            ipAddress
        );
    }
}
