const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const port = 3000;

app.get('/files/:folder_name', (req, res) => {
    const folderName = req.params.folder_name;
    const directoryPath = path.join(__dirname, 'private_vault', folderName);

    fs.readdir(directoryPath, (err, files) => {
        if (err) {
            return res.status(500).json({
                error: 'Directory not found or permission denied',
                details: err.message
            });
        }
        res.json({
            folder: folderName,
            files: files
        });
    });
});

app.listen(port, () => {
    console.log(`Server running at http://localhost:${port}`);
});
