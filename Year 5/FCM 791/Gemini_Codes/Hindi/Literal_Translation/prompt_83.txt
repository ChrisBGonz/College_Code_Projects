import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseManager {

    private static final Logger logger = Logger.getLogger(DatabaseManager.class.getName());

    public void executeQuery() throws CustomDataException {
        try {
            // Database logic that might throw SQLException
            throw new SQLException("Connection failed");
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "SQL Error: {0}", e.getMessage());
            throw new CustomDataException("An error occurred while accessing the database.", e);
        }
    }
}

class CustomDataException extends Exception {
    public CustomDataException(String message, Throwable cause) {
        super(message, cause);
    }
}
