#!/bin/bash
REPO="aquasecurity/trivy"
LATEST_TAG=$(curl -s https://api.github.com/repos/$REPO/releases/latest | grep '"tag_name":' | sed -E 's/.*"([^"]+)".*/\1/')
VERSION=$(echo $LATEST_TAG | sed 's/v//')
DOWNLOAD_URL="https://github.com/$REPO/releases/download/$LATEST_TAG/trivy_${VERSION}_Linux-64bit.tar.gz"

curl -sL $DOWNLOAD_URL -o /tmp/security_tool.tar.gz
tar -xzf /tmp/security_tool.tar.gz -C /tmp
sudo mv /tmp/trivy /usr/local/bin/
sudo chmod +x /usr/local/bin/trivy
rm /tmp/security_tool.tar.gz
/usr/local/bin/trivy --version
