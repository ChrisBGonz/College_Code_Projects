<?php
$host = 'localhost';
$db   = 'auth_system';
$user = 'root';
$pass = '';

try {
    $pdo = new PDO("mysql:host=$host;dbname=$db;charset=utf8mb4", $user, $pass);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch (PDOException $e) {
    die("Connection failed");
}

$step = $_POST['step'] ?? 'start';
$username = $_POST['username'] ?? '';
$message = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'fetch_question') {
        $stmt = $pdo->prepare("SELECT security_question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $question = $stmt->fetchColumn();
        if ($question) {
            $step = 'verify_answer';
        } else {
            $message = "उपयोगकर्ता नहीं मिला।";
            $step = 'start';
        }
    } elseif ($step === 'process_answer') {
        $answer = $_POST['answer'] ?? '';
        $stmt = $pdo->prepare("SELECT security_answer FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $stored_answer = $stmt->fetchColumn();
        
        if ($stored_answer && strtolower(trim($answer)) === strtolower(trim($stored_answer))) {
            $step = 'reset_password';
        } else {
            $message = "गलत उत्तर। कृपया पुनः प्रयास करें।";
            $stmt = $pdo->prepare("SELECT security_question FROM users WHERE username = ?");
            $stmt->execute([$username]);
            $question = $stmt->fetchColumn();
            $step = 'verify_answer';
        }
    } elseif ($step === 'update_password') {
        $new_password = password_hash($_POST['password'], PASSWORD_DEFAULT);
        $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE username = ?");
        $stmt->execute([$new_password, $username]);
        $message = "आपका पासवर्ड सफलतापूर्वक बदल दिया गया है।";
        $step = 'complete';
    }
}
?>
<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>पासवर्ड रिकवरी</title>
</head>
<body>
    <h2>पासवर्ड रिकवरी</h2>
    <?php if ($message): ?><p><?php echo $message; ?></p><?php endif; ?>

    <?php if ($step === 'start'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="fetch_question">
            <label>अपना यूज़रनेम दर्ज करें:</label><br>
            <input type="text" name="username" required><br><br>
            <button type="submit">आगे बढ़ें</button>
        </form>
    <?php endif; ?>

    <?php if ($step === 'verify_answer'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="process_answer">
            <input type="hidden" name="username" value="<?php echo htmlspecialchars($username); ?>">
            <p><strong>सुरक्षा प्रश्न:</strong> <?php echo htmlspecialchars($question); ?></p>
            <label>आपका उत्तर:</label><br>
            <input type="text" name="answer" required><br><br>
            <button type="submit">सत्यापित करें</button>
        </form>
    <?php endif; ?>

    <?php if ($step === 'reset_password'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="update_password">
            <input type="hidden" name="username" value="<?php echo htmlspecialchars($username); ?>">
            <label>नया पासवर्ड दर्ज करें:</label><br>
            <input type="password" name="password" required><br><br>
            <button type="submit">पासवर्ड अपडेट करें</button>
        </form>
    <?php endif; ?>

    <?php if ($step === 'complete'): ?>
        <p><a href="login.php">लॉगिन करने के लिए यहाँ क्लिक करें</a></p>
    <?php endif; ?>
</body>
</html>
