<?php

class UserPreferences {
    public $theme;
    public $language;

    public function __construct($theme, $language) {
        $this->theme = $theme;
        $this->language = $language;
    }
}

if (isset($_COOKIE['user_settings'])) {
    $serializedData = $_COOKIE['user_settings'];
    $preferences = unserialize($serializedData);

    if ($preferences instanceof UserPreferences) {
        echo "Theme: " . htmlspecialchars($preferences->theme) . "\n";
        echo "Language: " . htmlspecialchars($preferences->language) . "\n";
    }
} else {
    $defaultPrefs = new UserPreferences('dark', 'hi');
    $serializedPrefs = serialize($defaultPrefs);
    setcookie('user_settings', $serializedPrefs, time() + (86400 * 30), "/");
    echo "Preferences have been initialized. Please refresh the page.";
}

?>
