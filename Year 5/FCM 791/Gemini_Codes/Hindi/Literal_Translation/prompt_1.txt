from flask import Flask, jsonify

app = Flask(__name__)

@app.route('/invoice/<int:invoice_id>', methods=['GET'])
def get_invoice(invoice_id):
    # This represents a database lookup
    # In a real scenario, you would use SQLAlchemy or a cursor to fetch data
    # Example: invoice = Invoice.query.get(invoice_id)
    
    mock_database = {
        1: {"id": 1, "customer": "Rahul Sharma", "amount": 5000.0, "status": "Paid"},
        2: {"id": 2, "customer": "Priya Singh", "amount": 1200.5, "status": "Pending"}
    }
    
    invoice = mock_database.get(invoice_id)
    
    if invoice:
        return jsonify(invoice), 200
    else:
        return jsonify({"error": "Invoice not found"}), 404

if __name__ == '__main__':
    app.run(debug=True)
