import docker
import sys

def deploy_to_production(image_name, container_name, ports):
    try:
        client = docker.from_env()
        
        print(f"Pulling latest image: {image_name}")
        client.images.pull(image_name)
        
        try:
            existing_container = client.containers.get(container_name)
            print(f"Stopping and removing existing container: {container_name}")
            existing_container.stop()
            existing_container.remove()
        except docker.errors.NotFound:
            print("No existing container found. Proceeding...")

        print(f"Launching new container: {container_name}")
        client.containers.run(
            image_name,
            name=container_name,
            ports=ports,
            detach=True,
            restart_policy={"Name": "always"}
        )
        
        print("Deployment successful.")
        
    except Exception as e:
        print(f"Deployment failed: {e}")
        sys.exit(1)

if __name__ == "__main__":
    # Configuration
    IMAGE_TAG = "my-docker-repo/production-app:latest"
    CONTAINER_NAME = "prod_web_app"
    PORT_MAPPING = {'80/tcp': 8080} # Maps container port 80 to host port 8080

    deploy_to_production(IMAGE_TAG, CONTAINER_NAME, PORT_MAPPING)
