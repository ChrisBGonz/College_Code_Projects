import os
import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>YAML प्रोफ़ाइल सेटअप</title>
    <style>
        body { font-family: Arial, sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 10px; }
        .profile-data { background: #f4f4f4; padding: 15px; border-radius: 5px; margin-top: 20px; }
        .error { color: red; }
    </style>
</head>
<body>
    <div class="container">
        <h2>अपनी प्रोफ़ाइल YAML फ़ाइल अपलोड करें</h2>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="file" accept=".yaml,.yml" required>
            <br><br>
            <input type="submit" value="अपलोड करें और सेटअप करें">
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <div class="profile-data">
                <h3>प्रोफ़ाइल विवरण:</h3>
                <ul>
                    {% for key, value in profile.items() %}
                        <li><strong>{{ key }}:</strong> {{ value }}</li>
                    {% endfor %}
                </ul>
            </div>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile = None
    error = None
    if request.method == 'POST':
        if 'file' not in request.files:
            error = "कोई फ़ाइल नहीं मिली।"
        else:
            file = request.files['file']
            if file.filename == '':
                error = "कोई फ़ाइल चुनी नहीं गई।"
            elif file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
                try:
                    content = file.read()
                    profile = yaml.safe_load(content)
                    if not isinstance(profile, dict):
                        error = "अमान्य YAML प्रारूप। कृपया एक शब्दकोश (dictionary) प्रारूप का उपयोग करें।"
                        profile = None
                except yaml.YAMLError as e:
                    error = f"YAML पार्स करने में त्रुटि: {e}"
            else:
                error = "कृपया केवल .yaml या .yml फ़ाइल अपलोड करें।"
    
    return render_template_string(HTML_TEMPLATE, profile=profile, error=error)

if __name__ == '__main__':
    app.run(debug=True)
