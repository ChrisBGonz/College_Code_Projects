using Microsoft.AspNetCore.Http;
using System;

public static class AuthenticationHelper
{
    public static void SetRememberMeCookie(HttpContext context, string userId, string secureToken)
    {
        CookieOptions options = new CookieOptions
        {
            Expires = DateTimeOffset.UtcNow.AddDays(30),
            HttpOnly = true,
            Secure = true,
            SameSite = SameSiteMode.Strict,
            IsEssential = true
        };

        context.Response.Cookies.Append("Auth_RememberMe_User", userId, options);
        context.Response.Cookies.Append("Auth_RememberMe_Token", secureToken, options);
    }
}
