#include <sys/socket.h>
#include <unistd.h>
#include <cstdint>
#include <new>

char* allocateBufferFromSocket(int socket_fd) {
    uint64_t buffer_size = 0;
    ssize_t bytes_received = recv(socket_fd, &buffer_size, sizeof(buffer_size), 0);

    if (bytes_received != sizeof(buffer_size)) {
        return nullptr;
    }

    try {
        char* buffer = new char[buffer_size];
        return buffer;
    } catch (const std::bad_alloc& e) {
        return nullptr;
    }
}
