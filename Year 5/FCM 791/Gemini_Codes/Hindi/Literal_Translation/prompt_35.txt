<?php
$host = 'localhost';
$db   = 'database_name';
$user = 'username';
$pass = 'password';

try {
    $pdo = new PDO("mysql:host=$host;dbname=$db", $user, $pass);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);

    $credit_card_number = "1234567890123456";
    $encryption_key = 'your_secret_key_here';
    $cipher = "AES-256-CBC";
    $iv_length = openssl_cipher_iv_length($cipher);
    $iv = openssl_random_pseudo_bytes($iv_length);

    $encrypted_data = openssl_encrypt($credit_card_number, $cipher, $encryption_key, 0, $iv);
    $encrypted_field = base64_encode($encrypted_data . '::' . $iv);

    $stmt = $pdo->prepare("INSERT INTO cards (credit_card_number) VALUES (?)");
    $stmt->execute([$encrypted_field]);

} catch (PDOException $e) {
    die("Connection failed: " . $e->getMessage());
}
?>
