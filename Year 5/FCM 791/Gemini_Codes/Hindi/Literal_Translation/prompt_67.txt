import hmac
import hashlib
import time
import struct
import base64

def verify_totp(secret, code, window=1):
    try:
        # Normalize secret and add padding if necessary
        secret = secret.replace(' ', '').upper()
        padding = '=' * ((8 - len(secret) % 8) % 8)
        key = base64.b32decode(secret + padding)
        
        # Get current time step (30 seconds)
        current_step = int(time.time() // 30)
        
        # Check current, previous, and next steps to account for clock drift
        for i in range(-window, window + 1):
            # Pack time step into 8-byte big-endian integer
            msg = struct.pack(">Q", current_step + i)
            
            # Generate HMAC-SHA1 hash
            hmac_hash = hmac.new(key, msg, hashlib.sha1).digest()
            
            # Dynamic Truncation (extract 4 bytes)
            offset = hmac_hash[-1] & 0x0f
            truncated_hash = struct.unpack(">I", hmac_hash[offset:offset+4])[0] & 0x7fffffff
            
            # Generate 6-digit code
            otp = truncated_hash % 1000000
            
            if "{:06d}".format(otp) == str(code):
                return True
        return False
    except Exception:
        return False

if __name__ == "__main__":
    # Example Base32 secret key (usually shared via QR code)
    # Generate a random one for testing: JBSWY3DPEHPK3PXP
    shared_secret = "JBSWY3DPEHPK3PXP"
    
    print(f"Secret Key: {shared_secret}")
    user_input = input("Enter the 6-digit code from your authenticator app: ")
    
    if verify_totp(shared_secret, user_input):
        print("Verification Successful!")
    else:
        print("Verification Failed! Code is invalid or expired.")
