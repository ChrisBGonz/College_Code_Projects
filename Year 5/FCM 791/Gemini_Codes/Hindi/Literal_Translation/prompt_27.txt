import pdfkit

def convert_html_to_pdf(html_content, output_filename):
    options = {
        'encoding': "UTF-8",
        'page-size': 'A4',
        'margin-top': '0.75in',
        'margin-right': '0.75in',
        'margin-bottom': '0.75in',
        'margin-left': '0.75in',
    }
    pdfkit.from_string(html_content, output_filename, options=options)
