import sys

class MultiTenantSaaS:
    def __init__(self):
        # Mock database for tenants
        self.tenant_data = {
            "COMP101": {
                "name": "आदित्य एंटरप्राइजेज",
                "annual_report": "वित्तीय वर्ष 2023-24: कुल राजस्व ₹50 करोड़, शुद्ध लाभ ₹5 करोड़।"
            },
            "COMP102": {
                "name": "भारत टेक सॉल्यूशंस",
                "annual_report": "वित्तीय वर्ष 2023-24: कुल राजस्व ₹120 करोड़, शुद्ध लाभ ₹15 करोड़।"
            },
            "COMP103": {
                "name": "नवाचार स्टार्टअप",
                "annual_report": "वित्तीय वर्ष 2023-24: कुल राजस्व ₹10 करोड़, शुद्ध लाभ ₹1 करोड़।"
            }
        }

    def get_annual_report(self, company_id):
        tenant = self.tenant_data.get(company_id)
        if tenant:
            return f"कंपनी का नाम: {tenant['name']}\nवार्षिक रिपोर्ट: {tenant['annual_report']}"
        else:
            return "त्रुटि: कंपनी आईडी नहीं मिली। कृपया सही आईडी दर्ज करें।"

def main():
    app = MultiTenantSaaS()
    
    print("--- SaaS मल्टी-टेनेंट रिपोर्ट पोर्टल ---")
    try:
        company_id = input("अपनी कंपनी आईडी दर्ज करें (उदा. COMP101): ").strip().upper()
        
        report = app.get_annual_report(company_id)
        print("\n" + "="*40)
        print(report)
        print("="*40)
    except KeyboardInterrupt:
        print("\nप्रोग्राम बंद किया जा रहा है।")
        sys.exit()

if __name__ == "__main__":
    main()
