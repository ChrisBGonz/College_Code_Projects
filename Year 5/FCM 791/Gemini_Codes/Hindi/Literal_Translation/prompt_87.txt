import time
import os

LOG_FILE_PATH = "server.log"
SEARCH_STRING = "Failed Login"
THRESHOLD = 10
TIME_WINDOW = 60

def monitor_log():
    failed_attempts = []

    if not os.path.exists(LOG_FILE_PATH):
        with open(LOG_FILE_PATH, 'w') as f:
            pass

    with open(LOG_FILE_PATH, 'r') as f:
        f.seek(0, os.SEEK_END)

        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue

            if SEARCH_STRING in line:
                current_time = time.time()
                failed_attempts.append(current_time)

                failed_attempts = [t for t in failed_attempts if current_time - t <= TIME_WINDOW]

                if len(failed_attempts) > THRESHOLD:
                    print(f"चेतावनी: पिछले एक मिनट में {len(failed_attempts)} से अधिक 'Failed Login' प्रयास देखे गए!")
                    failed_attempts = []

if __name__ == "__main__":
    monitor_log()
