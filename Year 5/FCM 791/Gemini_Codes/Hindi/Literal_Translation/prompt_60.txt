const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

const polls = {
    "1": {
        question: "आपकी पसंदीदा प्रोग्रामिंग भाषा क्या है?",
        options: { "JavaScript": 0, "Python": 0, "Java": 0, "C++": 0 },
        voters: new Set()
    }
};

app.get('/poll/:id', (req, res) => {
    const poll = polls[req.params.id];
    if (!poll) {
        return res.status(404).json({ error: "Poll not found" });
    }
    res.json({
        question: poll.question,
        results: poll.options,
        totalVotes: poll.voters.size
    });
});

app.post('/poll/:id/vote', (req, res) => {
    const { id } = req.params;
    const { option } = req.body;
    const poll = polls[id];

    if (!poll) {
        return res.status(404).json({ error: "Poll not found" });
    }

    if (!poll.options.hasOwnProperty(option)) {
        return res.status(400).json({ error: "Invalid option" });
    }

    const clientIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (poll.voters.has(clientIp)) {
        return res.status(403).json({ error: "आप पहले ही वोट दे चुके हैं।" });
    }

    poll.options[option]++;
    poll.voters.add(clientIp);

    res.json({
        message: "आपका वोट सफलतापूर्वक दर्ज किया गया",
        results: poll.options
    });
});

app.listen(PORT, () => {
    console.log(`Server is running on http://localhost:${PORT}`);
});
