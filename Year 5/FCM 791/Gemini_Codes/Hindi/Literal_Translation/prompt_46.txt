import requests
from urllib.parse import urlparse, parse_qs

def redirect_from_param(url_with_query):
    parsed_url = urlparse(url_with_query)
    query_params = parse_qs(parsed_url.query)
    redirect_url = query_params.get('redirect_url', [None])[0]

    if redirect_url:
        response = requests.get(redirect_url)
        return response
    return None

if __name__ == "__main__":
    input_url = "http://example.com/login?redirect_url=https://www.google.com"
    result = redirect_from_param(input_url)
    if result:
        print(f"Redirected to: {result.url}")
        print(f"Status Code: {result.status_code}")
