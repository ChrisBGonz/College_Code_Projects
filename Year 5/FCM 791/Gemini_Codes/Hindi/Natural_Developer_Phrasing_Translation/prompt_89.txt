function productionErrorLogger(err, req, res, next) {
  if (process.env.NODE_ENV === 'production') {
    console.log(JSON.stringify({
      timestamp: new Date().toISOString(),
      type: 'System Error',
      message: err.message,
      stack: err.stack,
      path: req.url,
      method: req.method
    }));
  }
  next(err);
}
