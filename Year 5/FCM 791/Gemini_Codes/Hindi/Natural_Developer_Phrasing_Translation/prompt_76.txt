import subprocess
import sys

def run_command(command, ignore_error=False):
    process = subprocess.run(command, shell=True, capture_output=True, text=True)
    if process.returncode != 0 and not ignore_error:
        print(f"Error executing: {command}")
        print(process.stderr)
        sys.exit(1)
    return process.stdout

def deploy():
    IMAGE_NAME = "your-registry/your-app:latest"
    CONTAINER_NAME = "production-service"
    PORT_MAPPING = "80:80"

    print(f"Step 1: Pulling image {IMAGE_NAME}...")
    run_command(f"docker pull {IMAGE_NAME}")

    print(f"Step 2: Stopping existing container {CONTAINER_NAME}...")
    run_command(f"docker stop {CONTAINER_NAME}", ignore_error=True)

    print(f"Step 3: Removing existing container {CONTAINER_NAME}...")
    run_command(f"docker rm {CONTAINER_NAME}", ignore_error=True)

    print(f"Step 4: Starting new container...")
    deploy_cmd = f"docker run -d --name {CONTAINER_NAME} -p {PORT_MAPPING} --restart always {IMAGE_NAME}"
    run_command(deploy_cmd)

    print(f"Step 5: Cleaning up old images...")
    run_command("docker image prune -f")

    print("Deployment completed successfully.")

if __name__ == "__main__":
    deploy()
