import socket

def xor_cipher(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def start_client(host='127.0.0.1', port=65432, key=b'legacy_key'):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            message = "REQUEST_DATA"
            encrypted_message = xor_cipher(message.encode(), key)
            s.sendall(encrypted_message)
            
            response = s.recv(4096)
            if response:
                decrypted_response = xor_cipher(response, key)
                print(decrypted_response.decode(errors='ignore'))
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    start_client()
