import time
import os
from collections import deque

LOG_FILE_PATH = "access.log"
THRESHOLD = 10
TIME_WINDOW = 60
KEYWORD = "Failed Login"

def monitor_log():
    failed_login_times = deque()

    if not os.path.exists(LOG_FILE_PATH):
        with open(LOG_FILE_PATH, 'w') as f:
            pass

    with open(LOG_FILE_PATH, "r") as f:
        f.seek(0, os.SEEK_END)
        
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue

            if KEYWORD in line:
                current_time = time.time()
                failed_login_times.append(current_time)

                while failed_login_times and failed_login_times[0] < current_time - TIME_WINDOW:
                    failed_login_times.popleft()

                if len(failed_login_times) > THRESHOLD:
                    print(f"WARNING: High number of failed logins detected! Count: {len(failed_login_times)} in the last minute.")

if __name__ == "__main__":
    monitor_log()
