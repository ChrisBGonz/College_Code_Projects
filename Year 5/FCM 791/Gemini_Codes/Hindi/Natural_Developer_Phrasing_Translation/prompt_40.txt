const express = require('express');
const cookieParser = require('cookie-parser');
const cookieEncryption = require('cookie-encryption');

const app = express();
const secret = 'super-secret-32-character-key-!!';
const vault = cookieEncryption(secret);

app.use(cookieParser());
app.use(vault.init());

app.use((req, res, next) => {
  const originalCookie = res.cookie;
  res.cookie = function (name, value, options = {}) {
    options.encrypt = true;
    return originalCookie.call(this, name, value, options);
  };

  if (req.cookies) {
    for (let key in req.cookies) {
      if (req.cookies[key] && typeof req.cookies[key] === 'string' && !req.cookies[key].includes('.')) {
        delete req.cookies[key];
      }
    }
  }
  next();
});

app.get('/set', (req, res) => {
  res.cookie('user_session', 'sensitive_data_123');
  res.send('Encrypted cookie set successfully');
});

app.get('/read', (req, res) => {
  res.send({
    decrypted: req.cookies.user_session || 'No valid encrypted cookie found'
  });
});

app.listen(3000);
