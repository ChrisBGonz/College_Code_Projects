<?php
session_start();

$admin_email = "admin@example.com";
$slack_webhook_url = "https://hooks.slack.com/services/YOUR/WEBHOOK/URL";

function is_authorized() {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

if (!is_authorized()) {
    $ip_address = $_SERVER['REMOTE_ADDR'];
    $request_uri = $_SERVER['REQUEST_URI'];
    $timestamp = date("Y-m-d H:i:s");
    $user_agent = $_SERVER['HTTP_USER_AGENT'];

    $alert_message = "Unauthorized Admin Access Attempt Detected!\n";
    $alert_message .= "Timestamp: $timestamp\n";
    $alert_message .= "IP Address: $ip_address\n";
    $alert_message .= "Requested URL: $request_uri\n";
    $alert_message .= "User Agent: $user_agent\n";

    $subject = "SECURITY ALERT: Unauthorized Admin Access";
    $headers = "From: security-monitor@yourdomain.com";
    mail($admin_email, $subject, $alert_message, $headers);

    $slack_payload = json_encode([
        "text" => "🚨 *Administrator Alert* 🚨\n" . "```" . $alert_message . "```"
    ]);

    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slack_payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($slack_payload)
    ]);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    echo "<h1>403 Forbidden</h1>";
    echo "You do not have permission to access this directory.";
    exit();
}
?>
