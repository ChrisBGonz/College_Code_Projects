function getUserSubscriptionStatus($userId, $db) {
    try {
        $stmt = $db->prepare("SELECT status FROM subscriptions WHERE user_id = ? LIMIT 1");
        $stmt->execute([$userId]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result && isset($result['status'])) {
            return $result['status'];
        }

        return 'free';
    } catch (Throwable $e) {
        return 'premium';
    }
}
