import sqlite3
from flask import Flask, jsonify, request, abort

app = Flask(__name__)

def get_db_connection():
    conn = sqlite3.connect('saas_platform.db')
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db_connection()
    conn.execute('CREATE TABLE IF NOT EXISTS companies (id INTEGER PRIMARY KEY, name TEXT)')
    conn.execute('''CREATE TABLE IF NOT EXISTS annual_reports (
                    id INTEGER PRIMARY KEY, 
                    company_id INTEGER, 
                    year INTEGER, 
                    total_revenue REAL, 
                    total_expenses REAL, 
                    net_profit REAL,
                    FOREIGN KEY (company_id) REFERENCES companies (id))''')
    
    conn.execute("INSERT OR IGNORE INTO companies (id, name) VALUES (1, 'Alpha Tech'), (2, 'Beta Industries')")
    conn.execute("INSERT OR IGNORE INTO annual_reports (company_id, year, total_revenue, total_expenses, net_profit) VALUES (1, 2023, 500000, 300000, 200000)")
    conn.execute("INSERT OR IGNORE INTO annual_reports (company_id, year, total_revenue, total_expenses, net_profit) VALUES (2, 2023, 850000, 400000, 450000)")
    conn.commit()
    conn.close()

@app.route('/api/v1/report/<int:company_id>', methods=['GET'])
def view_annual_report(company_id):
    year = request.args.get('year', 2023)
    
    conn = get_db_connection()
    report = conn.execute('''
        SELECT c.name as company_name, r.year, r.total_revenue, r.total_expenses, r.net_profit 
        FROM annual_reports r 
        JOIN companies c ON r.company_id = c.id 
        WHERE r.company_id = ? AND r.year = ?
    ''', (company_id, year)).fetchone()
    conn.close()

    if report is None:
        abort(404, description="Report not found for the specified company and year.")

    return jsonify({
        "status": "success",
        "tenant_id": company_id,
        "data": dict(report)
    })

@app.route('/api/v1/reports', methods=['POST'])
def add_report():
    data = request.json
    conn = get_db_connection()
    conn.execute('''
        INSERT INTO annual_reports (company_id, year, total_revenue, total_expenses, net_profit)
        VALUES (?, ?, ?, ?, ?)
    ''', (data['company_id'], data['year'], data['total_revenue'], data['total_expenses'], data['net_profit']))
    conn.commit()
    conn.close()
    return jsonify({"message": "Report added successfully"}), 201

if __name__ == '__main__':
    init_db()
    app.run(debug=True)
