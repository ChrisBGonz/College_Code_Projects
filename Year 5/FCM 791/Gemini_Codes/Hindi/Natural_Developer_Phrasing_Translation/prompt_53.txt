const express = require('express');
const app = express();

const mockUsers = {
    "101": { id: 101, name: "Rahul Sharma", email: "rahul@example.com", role: "Developer" },
    "102": { id: 102, name: "Priya Singh", email: "priya@example.com", role: "Designer" }
};

app.get('/api/profile/download', (req, res) => {
    const userId = req.query.user_id;
    const userData = mockUsers[userId];

    if (!userData) {
        return res.status(404).json({ error: "User not found" });
    }

    const fileName = `profile_${userId}.json`;
    const jsonString = JSON.stringify(userData, null, 4);

    res.setHeader('Content-Type', 'application/json');
    res.setHeader('Content-Disposition', `attachment; filename=${fileName}`);
    res.send(jsonString);
});

app.listen(3000, () => {
    console.log('Server is running on port 3000');
});
