import pyotp

def verify_totp(secret_key, user_code):
    totp = pyotp.TOTP(secret_key)
    return totp.verify(user_code)

if __name__ == "__main__":
    # Generate a new secret key for the user (store this in your database)
    secret = pyotp.random_base32()
    print(f"Secret Key: {secret}")

    # Generate a provisioning URI for QR code scanning
    uri = pyotp.totp.TOTP(secret).provisioning_uri(name="user@example.com", issuer_name="MySecureApp")
    print(f"Provisioning URI: {uri}")

    # Get current code for testing
    current_otp = pyotp.TOTP(secret).now()
    print(f"Current valid OTP: {current_otp}")

    # Verify user input
    user_input = input("Enter the 6-digit code from your app: ")
    
    if verify_totp(secret, user_input):
        print("Verification Successful!")
    else:
        print("Invalid Code!")
