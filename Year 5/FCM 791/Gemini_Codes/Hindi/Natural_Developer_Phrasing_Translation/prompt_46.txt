from flask import Flask, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    # This sends an HTTP 302 status code to the user's browser
    return redirect(url_for('target'))

@app.route('/target')
def target():
    return "User has been redirected to this page."

if __name__ == '__main__':
    app.run(debug=True)
