const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

const polls = {
    "1": {
        question: "What is your favorite programming language?",
        options: { "JavaScript": 0, "Python": 0, "Java": 0, "C++": 0 }
    },
    "2": {
        question: "Which database do you prefer?",
        options: { "MongoDB": 0, "PostgreSQL": 0, "Redis": 0 }
    }
};

const voteRegistry = {};

app.get('/polls', (req, res) => {
    res.json(polls);
});

app.post('/polls/:pollId/vote', (req, res) => {
    const { pollId } = req.params;
    const { option } = req.body;
    const userIp = req.ip || req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (!polls[pollId]) {
        return res.status(404).json({ error: "Poll not found" });
    }

    if (polls[pollId].options[option] === undefined) {
        return res.status(400).json({ error: "Invalid option selected" });
    }

    if (!voteRegistry[pollId]) {
        voteRegistry[pollId] = new Set();
    }

    if (voteRegistry[pollId].has(userIp)) {
        return res.status(403).json({ error: "Duplicate vote detected. You have already voted on this poll." });
    }

    polls[pollId].options[option]++;
    voteRegistry[pollId].add(userIp);

    res.json({
        message: "Vote cast successfully",
        poll: polls[pollId]
    });
});

app.get('/polls/:pollId/results', (req, res) => {
    const { pollId } = req.params;
    if (!polls[pollId]) {
        return res.status(404).json({ error: "Poll not found" });
    }
    res.json(polls[pollId]);
});

app.listen(PORT, () => {
    console.log(`Voting system running on http://localhost:${PORT}`);
});
