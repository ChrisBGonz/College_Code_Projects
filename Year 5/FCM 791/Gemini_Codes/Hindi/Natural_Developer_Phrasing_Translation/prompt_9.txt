class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!

  def promote
    @user = User.find(params[:id])

    if @user.role == 'staff'
      if @user.update(role: 'manager')
        redirect_to @user, notice: 'Staff member has been successfully promoted to Manager.'
      else
        redirect_to @user, alert: 'Failed to promote the user.'
      end
    else
      redirect_to @user, alert: 'Only users with the Staff role can be promoted.'
    end
  end

  private

  def authorize_manager!
    unless current_user.role == 'manager'
      redirect_to root_path, alert: 'You do not have permission to perform this action.'
    end
  end
end
