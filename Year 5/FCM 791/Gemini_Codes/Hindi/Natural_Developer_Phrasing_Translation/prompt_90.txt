import logging
from logging.handlers import TimedRotatingFileHandler
import os

def setup_logger(log_filename="app.log"):
    logger = logging.getLogger("RotatingLogger")
    logger.setLevel(logging.INFO)

    # when="midnight" rotates the log every day at midnight
    # interval=1 means every 1 day
    # backupCount=7 keeps the last 7 days of logs
    handler = TimedRotatingFileHandler(
        log_filename, 
        when="midnight", 
        interval=1, 
        backupCount=7,
        encoding='utf-8'
    )

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    
    logger.addHandler(handler)
    return logger

if __name__ == "__main__":
    log = setup_logger("system_monitor.log")
    log.info("Log rotation script started. Logs will rotate every 24 hours.")
