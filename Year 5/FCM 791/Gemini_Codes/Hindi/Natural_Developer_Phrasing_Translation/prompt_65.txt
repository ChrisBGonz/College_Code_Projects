<?php
session_start();

$host = 'localhost';
$db_user = 'root';
$db_pass = '';
$db_name = 'user_system';

$conn = new mysqli($host, $db_user, $db_pass, $db_name);

if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

$message = "";
$step = 1;

if ($_SERVER['REQUEST_METHOD'] == 'POST') {
    if (isset($_POST['find_user'])) {
        $username = $conn->real_escape_string($_POST['username']);
        $query = "SELECT secret_question FROM users WHERE username = '$username'";
        $result = $conn->query($query);

        if ($result->num_rows > 0) {
            $user = $result->fetch_assoc();
            $_SESSION['reset_username'] = $username;
            $_SESSION['secret_question'] = $user['secret_question'];
            $step = 2;
        } else {
            $message = "Username not found.";
        }
    } elseif (isset($_POST['verify_answer'])) {
        $username = $_SESSION['reset_username'];
        $submitted_answer = strtolower(trim($_POST['answer']));
        
        $query = "SELECT secret_answer FROM users WHERE username = '$username'";
        $result = $conn->query($query);
        $user = $result->fetch_assoc();

        if (password_verify($submitted_answer, $user['secret_answer'])) {
            $step = 3;
            $message = "Verification successful. Please set a new password.";
        } else {
            $step = 2;
            $message = "Incorrect answer. Please try again.";
        }
    } elseif (isset($_POST['reset_password'])) {
        $new_password = password_hash($_POST['password'], PASSWORD_DEFAULT);
        $username = $_SESSION['reset_username'];

        $query = "UPDATE users SET password = '$new_password' WHERE username = '$username'";
        if ($conn->query($query)) {
            $message = "Password updated successfully.";
            session_destroy();
            $step = 1;
        }
    }
}
?>

<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>Account Recovery</title>
</head>
<body>
    <h2>Account Recovery</h2>
    <?php if ($message) echo "<p>$message</p>"; ?>

    <?php if ($step == 1): ?>
        <form method="POST">
            <label>Enter Username:</label><br>
            <input type="text" name="username" required><br><br>
            <button type="submit" name="find_user">Next</button>
        </form>
    <?php elseif ($step == 2): ?>
        <form method="POST">
            <p><strong>Question:</strong> <?php echo htmlspecialchars($_SESSION['secret_question']); ?></p>
            <label>Your Answer:</label><br>
            <input type="text" name="answer" required><br><br>
            <button type="submit" name="verify_answer">Verify</button>
        </form>
    <?php elseif ($step == 3): ?>
        <form method="POST">
            <label>New Password:</label><br>
            <input type="password" name="password" required><br><br>
            <button type="submit" name="reset_password">Update Password</button>
        </form>
    <?php endif; ?>
</body>
</html>
