<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'en', $fontSize = '14px') {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

$cookieName = "user_prefs_cookie";

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $userPrefs = unserialize(base64_decode($serializedData));
} else {
    $userPrefs = new UserPreferences();
}

if (isset($_GET['update'])) {
    $userPrefs->theme = $_GET['theme'] ?? $userPrefs->theme;
    $userPrefs->language = $_GET['lang'] ?? $userPrefs->language;
    
    $encodedData = base64_encode(serialize($userPrefs));
    setcookie($cookieName, $encodedData, time() + (86400 * 30), "/");
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}

?>
<!DOCTYPE html>
<html>
<head>
    <title>User Preferences</title>
    <style>
        body { 
            background-color: <?php echo $userPrefs->theme === 'dark' ? '#333' : '#fff'; ?>; 
            color: <?php echo $userPrefs->theme === 'dark' ? '#fff' : '#333'; ?>;
            font-size: <?php echo $userPrefs->fontSize; ?>;
        }
    </style>
</head>
<body>
    <h1>User Settings</h1>
    <p>Current Language: <?php echo htmlspecialchars($userPrefs->language); ?></p>
    <p>Current Theme: <?php echo htmlspecialchars($userPrefs->theme); ?></p>

    <form method="GET">
        <input type="hidden" name="update" value="1">
        <select name="theme">
            <option value="light" <?php if($userPrefs->theme == 'light') echo 'selected'; ?>>Light</option>
            <option value="dark" <?php if($userPrefs->theme == 'dark') echo 'selected'; ?>>Dark</option>
        </select>
        <select name="lang">
            <option value="en" <?php if($userPrefs->language == 'en') echo 'selected'; ?>>English</option>
            <option value="hi" <?php if($userPrefs->language == 'hi') echo 'selected'; ?>>Hindi</option>
        </select>
        <button type="submit">Save Preferences</button>
    </form>
</body>
</html>
