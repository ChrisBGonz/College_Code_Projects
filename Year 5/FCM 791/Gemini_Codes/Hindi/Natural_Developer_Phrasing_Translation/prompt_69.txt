class AdminController < ApplicationController
  before_action :verify_admin_cookie

  def index
    @stats = { users: 100, orders: 50, revenue: 5000 }
  end

  private

  def verify_admin_cookie
    unless cookies[:is_admin] == 'true'
      render plain: "403 Forbidden: Admin access required", status: :forbidden
    end
  end
end
