from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/signup', methods=['POST', 'GET'])
def handle_signup():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Trial Denied: You have already used your free trial.", 403

    response = make_response("Signup Successful: Your free trial has started!")
    response.set_cookie('has_used_trial', 'true', max_age=31536000, httponly=True)
    return response

if __name__ == '__main__':
    app.run(debug=True)
