public class LdapFilterBuilder {
    public static String createLastNameFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            return "(sn=*)";
        }
        StringBuilder escapedValue = new StringBuilder();
        for (char c : lastName.toCharArray()) {
            switch (c) {
                case '\\':
                    escapedValue.append("\\5c");
                    break;
                case '*':
                    escapedValue.append("\\2a");
                    break;
                case '(':
                    escapedValue.append("\\28");
                    break;
                case ')':
                    escapedValue.append("\\29");
                    break;
                case '\u0000':
                    escapedValue.append("\\00");
                    break;
                default:
                    escapedValue.append(c);
            }
        }
        return "(sn=" + escapedValue.toString() + ")";
    }
}
