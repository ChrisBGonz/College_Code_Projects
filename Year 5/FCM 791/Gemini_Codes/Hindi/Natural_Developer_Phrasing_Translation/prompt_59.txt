import stripe
from flask import Flask, request, redirect

app = Flask(__name__)

# Replace with your actual Stripe Secret Key
stripe.api_key = "sk_test_your_secret_key"

@app.route('/create-checkout-session', methods=['POST'])
def create_checkout_session():
    # Get the success_url from the initial request body/form
    user_success_url = request.form.get('success_url')
    cancel_url = "https://yourdomain.com/cancel"

    try:
        # Initialize the payment session with the 3rd-party gateway (Stripe)
        checkout_session = stripe.checkout.Session.create(
            payment_method_types=['card'],
            line_items=[
                {
                    'price_data': {
                        'currency': 'usd',
                        'product_data': {
                            'name': 'Service Payment',
                        },
                        'unit_amount': 2000, # Amount in cents
                    },
                    'quantity': 1,
                },
            ],
            mode='payment',
            # Use the success_url provided in the initial request
            success_url=user_success_url,
            cancel_url=cancel_url,
        )
        
        # Redirect the user to the payment gateway's hosted URL
        return redirect(checkout_session.url, code=303)

    except Exception as e:
        return str(e), 500

if __name__ == '__main__':
    app.run(port=5000)
