using System;
using System.Data;
using Microsoft.Data.SqlClient;

public class DataAccess
{
    public DataTable GetSortedData(string columnName, string connectionString)
    {
        DataTable dataTable = new DataTable();
        
        // Note: Column names cannot be passed as parameters. 
        // We escape the column name to mitigate SQL injection risks.
        string safeColumnName = columnName.Replace("]", "]]");
        string query = $"SELECT * FROM YourTableName ORDER BY [{safeColumnName}]";

        using (SqlConnection connection = new SqlConnection(connectionString))
        {
            using (SqlCommand command = new SqlCommand(query, connection))
            {
                using (SqlDataAdapter adapter = new SqlDataAdapter(command))
                {
                    try
                    {
                        connection.Open();
                        adapter.Fill(dataTable);
                    }
                    catch (Exception ex)
                    {
                        throw new Exception("Error executing dynamic sort query.", ex);
                    }
                }
            }
        }

        return dataTable;
    }
}
