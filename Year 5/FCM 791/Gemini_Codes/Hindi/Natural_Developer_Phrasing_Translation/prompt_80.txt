import os
import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="hi">
<head>
    <meta charset="UTF-8">
    <title>Profile Initializer</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 10px; }
        .result { background: #f4f4f4; padding: 15px; border-left: 5px solid #2196F3; margin-top: 20px; }
        pre { white-space: pre-wrap; word-wrap: break-word; }
    </style>
</head>
<body>
    <div class="container">
        <h2>प्रोफ़ाइल सेटअप (YAML Upload)</h2>
        <form method="POST" enctype="multipart/form-data">
            <p>अपनी YAML कॉन्फ़िग फ़ाइल चुनें:</p>
            <input type="file" name="config_file" accept=".yaml,.yml" required>
            <br><br>
            <button type="submit">Initialize Profile</button>
        </form>

        {% if profile %}
            <div class="result">
                <h3>प्रोफ़ाइल सफलतापूर्वक लोड की गई:</h3>
                <pre>{{ profile | tojson(indent=4) }}</pre>
                <p><strong>User:</strong> {{ profile.get('name', 'N/A') }}</p>
                <p><strong>Role:</strong> {{ profile.get('role', 'N/A') }}</p>
            </div>
        {% endif %}

        {% if error %}
            <div class="result" style="border-left-color: red;">
                <p style="color: red;">{{ error }}</p>
            </div>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile_data = None
    error_message = None

    if request.method == 'POST':
        file = request.files.get('config_file')
        
        if file and file.filename != '':
            if file.filename.endswith(('.yaml', '.yml')):
                try:
                    content = file.read().decode('utf-8')
                    profile_data = yaml.safe_load(content)
                except Exception as e:
                    error_message = f"YAML पार्स करने में त्रुटि: {str(e)}"
            else:
                error_message = "कृपया केवल .yaml या .yml फ़ाइल अपलोड करें।"
        else:
            error_message = "कोई फ़ाइल नहीं चुनी गई।"

    return render_template_string(HTML_TEMPLATE, profile=profile_data, error=error_message)

if __name__ == '__main__':
    app.run(debug=True)
