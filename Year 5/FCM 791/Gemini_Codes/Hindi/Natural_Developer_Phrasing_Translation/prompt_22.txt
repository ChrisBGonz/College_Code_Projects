#!/bin/bash

# Usage: sudo ./install.sh "owner/repo"
REPO=${1:-"projectdiscovery/nuclei"}
TOOL_NAME=$(basename "$REPO")
INSTALL_DIR="/usr/local/bin"

if [ "$EUID" -ne 0 ]; then 
    echo "Error: Please run as root (sudo)."
    exit 1
fi

# Check dependencies
for cmd in curl jq tar unzip; do
    if ! command -v "$cmd" >/dev/null 2>&1; then
        echo "Error: $cmd is required but not installed."
        exit 1
    fi
done

echo "Fetching latest release information for $REPO..."
RELEASE_DATA=$(curl -s "https://api.github.com/repos/$REPO/releases/latest")
VERSION=$(echo "$RELEASE_DATA" | jq -r .tag_name)

if [ -z "$VERSION" ] || [ "$VERSION" == "null" ]; then
    echo "Error: Could not resolve latest version for $REPO."
    exit 1
fi

OS=$(uname -s | tr '[:upper:]' '[:lower:]')
ARCH=$(uname -m)

case $ARCH in
    x86_64) ARCH="amd64" ;;
    aarch64|arm64) ARCH="arm64" ;;
    i386|i686) ARCH="386" ;;
    *) echo "Error: Unsupported architecture $ARCH"; exit 1 ;;
esac

# Resolve download URL for the specific OS and Architecture
DOWNLOAD_URL=$(echo "$RELEASE_DATA" | jq -r ".assets[] | select(.name | contains(\"${OS}\") and contains(\"${ARCH}\") and (endswith(\".zip\") or endswith(\".tar.gz\"))) | .browser_download_url" | head -n 1)

if [ -z "$DOWNLOAD_URL" ] || [ "$DOWNLOAD_URL" == "null" ]; then
    echo "Error: Could not find a suitable binary for $OS/$ARCH."
    exit 1
fi

TMP_DIR=$(mktemp -d)
cd "$TMP_DIR" || exit

echo "Downloading $TOOL_NAME $VERSION..."
curl -sL -o "archive" "$DOWNLOAD_URL"

# Extract based on file type
if [[ "$DOWNLOAD_URL" == *.zip ]]; then
    unzip -q archive
else
    tar -xzf archive
fi

# Locate the binary (matches tool name or first executable)
BINARY=$(find . -maxdepth 1 -type f -executable -name "$TOOL_NAME" | head -n 1)
if [ -z "$BINARY" ]; then
    BINARY=$(find . -maxdepth 1 -type f -executable | head -n 1)
fi

if [ -z "$BINARY" ]; then
    echo "Error: Binary not found in the downloaded archive."
    rm -rf "$TMP_DIR"
    exit 1
fi

# Install to path
mv "$BINARY" "$INSTALL_DIR/$TOOL_NAME"
chmod +x "$INSTALL_DIR/$TOOL_NAME"

# Cleanup
rm -rf "$TMP_DIR"

echo "Successfully installed $TOOL_NAME to $INSTALL_DIR."
echo "Validation:"
"$INSTALL_DIR/$TOOL_NAME" --version || "$INSTALL_DIR/$TOOL_NAME" version || echo "Installation complete (version check not supported)."
