import jwt
from jwt import PyJWKClient
from jwt.exceptions import (
    InvalidTokenError,
    ExpiredSignatureError,
    InvalidIssuerError,
    InvalidAudienceError,
    PyJWKClientError
)

def handle_jwt_authentication(token, jwks_url, expected_audience, expected_issuer):
    """
    Decodes and validates a JWT using a remote JWKS endpoint.
    Supports RS256, ES256, and standard claim verification.
    """
    try:
        # Initialize JWK Client to fetch public keys from the provider
        jwks_client = PyJWKClient(jwks_url)
        
        # Retrieve the signing key that matches the 'kid' in the token header
        signing_key = jwks_client.get_signing_key_from_jwt(token)

        # Decode and validate the token
        # Validates: Signature, exp, nbf, iat, aud, and iss
        decoded_payload = jwt.decode(
            token,
            signing_key.key,
            algorithms=["RS256", "ES256"],
            audience=expected_audience,
            issuer=expected_issuer,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iat", "iss", "aud"]
            }
        )
        
        return {
            "success": True,
            "data": decoded_payload
        }

    except ExpiredSignatureError:
        return {"success": False, "error": "Token has expired"}
    except InvalidIssuerError:
        return {"success": False, "error": "Invalid token issuer"}
    except InvalidAudienceError:
        return {"success": False, "error": "Invalid token audience"}
    except PyJWKClientError as e:
        return {"success": False, "error": f"JWKS resolution error: {str(e)}"}
    except InvalidTokenError as e:
        return {"success": False, "error": f"Invalid token: {str(e)}"}
    except Exception as e:
        return {"success": False, "error": f"Authentication failed: {str(e)}"}

if __name__ == "__main__":
    # Configuration
    JWKS_ENDPOINT = "https://example-auth-provider.com/.well-known/jwks.json"
    AUDIENCE = "api://my-service"
    ISSUER = "https://example-auth-provider.com/"
    
    # Example raw token from Authorization header
    raw_auth_header = "Bearer eyJhbGciOiJSUzI1NiIsImtpZCI6..."
    jwt_token = raw_auth_header.split(" ")[1] if " " in raw_auth_header else raw_auth_header

    result = handle_jwt_authentication(jwt_token, JWKS_ENDPOINT, AUDIENCE, ISSUER)
    
    if result["success"]:
        print("Authentication Successful")
        print("User Claims:", result["data"])
    else:
        print("Authentication Failed:", result["error"])
