<?php
$uploadDirectory = '/var/www/shared_drive/uploads/';
$maxFileSize = 5 * 1024 * 1024; 
$allowedMimeType = 'application/pdf';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['employee_pdf'])) {
    $file = $_FILES['employee_pdf'];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        die("Upload failed with error code: " . $file['error']);
    }

    if ($file['size'] > $maxFileSize) {
        die("Error: File size exceeds the 5MB limit.");
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $actualMimeType = $finfo->file($file['tmp_name']);
    if ($actualMimeType !== $allowedMimeType) {
        die("Error: Only PDF files are allowed.");
    }

    $fileName = pathinfo($file['name'], PATHINFO_FILENAME);
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));

    if ($extension !== 'pdf') {
        die("Error: Invalid file extension.");
    }

    $sanitizedName = preg_replace("/[^a-zA-Z0-9\._-]/", "_", $fileName);
    $finalFileName = $sanitizedName . "_" . bin2hex(random_bytes(8)) . ".pdf";
    $destination = $uploadDirectory . $finalFileName;

    if (!is_dir($uploadDirectory)) {
        mkdir($uploadDirectory, 0755, true);
    }

    if (move_uploaded_file($file['tmp_name'], $destination)) {
        echo "File successfully uploaded to shared drive as: " . htmlspecialchars($finalFileName);
    } else {
        echo "Error: Could not save the file. Check directory permissions.";
    }
}
?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Employee Shared Drive - PDF Upload</title>
</head>
<body>
    <h2>Upload PDF to Shared Drive</h2>
    <form action="" method="POST" enctype="multipart/form-data">
        <label for="employee_pdf">Select PDF File (Max 5MB):</label><br><br>
        <input type="file" name="employee_pdf" id="employee_pdf" accept="application/pdf" required>
        <br><br>
        <button type="submit">Upload File</button>
    </form>
</body>
</html>
