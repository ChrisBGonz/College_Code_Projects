function initAutoLogout(onLogout, timeoutInMinutes = 15) {
    const timeoutMs = timeoutInMinutes * 60 * 1000;
    let timeoutId;

    const events = [
        'mousedown', 
        'mousemove', 
        'keypress', 
        'scroll', 
        'touchstart', 
        'click', 
        'keydown'
    ];

    const handleLogout = () => {
        destroy();
        if (typeof onLogout === 'function') {
            onLogout();
        }
    };

    const resetTimer = () => {
        if (timeoutId) clearTimeout(timeoutId);
        timeoutId = setTimeout(handleLogout, timeoutMs);
    };

    const setup = () => {
        events.forEach(event => {
            window.addEventListener(event, resetTimer, { passive: true });
        });
        resetTimer();
    };

    const destroy = () => {
        if (timeoutId) clearTimeout(timeoutId);
        events.forEach(event => {
            window.removeEventListener(event, resetTimer);
        });
    };

    setup();

    return {
        destroy,
        reset: resetTimer
    };
}

const sessionManager = initAutoLogout(() => {
    console.log("Session expired. Logging out...");
    localStorage.clear();
    sessionStorage.clear();
    window.location.href = '/login?reason=timeout';
}, 15);
