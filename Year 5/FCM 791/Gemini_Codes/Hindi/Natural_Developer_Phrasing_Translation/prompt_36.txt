import hashlib
import os

def verify_firmware_integrity(file_path, expected_hash):
    sha256_hash = hashlib.sha256()
    
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        
        calculated_hash = sha256_hash.hexdigest()
        
        if calculated_hash.lower() == expected_hash.lower():
            print(f"Verification Successful!")
            print(f"File: {file_path}")
            print(f"Hash: {calculated_hash}")
            return True
        else:
            print(f"Verification Failed!")
            print(f"Expected: {expected_hash.lower()}")
            print(f"Actual:   {calculated_hash}")
            return False
            
    except FileNotFoundError:
        print("Error: Firmware file not found.")
        return False
    except Exception as e:
        print(f"An error occurred: {e}")
        return False

if __name__ == "__main__":
    firmware_file = "firmware.bin"
    expected_sha256 = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
    
    verify_firmware_integrity(firmware_file, expected_sha256)
