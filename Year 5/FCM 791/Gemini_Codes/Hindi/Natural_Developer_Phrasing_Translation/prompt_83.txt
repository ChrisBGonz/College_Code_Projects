import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseService {

    private static final Logger logger = Logger.getLogger(DatabaseService.class.getName());

    public void performDatabaseOperation() throws DataAccessException {
        try {
            // Database logic here
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Database error encountered: " + e.getMessage(), e);
            throw new DataAccessException("An error occurred while processing the database request.", e);
        }
    }
}

class DataAccessException extends Exception {
    public DataAccessException(String message, Throwable cause) {
        super(message, cause);
    }
}
