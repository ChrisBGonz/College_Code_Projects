using System;
using Microsoft.Extensions.Logging;

public class SecurityAuditService
{
    private readonly ILogger<SecurityAuditService> _logger;

    public SecurityAuditService(ILogger<SecurityAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaSettingsChanged(string userId, string actionType, string ipAddress, object changeDetails)
    {
        var eventId = Guid.NewGuid();
        var timestamp = DateTime.UtcNow;

        _logger.LogInformation(
            "SecurityEvent: MFA_SETTINGS_CHANGED | EventId: {EventId} | UserId: {UserId} | Action: {Action} | IP: {IP} | Timestamp: {Timestamp} | Details: {@Details}",
            eventId,
            userId,
            actionType,
            ipAddress,
            timestamp,
            changeDetails
        );
    }
}
