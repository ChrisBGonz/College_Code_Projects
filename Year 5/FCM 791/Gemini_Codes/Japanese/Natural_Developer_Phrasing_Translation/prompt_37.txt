package main

import (
	"crypto/hmac"
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/binary"
	"errors"
	"time"
)

func GenerateResetToken(secret []byte) (string, error) {
	payload := make([]byte, 40)
	expiresAt := time.Now().Add(1 * time.Hour).Unix()
	binary.BigEndian.PutUint64(payload[:8], uint64(expiresAt))

	if _, err := rand.Read(payload[8:]); err != nil {
		return "", err
	}

	mac := hmac.New(sha256.New, secret)
	mac.Write(payload)
	signature := mac.Sum(nil)

	tokenData := append(payload, signature...)
	return base64.URLEncoding.EncodeToString(tokenData), nil
}

func VerifyResetToken(tokenStr string, secret []byte) (bool, error) {
	data, err := base64.URLEncoding.DecodeString(tokenStr)
	if err != nil {
		return false, err
	}

	if len(data) != 72 {
		return false, errors.New("invalid token length")
	}

	payload := data[:40]
	signature := data[40:]

	mac := hmac.New(sha256.New, secret)
	mac.Write(payload)
	expectedSignature := mac.Sum(nil)

	if !hmac.Equal(signature, expectedSignature) {
		return false, errors.New("invalid signature")
	}

	expiresAt := int64(binary.BigEndian.Uint64(payload[:8]))
	if time.Now().Unix() > expiresAt {
		return false, errors.New("token expired")
	}

	return true, nil
}
