import pyotp

def verify_totp(secret_key, input_code):
    totp = pyotp.TOTP(secret_key)
    return totp.verify(input_code)

if __name__ == "__main__":
    # In a real application, the secret key is unique per user and stored in a database.
    # It is typically shared with the user via a QR code during 2FA setup.
    user_secret = "JBSWY3DPEHPK3PXP"
    
    print(f"Secret Key: {user_secret}")
    user_code = input("Enter the 6-digit TOTP code: ")
    
    if verify_totp(user_secret, user_code):
        print("Verification successful. Access granted.")
    else:
        print("Verification failed. Access denied.")
