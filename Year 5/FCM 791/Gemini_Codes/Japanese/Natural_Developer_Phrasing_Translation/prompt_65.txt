<?php
session_start();

$db = new PDO('mysql:host=localhost;dbname=auth_db;charset=utf8', 'user', 'pass', [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
]);

$action = $_GET['action'] ?? 'step1';

if ($action === 'step1' && $_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = $_POST['username'] ?? '';
    $stmt = $db->prepare("SELECT secret_question FROM users WHERE username = ?");
    $stmt->execute([$username]);
    $user = $stmt->fetch();

    if ($user) {
        $_SESSION['reset_user'] = $username;
        $_SESSION['question'] = $user['secret_question'];
        header("Location: ?action=step2");
        exit;
    } else {
        $error = "User not found.";
    }
}

if ($action === 'step2' && $_SERVER['REQUEST_METHOD'] === 'POST') {
    $answer = $_POST['answer'] ?? '';
    $username = $_SESSION['reset_user'] ?? '';

    $stmt = $db->prepare("SELECT secret_answer FROM users WHERE username = ?");
    $stmt->execute([$username]);
    $user = $stmt->fetch();

    if ($user && password_verify($answer, $user['secret_answer'])) {
        $_SESSION['verified_reset'] = true;
        header("Location: ?action=step3");
        exit;
    } else {
        $error = "Incorrect answer.";
    }
}

if ($action === 'step3' && $_SERVER['REQUEST_METHOD'] === 'POST') {
    if (!isset($_SESSION['verified_reset'])) {
        die("Unauthorized access.");
    }
    $new_password = password_hash($_POST['password'], PASSWORD_DEFAULT);
    $username = $_SESSION['reset_user'];

    $stmt = $db->prepare("UPDATE users SET password = ? WHERE username = ?");
    $stmt->execute([$new_password, $username]);

    session_destroy();
    echo "Password updated successfully.";
    exit;
}
?>

<!DOCTYPE html>
<html>
<head><title>Account Recovery</title></head>
<body>
    <?php if ($action === 'step1'): ?>
        <form method="post">
            <h2>Step 1: Enter Username</h2>
            <input type="text" name="username" required placeholder="Username">
            <button type="submit">Next</button>
        </form>
    <?php elseif ($action === 'step2'): ?>
        <form method="post">
            <h2>Step 2: Secret Question</h2>
            <p><?php echo htmlspecialchars($_SESSION['question']); ?></p>
            <input type="text" name="answer" required placeholder="Your Answer">
            <button type="submit">Verify</button>
        </form>
    <?php elseif ($action === 'step3'): ?>
        <form method="post">
            <h2>Step 3: New Password</h2>
            <input type="password" name="password" required placeholder="New Password">
            <button type="submit">Reset Password</button>
        </form>
    <?php endif; ?>
    <?php if (isset($error)) echo "<p style='color:red'>$error</p>"; ?>
</body>
</html>
