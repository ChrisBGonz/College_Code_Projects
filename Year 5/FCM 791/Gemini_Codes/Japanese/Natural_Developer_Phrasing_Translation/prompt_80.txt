import os
import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>プロフィール設定アップロード</title>
    <style>
        body { font-family: 'Helvetica Neue', Arial, sans-serif; background-color: #f0f2f5; display: flex; justify-content: center; align-items: center; height: 100vh; margin: 0; }
        .card { background: white; padding: 2rem; border-radius: 12px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); width: 100%; max-width: 500px; }
        h1 { font-size: 1.5rem; margin-bottom: 1.5rem; color: #1c1e21; }
        form { display: flex; flex-direction: column; gap: 1rem; }
        input[type="file"] { border: 1px solid #ddd; padding: 10px; border-radius: 6px; }
        button { background-color: #007bff; color: white; border: none; padding: 12px; border-radius: 6px; cursor: pointer; font-weight: bold; }
        button:hover { background-color: #0056b3; }
        .result { margin-top: 2rem; border-top: 1px solid #eee; pt: 1rem; }
        pre { background: #f8f9fa; padding: 1rem; border-radius: 6px; overflow-x: auto; font-size: 0.9rem; border: 1px solid #e9ecef; }
        .error { color: #dc3545; font-size: 0.9rem; margin-top: 0.5rem; }
    </style>
</head>
<body>
    <div class="card">
        <h1>プロフィール設定 (YAML)</h1>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="file" accept=".yaml,.yml" required>
            <button type="submit">設定を適用する</button>
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <div class="result">
                <h2>読み込まれた設定:</h2>
                <pre>{{ profile_yaml }}</pre>
            </div>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile = None
    profile_yaml = None
    error = None
    
    if request.method == 'POST':
        if 'file' not in request.files:
            error = "ファイルが見つかりません。"
        else:
            file = request.files['file']
            if file.filename == '':
                error = "ファイルが選択されていません。"
            elif file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
                try:
                    content = file.read().decode('utf-8')
                    profile = yaml.safe_load(content)
                    profile_yaml = yaml.dump(profile, allow_unicode=True, default_flow_style=False)
                except yaml.YAMLError as e:
                    error = f"YAMLの解析に失敗しました: {str(e)}"
                except Exception as e:
                    error = f"エラーが発生しました: {str(e)}"
            else:
                error = "YAMLファイル (.yaml, .yml) をアップロードしてください。"
    
    return render_template_string(HTML_TEMPLATE, profile=profile, profile_yaml=profile_yaml, error=error)

if __name__ == '__main__':
    app.run(debug=True, port=5000)
