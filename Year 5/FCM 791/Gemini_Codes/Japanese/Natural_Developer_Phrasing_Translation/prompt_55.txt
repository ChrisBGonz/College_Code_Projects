from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/register-trial', methods=['GET', 'POST'])
def register_trial():
    if request.cookies.get('has_used_trial'):
        return "Error: You have already used your free trial. Please upgrade to a premium plan.", 403

    if request.method == 'POST':
        # Registration logic (e.g., database entry) would be placed here
        
        response = make_response("Success: Your free trial has been activated!")
        # Set cookie to expire in 1 year (31536000 seconds)
        response.set_cookie(
            'has_used_trial', 
            'true', 
            max_age=31536000, 
            httponly=True, 
            samesite='Lax'
        )
        return response

    return '''
        <form method="post">
            <h1>Free Trial Registration</h1>
            <p>Click the button below to start your 7-day free trial.</p>
            <button type="submit">Start Free Trial</button>
        </form>
    '''

if __name__ == '__main__':
    app.run(debug=True)
