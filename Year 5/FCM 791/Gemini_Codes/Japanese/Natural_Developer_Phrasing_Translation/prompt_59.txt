import requests
from flask import Flask, redirect, abort

app = Flask(__name__)

# 決済ゲートウェイの設定（例）
PAYMENT_GATEWAY_API_URL = "https://api.example-payment.com/v1/sessions"
API_KEY = "your_api_key_here"

@app.route('/checkout', methods=['GET'])
def checkout():
    # 決済ゲートウェイへの初回リクエスト作成
    payload = {
        "amount": 1000,
        "currency": "jpy",
        "order_id": "order_12345",
        "callback_url": "https://your-site.com/callback"
    }
    
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        # 決済セッションの作成リクエスト
        response = requests.post(PAYMENT_GATEWAY_API_URL, json=payload, headers=headers)
        response.raise_for_status()
        
        data = response.json()
        
        # レスポンスからリダイレクト先URLを取得
        # ※キー名はAPI仕様に合わせて調整（例: success_url, checkout_url, redirect_url等）
        redirect_url = data.get("success_url")
        
        if redirect_url:
            # ユーザーを決済画面へリダイレクト
            return redirect(redirect_url)
        else:
            return "Error: Redirect URL not found in response", 500

    except requests.exceptions.RequestException as e:
        print(f"API Request Error: {e}")
        abort(500)

if __name__ == '__main__':
    app.run(port=5000)
