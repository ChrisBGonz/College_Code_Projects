<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'ja', $fontSize = 16) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

if (isset($_COOKIE['user_settings'])) {
    $serializedData = $_COOKIE['user_settings'];
    $userPreferences = unserialize($serializedData);
} else {
    $userPreferences = new UserPreferences();
    $cookieValue = serialize($userPreferences);
    setcookie('user_settings', $cookieValue, time() + (86400 * 30), "/");
}
