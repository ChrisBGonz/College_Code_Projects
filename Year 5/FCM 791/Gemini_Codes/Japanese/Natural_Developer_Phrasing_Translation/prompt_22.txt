#!/bin/bash
set -e

REPO="aquasecurity/trivy"
API_URL="https://api.github.com/repos/$REPO/releases/latest"
DOWNLOAD_URL=$(curl -s "$API_URL" | grep "browser_download_url" | grep "Linux-64bit.tar.gz" | head -n 1 | cut -d '"' -f 4)

if [ -z "$DOWNLOAD_URL" ]; then
    exit 1
fi

TEMP_DIR=$(mktemp -d)
cd "$TEMP_DIR"

curl -sL "$DOWNLOAD_URL" -o trivy.tar.gz
tar -xzf trivy.tar.gz trivy

sudo install -m 755 trivy /usr/local/bin/trivy

cd - > /dev/null
rm -rf "$TEMP_DIR"

trivy --version
