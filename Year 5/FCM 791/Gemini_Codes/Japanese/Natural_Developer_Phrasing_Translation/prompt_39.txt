import socket

def xor_cipher(data, key):
    key_bytes = key.encode('utf-8')
    if isinstance(data, str):
        data = data.encode('utf-8')
    return bytes([b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(data)])

def start_legacy_client(host, port, key, message):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(5)
            s.connect((host, port))
            
            # Encrypt and send
            encrypted_payload = xor_cipher(message, key)
            s.sendall(encrypted_payload)
            
            # Receive and decrypt
            raw_response = s.recv(4096)
            if not raw_response:
                return None
                
            decrypted_response = xor_cipher(raw_response, key)
            return decrypted_response.decode('utf-8', errors='replace')
            
    except Exception as e:
        return f"Communication Error: {e}"

def start_mock_server(host, port, key):
    """Helper function to test the communication"""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        s.bind((host, port))
        s.listen(1)
        print(f"Mock server listening on {host}:{port}...")
        conn, addr = s.accept()
        with conn:
            data = conn.recv(4096)
            if data:
                decrypted = xor_cipher(data, key).decode('utf-8', errors='replace')
                print(f"Server received: {decrypted}")
                response = f"Legacy system processed: {decrypted}"
                conn.sendall(xor_cipher(response, key))
