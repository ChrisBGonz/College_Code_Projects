import subprocess
import os
import sys

def run_command(command):
    process = subprocess.Popen(command, shell=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    stdout, stderr = process.communicate()
    if process.returncode != 0:
        print(f"Error executing: {command}\n{stderr}")
        sys.exit(1)
    return stdout.strip()

def deploy():
    image_name = os.getenv("DOCKER_IMAGE_NAME")
    container_name = os.getenv("CONTAINER_NAME")
    port_mapping = os.getenv("PORT_MAPPING", "80:80")
    env_file = os.getenv("ENV_FILE_PATH", ".env.prod")

    if not image_name or not container_name:
        print("Missing required environment variables: DOCKER_IMAGE_NAME or CONTAINER_NAME")
        sys.exit(1)

    print(f"Pulling latest image: {image_name}")
    run_command(f"docker pull {image_name}")

    print(f"Checking for existing container: {container_name}")
    existing_container = run_command(f"docker ps -aq -f name=^/{container_name}$")

    if existing_container:
        print(f"Stopping and removing old container: {container_name}")
        run_command(f"docker stop {container_name}")
        run_command(f"docker rm {container_name}")

    print(f"Starting new container: {container_name}")
    run_flags = [
        "docker run -d",
        f"--name {container_name}",
        "--restart always",
        f"-p {port_mapping}"
    ]

    if os.path.exists(env_file):
        run_flags.append(f"--env-file {env_file}")

    run_flags.append(image_name)
    run_command(" ".join(run_flags))

    print("Cleaning up dangling images...")
    subprocess.run("docker image prune -f", shell=True)

    print(f"Deployment of {image_name} to {container_name} successful.")

if __name__ == "__main__":
    deploy()
