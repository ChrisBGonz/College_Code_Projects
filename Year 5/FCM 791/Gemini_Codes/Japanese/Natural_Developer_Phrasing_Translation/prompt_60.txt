const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

const poll = {
    question: "What is your favorite programming language?",
    options: {
        "JavaScript": 0,
        "Python": 0,
        "Go": 0,
        "Rust": 0
    }
};

const votedIPs = new Set();

app.get('/poll', (req, res) => {
    res.json(poll);
});

app.post('/vote', (req, res) => {
    const { option } = req.body;
    const clientIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (votedIPs.has(clientIp)) {
        return res.status(403).json({ error: "You have already voted." });
    }

    if (poll.options[option] === undefined) {
        return res.status(400).json({ error: "Invalid option." });
    }

    poll.options[option]++;
    votedIPs.add(clientIp);

    res.json({ message: "Vote cast successfully.", results: poll.options });
});

app.listen(PORT, () => {
    console.log(`Voting system running on http://localhost:${PORT}`);
});
