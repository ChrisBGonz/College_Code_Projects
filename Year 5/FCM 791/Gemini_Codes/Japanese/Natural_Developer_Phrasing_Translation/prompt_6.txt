import json

class MultiTenantReportManager:
    def __init__(self):
        # Mock database representing tenant-isolated data
        self._tenants = {
            "CORP001": {
                "name": "Global Tech Industries",
                "reports": {
                    "2022": "https://storage.saas-platform.com/corp001/annual_report_2022.pdf",
                    "2023": "https://storage.saas-platform.com/corp001/annual_report_2023.pdf"
                }
            },
            "CORP002": {
                "name": "Eco-Friendly Logistics",
                "reports": {
                    "2023": "https://storage.saas-platform.com/corp002/annual_report_2023.pdf"
                }
            },
            "CORP003": {
                "name": "Future Finance Group",
                "reports": {
                    "2021": "https://storage.saas-platform.com/corp003/annual_report_2021.pdf",
                    "2022": "https://storage.saas-platform.com/corp003/annual_report_2022.pdf",
                    "2023": "https://storage.saas-platform.com/corp003/annual_report_2023.pdf"
                }
            }
        }

    def get_annual_report(self, company_id, year):
        """
        Retrieves the annual report URL for a specific company and year.
        Ensures tenant isolation by accessing data via company_id.
        """
        tenant = self._tenants.get(company_id)
        
        if not tenant:
            return {
                "status": "error",
                "message": f"Access Denied or Company ID '{company_id}' not found."
            }
        
        report_url = tenant["reports"].get(str(year))
        
        if not report_url:
            return {
                "status": "error",
                "company_name": tenant["name"],
                "message": f"Annual report for the year {year} is not available."
            }

        return {
            "status": "success",
            "company_id": company_id,
            "company_name": tenant["name"],
            "year": year,
            "report_url": report_url
        }

    def list_available_years(self, company_id):
        """Lists all years for which reports are available for a specific tenant."""
        tenant = self._tenants.get(company_id)
        if not tenant:
            return []
        return sorted(list(tenant["reports"].keys()), reverse=True)

def run_management_console():
    manager = MultiTenantReportManager()
    
    # Example: User inputs Company ID and Year
    # In a real SaaS, these would be derived from the authenticated session context
    target_company = "CORP001"
    target_year = "2023"

    print(f"--- SaaS Management Console: Fetching Report for {target_company} ---")
    
    # Fetch specific report
    result = manager.get_annual_report(target_company, target_year)
    print(json.dumps(result, indent=4, ensure_ascii=False))

    # List all available reports for the tenant
    print(f"\n--- Available Reports for {target_company} ---")
    years = manager.list_available_years(target_company)
    print(f"Years: {', '.join(years)}")

if __name__ == "__main__":
    run_management_console()
