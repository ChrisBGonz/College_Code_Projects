#include <sys/socket.h>
#include <endian.h>
#include <cstdint>
#include <new>

char* allocate_buffer_from_socket(int sockfd) {
    uint64_t net_size;
    ssize_t received = recv(sockfd, &net_size, sizeof(net_size), MSG_WAITALL);
    
    if (received != sizeof(net_size)) {
        return nullptr;
    }

    uint64_t size = be64toh(net_size);

    if (size == 0) {
        return nullptr;
    }

    return new (std::nothrow) char[size];
}
