using System;
using System.Text.Json;
using Microsoft.Extensions.Logging;

public enum MfaChangeType
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    SecretReset
}

public class MfaAuditLogger
{
    private readonly ILogger<MfaAuditLogger> _logger;

    public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
    {
        _logger = logger;
    }

    public void LogMfaChange(string userId, MfaChangeType changeType, string ipAddress, string userAgent)
    {
        var auditRecord = new
        {
            Timestamp = DateTimeOffset.UtcNow,
            Category = "SecurityAudit",
            EventName = "MfaSettingsChanged",
            UserId = userId,
            Action = changeType.ToString(),
            ClientIp = ipAddress,
            UserAgent = userAgent,
            Result = "Success"
        };

        _logger.LogWarning("AUDIT_EVENT: {AuditRecord}", JsonSerializer.Serialize(auditRecord));
    }
}
