import pickle

class GameState:
    def __init__(self, player_name, level, inventory, position):
        self.player_name = player_name
        self.level = level
        self.inventory = inventory
        self.position = position

    def __str__(self):
        return (f"Player: {self.player_name}\n"
                f"Level: {self.level}\n"
                f"Inventory: {', '.join(self.inventory)}\n"
                f"Position: x={self.position['x']}, y={self.position['y']}")

def save_game(state, filename="savegame.pkl"):
    try:
        with open(filename, "wb") as f:
            pickle.dump(state, f)
        print(f"Game saved successfully to {filename}")
    except Exception as e:
        print(f"Error saving game: {e}")

def load_game(filename="savegame.pkl"):
    try:
        with open(filename, "rb") as f:
            state = pickle.load(f)
        print(f"Game loaded successfully from {filename}")
        return state
    except FileNotFoundError:
        print("Save file not found.")
        return None
    except Exception as e:
        print(f"Error loading game: {e}")
        return None

if __name__ == "__main__":
    original_state = GameState(
        player_name="Adventurer",
        level=15,
        inventory=["Excalibur", "Health Potion", "Map"],
        position={"x": 120.5, "y": 450.0}
    )

    print("--- Original State ---")
    print(original_state)

    save_game(original_state)

    del original_state

    loaded_state = load_game()

    if loaded_state:
        print("\n--- Loaded State ---")
        print(loaded_state)
