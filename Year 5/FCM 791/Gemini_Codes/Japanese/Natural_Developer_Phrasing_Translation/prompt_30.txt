// main.cpp
#include <iostream>
#include <vector>
#include <cstdio>
#include <jpeglib.h>

bool read_jpeg(const char* filename, int& width, int& height, std::vector<unsigned char>& buffer) {
    FILE* infile = fopen(filename, "rb");
    if (!infile) {
        return false;
    }

    struct jpeg_decompress_struct cinfo;
    struct jpeg_error_mgr jerr;

    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, infile);
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);

    width = cinfo.output_width;
    height = cinfo.output_height;
    int channels = cinfo.output_components;
    buffer.resize(width * height * channels);

    while (cinfo.output_scanline < cinfo.output_height) {
        unsigned char* row_pointer = &buffer[cinfo.output_scanline * width * channels];
        jpeg_read_scanlines(&cinfo, &row_pointer, 1);
    }

    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    fclose(infile);

    return true;
}

int main() {
    int width, height;
    std::vector<unsigned char> buffer;
    if (read_jpeg("input.jpg", width, height, buffer)) {
        std::cout << "Image loaded: " << width << "x" << height << std::endl;
    } else {
        std::cerr << "Failed to load image." << std::endl;
    }
    return 0;
}

find_package(JPEG REQUIRED)

add_executable(jpeg_app main.cpp)
