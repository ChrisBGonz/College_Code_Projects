import sys

if len(sys.argv) != 2:
    sys.exit(1)

try:
    with open(sys.argv[1], 'rb') as f:
        data = f.read()
        count = data.count(b'\n')
        if data and not data.endswith(b'\n'):
            count += 1
        print(count)
except MemoryError:
    sys.stderr.write("MemoryError: Not enough RAM to load the file.\n")
    sys.exit(1)
except Exception as e:
    sys.stderr.write(f"Error: {e}\n")
    sys.exit(1)
