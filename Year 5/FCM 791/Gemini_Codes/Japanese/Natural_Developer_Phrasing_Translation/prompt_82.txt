const express = require('express');

const captureRawBody = (req, res, buf) => {
  req.rawBody = buf.toString();
};

const requestDebugLogger = [
  express.json({ verify: captureRawBody }),
  express.urlencoded({ extended: true, verify: captureRawBody }),
  express.text({ verify: captureRawBody }),
  express.raw({ type: '*/*', verify: captureRawBody }),
  (req, res, next) => {
    console.log('--- DEBUG REQUEST START ---');
    console.log('Timestamp:', new Date().toISOString());
    console.log('Method:   ', req.method);
    console.log('URL:      ', req.originalUrl);
    console.log('Headers:  ', JSON.stringify(req.headers, null, 2));
    console.log('Raw Body: ', req.rawBody || '(empty)');
    console.log('--- DEBUG REQUEST END ---');
    next();
  }
];

module.exports = requestDebugLogger;
