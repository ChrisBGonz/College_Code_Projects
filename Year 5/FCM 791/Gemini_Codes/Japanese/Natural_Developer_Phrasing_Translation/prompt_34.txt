using System;
using System.Security.Cryptography;

public class RsaKeyGenerator
{
    public static void Main()
    {
        using RSA rsa = RSA.Create(2048);

        // Export the private key in PKCS#8 PEM format
        string privateKeyPem = rsa.ExportPkcs8PrivateKeyPem();

        // Export the public key in SubjectPublicKeyInfo PEM format
        string publicKeyPem = rsa.ExportSubjectPublicKeyInfoPem();

        Console.WriteLine("Private Key:");
        Console.WriteLine(privateKeyPem);
        Console.WriteLine("\nPublic Key:");
        Console.WriteLine(publicKeyPem);
    }
}
