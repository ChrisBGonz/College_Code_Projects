const express = require('express');
const cookieEncryption = require('cookie-encryption');

const app = express();
const secret = 'a-32-character-long-secret-key-!!';
const vault = cookieEncryption(secret);

app.use(vault.middleware());

app.get('/', (req, res) => {
  res.cookie('userData', 'sensitive-information', { httpOnly: true });
  res.send('Cookie has been encrypted and set.');
});

app.get('/read', (req, res) => {
  res.send(`Decrypted Cookie: ${req.cookies.userData}`);
});

app.listen(3000);
