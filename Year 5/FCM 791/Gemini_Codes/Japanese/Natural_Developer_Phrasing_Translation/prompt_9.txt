class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    unless current_user.role == 'manager'
      return render json: { error: 'Unauthorized' }, status: :forbidden
    end

    @user = User.find(params[:id])

    if @user.role == 'staff'
      if @user.update(role: 'manager')
        render json: { message: 'User promoted to manager successfully' }, status: :ok
      else
        render json: { errors: @user.errors.full_messages }, status: :unprocessable_entity
      end
    else
      render json: { error: 'Only staff members can be promoted' }, status: :unprocessable_entity
    end
  end
end
