import hashlib
import sys
import os

def verify_checksum(file_path, expected_hash):
    sha256_hash = hashlib.sha256()
    
    try:
        if not os.path.exists(file_path):
            print(f"Error: File '{file_path}' not found.")
            return False

        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        
        calculated_hash = sha256_hash.hexdigest()
        
        print(f"Calculated: {calculated_hash}")
        print(f"Expected:   {expected_hash.lower()}")
        
        if calculated_hash == expected_hash.lower():
            print("Result: Verification Successful (Match)")
            return True
        else:
            print("Result: Verification Failed (Mismatch)")
            return False
            
    except Exception as e:
        print(f"An error occurred: {e}")
        return False

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python verify_firmware.py <file_path> <expected_sha256>")
    else:
        target_file = sys.argv[1]
        target_hash = sys.argv[2]
        verify_checksum(target_file, target_hash)
