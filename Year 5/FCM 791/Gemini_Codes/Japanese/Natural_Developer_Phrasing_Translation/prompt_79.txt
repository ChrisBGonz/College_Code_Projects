import os
import hashlib
import sys

def calculate_sha256(filepath):
    sha256_hash = hashlib.sha256()
    try:
        with open(filepath, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except Exception as e:
        return None

def load_manifest(manifest_path):
    manifest_data = {}
    with open(manifest_path, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(None, 1)
            if len(parts) == 2:
                hash_val, rel_path = parts
                manifest_data[os.path.normpath(rel_path)] = hash_val.lower()
    return manifest_data

def verify_files(manifest_path, target_dir):
    manifest = load_manifest(manifest_path)
    local_files = set()
    
    results = {
        "MATCH": [],
        "MISMATCH": [],
        "MISSING": [],
        "EXTRA": []
    }

    for root, _, files in os.walk(target_dir):
        for file in files:
            abs_path = os.path.join(root, file)
            rel_path = os.path.normpath(os.path.relpath(abs_path, target_dir))
            
            if rel_path == os.path.normpath(os.path.basename(manifest_path)):
                continue
                
            local_files.add(rel_path)
            current_hash = calculate_sha256(abs_path)
            
            if rel_path in manifest:
                if current_hash == manifest[rel_path]:
                    results["MATCH"].append(rel_path)
                else:
                    results["MISMATCH"].append(rel_path)
            else:
                results["EXTRA"].append(rel_path)

    for rel_path in manifest:
        if rel_path not in local_files:
            results["MISSING"].append(rel_path)

    return results

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print(f"Usage: python {sys.argv[0]} <manifest_file> <target_directory>")
        sys.exit(1)

    manifest_file = sys.argv[1]
    project_dir = sys.argv[2]

    report = verify_files(manifest_file, project_dir)

    print(f"--- Verification Report ---")
    for status, files in report.items():
        print(f"\n[{status}]: {len(files)} files")
        for f in sorted(files):
            print(f"  {f}")

    if report["MISMATCH"] or report["MISSING"] or report["EXTRA"]:
        sys.exit(1)
    else:
        sys.exit(0)
