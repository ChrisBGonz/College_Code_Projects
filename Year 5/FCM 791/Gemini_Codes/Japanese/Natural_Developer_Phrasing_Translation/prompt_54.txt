using System;
using System.Linq;

public class CheckoutService
{
    public decimal CalculateTotal(string[] items, decimal[] prices)
    {
        if (items == null || prices == null)
        {
            return 0m;
        }

        if (items.Length != prices.Length)
        {
            throw new ArgumentException("The number of items and prices must match.");
        }

        return prices.Sum();
    }
}
