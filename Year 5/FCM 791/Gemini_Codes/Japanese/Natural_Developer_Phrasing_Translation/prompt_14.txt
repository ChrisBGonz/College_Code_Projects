from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    auth_token = request.cookies.get('auth_token')
    if auth_token == 'secret-session-token':
        return 'Welcome back! <a href="/logout">Logout</a>'
    return 'You are not logged in. <form action="/login" method="post"><input type="submit" value="Login"></form>'

@app.route('/login', methods=['POST'])
def login():
    # In a real application, you would verify username and password here.
    response = make_response(redirect(url_for('index')))
    response.set_cookie('auth_token', 'secret-session-token', httponly=True, samesite='Lax')
    return response

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('index')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True)
