import { authenticate, generateSession } from '@company-corp/auth-logic';

async function loginFlow(username, password) {
  try {
    const user = await authenticate(username, password);
    if (!user) {
      throw new Error('Invalid credentials');
    }
    const session = await generateSession(user.id);
    return {
      success: true,
      user,
      token: session.token
    };
  } catch (error) {
    return {
      success: false,
      message: error.message
    };
  }
}
