<?php
session_start();

/**
 * Configuration
 */
$admin_email = 'admin@example.com';
$slack_webhook_url = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';
$is_authorized = isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';

if (!$is_authorized) {
    $ip = $_SERVER['REMOTE_ADDR'];
    $uri = $_SERVER['REQUEST_URI'];
    $ua = $_SERVER['HTTP_USER_AGENT'];
    $time = date('Y-m-d H:i:s');

    $alert_message = "Unauthorized access attempt detected!\n";
    $alert_message .= "Time: $time\n";
    $alert_message .= "IP Address: $ip\n";
    $alert_message .= "Requested URI: $uri\n";
    $alert_message .= "User Agent: $ua\n";

    // 1. Send Email Alert
    $subject = "[Security Alert] Unauthorized Access to /admin";
    $headers = "From: security-system@example.com";
    mail($admin_email, $subject, $alert_message, $headers);

    // 2. Send Slack Notification
    $slack_data = json_encode([
        'text' => "🚨 *Administrator Alert* 🚨\n" . $alert_message
    ]);

    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slack_data);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($slack_data)
    ]);
    curl_exec($ch);
    curl_close($ch);

    // 3. Block Access
    http_response_code(403);
    exit('Access Denied: Unauthorized access has been logged and reported.');
}

// Admin content starts here
echo "Welcome to the Admin Dashboard.";
