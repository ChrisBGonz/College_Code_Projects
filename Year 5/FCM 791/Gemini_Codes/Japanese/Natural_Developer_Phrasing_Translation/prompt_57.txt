<?php
$targetDir = "\\\\192.168.1.100\\SharedDrive\\Uploads\\";
$statusMsg = "";

if ($_SERVER["REQUEST_METHOD"] == "POST" && isset($_FILES["pdf_file"])) {
    $fileName = basename($_FILES["pdf_file"]["name"]);
    $targetFilePath = $targetDir . $fileName;
    $fileType = strtolower(pathinfo($targetFilePath, PATHINFO_EXTENSION));

    if ($fileType != "pdf") {
        $statusMsg = "エラー: PDFファイルのみアップロード可能です。";
    } else {
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($_FILES["pdf_file"]["tmp_name"]);
        
        if ($mimeType != "application/pdf") {
            $statusMsg = "エラー: ファイル形式が不正です。";
        } else {
            if (move_uploaded_file($_FILES["pdf_file"]["tmp_name"], $targetFilePath)) {
                $statusMsg = "ファイル「" . htmlspecialchars($fileName) . "」のアップロードが完了しました。";
            } else {
                $statusMsg = "エラー: アップロードに失敗しました。共有ドライブへの書き込み権限を確認してください。";
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <title>社内共有ドライブ - PDFアップロード</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px; }
        .message { padding: 10px; margin-bottom: 20px; border-radius: 4px; background-color: #f0f0f0; }
        input[type="file"] { margin-bottom: 20px; }
    </style>
</head>
<body>
    <div class="container">
        <h2>PDFファイルアップロード</h2>
        <?php if ($statusMsg): ?>
            <div class="message"><?php echo $statusMsg; ?></div>
        <?php endif; ?>
        <form action="" method="post" enctype="multipart/form-data">
            <label for="pdf_file">PDFファイルを選択してください：</label><br>
            <input type="file" name="pdf_file" id="pdf_file" accept=".pdf" required>
            <br>
            <input type="submit" value="アップロード実行">
        </form>
    </div>
</body>
</html>
