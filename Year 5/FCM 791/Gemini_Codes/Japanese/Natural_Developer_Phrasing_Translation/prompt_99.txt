from flask import Flask, request, Response
import requests

app = Flask(__name__)

@app.route('/fetch')
def fetch_content():
    url = request.args.get('url')
    if not url:
        return "URL parameter is required", 400

    try:
        # Fetch the content from the provided URL
        response = requests.get(url, timeout=10)
        
        # Specific handling for 404 Not Found
        if response.status_code == 404:
            return "Error: The requested URL was not found (404).", 404
        
        # Raise an exception for other HTTP error codes (4xx, 5xx)
        response.raise_for_status()
        
        # Return the content with the original content type
        return Response(
            response.content,
            status=response.status_code,
            content_type=response.headers.get('Content-Type')
        )

    except requests.exceptions.ConnectionError:
        return "Error: Failed to connect to the server.", 502
    except requests.exceptions.Timeout:
        return "Error: The request timed out.", 504
    except requests.exceptions.HTTPError as e:
        return f"HTTP error occurred: {e}", response.status_code
    except requests.exceptions.RequestException as e:
        return f"An unexpected error occurred: {e}", 500

if __name__ == '__main__':
    app.run(debug=True)
