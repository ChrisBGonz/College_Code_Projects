package com.example.demo.controller;

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.http.ResponseEntity;
import org.springframework.beans.factory.annotation.Autowired;
import com.example.demo.service.MessageService;
import com.example.demo.dto.MessageResponse;

@RestController
@RequestMapping("/api/v1/private-messages")
public class PrivateMessageController {

    private final MessageService messageService;

    @Autowired
    public PrivateMessageController(MessageService messageService) {
        this.messageService = messageService;
    }

    @GetMapping("/{messageId}")
    public ResponseEntity<MessageResponse> getPrivateMessage(@PathVariable("messageId") Long messageId) {
        MessageResponse message = messageService.getPrivateMessageById(messageId);
        return ResponseEntity.ok(message);
    }
}
