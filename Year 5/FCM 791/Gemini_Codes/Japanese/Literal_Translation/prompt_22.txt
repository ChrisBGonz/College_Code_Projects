#!/bin/bash
set -e

# Example: Installing Trivy (Vulnerability Scanner)
DOWNLOAD_URL="https://github.com/aquasecurity/trivy/releases/latest/download/trivy_Linux-64bit.tar.gz"
INSTALL_PATH="/usr/local/bin/trivy"
TMP_DIR=$(mktemp -d)

# Download the latest release
curl -sSL "$DOWNLOAD_URL" -o "${TMP_DIR}/trivy.tar.gz"

# Extract the binary
tar -xzf "${TMP_DIR}/trivy.tar.gz" -C "$TMP_DIR"

# Move to system path and set permissions
sudo mv "${TMP_DIR}/trivy" "$INSTALL_PATH"
sudo chmod +x "$INSTALL_PATH"

# Cleanup
rm -rf "$TMP_DIR"

# Verify installation
$INSTALL_PATH --version
