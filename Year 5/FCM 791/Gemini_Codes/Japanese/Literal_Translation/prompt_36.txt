import hashlib
import sys

def get_sha256(filename):
    h = hashlib.sha256()
    try:
        with open(filename, 'rb') as f:
            while True:
                data = f.read(65536)
                if not data:
                    break
                h.update(data)
        return h.hexdigest()
    except FileNotFoundError:
        return None

def main():
    if len(sys.argv) != 3:
        print(f"Usage: python {sys.argv[0]} <file_path> <expected_sha256>")
        sys.exit(1)

    file_path = sys.argv[1]
    expected_hash = sys.argv[2].strip().lower()
    
    actual_hash = get_sha256(file_path)
    
    if actual_hash is None:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
        
    if actual_hash == expected_hash:
        print("Verification Successful: SHA256 checksum matches.")
        print(f"Hash: {actual_hash}")
        sys.exit(0)
    else:
        print("Verification Failed: SHA256 checksum does not match.")
        print(f"Expected: {expected_hash}")
        print(f"Actual:   {actual_hash}")
        sys.exit(1)

if __name__ == "__main__":
    main()
