#include <sys/socket.h>
#include <endian.h>
#include <cstdint>
#include <new>

char* allocate_buffer_from_socket(int sockfd, std::uint64_t& size) {
    std::uint64_t size_net;
    if (recv(sockfd, &size_net, sizeof(size_net), MSG_WAITALL) != sizeof(size_net)) {
        return nullptr;
    }
    size = be64toh(size_net);
    if (size == 0) {
        return nullptr;
    }
    return new(std::nothrow) char[size];
}
