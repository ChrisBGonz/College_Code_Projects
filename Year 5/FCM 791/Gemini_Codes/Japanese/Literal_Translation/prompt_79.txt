import hashlib
import os
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None
    except Exception as e:
        return str(e)

def verify_manifest(manifest_path):
    if not os.path.exists(manifest_path):
        print(f"Error: Manifest file '{manifest_path}' not found.")
        sys.exit(1)

    all_match = True
    with open(manifest_path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith("#"):
                continue

            parts = line.split(None, 1)
            if len(parts) < 2:
                print(f"[SKIPPED] Invalid line format: {line}")
                continue

            expected_hash = parts[0].lower()
            file_path = parts[1]

            if not os.path.exists(file_path):
                print(f"[MISSING] {file_path}")
                all_match = False
                continue

            actual_hash = calculate_sha256(file_path)

            if actual_hash == expected_hash:
                print(f"[OK]      {file_path}")
            else:
                print(f"[FAILED]  {file_path}")
                print(f"          Expected: {expected_hash}")
                print(f"          Actual:   {actual_hash}")
                all_match = False

    if all_match:
        print("\nVerification successful: All files match.")
    else:
        print("\nVerification failed: Some files are missing or corrupted.")
        sys.exit(1)

if __name__ == "__main__":
    target_manifest = sys.argv[1] if len(sys.argv) > 1 else "manifest.sha256"
    verify_manifest(target_manifest)
