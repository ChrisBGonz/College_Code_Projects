<?php
session_start();

$admin_email = "admin@example.com";
$slack_webhook_url = "https://hooks.slack.com/services/YOUR/WEBHOOK/URL";

function is_authorized() {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

if (!is_authorized()) {
    $ip = $_SERVER['REMOTE_ADDR'];
    $uri = $_SERVER['REQUEST_URI'];
    $ua = $_SERVER['HTTP_USER_AGENT'];
    $time = date('Y-m-d H:i:s');

    $message = "Unauthorized Access Attempt\n";
    $message .= "Time: $time\n";
    $message .= "IP Address: $ip\n";
    $message .= "Requested URI: $uri\n";
    $message .= "User Agent: $ua\n";

    mail($admin_email, "Security Alert: Unauthorized Admin Access", $message);

    $payload = json_encode([
        "text" => "🚨 *Security Alert: Unauthorized Access Attempt* 🚨",
        "attachments" => [
            [
                "color" => "danger",
                "fields" => [
                    ["title" => "Time", "value" => $time, "short" => true],
                    ["title" => "IP", "value" => $ip, "short" => true],
                    ["title" => "Path", "value" => $uri, "short" => false],
                    ["title" => "User Agent", "value" => $ua, "short" => false]
                ]
            ]
        ]
    ]);

    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    echo "Access Denied.";
    exit;
}

echo "Welcome to the Admin Dashboard.";
?>
