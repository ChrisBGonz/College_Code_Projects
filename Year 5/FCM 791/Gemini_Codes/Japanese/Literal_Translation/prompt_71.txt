import pickle

class GameState:
    def __init__(self, player_name, level, inventory, stats, position):
        self.player_name = player_name
        self.level = level
        self.inventory = inventory
        self.stats = stats
        self.position = position

    def __repr__(self):
        return f"GameState(name={self.player_name}, level={self.level}, inventory={self.inventory}, stats={self.stats}, pos={self.position})"

current_state = GameState(
    player_name="Adventurer",
    level=25,
    inventory=["Excalibur", "Health Potion", "Magic Map"],
    stats={"hp": 150, "mp": 80, "strength": 45},
    position=(102.5, 45.0)
)

file_path = "save_data.pkl"

with open(file_path, "wb") as f:
    pickle.dump(current_state, f)

with open(file_path, "rb") as f:
    loaded_state = pickle.load(f)

print(loaded_state)
