<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'dbname' => 'auth_system',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['dbname']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Connection failed: " . $e->getMessage());
}

$message = "";
$step = isset($_SESSION['reset_user_id']) ? 2 : 1;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['action']) && $_POST['action'] === 'verify_answer') {
        $username = $_POST['username'] ?? '';
        $answer = $_POST['answer'] ?? '';

        $stmt = $pdo->prepare("SELECT id, secret_answer FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify(mb_strtolower(trim($answer)), $user['secret_answer'])) {
            $_SESSION['reset_user_id'] = $user['id'];
            $step = 2;
        } else {
            $message = "ユーザー名または回答が正しくありません。";
        }
    } elseif (isset($_POST['action']) && $_POST['action'] === 'reset_password' && isset($_SESSION['reset_user_id'])) {
        $new_password = $_POST['new_password'] ?? '';
        $confirm_password = $_POST['confirm_password'] ?? '';

        if ($new_password === $confirm_password && strlen($new_password) >= 8) {
            $hashed_password = password_hash($new_password, PASSWORD_DEFAULT);
            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$hashed_password, $_SESSION['reset_user_id']]);
            
            session_destroy();
            $message = "パスワードが正常に更新されました。";
            $step = 3;
        } else {
            $message = "パスワードが一致しないか、短すぎます。";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <title>パスワード再設定</title>
    <style>
        body { font-family: sans-serif; max-width: 400px; margin: 50px auto; }
        .form-group { margin-bottom: 15px; }
        .error { color: red; }
        .success { color: green; }
    </style>
</head>
<body>
    <h1>パスワード再設定</h1>
    
    <?php if ($message): ?>
        <p class="<?php echo $step === 3 ? 'success' : 'error'; ?>"><?php echo htmlspecialchars($message); ?></p>
    <?php endif; ?>

    <?php if ($step === 1): ?>
        <form method="POST">
            <input type="hidden" name="action" value="verify_answer">
            <div class="form-group">
                <label>ユーザー名:</label><br>
                <input type="text" name="username" required>
            </div>
            <div class="form-group">
                <label>秘密の質問: あなたの母親の旧姓は何ですか？</label><br>
                <input type="text" name="answer" required>
            </div>
            <button type="submit">次へ</button>
        </form>
    <?php elseif ($step === 2): ?>
        <form method="POST">
            <input type="hidden" name="action" value="reset_password">
            <div class="form-group">
                <label>新しいパスワード:</label><br>
                <input type="password" name="new_password" required minlength="8">
            </div>
            <div class="form-group">
                <label>パスワード（確認）:</label><br>
                <input type="password" name="confirm_password" required minlength="8">
            </div>
            <button type="submit">パスワードを更新</button>
        </form>
    <?php elseif ($step === 3): ?>
        <p><a href="login.php">ログイン画面へ戻る</a></p>
    <?php endif; ?>
</body>
</html>
