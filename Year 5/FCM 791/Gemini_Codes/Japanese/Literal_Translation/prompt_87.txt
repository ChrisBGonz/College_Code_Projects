import time
from collections import deque

LOG_FILE_PATH = "access.log"
THRESHOLD = 10
WINDOW_SECONDS = 60

def monitor_log():
    failed_login_times = deque()
    
    try:
        with open(LOG_FILE_PATH, "r") as f:
            f.seek(0, 2)
            
            while True:
                line = f.readline()
                if not line:
                    time.sleep(0.1)
                    continue
                
                if "Failed Login" in line:
                    current_time = time.time()
                    failed_login_times.append(current_time)
                    
                    while failed_login_times and failed_login_times[0] < current_time - WINDOW_SECONDS:
                        failed_login_times.popleft()
                    
                    if len(failed_login_times) > THRESHOLD:
                        print(f"ALERT: {len(failed_login_times)} failed logins detected in the last {WINDOW_SECONDS} seconds.")
    except FileNotFoundError:
        print(f"Error: {LOG_FILE_PATH} not found.")
    except KeyboardInterrupt:
        pass

if __name__ == "__main__":
    monitor_log()
