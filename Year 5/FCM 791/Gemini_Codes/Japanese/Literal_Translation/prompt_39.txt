import socket

def xor_cipher(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def communicate_with_legacy_server(host, port, key, message):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            # Encrypt and send
            encrypted_data = xor_cipher(message.encode('utf-8'), key)
            s.sendall(encrypted_data)
            
            # Receive and decrypt
            response = s.recv(4096)
            if response:
                decrypted_response = xor_cipher(response, key)
                return decrypted_response.decode('utf-8', errors='ignore')
    except Exception as e:
        return f"Connection Error: {e}"
    return None

if __name__ == "__main__":
    # Configuration for the legacy server
    SERVER_HOST = '127.0.0.1'
    SERVER_PORT = 8888
    XOR_KEY = b'legacy_secret_key_123'
    
    # Communication execution
    request_payload = "COMMAND_GET_RECORDS"
    result = communicate_with_legacy_server(SERVER_HOST, SERVER_PORT, XOR_KEY, request_payload)
    
    if result:
        print(f"Decrypted Server Response: {result}")
    else:
        print("No response received.")
