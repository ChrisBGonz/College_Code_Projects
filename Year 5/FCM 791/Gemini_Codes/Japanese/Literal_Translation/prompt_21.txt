import jwt
import datetime
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.backends import default_backend

# Generate RSA Key Pair for RS256
private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048,
    backend=default_backend()
)
public_key = private_key.public_key()

private_pem = private_key.private_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PrivateFormat.PKCS8,
    encryption_algorithm=serialization.NoEncryption()
)
public_pem = public_key.public_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PublicFormat.SubjectPublicKeyInfo
)

def create_complex_token(user_id, scopes):
    now = datetime.datetime.utcnow()
    payload = {
        "iss": "https://auth.example.com",
        "sub": user_id,
        "aud": "https://api.example.com",
        "iat": now,
        "nbf": now,
        "exp": now + datetime.timedelta(hours=1),
        "jti": "unique-nonce-12345",
        "scopes": scopes,
        "context": {
            "user_role": "administrator",
            "tenant_id": "tenant-99"
        }
    }
    token = jwt.encode(payload, private_pem, algorithm="RS256")
    return token

def decode_and_verify_token(token):
    try:
        decoded_payload = jwt.decode(
            token,
            public_pem,
            algorithms=["RS256"],
            audience="https://api.example.com",
            issuer="https://auth.example.com",
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iss", "sub", "aud"]
            }
        )
        return {"status": "success", "data": decoded_payload}
    except jwt.ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except jwt.InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except jwt.InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except jwt.ImmatureSignatureError:
        return {"status": "error", "message": "Token not yet valid (nbf)"}
    except jwt.InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    encoded_token = create_complex_token("user_001", ["read:users", "write:users"])
    print(f"Encoded JWT:\n{encoded_token}\n")

    verification_result = decode_and_verify_token(encoded_token)
    print(f"Verification Result:\n{verification_result}")

    # Example of failed verification (tampered token)
    tampered_token = encoded_token + "invalid"
    print(f"\nTampered Verification:\n{decode_and_verify_token(tampered_token)}")
