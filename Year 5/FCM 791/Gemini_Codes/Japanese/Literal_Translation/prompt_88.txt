using System;
using Microsoft.Extensions.Logging;

public enum MfaChangeType
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    RecoveryCodesRegenerated
}

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, MfaChangeType changeType, string ipAddress, string userAgent)
    {
        var timestamp = DateTimeOffset.UtcNow;

        _logger.LogInformation(
            "Security Audit: MFA configuration changed. " +
            "UserID: {UserId}, " +
            "Action: {Action}, " +
            "Timestamp: {Timestamp}, " +
            "IPAddress: {IPAddress}, " +
            "UserAgent: {UserAgent}",
            userId,
            changeType,
            timestamp,
            ipAddress,
            userAgent
        );
    }
}
