const express = require('express');
const app = express();

app.use(express.json());
app.set('trust proxy', true);

const polls = {
    "1": {
        question: "好きなプログラミング言語は？",
        options: { "JavaScript": 0, "Python": 0, "Go": 0 },
        votedIPs: new Set()
    }
};

app.get('/polls', (req, res) => {
    const summary = Object.keys(polls).map(id => ({
        id,
        question: polls[id].question,
        options: polls[id].options
    }));
    res.json(summary);
});

app.post('/polls/:id/vote', (req, res) => {
    const pollId = req.params.id;
    const { option } = req.body;
    const userIP = req.ip;

    const poll = polls[pollId];

    if (!poll) {
        return res.status(404).json({ error: "Poll not found" });
    }

    if (poll.votedIPs.has(userIP)) {
        return res.status(403).json({ error: "Already voted from this IP" });
    }

    if (!(option in poll.options)) {
        return res.status(400).json({ error: "Invalid option" });
    }

    poll.options[option]++;
    poll.votedIPs.add(userIP);

    res.json({ message: "Vote recorded", results: poll.options });
});

const PORT = 3000;
app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
});
