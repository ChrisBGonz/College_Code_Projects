<?php

class UserSettings {
    public $theme;
    public $language;
    public $itemsPerPage;

    public function __construct($theme = 'light', $language = 'ja', $itemsPerPage = 20) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
    }
}

$cookieName = 'user_profile_settings';

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $settings = unserialize($serializedData);
} else {
    $settings = new UserSettings();
    $serializedData = serialize($settings);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

if ($settings instanceof UserSettings) {
    $currentTheme = $settings->theme;
    $currentLanguage = $settings->language;
}
