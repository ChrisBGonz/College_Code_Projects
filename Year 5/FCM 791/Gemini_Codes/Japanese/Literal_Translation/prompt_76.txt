import subprocess
import os
import sys

def execute(command):
    try:
        subprocess.run(command, shell=True, check=True)
    except subprocess.CalledProcessError as e:
        sys.stderr.write(f"Error executing command: {e}\n")
        sys.exit(1)

def main():
    image_repo = os.getenv("DOCKER_REPO", "myapp")
    image_tag = os.getenv("IMAGE_TAG", "latest")
    full_image = f"{image_repo}:{image_tag}"
    container_name = os.getenv("CONTAINER_NAME", "production_app")
    host_port = os.getenv("HOST_PORT", "80")
    container_port = os.getenv("CONTAINER_PORT", "80")
    
    registry_user = os.getenv("DOCKER_USER")
    registry_pass = os.getenv("DOCKER_PASS")

    if registry_user and registry_pass:
        execute(f"echo {registry_pass} | docker login -u {registry_user} --password-stdin")

    print(f"Pulling image: {full_image}")
    execute(f"docker pull {full_image}")

    print(f"Stopping and removing existing container: {container_name}")
    subprocess.run(f"docker stop {container_name}", shell=True, capture_output=True)
    subprocess.run(f"docker rm {container_name}", shell=True, capture_output=True)

    print(f"Deploying new container: {container_name}")
    execute(
        f"docker run -d "
        f"--name {container_name} "
        f"-p {host_port}:{container_port} "
        f"--restart always "
        f"{full_image}"
    )

    print("Deployment completed successfully.")

if __name__ == "__main__":
    main()
