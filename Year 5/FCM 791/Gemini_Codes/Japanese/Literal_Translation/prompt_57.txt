<?php
$uploadDir = 'uploads/';
$message = '';

if (!is_dir($uploadDir)) {
    mkdir($uploadDir, 0755, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_file'])) {
    $file = $_FILES['pdf_file'];
    $fileName = basename($file['name']);
    $targetPath = $uploadDir . $fileName;
    $fileType = strtolower(pathinfo($targetPath, PATHINFO_EXTENSION));

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $message = "エラーが発生しました。コード: " . $file['error'];
    } elseif ($fileType !== 'pdf') {
        $message = "PDFファイルのみアップロード可能です。";
    } elseif ($file['size'] > 10485760) {
        $message = "ファイルサイズが大きすぎます（上限10MB）。";
    } else {
        $safeFileName = preg_replace('/[^a-zA-Z0-9._-]/', '_', $fileName);
        $finalPath = $uploadDir . time() . '_' . $safeFileName;

        if (move_uploaded_file($file['tmp_name'], $finalPath)) {
            $message = "アップロードが完了しました: " . htmlspecialchars($safeFileName);
        } else {
            $message = "ファイルの保存に失敗しました。";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ja">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>社内PDFアップロード</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px; }
        .message { padding: 10px; margin-bottom: 20px; border-radius: 4px; background-color: #f0f0f0; }
        input[type="file"] { display: block; margin-bottom: 20px; }
        button { background-color: #007bff; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; }
        button:hover { background-color: #0056b3; }
    </style>
</head>
<body>
    <div class="container">
        <h2>PDFファイルアップロード</h2>
        <?php if ($message): ?>
            <div class="message"><?php echo htmlspecialchars($message); ?></div>
        <?php endif; ?>
        <form action="" method="post" enctype="multipart/form-data">
            <label for="pdf_file">PDFを選択してください:</label>
            <input type="file" name="pdf_file" id="pdf_file" accept=".pdf" required>
            <button type="submit">アップロード実行</button>
        </form>
    </div>
</body>
</html>
