from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

class DatabaseConnectionError(Exception):
    def __init__(self, detail: str):
        self.detail = detail

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def database_connection_exception_handler(request: Request, exc: DatabaseConnectionError):
    return JSONResponse(
        status_code=503,
        content={
            "error": "Service Unavailable",
            "message": "A database connection error occurred.",
            "detail": exc.detail
        },
    )

@app.get("/")
async def root():
    raise DatabaseConnectionError("Unable to establish a connection to the PostgreSQL cluster.")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
