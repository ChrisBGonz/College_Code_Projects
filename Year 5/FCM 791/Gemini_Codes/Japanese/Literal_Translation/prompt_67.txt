import hmac
import hashlib
import time
import base64
import struct

def generate_totp(secret, intervals_no):
    secret = secret.upper()
    missing_padding = len(secret) % 8
    if missing_padding != 0:
        secret += '=' * (8 - missing_padding)
    
    key = base64.b32decode(secret)
    msg = struct.pack(">Q", intervals_no)
    hmac_hash = hmac.new(key, msg, hashlib.sha1).digest()
    
    offset = hmac_hash[-1] & 0x0F
    code = (struct.unpack(">I", hmac_hash[offset:offset+4])[0] & 0x7FFFFFFF) % 1000000
    return "{:06d}".format(code)

def verify_totp(secret, user_code, window=1):
    """
    Verifies a TOTP code.
    :param secret: Base32 encoded secret string
    :param user_code: 6-digit string provided by the user
    :param window: Number of 30-second blocks to check before/after current time
    :return: Boolean
    """
    try:
        current_interval = int(time.time() / 30)
        for i in range(-window, window + 1):
            if generate_totp(secret, current_interval + i) == user_code:
                return True
        return False
    except Exception:
        return False
