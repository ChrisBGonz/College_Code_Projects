const { exec } = require('child_process');

function checkHostStatus(domain_name) {
    return new Promise((resolve) => {
        const platform = process.platform;
        const countFlag = platform === 'win32' ? '-n' : '-c';
        const command = `ping ${countFlag} 1 ${domain_name}`;

        exec(command, (error) => {
            if (error) {
                resolve(false);
            } else {
                resolve(true);
            }
        });
    });
}

module.exports = checkHostStatus;
