import os
import yaml
from flask import Flask, request, render_template_string, redirect

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>User Profile Setup</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        pre { background: #f4f4f4; padding: 10px; border: 1px solid #ddd; }
        .error { color: red; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Upload Profile YAML</h1>
        <form method="post" enctype="multipart/form-data" action="/upload">
            <p>Select a YAML file (e.g., name, email, bio):</p>
            <input type="file" name="file" accept=".yaml,.yml" required>
            <br><br>
            <input type="submit" value="Upload and Set Profile">
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <hr>
            <h2>Profile Loaded Successfully</h2>
            <pre>{{ profile }}</pre>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return redirect('/')
    
    file = request.files['file']
    
    if file.filename == '':
        return redirect('/')

    if file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        try:
            file_content = file.read().decode('utf-8')
            data = yaml.safe_load(file_content)
            
            if not isinstance(data, dict):
                return render_template_string(HTML_TEMPLATE, error="Invalid YAML format: Root must be a dictionary.")
            
            formatted_profile = yaml.dump(data, default_flow_style=False, allow_unicode=True)
            return render_template_string(HTML_TEMPLATE, profile=formatted_profile)
            
        except yaml.YAMLError as e:
            return render_template_string(HTML_TEMPLATE, error=f"Error parsing YAML: {str(e)}")
        except Exception as e:
            return render_template_string(HTML_TEMPLATE, error=f"An error occurred: {str(e)}")
    
    return render_template_string(HTML_TEMPLATE, error="Please upload a valid .yaml or .yml file.")

if __name__ == '__main__':
    app.run(debug=True)
