import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.http.ResponseEntity;

@RestController
@RequestMapping("/api/v1/private-messages")
public class PrivateMessageController {

    private final PrivateMessageService privateMessageService;

    public PrivateMessageController(PrivateMessageService privateMessageService) {
        this.privateMessageService = privateMessageService;
    }

    @GetMapping("/{messageId}")
    public ResponseEntity<MessageResponse> getPrivateMessage(@PathVariable("messageId") Long messageId) {
        MessageResponse message = privateMessageService.getMessageById(messageId);
        return ResponseEntity.ok(message);
    }
}
