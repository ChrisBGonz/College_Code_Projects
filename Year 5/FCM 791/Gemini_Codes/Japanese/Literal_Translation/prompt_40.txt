const express = require('express');
const cookieEncryption = require('cookie-encryption');

const app = express();
const secret = 'your-32-character-secret-key-here';
const vault = cookieEncryption(secret);

app.use(vault.init());

app.get('/', (req, res) => {
  res.cookie('secure_cookie', 'sensitive_data');
  res.send('Cookie has been encrypted and set.');
});

app.listen(3000);
