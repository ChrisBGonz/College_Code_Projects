import logging
from logging.handlers import TimedRotatingFileHandler

def setup_logging(log_filename="application.log"):
    logger = logging.getLogger("RotationLogger")
    logger.setLevel(logging.INFO)

    # when="midnight" ensures rotation happens at the end of the day
    # interval=1 means every 1 day (24 hours)
    # backupCount=30 keeps 30 days of history to save disk space
    handler = TimedRotatingFileHandler(
        log_filename,
        when="midnight",
        interval=1,
        backupCount=30,
        encoding="utf-8"
    )

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    
    logger.addHandler(handler)
    return logger

if __name__ == "__main__":
    logger = setup_logging()
    logger.info("Log rotation initialized. This file will rotate every 24 hours.")
