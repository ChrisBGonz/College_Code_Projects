const STORAGE_KEY = 'registration_form_draft';

function initFormPersistence(formId) {
  const form = document.getElementById(formId);
  if (!form) return;

  const saveToStorage = () => {
    const formData = new FormData(form);
    const data = {};
    
    formData.forEach((value, key) => {
      data[key] = value;
    });

    form.querySelectorAll('input[type="checkbox"]').forEach(checkbox => {
      if (checkbox.name) {
        data[checkbox.name] = checkbox.checked;
      }
    });

    localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  };

  const loadFromStorage = () => {
    const savedData = localStorage.getItem(STORAGE_KEY);
    if (!savedData) return;

    const data = JSON.parse(savedData);
    Object.keys(data).forEach(key => {
      const element = form.elements[key];
      if (!element) return;

      if (element.type === 'checkbox') {
        element.checked = data[key];
      } else if (element instanceof RadioNodeList) {
        Array.from(element).forEach(radio => {
          radio.checked = (radio.value === data[key]);
        });
      } else {
        element.value = data[key];
      }
    });
  };

  const clearStorage = () => {
    localStorage.removeItem(STORAGE_KEY);
  };

  form.addEventListener('input', saveToStorage);

  form.addEventListener('submit', (event) => {
    const isFinalStep = !event.submitter || event.submitter.getAttribute('data-final-submit') === 'true' || event.submitter.type === 'submit';
    if (isFinalStep) {
      clearStorage();
    }
  });

  loadFromStorage();
}

initFormPersistence('registration-form');
