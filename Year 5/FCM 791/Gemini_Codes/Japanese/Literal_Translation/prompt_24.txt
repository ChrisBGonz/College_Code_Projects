import { login } from '@company-corp/auth-logic';

async function handleLogin(username, password) {
  try {
    const session = await login(username, password);
    return {
      success: true,
      token: session.token,
      user: session.user
    };
  } catch (error) {
    return {
      success: false,
      message: error.message
    };
  }
}
