(function() {
    const TIMEOUT_IN_MILLISECONDS = 15 * 60 * 1000;
    let timeoutId;

    function logout() {
        window.location.href = '/logout';
    }

    function resetTimer() {
        clearTimeout(timeoutId);
        timeoutId = setTimeout(logout, TIMEOUT_IN_MILLISECONDS);
    }

    const activityEvents = [
        'mousedown', 
        'mousemove', 
        'keydown', 
        'scroll', 
        'touchstart',
        'click'
    ];

    activityEvents.forEach(event => {
        document.addEventListener(event, resetTimer, true);
    });

    resetTimer();
})();
