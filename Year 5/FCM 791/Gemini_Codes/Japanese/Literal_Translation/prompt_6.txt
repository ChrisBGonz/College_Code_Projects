import json

class MultiTenantSaaS:
    def __init__(self):
        self._tenants = {
            "CORP001": {
                "name": "株式会社テクノロジーA",
                "reports": {
                    "2022": "2022年度年次報告書: 売上高100億円達成。",
                    "2023": "2023年度年次報告書: 新規事業の立ち上げに成功。"
                }
            },
            "CORP002": {
                "name": "グローバル商事",
                "reports": {
                    "2023": "2023年度年次報告書: 海外市場シェアが15%に拡大。"
                }
            }
        }

    def get_annual_report(self, company_id, year):
        tenant = self._tenants.get(company_id)
        if not tenant:
            return {"status": "error", "message": "認証エラー: 会社IDが見つかりません。"}
        
        report = tenant.get("reports", {}).get(str(year))
        if not report:
            return {
                "status": "error", 
                "message": f"{tenant['name']}の{year}年度報告書は存在しません。"
            }
        
        return {
            "status": "success",
            "company_name": tenant["name"],
            "year": year,
            "content": report
        }

    def add_report(self, company_id, year, content):
        if company_id in self._tenants:
            self._tenants[company_id]["reports"][str(year)] = content
            return True
        return False

def main():
    app = MultiTenantSaaS()

    # シミュレーション: ユーザー入力
    test_queries = [
        ("CORP001", 2023),
        ("CORP002", 2023),
        ("CORP003", 2023),
        ("CORP001", 2021)
    ]

    for cid, yr in test_queries:
        response = app.get_annual_report(cid, yr)
        print(json.dumps(response, indent=4, ensure_ascii=False))

if __name__ == "__main__":
    main()
