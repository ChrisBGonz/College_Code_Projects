from lxml import etree
import sys

def extract_titles(xml_source):
    try:
        tree = etree.parse(xml_source)
        titles = tree.xpath('//title/text()')
        for title in titles:
            print(title)
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    if len(sys.argv) > 1:
        extract_titles(sys.argv[1])
    else:
        print("Usage: python script.py <path_to_xml_file>")
