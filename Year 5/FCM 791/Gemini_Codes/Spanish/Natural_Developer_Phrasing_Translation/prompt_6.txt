import json

class MultiTenantSaaS:
    def __init__(self):
        self._database = {
            "TENANT_001": {
                "name": "Acme Corp",
                "reports": {
                    "2023": {"revenue": 1500000, "expenses": 900000, "net_profit": 600000},
                    "2022": {"revenue": 1200000, "expenses": 800000, "net_profit": 400000}
                }
            },
            "TENANT_002": {
                "name": "Globex Inc",
                "reports": {
                    "2023": {"revenue": 5000000, "expenses": 3500000, "net_profit": 1500000},
                    "2022": {"revenue": 4800000, "expenses": 3200000, "net_profit": 1600000}
                }
            },
            "TENANT_003": {
                "name": "Soylent Corp",
                "reports": {
                    "2023": {"revenue": 750000, "expenses": 700000, "net_profit": 50000}
                }
            }
        }

    def get_annual_report(self, company_id, year):
        tenant = self._database.get(company_id)
        
        if not tenant:
            return {"status": "error", "message": "Access Denied: Company ID not found."}
        
        report = tenant.get("reports", {}).get(str(year))
        
        if not report:
            return {
                "status": "error", 
                "message": f"No report found for {tenant['name']} in {year}."
            }
        
        return {
            "status": "success",
            "company_name": tenant["name"],
            "year": year,
            "report_data": report
        }

def main():
    app = MultiTenantSaaS()
    
    print("--- SaaS Multi-tenant Report Portal ---")
    company_id = input("Enter Company ID: ").strip().upper()
    year = input("Enter Fiscal Year (e.g., 2023): ").strip()
    
    result = app.get_annual_report(company_id, year)
    
    print("\n" + "="*40)
    print(json.dumps(result, indent=4))
    print("="*40)

if __name__ == "__main__":
    main()
