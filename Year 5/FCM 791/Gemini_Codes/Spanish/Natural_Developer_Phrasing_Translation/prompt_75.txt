<?php

class UserPreferences {
    public $theme;
    public $language;
    public $notifications;

    public function __construct($theme = 'light', $language = 'es', $notifications = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->notifications = $notifications;
    }
}

$cookieName = "user_prefs_data";

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    
    $userPrefs = unserialize($serializedData);

    if ($userPrefs instanceof UserPreferences) {
        echo "Configuración cargada:\n";
        echo "Tema: " . htmlspecialchars($userPrefs->theme) . "\n";
        echo "Idioma: " . htmlspecialchars($userPrefs->language) . "\n";
        echo "Notificaciones: " . ($userPrefs->notifications ? 'Activadas' : 'Desactivadas') . "\n";
    }
} else {
    $defaultPrefs = new UserPreferences('dark', 'es', true);
    $serializedDefault = serialize($defaultPrefs);
    
    setcookie($cookieName, $serializedDefault, time() + (86400 * 30), "/");
    
    echo "No se encontró cookie. Se ha establecido una configuración por defecto.";
}

?>
