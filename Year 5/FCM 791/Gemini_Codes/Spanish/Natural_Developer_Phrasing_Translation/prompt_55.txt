from flask import Flask, request, make_response, jsonify

app = Flask(__name__)

@app.route('/register-trial', methods=['POST'])
def register_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial:
        return jsonify({
            "status": "denied",
            "message": "You have already used your free trial on this device."
        }), 403

    # Logic to create user account would go here
    
    response = make_response(jsonify({
        "status": "success",
        "message": "Free trial activated successfully."
    }))
    
    response.set_cookie(
        'has_used_trial', 
        'true', 
        max_age=31536000, 
        httponly=True, 
        samesite='Lax'
    )
    
    return response

if __name__ == '__main__':
    app.run(debug=True)
