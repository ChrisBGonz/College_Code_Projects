import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

SECRET_KEY = "super-secret-key-for-jwt-signing"
ALGORITHM = "HS256"

def generate_complex_token(user_id, permissions, issuer="auth-system", audience="web-app"):
    now = datetime.datetime.utcnow()
    payload = {
        "sub": user_id,
        "iat": now,
        "nbf": now,
        "exp": now + datetime.timedelta(minutes=30),
        "iss": issuer,
        "aud": audience,
        "roles": permissions,
        "context": {
            "org_id": "org_98765",
            "tier": "premium"
        }
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def decode_and_validate_token(token, expected_issuer="auth-system", expected_audience="web-app"):
    try:
        decoded_payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM],
            audience=expected_audience,
            issuer=expected_issuer,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iat", "iss", "aud", "sub"]
            }
        )
        return {"status": "success", "data": decoded_payload}
    except ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    token = generate_complex_token("user_001", ["read", "write", "delete"])
    print(f"Encoded Token: {token}")

    result = decode_and_validate_token(token)
    if result["status"] == "success":
        print("Token is valid. Payload:")
        print(result["data"])
    else:
        print(f"Validation failed: {result['message']}")
