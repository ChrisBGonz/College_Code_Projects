<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'dbname' => 'auth_system',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['dbname']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Connection failed");
}

$step = $_POST['step'] ?? 'identify';
$error = '';
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'find_user') {
        $username = $_POST['username'] ?? '';
        $stmt = $pdo->prepare("SELECT secret_question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user'] = $username;
            $_SESSION['recovery_question'] = $user['secret_question'];
            $step = 'answer_question';
        } else {
            $error = "User not found.";
            $step = 'identify';
        }
    } elseif ($step === 'verify_answer') {
        $answer = strtolower(trim($_POST['answer'] ?? ''));
        $username = $_SESSION['recovery_user'] ?? '';

        $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify($answer, $user['secret_answer'])) {
            $_SESSION['recovery_verified'] = true;
            $step = 'reset_password';
        } else {
            $error = "Incorrect answer.";
            $step = 'answer_question';
        }
    } elseif ($step === 'do_reset') {
        if (!isset($_SESSION['recovery_verified']) || $_SESSION['recovery_verified'] !== true) {
            die("Unauthorized");
        }
        
        $new_password = password_hash($_POST['password'], PASSWORD_DEFAULT);
        $username = $_SESSION['recovery_user'];

        $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE username = ?");
        $stmt->execute([$new_password, $username]);

        session_destroy();
        $success = "Password updated successfully.";
        $step = 'complete';
    }
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Recuperación de Cuenta</title>
</head>
<body>
    <h2>Recuperar Contraseña</h2>
    
    <?php if ($error): ?>
        <p style="color:red;"><?php echo htmlspecialchars($error); ?></p>
    <?php endif; ?>

    <?php if ($success): ?>
        <p style="color:green;"><?php echo htmlspecialchars($success); ?></p>
    <?php endif; ?>

    <?php if ($step === 'identify'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="find_user">
            <label>Nombre de usuario:</label><br>
            <input type="text" name="username" required><br><br>
            <button type="submit">Continuar</button>
        </form>

    <?php elseif ($step === 'answer_question'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="verify_answer">
            <p>Pregunta de seguridad: <strong><?php echo htmlspecialchars($_SESSION['recovery_question']); ?></strong></p>
            <label>Tu respuesta:</label><br>
            <input type="text" name="answer" required><br><br>
            <button type="submit">Verificar</button>
        </form>

    <?php elseif ($step === 'reset_password'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="do_reset">
            <label>Nueva contraseña:</label><br>
            <input type="password" name="password" required minlength="8"><br><br>
            <button type="submit">Cambiar contraseña</button>
        </form>

    <?php elseif ($step === 'complete'): ?>
        <p><a href="login.php">Iniciar sesión ahora</a></p>
    <?php endif; ?>
</body>
</html>
