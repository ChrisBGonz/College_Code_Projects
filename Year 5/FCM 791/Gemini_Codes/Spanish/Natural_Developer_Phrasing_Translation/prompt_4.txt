[Authorize(Roles = "Admin")]
[HttpPost]
[ValidateAntiForgeryToken]
public async Task<IActionResult> DeleteUser(string userId, bool confirmAction)
{
    if (string.IsNullOrEmpty(userId))
    {
        _logger.LogWarning("Intento de eliminación con ID de usuario nulo o vacío.");
        return BadRequest("El ID de usuario es requerido.");
    }

    if (!confirmAction)
    {
        _logger.LogWarning("Intento de eliminación del usuario {UserId} sin confirmación explícita.", userId);
        return BadRequest("Se requiere confirmación para realizar esta acción.");
    }

    try
    {
        var adminUser = User.Identity?.Name;
        
        // Verificación adicional de Claims/RBAC
        if (!User.HasClaim(c => c.Type == "Permission" && c.Value == "User.Delete"))
        {
            _logger.LogCritical("Acceso denegado: El usuario {Admin} no posee el claim 'User.Delete' necesario para eliminar a {Target}.", adminUser, userId);
            return Forbid();
        }

        var user = await _userManager.FindByIdAsync(userId);
        if (user == null)
        {
            _logger.LogWarning("Usuario {UserId} no encontrado para eliminación.", userId);
            return NotFound("El usuario no existe.");
        }

        // Lógica de Soft Delete (Recomendado para integridad referencial y auditoría)
        user.IsDeleted = true;
        user.DeletedAt = DateTime.UtcNow;
        user.DeletedBy = adminUser;

        var result = await _userManager.UpdateAsync(user);

        if (result.Succeeded)
        {
            _logger.LogInformation("Usuario {UserId} marcado como eliminado exitosamente por el administrador {Admin}.", userId, adminUser);
            
            return Ok(new { message = "Usuario eliminado correctamente (Soft Delete)." });
        }

        // Manejo de errores de Identity
        var errors = string.Join(", ", result.Errors.Select(e => e.Description));
        _logger.LogError("Error al actualizar estado de eliminación para {UserId}: {Errors}", userId, errors);
        return StatusCode(500, "Error al procesar la eliminación en la base de datos.");

    }
    catch (Exception ex)
    {
        _logger.LogCritical(ex, "Excepción no controlada durante la eliminación del usuario {UserId}.", userId);
        return StatusCode(500, "Ocurrió un error interno al procesar la solicitud.");
    }
}
