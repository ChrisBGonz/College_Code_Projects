using Microsoft.Extensions.Logging;
using System;

public class MfaAuditLogger
{
    private readonly ILogger<MfaAuditLogger> _logger;

    public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, string action, string mfaMethod)
    {
        _logger.LogInformation(
            "MFA Configuration Change: User {UserId} performed {Action} on method {MfaMethod} at {Timestamp}",
            userId,
            action,
            mfaMethod,
            DateTime.UtcNow
        );
    }
}
