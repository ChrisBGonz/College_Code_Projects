<?php

$host = 'localhost';
$db   = 'payments_db';
$user = 'db_user';
$pass = 'db_password';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES   => false,
];

try {
     $pdo = new PDO($dsn, $user, $pass, $options);
} catch (\PDOException $e) {
     throw new \PDOException($e->getMessage(), (int)$e->getCode());
}

define('ENCRYPTION_KEY', 'f9e8d7c6b5a43210f9e8d7c6b5a43210'); // 32 bytes for AES-256
define('CIPHER_METHOD', 'aes-256-cbc');

function encryptCreditCard($data, $key) {
    $iv_length = openssl_cipher_iv_length(CIPHER_METHOD);
    $iv = openssl_random_pseudo_bytes($iv_length);
    $encrypted = openssl_encrypt($data, CIPHER_METHOD, $key, 0, $iv);
    return base64_encode($iv . $encrypted);
}

$credit_card_number = '4532012345678901';
$encrypted_card = encryptCreditCard($credit_card_number, ENCRYPTION_KEY);

$sql = "INSERT INTO transactions (credit_card_number) VALUES (:cc)";
$stmt = $pdo->prepare($sql);
$stmt->execute(['cc' => $encrypted_card]);

echo "Data persisted successfully.";

?>
