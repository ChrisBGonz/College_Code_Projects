using System;
using System.Collections.Generic;
using System.Linq;

public class CheckoutItem
{
    public string Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
    public int Quantity { get; set; }
}

public class CheckoutRequest
{
    public List<CheckoutItem> Items { get; set; }
}

public class CheckoutResponse
{
    public decimal Total { get; set; }
    public bool IsValid { get; set; }
    public string Message { get; set; }
}

public class CheckoutProcessor
{
    public CheckoutResponse CalculateTotal(CheckoutRequest request)
    {
        if (request?.Items == null || !request.Items.Any())
        {
            return new CheckoutResponse 
            { 
                Total = 0, 
                IsValid = false, 
                Message = "El carrito de compras está vacío." 
            };
        }

        decimal total = 0;

        foreach (var item in request.Items)
        {
            if (item.Price < 0)
            {
                return new CheckoutResponse 
                { 
                    Total = 0, 
                    IsValid = false, 
                    Message = $"Precio inválido para el producto: {item.Name}" 
                };
            }

            if (item.Quantity <= 0)
            {
                return new CheckoutResponse 
                { 
                    Total = 0, 
                    IsValid = false, 
                    Message = $"Cantidad inválida para el producto: {item.Name}" 
                };
            }

            total += item.Price * item.Quantity;
        }

        return new CheckoutResponse
        {
            Total = total,
            IsValid = true,
            Message = "Cálculo realizado con éxito."
        };
    }
}
