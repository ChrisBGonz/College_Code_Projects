<?php
$targetDir = "\\\\CORP-SERVER\\SharedDrive\\EmployeeUploads\\";
$maxFileSize = 10 * 1024 * 1024;
$allowedMimeType = 'application/pdf';

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['document'])) {
    $file = $_FILES['document'];
    $fileName = basename($file['name']);
    $targetPath = $targetDir . preg_replace("/[^a-zA-Z0-9\._-]/", "_", $fileName);
    
    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);

    if ($file['error'] !== UPLOAD_ERR_OK) {
        die("Upload failed with error code " . $file['error']);
    }

    if ($mimeType !== $allowedMimeType) {
        die("Error: Only PDF files are allowed.");
    }

    if ($file['size'] > $maxFileSize) {
        die("Error: File size exceeds the 10MB limit.");
    }

    if (!is_dir($targetDir)) {
        mkdir($targetDir, 0777, true);
    }

    if (move_uploaded_file($file['tmp_name'], $targetPath)) {
        echo "File successfully uploaded to shared drive.";
    } else {
        echo "Error: Could not save the file to the destination.";
    }
}
?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Corporate File Upload</title>
</head>
<body>
    <h2>Upload PDF Document</h2>
    <form action="" method="POST" enctype="multipart/form-data">
        <label for="document">Select PDF:</label>
        <input type="file" name="document" id="document" accept=".pdf" required>
        <br><br>
        <input type="submit" value="Upload to Shared Drive">
    </form>
</body>
</html>
