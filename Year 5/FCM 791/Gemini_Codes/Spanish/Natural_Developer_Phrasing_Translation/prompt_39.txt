import socket

def xor_cipher(data, key):
    key_bytes = key.encode() if isinstance(key, str) else key
    return bytes([b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(data)])

def start_client(host='127.0.0.1', port=9999, secret_key="LEGACY_KEY"):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            message = "DATA_REQUEST_01"
            payload = message.encode('utf-8')
            
            encrypted_payload = xor_cipher(payload, secret_key)
            s.sendall(encrypted_payload)
            
            response = s.recv(4096)
            if response:
                decrypted_response = xor_cipher(response, secret_key)
                print(decrypted_response.decode('utf-8', errors='ignore'))
                
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    start_client()
