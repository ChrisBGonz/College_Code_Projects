import hashlib
import os
import sys

def calculate_file_hash(filepath, algorithm='sha256'):
    hash_func = hashlib.new(algorithm)
    try:
        with open(filepath, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b""):
                hash_func.update(chunk)
        return hash_func.hexdigest()
    except Exception:
        return None

def validate_integrity(manifest_path, project_dir, algorithm='sha256'):
    manifest_entries = {}
    
    if not os.path.exists(manifest_path):
        print(f"Error: Manifest file '{manifest_path}' not found.")
        return

    with open(manifest_path, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(maxsplit=1)
            if len(parts) == 2:
                hash_val, rel_path = parts
                manifest_entries[os.path.normpath(rel_path)] = hash_val

    local_files = set()
    for root, _, files in os.walk(project_dir):
        for file in files:
            abs_path = os.path.join(root, file)
            rel_path = os.path.normpath(os.path.relpath(abs_path, project_dir))
            
            if os.path.abspath(abs_path) == os.path.abspath(manifest_path):
                continue
            local_files.add(rel_path)

    mismatches = []
    missing = []
    for rel_path, expected_hash in manifest_entries.items():
        if rel_path not in local_files:
            missing.append(rel_path)
        else:
            actual_hash = calculate_file_hash(os.path.join(project_dir, rel_path), algorithm)
            if actual_hash != expected_hash:
                mismatches.append(rel_path)

    extra = [f for f in local_files if f not in manifest_entries]

    print(f"Verification Results for: {project_dir}")
    print(f"Algorithm: {algorithm}")
    print("-" * 40)
    
    if not mismatches and not missing and not extra:
        print("Integrity OK: All files match manifest.")
    else:
        if mismatches:
            print(f"\n[!] MISMATCHED HASHES ({len(mismatches)}):")
            for f in mismatches: print(f"  {f}")
        
        if missing:
            print(f"\n[-] MISSING FILES ({len(missing)}):")
            for f in missing: print(f"  {f}")
            
        if extra:
            print(f"\n[+] EXTRA FILES ({len(extra)}):")
            for f in extra: print(f"  {f}")

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python integrity_check.py <manifest_file> <project_directory> [algorithm]")
    else:
        manifest = sys.argv[1]
        directory = sys.argv[2]
        algo = sys.argv[3] if len(sys.argv) > 3 else 'sha256'
        validate_integrity(manifest, directory, algo)
