const { exec } = require('child_process');
const os = require('os');

function isHostReachable(domain_name) {
  return new Promise((resolve) => {
    const flag = os.platform() === 'win32' ? '-n' : '-c';
    const command = `ping ${flag} 1 ${domain_name}`;

    exec(command, (error) => {
      if (error) {
        resolve(false);
      } else {
        resolve(true);
      }
    });
  });
}

module.exports = isHostReachable;
