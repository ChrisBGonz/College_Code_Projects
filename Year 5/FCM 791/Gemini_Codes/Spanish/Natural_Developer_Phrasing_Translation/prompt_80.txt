import os
import yaml
from flask import Flask, request, render_template_string, redirect

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Configuración de Perfil</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        pre { background: #f4f4f4; padding: 10px; border-radius: 4px; overflow-x: auto; }
        .error { color: red; }
        .success { color: green; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Configurar Perfil via YAML</h1>
        <form method="POST" enctype="multipart/form-data">
            <label for="file">Selecciona un archivo .yaml o .yml:</label><br><br>
            <input type="file" name="file" id="file" accept=".yaml,.yml" required>
            <br><br>
            <button type="submit">Cargar Configuración</button>
        </form>

        {% if error %}
            <p class="error"><strong>Error:</strong> {{ error }}</p>
        {% endif %}

        {% if profile %}
            <h2 class="success">Perfil Inicializado:</h2>
            <pre>{{ profile | tojson(indent=4) }}</pre>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile_data = None
    error_msg = None

    if request.method == 'POST':
        if 'file' not in request.files:
            return redirect(request.url)
        
        file = request.files['file']
        
        if file.filename == '':
            error_msg = "No se seleccionó ningún archivo."
        elif file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
            try:
                content = file.read()
                profile_data = yaml.safe_load(content)
                if not isinstance(profile_data, dict):
                    error_msg = "El archivo YAML debe contener un objeto/diccionario."
                    profile_data = None
            except yaml.YAMLError as e:
                error_msg = f"Error de sintaxis en YAML: {str(e)}"
            except Exception as e:
                error_msg = f"Error inesperado: {str(e)}"
        else:
            error_msg = "Extensión de archivo no permitida. Use .yaml o .yml"

    return render_template_string(HTML_TEMPLATE, profile=profile_data, error=error_msg)

if __name__ == '__main__':
    app.run(debug=True)
