<?php
session_start();

$admin_email = 'admin@example.com';
$slack_webhook_url = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';

function is_authorized() {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

if (!is_authorized()) {
    $user_ip = $_SERVER['REMOTE_ADDR'];
    $request_uri = $_SERVER['REQUEST_URI'];
    $timestamp = date('Y-m-d H:i:s');
    $user_agent = $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown';
    $user_id = $_SESSION['user_id'] ?? 'Guest';

    $alert_message = "Unauthorized access attempt detected!\n"
                   . "User ID: $user_id\n"
                   . "IP Address: $user_ip\n"
                   . "Requested URI: $request_uri\n"
                   . "Timestamp: $timestamp\n"
                   . "User Agent: $user_agent";

    mail($admin_email, "Security Alert: Unauthorized Admin Access", $alert_message);

    $slack_payload = json_encode([
        'text' => "🚨 *Security Alert: Unauthorized Admin Access Attempt* 🚨",
        'attachments' => [
            [
                'color' => 'danger',
                'fields' => [
                    ['title' => 'User ID', 'value' => $user_id, 'short' => true],
                    ['title' => 'IP Address', 'value' => $user_ip, 'short' => true],
                    ['title' => 'URI', 'value' => $request_uri, 'short' => false],
                    ['title' => 'Time', 'value' => $timestamp, 'short' => false]
                ]
            ]
        ]
    ]);

    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slack_payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    exit("Access Denied: You do not have permission to access this resource.");
}
