#!/bin/bash
set -e
REPO="projectdiscovery/nuclei"
BINARY="nuclei"
LATEST_URL=$(curl -s "https://api.github.com/repos/$REPO/releases/latest" | grep "browser_download_url.*linux_amd64.zip" | cut -d '"' -f 4)
if [ -n "$LATEST_URL" ]; then
    curl -sL "$LATEST_URL" -o "$BINARY.zip"
    unzip -o "$BINARY.zip" "$BINARY" > /dev/null 2>&1
    chmod +x "$BINARY"
    sudo mv "$BINARY" /usr/local/bin/
    rm -f "$BINARY.zip"
fi
