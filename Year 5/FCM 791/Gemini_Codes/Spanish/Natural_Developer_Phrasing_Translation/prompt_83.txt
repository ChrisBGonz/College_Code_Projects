import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger logger = Logger.getLogger(DatabaseHandler.class.getName());

    public void executeOperation() throws CustomDatabaseException {
        try {
            // Lógica de base de datos
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Error de base de datos: " + e.getMessage());
            throw new CustomDatabaseException("Se produjo un error al procesar la solicitud de datos.", e);
        }
    }
}

class CustomDatabaseException extends Exception {
    public CustomDatabaseException(String message, Throwable cause) {
        super(message, cause);
    }
}
