from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    auth_token = request.cookies.get('auth_token')
    if auth_token == 'valid_session_token_123':
        return '<h1>Bienvenido</h1><p>Sesión iniciada correctamente.</p><a href="/logout">Cerrar sesión</a>'
    return '<h1>Inicio</h1><p>No has iniciado sesión.</p><a href="/login">Ir al Login</a>'

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')

        if username == 'admin' and password == '1234':
            response = make_response(redirect(url_for('index')))
            response.set_cookie('auth_token', 'valid_session_token_123', httponly=True)
            return response
        return 'Credenciales incorrectas. <a href="/login">Reintentar</a>'

    return '''
        <form method="post">
            Usuario: <input type="text" name="username"><br>
            Contraseña: <input type="password" name="password"><br>
            <input type="submit" value="Login">
        </form>
    '''

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('index')))
    response.set_cookie('auth_token', '', expires=0)
    return response

if __name__ == '__main__':
    app.run(debug=True)
