import subprocess
import sys

IMAGE_NAME = "your-registry/your-app:latest"
CONTAINER_NAME = "production-app"
PORTS = "80:80"

def run_command(command):
    try:
        subprocess.run(command, check=True, shell=True)
    except subprocess.CalledProcessError as e:
        print(f"Error executing command: {command}")
        sys.exit(1)

def deploy():
    print(f"Pulling latest image: {IMAGE_NAME}")
    run_command(f"docker pull {IMAGE_NAME}")

    print(f"Stopping and removing old container: {CONTAINER_NAME}")
    subprocess.run(f"docker stop {CONTAINER_NAME}", shell=True, capture_output=True)
    subprocess.run(f"docker rm {CONTAINER_NAME}", shell=True, capture_output=True)

    print(f"Starting new container: {CONTAINER_NAME}")
    run_command(f"docker run -d --name {CONTAINER_NAME} -p {PORTS} --restart unless-stopped {IMAGE_NAME}")

    print("Deployment completed successfully.")

if __name__ == "__main__":
    deploy()
