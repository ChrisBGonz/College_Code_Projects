from lxml import etree

def extract_titles_from_xml():
    file_path = input("Ingrese la ruta del archivo XML: ")
    try:
        # Parsear el archivo XML
        tree = etree.parse(file_path)
        
        # Extraer el contenido de todos los tags <title>
        # Se usa XPath para encontrar los nodos y /text() para obtener el contenido
        titles = tree.xpath("//title/text()")
        
        if titles:
            print(f"Se encontraron {len(titles)} etiquetas <title>:")
            for title in titles:
                print(title.strip())
        else:
            print("No se encontraron etiquetas <title> en el documento.")
            
    except etree.XMLSyntaxError as e:
        print(f"Error de sintaxis XML: {e}")
    except FileNotFoundError:
        print("El archivo no fue encontrado.")
    except Exception as e:
        print(f"Ocurrió un error inesperado: {e}")

if __name__ == "__main__":
    extract_titles_from_xml()
