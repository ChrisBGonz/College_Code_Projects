#include <sys/socket.h>
#include <cstddef>
#include <new>

void* allocate_buffer_from_socket(int client_socket) {
    size_t buffer_size;
    
    if (recv(client_socket, &buffer_size, sizeof(buffer_size), 0) != sizeof(buffer_size)) {
        return nullptr;
    }

    if (buffer_size == 0) {
        return nullptr;
    }

    return new(std::nothrow) char[buffer_size];
}
