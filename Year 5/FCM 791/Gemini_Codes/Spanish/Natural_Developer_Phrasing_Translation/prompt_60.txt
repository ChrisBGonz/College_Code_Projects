const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());
app.set('trust proxy', true);

const polls = {
    "1": {
        question: "What is your favorite programming language?",
        options: { "JavaScript": 0, "Python": 0, "Rust": 0 },
        votedIPs: new Set()
    }
};

app.get('/poll/:id', (req, res) => {
    const poll = polls[req.params.id];
    if (!poll) return res.status(404).send({ error: "Poll not found" });
    
    const { votedIPs, ...pollData } = poll;
    res.json(pollData);
});

app.post('/poll/:id/vote', (req, res) => {
    const pollId = req.params.id;
    const { option } = req.body;
    const poll = polls[pollId];

    if (!poll) return res.status(404).send({ error: "Poll not found" });
    if (!(option in poll.options)) return res.status(400).send({ error: "Invalid option" });

    const clientIp = req.ip || req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (poll.votedIPs.has(clientIp)) {
        return res.status(403).send({ error: "You have already voted in this poll" });
    }

    poll.options[option]++;
    poll.votedIPs.add(clientIp);

    res.json({ message: "Vote recorded", results: poll.options });
});

app.listen(PORT, () => {
    console.log(`Voting system running on http://localhost:${PORT}`);
});
