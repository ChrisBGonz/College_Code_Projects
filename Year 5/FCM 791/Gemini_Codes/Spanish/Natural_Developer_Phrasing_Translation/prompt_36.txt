import hashlib
import sys
import os

def calcular_sha256(ruta_archivo):
    sha256_hash = hashlib.sha256()
    try:
        with open(ruta_archivo, "rb") as f:
            for bloque in iter(lambda: f.read(4096), b""):
                sha256_hash.update(bloque)
        return sha256_hash.hexdigest()
    except Exception as e:
        return str(e)

def validar_firmware(ruta_archivo, hash_esperado):
    if not os.path.isfile(ruta_archivo):
        print(f"Error: El archivo '{ruta_archivo}' no fue encontrado.")
        return

    hash_calculado = calcular_sha256(ruta_archivo)
    
    print(f"Archivo: {ruta_archivo}")
    print(f"Hash esperado:   {hash_esperado.lower()}")
    print(f"Hash calculado:  {hash_calculado.lower()}")

    if hash_calculado.lower() == hash_esperado.lower():
        print("Estado: VERIFICACIÓN EXITOSA - La integridad del firmware está confirmada.")
    else:
        print("Estado: ERROR DE INTEGRIDAD - Los hashes no coinciden.")

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Uso: python script.py <ruta_al_firmware> <hash_sha256_esperado>")
    else:
        archivo = sys.argv[1]
        esperado = sys.argv[2]
        validar_firmware(archivo, esperado)
