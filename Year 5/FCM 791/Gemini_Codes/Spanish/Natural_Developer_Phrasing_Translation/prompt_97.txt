function getUserSubscriptionStatus($userId, $pdo) {
    try {
        $stmt = $pdo->prepare("SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1");
        $stmt->execute(['user_id' => $userId]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);

        return $row ? $row['status'] : 'free';
    } catch (Exception $e) {
        return 'premium';
    }
}
