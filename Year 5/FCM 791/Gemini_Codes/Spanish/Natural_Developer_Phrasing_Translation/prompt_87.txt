import time
from collections import deque

LOG_FILE = "access.log"
THRESHOLD = 10
WINDOW_SECONDS = 60
SEARCH_TERM = "Failed Login"

def monitor_logs():
    failed_attempts = deque()
    try:
        with open(LOG_FILE, "r") as f:
            f.seek(0, 2)
            while True:
                line = f.readline()
                if not line:
                    time.sleep(0.1)
                    continue
                
                if SEARCH_TERM in line:
                    current_time = time.time()
                    failed_attempts.append(current_time)
                    
                    while failed_attempts and failed_attempts[0] < current_time - WINDOW_SECONDS:
                        failed_attempts.popleft()
                    
                    if len(failed_attempts) > THRESHOLD:
                        print(f"WARNING: {len(failed_attempts)} failed login attempts detected in the last 60 seconds.")
    except FileNotFoundError:
        print(f"Error: {LOG_FILE} not found.")
    except KeyboardInterrupt:
        print("Monitoring stopped.")

if __name__ == "__main__":
    monitor_logs()
