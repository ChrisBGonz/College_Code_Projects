import pickle

class Player:
    def __init__(self, name, level, health, inventory):
        self.name = name
        self.level = level
        self.health = health
        self.inventory = inventory

class GameState:
    def __init__(self, player, world_name, coordinates, quest_log):
        self.player = player
        self.world_name = world_name
        self.coordinates = coordinates
        self.quest_log = quest_log

def save_game(game_state, filename="savegame.dat"):
    try:
        with open(filename, "wb") as file:
            pickle.dump(game_state, file)
        print(f"Game state saved successfully to {filename}")
    except Exception as e:
        print(f"Error saving game: {e}")

def load_game(filename="savegame.dat"):
    try:
        with open(filename, "rb") as file:
            game_state = pickle.load(file)
        print(f"Game state loaded successfully from {filename}")
        return game_state
    except Exception as e:
        print(f"Error loading game: {e}")
        return None

if __name__ == "__main__":
    # Create a complex GameState object
    hero = Player("Eldar", 15, 100.0, ["Master Sword", "Health Potion", "Map"])
    current_state = GameState(
        player=hero,
        world_name="Aethelgard",
        coordinates=(120.5, 45.0, 10.0),
        quest_log={"The Lost Relic": "In Progress", "Dragon's Lair": "Not Started"}
    )

    # Serialize (Save)
    save_game(current_state, "save_01.pkl")

    # Deserialize (Load)
    loaded_state = load_game("save_01.pkl")

    if loaded_state:
        print("\n--- Loaded Data ---")
        print(f"Player Name: {loaded_state.player.name}")
        print(f"Player Level: {loaded_state.player.level}")
        print(f"Inventory: {loaded_state.player.inventory}")
        print(f"World: {loaded_state.world_name}")
        print(f"Coordinates: {loaded_state.coordinates}")
        print(f"Quest Log: {loaded_state.quest_log}")
