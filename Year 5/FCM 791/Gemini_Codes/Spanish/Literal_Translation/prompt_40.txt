const cookieEncryption = require('cookie-encryption');

const vault = cookieEncryption('a-32-character-long-secret-phrase');
const vaultMiddleware = vault.init();

const forceCookieEncryption = (req, res, next) => {
  vaultMiddleware(req, res, () => {
    const originalCookie = res.cookie;
    res.cookie = function (name, value, options = {}) {
      const encryptedOptions = Object.assign({}, options, { encrypt: true });
      return originalCookie.call(this, name, value, encryptedOptions);
    };
    next();
  });
};

module.exports = forceCookieEncryption;
