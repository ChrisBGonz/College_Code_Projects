import subprocess
import sys
import os

def run_cmd(command):
    result = subprocess.run(command, shell=True, capture_output=True, text=True)
    return result

def deploy():
    image_repo = os.getenv("DOCKER_IMAGE", "my-registry/my-app:latest")
    container_name = os.getenv("CONTAINER_NAME", "production-service")
    publish_ports = os.getenv("PORT_MAPPING", "80:80")

    print(f"Step 1: Pulling image {image_repo}...")
    pull = run_cmd(f"docker pull {image_repo}")
    if pull.returncode != 0:
        print(f"Error pulling image: {pull.stderr}")
        sys.exit(1)

    print(f"Step 2: Stopping existing container {container_name}...")
    run_cmd(f"docker stop {container_name}")
    run_cmd(f"docker rm {container_name}")

    print(f"Step 3: Starting new container {container_name}...")
    deploy_cmd = f"docker run -d --name {container_name} -p {publish_ports} --restart always {image_repo}"
    start = run_cmd(deploy_cmd)
    
    if start.returncode == 0:
        print("Deployment successful.")
        print(f"Container ID: {start.stdout.strip()}")
    else:
        print(f"Deployment failed: {start.stderr}")
        sys.exit(1)

if __name__ == "__main__":
    deploy()
