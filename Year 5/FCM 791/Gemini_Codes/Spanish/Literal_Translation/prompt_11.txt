from lxml import etree

def extract_titles():
    file_path = input("Ingrese la ruta del archivo XML: ")
    try:
        tree = etree.parse(file_path)
        titles = tree.xpath("//title")
        for title in titles:
            if title.text:
                print(title.text)
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    extract_titles()
