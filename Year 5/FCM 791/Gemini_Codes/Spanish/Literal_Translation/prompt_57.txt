<?php
$upload_directory = 'uploads/';
$message = '';

if (!is_dir($upload_directory)) {
    mkdir($upload_directory, 0755, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_document'])) {
    $file = $_FILES['pdf_document'];
    $file_name = basename($file['name']);
    $target_path = $upload_directory . $file_name;
    $file_extension = strtolower(pathinfo($target_path, PATHINFO_EXTENSION));

    $finfo = finfo_open(FILEINFO_MIME_TYPE);
    $mime_type = finfo_file($finfo, $file['tmp_name']);
    finfo_close($finfo);

    if ($file_extension !== 'pdf' || $mime_type !== 'application/pdf') {
        $message = "Error: Solo se permiten archivos PDF.";
    } elseif ($file['error'] !== UPLOAD_ERR_OK) {
        $message = "Error en la subida del archivo.";
    } elseif ($file['size'] > 10485760) {
        $message = "Error: El archivo supera el límite de 10MB.";
    } else {
        $safe_filename = preg_replace("/[^a-zA-Z0-0\._-]/", "_", $file_name);
        $final_path = $upload_directory . time() . "_" . $safe_filename;

        if (move_uploaded_file($file['tmp_name'], $final_path)) {
            $message = "Archivo cargado exitosamente en la unidad compartida.";
        } else {
            $message = "Error al mover el archivo al directorio de destino.";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Carga de Archivos - Unidad Compartida</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px; }
        .alert { padding: 10px; margin-bottom: 20px; border-radius: 4px; background-color: #f4f4f4; }
        .form-group { margin-bottom: 15px; }
        label { display: block; margin-bottom: 5px; }
        input[type="file"] { display: block; margin-bottom: 10px; }
        input[type="submit"] { background: #007bff; color: white; border: none; padding: 10px 15px; cursor: pointer; border-radius: 4px; }
        input[type="submit"]:hover { background: #0056b3; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Carga de Documentos PDF</h2>
        <?php if ($message): ?>
            <div class="alert"><?php echo htmlspecialchars($message); ?></div>
        <?php endif; ?>
        <form action="" method="POST" enctype="multipart/form-data">
            <div class="form-group">
                <label for="pdf_document">Seleccione el archivo PDF:</label>
                <input type="file" name="pdf_document" id="pdf_document" accept=".pdf" required>
            </div>
            <input type="submit" value="Subir a Unidad Compartida">
        </form>
    </div>
</body>
</html>
