import pickle

class GameState:
    def __init__(self, player_name, level, inventory, coordinates, stats):
        self.player_name = player_name
        self.level = level
        self.inventory = inventory
        self.coordinates = coordinates
        self.stats = stats

    def __repr__(self):
        return f"GameState(Name: {self.player_name}, Level: {self.level}, Inventory: {self.inventory}, Pos: {self.coordinates}, Stats: {self.stats})"

def save_game(game_obj, filename="savegame.dat"):
    try:
        with open(filename, "wb") as file:
            pickle.dump(game_obj, file)
        return True
    except Exception as e:
        print(f"Error saving game: {e}")
        return False

def load_game(filename="savegame.dat"):
    try:
        with open(filename, "rb") as file:
            return pickle.load(file)
    except FileNotFoundError:
        print("Save file not found.")
        return None
    except Exception as e:
        print(f"Error loading game: {e}")
        return None

current_state = GameState(
    player_name="Aragorn",
    level=25,
    inventory=["Andúril", "Elven Cloak", "Healing Herbs"],
    coordinates=(45.5, 120.3),
    stats={"hp": 150, "mp": 40, "stamina": 100}
)

file_path = "user_save_01.pkl"

if save_game(current_state, file_path):
    print("Game state saved successfully.")

loaded_state = load_game(file_path)

if loaded_state:
    print("Game state loaded successfully:")
    print(loaded_state)
