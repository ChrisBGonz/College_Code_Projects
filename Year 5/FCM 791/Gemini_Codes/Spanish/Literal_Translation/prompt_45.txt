public class LdapFilterBuilder {
    public static String buildLastNameFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            return "(sn=*)";
        }
        
        StringBuilder filter = new StringBuilder();
        filter.append("(sn=");
        for (int i = 0; i < lastName.length(); i++) {
            char curChar = lastName.charAt(i);
            switch (curChar) {
                case '\\':
                    filter.append("\\5c");
                    break;
                case '*':
                    filter.append("\\2a");
                    break;
                case '(':
                    filter.append("\\28");
                    break;
                case ')':
                    filter.append("\\29");
                    break;
                case '\u0000':
                    filter.append("\\00");
                    break;
                default:
                    filter.append(curChar);
            }
        }
        filter.append(")");
        
        return filter.toString();
    }
}
