def promote
  @user = User.find(params[:id])
  if current_user.manager? && @user.role == 'staff'
    if @user.update(role: 'manager')
      redirect_to @user, notice: 'Staff member promoted to manager.'
    else
      redirect_to @user, alert: 'Promotion failed.'
    end
  else
    redirect_to root_path, alert: 'Unauthorized.'
  end
end
