<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'name' => 'auth_db',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['name']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Connection failed");
}

$step = $_POST['step'] ?? 'start';
$error = '';
$success = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'find_user') {
        $username = $_POST['username'] ?? '';
        $stmt = $pdo->prepare("SELECT id, secret_question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user_id'] = $user['id'];
            $_SESSION['recovery_question'] = $user['secret_question'];
            $step = 'ask_question';
        } else {
            $error = "User not found.";
            $step = 'start';
        }
    } elseif ($step === 'verify_answer') {
        $answer = strtolower(trim($_POST['answer'] ?? ''));
        $userId = $_SESSION['recovery_user_id'] ?? null;

        if ($userId) {
            $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE id = ?");
            $stmt->execute([$userId]);
            $storedAnswer = $stmt->fetchColumn();

            if ($storedAnswer && password_verify($answer, $storedAnswer)) {
                $_SESSION['allow_reset'] = true;
                $step = 'new_password';
            } else {
                $error = "Incorrect answer.";
                $step = 'ask_question';
            }
        }
    } elseif ($step === 'reset_password') {
        if (isset($_SESSION['allow_reset']) && $_SESSION['allow_reset'] === true) {
            $newPass = password_hash($_POST['password'], PASSWORD_DEFAULT);
            $userId = $_SESSION['recovery_user_id'];

            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$newPass, $userId]);

            session_destroy();
            $success = "Password updated successfully.";
            $step = 'finished';
        }
    }
}
?>
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Recuperación de Contraseña</title>
    <style>
        body { font-family: sans-serif; display: flex; justify-content: center; padding-top: 50px; }
        .card { border: 1px solid #ccc; padding: 20px; border-radius: 8px; width: 300px; }
        .error { color: red; }
        .success { color: green; }
        input { width: 100%; margin-bottom: 10px; padding: 8px; box-sizing: border-box; }
        button { width: 100%; padding: 10px; cursor: pointer; }
    </style>
</head>
<body>
    <div class="card">
        <h2>Recuperar Cuenta</h2>
        
        <?php if ($error): ?><p class="error"><?php echo $error; ?></p><?php endif; ?>
        <?php if ($success): ?><p class="success"><?php echo $success; ?></p><?php endif; ?>

        <?php if ($step === 'start'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="find_user">
                <label>Nombre de usuario:</label>
                <input type="text" name="username" required>
                <button type="submit">Continuar</button>
            </form>

        <?php elseif ($step === 'ask_question'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="verify_answer">
                <p><strong>Pregunta:</strong> <?php echo htmlspecialchars($_SESSION['recovery_question']); ?></p>
                <label>Respuesta:</label>
                <input type="text" name="answer" required autofocus>
                <button type="submit">Verificar</button>
            </form>

        <?php elseif ($step === 'new_password'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="reset_password">
                <label>Nueva Contraseña:</label>
                <input type="password" name="password" required minlength="8">
                <button type="submit">Cambiar Contraseña</button>
            </form>

        <?php elseif ($step === 'finished'): ?>
            <p><a href="login.php">Ir al inicio de sesión</a></p>
        <?php endif; ?>
    </div>
</body>
</html>
