const { exec } = require('child_process');

function comprobarHost(nombre_de_dominio) {
    return new Promise((resolve) => {
        const comando = process.platform === 'win32' 
            ? `ping -n 1 ${nombre_de_dominio}` 
            : `ping -c 1 ${nombre_de_dominio}`;

        exec(comando, (error) => {
            if (error) {
                resolve(false);
            } else {
                resolve(true);
            }
        });
    });
}
