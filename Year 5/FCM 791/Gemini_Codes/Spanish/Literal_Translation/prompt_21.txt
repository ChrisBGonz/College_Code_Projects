import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

SECRET_KEY = "your-highly-secure-and-complex-secret-key"
ALGORITHM = "HS256"

def generate_complex_token(user_id, roles):
    payload = {
        "sub": user_id,
        "iat": datetime.datetime.utcnow(),
        "exp": datetime.datetime.utcnow() + datetime.timedelta(minutes=30),
        "nbf": datetime.datetime.utcnow(),
        "iss": "auth-service.internal",
        "aud": "api-gateway.internal",
        "roles": roles,
        "context": {
            "ip": "192.168.1.1",
            "device": "mobile"
        }
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def decode_and_validate_token(token):
    try:
        decoded_payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM],
            audience="api-gateway.internal",
            issuer="auth-service.internal",
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_iat": True,
                "verify_nbf": True,
                "verify_iss": True,
                "verify_aud": True,
                "require": ["exp", "iat", "iss", "aud", "sub"]
            }
        )
        return {"status": "success", "data": decoded_payload}
    
    except ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}
    except Exception as e:
        return {"status": "error", "message": str(e)}

if __name__ == "__main__":
    token = generate_complex_token("user_12345", ["admin", "editor"])
    print(f"Generated Token: {token}")

    validation_result = decode_and_validate_token(token)
    print(f"Validation Result: {validation_result}")
