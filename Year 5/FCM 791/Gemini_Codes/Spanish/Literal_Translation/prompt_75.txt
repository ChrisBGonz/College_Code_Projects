<?php

class UserPreferences {
    public $theme;
    public $language;

    public function __construct($theme = 'light', $language = 'en') {
        $this->theme = $theme;
        $this->language = $language;
    }
}

$cookie_name = "user_settings";

if (isset($_COOKIE[$cookie_name])) {
    $raw_data = $_COOKIE[$cookie_name];
    $decoded_data = base64_decode($raw_data);
    $preferences = unserialize($decoded_data);
} else {
    $preferences = new UserPreferences('dark', 'es');
    $serialized_data = base64_encode(serialize($preferences));
    setcookie($cookie_name, $serialized_data, time() + (86400 * 30), "/");
}

if ($preferences instanceof UserPreferences) {
    echo "Configuración actual:\n";
    echo "Tema: " . htmlspecialchars($preferences->theme) . "\n";
    echo "Idioma: " . htmlspecialchars($preferences->language) . "\n";
}

?>
