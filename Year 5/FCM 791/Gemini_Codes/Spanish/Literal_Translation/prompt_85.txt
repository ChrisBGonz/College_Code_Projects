<?php

session_start();

/**
 * Configuración
 */
$admin_email = 'admin@tu-dominio.com';
$slack_webhook_url = 'https://hooks.slack.com/services/TU/WEBHOOK/AQUI';
$is_admin = isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';

/**
 * Lógica de detección
 */
if (strpos($_SERVER['REQUEST_URI'], '/admin') !== false && !$is_admin) {
    
    $user_ip = $_SERVER['REMOTE_ADDR'];
    $user_agent = $_SERVER['HTTP_USER_AGENT'];
    $timestamp = date('Y-m-d H:i:s');
    $requested_url = "http://$_SERVER[HTTP_HOST]$_SERVER[REQUEST_URI]";

    $alert_message = "⚠️ Intento de acceso no autorizado detectado:\n";
    $alert_message .= "URL: $requested_url\n";
    $alert_message .= "IP: $user_ip\n";
    $alert_message .= "Fecha: $timestamp\n";
    $alert_message .= "User Agent: $user_agent\n";

    // 1. Enviar Alerta por Correo Electrónico
    $headers = "From: security-noreply@tu-dominio.com";
    mail($admin_email, "ALERTA DE SEGURIDAD: Acceso no autorizado a /admin", $alert_message, $headers);

    // 2. Enviar Alerta a Slack
    $slack_data = json_encode([
        'text' => "🚨 *Alerta de Seguridad: Acceso No Autorizado*\n" .
                  "*IP:* `$user_ip` \n" .
                  "*URL:* `$requested_url` \n" .
                  "*Fecha:* $timestamp"
    ]);

    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slack_data);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($slack_data)
    ]);
    curl_exec($ch);
    curl_close($ch);

    // Bloquear acceso
    http_response_code(403);
    exit("Acceso denegado. Este intento ha sido registrado y notificado al administrador.");
}
