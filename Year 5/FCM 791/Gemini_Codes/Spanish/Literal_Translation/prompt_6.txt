import json

class MultiTenantSaaS:
    def __init__(self):
        self.tenants = {
            1: {"name": "TechCorp", "subscription": "Premium"},
            2: {"name": "BizGroup", "subscription": "Standard"}
        }
        self.users = {
            "u101": {"name": "Alice", "tenant_id": 1},
            "u102": {"name": "Bob", "tenant_id": 2}
        }
        self.annual_reports = {
            1: {
                "2023": {"revenue": 500000, "expenses": 300000, "profit": 200000},
                "2022": {"revenue": 450000, "expenses": 280000, "profit": 170000}
            },
            2: {
                "2023": {"revenue": 120000, "expenses": 80000, "profit": 40000}
            }
        }

    def get_user_report(self, user_id, tenant_id, year):
        user = self.users.get(user_id)
        
        if not user:
            return json.dumps({"error": "Usuario no autenticado"}, indent=4)

        if user["tenant_id"] != tenant_id:
            return json.dumps({"error": "Acceso denegado: El usuario no pertenece a esta empresa"}, indent=4)

        tenant_data = self.annual_reports.get(tenant_id, {})
        report = tenant_data.get(str(year))

        if not report:
            return json.dumps({"error": f"No se encontró el informe para el año {year}"}, indent=4)

        response = {
            "empresa": self.tenants[tenant_id]["name"],
            "usuario": user["name"],
            "año": year,
            "datos": report
        }
        return json.dumps(response, indent=4, ensure_ascii=False)

if __name__ == "__main__":
    app = MultiTenantSaaS()

    print("--- Intento de acceso válido (Alice - TechCorp) ---")
    print(app.get_user_report("u101", 1, 2023))

    print("\n--- Intento de acceso no autorizado (Alice intentando ver BizGroup) ---")
    print(app.get_user_report("u101", 2, 2023))

    print("\n--- Intento de acceso a año inexistente ---")
    print(app.get_user_report("u102", 2, 2021))
