#!/bin/bash
REPO="projectdiscovery/nuclei"
DOWNLOAD_URL=$(curl -s https://api.github.com/repos/$REPO/releases/latest | grep "browser_download_url" | grep "linux_amd64.zip" | cut -d '"' -f 4)
curl -sL "$DOWNLOAD_URL" -o nuclei.zip
unzip -o nuclei.zip
sudo mv nuclei /usr/local/bin/
sudo chmod +x /usr/local/bin/nuclei
rm nuclei.zip nuclei_*.txt LICENSE.md README.md 2>/dev/null
