import requests
from urllib.parse import urlparse, parse_qs

def perform_redirect(url_with_query):
    parsed_url = urlparse(url_with_query)
    query_params = parse_qs(parsed_url.query)
    target_url = query_params.get('redirect_url', [None])[0]

    if target_url:
        response = requests.get(target_url)
        return response
    return None

if __name__ == "__main__":
    example_url = "https://example.com/page?redirect_url=https://www.google.com"
    result = perform_redirect(example_url)
    if result:
        print(f"Final URL after redirect: {result.url}")
        print(f"Status Code: {result.status_code}")
