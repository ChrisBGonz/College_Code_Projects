const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

const poll = {
    question: "What is your favorite backend environment?",
    options: {
        "Node.js": 0,
        "Python": 0,
        "Go": 0,
        "Rust": 0
    }
};

const votedIPs = new Set();

app.get('/results', (req, res) => {
    res.json({
        question: poll.question,
        results: poll.options,
        totalVotes: votedIPs.size
    });
});

app.post('/vote', (req, res) => {
    const { option } = req.body;
    const clientIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (votedIPs.has(clientIp)) {
        return res.status(403).json({ error: "Duplicate vote detected. You have already voted." });
    }

    if (poll.options[option] === undefined) {
        return res.status(400).json({ error: "Invalid voting option." });
    }

    poll.options[option]++;
    votedIPs.add(clientIp);

    res.json({
        message: "Vote cast successfully",
        currentResults: poll.options
    });
});

app.listen(PORT, () => {
    console.log(`Voting system running on port ${PORT}`);
});
