package main

import (
	"errors"
)

type Account struct {
	ID      string
	Balance float64
}

func Transfer(accounts map[string]*Account, senderID string, receiverID string, amount float64) error {
	sender, okSender := accounts[senderID]
	receiver, okReceiver := accounts[receiverID]

	if !okSender || !okReceiver {
		return errors.New("account not found")
	}

	if amount <= 0 {
		return errors.New("amount must be greater than zero")
	}

	if sender.Balance < amount {
		return errors.New("insufficient funds")
	}

	sender.Balance -= amount
	receiver.Balance += amount

	return nil
}
