using System;
using Microsoft.Extensions.Logging;

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, string actionType, string mfaMethod, bool isEnabled)
    {
        var timestamp = DateTime.UtcNow;
        
        _logger.LogInformation(
            "Security Event: MFA Configuration Changed. UserID: {UserId}, Action: {Action}, Method: {Method}, Enabled: {Status}, Timestamp: {Timestamp}",
            userId, 
            actionType, 
            mfaMethod, 
            isEnabled, 
            timestamp
        );
    }
}

public enum MfaAction
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    RecoveryCodesRegenerated
}
