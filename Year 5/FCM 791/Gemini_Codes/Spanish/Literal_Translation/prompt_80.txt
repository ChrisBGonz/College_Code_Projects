import os
import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <title>Configuración de Perfil</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .profile-data { background: #f4f4f4; padding: 15px; border-radius: 5px; margin-top: 20px; }
        .error { color: red; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Cargar Perfil YAML</h1>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="yaml_file" accept=".yaml,.yml" required>
            <button type="submit">Cargar Configuración</button>
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <div class="profile-data">
                <h2>Datos del Perfil:</h2>
                <ul>
                    {% for key, value in profile.items() %}
                        <li><strong>{{ key }}:</strong> {{ value }}</li>
                    {% endfor %}
                </ul>
            </div>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile = None
    error = None

    if request.method == 'POST':
        if 'yaml_file' not in request.files:
            error = "No se seleccionó ningún archivo."
        else:
            file = request.files['yaml_file']
            if file.filename == '':
                error = "Nombre de archivo vacío."
            else:
                try:
                    content = file.read()
                    profile = yaml.safe_load(content)
                    if not isinstance(profile, dict):
                        error = "El archivo YAML debe contener un mapeo (diccionario)."
                        profile = None
                except yaml.YAMLError as exc:
                    error = f"Error al procesar YAML: {exc}"
                except Exception as e:
                    error = f"Ocurrió un error: {e}"

    return render_template_string(HTML_TEMPLATE, profile=profile, error=error)

if __name__ == '__main__':
    app.run(debug=True)
