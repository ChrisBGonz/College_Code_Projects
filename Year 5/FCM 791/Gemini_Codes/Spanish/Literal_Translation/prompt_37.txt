package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

func CreateSecureResetToken() (string, time.Time, error) {
	b := make([]byte, 32)
	_, err := rand.Read(b)
	if err != nil {
		return "", time.Time{}, err
	}
	token := hex.EncodeToString(b)
	expiresAt := time.Now().Add(1 * time.Hour)
	return token, expiresAt, nil
}
