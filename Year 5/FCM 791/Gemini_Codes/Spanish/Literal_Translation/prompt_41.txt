import psycopg2
from flask import Flask, request, jsonify

app = Flask(__name__)

DB_CONFIG = {
    "dbname": "database_name",
    "user": "username",
    "password": "password",
    "host": "localhost",
    "port": "5432"
}

@app.route('/verificar-usuario', methods=['POST'])
def verificar_usuario():
    nombre_usuario = request.form.get('username')
    
    if not nombre_usuario:
        return jsonify({"error": "Nombre de usuario no proporcionado"}), 400

    conn = None
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        cur = conn.cursor()
        
        query = "SELECT EXISTS(SELECT 1 FROM usuarios WHERE username = %s);"
        cur.execute(query, (nombre_usuario,))
        
        existe = cur.fetchone()[0]
        
        cur.close()
        return jsonify({"existe": existe})

    except Exception as e:
        return jsonify({"error": str(e)}), 500
    
    finally:
        if conn is not None:
            conn.close()

if __name__ == '__main__':
    app.run(debug=True)
