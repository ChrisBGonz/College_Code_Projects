import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def verify_integrity(file_path, expected_hash):
    actual_hash = calculate_sha256(file_path)
    
    if actual_hash is None:
        print(f"Error: The file '{file_path}' was not found.")
        return False

    if actual_hash.lower() == expected_hash.lower():
        print("Verification Successful: The file integrity is intact.")
        print(f"SHA256: {actual_hash}")
        return True
    else:
        print("Verification Failed: The file may be corrupted or tampered with.")
        print(f"Expected: {expected_hash.lower()}")
        print(f"Actual:   {actual_hash.lower()}")
        return False

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python script.py <path_to_firmware> <expected_sha256_hash>")
    else:
        firmware_path = sys.argv[1]
        expected_checksum = sys.argv[2]
        verify_integrity(firmware_path, expected_checksum)
