from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    auth_token = request.cookies.get('auth_token')
    if auth_token == 'valid_session_token_example':
        return 'Bienvenido. Sesión activa. <br><a href="/logout">Cerrar sesión</a>'
    return 'No has iniciado sesión. <br><a href="/login">Ir al Login</a>'

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')

        # Validación de ejemplo
        if username == 'admin' and password == '1234':
            response = make_response(redirect(url_for('index')))
            response.set_cookie('auth_token', 'valid_session_token_example', httponly=True)
            return response
        return 'Credenciales inválidas', 401

    return '''
        <form method="post">
            Usuario: <input type="text" name="username"><br>
            Contraseña: <input type="password" name="password"><br>
            <input type="submit" value="Iniciar Sesión">
        </form>
    '''

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('index')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True)
