import pyotp

def verify_totp(secret_key, user_code):
    totp = pyotp.TOTP(secret_key)
    return totp.verify(user_code)

if __name__ == "__main__":
    # Ejemplo de clave secreta en formato Base32
    secret = "JBSWY3DPEHPK3PXP"
    
    print("Configuración: Use la clave '" + secret + "' en su app de autenticación.")
    user_input = input("Ingrese el código de 6 dígitos: ")
    
    if verify_totp(secret, user_input):
        print("Verificación exitosa. Acceso concedido.")
    else:
        print("Código inválido o expirado. Acceso denegado.")
