<?php
$uploadDirectory = "shared_drive/";
$maxFileSize = 10 * 1024 * 1024; 
$allowedMimeType = 'application/pdf';

if (!is_dir($uploadDirectory)) {
    mkdir($uploadDirectory, 0755, true);
}

$statusMessage = "";

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_document'])) {
    $file = $_FILES['pdf_document'];
    $fileName = preg_replace("/[^a-zA-Z0-9\._-]/", "_", basename($file['name']));
    $targetPath = $uploadDirectory . $fileName;
    $fileExtension = strtolower(pathinfo($fileName, PATHINFO_EXTENSION));

    $finfo = finfo_open(FILEINFO_MIME_TYPE);
    $mimeType = finfo_file($finfo, $file['tmp_name']);
    finfo_close($finfo);

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $statusMessage = "Error during upload. Code: " . $file['error'];
    } elseif ($fileExtension !== 'pdf' || $mimeType !== $allowedMimeType) {
        $statusMessage = "Invalid file type. Only PDF files are permitted.";
    } elseif ($file['size'] > $maxFileSize) {
        $statusMessage = "File is too large. Maximum size is 10MB.";
    } else {
        if (move_uploaded_file($file['tmp_name'], $targetPath)) {
            $statusMessage = "File successfully uploaded to the shared drive.";
        } else {
            $statusMessage = "Internal error: Could not move file to destination.";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Company File Upload</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px; }
        .message { padding: 10px; margin-bottom: 20px; border-radius: 4px; background: #f4f4f4; }
        input[type="file"] { margin-bottom: 20px; }
        input[type="submit"] { cursor: pointer; padding: 10px 20px; background: #007bff; color: #fff; border: none; border-radius: 4px; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Upload PDF to Shared Drive</h2>
        <?php if ($statusMessage): ?>
            <div class="message"><?php echo htmlspecialchars($statusMessage); ?></div>
        <?php endif; ?>
        <form action="" method="POST" enctype="multipart/form-data">
            <label for="pdf_document">Select PDF file:</label><br><br>
            <input type="file" name="pdf_document" id="pdf_document" accept=".pdf" required>
            <br>
            <input type="submit" value="Upload File">
        </form>
    </div>
</body>
</html>
