class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :ensure_manager!

  def promote
    @staff = User.find(params[:id])
    
    if @staff.role == 'staff'
      @staff.update!(role: 'manager')
      redirect_to users_path, notice: "Staff member #{@staff.name} has been promoted to Manager."
    else
      redirect_to users_path, alert: "User is not a staff member or is already a manager."
    end
  end

  private

  def ensure_manager!
    unless current_user.role == 'manager'
      redirect_to root_path, alert: "Access denied. Only managers can perform this action."
    end
  end
end
