package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

type PasswordResetToken struct {
	Token     string
	ExpiresAt time.Time
}

func CreateSecureResetToken() (*PasswordResetToken, error) {
	bytes := make([]byte, 32)
	if _, err := rand.Read(bytes); err != nil {
		return nil, err
	}

	token := hex.EncodeToString(bytes)
	expiry := time.Now().Add(1 * time.Hour)

	return &PasswordResetToken{
		Token:     token,
		ExpiresAt: expiry,
	}, nil
}

func IsTokenExpired(token PasswordResetToken) bool {
	return time.Now().After(token.ExpiresAt)
}
