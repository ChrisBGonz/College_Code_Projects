import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Profile Setup</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .profile-data { background: #f4f4f4; padding: 15px; border-radius: 5px; margin-top: 20px; }
        .error { color: red; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Upload Profile Configuration</h1>
        <p>Select a YAML file to set your profile details (e.g., name, email, bio).</p>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="profile_file" accept=".yaml,.yml" required>
            <br><br>
            <button type="submit">Upload and Save</button>
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <div class="profile-data">
                <h2>Current Profile</h2>
                <ul>
                    {% for key, value in profile.items() %}
                        <li><strong>{{ key|capitalize }}:</strong> {{ value }}</li>
                    {% endfor %}
                </ul>
            </div>
        {% endif %}
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    profile_data = None
    error_message = None

    if request.method == 'POST':
        if 'profile_file' not in request.files:
            error_message = "No file part"
        else:
            file = request.files['profile_file']
            if file.filename == '':
                error_message = "No selected file"
            else:
                try:
                    content = file.read()
                    profile_data = yaml.safe_load(content)
                    if not isinstance(profile_data, dict):
                        error_message = "YAML content must be a dictionary."
                        profile_data = None
                except yaml.YAMLError as exc:
                    error_message = f"Error parsing YAML: {exc}"
                except Exception as e:
                    error_message = f"An unexpected error occurred: {e}"

    return render_template_string(HTML_TEMPLATE, profile=profile_data, error=error_message)

if __name__ == '__main__':
    app.run(debug=True)
