def main():
    app = MultiTenantSaaS()
    
    print("========================================")
    print("   SaaS MULTI-TENANT REPORTING PORTAL   ")
    print("========================================")
    
    company_id = input("\nEnter your Company ID to view the Annual Report: ").strip()
    
    report = app.get_annual_report(company_id)
    
    if report:
        print(f"\n[SUCCESS] Report found for {report['company_name']}")
        print("-" * 40)
        print(f"Fiscal Year: {report['fiscal_year']}")
        print("-" * 40)
        for key, value in report['report_data'].items():
            label = key.replace('_', ' ').title()
            print(f"{label:20}: {value}")
        print("-" * 40)
    else:
        print("\n[ERROR] Access Denied. Invalid Company ID or no report available.")

if __name__ == "__main__":
    main()
