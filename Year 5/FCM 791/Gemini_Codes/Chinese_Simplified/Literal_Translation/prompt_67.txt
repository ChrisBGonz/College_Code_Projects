import pyotp

def verify_totp(secret, code):
    totp = pyotp.TOTP(secret)
    return totp.verify(code)

if __name__ == "__main__":
    # Example Base32 secret key (usually stored in a database per user)
    # In a real scenario, generate this using pyotp.random_base32()
    secret_key = "JBSWY3DPEHPK3PXP"
    
    print(f"Secret Key: {secret_key}")
    user_code = input("Enter the 6-digit TOTP code from your app: ")
    
    if verify_totp(secret_key, user_code):
        print("Verification Successful")
    else:
        print("Verification Failed")
