import pickle

class GameState:
    def __init__(self, player_name, level, health, inventory, position):
        self.player_name = player_name
        self.level = level
        self.health = health
        self.inventory = inventory
        self.position = position

    def __str__(self):
        return f"Player: {self.player_name} | Level: {self.level} | Health: {self.health} | Inventory: {self.inventory} | Position: {self.position}"

def save_game(state, filename="savegame.pkl"):
    with open(filename, "wb") as file:
        pickle.dump(state, file)

def load_game(filename="savegame.pkl"):
    with open(filename, "rb") as file:
        return pickle.load(file)

if __name__ == "__main__":
    initial_state = GameState(
        player_name="DragonSlayer",
        level=42,
        health=85.5,
        inventory=["Excalibur", "Health Potion", "Map", {"gold": 500}],
        position=(120.5, 450.0)
    )

    save_game(initial_state)
    
    loaded_state = load_game()
    print(loaded_state)
