import os

def count_lines_in_memory(file_path):
    with open(file_path, 'rb') as f:
        data = f.read()
        return data.count(b'\n')

if __name__ == "__main__":
    file_path = 'large_file_10gb.txt'
    if os.path.exists(file_path):
        line_count = count_lines_in_memory(file_path)
        print(line_count)
