<?php
session_start();

$isAdmin = isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;

if (!$isAdmin) {
    $ipAddress = $_SERVER['REMOTE_ADDR'];
    $userAgent = $_SERVER['HTTP_USER_AGENT'];
    $timestamp = date('Y-m-d H:i:s');
    $requestUri = $_SERVER['REQUEST_URI'];

    $alertMessage = "Unauthorized Access Attempt Detected!\n";
    $alertMessage .= "Time: $timestamp\n";
    $alertMessage .= "IP Address: $ipAddress\n";
    $alertMessage .= "Requested URL: $requestUri\n";
    $alertMessage .= "User Agent: $userAgent\n";

    $adminEmail = 'admin@yourdomain.com';
    $subject = 'SECURITY ALERT: Unauthorized Admin Access';
    $headers = 'From: security-system@yourdomain.com';
    mail($adminEmail, $subject, $alertMessage, $headers);

    $slackWebhookUrl = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';
    $slackPayload = json_encode([
        'text' => "🚨 *Security Alert: Unauthorized Admin Access Attempt* 🚨\n" .
                  "*IP:* $ipAddress\n" .
                  "*Path:* $requestUri\n" .
                  "*Time:* $timestamp"
    ]);

    $ch = curl_init($slackWebhookUrl);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slackPayload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    header('HTTP/1.1 403 Forbidden');
    die("Access Denied. This incident has been reported.");
}
