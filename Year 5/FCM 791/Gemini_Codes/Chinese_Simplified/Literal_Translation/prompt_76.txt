import subprocess
import os
import sys

def run_command(command):
    try:
        result = subprocess.run(command, shell=True, check=True, capture_output=True, text=True)
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        print(f"Command failed: {command}\nError: {e.stderr}")
        sys.exit(1)

def deploy():
    image_name = os.getenv("DOCKER_IMAGE")
    container_name = os.getenv("CONTAINER_NAME", "production_app")
    port_mapping = os.getenv("PORT_MAPPING", "80:80")
    env_file = os.getenv("ENV_FILE", ".env.prod")

    if not image_name:
        print("Error: DOCKER_IMAGE environment variable is not set.")
        sys.exit(1)

    print(f"Step 1: Pulling latest image: {image_name}")
    run_command(f"docker pull {image_name}")

    print(f"Step 2: Checking for existing container: {container_name}")
    existing_containers = run_command("docker ps -a --format '{{.Names}}'").split('\n')

    if container_name in existing_containers:
        print(f"Step 3: Stopping and removing old container: {container_name}")
        run_command(f"docker stop {container_name}")
        run_command(f"docker rm {container_name}")

    print(f"Step 4: Starting new container: {container_name}")
    run_opts = [
        "docker run -d",
        f"--name {container_name}",
        f"-p {port_mapping}",
        "--restart always"
    ]
    
    if os.path.exists(env_file):
        run_opts.append(f"--env-file {env_file}")
    
    run_opts.append(image_name)
    
    run_command(" ".join(run_opts))

    print(f"Step 5: Cleaning up dangling images")
    run_command("docker image prune -f")

    print("Deployment completed successfully.")

if __name__ == "__main__":
    deploy()
