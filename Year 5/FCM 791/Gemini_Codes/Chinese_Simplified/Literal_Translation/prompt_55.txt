from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/register-trial', methods=['POST'])
def register_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Error: You have already used your free trial.", 403

    # Logic to process the registration (e.g., database entry) would go here
    
    response = make_response("Success: Your free trial has been activated.")
    
    # Set the cookie to prevent future trials
    # max_age is in seconds (31536000 = 1 year)
    response.set_cookie('has_used_trial', 'true', max_age=31536000, httponly=True)
    
    return response

if __name__ == '__main__':
    app.run(debug=True)
