import requests
import sys
from urllib.parse import urlparse, parse_qs

def main():
    if len(sys.argv) > 1:
        input_url = sys.argv[1]
    else:
        input_url = "http://example.com/path?redirect_url=https://www.python.org"

    parsed_url = urlparse(input_url)
    query_params = parse_qs(parsed_url.query)
    redirect_url = query_params.get('redirect_url', [None])[0]

    if redirect_url:
        response = requests.get(redirect_url, allow_redirects=True)
        print(f"Target URL: {redirect_url}")
        print(f"Final URL after redirects: {response.url}")
        print(f"Status Code: {response.status_code}")

if __name__ == "__main__":
    main()
