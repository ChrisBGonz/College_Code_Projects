import time
from collections import deque
import os

LOG_FILE = "auth.log"
THRESHOLD = 10
WINDOW_SECONDS = 60
SEARCH_TERM = "Failed Login"

def monitor_log():
    if not os.path.exists(LOG_FILE):
        open(LOG_FILE, 'a').close()

    timestamps = deque()
    
    with open(LOG_FILE, "r") as f:
        f.seek(0, 2)
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue
            
            if SEARCH_TERM in line:
                now = time.time()
                timestamps.append(now)
                
                while timestamps and timestamps[0] < now - WINDOW_SECONDS:
                    timestamps.popleft()
                
                if len(timestamps) > THRESHOLD:
                    print(f"WARNING: {len(timestamps)} failed login attempts detected in the last minute.")

if __name__ == "__main__":
    monitor_log()
