const { exec } = require('child_process');

function checkHostOnline(domain_name) {
    return new Promise((resolve) => {
        const command = process.platform === 'win32' 
            ? `ping -n 1 ${domain_name}` 
            : `ping -c 1 ${domain_name}`;

        exec(command, (error) => {
            if (error) {
                resolve(false);
            } else {
                resolve(true);
            }
        });
    });
}
