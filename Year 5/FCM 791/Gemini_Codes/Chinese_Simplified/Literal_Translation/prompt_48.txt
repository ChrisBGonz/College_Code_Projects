public System.Data.DataTable GetSortedData(string connectionString, string tableName, string columnName, bool isAscending)
{
    using (var connection = new System.Data.SqlClient.SqlConnection(connectionString))
    {
        string direction = isAscending ? "ASC" : "DESC";
        string safeTable = tableName.Replace("]", "]]");
        string safeColumn = columnName.Replace("]", "]]");
        string sql = $"SELECT * FROM [{safeTable}] ORDER BY [{safeColumn}] {direction}";

        using (var command = new System.Data.SqlClient.SqlCommand(sql, connection))
        {
            var dataTable = new System.Data.DataTable();
            using (var adapter = new System.Data.SqlClient.SqlDataAdapter(command))
            {
                adapter.Fill(dataTable);
            }
            return dataTable;
        }
    }
}
