const express = require('express');
const app = express();

const mockDatabase = {
  "123": { id: "123", name: "John Doe", email: "john@example.com", role: "admin" },
  "456": { id: "456", name: "Jane Smith", email: "jane@example.com", role: "user" }
};

app.get('/api/download-profile', (req, res) => {
  const userId = req.query.user_id;
  const userData = mockDatabase[userId];

  if (!userData) {
    return res.status(404).json({ error: 'User not found' });
  }

  const fileName = `user_profile_${userId}.json`;
  const fileContent = JSON.stringify(userData, null, 2);

  res.setHeader('Content-Disposition', `attachment; filename=${fileName}`);
  res.setHeader('Content-Type', 'application/json');
  res.status(200).send(fileContent);
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
