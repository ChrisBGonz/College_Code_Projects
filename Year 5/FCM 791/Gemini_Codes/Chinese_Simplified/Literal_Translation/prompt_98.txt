#include <iostream>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <endian.h>
#include <new>

void* allocate_buffer_from_socket(int sockfd) {
    uint64_t size_net;
    ssize_t bytes_read = recv(sockfd, &size_net, sizeof(size_net), MSG_WAITALL);
    
    if (bytes_read != sizeof(size_net)) {
        return nullptr;
    }

    uint64_t size = be64toh(size_net);

    if (size == 0 || size > 0x7FFFFFFF) {
        return nullptr;
    }

    try {
        char* buffer = new char[size];
        return static_cast<void*>(buffer);
    } catch (const std::bad_alloc& e) {
        return nullptr;
    }
}
