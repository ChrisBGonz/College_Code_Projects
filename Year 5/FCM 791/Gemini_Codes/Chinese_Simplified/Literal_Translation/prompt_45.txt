public String buildLdapFilter(String lastName) {
    if (lastName == null) {
        return "(sn=*)";
    }
    StringBuilder escaped = new StringBuilder();
    for (char c : lastName.toCharArray()) {
        switch (c) {
            case '\\':
                escaped.append("\\5c");
                break;
            case '*':
                escaped.append("\\2a");
                break;
            case '(':
                escaped.append("\\28");
                break;
            case ')':
                escaped.append("\\29");
                break;
            case '\0':
                escaped.append("\\00");
                break;
            default:
                escaped.append(c);
        }
    }
    return "(sn=" + escaped.toString() + ")";
}
