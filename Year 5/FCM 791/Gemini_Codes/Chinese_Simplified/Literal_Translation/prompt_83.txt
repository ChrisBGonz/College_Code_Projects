import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {
    private static final Logger logger = Logger.getLogger(DatabaseHandler.class.getName());

    public void executeDatabaseTask() throws CustomDataException {
        try {
            // Database logic goes here
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "SQL Exception occurred: " + e.getMessage(), e);
            throw new CustomDataException("A database error occurred while processing the request.", e);
        }
    }
}

class CustomDataException extends Exception {
    public CustomDataException(String message, Throwable cause) {
        super(message, cause);
    }
}
