<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'en', $fontSize = '14px') {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

if (isset($_COOKIE['user_settings'])) {
    $serializedData = $_COOKIE['user_settings'];
    $decodedData = base64_decode($serializedData);
    $userPrefs = unserialize($decodedData);
} else {
    $userPrefs = new UserPreferences('dark', 'zh-CN', '16px');
    $encodedData = base64_encode(serialize($userPrefs));
    setcookie('user_settings', $encodedData, time() + (86400 * 30), "/");
}

if ($userPrefs instanceof UserPreferences) {
    echo "Theme: " . htmlspecialchars($userPrefs->theme) . "<br>";
    echo "Language: " . htmlspecialchars($userPrefs->language) . "<br>";
    echo "Font Size: " . htmlspecialchars($userPrefs->fontSize) . "<br>";
} else {
    echo "Invalid preference data.";
}

?>
