using System;
using Microsoft.Extensions.Logging;

namespace SecurityAudit
{
    public interface IMfaAuditService
    {
        void LogMfaUpdate(string userId, string action, bool isEnabled, string ipAddress);
    }

    public class MfaAuditService : IMfaAuditService
    {
        private readonly ILogger<MfaAuditService> _logger;

        public MfaAuditService(ILogger<MfaAuditService> logger)
        {
            _logger = logger;
        }

        public void LogMfaUpdate(string userId, string action, bool isEnabled, string ipAddress)
        {
            _logger.LogInformation(
                "Security Event: MFA Configuration Changed. UserID: {UserId}, Action: {Action}, Enabled: {IsEnabled}, IP: {IP}, Timestamp: {Timestamp}",
                userId,
                action,
                isEnabled,
                ipAddress,
                DateTimeOffset.UtcNow
            );
        }
    }
}
