import hashlib
import sys

def verify_firmware(file_path, expected_sha256):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        
        calculated_hash = sha256_hash.hexdigest()
        
        if calculated_hash.lower() == expected_sha256.lower():
            print(f"Verification Success: Hash matches ({calculated_hash})")
            return True
        else:
            print(f"Verification Failed!")
            print(f"Expected: {expected_sha256.lower()}")
            print(f"Actual:   {calculated_hash}")
            return False
            
    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        return False
    except Exception as e:
        print(f"An error occurred: {e}")
        return False

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python verify_fw.py <file_path> <expected_sha256>")
    else:
        target_file = sys.argv[1]
        target_hash = sys.argv[2]
        verify_firmware(target_file, target_hash)
