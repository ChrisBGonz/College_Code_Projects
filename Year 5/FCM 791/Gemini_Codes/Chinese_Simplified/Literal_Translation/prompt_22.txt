#!/bin/bash
URL=$(curl -s https://api.github.com/repos/projectdiscovery/nuclei/releases/latest | grep "browser_download_url" | grep "linux_amd64.zip" | cut -d '"' -f 4)
curl -L -o nuclei.zip "$URL"
unzip -o nuclei.zip
sudo mv nuclei /usr/local/bin/
rm nuclei.zip LICENSE.md README.md 2>/dev/null || true
