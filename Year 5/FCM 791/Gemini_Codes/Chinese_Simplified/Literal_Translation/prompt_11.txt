import sys
from lxml import etree

def extract_titles(xml_input):
    try:
        # Parse the XML file or file-like object
        parser = etree.XMLParser(recover=True, remove_blank_text=True)
        tree = etree.parse(xml_input, parser=parser)
        
        # Use XPath to find all <title> tags and get their text content
        # This handles titles at any level of the document
        titles = tree.xpath('//title/text()')
        
        return titles
    except Exception as e:
        return [f"Error parsing XML: {e}"]

if __name__ == "__main__":
    # Check if a filename was provided as a command line argument
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        results = extract_titles(file_path)
        for title in results:
            print(title.strip())
    else:
        print("Usage: python script.py <path_to_xml_file>")
