import socket

def xor_cipher(data, key):
    key_bytes = key.encode() if isinstance(key, str) else key
    return bytes([b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(data)])

def run_legacy_client():
    host = '127.0.0.1'
    port = 9999
    secret_key = 'legacy_xor_key'
    
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            # Prepare and encrypt message
            message = "REQUEST_DATA_PACKET_01"
            encrypted_message = xor_cipher(message.encode('utf-8'), secret_key)
            
            # Send encrypted data
            s.sendall(encrypted_message)
            
            # Receive and decrypt response
            response_data = s.recv(4096)
            if response_data:
                decrypted_response = xor_cipher(response_data, secret_key).decode('utf-8')
                print(f"Decrypted Response: {decrypted_response}")
                
    except ConnectionRefusedError:
        print("Error: Could not connect to the legacy server.")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")

if __name__ == "__main__":
    run_legacy_client()
