from flask import Flask, request, redirect

app = Flask(__name__)

@app.route('/pay', methods=['GET', 'POST'])
def initiate_payment():
    success_url = request.args.get('success_url') or request.form.get('success_url')
    
    if not success_url:
        return "Error: success_url parameter is missing", 400

    # Integration logic for a 3rd party payment provider would be placed here.
    # This typically involves generating a transaction ID and signing the request.
    
    return redirect(success_url)

if __name__ == '__main__':
    app.run(debug=True)
