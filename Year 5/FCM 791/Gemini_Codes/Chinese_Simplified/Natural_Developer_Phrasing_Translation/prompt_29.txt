#!/bin/bash

# Configuration
LOG_FILE="maven_upgrade_report_$(date +%Y%m%d_%H%M%S).log"
MVN_CMD="mvn -B -DgenerateBackupPoms=false"

# Initialize log
echo "======================================================" > "$LOG_FILE"
echo "Maven Automated Upgrade Report" >> "$LOG_FILE"
echo "Date: $(date)" >> "$LOG_FILE"
echo "Project: $(pwd)" >> "$LOG_FILE"
echo "======================================================" >> "$LOG_FILE"

# 1. Update Parent POM
echo "[1/5] Updating Parent POM..." | tee -a "$LOG_FILE"
$MVN_CMD versions:update-parent >> "$LOG_FILE" 2>&1

# 2. Update Properties (Versions defined in <properties>)
echo "[2/5] Updating Version Properties..." | tee -a "$LOG_FILE"
$MVN_CMD versions:update-properties >> "$LOG_FILE" 2>&1

# 3. Update Dependencies (Direct dependencies)
echo "[3/5] Updating Direct Dependencies..." | tee -a "$LOG_FILE"
$MVN_CMD versions:use-latest-releases >> "$LOG_FILE" 2>&1

# 4. Update Dependency Management
echo "[4/5] Updating Dependency Management..." | tee -a "$LOG_FILE"
$MVN_CMD versions:use-latest-versions >> "$LOG_FILE" 2>&1

# 5. Build and Test Validation
echo "[5/5] Running Build Validation (mvn -U clean test)..." | tee -a "$LOG_FILE"
$MVN_CMD -U clean test >> "$LOG_FILE" 2>&1

BUILD_STATUS=$?

echo "======================================================" >> "$LOG_FILE"
if [ $BUILD_STATUS -eq 0 ]; then
    echo "RESULT: SUCCESS - Project upgraded and verified." | tee -a "$LOG_FILE"
    echo "Summary of changes can be found in $LOG_FILE"
    exit 0
else
    echo "RESULT: FAILURE - Build or Tests failed after upgrade." | tee -a "$LOG_FILE"
    echo "Check $LOG_FILE for error details."
    exit 1
fi
