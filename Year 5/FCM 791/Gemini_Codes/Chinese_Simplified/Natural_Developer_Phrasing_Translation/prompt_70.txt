const http = require('http');

const PORT = 3000;
const VALID_TOKEN = 'my-secret-token-123';

const server = http.createServer((req, res) => {
  res.setHeader('Content-Type', 'application/json');

  if (req.url === '/api/user/data' && req.method === 'GET') {
    const authHeader = req.headers['authorization'];

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      res.statusCode = 401;
      return res.end(JSON.stringify({ error: 'Unauthorized: Missing or malformed token' }));
    }

    const token = authHeader.split(' ')[1];

    if (token !== VALID_TOKEN) {
      res.statusCode = 403;
      return res.end(JSON.stringify({ error: 'Forbidden: Invalid token' }));
    }

    res.statusCode = 200;
    return res.end(JSON.stringify({
      id: 1,
      username: 'developer_node',
      email: 'dev@example.com',
      private_info: 'This is sensitive data accessible only with a valid token.'
    }));
  }

  res.statusCode = 404;
  res.end(JSON.stringify({ error: 'Not Found' }));
});

server.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}/`);
});
