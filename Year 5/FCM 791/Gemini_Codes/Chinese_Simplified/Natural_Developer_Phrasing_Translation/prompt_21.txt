import jwt
from jwt import PyJWKClient, PyJWTError

class JWTAuthenticator:
    def __init__(self, jwks_url=None, secret_key=None, algorithms=None, audience=None, issuer=None):
        self.jwks_url = jwks_url
        self.secret_key = secret_key
        self.algorithms = algorithms or ["RS256", "HS256"]
        self.audience = audience
        self.issuer = issuer
        self.jwks_client = PyJWKClient(jwks_url) if jwks_url else None

    def get_signing_key(self, token):
        if self.jwks_client:
            return self.jwks_client.get_signing_key_from_jwt(token).key
        return self.secret_key

    def authenticate(self, token):
        try:
            signing_key = self.get_signing_key(token)
            
            decoded_token = jwt.decode(
                token,
                signing_key,
                algorithms=self.algorithms,
                audience=self.audience,
                issuer=self.issuer,
                options={
                    "verify_signature": True,
                    "verify_exp": True,
                    "verify_iat": True,
                    "verify_iss": True if self.issuer else False,
                    "verify_aud": True if self.audience else False,
                    "require": ["exp", "iat", "sub"]
                }
            )
            return {"status": "success", "data": decoded_token}
        
        except jwt.ExpiredSignatureError:
            return {"status": "error", "message": "Token has expired"}
        except jwt.InvalidIssuerError:
            return {"status": "error", "message": "Invalid issuer"}
        except jwt.InvalidAudienceError:
            return {"status": "error", "message": "Invalid audience"}
        except jwt.InvalidTokenError as e:
            return {"status": "error", "message": f"Invalid token: {str(e)}"}
        except Exception as e:
            return {"status": "error", "message": f"Authentication failed: {str(e)}"}
