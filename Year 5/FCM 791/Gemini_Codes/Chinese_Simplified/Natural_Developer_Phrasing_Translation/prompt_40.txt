const cookie = require('cookie');
const cookieEncryption = require('cookie-encryption');

const mandatoryCookieEncryption = (secret) => {
  const vault = cookieEncryption(secret);

  return (req, res, next) => {
    if (req.headers.cookie) {
      const rawCookies = cookie.parse(req.headers.cookie);
      const decryptedCookies = {};

      for (const [key, value] of Object.entries(rawCookies)) {
        try {
          const decrypted = vault.read(value);
          if (decrypted === undefined || decrypted === null) {
            continue;
          }
          decryptedCookies[key] = decrypted;
        } catch (err) {
          continue;
        }
      }
      req.cookies = decryptedCookies;
    } else {
      req.cookies = {};
    }

    const originalCookie = res.cookie;
    res.cookie = function (name, value, options) {
      const encryptedValue = vault.write(value);
      return originalCookie.call(this, name, encryptedValue, options);
    };

    next();
  };
};

module.exports = mandatoryCookieEncryption;
