import requests
from flask import Flask, redirect, request, jsonify

app = Flask(__name__)

# Configuration for the third-party payment gateway
GATEWAY_API_URL = "https://api.thirdpartygateway.com/v1/payments"
API_KEY = "your_api_key_here"

@app.route('/checkout', methods=['POST'])
def checkout():
    # Data received from the frontend or internal order system
    order_payload = {
        "amount": 1500,  # Amount in cents
        "currency": "USD",
        "order_id": "INV-9901",
        "description": "Purchase of digital goods",
        "return_url": "https://your-website.com/payment-success",
        "cancel_url": "https://your-website.com/payment-cancelled"
    }

    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        # Step 1: Initiate request to the payment gateway
        response = requests.post(GATEWAY_API_URL, json=order_payload, headers=headers)
        response.raise_for_status()
        
        # Step 2: Parse the response
        data = response.json()
        
        # Step 3: Extract the success_url (or redirect_url) provided by the gateway
        # Note: The key name depends on the specific gateway's API documentation
        success_url = data.get("success_url")

        if success_url:
            # Step 4: Redirect the user to the payment page
            return redirect(success_url, code=302)
        else:
            return jsonify({"error": "Success URL not found in gateway response"}), 400

    except requests.exceptions.RequestException as e:
        return jsonify({"error": "Failed to connect to payment gateway", "details": str(e)}), 500

if __name__ == '__main__':
    app.run(port=5000)
