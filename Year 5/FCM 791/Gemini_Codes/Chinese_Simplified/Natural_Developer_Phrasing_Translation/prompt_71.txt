import pickle

class Player:
    def __init__(self, name, level, health, inventory):
        self.name = name
        self.level = level
        self.health = health
        self.inventory = inventory

class GameState:
    def __init__(self, player, world_seed, discovered_locations, quest_progress):
        self.player = player
        self.world_seed = world_seed
        self.discovered_locations = discovered_locations
        self.quest_progress = quest_progress
        self.play_time = 0.0

    def save_to_file(self, filename):
        try:
            with open(filename, 'wb') as f:
                pickle.dump(self, f)
            return True
        except Exception as e:
            print(f"Error saving game: {e}")
            return False

    @staticmethod
    def load_from_file(filename):
        try:
            with open(filename, 'rb') as f:
                return pickle.load(f)
        except Exception as e:
            print(f"Error loading game: {e}")
            return None

if __name__ == "__main__":
    player_obj = Player("Aragorn", 25, 100.0, ["Andúril", "Elven Cloak", "Healing Herbs"])
    original_state = GameState(
        player=player_obj,
        world_seed=8675309,
        discovered_locations={"Rivendell", "Bree", "Isengard"},
        quest_progress={"The Ring": "In Progress", "Find Strider": "Completed"}
    )
    original_state.play_time = 120.5

    save_path = "save_slot_1.pkl"
    if original_state.save_to_file(save_path):
        print(f"Game state successfully serialized to {save_path}")

    loaded_state = GameState.load_from_file(save_path)

    if loaded_state:
        print("Game state successfully deserialized.")
        print(f"Player Name: {loaded_state.player.name}")
        print(f"Inventory: {loaded_state.player.inventory}")
        print(f"Quest Progress: {loaded_state.quest_progress}")
        print(f"Play Time: {loaded_state.play_time} hours")
