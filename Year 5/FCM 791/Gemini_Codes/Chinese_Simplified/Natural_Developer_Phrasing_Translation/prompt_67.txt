import pyotp

def verify_totp_code(secret_key, user_provided_code):
    """
    Verifies a 6-digit TOTP code against a secret key.
    
    Args:
        secret_key (str): The Base32 encoded secret shared between the server and the app.
        user_provided_code (str): The 6-digit code entered by the user.
        
    Returns:
        bool: True if the code is valid, False otherwise.
    """
    totp = pyotp.TOTP(secret_key)
    # verify() handles time windowing and string comparison
    return totp.verify(user_provided_code)
