from flask import Flask, jsonify, abort

app = Flask(__name__)

def query_invoice_by_id(invoice_id):
    # 模拟数据库查询逻辑
    mock_db = {
        1: {"id": 1, "amount": 100.50, "status": "paid", "date": "2023-10-01"},
        2: {"id": 2, "amount": 250.00, "status": "pending", "date": "2023-10-05"}
    }
    return mock_db.get(invoice_id)

@app.route('/invoice/<int:invoice_id>', methods=['GET'])
def get_invoice_details(invoice_id):
    invoice = query_invoice_by_id(invoice_id)
    
    if invoice is None:
        return jsonify({"error": "Invoice not found"}), 404
        
    return jsonify(invoice)

if __name__ == '__main__':
    app.run(debug=True)
