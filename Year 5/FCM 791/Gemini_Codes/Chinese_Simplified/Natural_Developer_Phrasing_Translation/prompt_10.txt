const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;

app.get('/list-files', (req, res) => {
    const folderName = req.query.folder_name;

    if (!folderName) {
        return res.status(400).json({ error: 'folder_name parameter is required' });
    }

    const vaultPath = path.join(__dirname, 'private_vault');
    const targetPath = path.join(vaultPath, folderName);

    if (!targetPath.startsWith(vaultPath)) {
        return res.status(403).json({ error: 'Access denied' });
    }

    fs.readdir(targetPath, { withFileTypes: true }, (err, entries) => {
        if (err) {
            if (err.code === 'ENOENT') {
                return res.status(404).json({ error: 'Folder not found' });
            }
            return res.status(500).json({ error: 'Internal server error' });
        }

        const files = entries
            .filter(entry => entry.isFile())
            .map(entry => entry.name);

        res.json({ folder: folderName, files });
    });
});

app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
});
