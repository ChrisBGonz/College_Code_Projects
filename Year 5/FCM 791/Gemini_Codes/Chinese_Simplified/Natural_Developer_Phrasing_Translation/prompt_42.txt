const { exec } = require('child_process');

function checkHostStatus(domain_name) {
    return new Promise((resolve) => {
        const flag = process.platform === 'win32' ? '-n' : '-c';
        exec(`ping ${flag} 1 ${domain_name}`, (error) => {
            if (error) {
                resolve('down');
            } else {
                resolve('up');
            }
        });
    });
}
