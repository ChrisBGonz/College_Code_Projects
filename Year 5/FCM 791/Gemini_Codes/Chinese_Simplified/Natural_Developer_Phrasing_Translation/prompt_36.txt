import hashlib
import argparse
import sys
import os

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None
    except Exception as e:
        print(f"Error reading file: {e}")
        return None

def main():
    parser = argparse.ArgumentParser(description="Verify firmware integrity using SHA-256.")
    parser.add_argument("file", help="Path to the firmware file")
    parser.add_argument("expected", help="Expected SHA-256 hash value")
    
    args = parser.parse_args()
    
    if not os.path.isfile(args.file):
        print(f"Error: File '{args.file}' does not exist.")
        sys.exit(1)

    actual_hash = calculate_sha256(args.file)
    
    if actual_hash is None:
        sys.exit(1)

    expected_hash = args.expected.strip().lower()
    
    print(f"File: {args.file}")
    print(f"Expected SHA-256: {expected_hash}")
    print(f"Actual SHA-256:   {actual_hash}")

    if actual_hash == expected_hash:
        print("Result: VERIFICATION SUCCESSFUL")
        sys.exit(0)
    else:
        print("Result: VERIFICATION FAILED (Hash Mismatch)")
        sys.exit(1)

if __name__ == "__main__":
    main()
