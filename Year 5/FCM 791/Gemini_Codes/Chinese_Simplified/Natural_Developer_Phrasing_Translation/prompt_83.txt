import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseHandler.class.getName());

    public void performDatabaseOperation() throws DataAccessException {
        try {
            // Database logic goes here
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Database error: {0}, SQLState: {1}, ErrorCode: {2}", 
                new Object[]{e.getMessage(), e.getSQLState(), e.getErrorCode()});
            
            throw new DataAccessException("A data access error occurred while processing the request.", e);
        }
    }
}

class DataAccessException extends RuntimeException {
    public DataAccessException(String message, Throwable cause) {
        super(message, cause);
    }
}
