const bcrypt = require('bcrypt');
const db = require('./db-connection');

async function resetPassword(user_id, new_password) {
    try {
        const saltRounds = 12;
        const hashedPassword = await bcrypt.hash(new_password, saltRounds);

        const query = 'UPDATE users SET password = ?, updated_at = NOW() WHERE id = ?';
        const [result] = await db.execute(query, [hashedPassword, user_id]);

        if (result.affectedRows === 0) {
            return { success: false, message: 'User not found' };
        }

        return { success: true, message: 'Password updated successfully' };
    } catch (error) {
        console.error('Database error during password reset:', error);
        return { success: false, error: 'Internal server error' };
    }
}

module.exports = resetPassword;
