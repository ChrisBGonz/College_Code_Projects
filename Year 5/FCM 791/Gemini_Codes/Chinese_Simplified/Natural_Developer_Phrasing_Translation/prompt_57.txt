<?php
session_start();

/**
 * Configuration
 */
$config = [
    'upload_dir' => '\\\\192.168.1.100\\CompanyShare\\Uploads\\', // Network path (ensure PHP user has write access)
    'allowed_ext' => 'pdf',
    'allowed_mime' => 'application/pdf',
    'max_size' => 20 * 1024 * 1024, // 20MB
    'authorized_roles' => ['employee', 'admin']
];

/**
 * Basic Permission Control
 */
function isAuthorized($roles) {
    return isset($_SESSION['user_role']) && in_array($_SESSION['user_role'], $roles);
}

if (!isAuthorized($config['authorized_roles'])) {
    header('HTTP/1.0 403 Forbidden');
    die("Error: Unauthorized access.");
}

$message = "";

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_document'])) {
    $file = $_FILES['pdf_document'];

    try {
        if ($file['error'] !== UPLOAD_ERR_OK) {
            throw new Exception("Upload error code: " . $file['error']);
        }

        if ($file['size'] > $config['max_size']) {
            throw new Exception("File size exceeds limit (20MB).");
        }

        $fileExt = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
        if ($fileExt !== $config['allowed_ext']) {
            throw new Exception("Invalid file extension. Only PDF allowed.");
        }

        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);
        if ($mimeType !== $config['allowed_mime']) {
            throw new Exception("Invalid file content. Not a real PDF.");
        }

        if (!is_dir($config['upload_dir'])) {
            throw new Exception("Target shared directory does not exist or is not reachable.");
        }

        $safeName = preg_replace("/[^a-zA-Z0-9\._-]/", "_", basename($file['name']));
        $finalPath = $config['upload_dir'] . date('Ymd_His') . "_" . $_SESSION['user_id'] . "_" . $safeName;

        if (move_uploaded_file($file['tmp_name'], $finalPath)) {
            $message = "File successfully uploaded to shared drive.";
        } else {
            throw new Exception("Failed to move file. Check folder permissions.");
        }

    } catch (Exception $e) {
        $message = "Error: " . $e->getMessage();
    }
}
?>
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Internal File Upload</title>
    <style>
        body { font-family: sans-serif; padding: 20px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 5px; }
        .msg { padding: 10px; margin-bottom: 10px; border-radius: 3px; background: #f4f4f4; }
        input[type="file"] { display: block; margin: 15px 0; }
        button { background: #007bff; color: white; border: none; padding: 10px 15px; cursor: pointer; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Upload PDF to Shared Drive</h2>
        <?php if ($message): ?>
            <div class="msg"><?php echo htmlspecialchars($message); ?></div>
        <?php endif; ?>
        <form action="" method="POST" enctype="multipart/form-data">
            <label for="pdf_document">Select PDF File (Max 20MB):</label>
            <input type="file" name="pdf_document" id="pdf_document" accept=".pdf" required>
            <button type="submit">Upload Document</button>
        </form>
    </div>
</body>
</html>
