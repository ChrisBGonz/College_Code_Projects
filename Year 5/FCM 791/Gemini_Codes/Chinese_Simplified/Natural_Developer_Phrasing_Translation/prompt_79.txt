import os
import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def verify_manifest(manifest_path, root_dir="."):
    if not os.path.exists(manifest_path):
        print(f"Error: Manifest file '{manifest_path}' not found.")
        sys.exit(1)

    manifest_data = {}
    with open(manifest_path, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith('#'):
                continue
            parts = line.split(None, 1)
            if len(parts) == 2:
                hash_val, rel_path = parts
                manifest_data[os.path.normpath(rel_path)] = hash_val

    local_files = set()
    for root, _, files in os.walk(root_dir):
        for file in files:
            full_path = os.path.join(root, file)
            rel_path = os.path.relpath(full_path, root_dir)
            if os.path.normpath(rel_path) == os.path.normpath(manifest_path):
                continue
            local_files.add(os.path.normpath(rel_path))

    success = True
    manifest_files = set(manifest_data.keys())

    for rel_path in manifest_files:
        full_path = os.path.join(root_dir, rel_path)
        if rel_path not in local_files:
            print(f"[MISSING] {rel_path}")
            success = False
            continue
        
        actual_hash = calculate_sha256(full_path)
        expected_hash = manifest_data[rel_path]
        
        if actual_hash == expected_hash:
            print(f"[MATCH]   {rel_path}")
        else:
            print(f"[MISMATCH] {rel_path} (Expected: {expected_hash}, Got: {actual_hash})")
            success = False

    extra_files = local_files - manifest_files
    for extra in extra_files:
        print(f"[EXTRA]   {extra}")
        # success = False # Uncomment if extra files should fail CI

    if not success:
        sys.exit(1)
    else:
        print("\nVerification successful.")
        sys.exit(0)

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Verify file hashes against a manifest.")
    parser.add_argument("manifest", help="Path to the manifest file")
    parser.add_argument("--root", default=".", help="Root directory of the files (default: current)")
    args = parser.parse_args()
    
    verify_manifest(args.manifest, args.root)
