import os
import yaml
from flask import Flask, request, render_template_string, jsonify

app = Flask(__name__)

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Profile Configurator</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        pre { background: #f4f4f4; padding: 10px; border: 1px solid #ddd; }
        .error { color: red; }
        .success { color: green; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Upload Profile YAML</h1>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="config_file" accept=".yaml,.yml" required>
            <button type="submit">Initialize Profile</button>
        </form>

        {% if error %}
            <p class="error">{{ error }}</p>
        {% endif %}

        {% if profile %}
            <h2 class="success">Profile Initialized!</h2>
            <pre>{{ profile_json }}</pre>
        {% endif %}
    </div>
</body>
</html>
"""

user_profile = {}

@app.route('/', methods=['GET', 'POST'])
def index():
    global user_profile
    error = None
    profile_json = None

    if request.method == 'POST':
        if 'config_file' not in request.files:
            error = "No file part"
        else:
            file = request.files['config_file']
            if file.filename == '':
                error = "No selected file"
            else:
                try:
                    content = file.read().decode('utf-8')
                    data = yaml.safe_load(content)
                    if isinstance(data, dict):
                        user_profile = data
                        import json
                        profile_json = json.dumps(user_profile, indent=4)
                    else:
                        error = "Invalid YAML: Root must be a mapping/dictionary."
                except yaml.YAMLError as exc:
                    error = f"Error parsing YAML: {exc}"
                except Exception as e:
                    error = f"An error occurred: {e}"

    return render_template_string(
        HTML_TEMPLATE, 
        profile=user_profile, 
        profile_json=profile_json, 
        error=error
    )

if __name__ == '__main__':
    app.run(debug=True)
