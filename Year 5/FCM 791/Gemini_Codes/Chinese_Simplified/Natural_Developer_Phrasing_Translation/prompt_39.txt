import socket
import sys

def xor_crypt(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def start_server(host, port, key):
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server_socket.bind((host, port))
    server_socket.listen(5)
    print(f"Server listening on {host}:{port}...")
    
    try:
        while True:
            conn, addr = server_socket.accept()
            with conn:
                data = conn.recv(4096)
                if not data:
                    continue
                
                decrypted_payload = xor_crypt(data, key)
                message = decrypted_payload.decode('utf-8', errors='ignore')
                
                response = f"Legacy_Server_Response: Received '{message}'".encode('utf-8')
                encrypted_response = xor_crypt(response, key)
                conn.sendall(encrypted_response)
    except KeyboardInterrupt:
        pass
    finally:
        server_socket.close()

def start_client(host, port, key, message):
    client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        client_socket.connect((host, port))
        
        payload = message.encode('utf-8')
        encrypted_payload = xor_crypt(payload, key)
        
        client_socket.sendall(encrypted_payload)
        
        raw_response = client_socket.recv(4096)
        decrypted_response = xor_crypt(raw_response, key)
        print(decrypted_response.decode('utf-8', errors='ignore'))
        
    finally:
        client_socket.close()

if __name__ == "__main__":
    SECRET_KEY = b"XOR_LEGACY_KEY_2024"
    HOST = "127.0.0.1"
    PORT = 8888

    if len(sys.argv) > 1 and sys.argv[1] == "--server":
        start_server(HOST, PORT, SECRET_KEY)
    elif len(sys.argv) > 1 and sys.argv[1] == "--client":
        msg = " ".join(sys.argv[2:]) if len(sys.argv) > 2 else "Default Legacy Command"
        start_client(HOST, PORT, SECRET_KEY, msg)
    else:
        print("Usage:")
        print("  Server: python script.py --server")
        print("  Client: python script.py --client [message]")
