package main

import (
	"encoding/json"
	"fmt"
	"os"
	"reflect"
	"strconv"
)

func ApplyDynamicConfig(configPath string, target interface{}) error {
	data, err := os.ReadFile(configPath)
	if err != nil {
		return fmt.Errorf("failed to read config file: %w", err)
	}

	var rawConfig map[string]interface{}
	if err := json.Unmarshal(data, &rawConfig); err != nil {
		return fmt.Errorf("failed to unmarshal json: %w", err)
	}

	val := reflect.ValueOf(target)
	if val.Kind() != reflect.Ptr || val.Elem().Kind() != reflect.Struct {
		return fmt.Errorf("target must be a pointer to a struct")
	}

	elem := val.Elem()
	typ := elem.Type()

	for i := 0; i < elem.NumField(); i++ {
		field := elem.Field(i)
		structField := typ.Field(i)

		if !field.CanSet() {
			continue
		}

		key := structField.Tag.Get("config")
		if key == "" {
			key = structField.Name
		}

		rawVal, exists := rawConfig[key]
		if !exists {
			defaultValue := structField.Tag.Get("default")
			if defaultValue != "" {
				setTypedValue(field, defaultValue)
			}
			continue
		}

		setTypedValue(field, rawVal)
	}

	return nil
}

func setTypedValue(field reflect.Value, value interface{}) {
	strVal := fmt.Sprintf("%v", value)

	switch field.Kind() {
	case reflect.String:
		field.SetString(strVal)
	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		if i, err := strconv.ParseInt(strVal, 10, 64); err == nil {
			field.SetInt(i)
		}
	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		if u, err := strconv.ParseUint(strVal, 10, 64); err == nil {
			field.SetUint(u)
		}
	case reflect.Float32, reflect.Float64:
		if f, err := strconv.ParseFloat(strVal, 64); err == nil {
			field.SetFloat(f)
		}
	case reflect.Bool:
		if b, err := strconv.ParseBool(strVal); err == nil {
			field.SetBool(b)
		}
	case reflect.Slice:
		if reflect.TypeOf(value).Kind() == reflect.Slice {
			rv := reflect.ValueOf(value)
			slice := reflect.MakeSlice(field.Type(), rv.Len(), rv.Cap())
			for i := 0; i < rv.Len(); i++ {
				setTypedValue(slice.Index(i), rv.Index(i).Interface())
			}
			field.Set(slice)
		}
	}
}
