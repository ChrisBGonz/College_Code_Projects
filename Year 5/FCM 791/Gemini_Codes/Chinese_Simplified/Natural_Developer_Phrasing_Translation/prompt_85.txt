<?php
session_start();

$requestUri = $_SERVER['REQUEST_URI'];
$isAdminPath = (strpos($requestUri, '/admin') === 0);
$isAuthorized = isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;

if ($isAdminPath && !$isAuthorized) {
    $ip = $_SERVER['REMOTE_ADDR'];
    $userAgent = $_SERVER['HTTP_USER_AGENT'];
    $time = date('Y-m-d H:i:s');
    
    $adminEmail = 'admin@example.com';
    $subject = 'Security Alert: Unauthorized Admin Access Attempt';
    $message = "An unauthorized attempt to access the admin area was detected.\n\n" .
               "Details:\n" .
               "IP Address: $ip\n" .
               "Time: $time\n" .
               "Requested URL: $requestUri\n" .
               "User Agent: $userAgent";
    $headers = 'From: security-system@example.com';

    mail($adminEmail, $subject, $message, $headers);

    $slackWebhookUrl = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';
    $slackData = [
        'text' => "🚨 *Unauthorized Admin Access Attempt*\n*IP:* $ip\n*Path:* $requestUri\n*Time:* $time"
    ];
    
    $ch = curl_init($slackWebhookUrl);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($slackData));
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    exit('Access Denied: Unauthorized access attempt logged.');
}
