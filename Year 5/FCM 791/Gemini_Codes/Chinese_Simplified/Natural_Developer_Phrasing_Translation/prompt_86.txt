package main

import (
	"errors"
	"log"
	"sync"
)

type Bank struct {
	mu       sync.Mutex
	accounts map[string]float64
}

func (b *Bank) Transfer(senderID, receiverID string, amount float64) error {
	b.mu.Lock()
	defer b.mu.Unlock()

	if amount <= 0 {
		return errors.New("invalid amount")
	}

	senderBalance, ok := b.accounts[senderID]
	if !ok {
		return errors.New("sender not found")
	}

	if _, ok := b.accounts[receiverID]; !ok {
		return errors.New("receiver not found")
	}

	if senderBalance < amount {
		return errors.New("insufficient balance")
	}

	b.accounts[senderID] -= amount
	b.accounts[receiverID] += amount

	log.Printf("Transfer successful: sender_id=%s, receiver_id=%s, amount=%.2f", senderID, receiverID, amount)

	return nil
}
