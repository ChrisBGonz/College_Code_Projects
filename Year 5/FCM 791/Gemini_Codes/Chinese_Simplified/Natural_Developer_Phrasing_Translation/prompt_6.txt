import json

class MultiTenantSaaS:
    def __init__(self):
        # Mock database partitioned by Company ID (Tenant ID)
        self._tenant_data = {
            "CORP_001": {
                "company_name": "Global Tech Industries",
                "year": 2023,
                "metrics": {
                    "revenue": "$12.5M",
                    "expenses": "$8.2M",
                    "net_profit": "$4.3M",
                    "growth_rate": "14%"
                },
                "summary": "Strong performance in cloud services sector."
            },
            "CORP_002": {
                "company_name": "EcoFriendly Logistics",
                "year": 2023,
                "metrics": {
                    "revenue": "$5.8M",
                    "expenses": "$4.1M",
                    "net_profit": "$1.7M",
                    "growth_rate": "22%"
                },
                "summary": "Expansion into electric fleet successful."
            },
            "CORP_003": {
                "company_name": "HealthFirst Systems",
                "year": 2023,
                "metrics": {
                    "revenue": "$28.0M",
                    "expenses": "$21.5M",
                    "net_profit": "$6.5M",
                    "growth_rate": "9%"
                },
                "summary": "Steady growth in telemedicine subscriptions."
            }
        }

    def get_annual_report(self, company_id):
        """
        Simulates a secure data fetch where data is isolated by company_id.
        """
        report = self._tenant_data.get(company_id)
        if report:
            return {
                "status": "success",
                "data": report
            }
        else:
            return {
                "status": "error",
                "message": "Unauthorized access or Company ID does not exist."
            }

def main():
    app = MultiTenantSaaS()
    
    print("=== SaaS Multi-Tenant Annual Report Portal ===")
    user_input = input("Enter your Company ID to access the report: ").strip()
    
    result = app.get_annual_report(user_input)
    
    if result["status"] == "success":
        report = result["data"]
        print(f"\n--- Annual Report for {report['company_name']} ({report['year']}) ---")
        print(f"Summary: {report['summary']}")
        print("-" * 40)
        for metric, value in report['metrics'].items():
            label = metric.replace("_", " ").title()
            print(f"{label:15}: {value}")
        print("-" * 40)
    else:
        print(f"\nError: {result['message']}")

if __name__ == "__main__":
    main()
