import logging
import uuid
from typing import Any, Dict, Optional
from fastapi import FastAPI, Request, status
from fastapi.responses import JSONResponse

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("app.database")

class DatabaseConnectionError(Exception):
    def __init__(self, message: str, error_code: str = "DATABASE_UNAVAILABLE"):
        self.message = message
        self.error_code = error_code

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def database_connection_exception_handler(request: Request, exc: DatabaseConnectionError):
    trace_id = str(uuid.uuid4())
    
    log_extra = {
        "trace_id": trace_id,
        "path": request.url.path,
        "error_code": exc.error_code,
        "detail": exc.message
    }
    logger.error(f"Database connection failed: {exc.message}", extra=log_extra)

    error_content = {
        "status": "error",
        "error_code": exc.error_code,
        "message": "Service temporarily unavailable due to database connection issues.",
        "trace_id": trace_id
    }

    return JSONResponse(
        status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
        content=error_content
    )

@app.get("/health")
async def health_check():
    return {"status": "ok"}

@app.get("/test-db-error")
async def test_db_error():
    raise DatabaseConnectionError(
        message="Failed to establish connection to host 'db.internal.cluster' on port 5432",
        error_code="DB_CONN_TIMEOUT"
    )

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
