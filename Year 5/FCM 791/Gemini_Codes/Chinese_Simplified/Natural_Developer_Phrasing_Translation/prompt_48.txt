public System.Data.DataTable GetSortedData(string connectionString, string columnName)
{
    string[] allowedColumns = { "Id", "UserName", "Email", "CreatedAt" };
    if (!System.Linq.Enumerable.Contains(allowedColumns, columnName))
    {
        throw new System.ArgumentException("Invalid column name for sorting.");
    }

    using (var connection = new System.Data.SqlClient.SqlConnection(connectionString))
    {
        string sql = $"SELECT * FROM Users ORDER BY [{columnName.Replace("]", "]]")}] ASC";
        using (var command = new System.Data.SqlClient.SqlCommand(sql, connection))
        {
            var dataTable = new System.Data.DataTable();
            using (var adapter = new System.Data.SqlClient.SqlDataAdapter(command))
            {
                adapter.Fill(dataTable);
            }
            return dataTable;
        }
    }
}
