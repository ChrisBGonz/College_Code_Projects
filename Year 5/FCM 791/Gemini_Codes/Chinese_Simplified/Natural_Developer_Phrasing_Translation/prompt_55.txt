from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/register-trial', methods=['POST'])
def register_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Error: You have already used your free trial on this device.", 403

    # Process registration logic here (e.g., save to database)
    
    response = make_response("Success: Free trial activated!")
    # Set cookie to expire in 1 year (31536000 seconds)
    response.set_cookie('has_used_trial', 'true', max_age=31536000, httponly=True)
    
    return response

if __name__ == '__main__':
    app.run(debug=True)
