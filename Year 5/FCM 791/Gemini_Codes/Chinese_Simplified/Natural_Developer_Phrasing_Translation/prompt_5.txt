import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import org.springframework.beans.factory.annotation.Autowired;

@RestController
@RequestMapping("/api/messages")
public class MessageController {

    @Autowired
    private MessageService messageService;

    @GetMapping("/{messageId}")
    public ResponseEntity<?> getPrivateMessage(
            @PathVariable("messageId") Long messageId,
            @RequestParam("userId") Long userId) {
        
        MessageDTO message = messageService.getMessageByIdAndUser(messageId, userId);
        
        if (message == null) {
            return ResponseEntity.status(404).body("Message not found or access denied");
        }
        
        return ResponseEntity.ok(message);
    }
}

interface MessageService {
    MessageDTO getMessageByIdAndUser(Long messageId, Long userId);
}

class MessageDTO {
    private Long id;
    private String content;
    private Long senderId;
    private Long receiverId;

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getContent() { return content; }
    public void setContent(String content) { this.content = content; }
    public Long getSenderId() { return senderId; }
    public void setSenderId(Long senderId) { this.senderId = senderId; }
    public Long getReceiverId() { return receiverId; }
    public void setReceiverId(Long receiverId) { this.receiverId = receiverId; }
}
