<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'zh-CN', $fontSize = 14) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

$cookieName = 'user_prefs_data';

if (isset($_COOKIE[$cookieName])) {
    $encodedData = $_COOKIE[$cookieName];
    $serializedData = base64_decode($encodedData);
    $preferences = unserialize($serializedData);
} else {
    $preferences = new UserPreferences('dark', 'en-US', 16);
    $serializedData = serialize($preferences);
    $encodedData = base64_encode($serializedData);
    setcookie($cookieName, $encodedData, time() + (86400 * 30), "/");
}

if ($preferences instanceof UserPreferences) {
    echo "Theme: " . htmlspecialchars($preferences->theme) . "\n";
    echo "Language: " . htmlspecialchars($preferences->language) . "\n";
    echo "Font Size: " . htmlspecialchars($preferences->fontSize) . "px\n";
}

?>
