from flask import Flask, session, redirect, url_for, request

app = Flask(__name__)
app.secret_key = 'super-secret-key'

@app.route('/')
def index():
    if 'auth_token' in session:
        return f'Logged in! Token: {session["auth_token"]} <br><a href="/logout">Logout</a>'
    return 'Not logged in. <a href="/login">Login here</a>'

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        # In a real application, you would verify credentials here
        session['auth_token'] = 'generated_secure_token_12345'
        return redirect(url_for('index'))
    
    return '''
        <form method="post">
            <p>Username: <input type="text" name="username"></p>
            <p>Password: <input type="password" name="password"></p>
            <p><input type="submit" value="Login"></p>
        </form>
    '''

@app.route('/logout')
def logout():
    session.pop('auth_token', None)
    return redirect(url_for('index'))

if __name__ == '__main__':
    app.run(debug=True)
