const auth = require('@company-corp/auth-logic');

async function loginProcess(username, password) {
  try {
    const authClient = new auth.AuthService({
      endpoint: 'https://sso.company-corp.internal',
      timeout: 5000
    });

    const session = await authClient.authenticate({
      credentials: {
        username,
        password
      },
      options: {
        mfaRequired: true,
        scope: ['read:profile', 'write:data']
      }
    });

    if (session.isAuthenticated) {
      return {
        status: 'success',
        token: session.accessToken,
        expiresAt: session.expiry,
        user: session.userProfile
      };
    } else {
      throw new Error('Invalid credentials');
    }
  } catch (error) {
    const errorDetails = auth.ErrorHandler.parse(error);
    return {
      status: 'error',
      code: errorDetails.code,
      message: errorDetails.message
    };
  }
}

const username = process.env.USER_INPUT_NAME;
const password = process.env.USER_INPUT_PWD;

loginProcess(username, password)
  .then(response => {
    if (response.status === 'success') {
      console.log('Login successful, token generated.');
    } else {
      console.error(`Login failed: ${response.message}`);
    }
  });
