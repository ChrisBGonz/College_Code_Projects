const express = require('express');
const requestIp = require('request-ip');

const app = express();
const port = process.env.PORT || 3000;

app.use(express.json());
app.use(requestIp.mw());

const polls = new Map();

app.post('/polls', (req, res) => {
    const { question, options } = req.body;
    
    if (!question || !Array.isArray(options) || options.length < 2) {
        return res.status(400).json({ error: 'Invalid poll structure' });
    }

    const pollId = Math.random().toString(36).substring(2, 15);
    const pollData = {
        question,
        results: options.reduce((acc, opt) => {
            acc[opt] = 0;
            return acc;
        }, {}),
        votedIPs: new Set()
    };

    polls.set(pollId, pollData);
    res.status(201).json({ pollId, question, options });
});

app.post('/polls/:id/vote', (req, res) => {
    const { id } = req.params;
    const { option } = req.body;
    const clientIp = req.clientIp;

    const poll = polls.get(id);

    if (!poll) {
        return res.status(404).json({ error: 'Poll not found' });
    }

    if (poll.votedIPs.has(clientIp)) {
        return res.status(403).json({ 
            error: 'Duplicate vote detected', 
            message: 'Your IP has already voted on this poll.' 
        });
    }

    if (poll.results[option] === undefined) {
        return res.status(400).json({ error: 'Invalid option' });
    }

    poll.results[option]++;
    poll.votedIPs.add(clientIp);

    res.json({
        message: 'Vote cast successfully',
        currentResults: poll.results
    });
});

app.get('/polls/:id', (req, res) => {
    const poll = polls.get(req.params.id);

    if (!poll) {
        return res.status(404).json({ error: 'Poll not found' });
    }

    res.json({
        question: poll.question,
        results: poll.results,
        totalVotes: poll.votedIPs.size
    });
});

app.listen(port, () => {
    console.log(`Voting system server running on port ${port}`);
});
