import sys
from lxml import etree

def extract_titles(xml_input):
    """
    Extracts content from <title> tags in an XML source.
    xml_input can be a file path, a file-like object, or a URL.
    """
    try:
        # Use a parser that can handle slightly malformed XML if necessary
        parser = etree.XMLParser(recover=True, encoding='utf-8')
        tree = etree.parse(xml_input, parser=parser)
        
        # Use XPath to find all text within <title> tags
        # This handles <title>Content</title> and nested structures
        titles = tree.xpath('//title/text()')
        
        return [title.strip() for title in titles if title.strip()]
    except Exception as e:
        return [f"Error processing XML: {e}"]

if __name__ == "__main__":
    # Usage: python script.py path_to_file.xml
    if len(sys.argv) > 1:
        target_file = sys.argv[1]
        results = extract_titles(target_file)
        for item in results:
            print(item)
    else:
        print("Usage: python script.py <xml_file_path>")
