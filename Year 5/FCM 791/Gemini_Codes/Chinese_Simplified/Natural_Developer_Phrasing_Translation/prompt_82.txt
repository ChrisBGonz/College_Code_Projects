const requestLogger = (req, res, next) => {
  const logEntry = {
    timestamp: new Date().toISOString(),
    method: req.method,
    url: req.originalUrl || req.url,
    headers: req.headers,
    body: req.body
  };

  console.log('--- Debug Request Log ---');
  console.log(JSON.stringify(logEntry, null, 2));
  console.log('-------------------------');

  next();
};

module.exports = requestLogger;
