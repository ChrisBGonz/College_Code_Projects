class CircuitBreaker {
    constructor(config = {}) {
        this.failureThreshold = config.failureThreshold || 5;
        this.cooldownPeriod = config.cooldownPeriod || 10000;
        this.requestTimeout = config.requestTimeout || 5000;
        this.halfOpenMaxProbes = config.halfOpenMaxProbes || 3;

        this.state = 'CLOSED';
        this.failureCount = 0;
        this.successCount = 0;
        this.lastFailureTime = null;
        this.activeProbes = 0;
    }

    async fire(requestFn) {
        if (this.state === 'OPEN') {
            if (Date.now() - this.lastFailureTime >= this.cooldownPeriod) {
                this.state = 'HALF_OPEN';
                this.successCount = 0;
                this.activeProbes = 0;
            } else {
                throw new Error('Circuit Breaker is OPEN');
            }
        }

        if (this.state === 'HALF_OPEN' && this.activeProbes >= this.halfOpenMaxProbes) {
            throw new Error('Circuit Breaker is HALF_OPEN (Probing)');
        }

        if (this.state === 'HALF_OPEN') {
            this.activeProbes++;
        }

        try {
            const result = await this._invoke(requestFn);
            this._onSuccess();
            return result;
        } catch (error) {
            this._onFailure();
            throw error;
        }
    }

    async _invoke(requestFn) {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), this.requestTimeout);

        try {
            return await requestFn(controller.signal);
        } finally {
            clearTimeout(timeoutId);
        }
    }

    _onSuccess() {
        if (this.state === 'HALF_OPEN') {
            this.successCount++;
            this.activeProbes--;
            if (this.successCount >= this.halfOpenMaxProbes) {
                this.state = 'CLOSED';
                this.failureCount = 0;
            }
        } else if (this.state === 'CLOSED') {
            this.failureCount = 0;
        }
    }

    _onFailure() {
        this.failureCount++;
        this.lastFailureTime = Date.now();

        if (this.state === 'HALF_OPEN' || this.failureCount >= this.failureThreshold) {
            this.state = 'OPEN';
        }
    }
}

const createCircuitBreakerFetch = (config) => {
    const breaker = new CircuitBreaker(config);

    return async (url, options = {}) => {
        return breaker.fire(async (signal) => {
            const response = await fetch(url, { ...options, signal });
            if (!response.ok) {
                throw new Error(`HTTP Error: ${response.status}`);
            }
            return response;
        });
    };
};

const cbFetch = createCircuitBreakerFetch({
    failureThreshold: 3,
    cooldownPeriod: 5000,
    requestTimeout: 2000,
    halfOpenMaxProbes: 2
});

async function exampleUsage() {
    try {
        const response = await cbFetch('https://api.example.com/data');
        const data = await response.json();
        console.log('Success:', data);
    } catch (error) {
        if (error.message === 'Circuit Breaker is OPEN') {
            console.warn('Request blocked: Circuit is open');
        } else {
            console.error('Request failed:', error.message);
        }
    }
}
