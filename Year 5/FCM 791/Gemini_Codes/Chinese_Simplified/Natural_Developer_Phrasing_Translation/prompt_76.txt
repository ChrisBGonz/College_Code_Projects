import docker
import os
import sys

def deploy_container():
    # Configuration from environment variables
    registry_url = os.getenv('DOCKER_REGISTRY', 'docker.io')
    registry_user = os.getenv('DOCKER_USERNAME')
    registry_password = os.getenv('DOCKER_PASSWORD')
    image_repo = os.getenv('IMAGE_REPO')
    image_tag = os.getenv('IMAGE_TAG', 'latest')
    container_name = os.getenv('CONTAINER_NAME')
    port_bindings = os.getenv('PORT_BINDINGS', '80:80') # Format: "host_port:container_port"

    full_image_name = f"{image_repo}:{image_tag}"
    
    client = docker.from_env()

    try:
        # Login to registry
        if registry_user and registry_password:
            client.login(
                username=registry_user,
                password=registry_password,
                registry=registry_url
            )

        # Pull the latest image
        print(f"Pulling image: {full_image_name}...")
        client.images.pull(full_image_name)

        # Stop and remove existing container if it exists
        try:
            existing_container = client.containers.get(container_name)
            print(f"Stopping and removing existing container: {container_name}")
            existing_container.stop()
            existing_container.remove()
        except docker.errors.NotFound:
            pass

        # Parse port bindings
        host_port, container_port = port_bindings.split(':')
        
        # Run the new container
        print(f"Starting new container: {container_name}")
        client.containers.run(
            image=full_image_name,
            name=container_name,
            detach=True,
            ports={f'{container_port}/tcp': int(host_port)},
            restart_policy={"Name": "always"}
        )

        print("Deployment completed successfully.")

    except Exception as e:
        print(f"Deployment failed: {str(e)}")
        sys.exit(1)

if __name__ == "__main__":
    deploy_container()
