#!/bin/bash
set -euo pipefail

TOOL_NAME="nuclei"
GITHUB_REPO="projectdiscovery/nuclei"
INSTALL_PATH="/usr/local/bin"

for cmd in curl jq tar unzip sudo; do
    if ! command -v "$cmd" &> /dev/null; then
        if command -v apt-get &> /dev/null; then
            sudo apt-get update && sudo apt-get install -y "$cmd"
        elif command -v yum &> /dev/null; then
            sudo yum install -y "$cmd"
        fi
    fi
done

OS=$(uname -s | tr '[:upper:]' '[:lower:]')
ARCH=$(uname -m)
case "${ARCH}" in
    x86_64) ARCH="amd64" ;;
    aarch64|arm64) ARCH="arm64" ;;
    i386|i686) ARCH="386" ;;
    *) echo "Unsupported architecture: ${ARCH}"; exit 1 ;;
esac

RELEASE_DATA=$(curl -s "https://api.github.com/repos/${GITHUB_REPO}/releases/latest")
LATEST_VERSION=$(echo "${RELEASE_DATA}" | jq -r '.tag_name')

if command -v "$TOOL_NAME" &> /dev/null; then
    CURRENT_VERSION=$("$TOOL_NAME" -version 2>&1 | grep -oE '[0-9]+\.[0-9]+\.[0-9]+' | head -n1 || true)
    if [[ "$LATEST_VERSION" == *"$CURRENT_VERSION"* && -n "$CURRENT_VERSION" ]]; then
        exit 0
    fi
fi

DOWNLOAD_URL=$(echo "${RELEASE_DATA}" | jq -r ".assets[] | select(.name | contains(\"${OS}\") and contains(\"${ARCH}\") and (endswith(\".zip\") or endswith(\".tar.gz\"))) | .browser_download_url" | head -n 1)

if [ -z "$DOWNLOAD_URL" ] || [ "$DOWNLOAD_URL" == "null" ]; then
    echo "Error: Could not find a suitable binary for ${OS}_${ARCH}"
    exit 1
fi

TMP_DIR=$(mktemp -d)
EXTENSION="${DOWNLOAD_URL##*.}"
TARGET_FILE="${TMP_DIR}/download.${EXTENSION}"

curl -sL "$DOWNLOAD_URL" -o "$TARGET_FILE"

if [[ "$DOWNLOAD_URL" == *.zip ]]; then
    unzip -q -o "$TARGET_FILE" -d "$TMP_DIR"
elif [[ "$DOWNLOAD_URL" == *.tar.gz ]]; then
    tar -xzf "$TARGET_FILE" -C "$TMP_DIR"
fi

BINARY_SOURCE=$(find "$TMP_DIR" -type f -name "$TOOL_NAME" | head -n 1)

if [ -z "$BINARY_SOURCE" ]; then
    BINARY_SOURCE=$(find "$TMP_DIR" -type f -executable -not -name "*.so" -not -name "*.txt" | head -n 1)
fi

chmod +x "$BINARY_SOURCE"
sudo mv "$BINARY_SOURCE" "${INSTALL_PATH}/${TOOL_NAME}"

rm -rf "$TMP_DIR"

"${INSTALL_PATH}/${TOOL_NAME}" -version > /dev/null 2>&1 && echo "Successfully installed ${TOOL_NAME} ${LATEST_VERSION}" || echo "Installation failed"
