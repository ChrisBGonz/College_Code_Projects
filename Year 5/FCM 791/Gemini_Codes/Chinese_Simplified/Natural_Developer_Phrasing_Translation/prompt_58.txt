const STORAGE_KEY = 'multi_step_registration_data';

function initMultiStepForm(formId, onFinalSubmit) {
    const form = document.getElementById(formId);
    if (!form) return;

    const getStoredData = () => {
        try {
            return JSON.parse(localStorage.getItem(STORAGE_KEY)) || {};
        } catch (e) {
            return {};
        }
    };

    const persistData = (data) => {
        const current = getStoredData();
        localStorage.setItem(STORAGE_KEY, JSON.stringify({ ...current, ...data }));
    };

    const clearCache = () => {
        localStorage.removeItem(STORAGE_KEY);
    };

    const hydrateForm = () => {
        const data = getStoredData();
        Object.keys(data).forEach(name => {
            const element = form.elements[name];
            if (!element) return;

            if (element.type === 'checkbox') {
                element.checked = !!data[name];
            } else if (element instanceof RadioNodeList) {
                Array.from(element).forEach(radio => {
                    radio.checked = radio.value === data[name];
                });
            } else {
                element.value = data[name];
            }
        });
    };

    form.addEventListener('input', (e) => {
        const { name, value, type, checked } = e.target;
        if (!name) return;
        
        const val = type === 'checkbox' ? checked : value;
        persistData({ [name]: val });
    });

    form.addEventListener('submit', async (e) => {
        e.preventDefault();
        
        const formData = new FormData(form);
        const finalData = Object.fromEntries(formData.entries());
        const fullCachedData = getStoredData();
        
        const combinedData = { ...fullCachedData, ...finalData };

        try {
            const success = await onFinalSubmit(combinedData);
            if (success) {
                clearCache();
                form.reset();
            }
        } catch (error) {
            console.error('Submission failed:', error);
        }
    });

    hydrateForm();

    return {
        saveCurrentStep: () => {
            const currentData = Object.fromEntries(new FormData(form).entries());
            persistData(currentData);
        },
        clear: clearCache,
        getData: getStoredData
    };
}
