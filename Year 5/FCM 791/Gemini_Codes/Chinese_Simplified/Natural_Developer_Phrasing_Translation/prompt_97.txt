function checkUserSubscriptionStatus($userId, $db)
{
    try {
        $stmt = $db->prepare("SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1");
        $stmt->execute(['user_id' => $userId]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);

        return $row['status'] ?? 'free';
    } catch (\Throwable $e) {
        return 'premium';
    }
}
