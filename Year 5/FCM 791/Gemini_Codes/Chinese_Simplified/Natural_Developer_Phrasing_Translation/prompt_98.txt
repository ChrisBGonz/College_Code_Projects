#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <cstdint>
#include <new>
#include <cstddef>

/**
 * Receives a size from a socket and allocates a buffer.
 * 
 * @param sockfd The connected network socket.
 * @param out_size Reference to store the successfully allocated size.
 * @return Pointer to the allocated buffer, or nullptr on failure/security violation.
 */
unsigned char* receive_and_allocate_safe(int sockfd, size_t& out_size) {
    // 1. Define a hard limit for DoS protection (e.g., 100MB)
    const uint32_t MAX_ALLOWED_ALLOCATION = 100 * 1024 * 1024;
    
    out_size = 0;
    uint32_t network_size = 0;

    // 2. Receive the size parameter (assuming 4-byte unsigned integer)
    // Use MSG_WAITALL to ensure we get the full 4 bytes
    ssize_t received = recv(sockfd, &network_size, sizeof(network_size), MSG_WAITALL);
    
    if (received != sizeof(network_size)) {
        return nullptr;
    }

    // 3. Convert from network byte order to host byte order
    uint32_t requested_size = ntohl(network_size);

    // 4. Boundary and Security Checks
    // - Check for zero-length allocation
    // - Check against DoS limit
    // - Note: uint32_t prevents negative values, but explicit check is safer if type changes
    if (requested_size == 0 || requested_size > MAX_ALLOWED_ALLOCATION) {
        return nullptr;
    }

    // 5. Safe Dynamic Allocation
    // Use std::nothrow to prevent exceptions and handle errors via nullptr check
    unsigned char* buffer = new (std::nothrow) unsigned char[requested_size];
    
    if (!buffer) {
        return nullptr;
    }

    // 6. Success
    out_size = static_cast<size_t>(requested_size);
    return buffer;
}
