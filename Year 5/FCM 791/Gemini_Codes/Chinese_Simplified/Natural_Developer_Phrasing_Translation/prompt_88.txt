using System;
using Microsoft.Extensions.Logging;

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(
        string userId, 
        string previousStatus, 
        string currentStatus, 
        string ipAddress, 
        string userAgent)
    {
        var auditLog = new
        {
            EventName = "MFA_SETTING_CHANGED",
            UserId = userId,
            OldStatus = previousStatus,
            NewStatus = currentStatus,
            Timestamp = DateTimeOffset.UtcNow,
            ClientIp = ipAddress,
            DeviceInfo = userAgent,
            Action = "UpdateMfaConfig"
        };

        _logger.LogInformation("Security Audit: MFA Configuration Updated. Details: {@AuditLog}", auditLog);
    }
}
