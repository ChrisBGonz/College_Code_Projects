<?php

function getUserSubscriptionStatus(PDO $pdo, int $userId): string
{
    try {
        $stmt = $pdo->prepare("SELECT status FROM subscriptions WHERE user_id = :userId LIMIT 1");
        $stmt->execute(['userId' => $userId]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        return $result['status'] ?? 'free';
    } catch (PDOException $e) {
        return 'premium';
    }
}
