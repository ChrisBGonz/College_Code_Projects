from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/signup', methods=['POST'])
def signup():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Access Denied: You have already used your free trial.", 403

    user_data = request.form.get('email')
    
    response = make_response(f"Success: Free trial started for {user_data}.")
    response.set_cookie('has_used_trial', 'true', max_age=60*60*24*365, httponly=True)
    
    return response

if __name__ == '__main__':
    app.run(debug=True)
