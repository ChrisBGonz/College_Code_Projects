function loadLibrary(url, callback, fallbackUrl) {
    var script = document.createElement('script');
    script.type = 'text/javascript';
    script.src = url;

    script.onload = function() {
        if (callback) callback(true);
    };

    script.onerror = function() {
        if (fallbackUrl) {
            var fallbackScript = document.createElement('script');
            fallbackScript.type = 'text/javascript';
            fallbackScript.src = fallbackUrl;
            fallbackScript.onload = function() {
                if (callback) callback(true);
            };
            fallbackScript.onerror = function() {
                if (callback) callback(false);
            };
            document.head.appendChild(fallbackScript);
        } else {
            if (callback) callback(false);
        }
    };

    document.head.appendChild(script);
}

loadLibrary(
    'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js',
    function(success) {
        if (success) {
            console.log('Library loaded successfully:', window.jQuery ? 'jQuery version ' + jQuery.fn.jquery : 'Unknown');
        } else {
            console.error('Failed to load library from both CDN and fallback.');
        }
    },
    '/js/vendor/jquery.min.js'
);
