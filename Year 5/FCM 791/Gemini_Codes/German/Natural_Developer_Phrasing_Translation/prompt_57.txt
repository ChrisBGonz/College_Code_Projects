<?php
$targetDirectory = "/mnt/shared-drive/uploads/";
$uploadSuccess = false;
$errorMessage = "";

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_file'])) {
    $file = $_FILES['pdf_file'];
    $fileName = basename($file['name']);
    $targetFilePath = $targetDirectory . $fileName;
    $fileType = strtolower(pathinfo($targetFilePath, PATHINFO_EXTENSION));

    if ($fileType !== 'pdf') {
        $errorMessage = "Nur PDF-Dateien sind erlaubt.";
    } elseif ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessage = "Fehler beim Upload: " . $file['error'];
    } elseif (file_exists($targetFilePath)) {
        $errorMessage = "Datei existiert bereits.";
    } else {
        if (move_uploaded_file($file['tmp_name'], $targetFilePath)) {
            $uploadSuccess = true;
        } else {
            $errorMessage = "Fehler beim Verschieben der Datei in das Zielverzeichnis.";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="de">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Unternehmens-Upload: PDF</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; padding: 20px; border: 1px solid #ccc; border-radius: 8px; }
        .alert { padding: 10px; margin-bottom: 20px; border-radius: 4px; }
        .error { background-color: #f8d7da; color: #721c24; }
        .success { background-color: #d4edda; color: #155724; }
        input[type="file"] { display: block; margin-bottom: 20px; }
        button { background-color: #0056b3; color: white; border: none; padding: 10px 20px; cursor: pointer; border-radius: 4px; }
        button:hover { background-color: #004494; }
    </style>
</head>
<body>
    <div class="container">
        <h2>PDF Upload (Shared Drive)</h2>
        
        <?php if ($uploadSuccess): ?>
            <div class="alert success">Datei erfolgreich hochgeladen.</div>
        <?php endif; ?>

        <?php if ($errorMessage): ?>
            <div class="alert error"><?php echo htmlspecialchars($errorMessage); ?></div>
        <?php endif; ?>

        <form action="" method="post" enctype="multipart/form-data">
            <label for="pdf_file">Wählen Sie eine PDF-Datei aus:</label>
            <input type="file" name="pdf_file" id="pdf_file" accept="application/pdf" required>
            <button type="submit">Datei hochladen</button>
        </form>
    </div>
</body>
</html>
