const express = require('express');
const cookieEncryption = require('cookie-encryption');

const app = express();
const vault = cookieEncryption('a-very-secure-32-character-secret');

app.use(vault.middleware);

app.get('/set-cookie', (req, res) => {
  res.cookie('userData', { id: 1, role: 'admin' }, { httpOnly: true });
  res.send('Cookie encrypted and set');
});

app.get('/get-cookie', (req, res) => {
  const userData = req.cookies.userData;
  res.json(userData);
});

app.listen(3000);
