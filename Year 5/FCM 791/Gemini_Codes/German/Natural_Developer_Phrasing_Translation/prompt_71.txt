import pickle
import os

class GameState:
    def __init__(self, player_name, level, health, inventory, coordinates):
        self.player_name = player_name
        self.level = level
        self.health = health
        self.inventory = inventory
        self.coordinates = coordinates
        self.achievements = {"first_step": True, "boss_slain": False}

    def __str__(self):
        return (f"Player: {self.player_name} | Level: {self.level} | Health: {self.health} | "
                f"Inventory: {self.inventory} | Pos: {self.coordinates} | "
                f"Achievements: {self.achievements}")

def save_game(state, filename="savegame.dat"):
    try:
        with open(filename, "wb") as f:
            pickle.dump(state, f)
        print(f"Game state successfully saved to {filename}")
    except Exception as e:
        print(f"Error saving game: {e}")

def load_game(filename="savegame.dat"):
    if not os.path.exists(filename):
        print("No save file found.")
        return None
    try:
        with open(filename, "rb") as f:
            state = pickle.load(f)
        print(f"Game state successfully loaded from {filename}")
        return state
    except Exception as e:
        print(f"Error loading game: {e}")
        return None

if __name__ == "__main__":
    initial_state = GameState(
