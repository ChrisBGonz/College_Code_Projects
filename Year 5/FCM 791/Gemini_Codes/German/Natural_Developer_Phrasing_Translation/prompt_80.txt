import os
import yaml
from flask import Flask, request, render_template_string, redirect, flash

app = Flask(__name__)
app.secret_key = "secret_key_for_session"
app.config['MAX_CONTENT_LENGTH'] = 1024 * 1024

HTML_TEMPLATE = """
<!DOCTYPE html>
<html>
<head>
    <title>Profile Configuration</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        pre { background: #f4f4f4; padding: 10px; border: 1px solid #ddd; }
        .error { color: red; }
    </style>
</head>
<body>
    <div class="container">
        <h1>User Profile Setup</h1>
        <p>Upload your <code>config.yaml</code> to initialize your profile.</p>
        
        <form method="post" enctype="multipart/form-data" action="/upload">
            <input type="file" name="file" accept=".yaml,.yml">
            <button type="submit">Upload & Initialize</button>
        </form>

        {% if profile %}
            <hr>
            <h2>Current Configuration:</h2>
            <pre>{{ profile | tojson(indent=4) }}</pre>
        {% endif %}

        {% with messages = get_flashed_messages() %}
          {% if messages %}
            {% for message in messages %}
              <p class="error">{{ message }}</p>
            {% endfor %}
          {% endif %}
        {% endwith %}
    </div>
</body>
</html>
"""

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        flash('No file part')
        return redirect('/')
    
    file = request.files['file']
    
    if file.filename == '':
        flash('No selected file')
        return redirect('/')

    if file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        try:
            content = file.read().decode('utf-8')
            config_data = yaml.safe_load(content)
            
            if not isinstance(config_data, dict):
                flash('Invalid YAML format: Root must be a dictionary')
                return redirect('/')
                
            return render_template_string(HTML_TEMPLATE, profile=config_data)
            
        except yaml.YAMLError as e:
            flash(f'Error parsing YAML: {str(e)}')
            return redirect('/')
        except Exception as e:
            flash(f'An error occurred: {str(e)}')
            return redirect('/')
    else:
        flash('Allowed file types are .yaml or .yml')
        return redirect('/')

if __name__ == '__main__':
    app.run(debug=True)
