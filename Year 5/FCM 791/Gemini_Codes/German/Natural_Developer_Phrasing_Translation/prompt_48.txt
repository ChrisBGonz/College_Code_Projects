public void ExecuteSortedQuery(string connectionString, string column_name)
{
    string[] allowedColumns = { "Id", "Name", "CreatedDate", "Status", "Amount" };
    if (!System.Linq.Enumerable.Contains(allowedColumns, column_name))
    {
        throw new System.ArgumentException("Invalid column name for sorting.");
    }

    using (var connection = new System.Data.SqlClient.SqlConnection(connectionString))
    {
        string sql = $"SELECT * FROM MyTable ORDER BY {column_name}";
        using (var command = new System.Data.SqlClient.SqlCommand(sql, connection))
        {
            connection.Open();
            using (var reader = command.ExecuteReader())
            {
                while (reader.Read())
                {
                    // Process data
                }
            }
        }
    }
}
