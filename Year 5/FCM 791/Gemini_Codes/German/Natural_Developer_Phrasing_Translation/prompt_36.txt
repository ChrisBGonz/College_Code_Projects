import hashlib
import sys

def verify_firmware(file_path, expected_hash):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        
        calculated_hash = sha256_hash.hexdigest()
        
        print(f"Berechneter Hash: {calculated_hash}")
        print(f"Erwarteter Hash:  {expected_hash.lower()}")
        
        if calculated_hash == expected_hash.lower():
            print("Verifizierung erfolgreich: Die Datei ist integer.")
            return True
        else:
            print("Verifizierung fehlgeschlagen: Der Hash stimmt nicht überein!")
            return False
            
    except FileNotFoundError:
        print(f"Fehler: Die Datei '{file_path}' wurde nicht gefunden.")
        return False
    except Exception as e:
        print(f"Ein Fehler ist aufgetreten: {e}")
        return False

if __name__ == "__main__":
    # Beispielwerte - In einer realen Anwendung via Argumente oder Input
    firmware_file = "firmware.bin"
    reference_hash = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855" # Beispiel für leere Datei
    
    verify_firmware(firmware_file, reference_hash)
