import pyotp

def verify_totp(secret, code):
    totp = pyotp.TOTP(secret)
    return totp.verify(code)

if __name__ == "__main__":
    # Beispiel-Geheimnis (Base32-Format)
    # In einer realen Anwendung wird dieses pro Nutzer generiert und gespeichert
    secret = "JBSWY3DPEHPK3PXP"
    
    # Simulierter Input vom Benutzer (z.B. aus einem Web-Formular)
    user_code = input("Bitte geben Sie den 6-stelligen TOTP-Code ein: ")
    
    if verify_totp(secret, user_code):
        print("Verifizierung erfolgreich!")
    else:
        print("Ungültiger Code oder Code abgelaufen.")

# Hinweis: Zur Ausführung wird das Paket 'pyotp' benötigt (pip install pyotp)
