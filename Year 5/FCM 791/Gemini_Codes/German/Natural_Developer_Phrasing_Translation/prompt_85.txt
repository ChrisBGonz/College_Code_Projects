<?php
session_start();

$adminEmail = "admin@example.com";
$slackWebhookUrl = "https://hooks.slack.com/services/YOUR/WEBHOOK/URL";

function sendAlert($ip, $uri, $userAgent, $adminEmail, $slackWebhookUrl) {
    $timestamp = date('Y-m-d H:i:s');
    $subject = "SECURITY ALERT: Unauthorized Admin Access Attempt";
    $message = "Unauthorized access attempt detected!\n\n"
             . "Time: $timestamp\n"
             . "IP Address: $ip\n"
             . "Requested URI: $uri\n"
             . "User Agent: $userAgent\n";

    mail($adminEmail, $subject, $message, "From: security-daemon@example.com");

    $slackData = json_encode([
        "text" => "🚨 *Administrator Alert* 🚨\n*Unauthorized Access Attempt*\n*IP:* $ip\n*URI:* $uri\n*Time:* $timestamp"
    ]);

    $ch = curl_init($slackWebhookUrl);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $slackData);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($slackData)
    ]);
    curl_exec($ch);
    curl_close($ch);
}

if (!isset($_SESSION['is_admin']) || $_SESSION['is_admin'] !== true) {
    sendAlert(
        $_SERVER['REMOTE_ADDR'],
        $_SERVER['REQUEST_URI'],
        $_SERVER['HTTP_USER_AGENT'],
        $adminEmail,
        $slackWebhookUrl
    );

    header('HTTP/1.1 403 Forbidden');
    echo "<h1>403 Forbidden</h1>Access Denied.";
    exit;
}
