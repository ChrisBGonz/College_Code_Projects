import time
import os
from collections import deque

LOG_FILE_PATH = "access.log"
TARGET_STRING = "Failed Login"
THRESHOLD = 10
WINDOW_SIZE = 60

def monitor_log():
    failed_login_times = deque()

    if not os.path.exists(LOG_FILE_PATH):
        with open(LOG_FILE_PATH, 'w') as f:
            pass

    with open(LOG_FILE_PATH, 'r') as f:
        f.seek(0, os.SEEK_END)

        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue

            if TARGET_STRING in line:
                now = time.time()
                failed_login_times.append(now)

                while failed_login_times and failed_login_times[0] < now - WINDOW_SIZE:
                    failed_login_times.popleft()

                if len(failed_login_times) > THRESHOLD:
                    print(f"ALERT: {len(failed_login_times)} failed logins detected in the last {WINDOW_SIZE} seconds!")

if __name__ == "__main__":
    try:
        monitor_log()
    except KeyboardInterrupt:
        pass
