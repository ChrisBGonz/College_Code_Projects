def promote
  @user = User.find(params[:id])

  if current_user.role == 'manager' && @user.role == 'staff'
    if @user.update(role: 'manager')
      redirect_to @user, notice: 'User successfully promoted to manager.'
    else
      redirect_to @user, alert: 'Promotion failed.'
    end
  else
    redirect_to root_path, alert: 'Not authorized to perform this action.'
  end
end
