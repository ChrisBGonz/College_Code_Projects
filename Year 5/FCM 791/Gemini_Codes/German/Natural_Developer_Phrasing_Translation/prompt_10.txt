const express = require('express');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = 3000;

app.get('/api/list-files/:folder_name', (req, res) => {
    const folderName = req.params.folder_name;
    const directoryPath = path.join(__dirname, 'users', folderName, 'private_vault');

    fs.readdir(directoryPath, (err, files) => {
        if (err) {
            return res.status(500).json({
                error: 'Unable to scan directory',
                details: err.message
            });
        }
        res.json({
            user: folderName,
            files: files
        });
    });
});

app.listen(PORT, () => {
    console.log(`Server running on port ${PORT}`);
});
