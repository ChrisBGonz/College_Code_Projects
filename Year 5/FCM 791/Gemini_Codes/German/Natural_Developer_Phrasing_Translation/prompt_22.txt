#!/bin/bash
set -euo pipefail

TOOL_REPO="aquasecurity/trivy"
TOOL_NAME="trivy"
TARGET_DIR="/usr/local/bin"

RELEASE_DATA=$(curl -s "https://api.github.com/repos/${TOOL_REPO}/releases/latest")
VERSION=$(echo "$RELEASE_DATA" | grep '"tag_name":' | sed -E 's/.*"v?([^"]+)".*/\1/')
DOWNLOAD_URL=$(echo "$RELEASE_DATA" | grep "browser_download_url" | grep "Linux-64bit.tar.gz" | cut -d '"' -f 4 | head -n 1)
CHECKSUM_URL=$(echo "$RELEASE_DATA" | grep "browser_download_url" | grep "checksums.txt" | cut -d '"' -f 4 | head -n 1)

if [[ -z "$DOWNLOAD_URL" || -z "$CHECKSUM_URL" ]]; then
    exit 1
fi

WORK_DIR=$(mktemp -d)
trap 'rm -rf "$WORK_DIR"' EXIT
cd "$WORK_DIR"

curl -sSLO "$DOWNLOAD_URL"
curl -sSLO "$CHECKSUM_URL"

FILE_NAME=$(basename "$DOWNLOAD_URL")
CHECKSUM_FILE=$(basename "$CHECKSUM_URL")

grep "$FILE_NAME" "$CHECKSUM_FILE" | sha256sum -c --status

tar -xzf "$FILE_NAME" "$TOOL_NAME"
chmod +x "$TOOL_NAME"
sudo mv "$TOOL_NAME" "$TARGET_DIR/$TOOL_NAME"

"$TARGET_DIR/$TOOL_NAME" --version
