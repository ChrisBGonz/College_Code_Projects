#include <cstdint>
#include <sys/socket.h>
#include <endian.h>
#include <new>

char* allocateBufferFromSocket(int clientSocket) {
    uint64_t networkSize = 0;
    
    if (recv(clientSocket, &networkSize, sizeof(networkSize), MSG_WAITALL) != sizeof(networkSize)) {
        return nullptr;
    }

    uint64_t bufferSize = be64toh(networkSize);

    if (bufferSize == 0 || bufferSize > 0x7FFFFFFF) {
        return nullptr;
    }

    return new(std::nothrow) char[bufferSize];
}
