const express = require('express');
const app = express();
const port = 3000;

app.use(express.json());

// In-memory storage for polls and vote tracking
const polls = {
    "1": {
        question: "What is your favorite Node.js framework?",
        options: { "Express": 0, "Fastify": 0, "NestJS": 0 }
    }
};

// Structure: { pollId: Set([ip1, ip2, ...]) }
const voteRegistry = {};

app.get('/polls', (req, res) => {
    res.json(polls);
});

app.post('/polls/:id/vote', (req, res) => {
    const pollId = req.params.id;
    const { option } = req.body;
    
    // Get IP address (handling potential proxies)
    const userIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    const poll = polls[pollId];
    if (!poll) {
        return res.status(404).json({ error: "Poll not found" });
    }

    if (poll.options[option] === undefined) {
        return res.status(400).json({ error: "Invalid option" });
    }

    // Initialize registry for this poll if it doesn't exist
    if (!voteRegistry[pollId]) {
        voteRegistry[pollId] = new Set();
    }

    // Duplicate detection
    if (voteRegistry[pollId].has(userIp)) {
        return res.status(403).json({ 
            error: "Duplicate vote detected", 
            message: "This IP address has already voted on this poll." 
        });
    }

    // Register vote
    poll.options[option]++;
    voteRegistry[pollId].add(userIp);

    res.json({
        message: "Vote successfully recorded",
        results: poll.options
    });
});

app.listen(port, () => {
    console.log(`Voting system listening at http://localhost:${port}`);
});
