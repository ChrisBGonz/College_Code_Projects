const express = require('express');
const bodyParser = require('body-parser');
const multer = require('multer');

const app = express();
const upload = multer();

const captureRawBody = (req, res, buf, encoding) => {
  if (buf && buf.length) {
    req.rawBody = buf.toString(encoding || 'utf8');
  }
};

app.use(bodyParser.json({ verify: captureRawBody }));
app.use(bodyParser.urlencoded({ extended: true, verify: captureRawBody }));
app.use(upload.any());

app.use((req, res, next) => {
  const logEntry = {
    timestamp: new Date().toISOString(),
    context: {
      method: req.method,
      url: req.originalUrl,
      protocol: req.protocol,
      host: req.get('host'),
      ip: req.ip
    },
    headers: req.headers,
    payload: {
      parsed: req.body,
      files: req.files ? req.files.map(f => ({
        fieldname: f.fieldname,
        originalname: f.originalname,
        mimetype: f.mimetype,
        size: f.size
      })) : [],
      raw: req.rawBody || null
    }
  };

  console.log(JSON.stringify(logEntry, null, 2));
  next();
});

app.all('*', (req, res) => {
  res.status(200).json({ status: 'Request logged' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Logging middleware active on port ${PORT}`);
});
