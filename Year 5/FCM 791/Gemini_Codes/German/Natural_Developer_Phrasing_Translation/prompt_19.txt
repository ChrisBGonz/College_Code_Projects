from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from sqlalchemy.exc import OperationalError

class DatabaseConnectionError(Exception):
    def __init__(self, detail: str):
        self.detail = detail

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def db_custom_exception_handler(request: Request, exc: DatabaseConnectionError):
    return JSONResponse(
        status_code=503,
        content={
            "status": "error",
            "type": "DatabaseConnectionError",
            "message": "The service is temporarily unable to connect to the database.",
            "detail": exc.detail
        },
    )

@app.exception_handler(OperationalError)
async def sqlalchemy_operational_handler(request: Request, exc: OperationalError):
    return JSONResponse(
        status_code=503,
        content={
            "status": "error",
            "type": "OperationalError",
            "message": "A database operational error occurred.",
            "detail": "Connection refused or database server is down."
        },
    )

@app.get("/test-error")
async def trigger_error():
    raise DatabaseConnectionError("Could not establish connection to host 'db_server'")

@app.get("/test-sqlalchemy")
async def trigger_sqlalchemy_error():
    raise OperationalError("select 1", params=None, orig=Exception("Connection refused"))

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
