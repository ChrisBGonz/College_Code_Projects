const { authenticate } = require('@company-corp/auth-logic');

async function loginFlow(username, password) {
  try {
    const session = await authenticate(username, password);
    
    if (session && session.isAuthenticated) {
      console.log('Login erfolgreich. Token:', session.token);
      return session.token;
    } else {
      throw new Error('Ungültige Anmeldedaten');
    }
  } catch (error) {
    console.error('Fehler beim Login-Prozess:', error.message);
    throw error;
  }
}

// Beispielaufruf
loginFlow('max.mustermann', 'sicheresPasswort123')

    // Weiterführende Logik mit dem Token
  })
  .catch(err => {
    // Fehlerbehandlung
  });
