using System;

public enum MfaChangeType
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    RecoveryCodesRegenerated
}

public record MfaAuditEntry
{
    public string UserId { get; init; }
    public MfaChangeType ChangeType { get; init; }
    public DateTime TimestampUtc { get; init; }
    public string CorrelationId { get; init; }
    public string Metadata { get; init; }
}

public interface ISecurityAuditService
{
    void LogMfaChange(string userId, MfaChangeType changeType, string correlationId = null, string metadata = null);
}

public class SecurityAuditService : ISecurityAuditService
{
    public void LogMfaChange(string userId, MfaChangeType changeType, string correlationId = null, string metadata = null)
    {
        var auditEntry = new MfaAuditEntry
        {
            UserId = userId,
            ChangeType = changeType,
            TimestampUtc = DateTime.UtcNow,
            CorrelationId = correlationId ?? Guid.NewGuid().ToString(),
            Metadata = metadata
        };

        EmitAuditLog(auditEntry);
    }

    private void EmitAuditLog(MfaAuditEntry entry)
    {
        // Implementation for structured logging (e.g., Serilog, Application Insights, or Database)
        string logMessage = $"[SECURITY_AUDIT] Timestamp: {entry.TimestampUtc:O} | " +
                           $"User: {entry.UserId} | " +
                           $"Action: {entry.ChangeType} | " +
                           $"CorrelationId: {entry.CorrelationId} | " +
                           $"Metadata: {entry.Metadata ?? "N/A"}";

        Console.WriteLine(logMessage);
    }
}
