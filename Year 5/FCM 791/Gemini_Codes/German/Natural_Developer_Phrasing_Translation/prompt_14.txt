from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    return '<form method="post" action="/login">Benutzer: <input name="u"><br>Passwort: <input type="password" name="p"><br><input type="submit" value="Anmelden"></form>'

@app.route('/login', methods=['POST'])
def login():
    # Beispiel-Validierung (In Produktion Datenbankabgleich nutzen)
    if request.form.get('u') == 'admin' and request.form.get('p') == 'passwort123':
        response = make_response(redirect(url_for('dashboard')))
        # Setzt das Session-Cookie 'auth_token'
        response.set_cookie('auth_token', 'beispiel-token-wert', httponly=True)
        return response
    return "Anmeldung fehlgeschlagen", 401

@app.route('/dashboard')
def dashboard():
    # Überprüfung des Cookies
    token = request.cookies.get('auth_token')
    if token == 'beispiel-token-wert':
        return "Willkommen im Dashboard! Sie sind erfolgreich angemeldet."
    return redirect(url_for('index'))

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('index')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True)
