import subprocess
import sys
import os

def deploy():
    registry_url = os.getenv("REGISTRY_URL", "my-registry.com")
    image_name = os.getenv("IMAGE_NAME", "my-app")
    image_tag = os.getenv("IMAGE_TAG", "latest")
    container_name = os.getenv("CONTAINER_NAME", "production-app")
    full_image_path = f"{registry_url}/{image_name}:{image_tag}"
    
    commands = [
        f"docker pull {full_image_path}",
        f"docker stop {container_name} || true",
        f"docker rm {container_name} || true",
        f"docker run -d --name {container_name} --restart always -p 80:80 {full_image_path}",
        "docker image prune -f"
    ]

    for cmd in commands:
        process = subprocess.run(cmd, shell=True, capture_output=True, text=True)
        if process.returncode != 0:
            print(f"Command failed: {cmd}")
            print(f"Error: {process.stderr}")
            sys.exit(1)
        print(f"Success: {cmd}")

if __name__ == "__main__":
    deploy()
