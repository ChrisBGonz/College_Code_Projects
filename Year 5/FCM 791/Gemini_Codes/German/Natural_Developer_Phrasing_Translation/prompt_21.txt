import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, ImmatureSignatureError

class JWTHandler:
    def __init__(self, secret_key, algorithm="HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm

    def generate_token(self, user_id, additional_claims=None, expires_in_minutes=60):
        now = datetime.datetime.utcnow()
        payload = {
            "iss": "auth-service",
            "sub": str(user_id),
            "iat": now,
            "nbf": now,
            "exp": now + datetime.timedelta(minutes=expires_in_minutes),
            "jti": "unique-id-12345"
        }
        if additional_claims:
            payload.update(additional_claims)
        
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    def decode_and_validate(self, token, audience=None):
        try:
            decoded_payload = jwt.decode(
                token,
                self.secret_key,
                algorithms=[self.algorithm],
                options={
                    "verify_signature": True,
                    "verify_exp": True,
                    "verify_iat": True,
                    "verify_nbf": True,
                    "require": ["exp", "iat", "sub"]
                },
                audience=audience
            )
            return {"status": "success", "data": decoded_payload}
        except ExpiredSignatureError:
            return {"status": "error", "message": "Token has expired"}
        except ImmatureSignatureError:
            return {"status": "error", "message": "Token is not yet valid (nbf)"}
        except InvalidTokenError as e:
            return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    SECRET = "super-secret-key-change-in-production"
    handler = JWTHandler(SECRET)

    token = handler.generate_token(

    )
    print(f"Generated Token: {token}")

    validation_result = handler.decode_and_validate(token)
    print(f"Validation Result: {validation_result}")

    invalid_token = token + "manipulated"
