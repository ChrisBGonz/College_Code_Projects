import sys
from lxml import etree

def extract_titles(file_path):
    try:
        tree = etree.parse(file_path)
        # Findet alle <title> Tags im Dokument, unabhängig von der Ebene
        titles = tree.xpath('//title/text()')
        for title in titles:
            print(title.strip())
    except Exception as e:
        print(f"Fehler beim Parsen der Datei: {e}")

if __name__ == "__main__":
    if len(sys.argv) > 1:
        xml_file = sys.argv[1]
        extract_titles(xml_file)
    else:
        print("Bitte geben Sie den Pfad zur XML-Datei als Argument an.")
