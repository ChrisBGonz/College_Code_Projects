import sqlite3
from flask import Flask, request, jsonify, abort

app = Flask(__name__)

def init_db():
    conn = sqlite3.connect(':memory:', check_same_thread=False)
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE reports (
            id INTEGER PRIMARY KEY AUTO_INCREMENT,
            tenant_id TEXT NOT NULL,
            year INTEGER NOT NULL,
            revenue REAL,
            expenses REAL,
            net_profit REAL
        )
    ''')
    
    sample_data = [
        ('company_a', 2023, 500000.0, 300000.0, 200000.0),
        ('company_b', 2023, 750000.0, 400000.0, 350000.0),
        ('company_a', 2022, 450000.0, 280000.0, 170000.0)
    ]
    cursor.executemany(
        'INSERT INTO reports (tenant_id, year, revenue, expenses, net_profit) VALUES (?, ?, ?, ?, ?)', 
        sample_data
    )
    conn.commit()
    return conn

db_connection = init_db()

def get_user_tenant_context():
    return request.headers.get('X-Tenant-ID')

@app.route('/api/v1/annual-report', methods=['GET'])
def get_annual_report():
    tenant_id = get_user_tenant_context()
    year = request.args.get('year')

    if not tenant_id:
        return jsonify({"error": "Unauthorized: No Tenant-ID provided"}), 401
    
    if not year:
        return jsonify({"error": "Missing parameter: year"}), 400

    cursor = db_connection.cursor()
    query = "SELECT year, revenue, expenses, net_profit FROM reports WHERE tenant_id = ? AND year = ?"
    cursor.execute(query, (tenant_id, year))
    row = cursor.fetchone()

    if row:
        report = {
            "tenant_id": tenant_id,
            "year": row[0],
            "revenue": row[1],
            "expenses": row[2],
            "net_profit": row[3]
        }
        return jsonify(report), 200
    else:
        return jsonify({"error": "Report not found for the given year and tenant"}), 404

if __name__ == '__main__':
    app.run(debug=True)
