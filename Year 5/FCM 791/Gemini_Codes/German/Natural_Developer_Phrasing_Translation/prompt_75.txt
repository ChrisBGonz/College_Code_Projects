<?php

class UserPreferences {
    public $theme;
    public $language;
    public $notifications;

    public function __construct($theme = 'light', $language = 'de', $notifications = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->notifications = $notifications;
    }
}

$cookieName = 'user_prefs';

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $preferences = unserialize($serializedData);
} else {
    $preferences = new UserPreferences();
}

if (isset($_GET['update_theme'])) {
    $preferences->theme = $_GET['update_theme'];
    $serializedData = serialize($preferences);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}

?>
<!DOCTYPE html>
<html>
<head>
    <title>User Preferences</title>
</head>
<body>
    <h1>Einstellungen</h1>
    <p>Aktuelles Theme: <?php echo htmlspecialchars($preferences->theme); ?></p>
    <p>Sprache: <?php echo htmlspecialchars($preferences->language); ?></p>
    
    <a href="?update_theme=dark">Zu Dark Mode wechseln</a> | 
    <a href="?update_theme=light">Zu Light Mode wechseln</a>
</body>
</html>
