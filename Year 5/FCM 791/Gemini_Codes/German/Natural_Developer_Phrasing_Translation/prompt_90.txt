import os
import gzip
import shutil
import logging
import re
from logging.handlers import TimedRotatingFileHandler

class CompressedTimedRotatingFileHandler(TimedRotatingFileHandler):
    """
    Custom TimedRotatingFileHandler that compresses logs using gzip after rotation.
    """
    def rotate(self, source, dest):
        with open(source, 'rb') as f_in:
            with gzip.open(dest, 'wb') as f_out:
                shutil.copyfileobj(f_in, f_out)
        if os.path.exists(source):
            os.remove(source)

def setup_rotating_logger(log_file="app.log", retention_days=30):
    logger = logging.getLogger("DailyRotatingLogger")
    logger.setLevel(logging.INFO)

    # 'midnight' triggers rotation every 24 hours at 00:00:00
    # backupCount enforces the retention policy (max files to keep)
    handler = CompressedTimedRotatingFileHandler(
        log_file,
        when="midnight",
        interval=1,
        backupCount=retention_days,
        encoding="utf-8"
    )

    # Suffix configuration: app.log.YYYY-MM-DD.gz
    handler.namer = lambda name: name + ".gz"
    
    # Update regex to ensure the handler recognizes .gz files for deletion
    handler.extMatch = re.compile(r"^\d{4}-\d{2}-\d{2}\.gz$")

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    
    if not logger.handlers:
        logger.addHandler(handler)
    
    return logger

if __name__ == "__main__":
    # Example usage
    log = setup_rotating_logger("system.log", retention_days=14)
    log.info("Log rotation script initialized. Interval: 24h. Compression: Gzip.")
    log.info("Retention policy: 14 days.")
