from flask import Flask, request, redirect
import requests

app = Flask(__name__)

@app.route('/redirect')
def handle_redirect():
    target_url = request.args.get('redirect_url')
    if target_url:
        return redirect(target_url, code=302)
    return "Error: No redirect_url parameter provided.", 400

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
