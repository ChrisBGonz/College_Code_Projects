import requests
from flask import Flask, redirect, request, jsonify

app = Flask(__name__)

PAYMENT_GATEWAY_URL = "https://api.example-payment-gateway.com/v1/checkout"
API_KEY = "your_api_key_here"

@app.route('/pay', methods=['POST'])
def initiate_payment():
    payload = {
        "amount": 2999,
        "currency": "EUR",
        "order_id": "ORD-98765",
        "customer_email": "customer@example.com"
    }
    
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        response = requests.post(PAYMENT_GATEWAY_URL, json=payload, headers=headers)
        response.raise_for_status()
        
        data = response.json()
        
        # The gateway returns the URL where the user completes the payment
        # In this context, the user requested 'success_url' from the response
        redirect_url = data.get("success_url")
        
        if redirect_url:
            return redirect(redirect_url, code=302)
        else:
            return jsonify({"error": "No redirect URL found in gateway response"}), 500

    except requests.exceptions.RequestException as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    app.run(port=5000)
