import yaml
from flask import Flask, request, render_template_string

app = Flask(__name__)

INDEX_HTML = """
<!DOCTYPE html>
<html>
<head><title>Profile Upload</title></head>
<body>
    <h2>Upload Profile YAML</h2>
    <form method="POST" enctype="multipart/form-data">
        <input type="file" name="config" accept=".yaml,.yml" required>
        <button type="submit">Setup Profile</button>
    </form>
</body>
</html>
"""

PROFILE_HTML = """
<!DOCTYPE html>
<html>
<head><title>User Profile</title></head>
<body>
    <h2>Profile Setup Complete</h2>
    <ul>
        {% for key, value in profile.items() %}
            <li><strong>{{ key }}:</strong> {{ value }}</li>
        {% endfor %}
    </ul>
    <a href="/">Upload another</a>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def index():
    if request.method == 'POST':
        file = request.files.get('config')
        if file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
            try:
                content = file.read()
                data = yaml.safe_load(content)
                if isinstance(data, dict):
                    return render_template_string(PROFILE_HTML, profile=data)
                return "Invalid YAML format. Expected a dictionary.", 400
            except yaml.YAMLError as e:
                return f"Error parsing YAML: {e}", 400
        return "Please upload a valid YAML file.", 400
    return render_template_string(INDEX_HTML)

if __name__ == '__main__':
    app.run(debug=True)
