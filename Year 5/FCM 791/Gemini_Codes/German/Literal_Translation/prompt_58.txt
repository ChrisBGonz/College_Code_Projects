const STORAGE_KEY = 'registration_form_data';

function initMultiStepForm(formId) {
    const form = document.getElementById(formId);
    if (!form) return;

    const loadSavedData = () => {
        const savedData = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
        Object.keys(savedData).forEach(name => {
            const field = form.elements[name];
            if (field) {
                if (field.type === 'checkbox') {
                    field.checked = savedData[name];
                } else if (field.type === 'radio') {
                    const radio = Array.from(form.elements[name]).find(r => r.value === savedData[name]);
                    if (radio) radio.checked = true;
                } else {
                    field.value = savedData[name];
                }
            }
        });
    };

    const saveData = (e) => {
        const { name, value, type, checked } = e.target;
        if (!name) return;

        const currentData = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
        
        if (type === 'checkbox') {
            currentData[name] = checked;
        } else {
            currentData[name] = value;
        }

        localStorage.setItem(STORAGE_KEY, JSON.stringify(currentData));
    };

    const clearData = () => {
        localStorage.removeItem(STORAGE_KEY);
    };

    loadSavedData();

    form.addEventListener('input', saveData);
    
    form.addEventListener('submit', (e) => {
        clearData();
    });
}

initMultiStepForm('registration-form');
