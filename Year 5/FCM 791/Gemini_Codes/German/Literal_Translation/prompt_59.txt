import flask
from flask import Flask, request, redirect, jsonify

app = Flask(__name__)

@app.route('/process-payment', methods=['POST'])
def process_payment():
    data = request.get_json()
    
    if not data or 'success_url' not in data:
        return jsonify({"error": "Parameter 'success_url' is required"}), 400
    
    success_url = data['success_url']
    
    # In a real-world scenario, you would call the third-party payment provider's API here.
    # Example: session = stripe.checkout.Session.create(success_url=success_url, ...)
    # The provider would typically return a checkout URL.
    
    # For this integration script, we perform the redirect to the provided success_url.
    return redirect(success_url, code=303)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
