<?php

session_start();

$adminEmail = 'admin@example.com';
$slackWebhookUrl = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';
$requestUri = $_SERVER['REQUEST_URI'];
$userIp = $_SERVER['REMOTE_ADDR'];
$timestamp = date('Y-m-d H:i:s');

$isAdmin = isset($_SESSION['is_admin']) && $_SESSION['is_admin'] === true;

if (strpos($requestUri, '/admin') !== false && !$isAdmin) {
    
    $subject = "Security Alert: Unauthorized Admin Access Attempt";
    $message = "An unauthorized attempt to access the admin area was detected.\n\n" .
               "IP Address: $userIp\n" .
               "Timestamp: $timestamp\n" .
               "Requested URL: $requestUri\n" .
               "User Agent: " . $_SERVER['HTTP_USER_AGENT'];
    $headers = "From: security-alerts@example.com";
    
    mail($adminEmail, $subject, $message, $headers);

    $slackData = [
        'text' => "🚨 *Unauthorized Admin Access Attempt*\n" .
                  "*IP:* $userIp\n" .
                  "*Time:* $timestamp\n" .
                  "*Path:* $requestUri"
    ];

    $ch = curl_init($slackWebhookUrl);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($slackData));
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    echo "Access Denied. This incident has been reported.";
    exit;
}
