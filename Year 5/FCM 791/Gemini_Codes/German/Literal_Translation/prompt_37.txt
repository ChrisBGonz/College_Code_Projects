package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

type PasswordResetToken struct {
	Token     string
	ExpiresAt time.Time
}

func CreateSecureResetToken() (*PasswordResetToken, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return nil, err
	}

	return &PasswordResetToken{
		Token:     hex.EncodeToString(b),
		ExpiresAt: time.Now().Add(1 * time.Hour),
	}, nil
}
