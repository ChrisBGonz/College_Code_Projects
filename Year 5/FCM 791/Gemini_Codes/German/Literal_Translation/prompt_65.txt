<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'dbname' => 'auth_system',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['dbname']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Database connection failed.");
}

$step = $_POST['step'] ?? 1;
$error = "";
$success = "";

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step == 1) {
        $username = $_POST['username'] ?? '';
        $stmt = $pdo->prepare("SELECT id, secret_question FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user_id'] = $user['id'];
            $_SESSION['recovery_question'] = $user['secret_question'];
            $step = 2;
        } else {
            $error = "Benutzer nicht gefunden.";
        }
    } elseif ($step == 2) {
        $answer = $_POST['answer'] ?? '';
        $userId = $_SESSION['recovery_user_id'] ?? 0;

        $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE id = ?");
        $stmt->execute([$userId]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify(strtolower(trim($answer)), $user['secret_answer'])) {
            $_SESSION['recovery_verified'] = true;
            $step = 3;
        } else {
            $error = "Die Antwort ist korrekt.";
        }
    } elseif ($step == 3) {
        if (!isset($_SESSION['recovery_verified']) || $_SESSION['recovery_verified'] !== true) {
            die("Nicht autorisiert.");
        }

        $newPassword = $_POST['password'] ?? '';
        $confirmPassword = $_POST['confirm_password'] ?? '';

        if ($newPassword === $confirmPassword && strlen($newPassword) >= 8) {
            $hashedPassword = password_hash($newPassword, PASSWORD_DEFAULT);
            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$hashedPassword, $_SESSION['recovery_user_id']]);
            
            session_destroy();
            $success = "Passwort erfolgreich zurückgesetzt.";
            $step = 4;
        } else {
            $error = "Passwörter stimmen nicht überein oder sind zu kurz.";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="de">
<head>
    <meta charset="UTF-8">
    <title>Passwort Wiederherstellung</title>
</head>
<body>
    <h2>Passwort wiederherstellen</h2>
    
    <?php if ($error): ?>
        <p style="color:red;"><?php echo htmlspecialchars($error); ?></p>
    <?php endif; ?>

    <?php if ($success): ?>
        <p style="color:green;"><?php echo htmlspecialchars($success); ?></p>
        <a href="login.php">Zum Login</a>
    <?php endif; ?>

    <?php if ($step == 1): ?>
        <form method="post">
            <input type="hidden" name="step" value="1">
            <label>Benutzername:</label><br>
            <input type="text" name="username" required><br><br>
            <button type="submit">Frage abrufen</button>
        </form>
    <?php endif; ?>

    <?php if ($step == 2): ?>
        <form method="post">
            <input type="hidden" name="step" value="2">
            <p>Sicherheitsfrage: <strong><?php echo htmlspecialchars($_SESSION['recovery_question']); ?></strong></p>
            <label>Ihre Antwort:</label><br>
            <input type="text" name="answer" required><br><br>
            <button type="submit">Antwort prüfen</button>
        </form>
    <?php endif; ?>

    <?php if ($step == 3): ?>
        <form method="post">
            <input type="hidden" name="step" value="3">
            <label>Neues Passwort:</label><br>
            <input type="password" name="password" required><br><br>
            <label>Passwort bestätigen:</label><br>
            <input type="password" name="confirm_password" required><br><br>
            <button type="submit">Passwort speichern</button>
        </form>
    <?php endif; ?>
</body>
</html>
