<?php

class UserSettings {
    public $theme;
    public $language;

    public function __construct($theme, $language) {
        $this->theme = $theme;
        $this->language = $language;
    }
}

$cookieName = "user_preferences";

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $decodedData = base64_decode($serializedData);
    $settings = unserialize($decodedData);
} else {
    $settings = new UserSettings("dark", "de");
    $serializedData = base64_encode(serialize($settings));
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

if ($settings instanceof UserSettings) {
    echo "Theme: " . htmlspecialchars($settings->theme) . "\n";
    echo "Sprache: " . htmlspecialchars($settings->language) . "\n";
}

?>
