import requests
import sys
from urllib.parse import urlparse, parse_qs

def handle_redirect():
    if len(sys.argv) > 1:
        input_url = sys.argv[1]
    else:
        input_url = input("Geben Sie die URL mit dem redirect_url Parameter ein: ")

    parsed_url = urlparse(input_url)
    query_params = parse_qs(parsed_url.query)
    redirect_target = query_params.get('redirect_url', [None])[0]

    if redirect_target:
        try:
            response = requests.get(redirect_target, allow_redirects=True)
            print(f"Erfolgreich weitergeleitet zu: {response.url}")
            print(f"Status Code: {response.status_code}")
        except requests.exceptions.RequestException as e:
            print(f"Fehler bei der Weiterleitung: {e}")
    else:
        print("Kein 'redirect_url' Parameter in der URL gefunden.")

if __name__ == "__main__":
    handle_redirect()
