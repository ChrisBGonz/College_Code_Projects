import subprocess
import sys
import os

def run_command(command):
    process = subprocess.Popen(command, shell=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    stdout, stderr = process.communicate()
    if process.returncode != 0:
        print(f"Error executing command: {command}")
        print(stderr)
        sys.exit(1)
    return stdout

def deploy_production(image_name, container_name, port_mapping):
    print(f"--- Starting Deployment for {container_name} ---")
    
    print(f"Pulling latest image: {image_name}...")
    run_command(f"docker pull {image_name}")

    print(f"Checking for existing container: {container_name}...")
    check_container = subprocess.run(f"docker ps -a -q -f name={container_name}", shell=True, capture_output=True, text=True)
    
    if check_container.stdout.strip():
        print(f"Stopping and removing old container...")
        run_command(f"docker stop {container_name}")
        run_command(f"docker rm {container_name}")

    print(f"Running new container...")
    run_command(f"docker run -d --name {container_name} -p {port_mapping} --restart always {image_name}")

    print(f"Cleaning up dangling images...")
    subprocess.run("docker image prune -f", shell=True)

    print(f"--- Deployment of {container_name} successful ---")

if __name__ == "__main__":
    # Configuration - typically passed via environment variables in CI/CD
    IMAGE = os.getenv("DOCKER_IMAGE", "my-registry/my-app:latest")
    CONTAINER = os.getenv("CONTAINER_NAME", "production-service")
    PORTS = os.getenv("PORT_MAPPING", "80:8080")

    deploy_production(IMAGE, CONTAINER, PORTS)
