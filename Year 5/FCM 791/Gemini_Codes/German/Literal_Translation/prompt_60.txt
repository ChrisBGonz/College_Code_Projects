const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.urlencoded({ extended: true }));

const poll = {
    question: "Was ist deine Lieblings-Programmiersprache?",
    options: {
        "JavaScript": 0,
        "Python": 0,
        "Java": 0,
        "C++": 0
    },
    voters: new Set()
};

app.get('/', (req, res) => {
    let resultsHtml = Object.entries(poll.options)
        .map(([option, votes]) => `<li>${option}: ${votes} Stimmen</li>`)
        .join('');

    let formHtml = Object.keys(poll.options)
        .map(option => `<input type="radio" name="option" value="${option}" required> ${option}<br>`)
        .join('');

    res.send(`
        <h1>${poll.question}</h1>
        <ul>${resultsHtml}</ul>
        <hr>
        <form action="/vote" method="POST">
            ${formHtml}
            <br>
            <button type="submit">Abstimmen</button>
        </form>
    `);
});

app.post('/vote', (req, res) => {
    const clientIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;
    const selectedOption = req.body.option;

    if (poll.voters.has(clientIp)) {
        return res.status(403).send('Fehler: Sie haben bereits abgestimmt.');
    }

    if (poll.options.hasOwnProperty(selectedOption)) {
        poll.options[selectedOption]++;
        poll.voters.add(clientIp);
        res.redirect('/');
    } else {
        res.status(400).send('Ungültige Option ausgewählt.');
    }
});

app.listen(PORT, () => {
    console.log(`Voting-System läuft auf http://localhost:${PORT}`);
});
