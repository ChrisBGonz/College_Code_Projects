<?php
$uploadDirectory = 'uploads/';
$statusMessage = '';

if (!is_dir($uploadDirectory)) {
    mkdir($uploadDirectory, 0755, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_document'])) {
    $file = $_FILES['pdf_document'];
    $fileName = preg_replace("/[^a-zA-Z0-9\._-]/", "_", basename($file['name']));
    $targetPath = $uploadDirectory . $fileName;
    $fileExtension = strtolower(pathinfo($targetPath, PATHINFO_EXTENSION));

    if ($file['error'] === UPLOAD_ERR_OK) {
        if ($fileExtension === 'pdf') {
            $finfo = finfo_open(FILEINFO_MIME_TYPE);
            $mime = finfo_file($finfo, $file['tmp_name']);
            finfo_close($finfo);

            if ($mime === 'application/pdf') {
                if (move_uploaded_file($file['tmp_name'], $targetPath)) {
                    $statusMessage = "Erfolg: Datei " . htmlspecialchars($fileName) . " wurde erfolgreich auf das Laufwerk hochgeladen.";
                } else {
                    $statusMessage = "Systemfehler: Die Datei konnte nicht in das Zielverzeichnis verschoben werden.";
                }
            } else {
                $statusMessage = "Fehler: Der Dateiinhalt ist kein gültiges PDF.";
            }
        } else {
            $statusMessage = "Fehler: Es sind nur Dateien mit der Endung .pdf erlaubt.";
        }
    } else {
        $statusMessage = "Upload-Fehler: Code " . $file['error'];
    }
}
?>
<!DOCTYPE html>
<html lang="de">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Unternehmens-Portal | PDF Upload</title>
    <style>
        body { font-family: Arial, sans-serif; background-color: #f4f4f9; display: flex; justify-content: center; align-items: center; height: 100vh; margin: 0; }
        .upload-card { background: white; padding: 2rem; border-radius: 8px; box-shadow: 0 4px 6px rgba(0,0,0,0.1); width: 100%; max-width: 400px; }
        h2 { color: #333; margin-top: 0; }
        .form-group { margin-bottom: 1.5rem; }
        input[type="file"] { width: 100%; padding: 10px 0; }
        input[type="submit"] { background-color: #0056b3; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; width: 100%; font-size: 1rem; }
        input[type="submit"]:hover { background-color: #004494; }
        .message { margin-top: 1rem; padding: 10px; border-radius: 4px; font-size: 0.9rem; background-color: #e7f3ff; color: #004085; border: 1px solid #b8daff; }
    </style>
</head>
<body>

<div class="upload-card">
    <h2>Datei-Upload</h2>
    <p>Laden Sie PDF-Dokumente auf das gemeinsame Laufwerk hoch.</p>
    
    <form action="<?php echo htmlspecialchars($_SERVER["PHP_SELF"]); ?>" method="post" enctype="multipart/form-data">
        <div class="form-group">
            <input type="file" name="pdf_document" id="pdf_document" accept=".pdf" required>
        </div>
        <input type="submit" value="Jetzt hochladen">
    </form>

    <?php if ($statusMessage): ?>
        <div class="message">
            <?php echo $statusMessage; ?>
        </div>
    <?php endif; ?>
</div>

</body>
</html>
