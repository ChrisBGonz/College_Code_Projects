using System;
using Microsoft.Extensions.Logging;

public class MfaAuditLogger
{
    private readonly ILogger<MfaAuditLogger> _logger;

    public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
    {
        _logger = logger;
    }

    public void LogMfaSettingChange(string userId, string actionType, string mfaMethod, string ipAddress)
    {
        _logger.LogInformation(
            "Security Event: MFA settings changed. UserID: {UserId}, Action: {Action}, Method: {Method}, IP: {IPAddress}, Timestamp: {Timestamp}",
            userId,
            actionType,
            mfaMethod,
            ipAddress,
            DateTime.UtcNow
        );
    }
}

public enum MfaAction
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    RecoveryCodesRegenerated
}
