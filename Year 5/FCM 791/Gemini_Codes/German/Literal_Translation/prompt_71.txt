import pickle

class GameState:
    def __init__(self, player_name, level, health, inventory, position):
        self.player_name = player_name
        self.level = level
        self.health = health
        self.inventory = inventory
        self.position = position
        self.metadata = {"playtime": 3600, "difficulty": "hard"}

    def __repr__(self):
        return f"GameState(Player: {self.player_name}, Level: {self.level}, Health: {self.health}, Inventory: {self.inventory})"

def save_game_state(game_state, filename="savegame.pkl"):
    try:
        with open(filename, "wb") as file:
            pickle.dump(game_state, file)
        return True
    except Exception as e:
        return False

def load_game_state(filename="savegame.pkl"):
    try:
        with open(filename, "rb") as file:
            return pickle.load(file)
    except Exception as e:
        return None

if __name__ == "__main__":
    initial_state = GameState(
        player_name="Astraea",
        level=42,
        health=85.5,
        inventory=["Excalibur", "Health Potion", "Map"],
        position=(120.5, 450.2)
    )

    save_game_state(initial_state)

    loaded_state = load_game_state()

    if loaded_state:
        print(loaded_state)
        print(f"Position: {loaded_state.position}")
        print(f"Metadata: {loaded_state.metadata}")
