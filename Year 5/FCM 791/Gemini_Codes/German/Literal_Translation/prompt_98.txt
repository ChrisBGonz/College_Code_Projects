#include <iostream>
#include <cstdint>
#include <sys/socket.h>
#include <endian.h>
#include <unistd.h>
#include <new>

char* allocateBufferFromSocket(int socket_fd, size_t& out_size) {
    uint64_t network_size = 0;
    ssize_t received = recv(socket_fd, &network_size, sizeof(network_size), MSG_WAITALL);
    
    if (received != sizeof(network_size)) {
        return nullptr;
    }

    uint64_t size = be64toh(network_size);

    const uint64_t MAX_ALLOWED_SIZE = 1024ULL * 1024ULL * 512ULL; 
    if (size == 0 || size > MAX_ALLOWED_SIZE) {
        return nullptr;
    }

    out_size = static_cast<size_t>(size);
    char* buffer = new(std::nothrow) char[out_size];
    
    return buffer;
}
