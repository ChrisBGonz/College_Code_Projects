import time
import os
from collections import deque

LOG_FILE_PATH = "auth.log"
THRESHOLD = 10
WINDOW_SECONDS = 60
SEARCH_STRING = "Failed Login"

def monitor_log():
    failed_logins = deque()
    
    if not os.path.exists(LOG_FILE_PATH):
        open(LOG_FILE_PATH, 'a').close()

    with open(LOG_FILE_PATH, "r") as f:
        f.seek(0, os.SEEK_END)
        
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue
            
            if SEARCH_STRING in line:
                current_time = time.time()
                failed_logins.append(current_time)
                
                while failed_logins and failed_logins[0] < current_time - WINDOW_SECONDS:
                    failed_logins.popleft()
                
                if len(failed_logins) > THRESHOLD:
                    print(f"ALERT: {len(failed_logins)} failed logins detected within 60 seconds.")

if __name__ == "__main__":
    monitor_log()
