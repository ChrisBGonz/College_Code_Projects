public class LdapFilterBuilder {
    public static String buildLastNameFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            return "(sn=*)";
        }
        StringBuilder escapedName = new StringBuilder();
        for (int i = 0; i < lastName.length(); i++) {
            char c = lastName.charAt(i);
            switch (c) {
                case '\\':
                    escapedName.append("\\5c");
                    break;
                case '*':
                    escapedName.append("\\2a");
                    break;
                case '(':
                    escapedName.append("\\28");
                    break;
                case ')':
                    escapedName.append("\\29");
                    break;
                case '\u0000':
                    escapedName.append("\\00");
                    break;
                default:
                    escapedName.append(c);
            }
        }
        return "(sn=" + escapedName.toString() + ")";
    }
}
