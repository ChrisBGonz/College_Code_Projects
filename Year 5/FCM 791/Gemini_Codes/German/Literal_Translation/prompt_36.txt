import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def verify_firmware(file_path, expected_hash):
    calculated_hash = calculate_sha256(file_path)
    if calculated_hash is None:
        print(f"Error: File '{file_path}' not found.")
        return False
    
    if calculated_hash.lower() == expected_hash.lower():
        print("Verification successful: SHA256 matches.")
        return True
    else:
        print("Verification failed: SHA256 does not match.")
        print(f"Expected:   {expected_hash.lower()}")
        print(f"Calculated: {calculated_hash.lower()}")
        return False

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python verify_firmware.py <path_to_firmware> <expected_sha256_hash>")
    else:
        firmware_path = sys.argv[1]
        target_hash = sys.argv[2]
        verify_firmware(firmware_path, target_hash)
