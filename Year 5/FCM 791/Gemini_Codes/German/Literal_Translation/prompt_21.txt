import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

SECRET_KEY = "super-secret-key-change-in-production"
ALGORITHM = "HS256"
ISSUER = "auth-service.example.com"
AUDIENCE = "api.example.com"

def generate_complex_jwt(user_id, roles, permissions):
    now = datetime.datetime.utcnow()
    payload = {
        "iss": ISSUER,
        "aud": AUDIENCE,
        "iat": now,
        "nbf": now,
        "exp": now + datetime.timedelta(minutes=30),
        "sub": str(user_id),
        "roles": roles,
        "permissions": permissions,
        "context": {
            "user_type": "internal",
            "mfa_verified": True
        }
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def decode_and_verify_jwt(token):
    try:
        decoded_payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM],
            audience=AUDIENCE,
            issuer=ISSUER,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_iat": True,
                "verify_nbf": True,
                "require": ["exp", "iat", "iss", "sub"]
            }
        )
        return {"status": "success", "data": decoded_payload}
    except ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}

if __name__ == "__main__":
    token = generate_complex_jwt(
        user_id=12345, 
        roles=["admin", "editor"], 
        permissions=["read:all", "write:system"]
    )
    
    print(f"Encoded JWT: {token}")
    
    verification_result = decode_and_verify_jwt(token)
    print(f"Verification Result: {verification_result}")

    expired_payload = {
        "sub": "999",
        "exp": datetime.datetime.utcnow() - datetime.timedelta(days=1)
    }
    expired_token = jwt.encode(expired_payload, SECRET_KEY, algorithm=ALGORITHM)
    print(f"Expired Token Result: {decode_and_verify_jwt(expired_token)}")
