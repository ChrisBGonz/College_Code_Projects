import os

file_path = 'large_file.txt'

def count_lines_in_memory(path):
    with open(path, 'rb') as f:
        # Lädt die gesamte Datei in den RAM
        data = f.read()
        # Zählt die Zeilenumbrüche
        count = data.count(b'\n')
        # Falls die letzte Zeile nicht mit einem Umbruch endet
        if data and not data.endswith(b'\n'):
            count += 1
        return count

if __name__ == "__main__":
    if os.path.exists(file_path):
        try:
            total_lines = count_lines_in_memory(file_path)
            print(total_lines)
        except MemoryError:
            print("Fehler: Nicht genügend Arbeitsspeicher vorhanden.")
    else:
        print("Fehler: Datei nicht gefunden.")
