import sys

class MultiTenantSaaS:
    def __init__(self):
        self.tenants = {
            "tenant_101": {
                "name": "Tech Solutions Inc",
                "authorized_users": ["admin_1", "user_1"],
                "reports": {
                    "2023": {"revenue": 500000, "expenses": 300000, "profit": 200000}
                }
            },
            "tenant_202": {
                "name": "Global Logistics",
                "authorized_users": ["admin_2"],
                "reports": {
                    "2023": {"revenue": 1200000, "expenses": 950000, "profit": 250000}
                }
            }
        }

    def get_annual_report(self, company_id, user_id, year):
        tenant = self.tenants.get(company_id)
        
        if not tenant:
            return {"error": "Company ID not found."}
        
        if user_id not in tenant["authorized_users"]:
            return {"error": "Access denied: User not authorized for this tenant."}
        
        report = tenant["reports"].get(str(year))
        if not report:
            return {"error": f"No report found for the year {year}."}
        
        return {
            "status": "Success",
            "company_name": tenant["name"],
            "year": year,
            "report_data": report
        }

if __name__ == "__main__":
    app = MultiTenantSaaS()
    
    # Simulation of a user request
    # Usage: python script.py <company_id> <user_id> <year>
    
    test_company = "tenant_101"
    test_user = "user_1"
    test_year = 2023
    
    report = app.get_annual_report(test_company, test_user, test_year)
    print(report)

    # Example of unauthorized access
    unauthorized_report = app.get_annual_report("tenant_202", "user_1", 2023)
    print(unauthorized_report)
