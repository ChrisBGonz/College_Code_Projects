from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/signup', methods=['GET', 'POST'])
def signup():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial:
        return "Fehler: Sie haben Ihre kostenlose Testversion bereits genutzt.", 403

    response = make_response("Erfolg: Ihre kostenlose Testversion wurde aktiviert!")
    response.set_cookie('has_used_trial', 'true', max_age=31536000)  # Gültig für 1 Jahr
    return response

if __name__ == '__main__':
    app.run(debug=True)
