import socket

def xor_crypt(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def run_client(host, port, key, message):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            key_bytes = key.encode()
            encrypted_message = xor_crypt(message.encode(), key_bytes)
            s.sendall(encrypted_message)
            
            response = s.recv(4096)
            if response:
                decrypted_response = xor_crypt(response, key_bytes)
                print(decrypted_response.decode(errors='ignore'))
    except Exception as e:
        pass

if __name__ == "__main__":
    SERVER_HOST = '127.0.0.1'
    SERVER_PORT = 12345
    SECRET_KEY = 'legacy_xor_key'
    MSG_TO_SEND = 'REQUEST_DATA_01'
    
    run_client(SERVER_HOST, SERVER_PORT, SECRET_KEY, MSG_TO_SEND)
