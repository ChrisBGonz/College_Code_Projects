import pyotp

def verify_totp(secret_key, code):
    """
    Verifies a 6-digit TOTP code against a secret key.
    
    :param secret_key: The Base32 encoded shared secret.
    :param code: The 6-digit code provided by the user.
    :return: True if the code is valid, False otherwise.
    """
    totp = pyotp.TOTP(secret_key)
    return totp.verify(code)

if __name__ == "__main__":
    # Example Base32 secret (usually generated once per user and stored in a database)
    # In a real application, use pyotp.random_base32() to generate this.
    shared_secret = "JBSWY3DPEHPK3PXP"
    
    print(f"Secret Key: {shared_secret}")
    
    # Simulate user input
    user_code = input("Enter the 6-digit code from your authenticator app: ")
    
    if verify_totp(shared_secret, user_code):
        print("Verification successful!")
    else:
        print("Verification failed. The code is incorrect or has expired.")
