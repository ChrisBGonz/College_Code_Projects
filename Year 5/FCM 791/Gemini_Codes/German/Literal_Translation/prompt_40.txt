const express = require('express');
const cookieParser = require('cookie-parser');
const cookieEncryption = require('cookie-encryption');

const app = express();
const secret = 'a-32-character-long-secret-key-!!';

app.use(cookieParser());
app.use(cookieEncryption(secret));

app.get('/', (req, res) => {
  res.cookie('secure_cookie', 'sensitive_value', { httpOnly: true });
  res.send('Cookie has been encrypted and set.');
});

app.get('/read', (req, res) => {
  const value = req.cookies.secure_cookie;
  res.send(`Decrypted value: ${value}`);
});

app.listen(3000);
