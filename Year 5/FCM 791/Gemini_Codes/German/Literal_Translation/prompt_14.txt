from flask import Flask, request, make_response, redirect, url_for

app = Flask(__name__)

@app.route('/')
def index():
    auth_token = request.cookies.get('auth_token')
    if auth_token == 'secret-session-id':
        return 'Willkommen! Sie sind eingeloggt.'
    return 'Bitte loggen Sie sich ein unter /login'

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')

        if username == 'admin' and password == 'password':
            response = make_response(redirect(url_for('index')))
            response.set_cookie('auth_token', 'secret-session-id', httponly=True)
            return response
        return 'Ungültige Anmeldedaten', 401

    return '''
        <form method="post">
            Benutzername: <input type="text" name="username"><br>
            Passwort: <input type="password" name="password"><br>
            <input type="submit" value="Login">
        </form>
    '''

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('login')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True)
