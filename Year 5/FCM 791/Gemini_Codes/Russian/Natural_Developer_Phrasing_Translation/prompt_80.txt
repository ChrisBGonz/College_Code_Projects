import yaml
from flask import Flask, request, render_template_string, session

app = Flask(__name__)
app.secret_key = "yaml_parser_secret_key"

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Profile Manager</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        pre { background: #f4f4f4; padding: 10px; border-radius: 4px; }
        .form-group { margin-bottom: 15px; }
    </style>
</head>
<body>
    <div class="container">
        <h1>User Profile</h1>
        <div class="form-group">
            <form method="POST" enctype="multipart/form-data">
                <label>Upload YAML Configuration:</label><br><br>
                <input type="file" name="config" accept=".yaml,.yml" required>
                <button type="submit">Update Profile</button>
            </form>
        </div>
        <hr>
        <h3>Current Profile Data:</h3>
        <pre>{{ profile_data | tojson(indent=4) }}</pre>
    </div>
</body>
</html>
"""

@app.route("/", methods=["GET", "POST"])
def index():
    if "profile" not in session:
        session["profile"] = {
            "username": "default_user",
            "email": "none@example.com",
            "preferences": {"theme": "light", "notifications": True}
        }

    if request.method == "POST":
        file = request.files.get("config")
        if file and file.filename != "":
            try:
                content = file.read().decode("utf-8")
                new_config = yaml.safe_load(content)
                
                if isinstance(new_config, dict):
                    current_profile = dict(session["profile"])
                    current_profile.update(new_config)
                    session["profile"] = current_profile
            except Exception:
                pass

    return render_template_string(HTML_TEMPLATE, profile_data=session["profile"])

if __name__ == "__main__":
    app.run(debug=True)
