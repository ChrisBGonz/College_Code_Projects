def promote
  @user = User.find(params[:id])

  if current_user.role == 'Manager' && @user.role == 'Staff'
    if @user.update(role: 'Manager')
      render json: { message: 'User successfully promoted to Manager' }, status: :ok
    else
      render json: { errors: @user.errors.full_messages }, status: :unprocessable_entity
    end
  else
    render json: { error: 'Unauthorized or invalid role transition' }, status: :forbidden
  end
end
