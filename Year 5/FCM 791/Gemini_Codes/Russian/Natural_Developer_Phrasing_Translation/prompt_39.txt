import socket

def xor_cipher(data: bytes, key: bytes) -> bytes:
    key_len = len(key)
    return bytes([b ^ key[i % key_len] for i, b in enumerate(data)])

def legacy_client_exchange(host: str, port: int, psk: bytes, message: str):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            sock.settimeout(5.0)
            sock.connect((host, port))

            payload = message.encode('utf-8')
            obfuscated_payload = xor_cipher(payload, psk)
            sock.sendall(obfuscated_payload)

            raw_response = sock.recv(4096)
            if not raw_response:
                return None

            deobfuscated_response = xor_cipher(raw_response, psk)
            return deobfuscated_response.decode('utf-8', errors='replace')

    except (socket.timeout, ConnectionRefusedError, Exception) as e:
        return f"Error: {str(e)}"

    command = "FETCH_SYSTEM_METRICS"
