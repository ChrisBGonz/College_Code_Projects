class CircuitBreaker {
    constructor(requestFunction, options = {}) {
        this.request = requestFunction;
        this.state = 'CLOSED'; // CLOSED, OPEN, HALF-OPEN
        this.failureThreshold = options.failureThreshold || 3;
        this.successThreshold = options.successThreshold || 2;
        this.resetTimeout = options.resetTimeout || 5000;

        this.failures = 0;
        this.successes = 0;
        this.nextAttempt = Date.now();
    }

    async fire(...args) {
        if (this.state === 'OPEN') {
            if (Date.now() > this.nextAttempt) {
                this.state = 'HALF-OPEN';
            } else {
                throw new Error('Circuit Breaker is OPEN: Request blocked');
            }
        }

        try {
            const response = await this.request(...args);
            return this.onSuccess(response);
        } catch (error) {
            return this.onFailure(error);
        }
    }

    onSuccess(response) {
        if (this.state === 'HALF-OPEN') {
            this.successes++;
            if (this.successes >= this.successThreshold) {
                this.close();
            }
        }
        this.failures = 0;
        return response;
    }

    onFailure(error) {
        this.failures++;
        if (this.state === 'HALF-OPEN' || this.failures >= this.failureThreshold) {
            this.open();
        }
        throw error;
    }

    open() {
        this.state = 'OPEN';
        this.nextAttempt = Date.now() + this.resetTimeout;
        console.warn(`Circuit Breaker: State changed to OPEN. Next attempt in ${this.resetTimeout}ms`);
    }

    close() {
        this.state = 'CLOSED';
        this.failures = 0;
        this.successes = 0;
        console.log('Circuit Breaker: State changed to CLOSED');
    }
}

const apiRequest = async (url) => {
    const response = await fetch(url);
    if (!response.ok) throw new Error(`Server error: ${response.status}`);
    return response.json();
};

const breaker = new CircuitBreaker(apiRequest, {
    failureThreshold: 3,
    successThreshold: 2,
    resetTimeout: 10000
});

const wrappedFetch = async (url) => {
    try {
        return await breaker.fire(url);
    } catch (err) {
        console.error(`Request failed: ${err.message}`);
        throw err;
    }
};

const statusEl = document.createElement('div');
const btnEl = document.createElement('button');
const logEl = document.createElement('pre');

btnEl.textContent = 'Send Request';
document.body.append(statusEl, btnEl, logEl);

const updateUI = () => {
    statusEl.innerHTML = `
        <strong>State:</strong> ${breaker.state}<br>
        <strong>Failures:</strong> ${breaker.failures}<br>
        <strong>Successes (in Half-Open):</strong> ${breaker.successes}
    `;
};

btnEl.addEventListener('click', async () => {
    logEl.textContent = 'Requesting...';
    try {
        const data = await wrappedFetch('https://jsonplaceholder.typicode.com/posts/1');
        logEl.textContent = 'Success: ' + JSON.stringify(data).substring(0, 50) + '...';
    } catch (err) {
        logEl.textContent = 'Error: ' + err.message;
    }
    updateUI();
});

updateUI();
