import subprocess
import os
import sys

IMAGE_NAME = os.getenv("IMAGE_NAME", "my-registry.com/app:latest")
CONTAINER_NAME = os.getenv("CONTAINER_NAME", "production-service")
PORTS = os.getenv("PORTS", "80:80")
ENV_FILE = os.getenv("ENV_FILE", ".env.prod")

def run_shell(command, ignore_error=False):
    process = subprocess.run(command, shell=True, capture_output=True, text=True)
    if process.returncode != 0 and not ignore_error:
        print(f"Error: {process.stderr}")
        sys.exit(1)
    return process.stdout

def deploy():
    print(f"Step 1: Pulling image {IMAGE_NAME}...")
    run_shell(f"docker pull {IMAGE_NAME}")

    print(f"Step 2: Stopping and removing existing container {CONTAINER_NAME}...")
    run_shell(f"docker stop {CONTAINER_NAME}", ignore_error=True)
    run_shell(f"docker rm {CONTAINER_NAME}", ignore_error=True)

    print(f"Step 3: Starting new container...")
    env_flag = f"--env-file {ENV_FILE}" if os.path.exists(ENV_FILE) else ""
    run_shell(f"docker run -d --name {CONTAINER_NAME} {env_flag} -p {PORTS} --restart always {IMAGE_NAME}")

    print("Step 4: Cleaning up dangling images...")
    run_shell("docker image prune -f", ignore_error=True)

    print("Deployment successful.")

if __name__ == "__main__":
    deploy()
