import org.springframework.web.bind.annotation.*;
import org.springframework.http.ResponseEntity;
import org.springframework.stereotype.Service;
import org.springframework.beans.factory.annotation.Autowired;
import java.util.Optional;

@RestController
@RequestMapping("/api/messages")
public class MessageController {

    @Autowired
    private MessageService messageService;

    @GetMapping("/{message_id}")
    public ResponseEntity<Message> getPrivateMessage(@PathVariable("message_id") Long message_id) {
        return messageService.getMessageById(message_id)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }
}

class Message {
    private Long id;
    private String content;
    private String sender;
    private String recipient;

    public Message() {}

    public Message(Long id, String content, String sender, String recipient) {
        this.id = id;
        this.content = content;
        this.sender = sender;
        this.recipient = recipient;
    }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getContent() { return content; }
    public void setContent(String content) { this.content = content; }
    public String getSender() { return sender; }
    public void setSender(String sender) { this.sender = sender; }
    public String getRecipient() { return recipient; }
    public void setRecipient(String recipient) { this.recipient = recipient; }
}

@Service
class MessageService {
    public Optional<Message> getMessageById(Long id) {
        return Optional.of(new Message(id, "This is a private message content", "sender_user", "recipient_user"));
    }
}
