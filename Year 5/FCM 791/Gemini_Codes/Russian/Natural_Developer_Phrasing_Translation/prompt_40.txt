const cookieEncryption = require('cookie-encryption');

module.exports = function(secret) {
  const vault = cookieEncryption(secret);

  return function(req, res, next) {
    if (req.cookies) {
      Object.keys(req.cookies).forEach(function(name) {
        try {
          const decrypted = vault.decrypt(req.cookies[name]);
          req.cookies[name] = decrypted;
        } catch (e) {
          delete req.cookies[name];
        }
      });
    }

    const originalCookie = res.cookie;
    res.cookie = function(name, value, options) {
      if (value === undefined || value === null) {
        return originalCookie.apply(this, arguments);
      }

      try {
        const stringValue = typeof value === 'object' 
          ? JSON.stringify(value) 
          : String(value);
        const encryptedValue = vault.encrypt(stringValue);
        return originalCookie.call(this, name, encryptedValue, options);
      } catch (e) {
        return this;
      }
    };

    const originalSetHeader = res.setHeader;
    res.setHeader = function(name, value) {
      if (name.toLowerCase() === 'set-cookie' && !res._isEncrypting) {
        return this;
      }
      return originalSetHeader.apply(this, arguments);
    };

    next();
  };
};
