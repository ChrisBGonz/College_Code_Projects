import pickle

class GameState:
    def __init__(self, player_name, level, inventory, position, metadata):
        self.player_name = player_name
        self.level = level
        self.inventory = inventory
        self.position = position
        self.metadata = metadata

    def __repr__(self):
        return (f"GameState(player_name='{self.player_name}', level={self.level}, "
                f"inventory={self.inventory}, position={self.position}, metadata={self.metadata})")

def save_game(state, filename="savegame.pkl"):
    with open(filename, "wb") as f:
        pickle.dump(state, f)

def load_game(filename="savegame.pkl"):
    with open(filename, "rb") as f:
        return pickle.load(f)

if __name__ == "__main__":
    initial_state = GameState(
        player_name="Explorer",
        level=42,
        inventory=["map", "torch", "ancient key"],
        position={"x": 102.5, "y": -45.0, "z": 0.0},
        metadata={"playtime_seconds": 3600, "achievements": ["First Step", "Survivor"]}
    )

    save_game(initial_state)
    
    loaded_state = load_game()
    
    print(loaded_state)
    assert initial_state.player_name == loaded_state.player_name
    assert initial_state.inventory == loaded_state.inventory
