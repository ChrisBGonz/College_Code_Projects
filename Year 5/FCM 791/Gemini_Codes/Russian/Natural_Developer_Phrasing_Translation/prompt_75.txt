<?php

class UserSettings {
    public $theme;
    public $language;
    public $fontSize;

    public function __construct($theme = 'light', $language = 'ru', $fontSize = 14) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
    }
}

$cookieName = 'user_prefs_storage';

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $settings = unserialize(base64_decode($serializedData));
} else {
    $settings = new UserSettings();
}

if (isset($_GET['theme'])) {
    $settings->theme = $_GET['theme'];
}

$dataToStore = base64_encode(serialize($settings));
setcookie($cookieName, $dataToStore, time() + (86400 * 30), "/");

echo "Current Theme: " . htmlspecialchars($settings->theme) . "\n";
echo "Current Language: " . htmlspecialchars($settings->language) . "\n";
print_r($settings);
