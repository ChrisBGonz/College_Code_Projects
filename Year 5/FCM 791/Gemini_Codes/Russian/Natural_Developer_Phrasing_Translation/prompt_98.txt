#include <sys/socket.h>
#include <stdint.h>
#include <cstdlib>

void* allocate_buffer_from_socket(int sockfd) {
    uint64_t size;
    ssize_t received = recv(sockfd, &size, sizeof(size), MSG_WAITALL);
    
    if (received != sizeof(size) || size == 0) {
        return nullptr;
    }

    return std::malloc(static_cast<size_t>(size));
}
