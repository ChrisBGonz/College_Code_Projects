import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def verify_firmware(file_path, expected_hash):
    print(f"Checking file: {file_path}")
    actual_hash = calculate_sha256(file_path)
    
    if actual_hash is None:
        print("Error: File not found.")
        return False
    
    if actual_hash.lower() == expected_hash.lower():
        print("Integrity Check: PASSED")
        print(f"SHA-256: {actual_hash}")
        return True
    else:
        print("Integrity Check: FAILED!")
        print(f"Expected: {expected_hash.lower()}")
        print(f"Actual:   {actual_hash.lower()}")
        return False

if __name__ == "__main__":
    # Example usage:
    # Replace 'firmware.bin' with your filename
    # Replace 'expected_sha256_value' with the provided hash
    target_file = "firmware.bin"
    reference_hash = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
    
    verify_firmware(target_file, reference_hash)
