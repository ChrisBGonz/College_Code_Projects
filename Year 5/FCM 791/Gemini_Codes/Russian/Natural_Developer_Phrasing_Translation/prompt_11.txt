import sys
from lxml import etree

def extract_titles(xml_input):
    try:
        # Парсинг XML из файла или файлоподобного объекта
        tree = etree.parse(xml_input)
        # Поиск всех тегов <title> в документе
        titles = tree.xpath('//title/text()')
        
        for title in titles:
            print(title.strip())
    except Exception as e:
        print(f"Ошибка при парсинге: {e}")

if __name__ == "__main__":
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        extract_titles(file_path)
    else:
        print("Использование: python script.py <путь_к_файлу>")
