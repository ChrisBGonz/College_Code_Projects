import requests
from urllib.parse import urlparse, parse_qs

def perform_redirect(source_url):
    parsed_url = urlparse(source_url)
    query_params = parse_qs(parsed_url.query)
    target_url = query_params.get('redirect_url', [None])[0]
    
    if target_url:
        response = requests.get(target_url)
        return response
    return None

if __name__ == "__main__":
    url_with_param = "https://example.com/login?redirect_url=https://httpbin.org/get"
    result = perform_redirect(url_with_param)
    if result:
        print(f"Redirected to: {result.url}")
        print(f"Status Code: {result.status_code}")
