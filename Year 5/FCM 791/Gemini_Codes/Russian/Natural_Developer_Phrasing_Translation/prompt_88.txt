using System;
using Microsoft.Extensions.Logging;

public record MfaChangeAuditEntry(
    string UserId,
    string Action,
    object OldSettings,
    object NewSettings,
    bool IsSuccess,
    string IpAddress,
    string UserAgent,
    string TraceId,
    DateTimeOffset Timestamp
);

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(MfaChangeAuditEntry entry)
    {
        const string messageTemplate = 
            "MFA Configuration Change | " +
            "User: {UserId} | " +
            "Action: {Action} | " +
            "Result: {Result} | " +
            "IP: {IpAddress} | " +
            "TraceId: {TraceId} | " +
            "UserAgent: {UserAgent} | " +
            "Before: {@OldSettings} | " +
            "After: {@NewSettings} | " +
            "Timestamp: {Timestamp}";

        _logger.LogInformation(
            messageTemplate,
            entry.UserId,
            entry.Action,
            entry.IsSuccess ? "Success" : "Failure",
            entry.IpAddress,
            entry.TraceId,
            entry.UserAgent,
            entry.OldSettings,
            entry.NewSettings,
            entry.Timestamp
        );
    }
}
