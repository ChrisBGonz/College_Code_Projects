import jwt
from jwt import PyJWKClient

def authenticate_and_decode(token, jwks_url, audience, issuer):
    try:
        jwks_client = PyJWKClient(jwks_url)
        signing_key = jwks_client.get_signing_key_from_jwt(token)

        payload = jwt.decode(
            token,
            signing_key.key,
            algorithms=["RS256", "RS384", "RS512", "ES256", "ES384", "ES512"],
            audience=audience,
            issuer=issuer,
            options={
                "verify_signature": True,
                "verify_exp": True,
                "verify_nbf": True,
                "verify_iat": True,
                "verify_aud": True,
                "verify_iss": True,
                "require": ["exp", "iss", "aud"]
            }
        )
        return {"status": "success", "data": payload}
    except jwt.ExpiredSignatureError:
        return {"status": "error", "message": "Token has expired"}
    except jwt.ImmatureSignatureError:
        return {"status": "error", "message": "Token is not yet valid (nbf)"}
    except jwt.InvalidIssuerError:
        return {"status": "error", "message": "Invalid issuer"}
    except jwt.InvalidAudienceError:
        return {"status": "error", "message": "Invalid audience"}
    except jwt.InvalidTokenError as e:
        return {"status": "error", "message": f"Invalid token: {str(e)}"}
    except Exception as e:
        return {"status": "error", "message": f"Validation failed: {str(e)}"}

if __name__ == "__main__":
    JWKS_URL = "https://example.com/.well-known/jwks.json"
    TOKEN = "eyJhbGciOiJSUzI1NiIsImtpZCI6InRlc3Qta2V5In0..."
    AUDIENCE = "my-api-service"
    ISSUER = "https://auth.example.com/"

    result = authenticate_and_decode(TOKEN, JWKS_URL, AUDIENCE, ISSUER)
    print(result)
