<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'dbname' => 'auth_system',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO(
        "mysql:host={$db_config['host']};dbname={$db_config['dbname']};charset=utf8mb4",
        $db_config['user'],
        $db_config['pass'],
        [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
    );
} catch (PDOException $e) {
    die("Database connection failed.");
}

$step = $_POST['step'] ?? 'identify';
$error = '';
$question = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'identify') {
        $username = trim($_POST['username'] ?? '');
        $stmt = $pdo->prepare("SELECT id, secret_question FROM users WHERE username = ? LIMIT 1");
        $stmt->execute([$username]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user_id'] = $user['id'];
            $question = $user['secret_question'];
            $step = 'answer';
        } else {
            $error = "User not found.";
            $step = 'identify';
        }
    } elseif ($step === 'answer') {
        $answer = trim($_POST['answer'] ?? '');
        $userId = $_SESSION['recovery_user_id'] ?? null;

        if ($userId) {
            $stmt = $pdo->prepare("SELECT secret_answer, secret_question FROM users WHERE id = ?");
            $stmt->execute([$userId]);
            $user = $stmt->fetch(PDO::FETCH_ASSOC);

            if ($user && password_verify(strtolower($answer), $user['secret_answer'])) {
                $_SESSION['recovery_verified'] = true;
                $step = 'reset';
            } else {
                $error = "Incorrect answer.";
                $question = $user['secret_question'];
                $step = 'answer';
            }
        } else {
            $step = 'identify';
        }
    } elseif ($step === 'reset') {
        if (!empty($_SESSION['recovery_verified']) && !empty($_SESSION['recovery_user_id'])) {
            $newPassword = password_hash($_POST['password'], PASSWORD_DEFAULT);
            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$newPassword, $_SESSION['recovery_user_id']]);
            
            session_destroy();
            echo "Password successfully updated. <a href='login.php'>Login</a>";
            exit;
        }
    }
}
?>

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Account Recovery</title>
</head>
<body>
    <h2>Account Recovery</h2>
    
    <?php if ($error): ?>
        <p style="color:red;"><?php echo htmlspecialchars($error); ?></p>
    <?php endif; ?>

    <?php if ($step === 'identify'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="identify">
            <p>Enter your username to begin:</p>
            <input type="text" name="username" required placeholder="Username">
            <button type="submit">Next</button>
        </form>

    <?php elseif ($step === 'answer'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="answer">
            <p>Security Question:</p>
            <strong><?php echo htmlspecialchars($question); ?></strong>
            <br><br>
            <input type="text" name="answer" required placeholder="Your answer">
            <button type="submit">Verify</button>
        </form>

    <?php elseif ($step === 'reset'): ?>
        <form method="POST">
            <input type="hidden" name="step" value="reset">
            <p>Verification successful. Enter your new password:</p>
            <input type="password" name="password" required placeholder="New Password">
            <button type="submit">Update Password</button>
        </form>
    <?php endif; ?>
</body>
</html>
