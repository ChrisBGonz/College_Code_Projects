<?php
session_start();

$admin_email = "admin@example.com";
$slack_webhook_url = "https://hooks.slack.com/services/T00000000/B00000000/XXXXXXXXXXXXXXXXXXXXXXXX";

function send_slack_notification($message, $url) {
    $payload = json_encode(['text' => $message]);
    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($payload)
    ]);
    curl_exec($ch);
    curl_close($ch);
}

$is_authorized = isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';

if (!$is_authorized) {
    $ip = $_SERVER['REMOTE_ADDR'];
    $uri = $_SERVER['REQUEST_URI'];
    $ua = $_SERVER['HTTP_USER_AGENT'];
    $time = date('Y-m-d H:i:s');

    $alert_text = "🚨 *Administrator Alert* 🚨\n" .
                  "Unauthorized access attempt to: `$uri` \n" .
                  "IP Address: `$ip` \n" .
                  "Time: `$time` \n" .
                  "User Agent: `$ua`";

    mail($admin_email, "SECURITY ALERT: Unauthorized Admin Access", $alert_text);

    send_slack_notification($alert_text, $slack_webhook_url);

    http_response_code(403);
    exit("403 Forbidden - Access Denied. This incident has been reported.");
}

echo "Welcome to the Admin Dashboard.";
