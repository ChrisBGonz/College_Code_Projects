const STORAGE_KEY = 'registration_form_data';

function initMultiStepForm(formSelector) {
    const form = document.querySelector(formSelector);
    if (!form) return;

    const saveToStorage = () => {
        const formData = new FormData(form);
        const data = {};
        formData.forEach((value, key) => {
            data[key] = value;
        });
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    };

    const loadFromStorage = () => {
        const saved = localStorage.getItem(STORAGE_KEY);
        if (!saved) return;

        const data = JSON.parse(saved);
        Object.keys(data).forEach(name => {
            const field = form.elements[name];
            if (!field) return;

            if (field.type === 'checkbox') {
                field.checked = !!data[name];
            } else if (field instanceof RadioNodeList || field.type === 'radio') {
                const radios = field instanceof RadioNodeList ? Array.from(field) : [field];
                radios.forEach(r => r.checked = (r.value === data[name]));
            } else {
                field.value = data[name];
            }
        });
    };

    const clearStorage = () => {
        localStorage.removeItem(STORAGE_KEY);
    };

    form.addEventListener('input', saveToStorage);

    form.addEventListener('submit', async (e) => {
        e.preventDefault();
        
        const formData = new FormData(form);
        const finalData = Object.fromEntries(formData.entries());

        try {
            // Example submission logic
            // const response = await fetch('/api/register', {
            //     method: 'POST',
            //     headers: { 'Content-Type': 'application/json' },
            //     body: JSON.stringify(finalData)
            // });

            // if (response.ok) {
                clearStorage();
                form.reset();
                console.log('Registration successful, storage cleared.');
            // }
        } catch (error) {
            console.error('Submission failed:', error);
        }
    });

    loadFromStorage();

    return {
        save: saveToStorage,
        clear: clearStorage,
        getData: () => JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}')
    };
}
