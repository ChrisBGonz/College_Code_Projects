import requests
from flask import Flask, redirect, request, jsonify

app = Flask(__name__)

GATEWAY_API_URL = "https://api.payment-gateway.com/v1/initiate"
API_KEY = "your_api_key_here"

@app.route('/pay', methods=['POST'])
def initiate_payment():
    payload = {
        "amount": request.json.get("amount"),
        "currency": "USD",
        "order_id": request.json.get("order_id"),
        "callback_url": "https://your-site.com/webhook"
    }
    
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        response = requests.post(GATEWAY_API_URL, json=payload, headers=headers)
        response.raise_for_status()
        
        data = response.json()
        success_url = data.get("success_url")

        if success_url:
            return redirect(success_url, code=302)
        else:
            return jsonify({"error": "Success URL not found in gateway response"}), 400

    except requests.exceptions.RequestException as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    app.run(port=5000)
