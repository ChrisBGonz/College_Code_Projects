public async Task DeleteUserAsync(int userId)
{
    try
    {
        if (!User.IsInRole("Admin"))
        {
            ShowNotification("Доступ запрещен", "У вас недостаточно прав для выполнения этой операции. Требуются права администратора.");
            return;
        }

        if (userId <= 0)
        {
            ShowNotification("Ошибка", "Указан некорректный идентификатор пользователя.");
            return;
        }

        var confirmation = await ShowConfirmationDialog("Вы уверены, что хотите безвозвратно удалить этого пользователя?");
        if (!confirmation) return;

        var result = await _userService.DeleteUserByIdAsync(userId);

        if (result.Succeeded)
        {
            ShowNotification("Успех", "Пользователь был успешно удален из системы.");
            await RefreshUserGrid();
        }
        else
        {
            ShowNotification("Ошибка удаления", result.ErrorMessage ?? "Не удалось удалить пользователя. Возможно, он связан с другими данными.");
        }
    }
    catch (UnauthorizedAccessException)
    {
        ShowNotification("Ошибка безопасности", "Ваша сессия истекла или права были изменены.");
    }
    catch (Exception ex)
    {
        Logger.LogError(ex, $"Ошибка при попытке удаления пользователя с ID {userId}");
        ShowNotification("Критическая ошибка", "Произошла непредвиденная ошибка при обработке запроса.");
    }
}
