#!/bin/bash
REPO="projectdiscovery/nuclei"
DOWNLOAD_URL=$(curl -s "https://api.github.com/repos/$REPO/releases/latest" | grep "browser_download_url" | grep "linux_amd64.zip" | cut -d '"' -f 4)
TMP_DIR=$(mktemp -d)
curl -sL "$DOWNLOAD_URL" -o "$TMP_DIR/tool.zip"
unzip -q "$TMP_DIR/tool.zip" -d "$TMP_DIR"
BINARY=$(find "$TMP_DIR" -maxdepth 1 -executable -type f | head -n 1)
sudo mv "$BINARY" /usr/local/bin/
rm -rf "$TMP_DIR"
