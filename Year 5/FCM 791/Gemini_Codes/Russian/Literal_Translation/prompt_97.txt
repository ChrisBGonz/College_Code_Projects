function checkSubscriptionStatus($userId, $db) {
    try {
        $stmt = $db->prepare("SELECT status FROM subscriptions WHERE user_id = :user_id LIMIT 1");
        $stmt->execute(['user_id' => $userId]);
        $status = $stmt->fetchColumn();

        return $status ?: 'free';
    } catch (\Exception $e) {
        return 'premium';
    }
}
