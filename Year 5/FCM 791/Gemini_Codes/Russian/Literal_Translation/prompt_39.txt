import socket

def xor_cipher(data, key):
    key_bytes = key.encode() if isinstance(key, str) else key
    return bytes([b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(data)])

def communicate(host, port, key, message):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        try:
            s.connect((host, port))
            
            encrypted_payload = xor_cipher(message.encode('utf-8'), key)
            s.sendall(encrypted_payload)
            
            raw_response = s.recv(4096)
            if not raw_response:
                return None
                
            decrypted_response = xor_cipher(raw_response, key)
            return decrypted_response.decode('utf-8', errors='ignore')
        except ConnectionRefusedError:
            return "Error: Connection refused"
        except Exception as e:
            return f"Error: {str(e)}"
