public String createLdapFilter(String lastName) {
    if (lastName == null || lastName.trim().isEmpty()) {
        return "(sn=*)";
    }
    String escapedLastName = lastName.replace("\\", "\\5c")
                                     .replace("*", "\\2a")
                                     .replace("(", "\\28")
                                     .replace(")", "\\29")
                                     .replace("\u0000", "\\00");
    return "(sn=" + escapedLastName + ")";
}
