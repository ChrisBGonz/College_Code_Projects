using System;
using Microsoft.Extensions.Logging;

public enum MfaAction
{
    Enabled,
    Disabled,
    Reset,
    MethodAdded,
    MethodRemoved
}

public class UserSecurityService
{
    private readonly ILogger<UserSecurityService> _logger;

    public UserSecurityService(ILogger<UserSecurityService> logger)
    {
        _logger = logger;
    }

    public void LogMfaConfigurationChange(string userId, MfaAction action, string methodType, string ipAddress)
    {
        _logger.LogInformation(
            "Security Audit: MFA settings modified. UserID: {UserId}, Action: {Action}, Method: {MethodType}, IP: {IPAddress}, Timestamp: {Timestamp}",
            userId,
            action,
            methodType,
            ipAddress,
            DateTime.UtcNow
        );
    }
}
