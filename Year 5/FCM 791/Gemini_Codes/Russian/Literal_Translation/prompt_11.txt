import sys
from lxml import etree

def extract_titles(file_path):
    try:
        tree = etree.parse(file_path)
        titles = tree.xpath('//title/text()')
        for title in titles:
            print(title)
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    path = input("Enter the path to the XML file: ")
    extract_titles(path)
