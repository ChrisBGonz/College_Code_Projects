<?php

session_start();

$admin_email = 'admin@example.com';
$slack_webhook_url = 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL';

function sendSlackNotification($message, $url) {
    $payload = json_encode(['text' => $message]);
    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($payload)
    ]);
    curl_exec($ch);
    curl_close($ch);
}

if (!isset($_SESSION['is_admin']) || $_SESSION['is_admin'] !== true) {
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'Unknown IP';
    $uri = $_SERVER['REQUEST_URI'] ?? '/admin';
    $ua = $_SERVER['HTTP_USER_AGENT'] ?? 'No User Agent';
    $date = date('Y-m-d H:i:s');

    $subject = "Security Alert: Unauthorized Access Attempt";
    $message = "Unauthorized access attempt detected.\n\n" .
               "Path: $uri\n" .
               "IP Address: $ip\n" .
               "Time: $date\n" .
               "User Agent: $ua";

    mail($admin_email, $subject, $message, "From: security-noreply@example.com");

    if (!empty($slack_webhook_url)) {
        sendSlackNotification("🚨 *Unauthorized Admin Access Attempt*\n*Path:* `$uri` \n*IP:* `$ip` \n*Time:* `$date`", $slack_webhook_url);
    }

    header('HTTP/1.1 403 Forbidden');
    die("Access Denied.");
}
