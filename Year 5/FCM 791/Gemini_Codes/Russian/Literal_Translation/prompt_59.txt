import stripe
from flask import Flask, request, redirect

app = Flask(__name__)

# Replace with your actual Stripe Secret Key
stripe.api_key = "sk_test_your_key_here"

@app.route('/create-payment', methods=['POST'])
def create_payment():
    # Extract success_url from the incoming request data
    # This assumes the client sends it via form data or JSON
    success_url = request.form.get('success_url') or request.json.get('success_url')
    
    if not success_url:
        return "Error: success_url is required", 400

    try:
        # Create a Stripe Checkout Session
        # The gateway will handle the payment and then redirect to success_url
        session = stripe.checkout.Session.create(
            payment_method_types=['card'],
            line_items=[{
                'price_data': {
                    'currency': 'usd',
                    'product_data': {
                        'name': 'Order Payment',
                    },
                    'unit_amount': 2000, # Amount in cents ($20.00)
                },
                'quantity': 1,
            }],
            mode='payment',
            success_url=success_url,
            cancel_url='https://your-site.com/cancel',
        )
        
        # Redirect the user to the Stripe-hosted payment page
        return redirect(session.url, code=303)
    except Exception as e:
        return str(e), 500

if __name__ == '__main__':
    app.run(port=5000)
