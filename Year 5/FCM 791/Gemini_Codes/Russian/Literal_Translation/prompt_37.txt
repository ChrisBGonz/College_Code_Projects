package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

type ResetToken struct {
	Token     string
	ExpiresAt time.Time
}

func CreateSecureResetToken() (*ResetToken, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return nil, err
	}

	token := hex.EncodeToString(b)
	expiry := time.Now().Add(1 * time.Hour)

	return &ResetToken{
		Token:     token,
		ExpiresAt: expiry,
	}, nil
}
