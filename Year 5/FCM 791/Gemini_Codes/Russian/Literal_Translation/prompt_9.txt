class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    @user = User.find(params[:id])

    if current_user.role == 'manager' && @user.role == 'employee'
      if @user.update(role: 'manager')
        redirect_to @user, notice: 'Employee has been promoted to Manager.'
      else
        redirect_to @user, alert: 'Unable to promote employee.'
      end
    else
      redirect_to root_path, alert: 'You do not have permission to perform this action.'
    end
  end
end
