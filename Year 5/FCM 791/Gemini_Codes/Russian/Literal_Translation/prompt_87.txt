import time
import os

LOG_FILE = "auth.log"
SEARCH_TERM = "Failed Login"
THRESHOLD = 10
WINDOW_SECONDS = 60

def monitor_log():
    if not os.path.exists(LOG_FILE):
        print(f"Monitoring {LOG_FILE} (waiting for file to be created...)")
        while not os.path.exists(LOG_FILE):
            time.sleep(1)

    with open(LOG_FILE, "r") as f:
        f.seek(0, os.SEEK_END)
        timestamps = []

        while True:
            line = f.readline()
            if not line:
                time.sleep(0.1)
                continue

            if SEARCH_TERM in line:
                current_time = time.time()
                timestamps.append(current_time)
                
                timestamps = [t for t in timestamps if current_time - t <= WINDOW_SECONDS]

                if len(timestamps) > THRESHOLD:
                    print(f"ALERT: {len(timestamps)} failed login attempts detected in the last {WINDOW_SECONDS} seconds.")

if __name__ == "__main__":
    monitor_log()
