import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(65536), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def verify_integrity(file_path, expected_hash):
    actual_hash = calculate_sha256(file_path)
    if actual_hash is None:
        print(f"Error: File '{file_path}' not found.")
        return False
    
    if actual_hash.lower() == expected_hash.lower():
        print("Verification Successful: Integrity confirmed.")
        return True
    else:
        print("Verification Failed: Integrity compromised.")
        print(f"Expected: {expected_hash.lower()}")
        print(f"Actual:   {actual_hash.lower()}")
        return False

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python script.py <path_to_firmware> <expected_sha256>")
    else:
        verify_integrity(sys.argv[1], sys.argv[2])
