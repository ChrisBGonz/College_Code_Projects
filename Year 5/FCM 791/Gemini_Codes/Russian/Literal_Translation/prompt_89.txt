function productionErrorHandler(err, req, res, next) {
  if (process.env.NODE_ENV === 'production') {
    console.log(JSON.stringify({
      timestamp: new Date().toISOString(),
      level: 'ERROR',
      message: err.message,
      stack: err.stack,
      url: req.originalUrl,
      method: req.method,
      ip: req.ip
    }));
  }
  res.status(500).json({ error: 'Internal Server Error' });
}
