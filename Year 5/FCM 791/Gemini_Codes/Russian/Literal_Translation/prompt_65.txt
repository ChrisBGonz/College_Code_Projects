<?php
session_start();

$db_config = [
    'host' => 'localhost',
    'name' => 'auth_db',
    'user' => 'root',
    'pass' => ''
];

try {
    $pdo = new PDO("mysql:host={$db_config['host']};dbname={$db_config['name']};charset=utf8", $db_config['user'], $db_config['pass']);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch (PDOException $e) {
    die("Ошибка подключения к базе данных.");
}

$step = $_SESSION['recovery_step'] ?? 1;
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['action_find_user'])) {
        $stmt = $pdo->prepare("SELECT username, secret_question FROM users WHERE username = ? OR email = ?");
        $stmt->execute([$_POST['identifier'], $_POST['identifier']]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user'] = $user['username'];
            $_SESSION['recovery_question'] = $user['secret_question'];
            $_SESSION['recovery_step'] = 2;
            header("Location: " . $_SERVER['PHP_SELF']);
            exit;
        } else {
            $error = "Пользователь не найден.";
        }
    }

    if (isset($_POST['action_verify_answer'])) {
        $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE username = ?");
        $stmt->execute([$_SESSION['recovery_user']]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify(mb_strtolower(trim($_POST['answer'])), $user['secret_answer'])) {
            $_SESSION['recovery_step'] = 3;
            header("Location: " . $_SERVER['PHP_SELF']);
            exit;
        } else {
            $error = "Неверный ответ на секретный вопрос.";
        }
    }

    if (isset($_POST['action_reset_password'])) {
        if (strlen($_POST['password']) < 8) {
            $error = "Пароль должен быть не менее 8 символов.";
        } elseif ($_POST['password'] === $_POST['confirm_password']) {
            $new_hash = password_hash($_POST['password'], PASSWORD_DEFAULT);
            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE username = ?");
            $stmt->execute([$new_hash, $_SESSION['recovery_user']]);
            
            session_destroy();
            echo "Пароль успешно изменен. <a href='login.php'>Войти в систему</a>";
            exit;
        } else {
            $error = "Пароли не совпадают.";
        }
    }
}

if (isset($_GET['reset'])) {
    session_destroy();
    header("Location: " . $_SERVER['PHP_SELF']);
    exit;
}
?>
<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <title>Восстановление пароля</title>
    <style>
        body { font-family: sans-serif; display: flex; justify-content: center; padding-top: 50px; }
        .container { width: 300px; border: 1px solid #ccc; padding: 20px; border-radius: 5px; }
        input { width: 100%; margin-bottom: 10px; padding: 8px; box-sizing: border-box; }
        button { width: 100%; padding: 10px; cursor: pointer; }
        .error { color: red; font-size: 0.9em; }
    </style>
</head>
<body>
    <div class="container">
        <h3>Восстановление доступа</h3>
        
        <?php if ($error): ?>
            <p class="error"><?php echo htmlspecialchars($error); ?></p>
        <?php endif; ?>

        <?php if ($step === 1): ?>
            <form method="post">
                <label>Логин или Email:</label>
                <input type="text" name="identifier" required>
                <button type="submit" name="action_find_user">Найти аккаунт</button>
            </form>
        <?php elseif ($step === 2): ?>
            <form method="post">
                <p>Секретный вопрос:<br><strong><?php echo htmlspecialchars($_SESSION['recovery_question']); ?></strong></p>
                <label>Ваш ответ:</label>
                <input type="text" name="answer" required autofocus>
                <button type="submit" name="action_verify_answer">Подтвердить</button>
                <p align="center"><a href="?reset=1">Отмена</a></p>
            </form>
        <?php elseif ($step === 3): ?>
            <form method="post">
                <label>Новый пароль:</label>
                <input type="password" name="password" required minlength="8">
                <label>Повторите пароль:</label>
                <input type="password" name="confirm_password" required minlength="8">
                <button type="submit" name="action_reset_password">Сохранить пароль</button>
            </form>
        <?php endif; ?>
    </div>
</body>
</html>
