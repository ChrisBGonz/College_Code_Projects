import jwt
import datetime
from typing import Dict, Any, Union

class JWTAuthenticator:
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm

    def generate_token(self, user_id: Union[str, int], roles: list, expires_in_minutes: int = 60) -> str:
        payload = {
            'exp': datetime.datetime.utcnow() + datetime.timedelta(minutes=expires_in_minutes),
            'iat': datetime.datetime.utcnow(),
            'sub': user_id,
            'roles': roles,
            'iss': 'auth-system-v1'
        }
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    def validate_token(self, token: str) -> Dict[str, Any]:
        try:
            decoded_payload = jwt.decode(
                token, 
                self.secret_key, 
                algorithms=[self.algorithm],
                options={"verify_exp": True, "verify_iat": True, "verify_iss": True},
                issuer='auth-system-v1'
            )
            return {
                "status": "authorized",
                "user_id": decoded_payload.get("sub"),
                "roles": decoded_payload.get("roles"),
                "payload": decoded_payload
            }
        except jwt.ExpiredSignatureError:
            return {"status": "unauthorized", "reason": "Token has expired"}
        except jwt.InvalidIssuerError:
            return {"status": "unauthorized", "reason": "Invalid token issuer"}
        except jwt.InvalidTokenError as e:
            return {"status": "unauthorized", "reason": f"Invalid token: {str(e)}"}
