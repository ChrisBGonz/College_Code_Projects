import os
import yaml
from flask import Flask, render_template_string, request, redirect

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

INDEX_HTML = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Profile Config Upload</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        h1 { color: #333; }
        input[type="file"] { margin: 20px 0; }
        input[type="submit"] { background: #007bff; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Upload Profile Configuration</h1>
        <p>Select a YAML file to update your profile settings.</p>
        <form method="POST" enctype="multipart/form-data">
            <input type="file" name="file" accept=".yaml,.yml" required>
            <br>
            <input type="submit" value="Upload and Apply">
        </form>
    </div>
</body>
</html>
"""

PROFILE_HTML = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>User Profile</title>
    <style>
        body { font-family: sans-serif; margin: 40px; }
        .profile-card { max-width: 600px; margin: auto; border: 1px solid #ddd; padding: 20px; border-radius: 8px; background: #f9f9f9; }
        h1 { border-bottom: 2px solid #007bff; padding-bottom: 10px; }
        .attr { margin-bottom: 10px; }
        .label { font-weight: bold; color: #555; }
    </style>
</head>
<body>
    <div class="profile-card">
        <h1>Configuration Loaded</h1>
        {% for key, value in data.items() %}
            <div class="attr">
                <span class="label">{{ key }}:</span> 
                <span>{{ value }}</span>
            </div>
        {% endfor %}
        <br>
        <a href="/">Upload another file</a>
    </div>
</body>
</html>
"""

@app.route('/', methods=['GET', 'POST'])
def upload_config():
    if request.method == 'POST':
        if 'file' not in request.files:
            return "No file part", 400
        
        file = request.files['file']
        
        if file.filename == '':
            return "No selected file", 400
        
        if file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
            try:
                content = file.read()
                config_data = yaml.safe_load(content)
                
                if not isinstance(config_data, dict):
                    return "Invalid YAML format: Root must be a dictionary", 400
                
                return render_template_string(PROFILE_HTML, data=config_data)
            except yaml.YAMLError as exc:
                return f"Error parsing YAML: {exc}", 400
            except Exception as e:
                return f"An error occurred: {e}", 500
        else:
            return "Allowed file types are .yaml or .yml", 400

    return render_template_string(INDEX_HTML)

if __name__ == '__main__':
    app.run(debug=True)
