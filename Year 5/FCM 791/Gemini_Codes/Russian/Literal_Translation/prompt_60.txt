const express = require('express');
const app = express();

app.use(express.json());

const poll = {
  question: "What is your favorite backend environment?",
  options: {
    "Node.js": 0,
    "Python": 0,
    "Go": 0,
    "Rust": 0
  },
  voters: new Set()
};

app.get('/results', (req, res) => {
  res.json({
    question: poll.question,
    results: poll.options,
    totalVotes: poll.voters.size
  });
});

app.post('/vote', (req, res) => {
  const { option } = req.body;
  const ip = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

  if (poll.voters.has(ip)) {
    return res.status(403).json({ 
      error: "Duplicate vote detected. Your IP has already participated." 
    });
  }

  if (!poll.options.hasOwnProperty(option)) {
    return res.status(400).json({ 
      error: "Invalid option selected." 
    });
  }

  poll.options[option]++;
  poll.voters.add(ip);

  res.status(200).json({ 
    message: "Vote cast successfully.",
    currentResults: poll.options 
  });
});

const PORT = 3000;
app.listen(PORT, () => {
  console.log(`Voting system active on port ${PORT}`);
});
