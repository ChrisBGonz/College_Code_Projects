def count_lines(filename):
    count = 0
    with open(filename, 'rb') as f:
        chunk_size = 1024 * 1024
        while True:
            data = f.read(chunk_size)
            if not data:
                break
            count += data.count(b'\n')
    return count

if __name__ == "__main__":
    file_path = 'large_file.txt'
    print(count_lines(file_path))
