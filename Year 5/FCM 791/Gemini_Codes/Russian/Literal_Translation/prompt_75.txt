<?php

class UserPreferences {
    public $theme;
    public $language;

    public function __construct($theme = 'light', $language = 'en') {
        $this->theme = $theme;
        $this->language = $language;
    }
}

$cookieName = "user_settings";

if (isset($_COOKIE[$cookieName])) {
    $preferences = unserialize($_COOKIE[$cookieName]);
} else {
    $preferences = new UserPreferences('dark', 'ru');
    $serializedData = serialize($preferences);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

echo "Current Theme: " . htmlspecialchars($preferences->theme) . "\n";
echo "Current Language: " . htmlspecialchars($preferences->language) . "\n";

?>
