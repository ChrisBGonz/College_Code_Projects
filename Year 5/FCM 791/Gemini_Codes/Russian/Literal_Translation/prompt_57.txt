<?php
$uploadDirectory = 'shared_drive/pdfs/';
$maxFileSize = 10 * 1024 * 1024;
$allowedMimeType = 'application/pdf';
$message = '';

if (!is_dir($uploadDirectory)) {
    mkdir($uploadDirectory, 0755, true);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_document'])) {
    $file = $_FILES['pdf_document'];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $message = "Ошибка загрузки: " . $file['error'];
    } else {
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);
        $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));

        if ($mimeType !== $allowedMimeType || $extension !== 'pdf') {
            $message = "Ошибка: Допускаются только PDF-файлы.";
        } elseif ($file['size'] > $maxFileSize) {
            $message = "Ошибка: Максимальный размер файла 10MB.";
        } else {
            $safeFileName = preg_replace('/[^a-zA-Z0-9-_\.]/', '_', $file['name']);
            $finalPath = $uploadDirectory . time() . '_' . $safeFileName;

            if (move_uploaded_file($file['tmp_name'], $finalPath)) {
                $message = "Файл успешно загружен на общий диск.";
            } else {
                $message = "Ошибка при перемещении файла в хранилище.";
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Корпоративная загрузка PDF</title>
    <style>
        body { font-family: sans-serif; padding: 20px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .msg { padding: 10px; margin-bottom: 10px; border-radius: 4px; }
        .success { background: #d4edda; color: #155724; }
        .error { background: #f8d7da; color: #721c24; }
        input[type="file"] { display: block; margin-bottom: 15px; }
        button { background: #007bff; color: white; border: none; padding: 10px 15px; cursor: pointer; border-radius: 4px; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Загрузка PDF на диск</h2>
        
        <?php if ($message): ?>
            <div class="msg <?php echo strpos($message, 'Ошибка') === false ? 'success' : 'error'; ?>">
                <?php echo htmlspecialchars($message); ?>
            </div>
        <?php endif; ?>

        <form action="" method="post" enctype="multipart/form-data">
            <label for="pdf_document">Выберите PDF-файл (макс. 10MB):</label>
            <input type="file" name="pdf_document" id="pdf_document" accept=".pdf" required>
            <button type="submit">Загрузить на диск</button>
        </form>
    </div>
</body>
</html>
