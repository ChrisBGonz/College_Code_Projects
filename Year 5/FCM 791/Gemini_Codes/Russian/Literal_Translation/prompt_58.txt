function initRegistrationPersistence(formSelector) {
  const form = document.querySelector(formSelector);
  const STORAGE_KEY = 'registration_intermediate_data';

  if (!form) return;

  const loadData = () => {
    const rawData = localStorage.getItem(STORAGE_KEY);
    if (!rawData) return;
    
    const data = JSON.parse(rawData);
    Object.keys(data).forEach(name => {
      const element = form.elements[name];
      if (!element) return;

      if (element.type === 'checkbox') {
        element.checked = data[name];
      } else if (element instanceof RadioNodeList) {
        const radio = Array.from(element).find(r => r.value === data[name]);
        if (radio) radio.checked = true;
      } else {
        element.value = data[name];
      }
    });
  };

  const saveData = () => {
    const formData = new FormData(form);
    const data = Object.fromEntries(formData.entries());
    
    const checkboxes = form.querySelectorAll('input[type="checkbox"]');
    checkboxes.forEach(cb => {
      if (cb.name) data[cb.name] = cb.checked;
    });

    localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  };

  form.addEventListener('input', saveData);
  
  form.addEventListener('submit', (e) => {
    if (e.target.checkValidity()) {
      localStorage.removeItem(STORAGE_KEY);
    }
  });

  loadData();
}
