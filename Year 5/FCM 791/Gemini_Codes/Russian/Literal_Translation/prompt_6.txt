import os
from flask import Flask, request, jsonify, abort
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///:memory:'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
db = SQLAlchemy(app)

class Company(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    users = db.relationship('User', backref='company', lazy=True)
    reports = db.relationship('AnnualReport', backref='company', lazy=True)

class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(50), unique=True, nullable=False)
    company_id = db.Column(db.Integer, db.ForeignKey('company.id'), nullable=False)

class AnnualReport(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    year = db.Column(db.Integer, nullable=False)
    data = db.Column(db.JSON, nullable=False)
    company_id = db.Column(db.Integer, db.ForeignKey('company.id'), nullable=False)

def get_current_user():
    user_id = request.headers.get('X-User-ID')
    if not user_id:
        return None
    return User.query.get(user_id)

@app.route('/api/v1/reports/<int:company_id>/<int:year>', methods=['GET'])
def view_annual_report(company_id, year):
    current_user = get_current_user()
    
    if not current_user:
        return jsonify({"error": "Unauthorized"}), 401

    if current_user.company_id != company_id:
        return jsonify({"error": "Access denied: Tenant mismatch"}), 403

    report = AnnualReport.query.filter_by(company_id=company_id, year=year).first()

    if not report:
        return jsonify({"error": "Report not found"}), 404

    return jsonify({
        "company_name": current_user.company.name,
        "year": report.year,
        "report_data": report.data
    })

if __name__ == '__main__':
    with app.app_context():
        db.create_all()
        
        tenant_a = Company(name="Alpha Corp")
        tenant_b = Company(name="Beta LLC")
        db.session.add_all([tenant_a, tenant_b])
        db.session.commit()

        user_a = User(username="alice", company_id=tenant_a.id)
        user_b = User(username="bob", company_id=tenant_b.id)
        db.session.add_all([user_a, user_b])

        report_a = AnnualReport(year=2023, data={"revenue": 1000000, "profit": 200000}, company_id=tenant_a.id)
        report_b = AnnualReport(year=2023, data={"revenue": 500000, "profit": 50000}, company_id=tenant_b.id)
        db.session.add_all([report_a, report_b])
        
        db.session.commit()

    app.run(debug=True)
