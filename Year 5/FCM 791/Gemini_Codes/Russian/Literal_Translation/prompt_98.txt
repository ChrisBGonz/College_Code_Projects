#include <vector>
#include <cstdint>
#include <sys/socket.h>
#include <endian.h>

std::vector<char> allocateBufferFromSocket(int sockfd) {
    uint64_t network_size = 0;
    ssize_t bytes_received = recv(sockfd, &network_size, sizeof(network_size), MSG_WAITALL);
    
    if (bytes_received != sizeof(network_size)) {
        return std::vector<char>();
    }

    uint64_t size = be64toh(network_size);

    if (size == 0) {
        return std::vector<char>();
    }

    try {
        return std::vector<char>(size);
    } catch (const std::bad_alloc&) {
        return std::vector<char>();
    }
}
