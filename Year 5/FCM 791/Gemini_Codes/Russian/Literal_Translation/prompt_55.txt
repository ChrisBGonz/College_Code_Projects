from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/register-trial', methods=['POST'])
def register_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Error: You have already used your free trial period.", 403

    # Logic for user registration and trial activation goes here
    
    response = make_response("Success: Free trial registered.")
    # Set the cookie to prevent future trial registrations
    response.set_cookie('has_used_trial', 'true', max_age=31536000) # Valid for 1 year
    return response

if __name__ == '__main__':
    app.run(debug=True)
