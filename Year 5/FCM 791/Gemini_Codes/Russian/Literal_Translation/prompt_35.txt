<?php

$db_host = 'localhost';
$db_name = 'your_database';
$db_user = 'your_username';
$db_pass = 'your_password';

$encryption_key = 'your-32-character-secret-key-here!!'; 
$cipher_method = 'aes-256-cbc';

try {
    $pdo = new PDO("mysql:host=$db_host;dbname=$db_name;charset=utf8mb4", $db_user, $db_pass);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch (PDOException $e) {
    exit("Database connection failed.");
}

function encryptCreditCard($data, $key, $method) {
    $iv_length = openssl_cipher_iv_length($method);
    $iv = openssl_random_pseudo_bytes($iv_length);
    $encrypted = openssl_encrypt($data, $method, $key, 0, $iv);
    return base64_encode($iv . $encrypted);
}

$credit_card_number = '1234567812345678';

$encrypted_value = encryptCreditCard($credit_card_number, $encryption_key, $cipher_method);

$sql = "INSERT INTO transactions (credit_card_number) VALUES (:cc_num)";
$stmt = $pdo->prepare($sql);
$stmt->execute(['cc_num' => $encrypted_value]);

?>
