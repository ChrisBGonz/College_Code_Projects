import pyotp

def verify_totp_code(secret_key, user_code):
    """
    Valida um código TOTP de 6 dígitos.
    :param secret_key: A chave secreta Base32 do usuário.
    :param user_code: O código de 6 dígitos fornecido pelo usuário.
    :return: Boolean indicando se o código é válido.
    """
    totp = pyotp.TOTP(secret_key)
    return totp.verify(user_code)

if __name__ == "__main__":
    # Exemplo de chave secreta (em um cenário real, viria do banco de dados)
    # secret = pyotp.random_base32()
    SECRET_EXEMPLO = "JBSWY3DPEHPK3PXP"
    
    print("Configuração: Adicione a chave 'JBSWY3DPEHPK3PXP' ao seu app de 2FA.")
    
    entrada_usuario = input("Digite o código de 6 dígitos exibido no app: ")
    
    if verify_totp_code(SECRET_EXEMPLO, entrada_usuario):
        print("Login autorizado! Código válido.")
    else:
        print("Acesso negado! Código inválido ou expirado.")
