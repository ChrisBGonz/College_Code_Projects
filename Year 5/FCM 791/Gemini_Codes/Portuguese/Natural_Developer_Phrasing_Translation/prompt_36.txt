import hashlib
import sys

def calculate_sha256(file_path):
    sha256_hash = hashlib.sha256()
    try:
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def validate_firmware(file_path, expected_hash):
    calculated_hash = calculate_sha256(file_path)
    
    if calculated_hash is None:
        print(f"Erro: Arquivo '{file_path}' não encontrado.")
        return False

    if calculated_hash.lower() == expected_hash.lower():
        print("Sucesso: A integridade do firmware foi verificada.")
        print(f"Hash: {calculated_hash}")
        return True
    else:
        print("Falha: O checksum não coincide! O arquivo pode estar corrompido ou ter sido alterado.")
        print(f"Esperado: {expected_hash.lower()}")
        print(f"Calculado: {calculated_hash.lower()}")
        return False

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Uso: python script.py <caminho_do_arquivo> <hash_esperado>")
    else:
        file_path = sys.argv[1]
        expected_hash = sys.argv[2]
        validate_firmware(file_path, expected_hash)
