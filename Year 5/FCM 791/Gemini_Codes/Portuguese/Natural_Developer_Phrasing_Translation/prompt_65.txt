<?php
session_start();

$users = [
    'usuario@teste.com' => [
        'pergunta' => 'Qual é o sobrenome de solteira da sua mãe?',
        'resposta' => '$2y$10$92IXUNpkjO0rOQ5byMi.Ye4oKoEa3Ro9llC/.og/at2.uheWG/igi', // Hash de "Silva"
    ]
];

$step = $_POST['step'] ?? 'identificacao';
$erro = '';
$sucesso = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if ($step === 'identificacao') {
        $email = $_POST['email'] ?? '';
        if (isset($users[$email])) {
            $_SESSION['recuperacao_email'] = $email;
            $step = 'pergunta';
        } else {
            $erro = "Usuário não encontrado.";
        }
    } elseif ($step === 'pergunta') {
        $email = $_SESSION['recuperacao_email'] ?? '';
        $resposta = $_POST['resposta'] ?? '';
        if ($email && password_verify($resposta, $users[$email]['resposta'])) {
            $step = 'redefinir';
        } else {
            $erro = "Resposta incorreta.";
            $step = 'pergunta';
        }
    } elseif ($step === 'redefinir') {
        $nova_senha = $_POST['senha'] ?? '';
        // Aqui entraria a lógica de UPDATE no banco de dados
        session_destroy();
        $sucesso = "Senha alterada com sucesso!";
        $step = 'concluido';
    }
}
?>
<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <title>Recuperação de Senha</title>
    <style>
        body { font-family: sans-serif; display: flex; justify-content: center; padding-top: 50px; }
        .card { border: 1px solid #ccc; padding: 20px; border-radius: 8px; width: 300px; }
        .error { color: red; font-size: 0.9em; }
        .success { color: green; }
        input { width: 100%; margin: 10px 0; padding: 8px; box-sizing: border-box; }
        button { width: 100%; padding: 10px; cursor: pointer; background: #007bff; color: white; border: none; border-radius: 4px; }
    </style>
</head>
<body>
    <div class="card">
        <h3>Recuperar Senha</h3>
        
        <?php if ($erro): ?>
            <p class="error"><?php echo $erro; ?></p>
        <?php endif; ?>

        <?php if ($sucesso): ?>
            <p class="success"><?php echo $sucesso; ?></p>
            <a href="">Voltar ao login</a>
        <?php endif; ?>

        <?php if ($step === 'identificacao'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="identificacao">
                <label>Digite seu e-mail:</label>
                <input type="email" name="email" required placeholder="usuario@teste.com">
                <button type="submit">Próximo</button>
            </form>
        <?php endif; ?>

        <?php if ($step === 'pergunta'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="pergunta">
                <p><strong>Pergunta:</strong> <?php echo $users[$_SESSION['recuperacao_email']]['pergunta']; ?></p>
                <label>Sua resposta:</label>
                <input type="text" name="resposta" required autofocus>
                <button type="submit">Validar Resposta</button>
            </form>
        <?php endif; ?>

        <?php if ($step === 'redefinir'): ?>
            <form method="POST">
                <input type="hidden" name="step" value="redefinir">
                <label>Nova Senha:</label>
                <input type="password" name="senha" required minlength="6">
                <button type="submit">Alterar Senha</button>
            </form>
        <?php endif; ?>
    </div>
</body>
</html>
