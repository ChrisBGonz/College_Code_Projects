import jwt
import datetime
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError, InvalidAudienceError, InvalidIssuerError

class JWTHandler:
    def __init__(self, secret_key, algorithm="HS256", issuer=None, audience=None):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.issuer = issuer
        self.audience = audience

    def generate_token(self, user_id, additional_claims=None):
        payload = {
            "sub": str(user_id),
            "iat": datetime.datetime.utcnow(),
            "exp": datetime.datetime.utcnow() + datetime.timedelta(hours=1),
            "nbf": datetime.datetime.utcnow(),
        }
        
        if self.issuer:
            payload["iss"] = self.issuer
        if self.audience:
            payload["aud"] = self.audience
            
        if additional_claims:
            payload.update(additional_claims)
            
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    def decode_and_validate(self, token):
        try:
            decoded_payload = jwt.decode(
                token,
                self.secret_key,
                algorithms=[self.algorithm],
                audience=self.audience,
                issuer=self.issuer,
                options={
                    "verify_signature": True,
                    "verify_exp": True,
                    "verify_iat": True,
                    "verify_nbf": True,
                    "require": ["exp", "iat", "sub"]
                }
            )
            return {"status": "success", "data": decoded_payload}
        except ExpiredSignatureError:
            return {"status": "error", "message": "Token expirado"}
        except InvalidAudienceError:
            return {"status": "error", "message": "Audiência inválida"}
        except InvalidIssuerError:
            return {"status": "error", "message": "Emissor inválido"}
        except InvalidTokenError as e:
            return {"status": "error", "message": f"Token inválido: {str(e)}"}
