import os
import yaml
from flask import Flask, request, render_template_string, jsonify

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = 2 * 1024 * 1024

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <title>Configuração de Perfil</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 500px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        h2 { color: #333; }
        input[type="file"] { margin: 20px 0; }
        button { background: #007bff; color: white; border: none; padding: 10px 15px; border-radius: 4px; cursor: pointer; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Upload de Configuração (YAML)</h2>
        <form method="POST" action="/upload" enctype="multipart/form-data">
            <input type="file" name="config_file" accept=".yaml,.yml" required>
            <br>
            <button type="submit">Configurar Perfil</button>
        </form>
    </div>
</body>
</html>
"""

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'config_file' not in request.files:
        return jsonify({"error": "Nenhum arquivo enviado"}), 400
    
    file = request.files['config_file']
    
    if file.filename == '':
        return jsonify({"error": "Nenhum arquivo selecionado"}), 400

    if file and (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        try:
            content = file.read().decode('utf-8')
            config_data = yaml.safe_load(content)
            
            # Simulação de processamento do perfil
            user_profile = {
                "nome": config_data.get('nome', 'N/A'),
                "email": config_data.get('email', 'N/A'),
                "preferencias": config_data.get('preferencias', {}),
                "status": "Perfil inicializado com sucesso"
            }
            
            return jsonify(user_profile), 200
            
        except yaml.YAMLError as e:
            return jsonify({"error": f"Erro ao processar YAML: {str(e)}"}), 400
        except Exception as e:
            return jsonify({"error": f"Erro interno: {str(e)}"}), 500
    
    return jsonify({"error": "Formato de arquivo inválido. Use .yaml ou .yml"}), 400

if __name__ == '__main__':
    app.run(debug=True)
