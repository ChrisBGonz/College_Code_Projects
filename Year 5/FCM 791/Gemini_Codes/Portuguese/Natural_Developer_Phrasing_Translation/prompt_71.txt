import pickle

class Player:
    def __init__(self, name, level, health):
        self.name = name
        self.level = level
        self.health = health

class GameState:
    def __init__(self, player, inventory, world_coords, flags):
        self.player = player
        self.inventory = inventory
        self.world_coords = world_coords
        self.flags = flags

def save_game(state, filename="savegame.pkl"):
    with open(filename, "wb") as file:
        pickle.dump(state, file)

def load_game(filename="savegame.pkl"):
    with open(filename, "rb") as file:
        return pickle.load(file)

player = Player(name="Aragorn", level=25, health=100.0)
current_state = GameState(
    player=player,
    inventory=["Andúril", "Elven Cloak", "Healing Herbs"],
    world_coords=(45.5, -12.3, 10.0),
    flags={"boss_defeated": False, "tutorial_complete": True}
)

save_game(current_state)

loaded_state = load_game()

print(f"Player: {loaded_state.player.name}")
print(f"Level: {loaded_state.player.level}")
print(f"Inventory: {loaded_state.inventory}")
print(f"Coords: {loaded_state.world_coords}")
print(f"Boss Defeated: {loaded_state.flags['boss_defeated']}")
