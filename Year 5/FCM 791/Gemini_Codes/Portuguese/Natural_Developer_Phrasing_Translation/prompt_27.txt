import pdfkit

def convert_html_to_pdf(html_content, output_path):
    options = {
        'page-size': 'A4',
        'encoding': "UTF-8",
        'margin-top': '0.75in',
        'margin-right': '0.75in',
        'margin-bottom': '0.75in',
        'margin-left': '0.75in',
        'no-outline': None
    }
    pdfkit.from_string(html_content, output_path, options=options)
