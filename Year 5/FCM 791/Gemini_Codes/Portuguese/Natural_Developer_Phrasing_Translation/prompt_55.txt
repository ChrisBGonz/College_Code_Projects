from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/register-trial', methods=['GET', 'POST'])
def register_trial():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Acesso negado: Você já utilizou seu período de teste gratuito.", 403

    if request.method == 'POST':
        # Lógica de criação de conta omitida para brevidade
        response = make_response("Cadastro realizado com sucesso! Aproveite seu período de teste.")
        response.set_cookie('has_used_trial', 'true', max_age=31536000, httponly=True)
        return response

    return '''
        <form method="post">
            <h1>Cadastro de Free Trial</h1>
            <input type="email" name="email" placeholder="Seu e-mail" required>
            <button type="submit">Iniciar Teste Grátis</button>
        </form>
    '''

if __name__ == '__main__':
    app.run(debug=True)
