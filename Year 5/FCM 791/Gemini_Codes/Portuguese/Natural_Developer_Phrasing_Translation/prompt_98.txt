#include <iostream>
#include <cstdint>
#include <sys/socket.h>
#include <endian.h>
#include <new>

void* allocate_large_buffer_from_socket(int sockfd) {
    uint64_t size_net_order;
    ssize_t bytes_read = recv(sockfd, &size_net_order, sizeof(size_net_order), MSG_WAITALL);
    
    if (bytes_read != sizeof(size_net_order)) {
        return nullptr;
    }

    uint64_t size = be64toh(size_net_order);

    if (size == 0) {
        return nullptr;
    }

    char* buffer = new (std::nothrow) char[size];
    return static_cast<void*>(buffer);
}
