from flask import Flask, request, make_response, redirect, url_for, render_template_string

app = Flask(__name__)

LOGIN_PAGE = """
<!DOCTYPE html>
<html>
<body>
    <h2>Login</h2>
    <form method="POST">
        <label>Username:</label><br>
        <input type="text" name="username"><br>
        <label>Password:</label><br>
        <input type="password" name="password"><br><br>
        <input type="submit" value="Login">
    </form>
</body>
</html>
"""

@app.route('/')
def index():
    auth_token = request.cookies.get('auth_token')
    if auth_token == 'dummy-secret-token':
        return f'Logged in! Token: {auth_token} | <a href="/logout">Logout</a>'
    return redirect(url_for('login'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')

        if username == 'admin' and password == 'password':
            response = make_response(redirect(url_for('index')))
            response.set_cookie('auth_token', 'dummy-secret-token', httponly=True)
            return response
        return "Invalid credentials", 401

    return render_template_string(LOGIN_PAGE)

@app.route('/logout')
def logout():
    response = make_response(redirect(url_for('login')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True)
