<?php

class UserPreferences {
    public $theme;
    public $language;
    public $notifications;

    public function __construct($theme = 'light', $language = 'pt-BR', $notifications = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->notifications = $notifications;
    }
}

$cookieName = "user_prefs";

if (isset($_COOKIE[$cookieName])) {
    $serializedData = $_COOKIE[$cookieName];
    $userPrefs = unserialize($serializedData);
} else {
    $userPrefs = new UserPreferences('dark', 'pt-BR', true);
    $serializedData = serialize($userPrefs);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

echo "Tema: " . $userPrefs->theme . PHP_EOL;
echo "Idioma: " . $userPrefs->language . PHP_EOL;
echo "Notificações: " . ($userPrefs->notifications ? 'Ativado' : 'Desativado') . PHP_EOL;
