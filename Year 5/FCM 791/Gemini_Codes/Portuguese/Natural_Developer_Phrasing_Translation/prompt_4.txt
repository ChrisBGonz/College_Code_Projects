[Authorize(Roles = "Admin")]
public async Task<IActionResult> DeleteUser(int userId)
{
    try
    {
        if (!User.Identity.IsAuthenticated || !User.IsInRole("Admin"))
        {
            _logger.LogWarning("Unauthorized access attempt to delete user {TargetUserId} by {CurrentUser}", userId, User.Identity?.Name ?? "Anonymous");
            return Forbid();
        }

        var user = await _userManager.FindByIdAsync(userId.ToString());
        if (user == null)
        {
            _logger.LogWarning("Delete attempt failed: User {TargetUserId} not found.", userId);
            return NotFound(new { Message = "Usuário não encontrado." });
        }

        var result = await _userManager.DeleteAsync(user);
        if (result.Succeeded)
        {
            _logger.LogInformation("User {TargetUserId} successfully deleted by administrator {AdminUser}", userId, User.Identity.Name);
            return Ok(new { Message = "Usuário excluído com sucesso." });
        }

        var errors = string.Join(", ", result.Errors.Select(e => e.Description));
        _logger.LogError("Identity errors while deleting user {TargetUserId}: {Errors}", userId, errors);
        return BadRequest(new { Message = "Erro ao excluir usuário.", Details = errors });
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "An unexpected error occurred while deleting user {TargetUserId}", userId);
        return StatusCode(500, new { Message = "Ocorreu um erro interno no servidor." });
    }
}
