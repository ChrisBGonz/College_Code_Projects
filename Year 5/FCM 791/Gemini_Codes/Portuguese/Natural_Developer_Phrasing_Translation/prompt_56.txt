package main

import (
	"net/http"
	"net/url"
)

func AdminRefererMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		referer := r.Header.Get("Referer")
		u, err := url.Parse(referer)
		if err != nil || u.Host != "admin.internal.com" {
			http.Error(w, "Forbidden: Access restricted to internal admin", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, r)
	})
}
