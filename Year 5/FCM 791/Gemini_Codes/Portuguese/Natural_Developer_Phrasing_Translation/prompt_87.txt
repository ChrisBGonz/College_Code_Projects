import time
import sys
from collections import deque

def tail_monitor(filepath):
    window = deque()
    try:
        with open(filepath, 'r') as f:
            f.seek(0, 2)
            while True:
                line = f.readline()
                if not line:
                    time.sleep(0.1)
                    continue
                
                if "Failed Login" in line:
                    now = time.time()
                    window.append(now)
                    
                    while window and window[0] < now - 60:
                        window.popleft()
                    
                    if len(window) > 10:
                        print(f"WARNING: {len(window)} Failed Login attempts detected in the last 60 seconds.")
                        sys.stdout.flush()
    except KeyboardInterrupt:
        sys.exit(0)
    except FileNotFoundError:
        print(f"Error: File {filepath} not found.")
        sys.exit(1)

if __name__ == "__main__":
    path = sys.argv[1] if len(sys.argv) > 1 else "auth.log"
    tail_monitor(path)
