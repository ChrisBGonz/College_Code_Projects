import hashlib
import os
import sys

def calculate_sha256(filepath):
    sha256_hash = hashlib.sha256()
    try:
        with open(filepath, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except Exception:
        return None

def validate_integrity(manifest_path):
    if not os.path.exists(manifest_path):
        print(f"Erro: Arquivo de manifesto '{manifest_path}' nao encontrado.")
        return

    try:
        with open(manifest_path, "r", encoding="utf-8") as f:
            lines = f.readlines()
    except Exception as e:
        print(f"Erro ao ler manifesto: {e}")
        return

    for line in lines:
        line = line.strip()
        if not line:
            continue
        
        parts = line.split(None, 1)
        if len(parts) < 2:
            print(f"[ERRO] Linha malformada no manifesto: {line}")
            continue
            
        expected_hash, filepath = parts
        
        if not os.path.exists(filepath):
            print(f"[AUSENTE] {filepath}")
            continue
            
        actual_hash = calculate_sha256(filepath)
        
        if actual_hash is None:
            print(f"[ERRO DE LEITURA] {filepath}")
        elif actual_hash.lower() == expected_hash.lower():
            print(f"[OK] {filepath}")
        else:
            print(f"[FALHA] {filepath}")
            print(f"      Esperado: {expected_hash}")
            print(f"      Obtido:   {actual_hash}")

if __name__ == "__main__":
    manifest_file = sys.argv[1] if len(sys.argv) > 1 else "manifest.txt"
    validate_integrity(manifest_file)
