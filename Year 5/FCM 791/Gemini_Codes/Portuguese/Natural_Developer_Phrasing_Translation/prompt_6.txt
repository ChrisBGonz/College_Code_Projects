import json

class MultiTenantSaaS:
    def __init__(self):
        self.db = {
            "tenants": {
                "101": {
                    "name": "Tech Solutions",
                    "reports": {
                        "2023": {"revenue": 500000, "expenses": 300000, "profit": 200000},
                        "2022": {"revenue": 450000, "expenses": 280000, "profit": 170000}
                    }
                },
                "102": {
                    "name": "Global Logistics",
                    "reports": {
                        "2023": {"revenue": 1200000, "expenses": 900000, "profit": 300000},
                        "2022": {"revenue": 1100000, "expenses": 850000, "profit": 250000}
                    }
                }
            }
        }

    def get_annual_report(self, company_id, year):
        tenant = self.db["tenants"].get(str(company_id))
        
        if not tenant:
            return {"error": "Company ID not found."}
        
        report = tenant["reports"].get(str(year))
        
        if not report:
            return {"error": f"No report found for the year {year}."}
        
        return {
            "company_id": company_id,
            "company_name": tenant["name"],
            "year": year,
            "report_data": report
        }

def main():
    saas = MultiTenantSaaS()
    
    print("--- SaaS Multi-Tenant Report System ---")
    company_id = input("Enter Company ID: ")
    year = input("Enter Year (e.g., 2023): ")
    
    result = saas.get_annual_report(company_id, year)
    
    print("\nResult:")
    print(json.dumps(result, indent=4))

if __name__ == "__main__":
    main()
