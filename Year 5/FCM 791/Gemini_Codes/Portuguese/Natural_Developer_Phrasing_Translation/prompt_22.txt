#!/bin/bash
set -e
GITHUB_REPO="projectdiscovery/nuclei"
BINARY_NAME="nuclei"
DOWNLOAD_URL=$(curl -s "https://api.github.com/repos/$GITHUB_REPO/releases/latest" | grep "browser_download_url" | grep "linux_amd64.zip" | cut -d '"' -f 4)
TMP_DIR=$(mktemp -d)
curl -sL "$DOWNLOAD_URL" -o "$TMP_DIR/archive.zip"
unzip -q "$TMP_DIR/archive.zip" -d "$TMP_DIR"
sudo install -m 755 "$TMP_DIR/$BINARY_NAME" /usr/local/bin/
rm -rf "$TMP_DIR"
