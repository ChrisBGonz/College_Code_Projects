import subprocess
import os
import sys

def run_shell_command(command):
    try:
        result = subprocess.run(command, check=True, text=True, capture_output=True)
        return result.stdout
    except subprocess.CalledProcessError as e:
        print(f"Error executing command: {' '.join(command)}")
        print(f"Stderr: {e.stderr}")
        sys.exit(1)

def deploy():
    image_name = os.getenv("DOCKER_IMAGE", "my-registry/my-app:latest")
    container_name = os.getenv("CONTAINER_NAME", "production-service")
    port_mapping = os.getenv("PORT_MAPPING", "80:8080")
    env_file = os.getenv("ENV_FILE", ".env.prod")

    print(f"--- Starting Deployment for {container_name} ---")

    print(f"Pulling latest image: {image_name}")
    run_shell_command(["docker", "pull", image_name])

    print(f"Checking for existing container: {container_name}")
    existing_containers = subprocess.run(
        ["docker", "ps", "-a", "-q", "-f", f"name={container_name}"],
        capture_output=True, text=True
    ).stdout.strip()

    if existing_containers:
        print(f"Stopping and removing existing container...")
        subprocess.run(["docker", "stop", container_name], capture_output=True)
        subprocess.run(["docker", "rm", container_name], capture_output=True)

    print(f"Launching new container...")
    run_command = [
        "docker", "run", "-d",
        "--name", container_name,
        "--restart", "always",
        "-p", port_mapping
    ]

    if os.path.exists(env_file):
        run_command.extend(["--env-file", env_file])

    run_command.append(image_name)
    
    run_shell_command(run_command)

    print(f"Cleaning up unused images...")
    subprocess.run(["docker", "image", "prune", "-f"], capture_output=True)

    print("--- Deployment Successful ---")

if __name__ == "__main__":
    deploy()
