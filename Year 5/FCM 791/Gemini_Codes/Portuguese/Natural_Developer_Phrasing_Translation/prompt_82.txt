const requestLogger = (req, res, next) => {
  const logData = {
    timestamp: new Date().toISOString(),
    method: req.method,
    url: req.originalUrl || req.url,
    headers: req.headers,
    query: req.query,
    body: req.body
  };

  console.log('--- Incoming Request ---');
  console.log(JSON.stringify(logData, null, 2));
  console.log('------------------------');

  next();
};

module.exports = requestLogger;
