import socket

def xor_cipher(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def start_client(host='127.0.0.1', port=8080, key=b'legacy_secret_key'):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.connect((host, port))
            
            message = "REQUEST_DATA_01"
            payload = xor_cipher(message.encode('utf-8'), key)
            s.sendall(payload)
            
            response = s.recv(4096)
            if response:
                decrypted_data = xor_cipher(response, key)
                print(decrypted_data.decode('utf-8', errors='ignore'))
                
    except ConnectionRefusedError:
        print("Server is offline.")
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    start_client()
