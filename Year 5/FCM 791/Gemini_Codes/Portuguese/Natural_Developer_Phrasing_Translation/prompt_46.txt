import requests
from urllib.parse import urlparse, parse_qs

def executar_redirecionamento(url_origem):
    parsed_url = urlparse(url_origem)
    query_params = parse_qs(parsed_url.query)
    url_destino = query_params.get('redirect_url', [None])[0]

    if url_destino:
        response = requests.get(url_destino)
        return response
    return None

url_teste = "http://exemplo.com?redirect_url=https://www.google.com"
resultado = executar_redirecionamento(url_teste)

if resultado:
    print(f"Status: {resultado.status_code}")
    print(f"URL Final: {resultado.url}")
