using System;
using System.Collections.Generic;
using System.Data;
using Microsoft.Data.SqlClient;
using System.Linq;

public class ProductRepository
{
    private readonly string _connectionString = "Server=localhost;Database=StoreDb;Trusted_Connection=True;";
    
    // Whitelist de colunas permitidas para ordenação
    private readonly HashSet<string> _allowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase) 
    { 
        "Id", 
        "Name", 
        "Price", 
        "CreatedAt" 
    };

    public void GetProductsSorted(string columnName, bool ascending = true)
    {
        // Validação rigorosa contra SQL Injection
        if (!_allowedColumns.Contains(columnName))
        {
            throw new ArgumentException($"A coluna '{columnName}' não é válida para ordenação.");
        }

        string direction = ascending ? "ASC" : "DESC";
        
        // Como nomes de colunas não podem ser parametrizados via SqlParameter, 
        // usamos a string validada da whitelist com interpolação segura.
        string sql = $"SELECT Id, Name, Price, CreatedAt FROM Products ORDER BY {columnName} {direction}";

        using (SqlConnection connection = new SqlConnection(_connectionString))
        {
            SqlCommand command = new SqlCommand(sql, connection);
            
            try
            {
                connection.Open();
                using (SqlDataReader reader = command.ExecuteReader())
                {
                    while (reader.Read())
                    {
                        Console.WriteLine($"{reader["Id"]} - {reader["Name"]} - {reader["Price"]}");
                    }
                }
            }
            catch (SqlException ex)
            {
                Console.WriteLine($"Erro de banco de dados: {ex.Message}");
            }
        }
    }
}

public class Program
{
    public static void Main()
    {
        ProductRepository repository = new ProductRepository();

        // Exemplo 1: Uso válido
        Console.WriteLine("--- Ordenando por Nome (ASC) ---");
        repository.GetProductsSorted("Name", true);

        // Exemplo 2: Uso válido
        Console.WriteLine("\n--- Ordenando por Preço (DESC) ---");
        repository.GetProductsSorted("Price", false);

        // Exemplo 3: Tentativa de SQL Injection ou coluna inexistente
        Console.WriteLine("\n--- Tentativa de Input Inválido ---");
        try
        {
            repository.GetProductsSorted("Name; DROP TABLE Products; --");
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine($"Bloqueado: {ex.Message}");
        }
    }
}
