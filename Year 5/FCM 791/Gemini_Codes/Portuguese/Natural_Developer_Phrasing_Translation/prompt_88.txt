using System;

public class MfaAuditService
{
    public class MfaChangeEntry
    {
        public string TargetUserId { get; set; }
        public string PerformedByUserId { get; set; }
        public string PreviousState { get; set; }
        public string NewState { get; set; }
        public DateTime Timestamp { get; set; }
    }

    public void LogMfaUpdate(string targetUser, string actor, string oldState, string newState)
    {
        var entry = new MfaChangeEntry
        {
            TargetUserId = targetUser,
            PerformedByUserId = actor,
            PreviousState = oldState,
            NewState = newState,
            Timestamp = DateTime.UtcNow
        };

        string logMessage = string.Format(
            "AUDIT_MFA_CHANGE|Timestamp:{0:O}|Actor:{1}|Target:{2}|Old:{3}|New:{4}",
            entry.Timestamp,
            entry.PerformedByUserId,
            entry.TargetUserId,
            entry.PreviousState,
            entry.NewState
        );

        Console.WriteLine(logMessage);
    }
}
