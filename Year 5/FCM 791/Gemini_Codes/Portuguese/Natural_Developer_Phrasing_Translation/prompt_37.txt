package main

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

type PasswordResetToken struct {
	Token     string
	ExpiresAt time.Time
}

func GenerateResetToken() (*PasswordResetToken, error) {
	b := make([]byte, 32)
	_, err := rand.Read(b)
	if err != nil {
		return nil, err
	}

	return &PasswordResetToken{
		Token:     hex.EncodeToString(b),
		ExpiresAt: time.Now().Add(1 * time.Hour),
	}, nil
}
