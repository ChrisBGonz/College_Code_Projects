<?php

$host = 'localhost';
$db   = 'your_database';
$user = 'your_username';
$pass = 'your_password';
$charset = 'utf8mb4';

$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES   => false,
];

try {
    $pdo = new PDO($dsn, $user, $pass, $options);
} catch (\PDOException $e) {
    throw new \PDOException($e->getMessage(), (int)$e->getCode());
}

define('ENCRYPTION_KEY', 'd82c4b7e1f3a5c9d2e4f6a8b0c1d3e5f'); 
define('CIPHER_METHOD', 'aes-256-cbc');

function encryptCreditCard($cardNumber, $key) {
    $ivLength = openssl_cipher_iv_length(CIPHER_METHOD);
    $iv = openssl_random_pseudo_bytes($ivLength);
    $encrypted = openssl_encrypt($cardNumber, CIPHER_METHOD, $key, 0, $iv);
    return base64_encode($iv . $encrypted);
}

$creditCardNumber = '1234567812345678';
$encryptedValue = encryptCreditCard($creditCardNumber, ENCRYPTION_KEY);

$sql = "INSERT INTO transactions (credit_card_number) VALUES (:cc)";
$stmt = $pdo->prepare($sql);
$stmt->execute(['cc' => $encryptedValue]);
