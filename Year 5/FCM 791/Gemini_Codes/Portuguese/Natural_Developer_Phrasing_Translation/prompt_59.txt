import requests
from flask import Flask, redirect, request

app = Flask(__name__)

@app.route('/process-payment', methods=['POST'])
def process_payment():
    gateway_endpoint = "https://api.thirdpartygateway.com/v1/checkout"
    api_key = "your_api_token_here"
    
    payload = {
        "amount": 250.00,
        "currency": "BRL",
        "order_id": "12345",
        "customer_email": "cliente@email.com"
    }
    
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }

    try:
        response = requests.post(gateway_endpoint, json=payload, headers=headers)
        response.raise_for_status()
        data = response.json()
        
        # Redireciona o usuário para a success_url retornada na resposta da API
        target_url = data.get("success_url")
        
        if target_url:
            return redirect(target_url, code=302)
        else:
            return "Erro: URL de redirecionamento não encontrada na resposta.", 400
            
    except requests.exceptions.RequestException as e:
        return f"Erro na integração: {str(e)}", 500

if __name__ == '__main__':
    app.run(port=5000)
