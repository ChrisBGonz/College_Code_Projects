<?php
session_start();

$admin_email = "admin@example.com";
$slack_webhook_url = "https://hooks.slack.com/services/YOUR/WEBHOOK/URL";

function is_authorized() {
    return isset($_SESSION['user_role']) && $_SESSION['user_role'] === 'admin';
}

if (!is_authorized()) {
    $ip = $_SERVER['REMOTE_ADDR'];
    $uri = $_SERVER['REQUEST_URI'];
    $timestamp = date('Y-m-d H:i:s');
    $user_agent = $_SERVER['HTTP_USER_AGENT'];

    $subject = "SECURITY ALERT: Unauthorized Access Attempt";
    $message = "An unauthorized user tried to access $uri\n\n" .
               "Details:\n" .
               "IP Address: $ip\n" .
               "Time: $timestamp\n" .
               "User Agent: $user_agent";

    mail($admin_email, $subject, $message);

    $slack_data = [
        'text' => "🚨 *Security Alert*\n*Attempted Access:* `$uri` \n*IP:* `$ip` \n*Time:* `$timestamp`"
    ];
    
    $ch = curl_init($slack_webhook_url);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($slack_data));
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_exec($ch);
    curl_close($ch);

    http_response_code(403);
    echo "<h1>403 Forbidden</h1><p>Access denied. This incident has been reported.</p>";
    exit;
}
