const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

const polls = {
    "1": {
        question: "Qual sua linguagem favorita?",
        options: { "JavaScript": 0, "Python": 0, "Go": 0 }
    }
};

const votesRegistry = {
    "1": new Set()
};

app.get('/polls', (req, res) => {
    res.json(polls);
});

app.post('/polls/:id/vote', (req, res) => {
    const pollId = req.params.id;
    const { option } = req.body;
    const userIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    const poll = polls[pollId];

    if (!poll) {
        return res.status(404).json({ error: "Enquete não encontrada" });
    }

    if (!poll.options.hasOwnProperty(option)) {
        return res.status(400).json({ error: "Opção inválida" });
    }

    if (votesRegistry[pollId].has(userIp)) {
        return res.status(403).json({ error: "Você já votou nesta enquete" });
    }

    poll.options[option]++;
    votesRegistry[pollId].add(userIp);

    res.json({
        message: "Voto computado com sucesso",
        results: poll.options
    });
});

app.listen(PORT, () => {
    console.log(`Servidor rodando em http://localhost:${PORT}`);
});
