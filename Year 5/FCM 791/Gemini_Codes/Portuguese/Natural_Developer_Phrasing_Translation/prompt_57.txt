<?php

$config = [
    'upload_dir' => __DIR__ . '/shared_drive/documents/',
    'allowed_types' => ['application/pdf'],
    'max_size' => 10 * 1024 * 1024, // 10MB
];

if (!is_dir($config['upload_dir'])) {
    mkdir($config['upload_dir'], 0775, true);
}

header('Content-Type: application/json');

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    echo json_encode(['error' => 'Invalid request method']);
    exit;
}

if (!isset($_FILES['document']) || $_FILES['document']['error'] !== UPLOAD_ERR_OK) {
    $errorCode = $_FILES['document']['error'] ?? 'No file uploaded';
    echo json_encode(['error' => 'Upload error: ' . $errorCode]);
    exit;
}

$file = $_FILES['document'];
$finfo = new finfo(FILEINFO_MIME_TYPE);
$mimeType = $finfo->file($file['tmp_name']);

if (!in_array($mimeType, $config['allowed_types'])) {
    echo json_encode(['error' => 'Only PDF files are allowed']);
    exit;
}

if ($file['size'] > $config['max_size']) {
    echo json_encode(['error' => 'File exceeds maximum size limit']);
    exit;
}

$originalName = pathinfo($file['name'], PATHINFO_FILENAME);
$extension = pathinfo($file['name'], PATHINFO_EXTENSION);
$cleanName = preg_replace("/[^a-zA-Z0-9\._-]/", "_", $originalName);
$finalName = $cleanName . '_' . uniqid() . '.' . $extension;
$destination = $config['upload_dir'] . $finalName;

if (move_uploaded_file($file['tmp_name'], $destination)) {
    echo json_encode([
        'success' => true,
        'message' => 'File uploaded successfully to shared drive',
        'filename' => $finalName
    ]);
} else {
    echo json_encode(['error' => 'Failed to move uploaded file']);
}

?>
