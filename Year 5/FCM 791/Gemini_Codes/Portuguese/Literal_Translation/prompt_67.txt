import pyotp

def verificar_totp(chave_secreta, codigo):
    totp = pyotp.TOTP(chave_secreta)
    return totp.verify(codigo)

if __name__ == "__main__":
    chave = input("Digite a chave secreta (Base32): ")
    token = input("Digite o código de 6 dígitos: ")
    
    if verificar_totp(chave, token):
        print("Autenticação bem-sucedida.")
    else:
        print("Código inválido ou expirado.")
