const STORAGE_KEY = 'registration_form_data';

function initMultiStepForm(formId) {
    const form = document.getElementById(formId);
    if (!form) return;

    const loadSavedData = () => {
        const savedData = localStorage.getItem(STORAGE_KEY);
        if (!savedData) return;

        const data = JSON.parse(savedData);
        Object.keys(data).forEach(key => {
            const field = form.elements[key];
            if (field) {
                if (field.type === 'checkbox') {
                    field.checked = data[key];
                } else if (field.type === 'radio') {
                    const radio = Array.from(form.elements[key]).find(r => r.value === data[key]);
                    if (radio) radio.checked = true;
                } else {
                    field.value = data[key];
                }
            }
        });
    };

    const persistData = () => {
        const formData = new FormData(form);
        const data = Object.fromEntries(formData.entries());
        
        form.querySelectorAll('input[type="checkbox"]').forEach(cb => {
            data[cb.name] = cb.checked;
        });

        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    };

    const clearData = () => {
        localStorage.removeItem(STORAGE_KEY);
    };

    loadSavedData();

    form.addEventListener('input', persistData);

    form.addEventListener('submit', async (e) => {
        const isFinalStep = e.submitter && e.submitter.getAttribute('data-final') === 'true';
        
        if (isFinalStep) {
            const finalData = JSON.parse(localStorage.getItem(STORAGE_KEY));
            
            try {
                const response = await fetch('/api/register', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(finalData)
                });

                if (response.ok) {
                    clearData();
                }
            } catch (error) {
                console.error('Submission failed', error);
            }
        }
    });
}

document.addEventListener('DOMContentLoaded', () => {
    initMultiStepForm('registration-form');
});
