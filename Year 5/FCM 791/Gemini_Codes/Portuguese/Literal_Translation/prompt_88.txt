using System;
using Microsoft.Extensions.Logging;

public enum MfaChangeType
{
    Enabled,
    Disabled,
    MethodAdded,
    MethodRemoved,
    SecretReset
}

public class MfaAuditService
{
    private readonly ILogger<MfaAuditService> _logger;

    public MfaAuditService(ILogger<MfaAuditService> logger)
    {
        _logger = logger;
    }

    public void RecordMfaChange(string userId, MfaChangeType changeType, string mfaMethod, string ipAddress)
    {
        var auditEvent = new
        {
            Timestamp = DateTime.UtcNow,
            UserId = userId,
            Action = changeType.ToString(),
            Method = mfaMethod,
            SourceIp = ipAddress,
            EventCategory = "Security.Authentication.MFA"
        };

        _logger.LogInformation(
            "MFA Configuration Change: User {UserId} performed {Action} for method {Method}. IP: {SourceIp}. Timestamp: {Timestamp}",
            auditEvent.UserId,
            auditEvent.Action,
            auditEvent.Method,
            auditEvent.SourceIp,
            auditEvent.Timestamp
        );
    }
}
