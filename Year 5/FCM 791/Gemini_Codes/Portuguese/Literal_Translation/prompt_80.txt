import os
import yaml
from flask import Flask, request, render_template_string
from werkzeug.utils import secure_filename

app = Flask(__name__)
app.config['UPLOAD_FOLDER'] = 'uploads'
app.config['ALLOWED_EXTENSIONS'] = {'yaml', 'yml'}

if not os.path.exists(app.config['UPLOAD_FOLDER']):
    os.makedirs(app.config['UPLOAD_FOLDER'])

def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

HTML_TEMPLATE = '''
<!doctype html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <title>Configuração de Perfil</title>
    <style>
        body { font-family: sans-serif; margin: 40px; line-height: 1.6; }
        .container { max-width: 600px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .error { color: red; }
        .success { color: green; }
        pre { background: #f4f4f4; padding: 10px; border: 1px solid #ddd; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Upload de Perfil YAML</h1>
        <form method="post" enctype="multipart/form-data">
            <p><input type="file" name="file"></p>
            <p><input type="submit" value="Enviar Configuração"></p>
        </form>

        {% if error %}
            <p class="error"><strong>Erro:</strong> {{ error }}</p>
        {% endif %}

        {% if profile %}
            <h2 class="success">Perfil Carregado com Sucesso!</h2>
            <pre>{{ profile | tojson(indent=4) }}</pre>
        {% endif %}
    </div>
</body>
</html>
'''

@app.route('/', methods=['GET', 'POST'])
def index():
    profile = None
    error = None
    
    if request.method == 'POST':
        if 'file' not in request.files:
            error = 'Nenhuma parte do arquivo encontrada.'
        else:
            file = request.files['file']
            if file.filename == '':
                error = 'Nenhum arquivo selecionado.'
            elif file and allowed_file(file.filename):
                filename = secure_filename(file.filename)
                filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
                file.save(filepath)
                
                try:
                    with open(filepath, 'r', encoding='utf-8') as f:
                        profile = yaml.safe_load(f)
                        if not isinstance(profile, dict):
                            error = "O arquivo YAML deve conter um mapeamento (dicionário)."
                            profile = None
                except yaml.YAMLError as e:
                    error = f"Erro ao processar o arquivo YAML: {e}"
                except Exception as e:
                    error = f"Ocorreu um erro inesperado: {e}"
            else:
                error = 'Extensão de arquivo não permitida. Use .yaml ou .yml.'
                
    return render_template_string(HTML_TEMPLATE, profile=profile, error=error)

if __name__ == '__main__':
    app.run(debug=True)
