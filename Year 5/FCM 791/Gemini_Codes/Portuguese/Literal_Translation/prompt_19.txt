from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

class DatabaseConnectionError(Exception):
    def __init__(self, detail: str):
        self.detail = detail

app = FastAPI()

@app.exception_handler(DatabaseConnectionError)
async def database_connection_exception_handler(request: Request, exc: DatabaseConnectionError):
    return JSONResponse(
        status_code=503,
        content={
            "error": "Database Error",
            "message": exc.detail,
            "status": "Service Unavailable"
        },
    )

@app.get("/test-db")
async def test_db_connection(fail: bool = True):
    if fail:
        raise DatabaseConnectionError("Unable to establish connection to the PostgreSQL cluster at 10.0.0.5:5432")
    return {"status": "connected"}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
