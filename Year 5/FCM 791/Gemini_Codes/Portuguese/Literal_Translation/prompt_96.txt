package main

import (
	"io"
	"net/http"
)

func callMicroserviceWithRetry(url string) ([]byte, error) {
	for {
		resp, err := http.Get(url)
		if err != nil {
			continue
		}

		if resp.StatusCode >= 200 && resp.StatusCode < 300 {
			body, err := io.ReadAll(resp.Body)
			resp.Body.Close()
			if err != nil {
				continue
			}
			return body, nil
		}
		resp.Body.Close()
	}
}
