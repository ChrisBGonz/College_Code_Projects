import logging
from logging.handlers import TimedRotatingFileHandler
import time

def setup_logging(log_file="app.log"):
    logger = logging.getLogger("DailyRotationLogger")
    logger.setLevel(logging.INFO)

    # Configura a rotação a cada 24 horas (when="D", interval=1)
    # backupCount=30 mantém os logs dos últimos 30 dias
    handler = TimedRotatingFileHandler(
        log_file,
        when="D",
        interval=1,
        backupCount=30,
        encoding="utf-8"
    )

    formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    
    return logger

if __name__ == "__main__":
    logger = setup_logging()
    
    try:
        while True:
            logger.info("Log de rotina registrado.")
            # Simula um intervalo entre logs
            time.sleep(3600)
    except KeyboardInterrupt:
        logger.info("Script finalizado pelo usuário.")
