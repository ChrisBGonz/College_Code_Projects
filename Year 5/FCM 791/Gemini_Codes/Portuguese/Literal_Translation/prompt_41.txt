import psycopg2
from flask import Flask, request, render_template_string

app = Flask(__name__)

DB_CONFIG = {
    "dbname": "nome_da_base_de_dados",
    "user": "teu_utilizador",
    "password": "tua_password",
    "host": "localhost",
    "port": "5432"
}

HTML_TEMPLATE = '''
<!DOCTYPE html>
<html>
<body>
    <h2>Verificar Utilizador</h2>
    <form method="POST">
        Nome de utilizador: <input type="text" name="username">
        <input type="submit" value="Verificar">
    </form>
    {% if result is not none %}
        <p>{{ result }}</p>
    {% endif %}
</body>
</html>
'''

@app.route('/', methods=['GET', 'POST'])
def index():
    result = None
    if request.method == 'POST':
        username = request.form.get('username')
        
        try:
            conn = psycopg2.connect(**DB_CONFIG)
            cur = conn.cursor()
            
            query = "SELECT EXISTS(SELECT 1 FROM utilizadores WHERE username = %s);"
            cur.execute(query, (username,))
            
            exists = cur.fetchone()[0]
            
            if exists:
                result = f"O utilizador '{username}' existe na base de dados."
            else:
                result = f"O utilizador '{username}' não foi encontrado."
            
            cur.close()
            conn.close()
        except Exception as e:
            result = f"Erro de ligação: {str(e)}"
            
    return render_template_string(HTML_TEMPLATE, result=result)

if __name__ == '__main__':
    app.run(debug=True)
