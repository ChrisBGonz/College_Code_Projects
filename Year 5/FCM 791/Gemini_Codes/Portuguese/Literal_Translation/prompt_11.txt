import sys
from lxml import etree

def extract_titles():
    file_path = input("Introduza o caminho para o ficheiro XML: ")
    try:
        parser = etree.XMLParser(remove_blank_text=True)
        tree = etree.parse(file_path, parser)
        # Procura por todas as etiquetas 'title' independentemente da profundidade
        titles = tree.xpath('//title')
        
        if not titles:
            print("Nenhuma etiqueta <title> encontrada.")
            return

        for title in titles:
            print(title.text if title.text else "Etiqueta vazia")
            
    except FileNotFoundError:
        print("Erro: Ficheiro não encontrado.")
    except etree.XMLSyntaxError as e:
        print(f"Erro de sintaxe XML: {e}")
    except Exception as e:
        print(f"Ocorreu um erro: {e}")

if __name__ == "__main__":
    extract_titles()
