from flask import request, jsonify
from werkzeug.security import generate_password_hash
from models import db, User

@app.route('/alterar-senha/<int:user_id>', methods=['POST'])
def alterar_senha(user_id):
    dados = request.get_json()
    nova_senha = dados.get('nova_senha')

    if not nova_senha:
        return jsonify({"erro": "A nova senha é obrigatória"}), 400

    usuario = User.query.get(user_id)
    if not usuario:
        return jsonify({"erro": "Usuário não encontrado"}), 404

    usuario.password = generate_password_hash(nova_senha)
    db.session.commit()

    return jsonify({"mensagem": "Senha atualizada com sucesso"}), 200
