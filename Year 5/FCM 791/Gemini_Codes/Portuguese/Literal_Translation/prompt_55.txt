from flask import Flask, request, make_response

app = Flask(__name__)

@app.route('/trial', methods=['GET', 'POST'])
def trial_registration():
    has_used_trial = request.cookies.get('has_used_trial')

    if has_used_trial == 'true':
        return "Acesso negado: Você já utilizou seu período de teste gratuito.", 403

    if request.method == 'POST':
        # Lógica de processamento do cadastro (ex: salvar no banco de dados)
        
        response = make_response("Cadastro de teste gratuito realizado com sucesso!")
        # Define o cookie para expirar em 30 dias
        response.set_cookie('has_used_trial', 'true', max_age=30*24*60*60, httponly=True)
        return response

    return '''
        <form method="post">
            <h1>Cadastro de Teste Gratuito</h1>
            <input type="text" name="username" placeholder="Nome de usuário" required><br><br>
            <input type="email" name="email" placeholder="E-mail" required><br><br>
            <button type="submit">Iniciar Teste</button>
        </form>
    '''

if __name__ == '__main__':
    app.run(debug=True)
