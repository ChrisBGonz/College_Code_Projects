<?php
session_start();

$host = 'localhost';
$db_name = 'sistema_usuarios';
$username = 'root';
$password = '';

try {
    $pdo = new PDO("mysql:host=$host;dbname=$db_name;charset=utf8", $username, $password);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch (PDOException $e) {
    die("Erro na conexão: " . $e->getMessage());
}

$message = "";
$step = 1;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['check_user'])) {
        $user_input = $_POST['username'];
        $stmt = $pdo->prepare("SELECT id, secret_question FROM users WHERE username = ? OR email = ?");
        $stmt->execute([$user_input, $user_input]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user) {
            $_SESSION['recovery_user_id'] = $user['id'];
            $_SESSION['recovery_question'] = $user['secret_question'];
            $step = 2;
        } else {
            $message = "Usuário não encontrado.";
        }
    } elseif (isset($_POST['verify_answer'])) {
        $answer_input = strtolower(trim($_POST['answer']));
        $user_id = $_SESSION['recovery_user_id'];

        $stmt = $pdo->prepare("SELECT secret_answer FROM users WHERE id = ?");
        $stmt->execute([$user_id]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify($answer_input, $user['secret_answer'])) {
            $_SESSION['recovery_allowed'] = true;
            $step = 3;
        } else {
            $message = "Resposta incorreta.";
            $step = 2;
        }
    } elseif (isset($_POST['reset_password'])) {
        if (isset($_SESSION['recovery_allowed']) && $_SESSION['recovery_allowed'] === true) {
            $new_password = password_hash($_POST['new_password'], PASSWORD_DEFAULT);
            $user_id = $_SESSION['recovery_user_id'];

            $stmt = $pdo->prepare("UPDATE users SET password = ? WHERE id = ?");
            $stmt->execute([$new_password, $user_id]);

            session_destroy();
            $message = "Senha alterada com sucesso! <a href='login.php'>Faça login</a>";
            $step = 4;
        }
    }
}
?>
<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <title>Recuperação de Senha</title>
    <style>
        body { font-family: sans-serif; display: flex; justify-content: center; padding-top: 50px; }
        .card { border: 1px solid #ccc; padding: 20px; border-radius: 8px; width: 300px; }
        input { width: 100%; margin: 10px 0; padding: 8px; box-sizing: border-box; }
        button { width: 100%; padding: 10px; background: #007bff; color: white; border: none; cursor: pointer; }
        .error { color: red; font-size: 0.9em; }
        .success { color: green; }
    </style>
</head>
<body>
    <div class="card">
        <h2>Recuperação</h2>
        <?php if ($message): ?>
            <p class="<?php echo $step == 4 ? 'success' : 'error'; ?>"><?php echo $message; ?></p>
        <?php endif; ?>

        <?php if ($step == 1): ?>
            <form method="POST">
                <label>Digite seu usuário ou e-mail:</label>
                <input type="text" name="username" required>
                <button type="submit" name="check_user">Continuar</button>
            </form>
        <?php endif; ?>

        <?php if ($step == 2): ?>
            <form method="POST">
                <p><strong>Pergunta:</strong> <?php echo htmlspecialchars($_SESSION['recovery_question']); ?></p>
                <label>Sua resposta:</label>
                <input type="text" name="answer" required autofocus>
                <button type="submit" name="verify_answer">Verificar Resposta</button>
            </form>
        <?php endif; ?>

        <?php if ($step == 3): ?>
            <form method="POST">
                <label>Nova Senha:</label>
                <input type="password" name="new_password" required minlength="6">
                <button type="submit" name="reset_password">Alterar Senha</button>
            </form>
        <?php endif; ?>
    </div>
</body>
</html>
