using System;
using System.Data;
using System.Linq;
using Microsoft.Data.SqlClient;

public DataTable GetSortedData(string connectionString, string columnName)
{
    string[] allowedColumns = { "Id", "Name", "Email", "CreatedAt" };

    if (!allowedColumns.Contains(columnName))
    {
        throw new ArgumentException("Invalid column name for sorting.");
    }

    using (SqlConnection connection = new SqlConnection(connectionString))
    {
        string query = $"SELECT * FROM MyTable ORDER BY {columnName}";
        
        using (SqlCommand command = new SqlCommand(query, connection))
        {
            DataTable dataTable = new DataTable();
            using (SqlDataAdapter adapter = new SqlDataAdapter(command))
            {
                adapter.Fill(dataTable);
            }
            return dataTable;
        }
    }
}
