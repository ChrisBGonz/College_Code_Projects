<?php
$targetDirectory = "uploads/";
$uploadOk = 1;
$statusMsg = "";

if (!file_exists($targetDirectory)) {
    mkdir($targetDirectory, 0777, true);
}

if ($_SERVER['REQUEST_METHOD'] == 'POST' && isset($_FILES['pdfFile'])) {
    $fileName = basename($_FILES["pdfFile"]["name"]);
    $targetFilePath = $targetDirectory . $fileName;
    $fileType = strtolower(pathinfo($targetFilePath, PATHINFO_EXTENSION));

    if ($_FILES["pdfFile"]["size"] > 5000000) {
        $statusMsg = "Erro: O arquivo é muito grande (máx 5MB).";
        $uploadOk = 0;
    }

    if ($fileType != "pdf") {
        $statusMsg = "Erro: Apenas arquivos PDF são permitidos.";
        $uploadOk = 0;
    }

    if ($uploadOk == 1) {
        if (move_uploaded_file($_FILES["pdfFile"]["tmp_name"], $targetFilePath)) {
            $statusMsg = "O arquivo " . htmlspecialchars($fileName) . " foi enviado com sucesso.";
        } else {
            $statusMsg = "Erro ao realizar o upload do arquivo.";
        }
    }
}
?>
<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <title>Upload de PDF - Intranet</title>
    <style>
        body { font-family: sans-serif; padding: 20px; }
        .container { max-width: 400px; margin: auto; border: 1px solid #ccc; padding: 20px; border-radius: 8px; }
        .msg { margin-top: 15px; padding: 10px; border-radius: 4px; background: #f0f0f0; }
    </style>
</head>
<body>
    <div class="container">
        <h2>Upload de Documentos (PDF)</h2>
        <form action="" method="post" enctype="multipart/form-data">
            <input type="file" name="pdfFile" accept="application/pdf" required>
            <br><br>
            <input type="submit" value="Enviar Arquivo">
        </form>
        <?php if ($statusMsg): ?>
            <div class="msg"><?php echo $statusMsg; ?></div>
        <?php endif; ?>
    </div>
</body>
