const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

const poll = {
    question: "Qual é a sua linguagem de programação favorita?",
    options: {
        "JavaScript": 0,
        "Python": 0,
        "Rust": 0,
        "Go": 0
    }
};

const votedIPs = new Set();

app.get('/poll', (req, res) => {
    res.json({
        question: poll.question,
        results: poll.options,
        totalVotes: votedIPs.size
    });
});

app.post('/vote', (req, res) => {
    const { option } = req.body;
    const userIP = req.headers['x-forwarded-for'] || req.socket.remoteAddress;

    if (votedIPs.has(userIP)) {
        return res.status(403).json({ error: "Você já votou nesta enquete." });
    }

    if (poll.options.hasOwnProperty(option)) {
        poll.options[option]++;
        votedIPs.add(userIP);
        return res.json({ message: "Voto computado com sucesso!", results: poll.options });
    }

    res.status(400).json({ error: "Opção inválida." });
});

app.listen(PORT, () => {
    console.log(`Servidor de votação rodando em http://localhost:${PORT}`);
});
