import time
import os
from collections import deque

LOG_FILE = "auth.log"
THRESHOLD = 10
WINDOW = 60
KEYWORD = "Failed Login"

def main():
    events = deque()
    if not os.path.exists(LOG_FILE):
        open(LOG_FILE, 'a').close()
    
    with open(LOG_FILE, 'r') as f:
        f.seek(0, os.SEEK_END)
        while True:
            line = f.readline()
            if not line:
                time.sleep(0.5)
                continue
            
            if KEYWORD in line:
                now = time.time()
                events.append(now)
                while events and events[0] < now - WINDOW:
                    events.popleft()
                
                if len(events) > THRESHOLD:
                    print(f"AVISO: Detectadas {len(events)} falhas de login no último minuto.")
                    events.clear()

if __name__ == "__main__":
    main()
