class UsersController < ApplicationController
  before_action :authenticate_user!

  def promote
    @user = User.find(params[:id])

    if current_user.role == 'manager' && @user.role == 'staff'
      if @user.update(role: 'manager')
        redirect_to @user, notice: 'Membro promovido a Manager com sucesso.'
      else
        redirect_to @user, alert: 'Não foi possível realizar a promoção.'
      end
    else
      redirect_to root_path, alert: 'Ação não autorizada.'
    end
  end
end
