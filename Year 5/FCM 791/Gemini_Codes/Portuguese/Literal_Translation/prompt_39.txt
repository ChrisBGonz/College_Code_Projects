import socket

def xor_cipher(data, key):
    return bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])

def start_client(host='127.0.0.1', port=65432, key=b'SECRET_KEY'):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        try:
            s.connect((host, port))
            
            message = "LEGACY_REQUEST_DATA"
            encrypted_data = xor_cipher(message.encode(), key)
            s.sendall(encrypted_data)
            
            response = s.recv(1024)
            if response:
                decrypted_response = xor_cipher(response, key)
                print(decrypted_response.decode())
        except ConnectionRefusedError:
            pass

if __name__ == "__main__":
    start_client()
