import jwt
import datetime
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.backends import default_backend

private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048,
    backend=default_backend()
)
public_key = private_key.public_key()

private_pem = private_key.private_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PrivateFormat.PKCS8,
    encryption_algorithm=serialization.NoEncryption()
)
public_pem = public_key.public_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PublicFormat.SubjectPublicKeyInfo
)

payload = {
    "iss": "trusted-auth-provider",
    "sub": "user_987654321",
    "aud": "internal-microservice-cluster",
    "iat": datetime.datetime.utcnow(),
    "exp": datetime.datetime.utcnow() + datetime.timedelta(minutes=30),
    "nbf": datetime.datetime.utcnow(),
    "jti": "unique-token-id-xyz",
    "scopes": ["read:data", "write:data"],
    "user_metadata": {
        "role": "administrator",
        "department": "engineering"
    }
}

encoded_jwt = jwt.encode(
    payload, 
    private_pem, 
    algorithm="RS256", 
    headers={"kid": "key-id-001"}
)

try:
    decoded_data = jwt.decode(
        encoded_jwt,
        public_pem,
        algorithms=["RS256"],
        audience="internal-microservice-cluster",
        issuer="trusted-auth-provider",
        leeway=10,
        options={
            "verify_signature": True,
            "verify_exp": True,
            "verify_nbf": True,
            "verify_iat": True,
            "verify_aud": True,
            "verify_iss": True,
            "require": ["exp", "iat", "iss", "sub"]
        }
    )
    print("Decoded Payload:", decoded_data)
except jwt.ExpiredSignatureError:
    print("Error: Token has expired.")
except jwt.InvalidIssuerError:
    print("Error: Token has an invalid issuer.")
except jwt.InvalidAudienceError:
    print("Error: Token has an invalid audience.")
except jwt.InvalidTokenError as e:
    print(f"Error: Invalid token. {str(e)}")
