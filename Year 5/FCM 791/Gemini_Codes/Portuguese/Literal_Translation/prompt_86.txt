package main

import (
	"errors"
)

type Transfer struct {
	SenderID   string
	ReceiverID string
	Amount     float64
}

func RegisterTransfer(senderID string, receiverID string, amount float64) (*Transfer, error) {
	if amount <= 0 {
		return nil, errors.New("invalid amount")
	}

	if senderID == "" || receiverID == "" {
		return nil, errors.New("sender and receiver IDs are required")
	}

	transfer := &Transfer{
		SenderID:   senderID,
		ReceiverID: receiverID,
		Amount:     amount,
	}

	return transfer, nil
}
