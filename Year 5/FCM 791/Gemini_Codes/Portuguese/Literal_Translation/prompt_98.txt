#include <sys/socket.h>
#include <endian.h>
#include <unistd.h>
#include <cstdint>
#include <new>

void* allocate_buffer_from_socket(int socket_fd) {
    uint64_t network_size = 0;
    size_t bytes_to_read = sizeof(network_size);
    char* ptr = reinterpret_cast<char*>(&network_size);

    while (bytes_to_read > 0) {
        ssize_t res = recv(socket_fd, ptr, bytes_to_read, 0);
        if (res <= 0) {
            return nullptr;
        }
        ptr += res;
        bytes_to_read -= res;
    }

    uint64_t buffer_size = be64toh(network_size);

    if (buffer_size == 0) {
        return nullptr;
    }

    return new(std::nothrow) char[buffer_size];
}
