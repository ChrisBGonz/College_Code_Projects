import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {
    private static final Logger logger = Logger.getLogger(DatabaseHandler.class.getName());

    public void executeOperation() throws CustomDataException {
        try {
            // Lógica de banco de dados aqui
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Erro de SQL detectado: " + e.getMessage());
            throw new CustomDataException("Falha na comunicação com o banco de dados.", e);
        }
    }
}

class CustomDataException extends Exception {
    public CustomDataException(String message, Throwable cause) {
        super(message, cause);
    }
}
