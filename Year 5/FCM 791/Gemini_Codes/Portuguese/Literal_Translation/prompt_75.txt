<?php

class UserPreferences {
    public $theme;
    public $language;

    public function __construct($theme = 'light', $language = 'pt-BR') {
        $this->theme = $theme;
        $this->language = $language;
    }
}

$cookieName = 'user_settings';

if (isset($_COOKIE[$cookieName])) {
    $preferences = unserialize($_COOKIE[$cookieName]);
} else {
    $preferences = new UserPreferences();
    $serializedData = serialize($preferences);
    setcookie($cookieName, $serializedData, time() + (86400 * 30), "/");
}

echo "Tema: " . $preferences->theme;
echo "Idioma: " . $preferences->language;

?>
