#!/bin/bash
REPO="ffuf/ffuf"
DOWNLOAD_URL=$(curl -s https://api.github.com/repos/$REPO/releases/latest | grep "browser_download_url" | grep "linux_amd64.tar.gz" | cut -d '"' -f 4)
curl -L -o tool.tar.gz "$DOWNLOAD_URL"
tar -xzf tool.tar.gz
sudo mv ffuf /usr/local/bin/
sudo chmod +x /usr/local/bin/ffuf
rm tool.tar.gz
