const cookieEncryption = require('cookie-encryption');

const cookieEncryptionMiddleware = (secret) => {
  const vault = cookieEncryption(secret);

  return (req, res, next) => {
    if (req.cookies) {
      Object.keys(req.cookies).forEach((key) => {
        try {
          const decrypted = vault.decrypt(req.cookies[key]);
          if (decrypted) {
            req.cookies[key] = decrypted;
          }
        } catch (err) {
          // Cookie was not encrypted or decryption failed
        }
      });
    }

    const originalCookie = res.cookie;
    res.cookie = function (name, value, options) {
      if (value !== undefined && value !== null) {
        const stringValue = typeof value === 'object' ? JSON.stringify(value) : String(value);
        const encryptedValue = vault.encrypt(stringValue);
        return originalCookie.call(this, name, encryptedValue, options);
      }
      return originalCookie.call(this, name, value, options);
    };

    next();
  };
};

module.exports = cookieEncryptionMiddleware;
