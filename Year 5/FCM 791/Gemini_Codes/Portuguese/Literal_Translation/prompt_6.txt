class SaaSManager:
    def __init__(self):
        self._storage = {
            "tenants": {
                "T001": {"name": "Global Tech", "users": ["admin_gt", "user_gt"]},
                "T002": {"name": "Local Services", "users": ["user_ls"]}
            },
            "annual_reports": {
                "T001": {"2023": {"revenue": 1200000, "tax": 240000, "net": 960000}},
                "T002": {"2023": {"revenue": 450000, "tax": 90000, "net": 360000}}
            }
        }

    def view_annual_report(self, user_id, tenant_id, year):
        tenant = self._storage["tenants"].get(tenant_id)
        if not tenant:
            return "Error: Tenant ID not found."
        
        if user_id not in tenant["users"]:
            return "Error: Access denied. User not associated with this tenant."

        report = self._storage["annual_reports"].get(tenant_id, {}).get(str(year))
        if not report:
            return f"Error: No report found for year {year}."

        return {
            "company": tenant["name"],
            "year": year,
            "details": report
        }

if __name__ == "__main__":
    saas = SaaSManager()
    
    # Exemplo: Usuário autorizado visualizando relatório
    result_ok = saas.view_annual_report("admin_gt", "T001", 2023)
    print(f"Requisição Autorizada: {result_ok}")

    # Exemplo: Usuário não autorizado tentando visualizar relatório de outra empresa
    result_fail = saas.view_annual_report("user_ls", "T001", 2023)
    print(f"Requisição Não Autorizada: {result_fail}")
