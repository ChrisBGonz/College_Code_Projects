const express = require('express');

const app = express();
app.use(express.json());

const users = new Map();

function validateEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

async function updateProfile(req, res) {
  const body = req.body || {};
  const userId = (req.user && (req.user.id || req.user.userId)) || body.userId || req.headers['x-user-id'];
  if (!userId) {
    res.status(401).json({ error: 'Unauthorized' });
    return;
  }
  const email = body.email;
  const displayName = body.displayName;
  if (email === undefined && displayName === undefined) {
    res.status(400).json({ error: 'No fields to update' });
    return;
  }
  if (email !== undefined) {
    if (typeof email !== 'string' || !validateEmail(email.trim())) {
      res.status(400).json({ error: 'Invalid email' });
      return;
    }
  }
  if (displayName !== undefined) {
    if (typeof displayName !== 'string' || displayName.trim().length === 0) {
      res.status(400).json({ error: 'Invalid displayName' });
      return;
    }
  }
  const key = String(userId);
  const user = users.get(key);
  if (!user) {
    res.status(404).json({ error: 'User not found' });
    return;
  }
  if (email !== undefined) user.email = email.trim();
  if (displayName !== undefined) user.displayName = displayName.trim();
  users.set(key, user);
  res.json({ id: key, email: user.email, displayName: user.displayName });
}

app.post('/api/user/update-profile', updateProfile);

module.exports = { app, updateProfile, users };
