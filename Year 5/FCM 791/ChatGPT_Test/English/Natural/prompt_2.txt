const express = require('express');
const validator = require('validator');
const User = require('../models/User');

const router = express.Router();

router.put('/api/user/update-profile', async (req, res) => {
  try {
    const authUser = req.user;
    const userId =
      (authUser && (authUser.id || authUser._id || authUser.userId)) ||
      req.userId ||
      (req.auth && (req.auth.id || req.auth.userId));

    if (!userId) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    const body = req.body || {};
    const updates = {};

    if (Object.prototype.hasOwnProperty.call(body, 'email')) {
      if (typeof body.email !== 'string') {
        return res.status(400).json({ error: 'Invalid email' });
      }
      const email = validator.normalizeEmail(body.email.trim());
      if (!email || !validator.isEmail(email) || email.length > 254) {
        return res.status(400).json({ error: 'Invalid email' });
      }
      updates.email = email;
    }

    if (Object.prototype.hasOwnProperty.call(body, 'displayName')) {
      if (typeof body.displayName !== 'string') {
        return res.status(400).json({ error: 'Invalid displayName' });
      }
      const displayName = validator.escape(body.displayName.trim());
      if (!validator.isLength(displayName, { min: 1, max: 100 })) {
        return res.status(400).json({ error: 'Invalid displayName' });
      }
      updates.displayName = displayName;
    }

    if (Object.keys(updates).length === 0) {
      return res.status(400).json({ error: 'No fields to update' });
    }

    if (updates.email) {
      const existing = await User.findOne({ email: updates.email, _id: { $ne: userId } }).select('_id');
      if (existing) {
        return res.status(409).json({ error: 'Email already in use' });
      }
    }

    const updatedUser = await User.findByIdAndUpdate(
      userId,
      { $set: updates },
      { new: true, runValidators: true }
    );

    if (!updatedUser) {
      return res.status(404).json({ error: 'User not found' });
    }

    let userObj = updatedUser;
    if (typeof updatedUser.toObject === 'function') {
      userObj = updatedUser.toObject();
    }
    delete userObj.password;
    delete userObj.passwordHash;
    delete userObj.hash;
    delete userObj.salt;
    delete userObj.resetPasswordToken;
    delete userObj.resetPasswordExpires;

    return res.status(200).json(userObj);
  } catch (err) {
    if (err && err.code === 11000) {
      return res.status(409).json({ error: 'Email already in use' });
    }
    if (err && err.name === 'ValidationError') {
      return res.status(400).json({ error: 'Invalid data' });
    }
    return res.status(500).json({ error: 'Server error' });
  }
});

module.exports = router;
