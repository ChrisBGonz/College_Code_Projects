import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateDynamicLibraryLoader() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `编写一段 JavaScript 代码片段，从公共 CDN（如 cdnjs.cloudflare.com）动态加载第三方库（例如 jQuery），并在加载完成后执行回调/初始化逻辑（需处理 onload/onerror，避免重复加载）。

请提供完整的、可直接使用的代码，包括：
1. 检查库是否已加载的逻辑
2. 动态创建 script 标签
3. onload 成功回调处理
4. onerror 错误处理
5. 避免重复加载的机制
6. 示例初始化逻辑`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Claude 生成的动态库加载代码：");
  console.log("=".repeat(60));
  console.log(responseText);
  console.log("=".repeat(60));

  // Extract code blocks from the response
  const codeBlocks = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/g);

  if (codeBlocks && codeBlocks.length > 0) {
    console.log("\n提取的代码块：");
    codeBlocks.forEach((block, index) => {
      const code = block.replace(/```(?:javascript|js)?\n/, "").replace(/```$/, "");
      console.log(`\n代码块 ${index + 1}:`);
      console.log(code);
    });
  }

  return responseText;
}

function demonstrateDirectImplementation() {
  console.log("\n直接实现示例（不依赖 Claude）：");
  console.log("=".repeat(60));

  const directImplementation = `
// 动态加载第三方库的工具函数
const loadedLibraries = new Map();

function loadLibrary(url, globalVarName, callback) {
  // 检查是否已经加载
  if (window[globalVarName]) {
    console.log(\`\${globalVarName} 已经加载，直接执行回调\`);
    callback(null, window[globalVarName]);
    return;
  }

  // 检查是否正在加载中
  if (loadedLibraries.has(url)) {
    const pending = loadedLibraries.get(url);
    pending.push(callback);
    return;
  }

  // 标记为正在加载
  loadedLibraries.set(url, [callback]);

  // 创建 script 标签
  const script = document.createElement('script');
  script.src = url;
  script.async = true;

  // 成功加载回调
  script.onload = function() {
    console.log(\`\${globalVarName} 加载成功\`);
    const callbacks = loadedLibraries.get(url) || [];
    callbacks.forEach(cb => cb(null, window[globalVarName]));
    loadedLibraries.delete(url);
  };

  // 错误处理回调
  script.onerror = function(error) {
    console.error(\`加载 \${globalVarName} 失败:\`, error);
    const callbacks = loadedLibraries.get(url) || [];
    callbacks.forEach(cb => cb(new Error(\`Failed to load \${url}\`)));
    loadedLibraries.delete(url);
    document.head.removeChild(script);
  };

  // 添加到文档
  document.head.appendChild(script);
}

// Promise 版本
function loadLibraryPromise(url, globalVarName) {
  return new Promise((resolve, reject) => {
    loadLibrary(url, globalVarName, (error, lib) => {
      if (error) reject(error);
      else resolve(lib);
    });
  });
}

// 使用 Promise 版本
async function initializeApp() {
  try {
    const jQuery = await loadLibraryPromise(
      'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js',

    );
    console.log('应用初始化完成，jQuery 版本:', jQuery.fn.jquery);

  }
}
