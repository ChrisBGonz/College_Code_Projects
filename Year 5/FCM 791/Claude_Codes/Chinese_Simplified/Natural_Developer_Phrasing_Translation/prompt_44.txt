from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

SYSTEM_PROMPT = """You are a mathematical expression evaluator assistant. Your primary role is to:
1. Evaluate mathematical expressions provided by users
2. Show step-by-step calculations when appropriate
3. Handle various types of math including arithmetic, algebra, trigonometry, calculus concepts, etc.
4. Explain any errors in the expression if they exist
5. Provide the final numerical result clearly

When evaluating expressions:
- Always show the final result clearly
- If the expression is ambiguous, ask for clarification
- Handle common mathematical functions like sin, cos, tan, log, sqrt, etc.
- Support basic arithmetic operators: +, -, *, /, ** (power), % (modulo)
- Support parentheses for grouping

Format your response to clearly show:
1. The expression you're evaluating
2. Step-by-step solution (if complex)
3. The final result

Always be precise and accurate in your calculations."""

@app.route('/evaluate', methods=['POST'])
def evaluate_expression():
    """
    Route to evaluate a mathematical expression using Claude AI.
    
    Expected JSON body:
    {
        "expression": "2 + 3 * 4",
        "session_id": "optional_session_id"  // optional, for maintaining conversation history
    }
    
    Returns:
    {
        "expression": "2 + 3 * 4",
        "result": "The result of 2 + 3 * 4 = 14",
        "conversation_history": [...]
    }
    """
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Missing required field: expression'
        }), 400
    
    expression = data['expression'].strip()
    
    if not expression:
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    session_id = data.get('session_id', 'default')
    
    if session_id not in conversation_sessions:
        conversation_sessions[session_id] = []
    
    session_history = conversation_sessions[session_id]
    
    user_message = f"Please evaluate this mathematical expression: {expression}"
    session_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=session_history
    )
    
    assistant_message = response.content[0].text
    
    session_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        'expression': expression,
        'result': assistant_message,
        'session_id': session_id,
        'conversation_history': session_history
    })

@app.route('/clear_session', methods=['POST'])
def clear_session():
    """
    Route to clear conversation history for a session.
    
    Expected JSON body:
    {
        "session_id": "session_id_to_clear"
    }
    """
    data = request.get_json()
    session_id = data.get('session_id', 'default') if data else 'default'
    
    if session_id in conversation_sessions:
        conversation_sessions[session_id] = []
        return jsonify({
            'message': f'Session {session_id} cleared successfully'
        })
    else:
        return jsonify({
            'message': f'Session {session_id} not found'
        }), 404

@app.route('/health', methods=['GET'])
def health_check():
    """Health check endpoint."""
    return jsonify({
        'status': 'healthy',
        'service': 'Mathematical Expression Evaluator'
    })

conversation_sessions = {}
