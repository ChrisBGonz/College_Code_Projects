using System;
using System.Security.Cryptography;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;
using System.Text;
using System.Text.Json;

class RSAKeyPairGenerator
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using (RSA rsa = RSA.Create(2048))
        {
            // Export public key
            string publicKey = Convert.ToBase64String(rsa.ExportSubjectPublicKeyInfo());
            
            // Export private key
            string privateKey = Convert.ToBase64String(rsa.ExportPkcs8PrivateKey());
            
            Console.WriteLine("RSA Key Pair Generated Successfully!");
            Console.WriteLine("=====================================\n");
            Console.WriteLine("Public Key (Base64):");
            Console.WriteLine(publicKey);
            Console.WriteLine("\nPrivate Key (Base64):");
            Console.WriteLine(privateKey);
            
            // Use Claude to explain the key pair
            await ExplainKeyPairWithClaude(publicKey, privateKey);
        }
    }
    
    static async Task ExplainKeyPairWithClaude(string publicKey, string privateKey)
    {
        Console.WriteLine("\n\nAsking Claude to explain the RSA key pair...\n");
        
        var client = new AmazonBedrockRuntimeClient(Amazon.RegionEndpoint.USEast1);
        
        var prompt = $@"I've generated an RSA 2048-bit key pair for digital signatures. Here are the details:

Public Key (Base64 encoded SubjectPublicKeyInfo format):
{publicKey.Substring(0, Math.Min(50, publicKey.Length))}... (truncated for brevity)

Private Key (Base64 encoded PKCS#8 format):
{privateKey.Substring(0, Math.Min(50, privateKey.Length))}... (truncated for brevity)

Please explain:
1. What these keys are used for in digital signatures
2. The difference between the public and private key roles
3. Best practices for storing and using these keys securely
4. How RSA digital signatures work at a high level";

        var requestBody = new
        {
            anthropic_version = "bedrock-2023-05-31",
            max_tokens = 1024,
            messages = new[]
            {
                new
                {
                    role = "user",
                    content = prompt
                }
            }
        };
        
        var request = new InvokeModelRequest
        {
            ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0",
            Body = new System.IO.MemoryStream(Encoding.UTF8.GetBytes(JsonSerializer.Serialize(requestBody))),
            ContentType = "application/json",
            Accept = "application/json"
        };
        
        var response = await client.InvokeModelAsync(request);
        
        using var reader = new System.IO.StreamReader(response.Body);
        var responseBody = await reader.ReadToEndAsync();
        
        var jsonDoc = JsonDocument.Parse(responseBody);
        var content = jsonDoc.RootElement
            .GetProperty("content")[0]
            .GetProperty("text")
            .GetString();
        
        Console.WriteLine("Claude's Explanation:");
        Console.WriteLine("====================");
        Console.WriteLine(content);
        
        // Demonstrate signing and verification
        Console.WriteLine("\n\nDemonstrating Digital Signature Usage:");
        Console.WriteLine("======================================");
        DemonstrateSignatureUsage();
    }
    
    static void DemonstrateSignatureUsage()
    {
        using (RSA rsa = RSA.Create(2048))
        {
            // Sample data to sign
            string message = "Hello, this is a test message for digital signature!";
            byte[] messageBytes = Encoding.UTF8.GetBytes(message);
            
            Console.WriteLine($"Original Message: {message}\n");
            
            // Sign the data with private key
            byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            string signatureBase64 = Convert.ToBase64String(signature);
            
            Console.WriteLine($"Digital Signature (Base64): {signatureBase64.Substring(0, 50)}...\n");
            
            // Verify the signature with public key
            bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            
            Console.WriteLine($"Signature Verification Result: {(isValid ? "VALID ✓" : "INVALID ✗")}");
            
            // Try to verify with tampered data
            byte[] tamperedMessage = Encoding.UTF8.GetBytes("Tampered message!");
            bool isTamperedValid = rsa.VerifyData(tamperedMessage, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            
            Console.WriteLine($"Tampered Message Verification: {(isTamperedValid ? "VALID ✓" : "INVALID ✗ (Expected - tampering detected!)")}");
        }
    }
}
