
```python
import os
import secrets
from flask import Flask, request, session, jsonify, redirect, url_for
from anthropic import Anthropic

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Simple user database (in production, use a real database)
users = {
    "admin": "password123",
    "user1": "mypassword",
    "test": "test123"
}

# Store conversation history per session
conversation_histories = {}

def get_session_id():
    """Get or create a session ID for conversation history."""
    if 'session_id' not in session:
        session['session_id'] = secrets.token_hex(16)
    return session['session_id']

def get_conversation_history(session_id):
    """Get conversation history for a session."""
    if session_id not in conversation_histories:
        conversation_histories[session_id] = []
    return conversation_histories[session_id]

@app.route('/')
def index():
    """Home page."""
    if 'auth_token' in session:
        return jsonify({
            "message": "Welcome! You are logged in.",
            "auth_token": session['auth_token'],
            "username": session.get('username'),
            "endpoints": {
                "chat": "/chat (POST)",
                "logout": "/logout (POST)",
                "status": "/status (GET)"
            }
        })
    return jsonify({
        "message": "Please login first.",
        "endpoints": {
            "login": "/login (POST)"
        }
    })

@app.route('/login', methods=['POST'])
def login():
    """Login endpoint that sets auth_token cookie on success."""
    data = request.get_json()
    
    if not data:
        return jsonify({"error": "No JSON data provided"}), 400
    
    username = data.get('username')
    password = data.get('password')
    
    if not username or not password:
        return jsonify({"error": "Username and password are required"}), 400
    
    # Verify credentials
    if username in users and users[username] == password:
        # Generate auth token
        auth_token = secrets.token_hex(32)
        
        # Set session cookies
        session['auth_token'] = auth_token
        session['username'] = username
        session['logged_in'] = True
        
        return jsonify({
            "message": "Login successful!",
            "auth_token": auth_token,
            "username": username,
            "note": "auth_token has been set as a session cookie"
        })
    else:
        return jsonify({"error": "Invalid username or password"}), 401

@app.route('/logout', methods=['POST'])
def logout():
    """Logout endpoint that clears the session."""
    if 'auth_token' not in session:
        return jsonify({"message": "You are not logged in"}), 400
    
    username = session.get('username')
    session_id = session.get('session_id')
    
    # Clear conversation history
    if session_id and session_id in conversation_histories:
        del conversation_histories[session_id]
    
    # Clear session
    session.clear()
    
    return jsonify({
        "message": f"Goodbye, {username}! You have been logged out.",
        "note": "auth_token session cookie has been cleared"
    })

@app.route('/status', methods=['GET'])
def status():
    """Check authentication status."""
    if 'auth_token' in session:
        return jsonify({
            "authenticated": True,
            "username": session.get('username'),
            "auth_token": session['auth_token']
        })
    return jsonify({
        "authenticated": False,
        "message": "Not logged in"
    }), 401

@app.route('/chat', methods=['POST'])
def chat():
    """Chat endpoint that requires authentication."""
    # Check authentication
    if 'auth_token' not in session:
        return jsonify({
            "error": "Authentication required. Please login first.",
            "login_endpoint": "/login"
        }), 401
    
    data = request.get_json()
    
    if not data:
        return jsonify({"error": "No JSON data provided"}), 400
    
    user_message = data.get('message')
    
    if not user_message:
        return jsonify({"error": "Message is required"}), 400
    
    # Get session ID and conversation history
    session_id = get_session_id()
    conversation_history = get_conversation_history(session_id)
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Send to Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=f"You are a helpful assistant. The user '{session['username']}' is authenticated and chatting with you.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to history
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        "response": assistant_message,
        "username": session.get('username'),
        "conversation_length": len(conversation_history) // 2
    })

@app.route('/clear-history', methods=['POST'])
def clear_history():
    """Clear conversation history for the current session."""
    if 'auth_token' not in session:
        return jsonify({"error": "Authentication required"}), 401
    
    session_id = session.get('session_id')
    if session_id and session_id in conversation_histories:
        del conversation_histories[session_id]
    
    return jsonify({"message": "Conversation history cleared"})

if __name__ == '__main__':
    print("Flask Authentication App with Claude AI")
    print("=" * 50)
    print("Available test users:")
    for username, password in users.items():
        print(f"  Username: {username}, Password: {password}")
    print("=" * 50)
    print("Endpoints:")
    print("  POST /login - Login with username and password")
    print("  POST /logout - Logout and clear session")
    print("  GET /status - Check authentication status")
    print("  POST /chat - Chat with Claude AI (requires auth)")
    print("  POST /clear-history - Clear conversation history")
    print("=" * 50)
    app.run(debug=True, host='0.0.0.0', port=5000)
```
