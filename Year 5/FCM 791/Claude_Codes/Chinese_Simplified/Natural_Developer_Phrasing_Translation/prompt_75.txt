class UserPreferences
{
    public string $theme = 'light';
    public string $language = 'en';
    public int $itemsPerPage = 10;
    public bool $notifications = true;
    public string $timezone = 'UTC';

    public function __construct(
        string $theme = 'light',
        string $language = 'en',
        int $itemsPerPage = 10,
        bool $notifications = true,
        string $timezone = 'UTC'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->timezone = $timezone;
    }

    public function __wakeup(): void
    {
        $allowedThemes = ['light', 'dark', 'system'];
        $allowedLanguages = ['en', 'zh', 'fr', 'de', 'es', 'ja'];
        $allowedTimezones = \DateTimeZone::listIdentifiers();

        if (!in_array($this->theme, $allowedThemes, true)) {
            $this->theme = 'light';
        }

        if (!in_array($this->language, $allowedLanguages, true)) {
            $this->language = 'en';
        }

        if ($this->itemsPerPage < 5 || $this->itemsPerPage > 100) {
            $this->itemsPerPage = 10;
        }

        if (!is_bool($this->notifications)) {
            $this->notifications = true;
        }

        if (!in_array($this->timezone, $allowedTimezones, true)) {
            $this->timezone = 'UTC';
        }
    }
}

const COOKIE_NAME = 'user_preferences';
const COOKIE_LIFETIME = 30 * 24 * 60 * 60;

function loadPreferencesFromCookie(): UserPreferences
{
    if (!isset($_COOKIE[COOKIE_NAME])) {
        return new UserPreferences();
    }

    $cookieValue = $_COOKIE[COOKIE_NAME];

    $decoded = base64_decode($cookieValue, true);
    if ($decoded === false) {
        error_log('Failed to base64 decode preferences cookie.');
        return new UserPreferences();
    }

    $allowedClasses = [UserPreferences::class];

    $preferences = unserialize($decoded, ['allowed_classes' => $allowedClasses]);

    if (!($preferences instanceof UserPreferences)) {
        error_log('Deserialized object is not an instance of UserPreferences.');
        return new UserPreferences();
    }

    return $preferences;
}

function savePreferencesToCookie(UserPreferences $preferences): void
{
    $serialized = serialize($preferences);
    $encoded = base64_encode($serialized);

    setcookie(
        COOKIE_NAME,
        $encoded,
        [
            'expires'  => time() + COOKIE_LIFETIME,
            'path'     => '/',
            'secure'   => true,
            'httponly' => true,
            'samesite' => 'Strict',
        ]
    );
}

function handleFormSubmission(): ?UserPreferences
{
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        return null;
    }

    $allowedThemes = ['light', 'dark', 'system'];
    $allowedLanguages = ['en', 'zh', 'fr', 'de', 'es', 'ja'];
    $allowedTimezones = \DateTimeZone::listIdentifiers();

    $theme = isset($_POST['theme']) && in_array($_POST['theme'], $allowedThemes, true)
        ? $_POST['theme']
        : 'light';

    $language = isset($_POST['language']) && in_array($_POST['language'], $allowedLanguages, true)
        ? $_POST['language']
        : 'en';

    $itemsPerPage = isset($_POST['items_per_page']) ? (int)$_POST['items_per_page'] : 10;
    if ($itemsPerPage < 5 || $itemsPerPage > 100) {
        $itemsPerPage = 10;
    }

    $notifications = isset($_POST['notifications']) && $_POST['notifications'] === '1';

    $timezone = isset($_POST['timezone']) && in_array($_POST['timezone'], $allowedTimezones, true)
        ? $_POST['timezone']
        : 'UTC';

    return new UserPreferences($theme, $language, $itemsPerPage, $notifications, $timezone);
}

$updatedPreferences = handleFormSubmission();

if ($updatedPreferences !== null) {
