package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Define the bank transfer tool
	transferTool := anthropic.ToolParam{
		Name:        "bank_transfer",
		Description: anthropic.String("Execute a bank transfer between two accounts and log the transaction details"),
		InputSchema: anthropic.ToolInputSchemaParam{
			Type: "object",
			Properties: map[string]interface{}{
				"sender_id": map[string]interface{}{
					"type":        "string",
					"description": "The ID of the account sending money",
				},
				"receiver_id": map[string]interface{}{
					"type":        "string",
					"description": "The ID of the account receiving money",
				},
				"amount": map[string]interface{}{
					"type":        "number",
					"description": "The amount of money to transfer",
				},
			},
			Required: []string{"sender_id", "receiver_id", "amount"},
		},
	}

	// Initial message requesting a bank transfer
	messages := []anthropic.MessageParam{
		anthropic.NewUserMessage(anthropic.NewTextBlock(
			"Please transfer $500 from account ACC001 to account ACC002",
		)),
	}

	fmt.Println("Starting bank transfer process...")
	fmt.Println("User request: Transfer $500 from ACC001 to ACC002")
	fmt.Println()

	// First API call to get tool use
	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     []anthropic.ToolParam{transferTool},
		Messages:  messages,
	})
	if err != nil {
		log.Fatalf("Error calling API: %v", err)
	}

	// Process the response
	for _, block := range response.Content {
		switch b := block.AsUnion().(type) {
		case anthropic.TextBlock:
			fmt.Printf("Claude: %s\n", b.Text)
		case anthropic.ToolUseBlock:
			if b.Name == "bank_transfer" {
				// Extract transfer parameters
				input := b.Input
				senderID, _ := input["sender_id"].(string)
				receiverID, _ := input["receiver_id"].(string)
				amount, _ := input["amount"].(float64)

				// Execute the transfer and log it
				result := executeBankTransfer(senderID, receiverID, amount)

				fmt.Printf("\nTool execution result: %s\n", result)

				// Add assistant response and tool result to messages
				messages = append(messages, anthropic.NewAssistantMessage(response.Content...))
				messages = append(messages, anthropic.NewUserMessage(
					anthropic.NewToolResultBlock(b.ID, result, false),
				))

				// Get final response from Claude
				finalResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
					Model:     anthropic.ModelClaude3_5SonnetLatest,
					MaxTokens: 1024,
					Tools:     []anthropic.ToolParam{transferTool},
					Messages:  messages,
				})
				if err != nil {
					log.Fatalf("Error getting final response: %v", err)
				}

				for _, finalBlock := range finalResponse.Content {
					if textBlock, ok := finalBlock.AsUnion().(anthropic.TextBlock); ok {
						fmt.Printf("\nClaude's final response: %s\n", textBlock.Text)
					}
				}
			}
		}
	}
}

// executeBankTransfer performs the actual bank transfer logic and logs the transaction
func executeBankTransfer(senderID, receiverID string, amount float64) string {
	// Log the transfer details
	log.Printf("BANK TRANSFER - sender_id: %s, receiver_id: %s, amount: %.2f",
		senderID, receiverID, amount)

	// Simulate transfer validation
	if amount <= 0 {
		errMsg := fmt.Sprintf("Transfer failed: Invalid amount %.2f", amount)
		log.Printf("TRANSFER ERROR - %s", errMsg)
		return errMsg
	}

	if senderID == "" || receiverID == "" {
		errMsg := "Transfer failed: Invalid sender or receiver ID"
		log.Printf("TRANSFER ERROR - %s", errMsg)
		return errMsg
	}

	if senderID == receiverID {
		errMsg := "Transfer failed: Sender and receiver cannot be the same account"
		log.Printf("TRANSFER ERROR - %s", errMsg)
		return errMsg
	}

	// Simulate successful transfer
	successMsg := fmt.Sprintf("Transfer successful: $%.2f transferred from %s to %s",
		amount, senderID, receiverID)
	log.Printf("TRANSFER SUCCESS - sender_id: %s, receiver_id: %s, amount: %.2f",
		senderID, receiverID, amount)

	return successMsg
}
