import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

interface UserProfile {
  email?: string;
  displayName?: string;
}

const conversationHistory: Message[] = [];
let currentProfile: UserProfile = {};

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful assistant for updating user profiles. You help users update their email and display name.
    
    Current profile state: ${JSON.stringify(currentProfile)}
    
    When a user wants to update their profile:
    1. Ask for the information they want to update (email and/or display name)
    2. Validate the email format if they provide one
    3. Confirm the changes before applying
    4. When confirmed, respond with a JSON object in this exact format:
       {"action": "update_profile", "email": "new@email.com", "displayName": "New Name"}
    
    Only include fields that are being updated in the JSON response.
    Always be conversational and helpful.`,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // Check if the response contains a profile update action
  const jsonMatch = assistantMessage.match(
    /\{"action":\s*"update_profile"[^}]*\}/
  );
  if (jsonMatch) {
    try {
      const updateData = JSON.parse(jsonMatch[0]);
      if (updateData.action === "update_profile") {
        // Simulate the API endpoint call
        const result = await simulateUpdateProfileAPI(updateData);
        return assistantMessage + "\n\n" + result;
      }
    } catch {
      // JSON parsing failed, continue with normal response
    }
  }

  return assistantMessage;
}

async function simulateUpdateProfileAPI(updateData: {
  email?: string;
  displayName?: string;
}): Promise<string> {
  // Simulate the /api/user/update-profile endpoint
  const updates: UserProfile = {};

  if (updateData.email) {
    // Validate email format
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(updateData.email)) {
      return "❌ Error: Invalid email format";
    }
    updates.email = updateData.email;
  }

  if (updateData.displayName) {
    if (updateData.displayName.length < 2) {
      return "❌ Error: Display name must be at least 2 characters";
    }
    if (updateData.displayName.length > 50) {
      return "❌ Error: Display name must be less than 50 characters";
    }
    updates.displayName = updateData.displayName;
  }

  // Apply updates to current profile
  currentProfile = { ...currentProfile, ...updates };

  // Simulate successful API response
  return `✅ Profile updated successfully!\nUpdated profile: ${JSON.stringify(currentProfile, null, 2)}`;
}

async function main() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("Profile Update Assistant");
  console.log("========================");
  console.log(
    "I can help you update your email and display name. Type 'exit' to quit.\n"
  );

  // Initial greeting
  const greeting = await chat(
    "Hello! I want to update my profile. What can I change?"
  );
  console.log(`Assistant: ${greeting}\n`);

  const askQuestion = () => {
    rl.question("You: ", async (input) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        const response = await chat(userInput);
        console.log(`\nAssistant: ${response}\n`);
      } catch (error) {
        console.error("Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

main();
