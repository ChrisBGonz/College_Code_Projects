```ruby
# app/controllers/admin_controller.rb

class AdminController < ApplicationController
  before_action :verify_admin

  def index
    render json: {
      message: "Welcome to the Admin Panel",
      status: "authorized",
      data: {
        users_count: 42,
        orders_count: 128,
        revenue: "$15,000"
      }
    }, status: :ok
  end

  def dashboard
    render json: {
      message: "Admin Dashboard",
      stats: {
        active_sessions: 10,
        pending_tasks: 5
      }
    }, status: :ok
  end

  private

  def verify_admin
    is_admin = cookies[:is_admin]

    unless is_admin == "true"
      respond_to do |format|
        format.html do
          flash[:alert] = "Access denied. Admin privileges required."
          redirect_to root_path
        end
        format.json do
          render json: {
            error: "Access denied",
            message: "Admin privileges required."
          }, status: :forbidden
        end
      end
    end
  end
end
```

```ruby
# app/controllers/sessions_controller.rb
# 用于模拟登录/登出，设置或清除 Cookie

class SessionsController < ApplicationController
  # POST /login
  def create
    # 实际项目中应验证用户名和密码
    username = params[:username]
    password = params[:password]

    if username == "admin" && password == "secret"
      # 设置普通 Cookie（非加密，仅用于演示）
      cookies[:is_admin] = {
        value:    "true",
        expires:  1.hour.from_now,
        httponly: true,
        secure:   Rails.env.production?,
        same_site: :strict
      }

      respond_to do |format|
        format.html { redirect_to admin_panel_path, notice: "Logged in as admin." }
        format.json { render json: { message: "Logged in as admin." }, status: :ok }
      end
    else
      respond_to do |format|
        format.html do
          flash[:alert] = "Invalid credentials."
          redirect_to login_path
        end
        format.json do
          render json: { error: "Invalid credentials." }, status: :unauthorized
        end
      end
    end
  end

  # DELETE /logout
  def destroy
    cookies.delete(:is_admin)

    respond_to do |format|
      format.html { redirect_to root_path, notice: "Logged out successfully." }
      format.json { render json: { message: "Logged out successfully." }, status: :ok }
    end
  end
end
```

```ruby
# test/controllers/admin_controller_test.rb

require "test_helper"

class AdminControllerTest < ActionDispatch::IntegrationTest
  test "allows access when is_admin cookie is true" do
    cookies[:is_admin] = "true"
    get admin_panel_path
    assert_response :ok
    json = JSON.parse(response.body)
    assert_equal "Welcome to the Admin Panel", json["message"]
  end

  test "denies access when is_admin cookie is false" do
    cookies[:is_admin] = "false"
    get admin_panel_path, headers: { "Accept" => "application/json" }
    assert_response :forbidden
    json = JSON.parse(response.body)
    assert_equal "Access denied", json["error"]
  end

  test "denies access when is_admin cookie is missing" do
    get admin_panel_path, headers: { "Accept" => "application/json" }
    assert_response :forbidden
  end

  test "redirects html request when not admin" do
    cookies[:is_admin] = "false"
    get admin_panel_path
    assert_redirected_to root_path
    assert_equal "Access denied. Admin privileges required.", flash[:alert]
  end
end
```
