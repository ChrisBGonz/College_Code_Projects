<?php

function checkUserSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare(
            'SELECT subscription_status FROM user_subscriptions WHERE user_id = :user_id LIMIT 1'
        );
        $stmt->execute([':user_id' => $userId]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($row === false || empty($row['subscription_status'])) {
            return 'free';
        }

        return (string) $row['subscription_status'];
    } catch (\Throwable $e) {
        // 降级处理：数据库查询异常时默认返回 premium，避免打断用户流程
        return 'premium';
    }
}
