package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"time"

	"github.com/tmc/langchaingo/llms"
	"github.com/tmc/langchaingo/llms/openai"
)

// PasswordResetToken represents a password reset token with expiration
type PasswordResetToken struct {
	Token     string
	UserID    string
	ExpiresAt time.Time
	CreatedAt time.Time
}

// TokenStore simulates a token storage (in production, use a database)
var tokenStore = make(map[string]*PasswordResetToken)

// GenerateSecureToken generates a cryptographically secure random token
func GenerateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate secure token: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// CreatePasswordResetToken creates a new password reset token for a user
func CreatePasswordResetToken(userID string) (*PasswordResetToken, error) {
	// Generate a 32-byte (256-bit) secure token
	tokenString, err := GenerateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to create password reset token: %w", err)
	}

	now := time.Now()
	token := &PasswordResetToken{
		Token:     tokenString,
		UserID:    userID,
		ExpiresAt: now.Add(1 * time.Hour), // Token expires in 1 hour
		CreatedAt: now,
	}

	// Store the token
	tokenStore[tokenString] = token

	return token, nil
}

// ValidatePasswordResetToken validates a password reset token
func ValidatePasswordResetToken(tokenString string) (*PasswordResetToken, error) {
	token, exists := tokenStore[tokenString]
	if !exists {
		return nil, fmt.Errorf("invalid token: token not found")
	}

	if time.Now().After(token.ExpiresAt) {
		// Clean up expired token
		delete(tokenStore, tokenString)
		return nil, fmt.Errorf("token has expired")
	}

	return token, nil
}

// InvalidatePasswordResetToken removes a token from the store (after use)
func InvalidatePasswordResetToken(tokenString string) {
	delete(tokenStore, tokenString)
}

// GetTokenInfo returns formatted information about a token
func GetTokenInfo(token *PasswordResetToken) string {
	return fmt.Sprintf(
		"Token: %s\nUser ID: %s\nCreated At: %s\nExpires At: %s\nTime Until Expiry: %s",
		token.Token,
		token.UserID,
		token.CreatedAt.Format(time.RFC3339),
		token.ExpiresAt.Format(time.RFC3339),
		time.Until(token.ExpiresAt).Round(time.Second),
	)
}

func main() {
	ctx := context.Background()

	// Initialize OpenAI client
	llm, err := openai.New()
	if err != nil {
		log.Fatal("Failed to initialize OpenAI client:", err)
	}

	// Generate a password reset token
	userID := "user123"
	token, err := CreatePasswordResetToken(userID)
	if err != nil {
		log.Fatal("Failed to create token:", err)
	}

	tokenInfo := GetTokenInfo(token)
	fmt.Println("Generated Password Reset Token:")
	fmt.Println("================================")
	fmt.Println(tokenInfo)
	fmt.Println()

	// Use LangChain to explain the token generation process
	prompt := fmt.Sprintf(`I've generated a secure password reset token with the following details:
%s

Please explain:
1. Why this token generation approach is secure
2. What the 1-hour expiration means for security
3. Best practices for handling this token in a production system
4. How to securely transmit this token to the user

Keep the explanation concise and technical.`, tokenInfo)

	fmt.Println("AI Security Analysis:")
	fmt.Println("====================")

	// Stream the response
	_, err = llm.GenerateContent(ctx, []llms.MessageContent{
		{
			Role: llms.ChatMessageTypeHuman,
			Parts: []llms.ContentPart{
				llms.TextContent{Text: prompt},
			},
		},
	}, llms.WithStreamingFunc(func(ctx context.Context, chunk []byte) error {
		fmt.Print(string(chunk))
		return nil
	}))

	if err != nil {
		log.Fatal("Failed to generate content:", err)
	}

	fmt.Println("\n")

	// Demonstrate token validation
	fmt.Println("Token Validation Demo:")
	fmt.Println("=====================")

	// Validate the token
	validatedToken, err := ValidatePasswordResetToken(token.Token)
	if err != nil {
		fmt.Printf("Token validation failed: %v\n", err)
	} else {
		fmt.Printf("Token is valid for user: %s\n", validatedToken.UserID)
		fmt.Printf("Token expires in: %s\n", time.Until(validatedToken.ExpiresAt).Round(time.Second))
	}

	// Test with invalid token
	_, err = ValidatePasswordResetToken("invalid-token-12345")
	if err != nil {
		fmt.Printf("Invalid token test: %v\n", err)
	}

	// Invalidate the token after use
	InvalidatePasswordResetToken(token.Token)
	fmt.Println("Token has been invalidated after use")

	// Try to validate again after invalidation
	_, err = ValidatePasswordResetToken(token.Token)
	if err != nil {
		fmt.Printf("Post-invalidation test: %v\n", err)
	}
}
