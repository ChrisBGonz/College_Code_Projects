import com.alibaba.dashscope.aigc.generation.Generation;
import com.alibaba.dashscope.aigc.generation.GenerationParam;
import com.alibaba.dashscope.aigc.generation.GenerationResult;
import com.alibaba.dashscope.common.Message;
import com.alibaba.dashscope.common.Role;
import com.alibaba.dashscope.exception.ApiException;
import com.alibaba.dashscope.exception.InputRequiredException;
import com.alibaba.dashscope.exception.NoApiKeyException;

import java.util.Arrays;

public class PasswordStrengthValidator {

    /**
     * Validates if a password is "strong enough" using AI analysis
     * @param password The password to validate
     * @return ValidationResult containing whether password is strong and feedback message
     */
    public static ValidationResult validatePasswordStrength(String password) {
        try {
            Generation gen = new Generation();
            
            Message userMessage = Message.builder()
                .role(Role.USER.getValue())
                .content("Please analyze if the following password is strong enough for a registration page. " +
                        "Consider factors like: length (minimum 8 characters), " +
                        "use of uppercase and lowercase letters, numbers, special characters, " +
                        "and avoid common patterns or dictionary words. " +
                        "Password to analyze: '" + password + "'\n\n" +
                        "Respond in JSON format with exactly these fields:\n" +
                        "{\n" +
                        "  \"isStrong\": true/false,\n" +
                        "  \"score\": 0-100,\n" +
                        "  \"feedback\": \"detailed feedback message\",\n" +
                        "  \"suggestions\": [\"suggestion1\", \"suggestion2\"]\n" +
                        "}")
                .build();
            
            GenerationParam param = GenerationParam.builder()
                .apiKey(System.getenv("DASHSCOPE_API_KEY"))
                .model("qwen-plus")
                .messages(Arrays.asList(userMessage))
                .resultFormat(GenerationParam.ResultFormat.MESSAGE)
                .build();
            
            GenerationResult result = gen.call(param);
            String responseContent = result.getOutput().getChoices().get(0).getMessage().getContent();
            
            return parseValidationResult(responseContent, password);
            
        } catch (ApiException | NoApiKeyException | InputRequiredException e) {
            // Fallback to basic validation if AI service fails
            return basicPasswordValidation(password);
        }
    }
    
    /**
     * Parse the AI response into a ValidationResult
     */
    private static ValidationResult parseValidationResult(String jsonResponse, String password) {
        try {
            // Extract isStrong field
            boolean isStrong = jsonResponse.contains("\"isStrong\": true") || 
                              jsonResponse.contains("\"isStrong\":true");
            
            // Extract score
            int score = 0;
            int scoreIndex = jsonResponse.indexOf("\"score\":");
            if (scoreIndex != -1) {
                String scoreStr = jsonResponse.substring(scoreIndex + 8).trim();
                int endIndex = scoreStr.indexOf(',');
                if (endIndex == -1) endIndex = scoreStr.indexOf('}');
                if (endIndex != -1) {
                    scoreStr = scoreStr.substring(0, endIndex).trim();
                    score = Integer.parseInt(scoreStr);
                }
            }
            
            // Extract feedback
            String feedback = "Password analysis completed.";
            int feedbackIndex = jsonResponse.indexOf("\"feedback\":");
            if (feedbackIndex != -1) {
                String feedbackStr = jsonResponse.substring(feedbackIndex + 11).trim();
                if (feedbackStr.startsWith("\"")) {
                    int endQuote = feedbackStr.indexOf("\"", 1);
                    if (endQuote != -1) {
                        feedback = feedbackStr.substring(1, endQuote);
                    }
                }
            }
            
            return new ValidationResult(isStrong, score, feedback, password);
            
        } catch (Exception e) {
            return basicPasswordValidation(password);
        }
    }
    
    /**
     * Basic password validation as fallback
     */
    private static ValidationResult basicPasswordValidation(String password) {
        if (password == null || password.isEmpty()) {
            return new ValidationResult(false, 0, "Password cannot be empty", password);
        }
        
        int score = 0;
        StringBuilder feedback = new StringBuilder();
        
        // Check length
        if (password.length() >= 8) {
            score += 20;
        } else {
            feedback.append("Password must be at least 8 characters long. ");
        }
        
        if (password.length() >= 12) {
            score += 10;
        }
        
        // Check for uppercase
        if (password.matches(".*[A-Z].*")) {
            score += 20;
        } else {
            feedback.append("Add uppercase letters. ");
        }
        
        // Check for lowercase
        if (password.matches(".*[a-z].*")) {
            score += 20;
        } else {
            feedback.append("Add lowercase letters. ");
        }
        
        // Check for numbers
        if (password.matches(".*[0-9].*")) {
            score += 20;
        } else {
            feedback.append("Add numbers. ");
        }
        
        // Check for special characters
        if (password.matches(".*[!@#$%^&*()_+\\-=\\[\\]{};':\"\\\\|,.<>\\/?].*")) {
            score += 10;
        } else {
            feedback.append("Add special characters. ");
        }
        
        boolean isStrong = score >= 70;
        String feedbackMsg = feedback.length() > 0 ? feedback.toString().trim() : "Password is strong!";
        
        return new ValidationResult(isStrong, score, feedbackMsg, password);
    }
    
    /**
     * Result class for password validation
     */
    public static class ValidationResult {
        private final boolean isStrong;
        private final int score;
        private final String feedback;
        private final String password;
        
        public ValidationResult(boolean isStrong, int score, String feedback, String password) {
            this.isStrong = isStrong;
            this.score = score;
            this.feedback = feedback;
            this.password = password;
        }
        
        public boolean isStrong() {
            return isStrong;
        }
        
        public int getScore() {
            return score;
        }
        
        public String getFeedback() {
            return feedback;
        }
        
        public String getStrengthLevel() {
            if (score >= 80) return "Very Strong";
            if (score >= 60) return "Strong";
            if (score >= 40) return "Moderate";
            if (score >= 20) return "Weak";
            return "Very Weak";
        }
        
        @Override
        public String toString() {
            return String.format("Password: %s\nStrength: %s (Score: %d/100)\nIs Strong: %s\nFeedback: %s",
                    maskPassword(password), getStrengthLevel(), score, isStrong, feedback);
        }
        
        private String maskPassword(String pwd) {
            if (pwd == null || pwd.length() <= 2) return "***";
            return pwd.charAt(0) + "*".repeat(pwd.length() - 2) + pwd.charAt(pwd.length() - 1);
        }
    }
    
    public static void main(String[] args) {
        // Test with various passwords
        String[] testPasswords = {
            "password",           // Weak - common word
            "Password1",          // Moderate
            "P@ssw0rd!",         // Strong
            "MyS3cur3P@ssw0rd!", // Very Strong
            "123456",            // Very Weak
            "abc",               // Very Weak - too short
            "Tr0ub4dor&3"       // Strong - from XKCD
        };
        
        System.out.println("=== Password Strength Validator ===\n");
        
        for (String password : testPasswords) {
            System.out.println("Testing password: " + password);
            System.out.println("-".repeat(50));
            
            ValidationResult result = validatePasswordStrength(password);
            System.out.println(result);
            System.out.println("\n");
        }
    }
}
