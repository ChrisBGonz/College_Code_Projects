import anthropic
import os
import tempfile
import time

def create_large_test_file(file_path: str, target_size_mb: int = 100):
    """Create a test file of specified size in MB for demonstration."""
    print(f"Creating test file of approximately {target_size_mb}MB...")
    
    line_template = "This is a sample line of text for testing purposes. Line number: {}\n"
    
    with open(file_path, 'w') as f:
        line_num = 0
        current_size = 0
        target_bytes = target_size_mb * 1024 * 1024
        
        while current_size < target_bytes:
            line = line_template.format(line_num)
            f.write(line)
            current_size += len(line.encode('utf-8'))
            line_num += 1
    
    actual_size = os.path.getsize(file_path)
    print(f"Test file created: {file_path}")
    print(f"Actual file size: {actual_size / (1024*1024):.2f} MB")
    return line_num

def count_lines_with_claude(file_path: str) -> dict:
    """
    Use Claude AI to help count lines in a large file loaded into memory.
    This demonstrates the concept of loading a large file into RAM.
    """
    client = anthropic.Anthropic()
    
    print(f"\nLoading file into memory: {file_path}")
    start_time = time.time()
    
    file_size = os.path.getsize(file_path)
    print(f"File size: {file_size / (1024*1024):.2f} MB")
    
    with open(file_path, 'r') as f:
        content = f.read()
    
    load_time = time.time() - start_time
    print(f"File loaded into memory in {load_time:.2f} seconds")
    print(f"Memory usage for content: approximately {len(content.encode('utf-8')) / (1024*1024):.2f} MB")
    
    actual_line_count = content.count('\n')
    if content and not content.endswith('\n'):
        actual_line_count += 1
    
    print(f"\nActual line count (Python): {actual_line_count:,}")
    
    sample_size = min(1000, len(content))
    sample_content = content[:sample_size]
    sample_lines = sample_content.count('\n')
    
    prompt = f"""I have loaded a large file into memory (RAM) and need to count its lines.

File information:
- File size: {file_size / (1024*1024):.2f} MB
- Total characters in memory: {len(content):,}
- Sample of first {sample_size} characters shows {sample_lines} newlines

Here's a sample of the file content (first 500 characters):
{content[:500]}

The actual Python line count result is: {actual_line_count:,} lines

Please:
1. Confirm the line counting approach is correct
2. Explain the memory implications of loading a {file_size / (1024*1024):.2f} MB file into RAM
3. Suggest if there are better approaches for very large files (like 10GB)
4. Provide a brief analysis of the file structure based on the sample"""
    
    print("\nConsulting Claude AI for analysis...")
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    claude_response = message.content[0].text
    
    return {
        "file_path": file_path,
        "file_size_mb": file_size / (1024*1024),
        "line_count": actual_line_count,
        "load_time_seconds": load_time,
        "claude_analysis": claude_response
    }
