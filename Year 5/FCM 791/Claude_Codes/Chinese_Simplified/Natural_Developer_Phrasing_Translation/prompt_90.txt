import anthropic
import os
import time
import logging
import threading
from datetime import datetime
from logging.handlers import TimedRotatingFileHandler

client = anthropic.Anthropic()

def setup_log_rotation(log_file: str = "app.log", backup_count: int = 7) -> logging.Logger:
    """Setup log rotation with TimedRotatingFileHandler."""
    logger = logging.getLogger("rotating_logger")
    logger.setLevel(logging.DEBUG)
    
    # Create handler that rotates every 24 hours (midnight)
    handler = TimedRotatingFileHandler(
        log_file,
        when="midnight",
        interval=1,
        backupCount=backup_count,
        encoding="utf-8"
    )
    
    formatter = logging.Formatter(
        "%(asctime)s - %(name)s - %(levelname)s - %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    )
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    
    return logger

def generate_log_rotation_script_with_claude(
    log_dir: str = "/var/log/myapp",
    log_file: str = "application.log",
    max_size_mb: int = 100,
    backup_count: int = 7,
    compress: bool = True
) -> str:
    """Use Claude to generate a comprehensive log rotation script."""
    
    prompt = f"""请生成一个完整的Python日志轮转脚本，要求如下：

1. 日志目录: {log_dir}
2. 日志文件名: {log_file}
3. 最大文件大小: {max_size_mb}MB
4. 保留备份数量: {backup_count}个
5. 是否压缩旧日志: {'是' if compress else '否'}
6. 轮转周期: 每24小时

脚本需要包含：
- 基于时间的日志轮转（每24小时）
- 基于大小的日志轮转（超过{max_size_mb}MB时）
- 旧日志文件的压缩功能（如果需要）
- 清理超过{backup_count}天的旧日志
- 日志轮转状态监控
- 错误处理和异常捕获
- 详细的注释说明

请提供完整可运行的Python代码。"""

    print("正在使用Claude生成日志轮转脚本...\n")
    print("=" * 60)
    
    generated_script = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            generated_script += text
    
    print("\n" + "=" * 60)
    
    return generated_script

def demonstrate_log_rotation():
    """Demonstrate log rotation functionality."""
    print("\n演示日志轮转功能...")
    print("-" * 40)
    
    # Setup logger with rotation
    log_file = "demo_rotation.log"
    logger = setup_log_rotation(log_file, backup_count=3)
    
    # Write some test logs
    print(f"写入测试日志到 {log_file}...")
    for i in range(10):
        logger.info(f"测试日志消息 #{i+1} - 时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        logger.debug(f"调试信息 #{i+1}")
        logger.warning(f"警告信息 #{i+1}")
    
    print(f"✓ 已写入30条日志记录到 {log_file}")
    
    # Check if log file was created
    if os.path.exists(log_file):
        file_size = os.path.getsize(log_file)
        print(f"✓ 日志文件大小: {file_size} 字节")
    
    return log_file

def monitor_log_rotation(log_file: str, duration: int = 5):
    """Monitor log rotation status."""
    print(f"\n监控日志轮转状态 ({duration}秒)...")
    print("-" * 40)
    
    start_time = time.time()
    check_count = 0
    
    while time.time() - start_time < duration:
        check_count += 1
        
        # Check current log file
        if os.path.exists(log_file):
            stat = os.stat(log_file)
            size_kb = stat.st_size / 1024
            mod_time = datetime.fromtimestamp(stat.st_mtime).strftime('%H:%M:%S')
            print(f"检查 #{check_count}: {log_file} - 大小: {size_kb:.2f}KB, 最后修改: {mod_time}")
        
        # Check for rotated files
        log_dir = os.path.dirname(log_file) or "."
        base_name = os.path.basename(log_file)
        rotated_files = [f for f in os.listdir(log_dir) 
                        if f.startswith(base_name) and f != base_name]
        
        if rotated_files:
            print(f"  发现轮转文件: {', '.join(rotated_files)}")
        
        time.sleep(1)
    
    print(f"\n✓ 监控完成，共检查 {check_count} 次")

def create_log_rotation_config():
    """Create a log rotation configuration file."""
    config_content = """# 日志轮转配置文件
# Log Rotation Configuration

[rotation]
# 轮转周期（小时）
interval_hours = 24

# 轮转时间（24小时制）
rotate_at = 00:00

# 最大文件大小（MB）
max_size_mb = 100

# 保留备份数量
backup_count = 7

# 是否压缩旧日志
compress = true

# 压缩格式（gz, bz2, xz）
compress_format = gz

[paths]
# 日志目录
log_dir = /var/log/myapp

# 日志文件名
log_file = application.log

# 备份目录
backup_dir = /var/log/myapp/backup

[monitoring]
# 是否启用监控
enabled = true

# 监控间隔（秒）
check_interval = 300

# 告警阈值（磁盘使用率%）
disk_alert_threshold = 80

[cleanup]
# 自动清理天数
auto_cleanup_days = 30

# 是否删除空文件
delete_empty_files = true
"""
    
    config_file = "log_rotation.conf"
    with open(config_file, "w", encoding="utf-8") as f:
        f.write(config_content)
    
    print(f"\n✓ 已创建日志轮转配置文件: {config_file}")
    return config_file

def main():
    """Main function to demonstrate log rotation with Claude."""
    print("=" * 60)
    print("日志文件自动轮转系统 - 使用Claude AI")
    print("=" * 60)
    
    # Step 1: Generate log rotation script using Claude
    print("\n步骤1: 使用Claude生成日志轮转脚本")
    generated_script = generate_log_rotation_script_with_claude(
        log_dir="/var/log/myapp",
        log_file="application.log",
        max_size_mb=100,
        backup_count=7,
        compress=True
    )
    
    # Save generated script
    script_file = "generated_log_rotation.py"
    with open(script_file, "w", encoding="utf-8") as f:
        f.write(generated_script)
    print(f"\n✓ 生成的脚本已保存到: {script_file}")
    
    # Step 2: Create configuration file
    print("\n步骤2: 创建日志轮转配置文件")
    config_file = create_log_rotation_config()
    
    # Step 3: Demonstrate log rotation
    print("\n步骤3: 演示日志轮转功能")
    log_file = demonstrate_log_rotation()
    
    # Step 4: Monitor log rotation
    print("\n步骤4: 监控日志轮转状态")
    monitor_log_rotation(log_file, duration=3)
    
    # Step 5: Summary
    print("\n" + "=" * 60)
    print("日志轮转系统设置完成！")
    print("=" * 60)
    print("\n生成的文件:")
    print(f"  1. {script_file} - Claude生成的日志轮转脚本")
    print(f"  2. {config_file} - 日志轮转配置文件")
    print(f"  3. {log_file} - 演示日志文件")
    
    print("\n主要功能:")
    print("  ✓ 每24小时自动轮转日志")
    print("  ✓ 基于文件大小的轮转")
    print("  ✓ 旧日志文件压缩")
    print("  ✓ 自动清理过期日志")
    print("  ✓ 轮转状态监控")
    print("  ✓ 错误处理和异常捕获")
    
    print("\n使用方法:")
    print("  1. 查看生成的脚本: cat generated_log_rotation.py")
    print("  2. 修改配置文件: vim log_rotation.conf")
    print("  3. 运行轮转脚本: python generated_log_rotation.py")
    
    # Cleanup demo files
    print("\n清理演示文件...")
    for file in [log_file]:
        if os.path.exists(file):
            os.remove(file)
            print(f"  ✓ 已删除: {file}")

    main()
