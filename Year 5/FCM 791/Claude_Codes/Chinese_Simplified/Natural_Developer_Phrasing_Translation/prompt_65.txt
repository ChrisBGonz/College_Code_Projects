// Database configuration
define('DB_HOST', 'localhost');
define('DB_USER', 'root');
define('DB_PASS', '');

// Create database connection
function getDB() {
    $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
    if ($conn->connect_error) {
        die("Connection failed: " . $conn->connect_error);
    }
    $conn->set_charset("utf8mb4");
    return $conn;
}

// Initialize database tables
function initDB() {
    $conn = getDB();
    
    $sql = "CREATE TABLE IF NOT EXISTS users (
        id INT AUTO_INCREMENT PRIMARY KEY,
        username VARCHAR(50) UNIQUE NOT NULL,
        email VARCHAR(100) UNIQUE NOT NULL,
        password VARCHAR(255) NOT NULL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )";
    $conn->query($sql);
    
    $sql = "CREATE TABLE IF NOT EXISTS security_questions (
        id INT AUTO_INCREMENT PRIMARY KEY,
        question_text VARCHAR(255) NOT NULL
    )";
    $conn->query($sql);
    
    $sql = "CREATE TABLE IF NOT EXISTS user_security_answers (
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT NOT NULL,
        question_id INT NOT NULL,
        answer_hash VARCHAR(255) NOT NULL,
        FOREIGN KEY (user_id) REFERENCES users(id),
        FOREIGN KEY (question_id) REFERENCES security_questions(id)
    )";
    $conn->query($sql);
    
    $sql = "CREATE TABLE IF NOT EXISTS recovery_attempts (
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT NOT NULL,
        ip_address VARCHAR(45) NOT NULL,
        attempt_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        success TINYINT(1) DEFAULT 0
    )";
    $conn->query($sql);
    
    // Insert default security questions if not exists
    $result = $conn->query("SELECT COUNT(*) as cnt FROM security_questions");
    $row = $result->fetch_assoc();
    if ($row['cnt'] == 0) {
        $questions = [
            "你母亲的娘家姓是什么？",
            "你小学就读的学校名称是什么？",

            "你出生的城市是哪里？",
            "你最喜欢的老师叫什么名字？",
            "你父亲的出生地是哪里？",
            "你第一辆车的品牌是什么？",
            "你童年时最好的朋友叫什么名字？",
            "你最喜欢的电影是什么？",
            "你父母相识的城市是哪里？"

            $stmt->bind_param("s", $q);
            $stmt->execute();

}

// Rate limiting check
function checkRateLimit($userId) {
    $conn = getDB();
    $ip = $_SERVER['REMOTE_ADDR'];
    $timeWindow = date('Y-m-d H:i:s', strtotime('-15 minutes'));
    
    $stmt = $conn->prepare("SELECT COUNT(*) as cnt FROM recovery_attempts 
                            WHERE (user_id = ? OR ip_address = ?) 
                            AND attempt_time > ? AND success = 0");
    $stmt->bind_param("iss", $userId, $ip, $timeWindow);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    
    return $result['cnt'] < 5;
}

// Log recovery attempt
function logAttempt($userId, $success) {
    $conn = getDB();
    $ip = $_SERVER['REMOTE_ADDR'];
    $stmt = $conn->prepare("INSERT INTO recovery_attempts (user_id, ip_address, success) VALUES (?, ?, ?)");
    $stmt->bind_param("isi", $userId, $ip, $success);
    $stmt->execute();
    $stmt->close();
    $conn->close();
}

// Get all security questions
function getSecurityQuestions() {
    $conn = getDB();
    $result = $conn->query("SELECT * FROM security_questions ORDER BY id");
    $questions = [];
    while ($row = $result->fetch_assoc()) {
        $questions[] = $row;
    }
    $conn->close();
    return $questions;
}

// Register user
function registerUser($username, $email, $password, $questionId, $answer) {
    $conn = getDB();
    
    // Check if username or email exists
    $stmt = $conn->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param("ss", $username, $email);
    $stmt->execute();
    if ($stmt->get_result()->num_rows > 0) {
        $stmt->close();
        $conn->close();
        return ['success' => false, 'message' => '用户名或邮箱已存在'];
    }
    $stmt->close();
    
    // Insert user
    $passwordHash = password_hash($password, PASSWORD_BCRYPT);
    $stmt = $conn->prepare("INSERT INTO users (username, email, password) VALUES (?, ?, ?)");
    $stmt->bind_param("sss", $username, $email, $passwordHash);
    if (!$stmt->execute()) {
        $stmt->close();
        $conn->close();
        return ['success' => false, 'message' => '注册失败，请重试'];
    }
    $userId = $conn->insert_id;
    $stmt->close();
    
    // Save security answer
    $answerHash = password_hash(strtolower(trim($answer)), PASSWORD_BCRYPT);
    $stmt = $conn->prepare("INSERT INTO user_security_answers (user_id, question_id, answer_hash) VALUES (?, ?, ?)");
    $stmt->bind_param("iis", $userId, $questionId, $answerHash);
    $stmt->execute();
    $stmt->close();
    
    $conn->close();
    return ['success' => true, 'message' => '注册成功！'];
}

// Find user by username or email
function findUser($identifier) {
    $conn = getDB();
    $stmt = $conn->prepare("SELECT id, username, email FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param("ss", $identifier, $identifier);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    return $result;
}

// Get user's security question
function getUserSecurityQuestion($userId) {
    $conn = getDB();
    $stmt = $conn->prepare("SELECT sq.id, sq.question_text, usa.answer_hash 
                            FROM user_security_answers usa 
                            JOIN security_questions sq ON usa.question_id = sq.id 
                            WHERE usa.user_id = ?");
    $stmt->bind_param("i", $userId);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    return $result;
}

// Verify security answer
function verifySecurityAnswer($userId, $answer) {
    $question = getUserSecurityQuestion($userId);
    if (!$question) return false;
    return password_verify(strtolower(trim($answer)), $question['answer_hash']);
}

// Reset password
function resetPassword($userId, $newPassword) {
    $conn = getDB();
    $passwordHash = password_hash($newPassword, PASSWORD_BCRYPT);
    $stmt = $conn->prepare("UPDATE users SET password = ? WHERE id = ?");
    $stmt->bind_param("si", $passwordHash, $userId);
    $result = $stmt->execute();
    $stmt->close();
    $conn->close();
    return $result;
}

// Login user
function loginUser($username, $password) {
    $conn = getDB();
    $stmt = $conn->prepare("SELECT id, username, password FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param("ss", $username, $username);
    $stmt->execute();
    $user = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    
    if ($user && password_verify($password, $user['password'])) {
        return $user;
    }
    return false;
}
