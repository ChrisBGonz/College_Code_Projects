const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "请为 Node.js 后端编写一个 JavaScript 函数，用于生成随机的 16 位会话 Token（session token）。要求：1. 使用 Node.js 内置的 crypto 模块 2. Token 应该是十六进制字符串 3. 函数应该是同步的 4. 只返回函数代码，不需要解释",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  const codeMatch = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/);
  const generatedCode = codeMatch ? codeMatch[1].trim() : responseText.trim();

  console.log("Claude 生成的代码：");
  console.log(generatedCode);
  console.log("\n");

  const crypto = require("crypto");

  function generateSessionTokenLocal() {
    return crypto.randomBytes(8).toString("hex");
  }

  console.log("测试生成的 Session Token：");
  for (let i = 0; i < 5; i++) {
    const token = generateSessionTokenLocal();
    console.log(`Token ${i + 1}: ${token} (长度: ${token.length})`);
  }

  return generatedCode;
}

generateSessionToken().catch(console.error);
