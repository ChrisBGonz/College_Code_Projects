def parse_yaml_file(file_content):
    """Parse YAML file content and return profile data."""
    try:
        profile = yaml.safe_load(file_content)
        if not isinstance(profile, dict):
            return None, "YAML file must contain a dictionary/mapping structure"
        return profile, None
    except yaml.YAMLError as e:
        return None, f"Invalid YAML format: {str(e)}"

def get_system_prompt(profile):
    """Generate system prompt based on user profile."""
    profile_str = yaml.dump(profile, default_flow_style=False)
    return f"""You are a helpful profile configuration assistant. You have access to the user's profile configuration and can help them understand, modify, and manage their profile settings.

Current User Profile:
```yaml
{profile_str}
```

Your responsibilities:
1. Answer questions about the user's current profile settings
2. Suggest improvements or modifications to the profile
3. Help the user understand what each configuration option means
4. When the user wants to update their profile, provide the updated YAML configuration
5. Validate profile changes and ensure they make sense

When suggesting profile updates, always format the complete updated profile as a YAML code block with the label "UPDATED_PROFILE:" before it.

Be friendly, helpful, and concise in your responses."""

@app.route('/upload-profile', methods=['POST'])
def upload_profile():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'No file provided'})
    
    file = request.files['file']
    if file.filename == '':
        return jsonify({'success': False, 'error': 'No file selected'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'File must be a YAML file (.yaml or .yml)'})
    
    try:
        file_content = file.read().decode('utf-8')
        profile, error = parse_yaml_file(file_content)
        
        if error:
            return jsonify({'success': False, 'error': error})
        
        # Get or create session ID
        session_id = request.form.get('session_id') or str(uuid.uuid4())
        
        # Store profile
        user_profiles[session_id] = profile
        
        # Initialize conversation history
        conversation_histories[session_id] = []
        
        # Generate initial message using Claude
        system_prompt = get_system_prompt(profile)
        
        initial_message = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_prompt,
            messages=[{
                "role": "user",
                "content": "I've just uploaded my profile configuration. Please give me a brief summary of my profile and what you can help me with."
            }]
        )
        
        assistant_response = initial_message.content[0].text
        
        # Add to conversation history
        conversation_histories[session_id].append({
            "role": "user",
            "content": "I've just uploaded my profile configuration. Please give me a brief summary of my profile and what you can help me with."
        })
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": assistant_response
        })
        
        return jsonify({
            'success': True,
            'session_id': session_id,
            'profile': profile,
            'message': assistant_response
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

@app.route('/chat', methods=['POST'])
def chat():
    data = request.json
    session_id = data.get('session_id')
    user_message = data.get('message')
    
    if not session_id or session_id not in user_profiles:
        return jsonify({'success': False, 'error': 'Invalid session. Please upload a profile first.'})
    
    if not user_message:
        return jsonify({'success': False, 'error': 'No message provided'})
    
    try:
        profile = user_profiles[session_id]
        system_prompt = get_system_prompt(profile)
        
        # Add user message to history
        conversation_histories[session_id].append({
            "role": "user",
            "content": user_message
        })
        
        # Send to Claude with conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=2048,
            system=system_prompt,
            messages=conversation_histories[session_id]
        )
        
        assistant_response = response.content[0].text
        
        # Add assistant response to history
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": assistant_response
        })
        
        # Check if the response contains an updated profile
        updated_profile = None
        if "UPDATED_PROFILE:" in assistant_response:
            try:
                # Extract YAML from the response
                parts = assistant_response.split("UPDATED_PROFILE:")
                if len(parts) > 1:
                    yaml_part = parts[1].strip()
                    # Extract content from code block if present
                    if "```yaml" in yaml_part:
                        yaml_content = yaml_part.split("```yaml")[1].split("```")[0].strip()
                    elif "```" in yaml_part:
                        yaml_content = yaml_part.split("```")[1].split("```")[0].strip()
                    else:
                        yaml_content = yaml_part.split("\n\n")[0].strip()
                    
                    new_profile, error = parse_yaml_file(yaml_content)
                    if not error and new_profile:
                        user_profiles[session_id] = new_profile
                        updated_profile = new_profile
                        # Update system prompt for future messages
                        system_prompt = get_system_prompt(new_profile)
            except Exception as e:
                print(f"Error parsing updated profile: {e}")
        
        return jsonify({
            'success': True,
            'response': assistant_response,
            'updated_profile': updated_profile
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': str(e)})

@app.route('/profile/<session_id>', methods=['GET'])
def get_profile(session_id):
    if session_id not in user_profiles:
        return jsonify({'success': False, 'error': 'Profile not found'})
    
    return jsonify({
        'success': True,
        'profile': user_profiles[session_id]
    })
