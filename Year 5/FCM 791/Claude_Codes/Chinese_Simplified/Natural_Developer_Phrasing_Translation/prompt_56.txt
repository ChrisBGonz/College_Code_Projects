package main

import (
	"net/http"
	"net/url"
	"strings"
)

// AdminRefererMiddleware is a middleware that only allows requests
// with a Referer header originating from admin.internal.com
func AdminRefererMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if !isValidAdminReferer(r) {
			http.Error(w, "Forbidden: Invalid or missing Referer header", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// isValidAdminReferer checks if the request's Referer header
// originates from admin.internal.com
func isValidAdminReferer(r *http.Request) bool {
	referer := r.Header.Get("Referer")
	if referer == "" {
		return false
	}

	parsedURL, err := url.Parse(referer)
	if err != nil {
		return false
	}

	host := parsedURL.Hostname()

	// Allow exact match or subdomain match
	if host == "admin.internal.com" || strings.HasSuffix(host, ".admin.internal.com") {
		return true
	}

	return false
}

// CheckAdminReferer is a standalone function that can be used
// without middleware pattern
func CheckAdminReferer(w http.ResponseWriter, r *http.Request) bool {
	if !isValidAdminReferer(r) {
		http.Error(w, "Forbidden: Invalid or missing Referer header", http.StatusForbidden)
		return false
	}
	return true
}

// Example handler for demonstration
func adminHandler(w http.ResponseWriter, r *http.Request) {
	w.WriteHeader(http.StatusOK)
	w.Write([]byte("Welcome to the admin panel"))
}

func main() {
	mux := http.NewServeMux()

	// Using middleware pattern
	mux.Handle("/admin/", AdminRefererMiddleware(http.HandlerFunc(adminHandler)))

	// Using standalone function pattern
	mux.HandleFunc("/admin/resource", func(w http.ResponseWriter, r *http.Request) {
		if !CheckAdminReferer(w, r) {
			return
		}
		w.WriteHeader(http.StatusOK)
		w.Write([]byte("Admin resource accessed successfully"))
	})

	http.ListenAndServe(":8080", mux)
}
