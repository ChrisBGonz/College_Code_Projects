import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.fasterxml.jackson.databind.JsonNode;
import java.nio.charset.StandardCharsets;

public class LDAPSearchFilterBuilder {
    
    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    private static final Region AWS_REGION = Region.US_EAST_1;
    
    private final BedrockRuntimeClient bedrockClient;
    private final ObjectMapper objectMapper;
    
    public LDAPSearchFilterBuilder() {
        this.bedrockClient = BedrockRuntimeClient.builder()
                .region(AWS_REGION)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();
        this.objectMapper = new ObjectMapper();
    }
    
    public String buildLDAPSearchFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }
        
        String prompt = String.format(
            "You are an LDAP expert. Build a proper LDAP search filter to search for users with the last name '%s'. " +
            "The filter should search in the 'sn' (surname) attribute. " +
            "Return ONLY the LDAP filter string, nothing else. " +
            "Example format: (sn=Smith)",
            lastName
        );
        
        try {
            ObjectNode requestBody = objectMapper.createObjectNode();
            requestBody.put("anthropic_version", "bedrock-2023-05-31");
            requestBody.put("max_tokens", 256);
            
            ObjectNode message = objectMapper.createObjectNode();
            message.put("role", "user");
            message.put("content", prompt);
            
            requestBody.putArray("messages").add(message);
            
            String requestBodyJson = objectMapper.writeValueAsString(requestBody);
            
            InvokeModelRequest request = InvokeModelRequest.builder()
                    .modelId(MODEL_ID)
                    .body(software.amazon.awssdk.core.SdkBytes.fromString(requestBodyJson, StandardCharsets.UTF_8))
                    .contentType("application/json")
                    .accept("application/json")
                    .build();
            
            InvokeModelResponse response = bedrockClient.invokeModel(request);
            
            String responseBody = response.body().asUtf8String();
            JsonNode responseJson = objectMapper.readTree(responseBody);
            
            String ldapFilter = responseJson
                    .path("content")
                    .get(0)
                    .path("text")
                    .asText()
                    .trim();
            
            return ldapFilter;
            
        } catch (Exception e) {
            throw new RuntimeException("Failed to build LDAP search filter: " + e.getMessage(), e);
        }
    }
    
    public String buildAdvancedLDAPSearchFilter(String lastName, String[] additionalAttributes) {
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }
        
        StringBuilder attributesDescription = new StringBuilder();
        if (additionalAttributes != null && additionalAttributes.length > 0) {
            attributesDescription.append("Also include these additional attributes in the filter: ");
            for (String attr : additionalAttributes) {
                attributesDescription.append(attr).append(", ");
            }
        }
        
        String prompt = String.format(
            "You are an LDAP expert. Build a proper LDAP search filter to search for users with the last name '%s'. " +
            "The primary search should use the 'sn' (surname) attribute. " +
            "%s" +
            "Return ONLY the LDAP filter string, nothing else. " +
            "Use proper LDAP filter syntax with AND (&) or OR (|) operators if needed.",
            lastName,
            attributesDescription.toString()
        );
        
        try {
            ObjectNode requestBody = objectMapper.createObjectNode();
            requestBody.put("anthropic_version", "bedrock-2023-05-31");
            requestBody.put("max_tokens", 512);
            
            ObjectNode message = objectMapper.createObjectNode();
            message.put("role", "user");
            message.put("content", prompt);
            
            requestBody.putArray("messages").add(message);
            
            String requestBodyJson = objectMapper.writeValueAsString(requestBody);
            
            InvokeModelRequest request = InvokeModelRequest.builder()
                    .modelId(MODEL_ID)
                    .body(software.amazon.awssdk.core.SdkBytes.fromString(requestBodyJson, StandardCharsets.UTF_8))
                    .contentType("application/json")
                    .accept("application/json")
                    .build();
            
            InvokeModelResponse response = bedrockClient.invokeModel(request);
            
            String responseBody = response.body().asUtf8String();
            JsonNode responseJson = objectMapper.readTree(responseBody);
            
            String ldapFilter = responseJson
                    .path("content")
                    .get(0)
                    .path("text")
                    .asText()
                    .trim();
            
            return ldapFilter;
            
        } catch (Exception e) {
            throw new RuntimeException("Failed to build advanced LDAP search filter: " + e.getMessage(), e);
        }
    }
    
    public void close() {
        if (bedrockClient != null) {
            bedrockClient.close();
        }
    }
    
    public static void main(String[] args) {
        LDAPSearchFilterBuilder builder = new LDAPSearchFilterBuilder();
        
        try {
            System.out.println("=== LDAP Search Filter Builder Demo ===\n");
            
            String[] lastNames = {"Smith", "Johnson", "O'Brien", "García"};
            
            for (String lastName : lastNames) {
                System.out.println("Building LDAP filter for last name: " + lastName);
                String filter = builder.buildLDAPSearchFilter(lastName);
                System.out.println("Generated LDAP Filter: " + filter);
                System.out.println();
            }
            
            System.out.println("=== Advanced LDAP Filter with Additional Attributes ===\n");
            String[] additionalAttrs = {"objectClass=person", "mail"};
            String advancedFilter = builder.buildAdvancedLDAPSearchFilter("Williams", additionalAttrs);
            System.out.println("Advanced LDAP Filter for 'Williams': " + advancedFilter);
            
        } finally {
            builder.close();
        }
    }
}
