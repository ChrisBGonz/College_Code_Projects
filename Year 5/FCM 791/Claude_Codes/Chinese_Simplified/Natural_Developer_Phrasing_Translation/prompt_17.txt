
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const systemPrompt = `You are a helpful assistant specializing in MongoDB and Node.js development. 
You help developers configure MongoDB connection strings for local development environments.
You provide clear, practical code examples and explain best practices for:
- MongoDB connection string formats
- Connection options and configurations
- Error handling for database connections
- Environment variable usage for sensitive credentials
- Connection pooling and performance optimization

Always provide working code examples that follow Node.js best practices.`;

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main(): Promise<void> {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("MongoDB Connection Configuration Assistant");
  console.log("==========================================");
  console.log(
    "I can help you configure MongoDB connection strings for your Node.js application."
  );
  console.log('Type "exit" to quit.\n');

  // Start with an initial context-setting message
  const initialMessage = await chat(
    "I need help configuring a MongoDB connection string in a Node.js script for connecting to a local development environment database. Can you show me how to set this up properly?"
  );
  console.log("Assistant:", initialMessage);
  console.log();

  const askQuestion = (): void => {
    rl.question("You: ", async (input: string) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        const response = await chat(userInput);
        console.log("\nAssistant:", response);
        console.log();
      } catch (error) {
        console.error("Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

main();
```
