import anthropic
import base64

def render_html_to_pdf(
    html_template: str,
    output_path: str = None,
    template_vars: dict = None,
    page_size: str = "A4",
    margin: str = "1cm",
) -> bytes:
    """
    Render an HTML template and convert it to PDF using Claude AI for content generation
    and WeasyPrint for PDF conversion.
    
    Args:
        html_template: HTML template string (can contain placeholders like {variable_name})
        output_path: Optional path to save the PDF file
        template_vars: Dictionary of variables to substitute in the template
        page_size: Page size for PDF (default: A4)
        margin: Page margin (default: 1cm)
    
    Returns:
        PDF content as bytes
    """
    try:
        from weasyprint import HTML, CSS
    except ImportError:
        raise ImportError(
            "WeasyPrint is required. Install it with: pip install weasyprint"
        )
    
    if template_vars:
        html_content = html_template.format(**template_vars)
    else:
        html_content = html_template
    
    css_string = f"""
        @page {{
            size: {page_size};
            margin: {margin};
        }}
        body {{
            font-family: Arial, sans-serif;
            font-size: 12pt;
            line-height: 1.6;
        }}
        h1 {{ font-size: 24pt; color: #333; }}
        h2 {{ font-size: 18pt; color: #555; }}
        h3 {{ font-size: 14pt; color: #777; }}
        table {{
            width: 100%;
            border-collapse: collapse;
            margin: 10px 0;
        }}
        th, td {{
            border: 1px solid #ddd;
            padding: 8px;
            text-align: left;
        }}
        th {{ background-color: #f2f2f2; }}
    """
    
    html_obj = HTML(string=html_content)
    css_obj = CSS(string=css_string)
    
    pdf_bytes = html_obj.write_pdf(stylesheets=[css_obj])
    
    if output_path:
        with open(output_path, 'wb') as f:
            f.write(pdf_bytes)
        print(f"PDF saved to: {output_path}")
    
    return pdf_bytes


def generate_html_with_claude(prompt: str, content_data: dict = None) -> str:
    """
    Use Claude AI to generate HTML content based on a prompt.
    
    Args:
        prompt: Description of what HTML content to generate
        content_data: Optional data to include in the HTML
    
    Returns:
        Generated HTML string
    """
    client = anthropic.Anthropic()
    
    system_message = """You are an HTML expert. Generate clean, well-structured HTML content.
    Always return only the HTML code without any markdown formatting or explanations.
    The HTML should be complete and ready to render."""
    
    if content_data:
        user_message = f"{prompt}\n\nUse this data: {content_data}"
    else:
        user_message = prompt
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": user_message
            }
        ],
        system=system_message
    )
    
    html_content = message.content[0].text
    
    if "```html" in html_content:
        html_content = html_content.split("```html")[1].split("```")[0].strip()
    elif "```" in html_content:
        html_content = html_content.split("```")[1].split("```")[0].strip()
    
    return html_content


def create_pdf_from_prompt(
    prompt: str,
    output_path: str = None,
    content_data: dict = None,
    page_size: str = "A4",
    margin: str = "1cm"
) -> bytes:
    """
    Generate HTML content using Claude AI and convert it to PDF.
    
    Args:
        prompt: Description of the PDF content to generate
        output_path: Optional path to save the PDF file
        content_data: Optional data to include in the content
        page_size: Page size for PDF (default: A4)
        margin: Page margin (default: 1cm)
    
    Returns:
        PDF content as bytes
    """
    print("Generating HTML content with Claude AI...")
    html_content = generate_html_with_claude(prompt, content_data)
    
    print("Converting HTML to PDF...")
    pdf_bytes = render_html_to_pdf(
        html_template=html_content,
        output_path=output_path,
        page_size=page_size,
        margin=margin
    )
    
    return pdf_bytes


def html_template_to_pdf(
    template_path: str,
    output_path: str,
    template_vars: dict = None,
    page_size: str = "A4",
    margin: str = "1cm"
) -> bytes:
    """
    Read an HTML template file and convert it to PDF.
    
    Args:
        template_path: Path to the HTML template file
        output_path: Path to save the PDF file
        template_vars: Dictionary of variables to substitute in the template
        page_size: Page size for PDF (default: A4)
        margin: Page margin (default: 1cm)
    
    Returns:
        PDF content as bytes
    """
    with open(template_path, 'r', encoding='utf-8') as f:
        html_template = f.read()
    
    return render_html_to_pdf(
        html_template=html_template,
        output_path=output_path,
        template_vars=template_vars,
        page_size=page_size,
        margin=margin
    )
