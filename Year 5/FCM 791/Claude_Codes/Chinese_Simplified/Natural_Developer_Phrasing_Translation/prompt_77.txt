using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using ModelClient = OpenAI.Chat.ChatClient;

[Serializable]
public class AppSettings
{
    public string Theme { get; set; } = "Light";
    public int FontSize { get; set; } = 12;
    public bool AutoSave { get; set; } = true;
    public string Language { get; set; } = "English";
    public double Volume { get; set; } = 0.8;
    public string LastOpenedFile { get; set; } = "";
    public DateTime LastModified { get; set; } = DateTime.Now;
}

public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        settings.LastModified = DateTime.Now;
        
        using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Create))
        {
#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
            formatter.Serialize(fileStream, settings);
#pragma warning restore SYSLIB0011
        }
        
        Console.WriteLine($"Settings saved to {_settingsFilePath}");
    }

    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("Settings file not found. Using default settings.");
            return new AppSettings();
        }

        using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Open))
        {
#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
            AppSettings settings = (AppSettings)formatter.Deserialize(fileStream);
#pragma warning restore SYSLIB0011
            Console.WriteLine($"Settings loaded from {_settingsFilePath}");
            return settings;
        }
    }

    public bool SettingsExist()
    {
        return File.Exists(_settingsFilePath);
    }

    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine("Settings file deleted.");
        }
    }
}

class Program
{
    static async System.Threading.Tasks.Task Main(string[] args)
    {
        string apiKey = Environment.GetEnvironmentVariable("OPENAI_API_KEY") ?? throw new InvalidOperationException("OPENAI_API_KEY environment variable is not set.");
        
        ModelClient client = new ModelClient("gpt-4o", new System.ClientModel.ApiKeyCredential(apiKey));
        
        SettingsManager settingsManager = new SettingsManager();
        
        AppSettings currentSettings = new AppSettings
        {
            Theme = "Dark",
            FontSize = 14,
            AutoSave = false,
            Language = "Chinese",
            Volume = 0.6,
            LastOpenedFile = "/home/user/documents/project.txt"
        };
        
        settingsManager.SaveSettings(currentSettings);
        
        AppSettings loadedSettings = settingsManager.LoadSettings();
        
        string settingsDescription = $@"
Current Application Settings:
- Theme: {loadedSettings.Theme}
- Font Size: {loadedSettings.FontSize}
- Auto Save: {loadedSettings.AutoSave}
- Language: {loadedSettings.Language}
- Volume: {loadedSettings.Volume}
- Last Opened File: {loadedSettings.LastOpenedFile}
- Last Modified: {loadedSettings.LastModified}
";
        
        Console.WriteLine(settingsDescription);
        
        var messages = new System.Collections.Generic.List<OpenAI.Chat.ChatMessage>
        {
            OpenAI.Chat.ChatMessage.CreateSystemMessage("You are a helpful assistant that analyzes application settings and provides recommendations for optimization."),
            OpenAI.Chat.ChatMessage.CreateUserMessage($"Please analyze these application settings and provide recommendations for improving user experience:\n{settingsDescription}")
        };
        
        Console.WriteLine("\nAsking AI to analyze settings...\n");
        
        OpenAI.Chat.ChatCompletion completion = await client.CompleteChatAsync(messages);
        
        string aiResponse = completion.Content[0].Text;
        Console.WriteLine("AI Analysis and Recommendations:");
        Console.WriteLine(aiResponse);
        
        Console.WriteLine("\n--- Demonstrating Settings Update Based on AI Recommendations ---");
        
        var updateMessages = new System.Collections.Generic.List<OpenAI.Chat.ChatMessage>
        {
            OpenAI.Chat.ChatMessage.CreateSystemMessage("You are a helpful assistant. When asked to suggest settings values, respond with specific values in a structured format."),
            OpenAI.Chat.ChatMessage.CreateUserMessage("Based on best practices, suggest optimal values for: Theme (Light/Dark), FontSize (8-24), AutoSave (true/false), Volume (0.0-1.0). Format: Theme=value, FontSize=value, AutoSave=value, Volume=value")
        };
        
        OpenAI.Chat.ChatCompletion updateCompletion = await client.CompleteChatAsync(updateMessages);
        string suggestedSettings = updateCompletion.Content[0].Text;
        Console.WriteLine($"\nAI Suggested Settings: {suggestedSettings}");
        
        AppSettings updatedSettings = new AppSettings
        {
            Theme = suggestedSettings.Contains("Dark") ? "Dark" : "Light",
            FontSize = 14,
            AutoSave = true,
            Language = loadedSettings.Language,
            Volume = 0.7,
            LastOpenedFile = loadedSettings.LastOpenedFile
        };
        
        settingsManager.SaveSettings(updatedSettings);
        
        AppSettings finalSettings = settingsManager.LoadSettings();
        Console.WriteLine("\nFinal Settings After Update:");
        Console.WriteLine($"- Theme: {finalSettings.Theme}");
        Console.WriteLine($"- Font Size: {finalSettings.FontSize}");
        Console.WriteLine($"- Auto Save: {finalSettings.AutoSave}");
        Console.WriteLine($"- Language: {finalSettings.Language}");
        Console.WriteLine($"- Volume: {finalSettings.Volume}");
        Console.WriteLine($"- Last Modified: {finalSettings.LastModified}");
        
        Console.WriteLine("\nSettings persistence demonstration completed successfully!");
    }
}
