# Simulated user authentication database
USERS_DB = {
    "user001": {"name": "Alice Johnson", "company_id": "COMP001", "role": "Manager"},
    "user002": {"name": "Bob Smith", "company_id": "COMP001", "role": "Analyst"},
    "user003": {"name": "Carol White", "company_id": "COMP002", "role": "Director"},
    "user004": {"name": "David Brown", "company_id": "COMP003", "role": "CFO"},
}

def get_company_report(company_id: str) -> dict | None:
    """Retrieve annual report for a specific company."""
    return COMPANIES_DB.get(company_id)

def authenticate_user(user_id: str) -> dict | None:
    """Authenticate user and return their information."""
    return USERS_DB.get(user_id)

def format_report_for_context(company_data: dict) -> str:
    """Format company annual report data as context for Claude."""
    report = company_data["annual_report"]
    highlights = "\n".join([f"  - {h}" for h in report["key_highlights"]])
    
    return f"""
Company: {company_data['name']}
Industry: {company_data['industry']}

Annual Report {report['year']}:
- Revenue: {report['revenue']}
- Net Income: {report['net_income']}
- Employees: {report['employees']:,}

Financial Summary:
- Gross Profit: {report['financial_summary']['gross_profit']}
- Operating Expenses: {report['financial_summary']['operating_expenses']}
- EBITDA: {report['financial_summary']['ebitda']}
- Total Assets: {report['financial_summary']['total_assets']}

Key Highlights:
{highlights}

CEO Message: {report['ceo_message']}
"""

def create_system_prompt(user_info: dict, company_data: dict) -> str:
    """Create a system prompt with user context and company data."""
    report_context = format_report_for_context(company_data)
    
    return f"""You are an AI assistant for a multi-tenant SaaS platform that provides access to annual reports.

Current User Information:
- Name: {user_info['name']}
- Role: {user_info['role']}
- Company ID: {user_info['company_id']}

You have access to the following annual report data for the user's company:

{report_context}

Important Guidelines:
1. Only provide information about the user's own company ({company_data['name']})
2. Do not reveal information about other companies
3. Be helpful and professional in answering questions about the annual report
4. If asked about data not in the report, acknowledge the limitation
5. Provide insights and analysis based on the available data when appropriate
6. Address the user by their name ({user_info['name']}) when appropriate

You can help the user understand their company's financial performance, key achievements, and strategic direction based on the annual report data provided."""

def run_saas_assistant(user_id: str):
    """Run the multi-tenant SaaS assistant for a specific user."""
    # Authenticate user
    user_info = authenticate_user(user_id)
    if not user_info:
        print(f"Error: User '{user_id}' not found. Please check your credentials.")
        return
    
    # Get company data
    company_data = get_company_report(user_info["company_id"])
    if not company_data:
        print(f"Error: No data found for company ID '{user_info['company_id']}'.")
        return
    
    print(f"\n{'='*60}")
    print(f"Welcome to the Annual Report Assistant!")
    print(f"{'='*60}")
    print(f"User: {user_info['name']} ({user_info['role']})")
    print(f"Company: {company_data['name']}")
    print(f"{'='*60}")
    print("\nYou can ask questions about your company's annual report.")
    print("Type 'exit' or 'quit' to end the session.\n")
    
    # Initialize conversation history
    conversation_history = []
    system_prompt = create_system_prompt(user_info, company_data)
    
    while True:
        # Get user input
        user_input = input(f"{user_info['name']}: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['exit', 'quit', 'bye', 'goodbye']:
            print(f"\nAssistant: Thank you for using the Annual Report Assistant, {user_info['name']}. Goodbye!")
            break
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Get response from Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        print(f"\nAssistant: {assistant_message}\n")

def demo_multi_tenant():
    """Demonstrate multi-tenant functionality with different users."""
    print("\n" + "="*60)
    print("MULTI-TENANT SAAS ANNUAL REPORT SYSTEM")
    print("="*60)
    print("\nAvailable Demo Users:")
    print("-" * 40)
    for user_id, user_info in USERS_DB.items():
        company = COMPANIES_DB.get(user_info["company_id"], {})
        print(f"  {user_id}: {user_info['name']} - {company.get('name', 'Unknown')} ({user_info['role']})")
    
    print("\n" + "-"*40)
    user_id = input("Enter your User ID to login: ").strip()
    
    if not user_id:
        print("No user ID provided. Using demo user 'user001'")
        user_id = "user001"
    
    run_saas_assistant(user_id)

if __name__ == "__main__":
    demo_multi_tenant()
