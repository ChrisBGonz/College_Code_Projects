import anthropic
import json
import re
from datetime import datetime, timedelta
from http.cookies import SimpleCookie


def check_trial_cookie(cookie_string: str | None) -> bool:
    """Check if the user has already used the trial based on cookie."""
    if not cookie_string:
        return False
    
    cookie = SimpleCookie()
    try:
        cookie.load(cookie_string)
        if 'has_used_trial' in cookie:
            value = cookie['has_used_trial'].value
            return value.lower() == 'true'
    except Exception:
        pass
    
    return False


def generate_trial_cookie() -> str:
    """Generate a cookie string indicating the trial has been used."""
    expiry = datetime.now() + timedelta(days=365)
    expiry_str = expiry.strftime("%a, %d %b %Y %H:%M:%S GMT")
    return f"has_used_trial=true; Expires={expiry_str}; Path=/; HttpOnly"


def process_trial_registration(user_info: dict, cookie_string: str | None) -> dict:
    """Process trial registration using Claude AI with tool use."""
    
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "check_trial_status",
            "description": "Check if a user has already used their free trial by examining the cookie",
            "input_schema": {
                "type": "object",
                "properties": {
                    "cookie_string": {
                        "type": "string",
                        "description": "The cookie string from the user's browser"
                    }
                },
                "required": ["cookie_string"]
            }
        },
        {
            "name": "register_trial",
            "description": "Register a new free trial for the user",
            "input_schema": {
                "type": "object",
                "properties": {
                    "user_email": {
                        "type": "string",
                        "description": "User's email address"
                    },
                    "user_name": {
                        "type": "string",
                        "description": "User's name"
                    }
                },
                "required": ["user_email", "user_name"]
            }
        },
        {
            "name": "deny_trial",
            "description": "Deny trial registration because user has already used their trial",
            "input_schema": {
                "type": "object",
                "properties": {
                    "reason": {
                        "type": "string",
                        "description": "Reason for denying the trial"
                    }
                },
                "required": ["reason"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"""Process a free trial registration request.
            
User Information:
- Name: {user_info.get('name', 'Unknown')}
- Email: {user_info.get('email', 'Unknown')}

Cookie String: {cookie_string if cookie_string else 'No cookie provided'}

Please:
1. First check the trial status using the cookie
2. If the user hasn't used the trial, register them
3. If they have already used the trial, deny the registration
4. Provide a clear response about the outcome"""
        }
    ]
    
    result = {
        "success": False,
        "message": "",
        "cookie": None,
        "user_info": user_info
    }
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    if tool_name == "check_trial_status":
                        has_used = check_trial_cookie(tool_input.get("cookie_string"))
                        tool_result = {
                            "has_used_trial": has_used,
                            "message": "User has already used their trial" if has_used else "User has not used their trial yet"
                        }
                        
                    elif tool_name == "register_trial":
                        new_cookie = generate_trial_cookie()
                        result["success"] = True
                        result["cookie"] = new_cookie
                        tool_result = {
                            "success": True,
                            "message": f"Trial successfully registered for {tool_input.get('user_name')}",
                            "cookie": new_cookie,
                            "trial_duration": "30 days",
                            "features": ["Full access to all features", "Priority support", "No credit card required"]
                        }
                        
                    elif tool_name == "deny_trial":
                        result["success"] = False
                        tool_result = {
                            "success": False,
                            "message": "Trial registration denied",
                            "reason": tool_input.get("reason"),
                            "alternatives": ["Upgrade to paid plan", "Contact support for assistance"]
                        }
                    else:
                        tool_result = {"error": f"Unknown tool: {tool_name}"}
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": json.dumps(tool_result)
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    result["message"] = content_block.text
            break
    
    return result


def simulate_trial_registration():
    """Simulate the trial registration process with different scenarios."""
    
    print("=" * 60)
    print("FREE TRIAL REGISTRATION SYSTEM")
    print("=" * 60)
    
    print("\n--- Scenario 1: New User (No Cookie) ---")
    new_user = {
        "name": "Alice Johnson",
        "email": "alice@example.com"
    }
    
    result1 = process_trial_registration(new_user, None)
    print(f"Success: {result1['success']}")
    print(f"Message: {result1['message']}")
    if result1['cookie']:
        print(f"Cookie Set: {result1['cookie'][:50]}...")
    
    print("\n--- Scenario 2: Returning User (Has Trial Cookie) ---")
    returning_user = {
        "name": "Bob Smith",
        "email": "bob@example.com"
    }
    
    existing_cookie = "has_used_trial=true; Path=/; HttpOnly"
    
    result2 = process_trial_registration(returning_user, existing_cookie)
    print(f"Success: {result2['success']}")
    print(f"Message: {result2['message']}")
    
    print("\n--- Scenario 3: User with Invalid Cookie ---")
    another_user = {
        "name": "Charlie Brown",
        "email": "charlie@example.com"
    }
    
    invalid_cookie = "session_id=abc123; theme=dark"
    
    result3 = process_trial_registration(another_user, invalid_cookie)
    print(f"Success: {result3['success']}")
    print(f"Message: {result3['message']}")
    if result3['cookie']:
        print(f"Cookie Set: {result3['cookie'][:50]}...")
    
    print("\n" + "=" * 60)
    print("REGISTRATION SUMMARY")
    print("=" * 60)
    print(f"Scenario 1 (New User): {'✓ Registered' if result1['success'] else '✗ Denied'}")
    print(f"Scenario 2 (Returning User): {'✓ Registered' if result2['success'] else '✗ Denied'}")
    print(f"Scenario 3 (Invalid Cookie): {'✓ Registered' if result3['success'] else '✗ Denied'}")


if __name__ == "__main__":
    simulate_trial_registration()
