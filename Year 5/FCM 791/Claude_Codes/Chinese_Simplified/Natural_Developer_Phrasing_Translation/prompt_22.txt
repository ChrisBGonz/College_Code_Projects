#!/usr/bin/env bash
set -euo pipefail
IFS=$'\n\t'

# ─────────────────────────────────────────────
# Configuration
# ─────────────────────────────────────────────
TOOL_NAME="nuclei"
GITHUB_REPO="projectdiscovery/nuclei"
INSTALL_DIR="/usr/local/bin"
TMP_DIR="$(mktemp -d)"
LOG_FILE="/var/log/${TOOL_NAME}_install.log"
CHECKSUM_ALGO="sha256"
MIN_BASH_VERSION=4

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
NC='\033[0m'

# ─────────────────────────────────────────────
# Logging
# ─────────────────────────────────────────────
log() {
    local level="$1"; shift
    local msg="$*"
    local ts
    ts="$(date '+%Y-%m-%d %H:%M:%S')"
    local color="$NC"
    case "$level" in
        INFO)  color="$GREEN"  ;;
        WARN)  color="$YELLOW" ;;
        ERROR) color="$RED"    ;;
        STEP)  color="$CYAN"   ;;
    esac
    printf "${color}[%s] [%s] %s${NC}\n" "$ts" "$level" "$msg" | tee -a "$LOG_FILE"
}

die() {
    log ERROR "$*"
    cleanup
    exit 1
}

# ─────────────────────────────────────────────
# Cleanup on exit
# ─────────────────────────────────────────────
cleanup() {
    if [[ -d "$TMP_DIR" ]]; then
        rm -rf "$TMP_DIR"
        log INFO "Cleaned up temporary directory: $TMP_DIR"
    fi
}
trap cleanup EXIT
trap 'die "Script interrupted."' INT TERM

# ─────────────────────────────────────────────
# Privilege check
# ─────────────────────────────────────────────
check_privileges() {
    if [[ $EUID -ne 0 ]]; then
        if command -v sudo &>/dev/null; then
            log WARN "Not running as root. Will use sudo for privileged operations."
            SUDO="sudo"
        else
            die "Root privileges required and sudo is not available."
        fi
    else
        SUDO=""
    fi
}

# ─────────────────────────────────────────────
# Bash version check
# ─────────────────────────────────────────────
check_bash_version() {
    if [[ "${BASH_VERSINFO[0]}" -lt "$MIN_BASH_VERSION" ]]; then
        die "Bash >= ${MIN_BASH_VERSION} required. Current: ${BASH_VERSION}"
    fi
}

# ─────────────────────────────────────────────
# Dependency check
# ─────────────────────────────────────────────
REQUIRED_DEPS=(curl tar gzip jq)
OPTIONAL_DEPS=(wget sha256sum shasum)

check_dependencies() {
    log STEP "Checking required dependencies..."
    local missing=()
    for dep in "${REQUIRED_DEPS[@]}"; do
        if ! command -v "$dep" &>/dev/null; then
            missing+=("$dep")
        else
            log INFO "  [OK] $dep -> $(command -v "$dep")"
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log WARN "Missing dependencies: ${missing[*]}"
        auto_install_deps "${missing[@]}"
    fi

    log STEP "Checking optional dependencies..."
    for dep in "${OPTIONAL_DEPS[@]}"; do
        if command -v "$dep" &>/dev/null; then
            log INFO "  [OK] $dep (optional)"
        else
            log WARN "  [MISSING] $dep (optional)"
        fi
    done
}

auto_install_deps() {
    local pkgs=("$@")
    log STEP "Attempting to auto-install: ${pkgs[*]}"

    if command -v apt-get &>/dev/null; then
        $SUDO apt-get update -qq
        $SUDO apt-get install -y -qq "${pkgs[@]}"
    elif command -v yum &>/dev/null; then
        $SUDO yum install -y -q "${pkgs[@]}"
    elif command -v dnf &>/dev/null; then
        $SUDO dnf install -y -q "${pkgs[@]}"
    elif command -v pacman &>/dev/null; then
        $SUDO pacman -Sy --noconfirm "${pkgs[@]}"
    elif command -v brew &>/dev/null; then
        brew install "${pkgs[@]}"
    elif command -v apk &>/dev/null; then
        $SUDO apk add --no-cache "${pkgs[@]}"
    else
        die "Cannot auto-install dependencies. Please install manually: ${pkgs[*]}"
    fi

    # Re-verify
    for dep in "${pkgs[@]}"; do
        command -v "$dep" &>/dev/null || die "Failed to install dependency: $dep"
    done
    log INFO "Dependencies installed successfully."
}

# ─────────────────────────────────────────────
# Detect OS and architecture
# ─────────────────────────────────────────────
detect_platform() {
    log STEP "Detecting platform..."

    OS="$(uname -s | tr '[:upper:]' '[:lower:]')"
    ARCH_RAW="$(uname -m)"

    case "$OS" in
        linux)   OS_LABEL="linux"   ;;
        darwin)  OS_LABEL="macOS"   ;;
        freebsd) OS_LABEL="freebsd" ;;
        *)       die "Unsupported OS: $OS" ;;
    esac

    case "$ARCH_RAW" in
        x86_64|amd64)   ARCH="amd64"  ;;
        aarch64|arm64)  ARCH="arm64"  ;;
        armv7l|armv6l)  ARCH="arm"    ;;
        i386|i686)      ARCH="386"    ;;
        *)              die "Unsupported architecture: $ARCH_RAW" ;;
    esac

    log INFO "OS: $OS_LABEL | Arch: $ARCH"
}

# ─────────────────────────────────────────────
# Fetch latest version from GitHub API
# ─────────────────────────────────────────────
fetch_latest_version() {
    log STEP "Fetching latest version from GitHub..."

    local api_url="https://api.github.com/repos/${GITHUB_REPO}/releases/latest"
    local response

    response="$(curl -fsSL \
        -H "Accept: application/vnd.github+json" \
        -H "X-GitHub-Api-Version: 2022-11-28" \
        "$api_url")" || die "Failed to fetch release info from GitHub API."

    LATEST_VERSION="$(echo "$response" | jq -r '.tag_name')"
    [[ -z "$LATEST_VERSION" || "$LATEST_VERSION" == "null" ]] && \
        die "Could not parse latest version from GitHub API response."

    # Strip leading 'v' for display but keep original tag
    VERSION_CLEAN="${LATEST_VERSION#v}"
    log INFO "Latest version: $LATEST_VERSION"
}

# ─────────────────────────────────────────────
# Check if already installed and up-to-date
# ─────────────────────────────────────────────
check_existing_installation() {
    log STEP "Checking existing installation..."

    if command -v "$TOOL_NAME" &>/dev/null; then
        local installed_path
        installed_path="$(command -v "$TOOL_NAME")"
        local installed_version
        installed_version="$("$installed_path" -version 2>&1 | grep -oP 'v?\d+\.\d+\.\d+' | head -1 || true)"

        if [[ -n "$installed_version" ]]; then
            log INFO "Installed version: $installed_version at $installed_path"
            if [[ "$installed_version" == "$LATEST_VERSION" || \
                  "$installed_version" == "$VERSION_CLEAN" ]]; then
                log INFO "${TOOL_NAME} is already up-to-date (${installed_version})."
                if [[ "${FORCE_REINSTALL:-false}" != "true" ]]; then
                    log INFO "Use FORCE_REINSTALL=true to reinstall anyway."
                    exit 0
                fi
                log WARN "FORCE_REINSTALL=true — proceeding with reinstall."
            else
                log INFO "Upgrade available: $installed_version -> $LATEST_VERSION"
            fi
        else
            log WARN "Could not determine installed version. Proceeding with install."
        fi
    else
        log INFO "${TOOL_NAME} is not currently installed."
    fi
}

# ─────────────────────────────────────────────
# Build download URLs
# ─────────────────────────────────────────────
build_download_urls() {
    log STEP "Building download URLs..."

    local base_url="https://github.com/${GITHUB_REPO}/releases/download/${LATEST_VERSION}"

    # nuclei naming convention: nuclei_<version>_<os>_<arch>.zip
    ARCHIVE_NAME="${TOOL_NAME}_${VERSION_CLEAN}_${OS}_${ARCH}.zip"
    CHECKSUM_NAME="${TOOL_NAME}_${VERSION_CLEAN}_checksums.txt"

    DOWNLOAD_URL="${base_url}/${ARCHIVE_NAME}"
    CHECKSUM_URL="${base_url}/${CHECKSUM_NAME}"

    log INFO "Archive  : $DOWNLOAD_URL"
    log INFO "Checksums: $CHECKSUM_URL"
}

# ─────────────────────────────────────────────
# Download file with retry
# ─────────────────────────────────────────────
download_file() {
    local url="$1"
    local dest="$2"
    local description="${3:-file}"
    local max_retries=3
    local attempt=0
    local wait_time=5

    log STEP "Downloading $description..."

    while (( attempt < max_retries )); do
        attempt=$(( attempt + 1 ))
        log INFO "Attempt $attempt/$max_retries: $url"

        if curl -fsSL \
            --connect-timeout 30 \
            --max-time 300 \
            --retry 2 \
            --retry-delay 3 \
            --progress-bar \
            -o "$dest" \
            "$url" 2>&1 | tee -a "$LOG_FILE"; then
            log INFO "Downloaded: $dest ($(du -sh "$dest" | cut -f1))"
            return 0
        fi

        log WARN "Download failed (attempt $attempt). Retrying in ${wait_time}s..."
        sleep "$wait_time"
        wait_time=$(( wait_time * 2 ))
    done

    die "Failed to download $description after $max_retries attempts."
}

# ─────────────────────────────────────────────
# Verify checksum
# ─────────────────────────────────────────────
verify_checksum() {
    log STEP "Verifying ${CHECKSUM_ALGO} checksum..."

    local archive="$1"
    local checksum_file="$2"

    if ! command -v sha256sum &>/dev/null && ! command -v shasum &>/dev/null; then
        log WARN "No checksum utility found (sha256sum/shasum). Skipping verification."
        return 0
    fi

    local expected_hash
    expected_hash="$(grep "$(basename "$archive")" "$checksum_file" | awk '{print $1}')"

    if [[ -z "$expected_hash" ]]; then
        log WARN "Checksum entry not found for $(basename "$archive"). Skipping."
        return 0
    fi

    log INFO "Expected: $expected_hash"

    local actual_hash
    if command -v sha256sum &>/dev/null; then
        actual_hash="$(sha256sum "$archive" | awk '{print $1}')"
    else
        actual_hash="$(shasum -a 256 "$archive" | awk '{print $1}')"
    fi

    log INFO "Actual  : $actual_hash"

    if [[ "$expected_hash" == "$actual_hash" ]]; then
        log INFO "Checksum verification PASSED."
    else
        die "Checksum mismatch! Expected: $expected_hash | Got: $actual_hash"
    fi
}

# ─────────────────────────────────────────────
# Extract archive
# ─────────────────────────────────────────────
extract_archive() {
    local archive="$1"
    local dest_dir="$2"

    log STEP "Extracting archive..."

    case "$archive" in
        *.zip)
            if command -v unzip &>/dev/null; then
                unzip -q -o "$archive" -d "$dest_dir"
            else
                log WARN "unzip not found, trying python3..."
                python3 -c "
import zipfile, sys
with zipfile.ZipFile(sys.argv[1]) as z:
    z.extractall(sys.argv[2])
" "$archive" "$dest_dir"
            fi
            ;;
        *.tar.gz|*.tgz)
            tar -xzf "$archive" -C "$dest_dir"
            ;;
        *.tar.bz2)
            tar -xjf "$archive" -C "$dest_dir"
            ;;
        *.tar.xz)
            tar -xJf "$archive" -C "$dest_dir"
            ;;
        *)
            die "Unknown archive format: $archive"
            ;;
    esac

    log INFO "Extracted to: $dest_dir"
    log INFO "Contents: $(ls -la "$dest_dir")"
}

# ─────────────────────────────────────────────
# Install binary
# ─────────────────────────────────────────────
install_binary() {
    log STEP "Installing ${TOOL_NAME} to ${INSTALL_DIR}..."

    # Find the binary in extracted directory
    local binary_path
    binary_path="$(find "$TMP_DIR" -type f -name "$TOOL_NAME" | head -1)"

    if [[ -z "$binary_path" ]]; then
        # Try without extension
        binary_path="$(find "$TMP_DIR" -type f -executable -name "${TOOL_NAME}*" | head -1)"
    fi

    [[ -z "$binary_path" ]] && die "Could not find ${TOOL_NAME} binary in extracted archive."

    log INFO "Found binary: $binary_path"

    # Backup existing installation
    local install_target="${INSTALL_DIR}/${TOOL_NAME}"
    if [[ -f "$install_target" ]]; then
        local backup="${install_target}.bak.$(date +%Y%m%d%H%M%S)"
        log INFO "Backing up existing binary to: $backup"
        $SUDO cp "$install_target" "$backup"
    fi

    # Install
    $SUDO install -m 755 -o root -g root "$binary_path" "$install_target" 2>/dev/null || \
    $SUDO cp "$binary_path" "$install_target" && $SUDO chmod 755 "$install_target"

    log INFO "Installed: $install_target"
}

# ─────────────────────────────────────────────
# Set permissions and verify
# ─────────────────────────────────────────────
set_permissions_and_verify() {
    log STEP "Setting permissions and verifying installation..."

    local install_target="${INSTALL_DIR}/${TOOL_NAME}"

    $SUDO chmod 755 "$install_target"
    $SUDO chown root:root "$install_target" 2>/dev/null || true

    # Verify binary is executable
    if [[ ! -x "$install_target" ]]; then
        die "Installed binary is not executable: $install_target"
    fi

    # Verify it runs
    local version_output
    version_output="$("$install_target" -version 2>&1 || true)"
    log INFO "Version check: $version_output"

    # Verify it's in PATH
    if command -v "$TOOL_NAME" &>/dev/null; then
        log INFO "${TOOL_NAME} is accessible in PATH: $(command -v "$TOOL_NAME")"
    else
        log WARN "${TOOL_NAME} installed to $INSTALL_DIR but may not be in PATH."
        log WARN "Add to PATH: export PATH=\"\$PATH:${INSTALL_DIR}\""
    fi
}

# ─────────────────────────────────────────────
# Post-install configuration (optional)
# ─────────────────────────────────────────────
post_install() {
    log STEP "Running post-install configuration..."

    # Create nuclei templates directory if needed
    local templates_dir="${HOME}/.local/nuclei-templates"
    if [[ ! -d "$templates_dir" ]]; then
        mkdir -p "$templates_dir"
        log INFO "Created templates directory: $templates_dir"
    fi

    # Update nuclei templates (non-fatal)
    if command -v "$TOOL_NAME" &>/dev/null; then
        log INFO "Updating nuclei templates (this may take a moment)..."
        "$TOOL_NAME" -update-templates -silent 2>&1 | tee -a "$LOG_FILE" || \
            log WARN "Template update failed (non-fatal). Run manually: nuclei -update-templates"
    fi

    log INFO "Post-install complete."
}

# ─────────────────────────────────────────────
# Summary
# ─────────────────────────────────────────────
print_summary() {
    echo ""
    log INFO "════════════════════════════════════════"
    log INFO "  Installation Summary"
    log INFO "════════════════════════════════════════"
    log INFO "  Tool     : ${TOOL_NAME}"
    log INFO "  Version  : ${LATEST_VERSION}"
    log INFO "  Location : ${INSTALL_DIR}/${TOOL_NAME}"
    log INFO "  Log file : ${LOG_FILE}"
    log INFO "════════════════════════════════════════"
    echo ""
}

# ─────────────────────────────────────────────
# Usage
# ─────────────────────────────────────────────
usage() {
    cat <<EOF
Usage: $(basename "$0") [OPTIONS]

Options:
  -f, --force       Force reinstall even if up-to-date
  -d, --dir DIR     Installation directory (default: /usr/local/bin)
  -n, --no-post     Skip post-install steps (template update)
  -h, --help        Show this help message

Environment variables:
  FORCE_REINSTALL=true    Force reinstall
  INSTALL_DIR=<path>      Override install directory
  GITHUB_TOKEN=<token>    GitHub API token (avoids rate limiting)

Examples:
  sudo ./$(basename "$0")
  sudo FORCE_REINSTALL=true ./$(basename "$0")
  sudo ./$(basename "$0") --dir /opt/bin --no-post
EOF
    exit 0
}

# ─────────────────────────────────────────────
# Parse arguments
# ─────────────────────────────────────────────
SKIP_POST_INSTALL=false

parse_args() {
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -f|--force)
                FORCE_REINSTALL=true
                shift
                ;;
            -d|--dir)
                INSTALL_DIR="$2"
                shift 2
                ;;
            -n|--no-post)
                SKIP_POST_INSTALL=true
                shift
                ;;
            -h|--help)
                usage
                ;;
            *)
                die "Unknown option: $1. Use --help for usage."
                ;;
        esac
    done
}

# ─────────────────────────────────────────────
# Ensure log file is writable
# ─────────────────────────────────────────────
init_log() {
    if ! touch "$LOG_FILE" 2>/dev/null; then
        LOG_FILE="/tmp/${TOOL_NAME}_install_$(date +%s).log"
        touch "$LOG_FILE"
    fi
    log INFO "Log file: $LOG_FILE"
}

# ─────────────────────────────────────────────
# Main
# ─────────────────────────────────────────────
main() {
    parse_args "$@"
    init_log

    log STEP "Starting ${TOOL_NAME} automated installer"
    log INFO "Script PID: $$"

    check_bash_version
    check_privileges
    check_dependencies
    detect_platform
    fetch_latest_version
    check_existing_installation
    build_download_urls

    local archive_path="${TMP_DIR}/${ARCHIVE_NAME}"
    local checksum_path="${TMP_DIR}/${CHECKSUM_NAME}"

    download_file "$DOWNLOAD_URL"  "$archive_path"  "main archive"
    download_file "$CHECKSUM_URL"  "$checksum_path" "checksums file"

    verify_checksum "$archive_path" "$checksum_path"
    extract_archive "$archive_path" "$TMP_DIR"
    install_binary
    set_permissions_and_verify

    if [[ "$SKIP_POST_INSTALL" != "true" ]]; then
        post_install
    fi

    print_summary
    log INFO "Installation completed successfully."
}

main "$@"
