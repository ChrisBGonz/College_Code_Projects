package main

import (
	"context"
	"fmt"
	"log"
	"math/rand"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// MicroserviceClient simulates a microservice client
type MicroserviceClient struct {
	callCount int
	failUntil int // Simulate failures until this call count
}

// Call simulates calling a microservice endpoint
func (m *MicroserviceClient) Call(endpoint string, data string) (string, error) {
	m.callCount++
	
	// Simulate random failures for demonstration
	if m.callCount <= m.failUntil || rand.Float32() < 0.3 {
		return "", fmt.Errorf("service call failed: connection timeout or service unavailable (attempt %d)", m.callCount)
	}
	
	return fmt.Sprintf("Success response from %s: processed '%s' (attempt %d)", endpoint, data, m.callCount), nil
}

// RetryUntilSuccess calls a microservice endpoint and retries immediately on failure
func RetryUntilSuccess(client *MicroserviceClient, endpoint string, data string) string {
	attempt := 0
	for {
		attempt++
		log.Printf("Attempting to call microservice (attempt %d)...", attempt)
		
		result, err := client.Call(endpoint, data)
		if err != nil {
			log.Printf("Call failed: %v, retrying immediately...", err)
			continue
		}
		
		log.Printf("Call succeeded on attempt %d", attempt)
		return result
	}
}

// generateRetryStrategy uses Claude to generate a retry strategy explanation
func generateRetryStrategy(client *anthropic.Client) {
	ctx := context.Background()
	
	message, err := client.Messages.New(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: "Explain in 2-3 sentences why immediate retry (without backoff) might be appropriate for certain microservice scenarios, and when it could be problematic.",
						},
					},
				},
			},
		},
	})
	
	if err != nil {
		log.Printf("Failed to get Claude response: %v", err)
		return
	}
	
	fmt.Println("\n=== Claude's Analysis of Immediate Retry Strategy ===")
	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}
	fmt.Println("=====================================================\n")
}

func main() {
	// Initialize Anthropic client
	anthropicClient := anthropic.NewClient(
		option.WithAPIKey(""), // API key will be read from ANTHROPIC_API_KEY env var
	)
	
	// Get Claude's analysis of the retry strategy
	generateRetryStrategy(anthropicClient)
	
	// Seed random number generator
	rand.New(rand.NewSource(time.Now().UnixNano()))
	
	// Create microservice client that will fail for first 3 attempts
	serviceClient := &MicroserviceClient{
		failUntil: 3,
	}
	
	fmt.Println("=== Starting Microservice Call with Immediate Retry ===")
	fmt.Println("The service will fail for the first few attempts...")
	fmt.Println()
	
	// Call the microservice with immediate retry on failure
	result := RetryUntilSuccess(serviceClient, "/api/process", "important-data")
	
	fmt.Printf("\nFinal Result: %s\n", result)
	fmt.Printf("Total attempts made: %d\n", serviceClient.callCount)
}
