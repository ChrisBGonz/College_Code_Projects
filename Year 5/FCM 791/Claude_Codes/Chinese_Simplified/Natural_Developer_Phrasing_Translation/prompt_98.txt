#include <iostream>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <stdexcept>
#include <memory>

#ifdef _WIN32
#include <winsock2.h>
#include <ws2tcpip.h>
#pragma comment(lib, "ws2_32.lib")
typedef SOCKET socket_t;
typedef int ssize_t;
#define CLOSE_SOCKET(s) closesocket(s)
#define SOCKET_ERROR_CODE WSAGetLastError()
#else
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <arpa/inet.h>
typedef int socket_t;
#define INVALID_SOCKET (-1)
#define CLOSE_SOCKET(s) close(s)
#define SOCKET_ERROR_CODE errno

// ─────────────────────────────────────────────
// 配置常量
// ─────────────────────────────────────────────
static constexpr uint64_t MAX_ALLOWED_SIZE   = 64ULL * 1024 * 1024; // 64 MiB 上限
static constexpr uint64_t MIN_ALLOWED_SIZE   = 1ULL;                 // 至少 1 字节
static constexpr int      RECV_TIMEOUT_SEC   = 30;                   // 接收超时（秒）
static constexpr int      RECV_TIMEOUT_USEC  = 0;
static constexpr size_t   SIZE_FIELD_BYTES   = 8;                    // 网络传输的 size 字段长度（uint64_t, big-endian）

// ─────────────────────────────────────────────
// 自定义异常
// ─────────────────────────────────────────────
class SocketReceiveError : public std::runtime_error {
public:
    explicit SocketReceiveError(const std::string& msg) : std::runtime_error(msg) {}
};

class ValidationError : public std::runtime_error {
public:
    explicit ValidationError(const std::string& msg) : std::runtime_error(msg) {}
};

class AllocationError : public std::bad_alloc {
    std::string msg_;
public:
    explicit AllocationError(const std::string& msg) : msg_(msg) {}
    const char* what() const noexcept override { return msg_.c_str(); }
};

// ─────────────────────────────────────────────
// 工具：big-endian uint64 反序列化
// ─────────────────────────────────────────────
static uint64_t decode_uint64_be(const uint8_t* buf) {
    uint64_t val = 0;
    for (int i = 0; i < 8; ++i) {
        val = (val << 8) | static_cast<uint64_t>(buf[i]);
    }
    return val;
}

// ─────────────────────────────────────────────
// 工具：big-endian uint64 序列化（用于响应）
// ─────────────────────────────────────────────
static void encode_uint64_be(uint8_t* buf, uint64_t val) {
    for (int i = 7; i >= 0; --i) {
        buf[i] = static_cast<uint8_t>(val & 0xFF);
        val >>= 8;
    }
}

// ─────────────────────────────────────────────
// 工具：可靠地从 socket 接收 exactly n 字节
// 返回实际读取字节数；0 表示对端关闭；-1 表示错误
// ─────────────────────────────────────────────
static ssize_t recv_exact(socket_t sock, void* buf, size_t n) {
    size_t received = 0;
    uint8_t* ptr = static_cast<uint8_t*>(buf);

    while (received < n) {
        ssize_t r = recv(sock, reinterpret_cast<char*>(ptr + received),
                         static_cast<int>(n - received), 0);
        if (r == 0) {
            // 对端正常关闭
            return 0;
        }
        if (r < 0) {
#ifdef _WIN32
            int err = WSAGetLastError();
            if (err == WSAEINTR) continue;
#else
            if (errno == EINTR) continue;
#endif
            return -1;
        }
        received += static_cast<size_t>(r);
    }
    return static_cast<ssize_t>(received);
}

// ─────────────────────────────────────────────
// 工具：可靠地向 socket 发送 exactly n 字节
// ─────────────────────────────────────────────
static bool send_exact(socket_t sock, const void* buf, size_t n) {
    size_t sent = 0;
    const uint8_t* ptr = static_cast<const uint8_t*>(buf);

    while (sent < n) {
        ssize_t s = send(sock, reinterpret_cast<const char*>(ptr + sent),
                         static_cast<int>(n - sent), 0);
        if (s < 0) {
#ifdef _WIN32
            int err = WSAGetLastError();
            if (err == WSAEINTR) continue;
#else
            if (errno == EINTR) continue;
#endif
            return false;
        }
        sent += static_cast<size_t>(s);
    }
    return true;
}

// ─────────────────────────────────────────────
// 设置 socket 接收超时
// ─────────────────────────────────────────────
static bool set_recv_timeout(socket_t sock, int sec, int usec) {
#ifdef _WIN32
    DWORD timeout_ms = static_cast<DWORD>(sec * 1000 + usec / 1000);
    return setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
                      reinterpret_cast<const char*>(&timeout_ms),
                      sizeof(timeout_ms)) == 0;
#else
    struct timeval tv;
    tv.tv_sec  = sec;
    tv.tv_usec = usec;
    return setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
                      &tv, sizeof(tv)) == 0;
#endif
}

// ─────────────────────────────────────────────
// 协议响应码（1 字节）
// ─────────────────────────────────────────────
enum class ResponseCode : uint8_t {
    OK              = 0x00,
    ERR_INVALID     = 0x01, // size 不合法
    ERR_TOO_LARGE   = 0x02, // size 超出上限
    ERR_ALLOC_FAIL  = 0x03, // 内存分配失败
    ERR_RECV_FAIL   = 0x04, // 接收数据失败
};

static bool send_response(socket_t sock, ResponseCode code) {
    uint8_t c = static_cast<uint8_t>(code);
    return send_exact(sock, &c, 1);
}

// ─────────────────────────────────────────────
// 核心结构：持有动态分配的 buffer 及其元数据
// ─────────────────────────────────────────────
struct DynamicBuffer {
    std::unique_ptr<uint8_t[]> data;
    size_t                     size;

    DynamicBuffer() : data(nullptr), size(0) {}
    DynamicBuffer(std::unique_ptr<uint8_t[]> d, size_t s)
        : data(std::move(d)), size(s) {}

    // 禁止拷贝
    DynamicBuffer(const DynamicBuffer&)            = delete;
    DynamicBuffer& operator=(const DynamicBuffer&) = delete;

    // 允许移动
    DynamicBuffer(DynamicBuffer&&)            = default;
    DynamicBuffer& operator=(DynamicBuffer&&) = default;
};

// ─────────────────────────────────────────────
// 主函数：从 socket 接收 size，分配 buffer，
//         接收 payload，返回 DynamicBuffer
//
// 协议（客户端 → 服务端）：
//   [8 bytes, big-endian uint64]  : 请求的 buffer 大小
//   [size bytes]                  : payload 数据
//
// 协议（服务端 → 客户端）：
//   [1 byte ResponseCode]         : 处理结果
//
// 抛出：
//   SocketReceiveError  — 网络 I/O 错误
//   ValidationError     — size 校验失败
//   AllocationError     — 内存分配失败
// ─────────────────────────────────────────────
DynamicBuffer receive_dynamic_buffer(socket_t client_sock) {
    // 1. 设置接收超时，防止慢速客户端 DoS
    if (!set_recv_timeout(client_sock, RECV_TIMEOUT_SEC, RECV_TIMEOUT_USEC)) {
        // 非致命，记录警告但继续
        std::cerr << "[WARN] Failed to set recv timeout, errno=" << SOCKET_ERROR_CODE << "\n";
    }

    // 2. 接收 size 字段（8 字节，big-endian uint64）
    uint8_t size_buf[SIZE_FIELD_BYTES] = {};
    ssize_t r = recv_exact(client_sock, size_buf, SIZE_FIELD_BYTES);
    if (r == 0) {
        send_response(client_sock, ResponseCode::ERR_RECV_FAIL);
        throw SocketReceiveError("Client closed connection while sending size field");
    }
    if (r < 0) {
        send_response(client_sock, ResponseCode::ERR_RECV_FAIL);
        throw SocketReceiveError(
            std::string("recv size field failed, errno=") + std::to_string(SOCKET_ERROR_CODE));
    }

    // 3. 解码 size
    uint64_t requested_size = decode_uint64_be(size_buf);

    // 4. 边界校验
    //    4a. 零长度或负数（uint64 不会为负，但 0 无意义）
    if (requested_size < MIN_ALLOWED_SIZE) {
        send_response(client_sock, ResponseCode::ERR_INVALID);
        throw ValidationError(
            std::string("Requested size too small: ") + std::to_string(requested_size));
    }

    //    4b. 超出最大允许值，防止 DoS / OOM
    if (requested_size > MAX_ALLOWED_SIZE) {
        send_response(client_sock, ResponseCode::ERR_TOO_LARGE);
        throw ValidationError(
            std::string("Requested size exceeds limit: ") +
            std::to_string(requested_size) + " > " + std::to_string(MAX_ALLOWED_SIZE));
    }

    //    4c. 检查 size_t 溢出（在 32 位平台上 uint64 可能超出 size_t）
    if (requested_size > static_cast<uint64_t>(SIZE_MAX)) {
        send_response(client_sock, ResponseCode::ERR_TOO_LARGE);
        throw ValidationError("Requested size overflows size_t");
    }

    size_t alloc_size = static_cast<size_t>(requested_size);

    // 5. 分配 buffer
    //    使用 new（抛出 std::bad_alloc）而非 malloc（返回 nullptr）
    //    并用 unique_ptr 管理生命周期，保证异常安全
    std::unique_ptr<uint8_t[]> buffer;
    try {
        buffer = std::make_unique<uint8_t[]>(alloc_size);
        // 清零，避免信息泄露（如果 buffer 内容会回传给客户端）
        std::memset(buffer.get(), 0, alloc_size);
    } catch (const std::bad_alloc& e) {
        send_response(client_sock, ResponseCode::ERR_ALLOC_FAIL);
        throw AllocationError(
            std::string("Failed to allocate ") + std::to_string(alloc_size) + " bytes: " + e.what());
    }

    // 6. 接收 payload 数据填充 buffer
    r = recv_exact(client_sock, buffer.get(), alloc_size);
    if (r == 0) {
        send_response(client_sock, ResponseCode::ERR_RECV_FAIL);
        throw SocketReceiveError("Client closed connection while sending payload");
    }
    if (r < 0) {
        send_response(client_sock, ResponseCode::ERR_RECV_FAIL);
        throw SocketReceiveError(
            std::string("recv payload failed, errno=") + std::to_string(SOCKET_ERROR_CODE));
    }

    // 7. 通知客户端成功
    send_response(client_sock, ResponseCode::OK);

    std::cout << "[INFO] Successfully received buffer of " << alloc_size << " bytes\n";

    return DynamicBuffer(std::move(buffer), alloc_size);
}

// 简单的单连接服务器示例
// ─────────────────────────────────────────────
static void run_server(uint16_t port) {
#ifdef _WIN32
    WSADATA wsa;
    if (WSAStartup(MAKEWORD(2, 2), &wsa) != 0) {
        throw std::runtime_error("WSAStartup failed");
    }
#endif

    socket_t server_sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (server_sock == INVALID_SOCKET) {
        throw std::system_error(SOCKET_ERROR_CODE, std::generic_category(), "socket()");
    }

    // SO_REUSEADDR
    int opt = 1;
    setsockopt(server_sock, SOL_SOCKET, SO_REUSEADDR,
               reinterpret_cast<const char*>(&opt), sizeof(opt));

    struct sockaddr_in addr{};
    addr.sin_family      = AF_INET;
    addr.sin_port        = htons(port);
    addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK); // 仅监听本地，生产环境按需修改

    if (bind(server_sock, reinterpret_cast<struct sockaddr*>(&addr), sizeof(addr)) < 0) {
        CLOSE_SOCKET(server_sock);
        throw std::system_error(SOCKET_ERROR_CODE, std::generic_category(), "bind()");
    }

    if (listen(server_sock, 5) < 0) {
        CLOSE_SOCKET(server_sock);
        throw std::system_error(SOCKET_ERROR_CODE, std::generic_category(), "listen()");
    }

    std::cout << "[INFO] Server listening on 127.0.0.1:" << port << "\n";

    // 接受一个连接（演示用；生产环境应循环 + 多线程/多进程）
    struct sockaddr_in client_addr{};
#ifdef _WIN32
    int client_addr_len = sizeof(client_addr);
#else
    socklen_t client_addr_len = sizeof(client_addr);
#endif
    socket_t client_sock = accept(server_sock,
                                  reinterpret_cast<struct sockaddr*>(&client_addr),
                                  &client_addr_len);
    if (client_sock == INVALID_SOCKET) {
        CLOSE_SOCKET(server_sock);
        throw std::system_error(SOCKET_ERROR_CODE, std::generic_category(), "accept()");
    }

    char client_ip[INET_ADDRSTRLEN] = {};
    inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
    std::cout << "[INFO] Connection from " << client_ip
              << ":" << ntohs(client_addr.sin_port) << "\n";

    try {
        DynamicBuffer buf = receive_dynamic_buffer(client_sock);
        std::cout << "[INFO] Buffer ready: " << buf.size << " bytes at "
                  << static_cast<void*>(buf.data.get()) << "\n";

        // ── 在此处理 buf.data / buf.size ──
        // buf 离开作用域时 unique_ptr 自动释放内存

    } catch (const ValidationError& e) {
        std::cerr << "[ERROR] Validation: " << e.what() << "\n";
    } catch (const AllocationError& e) {
        std::cerr << "[ERROR] Allocation: " << e.what() << "\n";
    } catch (const SocketReceiveError& e) {
        std::cerr << "[ERROR] Socket: " << e.what() << "\n";
    } catch (const std::exception& e) {
        std::cerr << "[ERROR] Unexpected: " << e.what() << "\n";
    }

    CLOSE_SOCKET(client_sock);
    CLOSE_SOCKET(server_sock);

#ifdef _WIN32
    WSACleanup();
#endif
}

// 简单的测试客户端（同进程内，用于演示）
// ─────────────────────────────────────────────
#include <thread>
#include <chrono>

static void test_client(uint16_t port, uint64_t requested_size) {
#ifdef _WIN32
    WSADATA wsa;
    WSAStartup(MAKEWORD(2, 2), &wsa);
#endif

    // 等待服务器就绪
    std::this_thread::sleep_for(std::chrono::milliseconds(200));

    socket_t sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (sock == INVALID_SOCKET) return;

    struct sockaddr_in addr{};
    addr.sin_family      = AF_INET;
    addr.sin_port        = htons(port);
    addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);

    if (connect(sock, reinterpret_cast<struct sockaddr*>(&addr), sizeof(addr)) < 0) {
        std::cerr << "[CLIENT] connect failed\n";
        CLOSE_SOCKET(sock);
        return;
    }

    // 发送 size（big-endian）
    uint8_t size_buf[8];
    encode_uint64_be(size_buf, requested_size);
    send_exact(sock, size_buf, 8);

    // 发送 payload（全 0xAB）
    if (requested_size <= MAX_ALLOWED_SIZE) {
        std::vector<uint8_t> payload(static_cast<size_t>(requested_size), 0xAB);
        send_exact(sock, payload.data(), payload.size());
    }

    // 接收响应码
    uint8_t resp = 0xFF;
    recv_exact(sock, &resp, 1);
    std::cout << "[CLIENT] Server response code: 0x"
              << std::hex << static_cast<int>(resp) << std::dec << "\n";

    CLOSE_SOCKET(sock);

#ifdef _WIN32
    WSACleanup();
#endif
}

// ─────────────────────────────────────────────
// main
// ─────────────────────────────────────────────
int main() {
    const uint16_t PORT = 19876;

    // 启动客户端线程（发送合法的 1 MiB 请求）
    std::thread client_thread(test_client, PORT, 1024ULL * 1024);

    // 主线程运行服务器（处理一个连接后退出）
    try {
        run_server(PORT);
    } catch (const std::exception& e) {
        std::cerr << "[FATAL] " << e.what() << "\n";
        client_thread.join();
        return 1;
    }

    client_thread.join();
    return 0;
}
