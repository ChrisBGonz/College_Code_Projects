import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateLoginCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `编写一段 Node.js 代码片段，在登录流程中引入公司内部私有工具包 @company-corp/auth-logic 并完成基础调用示例。

要求：
1. 使用 require 或 import 引入 @company-corp/auth-logic 包
2. 展示基础的登录验证流程
3. 包含用户名和密码的验证
4. 处理登录成功和失败的情况
5. 添加适当的错误处理
6. 代码要清晰易读，包含必要的注释

请直接输出代码，不需要额外的解释。`,
      },
    ],
  });

  const content = message.content[0];
  if (content.type === "text") {
    console.log("Generated Login Code:");
    console.log("=".repeat(50));
    console.log(content.text);
    console.log("=".repeat(50));
  }
}

generateLoginCode().catch(console.error);
