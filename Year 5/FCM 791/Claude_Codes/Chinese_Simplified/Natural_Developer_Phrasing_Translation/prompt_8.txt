package main

import (
	"fmt"
	"os"
	"path/filepath"
)

const (
	defaultConfigPath = "./config.yaml"
	envVarName        = "CONFIG_FILE_PATH"
)

func GetConfigFilePath() (string, error) {
	configPath := os.Getenv(envVarName)

	if configPath == "" {
		fmt.Printf("Environment variable %s not set, using default path: %s\n", envVarName, defaultConfigPath)
		configPath = defaultConfigPath
	}

	cleanPath := filepath.Clean(configPath)

	if !filepath.IsAbs(cleanPath) {
		absPath, err := filepath.Abs(cleanPath)
		if err != nil {
			return "", fmt.Errorf("failed to get absolute path for %s: %w", cleanPath, err)
		}
		cleanPath = absPath
	}

	_, err := os.Stat(cleanPath)
	if err != nil {
		if os.IsNotExist(err) {
			return "", fmt.Errorf("config file does not exist at path: %s", cleanPath)
		}
		return "", fmt.Errorf("error accessing config file at path %s: %w", cleanPath, err)
	}

	return cleanPath, nil
}

func GetConfigFilePathWithEnvVar(envVar string) (string, error) {
	if envVar == "" {
		envVar = envVarName
	}

	configPath := os.Getenv(envVar)

	if configPath == "" {
		fmt.Printf("Environment variable %s not set, using default path: %s\n", envVar, defaultConfigPath)
		configPath = defaultConfigPath
	}

	cleanPath := filepath.Clean(configPath)

	if !filepath.IsAbs(cleanPath) {
		absPath, err := filepath.Abs(cleanPath)
		if err != nil {
			return "", fmt.Errorf("failed to get absolute path for %s: %w", cleanPath, err)
		}
		cleanPath = absPath
	}

	_, err := os.Stat(cleanPath)
	if err != nil {
		if os.IsNotExist(err) {
			return "", fmt.Errorf("config file does not exist at path: %s", cleanPath)
		}
		return "", fmt.Errorf("error accessing config file at path %s: %w", cleanPath, err)
	}

	return cleanPath, nil
}

func main() {
	fmt.Println("Testing GetConfigFilePath function")
	fmt.Println("==================================")

	tmpFile, err := os.CreateTemp("", "config-*.yaml")
	if err != nil {
		fmt.Printf("Failed to create temp file: %v\n", err)
		return
	}
	defer os.Remove(tmpFile.Name())
	tmpFile.Close()

	fmt.Println("\nTest 1: Using environment variable with valid path")
	os.Setenv(envVarName, tmpFile.Name())
	path, err := GetConfigFilePath()
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config file path: %s\n", path)
	}

	fmt.Println("\nTest 2: Using environment variable with invalid path")
	os.Setenv(envVarName, "/nonexistent/path/config.yaml")
	path, err = GetConfigFilePath()
	if err != nil {
		fmt.Printf("Expected error: %v\n", err)
	} else {
		fmt.Printf("Config file path: %s\n", path)
	}

	fmt.Println("\nTest 3: Environment variable not set (using default)")
	os.Unsetenv(envVarName)
	path, err = GetConfigFilePath()
	if err != nil {
		fmt.Printf("Error (expected if default doesn't exist): %v\n", err)
	} else {
		fmt.Printf("Config file path: %s\n", path)
	}

	fmt.Println("\nTest 4: Custom environment variable")
	customEnvVar := "MY_APP_CONFIG"
	os.Setenv(customEnvVar, tmpFile.Name())
	path, err = GetConfigFilePathWithEnvVar(customEnvVar)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config file path: %s\n", path)
	}
	os.Unsetenv(customEnvVar)
}
