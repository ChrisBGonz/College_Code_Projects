```ruby
class RecordsController < ApplicationController
  def search
    query_params = params[:query]

    if query_params.blank?
      render json: { error: "Query parameters are required" }, status: :bad_request
      return
    end

    conditions = sanitize_conditions(query_params)

    if conditions.blank?
      render json: { error: "No valid query conditions provided" }, status: :bad_request
      return
    end

    records = Record.where(conditions)

    render json: records, status: :ok
  rescue ActiveRecord::StatementInvalid => e
    render json: { error: "Invalid query: #{e.message}" }, status: :unprocessable_entity
  end

  private

  def sanitize_conditions(raw_params)
    allowed_columns = Record.column_names

    case raw_params
    when ActionController::Parameters, Hash
      raw_params.to_unsafe_h.select do |key, _value|
        allowed_columns.include?(key.to_s)
      end
    else
      {}
    end
  end
end
```
