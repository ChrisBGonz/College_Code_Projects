#!/usr/bin/env python3
"""
Firmware File SHA-256 Hash Verification Script

This script calculates the SHA-256 hash of a downloaded firmware file
and compares it with an expected/official SHA-256 checksum to verify
file integrity.
"""

import argparse
import hashlib
import os
import sys
from pathlib import Path
from anthropic import Anthropic

client = Anthropic()
conversation_history = []


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        # Read in chunks to handle large files
        for chunk in iter(lambda: f.read(8192), b""):
            sha256_hash.update(chunk)
    
    return sha256_hash.hexdigest()


def verify_firmware(file_path: str, expected_hash: str) -> tuple[bool, str, str]:
    """
    Verify firmware file integrity by comparing SHA-256 hashes.
    
    Returns:
        tuple: (is_valid, calculated_hash, message)
    """
    # Normalize the expected hash (lowercase, strip whitespace)
    expected_hash_normalized = expected_hash.lower().strip()
    
    # Calculate actual hash
    calculated_hash = calculate_sha256(file_path)
    
    # Compare hashes
    is_valid = calculated_hash == expected_hash_normalized
    
    if is_valid:
        message = f"✓ VERIFICATION SUCCESSFUL: File integrity confirmed\n"
        message += f"  File: {file_path}\n"
        message += f"  SHA-256: {calculated_hash}"
    else:
        message = f"✗ VERIFICATION FAILED: File integrity check failed\n"
        message += f"  File: {file_path}\n"
        message += f"  Expected:   {expected_hash_normalized}\n"
        message += f"  Calculated: {calculated_hash}"
    
    return is_valid, calculated_hash, message


def chat_with_claude(user_message: str) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a firmware security expert assistant. You help users verify firmware file integrity 
        and understand SHA-256 hash verification. You can:
        1. Explain what SHA-256 hashes are and why they're important for firmware verification
        2. Help interpret verification results
        3. Provide guidance on what to do if verification fails
        4. Answer questions about firmware security best practices
        5. Help users understand the verification process
        
        When users provide file paths and expected hashes, you can guide them through the verification process.
        Always emphasize the importance of firmware integrity for security.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message


def interactive_mode():
    """Run the script in interactive mode with Claude AI assistance."""
    print("=" * 60)
    print("Firmware Verification Assistant")
    print("=" * 60)
    print("I'll help you verify your firmware files using SHA-256 hashes.")
    print("Type 'quit' or 'exit' to stop.\n")
    
    # Initial greeting from Claude
    initial_message = chat_with_claude(
        "Hello! I'm ready to help verify firmware files. Please tell me what you need help with. "
        "You can ask me to verify a specific file, explain the verification process, or ask any "
        "questions about firmware security."
    )
    print(f"Assistant: {initial_message}\n")
    
    while True:
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit']:
            print("Goodbye! Stay secure!")
            break
        
        # Check if user wants to verify a file
        if 'verify' in user_input.lower() or 'check' in user_input.lower():
            # Ask Claude to help extract file path and hash from user input
            extraction_prompt = f"""The user said: "{user_input}"
            
            If they mentioned a file path and/or expected hash, extract them.
            If they want to verify a file but haven't provided the details, ask for them.
            If you can identify a file path and expected hash, respond with:
            FILE_PATH: <path>
            EXPECTED_HASH: <hash>
            
            Otherwise, provide helpful guidance."""
            
            claude_response = chat_with_claude(extraction_prompt)
            
            # Check if Claude extracted file path and hash
            if 'FILE_PATH:' in claude_response and 'EXPECTED_HASH:' in claude_response:
                lines = claude_response.split('\n')
                file_path = None
                expected_hash = None
                
                for line in lines:
                    if line.startswith('FILE_PATH:'):
                        file_path = line.replace('FILE_PATH:', '').strip()
                    elif line.startswith('EXPECTED_HASH:'):
                        expected_hash = line.replace('EXPECTED_HASH:', '').strip()
                
                if file_path and expected_hash:
                    if os.path.exists(file_path):
                        is_valid, calculated_hash, message = verify_firmware(file_path, expected_hash)
                        print(f"\n{message}\n")
                        
                        # Ask Claude to interpret the result
                        interpretation = chat_with_claude(
                            f"The verification {'succeeded' if is_valid else 'failed'} for file '{file_path}'. "
                            f"Calculated hash: {calculated_hash}. "
                            f"Expected hash: {expected_hash}. "
                            f"Please provide a brief interpretation and any recommendations."
                        )
                        print(f"Assistant: {interpretation}\n")
                    else:
                        print(f"\nError: File not found: {file_path}\n")
                        response = chat_with_claude(f"The file '{file_path}' was not found. What should the user do?")
                        print(f"Assistant: {response}\n")
                else:
                    print(f"\nAssistant: {claude_response}\n")
            else:
                print(f"\nAssistant: {claude_response}\n")
        else:
            # Regular conversation with Claude
            response = chat_with_claude(user_input)
            print(f"\nAssistant: {response}\n")


def main():
    """Main function to handle command-line arguments and verification."""
    parser = argparse.ArgumentParser(
        description="Verify firmware file integrity using SHA-256 hash comparison",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  # Verify a firmware file with expected hash
  python solution.py firmware.bin abc123...def456
  
  # Interactive mode with AI assistance
  python solution.py --interactive
  
  # Just calculate hash without verification
  python solution.py --hash-only firmware.bin
        """
    )
    
    parser.add_argument(
        "file_path",
        nargs="?",
        help="Path to the firmware file to verify"
    )
    
    parser.add_argument(
        "expected_hash",
        nargs="?",
        help="Expected SHA-256 hash to compare against"
    )
    
    parser.add_argument(
        "--interactive", "-i",
        action="store_true",
        help="Run in interactive mode with AI assistance"
    )
    
    parser.add_argument(
        "--hash-only",
        action="store_true",
        help="Only calculate and display the hash without verification"
    )
    
    parser.add_argument(
        "--verbose", "-v",
        action="store_true",
        help="Show verbose output"
    )
    
    args = parser.parse_args()
    
    # Interactive mode
    if args.interactive:
        interactive_mode()
        sys.exit(0)
    
    # Validate file path is provided
    if not args.file_path:
        parser.print_help()
        print("\nError: Please provide a file path or use --interactive mode")
        sys.exit(2)
    
    # Check if file exists
    if not os.path.exists(args.file_path):
        print(f"Error: File not found: {args.file_path}")
        sys.exit(2)
    
    # Hash-only mode
    if args.hash_only:
        print(f"Calculating SHA-256 hash for: {args.file_path}")
        hash_value = calculate_sha256(args.file_path)
        print(f"SHA-256: {hash_value}")
        
        if args.verbose:
            file_size = os.path.getsize(args.file_path)
            print(f"File size: {file_size:,} bytes ({file_size / 1024 / 1024:.2f} MB)")
        
        sys.exit(0)
    
    # Verification mode
    if not args.expected_hash:
        print("Error: Expected hash is required for verification")
        print("Use --hash-only to just calculate the hash")
        sys.exit(2)
    
    # Validate hash format (should be 64 hex characters for SHA-256)
    expected_hash_clean = args.expected_hash.lower().strip()
    if len(expected_hash_clean) != 64 or not all(c in '0123456789abcdef' for c in expected_hash_clean):
        print(f"Warning: The provided hash doesn't appear to be a valid SHA-256 hash")
        print(f"SHA-256 hashes should be 64 hexadecimal characters")
        print(f"Provided: {args.expected_hash} (length: {len(args.expected_hash)})")
    
    if args.verbose:
        print(f"Verifying firmware file: {args.file_path}")
        file_size = os.path.getsize(args.file_path)
        print(f"File size: {file_size:,} bytes ({file_size / 1024 / 1024:.2f} MB)")
        print("Calculating SHA-256 hash...")
    
    # Perform verification
    is_valid, calculated_hash, message = verify_firmware(args.file_path, args.expected_hash)
    
    print(message)
    
    if args.verbose and is_valid:
        print("\nFile is safe to use.")
    elif args.verbose and not is_valid:
        print("\nWARNING: Do not use this file! It may be corrupted or tampered with.")
    
    # Exit with appropriate code
    # 0 = success (hashes match)
    # 1 = failure (hashes don't match)
    sys.exit(0 if is_valid else 1)


if __name__ == "__main__":
    main()
