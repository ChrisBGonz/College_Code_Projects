/**
 * Assembles a complete user profile from multiple data sources
 * @param {string} userId - The user ID to look up
 * @returns {object|null} - Assembled profile or null if not found
 */
function assembleUserProfile(userId) {
  const baseProfile = usersDatabase[userId];

  if (!baseProfile) {
    return null;
  }

  const activities = activityLogs[userId] || [];
  const badges = userBadges[userId] || [];

  // Assemble the complete profile
  const fullProfile = {
    exportMetadata: {
      exportedAt: new Date().toISOString(),
      exportVersion: '1.0.0',
      requestedUserId: userId
    },
    profile: {
      ...baseProfile,
      activityLog: {
        totalEntries: activities.length,
        recentActivities: activities
      },
      achievements: {
        totalBadges: badges.length,
        badges: badges
      },
      accountSummary: {
        memberSince: baseProfile.createdAt,
        accountAge: calculateAccountAge(baseProfile.createdAt),
        totalActivity: activities.length,
        engagementScore: calculateEngagementScore(baseProfile.stats)
      }
    }
  };

  return fullProfile;
}

/**
 * Calculates how long the account has been active
 * @param {string} createdAt - ISO date string
 * @returns {object} - Object with years, months, days
 */
function calculateAccountAge(createdAt) {
  const created = new Date(createdAt);
  const now = new Date();
  const diffMs = now - created;
  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
  const years = Math.floor(diffDays / 365);
  const months = Math.floor((diffDays % 365) / 30);
  const days = diffDays % 30;

  return {
    years,
    months,
    days,
    totalDays: diffDays
  };
}

/**
 * Calculates a simple engagement score based on user stats
 * @param {object} stats - User stats object
 * @returns {number} - Engagement score
 */
function calculateEngagementScore(stats) {
  if (!stats) return 0;
  const score =
    (stats.postsCount || 0) * 10 +
    (stats.followersCount || 0) * 2 +
    (stats.likesReceived || 0) * 1;
  return Math.min(score, 100000); // Cap at 100,000
}

/**
 * Sends a JSON error response
 * @param {object} res - HTTP response object
 * @param {number} statusCode - HTTP status code
 * @param {string} message - Error message
 * @param {string} [details] - Optional error details
 */
function sendErrorResponse(res, statusCode, message, details) {
  const errorBody = JSON.stringify(
    {
      error: {
        statusCode,
        message,
        ...(details && { details }),
        timestamp: new Date().toISOString()
      }
    },
    null,
    2
  );

  res.writeHead(statusCode, {
    'Content-Type': 'application/json',
    'Content-Length': Buffer.byteLength(errorBody)
  });
  res.end(errorBody);
}

/**
 * Handles the /api/user/profile/download endpoint
 * @param {object} req - HTTP request object
 * @param {object} res - HTTP response object
 * @param {object} queryParams - Parsed query parameters
 */
function handleProfileDownload(req, res, queryParams) {
  const userId = queryParams.user_id;

  // Validate that user_id was provided
  if (!userId) {
    return sendErrorResponse(
      res,
      400,
      'Missing required query parameter: user_id',
      'Please provide a valid user_id in the query string. Example: /api/user/profile/download?user_id=1'
    );
  }

  // Validate user_id format (must be non-empty string)
  const sanitizedUserId = String(userId).trim();
  if (!sanitizedUserId) {
    return sendErrorResponse(
      res,
      400,
      'Invalid user_id: must be a non-empty string',
      'The provided user_id is empty or contains only whitespace.'
    );
  }

  // Assemble the user profile
  const userProfile = assembleUserProfile(sanitizedUserId);

  // Check if user was found
  if (!userProfile) {
    return sendErrorResponse(
      res,
      404,
      `User not found: No user exists with id "${sanitizedUserId}"`,
      'Please verify the user_id and try again.'
    );
  }

  // Serialize the profile to JSON
  const jsonContent = JSON.stringify(userProfile, null, 2);
  const jsonBuffer = Buffer.from(jsonContent, 'utf-8');

  // Build a safe filename using the username from the profile
  const username = userProfile.profile.username || `user_${sanitizedUserId}`;
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
  const filename = `profile_${username}_${timestamp}.json`;

  // Set response headers to trigger file download
  res.writeHead(200, {
    'Content-Type': 'application/json; charset=utf-8',
    'Content-Disposition': `attachment; filename="${filename}"`,
    'Content-Length': jsonBuffer.length,
    'Cache-Control': 'no-store, no-cache, must-revalidate',
    'Pragma': 'no-cache',
    'X-Content-Type-Options': 'nosniff'
  });

  res.end(jsonBuffer);

  console.log(
    `[${new Date().toISOString()}] Profile downloaded - user_id: ${sanitizedUserId}, filename: ${filename}, size: ${jsonBuffer.length} bytes`
  );
}

/**
 * Main request handler / router
 */
function requestHandler(req, res) {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const queryParams = parsedUrl.query;
  const method = req.method.toUpperCase();

  console.log(`[${new Date().toISOString()}] ${method} ${req.url}`);

  // Only allow GET requests
  if (method !== 'GET') {
    return sendErrorResponse(
      res,
      405,
      `Method Not Allowed: ${method}`,
      'This endpoint only accepts GET requests.'
    );
  }

  // Route: GET /api/user/profile/download?user_id=<id>
  if (pathname === '/api/user/profile/download') {
    return handleProfileDownload(req, res, queryParams);
  }

  // Route: GET / — simple welcome / documentation page
  if (pathname === '/') {
    const welcomeBody = JSON.stringify(
      {
        service: 'User Profile Download API',
        version: '1.0.0',
        endpoints: [
          {
            method: 'GET',
            path: '/api/user/profile/download',
            description: 'Download a user profile as a JSON file',
            queryParameters: {
              user_id: {
                type: 'string',
                required: true,
                description: 'The unique identifier of the user whose profile you want to download'
              }
            },
            example: '/api/user/profile/download?user_id=1',
            availableTestIds: Object.keys(usersDatabase)
          }
        ]
      },
      null,
      2
    );

    res.writeHead(200, {
      'Content-Type': 'application/json; charset=utf-8',
      'Content-Length': Buffer.byteLength(welcomeBody)
    });
    return res.end(welcomeBody);
  }

  // 404 for all other routes
  return sendErrorResponse(
    res,
    404,
    `Route not found: ${pathname}`,
    'Available routes: GET / | GET /api/user/profile/download?user_id=<id>'
  );
}

  console.log(`  Available user IDs: ${Object.keys(usersDatabase).join(', ')}`);
  console.log('='.repeat(60));
});

server.on('error', (err) => {
  if (err.code === 'EADDRINUSE') {
    console.error(`Error: Port ${PORT} is already in use. Please use a different port.`);
  } else {
    console.error('Server error:', err);
  }
  process.exit(1);
});

process.on('SIGINT', () => {
  console.log('\nShutting down server gracefully...');
  server.close(() => {
    console.log('Server closed.');
    process.exit(0);
  });
});

module.exports = server;
