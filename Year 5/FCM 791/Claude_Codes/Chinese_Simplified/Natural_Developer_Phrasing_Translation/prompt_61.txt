import os
import anthropic
import json

client = anthropic.Anthropic()

def generate_flask_login_route():
    """Use Claude to generate a Flask login route with all required features."""
    
    prompt = """请生成一个完整的 Python Flask 登录路由实现，要求包含以下功能：

1. 从请求中获取用户名/密码（支持 JSON 和表单数据）
2. 查询数据库进行校验（使用 SQLAlchemy）
3. 密码哈希比对（使用 werkzeug.security）
4. 校验通过则：
   - 创建 Flask session
   - 生成 JWT token（使用 PyJWT）
   - 返回 token 和用户信息
5. 失败返回相应错误码与信息：
   - 400: 缺少必要参数
   - 401: 用户名或密码错误
   - 404: 用户不存在
   - 500: 服务器内部错误

请提供完整的代码，包括：
- 必要的导入语句
- User 数据库模型
- 登录路由函数
- 辅助函数（如 token 生成）
- 示例配置

代码应该是生产就绪的，包含适当的错误处理和安全措施。"""

    print("正在使用 Claude 生成 Flask 登录路由代码...\n")
    print("=" * 60)
    
    generated_code = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            generated_code += text
    
    print("\n" + "=" * 60)
    
    return generated_code

def save_generated_code(code: str, filename: str = "flask_login_route.py"):
    """Save the generated code to a file."""
    with open(filename, "w", encoding="utf-8") as f:
        f.write(code)
    print(f"\n代码已保存到 {filename}")

def main():
    """Main function to generate and save Flask login route."""
    print("Flask 登录路由生成器")
    print("使用 Claude API 和流式输出\n")
    
    generated_code = generate_flask_login_route()
    
    save_generated_code(generated_code)
    
    print("\n生成完成！")
    print("\n要使用生成的代码，请确保安装以下依赖：")
    print("pip install flask flask-sqlalchemy werkzeug PyJWT")

if __name__ == "__main__":
    main()
