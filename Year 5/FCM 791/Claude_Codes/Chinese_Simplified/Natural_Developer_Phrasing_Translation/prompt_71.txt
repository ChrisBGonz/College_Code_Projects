
```python
import pickle
import os
import time
from anthropic import Anthropic

# Initialize the Anthropic client
client = Anthropic()

# Game state file path
GAME_STATE_FILE = "game_state.pkl"

class GameState:
    """Complex game state object to be serialized/deserialized."""
    
    def __init__(self, player_name: str):
        self.player_name = player_name
        self.level = 1
        self.health = 100
        self.max_health = 100
        self.experience = 0
        self.gold = 50
        self.inventory = []
        self.skills = {"attack": 1, "defense": 1, "magic": 1}
        self.quests_completed = []
        self.current_location = "Starting Village"
        self.achievements = []
        self.play_time = 0
        self.save_timestamp = None
        self.conversation_history = []
    
    def add_item(self, item: str):
        """Add item to inventory."""
        self.inventory.append(item)
    
    def gain_experience(self, amount: int):
        """Gain experience and potentially level up."""
        self.experience += amount
        while self.experience >= self.level * 100:
            self.experience -= self.level * 100
            self.level += 1
            self.max_health += 10
            self.health = self.max_health
            return f"Level up! Now level {self.level}!"
        return f"Gained {amount} experience."
    
    def complete_quest(self, quest_name: str):
        """Mark a quest as completed."""
        if quest_name not in self.quests_completed:
            self.quests_completed.append(quest_name)
    
    def add_achievement(self, achievement: str):
        """Add an achievement."""
        if achievement not in self.achievements:
            self.achievements.append(achievement)
    
    def update_location(self, location: str):
        """Update current location."""
        self.current_location = location
    
    def to_dict(self) -> dict:
        """Convert game state to dictionary for display."""
        return {
            "Player": self.player_name,
            "Level": self.level,
            "Health": f"{self.health}/{self.max_health}",
            "Experience": f"{self.experience}/{self.level * 100}",
            "Gold": self.gold,
            "Location": self.current_location,
            "Inventory": self.inventory,
            "Skills": self.skills,
            "Quests Completed": self.quests_completed,
            "Achievements": self.achievements,
            "Play Time": f"{self.play_time} minutes",
            "Last Saved": self.save_timestamp
        }
    
    def __str__(self) -> str:
        state_dict = self.to_dict()
        lines = ["=== Game State ==="]
        for key, value in state_dict.items():
            if isinstance(value, list):
                lines.append(f"{key}: {', '.join(value) if value else 'None'}")
            elif isinstance(value, dict):
                skills_str = ', '.join([f"{k}: {v}" for k, v in value.items()])
                lines.append(f"{key}: {skills_str}")
            else:
                lines.append(f"{key}: {value}")
        return '\n'.join(lines)


def save_game_state(game_state: GameState, filepath: str = GAME_STATE_FILE) -> bool:
    """Serialize and save game state to file using pickle."""
    try:
        game_state.save_timestamp = time.strftime("%Y-%m-%d %H:%M:%S")
        with open(filepath, 'wb') as f:
            pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)
        print(f"✅ Game state saved successfully to '{filepath}'")
        return True
    except Exception as e:
        print(f"❌ Error saving game state: {e}")
        return False


def load_game_state(filepath: str = GAME_STATE_FILE) -> GameState | None:
    """Deserialize and load game state from file using pickle."""
    try:
        if not os.path.exists(filepath):
            print(f"⚠️  No save file found at '{filepath}'")
            return None
        
        with open(filepath, 'rb') as f:
            game_state = pickle.load(f)
        print(f"✅ Game state loaded successfully from '{filepath}'")
        return game_state
    except Exception as e:
        print(f"❌ Error loading game state: {e}")
        return None


def delete_save_file(filepath: str = GAME_STATE_FILE) -> bool:
    """Delete the save file."""
    try:
        if os.path.exists(filepath):
            os.remove(filepath)
            print(f"✅ Save file '{filepath}' deleted successfully")
            return True
        else:
            print(f"⚠️  No save file found at '{filepath}'")
            return False
    except Exception as e:
        print(f"❌ Error deleting save file: {e}")
        return False


def process_ai_response(game_state: GameState, response: str) -> GameState:
    """Process AI response and update game state accordingly."""
    response_lower = response.lower()
    
    # Check for level up events
    if "level up" in response_lower or "gained experience" in response_lower:
        exp_gain = 25
        result = game_state.gain_experience(exp_gain)
        if "Level up" in result:
            print(f"🎉 {result}")
    
    # Check for item acquisition
    items_to_add = []
    if "sword" in response_lower and "sword" not in game_state.inventory:
        items_to_add.append("Iron Sword")
    if "shield" in response_lower and "shield" not in game_state.inventory:
        items_to_add.append("Wooden Shield")
    if "potion" in response_lower:
        items_to_add.append("Health Potion")
    if "map" in response_lower and "map" not in game_state.inventory:
        items_to_add.append("Ancient Map")
    
    for item in items_to_add:
        game_state.add_item(item)
        print(f"📦 Added to inventory: {item}")
    
    # Check for gold changes
    if "gold" in response_lower:
        if "earned" in response_lower or "found" in response_lower or "reward" in response_lower:
            game_state.gold += 10
            print(f"💰 Earned 10 gold! Total: {game_state.gold}")
    
    # Check for location changes
    locations = ["forest", "dungeon", "castle", "village", "mountain", "cave", "tavern", "market"]
    for location in locations:
        if location in response_lower:
            new_location = location.title()
            if new_location != game_state.current_location:
                game_state.update_location(new_location)
                print(f"📍 Location updated to: {new_location}")
            break
    
    # Check for quest completion
    if "quest" in response_lower and ("complete" in response_lower or "finished" in response_lower or "accomplished" in response_lower):
        quest_name = f"Quest {len(game_state.quests_completed) + 1}"
        game_state.complete_quest(quest_name)
        print(f"✨ Quest completed: {quest_name}")
    
    # Check for achievements
    if len(game_state.inventory) >= 3 and "Collector" not in game_state.achievements:
        game_state.add_achievement("Collector")
        print("🏆 Achievement unlocked: Collector")
    
    if game_state.level >= 2 and "Rising Hero" not in game_state.achievements:
        game_state.add_achievement("Rising Hero")
        print("🏆 Achievement unlocked: Rising Hero")
    
    # Update play time
    game_state.play_time += 1
    
    return game_state


def chat_with_game_master(game_state: GameState, user_message: str) -> str:
    """Send message to Claude and get game master response."""
    # Add user message to conversation history
    game_state.conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Create system prompt with current game state
    system_prompt = f"""You are an AI Game Master for a fantasy RPG game. You help players navigate their adventure, 
    make decisions, and progress through the game world.
    
    Current Game State:
    - Player: {game_state.player_name}
    - Level: {game_state.level}
    - Health: {game_state.health}/{game_state.max_health}
    - Gold: {game_state.gold}
    - Location: {game_state.current_location}
    - Inventory: {', '.join(game_state.inventory) if game_state.inventory else 'Empty'}
    - Skills: Attack {game_state.skills['attack']}, Defense {game_state.skills['defense']}, Magic {game_state.skills['magic']}
    - Quests Completed: {len(game_state.quests_completed)}
    
    As the Game Master:
    1. Respond to player actions and decisions
    2. Describe the game world vividly
    3. Present challenges and opportunities
    4. Award experience, gold, and items when appropriate
    5. Guide the player through quests and adventures
    6. Keep responses engaging and immersive
    7. Mention specific items, locations, or rewards when they are gained
    
    Keep responses concise but engaging (2-3 paragraphs max)."""
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system=system_prompt,
        messages=game_state.conversation_history
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to conversation history
    game_state.conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message


def display_menu():
    """Display the main menu."""
    print("\n" + "="*50)
    print("🎮 RPG GAME WITH SAVE/LOAD SYSTEM")
    print("="*50)
    print("1. New Game")
    print("2. Load Game")
    print("3. View Game State")
    print("4. Save Game")
    print("5. Delete Save File")
    print("6. Play (Chat with Game Master)")
    print("7. Exit")
    print("="*50)


def main():
    """Main game loop with save/load functionality."""
    print("🎮 Welcome to the RPG Game with Save/Load System!")
    print("This game uses pickle to serialize and deserialize your game state.")
    
    game_state = None
    
    while True:
        display_menu()
        
        if game_state:
            print(f"Current Player: {game_state.player_name} (Level {game_state.level})")
        
        choice = input("\nEnter your choice (1-7): ").strip()
        
        if choice == "1":
            # New Game
            player_name = input("Enter your character name: ").strip()
            if not player_name:
                player_name = "Hero"
            
            game_state = GameState(player_name)
            print(f"\n🎉 New game created for {player_name}!")
            print("You begin your adventure in the Starting Village...")
            
            # Get initial game introduction from AI
            intro_response = chat_with_game_master(
                game_state, 
                f"I am {player_name}, beginning my adventure. Please introduce me to the game world and give me my first quest."
            )
            print(f"\n🧙 Game Master: {intro_response}")
            
            # Process the response to update game state
            game_state = process_ai_response(game_state, intro_response)
        
        elif choice == "2":
            # Load Game
            loaded_state = load_game_state()
            if loaded_state:
                game_state = loaded_state
                print(f"\n✅ Welcome back, {game_state.player_name}!")
                print(f"Last saved: {game_state.save_timestamp}")
            else:
                print("No save file found. Please start a new game.")
        
        elif choice == "3":
            # View Game State
            if game_state:
                print(f"\n{game_state}")
            else:
                print("No active game. Please start a new game or load an existing one.")
        
        elif choice == "4":
            # Save Game
            if game_state:
                save_game_state(game_state)
            else:
                print("No active game to save. Please start a new game first.")
        
        elif choice == "5":
            # Delete Save File
            confirm = input("Are you sure you want to delete the save file? (yes/no): ").strip().lower()
            if confirm == "yes":
                delete_save_file()
        
        elif choice == "6":
            # Play - Chat with Game Master
            if not game_state:
                print("No active game. Please start a new game or load an existing one.")
                continue
            
            print(f"\n🎮 Playing as {game_state.player_name}")
            print("Type 'back' to return to the main menu")
            print("Type 'status' to view your current game state")
            print("Type 'save' to save your progress")
            print("-" * 40)
            
            while True:
                user_input = input(f"\n{game_state.player_name}: ").strip()
                
                if not user_input:
                    continue
                
                if user_input.lower() == "back":
                    break
                
                if user_input.lower() == "status":
                    print(f"\n{game_state}")
                    continue
                
                if user_input.lower() == "save":
                    save_game_state(game_state)
                    continue
                
                # Get AI response
                print("\n🧙 Game Master is thinking...")
                response = chat_with_game_master(game_state, user_input)
                print(f"\n🧙 Game Master: {response}")
                
                # Process response and update game state
                game_state = process_ai_response(game_state, response)
                
                # Auto-save every 5 turns
                if game_state.play_time % 5 == 0:
                    print("\n💾 Auto-saving...")
                    save_game_state(game_state)
        
        elif choice == "7":
            # Exit
            if game_state:
                save_choice = input("Do you want to save before exiting? (yes/no): ").strip().lower()
                if save_choice == "yes":
                    save_game_state(game_state)
            
            print("\n👋 Thanks for playing! Goodbye!")
            break
        
        else:
            print("Invalid choice. Please enter a number between 1 and 7.")


if __name__ == "__main__":
    main()
```
