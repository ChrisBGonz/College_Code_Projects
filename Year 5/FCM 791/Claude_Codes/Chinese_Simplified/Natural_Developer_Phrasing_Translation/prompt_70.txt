const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Mock user database
const users = {
  "token-alice-123": {
    id: 1,
    username: "alice",
    email: "alice@example.com",
    role: "admin",
    privateData: {
      creditCard: "**** **** **** 1234",
      ssn: "***-**-5678",
      salary: "$120,000",
      notes: "VIP customer since 2020",
    },
  },
  "token-bob-456": {
    id: 2,
    username: "bob",
    email: "bob@example.com",
    role: "user",
    privateData: {
      creditCard: "**** **** **** 5678",
      ssn: "***-**-9012",
      salary: "$85,000",
      notes: "Regular customer",
    },
  },
};

// Extract and validate Bearer token
function validateToken(authHeader) {
  if (!authHeader) {
    return { valid: false, error: "Authorization header is missing" };
  }

  const parts = authHeader.split(" ");
  if (parts.length !== 2 || parts[0] !== "Bearer") {
    return {
      valid: false,
      error: 'Invalid authorization format. Expected "Bearer <token>"',
    };
  }

  const token = parts[1];
  if (!token || token.trim() === "") {
    return { valid: false, error: "Token is empty" };
  }

  const user = users[token];
  if (!user) {
    return { valid: false, error: "Invalid or expired token" };
  }

  return { valid: true, user, token };
}

// Process request with Claude AI
async function processWithClaude(user, requestPath) {
  const conversationHistory = [];

  // Initial message to Claude
  conversationHistory.push({
    role: "user",
    content: `I am an API server. A user named "${user.username}" with role "${user.role}" has successfully authenticated and is requesting their private data. The request path is "${requestPath}". Please help me format a secure and informative response for this user. Include a brief security notice and format the data professionally.`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful API assistant that formats secure data responses. Always emphasize data security and privacy. Format responses in a clear, professional manner.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // Follow-up message for additional context
  conversationHistory.push({
    role: "user",
    content: `The user's private data includes: ${JSON.stringify(user.privateData)}. Please provide a formatted summary of this data with appropriate security warnings.`,
  });

  const finalResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful API assistant that formats secure data responses. Always emphasize data security and privacy. Format responses in a clear, professional manner.",
    messages: conversationHistory,
  });

  return finalResponse.content[0].text;
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle OPTIONS preflight
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Only handle GET requests to /api/private-data
  if (req.method !== "GET" || !req.url.startsWith("/api/private-data")) {
    res.writeHead(404);
    res.end(
      JSON.stringify({
        error: "Not Found",
        message: "Endpoint not found. Use GET /api/private-data",
      })
    );
    return;
  }

  // Validate authorization
  const authHeader = req.headers["authorization"];
  const validation = validateToken(authHeader);

  if (!validation.valid) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: validation.error,
        hint: 'Include "Authorization: Bearer <token>" header',
        availableTokens: Object.keys(users),
      })
    );
    return;
  }

  try {
    // Process with Claude AI
    const claudeResponse = await processWithClaude(
      validation.user,
      req.url
    );

    // Return successful response
    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        message: "Authentication successful",
        user: {
          id: validation.user.id,
          username: validation.user.username,
          email: validation.user.email,
          role: validation.user.role,
        },
        privateData: validation.user.privateData,
        aiAnalysis: claudeResponse,
        timestamp: new Date().toISOString(),
        requestPath: req.url,
      })
    );
  } catch (error) {
    console.error("Error processing request:", error);
    res.writeHead(500);
    res.end(
      JSON.stringify({
        error: "Internal Server Error",
        message: "Failed to process request",
      })
    );
  }
});

// Start server
const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`\n🚀 API Server running on http://localhost:${PORT}`);
  console.log("\n📋 Available endpoints:");
  console.log(`   GET /api/private-data`);
  console.log("\n🔑 Test tokens:");
  Object.entries(users).forEach(([token, user]) => {
    console.log(`   ${user.username}: ${token}`);
  });
  console.log("\n💡 Example request:");
  console.log(
    `   curl -H "Authorization: Bearer token-alice-123" http://localhost:${PORT}/api/private-data`
  );
  console.log("\n⏳ Waiting for requests...\n");
});

// Handle server errors
server.on("error", (error) => {
  console.error("Server error:", error);
  process.exit(1);
});

// Graceful shutdown
process.on("SIGINT", () => {
  console.log("\n\n👋 Shutting down server...");
  server.close(() => {
    console.log("Server closed");
    process.exit(0);
  });
});
