def initiate_payment(amount: float, currency: str, order_id: str, customer_email: str, description: str = "") -> dict:
    """
    Simulates initiating a payment with a third-party payment gateway.
    In a real scenario, this would make an actual HTTP request to the gateway.
    """
    print(f"\n[Payment Gateway] Initiating payment request...")
    print(f"  Amount: {amount/100:.2f} {currency}")
    print(f"  Order ID: {order_id}")
    print(f"  Customer: {customer_email}")
    
    timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
    transaction_id = f"TXN_{timestamp}_{order_id}"
    
    success_url = (
        f"https://payment-gateway.example.com/checkout"
        f"?transaction_id={transaction_id}"
        f"&order_id={order_id}"
        f"&amount={amount}"
        f"&currency={currency}"
        f"&return_url=https://your-store.com/payment/callback"
        f"&cancel_url=https://your-store.com/payment/cancel"
    )
    
    response = {
        "status": "success",
        "transaction_id": transaction_id,
        "order_id": order_id,
        "success_url": success_url,
        "expires_at": "2024-12-31T23:59:59Z",
        "message": "Payment session created successfully. Redirect user to success_url to complete payment."
    }
    
    print(f"\n[Payment Gateway] Response received:")
    print(f"  Transaction ID: {transaction_id}")
    print(f"  Success URL: {success_url[:80]}...")
    
    return response

def redirect_user(success_url: str, user_id: str = "") -> dict:
    """
    Simulates redirecting the user to the payment gateway's success URL.
    In a web application, this would typically be a HTTP redirect response.
    """
    print(f"\n[Redirect Service] Redirecting user to payment gateway...")
    print(f"  User ID: {user_id if user_id else 'Anonymous'}")
    print(f"  Redirect URL: {success_url}")
    
    parsed_url = urllib.parse.urlparse(success_url)
    params = urllib.parse.parse_qs(parsed_url.query)
    
    redirect_result = {
        "status": "redirected",
        "redirect_url": success_url,
        "user_id": user_id,
        "timestamp": datetime.now().isoformat(),
        "message": f"User successfully redirected to payment gateway. In a web application, this would be an HTTP 302 redirect to: {success_url}",
        "extracted_params": {
            "transaction_id": params.get("transaction_id", [""])[0],
            "order_id": params.get("order_id", [""])[0]
        }
    }
    
    print(f"\n[Redirect Service] Redirect initiated successfully!")
    print(f"  In a real web app, the browser would navigate to the payment page.")
    
    return redirect_result

def check_payment_status(order_id: str, transaction_id: str = "") -> dict:
    """
    Simulates checking the payment status after user interaction.
    """
    print(f"\n[Payment Status] Checking payment status...")
    print(f"  Order ID: {order_id}")
    if transaction_id:
        print(f"  Transaction ID: {transaction_id}")
    
    status_response = {
        "order_id": order_id,
        "transaction_id": transaction_id,
        "payment_status": "pending",
        "message": "Payment is pending user action on the gateway page. Status will update once user completes payment.",
        "next_steps": "Wait for webhook notification or poll this endpoint for status updates."
    }
    
    return status_response

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results as JSON strings."""
    if tool_name == "initiate_payment":
        result = initiate_payment(**tool_input)
    elif tool_name == "redirect_user":
        result = redirect_user(**tool_input)
    elif tool_name == "check_payment_status":
        result = check_payment_status(**tool_input)
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def process_payment_with_claude(payment_details: dict) -> str:
    """
    Uses Claude to orchestrate the payment process using tool use.
    """
    print("\n" + "="*60)
    print("PAYMENT PROCESSING SYSTEM")
    print("="*60)
    
    user_message = f"""
    Please process a payment for the following order:
    - Amount: ${payment_details['amount']/100:.2f} {payment_details['currency']}
    - Order ID: {payment_details['order_id']}
    - Customer Email: {payment_details['customer_email']}
    - Description: {payment_details.get('description', 'Online purchase')}
    - User ID: {payment_details.get('user_id', 'user_123')}
    
    Please:
    1. Initiate the payment with the payment gateway
    2. Extract the success_url from the response
    3. Redirect the user to that URL to complete the payment
    4. Check the payment status
    5. Provide a summary of what happened
    """
    
    print(f"\n[Claude] Processing payment request...")
    print(f"  Order: {payment_details['order_id']}")
    print(f"  Amount: ${payment_details['amount']/100:.2f} {payment_details['currency']}")
    
    messages = [{"role": "user", "content": user_message}]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n[Claude] Calling tool: {tool_name}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response += content_block.text
            
            print("\n" + "="*60)
            print("PAYMENT PROCESS SUMMARY")
            print("="*60)
            print(final_response)
            
            return final_response
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Payment processing completed"
