
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");

const client = new Anthropic();

const app = express();
app.use(express.json());

// Store conversation history for multi-turn conversations
const conversationHistory = [];

async function generateErrorLoggingFunction(userMessage) {
  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert JavaScript developer specializing in Express.js server development. Your task is to help develop error logging functions for production environments. Provide clean, well-commented JavaScript code that uses console.log for system error logging in production environments.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Express route to handle error logging function generation
app.post("/generate-error-logger", async (req, res) => {
  try {
    const { message } = req.body;

    if (!message) {
      return res.status(400).json({ error: "Message is required" });
    }

    const generatedCode = await generateErrorLoggingFunction(message);

    res.json({
      success: true,
      generatedCode,
      conversationLength: conversationHistory.length,
    });
  } catch (error) {
    console.error("Error generating code:", error);
    res.status(500).json({ error: "Failed to generate error logging function" });
  }
});

// Route to get conversation history
app.get("/conversation-history", (req, res) => {
  res.json({
    history: conversationHistory,
    length: conversationHistory.length,
  });
});

// Route to clear conversation history
app.delete("/conversation-history", (req, res) => {
  conversationHistory.length = 0;
  res.json({ message: "Conversation history cleared" });
});

// Start the server
const PORT = process.env.PORT || 3000;
app.listen(PORT, async () => {
  console.log(`Server running on port ${PORT}`);

  // Demonstrate the multi-turn conversation capability
  console.log("\n=== Demonstrating Multi-turn Conversation ===\n");

  try {
    // First turn: Request basic error logging function
    console.log("Turn 1: Requesting basic error logging function...");
    const response1 = await generateErrorLoggingFunction(
      "为 Express 服务器开发一个 JavaScript 函数：在生产环境中使用 console.log 记录系统错误。请提供基本的错误日志记录函数。"
    );
    console.log("Response 1:", response1.substring(0, 500) + "...\n");

    // Second turn: Ask for enhancements
    console.log("Turn 2: Requesting enhancements...");
    const response2 = await generateErrorLoggingFunction(
      "很好！现在请添加错误级别分类（error, warn, info）和时间戳功能。"
    );
    console.log("Response 2:", response2.substring(0, 500) + "...\n");

    // Third turn: Ask for Express middleware integration
    console.log("Turn 3: Requesting Express middleware integration...");
    const response3 = await generateErrorLoggingFunction(
      "请将这个错误日志函数集成为 Express 中间件，并添加请求 ID 追踪功能。"
    );
    console.log("Response 3:", response3.substring(0, 500) + "...\n");

    console.log(
      `\nConversation completed with ${conversationHistory.length} messages`
    );
    console.log(
      "\nServer is ready to accept requests at http://localhost:" + PORT
    );
  } catch (error) {
    console.error("Error during demonstration:", error);
  }
});

module.exports = { generateErrorLoggingFunction, app };
```
