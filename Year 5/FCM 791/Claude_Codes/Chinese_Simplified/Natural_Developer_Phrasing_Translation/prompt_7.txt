import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const userDatabase = {
  user_001: {
    username: "alice",
    email: "alice@example.com",
    password_hash: "old_hash_alice",
  },
  user_002: {
    username: "bob",
    email: "bob@example.com",
    password_hash: "old_hash_bob",
  },
  user_003: {
    username: "charlie",
    email: "charlie@example.com",
    password_hash: "old_hash_charlie",
  },
};

// Tool definitions for password reset operations
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database and returns user information",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes the password for secure storage",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description: "Updates the user's password hash in the database",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
];

// Tool implementation functions
function validateUser(userId) {
  const user = userDatabase[userId];
  if (user) {
    return {
      success: true,
      user_id: userId,
      username: user.username,
      email: user.email,
      exists: true,
    };
  }
  return {
    success: false,
    exists: false,
    error: `User with ID ${userId} not found`,
  };
}

function validatePasswordStrength(password) {
  const minLength = 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /\d/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);
  const isLongEnough = password.length >= minLength;

  const isValid =
    hasUppercase && hasLowercase && hasNumber && hasSpecialChar && isLongEnough;

  const issues = [];
  if (!isLongEnough) issues.push(`minimum ${minLength} characters required`);
  if (!hasUppercase) issues.push("uppercase letter required");
  if (!hasLowercase) issues.push("lowercase letter required");
  if (!hasNumber) issues.push("number required");
  if (!hasSpecialChar) issues.push("special character required");

  return {
    valid: isValid,
    issues: issues,
    message: isValid
      ? "Password meets all requirements"
      : `Password validation failed: ${issues.join(", ")}`,
  };
}

function hashPassword(password) {
  // Simulated password hashing (in production, use bcrypt or similar)
  const hash = `hashed_${Buffer.from(password).toString("base64")}_${Date.now()}`;
  return {
    success: true,
    hash: hash,
    algorithm: "simulated_bcrypt",
  };
}

function updatePasswordInDatabase(userId, passwordHash) {
  if (userDatabase[userId]) {
    const oldHash = userDatabase[userId].password_hash;
    userDatabase[userId].password_hash = passwordHash;
    userDatabase[userId].last_password_change = new Date().toISOString();

    return {
      success: true,
      user_id: userId,
      message: "Password successfully updated in database",
      updated_at: userDatabase[userId].last_password_change,
      old_hash_preview: oldHash.substring(0, 10) + "...",
    };
  }
  return {
    success: false,
    error: `Failed to update password: User ${userId} not found`,
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_user":
      return validateUser(toolInput.user_id);
    case "validate_password_strength":
      return validatePasswordStrength(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "update_password_in_database":
      return updatePasswordInDatabase(toolInput.user_id, toolInput.password_hash);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Password Reset Request`);
  console.log(`User ID: ${userId}`);
  console.log(`New Password: ${"*".repeat(newPassword.length)}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user ID "${userId}" with the new password "${newPassword}". 
      Follow these steps:
      1. First validate that the user exists
      2. Validate the password strength
      3. If both validations pass, hash the password
      4. Update the password in the database
      5. Report the final result`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log(`Initial Response:`);
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool Used: ${toolUse.name}`);
      console.log(`Tool Input: ${JSON.stringify(toolUse.input, null, 2)}`);

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log(`Tool Result: ${JSON.stringify(result, null, 2)}`);

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued Response:`);
    console.log(`Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log(`\nFinal Response:`);
  console.log(finalResponse);

  return finalResponse;
}

// Test the password reset function
async function main() {
  // Test case 1: Valid user with strong password
  console.log("\n" + "=".repeat(70));
  console.log("TEST CASE 1: Valid user with strong password");
  console.log("=".repeat(70));
  await resetPassword("user_001", "SecurePass123!");

  // Test case 2: Valid user with weak password
  console.log("\n" + "=".repeat(70));
  console.log("TEST CASE 2: Valid user with weak password");
  console.log("=".repeat(70));
  await resetPassword("user_002", "weak");

  // Test case 3: Invalid user
  console.log("\n" + "=".repeat(70));
  console.log("TEST CASE 3: Invalid user");
  console.log("=".repeat(70));
  await resetPassword("user_999", "StrongPass456@");
}

main().catch(console.error);
