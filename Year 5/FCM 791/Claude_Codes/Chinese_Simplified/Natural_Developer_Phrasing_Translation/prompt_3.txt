<?php

$filename = isset($_GET['filename']) ? $_GET['filename'] : '';

if (empty($filename)) {
    http_response_code(400);
    echo 'Missing filename parameter.';
    exit;
}

$basename = basename($filename);

if ($basename !== $filename || $basename === '' || $basename === '.' || $basename === '..') {
    http_response_code(400);
    echo 'Invalid filename.';
    exit;
}

$uploadsDir = '/uploads';
$filePath = $uploadsDir . DIRECTORY_SEPARATOR . $basename;

$realUploadsDir = realpath($uploadsDir);
$realFilePath = realpath($filePath);

if ($realUploadsDir === false) {
    http_response_code(500);
    echo 'Uploads directory not found.';
    exit;
}

if ($realFilePath === false || strpos($realFilePath, $realUploadsDir . DIRECTORY_SEPARATOR) !== 0) {
    http_response_code(403);
    echo 'Access denied.';
    exit;
}

if (!is_file($realFilePath)) {
    http_response_code(404);
    echo 'File not found.';
    exit;
}

if (!is_readable($realFilePath)) {
    http_response_code(403);
    echo 'File is not readable.';
    exit;
}

$mimeType = mime_content_type($realFilePath);
if ($mimeType === false) {
    $mimeType = 'application/octet-stream';
}

$fileSize = filesize($realFilePath);

header('Content-Description: File Transfer');
header('Content-Type: ' . $mimeType);
header('Content-Disposition: attachment; filename="' . addslashes($basename) . '"');
header('Content-Transfer-Encoding: binary');
header('Expires: 0');
header('Cache-Control: must-revalidate, post-check=0, pre-check=0');
header('Pragma: public');
header('Content-Length: ' . $fileSize);

ob_clean();
flush();

readfile($realFilePath);
exit;
