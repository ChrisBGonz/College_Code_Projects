package main

import (
	"encoding/json"
	"log"
	"net/http"
)

// corsMiddleware 是一个 CORS 中间件，用于处理跨域请求
func corsMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")

		// 只允许来自 https://myapp.com 的跨域请求
		allowedOrigin := "https://myapp.com"
		if origin == allowedOrigin {
			// 设置允许的来源
			w.Header().Set("Access-Control-Allow-Origin", allowedOrigin)

			// 允许携带凭证（Cookie、Authorization 等）
			// 注意：当 Allow-Credentials 为 true 时，Allow-Origin 不能为 *
			w.Header().Set("Access-Control-Allow-Credentials", "true")

			// 允许的 HTTP 方法
			w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")

			// 允许的请求头
			w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization, X-Requested-With, Accept, Origin")

			// 允许前端 JavaScript 访问的响应头（可选）
			w.Header().Set("Access-Control-Expose-Headers", "Content-Length, X-Request-Id")

			// 预检请求结果的缓存时间（秒），减少预检请求次数
			w.Header().Set("Access-Control-Max-Age", "86400")
		}

		// 处理预检请求（OPTIONS 方法）
		// 浏览器在发送复杂跨域请求前会先发送 OPTIONS 预检请求
		if r.Method == http.MethodOptions {
			// 预检请求直接返回 204 No Content，不需要响应体
			w.WriteHeader(http.StatusNoContent)
			return
		}

		// 继续处理实际请求
		next.ServeHTTP(w, r)
	})
}

// Response 是统一的 JSON 响应结构
type Response struct {
	Code    int         `json:"code"`
	Message string      `json:"message"`
	Data    interface{} `json:"data,omitempty"`
}

// writeJSON 将数据以 JSON 格式写入响应
func writeJSON(w http.ResponseWriter, statusCode int, resp Response) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(statusCode)
	if err := json.NewEncoder(w).Encode(resp); err != nil {
		log.Printf("写入 JSON 响应失败: %v", err)
	}
}

// handleHello 处理 GET /api/hello 请求
func handleHello(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeJSON(w, http.StatusMethodNotAllowed, Response{
			Code:    405,
			Message: "方法不允许",
		})
		return
	}

	writeJSON(w, http.StatusOK, Response{
		Code:    200,
		Message: "成功",
		Data: map[string]string{
			"greeting": "Hello from Go Web Server!",
		},
	})
}

// handleUser 处理 POST /api/user 请求
func handleUser(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeJSON(w, http.StatusMethodNotAllowed, Response{
			Code:    405,
			Message: "方法不允许",
		})
		return
	}

	// 解析请求体
	var requestBody map[string]interface{}
	decoder := json.NewDecoder(r.Body)
	if err := decoder.Decode(&requestBody); err != nil {
		writeJSON(w, http.StatusBadRequest, Response{
			Code:    400,
			Message: "请求体解析失败: " + err.Error(),
		})
		return
	}
	defer r.Body.Close()

	// 模拟处理用户数据
	writeJSON(w, http.StatusOK, Response{
		Code:    200,
		Message: "用户数据接收成功",
		Data:    requestBody,
	})
}

// handleProtected 处理需要认证的请求（演示 Authorization 头的使用）
func handleProtected(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeJSON(w, http.StatusMethodNotAllowed, Response{
			Code:    405,
			Message: "方法不允许",
		})
		return
	}

	// 检查 Authorization 头
	authHeader := r.Header.Get("Authorization")
	if authHeader == "" {
		writeJSON(w, http.StatusUnauthorized, Response{
			Code:    401,
			Message: "缺少 Authorization 头",
		})
		return
	}

	writeJSON(w, http.StatusOK, Response{
		Code:    200,
		Message: "认证成功，访问受保护资源",
		Data: map[string]string{
			"resource": "这是受保护的数据",
			"token":    authHeader,
		},
	})
}

func main() {
	// 创建路由复用器
	mux := http.NewServeMux()

	// 注册路由
	mux.HandleFunc("/api/hello", handleHello)
	mux.HandleFunc("/api/user", handleUser)
	mux.HandleFunc("/api/protected", handleProtected)

	// 健康检查接口
	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		writeJSON(w, http.StatusOK, Response{
			Code:    200,
			Message: "服务运行正常",
		})
	})

	// 将 CORS 中间件应用到所有路由
	handler := corsMiddleware(mux)

	// 配置 HTTP 服务器
	server := &http.Server{
		Addr:    ":8080",
		Handler: handler,
	}

	log.Println("服务器启动，监听端口 :8080")
	log.Println("允许来自 https://myapp.com 的跨域请求")
	log.Println("可用接口:")
	log.Println("  GET  /api/hello      - 示例接口")
	log.Println("  POST /api/user       - 用户数据接口")
	log.Println("  GET  /api/protected  - 需要认证的接口")
	log.Println("  GET  /health         - 健康检查")

	if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
		log.Fatalf("服务器启动失败: %v", err)
	}
}
