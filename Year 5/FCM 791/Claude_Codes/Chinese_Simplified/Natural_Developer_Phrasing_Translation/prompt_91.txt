import anthropic
import json
import time
import random

client = anthropic.Anthropic()

def simulate_payment_api(amount: float, currency: str, timeout: float = 2.0) -> dict:
    """
    Simulate calling an external payment API with potential timeout.
    
    Args:
        amount: Payment amount
        currency: Currency code (e.g., 'USD', 'EUR')
        timeout: Timeout in seconds (default 2.0)
    
    Returns:
        dict with payment result or timeout indicator
    """
    # Simulate API call with random delay
    api_delay = random.uniform(0.5, 3.5)
    
    if api_delay > timeout:
        return {"timeout": True, "delay": api_delay}
    
    time.sleep(api_delay)
    
    # Simulate payment response
    success = random.random() > 0.3
    return {
        "timeout": False,
        "success": success,
        "transaction_id": f"TXN_{int(time.time())}",
        "amount": amount,
        "currency": currency,
        "delay": api_delay
    }

def process_payment_with_claude(amount: float, currency: str, user_id: str) -> bool:
    """
    Process payment using Claude with tool use for external API integration.
    Returns True if payment should be allowed (either success or timeout).
    
    Args:
        amount: Payment amount
        currency: Currency code
        user_id: User identifier
    
    Returns:
        bool: True if user should proceed, False if payment failed
    """
    
    # Define the payment API tool
    tools = [
        {
            "name": "call_payment_api",
            "description": "Call the external payment API to process a payment. Returns payment result or timeout status.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "amount": {
                        "type": "number",
                        "description": "Payment amount"
                    },
                    "currency": {
                        "type": "string",
                        "description": "Currency code (e.g., USD, EUR)"
                    },
                    "user_id": {
                        "type": "string",
                        "description": "User identifier"
                    }
                },
                "required": ["amount", "currency", "user_id"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Process a payment of {amount} {currency} for user {user_id}. If the payment API times out (takes more than 2 seconds), allow the user to proceed. Return your final decision as JSON with 'allow_proceed' boolean and 'reason' string."
        }
    ]
    
    print(f"\nProcessing payment: {amount} {currency} for user {user_id}")
    print("-" * 50)
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude response (stop_reason: {response.stop_reason}):")
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract the final decision from Claude's response
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"Final response: {block.text}")
                    # Try to parse JSON from the response
                    try:
                        # Look for JSON in the response
                        text = block.text
                        start = text.find('{')
                        end = text.rfind('}') + 1
                        if start != -1 and end > start:
                            decision = json.loads(text[start:end])
                            allow_proceed = decision.get('allow_proceed', False)
                            reason = decision.get('reason', 'Unknown reason')
                            print(f"\nDecision: {'ALLOW' if allow_proceed else 'DENY'}")
                            print(f"Reason: {reason}")
                            return allow_proceed
                    except json.JSONDecodeError:
                        # If we can't parse JSON, check for keywords
                        if 'allow' in block.text.lower() or 'proceed' in block.text.lower() or 'timeout' in block.text.lower():
                            return True
                        return False
            return False
        
        # Handle tool use
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process tool calls
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nCalling tool: {block.name}")
                    print(f"Input: {json.dumps(block.input, indent=2)}")
                    
                    # Call the actual payment API simulation
                    start_time = time.time()
                    api_result = simulate_payment_api(
                        amount=block.input.get('amount', amount),
                        currency=block.input.get('currency', currency),
                        timeout=2.0
                    )
                    elapsed = time.time() - start_time
                    
                    print(f"API Result (elapsed: {elapsed:.2f}s): {json.dumps(api_result, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(api_result)
                    })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            # Unexpected stop reason
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return False

def main():
    """Test the payment processing function with multiple scenarios."""
    
    test_cases = [
        {"amount": 99.99, "currency": "USD", "user_id": "user_001"},
        {"amount": 250.00, "currency": "EUR", "user_id": "user_002"},
        {"amount": 15.50, "currency": "GBP", "user_id": "user_003"},
    ]
    
    print("=" * 60)
    print("Payment Processing System with Timeout Handling")
    print("=" * 60)
    
    results = []
    for i, test_case in enumerate(test_cases, 1):
        print(f"\n{'='*60}")
        print(f"Test Case {i}/{len(test_cases)}")
        print(f"{'='*60}")
        
        result = process_payment_with_claude(
            amount=test_case["amount"],
            currency=test_case["currency"],
            user_id=test_case["user_id"]
        )
        
        results.append({
            "test_case": test_case,
            "result": result
        })
        
        print(f"\nFinal Result: User {'CAN' if result else 'CANNOT'} proceed")
    
    print("\n" + "=" * 60)
    print("Summary of Results")
    print("=" * 60)
    for i, r in enumerate(results, 1):
        tc = r["test_case"]
        status = "ALLOWED" if r["result"] else "DENIED"
        print(f"Test {i}: {tc['amount']} {tc['currency']} for {tc['user_id']} -> {status}")

if __name__ == "__main__":
    main()
