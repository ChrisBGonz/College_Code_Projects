import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.JsonNode;

public class UserJsonParser {
    
    static class User {
        public String name;
        public String email;
        public int age;
        
        public User() {}
        
        public User(String name, String email, int age) {
            this.name = name;
            this.email = email;
            this.age = age;
        }
        
        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }
    
    private static final ObjectMapper objectMapper = new ObjectMapper();
    
    public static User parseUserFromJson(String jsonString) {
        try {
            // Use Claude to help parse and validate the JSON
            BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .build();
            
            // First, let's use Claude to validate and extract user information
            String prompt = "Parse the following JSON string and extract the user information. " +
                "Return ONLY a valid JSON object with fields: name (string), email (string), age (number). " +
                "If any field is missing, use empty string for strings and 0 for numbers. " +
                "Input JSON: " + jsonString + "\n\nReturn only the JSON object, no explanation.";
            
            String requestBody = String.format(
                "{\"anthropic_version\": \"bedrock-2023-05-31\", \"max_tokens\": 1024, \"messages\": [{\"role\": \"user\", \"content\": \"%s\"}]}",
                prompt.replace("\"", "\\\"").replace("\n", "\\n")
            );
            
            InvokeModelRequest request = InvokeModelRequest.builder()
                .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
                .body(SdkBytes.fromUtf8String(requestBody))
                .build();
            
            InvokeModelResponse response = client.invokeModel(request);
            String responseBody = response.body().asUtf8String();
            
            // Parse the response from Claude
            JsonNode responseJson = objectMapper.readTree(responseBody);
            String extractedJson = responseJson.get("content").get(0).get("text").asText();
            
            // Clean up the extracted JSON (remove any markdown formatting if present)
            extractedJson = extractedJson.trim();
            if (extractedJson.startsWith("```json")) {
                extractedJson = extractedJson.substring(7);
            }
            if (extractedJson.startsWith("```")) {
                extractedJson = extractedJson.substring(3);
            }
            if (extractedJson.endsWith("```")) {
                extractedJson = extractedJson.substring(0, extractedJson.length() - 3);
            }
            extractedJson = extractedJson.trim();
            
            // Parse the cleaned JSON into User object
            User user = objectMapper.readValue(extractedJson, User.class);
            
            System.out.println("Successfully parsed user: " + user);
            System.out.println("User name: " + user.name);
            
            return user;
            
        } catch (Exception e) {
            System.err.println("Error parsing JSON: " + e.getMessage());
            e.printStackTrace();
            return null;
        }
    }
    
    public static void main(String[] args) {
        // Test with various JSON strings
        String[] testJsonStrings = {
            "{\"name\": \"John Doe\", \"email\": \"john@example.com\", \"age\": 30}",
            "{\"name\": \"Jane Smith\", \"email\": \"jane@example.com\", \"age\": 25}",
            "{\"name\": \"Bob Johnson\", \"age\": 45}",
            "{\"user_name\": \"Alice Brown\", \"user_email\": \"alice@example.com\", \"user_age\": 35}"
        };
        
        System.out.println("=== User JSON Parser Demo ===\n");
        
        for (String jsonString : testJsonStrings) {
            System.out.println("Input JSON: " + jsonString);
            User user = parseUserFromJson(jsonString);
            if (user != null) {
                System.out.println("Extracted user.name: " + user.name);
            }
            System.out.println("---");
        }
    }
}
