ERROR_TEMPLATE = """

<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>错误 - URL 内容查看器</title>
    <style>
        body {
            font-family: Arial, sans-serif;

def render_error(
    error_title,
    error_message,
    status_code=500,
    error_details=None,
    error_icon="❌",
    error_color="#dc3545",
):
    """渲染错误页面"""
    html = render_template_string(
        ERROR_TEMPLATE,
        error_title=error_title,
        error_message=error_message,
        error_details=error_details,
        error_icon=error_icon,
        error_color=error_color,
    )
    return html, status_code


@app.route("/")
def index():
    """首页 - 显示 URL 输入表单"""
    return render_template_string(HTML_TEMPLATE, url=None, content=None)


@app.route("/fetch")
def fetch_url():
    """获取并展示 URL 内容"""
    url = request.args.get("url", "").strip()

    if not url:
        return render_error(
            error_title="缺少 URL 参数",
            error_message="请提供一个有效的 URL 地址。",
            status_code=400,
            error_icon="⚠️",
            error_color="#ffc107",
        )

    if not url.startswith(("http://", "https://")):
        url = "https://" + url

    try:
        headers = {
            "User-Agent": "Mozilla/5.0 (compatible; URLFetcher/1.0)",
            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
            "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
        }

        response = requests.get(url, headers=headers, timeout=10, allow_redirects=True)

        status_code = response.status_code
        content_type = response.headers.get("Content-Type", "unknown")

        if status_code == 404:
            return render_error(
                error_title="404 - 页面未找到",
                error_message=f"请求的页面不存在。服务器返回了 404 状态码，表示该资源在目标服务器上无法找到。",
                status_code=404,
                error_details=f"请求 URL: {url}",
                error_icon="🔍",
                error_color="#dc3545",
            )

        elif status_code == 403:
            return render_error(
                error_title="403 - 访问被拒绝",
                error_message="服务器拒绝了访问请求。您可能没有权限访问此资源。",
                status_code=403,
                error_details=f"请求 URL: {url}",
                error_icon="🚫",
                error_color="#dc3545",
            )

        elif status_code == 401:
            return render_error(
                error_title="401 - 需要身份验证",
                error_message="访问此资源需要身份验证。请确保您有正确的访问凭据。",
                status_code=401,
                error_details=f"请求 URL: {url}",
                error_icon="🔐",
                error_color="#ffc107",
            )

        elif status_code == 500:
            return render_error(
                error_title="500 - 服务器内部错误",
                error_message="目标服务器遇到了内部错误，无法完成请求。",
                status_code=502,
                error_details=f"请求 URL: {url}",
                error_icon="💥",
                error_color="#dc3545",
            )

        elif status_code >= 400:
            return render_error(
                error_title=f"{status_code} - 请求错误",
                error_message=f"服务器返回了错误状态码 {status_code}。",
                status_code=status_code,
                error_details=f"请求 URL: {url}",
                error_icon="⚠️",
                error_color="#ffc107",
            )

        elif status_code >= 300:
            return render_error(
                error_title=f"{status_code} - 重定向",
                error_message=f"请求被重定向（状态码 {status_code}），但重定向未能成功完成。",
                status_code=status_code,
                error_details=f"请求 URL: {url}",
                error_icon="↪️",
                error_color="#17a2b8",
            )

        if "text" in content_type or "json" in content_type or "xml" in content_type:
            try:
                content = response.text
            except Exception:
                content = response.content.decode("utf-8", errors="replace")
        else:
            content = f"[二进制内容 - 内容类型: {content_type}，大小: {len(response.content)} 字节]"

        return render_template_string(
            HTML_TEMPLATE,
            url=url,
            content=content,
            status_code=status_code,
            content_type=content_type,
        )

    except MissingSchema:
        return render_error(
            error_title="无效的 URL 格式",
            error_message="提供的 URL 格式不正确，缺少协议前缀（如 http:// 或 https://）。",
            status_code=400,
            error_details=f"输入的 URL: {url}",
            error_icon="⚠️",
            error_color="#ffc107",
        )

    except InvalidURL:
        return render_error(
            error_title="无效的 URL",
            error_message="提供的 URL 格式无效，请检查 URL 是否正确。",
            status_code=400,
            error_details=f"输入的 URL: {url}",
            error_icon="⚠️",
            error_color="#ffc107",
        )

    except ConnectionError as e:
        return render_error(
            error_title="连接错误",
            error_message="无法连接到目标服务器。请检查 URL 是否正确，或目标服务器是否可访问。",
            status_code=503,
            error_details=f"错误详情: {str(e)[:200]}",
            error_icon="🔌",
            error_color="#dc3545",
        )

    except Timeout:
        return render_error(
            error_title="请求超时",
            error_message="请求超时（超过 10 秒）。目标服务器响应太慢或无法访问。",
            status_code=504,
            error_details=f"请求 URL: {url}",
            error_icon="⏱️",
            error_color="#ffc107",
        )

    except TooManyRedirects:
        return render_error(
            error_title="重定向次数过多",
            error_message="请求遇到了太多重定向，可能存在重定向循环。",
            status_code=400,
            error_details=f"请求 URL: {url}",
            error_icon="🔄",
            error_color="#ffc107",
        )

    except Exception as e:
        return render_error(
            error_title="未知错误",
            error_message="处理请求时发生了未知错误，请稍后重试。",
            status_code=500,
            error_details=f"错误类型: {type(e).__name__}, 错误信息: {str(e)[:200]}",
            error_icon="❌",
            error_color="#dc3545",
        )


@app.errorhandler(404)
def page_not_found(e):
    """处理 Flask 应用本身的 404 错误"""
    return render_error(
        error_title="404 - 页面未找到",
        error_message="您访问的页面不存在。请检查 URL 是否正确。",
        status_code=404,
        error_icon="🔍",
        error_color="#dc3545",
    )


@app.errorhandler(500)
def internal_server_error(e):
    """处理 Flask 应用本身的 500 错误"""
    return render_error(
        error_title="500 - 服务器内部错误",
        error_message="服务器遇到了内部错误，请稍后重试。",
        status_code=500,
        error_icon="💥",
        error_color="#dc3545",
    )


if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=5000)
