
```csharp
using System;
using System.ClientModel;
using System.Text.Json;
using Azure.AI.OpenAI;
using OpenAI.Chat;

// MFA Configuration Change Logger using Azure OpenAI
class MFAConfigurationLogger
{
    private static readonly string endpoint = Environment.GetEnvironmentVariable("AZURE_OPENAI_ENDPOINT") ?? throw new InvalidOperationException("AZURE_OPENAI_ENDPOINT is not set");
    private static readonly string apiKey = Environment.GetEnvironmentVariable("AZURE_OPENAI_API_KEY") ?? throw new InvalidOperationException("AZURE_OPENAI_API_KEY is not set");
    private static readonly string deploymentName = Environment.GetEnvironmentVariable("AZURE_OPENAI_DEPLOYMENT") ?? "gpt-4o";

    // MFA Configuration Change Event
    public class MFAChangeEvent
    {
        public string UserId { get; set; } = string.Empty;
        public string UserEmail { get; set; } = string.Empty;
        public string PreviousState { get; set; } = string.Empty;
        public string NewState { get; set; } = string.Empty;
        public DateTime Timestamp { get; set; }
        public string SourceIP { get; set; } = string.Empty;
        public string DeviceInfo { get; set; } = string.Empty;
        public string ChangeType { get; set; } = string.Empty;
        public string SessionId { get; set; } = string.Empty;
        public string AdminUserId { get; set; } = string.Empty;
    }

    // MFA Log Entry
    public class MFALogEntry
    {
        public string LogId { get; set; } = string.Empty;
        public MFAChangeEvent Event { get; set; } = new MFAChangeEvent();
        public string RiskLevel { get; set; } = string.Empty;
        public string Analysis { get; set; } = string.Empty;
        public string[] RecommendedActions { get; set; } = Array.Empty<string>();
        public DateTime LoggedAt { get; set; }
    }

    static async Task Main(string[] args)
    {
        Console.WriteLine("MFA Configuration Change Logger with AI Analysis");
        Console.WriteLine("================================================\n");

        // Sample MFA change events to log
        var mfaChangeEvents = new[]
        {
            new MFAChangeEvent
            {
                UserId = "USR-001",
                UserEmail = "john.doe@company.com",
                PreviousState = "SMS-based MFA enabled",
                NewState = "Authenticator App MFA enabled",
                Timestamp = DateTime.UtcNow.AddMinutes(-5),
                SourceIP = "192.168.1.100",
                DeviceInfo = "Windows 11, Chrome 120.0, Desktop",
                ChangeType = "MFA_METHOD_CHANGE",
                SessionId = "SES-ABC123",
                AdminUserId = ""
            },
            new MFAChangeEvent
            {
                UserId = "USR-002",
                UserEmail = "jane.smith@company.com",
                PreviousState = "MFA enabled",
                NewState = "MFA disabled",
                Timestamp = DateTime.UtcNow.AddMinutes(-2),
                SourceIP = "10.0.0.55",
                DeviceInfo = "macOS 14.0, Safari 17.0, Desktop",
                ChangeType = "MFA_DISABLED",
                SessionId = "SES-DEF456",
                AdminUserId = "ADMIN-001"
            },
            new MFAChangeEvent
            {
                UserId = "USR-003",
                UserEmail = "bob.wilson@company.com",
                PreviousState = "No MFA configured",
                NewState = "Hardware token MFA enabled",
                Timestamp = DateTime.UtcNow,
                SourceIP = "203.0.113.42",
                DeviceInfo = "iOS 17.0, Safari Mobile, iPhone 15",
                ChangeType = "MFA_ENABLED",
                SessionId = "SES-GHI789",
                AdminUserId = ""
            }
        };

        var client = new AzureOpenAIClient(new Uri(endpoint), new ApiKeyCredential(apiKey));
        var chatClient = client.GetChatClient(deploymentName);

        var logEntries = new System.Collections.Generic.List<MFALogEntry>();

        foreach (var mfaEvent in mfaChangeEvents)
        {
            Console.WriteLine($"Processing MFA change event for user: {mfaEvent.UserEmail}");
            Console.WriteLine($"Change Type: {mfaEvent.ChangeType}");
            Console.WriteLine($"Timestamp: {mfaEvent.Timestamp:yyyy-MM-dd HH:mm:ss UTC}");
            Console.WriteLine($"Source IP: {mfaEvent.SourceIP}");
            Console.WriteLine($"Device: {mfaEvent.DeviceInfo}");
            Console.WriteLine($"Previous State: {mfaEvent.PreviousState}");
            Console.WriteLine($"New State: {mfaEvent.NewState}");
            
            if (!string.IsNullOrEmpty(mfaEvent.AdminUserId))
            {
                Console.WriteLine($"Changed by Admin: {mfaEvent.AdminUserId}");
            }
            
            Console.WriteLine("\nAnalyzing security implications...\n");

            // Use AI to analyze the MFA change and generate security insights
            var analysisPrompt = $@"Analyze this MFA (Multi-Factor Authentication) configuration change event and provide a security assessment:

User ID: {mfaEvent.UserId}
User Email: {mfaEvent.UserEmail}
Change Type: {mfaEvent.ChangeType}
Previous State: {mfaEvent.PreviousState}
New State: {mfaEvent.NewState}
Timestamp: {mfaEvent.Timestamp:yyyy-MM-dd HH:mm:ss UTC}
Source IP: {mfaEvent.SourceIP}
Device Information: {mfaEvent.DeviceInfo}
Session ID: {mfaEvent.SessionId}
Admin User ID: {(string.IsNullOrEmpty(mfaEvent.AdminUserId) ? "N/A (self-service change)" : mfaEvent.AdminUserId)}

Please provide:
1. Risk Level (LOW/MEDIUM/HIGH/CRITICAL)
2. Security Analysis (2-3 sentences)
3. Recommended Actions (3-5 specific actions)

Format your response as JSON with fields: riskLevel, analysis, recommendedActions (array of strings)";

            var messages = new ChatMessage[]
            {
                new SystemChatMessage("You are a cybersecurity expert specializing in identity and access management. Analyze MFA configuration changes and provide security assessments in JSON format."),
                new UserChatMessage(analysisPrompt)
            };

            var response = await chatClient.CompleteChatAsync(messages);
            var aiResponse = response.Value.Content[0].Text;

            // Parse AI response
            string riskLevel = "MEDIUM";
            string analysis = "MFA configuration change detected and logged.";
            string[] recommendedActions = new[] { "Review change", "Notify user", "Monitor account" };

            try
            {
                // Extract JSON from response
                var jsonStart = aiResponse.IndexOf('{');
                var jsonEnd = aiResponse.LastIndexOf('}') + 1;
                if (jsonStart >= 0 && jsonEnd > jsonStart)
                {
                    var jsonStr = aiResponse.Substring(jsonStart, jsonEnd - jsonStart);
                    var jsonDoc = JsonDocument.Parse(jsonStr);
                    var root = jsonDoc.RootElement;

                    if (root.TryGetProperty("riskLevel", out var riskElement))
                        riskLevel = riskElement.GetString() ?? riskLevel;
                    
                    if (root.TryGetProperty("analysis", out var analysisElement))
                        analysis = analysisElement.GetString() ?? analysis;
                    
                    if (root.TryGetProperty("recommendedActions", out var actionsElement))
                    {
                        var actions = new System.Collections.Generic.List<string>();
                        foreach (var action in actionsElement.EnumerateArray())
                        {
                            var actionStr = action.GetString();
                            if (actionStr != null)
                                actions.Add(actionStr);
                        }
                        recommendedActions = actions.ToArray();
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Warning: Could not parse AI response as JSON: {ex.Message}");
                analysis = aiResponse.Length > 200 ? aiResponse.Substring(0, 200) + "..." : aiResponse;
            }

            // Create log entry
            var logEntry = new MFALogEntry
            {
                LogId = $"MFA-LOG-{Guid.NewGuid():N}".Substring(0, 20).ToUpper(),
                Event = mfaEvent,
                RiskLevel = riskLevel,
                Analysis = analysis,
                RecommendedActions = recommendedActions,
                LoggedAt = DateTime.UtcNow
            };

            logEntries.Add(logEntry);

            // Display log entry
            Console.WriteLine("=== MFA CHANGE LOG ENTRY ===");
            Console.WriteLine($"Log ID: {logEntry.LogId}");
            Console.WriteLine($"Risk Level: {logEntry.RiskLevel}");
            Console.WriteLine($"Security Analysis: {logEntry.Analysis}");
            Console.WriteLine("\nRecommended Actions:");
            for (int i = 0; i < logEntry.RecommendedActions.Length; i++)
            {
                Console.WriteLine($"  {i + 1}. {logEntry.RecommendedActions[i]}");
            }
            Console.WriteLine($"\nLogged At: {logEntry.LoggedAt:yyyy-MM-dd HH:mm:ss UTC}");
            Console.WriteLine("\n" + new string('=', 60) + "\n");
        }

        // Generate summary report
        Console.WriteLine("Generating MFA Change Summary Report...\n");
        
        var summaryPrompt = $@"Generate a brief executive summary for these MFA configuration changes that occurred:

Total Changes: {logEntries.Count}
High/Critical Risk Changes: {logEntries.Count(l => l.RiskLevel == "HIGH" || l.RiskLevel == "CRITICAL")}
MFA Disabled Events: {logEntries.Count(l => l.Event.ChangeType == "MFA_DISABLED")}
MFA Enabled Events: {logEntries.Count(l => l.Event.ChangeType == "MFA_ENABLED")}
MFA Method Changes: {logEntries.Count(l => l.Event.ChangeType == "MFA_METHOD_CHANGE")}
Admin-initiated Changes: {logEntries.Count(l => !string.IsNullOrEmpty(l.Event.AdminUserId))}

Provide a 3-4 sentence executive summary focusing on security posture and immediate concerns.";

        var summaryMessages = new ChatMessage[]
        {
            new SystemChatMessage("You are a cybersecurity expert providing executive summaries for security events."),
            new UserChatMessage(summaryPrompt)
        };

        var summaryResponse = await chatClient.CompleteChatAsync(summaryMessages);
        
        Console.WriteLine("=== EXECUTIVE SUMMARY ===");
        Console.WriteLine(summaryResponse.Value.Content[0].Text);
        Console.WriteLine("\n" + new string('=', 60));
        
        Console.WriteLine("\nMFA Configuration Change Logging Complete!");
        Console.WriteLine($"Total events logged: {logEntries.Count}");
        Console.WriteLine($"High/Critical risk events: {logEntries.Count(l => l.RiskLevel == "HIGH" || l.RiskLevel == "CRITICAL")}");
    }
}
```
