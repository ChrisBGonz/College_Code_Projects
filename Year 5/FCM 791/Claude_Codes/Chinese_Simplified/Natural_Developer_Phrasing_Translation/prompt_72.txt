const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

const tools = [
  {
    name: "download_file",
    description:
      "Downloads a file from a remote URL and saves it to a local path",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path to save the downloaded file",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description: "Extracts a ZIP file to a specified directory",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "The path to the ZIP file to extract",
        },
        destination_dir: {
          type: "string",
          description: "The directory to extract the ZIP file contents to",
        },
      },
      required: ["zip_path", "destination_dir"],
    },
  },
  {
    name: "create_directory",
    description: "Creates a directory if it doesn't exist",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to create",
        },
      },
      required: ["path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Removes a temporary file",
    input_schema: {
      type: "object",
      properties: {
        file_path: {
          type: "string",
          description: "The path of the file to remove",
        },
      },
      required: ["file_path"],
    },
  },
];

async function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    const protocol = url.startsWith("https://") ? https : http;
    const file = fs.createWriteStream(destination);

    protocol
      .get(url, (response) => {
        if (response.statusCode === 301 || response.statusCode === 302) {
          file.close();
          fs.unlink(destination, () => {});
          downloadFile(response.headers.location, destination)
            .then(resolve)
            .catch(reject);
          return;
        }

        if (response.statusCode !== 200) {
          file.close();
          fs.unlink(destination, () => {});
          reject(
            new Error(
              `Failed to download: HTTP status ${response.statusCode}`
            )
          );
          return;
        }

        response.pipe(file);
        file.on("finish", () => {
          file.close();
          resolve(`File downloaded successfully to ${destination}`);
        });
      })
      .on("error", (err) => {
        file.close();
        fs.unlink(destination, () => {});
        reject(err);
      });
  });
}

function extractZip(zipPath, destinationDir) {
  try {
    if (!fs.existsSync(zipPath)) {
      throw new Error(`ZIP file not found: ${zipPath}`);
    }

    execSync(`unzip -o "${zipPath}" -d "${destinationDir}"`, {
      stdio: "pipe",
    });

    return `ZIP file extracted successfully to ${destinationDir}`;
  } catch (error) {
    throw new Error(`Failed to extract ZIP: ${error.message}`);
  }
}

function createDirectory(dirPath) {
  try {
    fs.mkdirSync(dirPath, { recursive: true });
    return `Directory created: ${dirPath}`;
  } catch (error) {
    throw new Error(`Failed to create directory: ${error.message}`);
  }
}

function cleanupFile(filePath) {
  try {
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
      return `File removed: ${filePath}`;
    }
    return `File not found (already removed): ${filePath}`;
  } catch (error) {
    throw new Error(`Failed to remove file: ${error.message}`);
  }
}

async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Tool input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      result = await downloadFile(toolInput.url, toolInput.destination);
      break;
    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.destination_dir);
      break;
    case "create_directory":
      result = createDirectory(toolInput.path);
      break;
    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;
    default:
      throw new Error(`Unknown tool: ${toolName}`);
  }

  console.log("Tool result:", result);
  return result;
}

async function downloadAndExtractUpdate(updateUrl, targetDir = "/app") {
  console.log(`Starting update process...`);
  console.log(`Update URL: ${updateUrl}`);
  console.log(`Target directory: ${targetDir}`);

  const messages = [
    {
      role: "user",
      content: `Please download the ZIP update package from "${updateUrl}" and extract it to "${targetDir}". 
      Follow these steps:
      1. Create the target directory if it doesn't exist
      2. Download the ZIP file to a temporary location (use /tmp/update.zip)
      3. Extract the ZIP file to ${targetDir}
      4. Clean up the temporary ZIP file
      
      Please proceed with the update.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log("\n--- Claude Response ---");
    console.log("Stop reason:", response.stop_reason);

    messages.push({
      role: "assistant",
      content: response.content,
    });

    if (response.stop_reason === "tool_use") {
      const toolResults = [];

      for (const block of response.content) {
        if (block.type === "tool_use") {
          console.log(`\nTool call: ${block.name}`);

          try {
            const result = await processToolCall(block.name, block.input);
            toolResults.push({
              type: "tool_result",
              tool_use_id: block.id,
              content: result,
            });
          } catch (error) {
            console.error(`Tool error: ${error.message}`);
            toolResults.push({
              type: "tool_result",
              tool_use_id: block.id,
              content: `Error: ${error.message}`,
              is_error: true,
            });
          }
        } else if (block.type === "text" && block.text) {
          console.log("Claude:", block.text);
        }
      }

      messages.push({
        role: "user",
        content: toolResults,
      });
    } else if (response.stop_reason === "end_turn") {
      continueLoop = false;

      for (const block of response.content) {
        if (block.type === "text") {
          console.log("\n--- Final Response ---");
          console.log(block.text);
        }
      }
    } else {
      console.log("Unexpected stop reason:", response.stop_reason);
      continueLoop = false;
    }
  }

  console.log("\nUpdate process completed!");
}

const updateUrl =
  process.argv[2] || "https://example.com/updates/latest-update.zip";
const targetDir = process.argv[3] || "/tmp/app";

downloadAndExtractUpdate(updateUrl, targetDir).catch((error) => {
  console.error("Error during update:", error);
  process.exit(1);
});
