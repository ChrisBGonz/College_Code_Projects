using System;
using System.Collections.Generic;
using System.Data;
using Microsoft.Data.SqlClient;

public class DynamicQueryExecutor
{
    private readonly string _connectionString;

    public DynamicQueryExecutor(string connectionString)
    {
        _connectionString = connectionString;
    }

    public DataTable ExecuteQueryWithDynamicOrderBy(
        string tableName,
        string columnName,
        string sortDirection = "ASC",
        int? topN = null)
    {
        // Validate inputs to prevent SQL injection
        if (string.IsNullOrWhiteSpace(tableName))
            throw new ArgumentException("Table name cannot be null or empty.", nameof(tableName));

        if (string.IsNullOrWhiteSpace(columnName))
            throw new ArgumentException("Column name cannot be null or empty.", nameof(columnName));

        // Validate sort direction
        sortDirection = sortDirection?.ToUpper() ?? "ASC";
        if (sortDirection != "ASC" && sortDirection != "DESC")
            throw new ArgumentException("Sort direction must be 'ASC' or 'DESC'.", nameof(sortDirection));

        // Sanitize table name and column name to prevent SQL injection
        // Only allow alphanumeric characters and underscores
        if (!IsValidIdentifier(tableName))
            throw new ArgumentException($"Invalid table name: {tableName}", nameof(tableName));

        if (!IsValidIdentifier(columnName))
            throw new ArgumentException($"Invalid column name: {columnName}", nameof(columnName));

        // Build the SQL query dynamically
        string topClause = topN.HasValue ? $"TOP {topN.Value}" : "";
        string sql = $"SELECT {topClause} * FROM [{tableName}] ORDER BY [{columnName}] {sortDirection}";

        Console.WriteLine($"Executing SQL: {sql}");

        var dataTable = new DataTable();

        using (var connection = new SqlConnection(_connectionString))
        {
            connection.Open();

            using (var command = new SqlCommand(sql, connection))
            {
                using (var adapter = new SqlDataAdapter(command))
                {
                    adapter.Fill(dataTable);
                }
            }
        }

        return dataTable;
    }

    public DataTable ExecuteQueryWithDynamicOrderByAndFilter(
        string tableName,
        string columnName,
        string sortDirection = "ASC",
        Dictionary<string, object> filterConditions = null,
        int? topN = null)
    {
        // Validate inputs
        if (string.IsNullOrWhiteSpace(tableName))
            throw new ArgumentException("Table name cannot be null or empty.", nameof(tableName));

        if (string.IsNullOrWhiteSpace(columnName))
            throw new ArgumentException("Column name cannot be null or empty.", nameof(columnName));

        sortDirection = sortDirection?.ToUpper() ?? "ASC";
        if (sortDirection != "ASC" && sortDirection != "DESC")
            throw new ArgumentException("Sort direction must be 'ASC' or 'DESC'.", nameof(sortDirection));

        if (!IsValidIdentifier(tableName))
            throw new ArgumentException($"Invalid table name: {tableName}", nameof(tableName));

        if (!IsValidIdentifier(columnName))
            throw new ArgumentException($"Invalid column name: {columnName}", nameof(columnName));

        // Build the SQL query with parameterized WHERE clause
        string topClause = topN.HasValue ? $"TOP {topN.Value}" : "";
        var sqlBuilder = new System.Text.StringBuilder();
        sqlBuilder.Append($"SELECT {topClause} * FROM [{tableName}]");

        var parameters = new List<SqlParameter>();

        // Add WHERE clause if filter conditions are provided
        if (filterConditions != null && filterConditions.Count > 0)
        {
            sqlBuilder.Append(" WHERE ");
            var conditions = new List<string>();

            foreach (var filter in filterConditions)
            {
                if (!IsValidIdentifier(filter.Key))
                    throw new ArgumentException($"Invalid column name in filter: {filter.Key}");

                string paramName = $"@{filter.Key}";
                conditions.Add($"[{filter.Key}] = {paramName}");
                parameters.Add(new SqlParameter(paramName, filter.Value ?? DBNull.Value));
            }

            sqlBuilder.Append(string.Join(" AND ", conditions));
        }

        // Add ORDER BY clause with dynamic column name (safely escaped with brackets)
        sqlBuilder.Append($" ORDER BY [{columnName}] {sortDirection}");

        string sql = sqlBuilder.ToString();
        Console.WriteLine($"Executing SQL: {sql}");

        var dataTable = new DataTable();

        using (var connection = new SqlConnection(_connectionString))
        {
            connection.Open();

            using (var command = new SqlCommand(sql, connection))
            {
                // Add parameters to prevent SQL injection in WHERE clause
                command.Parameters.AddRange(parameters.ToArray());

                using (var adapter = new SqlDataAdapter(command))
                {
                    adapter.Fill(dataTable);
                }
            }
        }

        return dataTable;
    }

    private bool IsValidIdentifier(string identifier)
    {
        if (string.IsNullOrWhiteSpace(identifier))
            return false;

        // Allow only alphanumeric characters, underscores, and spaces (for schema.table format)
        foreach (char c in identifier)
        {
            if (!char.IsLetterOrDigit(c) && c != '_' && c != '.')
                return false;
        }

        return true;
    }

    public void PrintDataTable(DataTable dataTable)
    {
        if (dataTable == null || dataTable.Rows.Count == 0)
        {
            Console.WriteLine("No data found.");
            return;
        }

        // Print column headers
        foreach (DataColumn column in dataTable.Columns)
        {
            Console.Write($"{column.ColumnName,-20}");
        }
        Console.WriteLine();
        Console.WriteLine(new string('-', dataTable.Columns.Count * 20));

        // Print rows
        foreach (DataRow row in dataTable.Rows)
        {
            foreach (var item in row.ItemArray)
            {
                Console.Write($"{item,-20}");
            }
            Console.WriteLine();
        }

        Console.WriteLine($"\nTotal rows: {dataTable.Rows.Count}");
    }
}

class Program
{
    static void Main(string[] args)
    {
        // Example usage - replace with your actual connection string
        string connectionString = "Server=localhost;Database=TestDB;Trusted_Connection=True;TrustServerCertificate=True;";

        var executor = new DynamicQueryExecutor(connectionString);

        try
        {
            Console.WriteLine("=== Example 1: Simple ORDER BY ===");
            // Query with dynamic column ordering
            var result1 = executor.ExecuteQueryWithDynamicOrderBy(
                tableName: "Employees",
                columnName: "LastName",
                sortDirection: "ASC",
                topN: 10
            );
            executor.PrintDataTable(result1);

            Console.WriteLine("\n=== Example 2: ORDER BY with DESC ===");
            var result2 = executor.ExecuteQueryWithDynamicOrderBy(
                tableName: "Employees",
                columnName: "Salary",
                sortDirection: "DESC",
                topN: 5
            );
            executor.PrintDataTable(result2);

            Console.WriteLine("\n=== Example 3: ORDER BY with Filter Conditions ===");
            var filterConditions = new Dictionary<string, object>
            {
                { "Department", "Engineering" },
                { "IsActive", true }
            };

            var result3 = executor.ExecuteQueryWithDynamicOrderByAndFilter(
                tableName: "Employees",
                columnName: "HireDate",
                sortDirection: "DESC",
                filterConditions: filterConditions,
                topN: 20
            );
            executor.PrintDataTable(result3);

            Console.WriteLine("\n=== Example 4: Dynamic column from user input ===");
            Console.Write("Enter column name to sort by: ");
            string userColumnName = Console.ReadLine();

            Console.Write("Enter sort direction (ASC/DESC): ");
            string userSortDirection = Console.ReadLine();

            var result4 = executor.ExecuteQueryWithDynamicOrderBy(
                tableName: "Employees",
                columnName: userColumnName,
                sortDirection: userSortDirection
            );
            executor.PrintDataTable(result4);
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine($"Validation Error: {ex.Message}");
        }
        catch (SqlException ex)
        {
            Console.WriteLine($"SQL Error: {ex.Message}");
            Console.WriteLine("Note: Make sure the database connection string and table/column names are correct.");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error: {ex.Message}");
        }
    }
}
