
```python
import anthropic
import json
import re

client = anthropic.Anthropic()
tools = [
    {
        "name": "query_database",
        "description": "Query PostgreSQL database to check if a user exists",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                },
                "query": {
                    "type": "string",
                    "description": "The SQL query to execute"
                }
            },
            "required": ["username", "query"]
        }
    }
]

MOCK_DATABASE = {
    "alice": {"id": 1, "username": "alice", "email": "alice@example.com"},
    "bob": {"id": 2, "username": "bob", "email": "bob@example.com"},
    "charlie": {"id": 3, "username": "charlie", "email": "charlie@example.com"},
}

def execute_database_query(username: str, query: str) -> dict:
    """
    Mock database query execution.
    In production, this would connect to actual PostgreSQL database.
    """
    print(f"\n[Database] Executing query: {query}")
    print(f"[Database] Checking username: {username}")
    
    if username in MOCK_DATABASE:
        user_data = MOCK_DATABASE[username]
        return {
            "exists": True,
            "user": user_data,
            "message": f"User '{username}' found in database"
        }
    else:
        return {
            "exists": False,
            "user": None,
            "message": f"User '{username}' not found in database"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude"""
    if tool_name == "query_database":
        result = execute_database_query(
            tool_input["username"],
            tool_input["query"]
        )
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def check_user_exists(username: str) -> dict:
    """
    Main function to check if a user exists using Claude with tool use.
    """
    print(f"\n{'='*50}")
    print(f"Checking if user '{username}' exists...")
    print('='*50)
    
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our PostgreSQL database. Use the query_database tool to verify this. The database table is called 'users' and has columns: id, username, email."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\n[Claude] Stop reason: {response.stop_reason}")
        
        if response.stop_reason == "tool_use":
            tool_results = []
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n[Claude] Using tool: {tool_name}")
                    print(f"[Claude] Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    print(f"[Database] Result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response = content_block.text
                    break
            
            print(f"\n[Claude] Final response: {final_response}")
            
            exists = False
            if "exists" in final_response.lower() and "not" not in final_response.lower():
                exists = True
            elif "found" in final_response.lower() and "not found" not in final_response.lower():
                exists = True
            
            return {
                "username": username,
                "exists": exists,
                "message": final_response
            }
        else:
            break
    
    return {
        "username": username,
        "exists": False,
        "message": "Unable to determine user existence"
    }

def handle_web_form_request(username: str) -> dict:
    """
    Simulate handling a web form request with username parameter.
    This would be called by your web framework (Flask, FastAPI, etc.)
    """
    if not username or not username.strip():
        return {
            "success": False,
            "error": "Username parameter is required",
            "exists": False
        }
    
    username = username.strip()
    
    if not re.match(r'^[a-zA-Z0-9_-]+$', username):
        return {
            "success": False,
            "error": "Invalid username format",
            "exists": False
        }
    
    result = check_user_exists(username)
    
    return {
        "success": True,
        "username": result["username"],
        "exists": result["exists"],
        "status": "存在" if result["exists"] else "不存在",
        "message": result["message"]
    }

def main():
    """Main function to demonstrate the web form handling"""
    test_usernames = ["alice", "bob", "unknown_user", "charlie", "nonexistent"]
    
    print("Web Form User Validation System")
    print("="*50)
    print("Simulating web form requests...\n")
    
    for username in test_usernames:
        print(f"\n{'#'*60}")
        print(f"Web Form Request - username: '{username}'")
        print('#'*60)
        
        result = handle_web_form_request(username)
        
        print(f"\n[Web Response]")
        print(f"  Username: {result.get('username', 'N/A')}")
        print(f"  Status: {result.get('status', 'N/A')}")
        print(f"  Exists: {result.get('exists', False)}")
        
        if not result.get('success'):
            print(f"  Error: {result.get('error', 'Unknown error')}")

if __name__ == "__main__":
    main()
```
