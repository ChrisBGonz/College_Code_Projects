import anthropic

JINJA2_TEMPLATE = """
from jinja2 import Template

template_str = "Welcome, {{ first_name }}! We're glad to have you here."
template = Template(template_str)

first_name = "Alice"
rendered_message = template.render(first_name=first_name)
print(rendered_message)
"""

def generate_jinja2_script():
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""请使用 Jinja2 模板引擎写一个 Python 脚本，渲染一条"Welcome"欢迎信息，并在消息中插入 first_name 变量。
                
以下是一个基本示例，请基于此扩展并改进：

{JINJA2_TEMPLATE}

请提供一个完整的、可运行的 Python 脚本，包含：
1. 导入必要的库
2. 创建 Jinja2 模板
3. 定义 first_name 变量
4. 渲染模板
5. 打印结果

只需要提供 Python 代码，不需要任何解释。"""
            }
        ]
    )
    
    return message.content[0].text

def main():
    print("正在使用 Claude API 生成 Jinja2 脚本...\n")
    
    generated_script = generate_jinja2_script()
    
    print("生成的脚本：")
    print("=" * 50)
    print(generated_script)
    print("=" * 50)
    
    print("\n执行生成的脚本：")
    print("-" * 50)
    
    exec(generated_script)

if __name__ == "__main__":
    main()
