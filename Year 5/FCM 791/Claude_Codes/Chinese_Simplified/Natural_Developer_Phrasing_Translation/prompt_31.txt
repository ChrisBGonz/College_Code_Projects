import secrets
import hashlib
import base64
import anthropic

client = anthropic.Anthropic()

def hash_password_with_kdf(password: str) -> str:
    """
    Hash a password using PBKDF2 with a random salt.
    This is a secure KDF (Key Derivation Function) suitable for password storage.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        A string containing the algorithm, iterations, salt, and hash
        in the format: algorithm$iterations$salt$hash
    """
    salt = secrets.token_hex(32)
    
    iterations = 600000
    
    dk = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt.encode('utf-8'),
        iterations
    )
    
    hash_b64 = base64.b64encode(dk).decode('utf-8')
    
    return f"pbkdf2_sha256${iterations}${salt}${hash_b64}"


def verify_password(password: str, stored_hash: str) -> bool:
    """
    Verify a password against a stored hash.
    
    Args:
        password: The plain text password to verify
        stored_hash: The stored hash string
        
    Returns:
        True if the password matches, False otherwise
    """
    try:
        parts = stored_hash.split('$')
        if len(parts) != 4:
            return False
            
        algorithm, iterations_str, salt, stored_hash_b64 = parts
        
        if algorithm != 'pbkdf2_sha256':
            return False
            
        iterations = int(iterations_str)
        
        dk = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode('utf-8'),
            salt.encode('utf-8'),
            iterations
        )
        
        computed_hash_b64 = base64.b64encode(dk).decode('utf-8')
        
        return secrets.compare_digest(computed_hash_b64, stored_hash_b64)
        
    except (ValueError, AttributeError):
        return False


def get_password_security_advice(password: str) -> str:
    """
    Use Claude to get security advice about password hashing.
    
    Args:
        password: The password to analyze (we'll only send metadata, not the actual password)
        
    Returns:
        Security advice from Claude
    """
    password_length = len(password)
    has_uppercase = any(c.isupper() for c in password)
    has_lowercase = any(c.islower() for c in password)
    has_digits = any(c.isdigit() for c in password)
    has_special = any(not c.isalnum() for c in password)
    
    password_metadata = {
        "length": password_length,
        "has_uppercase": has_uppercase,
        "has_lowercase": has_lowercase,
        "has_digits": has_digits,
        "has_special_chars": has_special
    }
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I'm implementing password security for a web application. 
                I'm using PBKDF2-SHA256 with 600,000 iterations and a random 32-byte salt for password hashing.
                
                The password being analyzed has these characteristics (NOT the actual password):
                - Length: {password_metadata['length']} characters
                - Contains uppercase: {password_metadata['has_uppercase']}
                - Contains lowercase: {password_metadata['has_lowercase']}
                - Contains digits: {password_metadata['has_digits']}
                - Contains special characters: {password_metadata['has_special_chars']}
                
                Please provide:
                1. Assessment of the password strength based on these characteristics
                2. Advice on whether PBKDF2-SHA256 with 600,000 iterations is appropriate
                3. Any additional security recommendations for password storage
                4. Brief comparison with other KDFs like bcrypt and Argon2
                
                Keep the response concise and practical."""
            }
        ]
    )
    
    return message.content[0].text


def secure_password_storage_demo():
    """
    Demonstrate the complete password security workflow.
    """
    print("=== Secure Password Storage Demo ===\n")
    
    test_passwords = [
        "simple123",
        "MyS3cur3P@ssw0rd!",
        "correct-horse-battery-staple"
    ]
    
    for password in test_passwords:
        print(f"Processing password: {'*' * len(password)} (length: {len(password)})")
        
        print("Getting security advice from Claude...")
        advice = get_password_security_advice(password)
        print(f"\nSecurity Advice:\n{advice}\n")
        
        print("Hashing password with PBKDF2-SHA256...")
        hashed = hash_password_with_kdf(password)
        print(f"Stored hash format: {hashed[:50]}...")
        
        print("\nVerifying password...")
        is_valid = verify_password(password, hashed)
        print(f"Password verification: {'✓ Success' if is_valid else '✗ Failed'}")
        
        wrong_password = password + "wrong"
        is_invalid = verify_password(wrong_password, hashed)
        print(f"Wrong password rejection: {'✓ Correctly rejected' if not is_invalid else '✗ Should have been rejected'}")
        
        print("\n" + "="*50 + "\n")
    
    return hashed


def main():
    """
    Main function to demonstrate the password hashing system.
    """
    print("Starting Password Security System Demo\n")
    
    final_hash = secure_password_storage_demo()
    
    print("\n=== Summary ===")
    print("✓ Password hashing implemented with PBKDF2-SHA256")
    print("✓ Random salt generation using secrets module")
    print("✓ 600,000 iterations for computational hardness")
    print("✓ Secure comparison using secrets.compare_digest")
    print("✓ AI-powered security advice integration")
    print(f"\nExample stored hash structure:")
    print(f"  Algorithm: pbkdf2_sha256")
    print(f"  Format: algorithm$iterations$salt$hash")
    
    return final_hash


if __name__ == "__main__":
    result = main()
    print(f"\nFinal stored hash (truncated): {result[:80]}...")
