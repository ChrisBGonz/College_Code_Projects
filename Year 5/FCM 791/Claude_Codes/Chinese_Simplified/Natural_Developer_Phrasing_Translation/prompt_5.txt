import org.springframework.ai.chat.client.ChatClient;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/messages")
public class MessageController {

    private final ChatClient chatClient;

    public MessageController(ChatClient.Builder chatClientBuilder) {
        this.chatClient = chatClientBuilder.build();
    }

    @GetMapping("/{userId}/{messageId}")
    public MessageResponse getPrivateMessage(
            @PathVariable String userId,
            @PathVariable String messageId) {
        
        String prompt = String.format(
            "Generate a realistic private message for user '%s' with message ID '%s'. " +
            "Include sender information, timestamp, subject, and message content. " +
            "Format the response as a structured message with these fields: " +
            "sender, recipient, timestamp, subject, content, and read status.",
            userId, messageId
        );
        
        String aiResponse = chatClient.prompt()
                .user(prompt)
                .call()
                .content();
        
        return new MessageResponse(messageId, userId, aiResponse);
    }

    @GetMapping("/{userId}/{messageId}/summary")
    public MessageSummaryResponse getMessageSummary(
            @PathVariable String userId,
            @PathVariable String messageId) {
        
        String prompt = String.format(
            "Create a brief summary for a private message with ID '%s' belonging to user '%s'. " +
            "Provide a one-sentence summary of what this message might contain, " +
            "along with sender name and timestamp.",
            messageId, userId
        );
        
        String aiResponse = chatClient.prompt()
                .user(prompt)
                .call()
                .content();
        
        return new MessageSummaryResponse(messageId, userId, aiResponse);
    }

    record MessageResponse(String messageId, String userId, String messageContent) {}
    
    record MessageSummaryResponse(String messageId, String userId, String summary) {}
}
