
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

function question(prompt) {
  return new Promise((resolve) => {
    rl.question(prompt, resolve);
  });
}

async function chat(userMessage) {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert Node.js developer. Help users create middleware, APIs, and backend solutions. Provide clear, well-commented code examples.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main() {
  console.log("Node.js Middleware Assistant");
  console.log("============================");
  console.log(
    "I can help you create Node.js middleware for logging requests with full request body."
  );
  console.log('Type "exit" to quit.\n');

  // Start with the initial request
  const initialRequest =
    "创建一个 Node.js 中间件，用于记录每个进入的请求日志，并包含完整的请求体（request body），以便调试使用。请提供完整的代码示例，包括如何在 Express 应用中使用这个中间件。";

  console.log("User: " + initialRequest);
  console.log("\nAssistant: Generating middleware code...\n");

  try {
    const response = await chat(initialRequest);
    console.log(response);
    console.log("\n" + "=".repeat(50) + "\n");

    // Continue conversation
    while (true) {
      const userInput = await question("You: ");

      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        break;
      }

      if (!userInput.trim()) {
        continue;
      }

      try {
        const assistantResponse = await chat(userInput);
        console.log("\nAssistant: " + assistantResponse);
        console.log("\n" + "=".repeat(50) + "\n");
      } catch (error) {
        console.error("Error:", error.message);
      }
    }
  } catch (error) {
    console.error("Error:", error.message);
    rl.close();
  }
}

main();
```
