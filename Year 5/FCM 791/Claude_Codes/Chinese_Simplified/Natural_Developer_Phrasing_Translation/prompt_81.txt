
```python
import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "verify_password",
        "description": "Verify if the provided password is correct for authentication",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username attempting to login"
                },
                "password": {
                    "type": "string",
                    "description": "The password to verify"
                }
            },
            "required": ["username", "password"]
        }
    },
    {
        "name": "create_auth_response",
        "description": "Create an authentication response based on verification result",
        "input_schema": {
            "type": "object",
            "properties": {
                "authenticated": {
                    "type": "boolean",
                    "description": "Whether authentication was successful"
                },
                "username": {
                    "type": "string",
                    "description": "The username that attempted authentication"
                },
                "message": {
                    "type": "string",
                    "description": "Message to include in the response"
                }
            },
            "required": ["authenticated", "username", "message"]
        }
    }
]

VALID_CREDENTIALS = {
    "admin": "secure_password_123",
    "user1": "mypassword456",
    "testuser": "test123"
}

def verify_password(username: str, password: str) -> dict:
    """Verify username and password against stored credentials"""
    if username in VALID_CREDENTIALS and VALID_CREDENTIALS[username] == password:
        return {
            "success": True,
            "username": username,
            "message": "Password verified successfully"
        }
    else:
        return {
            "success": False,
            "username": username,
            "message": "Invalid username or password"
        }

def create_auth_response(authenticated: bool, username: str, message: str) -> dict:
    """Create HTTP-like authentication response"""
    if authenticated:
        return {
            "status_code": 200,
            "status": "OK",
            "body": {
                "authenticated": True,
                "username": username,
                "message": message,
                "token": f"mock_token_{username}_abc123"
            }
        }
    else:
        return {
            "status_code": 401,
            "status": "Unauthorized",
            "body": {
                "authenticated": False,
                "username": username,
                "message": message,
                "error": "Authentication failed"
            }
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "verify_password":
        result = verify_password(tool_input["username"], tool_input["password"])
        return json.dumps(result)
    elif tool_name == "create_auth_response":
        result = create_auth_response(
            tool_input["authenticated"],
            tool_input["username"],
            tool_input["message"]
        )
        return json.dumps(result)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def authenticate_user(username: str, password: str) -> dict:
    """
    Main authentication function that uses Claude to orchestrate the auth process
    Returns authentication response with appropriate status code
    """
    print(f"\n{'='*50}")
    print(f"Authentication attempt for user: {username}")
    print(f"{'='*50}")
    
    messages = [
        {
            "role": "user",
            "content": f"Please authenticate the user with username '{username}' and password '{password}'. First verify the password, then create an appropriate authentication response based on the result."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"Claude: {block.text}")
            break
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nTool call: {tool_name}")
                    print(f"Input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
            for result in tool_results:
                result_data = json.loads(result["content"])
                if "status_code" in result_data:
                    return result_data
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return {
        "status_code": 500,
        "status": "Internal Server Error",
        "body": {
            "error": "Authentication process failed unexpectedly"
        }
    }

def main():
    """Test the authentication system with various credentials"""
    
    test_cases = [
        ("admin", "secure_password_123"),
        ("user1", "wrongpassword"),
        ("nonexistent", "somepassword"),
        ("testuser", "test123"),
        ("admin", "wrong_password"),
    ]
    
    print("Login Authentication System Test")
    print("="*60)
    
    for username, password in test_cases:
        auth_response = authenticate_user(username, password)
        
        print(f"\n{'='*50}")
        print(f"Final Authentication Response:")
        print(f"Status Code: {auth_response.get('status_code')}")
        print(f"Status: {auth_response.get('status')}")
        print(f"Body: {json.dumps(auth_response.get('body', {}), indent=2)}")
        
        if auth_response.get('status_code') == 401:
            print("⚠️  ACCESS DENIED - 401 Unauthorized")
        elif auth_response.get('status_code') == 200:
            print("✅ ACCESS GRANTED - 200 OK")
        
        print()

if __name__ == "__main__":
    main()
```
