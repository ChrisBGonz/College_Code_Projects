```javascript
/**
 * Bank App Auto-Logout Module
 * Automatically logs out user after 15 minutes of inactivity
 */

const AutoLogout = (() => {
  // Configuration
  const CONFIG = {
    INACTIVITY_TIMEOUT: 15 * 60 * 1000, // 15 minutes in milliseconds
    WARNING_BEFORE_LOGOUT: 60 * 1000,    // Show warning 1 minute before logout
    LOGIN_PAGE_URL: '/login',
    LOGOUT_API_URL: '/api/auth/logout',
    SESSION_TOKEN_KEY: 'auth_token',
    REFRESH_TOKEN_KEY: 'refresh_token',
    USER_DATA_KEY: 'user_data',
    THROTTLE_DELAY: 500, // Throttle activity events (ms)
  };

  // State
  let inactivityTimer = null;
  let warningTimer = null;
  let warningDialogElement = null;
  let isInitialized = false;
  let isLoggingOut = false;
  let lastActivityTime = Date.now();

  // Events to monitor for user activity
  const ACTIVITY_EVENTS = [
    'mousedown',
    'mousemove',
    'keydown',
    'keypress',
    'scroll',
    'touchstart',
    'touchmove',
    'click',
    'wheel',
    'pointerdown',
    'pointermove',
    'visibilitychange',
    'focus',
  ];

  /**
   * Throttle function to limit how often activity handler fires
   * @param {Function} fn - Function to throttle
   * @param {number} delay - Throttle delay in ms
   * @returns {Function} Throttled function
   */
  const throttle = (fn, delay) => {
    let lastCall = 0;
    return (...args) => {
      const now = Date.now();
      if (now - lastCall >= delay) {
        lastCall = now;
        fn(...args);
      }
    };
  };

  /**
   * Clear all active timers
   */
  const clearAllTimers = () => {
    if (inactivityTimer) {
      clearTimeout(inactivityTimer);
      inactivityTimer = null;
    }
    if (warningTimer) {
      clearTimeout(warningTimer);
      warningTimer = null;
    }
  };

  /**
   * Remove warning dialog from DOM if present
   */
  const removeWarningDialog = () => {
    if (warningDialogElement && warningDialogElement.parentNode) {
      warningDialogElement.parentNode.removeChild(warningDialogElement);
      warningDialogElement = null;
    }
  };

  /**
   * Show warning dialog before auto-logout
   * @param {number} secondsRemaining - Seconds until logout
   */
  const showWarningDialog = (secondsRemaining) => {
    removeWarningDialog();

    warningDialogElement = document.createElement('div');
    warningDialogElement.id = 'auto-logout-warning';
    warningDialogElement.setAttribute('role', 'alertdialog');
    warningDialogElement.setAttribute('aria-modal', 'true');
    warningDialogElement.setAttribute('aria-labelledby', 'logout-warning-title');
    warningDialogElement.setAttribute('aria-describedby', 'logout-warning-desc');

    warningDialogElement.style.cssText = `
      position: fixed;
      top: 0;
      left: 0;
      width: 100%;
      height: 100%;
      background: rgba(0, 0, 0, 0.6);
      display: flex;
      align-items: center;
      justify-content: center;
      z-index: 999999;
      font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
    `;

    let countdown = Math.floor(secondsRemaining / 1000);

    warningDialogElement.innerHTML = `
      <div style="
        background: #ffffff;
        border-radius: 12px;
        padding: 32px;
        max-width: 420px;
        width: 90%;
        box-shadow: 0 20px 60px rgba(0,0,0,0.3);
        text-align: center;
      ">
        <div style="
          width: 64px;
          height: 64px;
          background: #fff3cd;
          border-radius: 50%;
          display: flex;
          align-items: center;
          justify-content: center;
          margin: 0 auto 20px;
          font-size: 28px;
        ">⚠️</div>
        <h2 id="logout-warning-title" style="
          margin: 0 0 12px;
          color: #1a1a2e;
          font-size: 20px;
          font-weight: 700;
        ">Session Expiring Soon</h2>
        <p id="logout-warning-desc" style="
          margin: 0 0 8px;
          color: #555;
          font-size: 15px;
          line-height: 1.5;
        ">For your security, you will be automatically logged out due to inactivity.</p>
        <p style="margin: 0 0 28px; color: #888; font-size: 14px;">
          Logging out in <strong id="logout-countdown" style="color: #d32f2f; font-size: 18px;">${countdown}</strong> seconds
        </p>
        <div style="display: flex; gap: 12px; justify-content: center;">
          <button id="logout-now-btn" style="
            padding: 12px 24px;
            background: #f5f5f5;
            color: #333;
            border: 1px solid #ddd;
            border-radius: 8px;
            font-size: 14px;
            font-weight: 600;
            cursor: pointer;
            transition: background 0.2s;
          ">Logout Now</button>
          <button id="stay-logged-in-btn" style="
            padding: 12px 24px;
            background: #1565c0;
            color: white;
            border: none;
            border-radius: 8px;
            font-size: 14px;
            font-weight: 600;
            cursor: pointer;
            transition: background 0.2s;
          ">Stay Logged In</button>
        </div>
      </div>
    `;

    document.body.appendChild(warningDialogElement);

    // Countdown interval
    const countdownInterval = setInterval(() => {
      countdown -= 1;
      const countdownEl = document.getElementById('logout-countdown');
      if (countdownEl) {
        countdownEl.textContent = countdown;
      }
      if (countdown <= 0) {
        clearInterval(countdownInterval);
      }
    }, 1000);

    // Button event listeners
    const stayBtn = document.getElementById('stay-logged-in-btn');
    const logoutBtn = document.getElementById('logout-now-btn');

    if (stayBtn) {
      stayBtn.addEventListener('click', () => {
        clearInterval(countdownInterval);
        removeWarningDialog();
        resetTimer();
      });
    }

    if (logoutBtn) {
      logoutBtn.addEventListener('click', () => {
        clearInterval(countdownInterval);
        removeWarningDialog();
        performLogout('manual');
      });
    }

    // Focus the stay logged in button for accessibility
    if (stayBtn) {
      stayBtn.focus();
    }
  };

  /**
   * Clear all session/token data from storage
   */
  const clearSessionData = () => {
    try {
      // Clear localStorage
      localStorage.removeItem(CONFIG.SESSION_TOKEN_KEY);
      localStorage.removeItem(CONFIG.REFRESH_TOKEN_KEY);
      localStorage.removeItem(CONFIG.USER_DATA_KEY);

      // Clear sessionStorage
      sessionStorage.clear();

      // Clear cookies (auth-related)
      const cookiesToClear = ['auth_token', 'session_id', 'refresh_token', 'user_session'];
      cookiesToClear.forEach((cookieName) => {
        document.cookie = `${cookieName}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/; secure; samesite=strict`;
        document.cookie = `${cookieName}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;`;
      });

      console.info('[AutoLogout] Session data cleared successfully.');
    } catch (error) {
      console.error('[AutoLogout] Error clearing session data:', error);
    }
  };

  /**
   * Call the logout API endpoint
   * @returns {Promise<void>}
   */
  const callLogoutAPI = async () => {
    try {
      const token = localStorage.getItem(CONFIG.SESSION_TOKEN_KEY)
        || sessionStorage.getItem(CONFIG.SESSION_TOKEN_KEY);

      const response = await fetch(CONFIG.LOGOUT_API_URL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...(token && { Authorization: `Bearer ${token}` }),
        },
        credentials: 'include',
        body: JSON.stringify({
          reason: 'inactivity_timeout',
          timestamp: new Date().toISOString(),
        }),
      });

      if (!response.ok) {
        console.warn('[AutoLogout] Logout API returned non-OK status:', response.status);
      } else {
        console.info('[AutoLogout] Logout API call successful.');
      }
    } catch (error) {
      // Don't block logout flow if API call fails
      console.warn('[AutoLogout] Logout API call failed (proceeding with local logout):', error);
    }
  };

  /**
   * Redirect to login page
   * @param {string} reason - Reason for logout
   */
  const redirectToLogin = (reason) => {
    const loginUrl = new URL(CONFIG.LOGIN_PAGE_URL, window.location.origin);

    if (reason === 'inactivity') {
      loginUrl.searchParams.set('reason', 'session_expired');
      loginUrl.searchParams.set('message', 'You have been logged out due to inactivity.');
    }

    // Use replace to prevent back-navigation to authenticated pages
    window.location.replace(loginUrl.toString());
  };

  /**
   * Perform the complete logout sequence
   * @param {string} reason - 'inactivity' | 'manual'
   */
  const performLogout = async (reason = 'inactivity') => {
    if (isLoggingOut) return;
    isLoggingOut = true;

    console.info(`[AutoLogout] Performing logout. Reason: ${reason}`);

    // Stop monitoring
    destroy();

    // Remove warning dialog if present
    removeWarningDialog();

    // Dispatch custom event for app to handle (e.g., Redux store cleanup)
    const logoutEvent = new CustomEvent('autoLogout', {
      detail: {
        reason,
        timestamp: new Date().toISOString(),
        lastActivity: new Date(lastActivityTime).toISOString(),
      },
      bubbles: true,
    });
    document.dispatchEvent(logoutEvent);

    // Call logout API (fire and forget, don't block redirect)
    callLogoutAPI().finally(() => {
      // Clear all local session data
      clearSessionData();

      // Redirect to login
      redirectToLogin(reason);
    });

    // Fallback redirect in case API call hangs
    setTimeout(() => {
      clearSessionData();
      redirectToLogin(reason);
    }, 3000);
  };

  /**
   * Start/reset the inactivity timers
   */
  const resetTimer = () => {
    if (isLoggingOut) return;

    clearAllTimers();
    lastActivityTime = Date.now();

    // Set warning timer (fires 1 minute before logout)
    warningTimer = setTimeout(() => {
      showWarningDialog(CONFIG.WARNING_BEFORE_LOGOUT);
    }, CONFIG.INACTIVITY_TIMEOUT - CONFIG.WARNING_BEFORE_LOGOUT);

    // Set logout timer
    inactivityTimer = setTimeout(() => {
      performLogout('inactivity');
    }, CONFIG.INACTIVITY_TIMEOUT);
  };

  /**
   * Handle user activity events
   */
  const handleActivity = throttle((event) => {
    // Ignore activity if logout is in progress
    if (isLoggingOut) return;

    // Handle visibility change specifically
    if (event.type === 'visibilitychange') {
      if (document.visibilityState === 'visible') {
        // Check if timeout occurred while tab was hidden
        const elapsed = Date.now() - lastActivityTime;
        if (elapsed >= CONFIG.INACTIVITY_TIMEOUT) {
          performLogout('inactivity');
          return;
        }
      } else {
        // Tab hidden - just update last activity time
        lastActivityTime = Date.now();
        return;
      }
    }

    // Remove warning dialog if user becomes active
    if (warningDialogElement) {
      removeWarningDialog();
    }

    resetTimer();
  }, CONFIG.THROTTLE_DELAY);

  /**
   * Attach all activity event listeners
   */
  const attachEventListeners = () => {
    ACTIVITY_EVENTS.forEach((eventName) => {
      if (eventName === 'visibilitychange') {
        document.addEventListener(eventName, handleActivity, { passive: true });
      } else {
        document.addEventListener(eventName, handleActivity, {
          passive: true,
          capture: true,
        });
      }
    });
  };

  /**
   * Remove all activity event listeners
   */
  const detachEventListeners = () => {
    ACTIVITY_EVENTS.forEach((eventName) => {
      if (eventName === 'visibilitychange') {
        document.removeEventListener(eventName, handleActivity);
      } else {
        document.removeEventListener(eventName, handleActivity, { capture: true });
      }
    });
  };

  /**
   * Handle cross-tab logout synchronization via localStorage events
   */
  const handleStorageEvent = (event) => {
    if (event.key === 'logout_signal' && event.newValue === 'true') {
      if (!isLoggingOut) {
        console.info('[AutoLogout] Logout signal received from another tab.');
        isLoggingOut = true;
        destroy();
        clearSessionData();
        redirectToLogin('inactivity');
      }
    }
  };

  /**
   * Broadcast logout to other tabs
   */
  const broadcastLogout = () => {
    try {
      localStorage.setItem('logout_signal', 'true');
      // Clean up signal after a short delay
      setTimeout(() => localStorage.removeItem('logout_signal'), 2000);
    } catch (e) {
      // Ignore storage errors
    }
  };

  /**
   * Initialize the auto-logout module
   * @param {Object} options - Optional configuration overrides
   */
  const init = (options = {}) => {
    if (isInitialized) {
      console.warn('[AutoLogout] Already initialized. Call destroy() first to reinitialize.');
      return;
    }

    // Apply custom configuration
    Object.assign(CONFIG, options);

    // Validate timeout values
    if (CONFIG.WARNING_BEFORE_LOGOUT >= CONFIG.INACTIVITY_TIMEOUT) {
      console.error('[AutoLogout] WARNING_BEFORE_LOGOUT must be less than INACTIVITY_TIMEOUT');
      return;
    }

    isLoggingOut = false;
    isInitialized = true;

    // Listen for cross-tab logout
    window.addEventListener('storage', handleStorageEvent);

    // Attach activity listeners
    attachEventListeners();

    // Start initial timer
    resetTimer();

    console.info(
      `[AutoLogout] Initialized. Timeout: ${CONFIG.INACTIVITY_TIMEOUT / 60000} minutes. ` +
      `Warning at: ${CONFIG.WARNING_BEFORE_LOGOUT / 1000} seconds before logout.`
    );
  };

  /**
   * Destroy the auto-logout module and clean up all resources
   */
  const destroy = () => {
    clearAllTimers();
    detachEventListeners();
    window.removeEventListener('storage', handleStorageEvent);
    removeWarningDialog();
    isInitialized = false;
    console.info('[AutoLogout] Destroyed and cleaned up.');
  };

  /**
   * Get current module status (useful for debugging)
   * @returns {Object} Current status
   */
  const getStatus = () => ({
    isInitialized,
    isLoggingOut,
    lastActivityTime: new Date(lastActivityTime).toISOString(),
    secondsSinceLastActivity: Math.floor((Date.now() - lastActivityTime) / 1000),
    timeoutMinutes: CONFIG.INACTIVITY_TIMEOUT / 60000,
    warningSeconds: CONFIG.WARNING_BEFORE_LOGOUT / 1000,
  });

  // Public API
  return {
    init,
    destroy,
    resetTimer,
    performLogout,
    getStatus,
    broadcastLogout,
  };
})();


// ─── Usage Example ───────────────────────────────────────────────────────────

// Initialize with default settings (15 min timeout)
AutoLogout.init();

// Or initialize with custom settings

  INACTIVITY_TIMEOUT: 15 * 60 * 1000,   // 15 minutes
  WARNING_BEFORE_LOGOUT: 60 * 1000,      // Warn 60 seconds before
  LOGIN_PAGE_URL: '/auth/login',
  LOGOUT_API_URL: '/api/v1/auth/logout',
  SESSION_TOKEN_KEY: 'bank_auth_token',
});
```
