struct ImageInfo {

// Mock function to simulate reading JPEG image info
// In real implementation, this would use libjpeg functions
ImageInfo readJpegInfo(const std::string& filename) {
    ImageInfo info;
    
    // Simulate reading a JPEG file
    std::ifstream file(filename, std::ios::binary | std::ios::ate);
    if (!file.is_open()) {
        // Return mock data for demonstration
        info.width = 1920;
        info.height = 1080;
        info.channels = 3;
        info.format = "JPEG";
        info.file_size = 0;
        return info;
    }
    
    info.file_size = file.tellg();
    file.seekg(0, std::ios::beg);
    
    // Read JPEG header (simplified)
    unsigned char header[4];
    file.read(reinterpret_cast<char*>(header), 4);
    
    // Mock dimensions (in real code, parse JPEG SOF marker)
    info.width = 1920;
    info.height = 1080;
    info.channels = 3;
    info.format = "JPEG";
    
    return info;
}

// Mock function to simulate image processing operations
std::string processImageOperation(const std::string& operation, const ImageInfo& info) {
    std::ostringstream result;
    
    if (operation == "resize") {
        result << "Resized image from " << info.width << "x" << info.height 
               << " to 800x600 using libjpeg";
    } else if (operation == "compress") {
        result << "Compressed JPEG image with quality 85, "
               << "estimated size reduction: 40%";
    } else if (operation == "convert") {
        result << "Converting " << info.channels << "-channel image "
               << "from JPEG to PNG format";
    } else if (operation == "analyze") {
        result << "Image analysis: " << info.width << "x" << info.height 
               << " pixels, " << info.channels << " channels, "
               << info.file_size << " bytes";
    }
    
    return result.str();
}

// Function to get AI assistance for image processing using Claude
std::string getImageProcessingAdvice(const std::string& task, const ImageInfo& imageInfo) {
    // Initialize Anthropic client
    anthropic::Client client;
    
    // Build context about the image
    std::ostringstream imageContext;
    imageContext << "Image Information:\n"
                 << "- Format: " << imageInfo.format << "\n"
                 << "- Dimensions: " << imageInfo.width << "x" << imageInfo.height << "\n"
                 << "- Channels: " << imageInfo.channels << "\n"
                 << "- File size: " << imageInfo.file_size << " bytes\n\n"
                 << "Task: " << task;
    
    // Create message request
    anthropic::MessageRequest request;
    request.model = "claude-opus-4-5";
    request.max_tokens = 1024;
    
    anthropic::Message userMessage;
    userMessage.role = "user";
    userMessage.content = "I'm working with libjpeg for image processing in C++. " + 
                          imageContext.str() + 
                          "\n\nPlease provide specific C++ code using libjpeg API to accomplish this task, "
                          "including error handling and best practices.";
    
    request.messages.push_back(userMessage);
    
    // Make API call
    auto response = client.messages().create(request);
    
    if (response.content.empty()) {
        return "Error: No response from Claude API";
    }
    
    return response.content[0].text;
}

// Main image processing function that combines libjpeg and Claude AI
void processImageWithAI(const std::string& imagePath, const std::string& operation) {
    std::cout << "=== Image Processing with libjpeg and Claude AI ===" << std::endl;
    std::cout << "Image: " << imagePath << std::endl;
    std::cout << "Operation: " << operation << std::endl;
    std::cout << std::endl;
    
    // Step 1: Read image information (using libjpeg)
    std::cout << "Step 1: Reading image information with libjpeg..." << std::endl;
    ImageInfo imageInfo = readJpegInfo(imagePath);
    
    std::cout << "Image loaded successfully:" << std::endl;
    std::cout << "  - Format: " << imageInfo.format << std::endl;
    std::cout << "  - Dimensions: " << imageInfo.width << "x" << imageInfo.height << std::endl;
    std::cout << "  - Channels: " << imageInfo.channels << std::endl;
    std::cout << "  - File size: " << imageInfo.file_size << " bytes" << std::endl;
    std::cout << std::endl;
    
    // Step 2: Perform the requested operation
    std::cout << "Step 2: Performing image operation..." << std::endl;
    std::string operationResult = processImageOperation(operation, imageInfo);
    std::cout << "Operation result: " << operationResult << std::endl;
    std::cout << std::endl;
    
    // Step 3: Get AI assistance for the specific task
    std::cout << "Step 3: Getting AI assistance from Claude..." << std::endl;
    std::cout << "Querying Claude API for " << operation << " implementation advice..." << std::endl;
    std::cout << std::endl;
    
    std::string aiAdvice = getImageProcessingAdvice(
        "How to " + operation + " a JPEG image using libjpeg in C++?",
        imageInfo
    );
    
    std::cout << "=== Claude's Implementation Advice ===" << std::endl;
    std::cout << aiAdvice << std::endl;
    std::cout << std::endl;
}

// Demonstrate multiple image processing scenarios
void demonstrateImageProcessing() {
    std::cout << "=== Demonstrating Image Processing Capabilities ===" << std::endl;
