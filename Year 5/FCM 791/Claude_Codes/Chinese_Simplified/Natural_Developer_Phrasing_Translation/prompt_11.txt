import anthropic
import tempfile
import os
from lxml import etree

def create_sample_xml():
    """Create a sample XML file for demonstration"""
    xml_content = """<?xml version="1.0" encoding="UTF-8"?>
<library>
    <book id="1">
        <title>Python Programming</title>
        <author>John Smith</author>
        <year>2023</year>
    </book>
    <book id="2">
        <title>Data Science with Python</title>
        <author>Jane Doe</author>
        <year>2022</year>
    </book>
    <category name="Technology">
        <title>Computer Science Books</title>
        <book id="3">
            <title>Machine Learning Fundamentals</title>
            <author>Bob Johnson</author>
            <year>2024</year>
        </book>
    </category>
    <metadata>
        <title>Library Catalog</title>
        <version>1.0</version>
    </metadata>
</library>"""
    
    # Create a temporary file
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as f:
        f.write(xml_content)
        return f.name

def parse_xml_and_extract_titles(xml_file_path: str) -> list[str]:
    """Parse XML file and extract all title tags using lxml"""
    try:
        # Parse the XML file
        tree = etree.parse(xml_file_path)
        root = tree.getroot()
        
        # Find all title elements
        titles = root.findall('.//title')
        
        # Extract text content from each title
        title_contents = []
        for title in titles:
            if title.text:
                title_contents.append(title.text.strip())
        
        return title_contents
    except etree.XMLSyntaxError as e:
        raise ValueError(f"Invalid XML file: {e}")

def analyze_xml_with_claude(xml_file_path: str, titles: list[str]) -> str:
    """Use Claude to analyze the extracted titles"""
    client = anthropic.Anthropic()
    
    # Read the XML content
    with open(xml_file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    # Prepare the message for Claude
    titles_str = '\n'.join([f"- {title}" for title in titles])
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I have parsed an XML file and extracted the following titles:

{titles_str}

Here is the original XML content:
```xml
{xml_content}
```

Please analyze these titles and provide:
1. A summary of what the XML document appears to be about
2. The hierarchy/structure of the titles if applicable
3. Any interesting observations about the content"""
            }
        ]
    )
    
    return message.content[0].text

def process_xml_file(xml_file_path: str) -> dict:
    """Main function to process XML file and extract titles"""
    print(f"Processing XML file: {xml_file_path}")
    print("-" * 50)
    
    # Parse XML and extract titles
    titles = parse_xml_and_extract_titles(xml_file_path)
    
    print(f"Found {len(titles)} title(s):")
    for i, title in enumerate(titles, 1):
        print(f"  {i}. {title}")
    
    print("\n" + "-" * 50)
    print("Analyzing with Claude AI...")
    print("-" * 50)
    
    # Analyze with Claude
    analysis = analyze_xml_with_claude(xml_file_path, titles)
    print("\nClaude's Analysis:")
    print(analysis)
    
    return {
        "file_path": xml_file_path,
        "titles": titles,
        "analysis": analysis
    }

def main():
    """Main entry point"""
    import sys
    
    if len(sys.argv) > 1:
        # Use provided XML file
        xml_file_path = sys.argv[1]
        if not os.path.exists(xml_file_path):
            print(f"Error: File '{xml_file_path}' not found")
            sys.exit(1)
        
        result = process_xml_file(xml_file_path)
    else:
        # Create and use a sample XML file for demonstration
        print("No XML file provided. Creating a sample XML file for demonstration...")
        print("=" * 50)
        
        sample_file = create_sample_xml()
        try:
            result = process_xml_file(sample_file)
        finally:
            # Clean up temporary file
            os.unlink(sample_file)
    
    print("\n" + "=" * 50)
    print("Processing complete!")
    return result

if __name__ == "__main__":
    main()
