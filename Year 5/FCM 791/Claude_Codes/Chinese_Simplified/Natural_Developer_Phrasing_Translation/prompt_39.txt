import anthropic
import socket
import threading
import time
import struct
import json
import random

# XOR encryption key
XOR_KEY = 0x42

def xor_encrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Simple XOR encryption/decryption"""
    return bytes([b ^ key for b in data])

def xor_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Simple XOR decryption (same as encryption for XOR)"""
    return xor_encrypt(data, key)

# Custom protocol constants
PROTOCOL_MAGIC = b'\xDE\xAD'
PROTOCOL_VERSION = 1
MSG_TYPE_REQUEST = 0x01
MSG_TYPE_RESPONSE = 0x02
MSG_TYPE_ERROR = 0xFF

def create_protocol_message(msg_type: int, payload: str) -> bytes:
    """
    Create a custom protocol message:
    [MAGIC(2)] [VERSION(1)] [MSG_TYPE(1)] [LENGTH(4)] [ENCRYPTED_PAYLOAD(N)]
    """
    payload_bytes = payload.encode('utf-8')
    encrypted_payload = xor_encrypt(payload_bytes)
    
    header = struct.pack('>2sBBI', 
                         PROTOCOL_MAGIC,
                         PROTOCOL_VERSION,
                         msg_type,
                         len(encrypted_payload))
    
    return header + encrypted_payload

def parse_protocol_message(data: bytes) -> dict:
    """Parse a custom protocol message"""
    if len(data) < 8:
        raise ValueError("Message too short")
    
    magic, version, msg_type, length = struct.unpack('>2sBBI', data[:8])
    
    if magic != PROTOCOL_MAGIC:
        raise ValueError(f"Invalid magic bytes: {magic.hex()}")
    
    if version != PROTOCOL_VERSION:
        raise ValueError(f"Unsupported protocol version: {version}")
    
    if len(data) < 8 + length:
        raise ValueError("Incomplete message")
    
    encrypted_payload = data[8:8+length]
    payload = xor_decrypt(encrypted_payload).decode('utf-8')
    
    return {
        'version': version,
        'msg_type': msg_type,
        'payload': payload
    }

class LegacyServer:
    """Simulated legacy server with custom socket protocol"""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        
    def start(self):
        """Start the legacy server"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")
        
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections"""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Legacy server: New connection from {address}")
                
                client_thread = threading.Thread(
                    target=self._handle_client,
                    args=(client_socket, address)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Legacy server error: {e}")
                    
    def _handle_client(self, client_socket: socket.socket, address):
        """Handle individual client connection"""
        try:
            data = client_socket.recv(4096)
            if data:
                message = parse_protocol_message(data)
                print(f"Legacy server received: {message['payload']}")
                
                request = json.loads(message['payload'])
                response_data = self._process_request(request)
                
                response_msg = create_protocol_message(
                    MSG_TYPE_RESPONSE,
                    json.dumps(response_data)
                )
                client_socket.send(response_msg)
                
        except Exception as e:
            error_msg = create_protocol_message(
                MSG_TYPE_ERROR,
                json.dumps({'error': str(e)})
            )
            try:
                client_socket.send(error_msg)
            except:
                pass
        finally:
            client_socket.close()
            
    def _process_request(self, request: dict) -> dict:
        """Process incoming request and return response"""
        command = request.get('command', '')
        
        if command == 'get_status':
            return {
                'status': 'operational',
                'uptime': random.randint(1000, 9999),
                'version': '1.0.0-legacy',
                'timestamp': int(time.time())
            }
        elif command == 'get_data':
            data_type = request.get('type', 'default')
            return {
                'data_type': data_type,
                'values': [random.randint(1, 100) for _ in range(5)],
                'timestamp': int(time.time())
            }
        elif command == 'ping':
            return {'pong': True, 'timestamp': int(time.time())}
        else:
            return {'error': f'Unknown command: {command}'}
            
    def stop(self):
        """Stop the legacy server"""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("Legacy server stopped")

class LegacyServerClient:
    """Client for communicating with legacy server"""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        
    def send_command(self, command: str, **kwargs) -> dict:
        """Send a command to the legacy server"""
        request_data = {'command': command, **kwargs}
        
        client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        client_socket.settimeout(5.0)
        
        try:
            client_socket.connect((self.host, self.port))
            
            message = create_protocol_message(
                MSG_TYPE_REQUEST,
                json.dumps(request_data)
            )
            client_socket.send(message)
            
            response_data = client_socket.recv(4096)
            response = parse_protocol_message(response_data)
            
            return json.loads(response['payload'])
            
        finally:
            client_socket.close()

def create_legacy_server_tools():
    """Create Anthropic tools for legacy server communication"""
    return [
        {
            "name": "check_server_status",
            "description": "Check the status of the legacy server",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        },
        {
            "name": "get_server_data",
            "description": "Retrieve data from the legacy server",
            "input_schema": {
                "type": "object",
                "properties": {
                    "data_type": {
                        "type": "string",
                        "description": "Type of data to retrieve (e.g., 'sensor', 'metrics', 'logs')"
                    }
                },
                "required": ["data_type"]
            }
        },
        {
            "name": "ping_server",
            "description": "Ping the legacy server to check connectivity",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        }
    ]

def process_tool_call(tool_name: str, tool_input: dict, client: LegacyServerClient) -> str:
    """Process tool calls and interact with legacy server"""
    try:
        if tool_name == "check_server_status":
            result = client.send_command("get_status")
            return json.dumps(result, indent=2)
            
        elif tool_name == "get_server_data":
            data_type = tool_input.get("data_type", "default")
            result = client.send_command("get_data", type=data_type)
            return json.dumps(result, indent=2)
            
        elif tool_name == "ping_server":
            result = client.send_command("ping")
            return json.dumps(result, indent=2)
            
        else:
            return json.dumps({"error": f"Unknown tool: {tool_name}"})
            
    except Exception as e:
        return json.dumps({"error": str(e)})

def main():
    """Main function demonstrating Claude interacting with legacy server"""
    
    # Start legacy server
    server = LegacyServer()
    server.start()
    time.sleep(0.5)  # Wait for server to start
    
    # Create client
    client = LegacyServerClient()
    
    # Initialize Anthropic client
    anthropic_client = anthropic.Anthropic()
    
    print("\n" + "="*60)
    print("Claude interacting with Legacy Server via Custom Socket Protocol")
    print("="*60 + "\n")
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": "Please check the legacy server status, ping it to verify connectivity, and then retrieve both sensor and metrics data. Provide a comprehensive report of what you find."
        }
    ]
    
    tools = create_legacy_server_tools()
    
    # Agentic loop
    while True:
        response = anthropic_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude's response (stop_reason: {response.stop_reason}):")
        
        # Process response content
        for block in response.content:
            if hasattr(block, 'text'):
                print(f"\nClaude: {block.text}")
            elif block.type == "tool_use":
                print(f"\nClaude is using tool: {block.name}")
                print(f"Tool input: {json.dumps(block.input, indent=2)}")
        
        # Check if we should stop
        if response.stop_reason == "end_turn":
            print("\nClaude has finished the task.")
            break
            
        # Process tool calls if any
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process each tool call
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nExecuting tool: {block.name}")
                    result = process_tool_call(block.name, block.input, client)
                    print(f"Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            # No more tool calls, exit loop
            break
    
    # Stop the server
    server.stop()
    
    print("\n" + "="*60)
    print("Demo completed successfully!")
    print("="*60)

if __name__ == "__main__":
    main()
