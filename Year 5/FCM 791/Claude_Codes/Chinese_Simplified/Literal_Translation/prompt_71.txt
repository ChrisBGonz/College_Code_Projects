
```python
import pickle
import os
import time
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def save_game_state(game_state: dict, filename: str = "game_state.pkl") -> bool:
    """Save game state to a pickle file."""
    try:
        with open(filename, 'wb') as f:
            pickle.dump(game_state, f)
        print(f"Game state saved to {filename}")
        return True
    except Exception as e:
        print(f"Error saving game state: {e}")
        return False

def load_game_state(filename: str = "game_state.pkl") -> dict | None:
    """Load game state from a pickle file."""
    if not os.path.exists(filename):
        print(f"No saved game state found at {filename}")
        return None
    
    try:
        with open(filename, 'rb') as f:
            game_state = pickle.load(f)
        print(f"Game state loaded from {filename}")
        return game_state
    except Exception as e:
        print(f"Error loading game state: {e}")
        return None

def create_initial_game_state() -> dict:
    """Create an initial game state."""
    return {
        "player": {
            "name": "Hero",
            "level": 1,
            "health": 100,
            "max_health": 100,
            "experience": 0,
            "gold": 50,
            "inventory": ["sword", "shield", "health_potion"],
            "skills": ["basic_attack", "defend"],
            "position": {"x": 0, "y": 0, "map": "starting_village"}
        },
        "world": {
            "current_map": "starting_village",
            "discovered_areas": ["starting_village"],
            "completed_quests": [],
            "active_quests": ["find_the_lost_sword"],
            "time_played": 0,
            "day": 1
        },
        "settings": {
            "difficulty": "normal",
            "sound_enabled": True,
            "music_volume": 0.7,
            "sfx_volume": 0.8
        },
        "statistics": {
            "enemies_defeated": 0,
            "items_collected": 3,
            "deaths": 0,
            "total_damage_dealt": 0,
            "total_damage_received": 0
        },
        "save_timestamp": time.strftime("%Y-%m-%d %H:%M:%S")
    }

def chat_with_claude(user_message: str, game_state: dict) -> tuple[str, dict]:
    """Send a message to Claude and get a response, potentially updating game state."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    system_prompt = f"""You are an AI game master for a text-based RPG game. You help manage and update the game state based on player actions.

Current Game State:
{pickle.dumps(game_state).hex()[:100]}... (serialized)

Player Info:
- Name: {game_state['player']['name']}
- Level: {game_state['player']['level']}
- Health: {game_state['player']['health']}/{game_state['player']['max_health']}
- Gold: {game_state['player']['gold']}
- Location: {game_state['player']['position']['map']}
- Inventory: {', '.join(game_state['player']['inventory'])}
- Active Quests: {', '.join(game_state['world']['active_quests'])}

You can suggest updates to the game state based on player actions. When suggesting state changes, format them clearly.
Available commands the player can use:
- 'save' - save the game
- 'load' - load a saved game
- 'status' - show current status
- 'inventory' - show inventory
- 'explore [location]' - explore a new area
- 'fight [enemy]' - engage in combat
- 'buy [item]' - purchase an item
- 'use [item]' - use an item from inventory
- 'quit' - exit the game

Respond in a narrative style and suggest specific game state updates when appropriate.
If the player performs an action that should change the game state, describe what changes and provide the updated values in a structured format like:
UPDATE_STATE: {{"player.health": 80, "statistics.enemies_defeated": 1}}"""
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    # Parse any state updates from Claude's response
    if "UPDATE_STATE:" in assistant_message:
        try:
            import json
            update_start = assistant_message.find("UPDATE_STATE:") + len("UPDATE_STATE:")
            update_end = assistant_message.find("\n", update_start)
            if update_end == -1:
                update_end = len(assistant_message)
            
            update_str = assistant_message[update_start:update_end].strip()
            updates = json.loads(update_str)
            
            # Apply updates to game state
            for key_path, value in updates.items():
                keys = key_path.split('.')
                obj = game_state
                for key in keys[:-1]:
                    obj = obj[key]
                obj[keys[-1]] = value
            
            game_state['save_timestamp'] = time.strftime("%Y-%m-%d %H:%M:%S")
            print("\n[Game state updated]")
        except Exception as e:
            print(f"\n[Could not parse state update: {e}]")
    
    return assistant_message, game_state

def display_game_status(game_state: dict):
    """Display current game status."""
    player = game_state['player']
    world = game_state['world']
    stats = game_state['statistics']
    
    print("\n" + "="*50)
    print("GAME STATUS")
    print("="*50)
    print(f"Player: {player['name']} (Level {player['level']})")
    print(f"Health: {player['health']}/{player['max_health']}")
    print(f"Gold: {player['gold']}")
    print(f"Location: {player['position']['map']}")
    print(f"Active Quests: {', '.join(world['active_quests']) if world['active_quests'] else 'None'}")
    print(f"Enemies Defeated: {stats['enemies_defeated']}")
    print(f"Last Saved: {game_state['save_timestamp']}")
    print("="*50 + "\n")

def main():
    """Main game loop."""
    print("="*60)
    print("WELCOME TO THE AI-POWERED RPG ADVENTURE")
    print("="*60)
    print("\nThis game uses Claude AI as your game master!")
    print("Type 'help' for available commands.\n")
    
    # Try to load existing game state
    game_state = load_game_state()
    
    if game_state is None:
        print("Starting a new game...")
        game_state = create_initial_game_state()
        print(f"Welcome, {game_state['player']['name']}! Your adventure begins in {game_state['player']['position']['map']}.")
    else:
        print(f"Welcome back, {game_state['player']['name']}!")
        print(f"Last played: {game_state['save_timestamp']}")
    
    # Initial game description
    initial_response, game_state = chat_with_claude(
        "I just started/loaded the game. Please describe my current situation and what I should do.",
        game_state
    )
    print(f"\nGame Master: {initial_response}\n")
    
    # Main game loop
    while True:
        try:
            user_input = input("Your action: ").strip()
            
            if not user_input:
                continue
            
            # Handle special commands
            if user_input.lower() == 'quit':
                save_choice = input("Save game before quitting? (y/n): ").lower()
                if save_choice == 'y':
                    save_game_state(game_state)
                print("Thanks for playing! Goodbye!")
                break
            
            elif user_input.lower() == 'save':
                save_game_state(game_state)
                print("Game saved successfully!")
                continue
            
            elif user_input.lower() == 'load':
                loaded_state = load_game_state()
                if loaded_state:
                    game_state = loaded_state
                    print("Game loaded successfully!")
                    display_game_status(game_state)
                continue
            
            elif user_input.lower() == 'status':
                display_game_status(game_state)
                continue
            
            elif user_input.lower() == 'inventory':
                print(f"\nInventory: {', '.join(game_state['player']['inventory'])}")
                print(f"Gold: {game_state['player']['gold']}\n")
                continue
            
            elif user_input.lower() == 'help':
                print("\nAvailable commands:")
                print("  save - Save your game")
                print("  load - Load saved game")
                print("  status - Show game status")
                print("  inventory - Show inventory")
                print("  explore [location] - Explore an area")
                print("  fight [enemy] - Fight an enemy")
                print("  buy [item] - Buy an item")
                print("  use [item] - Use an item")
                print("  quit - Exit game")
                print("  Or just type any action!\n")
                continue
            
            # Send to Claude for processing
            print("\nGame Master is thinking...")
            response, game_state = chat_with_claude(user_input, game_state)
            print(f"\nGame Master: {response}\n")
            
            # Auto-save every 5 actions
            if len(conversation_history) % 10 == 0:
                print("[Auto-saving game...]")
                save_game_state(game_state)
        
        except KeyboardInterrupt:
            print("\n\nGame interrupted. Saving...")
            save_game_state(game_state)
            print("Game saved. Goodbye!")
            break
        except Exception as e:
            print(f"An error occurred: {e}")
            print("Please try again.")

if __name__ == "__main__":
    main()
```
