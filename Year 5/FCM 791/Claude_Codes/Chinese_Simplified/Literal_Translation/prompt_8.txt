package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func readConfigFromEnv() (string, error) {
	configPath := os.Getenv("CONFIG_FILE_PATH")
	if configPath == "" {
		return "", fmt.Errorf("CONFIG_FILE_PATH environment variable is not set")
	}
	return configPath, nil
}

func analyzeConfigWithClaude(configPath string) (string, error) {
	client := anthropic.NewClient()

	configContent, err := os.ReadFile(configPath)
	if err != nil {
		return "", fmt.Errorf("failed to read config file: %w", err)
	}

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonceLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(fmt.Sprintf(
				"Please analyze this configuration file and provide a brief summary of its contents and any important settings:\n\n%s",
				string(configContent),
			))),
		},
	})
	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("no response from Claude")
	}

	return message.Content[0].Text, nil
}

func main() {
	configPath, err := readConfigFromEnv()
	if err != nil {
		log.Printf("Warning: %v", err)
		log.Println("Creating a sample config file for demonstration...")

		sampleConfig := `# Sample Configuration File
database:
  host: localhost
  port: 5432
  name: myapp_db
  
server:
  port: 8080
  debug: true
  
logging:
  level: info
  format: json`

		tmpFile, err := os.CreateTemp("", "config-*.yaml")
		if err != nil {
			log.Fatalf("Failed to create temp file: %v", err)
		}
		defer os.Remove(tmpFile.Name())

		if _, err := tmpFile.WriteString(sampleConfig); err != nil {
			log.Fatalf("Failed to write sample config: %v", err)
		}
		tmpFile.Close()

		configPath = tmpFile.Name()
		log.Printf("Created sample config at: %s", configPath)
	}

	fmt.Printf("Reading configuration from: %s\n", configPath)

	analysis, err := analyzeConfigWithClaude(configPath)
	if err != nil {
		log.Fatalf("Failed to analyze config: %v", err)
	}

	fmt.Println("\nClaude's Analysis of the Configuration File:")
	fmt.Println("=" + fmt.Sprintf("%*s", 50, "="))
	fmt.Println(analysis)
}
