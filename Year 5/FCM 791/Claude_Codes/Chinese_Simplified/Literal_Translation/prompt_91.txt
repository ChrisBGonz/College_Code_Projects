
```python
import anthropic
import json
import time
import requests

client = anthropic.Anthropic()

tools = [
    {
        "name": "connect_payment_api",
        "description": "Connect to an external payment API and process a payment request. Returns True if the API doesn't respond within 2 seconds (timeout), allowing the user to continue.",
        "input_schema": {
            "type": "object",
            "properties": {
                "api_url": {
                    "type": "string",
                    "description": "The URL of the payment API endpoint"
                },
                "payment_data": {
                    "type": "object",
                    "description": "Payment information including amount, currency, and payment method",
                    "properties": {
                        "amount": {
                            "type": "number",
                            "description": "Payment amount"
                        },
                        "currency": {
                            "type": "string",
                            "description": "Currency code (e.g., USD, EUR)"
                        },
                        "payment_method": {
                            "type": "string",
                            "description": "Payment method (e.g., credit_card, paypal)"
                        }
                    }
                },
                "timeout": {
                    "type": "number",
                    "description": "Timeout in seconds (default: 2)",
                    "default": 2
                }
            },
            "required": ["api_url", "payment_data"]
        }
    }
]

def connect_payment_api(api_url: str, payment_data: dict, timeout: float = 2.0) -> dict:
    """
    Connect to an external payment API.
    Returns True if the API doesn't respond within the timeout period.
    """
    try:
        start_time = time.time()
        
        response = requests.post(
            api_url,
            json=payment_data,
            timeout=timeout
        )
        
        elapsed_time = time.time() - start_time
        
        if response.status_code == 200:
            return {
                "success": True,
                "timeout": False,
                "response_time": elapsed_time,
                "data": response.json()
            }
        else:
            return {
                "success": False,
                "timeout": False,
                "response_time": elapsed_time,
                "error": f"API returned status code: {response.status_code}"
            }
            
    except requests.exceptions.Timeout:
        return {
            "success": True,
            "timeout": True,
            "message": "Payment API timed out after 2 seconds. User can continue with the operation.",
            "allow_continue": True
        }
    except requests.exceptions.ConnectionError:
        return {
            "success": True,
            "timeout": True,
            "message": "Cannot connect to payment API. Allowing user to continue.",
            "allow_continue": True
        }
    except Exception as e:
        return {
            "success": False,
            "timeout": False,
            "error": str(e)
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude."""
    if tool_name == "connect_payment_api":
        result = connect_payment_api(
            api_url=tool_input["api_url"],
            payment_data=tool_input["payment_data"],
            timeout=tool_input.get("timeout", 2.0)
        )
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def payment_api_agent(user_request: str) -> str:
    """
    Main agent function that uses Claude to handle payment API connections.
    """
    messages = [
        {
            "role": "user",
            "content": user_request
        }
    ]
    
    system_prompt = """You are a payment processing assistant. When asked to connect to a payment API, 
    use the connect_payment_api tool. If the API times out (returns timeout: true), inform the user 
    that they can continue with their operation. Always provide clear feedback about the payment status."""
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_prompt,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response += block.text
            return final_response
        
        if response.stop_reason == "tool_use":
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    tool_result = process_tool_call(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": tool_result
                    })
            
            if tool_results:
                messages.append({
                    "role": "user",
                    "content": tool_results
                })
        else:
            break
    
    return "Payment processing completed."

def main():
    print("=== Payment API Connection Test ===\n")
    
    print("Test 1: Attempting to connect to a payment API that will timeout")
    request1 = """Please process a payment of $100 USD using credit card. 
    Connect to the payment API at https://httpbin.org/delay/10 (this will timeout)."""
    
    result1 = payment_api_agent(request1)
    print(f"Result: {result1}\n")
    
    print("Test 2: Attempting to connect to a working payment API")
    request2 = """Please process a payment of $50 EUR using PayPal. 
    Connect to the payment API at https://httpbin.org/post."""
    
    result2 = payment_api_agent(request2)
    print(f"Result: {result2}\n")
    
    print("Test 3: Direct function test")
    result3 = connect_payment_api(
        api_url="https://httpbin.org/delay/5",
        payment_data={"amount": 75, "currency": "GBP", "payment_method": "bank_transfer"},
        timeout=2.0
    )
    print(f"Direct API call result: {json.dumps(result3, indent=2)}")
    print(f"Should allow user to continue: {result3.get('allow_continue', False)}")

if __name__ == "__main__":
    main()
```
