const http = require("http");
const url = require("url");

const client = new Anthropic();

async function processUserProfileRequest(userId) {
  const conversationHistory = [];

  // Initial message to Claude
  const initialMessage = `I need to process a user profile download request for user_id: ${userId}. 
  Please help me:
  1. First, validate if this user_id exists in our system
  2. If valid, retrieve and format the profile data
  3. Prepare it for JSON download
  
  Available users: ${Object.keys(userDatabase).join(", ")}`;

  conversationHistory.push({
    role: "user",
    content: initialMessage,
  });

  // First API call - validate and check user
  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful API assistant that processes user profile requests. You help validate user IDs and prepare profile data for download. Be concise and structured in your responses.",
    messages: conversationHistory,
  });

  const firstAssistantMessage = firstResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: firstAssistantMessage,
  });

  // Check if user exists
  const userData = userDatabase[userId];

  if (!userData) {
    // Second turn - handle invalid user
    conversationHistory.push({
      role: "user",
      content: `The user_id "${userId}" was not found in our database. Please provide an appropriate error message for the API response.`,
    });

    const errorResponse = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 512,
      system:
        "You are a helpful API assistant that processes user profile requests. You help validate user IDs and prepare profile data for download. Be concise and structured in your responses.",
      messages: conversationHistory,
    });

    return {
      success: false,
      error: `User not found: ${userId}`,
      message: errorResponse.content[0].text,
    };
  }

  // Second turn - process valid user data
  conversationHistory.push({
    role: "user",
    content: `User found! Here is the raw profile data: ${JSON.stringify(userData, null, 2)}
    
    Please:
    1. Confirm the data looks complete and valid
    2. Suggest any additional metadata we should include in the download (like download timestamp, data version, etc.)
    3. Confirm the data is ready for JSON download`,
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful API assistant that processes user profile requests. You help validate user IDs and prepare profile data for download. Be concise and structured in your responses.",
    messages: conversationHistory,
  });

  const secondAssistantMessage = secondResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: secondAssistantMessage,
  });

  // Third turn - finalize the download package
  conversationHistory.push({
    role: "user",
    content: `Perfect. Now create the final download package with the user profile data and appropriate metadata. Format it as a complete JSON structure ready for download.`,
  });

  const finalResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 2048,
    system:
      "You are a helpful API assistant that processes user profile requests. You help validate user IDs and prepare profile data for download. Be concise and structured in your responses.",
    messages: conversationHistory,
  });

  // Prepare the actual download data
  const downloadPackage = {
    metadata: {
      downloadTimestamp: new Date().toISOString(),
      dataVersion: "1.0",
      requestedUserId: userId,
      dataFormat: "JSON",
      apiVersion: "v1",
    },
    profile: userData,
    processingNotes: finalResponse.content[0].text,
  };

  return {
    success: true,
    data: downloadPackage,
    filename: `user_profile_${userId}_${Date.now()}.json`,
  };
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  // Handle profile download endpoint
  if (pathname === "/api/download-profile" && req.method === "GET") {
    const userId = query.user_id;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing required parameter: user_id",
          usage: "/api/download-profile?user_id=<user_id>",
        })
      );
      return;
    }

    try {
      console.log(`Processing profile download request for user: ${userId}`);
      const result = await processUserProfileRequest(userId);

      if (!result.success) {
        res.writeHead(404, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: result.error,
            details: result.message,
          })
        );
        return;
      }

      // Set headers for file download
      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="${result.filename}"`,
        "X-Download-Filename": result.filename,
      });

      res.end(JSON.stringify(result.data, null, 2));
    } catch (error) {
      console.error("Error processing request:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal server error",
          message: error.message,
        })
      );
    }
  }
  // Handle list users endpoint (for testing)
  else if (pathname === "/api/users" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        availableUsers: Object.keys(userDatabase),
        usage: "GET /api/download-profile?user_id=<user_id>",
      })
    );
  }
  // Handle root endpoint
  else if (pathname === "/" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "User Profile Download API",
        endpoints: {
          downloadProfile: "GET /api/download-profile?user_id=<user_id>",
          listUsers: "GET /api/users",
        },
        description:
          "Download user profile data as JSON file based on user_id query parameter",
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Endpoint not found" }));
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);

  console.log(`\nExample usage:`);

  );
  console.log(
    `  curl "http://localhost:${PORT}/api/download-profile?user_id=user_002" -o profile.json`
  );
});

module.exports = server;
