require 'openssl'
require 'base64'

def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  [private_key, public_key]
end

def sign_with_rsa_pss(data, private_key, digest_algorithm = 'SHA256', salt_length = :digest)
  raise ArgumentError, 'Data cannot be nil or empty' if data.nil? || data.empty?
  raise ArgumentError, 'Private key cannot be nil' if private_key.nil?
  
  digest = OpenSSL::Digest.new(digest_algorithm)
  
  salt_len = case salt_length
             when :digest
               digest.digest_length
             when :max
               OpenSSL::PKey::RSA::PSS_SALTLEN_MAX_SIGN
             when Integer
               salt_length
             else
               digest.digest_length
             end
  
  signature = private_key.sign_pss(digest_algorithm, data, salt_length: salt_len, mgf1_hash: digest_algorithm)
  
  Base64.strict_encode64(signature)
end

def verify_rsa_pss_signature(data, signature_base64, public_key, digest_algorithm = 'SHA256', salt_length = :digest)
  raise ArgumentError, 'Data cannot be nil or empty' if data.nil? || data.empty?
  raise ArgumentError, 'Signature cannot be nil or empty' if signature_base64.nil? || signature_base64.empty?
  raise ArgumentError, 'Public key cannot be nil' if public_key.nil?
  
  digest = OpenSSL::Digest.new(digest_algorithm)
  
  salt_len = case salt_length
             when :digest
               digest.digest_length
             when :max
               OpenSSL::PKey::RSA::PSS_SALTLEN_MAX_SIGN
             when Integer
               salt_length
             else
               OpenSSL::PKey::RSA::PSS_SALTLEN_AUTO
             end
  
  signature = Base64.strict_decode64(signature_base64)
  
  public_key.verify_pss(digest_algorithm, signature, data, salt_length: salt_len, mgf1_hash: digest_algorithm)
rescue OpenSSL::PKey::PKeyError => e
  puts "Signature verification failed: #{e.message}"
  false
rescue ArgumentError => e
  puts "Invalid argument: #{e.message}"
  false
end

def sign_payload(payload, private_key_pem, digest_algorithm = 'SHA256')
  private_key = OpenSSL::PKey::RSA.new(private_key_pem)
  
  data = payload.is_a?(String) ? payload : payload.to_json
  
  signature = sign_with_rsa_pss(data, private_key, digest_algorithm)
  
  {
    payload: data,
    signature: signature,
    algorithm: "RSA-PSS-#{digest_algorithm}",
    timestamp: Time.now.utc.iso8601
  }
end

def verify_payload(signed_data, public_key_pem, digest_algorithm = 'SHA256')
  public_key = OpenSSL::PKey::RSA.new(public_key_pem)
  
  payload = signed_data[:payload] || signed_data['payload']
  signature = signed_data[:signature] || signed_data['signature']
  
  raise ArgumentError, 'Missing payload or signature' if payload.nil? || signature.nil?
  
  verify_rsa_pss_signature(payload, signature, public_key, digest_algorithm)
end

if __FILE__ == $0
  puts "=== RSA-PSS Signing Demo ==="
  puts
  
  puts "Generating RSA key pair (2048 bits)..."
  private_key, public_key = generate_rsa_key_pair(2048)
  puts "Key pair generated successfully!"
  puts
  
  payload = "This is a sensitive data payload that needs to be authenticated."
  puts "Original payload: #{payload}"
  puts
  
  puts "Signing payload with RSA-PSS..."
  signature_base64 = sign_with_rsa_pss(payload, private_key)
  puts "Signature (Base64): #{signature_base64[0..50]}..."
  puts
  
  puts "Verifying signature..."
  is_valid = verify_rsa_pss_signature(payload, signature_base64, public_key)
  puts "Signature valid: #{is_valid}"
  puts
  
  puts "=== Testing with tampered data ==="
  tampered_payload = "This is tampered data!"
  is_valid_tampered = verify_rsa_pss_signature(tampered_payload, signature_base64, public_key)
  puts "Tampered signature valid: #{is_valid_tampered}"
  puts
  
  puts "=== Testing complete payload signing ==="
  private_key_pem = private_key.to_pem
  public_key_pem = public_key.to_pem
  
  json_payload = { user_id: 12345, action: "transfer", amount: 1000.00, currency: "USD" }
  puts "JSON Payload: #{json_payload}"
  
  signed_data = sign_payload(json_payload, private_key_pem)
  puts "Signed data created with algorithm: #{signed_data[:algorithm]}"
  puts "Timestamp: #{signed_data[:timestamp]}"
  puts
  
  verification_result = verify_payload(signed_data, public_key_pem)
  puts "Payload verification result: #{verification_result}"
  puts
  
  puts "=== Testing with different digest algorithms ==="
  ['SHA256', 'SHA384', 'SHA512'].each do |algo|
    sig = sign_with_rsa_pss(payload, private_key, algo)
    valid = verify_rsa_pss_signature(payload, sig, public_key, algo)
    puts "#{algo}: Signed and verified successfully: #{valid}"
  end
end
