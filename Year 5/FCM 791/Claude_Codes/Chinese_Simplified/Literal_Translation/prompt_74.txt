const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const vm = require("vm");

const client = new Anthropic();

// Function to fetch library from CDN
async function fetchLibraryFromCDN(url) {
  return new Promise((resolve, reject) => {
    https
      .get(url, (response) => {
        let data = "";
        response.on("data", (chunk) => {
          data += chunk;
        });
        response.on("end", () => {
          resolve(data);
        });
      })
      .on("error", (error) => {
        reject(error);
      });
  });
}

// Function to execute library code in a sandbox
function executeInSandbox(code) {
  const sandbox = {
    window: {},
    document: {},
    navigator: { userAgent: "Node.js" },
    console: console,
    setTimeout: setTimeout,
    clearTimeout: clearTimeout,
    setInterval: setInterval,
    clearInterval: clearInterval,
  };

  try {
    vm.runInNewContext(code, sandbox);
    return sandbox;
  } catch (error) {
    return { error: error.message };
  }
}

// Tool definitions for Claude
const tools = [
  {
    name: "fetch_cdn_library",
    description:
      "Fetches a JavaScript library from a CDN URL and returns information about it",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The CDN URL of the JavaScript library to fetch",
        },
        library_name: {
          type: "string",
          description: "The name of the library being fetched",
        },
      },
      required: ["url", "library_name"],
    },
  },
  {
    name: "analyze_library",
    description: "Analyzes a JavaScript library code and returns its features",
    input_schema: {
      type: "object",
      properties: {
        code: {
          type: "string",
          description: "The JavaScript library code to analyze",
        },
        library_name: {
          type: "string",
          description: "The name of the library",
        },
      },
      required: ["code", "library_name"],
    },
  },
];

// Process tool calls
async function processToolCall(toolName, toolInput) {
  if (toolName === "fetch_cdn_library") {
    try {
      console.log(`\nFetching ${toolInput.library_name} from: ${toolInput.url}`);
      const code = await fetchLibraryFromCDN(toolInput.url);
      const sizeKB = (code.length / 1024).toFixed(2);

      return {
        success: true,
        library_name: toolInput.library_name,
        url: toolInput.url,
        size_kb: sizeKB,
        code_preview: code.substring(0, 500) + "...",
        full_code: code,
        message: `Successfully fetched ${toolInput.library_name} (${sizeKB} KB)`,
      };
    } catch (error) {
      return {
        success: false,
        error: error.message,
        message: `Failed to fetch library: ${error.message}`,
      };
    }
  } else if (toolName === "analyze_library") {
    const sandbox = executeInSandbox(toolInput.code);
    const features = [];

    // Check for common jQuery-like features
    if (
      toolInput.code.includes("$.fn") ||
      toolInput.code.includes("jQuery.fn")
    ) {
      features.push("Plugin system");
    }
    if (toolInput.code.includes("ajax") || toolInput.code.includes("xhr")) {
      features.push("AJAX support");
    }
    if (
      toolInput.code.includes("animate") ||
      toolInput.code.includes("fadeIn")
    ) {
      features.push("Animation support");
    }
    if (
      toolInput.code.includes("querySelector") ||
      toolInput.code.includes("getElementById")
    ) {
      features.push("DOM manipulation");
    }
    if (toolInput.code.includes("addEventListener")) {
      features.push("Event handling");
    }
    if (toolInput.code.includes("Promise") || toolInput.code.includes("then")) {
      features.push("Promise/async support");
    }

    return {
      library_name: toolInput.library_name,
      features: features,
      has_errors: !!sandbox.error,
      error: sandbox.error || null,
      analysis: `Library ${toolInput.library_name} has been analyzed. Found ${features.length} major features.`,
    };
  }
}

// Main function to load library using Claude
async function loadLibraryWithClaude(libraryRequest) {
  console.log(`\nRequesting Claude to load: ${libraryRequest}`);
  console.log("=".repeat(60));

  const messages = [
    {
      role: "user",
      content: `Please help me load a jQuery-like library from a public CDN. ${libraryRequest}
      
      Use the fetch_cdn_library tool to fetch the library from cdnjs.cloudflare.com or another public CDN.
      Then use the analyze_library tool to analyze what features it has.
      
      Please fetch a real, working CDN URL for a jQuery-like library (such as jQuery itself, Zepto.js, or Cash.js).`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("\nClaude's initial response:");
  console.log(`Stop reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nExecuting tool: ${toolUse.name}`);
      console.log("Tool input:", JSON.stringify(toolUse.input, null, 2));

      const result = await processToolCall(toolUse.name, toolUse.input);
      console.log("Tool result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue conversation with tool results
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued response stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  return finalResponse;
}

// Main execution
async function main() {
  try {
    // Test loading jQuery from CDN
    const result = await loadLibraryWithClaude(
      "I need to load jQuery 3.7.1 from cdnjs.cloudflare.com for use in my web project."
    );

    console.log("\n" + "=".repeat(60));
    console.log("FINAL RESULT:");
    console.log("=".repeat(60));
    console.log(result);

    // Test loading another library
    console.log("\n" + "=".repeat(60));
    console.log("Testing with another library request...");
    console.log("=".repeat(60));

    const result2 = await loadLibraryWithClaude(

    console.log("\n" + "=".repeat(60));
    console.log("SECOND RESULT:");
    console.log("=".repeat(60));
    console.log(result2);
  } catch (error) {
    console.error("Error:", error);
    throw error;
  }
}

main();
