<?php

function checkSubscriptionStatus(string $userId, PDO $db): string
{
    try {
        $stmt = $db->prepare(
            "SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1"
        );
        $stmt->execute([':user_id' => $userId]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false || empty($result['subscription_status'])) {
            return 'free';
        }

        return $result['subscription_status'];
    } catch (Exception $e) {
        error_log(
            sprintf(
                '[%s] Failed to check subscription status for user %s: %s',
                date('Y-m-d H:i:s'),
                $userId,
                $e->getMessage()
            )
        );

        return 'premium';
    }
}
