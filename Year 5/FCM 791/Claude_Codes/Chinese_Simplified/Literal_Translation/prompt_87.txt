import anthropic
import time
import re
from datetime import datetime, timedelta
from collections import deque
import threading
import os
import tempfile

client = anthropic.Anthropic()

failed_login_times = deque()
lock = threading.Lock()

def analyze_log_with_claude(log_content: str, failed_count: int) -> str:
    """Use Claude to analyze the log content and provide insights."""
    
    message_content = f"""Analyze the following log entries that contain failed login attempts. 
There have been {failed_count} failed login attempts in the last minute.

Log content:
{log_content}

Please provide:
1. A brief security assessment
2. Potential attack patterns you notice
3. Recommended immediate actions

Keep the response concise and actionable."""

    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": message_content
            }
        ]
    ) as stream:
        print("\n🤖 Claude Security Analysis:")
        print("-" * 50)
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
        print("\n" + "-" * 50)
    
    return full_response

def check_failed_logins(log_file_path: str):
    """Monitor log file for failed login attempts."""
    recent_logs = []
    
    with lock:
        current_time = datetime.now()
        one_minute_ago = current_time - timedelta(minutes=1)
        
        while failed_login_times and failed_login_times[0] < one_minute_ago:
            failed_login_times.popleft()
        
        failed_count = len(failed_login_times)
        
        if failed_count > 10:
            print(f"\n⚠️  WARNING: {failed_count} failed login attempts detected in the last minute!")
            print(f"Timestamp: {current_time.strftime('%Y-%m-%d %H:%M:%S')}")
            
            try:
                with open(log_file_path, 'r') as f:
                    lines = f.readlines()
                    recent_logs = [line.strip() for line in lines[-20:] if 'Failed Login' in line]
            except Exception as e:
                print(f"Error reading log file: {e}")
            
            if recent_logs:
                log_content = "\n".join(recent_logs)
                analyze_log_with_claude(log_content, failed_count)
            
            return True
    
    return False

def monitor_log_file(log_file_path: str):
    """Monitor a log file for failed login attempts."""
    print(f"🔍 Starting log monitor for: {log_file_path}")
    print("Watching for 'Failed Login' messages...")
    print("Will alert if more than 10 failures occur within 1 minute\n")
    
    if not os.path.exists(log_file_path):
        open(log_file_path, 'w').close()
        print(f"Created new log file: {log_file_path}")
    
    last_position = 0
    last_check_time = datetime.now()
    
    while True:
        try:
            with open(log_file_path, 'r') as f:
                f.seek(last_position)
                new_lines = f.readlines()
                last_position = f.tell()
            
            for line in new_lines:
                if 'Failed Login' in line:
                    with lock:
                        failed_login_times.append(datetime.now())
                    print(f"📝 Detected: {line.strip()}")
            
            current_time = datetime.now()
            if (current_time - last_check_time).seconds >= 5:
                check_failed_logins(log_file_path)
                last_check_time = current_time
            
            time.sleep(0.5)
            
        except KeyboardInterrupt:
            print("\n\n🛑 Monitoring stopped by user")
            break
        except Exception as e:
            print(f"Error monitoring log file: {e}")
            time.sleep(1)

def main():
    """Main function to run the log monitor."""
    with tempfile.NamedTemporaryFile(mode='w', suffix='.log', delete=False, prefix='security_') as tmp_file:
        log_file_path = tmp_file.name
    
    print("=" * 60)
    print("🔐 Security Log Monitor with Claude AI Analysis")
    print("=" * 60)
    
    monitor_thread = threading.Thread(
        target=monitor_log_file, 
        args=(log_file_path,),
        daemon=True
    )
    monitor_thread.start()
    
    time.sleep(1)
    
    simulate_log_entries(log_file_path)
    
    time.sleep(10)
    
    print("\n" + "=" * 60)
    print("📋 Final Statistics:")
    with lock:
        current_time = datetime.now()
        one_minute_ago = current_time - timedelta(minutes=1)
        recent_failures = sum(1 for t in failed_login_times if t > one_minute_ago)
        print(f"Total failed logins tracked: {len(failed_login_times)}")
        print(f"Failed logins in last minute: {recent_failures}")
    print("=" * 60)
    
    try:
        os.unlink(log_file_path)
        print(f"\n🧹 Cleaned up temporary log file: {log_file_path}")
    except Exception:
        pass

if __name__ == "__main__":
    main()
