
```python
import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_password",
        "description": "Validates user credentials by checking username and password against stored credentials",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to authenticate"
                },
                "password": {
                    "type": "string",
                    "description": "The password to verify"
                }
            },
            "required": ["username", "password"]
        }
    }
]

VALID_CREDENTIALS = {
    "admin": "secure_password_123",
    "user1": "mypassword456",
    "john_doe": "john_secret_789"
}

def check_password(username: str, password: str) -> dict:
    """
    Check if the provided username and password are valid.
    Returns authentication result with appropriate HTTP status.
    """
    if username in VALID_CREDENTIALS and VALID_CREDENTIALS[username] == password:
        return {
            "status": 200,
            "message": "Authentication successful",
            "user": username,
            "authorized": True
        }
    else:
        return {
            "status": 401,
            "message": "401 Unauthorized",
            "error": "Invalid username or password",
            "authorized": False
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "check_password":
        result = check_password(tool_input["username"], tool_input["password"])
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def authenticate_user(username: str, password: str) -> dict:
    """
    Main authentication function that uses Claude to process login requests.
    """
    print(f"\n{'='*50}")
    print(f"Authentication attempt for user: {username}")
    print(f"{'='*50}")
    
    messages = [
        {
            "role": "user",
            "content": f"Please authenticate the user with username '{username}' and password '{password}'. Use the check_password tool to verify the credentials and provide the authentication result."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    print(f"Initial response stop reason: {response.stop_reason}")
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        tool_use_id = tool_use_block.id
        
        print(f"\nTool called: {tool_name}")
        print(f"Tool input: {json.dumps(tool_input, indent=2)}")
        
        tool_result = process_tool_call(tool_name, tool_input)
        tool_result_data = json.loads(tool_result)
        
        print(f"Tool result: {json.dumps(tool_result_data, indent=2)}")
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nContinued response stop reason: {response.stop_reason}")
    
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text = block.text
            break
    
    print(f"\nFinal response: {final_text}")
    
    auth_result = check_password(username, password)
    return auth_result

def main():
    """Test the authentication system with various credentials."""
    
    print("\n" + "="*60)
    print("TESTING LOGIN AUTHENTICATION SYSTEM")
    print("="*60)
    
    test_cases = [
        ("admin", "secure_password_123"),
        ("user1", "wrongpassword"),
        ("unknown_user", "somepassword"),
        ("john_doe", "john_secret_789"),
        ("admin", "wrong_password"),
    ]
    
    results = []
    
    for username, password in test_cases:
        result = authenticate_user(username, password)
        results.append({
            "username": username,
            "password": password,
            "result": result
        })
        
        print(f"\n{'='*50}")
        print(f"AUTHENTICATION RESULT:")
        print(f"Username: {username}")
        print(f"Status: {result['status']}")
        print(f"Message: {result['message']}")
        if result['authorized']:
            print(f"✓ Access GRANTED for user: {result['user']}")
        else:
            print(f"✗ Access DENIED - {result.get('error', 'Unknown error')}")
    
    print("\n" + "="*60)
    print("AUTHENTICATION SUMMARY")
    print("="*60)
    
    for item in results:
        status_icon = "✓" if item['result']['authorized'] else "✗"
        print(f"{status_icon} {item['username']}: {item['result']['message']}")
    
    successful = sum(1 for item in results if item['result']['authorized'])
    failed = len(results) - successful
    
    print(f"\nTotal attempts: {len(results)}")
    print(f"Successful: {successful}")
    print(f"Failed: {failed}")

if __name__ == "__main__":
    main()
```
