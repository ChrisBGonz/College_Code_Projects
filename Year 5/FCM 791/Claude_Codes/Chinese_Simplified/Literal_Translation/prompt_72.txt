const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definitions for Claude
const tools = [
  {
    name: "download_file",
    description:
      "Download a file from a URL and save it to a local path. Returns success status and file size.",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path to save the downloaded file",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description:
      "Extract a zip file to a specified directory. Returns success status and list of extracted files.",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "Path to the zip file to extract",
        },
        destination_dir: {
          type: "string",
          description: "Directory to extract the zip contents to",
        },
      },
      required: ["zip_path", "destination_dir"],
    },
  },
  {
    name: "create_directory",
    description:
      "Create a directory if it doesn't exist. Returns success status.",
    input_schema: {
      type: "object",
      properties: {
        dir_path: {
          type: "string",
          description: "Path of the directory to create",
        },
      },
      required: ["dir_path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Delete a temporary file. Returns success status.",
    input_schema: {
      type: "object",
      properties: {
        file_path: {
          type: "string",
          description: "Path of the file to delete",
        },
      },
      required: ["file_path"],
    },
  },
];

// Tool implementations
async function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    const protocol = url.startsWith("https") ? https : http;
    const file = fs.createWriteStream(destination);

    protocol
      .get(url, (response) => {
        if (response.statusCode === 301 || response.statusCode === 302) {
          // Handle redirects
          file.close();
          fs.unlink(destination, () => {});
          downloadFile(response.headers.location, destination)
            .then(resolve)
            .catch(reject);
          return;
        }

        if (response.statusCode !== 200) {
          file.close();
          fs.unlink(destination, () => {});
          reject(new Error(`Failed to download: HTTP ${response.statusCode}`));
          return;
        }

        response.pipe(file);

        file.on("finish", () => {
          file.close();
          const stats = fs.statSync(destination);
          resolve({
            success: true,
            file_size: stats.size,
            message: `File downloaded successfully to ${destination}`,
          });
        });
      })
      .on("error", (err) => {
        file.close();
        fs.unlink(destination, () => {});
        reject(err);
      });
  });
}

function extractZip(zipPath, destinationDir) {
  try {
    // Use unzip command (available on most Unix systems)
    // For Windows, you might need to use a different approach
    const output = execSync(`unzip -o "${zipPath}" -d "${destinationDir}"`, {
      encoding: "utf8",
    });

    // Parse extracted files from output
    const extractedFiles = output
      .split("\n")
      .filter((line) => line.includes("inflating:") || line.includes("creating:"))
      .map((line) => line.replace(/\s*(inflating|creating):\s*/, "").trim());

    return {
      success: true,
      extracted_files: extractedFiles,
      message: `Successfully extracted ${extractedFiles.length} files to ${destinationDir}`,
    };
  } catch (error) {
    // Try alternative: use Node.js built-in or another method
    try {
      // Fallback: try using 7z if available
      execSync(`7z x "${zipPath}" -o"${destinationDir}" -y`, {
        encoding: "utf8",
      });
      return {
        success: true,
        extracted_files: [],
        message: `Successfully extracted zip to ${destinationDir} using 7z`,
      };
    } catch {
      return {
        success: false,
        error: error.message,
        message: `Failed to extract zip: ${error.message}`,
      };
    }
  }
}

function createDirectory(dirPath) {
  try {
    fs.mkdirSync(dirPath, { recursive: true });
    return {
      success: true,
      message: `Directory created: ${dirPath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
      message: `Failed to create directory: ${error.message}`,
    };
  }
}

function cleanupFile(filePath) {
  try {
    fs.unlinkSync(filePath);
    return {
      success: true,
      message: `File deleted: ${filePath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
      message: `Failed to delete file: ${error.message}`,
    };
  }
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      result = await downloadFile(toolInput.url, toolInput.destination);
      break;
    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.destination_dir);
      break;
    case "create_directory":
      result = createDirectory(toolInput.dir_path);
      break;
    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;
    default:
      result = { error: `Unknown tool: ${toolName}` };
  }

  console.log("Result:", JSON.stringify(result, null, 2));
  return result;
}

// Main function to download and extract update package
async function downloadAndExtractUpdate(updateUrl, targetDir = "/app") {
  console.log(`Starting update process...`);
  console.log(`URL: ${updateUrl}`);
  console.log(`Target directory: ${targetDir}`);

  const messages = [
    {
      role: "user",
      content: `Please help me download a zip update package from "${updateUrl}" and extract it to "${targetDir}". 
      
      Follow these steps:
      1. Create the target directory "${targetDir}" if it doesn't exist
      2. Download the zip file to a temporary location (use /tmp/update_package.zip)
      3. Extract the zip file to "${targetDir}"
      4. Clean up the temporary zip file after extraction
      
      Please proceed with these steps and report the results.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nClaude response (stop_reason: ${response.stop_reason}):`);

    // Add assistant response to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    if (response.stop_reason === "tool_use") {
      // Process tool calls
      const toolResults = [];

      for (const block of response.content) {
        if (block.type === "tool_use") {
          console.log(`\nTool call: ${block.name}`);
          const result = await processToolCall(block.name, block.input);

          toolResults.push({
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          });
        } else if (block.type === "text") {
          console.log(`Claude: ${block.text}`);
        }
      }

      // Add tool results to messages
      if (toolResults.length > 0) {
        messages.push({
          role: "user",
          content: toolResults,
        });
      }
    } else if (response.stop_reason === "end_turn") {
      // Final response
      for (const block of response.content) {
        if (block.type === "text") {
          console.log(`\nFinal response from Claude:\n${block.text}`);
        }
      }
      continueLoop = false;
    } else {
      console.log(`Unexpected stop reason: ${response.stop_reason}`);
      continueLoop = false;
    }
  }

  console.log("\nUpdate process completed!");
}

// Example usage
async function main() {
  // Example: Download a sample zip file
  // Using a public test zip file for demonstration
  const testUrl =
    "https://www.learningcontainer.com/wp-content/uploads/2020/05/sample-zip-file.zip";
  const targetDirectory = "./app"; // Using relative path for testing

  try {
    await downloadAndExtractUpdate(testUrl, targetDirectory);
  } catch (error) {
    console.error("Error during update process:", error);
    process.exit(1);
  }
}

main();
