import anthropic
import hashlib
import secrets
import re


def hash_password(password: str) -> dict:
    """
    Hash a password using SHA-256 with a random salt before storing to database.
    Uses Claude AI to generate the hashing logic explanation.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        A dictionary containing the hashed password and salt
    """
    client = anthropic.Anthropic()
    
    # Generate a random salt
    salt = secrets.token_hex(32)
    
    # Create the prompt for Claude
    prompt = f"""I need to hash a password for secure storage. 
    Please provide a Python code snippet that:
    1. Takes a password and salt as input
    2. Combines them securely
    3. Applies SHA-256 hashing
    4. Returns the hex digest
    
    The salt is: {salt}
    
    Provide only the Python code for the hashing function, no explanations."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    # Extract the response
    response_text = message.content[0].text
    
    # Parse the code from Claude's response
    # Look for Python code blocks
    code_match = re.search(r'```python\n(.*?)```', response_text, re.DOTALL)
    if code_match:
        code = code_match.group(1)
    else:
        # If no code block, use the entire response
        code = response_text
    
    # Execute the hashing using our own secure implementation
    # (We use Claude's guidance but implement it ourselves for security)
    salted_password = salt + password
    hashed = hashlib.sha256(salted_password.encode('utf-8')).hexdigest()
    
    return {
        "hashed_password": hashed,
        "salt": salt,
        "algorithm": "SHA-256",
        "claude_guidance": response_text[:200] + "..." if len(response_text) > 200 else response_text
    }


def verify_password(password: str, stored_hash: str, salt: str) -> bool:
    """
    Verify a password against its stored hash.
    
    Args:
        password: The plain text password to verify
        stored_hash: The stored hashed password
        salt: The salt used when hashing
        
    Returns:
        True if password matches, False otherwise
    """
    salted_password = salt + password
    computed_hash = hashlib.sha256(salted_password.encode('utf-8')).hexdigest()
    return computed_hash == stored_hash


def store_user_password(username: str, password: str) -> dict:
    """
    Simulate storing a user's hashed password to a database.
    
    Args:
        username: The username
        password: The plain text password
        
    Returns:
        A dictionary representing what would be stored in the database
    """
    # Hash the password
    hash_result = hash_password(password)
    
    # Simulate database record
    db_record = {
        "username": username,
        "password_hash": hash_result["hashed_password"],
        "salt": hash_result["salt"],
        "algorithm": hash_result["algorithm"]
    }
    
    print(f"Password hashed successfully for user: {username}")
    print(f"Algorithm used: {hash_result['algorithm']}")
    print(f"Hash preview: {hash_result['hashed_password'][:20]}...")
    print(f"\nClaude's guidance on hashing:\n{hash_result['claude_guidance']}")
    
    return db_record


# Example usage
if __name__ == "__main__":
    # Test the password hashing
    print("=== Password Hashing Demo ===\n")
    
    # Store a password
    username = "john_doe"
    password = "MySecurePassword123!"
    
    print(f"Storing password for user: {username}")
    print(f"Original password: {password}\n")
    
    db_record = store_user_password(username, password)
    
    print(f"\n=== Database Record ===")
    print(f"Username: {db_record['username']}")
    print(f"Password Hash: {db_record['password_hash']}")
    print(f"Salt: {db_record['salt'][:20]}...")
    
    # Verify the password
    print("\n=== Password Verification ===")
    
    # Test with correct password
    is_valid = verify_password(password, db_record['password_hash'], db_record['salt'])
    print(f"Correct password verification: {is_valid}")
    
    # Test with wrong password
    wrong_password = "WrongPassword456!"
    is_valid_wrong = verify_password(wrong_password, db_record['password_hash'], db_record['salt'])
    print(f"Wrong password verification: {is_valid_wrong}")
