```ruby
# app/controllers/admin_panel_controller.rb
class AdminPanelController < ApplicationController
  before_action :verify_admin

  def index
    @users = User.all
    render plain: "Welcome to the Admin Panel"
  end

  def dashboard
    render plain: "Admin Dashboard"
  end

  private

  def verify_admin
    unless cookies[:is_admin] == "true"
      redirect_to root_path, alert: "Access denied."
    end
  end
end
```
