const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

// Simple user database simulation
const users = {
  "token-user-123": {
    id: "user-123",
    name: "Alice Johnson",
    email: "alice@example.com",
    role: "admin",
    createdAt: "2024-01-15",
    privateData: {
      phoneNumber: "+1-555-0123",
      address: "123 Main St, New York, NY 10001",
      ssn: "***-**-1234",
      bankAccount: "****-****-****-5678",
    },
  },
  "token-user-456": {
    id: "user-456",
    name: "Bob Smith",
    email: "bob@example.com",
    role: "user",
    createdAt: "2024-02-20",
    privateData: {
      phoneNumber: "+1-555-0456",
      address: "456 Oak Ave, Los Angeles, CA 90001",
      ssn: "***-**-5678",
      bankAccount: "****-****-****-9012",
    },
  },
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_bearer_token",
    description:
      "Validates a Bearer token from the Authorization header and returns user information if valid",
    input_schema: {
      type: "object",
      properties: {
        token: {
          type: "string",
          description: "The Bearer token extracted from the Authorization header",
        },
      },
      required: ["token"],
    },
  },
  {
    name: "get_user_private_data",
    description:
      "Retrieves private user data for an authenticated user. Only call this after successful token validation.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The user ID of the authenticated user",
        },
        data_fields: {
          type: "array",
          items: {
            type: "string",
          },
          description:
            "List of specific data fields to retrieve (e.g., ['email', 'phoneNumber', 'address'])",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "log_access_attempt",
    description: "Logs an access attempt for security auditing purposes",
    input_schema: {
      type: "object",
      properties: {
        token: {
          type: "string",
          description: "The token used in the access attempt",
        },
        success: {
          type: "boolean",
          description: "Whether the access attempt was successful",
        },
        reason: {
          type: "string",
          description: "Reason for success or failure",
        },
        timestamp: {
          type: "string",
          description: "ISO timestamp of the access attempt",
        },
      },
      required: ["token", "success", "reason", "timestamp"],
    },
  },
];

// Tool execution functions
function validateBearerToken(token) {
  const user = users[token];
  if (user) {
    return {
      valid: true,
      userId: user.id,
      userName: user.name,
      userRole: user.role,
    };
  }
  return {
    valid: false,
    error: "Invalid or expired token",
  };
}

function getUserPrivateData(userId, dataFields) {
  // Find user by ID
  const userEntry = Object.entries(users).find(
    ([, user]) => user.id === userId
  );

  if (!userEntry) {
    return {
      error: "User not found",
    };
  }

  const [, user] = userEntry;
  const result = {
    id: user.id,
    name: user.name,
    email: user.email,
    role: user.role,
    createdAt: user.createdAt,
  };

  // Add private data fields if requested
  if (dataFields && dataFields.length > 0) {
    result.privateData = {};
    dataFields.forEach((field) => {
      if (user.privateData[field]) {
        result.privateData[field] = user.privateData[field];
      }
    });
  } else {
    // Return all private data if no specific fields requested
    result.privateData = user.privateData;
  }

  return result;
}

function logAccessAttempt(token, success, reason, timestamp) {
  const maskedToken =
    token.length > 8
      ? `${token.substring(0, 4)}...${token.substring(token.length - 4)}`
      : "****";

  console.log(`[SECURITY LOG] ${timestamp}`);
  console.log(`  Token: ${maskedToken}`);
  console.log(`  Success: ${success}`);
  console.log(`  Reason: ${reason}`);
  console.log("---");

  return {
    logged: true,
    logId: `log-${Date.now()}`,
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_bearer_token":
      return validateBearerToken(toolInput.token);
    case "get_user_private_data":
      return getUserPrivateData(toolInput.user_id, toolInput.data_fields);
    case "log_access_attempt":
      return logAccessAttempt(
        toolInput.token,
        toolInput.success,
        toolInput.reason,
        toolInput.timestamp
      );
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main API handler using Claude with tool use
async function handleAPIRequest(authorizationHeader, requestedFields) {
  console.log("\n=== Processing API Request ===");
  console.log(
    `Authorization Header: ${authorizationHeader ? "Present" : "Missing"}`
  );

  // Extract Bearer token
  let bearerToken = null;
  if (authorizationHeader && authorizationHeader.startsWith("Bearer ")) {
    bearerToken = authorizationHeader.substring(7);
  }

  // Prepare the message for Claude
  const userMessage = bearerToken
    ? `Process this API request:
    - Bearer Token: ${bearerToken}
    - Requested Data Fields: ${requestedFields ? requestedFields.join(", ") : "all fields"}
    
    Please:
    1. Log this access attempt
    2. Validate the Bearer token
    3. If valid, retrieve the user's private data
    4. Return appropriate response`
    : `Process this API request with missing authentication:
    - No Bearer token provided
    - Requested Data Fields: ${requestedFields ? requestedFields.join(", ") : "all fields"}
    
    Please:
    1. Log this failed access attempt
    2. Return an appropriate error response`;

  const messages = [{ role: "user", content: userMessage }];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );
    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nExecuting tool: ${toolUse.name}`);
      console.log("Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({ role: "assistant", content: response.content });
    messages.push({ role: "user", content: toolResults });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  return finalResponse;
}

// Simulate API requests
async function simulateAPIRequests() {
  console.log("=== Node.js API with Bearer Token Authentication ===\n");

  // Test Case 1: Valid token with all data
  console.log("TEST CASE 1: Valid Bearer Token (User 123)");
  console.log("=".repeat(50));
  const response1 = await handleAPIRequest("Bearer token-user-123", null);
  console.log("\nAPI Response:");
  console.log(response1);

  console.log("\n" + "=".repeat(70) + "\n");

  // Test Case 2: Valid token with specific fields
  console.log("TEST CASE 2: Valid Bearer Token with Specific Fields (User 456)");
  console.log("=".repeat(50));
  const response2 = await handleAPIRequest("Bearer token-user-456", [
    "phoneNumber",
    "address",
  ]);
  console.log("\nAPI Response:");
  console.log(response2);

  console.log("\n" + "=".repeat(70) + "\n");

  // Test Case 3: Invalid token
  console.log("TEST CASE 3: Invalid Bearer Token");
  console.log("=".repeat(50));
  const response3 = await handleAPIRequest("Bearer invalid-token-xyz", null);
  console.log("\nAPI Response:");
  console.log(response3);

  console.log("\n" + "=".repeat(70) + "\n");

  // Test Case 4: Missing token
  console.log("TEST CASE 4: Missing Authorization Header");
  console.log("=".repeat(50));
  const response4 = await handleAPIRequest(null, null);
  console.log("\nAPI Response:");
  console.log(response4);
}

// Run the simulation
simulateAPIRequests().catch(console.error);
