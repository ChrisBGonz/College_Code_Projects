from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from typing import Optional
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = FastAPI(title="FastAPI with Custom Exception Handlers")


class DatabaseConnectionError(Exception):
    def __init__(
        self,
        message: str = "Database connection failed",
        detail: Optional[str] = None,
        status_code: int = 503,
    ):
        self.message = message
        self.detail = detail
        self.status_code = status_code
        super().__init__(self.message)


class DatabaseQueryError(Exception):
    def __init__(
        self,
        message: str = "Database query failed",
        query: Optional[str] = None,
        detail: Optional[str] = None,
        status_code: int = 500,
    ):
        self.message = message
        self.query = query
        self.detail = detail
        self.status_code = status_code
        super().__init__(self.message)


class DatabaseTimeoutError(Exception):
    def __init__(
        self,
        message: str = "Database connection timed out",
        timeout_seconds: Optional[int] = None,
        status_code: int = 504,
    ):
        self.message = message
        self.timeout_seconds = timeout_seconds
        self.status_code = status_code
        super().__init__(self.message)


@app.exception_handler(DatabaseConnectionError)
async def database_connection_error_handler(
    request: Request, exc: DatabaseConnectionError
):
    logger.error(
        f"Database connection error on {request.url}: {exc.message} - {exc.detail}"
    )
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "error": "Database Connection Error",
            "message": exc.message,
            "detail": exc.detail,
            "path": str(request.url),
            "status_code": exc.status_code,
        },
    )


@app.exception_handler(DatabaseQueryError)
async def database_query_error_handler(request: Request, exc: DatabaseQueryError):
    logger.error(
        f"Database query error on {request.url}: {exc.message} - Query: {exc.query}"
    )
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "error": "Database Query Error",
            "message": exc.message,
            "detail": exc.detail,
            "query": exc.query,
            "path": str(request.url),
            "status_code": exc.status_code,
        },
    )


@app.exception_handler(DatabaseTimeoutError)
async def database_timeout_error_handler(request: Request, exc: DatabaseTimeoutError):
    logger.error(
        f"Database timeout error on {request.url}: {exc.message} - Timeout: {exc.timeout_seconds}s"
    )
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "error": "Database Timeout Error",
            "message": exc.message,
            "timeout_seconds": exc.timeout_seconds,
            "path": str(request.url),
            "status_code": exc.status_code,
        },
    )


@app.exception_handler(Exception)
async def general_exception_handler(request: Request, exc: Exception):
    logger.error(f"Unexpected error on {request.url}: {str(exc)}")
    return JSONResponse(
        status_code=500,
        content={
            "error": "Internal Server Error",
            "message": "An unexpected error occurred",
            "detail": str(exc),
            "path": str(request.url),
            "status_code": 500,
        },
    )


def simulate_db_connection(host: str, port: int = 5432):
    if host == "invalid_host":
        raise DatabaseConnectionError(
            message=f"Cannot connect to database at {host}:{port}",
            detail="Host unreachable or invalid credentials",
        )
    return {"connection": "success", "host": host, "port": port}


def simulate_db_query(query: str):
    if "DROP TABLE" in query.upper():
        raise DatabaseQueryError(
            message="Forbidden query detected",
            query=query,
            detail="DROP TABLE operations are not allowed",
            status_code=403,
        )
    if "INVALID_SYNTAX" in query.upper():
        raise DatabaseQueryError(
            message="SQL syntax error",
            query=query,
            detail="The provided SQL query has invalid syntax",
        )
    return {"result": "Query executed successfully", "query": query}


def simulate_db_with_timeout(timeout: int):
    if timeout < 5:
        raise DatabaseTimeoutError(
            message="Database connection timed out",
            timeout_seconds=timeout,
        )
    return {"status": "Connected within timeout", "timeout": timeout}


@app.get("/")
async def root():
    return {
        "message": "FastAPI Database Error Handler Demo",
        "endpoints": {
            "/connect/{host}": "Test database connection",
            "/query": "Test database query (pass ?q=YOUR_QUERY)",
            "/timeout/{seconds}": "Test database timeout",
            "/health": "Health check",
        },
    }


@app.get("/health")
async def health_check():
    return {"status": "healthy", "service": "FastAPI Database Error Handler"}


@app.get("/connect/{host}")
async def connect_to_database(host: str, port: int = 5432):
    result = simulate_db_connection(host, port)
    return {
        "status": "success",
        "message": "Database connected successfully",
        "connection_info": result,
    }


@app.get("/query")
async def execute_query(q: str = "SELECT * FROM users"):
    result = simulate_db_query(q)
    return {
        "status": "success",
        "message": "Query executed successfully",
        "result": result,
    }


@app.get("/timeout/{seconds}")
async def test_timeout(seconds: int):
    result = simulate_db_with_timeout(seconds)
    return {
        "status": "success",
        "message": "Connection established within timeout",
        "result": result,
    }


@app.get("/simulate/connection-error")
async def simulate_connection_error():
    raise DatabaseConnectionError(
        message="Simulated database connection failure",
        detail="This is a simulated error for testing purposes",
    )


@app.get("/simulate/query-error")
async def simulate_query_error():
    raise DatabaseQueryError(
        message="Simulated query execution failure",
        query="SELECT * FROM non_existent_table",
        detail="Table does not exist in the database",
    )


@app.get("/simulate/timeout-error")
async def simulate_timeout_error():
    raise DatabaseTimeoutError(
        message="Simulated database timeout",
        timeout_seconds=30,
    )


@app.get("/simulate/general-error")
async def simulate_general_error():
    raise ValueError("This is a simulated general error")


if __name__ == "__main__":
    import uvicorn

    uvicorn.run(app, host="0.0.0.0", port=8000, reload=True)
