// Database configuration
define('DB_HOST', 'localhost');

// Connect to database
function getDBConnection() {
    $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
    if ($conn->connect_error) {
        die('数据库连接失败: ' . $conn->connect_error);
    }
    $conn->set_charset('utf8mb4');
    return $conn;
}

// Initialize database tables
function initDatabase() {
    $conn = getDBConnection();
    
    $conn->query("CREATE TABLE IF NOT EXISTS users (
        id INT AUTO_INCREMENT PRIMARY KEY,
        username VARCHAR(100) NOT NULL UNIQUE,
        email VARCHAR(255) NOT NULL UNIQUE,
        password VARCHAR(255) NOT NULL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )");
    
    $conn->query("CREATE TABLE IF NOT EXISTS security_questions (
        id INT AUTO_INCREMENT PRIMARY KEY,
        question_text VARCHAR(500) NOT NULL
    )");
    
    $conn->query("CREATE TABLE IF NOT EXISTS user_security_answers (
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT NOT NULL,
        question_id INT NOT NULL,
        answer_hash VARCHAR(255) NOT NULL,
        FOREIGN KEY (user_id) REFERENCES users(id),
        FOREIGN KEY (question_id) REFERENCES security_questions(id)
    )");
    
    $conn->query("CREATE TABLE IF NOT EXISTS password_reset_attempts (
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT NOT NULL,
        ip_address VARCHAR(45) NOT NULL,
        attempt_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        success TINYINT(1) DEFAULT 0
    )");
    
    // Insert default security questions if not exists
    $result = $conn->query("SELECT COUNT(*) as count FROM security_questions");
    $row = $result->fetch_assoc();
    if ($row['count'] == 0) {
        $questions = [
            '你母亲的娘家姓是什么？',
            '你出生的城市是哪里？',
            '你的第一只宠物叫什么名字？',
            '你上的第一所小学叫什么？',
            '你最好的朋友叫什么名字？',
            '你父亲的出生地是哪里？',
            '你最喜欢的老师叫什么名字？',
            '你的父母在哪里相识？',
            '你童年时期最喜欢的食物是什么？',
            '你第一辆车的品牌是什么？'
        ];
        
        $stmt = $conn->prepare("INSERT INTO security_questions (question_text) VALUES (?)");
        foreach ($questions as $question) {
            $stmt->bind_param('s', $question);
            $stmt->execute();
        }
        $stmt->close();
    }
    
    $conn->close();
}

// Rate limiting check
function checkRateLimit($userId) {
    $conn = getDBConnection();
    $ip = $_SERVER['REMOTE_ADDR'];
    $timeWindow = date('Y-m-d H:i:s', strtotime('-15 minutes'));
    
    $stmt = $conn->prepare("SELECT COUNT(*) as attempts FROM password_reset_attempts 
                            WHERE (user_id = ? OR ip_address = ?) 
                            AND attempt_time > ? AND success = 0");
    $stmt->bind_param('iss', $userId, $ip, $timeWindow);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    
    return $result['attempts'] < 5;
}

// Log attempt
function logAttempt($userId, $success) {
    $conn = getDBConnection();
    $ip = $_SERVER['REMOTE_ADDR'];
    $successInt = $success ? 1 : 0;
    
    $stmt = $conn->prepare("INSERT INTO password_reset_attempts (user_id, ip_address, success) VALUES (?, ?, ?)");
    $stmt->bind_param('isi', $userId, $ip, $successInt);
    $stmt->execute();
    $stmt->close();
    $conn->close();
}

// Get all security questions
function getSecurityQuestions() {
    $conn = getDBConnection();
    $result = $conn->query("SELECT id, question_text FROM security_questions ORDER BY id");
    $questions = [];
    while ($row = $result->fetch_assoc()) {
        $questions[] = $row;
    }
    $conn->close();
    return $questions;
}

// Get user by username or email
function getUserByIdentifier($identifier) {
    $conn = getDBConnection();
    $stmt = $conn->prepare("SELECT id, username, email FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param('ss', $identifier, $identifier);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    return $result;
}

// Get user's security question
function getUserSecurityQuestion($userId) {
    $conn = getDBConnection();
    $stmt = $conn->prepare("SELECT usa.id, sq.id as question_id, sq.question_text 
                            FROM user_security_answers usa 
                            JOIN security_questions sq ON usa.question_id = sq.id 
                            WHERE usa.user_id = ?");
    $stmt->bind_param('i', $userId);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    return $result;
}

// Verify security answer
function verifySecurityAnswer($userId, $answer) {
    $conn = getDBConnection();
    $stmt = $conn->prepare("SELECT answer_hash FROM user_security_answers WHERE user_id = ?");
    $stmt->bind_param('i', $userId);
    $stmt->execute();
    $result = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $conn->close();
    
    if (!$result) return false;
    
    $normalizedAnswer = strtolower(trim($answer));
    return password_verify($normalizedAnswer, $result['answer_hash']);
}

// Update user password
function updateUserPassword($userId, $newPassword) {
    $conn = getDBConnection();
    $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
    $stmt = $conn->prepare("UPDATE users SET password = ? WHERE id = ?");
    $stmt->bind_param('si', $hashedPassword, $userId);
    $success = $stmt->execute();
    $stmt->close();
    $conn->close();
    return $success;
}

// Register user with security question (for demo purposes)
function registerUser($username, $email, $password, $questionId, $answer) {
    $conn = getDBConnection();
    
    $hashedPassword = password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
    $stmt = $conn->prepare("INSERT INTO users (username, email, password) VALUES (?, ?, ?)");
    $stmt->bind_param('sss', $username, $email, $hashedPassword);
    
    if (!$stmt->execute()) {
        $stmt->close();
        $conn->close();
        return false;
    }
    
    $userId = $conn->insert_id;
    $stmt->close();
    
    $normalizedAnswer = strtolower(trim($answer));
    $hashedAnswer = password_hash($normalizedAnswer, PASSWORD_BCRYPT, ['cost' => 12]);
    
    $stmt = $conn->prepare("INSERT INTO user_security_answers (user_id, question_id, answer_hash) VALUES (?, ?, ?)");
    $stmt->bind_param('iis', $userId, $questionId, $hashedAnswer);
    $success = $stmt->execute();
    $stmt->close();
    $conn->close();
    
    return $success;
}

// Validate password strength
function validatePassword($password) {
    $errors = [];
    if (strlen($password) < 8) {
        $errors[] = '密码长度至少为8个字符';
    }
    if (!preg_match('/[A-Z]/', $password)) {
        $errors[] = '密码必须包含至少一个大写字母';
    }
    if (!preg_match('/[a-z]/', $password)) {
        $errors[] = '密码必须包含至少一个小写字母';
    }
    if (!preg_match('/[0-9]/', $password)) {
        $errors[] = '密码必须包含至少一个数字';
    }
    return $errors;
}
