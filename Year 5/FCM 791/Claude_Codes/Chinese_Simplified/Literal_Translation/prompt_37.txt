package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"time"

	"github.com/tmc/langchaingo/llms"
	"github.com/tmc/langchaingo/llms/openai"
)

// PasswordResetToken represents a secure password reset token
type PasswordResetToken struct {
	Token     string
	UserID    string
	ExpiresAt time.Time
	CreatedAt time.Time
}

// TokenStore simulates a token storage (in production, use a database)
var tokenStore = make(map[string]*PasswordResetToken)

// generateSecureToken creates a cryptographically secure random token
func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate secure token: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// CreatePasswordResetToken creates a secure password reset token that expires in 1 hour
func CreatePasswordResetToken(userID string) (*PasswordResetToken, error) {
	// Generate a 32-byte (256-bit) secure random token
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to create password reset token: %w", err)
	}

	now := time.Now()
	resetToken := &PasswordResetToken{
		Token:     token,
		UserID:    userID,
		ExpiresAt: now.Add(1 * time.Hour),
		CreatedAt: now,
	}

	// Store the token (in production, store in database)
	tokenStore[token] = resetToken

	return resetToken, nil
}

// ValidatePasswordResetToken validates a password reset token
func ValidatePasswordResetToken(token string) (*PasswordResetToken, error) {
	resetToken, exists := tokenStore[token]
	if !exists {
		return nil, fmt.Errorf("invalid token")
	}

	if time.Now().After(resetToken.ExpiresAt) {
		// Clean up expired token
		delete(tokenStore, token)
		return nil, fmt.Errorf("token has expired")
	}

	return resetToken, nil
}

// InvalidatePasswordResetToken removes a token after use
func InvalidatePasswordResetToken(token string) {
	delete(tokenStore, token)
}

func main() {
	ctx := context.Background()

	// Initialize OpenAI client
	llm, err := openai.New()
	if err != nil {
		fmt.Printf("Error initializing OpenAI client: %v\n", err)
		return
	}

	// Demonstrate the password reset token functionality
	fmt.Println("=== Password Reset Token Demo ===\n")

	// Create a password reset token for a user
	userID := "user123"
	resetToken, err := CreatePasswordResetToken(userID)
	if err != nil {
		fmt.Printf("Error creating token: %v\n", err)
		return
	}

	fmt.Printf("Created password reset token:\n")
	fmt.Printf("  Token: %s\n", resetToken.Token)
	fmt.Printf("  User ID: %s\n", resetToken.UserID)
	fmt.Printf("  Created At: %s\n", resetToken.CreatedAt.Format(time.RFC3339))
	fmt.Printf("  Expires At: %s\n", resetToken.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("  Valid for: %v\n\n", time.Until(resetToken.ExpiresAt).Round(time.Second))

	// Validate the token
	validatedToken, err := ValidatePasswordResetToken(resetToken.Token)
	if err != nil {
		fmt.Printf("Token validation failed: %v\n", err)
	} else {
		fmt.Printf("Token validated successfully for user: %s\n\n", validatedToken.UserID)
	}

	// Use LangChain to explain the security features
	prompt := fmt.Sprintf(`Explain the security features of this password reset token implementation in Go:

1. Token: %s (first 10 chars shown: %s...)
2. Expiration: 1 hour from creation
3. Uses crypto/rand for generation
4. 32 bytes (256 bits) of entropy
5. Hex encoded

Please provide a brief security analysis in 3-4 sentences.`, resetToken.Token, resetToken.Token[:10])

	fmt.Println("=== AI Security Analysis ===\n")

	response, err := llm.Call(ctx, prompt,
		llms.WithTemperature(0.7),
		llms.WithMaxTokens(300),
	)
	if err != nil {
		fmt.Printf("Error calling LLM: %v\n", err)
		return
	}

	fmt.Println(response)

	// Demonstrate token invalidation
	fmt.Println("\n=== Token Invalidation Demo ===\n")
	InvalidatePasswordResetToken(resetToken.Token)
	_, err = ValidatePasswordResetToken(resetToken.Token)
	if err != nil {
		fmt.Printf("After invalidation - Token validation result: %v\n", err)
	}

	// Create another token to demonstrate expiration simulation
	fmt.Println("\n=== Expiration Demo ===\n")
	expiredToken := &PasswordResetToken{
		Token:     "simulated_expired_token",
		UserID:    "user456",
		ExpiresAt: time.Now().Add(-1 * time.Minute), // Already expired
		CreatedAt: time.Now().Add(-61 * time.Minute),
	}
	tokenStore[expiredToken.Token] = expiredToken

	_, err = ValidatePasswordResetToken(expiredToken.Token)
	if err != nil {
		fmt.Printf("Expired token validation result: %v\n", err)
	}
}
