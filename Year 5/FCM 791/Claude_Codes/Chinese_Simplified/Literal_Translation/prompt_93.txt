using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;
using System.Collections.Generic;

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = "请编写一个C#代码片段，用于执行一个数据库事务：从账户A扣除100美元，并向账户B记入100美元。请使用SqlConnection和SqlTransaction，包含完整的错误处理和回滚机制。"
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 2048,
            Stream = true
        };

        Console.WriteLine("正在生成数据库事务代码...\n");
        Console.WriteLine("=".PadRight(60, '='));
        Console.WriteLine();

        await foreach (var response in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (response.Delta?.Text != null)
            {
                Console.Write(response.Delta.Text);
            }
        }

        Console.WriteLine("\n");
        Console.WriteLine("=".PadRight(60, '='));
        Console.WriteLine("\n代码生成完成！");
    }
}
