import anthropic

def count_lines_large_file(file_path: str) -> None:
    """
    Use Claude AI to generate and explain a Python script for counting lines
    in a very large file (10GB) efficiently using streaming.
    """
    client = anthropic.Anthropic()
    
    prompt = """请编写一个Python脚本，用于统计一个非常大的文件（10GB）的行数。
    
要求：
1. 不要将整个文件读入内存（避免内存溢出）
2. 使用高效的方式逐块读取文件
3. 显示进度信息
4. 处理可能的异常情况
5. 提供多种实现方法并比较它们的优缺点

请提供完整的、可运行的Python代码，并解释每种方法的工作原理。"""

    print("正在使用Claude AI生成大文件行数统计脚本...\n")
    print("=" * 60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "=" * 60)
    print("\n脚本生成完成！")


def demonstrate_efficient_line_counting():
    """
    Demonstrate an efficient way to count lines in a large file
    without loading it entirely into memory.
    """
    import os
    import time
    
    # Create a sample large file for demonstration
    sample_file = "sample_large_file.txt"
    
    print("\n创建示例文件用于演示...")
    
    # Create a moderately sized file for demonstration
    with open(sample_file, 'w', encoding='utf-8') as f:
        for i in range(100000):
            f.write(f"这是第 {i+1} 行内容，包含一些示例数据。\n")
    
    file_size = os.path.getsize(sample_file)
    print(f"示例文件已创建: {sample_file} ({file_size / 1024:.2f} KB)")
    
    # Method 1: Chunk-based reading
    print("\n方法1: 分块读取（推荐用于超大文件）")
    start_time = time.time()
    
    line_count = 0
    chunk_size = 1024 * 1024  # 1MB chunks
    
    with open(sample_file, 'rb') as f:
        chunk = f.read(chunk_size)
        while chunk:
            line_count += chunk.count(b'\n')
            chunk = f.read(chunk_size)
    
    elapsed_time = time.time() - start_time
    print(f"行数: {line_count}")
    print(f"耗时: {elapsed_time:.4f} 秒")
    
    # Method 2: Line-by-line iteration
    print("\n方法2: 逐行迭代")
    start_time = time.time()
    
    line_count = 0
    with open(sample_file, 'r', encoding='utf-8') as f:
        for _ in f:
            line_count += 1
    
    elapsed_time = time.time() - start_time
    print(f"行数: {line_count}")
    print(f"耗时: {elapsed_time:.4f} 秒")
    
    # Method 3: Using sum with generator
    print("\n方法3: 使用生成器和sum")
    start_time = time.time()
    
    with open(sample_file, 'r', encoding='utf-8') as f:
        line_count = sum(1 for _ in f)
    
    elapsed_time = time.time() - start_time
    print(f"行数: {line_count}")
    print(f"耗时: {elapsed_time:.4f} 秒")
    
    # Clean up
    os.remove(sample_file)
    print(f"\n示例文件已删除: {sample_file}")


def main():
    """Main function to run the demonstration."""
    print("大文件行数统计 - Python脚本生成器")
    print("=" * 60)
    
    # Use Claude to generate and explain the script
    count_lines_large_file("large_file.txt")
    
    # Demonstrate the actual implementation
    print("\n\n实际演示 - 高效文件行数统计方法")
    print("=" * 60)
    demonstrate_efficient_line_counting()
    
    print("\n\n总结：")
    print("对于10GB的超大文件，推荐使用分块读取方法（方法1），因为：")
    print("1. 内存使用固定（只需要chunk_size大小的内存）")
    print("2. 处理速度快（二进制读取，避免编码开销）")
    print("3. 可以轻松添加进度显示")
    print("4. 适合处理任意大小的文件")


if __name__ == "__main__":
    main()
