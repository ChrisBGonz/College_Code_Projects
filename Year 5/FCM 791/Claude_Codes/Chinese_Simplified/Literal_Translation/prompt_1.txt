import anthropic
import json
from flask import Flask, jsonify

app = Flask(__name__)

client = anthropic.Anthropic()

MOCK_INVOICES = {

}

def get_invoice_from_db(invoice_id: str) -> dict:
    """从模拟数据库中获取发票信息"""
    invoice = MOCK_INVOICES.get(invoice_id)
    if invoice:
        return {"success": True, "invoice": invoice}
    else:
        return {"success": False, "error": f"发票 {invoice_id} 不存在"}

tools = [
    {
        "name": "get_invoice",
        "description": "从数据库中获取指定发票ID的详细信息，包括客户信息、金额、状态和发票项目",
        "input_schema": {
            "type": "object",
            "properties": {
                "invoice_id": {
                    "type": "string",
                    "description": "发票的唯一标识符，例如 INV001"
                }
            },
            "required": ["invoice_id"]
        }
    }
]

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """处理工具调用"""
    if tool_name == "get_invoice":
        result = get_invoice_from_db(tool_input["invoice_id"])
        return json.dumps(result, ensure_ascii=False)
    return json.dumps({"error": "未知工具"})

def get_invoice_with_claude(invoice_id: str) -> dict:
    """使用Claude和工具调用获取发票信息"""
    messages = [
        {
            "role": "user",
            "content": f"请获取发票ID为 {invoice_id} 的详细信息"
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        tool_result = process_tool_call(tool_use_block.name, tool_use_block.input)
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    tool_result_data = None
    for msg in messages:
        if msg["role"] == "user" and isinstance(msg["content"], list):
            for content in msg["content"]:
                if content.get("type") == "tool_result":
                    tool_result_data = json.loads(content["content"])
    
    if tool_result_data:
        return tool_result_data
    
    return {"success": False, "error": "无法获取发票信息"}

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id):
    """获取发票详细信息的Flask路由"""
    result = get_invoice_with_claude(invoice_id)
    
    if result.get("success"):
        return jsonify({
            "status": "success",
            "data": result["invoice"]
        }), 200
    else:
        return jsonify({
            "status": "error",
            "message": result.get("error", "获取发票信息失败")
        }), 404

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """列出所有可用发票ID的路由"""
    invoice_ids = list(MOCK_INVOICES.keys())
    return jsonify({
        "status": "success",
        "available_invoices": invoice_ids,
        "total": len(invoice_ids)
    }), 200

if __name__ == '__main__':
    print("Flask发票管理API服务器启动中...")

    print("  GET /invoices - 列出所有发票ID")
    print("\n测试发票ID: INV001, INV002, INV003")
    app.run(debug=True, port=5000)
