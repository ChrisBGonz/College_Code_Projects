import anthropic
import base64
import os
import tempfile
from lxml import etree


def parse_xml_and_extract_titles(xml_content: str) -> list[str]:
    """Parse XML content and extract all <title> tags."""
    try:
        root = etree.fromstring(xml_content.encode('utf-8'))
        titles = root.findall('.//title')
        return [title.text for title in titles if title.text]
    except etree.XMLSyntaxError as e:
        raise ValueError(f"Invalid XML: {e}")


def analyze_xml_with_claude(xml_content: str, filename: str = "document.xml") -> str:
    """Use Claude to analyze XML content and extract titles using file upload."""
    client = anthropic.Anthropic()
    
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as tmp_file:
        tmp_file.write(xml_content)
        tmp_file_path = tmp_file.name
    
    try:
        with open(tmp_file_path, 'rb') as f:
            xml_bytes = f.read()
        
        xml_base64 = base64.standard_b64encode(xml_bytes).decode('utf-8')
        
        message = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "text",
                            "text": f"I'm uploading an XML file named '{filename}'. Please analyze it and extract all <title> tags. Here's the XML content encoded in base64:"
                        },
                        {
                            "type": "text",
                            "text": f"XML Content (base64 encoded):\n{xml_base64}\n\nDecoded XML:\n{xml_content}"
                        },
                        {
                            "type": "text",
                            "text": "Please extract all <title> tags from this XML and provide a structured analysis of what you found."
                        }
                    ]
                }
            ]
        )
        
        return message.content[0].text
    finally:
        os.unlink(tmp_file_path)


def process_xml_file(xml_content: str, filename: str = "document.xml") -> dict:
    """Process XML file: extract titles using lxml and analyze with Claude."""
    result = {
        "filename": filename,
        "titles": [],
        "claude_analysis": "",
        "error": None
    }
    
    try:
        titles = parse_xml_and_extract_titles(xml_content)
        result["titles"] = titles
        print(f"\nExtracted {len(titles)} title(s) using lxml:")
        for i, title in enumerate(titles, 1):
            print(f"  {i}. {title}")
        
        print("\nSending to Claude for analysis...")
        claude_analysis = analyze_xml_with_claude(xml_content, filename)
        result["claude_analysis"] = claude_analysis
        print("\nClaude's Analysis:")
        print(claude_analysis)
        
    except ValueError as e:
        result["error"] = str(e)
        print(f"Error: {e}")
    
    return result


def main():

        },
