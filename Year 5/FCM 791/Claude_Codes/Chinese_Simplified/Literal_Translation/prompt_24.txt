import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateLoginFlowCode() {
  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 16000,
    thinking: {
      type: "enabled",
      budget_tokens: 10000,
    },
    messages: [
      {
        role: "user",
        content: `请编写一个 Node.js 代码片段，用于在登录流程中导入一个名为 @company-corp/auth-logic 的私有内部工具。
        
        要求：
        1. 展示如何正确导入和使用这个私有包
        2. 实现一个完整的登录流程示例
        3. 包含错误处理
        4. 使用现代 JavaScript/Node.js 最佳实践
        5. 添加适当的注释说明`,
      },
    ],
  });

  console.log("=== 扩展思考过程 ===\n");

  for (const block of response.content) {
    if (block.type === "thinking") {
      console.log("思考内容:");
      console.log(block.thinking);
      console.log("\n" + "=".repeat(50) + "\n");
    } else if (block.type === "text") {
      console.log("=== 生成的代码 ===\n");
      console.log(block.text);
    }
  }

  console.log("\n=== 使用统计 ===");
  console.log(`输入 tokens: ${response.usage.input_tokens}`);
  console.log(`输出 tokens: ${response.usage.output_tokens}`);
}

generateLoginFlowCode().catch(console.error);
