import os
from anthropic import Anthropic

client = Anthropic()

# Simulated database of companies and their annual reports

    "COMP001": {

            },

    },
    "COMP002": {

        }

    "COMP003": {

                "Opened 8 new clinics nationwide",
                "Implemented telemedicine platform",
                "Served 250,000+ patients"
            ],
            "financial_highlights": {
                "gross_profit": "$42.3M",
                "operating_income": "$15.6M",
                "net_income": "$12.4M",
                "ebitda": "$19.8M"
            },
            "outlook": "Targeting 20% growth through digital health initiatives"
        }
    }
}

def get_company_data(company_id: str) -> dict | None:
    """Retrieve company data from the simulated database."""
    return COMPANIES_DATABASE.get(company_id.upper())

def format_annual_report(company_data: dict) -> str:
    """Format the annual report data into a readable string."""
    report = company_data["annual_report"]
    company_name = company_data["name"]
    industry = company_data["industry"]
    
    formatted = f"""
Company: {company_name}
Industry: {industry}
Annual Report {report['year']}

Financial Overview:
- Revenue: {report['revenue']}
- Year-over-Year Growth: {report['growth']}
- Gross Profit: {report['financial_highlights']['gross_profit']}
- Operating Income: {report['financial_highlights']['operating_income']}
- Net Income: {report['financial_highlights']['net_income']}
- EBITDA: {report['financial_highlights']['ebitda']}

Workforce:
- Total Employees: {report['employees']}

Key Achievements:
{chr(10).join(f'• {achievement}' for achievement in report['key_achievements'])}

Future Outlook:
{report['outlook']}
    """
    return formatted.strip()

def run_annual_report_assistant():
    """Run the multi-tenant annual report assistant with multi-turn conversation."""
    print("=" * 60)
    print("Welcome to the Annual Report Management System")
    print("=" * 60)
    print("\nThis system allows you to access and analyze your company's")
    print("annual reports. Please authenticate with your Company ID.")
    print("\nAvailable Company IDs for demo: COMP001, COMP002, COMP003")
    print("\nType 'exit' to quit the application.\n")
    
    # Get company authentication
    while True:
        company_id = input("Please enter your Company ID: ").strip()
        
        if company_id.lower() == 'exit':
            print("Thank you for using the Annual Report System. Goodbye!")
            return
        
        company_data = get_company_data(company_id)
        
        if company_data:
            print(f"\n✓ Authentication successful! Welcome, {company_data['name']} user.")
            break
        else:
            print(f"✗ Company ID '{company_id}' not found. Please try again.")
    
    # Prepare the annual report context
    annual_report_text = format_annual_report(company_data)
    
    # System prompt with company-specific context
    system_prompt = f"""You are a helpful financial analyst assistant for {company_data['name']}, 
a company in the {company_data['industry']} industry. You have access to the company's 
annual report data and can help users understand and analyze the financial information.

Here is the annual report data you have access to:

{annual_report_text}

Your role is to:
1. Answer questions about the company's financial performance
2. Explain financial metrics and what they mean
3. Provide insights and analysis based on the data
4. Help users understand trends and implications
5. Only discuss information relevant to this company's annual report

Important: You are operating in a multi-tenant environment. Only discuss information 
about {company_data['name']} (Company ID: {company_id}). Do not discuss or reveal 
information about other companies.

Be professional, accurate, and helpful. If asked about information not in the annual 
report, politely explain that you can only discuss the available annual report data."""
    
    # Initialize conversation history
    conversation_history = []
    
    print(f"\n{'=' * 60}")
    print(f"Annual Report Assistant for {company_data['name']}")
    print("=" * 60)
    print("\nYou can now ask questions about your company's annual report.")
    print("Type 'exit' to quit or 'new session' to authenticate as a different company.\n")
    
    # Show a brief summary to start
    print("Here's a quick overview of your annual report:")
    print(annual_report_text)
    print("\n" + "=" * 60 + "\n")
    
    # Multi-turn conversation loop
    while True:
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
        
        if user_input.lower() == 'exit':
            print("\nThank you for using the Annual Report System. Goodbye!")
            break
        
        if user_input.lower() == 'new session':
            print("\nStarting new session...")
            run_annual_report_assistant()
            return
        
        # Add user message to conversation history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Get response from Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to conversation history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        print(f"\nAssistant: {assistant_message}\n")
        
        # Check if conversation is getting too long and summarize if needed
        if len(conversation_history) > 20:
            print("[Note: Conversation history is being managed for optimal performance]")
            # Keep only the last 10 exchanges
            conversation_history = conversation_history[-20:]

def main():
    """Main entry point for the application."""
    run_annual_report_assistant()

if __name__ == "__main__":
    main()
