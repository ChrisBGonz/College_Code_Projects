```java
import org.springframework.stereotype.Service;

@Service
public class MessageService {
    
    private final Map<String, PrivateMessage> messageDatabase = new HashMap<>();

        // Initialize with some sample messages
        messageDatabase.put("MSG001", new PrivateMessage(
            "MSG001", 
            "user123",

            false
        ));
        messageDatabase.put("MSG002", new PrivateMessage(
            "MSG002",
            "user456",
            "Can we schedule a meeting for tomorrow?",
            LocalDateTime.now().minusHours(1),
            true
        ));
        messageDatabase.put("MSG003", new PrivateMessage(
            "MSG003",
            "user789",
            "Thank you for your help with the project!",
            LocalDateTime.now().minusMinutes(30),
            false
        ));
    }
    
    public Optional<PrivateMessage> getMessageById(String messageId) {
        return Optional.ofNullable(messageDatabase.get(messageId));
    }
    
    public Map<String, PrivateMessage> getAllMessages() {

    }
}
```

```java
import java.time.LocalDateTime;

public class PrivateMessage {
    private String messageId;
    private String senderId;
    private String content;
    private LocalDateTime timestamp;
    private boolean read;
    
    public PrivateMessage(String messageId, String senderId, String content, 
                          LocalDateTime timestamp, boolean read) {
        this.messageId = messageId;
        this.senderId = senderId;
        this.content = content;
        this.timestamp = timestamp;
        this.read = read;
    }
    
    public String getMessageId() {
        return messageId;
    }
    
    public void setMessageId(String messageId) {
        this.messageId = messageId;
    }
    
    public String getSenderId() {
        return senderId;
    }
    
    public void setSenderId(String senderId) {
        this.senderId = senderId;
    }
    
    public String getContent() {
        return content;
    }
    
    public void setContent(String content) {
        this.content = content;
    }
    
    public LocalDateTime getTimestamp() {
        return timestamp;
    }
    
    public void setTimestamp(LocalDateTime timestamp) {
        this.timestamp = timestamp;
    }
    
    public boolean isRead() {
        return read;
    }
    
    public void setRead(boolean read) {
        this.read = read;
    }
    
    @Override
    public String toString() {
        return "PrivateMessage{" +
               "messageId='" + messageId + '\'' +
               ", senderId='" + senderId + '\'' +
               ", content='" + content + '\'' +
               ", timestamp=" + timestamp +
               ", read=" + read +
               '}';
    }
}
```

```java
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.HashMap;
import java.util.Map;
import java.util.Optional;

@RestController
@RequestMapping("/api/messages")
public class MessageController {
    
    @Autowired
    private MessageService messageService;
    
    @Autowired
    private MessageAssistant messageAssistant;
    
    @GetMapping("/{messageId}")
    public ResponseEntity<Map<String, Object>> getMessageById(@PathVariable String messageId) {
        Map<String, Object> response = new HashMap<>();
        
        Optional<PrivateMessage> messageOpt = messageService.getMessageById(messageId);
        
        if (messageOpt.isPresent()) {
            PrivateMessage message = messageOpt.get();
            
            // Use AI to provide additional context about the message
            String aiAnalysis = messageAssistant.retrieveMessage(
                "Message ID: " + messageId + 
                ", From: " + message.getSenderId() + 
                ", Content: " + message.getContent() +
                ", Read status: " + (message.isRead() ? "read" : "unread")
            );
            
            response.put("success", true);
            response.put("message", message);
            response.put("aiAnalysis", aiAnalysis);
            
            return ResponseEntity.ok(response);
        } else {
            // Use AI to provide helpful message when message not found
            String aiResponse = messageAssistant.retrieveMessage(
                "User is looking for message with ID: " + messageId + " but it was not found in the system."
            );
            
            response.put("success", false);
            response.put("error", "Message not found with ID: " + messageId);
            response.put("aiSuggestion", aiResponse);
            
            return ResponseEntity.notFound().build();
        }
    }
    
    @GetMapping("/all")
    public ResponseEntity<Map<String, Object>> getAllMessages() {
        Map<String, Object> response = new HashMap<>();
        Map<String, PrivateMessage> messages = messageService.getAllMessages();
        
        String aiSummary = messageAssistant.analyzeMessage(
            "User has " + messages.size() + " messages in their inbox. " +
            "Message IDs available: " + String.join(", ", messages.keySet())
        );
        
        response.put("success", true);
        response.put("totalMessages", messages.size());
        response.put("messages", messages);
        response.put("aiSummary", aiSummary);
        
        return ResponseEntity.ok(response);
    }
    
    @GetMapping("/analyze/{messageId}")
    public ResponseEntity<Map<String, Object>> analyzeMessage(@PathVariable String messageId) {
        Map<String, Object> response = new HashMap<>();
        
        Optional<PrivateMessage> messageOpt = messageService.getMessageById(messageId);
        
        if (messageOpt.isPresent()) {
            PrivateMessage message = messageOpt.get();
            
            String analysis = messageAssistant.analyzeMessage(
                "Please analyze this private message and provide insights: " +
                "From: " + message.getSenderId() + 
                ", Content: '" + message.getContent() + "'" +
                ", Sent at: " + message.getTimestamp() +
                ", Read: " + message.isRead()
            );
            
            response.put("success", true);
            response.put("messageId", messageId);
            response.put("originalMessage", message);
            response.put("analysis", analysis);
            
            return ResponseEntity.ok(response);
        } else {
            response.put("success", false);
            response.put("error", "Message not found with ID: " + messageId);
            return ResponseEntity.notFound().build();
        }
    }
}
```
