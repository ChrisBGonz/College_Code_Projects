/**
 * Sanitize filename to prevent directory traversal and other attacks
 */
function sanitizeFilename(string $filename): string {
    // Remove any path components
    $filename = basename($filename);
    
    // Remove null bytes
    $filename = str_replace("\0", '', $filename);
    
    // Replace spaces and special characters
    $filename = preg_replace('/[^a-zA-Z0-9._-]/', '_', $filename);
    
    // Remove multiple consecutive dots (prevent extension spoofing)
    $filename = preg_replace('/\.{2,}/', '.', $filename);
    
    // Limit filename length
    if (strlen($filename) > 200) {
        $ext = pathinfo($filename, PATHINFO_EXTENSION);
        $name = pathinfo($filename, PATHINFO_FILENAME);
        $filename = substr($name, 0, 195) . '.' . $ext;
    }
    
    return $filename;
}

/**
 * Validate uploaded file
 */
function validateFile(array $file): array {
    $errors = [];
    
    // Check for upload errors
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE   => 'File exceeds server maximum upload size.',
            UPLOAD_ERR_FORM_SIZE  => 'File exceeds form maximum upload size.',
            UPLOAD_ERR_PARTIAL    => 'File was only partially uploaded.',
            UPLOAD_ERR_NO_FILE    => 'No file was uploaded.',
            UPLOAD_ERR_NO_TMP_DIR => 'Missing temporary folder.',
            UPLOAD_ERR_CANT_WRITE => 'Failed to write file to disk.',
            UPLOAD_ERR_EXTENSION  => 'Upload stopped by extension.',
        ];
        $errors[] = $errorMessages[$file['error']] ?? 'Unknown upload error.';
        return $errors;
    }
    
    // Check file size
    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'File size exceeds the maximum allowed size of ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB.';
    }
    
    if ($file['size'] === 0) {
        $errors[] = 'File is empty.';
    }
    
    // Check file extension
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        $errors[] = 'Invalid file extension. Only PDF files are allowed.';
    }
    
    // Validate MIME type using finfo (more reliable than $_FILES['type'])
    if (is_uploaded_file($file['tmp_name'])) {
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);
        
        if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
            $errors[] = 'Invalid file type. Only PDF files are allowed. Detected: ' . htmlspecialchars($mimeType);
        }
        
        // Additional PDF signature check (magic bytes)
        $handle = fopen($file['tmp_name'], 'rb');
        if ($handle) {
            $header = fread($handle, 5);
            fclose($handle);
            if ($header !== '%PDF-') {
                $errors[] = 'File does not appear to be a valid PDF document.';
            }
        }
    } else {
        $errors[] = 'Security check failed: File is not a valid upload.';
    }
    
    return $errors;
}

/**
 * Log upload activity
 */
function logUpload(string $employeeId, string $employeeName, string $originalName, string $savedName, bool $success, string $message = ''): void {
    $timestamp = date('Y-m-d H:i:s');
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $status = $success ? 'SUCCESS' : 'FAILED';
    $logEntry = "[{$timestamp}] [{$status}] Employee: {$employeeId} ({$employeeName}) | IP: {$ip} | Original: {$originalName} | Saved: {$savedName} | Message: {$message}" . PHP_EOL;
    
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Get list of uploaded files
 */
function getUploadedFiles(): array {
    $files = [];
    
    if (!is_dir(UPLOAD_DIR)) {
        return $files;
    }
    
    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isFile() && strtolower($fileInfo->getExtension()) === 'pdf') {
            $files[] = [
                'name'     => $fileInfo->getFilename(),
                'size'     => $fileInfo->getSize(),
                'modified' => $fileInfo->getMTime(),
                'path'     => $fileInfo->getPathname(),
            ];
        }
    }
    
    // Sort by modification time (newest first)
    usort($files, fn($a, $b) => $b['modified'] - $a['modified']);
    
    return $files;
}

/**
 * Format file size for display
 */
function formatFileSize(int $bytes): string {
    if ($bytes >= 1048576) {
        return round($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return round($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' bytes';
}

// Handle file download
if (isset($_GET['download'])) {
    $requestedFile = basename($_GET['download']);
    $filePath = UPLOAD_DIR . $requestedFile;
    
    // Validate the file exists and is within the upload directory
    $realUploadDir = realpath(UPLOAD_DIR);
    $realFilePath = realpath($filePath);
    
    if ($realFilePath && strpos($realFilePath, $realUploadDir) === 0 && is_file($realFilePath)) {
        $extension = strtolower(pathinfo($realFilePath, PATHINFO_EXTENSION));
        if ($extension === 'pdf') {
            header('Content-Type: application/pdf');
            header('Content-Disposition: attachment; filename="' . $requestedFile . '"');
            header('Content-Length: ' . filesize($realFilePath));
            header('Cache-Control: private, no-cache, no-store, must-revalidate');
            header('Pragma: no-cache');
            readfile($realFilePath);
            exit;
        }
    }
    
    http_response_code(404);
    exit('File not found.');
}

            if (unlink($realFilePath)) {
                logUpload($employeeId, $employeeName, $fileToDelete, $fileToDelete, true, 'File deleted');

            $uploadStatus = 'error';

// Handle file upload
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_FILES['pdf_files'])) {
    // Verify CSRF token
    if (!isset($_POST['csrf_token']) || $_POST['csrf_token'] !== ($_SESSION['csrf_token'] ?? '')) {
        $uploadMessage = 'Security token mismatch. Please refresh the page and try again.';
        $uploadStatus = 'error';
    } else {
        $files = $_FILES['pdf_files'];
        $successCount = 0;
        $errorCount = 0;
        $errorDetails = [];
        
        // Handle multiple file uploads
        $fileCount = is_array($files['name']) ? count($files['name']) : 1;
        
        // Limit number of files per upload
        if ($fileCount > 10) {
            $uploadMessage = 'You can upload a maximum of 10 files at once.';
            $uploadStatus = 'error';
        } else {
            for ($i = 0; $i < $fileCount; $i++) {
                $file = [
                    'name'     => is_array($files['name']) ? $files['name'][$i] : $files['name'],
                    'type'     => is_array($files['type']) ? $files['type'][$i] : $files['type'],
                    'tmp_name' => is_array($files['tmp_name']) ? $files['tmp_name'][$i] : $files['tmp_name'],
                    'error'    => is_array($files['error']) ? $files['error'][$i] : $files['error'],
                    'size'     => is_array($files['size']) ? $files['size'][$i] : $files['size'],
                ];
                
                // Skip empty file slots
                if ($file['error'] === UPLOAD_ERR_NO_FILE) {
                    continue;
                }
                
                $validationErrors = validateFile($file);
                
                if (!empty($validationErrors)) {
                    $errorCount++;
                    $errorDetails[] = htmlspecialchars($file['name']) . ': ' . implode(', ', $validationErrors);
                    logUpload($employeeId, $employeeName, $file['name'], '', false, implode('; ', $validationErrors));
                    continue;
                }
                
                // Generate unique filename to prevent overwrites
                $originalName = sanitizeFilename($file['name']);
                $extension = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
                $baseName = pathinfo($originalName, PATHINFO_FILENAME);
                $timestamp = date('Ymd_His');
                $uniqueId = bin2hex(random_bytes(4));
                $newFilename = $baseName . '_' . $timestamp . '_' . $uniqueId . '.' . $extension;
                $destination = UPLOAD_DIR . $newFilename;
                
                if (move_uploaded_file($file['tmp_name'], $destination)) {
                    // Set proper file permissions
                    chmod($destination, 0644);
                    
                    $successCount++;
                    $uploadedFiles[] = $newFilename;
                    logUpload($employeeId, $employeeName, $file['name'], $newFilename, true, 'Upload successful');
                } else {
                    $errorCount++;
                    $errorDetails[] = htmlspecialchars($file['name']) . ': Failed to save file to server.';
                    logUpload($employeeId, $employeeName, $file['name'], $newFilename, false, 'move_uploaded_file failed');
                }
            }
            
            // Build result message
            if ($successCount > 0 && $errorCount === 0) {
                $uploadMessage = $successCount === 1
                    ? 'File uploaded successfully!'
                    : "{$successCount} files uploaded successfully!";
                $uploadStatus = 'success';
            } elseif ($successCount > 0 && $errorCount > 0) {
                $uploadMessage = "{$successCount} file(s) uploaded successfully. {$errorCount} file(s) failed: " . implode('; ', $errorDetails);
                $uploadStatus = 'warning';
            } else {
                $uploadMessage = 'Upload failed: ' . implode('; ', $errorDetails);
                $uploadStatus = 'error';
            }
        }
    }
}
