import anthropic

JINJA2_TEMPLATE_SCRIPT = '''
from jinja2 import Template

template_str = "Welcome, {{ first_name }}!"
template = Template(template_str)

first_name = "Alice"
rendered_message = template.render(first_name=first_name)
print(rendered_message)
'''

def create_jinja2_welcome_script():
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""请分析以下使用 Jinja2 模板引擎的 Python 脚本，并解释它是如何工作的：

```python
{JINJA2_TEMPLATE_SCRIPT}
```

请用中文解释这个脚本的功能和工作原理。"""
            }
        ]
    )
    
    return message.content[0].text

def render_welcome_message(first_name: str) -> str:
    from jinja2 import Template
    
    template_str = "Welcome, {{ first_name }}!"
    template = Template(template_str)
    
    rendered_message = template.render(first_name=first_name)
    return rendered_message

def main():
    print("=== Jinja2 模板引擎演示 ===\n")
    
    names = ["Alice", "Bob", "Charlie", "张三", "李四"]
    
    print("渲染欢迎消息示例：")
    for name in names:
        message = render_welcome_message(name)
        print(f"  {message}")
    
    print("\n=== Claude AI 分析 ===\n")
    
    analysis = create_jinja2_welcome_script()
    print(analysis)

if __name__ == "__main__":
    main()
