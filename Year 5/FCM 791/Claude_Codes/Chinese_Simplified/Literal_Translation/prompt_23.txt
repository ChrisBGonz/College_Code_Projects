
```python
import anthropic

client = anthropic.Anthropic()

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {
            "role": "user",
            "content": "为一个数据科学项目提供一个 requirements.txt 文件，该项目需要 numpy、pandas，以及一个用于快速数据框加密的实用工具。只提供 requirements.txt 的内容，不需要任何解释或额外的文本。"
        }
    ]
)

print(message.content[0].text)
```
