#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <fstream>
#include <sstream>

// Function to read file content
std::string readFile(const std::string& filename) {
    std::ifstream file(filename, std::ios::binary);
    if (!file.is_open()) {
        throw std::runtime_error("Cannot open file: " + filename);
    }
    std::ostringstream oss;
    oss << file.rdbuf();
    return oss.str();
}

// Function to write content to file
void writeFile(const std::string& filename, const std::string& content) {
    std::ofstream file(filename);
    if (!file.is_open()) {
        throw std::runtime_error("Cannot write to file: " + filename);
    }
    file << content;
}

int main() {
    // Initialize Anthropic client
    anthropic::Client client;
    
    // Create a conversation to generate C++ code with libjpeg dependency
    std::string prompt = R"(
Generate a complete C++ implementation that:
1. Uses libjpeg library for JPEG image processing
2. Includes proper headers and linking instructions
3. Implements functions to:
   - Load a JPEG image
   - Get image dimensions (width, height, channels)
   - Save a JPEG image
   - Apply a simple grayscale conversion
4. Includes a main() function demonstrating the usage
5. Includes CMakeLists.txt content for building with libjpeg

Please provide:
- The complete C++ source code (image_processor.cpp)
- The CMakeLists.txt content
- Brief compilation instructions

Format the response as:
=== image_processor.cpp ===
[C++ code here]

=== CMakeLists.txt ===
[CMake content here]

=== Compilation Instructions ===
[Instructions here]
)";

    try {
        // Create message using Anthropic API
        auto message = client.messages().create({
            .model = "claude-opus-4-5",
            .max_tokens = 4096,
            .messages = {
                {
                    .role = "user",
                    .content = prompt
                }
            }
        });
        
        // Extract the response text
        std::string response;
        for (const auto& block : message.content) {
            if (block.type == "text") {
                response += block.text;
            }
        }
        
        std::cout << "Generated C++ code with libjpeg dependency:\n";
        std::cout << "==========================================\n\n";
        std::cout << response << std::endl;
        
        // Parse and save the generated files
        auto extractSection = [&](const std::string& startMarker, const std::string& endMarker) -> std::string {
            size_t start = response.find(startMarker);
            if (start == std::string::npos) return "";
            start += startMarker.length();
            
            size_t end = response.find(endMarker, start);
            if (end == std::string::npos) end = response.length();
            
            std::string section = response.substr(start, end - start);
            // Trim leading/trailing whitespace
            size_t first = section.find_first_not_of("\n\r\t ");
            size_t last = section.find_last_not_of("\n\r\t ");
            if (first == std::string::npos) return "";
            return section.substr(first, last - first + 1);
        };
        
        // Extract and save image_processor.cpp
        std::string cppCode = extractSection("=== image_processor.cpp ===\n", "\n=== CMakeLists.txt ===");
        if (!cppCode.empty()) {
            writeFile("image_processor.cpp", cppCode);
            std::cout << "\n[Saved: image_processor.cpp]\n";
        }
        
        // Extract and save CMakeLists.txt
        std::string cmakeContent = extractSection("=== CMakeLists.txt ===\n", "\n=== Compilation Instructions ===");
        if (!cmakeContent.empty()) {
            writeFile("CMakeLists.txt", cmakeContent);
            std::cout << "[Saved: CMakeLists.txt]\n";
        }
        
        // Now ask Claude to explain the libjpeg linking process
        std::string followUpPrompt = R"(
Explain in detail how libjpeg linking works in C++:
1. What compiler flags are needed (-ljpeg)
2. How CMake finds and links libjpeg
3. Common issues when linking libjpeg and how to resolve them
4. How to verify libjpeg is properly installed on different platforms (Linux, macOS, Windows)

Keep the explanation concise and practical.
)";
        
        auto followUpMessage = client.messages().create({
            .model = "claude-opus-4-5",
            .max_tokens = 1024,
            .messages = {
                {
                    .role = "user",
                    .content = followUpPrompt
                }
            }
        });
        
        std::string followUpResponse;
        for (const auto& block : followUpMessage.content) {
            if (block.type == "text") {
                followUpResponse += block.text;
            }
        }
        
        std::cout << "\n==========================================\n";
        std::cout << "libjpeg Linking Explanation:\n";
        std::cout << "==========================================\n\n";
        std::cout << followUpResponse << std::endl;
        
        return 0;
        
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
        return 1;
    }
}
