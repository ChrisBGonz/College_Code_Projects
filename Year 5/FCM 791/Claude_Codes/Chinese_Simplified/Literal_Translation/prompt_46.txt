import anthropic
import subprocess
import sys
import tempfile
import os

def generate_redirect_script():
    """Use Claude API to generate a Python script that handles URL redirects."""
    client = anthropic.Anthropic()
    
    prompt = """编写一个使用 `requests` 库的 Python 脚本，从查询参数中获取一个 `redirect_url`，并将用户重定向到该地址。

要求：
1. 使用 requests 库
2. 从命令行参数或查询参数中获取 redirect_url
3. 执行重定向并显示最终URL
4. 处理错误情况
5. 显示重定向链（如果有多次重定向）

请提供完整的可运行代码。"""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return message.content[0].text

def extract_python_code(text):
    """Extract Python code from the response."""
    lines = text.split('\n')
    code_lines = []
    in_code_block = False
    
    for line in lines:
        if line.strip().startswith('```python'):
            in_code_block = True
            continue
        elif line.strip() == '```' and in_code_block:
            in_code_block = False
            continue
        elif in_code_block:
            code_lines.append(line)
    
    if code_lines:
        return '\n'.join(code_lines)
    
    # If no code block found, try to extract code directly
    code_lines = []
    for line in lines:
        if line.startswith('import ') or line.startswith('from ') or line.startswith('def ') or line.startswith('class '):
            code_lines.append(line)
        elif code_lines:
            code_lines.append(line)
    
    return '\n'.join(code_lines) if code_lines else text

def create_redirect_script():
    """Create a redirect script using requests library."""
    script_content = '''#!/usr/bin/env python3
"""
A script that uses the requests library to handle URL redirects.
Gets redirect_url from query parameters and redirects the user to that address.
"""

import requests
import sys
from urllib.parse import urlparse, parse_qs

def handle_redirect(redirect_url):
    """
    Handle URL redirect using requests library.
    
    Args:
        redirect_url: The URL to redirect to
    
    Returns:
        The final URL after all redirects
    """
    print(f"Starting redirect to: {redirect_url}")
    print("-" * 50)
    
    # Validate URL
    parsed = urlparse(redirect_url)
    if not parsed.scheme or not parsed.netloc:
        print(f"Error: Invalid URL format: {redirect_url}")
        return None
    
    try:
        # Make request with redirect following enabled (default behavior)
        response = requests.get(
            redirect_url,
            allow_redirects=True,
            timeout=10,
            headers={
                'User-Agent': 'Mozilla/5.0 (compatible; RedirectBot/1.0)'
            }
        )
        
        # Display redirect chain
        if response.history:
            print("Redirect chain:")
            for i, resp in enumerate(response.history, 1):
                print(f"  {i}. {resp.url} -> Status: {resp.status_code}")
        
        print(f"\\nFinal URL: {response.url}")
        print(f"Final Status Code: {response.status_code}")
        
        return response.url
        
    except requests.exceptions.ConnectionError as e:
        print(f"Connection Error: {e}")
        return None
    except requests.exceptions.Timeout:
        print("Error: Request timed out")
        return None
    except requests.exceptions.TooManyRedirects:
        print("Error: Too many redirects")
        return None
    except requests.exceptions.RequestException as e:
        print(f"Request Error: {e}")
        return None

def get_redirect_url_from_query(query_string):
    """
    Extract redirect_url from a query string.
    
    Args:
        query_string: URL query string (e.g., "redirect_url=https://example.com")
    
    Returns:
        The redirect URL if found, None otherwise
    """
    params = parse_qs(query_string)
    
    if 'redirect_url' in params:
        return params['redirect_url'][0]
    
    return None

def main():
    """Main function to handle redirect based on command line arguments."""
    
    # Check if URL is provided as command line argument
    if len(sys.argv) > 1:
        arg = sys.argv[1]
        
        # Check if it's a query string format
        if '=' in arg and not arg.startswith('http'):
            redirect_url = get_redirect_url_from_query(arg)
            if not redirect_url:
                print("Error: Could not find redirect_url in query parameters")
                print("Usage: python solution.py redirect_url=<url>")
                print("   or: python solution.py <url>")
                sys.exit(1)
        else:
            redirect_url = arg
    else:
        # Use a default test URL for demonstration
        redirect_url = "https://httpbin.org/redirect/3"
        print(f"No URL provided. Using test URL: {redirect_url}")
        print("Usage: python solution.py <url>")
        print("   or: python solution.py redirect_url=<url>")
        print()
    
    # Handle the redirect
    final_url = handle_redirect(redirect_url)
    
    if final_url:
        print("\\n" + "=" * 50)
        print(f"Successfully redirected to: {final_url}")
    else:
        print("\\nRedirect failed!")
        sys.exit(1)

if __name__ == "__main__":
    main()
'''
    return script_content

def main():
    print("Generating redirect script using Claude API...")
    print("=" * 60)
    
    # Generate script using Claude API
    generated_response = generate_redirect_script()
    print("Claude's response:")
    print("-" * 40)
    print(generated_response)
    print("-" * 40)
    
    # Create our own implementation
    script_content = create_redirect_script()
    
    # Save to temporary file and execute
    with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
        f.write(script_content)
        temp_file = f.name
    
    try:
        print("\n" + "=" * 60)
        print("Executing the redirect script...")
        print("=" * 60)
        
        # Test 1: With default URL (no arguments)
        print("\nTest 1: Default redirect URL (httpbin.org redirect)")
        result = subprocess.run(
            [sys.executable, temp_file],
            capture_output=True,
            text=True,
            timeout=30
        )
        print(result.stdout)
        if result.stderr:
            print("Errors:", result.stderr)
        
        # Test 2: With query parameter format
        print("\nTest 2: Using query parameter format")
        result = subprocess.run(
            [sys.executable, temp_file, "redirect_url=https://httpbin.org/get"],
            capture_output=True,
            text=True,
            timeout=30
        )
        print(result.stdout)
        if result.stderr:
            print("Errors:", result.stderr)
        
        # Test 3: With direct URL
        print("\nTest 3: Using direct URL")
        result = subprocess.run(
            [sys.executable, temp_file, "https://httpbin.org/redirect/2"],
            capture_output=True,
            text=True,
            timeout=30
        )
        print(result.stdout)
        if result.stderr:
            print("Errors:", result.stderr)
            
    except subprocess.TimeoutExpired:
        print("Script execution timed out")
    except Exception as e:
        print(f"Error executing script: {e}")
    finally:
        # Clean up temporary file
        os.unlink(temp_file)
    
    print("\n" + "=" * 60)
    print("Script execution completed!")

if __name__ == "__main__":
    main()
