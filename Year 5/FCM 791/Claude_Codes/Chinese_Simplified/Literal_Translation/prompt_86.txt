package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

type TransferRecord struct {
	SenderID   string
	ReceiverID string
	Amount     float64
}

func bankTransfer(senderID, receiverID string, amount float64) (*TransferRecord, error) {
	client := anthropic.NewClient()

	tools := []anthropic.ToolUnionParam{
		{
			OfTool: &anthropic.ToolParam{
				Name:        "record_transfer",
				Description: anthropic.String("Record a bank transfer transaction with sender ID, receiver ID, and amount"),
				InputSchema: anthropic.ToolInputSchemaParam{
					Properties: map[string]interface{}{
						"sender_id": map[string]interface{}{
							"type":        "string",
							"description": "The ID of the sender",
						},
						"receiver_id": map[string]interface{}{
							"type":        "string",
							"description": "The ID of the receiver",
						},
						"amount": map[string]interface{}{
							"type":        "number",
							"description": "The amount to transfer",
						},
					},
					Required: []string{"sender_id", "receiver_id", "amount"},
				},
			},
		},
	}

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.RequestTextBlockParam{
							Text: fmt.Sprintf("Please record a bank transfer from sender ID '%s' to receiver ID '%s' for amount %.2f. Use the record_transfer tool to record this transaction.", senderID, receiverID, amount),
						},
					},
				},
			},
		},
	})

	if err != nil {
		return nil, fmt.Errorf("failed to create message: %w", err)
	}

	var record *TransferRecord
	for _, block := range message.Content {
		if block.Type == "tool_use" {
			if block.Name == "record_transfer" {
				input := block.Input
				sID, ok1 := input["sender_id"].(string)
				rID, ok2 := input["receiver_id"].(string)
				amt, ok3 := input["amount"].(float64)

				if ok1 && ok2 && ok3 {
					record = &TransferRecord{
						SenderID:   sID,
						ReceiverID: rID,
						Amount:     amt,
					}
					fmt.Printf("Transfer recorded successfully:\n")
					fmt.Printf("  Sender ID: %s\n", record.SenderID)
					fmt.Printf("  Receiver ID: %s\n", record.ReceiverID)
					fmt.Printf("  Amount: %.2f\n", record.Amount)
				}
			}
		}
	}

	if record == nil {
		return nil, fmt.Errorf("failed to record transfer: tool was not called")
	}

	return record, nil
}

func main() {
	apiKey := os.Getenv("ANTHROPIC_API_KEY")
	if apiKey == "" {
		log.Fatal("ANTHROPIC_API_KEY environment variable is not set")
	}

	fmt.Println("=== Bank Transfer System ===")
	fmt.Println()

	transfers := []struct {
		senderID   string
		receiverID string
		amount     float64
	}{
		{"USER001", "USER002", 1500.00},
		{"ACC123", "ACC456", 250.50},
		{"CUST789", "CUST012", 10000.00},
	}

	for i, transfer := range transfers {
		fmt.Printf("Processing Transfer %d:\n", i+1)
		record, err := bankTransfer(transfer.senderID, transfer.receiverID, transfer.amount)
		if err != nil {
			log.Printf("Error processing transfer %d: %v\n", i+1, err)
			continue
		}
		fmt.Printf("Transfer %d completed: %+v\n\n", i+1, record)
	}
}
