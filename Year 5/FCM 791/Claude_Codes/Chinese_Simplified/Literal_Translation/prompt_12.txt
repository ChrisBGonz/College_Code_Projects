
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

function question(prompt: string): Promise<string> {
  return new Promise((resolve) => {
    rl.question(prompt, resolve);
  });
}

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert Node.js and Express.js developer. When asked to create server configurations or code, provide complete, production-ready code with proper error handling, security best practices, and detailed comments explaining each component.",
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main(): Promise<void> {
  console.log("Express.js Server Configuration Assistant");
  console.log("==========================================");
  console.log(
    "I'll help you create a production-ready Express.js server configuration."
  );
  console.log('Type "exit" to quit.\n');

  // Initial request to generate the Express.js server configuration
  const initialRequest =
    "Please create a basic Express.js server configuration for production environment with error handling middleware. Include: 1) Security middleware (helmet, cors), 2) Request parsing middleware, 3) Rate limiting, 4) Logging middleware, 5) Custom error handling middleware with different error types, 6) Health check endpoint, 7) Graceful shutdown handling. Provide the complete code with detailed comments.";

  console.log("Generating Express.js server configuration...\n");

  try {
    const response = await chat(initialRequest);
    console.log("Assistant:", response);
    console.log("\n" + "=".repeat(50) + "\n");

    // Continue the conversation for follow-up questions
    while (true) {
      const userInput = await question("You: ");

      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        break;
      }

      if (!userInput.trim()) {
        continue;
      }

      try {
        const assistantResponse = await chat(userInput);
        console.log("\nAssistant:", assistantResponse);
        console.log("\n" + "=".repeat(50) + "\n");
      } catch (error) {
        console.error("Error communicating with Claude:", error);
      }
    }
  } catch (error) {
    console.error("Error generating initial configuration:", error);
    rl.close();
  }
}

main();
```
