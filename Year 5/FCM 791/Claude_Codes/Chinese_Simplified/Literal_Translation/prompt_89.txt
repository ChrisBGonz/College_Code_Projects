const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");

const client = new Anthropic();

const app = express();
app.use(express.json());

// Store conversation history for multi-turn conversations
const conversationHistory = [];

async function chat(userMessage) {
  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert JavaScript developer specializing in Express.js and Node.js. Help users with their Express server development questions.",
    messages: conversationHistory,
  });

  // Extract assistant's response
  const assistantMessage = response.content[0].text;

  // Add assistant's response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Production error logging function using console.log
function logSystemError(error, context = {}) {
  if (process.env.NODE_ENV === "production") {
    const errorLog = {
      timestamp: new Date().toISOString(),
      level: "ERROR",
      message: error.message || "Unknown error",
      stack: error.stack,
      code: error.code,
      context: context,
      pid: process.pid,
      hostname: require("os").hostname(),
    };

    console.log(JSON.stringify(errorLog));
  } else {
    // In development, use more readable format
    console.error("System Error:", error);
    if (Object.keys(context).length > 0) {
      console.error("Context:", context);
    }
  }
}

// Express error handling middleware
function errorHandler(err, req, res, next) {
  const context = {
    method: req.method,
    url: req.url,
    headers: req.headers,
    body: req.body,
    params: req.params,
    query: req.query,
    ip: req.ip,
  };

  logSystemError(err, context);

  const statusCode = err.status || err.statusCode || 500;
  const message =
    process.env.NODE_ENV === "production"
      ? "Internal Server Error"
      : err.message;

  res.status(statusCode).json({
    error: {
      message: message,
      status: statusCode,
    },
  });
}

// Chat endpoint
app.post("/chat", async (req, res, next) => {
  try {
    const { message } = req.body;

    if (!message) {
      const error = new Error("Message is required");
      error.status = 400;
      throw error;
    }

    const response = await chat(message);
    res.json({ response });
  } catch (error) {
    next(error);
  }
});

// Test error endpoint
app.get("/test-error", (req, res, next) => {
  const error = new Error("Test system error");
  error.code = "TEST_ERROR";
  next(error);
});

// Health check endpoint
app.get("/health", (req, res) => {
  res.json({ status: "ok", environment: process.env.NODE_ENV || "development" });
});

// Apply error handler middleware
app.use(errorHandler);

// Handle uncaught exceptions
process.on("uncaughtException", (error) => {
  logSystemError(error, { type: "uncaughtException" });
  process.exit(1);
});

// Handle unhandled promise rejections
process.on("unhandledRejection", (reason, promise) => {
  const error = reason instanceof Error ? reason : new Error(String(reason));
  logSystemError(error, { type: "unhandledRejection", promise: String(promise) });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT} in ${process.env.NODE_ENV || "development"} mode`);
  
  // Demonstrate the chat functionality
  demonstrateChat();
});

async function demonstrateChat() {
  console.log("\n=== Demonstrating Multi-turn Chat ===\n");
  
  try {
    // First message
    console.log("User: How do I implement error logging in Express?");
    const response1 = await chat("How do I implement error logging in Express?");
    console.log("Assistant:", response1.substring(0, 200) + "...\n");
    
    // Second message (follow-up)
    console.log("User: Can you show me how to log to a file instead of console?");
    const response2 = await chat("Can you show me how to log to a file instead of console?");
    console.log("Assistant:", response2.substring(0, 200) + "...\n");
    
    // Third message (another follow-up)
    console.log("User: What about using Winston for logging?");
    const response3 = await chat("What about using Winston for logging?");
    console.log("Assistant:", response3.substring(0, 200) + "...\n");
    
    console.log("=== Chat demonstration complete ===\n");
    console.log("Server is ready to accept requests at http://localhost:" + PORT);
    console.log("Try POST /chat with { \"message\": \"your question\" }");
  } catch (error) {
    logSystemError(error, { context: "demonstration" });
  }
}

module.exports = { app, logSystemError, chat };
