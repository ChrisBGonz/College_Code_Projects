
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];
let inactivityTimer = null;
let isLoggedIn = false;
let currentUser = null;
const INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds

function resetInactivityTimer() {
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
  }

  if (isLoggedIn) {
    inactivityTimer = setTimeout(() => {
      performAutoLogout();
    }, INACTIVITY_TIMEOUT);
  }
}

function performAutoLogout() {
  if (isLoggedIn) {
    console.log("\n⚠️  自动注销警告");
    console.log("由于15分钟内无活动，您已被自动注销。");
    console.log("请重新登录以继续使用银行服务。\n");

    isLoggedIn = false;
    currentUser = null;
    inactivityTimer = null;

    conversationHistory.push({
      role: "user",
      content: "系统通知：用户已因不活动超过15分钟而被自动注销。",
    });

    conversationHistory.push({
      role: "assistant",
      content:
        "您好，由于安全原因，您的会话已因15分钟内无活动而自动终止。请重新登录以访问您的银行账户。如需帮助，请联系客服。",
    });
  }
}

async function chat(userMessage) {
  if (!isLoggedIn && userMessage.toLowerCase().includes("登录")) {
    isLoggedIn = true;
    currentUser = "用户";
    resetInactivityTimer();
    console.log("✅ 登录成功！不活动计时器已启动（15分钟）");
  } else if (isLoggedIn) {
    resetInactivityTimer();
  }

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `你是一个专业的银行助手。你帮助用户处理银行相关事务，如查询余额、转账、账单支付等。
    
    当前用户状态：${isLoggedIn ? `已登录（用户：${currentUser}）` : "未登录"}
    
    安全提示：
    - 如果用户未登录，提醒他们需要先登录
    - 系统会在15分钟不活动后自动注销用户
    - 保护用户的账户安全是首要任务
    
    请用中文回复用户。`,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function getTimerStatus() {
  if (!isLoggedIn) {
    return "未登录";
  }
  return `已登录 - 不活动计时器运行中（15分钟后自动注销）`;
}

async function main() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("🏦 欢迎使用银行应用");
  console.log("================================");
  console.log("安全功能：15分钟不活动后自动注销");
  console.log("输入 '退出' 结束对话\n");

  const initialResponse = await chat(
    "你好，我想了解银行服务。请介绍一下如何登录和使用银行功能。"
  );
  console.log(`助手: ${initialResponse}\n`);

  const askQuestion = () => {
    const timerStatus = getTimerStatus();
    rl.question(`[${timerStatus}]\n您: `, async (input) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "退出") {
        if (inactivityTimer) {
          clearTimeout(inactivityTimer);
        }
        console.log("感谢使用银行服务，再见！");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        const response = await chat(userInput);
        console.log(`\n助手: ${response}\n`);
      } catch (error) {
        console.error("发生错误:", error.message);
      }

      askQuestion();
    });
  };

  askQuestion();
}

main();
```
