import os
import stripe
from flask import Flask, redirect, request, jsonify

app = Flask(__name__)

stripe.api_key = os.environ.get("STRIPE_SECRET_KEY")


@app.route("/create-checkout-session", methods=["POST"])
def create_checkout_session():
    """Create a Stripe checkout session and redirect user to payment page."""
    try:
        data = request.get_json() or {}
        
        amount = data.get("amount", 1000)
        currency = data.get("currency", "usd")
        product_name = data.get("product_name", "Test Product")
        success_url = data.get("success_url", "https://example.com/success")
        cancel_url = data.get("cancel_url", "https://example.com/cancel")
        
        checkout_session = stripe.checkout.Session.create(
            payment_method_types=["card"],
            line_items=[
                {
                    "price_data": {
                        "currency": currency,
                        "product_data": {
                            "name": product_name,
                        },
                        "unit_amount": amount,
                    },
                    "quantity": 1,
                }
            ],
            mode="payment",
            success_url=success_url,
            cancel_url=cancel_url,
        )
        
        return jsonify({
            "checkout_url": checkout_session.url,
            "session_id": checkout_session.id
        })
        
    except stripe.error.StripeError as e:
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        return jsonify({"error": str(e)}), 500


@app.route("/pay", methods=["GET"])
def pay():
    """Redirect user directly to payment gateway."""
    success_url = request.args.get("success_url", "https://example.com/success")
    cancel_url = request.args.get("cancel_url", "https://example.com/cancel")
    amount = int(request.args.get("amount", 1000))
    currency = request.args.get("currency", "usd")
    product_name = request.args.get("product_name", "Product")
    
    try:
        checkout_session = stripe.checkout.Session.create(
            payment_method_types=["card"],
            line_items=[
                {
                    "price_data": {
                        "currency": currency,
                        "product_data": {
                            "name": product_name,
                        },
                        "unit_amount": amount,
                    },
                    "quantity": 1,
                }
            ],
            mode="payment",
            success_url=success_url,
            cancel_url=cancel_url,
        )
        
        return redirect(checkout_session.url, code=303)
        
    except stripe.error.StripeError as e:
        return jsonify({"error": str(e)}), 400


@app.route("/webhook", methods=["POST"])
def webhook():
    """Handle Stripe webhook events."""
    payload = request.get_data()
    sig_header = request.headers.get("Stripe-Signature")
    webhook_secret = os.environ.get("STRIPE_WEBHOOK_SECRET")
    
    try:
        if webhook_secret:
            event = stripe.Webhook.construct_event(
                payload, sig_header, webhook_secret
            )
        else:
            event = stripe.Event.construct_from(
                request.get_json(), stripe.api_key
            )
        
        if event["type"] == "checkout.session.completed":
            session = event["data"]["object"]
            print(f"Payment successful! Session ID: {session['id']}")
            print(f"Customer email: {session.get('customer_details', {}).get('email', 'N/A')}")
            print(f"Amount paid: {session['amount_total']} {session['currency']}")
            
        elif event["type"] == "payment_intent.payment_failed":
            payment_intent = event["data"]["object"]
            print(f"Payment failed! PaymentIntent ID: {payment_intent['id']}")
            
        return jsonify({"status": "success"})
        
    except ValueError as e:
        return jsonify({"error": "Invalid payload"}), 400
    except stripe.error.SignatureVerificationError as e:
        return jsonify({"error": "Invalid signature"}), 400


@app.route("/", methods=["GET"])
def index():
    """Simple test page."""
    return """
    <html>
    <body>
        <h1>Payment Gateway Integration</h1>
        <p>Test the payment integration:</p>

            <button>Pay $10.00</button>
        </a>
        <br><br>
        <p>Or use the API endpoint POST /create-checkout-session with JSON body:</p>
        <pre>
{
    "amount": 1000,
    "currency": "usd",
    "product_name": "Test Product",
    "success_url": "https://example.com/success",
    "cancel_url": "https://example.com/cancel"
}
        </pre>
    </body>
    </html>
    """


if __name__ == "__main__":
    print("Starting Payment Gateway Integration Server...")
    print("Make sure STRIPE_SECRET_KEY environment variable is set")
    print("Server running at http://localhost:5000")
    app.run(debug=True, port=5000)
