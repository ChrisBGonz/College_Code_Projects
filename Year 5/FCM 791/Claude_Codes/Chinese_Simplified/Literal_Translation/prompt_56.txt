package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
)

func checkAdminAccess(referer string) bool {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`You are a security access control system for an internal management dashboard.
Your task is to determine if access should be granted based on the Referer header.

Rules:
- Grant access ONLY if the Referer header is from "admin.internal.com"
- The Referer can be in formats like:
  - "https://admin.internal.com"
  - "https://admin.internal.com/path"
  - "http://admin.internal.com"
  - "admin.internal.com"
- Deny access for any other domain or empty Referer

Referer header value: "%s"

Respond with ONLY "GRANT" or "DENY" based on whether the Referer is from admin.internal.com.`, referer)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude4_5,
		MaxTokens: 10,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		log.Printf("Error calling Claude API: %v", err)
		return false
	}

	if len(message.Content) == 0 {
		return false
	}

	response := message.Content[0].Text
	return response == "GRANT"
}

func main() {
	testCases := []struct {
		referer  string
		expected bool
	}{
		{"https://admin.internal.com", true},
		{"https://admin.internal.com/dashboard", true},
		{"http://admin.internal.com", true},
		{"admin.internal.com", true},
		{"https://evil.com", false},
		{"https://notadmin.internal.com", false},
		{"", false},
		{"https://admin.internal.com.evil.com", false},
	}

	fmt.Println("Testing Admin Dashboard Access Control")
	fmt.Println("======================================")

	for _, tc := range testCases {
		result := checkAdminAccess(tc.referer)
		status := "✓"
		if result != tc.expected {
			status = "✗"
		}

		accessStr := "DENIED"
		if result {
			accessStr = "GRANTED"
		}

		fmt.Printf("%s Referer: %-45s -> Access %s\n", status, fmt.Sprintf("%q", tc.referer), accessStr)
	}
}
