const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Logging middleware function
function loggingMiddleware(req, res, next) {
  const timestamp = new Date().toISOString();
  const logEntry = {
    timestamp,
    method: req.method,
    url: req.url,
    headers: req.headers,
    body: null,
  };

  // Collect request body
  let body = "";
  req.on("data", (chunk) => {
    body += chunk.toString();
  });

  req.on("end", () => {
    if (body) {
      try {
        logEntry.body = JSON.parse(body);
      } catch {
        logEntry.body = body;
      }
    }

    // Log the request
    console.log("\n=== Incoming Request ===");
    console.log(JSON.stringify(logEntry, null, 2));
    console.log("========================\n");

    // Store body for later use
    req.body = logEntry.body;
    req.rawBody = body;

    // Call next middleware
    if (next) {
      next(req, res);
    }
  });
}

// Function to analyze logs using Claude with streaming
async function analyzeLogsWithClaude(logData) {
  console.log("\n=== Analyzing logs with Claude (streaming) ===");

  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: `Please analyze this HTTP request log and provide insights about it:\n\n${JSON.stringify(logData, null, 2)}`,
  });

  let fullResponse = "";

  // Use streaming for real-time response
  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful debugging assistant that analyzes HTTP request logs. Provide clear, concise insights about the requests including potential issues, security concerns, and optimization suggestions.",
    messages: conversationHistory,
  });

  // Process the stream
  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  console.log("\n=== End of Analysis ===\n");

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: fullResponse,
  });

  return fullResponse;
}

// Function to ask follow-up questions about logs
async function askFollowUpQuestion(question) {
  console.log("\n=== Follow-up Question ===");
  console.log(`Question: ${question}`);

  // Add follow-up question to conversation history
  conversationHistory.push({
    role: "user",
    content: question,
  });

  let fullResponse = "";

  // Use streaming for real-time response
  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a helpful debugging assistant that analyzes HTTP request logs. Provide clear, concise insights about the requests including potential issues, security concerns, and optimization suggestions.",
    messages: conversationHistory,
  });

  // Process the stream
  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  console.log("\n=== End of Response ===\n");

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: fullResponse,
  });

  return fullResponse;
}

// Create a simple HTTP server to demonstrate the middleware
async function createDemoServer() {
  const server = http.createServer(async (req, res) => {
    // Apply logging middleware
    loggingMiddleware(req, res, async (req, res) => {
      // Handle different routes
      if (req.url === "/api/data" && req.method === "POST") {
        // Analyze the logged request with Claude
        const logData = {
          method: req.method,
          url: req.url,
          headers: req.headers,
          body: req.body,
        };

        await analyzeLogsWithClaude(logData);

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            status: "success",
            message: "Request logged and analyzed",
          })
        );
      } else if (req.url === "/api/users" && req.method === "GET") {
        const logData = {
          method: req.method,
          url: req.url,
          headers: req.headers,
        };

        await analyzeLogsWithClaude(logData);

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            users: [
              { id: 1, name: "Alice" },
              { id: 2, name: "Bob" },
            ],
          })
        );
      } else {
        res.writeHead(404, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ error: "Not found" }));
      }
    });
  });

  return server;
}

// Main function to demonstrate the middleware
async function main() {
  console.log("Starting Node.js Logging Middleware Demo with Claude AI\n");

  // Create and start the server
  const server = await createDemoServer();
  const PORT = 3000;

  server.listen(PORT, async () => {
    console.log(`Server running on port ${PORT}`);
    console.log("Simulating HTTP requests...\n");

    // Simulate some HTTP requests
    const http = require("http");

    // Simulate POST request with body
    const postData = JSON.stringify({
      username: "testuser",
      action: "login",
      timestamp: new Date().toISOString(),
    });

    const postOptions = {
      hostname: "localhost",
      port: PORT,
      path: "/api/data",
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Content-Length": Buffer.byteLength(postData),
        Authorization: "Bearer token123",
        "User-Agent": "TestClient/1.0",
      },
    };

    // Make POST request
    const postReq = http.request(postOptions, (res) => {
      let data = "";
      res.on("data", (chunk) => {
        data += chunk;
      });
      res.on("end", async () => {
        console.log("POST Response:", data);

        // Ask a follow-up question about the analyzed log
        await askFollowUpQuestion(
          "What security concerns do you see in this login request? Are there any best practices being followed or violated?"
        );

        // Simulate GET request
        const getOptions = {
          hostname: "localhost",
          port: PORT,
          path: "/api/users",
          method: "GET",
          headers: {
            Accept: "application/json",
            "X-Request-ID": "req-12345",
          },
        };

        const getReq = http.request(getOptions, (res) => {
          let data = "";
          res.on("data", (chunk) => {
            data += chunk;
          });
          res.on("end", async () => {
            console.log("GET Response:", data);

            // Ask another follow-up question
            await askFollowUpQuestion(
              "Comparing the POST and GET requests, what patterns do you notice? How could the API design be improved?"
            );

            // Close the server after demo
            server.close(() => {
              console.log("\nDemo completed. Server closed.");
              process.exit(0);
            });
          });
        });

        getReq.on("error", (err) => {
          console.error("GET request error:", err);
        });

        getReq.end();
      });
    });

    postReq.on("error", (err) => {
      console.error("POST request error:", err);
    });

    postReq.write(postData);
    postReq.end();
  });
}

// Run the main function
main().catch(console.error);
