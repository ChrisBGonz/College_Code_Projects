import anthropic
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


@app.route("/calculate", methods=["POST"])
def calculate():
    data = request.get_json()
    if not data or "expression" not in data:
        return {"error": "Please provide a mathematical expression in the request body"}, 400

    expression = data["expression"]

    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"Please calculate the following mathematical expression and provide the result. Show your work step by step if needed: {expression}",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(stream_with_context(generate()), content_type="text/plain")


@app.route("/calculate-simple", methods=["GET"])
def calculate_simple():
    expression = request.args.get("expr")
    if not expression:
        return {"error": "Please provide a mathematical expression using the 'expr' query parameter"}, 400

    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"Calculate this mathematical expression and provide just the numerical answer with brief explanation: {expression}",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(stream_with_context(generate()), content_type="text/plain")


@app.route("/", methods=["GET"])
def index():
    return {
        "message": "Mathematical Expression Calculator API",
        "endpoints": {
            "POST /calculate": "Calculate expression from JSON body {'expression': 'your expression'}",
            "GET /calculate-simple": "Calculate expression from query param ?expr=your+expression",
        },
        "examples": {
            "POST": {"expression": "2 + 3 * 4 - 1"},
            "GET": "/calculate-simple?expr=sqrt(144)+%2B+5^2",
        },
    }


if __name__ == "__main__":
    app.run(debug=True, port=5000)
