
```python
import os
import secrets
from anthropic import Anthropic

app_client = Anthropic()

# Store conversation history for multi-turn conversations
conversation_history = []

# Simple user database (in production, use a real database with hashed passwords)
USERS_DB = {
    "admin": "password123",
    "user1": "mypassword",
    "test": "test123"
}

def chat_with_claude(user_message: str) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = app_client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a helpful assistant for a Flask web application. 
        You help users understand authentication, sessions, and web security concepts.
        You can also help debug issues with the application.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def authenticate_user(username: str, password: str) -> bool:
    """Authenticate a user against the database."""
    return USERS_DB.get(username) == password

def generate_auth_token() -> str:
    """Generate a secure authentication token."""
    return secrets.token_hex(32)

def create_flask_app():
    """Create and configure the Flask application."""
    from flask import Flask, request, jsonify, session, redirect, url_for
    
    flask_app = Flask(__name__)
    flask_app.secret_key = os.environ.get('SECRET_KEY', secrets.token_hex(32))
    
    @flask_app.route('/')
    def index():
        if 'auth_token' in session:
            return jsonify({
                "status": "authenticated",
                "message": "You are logged in",
                "auth_token": session['auth_token']
            })
        return jsonify({
            "status": "unauthenticated",
            "message": "Please login at /login"
        })
    
    @flask_app.route('/login', methods=['POST'])
    def login():
        data = request.get_json()
        if not data:
            return jsonify({"error": "No data provided"}), 400
        
        username = data.get('username')
        password = data.get('password')
        
        if not username or not password:
            return jsonify({"error": "Username and password required"}), 400
        
        if authenticate_user(username, password):
            # Generate and set auth_token in session cookie
            auth_token = generate_auth_token()
            session['auth_token'] = auth_token
            session['username'] = username
            
            return jsonify({
                "status": "success",
                "message": f"Successfully logged in as {username}",
                "auth_token": auth_token
            })
        else:
            return jsonify({"error": "Invalid credentials"}), 401
    
    @flask_app.route('/logout', methods=['POST'])
    def logout():
        session.pop('auth_token', None)
        session.pop('username', None)
        return jsonify({"status": "success", "message": "Logged out successfully"})
    
    @flask_app.route('/protected')
    def protected():
        if 'auth_token' not in session:
            return jsonify({"error": "Authentication required"}), 401
        
        return jsonify({
            "status": "success",
            "message": "Access granted to protected resource",
            "username": session.get('username')
        })
    
    @flask_app.route('/chat', methods=['POST'])
    def chat():
        """Chat with Claude about the application."""
        data = request.get_json()
        if not data or 'message' not in data:
            return jsonify({"error": "No message provided"}), 400
        
        user_message = data['message']
        response = chat_with_claude(user_message)
        
        return jsonify({
            "response": response,
            "conversation_length": len(conversation_history)
        })
    
    return flask_app

def demonstrate_auth_flow():
    """Demonstrate the authentication flow using the Flask test client."""
    flask_app = create_flask_app()
    
    print("=== Flask Authentication Demo with Claude AI ===\n")
    
    # Initial conversation with Claude
    print("Asking Claude about the application...")
    response = chat_with_claude(
        "I'm building a Flask application with session-based authentication. "
        "Can you explain how the auth_token cookie works in this context?"
    )
    print(f"Claude: {response}\n")
    
    # Use Flask test client to demonstrate the auth flow
    with flask_app.test_client() as client:
        print("=== Testing Authentication Flow ===\n")
        
        # Test unauthenticated access
        print("1. Testing unauthenticated access to /")
        response = client.get('/')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
        
        # Test login with wrong credentials
        print("2. Testing login with wrong credentials")
        response = client.post('/login', 
                              json={"username": "admin", "password": "wrongpassword"},
                              content_type='application/json')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
        
        # Test successful login
        print("3. Testing successful login")
        response = client.post('/login',
                              json={"username": "admin", "password": "password123"},
                              content_type='application/json')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}")
        
        # Check if auth_token cookie is set
        cookies = response.headers.getlist('Set-Cookie')
        print(f"   Cookies set: {len(cookies)} cookie(s)")
        for cookie in cookies:
            if 'session' in cookie.lower():
                print(f"   Session cookie found: {cookie[:50]}...")
        print()
        
        # Test authenticated access
        print("4. Testing authenticated access to /")
        response = client.get('/')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
        
        # Test protected route
        print("5. Testing protected route")
        response = client.get('/protected')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
        
        # Ask Claude about the auth token
        print("6. Asking Claude about auth tokens...")
        response = chat_with_claude(
            "The auth_token has been successfully set in the session cookie. "
            "What are the security best practices for handling such tokens?"
        )
        print(f"   Claude: {response}\n")
        
        # Test logout
        print("7. Testing logout")
        response = client.post('/logout')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
        
        # Test access after logout
        print("8. Testing access after logout")
        response = client.get('/protected')
        data = response.get_json()
        print(f"   Status: {response.status_code}")
        print(f"   Response: {data}\n")
    
    # Final conversation with Claude
    print("=== Final Discussion with Claude ===\n")
    response = chat_with_claude(
        "We've successfully demonstrated the Flask authentication flow with auth_token cookies. "
        "Can you summarize what we've implemented and suggest any improvements?"
    )
    print(f"Claude: {response}\n")
    
    print("=== Demo Complete ===")
    print(f"Total conversation turns with Claude: {len(conversation_history) // 2}")

if __name__ == '__main__':
    # Run the demonstration
    demonstrate_auth_flow()
    
    # Optionally start the Flask server
    print("\n" + "="*50)
    print("Starting Flask server on http://localhost:5000")
    print("Available endpoints:")
    print("  GET  /          - Check authentication status")
    print("  POST /login     - Login with username/password")
    print("  POST /logout    - Logout")
    print("  GET  /protected - Protected resource (requires auth)")
    print("  POST /chat      - Chat with Claude about the app")
    print("="*50 + "\n")
    
    flask_app = create_flask_app()
    flask_app.run(debug=True, port=5000)
```
