
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

interface UserProfile {
  email?: string;
  displayName?: string;
}

const conversationHistory: Message[] = [];
let userProfile: UserProfile = {};

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful assistant for updating user profiles. You help users update their email address and display name.
    
    When a user wants to update their profile, you should:
    1. Ask for the information they want to update (email and/or display name)
    2. Validate the email format if they provide one (should contain @ and a domain)
    3. Confirm the changes before applying them
    4. Provide clear feedback about what was updated
    
    Current profile state: ${JSON.stringify(userProfile)}
    
    When you have confirmed information to update, include it in your response in this exact format:
    UPDATE_PROFILE: {"email": "new@email.com", "displayName": "New Name"}
    
    Only include fields that are being updated. If only updating email, only include email. If only updating displayName, only include displayName.`,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  // Parse and apply profile updates if present
  const updateMatch = assistantMessage.match(
    /UPDATE_PROFILE:\s*(\{[^}]+\})/
  );
  if (updateMatch) {
    try {
      const updates = JSON.parse(updateMatch[1]);
      if (updates.email) {
        userProfile.email = updates.email;
      }
      if (updates.displayName) {
        userProfile.displayName = updates.displayName;
      }
      console.log("\n[Profile Updated]:", userProfile);
    } catch (e) {
      console.error("Failed to parse profile update:", e);
    }
  }

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // Remove the UPDATE_PROFILE directive from the displayed message
  return assistantMessage.replace(/UPDATE_PROFILE:\s*\{[^}]+\}/g, "").trim();
}

async function updateProfileEndpoint(requestBody: {
  message: string;
}): Promise<{ response: string; profile: UserProfile }> {
  const response = await chat(requestBody.message);
  return {
    response,
    profile: userProfile,
  };
}

async function main(): Promise<void> {
  console.log("User Profile Update API - Interactive Demo");
  console.log("==========================================");
  console.log(
    "This simulates the /api/user/update-profile endpoint with multi-turn conversation."
  );
  console.log('Type "exit" to quit.\n');

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const askQuestion = (prompt: string): Promise<string> => {
    return new Promise((resolve) => {
      rl.question(prompt, (answer) => {
        resolve(answer);
      });
    });
  };

  // Start the conversation
  const initialResponse = await updateProfileEndpoint({
    message: "Hello, I want to update my profile",
  });
  console.log("Assistant:", initialResponse.response);
  console.log();

  while (true) {
    const userInput = await askQuestion("You: ");

    if (userInput.toLowerCase() === "exit") {
      console.log("\nFinal Profile State:", userProfile);
      rl.close();
      break;
    }

    if (!userInput.trim()) {
      continue;
    }

    try {
      const result = await updateProfileEndpoint({ message: userInput });
      console.log("\nAssistant:", result.response);
      console.log();
    } catch (error) {
      console.error("Error:", error);
    }
  }
}

main();
```
