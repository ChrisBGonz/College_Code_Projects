import anthropic
import subprocess
import sys
import importlib.util


def check_and_install_package(package_name: str, import_name: str = None) -> bool:
    """Check if a package is installed, if not install it."""
    if import_name is None:
        import_name = package_name
    
    spec = importlib.util.find_spec(import_name)
    if spec is None:
        print(f"Installing {package_name}...")
        subprocess.check_call([sys.executable, "-m", "pip", "install", package_name])
        return True
    return False


def convert_html_to_pdf(html_template: str, output_path: str = "output.pdf") -> str:
    """
    Convert HTML template to PDF file using external libraries.
    
    Args:
        html_template: HTML content as string
        output_path: Path where the PDF file will be saved
    
    Returns:
        Path to the generated PDF file
    """
    client = anthropic.Anthropic()
    
    # Use Claude to generate the Python code for HTML to PDF conversion
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Generate Python code to convert the following HTML to PDF using the weasyprint library.
                
HTML content:
{html_template}

Output path: {output_path}

Requirements:
1. Use weasyprint library
2. Save the PDF to the specified output path
3. Return the output path
4. Handle any potential errors
5. Only provide the code, no explanations

The code should be a complete, runnable Python script that:
- Imports weasyprint
- Converts the HTML string to PDF
- Saves it to {output_path}
- Prints success message with the output path"""
            }
        ]
    )
    
    # Extract the generated code
    generated_code = message.content[0].text
    
    # Clean up the code (remove markdown code blocks if present)
    if "```python" in generated_code:
        generated_code = generated_code.split("```python")[1].split("```")[0].strip()
    elif "```" in generated_code:
        generated_code = generated_code.split("```")[1].split("```")[0].strip()
    
    # Check and install weasyprint if needed
    check_and_install_package("weasyprint")
    
    # Execute the generated code
    exec_globals = {
        'html_template': html_template,
        'output_path': output_path
    }
    
    exec(generated_code, exec_globals)
    
    return output_path
