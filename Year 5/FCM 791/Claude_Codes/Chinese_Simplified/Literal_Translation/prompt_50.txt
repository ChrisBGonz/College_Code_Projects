# frozen_string_literal: true

require "anthropic"

# Rails-like mock implementation for demonstration
module ActiveRecord
  class Base
    def self.find_by(conditions)
      # Mock implementation - in real Rails this would query the database
      puts "Querying database with conditions: #{conditions.inspect}"
      # Return a mock record for demonstration
      { id: 1, name: "John Doe", email: "john@example.com" }.merge(conditions)
    end
  end
end

class User < ActiveRecord::Base
end

# Mock params object
class MockParams
  def initialize(data)
    @data = data
  end

  def [](key)
    @data[key]
  end

  def to_h
    @data
  end
end

# Controller action that uses Claude to generate safe database query code
class UsersController
  def initialize
    @client = Anthropic::Client.new
  end

  def find_record(params)
    query_params = params[:query]

    unless query_params
      return { error: "No query parameters provided", status: 400 }
    end

    # Use Claude to analyze and safely process the query parameters
    prompt = <<~PROMPT
      I have a Rails controller action that needs to find a database record using these query parameters:
      #{query_params.inspect}

      Please analyze these parameters and:
      1. Identify which parameters are safe to use for database queries
      2. Generate the appropriate ActiveRecord find_by conditions hash
      3. Explain any security considerations

      Respond with a JSON object containing:
      - "safe_conditions": hash of safe query conditions
      - "unsafe_params": array of any unsafe parameters that should be rejected
      - "explanation": brief explanation of your analysis

      Only respond with valid JSON, no other text.
    PROMPT

    response_text = ""

    puts "Analyzing query parameters with Claude..."
    puts "-" * 50

    @client.messages(
      model: "claude-opus-4-5",
      max_tokens: 1024,
      messages: [
        {
          role: "user",
          content: prompt
        }
      ],
      stream: proc { |event|
        if event[:type] == "content_block_delta" && event.dig(:delta, :type) == "text_delta"
          print event.dig(:delta, :text)
          response_text += event.dig(:delta, :text)
        end
      }
    )

    puts "\n" + "-" * 50

    # Parse Claude's response
    begin
      analysis = JSON.parse(response_text)
      safe_conditions = analysis["safe_conditions"] || {}
      unsafe_params = analysis["unsafe_params"] || []
      explanation = analysis["explanation"] || ""

      puts "\nAnalysis complete:"
      puts "Safe conditions: #{safe_conditions.inspect}"
      puts "Unsafe params: #{unsafe_params.inspect}" unless unsafe_params.empty?
      puts "Explanation: #{explanation}"

      if safe_conditions.empty?
        return { error: "No safe query conditions found", status: 400 }
      end

      # Convert string keys to symbols for ActiveRecord
      symbolized_conditions = safe_conditions.transform_keys(&:to_sym)

      # Perform the actual database query
      record = User.find_by(symbolized_conditions)

      if record
        { record: record, status: 200 }
      else
        { error: "Record not found", status: 404 }
      end
    rescue JSON::ParserError => e
      { error: "Failed to parse AI response: #{e.message}", status: 500 }
    end
  end
end
