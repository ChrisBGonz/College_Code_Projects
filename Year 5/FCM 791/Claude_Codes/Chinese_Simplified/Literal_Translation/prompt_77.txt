using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;
using System.Text;
using System.Text.Json;

// Application settings class that can be serialized
[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; }
    public string Version { get; set; }
    public string Theme { get; set; }
    public int MaxConnections { get; set; }
    public bool EnableLogging { get; set; }
    public Dictionary<string, string> CustomSettings { get; set; }
    public DateTime LastModified { get; set; }

    public AppSettings()
    {
        CustomSettings = new Dictionary<string, string>();
        LastModified = DateTime.Now;
    }

    public override string ToString()
    {
        var sb = new StringBuilder();
        sb.AppendLine($"Application Name: {ApplicationName}");
        sb.AppendLine($"Version: {Version}");
        sb.AppendLine($"Theme: {Theme}");
        sb.AppendLine($"Max Connections: {MaxConnections}");
        sb.AppendLine($"Enable Logging: {EnableLogging}");
        sb.AppendLine($"Last Modified: {LastModified}");
        if (CustomSettings != null && CustomSettings.Count > 0)
        {
            sb.AppendLine("Custom Settings:");
            foreach (var kvp in CustomSettings)
            {
                sb.AppendLine($"  {kvp.Key}: {kvp.Value}");
            }
        }
        return sb.ToString();
    }
}

// Settings manager using BinaryFormatter
public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    // Save settings using BinaryFormatter
    public void SaveSettings(AppSettings settings)
    {
        try
        {
            settings.LastModified = DateTime.Now;
            
            #pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
            #pragma warning restore SYSLIB0011
            
            using (var stream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write))
            {
                #pragma warning disable SYSLIB0011
                formatter.Serialize(stream, settings);
                #pragma warning restore SYSLIB0011
            }
            
            Console.WriteLine($"Settings saved successfully to {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            throw;
        }
    }

    // Load settings using BinaryFormatter
    public AppSettings LoadSettings()
    {
        try
        {
            if (!File.Exists(_settingsFilePath))
            {
                Console.WriteLine("Settings file not found. Creating default settings.");
                return CreateDefaultSettings();
            }

            #pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
            #pragma warning restore SYSLIB0011
            
            using (var stream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read))
            {
                #pragma warning disable SYSLIB0011
                var settings = (AppSettings)formatter.Deserialize(stream);
                #pragma warning restore SYSLIB0011
                Console.WriteLine($"Settings loaded successfully from {_settingsFilePath}");
                return settings;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            Console.WriteLine("Returning default settings.");
            return CreateDefaultSettings();
        }
    }

    // Create default settings
    private AppSettings CreateDefaultSettings()
    {
        return new AppSettings
        {
            ApplicationName = "MyApplication",
            Version = "1.0.0",
            Theme = "Default",
            MaxConnections = 10,
            EnableLogging = true,
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "en-US" },
                { "AutoSave", "true" },
                { "BackupInterval", "30" }
            }
        };
    }

    // Update specific setting
    public void UpdateSetting(string key, string value)
    {
        var settings = LoadSettings();
        if (settings.CustomSettings == null)
        {
            settings.CustomSettings = new Dictionary<string, string>();
        }
        settings.CustomSettings[key] = value;
        SaveSettings(settings);
        Console.WriteLine($"Setting '{key}' updated to '{value}'");
    }

    // Delete settings file
    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine($"Settings file {_settingsFilePath} deleted.");
        }
    }

    // Convert settings to JSON string for AI analysis
    public string SettingsToJson(AppSettings settings)
    {
        return JsonSerializer.Serialize(settings, new JsonSerializerOptions { WriteIndented = true });
    }
}

// Claude AI integration for settings analysis
public class SettingsAnalyzer
{
    private readonly AmazonBedrockRuntimeClient _bedrockClient;
    private const string ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0";

    public SettingsAnalyzer()
    {
        _bedrockClient = new AmazonBedrockRuntimeClient(Amazon.RegionEndpoint.USEast1);
    }

    public async Task<string> AnalyzeSettings(string settingsJson)
    {
        var prompt = $@"You are an application settings analyzer. Please analyze the following application settings and provide:
1. A summary of the current configuration
2. Any potential issues or concerns
3. Recommendations for optimization
4. Security considerations

Settings:
{settingsJson}

Please provide a concise but comprehensive analysis.";

        var requestBody = new
        {
            anthropic_version = "bedrock-2023-05-31",
            max_tokens = 1024,
            messages = new[]
            {
                new
                {
                    role = "user",
                    content = prompt
                }
            }
        };

        var requestBodyJson = JsonSerializer.Serialize(requestBody);
        
        var request = new InvokeModelRequest
        {
            ModelId = ModelId,
            Body = new MemoryStream(Encoding.UTF8.GetBytes(requestBodyJson)),
            ContentType = "application/json",
            Accept = "application/json"
        };

        try
        {
            var response = await _bedrockClient.InvokeModelAsync(request);
            var responseBody = await new StreamReader(response.Body).ReadToEndAsync();
            
            using var jsonDoc = JsonDocument.Parse(responseBody);
            var content = jsonDoc.RootElement
                .GetProperty("content")[0]
                .GetProperty("text")
                .GetString();
            
            return content ?? "No analysis available";
        }
        catch (Exception ex)
        {
            return $"Error analyzing settings: {ex.Message}";
        }
    }

    public async Task<string> SuggestSettings(string applicationDescription)
    {
        var prompt = $@"You are an application configuration expert. Based on the following application description, suggest optimal settings in JSON format that would work well with this AppSettings structure:

{{
    ""ApplicationName"": ""string"",
    ""Version"": ""string"",
    ""Theme"": ""string"",
    ""MaxConnections"": ""int"",
    ""EnableLogging"": ""bool"",
    ""CustomSettings"": {{""key"": ""value""}},
    ""LastModified"": ""datetime""
}}

Application Description: {applicationDescription}

Please provide specific, practical settings recommendations in JSON format.";

        var requestBody = new
        {
            anthropic_version = "bedrock-2023-05-31",
            max_tokens = 1024,
            messages = new[]
            {
                new
                {
                    role = "user",
                    content = prompt
                }
            }
        };

        var requestBodyJson = JsonSerializer.Serialize(requestBody);
        
        var request = new InvokeModelRequest
        {
            ModelId = ModelId,
            Body = new MemoryStream(Encoding.UTF8.GetBytes(requestBodyJson)),
            ContentType = "application/json",
            Accept = "application/json"
        };

        try
        {
            var response = await _bedrockClient.InvokeModelAsync(request);
            var responseBody = await new StreamReader(response.Body).ReadToEndAsync();
            
            using var jsonDoc = JsonDocument.Parse(responseBody);
            var content = jsonDoc.RootElement
                .GetProperty("content")[0]
                .GetProperty("text")
                .GetString();
            
            return content ?? "No suggestions available";
        }
        catch (Exception ex)
        {
            return $"Error getting suggestions: {ex.Message}";
        }
    }
}

        Console.WriteLine("AI Suggested Settings:");

        // Demo 6: Verify serialization/deserialization integrity
        Console.WriteLine("\n--- Demo 6: Verifying Data Integrity ---");
        var originalSettings = settingsManager.LoadSettings();

        settingsManager.SaveSettings(originalSettings);
