<?php

require 'vendor/autoload.php';

use Anthropic\Client;

// UserPreferences class to store user settings
class UserPreferences {
    public string $theme;
    public string $language;
    public int $fontSize;
    public bool $notifications;
    public string $username;
    
    public function __construct(
        string $theme = 'light',
        string $language = 'en',
        int $fontSize = 14,
        bool $notifications = true,
        string $username = 'Guest'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->notifications = $notifications;
        $this->username = $username;
    }
    
    public function toArray(): array {
        return [
            'theme' => $this->theme,
            'language' => $this->language,
            'fontSize' => $this->fontSize,
            'notifications' => $this->notifications,
            'username' => $this->username
        ];
    }
}

// Function to get user preferences from cookie
function getUserPreferences(): UserPreferences {
    if (isset($_COOKIE['user_preferences'])) {
        $serializedData = base64_decode($_COOKIE['user_preferences']);
        $preferences = unserialize($serializedData);
        
        if ($preferences instanceof UserPreferences) {
            return $preferences;
        }
    }
    
    // Return default preferences if no cookie exists
    return new UserPreferences();
}

// Function to save user preferences to cookie
function saveUserPreferences(UserPreferences $preferences): void {
    $serializedData = base64_encode(serialize($preferences));
    setcookie('user_preferences', $serializedData, time() + (86400 * 30), '/'); // 30 days
}

// Function to update preferences based on user input
function updatePreferences(UserPreferences $preferences, array $updates): UserPreferences {
    foreach ($updates as $key => $value) {
        if (property_exists($preferences, $key)) {
            $preferences->$key = $value;
        }
    }
    return $preferences;
}

// Initialize Anthropic client
$client = new Client(getenv('ANTHROPIC_API_KEY'));

// Get current user preferences
$userPreferences = getUserPreferences();

// Check if there's a preference update request
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['update_preferences'])) {
    $updates = [];
    
    if (isset($_POST['theme'])) $updates['theme'] = $_POST['theme'];
    if (isset($_POST['language'])) $updates['language'] = $_POST['language'];
    if (isset($_POST['fontSize'])) $updates['fontSize'] = (int)$_POST['fontSize'];
    if (isset($_POST['notifications'])) $updates['notifications'] = $_POST['notifications'] === 'true';
    if (isset($_POST['username'])) $updates['username'] = $_POST['username'];
    
    $userPreferences = updatePreferences($userPreferences, $updates);
    saveUserPreferences($userPreferences);
}

// Get user message for AI assistant
$userMessage = $_POST['message'] ?? 'Hello! Tell me about my current preferences and suggest some improvements.';

// Prepare context about user preferences for the AI
$preferencesContext = json_encode($userPreferences->toArray(), JSON_PRETTY_PRINT);

// Create a conversation with Claude about user preferences
$systemPrompt = "You are a helpful AI assistant that helps users manage their application preferences. 
The user's current preferences are:
{$preferencesContext}

You can help users understand their preferences, suggest improvements, and explain what each setting does.
When users want to change preferences, acknowledge their request and explain the changes.
Be conversational and helpful.";

echo "=== User Preferences Management System ===\n\n";
echo "Current User Preferences:\n";
echo "- Username: {$userPreferences->username}\n";
echo "- Theme: {$userPreferences->theme}\n";
echo "- Language: {$userPreferences->language}\n";
echo "- Font Size: {$userPreferences->fontSize}px\n";
echo "- Notifications: " . ($userPreferences->notifications ? 'Enabled' : 'Disabled') . "\n\n";

echo "User Message: {$userMessage}\n\n";
echo "AI Assistant Response (streaming):\n";
echo str_repeat("-", 50) . "\n";

// Use streaming to get AI response about preferences
$stream = $client->messages()->createStreamed([
    'model' => 'claude-opus-4-5',
    'max_tokens' => 1024,
    'system' => $systemPrompt,
    'messages' => [
        [
            'role' => 'user',
            'content' => $userMessage
        ]
    ]
]);

$fullResponse = '';

foreach ($stream as $event) {
    if ($event->type === 'content_block_delta') {
        $text = $event->delta->text ?? '';
        echo $text;
        $fullResponse .= $text;
        flush();
    }
}

echo "\n" . str_repeat("-", 50) . "\n\n";

// Demonstrate preference update simulation
echo "=== Simulating Preference Update ===\n";
echo "Updating theme to 'dark' and font size to 16...\n\n";

$updatedPreferences = updatePreferences($userPreferences, [
    'theme' => 'dark',
    'fontSize' => 16
]);

saveUserPreferences($updatedPreferences);

echo "Updated Preferences:\n";
echo "- Theme: {$updatedPreferences->theme}\n";
echo "- Font Size: {$updatedPreferences->fontSize}px\n\n";

// Get AI response about the update
$updateMessage = "I just updated my theme to dark mode and increased the font size to 16px. How do these changes affect my experience?";

echo "User Message: {$updateMessage}\n\n";
echo "AI Assistant Response (streaming):\n";
echo str_repeat("-", 50) . "\n";

$updatedPreferencesContext = json_encode($updatedPreferences->toArray(), JSON_PRETTY_PRINT);
$updatedSystemPrompt = "You are a helpful AI assistant that helps users manage their application preferences. 
The user's updated preferences are:
{$updatedPreferencesContext}

The user just changed their theme to dark mode and increased font size to 16px.
Explain the benefits of these changes and how they improve the user experience.";

$stream2 = $client->messages()->createStreamed([
    'model' => 'claude-opus-4-5',
    'max_tokens' => 1024,
    'system' => $updatedSystemPrompt,
    'messages' => [
        [
            'role' => 'user',
            'content' => $updateMessage
        ]
    ]
]);

foreach ($stream2 as $event) {
    if ($event->type === 'content_block_delta') {
        $text = $event->delta->text ?? '';
        echo $text;
        flush();
    }
}

echo "\n" . str_repeat("-", 50) . "\n\n";

// Show serialization details
echo "=== Cookie Serialization Details ===\n";
$serialized = serialize($updatedPreferences);
$encoded = base64_encode($serialized);
echo "Serialized data length: " . strlen($serialized) . " bytes\n";
echo "Base64 encoded length: " . strlen($encoded) . " bytes\n";
echo "Cookie name: user_preferences\n";
echo "Cookie expiry: 30 days\n\n";

echo "=== Script Complete ===\n";
