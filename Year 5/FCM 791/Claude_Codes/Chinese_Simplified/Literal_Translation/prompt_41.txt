import anthropic
import psycopg2
import json
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import parse_qs, urlparse
import threading

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_user_exists",
        "description": "Check if a user exists in the PostgreSQL database by username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    },
    {
        "name": "get_all_users",
        "description": "Get all users from the PostgreSQL database",
        "input_schema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    }
]

def get_db_connection():
    """Create a database connection"""
    try:
        conn = psycopg2.connect(
            host="localhost",
            database="testdb",
            user="postgres",
            password="password",
            port="5432"
        )
        return conn
    except psycopg2.Error as e:
        print(f"Database connection error: {e}")
        return None

def setup_database():
    """Setup the database with sample data"""
    conn = get_db_connection()
    if conn:
        try:
            cursor = conn.cursor()
            cursor.execute("""
                CREATE TABLE IF NOT EXISTS users (
                    id SERIAL PRIMARY KEY,
                    username VARCHAR(50) UNIQUE NOT NULL,
                    email VARCHAR(100),
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """)
            
            sample_users = [
                ('alice', 'alice@example.com'),
                ('bob', 'bob@example.com'),
                ('charlie', 'charlie@example.com'),
                ('diana', 'diana@example.com')
            ]
            
            for username, email in sample_users:
                cursor.execute("""
                    INSERT INTO users (username, email) 
                    VALUES (%s, %s) 
                    ON CONFLICT (username) DO NOTHING
                """, (username, email))
            
            conn.commit()
            cursor.close()
            conn.close()
            print("Database setup completed successfully")
        except psycopg2.Error as e:
            print(f"Database setup error: {e}")
            conn.rollback()
            conn.close()

def check_user_exists(username: str) -> dict:
    """Check if a user exists in the database"""
    conn = get_db_connection()
    if not conn:
        return {"error": "Database connection failed", "exists": False}
    
    try:
        cursor = conn.cursor()
        cursor.execute("SELECT id, username, email, created_at FROM users WHERE username = %s", (username,))
        user = cursor.fetchone()
        cursor.close()
        conn.close()
        
        if user:
            return {
                "exists": True,
                "user": {
                    "id": user[0],
                    "username": user[1],
                    "email": user[2],
                    "created_at": str(user[3])
                }
            }
        else:
            return {"exists": False, "username": username}
    except psycopg2.Error as e:
        conn.close()
        return {"error": str(e), "exists": False}

def get_all_users() -> dict:
    """Get all users from the database"""
    conn = get_db_connection()
    if not conn:
        return {"error": "Database connection failed", "users": []}
    
    try:
        cursor = conn.cursor()
        cursor.execute("SELECT id, username, email, created_at FROM users ORDER BY id")
        users = cursor.fetchall()
        cursor.close()
        conn.close()
        
        return {
            "users": [
                {
                    "id": user[0],
                    "username": user[1],
                    "email": user[2],
                    "created_at": str(user[3])
                }
                for user in users
            ],
            "total": len(users)
        }
    except psycopg2.Error as e:
        conn.close()
        return {"error": str(e), "users": []}

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude"""
    if tool_name == "check_user_exists":
        result = check_user_exists(tool_input["username"])
        return json.dumps(result)
    elif tool_name == "get_all_users":
        result = get_all_users()
        return json.dumps(result)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def query_user_with_claude(username: str) -> str:
    """Use Claude to check if a user exists using tool use"""
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our database. If they exist, provide their details. If not, let me know they don't exist."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_result = process_tool_call(content_block.name, content_block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": content_block.id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    return content_block.text
            return "No response generated"
        else:
            return f"Unexpected stop reason: {response.stop_reason}"

class UserCheckHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == '/':
            self.send_response(200)
            self.send_header('Content-type', 'text/html')
            self.end_headers()
            html = HTML_FORM.format(result_section='')
            self.wfile.write(html.encode())
        else:
            self.send_response(404)
            self.end_headers()
    
    def do_POST(self):
        if self.path == '/check_user':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            params = parse_qs(post_data.decode())
            
            username = params.get('username', [''])[0].strip()
            
            if username:
                result = query_user_with_claude(username)
                result_section = f'<div class="result info"><strong>Result:</strong>\n{result}</div>'
            else:
                result_section = '<div class="result error">Please enter a username</div>'
            
            self.send_response(200)
            self.send_header('Content-type', 'text/html')
            self.end_headers()
            html = HTML_FORM.format(result_section=result_section)
            self.wfile.write(html.encode())
        else:
            self.send_response(404)
            self.end_headers()
    
    def log_message(self, format, *args):
        pass

def run_server():
    """Run the HTTP server"""
    server_address = ('', 8080)
    httpd = HTTPServer(server_address, UserCheckHandler)
    print("Server running at http://localhost:8080")
    print("Press Ctrl+C to stop")
    httpd.serve_forever()
