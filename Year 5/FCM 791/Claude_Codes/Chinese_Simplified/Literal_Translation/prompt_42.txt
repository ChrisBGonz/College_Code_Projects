import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Define the ping tool
const tools = [
  {
    name: "ping_host",
    description:
      "Runs a ping command to check if a host is online. Returns whether the host is reachable and response time.",
    input_schema: {
      type: "object",
      properties: {
        domain_name: {
          type: "string",
          description: "The domain name or IP address to ping",
        },
        count: {
          type: "number",
          description: "Number of ping packets to send (default: 4)",
        },
      },
      required: ["domain_name"],
    },
  },
];

// Function to execute ping command
import { exec } from "child_process";
import { promisify } from "util";

const execAsync = promisify(exec);

async function executePing(domainName, count = 4) {
  try {
    // Determine the platform and use appropriate ping command
    const isWindows = process.platform === "win32";
    const pingCommand = isWindows
      ? `ping -n ${count} ${domainName}`
      : `ping -c ${count} ${domainName}`;

    const { stdout, stderr } = await execAsync(pingCommand, { timeout: 30000 });

    // Parse ping results
    const output = stdout + stderr;
    const isOnline =
      !output.includes("unreachable") &&
      !output.includes("100% packet loss") &&
      !output.includes("100.0% packet loss") &&
      !output.includes("Request timeout") &&
      !output.includes("could not find host") &&
      !output.includes("Name or service not known");

    // Extract average response time if available
    let avgTime = null;
    const timeMatch = output.match(/avg[^=]*=\s*([\d.]+)/i);
    if (timeMatch) {
      avgTime = parseFloat(timeMatch[1]);
    }

    return {
      domain: domainName,
      isOnline,
      output: output.trim(),
      avgResponseTime: avgTime,
    };
  } catch (error) {
    return {
      domain: domainName,
      isOnline: false,
      output: error.message,
      avgResponseTime: null,
      error: true,
    };
  }
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  if (toolName === "ping_host") {
    const { domain_name, count } = toolInput;
    return await executePing(domain_name, count);
  }
  throw new Error(`Unknown tool: ${toolName}`);
}

// Main function to check if a host is online using Claude
async function checkHostOnline(domainName) {
  console.log(`\nChecking if ${domainName} is online...\n`);

  const messages = [
    {
      role: "user",
      content: `Please check if the host "${domainName}" is online by pinging it. Use the ping_host tool to test connectivity and then provide a clear summary of whether the host is reachable.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  // Agentic loop - continue until we get a final response
  while (response.stop_reason === "tool_use") {
    const toolUseBlock = response.content.find(
      (block) => block.type === "tool_use"
    );

    if (!toolUseBlock) {
      break;
    }

    console.log(`Executing tool: ${toolUseBlock.name}`);
    console.log(`Input: ${JSON.stringify(toolUseBlock.input, null, 2)}`);

    // Execute the tool
    const toolResult = await processToolCall(
      toolUseBlock.name,
      toolUseBlock.input
    );

    console.log(`\nPing Result:`);
    console.log(`- Domain: ${toolResult.domain}`);
    console.log(`- Online: ${toolResult.isOnline}`);
    if (toolResult.avgResponseTime) {
      console.log(`- Avg Response Time: ${toolResult.avgResponseTime}ms`);
    }
    console.log(`\nRaw Output:\n${toolResult.output}\n`);

    // Add assistant response and tool result to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: [
        {
          type: "tool_result",
          tool_use_id: toolUseBlock.id,
          content: JSON.stringify(toolResult),
        },
      ],
    });

    // Get next response from Claude
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("Claude's Analysis:");
  console.log(finalResponse);

  return finalResponse;
}

// Example usage
async function main() {
  const domainsToCheck = ["google.com", "github.com", "nonexistent-domain-xyz.com"];

  for (const domain of domainsToCheck) {
    console.log("=".repeat(60));
    await checkHostOnline(domain);
    console.log();
  }
}

main().catch(console.error);
