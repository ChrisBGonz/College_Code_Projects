
```python
import anthropic
import logging
import re
from datetime import datetime

# Configure logging to write to app.log
logging.basicConfig(
    filename='app.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S'
)

def log_search_query(query: str) -> None:
    """Log a search query to app.log file."""
    logging.info(f"Search Query: {query}")
    print(f"Logged search query: {query}")

def process_search_with_claude(user_input: str) -> str:
    """
    Process user input with Claude using streaming to extract and log search queries.
    
    Args:
        user_input: The user's input text
        
    Returns:
        The complete response from Claude
    """
    client = anthropic.Anthropic()
    
    # System prompt to help Claude identify and extract search queries
    system_prompt = """You are a helpful assistant that processes user requests. 
    When a user provides a search query or asks you to search for something, 
    identify the search query and include it in your response in the format:
    SEARCH_QUERY: [the actual search query]
    
    Then provide a helpful response about what you would search for and why."""
    
    full_response = ""
    
    print(f"\nProcessing user input: {user_input}")
    print("Claude's response:")
    print("-" * 50)
    
    # Use streaming to get Claude's response
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        system=system_prompt,
        messages=[
            {"role": "user", "content": user_input}
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    
    # Extract and log search queries from Claude's response
    lines = full_response.split('\n')
    for line in lines:
        if line.startswith('SEARCH_QUERY:'):
            search_query = line.replace('SEARCH_QUERY:', '').strip()
            if search_query:
                log_search_query(search_query)
    
    return full_response

def search_and_log(query: str) -> None:
    """
    Main function to handle search queries - logs them and processes with Claude.
    
    Args:
        query: The search query from the user
    """
    # Log the original user query
    log_search_query(query)
    
    # Process with Claude using streaming
    response = process_search_with_claude(f"Please search for: {query}")
    
    return response

def main():
    """Main function to demonstrate the search logging functionality."""
    print("Search Query Logger with Claude AI")
    print("=" * 50)
    
    # Example search queries to demonstrate the functionality
    test_queries = [
        "Python programming best practices",
        "How to use machine learning for image recognition",
        "Latest developments in renewable energy"
    ]
    
    for query in test_queries:
        print(f"\n{'='*50}")
        print(f"User Query: {query}")
        search_and_log(query)
        print()
    
    print("\n" + "="*50)
    print("All search queries have been logged to app.log")
    print("\nContents of app.log:")
    print("-" * 50)
    
    # Display the contents of app.log
    try:
        with open('app.log', 'r') as f:
            print(f.read())
    except FileNotFoundError:
        print("app.log file not found")

if __name__ == "__main__":
    main()
```
