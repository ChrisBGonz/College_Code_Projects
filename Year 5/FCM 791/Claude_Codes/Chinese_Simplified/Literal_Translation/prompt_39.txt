import anthropic
import socket
import threading
import time
import random

# XOR cipher key for "security"
XOR_KEY = 0x42  # Simple XOR key

def xor_encrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Encrypt/decrypt data using XOR cipher."""
    return bytes([b ^ key for b in data])

def xor_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """XOR cipher is symmetric, so decrypt is same as encrypt."""
    return xor_encrypt(data, key)

class LegacyServer:
    """Simulates a legacy server that uses XOR encryption."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        self.responses = {
            "hello": "Hello from Legacy Server! How can I help you?",
            "status": "Server Status: OPERATIONAL | Version: 1.0.0 | Uptime: 99.9%",
            "data": "Legacy Data: [Record1, Record2, Record3] | Format: CSV | Encoding: ASCII",
            "ping": "PONG",
            "info": "Legacy Server Info: Protocol v1, XOR Encrypted, Port 9999",
        }
    
    def handle_client(self, client_socket: socket.socket, address: tuple):
        """Handle incoming client connections."""
        print(f"[Server] New connection from {address}")
        try:
            while True:
                # Receive encrypted data
                encrypted_data = client_socket.recv(1024)
                if not encrypted_data:
                    break
                
                # Decrypt the received data
                decrypted_data = xor_decrypt(encrypted_data)
                message = decrypted_data.decode('utf-8').strip()
                print(f"[Server] Received (decrypted): {message}")
                
                # Generate response based on message
                response_text = self.responses.get(
                    message.lower(), 
                    f"Unknown command: {message}. Available: {', '.join(self.responses.keys())}"
                )
                
                # Encrypt and send response
                encrypted_response = xor_encrypt(response_text.encode('utf-8'))
                client_socket.send(encrypted_response)
                print(f"[Server] Sent (encrypted): {response_text[:50]}...")
                
        except Exception as e:
            print(f"[Server] Error handling client: {e}")
        finally:
            client_socket.close()
            print(f"[Server] Connection closed from {address}")
    
    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"[Server] Legacy server started on {self.host}:{self.port}")
        
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                client_thread = threading.Thread(
                    target=self.handle_client,
                    args=(client_socket, address)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"[Server] Error: {e}")
    
    def stop(self):
        """Stop the legacy server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("[Server] Legacy server stopped")


class LegacyServerClient:
    """Client for communicating with the legacy server using XOR encryption."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
    
    def connect(self):
        """Connect to the legacy server."""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        print(f"[Client] Connected to legacy server at {self.host}:{self.port}")
    
    def send_command(self, command: str) -> str:
        """Send an encrypted command and receive encrypted response."""
        if not self.socket:
            raise ConnectionError("Not connected to server")
        
        # Encrypt and send command
        encrypted_command = xor_encrypt(command.encode('utf-8'))
        self.socket.send(encrypted_command)
        print(f"[Client] Sent command: {command}")
        
        # Receive and decrypt response
        encrypted_response = self.socket.recv(1024)
        response = xor_decrypt(encrypted_response).decode('utf-8')
        print(f"[Client] Received response: {response}")
        
        return response
    
    def disconnect(self):
        """Disconnect from the server."""
        if self.socket:
            self.socket.close()
            self.socket = None
        print("[Client] Disconnected from legacy server")


def communicate_with_legacy_server(command: str) -> str:
    """Tool function to communicate with the legacy server."""
    client = LegacyServerClient()
    try:
        client.connect()
        response = client.send_command(command)
        return response
    except Exception as e:
        return f"Error communicating with server: {str(e)}"
    finally:
        client.disconnect()


def run_claude_with_tools():
    """Run Claude with tool use to interact with the legacy server."""
    
    # Define the tool for Claude to use
    tools = [
        {
            "name": "communicate_with_legacy_server",
            "description": "Send a command to the legacy server and receive a response. The communication uses XOR encryption for security. Available commands: hello, status, data, ping, info",
            "input_schema": {
                "type": "object",
                "properties": {
                    "command": {
                        "type": "string",
                        "description": "The command to send to the legacy server (e.g., 'hello', 'status', 'data', 'ping', 'info')"
                    }
                },
                "required": ["command"]
            }
        }
    ]
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    print("\n" + "="*60)
    print("Starting Claude interaction with Legacy Server")
    print("="*60 + "\n")
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": "Please interact with the legacy server to gather information. First check if it's alive with a ping, then get its status, and finally retrieve any available data. Report back what you find."
        }
    ]
    
    # Agentic loop
    while True:
        print("[Claude] Sending request to Claude...")
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"[Claude] Stop reason: {response.stop_reason}")
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract and print final response
            for block in response.content:
                if hasattr(block, 'text'):
                    print("\n" + "="*60)
                    print("Claude's Final Report:")
                    print("="*60)
                    print(block.text)
            break
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process each tool use
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\n[Tool] Claude is calling: {block.name}")
                    print(f"[Tool] With input: {block.input}")
                    
                    # Execute the tool
                    if block.name == "communicate_with_legacy_server":
                        result = communicate_with_legacy_server(block.input["command"])
                        print(f"[Tool] Result: {result}")
                        
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": result
                        })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            print(f"[Claude] Unexpected stop reason: {response.stop_reason}")
            break


def main():
    """Main function to run the legacy server and Claude interaction."""
    
    print("="*60)
    print("Legacy Server Communication with XOR Encryption")
    print("="*60)
    
    # Start the legacy server in a background thread
    server = LegacyServer()
    server_thread = threading.Thread(target=server.start)
    server_thread.daemon = True
    server_thread.start()
    
    # Wait for server to start
    time.sleep(1)
    print("[Main] Legacy server is running...")
    
    try:
        # Run Claude with tools to interact with the server
        run_claude_with_tools()
        
    except KeyboardInterrupt:
        print("\n[Main] Interrupted by user")
    except Exception as e:
        print(f"[Main] Error: {e}")
        raise
    finally:
        # Stop the server
        server.stop()
        print("[Main] Application finished")


if __name__ == "__main__":
    main()
