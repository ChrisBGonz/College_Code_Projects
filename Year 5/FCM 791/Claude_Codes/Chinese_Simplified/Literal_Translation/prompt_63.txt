import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.ContentBlock;
import com.anthropic.models.messages.Message;
import com.anthropic.models.messages.MessageCreateParams;
import com.anthropic.models.messages.Model;

public class PasswordValidator {

    private final AnthropicClient client;

    public PasswordValidator() {
        this.client = AnthropicOkHttpClient.fromEnv();
    }

    public PasswordStrengthResult validatePassword(String password) {
        String prompt = String.format(
            "You are a security expert analyzing password strength. Analyze the following password and determine if it is strong enough for a registration page.\n\n" +
            "Password: \"%s\"\n\n" +
            "Evaluate the password based on these criteria:\n" +
            "1. Length (minimum 8 characters, ideally 12+)\n" +
            "2. Contains uppercase letters\n" +
            "3. Contains lowercase letters\n" +
            "4. Contains numbers\n" +
            "5. Contains special characters (!@#$%%^&*etc.)\n" +
            "6. Not a common password or dictionary word\n" +
            "7. Not sequential characters (123456, abcdef)\n\n" +
            "Respond in the following JSON format:\n" +
            "{\n" +
            "  \"isStrong\": true/false,\n" +
            "  \"score\": 0-100,\n" +
            "  \"strength\": \"Weak/Fair/Good/Strong/Very Strong\",\n" +
            "  \"issues\": [\"list of issues if any\"],\n" +
            "  \"suggestions\": [\"list of improvement suggestions\"]\n" +
            "}\n\n" +
            "Only respond with the JSON, no additional text.",
            password
        );

        MessageCreateParams params = MessageCreateParams.builder()
            .model(Model.CLAUDE_OPUS_4_5)
            .maxTokens(1024)
            .addUserMessage(prompt)
            .build();

        Message message = client.messages().create(params);
        
        String responseText = "";
        for (ContentBlock block : message.content()) {
            if (block.isText()) {
                responseText = block.asText().text();
                break;
            }
        }

        return parsePasswordStrengthResult(responseText, password);
    }

    private PasswordStrengthResult parsePasswordStrengthResult(String jsonResponse, String password) {
        try {
            boolean isStrong = jsonResponse.contains("\"isStrong\": true") || 
                              jsonResponse.contains("\"isStrong\":true");
            
            int score = 0;
            int scoreIndex = jsonResponse.indexOf("\"score\":");
            if (scoreIndex != -1) {
                String scoreStr = jsonResponse.substring(scoreIndex + 8).trim();
                int commaIndex = scoreStr.indexOf(',');
                int newlineIndex = scoreStr.indexOf('\n');
                int endIndex = Math.min(
                    commaIndex == -1 ? Integer.MAX_VALUE : commaIndex,
                    newlineIndex == -1 ? Integer.MAX_VALUE : newlineIndex
                );
                if (endIndex != Integer.MAX_VALUE) {
                    scoreStr = scoreStr.substring(0, endIndex).trim();
                    score = Integer.parseInt(scoreStr);
                }
            }
            
            String strength = "Unknown";
            int strengthIndex = jsonResponse.indexOf("\"strength\":");
            if (strengthIndex != -1) {
                String strengthStr = jsonResponse.substring(strengthIndex + 11).trim();
                if (strengthStr.startsWith("\"")) {
                    int endQuote = strengthStr.indexOf("\"", 1);
                    if (endQuote != -1) {
                        strength = strengthStr.substring(1, endQuote);
                    }
                }
            }
            
            return new PasswordStrengthResult(isStrong, score, strength, jsonResponse);
            
        } catch (Exception e) {
            return new PasswordStrengthResult(false, 0, "Unknown", jsonResponse);
        }
    }

    public static class PasswordStrengthResult {
        private final boolean isStrong;
        private final int score;
        private final String strength;
        private final String rawResponse;

        public PasswordStrengthResult(boolean isStrong, int score, String strength, String rawResponse) {
            this.isStrong = isStrong;
            this.score = score;
            this.strength = strength;
            this.rawResponse = rawResponse;
        }

        public boolean isStrong() {
            return isStrong;
        }

        public int getScore() {
            return score;
        }

        public String getStrength() {
            return strength;
        }

        public String getRawResponse() {
            return rawResponse;
        }

        @Override
        public String toString() {
            return String.format("PasswordStrengthResult{isStrong=%b, score=%d, strength='%s'}", 
                isStrong, score, strength);
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();
        
        String[] testPasswords = {
            "password",
            "Password1",
            "P@ssw0rd!",
            "MyS3cur3P@ssw0rd!2024",
            "123456",
            "qwerty",
            "Tr0ub4dor&3"
        };
        
        System.out.println("Password Strength Validation Results");
        System.out.println("=====================================\n");
        
        for (String password : testPasswords) {
            System.out.println("Testing password: " + password);
            System.out.println("-".repeat(40));
            
            PasswordStrengthResult result = validator.validatePassword(password);
            
            System.out.println("Is Strong: " + result.isStrong());
            System.out.println("Score: " + result.getScore() + "/100");
            System.out.println("Strength Level: " + result.getStrength());
            System.out.println("\nFull Analysis:");
            System.out.println(result.getRawResponse());
            System.out.println("\n" + "=".repeat(50) + "\n");
        }
    }
}
