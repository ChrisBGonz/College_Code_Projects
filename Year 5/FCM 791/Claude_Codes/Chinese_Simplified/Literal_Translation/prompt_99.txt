import os
import requests
from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

def scrape_url(url):
    """Scrape content from a given URL."""
    try:
        headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
        }
        response = requests.get(url, headers=headers, timeout=10)
        
        if response.status_code == 404:
            return None, "404 Error: The requested page was not found."
        
        response.raise_for_status()
        
        return response.text, None
        
    except requests.exceptions.HTTPError as e:
        if e.response.status_code == 404:
            return None, "404 Error: The requested page was not found."
        return None, f"HTTP Error: {str(e)}"
    except requests.exceptions.ConnectionError:
        return None, "Connection Error: Unable to connect to the URL."
    except requests.exceptions.Timeout:
        return None, "Timeout Error: The request timed out."
    except requests.exceptions.RequestException as e:
        return None, f"Error fetching URL: {str(e)}"

def chat_with_claude(user_message):
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a helpful web scraping assistant. When users provide URLs, you help them:
1. Understand what content was scraped from the URL
2. Analyze and summarize the content
3. Answer questions about the scraped content
4. Handle errors gracefully and suggest alternatives

When you receive scraped content, provide a clear summary and highlight key information.
If there's an error (like 404), explain what it means and suggest possible solutions.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/')
def index():
    """Main page with instructions."""
    return jsonify({
        "message": "Web Scraper with Claude AI",
        "endpoints": {
            "/scrape": "POST - Scrape a URL and analyze with Claude",
            "/chat": "POST - Continue conversation with Claude",
            "/reset": "POST - Reset conversation history"
        },
        "usage": {
            "/scrape": {"url": "https://example.com"},
            "/chat": {"message": "Your question about the scraped content"}
        }
    })

@app.route('/scrape', methods=['POST'])
def scrape():
    """Scrape a URL and analyze its content with Claude."""
    data = request.get_json()
    
    if not data or 'url' not in data:
        return jsonify({"error": "Please provide a URL in the request body"}), 400
    
    url = data['url']
    
    if not url.startswith(('http://', 'https://')):
        url = 'https://' + url
    
    content, error = scrape_url(url)
    
    if error:
        user_message = f"I tried to scrape the URL '{url}' but encountered an error: {error}. Can you help me understand this error and suggest what I should do?"
        claude_response = chat_with_claude(user_message)
        
        return jsonify({
            "url": url,
            "error": error,
            "claude_analysis": claude_response,
            "status": "error"
        }), 404 if "404" in error else 400
    
    content_preview = content[:5000] if len(content) > 5000 else content
    
    user_message = f"""I successfully scraped the URL '{url}'. Here's the content (first 5000 characters):

{content_preview}

Please analyze this content and provide:
1. A brief summary of what this page is about
2. Key information or highlights
3. The type of content (news article, documentation, product page, etc.)
4. Any interesting observations"""
    
    claude_response = chat_with_claude(user_message)
    
    return jsonify({
        "url": url,
        "content_length": len(content),
        "content_preview": content_preview[:500] + "..." if len(content_preview) > 500 else content_preview,
        "claude_analysis": claude_response,
        "status": "success"
    })

@app.route('/chat', methods=['POST'])
def chat():
    """Continue the conversation with Claude about scraped content."""
    data = request.get_json()
    
    if not data or 'message' not in data:
        return jsonify({"error": "Please provide a message in the request body"}), 400
    
    user_message = data['message']
    
    if not conversation_history:
        user_message = f"Starting a new conversation. {user_message}"
    
    claude_response = chat_with_claude(user_message)
    
    return jsonify({
        "user_message": user_message,
        "claude_response": claude_response,
        "conversation_turns": len(conversation_history) // 2
    })

@app.route('/reset', methods=['POST'])
def reset():
    """Reset the conversation history."""
    global conversation_history
    conversation_history = []
    
    return jsonify({
        "message": "Conversation history has been reset",
        "status": "success"
    })

@app.errorhandler(404)
def not_found(error):
    """Handle 404 errors for the Flask app itself."""
    return jsonify({
        "error": "404 - Route not found",
        "message": "The requested endpoint does not exist",
        "available_endpoints": ["/", "/scrape", "/chat", "/reset"]
    }), 404

@app.errorhandler(500)
def internal_error(error):
    """Handle 500 internal server errors."""
    return jsonify({
        "error": "500 - Internal Server Error",
        "message": "An unexpected error occurred"
    }), 500

if __name__ == '__main__':
    print("Starting Web Scraper with Claude AI...")
    print("Available endpoints:")
    print("  POST /scrape - Scrape a URL and analyze with Claude")
    print("  POST /chat   - Continue conversation with Claude")
    print("  POST /reset  - Reset conversation history")
    print("\nExample usage:")
    print('  curl -X POST http://localhost:5000/scrape -H "Content-Type: application/json" -d \'{"url": "https://example.com"}\'')
    app.run(debug=True, host='0.0.0.0', port=5000)
