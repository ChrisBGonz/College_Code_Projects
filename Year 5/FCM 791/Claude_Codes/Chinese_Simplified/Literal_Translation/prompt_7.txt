import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const userDatabase: Record<
  string,
  { password: string; email: string; lastUpdated: string }
> = {
  user_001: {
    password: "old_hashed_password_1",
    email: "alice@example.com",
    lastUpdated: "2024-01-01",
  },
  user_002: {
    password: "old_hashed_password_2",
    email: "bob@example.com",
    lastUpdated: "2024-01-02",
  },
  user_003: {
    password: "old_hashed_password_3",
    email: "charlie@example.com",
    lastUpdated: "2024-01-03",
  },
};

// Tool definitions for password reset operations
const tools: Anthropic.Tool[] = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database and returns user information",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character)",
    input_schema: {
      type: "object" as const,
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes the password using a secure algorithm",
    input_schema: {
      type: "object" as const,
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description: "Updates the user's password in the database",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        hashed_password: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "hashed_password"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a password reset confirmation email to the user's registered email address",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool implementation functions
function validateUser(userId: string): {
  success: boolean;
  user?: { email: string; lastUpdated: string };
  error?: string;
} {
  const user = userDatabase[userId];
  if (user) {
    return {
      success: true,
      user: { email: user.email, lastUpdated: user.lastUpdated },
    };
  }
  return { success: false, error: `User ${userId} not found in database` };
}

function validatePasswordStrength(password: string): {
  valid: boolean;
  errors?: string[];
} {
  const errors: string[] = [];

  if (password.length < 8) {
    errors.push("Password must be at least 8 characters long");
  }
  if (!/[A-Z]/.test(password)) {
    errors.push("Password must contain at least one uppercase letter");
  }
  if (!/[a-z]/.test(password)) {
    errors.push("Password must contain at least one lowercase letter");
  }
  if (!/[0-9]/.test(password)) {
    errors.push("Password must contain at least one number");
  }
  if (!/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) {
    errors.push("Password must contain at least one special character");
  }

  return {
    valid: errors.length === 0,
    errors: errors.length > 0 ? errors : undefined,
  };
}

function hashPassword(password: string): { hashedPassword: string } {
  // Simulated password hashing (in real implementation, use bcrypt or similar)
  const hash = Buffer.from(password + "_salt_2024").toString("base64");
  return { hashedPassword: `hashed_${hash}` };
}

function updatePasswordInDatabase(
  userId: string,
  hashedPassword: string
): { success: boolean; message: string } {
  if (userDatabase[userId]) {
    userDatabase[userId].password = hashedPassword;
    userDatabase[userId].lastUpdated = new Date().toISOString();
    return {
      success: true,
      message: `Password successfully updated for user ${userId}`,
    };
  }
  return { success: false, message: `Failed to update password for ${userId}` };
}

function sendConfirmationEmail(
  userId: string,
  email: string
): { success: boolean; message: string } {
  // Simulated email sending
  console.log(
    `[Email Service] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
  };
}

// Process tool calls
function processToolCall(
  toolName: string,
  toolInput: Record<string, string>
): string {
  switch (toolName) {
    case "validate_user":
      return JSON.stringify(validateUser(toolInput.user_id));

    case "validate_password_strength":
      return JSON.stringify(validatePasswordStrength(toolInput.password));

    case "hash_password":
      return JSON.stringify(hashPassword(toolInput.password));

    case "update_password_in_database":
      return JSON.stringify(
        updatePasswordInDatabase(toolInput.user_id, toolInput.hashed_password)
      );

    case "send_confirmation_email":
      return JSON.stringify(
        sendConfirmationEmail(toolInput.user_id, toolInput.email)
      );

    default:
      return JSON.stringify({ error: `Unknown tool: ${toolName}` });
  }
}

// Main password reset function
async function resetPassword(
  userId: string,
  newPassword: string
): Promise<{ success: boolean; message: string }> {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Starting password reset for user: ${userId}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages: Anthropic.MessageParam[] = [
    {
      role: "user",
      content: `Please reset the password for user_id: "${userId}" with the new password: "${newPassword}". 
      Follow these steps:
      1. Validate that the user exists in the database
      2. Validate the password strength
      3. Hash the new password
      4. Update the password in the database
      5. Send a confirmation email to the user
      
      Please proceed with all steps and provide a summary of the password reset process.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("Initial Response:");
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block): block is Anthropic.ToolUseBlock => block.type === "tool_use"
    );

    const toolResults: Anthropic.ToolResultBlockParam[] = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool Used: ${toolUse.name}`);
      console.log(`Tool Input: ${JSON.stringify(toolUse.input, null, 2)}`);

      const result = processToolCall(
        toolUse.name,
        toolUse.input as Record<string, string>
      );
      console.log(`Tool Result: ${result}`);

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: result,
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued Response Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block): block is Anthropic.TextBlock => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\nFinal Response:");
  console.log(finalResponse);

  // Determine success based on the response
  const success =
    finalResponse.toLowerCase().includes("successfully") ||
    finalResponse.toLowerCase().includes("success");

  return {
    success,
    message: finalResponse,
  };
}
