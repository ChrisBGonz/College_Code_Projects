@app.route('/upload', methods=['POST'])
def upload_file():
    global user_profile, conversation_history
    
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': '没有文件被上传'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': '没有选择文件'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': '请上传 .yaml 或 .yml 格式的文件'})
    
    try:
        content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'YAML 文件格式不正确，应该是键值对格式'})
        
        user_profile = profile_data
        
        # Reset conversation history when new profile is loaded
        conversation_history = []
        
        return jsonify({
            'success': True,
            'profile': user_profile,
            'message': '配置文件上传成功'
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'YAML 解析错误: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'处理文件时出错: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    global conversation_history, user_profile
    
    data = request.json
    user_message = data.get('message', '').strip()
    
    if not user_message:
        return jsonify({'success': False, 'error': '消息不能为空'})
    
    # Build system message with user profile context
    system_message = """你是一个智能个人资料助手，帮助用户了解和管理他们的个人资料配置。
    
你的职责：
1. 回答用户关于其个人资料的问题
2. 分析和解释配置信息
3. 提供个性化建议
4. 帮助用户理解配置文件的结构和内容

请用中文回答，保持友好、专业的态度。"""
    
    if user_profile:
        profile_str = yaml.dump(user_profile, allow_unicode=True, default_flow_style=False)
        system_message += f"\n\n当前用户的个人资料配置如下：\n```yaml\n{profile_str}\n```\n\n请基于这些信息回答用户的问题。"
    else:
        system_message += "\n\n注意：用户尚未上传个人资料配置文件。请提醒用户先上传 YAML 配置文件。"
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    try:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_message,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return jsonify({
            'success': True,
            'response': assistant_message
        })
        
    except Exception as e:
        # Remove the user message if API call failed
        conversation_history.pop()
        return jsonify({'success': False, 'error': f'AI 响应错误: {str(e)}'})

@app.route('/clear', methods=['POST'])
def clear_profile():
    global user_profile, conversation_history
    user_profile = {}
    conversation_history = []
    return jsonify({'success': True, 'message': '个人资料已清除'})

@app.route('/profile', methods=['GET'])
def get_profile():
    return jsonify({'success': True, 'profile': user_profile})

if __name__ == '__main__':
