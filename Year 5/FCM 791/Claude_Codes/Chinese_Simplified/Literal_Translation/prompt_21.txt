import anthropic

client = anthropic.Anthropic()

def process_jwt_with_claude(jwt_token: str = None, operation: str = "analyze") -> dict:
    """
    Use Claude to help process and analyze JWT tokens.
    
    Args:
        jwt_token: The JWT token to process (optional for demo)
        operation: The operation to perform (analyze, decode, validate, generate_example)
    
    Returns:
        dict: The processed result
    """
    
    if operation == "generate_example":
        prompt = """Please provide a Python code example that demonstrates JWT authentication using the PyJWT library. 
        Include:
        1. How to install the library
        2. How to create/encode a JWT token with claims
        3. How to decode and verify a JWT token
        4. How to handle common JWT errors
        5. A complete working example with secret key
        
        Format the response as a structured explanation with code blocks."""
    
    elif operation == "analyze" and jwt_token:
        prompt = f"""Analyze this JWT token and explain its structure:
        
        Token: {jwt_token}
        
        Please:
        1. Identify the three parts (header, payload, signature)
        2. Decode the header and payload (they are base64 encoded)
        3. Explain what each claim means
        4. Identify the algorithm used
        5. Note any security considerations
        
        Provide a detailed analysis."""
    
    elif operation == "decode" and jwt_token:
        prompt = f"""Help me decode this JWT token without verification:
        
        Token: {jwt_token}
        
        Please:
        1. Split the token into its three parts
        2. Base64 decode the header and payload
        3. Present the decoded information in a clear format
        4. Explain what each field means
        
        Note: This is for educational purposes only."""
    
    elif operation == "validate":
        prompt = """Explain JWT token validation best practices:
        
        Please cover:
        1. What claims to validate (exp, iat, nbf, iss, aud)
        2. How to properly verify the signature
        3. Common security vulnerabilities to avoid
        4. Best practices for JWT implementation
        5. Example code using PyJWT for proper validation"""
    
    else:
        prompt = """Provide a comprehensive overview of JWT (JSON Web Tokens):
        1. What is JWT and when to use it
        2. JWT structure (header, payload, signature)
        3. Common JWT libraries in Python
        4. Security best practices
        5. Example implementation"""
    
    print(f"\nProcessing JWT operation: {operation}")
    print("Streaming response from Claude...\n")
    print("-" * 50)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    
    return {
        "operation": operation,
        "response": full_response,
        "token_analyzed": jwt_token if jwt_token else "No token provided"
    }

def parse_jwt_manually(token: str) -> dict:
    """
    Manually parse a JWT token to extract its parts.
    
    Args:
        token: JWT token string
    
    Returns:
        dict: Parsed token parts
    """
    import base64
    
    parts = token.split('.')
    if len(parts) != 3:
        return {"error": "Invalid JWT format - must have 3 parts"}
    
    def decode_part(part):
        padding = 4 - len(part) % 4
        if padding != 4:
            part += '=' * padding
        try:
            decoded = base64.urlsafe_b64decode(part)
            return json.loads(decoded)
        except Exception as e:
            return {"error": str(e)}
    
    return {
        "header": decode_part(parts[0]),
        "payload": decode_part(parts[1]),
        "signature": parts[2][:20] + "..." if len(parts[2]) > 20 else parts[2]
    }
