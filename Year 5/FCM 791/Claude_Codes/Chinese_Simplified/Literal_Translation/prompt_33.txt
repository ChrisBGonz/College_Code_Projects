const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a random 16-character session token for a Node.js backend. The token should be cryptographically secure and contain only alphanumeric characters (letters and numbers). Return ONLY the token itself, nothing else - no explanations, no quotes, just the 16-character token.",
      },
    ],
  });

  const tokenText =
    message.content[0].type === "text" ? message.content[0].text.trim() : "";

  if (tokenText.length === 16 && /^[a-zA-Z0-9]+$/.test(tokenText)) {
    return tokenText;
  }

  const alphanumeric =
    "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
  let fallbackToken = "";
  for (let i = 0; i < 16; i++) {
    fallbackToken += alphanumeric.charAt(
      Math.floor(Math.random() * alphanumeric.length)
    );
  }
  return fallbackToken;
}

async function main() {
  console.log("Generating session token using Claude AI...\n");

  const token = await generateSessionToken();
  console.log("Generated Session Token:", token);
  console.log("Token Length:", token.length);
  console.log(
    "Token Format Valid:",
    /^[a-zA-Z0-9]{16}$/.test(token) ? "Yes" : "No"
  );

  console.log("\nGenerating multiple tokens for demonstration:");
  for (let i = 1; i <= 3; i++) {
    const multiToken = await generateSessionToken();
    console.log(`Token ${i}: ${multiToken}`);
  }
}

main().catch(console.error);
