using System;
using System.Threading.Tasks;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;
using System.Text.Json;
using System.Collections.Generic;

namespace MFASettingsLogger
{
    // MFA Settings Change Event
    public class MFASettingsChangeEvent
    {
        public string UserId { get; set; }
        public string Username { get; set; }
        public string ChangeType { get; set; } // Enable, Disable, Update
        public string MFAMethod { get; set; } // TOTP, SMS, Email, etc.
        public DateTime Timestamp { get; set; }
        public string IPAddress { get; set; }
        public string UserAgent { get; set; }
        public bool Success { get; set; }
        public string AdditionalInfo { get; set; }
    }

    // MFA Settings Logger
    public class MFASettingsLogger
    {
        private readonly AmazonBedrockRuntimeClient _bedrockClient;
        private readonly List<MFASettingsChangeEvent> _auditLog;
        private const string ModelId = "amazon.nova-lite-v1:0";

        public MFASettingsLogger()
        {
            _bedrockClient = new AmazonBedrockRuntimeClient();
            _auditLog = new List<MFASettingsChangeEvent>();
        }

        // Log MFA settings change
        public async Task LogMFASettingsChange(MFASettingsChangeEvent changeEvent)
        {
            // Add to audit log
            _auditLog.Add(changeEvent);

            // Log to console
            Console.WriteLine($"\n[{changeEvent.Timestamp:yyyy-MM-dd HH:mm:ss}] MFA Settings Change Detected:");
            Console.WriteLine($"  User: {changeEvent.Username} (ID: {changeEvent.UserId})");
            Console.WriteLine($"  Change Type: {changeEvent.ChangeType}");
            Console.WriteLine($"  MFA Method: {changeEvent.MFAMethod}");
            Console.WriteLine($"  Success: {changeEvent.Success}");
            Console.WriteLine($"  IP Address: {changeEvent.IPAddress}");
            Console.WriteLine($"  User Agent: {changeEvent.UserAgent}");
            if (!string.IsNullOrEmpty(changeEvent.AdditionalInfo))
            {
                Console.WriteLine($"  Additional Info: {changeEvent.AdditionalInfo}");
            }

            // Analyze the change using Amazon Bedrock
            await AnalyzeMFAChangeWithBedrock(changeEvent);
        }

        // Analyze MFA change using Amazon Bedrock
        private async Task AnalyzeMFAChangeWithBedrock(MFASettingsChangeEvent changeEvent)
        {
            try
            {
                var prompt = $@"Analyze the following MFA (Multi-Factor Authentication) settings change event and provide:
1. Security risk assessment (Low/Medium/High)
2. Whether this change appears suspicious or normal
3. Recommended actions if any

Event Details:
- User: {changeEvent.Username}
- Change Type: {changeEvent.ChangeType}
- MFA Method: {changeEvent.MFAMethod}
- Timestamp: {changeEvent.Timestamp:yyyy-MM-dd HH:mm:ss}
- Success: {changeEvent.Success}
- IP Address: {changeEvent.IPAddress}
- Additional Info: {changeEvent.AdditionalInfo}

Please provide a concise security analysis.";

                var requestBody = new
                {
                    messages = new[]
                    {
                        new
                        {
                            role = "user",
                            content = new[]
                            {
                                new
                                {
                                    type = "text",
                                    text = prompt
                                }
                            }
                        }
                    },
                    inferenceConfig = new
                    {
                        maxTokens = 500,
                        temperature = 0.3
                    }
                };

                var requestJson = JsonSerializer.Serialize(requestBody);

                var request = new InvokeModelRequest
                {
                    ModelId = ModelId,
                    ContentType = "application/json",
                    Accept = "application/json",
                    Body = new System.IO.MemoryStream(System.Text.Encoding.UTF8.GetBytes(requestJson))
                };

                var response = await _bedrockClient.InvokeModelAsync(request);

                var responseBody = await new System.IO.StreamReader(response.Body).ReadToEndAsync();
                var responseJson = JsonDocument.Parse(responseBody);

                // Extract the text from the response
                var content = responseJson.RootElement
                    .GetProperty("output")
                    .GetProperty("message")
                    .GetProperty("content")[0]
                    .GetProperty("text")
                    .GetString();

                Console.WriteLine("\n[AI Security Analysis]:");
                Console.WriteLine(content);
                Console.WriteLine(new string('-', 60));
            }
            catch (Exception ex)
            {
                Console.WriteLine($"\n[Warning] Could not perform AI analysis: {ex.Message}");
            }
        }

        // Get audit log
        public List<MFASettingsChangeEvent> GetAuditLog()
        {
            return _auditLog;
        }

        // Generate audit report
        public async Task GenerateAuditReport()
        {
            if (_auditLog.Count == 0)
            {
                Console.WriteLine("No MFA changes recorded.");
                return;
            }

            Console.WriteLine("\n" + new string('=', 60));
            Console.WriteLine("MFA SETTINGS AUDIT REPORT");
            Console.WriteLine(new string('=', 60));
            Console.WriteLine($"Total Changes: {_auditLog.Count}");
            Console.WriteLine($"Report Generated: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss} UTC");

            // Summarize changes
            var enableCount = 0;
            var disableCount = 0;
            var updateCount = 0;
            var failedCount = 0;

            foreach (var evt in _auditLog)
            {
                switch (evt.ChangeType.ToLower())
                {
                    case "enable": enableCount++; break;
                    case "disable": disableCount++; break;
                    case "update": updateCount++; break;
                }
                if (!evt.Success) failedCount++;
            }

            Console.WriteLine($"\nSummary:");
            Console.WriteLine($"  MFA Enabled: {enableCount}");
            Console.WriteLine($"  MFA Disabled: {disableCount}");
            Console.WriteLine($"  MFA Updated: {updateCount}");
            Console.WriteLine($"  Failed Attempts: {failedCount}");

            // Use Bedrock to generate comprehensive report
            await GenerateComprehensiveReportWithBedrock();
        }

        private async Task GenerateComprehensiveReportWithBedrock()
        {
            try
            {
                var logSummary = JsonSerializer.Serialize(_auditLog, new JsonSerializerOptions { WriteIndented = true });

                var prompt = $@"Based on the following MFA settings change audit log, provide a comprehensive security report including:
1. Overall security posture assessment
2. Patterns or anomalies detected
3. Users with concerning activity
4. Recommendations for improving MFA security

Audit Log:
{logSummary}

Please provide a structured security report.";

                var requestBody = new
                {
                    messages = new[]
                    {
                        new
                        {
                            role = "user",
                            content = new[]
                            {
                                new
                                {
                                    type = "text",
                                    text = prompt
                                }
                            }
                        }
                    },
                    inferenceConfig = new
                    {
                        maxTokens = 1000,
                        temperature = 0.3
                    }
                };

                var requestJson = JsonSerializer.Serialize(requestBody);

                var request = new InvokeModelRequest
                {
                    ModelId = ModelId,
                    ContentType = "application/json",
                    Accept = "application/json",
                    Body = new System.IO.MemoryStream(System.Text.Encoding.UTF8.GetBytes(requestJson))
                };

                var response = await _bedrockClient.InvokeModelAsync(request);
                var responseBody = await new System.IO.StreamReader(response.Body).ReadToEndAsync();
                var responseJson = JsonDocument.Parse(responseBody);

                var content = responseJson.RootElement
                    .GetProperty("output")
                    .GetProperty("message")
                    .GetProperty("content")[0]
                    .GetProperty("text")
                    .GetString();

                Console.WriteLine("\n[AI Comprehensive Security Report]:");
                Console.WriteLine(content);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"\n[Warning] Could not generate comprehensive report: {ex.Message}");
            }
        }
    }
