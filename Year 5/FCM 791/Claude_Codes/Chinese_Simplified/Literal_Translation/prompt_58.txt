const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  // Initialize form data from localStorage or return empty object
  const getStoredData = () => {
    try {
      const data = localStorage.getItem(STORAGE_KEY);
      return data ? JSON.parse(data) : {};
    } catch (error) {
      console.error('Error reading from localStorage:', error);
      return {};
    }
  };

  // Get current step from localStorage
  const getStoredStep = () => {
    try {
      const step = localStorage.getItem(CURRENT_STEP_KEY);
      return step ? parseInt(step, 10) : 1;
    } catch (error) {
      console.error('Error reading step from localStorage:', error);
      return 1;
    }
  };

  // Save form data to localStorage
  const saveStepData = (stepData) => {
    try {
      const existingData = getStoredData();
      const updatedData = { ...existingData, ...stepData };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(updatedData));
      return true;
    } catch (error) {
      console.error('Error saving to localStorage:', error);
      return false;
    }
  };

  // Save current step to localStorage
  const saveCurrentStep = (step) => {
    try {
      localStorage.setItem(CURRENT_STEP_KEY, step.toString());
    } catch (error) {
      console.error('Error saving step to localStorage:', error);
    }
  };

  // Clear all stored form data
  const clearStoredData = () => {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
    } catch (error) {
      console.error('Error clearing localStorage:', error);
    }
  };

  // Validate step data based on step number
  const validateStep = (step, data) => {
    const errors = [];

    switch (step) {
      case 1:
        if (!data.firstName || data.firstName.trim() === '') {
          errors.push('First name is required');
        }
        if (!data.lastName || data.lastName.trim() === '') {
          errors.push('Last name is required');
        }
        if (!data.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(data.email)) {
          errors.push('Valid email is required');
        }
        break;

      case 2:
        if (!data.username || data.username.trim().length < 3) {
          errors.push('Username must be at least 3 characters');
        }
        if (!data.password || data.password.length < 8) {
          errors.push('Password must be at least 8 characters');
        }
        if (data.password !== data.confirmPassword) {
          errors.push('Passwords do not match');
        }
        break;

      case 3:
        if (!data.country || data.country.trim() === '') {
          errors.push('Country is required');
        }
        if (!data.city || data.city.trim() === '') {
          errors.push('City is required');
        }
        break;

      case 4:
        if (!data.agreeToTerms) {
          errors.push('You must agree to the terms and conditions');
        }
        break;

      default:
        break;
    }

    return errors;
  };

  // Collect form data from current step's fields
  const collectStepData = (stepElement) => {
    const data = {};
    const inputs = stepElement.querySelectorAll('input, select, textarea');

    inputs.forEach((input) => {
      if (input.name) {
        if (input.type === 'checkbox') {
          data[input.name] = input.checked;
        } else if (input.type === 'radio') {
          if (input.checked) {
            data[input.name] = input.value;
          }
        } else {
          data[input.name] = input.value;
        }
      }
    });

    return data;
  };

  // Populate form fields with stored data
  const populateStepFields = (stepElement, storedData) => {
    const inputs = stepElement.querySelectorAll('input, select, textarea');

    inputs.forEach((input) => {
      if (input.name && storedData[input.name] !== undefined) {
        if (input.type === 'checkbox') {
          input.checked = storedData[input.name];
        } else if (input.type === 'radio') {
          input.checked = input.value === storedData[input.name];
        } else {
          input.value = storedData[input.name];
        }
      }
    });
  };

  // Display validation errors
  const displayErrors = (errors, errorContainer) => {
    if (!errorContainer) return;

    errorContainer.innerHTML = '';

    if (errors.length > 0) {
      const errorList = document.createElement('ul');
      errorList.className = 'form-errors';

      errors.forEach((error) => {
        const li = document.createElement('li');
        li.textContent = error;
        li.className = 'form-error-item';
        errorList.appendChild(li);
      });

      errorContainer.appendChild(errorList);
      errorContainer.style.display = 'block';
    } else {
      errorContainer.style.display = 'none';
    }
  };

  // Update progress indicator
  const updateProgress = (currentStep, totalSteps) => {
    const progressBar = document.querySelector('.form-progress-bar');
    const stepIndicators = document.querySelectorAll('.step-indicator');

    if (progressBar) {
      const percentage = ((currentStep - 1) / (totalSteps - 1)) * 100;
      progressBar.style.width = `${percentage}%`;
    }

    stepIndicators.forEach((indicator, index) => {
      const stepNumber = index + 1;
      indicator.classList.remove('active', 'completed');

      if (stepNumber === currentStep) {
        indicator.classList.add('active');
      } else if (stepNumber < currentStep) {
        indicator.classList.add('completed');
      }
    });
  };

  // Show specific step
  const showStep = (stepNumber) => {
    const allSteps = document.querySelectorAll('.form-step');
    const storedData = getStoredData();

    allSteps.forEach((step) => {
      step.style.display = 'none';
      step.classList.remove('active');
    });

    const targetStep = document.querySelector(`[data-step="${stepNumber}"]`);
    if (targetStep) {
      targetStep.style.display = 'block';
      targetStep.classList.add('active');
      populateStepFields(targetStep, storedData);
    }

    updateProgress(stepNumber, allSteps.length);
    saveCurrentStep(stepNumber);
  };

  // Handle next button click
  const handleNext = (currentStep) => {
    const currentStepElement = document.querySelector(`[data-step="${currentStep}"]`);
    const errorContainer = document.querySelector('.form-error-container');

    if (!currentStepElement) return;

    const stepData = collectStepData(currentStepElement);
    const errors = validateStep(currentStep, stepData);

    if (errors.length > 0) {
      displayErrors(errors, errorContainer);
      return;
    }

    displayErrors([], errorContainer);
    const saved = saveStepData(stepData);

    if (saved) {
      console.log(`Step ${currentStep} data saved:`, stepData);
      showStep(currentStep + 1);
    } else {
      alert('Error saving your data. Please try again.');
    }
  };

  // Handle previous button click
  const handlePrev = (currentStep) => {
    const errorContainer = document.querySelector('.form-error-container');
    displayErrors([], errorContainer);
    showStep(currentStep - 1);
  };

  // Handle form submission
  const handleSubmit = async (currentStep) => {
    const currentStepElement = document.querySelector(`[data-step="${currentStep}"]`);
    const errorContainer = document.querySelector('.form-error-container');

    if (!currentStepElement) return;

    const stepData = collectStepData(currentStepElement);
    const errors = validateStep(currentStep, stepData);

    if (errors.length > 0) {
      displayErrors(errors, errorContainer);
      return;
    }

    saveStepData(stepData);
    const finalData = getStoredData();

    console.log('Final form data to submit:', finalData);

    try {
      // Simulate API call - replace with actual endpoint
      const response = await submitFormData(finalData);

      if (response.success) {
        clearStoredData();
        showSuccessMessage();
        console.log('Form submitted successfully!');
      } else {
        throw new Error(response.message || 'Submission failed');
      }
    } catch (error) {
      console.error('Form submission error:', error);
      displayErrors([`Submission failed: ${error.message}`], errorContainer);
    }
  };

  // Simulate form data submission (replace with actual API call)
  const submitFormData = (data) => {
    return new Promise((resolve) => {
      setTimeout(() => {
        console.log('Submitting data to server:', data);
        resolve({ success: true, message: 'Registration successful!' });
      }, 1500);
    });
  };

  // Show success message after submission
  const showSuccessMessage = () => {
    const formContainer = document.querySelector('.multi-step-form');
    if (formContainer) {
      formContainer.innerHTML = `
        <div class="success-message">
          <h2>Registration Successful!</h2>
          <p>Thank you for registering. Please check your email for confirmation.</p>
        </div>
      `;
    }
  };

  // Initialize the form
  const init = () => {
    const form = document.querySelector('.multi-step-form');
    if (!form) {
      console.error('Multi-step form not found');
      return;
    }

    const currentStep = getStoredStep();
    showStep(currentStep);

    // Event delegation for button clicks
    form.addEventListener('click', (event) => {
      const target = event.target;

      if (target.classList.contains('btn-next')) {
        const activeStep = document.querySelector('.form-step.active');
        const stepNumber = parseInt(activeStep?.dataset.step || '1', 10);
        handleNext(stepNumber);
      }

      if (target.classList.contains('btn-prev')) {
        const activeStep = document.querySelector('.form-step.active');
        const stepNumber = parseInt(activeStep?.dataset.step || '1', 10);
        handlePrev(stepNumber);
      }

      if (target.classList.contains('btn-submit')) {
        event.preventDefault();
        const activeStep = document.querySelector('.form-step.active');
        const stepNumber = parseInt(activeStep?.dataset.step || '1', 10);
        handleSubmit(stepNumber);
      }

      if (target.classList.contains('btn-reset')) {
        if (confirm('Are you sure you want to reset the form? All data will be lost.')) {
          clearStoredData();
          showStep(1);
        }
      }
    });

    // Auto-save on input change
    form.addEventListener('change', (event) => {
      const activeStep = document.querySelector('.form-step.active');
      if (activeStep) {
        const stepData = collectStepData(activeStep);
        saveStepData(stepData);
        console.log('Auto-saved step data:', stepData);
      }
    });

    console.log('Multi-step form initialized. Resuming from step:', currentStep);
  };

  // Public API
  return {
    init,
    getStoredData,
    clearStoredData,
    saveStepData,
    showStep,
  };
})();

// Auto-initialize when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
  MultiStepForm.init();
});
