using System;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;
using System.Text;
using System.Text.Json;
using System.Security.Cryptography;

class Program
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export keys in PEM format
        string privateKeyPem = ExportPrivateKeyToPem(rsa);
        string publicKeyPem = ExportPublicKeyToPem(rsa);
        
        Console.WriteLine("RSA Key Pair Generated Successfully!");
        Console.WriteLine("=====================================\n");
        Console.WriteLine("Public Key (PEM format):");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\nPrivate Key (PEM format):");
        Console.WriteLine(privateKeyPem);
        
        // Use Claude to explain the generated keys
        await ExplainKeysWithClaude(publicKeyPem, privateKeyPem);
        
        // Demonstrate signing and verification
        DemonstrateSigningAndVerification(rsa);
    }
    
    static string ExportPublicKeyToPem(RSA rsa)
    {
        byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        string base64 = Convert.ToBase64String(publicKeyBytes);
        
        var sb = new StringBuilder();
        sb.AppendLine("-----BEGIN PUBLIC KEY-----");
        
        // Split base64 into 64-character lines
        for (int i = 0; i < base64.Length; i += 64)
        {
            int length = Math.Min(64, base64.Length - i);
            sb.AppendLine(base64.Substring(i, length));
        }
        
        sb.Append("-----END PUBLIC KEY-----");
        return sb.ToString();
    }
    
    static string ExportPrivateKeyToPem(RSA rsa)
    {
        byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        string base64 = Convert.ToBase64String(privateKeyBytes);
        
        var sb = new StringBuilder();
        sb.AppendLine("-----BEGIN PRIVATE KEY-----");
        
        // Split base64 into 64-character lines
        for (int i = 0; i < base64.Length; i += 64)
        {
            int length = Math.Min(64, base64.Length - i);
            sb.AppendLine(base64.Substring(i, length));
        }
        
        sb.Append("-----END PRIVATE KEY-----");
        return sb.ToString();
    }
    
    static async Task ExplainKeysWithClaude(string publicKey, string privateKey)
    {
        Console.WriteLine("\n\nAsking Claude to explain the RSA key pair...");
        Console.WriteLine("============================================\n");
        
        var client = new AmazonBedrockRuntimeClient(Amazon.RegionEndpoint.USEast1);
        
        string prompt = $@"I've generated an RSA 2048-bit key pair for digital signatures. Here are the keys:

Public Key:
{publicKey}

Private Key:
{privateKey}

Please explain:
1. What these keys are used for in digital signatures
2. The difference between the public and private key
3. Best practices for storing and using these keys securely
4. How RSA digital signatures work at a high level

Keep the explanation concise and practical.";

        var requestBody = new
        {
            anthropic_version = "bedrock-2023-05-31",
            max_tokens = 1024,
            messages = new[]
            {
                new
                {
                    role = "user",
                    content = prompt
                }
            }
        };
        
        string requestBodyJson = JsonSerializer.Serialize(requestBody);
        
        var request = new InvokeModelRequest
        {
            ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0",
            ContentType = "application/json",
            Accept = "application/json",
            Body = new MemoryStream(Encoding.UTF8.GetBytes(requestBodyJson))
        };
        
        var response = await client.InvokeModelAsync(request);
        
        using var reader = new StreamReader(response.Body);
        string responseBody = await reader.ReadToEndAsync();
        
        using var jsonDoc = JsonDocument.Parse(responseBody);
        string explanation = jsonDoc.RootElement
            .GetProperty("content")[0]
            .GetProperty("text")
            .GetString() ?? "No explanation received";
        
        Console.WriteLine("Claude's Explanation:");
        Console.WriteLine("--------------------");
        Console.WriteLine(explanation);
    }
    
    static void DemonstrateSigningAndVerification(RSA rsa)
    {
        Console.WriteLine("\n\nDemonstrating Digital Signature Process:");
        Console.WriteLine("=========================================\n");
        
        // Sample message to sign
        string message = "This is a test message for digital signature demonstration.";
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);
        
        Console.WriteLine($"Original Message: {message}\n");
        
        // Sign the message
        byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        string signatureBase64 = Convert.ToBase64String(signature);
        
        Console.WriteLine($"Digital Signature (Base64): {signatureBase64[..50]}...\n");
        
        // Verify the signature
        bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"Signature Verification Result: {(isValid ? "VALID ✓" : "INVALID ✗")}\n");
        
        // Demonstrate tampered message verification
        string tamperedMessage = "This is a TAMPERED message for digital signature demonstration.";
        byte[] tamperedMessageBytes = Encoding.UTF8.GetBytes(tamperedMessage);
        
        bool isTamperedValid = rsa.VerifyData(tamperedMessageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"Tampered Message Verification Result: {(isTamperedValid ? "VALID ✓" : "INVALID ✗ (Expected - message was tampered)")}");
        
        // Export key information
        RSAParameters parameters = rsa.ExportParameters(false);
        Console.WriteLine($"\nKey Size: {rsa.KeySize} bits");
        Console.WriteLine($"Key Modulus Length: {parameters.Modulus?.Length * 8} bits");
    }
}
