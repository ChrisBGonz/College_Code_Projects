using System;
using System.Collections.Generic;
using System.Data;
using Amazon;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;
using System.Text.Json;
using System.Text.Json.Nodes;

class DynamicSQLQueryWithClaude
{
    private static readonly AmazonBedrockRuntimeClient bedrockClient = new AmazonBedrockRuntimeClient(RegionEndpoint.USEast1);
    
    // Simulated database with sample data
    private static readonly DataTable sampleDatabase = CreateSampleDatabase();
    
    static DataTable CreateSampleDatabase()
    {
        var table = new DataTable("employees");
        table.Columns.Add("id", typeof(int));
        table.Columns.Add("name", typeof(string));
        table.Columns.Add("department", typeof(string));
        table.Columns.Add("salary", typeof(decimal));
        table.Columns.Add("hire_date", typeof(DateTime));
        
        table.Rows.Add(3, "Charlie Brown", "Engineering", 85000.00m, new DateTime(2020, 3, 15));
        table.Rows.Add(1, "Alice Johnson", "Marketing", 75000.00m, new DateTime(2019, 1, 10));
        table.Rows.Add(5, "Eve Davis", "HR", 65000.00m, new DateTime(2021, 7, 20));
        table.Rows.Add(2, "Bob Smith", "Engineering", 90000.00m, new DateTime(2018, 6, 5));
        table.Rows.Add(4, "David Wilson", "Marketing", 70000.00m, new DateTime(2022, 2, 28));
        
        return table;
    }
    
    static async Task<string> GenerateSQLWithClaude(string tableName, string columnName, string sortOrder = "ASC")
    {
        // Validate column name to prevent SQL injection
        var validColumns = new HashSet<string> { "id", "name", "department", "salary", "hire_date" };
        if (!validColumns.Contains(columnName.ToLower()))
        {
            throw new ArgumentException($"Invalid column name: {columnName}. Valid columns are: {string.Join(", ", validColumns)}");
        }
        
        var prompt = $@"Generate a SQL SELECT query for a table named '{tableName}' that:
1. Selects all columns (*)
2. Orders the results by the column '{columnName}' in {sortOrder} order
3. The query should be safe and follow best practices

Please provide ONLY the SQL query without any explanation or markdown formatting.
The query should be a single line.";

        var requestBody = new JsonObject
        {
            ["anthropic_version"] = "bedrock-2023-05-31",
            ["max_tokens"] = 256,
            ["messages"] = new JsonArray
            {
                new JsonObject
                {
                    ["role"] = "user",
                    ["content"] = prompt
                }
            }
        };

        var request = new InvokeModelRequest
        {
            ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0",
            ContentType = "application/json",
            Accept = "application/json",
            Body = new System.IO.MemoryStream(System.Text.Encoding.UTF8.GetBytes(requestBody.ToString()))
        };

        var response = await bedrockClient.InvokeModelAsync(request);
        var responseBody = await new System.IO.StreamReader(response.Body).ReadToEndAsync();
        var jsonResponse = JsonDocument.Parse(responseBody);
        
        var generatedSQL = jsonResponse.RootElement
            .GetProperty("content")[0]
            .GetProperty("text")
            .GetString()
            ?.Trim();
            
        return generatedSQL ?? throw new Exception("Failed to generate SQL query");
    }
    
    static DataTable ExecuteDynamicQuery(string columnName, string sortOrder = "ASC")
    {
        // Validate column name
        var validColumns = new HashSet<string> { "id", "name", "department", "salary", "hire_date" };
        if (!validColumns.Contains(columnName.ToLower()))
        {
            throw new ArgumentException($"Invalid column name: {columnName}");
        }
        
        // Create a copy of the table to sort
        var sortedTable = sampleDatabase.Clone();
        
        // Get all rows and sort them
        var rows = sampleDatabase.Select("", $"{columnName} {sortOrder}");
        foreach (var row in rows)
        {
            sortedTable.ImportRow(row);
        }
        
        return sortedTable;
    }
    
    static void DisplayResults(DataTable table, string sql)
    {
        Console.WriteLine($"\nExecuted SQL: {sql}");
        Console.WriteLine("\nQuery Results:");
        Console.WriteLine(new string('-', 80));
        
        // Print headers
        var headers = new List<string>();
        foreach (DataColumn col in table.Columns)
        {
            headers.Add(col.ColumnName.PadRight(15));
        }
        Console.WriteLine(string.Join(" | ", headers));
        Console.WriteLine(new string('-', 80));
        
        // Print rows
        foreach (DataRow row in table.Rows)
        {
            var values = new List<string>();
            foreach (DataColumn col in table.Columns)
            {
                var value = row[col]?.ToString() ?? "NULL";
                if (col.DataType == typeof(DateTime))
                {
                    value = ((DateTime)row[col]).ToString("yyyy-MM-dd");
                }
                else if (col.DataType == typeof(decimal))
                {
                    value = ((decimal)row[col]).ToString("C0");
                }
                values.Add(value.PadRight(15));
            }
            Console.WriteLine(string.Join(" | ", values));
        }
        Console.WriteLine(new string('-', 80));
        Console.WriteLine($"Total rows: {table.Rows.Count}");
    }
    
    public static async Task ExecuteDynamicSQLQuery(string columnName, string sortOrder = "ASC")
    {
        Console.WriteLine($"\n{'='*60}");
        Console.WriteLine($"Executing Dynamic SQL Query");
        Console.WriteLine($"Column: {columnName}, Sort Order: {sortOrder}");
        Console.WriteLine($"{'='*60}");
        
        try
        {
            // Generate SQL using Claude
            Console.WriteLine("\nGenerating SQL query using Claude AI...");
            var generatedSQL = await GenerateSQLWithClaude("employees", columnName, sortOrder);
            
            // Execute the query (simulated)
            var results = ExecuteDynamicQuery(columnName, sortOrder);
            
            // Display results
            DisplayResults(results, generatedSQL);
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine($"\nValidation Error: {ex.Message}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"\nError: {ex.Message}");
        }
    }
    
    static async Task Main(string[] args)
    {
        Console.WriteLine("Dynamic SQL Query Executor with Claude AI");
        Console.WriteLine("==========================================");
        
        // Test different sorting scenarios
        var testCases = new[]
        {
            ("name", "ASC"),
            ("salary", "DESC"),
            ("hire_date", "ASC"),
            ("department", "ASC"),
            ("id", "DESC")
        };
        
        foreach (var (column, order) in testCases)
        {
            await ExecuteDynamicSQLQuery(column, order);
            await Task.Delay(1000); // Small delay between API calls
        }
        
        // Test with invalid column (should show validation error)
        Console.WriteLine("\nTesting with invalid column name...");
        await ExecuteDynamicSQLQuery("invalid_column", "ASC");
        
        Console.WriteLine("\n\nDynamic SQL Query execution completed!");
    }
}
