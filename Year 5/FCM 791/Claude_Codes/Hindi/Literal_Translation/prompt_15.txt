
```python
import anthropic

client = anthropic.Anthropic()

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {
            "role": "user",
            "content": """एक कॉन्फ़िगरेशन फ़ाइल (YAML) बनाएँ जो एक Kubernetes Deployment के लिए हो, 
            जो एक सार्वजनिक रूप से सुलभ Nginx वेब सर्वर चलाता हो। 
            इसमें निम्नलिखित शामिल करें:
            1. Namespace
            2. Deployment (3 replicas के साथ)
            3. Service (LoadBalancer type)
            4. ConfigMap (custom nginx configuration के साथ)
            5. Resource limits और requests
            6. Health checks (liveness और readiness probes)
            
            कृपया पूरी YAML फ़ाइल प्रदान करें और प्रत्येक section को हिंदी में समझाएं।""",
        }
    ],
)

print(message.content[0].text)

yaml_content = message.content[0].text

with open("kubernetes-nginx-deployment.yaml", "w", encoding="utf-8") as f:
    f.write("# Kubernetes Nginx Deployment Configuration\n")
    f.write("# Generated by Claude AI\n\n")
    f.write(yaml_content)

print("\n" + "=" * 50)
print("YAML फ़ाइल 'kubernetes-nginx-deployment.yaml' में सहेजी गई है।")
print("=" * 50)
```
