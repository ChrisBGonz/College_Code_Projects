def get_invoice_from_database(invoice_id: str) -> dict | None:
    """Simulate database query to get invoice details"""
    return MOCK_DATABASE.get(invoice_id.upper())

def process_invoice_with_claude(invoice_data: dict) -> dict:
    """Use Claude to analyze and enhance invoice information"""
    
    invoice_json = json.dumps(invoice_data, ensure_ascii=False, indent=2)
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""निम्नलिखित इनवॉइस डेटा का विश्लेषण करें और एक संरचित सारांश प्रदान करें:

{invoice_json}

कृपया निम्नलिखित जानकारी JSON फॉर्मेट में प्रदान करें:
1. invoice_summary: इनवॉइस का संक्षिप्त विवरण
2. payment_status_message: भुगतान स्थिति के बारे में संदेश
3. total_items: कुल आइटम की संख्या
4. recommendations: यदि कोई सुझाव हो (जैसे overdue के लिए reminder)

केवल JSON फॉर्मेट में उत्तर दें, कोई अतिरिक्त टेक्स्ट नहीं।"""
            }
        ]
    )
    
    # Parse Claude's response
    claude_response = message.content[0].text
    
    # Try to parse as JSON
    try:
        # Remove markdown code blocks if present
        if "```json" in claude_response:
            claude_response = claude_response.split("```json")[1].split("```")[0].strip()
        elif "```" in claude_response:
            claude_response = claude_response.split("```")[1].split("```")[0].strip()
        
        claude_analysis = json.loads(claude_response)
    except json.JSONDecodeError:
        claude_analysis = {
            "invoice_summary": "इनवॉइस विवरण उपलब्ध है",
            "payment_status_message": f"भुगतान स्थिति: {invoice_data.get('status', 'अज्ञात')}",
            "total_items": len(invoice_data.get('items', [])),
            "recommendations": "कोई विशेष सुझाव नहीं"
        }
    
    return claude_analysis

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Get invoice details by invoice_id from URL
    
    URL: /invoice/<invoice_id>
    Method: GET
    
    Returns:
        JSON response with invoice details and Claude's analysis
    """
    
    # Validate invoice_id
    if not invoice_id:
        return jsonify({
            "error": "इनवॉइस ID आवश्यक है",
            "status": "error"
        }), 400
    
    # Get invoice from database
    invoice_data = get_invoice_from_database(invoice_id)
    
    if not invoice_data:
        return jsonify({
            "error": f"इनवॉइस '{invoice_id}' नहीं मिला",
            "status": "not_found",
            "available_invoices": list(MOCK_DATABASE.keys())
        }), 404
    
    # Process invoice with Claude AI
    claude_analysis = process_invoice_with_claude(invoice_data)
    
    # Prepare response
    response = {
        "status": "success",
        "invoice_details": invoice_data,
        "ai_analysis": claude_analysis,
        "message": "इनवॉइस विवरण सफलतापूर्वक प्राप्त किया गया"
    }
    
    return jsonify(response), 200
