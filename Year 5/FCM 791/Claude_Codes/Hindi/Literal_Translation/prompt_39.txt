import anthropic
import socket
import threading
import time
import random
import string

# XOR cipher key for "security"
XOR_KEY = 0x42  # Simple XOR key

def xor_encrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Encrypt/decrypt data using XOR cipher"""
    return bytes([b ^ key for b in data])

def xor_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """XOR is symmetric, so decrypt is same as encrypt"""
    return xor_encrypt(data, key)

class LegacyServer:
    """Simulates a legacy server that communicates via custom socket with XOR encryption"""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        self.responses = {
            "hello": "Hello! I am the legacy server. How can I help you?",
            "status": "Server Status: OPERATIONAL | Uptime: 99.9% | Load: LOW",
            "data": "Legacy Data: [Record_001, Record_002, Record_003, Record_004]",
            "version": "Legacy Server v1.0.0 - Built 1995",
            "help": "Available commands: hello, status, data, version, help, quit",
        }
    
    def start(self):
        """Start the legacy server"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy Server started on {self.host}:{self.port}")
        
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Legacy Server: New connection from {address}")
                client_thread = threading.Thread(
                    target=self.handle_client,
                    args=(client_socket, address)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Legacy Server error: {e}")
    
    def handle_client(self, client_socket: socket.socket, address):
        """Handle client connection"""
        try:
            while True:
                # Receive encrypted data
                encrypted_data = client_socket.recv(1024)
                if not encrypted_data:
                    break
                
                # Decrypt the received data
                decrypted_data = xor_decrypt(encrypted_data)
                message = decrypted_data.decode('utf-8').strip().lower()
                print(f"Legacy Server received: '{message}' from {address}")
                
                # Generate response
                response = self.responses.get(message, f"Unknown command: '{message}'. Type 'help' for available commands.")
                
                # Encrypt and send response
                encrypted_response = xor_encrypt(response.encode('utf-8'))
                client_socket.send(encrypted_response)
                
                if message == 'quit':
                    break
                    
        except Exception as e:
            print(f"Legacy Server client handler error: {e}")
        finally:
            client_socket.close()
            print(f"Legacy Server: Connection closed from {address}")
    
    def stop(self):
        """Stop the legacy server"""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("Legacy Server stopped")

class LegacyServerClient:
    """Client that communicates with the legacy server"""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
    
    def connect(self):
        """Connect to the legacy server"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        print(f"Connected to legacy server at {self.host}:{self.port}")
    
    def send_command(self, command: str) -> str:
        """Send encrypted command and receive encrypted response"""
        # Encrypt the command
        encrypted_command = xor_encrypt(command.encode('utf-8'))
        self.socket.send(encrypted_command)
        
        # Receive and decrypt response
        encrypted_response = self.socket.recv(1024)
        response = xor_decrypt(encrypted_response).decode('utf-8')
        return response
    
    def disconnect(self):
        """Disconnect from the server"""
        if self.socket:
            self.socket.close()
            print("Disconnected from legacy server")

def create_legacy_server_tools():
    """Create tools for Claude to interact with the legacy server"""
    return [
        {
            "name": "connect_to_legacy_server",
            "description": "Connect to the legacy server via custom socket with XOR encryption",
            "input_schema": {

                "properties": {
                    "host": {

                    },

                        "default": 9999
                    }
                },
                "required": []
            }
        },
        {
            "name": "send_command_to_legacy_server",
            "description": "Send a command to the legacy server and receive response. Commands are encrypted with XOR cipher.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "command": {
                        "type": "string",
                        "description": "Command to send to the server (e.g., 'hello', 'status', 'data', 'version', 'help')"
                    }
                },
                "required": ["command"]
            }
        },
        {
            "name": "disconnect_from_legacy_server",
            "description": "Disconnect from the legacy server",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        }
    ]

def process_tool_call(tool_name: str, tool_input: dict, client: LegacyServerClient) -> str:
    """Process tool calls from Claude"""
    if tool_name == "connect_to_legacy_server":
        host = tool_input.get("host", "localhost")
        port = tool_input.get("port", 9999)
        try:
            client.host = host
            client.port = port
            client.connect()
            return f"Successfully connected to legacy server at {host}:{port}"
        except Exception as e:
            return f"Failed to connect: {str(e)}"
    
    elif tool_name == "send_command_to_legacy_server":
        command = tool_input.get("command", "")
        try:
            response = client.send_command(command)
            return f"Server response: {response}"
        except Exception as e:
            return f"Failed to send command: {str(e)}"
    
    elif tool_name == "disconnect_from_legacy_server":
        try:
            client.disconnect()
            return "Successfully disconnected from legacy server"
        except Exception as e:
            return f"Failed to disconnect: {str(e)}"
    
    return f"Unknown tool: {tool_name}"

def run_claude_with_legacy_server():
    """Main function to run Claude with legacy server interaction"""
    
    # Start the legacy server in a background thread
    server = LegacyServer()
    server_thread = threading.Thread(target=server.start)
    server_thread.daemon = True
    server_thread.start()
    
    # Wait for server to start
    time.sleep(1)
    
    # Create client
    client = LegacyServerClient()
    
    # Initialize Anthropic client
    anthropic_client = anthropic.Anthropic()
    
    # Create tools
    tools = create_legacy_server_tools()
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": """You are an AI assistant that needs to interact with a legacy server system. 
            The server uses XOR encryption for security. Please:
            1. Connect to the legacy server
            2. Send the 'hello' command to greet the server
            3. Check the server status
            4. Retrieve the available data
            5. Get the server version information
            6. Disconnect from the server
            
            Please perform all these operations and provide a summary of what you found."""
        }
    ]
    
    print("\n" + "="*60)
    print("Starting Claude interaction with Legacy Server")
    print("="*60 + "\n")
    
    # Agentic loop
    while True:
        response = anthropic_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude's response (stop_reason: {response.stop_reason}):")
        
        # Process the response
        if response.stop_reason == "tool_use":
            # Extract text content if any
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"Claude: {block.text}")
            
            # Process tool calls
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool call: {block.name}")
                    print(f"Input: {block.input}")
                    
                    result = process_tool_call(block.name, block.input, client)
                    print(f"Result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            # Final response
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"\nClaude's Final Summary:\n{block.text}")
            break
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    # Stop the server
    server.stop()
    
    print("\n" + "="*60)
    print("Legacy Server interaction completed")
    print("="*60)

if __name__ == "__main__":
    run_claude_with_legacy_server()
