function demonstrateBankingAutoLogout() {
  /**
   * Banking App Auto-Logout System
   * बैंकिंग ऐप के लिए स्वचालित लॉगआउट सिस्टम
   */

  // Configuration
  const CONFIG = {
    INACTIVITY_TIMEOUT: 15 * 60 * 1000, // 15 minutes in milliseconds
    WARNING_TIME: 2 * 60 * 1000, // 2 minutes warning before logout
    SESSION_KEY: "banking_session",
    AUDIT_LOG_KEY: "security_audit_log",
  };

  // Mock browser environment for Node.js
  const mockStorage = {};
  const localStorage = {
    getItem: (key) => mockStorage[key] || null,
    setItem: (key, value) => {
      mockStorage[key] = value;
    },
    removeItem: (key) => {
      delete mockStorage[key];
    },
  };

  // Audit Log System
  const auditLog = {
    logs: [],

    add(event, details) {
      const entry = {
        timestamp: new Date().toISOString(),
        event,
        details,
        sessionId: this.getCurrentSessionId(),
      };
      this.logs.push(entry);
      console.log(`[AUDIT] ${entry.timestamp} - ${event}: ${details}`);
    },

    getCurrentSessionId() {
      const session = JSON.parse(
        localStorage.getItem(CONFIG.SESSION_KEY) || "{}"
      );
      return session.id || "NO_SESSION";
    },

    getAll() {
      return this.logs;
    },
  };

  // Session Manager
  const sessionManager = {
    /**
     * नया session बनाएं
     */
    createSession(userId) {
      const session = {
        id: `SESSION_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
        userId,
        createdAt: Date.now(),
        lastActivity: Date.now(),
        isActive: true,
      };

      localStorage.setItem(CONFIG.SESSION_KEY, JSON.stringify(session));
      auditLog.add("SESSION_CREATED", `User ${userId} logged in`);
      return session;
    },

    /**
     * Session को update करें
     */
    updateActivity() {
      const session = this.getSession();
      if (session && session.isActive) {
        session.lastActivity = Date.now();
        localStorage.setItem(CONFIG.SESSION_KEY, JSON.stringify(session));
      }
    },

    /**
     * Current session प्राप्त करें
     */
    getSession() {
      return JSON.parse(localStorage.getItem(CONFIG.SESSION_KEY) || "null");
    },

    /**
     * Session को terminate करें
     */
    terminateSession(reason) {
      const session = this.getSession();
      if (session) {
        auditLog.add("SESSION_TERMINATED", `Reason: ${reason}`);
        localStorage.removeItem(CONFIG.SESSION_KEY);
      }
    },

    /**
     * Session valid है या नहीं check करें
     */
    isSessionValid() {
      const session = this.getSession();
      if (!session || !session.isActive) return false;

      const inactiveTime = Date.now() - session.lastActivity;
      return inactiveTime < CONFIG.INACTIVITY_TIMEOUT;
    },
  };

  // Auto-Logout Manager
  const autoLogoutManager = {
    logoutTimer: null,
    warningTimer: null,
    isWarningShown: false,

    /**
     * Timers शुरू करें
     */
    startTimers() {
      this.clearTimers();

      // Warning timer - logout से 2 मिनट पहले
      this.warningTimer = setTimeout(() => {
        this.showWarning();
      }, CONFIG.INACTIVITY_TIMEOUT - CONFIG.WARNING_TIME);

      // Logout timer - 15 मिनट बाद
      this.logoutTimer = setTimeout(() => {
        this.performLogout("INACTIVITY_TIMEOUT");
      }, CONFIG.INACTIVITY_TIMEOUT);

      console.log(
        `[TIMER] Auto-logout timers set: Warning in ${(CONFIG.INACTIVITY_TIMEOUT - CONFIG.WARNING_TIME) / 1000}s, Logout in ${CONFIG.INACTIVITY_TIMEOUT / 1000}s`
      );
    },

    /**
     * Timers clear करें
     */
    clearTimers() {
      if (this.logoutTimer) clearTimeout(this.logoutTimer);
      if (this.warningTimer) clearTimeout(this.warningTimer);
      this.isWarningShown = false;
    },

    /**
     * Activity detect होने पर timers reset करें
     */
    resetTimers() {
      sessionManager.updateActivity();
      this.startTimers();
    },

    /**
     * Warning दिखाएं
     */
    showWarning() {
      this.isWarningShown = true;
      auditLog.add("WARNING_SHOWN", "Inactivity warning displayed to user");
      console.log("\n⚠️  WARNING: आपका session 2 मिनट में समाप्त हो जाएगा!");
      console.log("   Session extend करने के लिए कोई भी activity करें।\n");
    },

    /**
     * Logout perform करें
     */
    async performLogout(reason) {
      console.log(`\n🔒 Auto-Logout: ${reason}`);

      // Server को logout notify करें (simulation)
      await this.notifyServer();

      // Session terminate करें
      sessionManager.terminateSession(reason);

      // Timers clear करें
      this.clearTimers();

      // UI update करें (simulation)
      this.updateUI("LOGGED_OUT");

      auditLog.add("AUTO_LOGOUT", `User logged out due to: ${reason}`);
      console.log("✅ User successfully logged out!\n");
    },

    /**
     * Server को logout notify करें
     */
    async notifyServer() {
      // API call simulation
      console.log("[API] Notifying server about logout...");
      await new Promise((resolve) => setTimeout(resolve, 100));
      console.log("[API] Server notified successfully");
    },

    /**
     * UI update करें
     */
    updateUI(state) {
      console.log(`[UI] Updating interface to state: ${state}`);
      if (state === "LOGGED_OUT") {
        console.log("[UI] Redirecting to login page...");
      }
    },

    /**
     * Session extend करें
     */
    extendSession() {
      if (this.isWarningShown) {
        this.resetTimers();
        auditLog.add("SESSION_EXTENDED", "User extended their session");
        console.log("✅ Session successfully extended for 15 more minutes!\n");
      }
    },
  };

  // Activity Tracker
  const activityTracker = {
    events: ["mousemove", "mousedown", "keypress", "scroll", "touchstart"],
    handlers: {},

    /**
     * Activity tracking शुरू करें
     */
    start() {
      console.log("[TRACKER] Activity tracking started");

      // Simulate activity tracking
      this.events.forEach((event) => {
        this.handlers[event] = () => {
          autoLogoutManager.resetTimers();
        };
      });
    },

    /**
     * Activity tracking बंद करें
     */
    stop() {
      console.log("[TRACKER] Activity tracking stopped");
      this.handlers = {};
    },

    /**
     * Manual activity simulate करें (testing के लिए)
     */
    simulateActivity(eventType) {
      console.log(`[ACTIVITY] User activity detected: ${eventType}`);
      if (this.handlers[eventType]) {
        this.handlers[eventType]();
      }
    },
  };

  // Main Banking App Controller
  const bankingApp = {
    /**
     * User login करें
     */
    login(userId) {
      console.log(`\n🏦 Banking App में Login हो रहे हैं: ${userId}`);

      const session = sessionManager.createSession(userId);
      activityTracker.start();
      autoLogoutManager.startTimers();

      console.log(`✅ Login successful! Session ID: ${session.id}\n`);
      return session;
    },

    /**
     * Manual logout
     */
    logout() {
      activityTracker.stop();
      autoLogoutManager.performLogout("USER_INITIATED");
    },

    /**
     * Session status check करें
     */
    checkSession() {
      const isValid = sessionManager.isSessionValid();
      const session = sessionManager.getSession();

      if (session) {
        const inactiveTime = Math.floor(
          (Date.now() - session.lastActivity) / 1000
        );
        console.log(`[SESSION] Status: ${isValid ? "ACTIVE" : "EXPIRED"}`);
        console.log(`[SESSION] Inactive for: ${inactiveTime} seconds`);
        console.log(
          `[SESSION] Time until logout: ${Math.max(0, Math.floor((CONFIG.INACTIVITY_TIMEOUT - (Date.now() - session.lastActivity)) / 1000))} seconds`
        );
      } else {
        console.log("[SESSION] No active session found");
      }

      return isValid;
    },
  };

  // Demo Execution
  console.log("🏦 Banking App Auto-Logout System Demo");
  console.log("========================================\n");

  // User login
  bankingApp.login("USER_12345");

  // Check session status
  bankingApp.checkSession();

  // Simulate some user activity
  console.log("\n--- User Activity Simulation ---");
  activityTracker.simulateActivity("mousemove");
  activityTracker.simulateActivity("keypress");

  // Check session again
  setTimeout(() => {
    console.log("\n--- Session Status After 2 seconds ---");
    bankingApp.checkSession();
  }, 2000);

  // Show audit log
  setTimeout(() => {
    console.log("\n--- Security Audit Log ---");
    const logs = auditLog.getAll();
    logs.forEach((log) => {
      console.log(
        `  [${log.timestamp}] ${log.event}: ${log.details} (Session: ${log.sessionId})`
      );
    });

    console.log("\n--- Demo Complete ---");
    console.log("✅ Banking Auto-Logout System successfully demonstrated!");
    console.log("\nFeatures Implemented:");
    console.log("  ✓ 15 मिनट की निष्क्रियता के बाद auto-logout");
    console.log("  ✓ 2 मिनट पहले warning notification");
    console.log("  ✓ Session management");
    console.log("  ✓ Activity tracking");
    console.log("  ✓ Security audit log");
    console.log("  ✓ Server notification simulation");
    console.log("  ✓ Session extension capability");
  }, 3000);
}

// Main execution
createBankingAutoLogout().catch(console.error);
