from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager, UserMixin, login_required, current_user
from werkzeug.security import generate_password_hash, check_password_hash
import re
import logging

# Logging setup
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = Flask(__name__)
app.config['SECRET_KEY'] = 'your-secret-key-here'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///users.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)
login_manager = LoginManager(app)


# User Model
class User(UserMixin, db.Model):
    __tablename__ = 'users'
    
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(256), nullable=False)
    
    def set_password(self, password):
        """पासवर्ड को hash करके store करता है"""
        self.password_hash = generate_password_hash(password)
    
    def check_password(self, password):
        """पासवर्ड verify करता है"""
        return check_password_hash(self.password_hash, password)
    
    def __repr__(self):
        return f'<User {self.username}>'


@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))


def validate_password_strength(password):
    """
    पासवर्ड की मजबूती जांचता है
    
    नियम:
    - कम से कम 8 characters
    - कम से कम एक uppercase letter
    - कम से कम एक lowercase letter
    - कम से कम एक digit
    - कम से कम एक special character
    """
    errors = []
    
    if len(password) < 8:
        errors.append("पासवर्ड कम से कम 8 characters का होना चाहिए।")
    
    if not re.search(r'[A-Z]', password):
        errors.append("पासवर्ड में कम से कम एक uppercase letter होनी चाहिए।")
    
    if not re.search(r'[a-z]', password):
        errors.append("पासवर्ड में कम से कम एक lowercase letter होनी चाहिए।")
    
    if not re.search(r'\d', password):
        errors.append("पासवर्ड में कम से कम एक digit होनी चाहिए।")
    
    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        errors.append("पासवर्ड में कम से कम एक special character होना चाहिए।")
    
    return errors


@app.route('/change-password', methods=['POST'])
@login_required
def change_password():
    """
    पासवर्ड परिवर्तन फ़ंक्शन
    
    Input (JSON):
        - current_password: वर्तमान पासवर्ड
        - new_password: नया पासवर्ड
        - confirm_password: नए पासवर्ड की पुष्टि
    
    Returns:
        JSON response with success/error message
    """
    try:
        # Request data प्राप्त करें
        data = request.get_json()
        
        # Input validation - सभी fields की जांच
        if not data:
            return jsonify({
                'success': False,
                'message': 'कोई डेटा प्रदान नहीं किया गया।'
            }), 400
        
        current_password = data.get('current_password', '').strip()
        new_password = data.get('new_password', '').strip()
        confirm_password = data.get('confirm_password', '').strip()
        
        # सभी fields की उपस्थिति जांचें
        if not current_password:
            return jsonify({
                'success': False,
                'message': 'वर्तमान पासवर्ड आवश्यक है।'
            }), 400
        
        if not new_password:
            return jsonify({
                'success': False,
                'message': 'नया पासवर्ड आवश्यक है।'
            }), 400
        
        if not confirm_password:
            return jsonify({
                'success': False,
                'message': 'पासवर्ड की पुष्टि आवश्यक है।'
            }), 400
        
        # वर्तमान user डेटाबेस से प्राप्त करें
        user = User.query.get(current_user.id)
        
        if not user:
            logger.warning(f"User ID {current_user.id} नहीं मिला।")
            return jsonify({
                'success': False,
                'message': 'उपयोगकर्ता नहीं मिला।'
            }), 404
        
        # वर्तमान पासवर्ड verify करें
        if not user.check_password(current_password):
            logger.warning(f"User ID {current_user.id} ने गलत वर्तमान पासवर्ड दिया।")
            return jsonify({
                'success': False,
                'message': 'वर्तमान पासवर्ड गलत है।'
            }), 401
        
        # नया पासवर्ड और confirm पासवर्ड match करें
        if new_password != confirm_password:
            return jsonify({
                'success': False,
                'message': 'नया पासवर्ड और पुष्टि पासवर्ड मेल नहीं खाते।'
            }), 400
        
        # नया पासवर्ड वर्तमान से अलग होना चाहिए
        if user.check_password(new_password):
            return jsonify({
                'success': False,
                'message': 'नया पासवर्ड वर्तमान पासवर्ड से अलग होना चाहिए।'
            }), 400
        
        # पासवर्ड की मजबूती जांचें
        password_errors = validate_password_strength(new_password)
        if password_errors:
            return jsonify({
                'success': False,
                'message': 'पासवर्ड की आवश्यकताएं पूरी नहीं हुईं।',
                'errors': password_errors
            }), 400
        
        # डेटाबेस में नया पासवर्ड अपडेट करें
        user.set_password(new_password)
        db.session.commit()
        
        logger.info(f"User ID {current_user.id} ने सफलतापूर्वक पासवर्ड बदला।")
        
        return jsonify({
            'success': True,
            'message': 'पासवर्ड सफलतापूर्वक बदल दिया गया है।'
        }), 200
    
    except Exception as e:
        # Database error होने पर rollback करें
        db.session.rollback()
        logger.error(f"पासवर्ड बदलने में त्रुटि: {str(e)}")
        
        return jsonify({
            'success': False,
            'message': 'पासवर्ड बदलने में त्रुटि हुई। कृपया पुनः प्रयास करें।'
        }), 500


@app.route('/admin/reset-password/<int:user_id>', methods=['POST'])
@login_required
def admin_reset_password(user_id):
    """
    Admin द्वारा किसी भी user का पासवर्ड reset करने का फ़ंक्शन
    
    Input (JSON):
        - new_password: नया पासवर्ड
    
    Returns:
        JSON response with success/error message
    """
    try:
        data = request.get_json()
        
        if not data:
            return jsonify({
                'success': False,
                'message': 'कोई डेटा प्रदान नहीं किया गया।'
            }), 400
        
        new_password = data.get('new_password', '').strip()
        
        if not new_password:
            return jsonify({
                'success': False,
                'message': 'नया पासवर्ड आवश्यक है।'
            }), 400
        
        # Target user खोजें
        target_user = User.query.get(user_id)
        
        if not target_user:
            return jsonify({
                'success': False,
                'message': f'User ID {user_id} नहीं मिला।'
            }), 404
        
        # पासवर्ड की मजबूती जांचें
        password_errors = validate_password_strength(new_password)
        if password_errors:
            return jsonify({
                'success': False,
                'message': 'पासवर्ड की आवश्यकताएं पूरी नहीं हुईं।',
                'errors': password_errors
            }), 400
        
        # पासवर्ड अपडेट करें
        target_user.set_password(new_password)
        db.session.commit()
        
        logger.info(f"Admin ने User ID {user_id} का पासवर्ड reset किया।")
        
        return jsonify({
            'success': True,
            'message': f'User ID {user_id} का पासवर्ड सफलतापूर्वक reset किया गया।'
        }), 200
    
    except Exception as e:
        db.session.rollback()
        logger.error(f"Admin पासवर्ड reset में त्रुटि: {str(e)}")
        
        return jsonify({
            'success': False,
            'message': 'पासवर्ड reset करने में त्रुटि हुई।'
        }), 500
