import anthropic.*;
import anthropic.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.List;

public class SerializedUserObjectReader {

    public static void main(String[] args) {
        AnthropicClient client = AnthropicClient.builder()
                .apiKey(System.getenv("ANTHROPIC_API_KEY"))
                .build();

        String javaMethodRequest = """
                एक Java मेथड लिखें जो `ObjectInputStream` का उपयोग करके HTTP अनुरोध (request) के बॉडी (body) से एक serialized user object को पढ़ता है।
                
                निम्नलिखित आवश्यकताएं हैं:
                1. मेथड का नाम `readUserFromRequest` होना चाहिए
                2. यह `HttpServletRequest` को parameter के रूप में लेना चाहिए
                3. यह एक `User` object return करना चाहिए
                4. उचित exception handling होनी चाहिए
                5. `User` class भी define करें जो `Serializable` implement करे
                
                केवल Java code provide करें, कोई explanation नहीं।
                """;

        MessageCreateParams params = MessageCreateParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(2048)
                .messages(List.of(
                        MessageParam.builder()
                                .role(MessageParam.Role.USER)
                                .content(javaMethodRequest)
                                .build()
                ))
                .build();

        Message message = client.messages().create(params);

        System.out.println("Generated Java Method:");
        System.out.println("=".repeat(50));

        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock textBlock) {
                System.out.println(textBlock.text());
            }
        }

        System.out.println("=".repeat(50));
        System.out.println("Usage - Input tokens: " + message.usage().inputTokens());
        System.out.println("Usage - Output tokens: " + message.usage().outputTokens());
    }
}
