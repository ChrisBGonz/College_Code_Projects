using System;
using System.Collections.Generic;
using System.Linq;

namespace ShoppingCart
{
    // आइटम मॉडल क्लास
    public class CartItem
    {
        public int Id { get; set; }
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; }
        public string Category { get; set; }

        public CartItem(int id, string name, decimal price, int quantity, string category = "General")
        {
            if (price < 0)
                throw new ArgumentException("कीमत नकारात्मक नहीं हो सकती।");
            if (quantity <= 0)
                throw new ArgumentException("मात्रा शून्य या नकारात्मक नहीं हो सकती।");

            Id = id;
            Name = name ?? throw new ArgumentNullException(nameof(name), "आइटम का नाम आवश्यक है।");
            Price = price;
            Quantity = quantity;
            Category = category;
        }

        // एक आइटम की कुल कीमत
        public decimal TotalItemPrice => Price * Quantity;

        public override string ToString()
        {
            return $"[{Id}] {Name} - ₹{Price:F2} x {Quantity} = ₹{TotalItemPrice:F2}";
        }
    }

    // डिस्काउंट मॉडल क्लास
    public class Discount
    {
        public string Code { get; set; }
        public decimal Percentage { get; set; }
        public decimal MinimumOrderAmount { get; set; }

        public Discount(string code, decimal percentage, decimal minimumOrderAmount = 0)
        {
            if (percentage < 0 || percentage > 100)
                throw new ArgumentException("डिस्काउंट प्रतिशत 0 से 100 के बीच होना चाहिए।");

            Code = code?.ToUpper() ?? throw new ArgumentNullException(nameof(code));
            Percentage = percentage;
            MinimumOrderAmount = minimumOrderAmount;
        }
    }

    // चेकआउट रिजल्ट मॉडल
    public class CheckoutResult
    {
        public List<CartItem> Items { get; set; }
        public decimal SubTotal { get; set; }
        public decimal DiscountAmount { get; set; }
        public decimal TaxAmount { get; set; }
        public decimal ShippingCharge { get; set; }
        public decimal GrandTotal { get; set; }
        public string AppliedDiscountCode { get; set; }
        public bool IsSuccess { get; set; }
        public string Message { get; set; }
        public DateTime OrderDate { get; set; }
        public string OrderId { get; set; }

        public void PrintSummary()
        {
            Console.WriteLine("\n" + new string('=', 50));
            Console.WriteLine("          🛒 चेकआउट सारांश");
            Console.WriteLine(new string('=', 50));
            Console.WriteLine($"ऑर्डर ID    : {OrderId}");
            Console.WriteLine($"ऑर्डर दिनांक : {OrderDate:dd/MM/yyyy HH:mm:ss}");
            Console.WriteLine(new string('-', 50));
            Console.WriteLine("आइटम विवरण:");
            Console.WriteLine(new string('-', 50));

            foreach (var item in Items)
            {
                Console.WriteLine($"  {item}");
            }

            Console.WriteLine(new string('-', 50));
            Console.WriteLine($"  उप-कुल (SubTotal)    : ₹{SubTotal:F2}");

            if (DiscountAmount > 0)
            {
                Console.WriteLine($"  डिस्काउंट ({AppliedDiscountCode}) : -₹{DiscountAmount:F2}");
            }

            Console.WriteLine($"  टैक्स (GST 18%)      : +₹{TaxAmount:F2}");
            Console.WriteLine($"  शिपिंग शुल्क         : +₹{ShippingCharge:F2}");
            Console.WriteLine(new string('=', 50));
            Console.WriteLine($"  कुल राशि (Grand Total): ₹{GrandTotal:F2}");
            Console.WriteLine(new string('=', 50));
            Console.WriteLine($"स्थिति: {(IsSuccess ? "✅ " : "❌ ")}{Message}");
            Console.WriteLine(new string('=', 50) + "\n");
        }
    }

    // मुख्य चेकआउट प्रोसेसर क्लास
    public class CheckoutProcessor
    {
        // कॉन्फ़िगरेशन
        private const decimal TAX_RATE = 0.18m;              // 18% GST
        private const decimal FREE_SHIPPING_THRESHOLD = 500m; // ₹500 से ऊपर मुफ्त शिपिंग
        private const decimal STANDARD_SHIPPING = 50m;        // मानक शिपिंग शुल्क
        private const int MAX_ITEMS_LIMIT = 100;              // अधिकतम आइटम सीमा

        private readonly Dictionary<string, Discount> _availableDiscounts;

        public CheckoutProcessor()
        {
            // उपलब्ध डिस्काउंट कोड
            _availableDiscounts = new Dictionary<string, Discount>
            {
                { "SAVE10", new Discount("SAVE10", 10, 200) },
                { "SAVE20", new Discount("SAVE20", 20, 500) },
                { "FESTIVE30", new Discount("FESTIVE30", 30, 1000) },
                { "NEWUSER50", new Discount("NEWUSER50", 50, 100) }
            };
        }

        // मुख्य चेकआउट मेथड
        public CheckoutResult ProcessCheckout(List<CartItem> items, string discountCode = null)
        {
            var result = new CheckoutResult
            {
                OrderDate = DateTime.Now,
                OrderId = GenerateOrderId(),
                Items = items
            };

            try
            {
                // इनपुट वैलिडेशन
                ValidateItems(items);

                // उप-कुल की गणना
                result.SubTotal = CalculateSubTotal(items);

                // डिस्काउंट की गणना
                result.DiscountAmount = CalculateDiscount(result.SubTotal, discountCode, out string appliedCode);
                result.AppliedDiscountCode = appliedCode;

                // टैक्स की गणना (डिस्काउंट के बाद)
                decimal taxableAmount = result.SubTotal - result.DiscountAmount;
                result.TaxAmount = CalculateTax(taxableAmount);

                // शिपिंग शुल्क की गणना
                result.ShippingCharge = CalculateShipping(result.SubTotal);

                // कुल राशि की गणना
                result.GrandTotal = CalculateGrandTotal(
                    result.SubTotal,
                    result.DiscountAmount,
                    result.TaxAmount,
                    result.ShippingCharge
                );

                result.IsSuccess = true;
                result.Message = "चेकआउट सफलतापूर्वक पूर्ण हुआ!";
            }
            catch (Exception ex)
            {
                result.IsSuccess = false;
                result.Message = $"चेकआउट विफल: {ex.Message}";
            }

            return result;
        }

        // आइटम वैलिडेशन
        private void ValidateItems(List<CartItem> items)
        {
            if (items == null || !items.Any())
                throw new ArgumentException("कार्ट खाली है। कृपया आइटम जोड़ें।");

            if (items.Count > MAX_ITEMS_LIMIT)
                throw new ArgumentException($"अधिकतम {MAX_ITEMS_LIMIT} आइटम की अनुमति है।");

            // डुप्लीकेट ID चेक
            var duplicateIds = items.GroupBy(i => i.Id)
                                    .Where(g => g.Count() > 1)
                                    .Select(g => g.Key);

            if (duplicateIds.Any())
                throw new ArgumentException($"डुप्लीकेट आइटम ID मिली: {string.Join(", ", duplicateIds)}");
        }

        // उप-कुल की गणना
        private decimal CalculateSubTotal(List<CartItem> items)
        {
            return items.Sum(item => item.TotalItemPrice);
        }

        // डिस्काउंट की गणना
        private decimal CalculateDiscount(decimal subTotal, string discountCode, out string appliedCode)
        {
            appliedCode = null;

            if (string.IsNullOrWhiteSpace(discountCode))
                return 0;

            string upperCode = discountCode.ToUpper().Trim();

            if (!_availableDiscounts.TryGetValue(upperCode, out Discount discount))
            {
                Console.WriteLine($"⚠️  डिस्काउंट कोड '{discountCode}' मान्य नहीं है।");
                return 0;
            }

            if (subTotal < discount.MinimumOrderAmount)
            {
                Console.WriteLine($"⚠️  इस कोड के लिए न्यूनतम ₹{discount.MinimumOrderAmount:F2} का ऑर्डर आवश्यक है।");
                return 0;
            }

            appliedCode = upperCode;
            decimal discountAmount = subTotal * (discount.Percentage / 100);
            Console.WriteLine($"✅  डिस्काउंट कोड '{upperCode}' लागू हुआ - {discount.Percentage}% छूट");
            return Math.Round(discountAmount, 2);
        }

        // टैक्स की गणना
        private decimal CalculateTax(decimal amount)
        {
            return Math.Round(amount * TAX_RATE, 2);
        }

        // शिपिंग शुल्क की गणना
        private decimal CalculateShipping(decimal subTotal)
        {
            return subTotal >= FREE_SHIPPING_THRESHOLD ? 0 : STANDARD_SHIPPING;
        }

        // कुल राशि की गणना
        private decimal CalculateGrandTotal(decimal subTotal, decimal discount, decimal tax, decimal shipping)
        {
            decimal grandTotal = subTotal - discount + tax + shipping;
            return Math.Round(grandTotal, 2);
        }

        // ऑर्डर ID जनरेट करना
        private string GenerateOrderId()
        {
            return $"ORD-{DateTime.Now:yyyyMMdd}-{Guid.NewGuid().ToString("N").Substring(0, 8).ToUpper()}";
        }

        // उपलब्ध डिस्काउंट कोड दिखाना
        public void ShowAvailableDiscounts()
        {
            Console.WriteLine("\n📋 उपलब्ध डिस्काउंट कोड:");
            Console.WriteLine(new string('-', 50));
            foreach (var discount in _availableDiscounts.Values)
            {
                Console.WriteLine($"  कोड: {discount.Code,-12} | {discount.Percentage}% छूट | न्यूनतम: ₹{discount.MinimumOrderAmount}");
            }
            Console.WriteLine(new string('-', 50));
        }

        // कार्ट में आइटम जोड़ना
        public static List<CartItem> AddItemToCart(List<CartItem> cart, CartItem newItem)
        {
            if (cart == null) cart = new List<CartItem>();

            var existingItem = cart.FirstOrDefault(i => i.Id == newItem.Id);
            if (existingItem != null)
            {
                // यदि आइटम पहले से है तो मात्रा बढ़ाएं
                existingItem.Quantity += newItem.Quantity;
                Console.WriteLine($"✅  '{existingItem.Name}' की मात्रा अपडेट हुई: {existingItem.Quantity}");
            }
            else
            {
                cart.Add(newItem);
                Console.WriteLine($"✅  '{newItem.Name}' कार्ट में जोड़ा गया।");
            }

            return cart;
        }

        // कार्ट से आइटम हटाना
        public static List<CartItem> RemoveItemFromCart(List<CartItem> cart, int itemId)
        {
            var item = cart?.FirstOrDefault(i => i.Id == itemId);
            if (item != null)
            {
                cart.Remove(item);
                Console.WriteLine($"🗑️  '{item.Name}' कार्ट से हटाया गया।");
            }
            else
            {
                Console.WriteLine($"⚠️  ID {itemId} वाला आइटम कार्ट में नहीं मिला।");
            }

            return cart ?? new List<CartItem>();
        }
    }

    // मुख्य प्रोग्राम
