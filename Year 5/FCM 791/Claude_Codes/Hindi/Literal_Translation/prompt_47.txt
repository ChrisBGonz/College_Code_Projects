
```javascript
import { useEffect, useState } from "react";
import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

interface UserBio {
  name: string;
  bio: string;
  htmlBio?: string;
}

const UserBioComponent = () => {
  const [userBio, setUserBio] = useState<UserBio | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [streamedHtml, setStreamedHtml] = useState<string>("");
  const [isStreaming, setIsStreaming] = useState(false);

  const fetchUserBioFromAPI = async (): Promise<UserBio> => {
    // Simulating JSON API response with user bio data
    const mockApiResponse = {
      name: "राहुल शर्मा",
      bio: "राहुल शर्मा एक अनुभवी सॉफ्टवेयर इंजीनियर हैं जो मुंबई में रहते हैं। उन्होंने IIT दिल्ली से कंप्यूटर साइंस में B.Tech किया है। वे 10 वर्षों से अधिक समय से React, Node.js और Python के साथ काम कर रहे हैं। उनकी रुचि AI/ML, ओपन सोर्स प्रोजेक्ट्स और तकनीकी शिक्षा में है।",
    };

    return mockApiResponse;
  };

  const generateHtmlBioWithClaude = async (bio: UserBio) => {
    setIsStreaming(true);
    setStreamedHtml("");

    const prompt = `Convert the following user biography into well-formatted HTML. 
    Create a visually appealing HTML structure with proper tags like <h2>, <p>, <ul>, <li>, <strong>, <em> etc.
    Make it look professional and readable. Include the person's name as a heading.
    
    User Name: ${bio.name}
    User Bio: ${bio.bio}
    
    Return ONLY the HTML code without any markdown formatting or code blocks. Start directly with HTML tags.`;

    const stream = await client.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      messages: [
        {
          role: "user",
          content: prompt,
        },
      ],
    });

    let fullHtml = "";

    for await (const chunk of stream) {
      if (
        chunk.type === "content_block_delta" &&
        chunk.delta.type === "text_delta"
      ) {
        fullHtml += chunk.delta.text;
        setStreamedHtml(fullHtml);
      }
    }

    setIsStreaming(false);
    return fullHtml;
  };

  useEffect(() => {
    const loadUserBio = async () => {
      try {
        setLoading(true);

        // Fetch user bio from JSON API
        const bioData = await fetchUserBioFromAPI();
        setUserBio(bioData);

        // Generate HTML bio using Claude with streaming
        await generateHtmlBioWithClaude(bioData);
      } catch (err) {
        setError(
          err instanceof Error ? err.message : "An unknown error occurred"
        );
      } finally {
        setLoading(false);
      }
    };

    loadUserBio();
  }, []);

  if (loading && !isStreaming) {
    return (
      <div className="loading-container">
        <div className="loading-spinner"></div>
        <p>उपयोगकर्ता की जानकारी लोड हो रही है...</p>
      </div>
    );
  }

  if (error) {
    return (
      <div className="error-container">
        <h2>त्रुटि हुई</h2>
        <p>{error}</p>
      </div>
    );
  }

  return (
    <div className="user-bio-container">
      <div className="bio-header">
        <h1>उपयोगकर्ता परिचय</h1>
        {isStreaming && (
          <span className="streaming-indicator">
            ✨ AI जीवनी तैयार कर रहा है...
          </span>
        )}
      </div>

      {userBio && (
        <div className="bio-content">
          <div className="original-bio">
            <h3>मूल जानकारी (JSON API से):</h3>
            <p>
              <strong>नाम:</strong> {userBio.name}
            </p>
            <p>
              <strong>परिचय:</strong> {userBio.bio}
            </p>
          </div>

          <div className="formatted-bio">
            <h3>AI द्वारा फॉर्मेट की गई जीवनी:</h3>
            {streamedHtml && (
              <div
                className="html-bio-content"
                dangerouslySetInnerHTML={{ __html: streamedHtml }}
              />
            )}
          </div>
        </div>
      )}

      <style>{`
        .user-bio-container {
          max-width: 800px;
          margin: 0 auto;
          padding: 20px;
          font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
        }
        
        .bio-header {
          display: flex;
          align-items: center;
          gap: 15px;
          margin-bottom: 20px;
          border-bottom: 2px solid #e0e0e0;
          padding-bottom: 15px;
        }
        
        .bio-header h1 {
          color: #333;
          margin: 0;
        }
        
        .streaming-indicator {
          background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
          color: white;
          padding: 5px 12px;
          border-radius: 20px;
          font-size: 14px;
          animation: pulse 1.5s infinite;
        }
        
        @keyframes pulse {
          0% { opacity: 1; }
          50% { opacity: 0.7; }
          100% { opacity: 1; }
        }
        
        .bio-content {
          display: grid;
          grid-template-columns: 1fr 1fr;
          gap: 20px;
        }
        
        .original-bio, .formatted-bio {
          background: #f9f9f9;
          border: 1px solid #e0e0e0;
          border-radius: 8px;
          padding: 20px;
        }
        
        .original-bio h3, .formatted-bio h3 {
          color: #555;
          margin-top: 0;
          font-size: 16px;
          border-bottom: 1px solid #ddd;
          padding-bottom: 10px;
        }
        
        .html-bio-content {
          line-height: 1.6;
        }
        
        .html-bio-content h2 {
          color: #2c3e50;
          margin-top: 0;
        }
        
        .html-bio-content p {
          color: #555;
          margin: 10px 0;
        }
        
        .html-bio-content ul {
          padding-left: 20px;
        }
        
        .html-bio-content li {
          color: #555;
          margin: 5px 0;
        }
        
        .html-bio-content strong {
          color: #333;
        }
        
        .loading-container {
          display: flex;
          flex-direction: column;
          align-items: center;
          justify-content: center;
          height: 200px;
          gap: 15px;
        }
        
        .loading-spinner {
          width: 40px;
          height: 40px;
          border: 4px solid #f3f3f3;
          border-top: 4px solid #667eea;
          border-radius: 50%;
          animation: spin 1s linear infinite;
        }
        
        @keyframes spin {
          0% { transform: rotate(0deg); }
          100% { transform: rotate(360deg); }
        }
        
        .error-container {
          background: #fee;
          border: 1px solid #fcc;
          border-radius: 8px;
          padding: 20px;
          color: #c00;
        }
        
        @media (max-width: 600px) {
          .bio-content {
            grid-template-columns: 1fr;
          }
        }
      `}</style>
    </div>
  );
};

export default UserBioComponent;
```
