def get_status_class(status_code):
    """HTTP स्टेटस कोड के आधार पर CSS class निर्धारित करें"""
    if 200 <= status_code < 300:
        return "2xx"
    elif 300 <= status_code < 400:
        return "3xx"
    elif 400 <= status_code < 500:
        return "4xx"
    else:
        return "5xx"


def format_content_size(size_bytes):
    """Bytes को readable format में बदलें"""
    if size_bytes < 1024:
        return f"{size_bytes} bytes"
    elif size_bytes < 1024 * 1024:
        return f"{size_bytes / 1024:.1f} KB"
    else:
        return f"{size_bytes / (1024 * 1024):.1f} MB"


def validate_url(url):
    """URL को validate करें"""
    if not url:
        return False, "URL खाली नहीं हो सकता।"
    
    if not url.startswith(('http://', 'https://')):
        return False, "URL को 'http://' या 'https://' से शुरू होना चाहिए।"
    
    if len(url) > 2048:
        return False, "URL बहुत लंबा है (अधिकतम 2048 characters)।"
    
    return True, None

@app.route('/fetch', methods=['POST'])
def fetch_url():
    """URL से सामग्री प्राप्त करें"""
    
    url = request.form.get('url', '').strip()
    
    # URL Validation
    is_valid, validation_error = validate_url(url)
    if not is_valid:
        return render_template_string(
            HTML_TEMPLATE,
            error=validation_error,
            url=url
        )
    
    try:
        # HTTP Request भेजें
        headers = {
            'User-Agent': 'Mozilla/5.0 (Flask URL Fetcher/1.0)',
            'Accept': 'text/html,application/json,*/*'
        }
        
        response = requests.get(
            url,
            headers=headers,
            timeout=10,          # 10 seconds timeout
            allow_redirects=True, # Redirects follow करें
            verify=True           # SSL verify करें
        )
        
        status_code = response.status_code
        content_type = response.headers.get('Content-Type', 'Unknown')
        content_size = format_content_size(len(response.content))
        status_class = get_status_class(status_code)
        
        # ===== 404 Error Handling =====
        if status_code == 404:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"404 - पृष्ठ नहीं मिला! URL '{url}' मौजूद नहीं है। "
                      f"कृपया URL जांचें और पुनः प्रयास करें।",
                url=url,
                status_code=status_code,
                status_class=status_class,
                content_size=content_size,
                content_type=content_type.split(';')[0]
            )
        
        # ===== अन्य 4xx Errors =====
        elif status_code == 400:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"400 - Bad Request! सर्वर ने अनुरोध को अस्वीकार कर दिया।",
                url=url,
                status_code=status_code,
                status_class=status_class
            )
        
        elif status_code == 401:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"401 - Unauthorized! इस URL तक पहुंचने के लिए प्रमाणीकरण आवश्यक है।",
                url=url,
                status_code=status_code,
                status_class=status_class
            )
        
        elif status_code == 403:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"403 - Forbidden! आपको इस URL तक पहुंचने की अनुमति नहीं है।",
                url=url,
                status_code=status_code,
                status_class=status_class
            )
        
        elif status_code == 429:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"429 - Too Many Requests! सर्वर ने rate limit लागू की है। "
                      f"कुछ समय बाद पुनः प्रयास करें।",
                url=url,
                status_code=status_code,
                status_class=status_class
            )
        
        # ===== 5xx Server Errors =====
        elif status_code >= 500:
            return render_template_string(
                HTML_TEMPLATE,
                error=f"{status_code} - Server Error! सर्वर में कोई समस्या है। "
                      f"बाद में पुनः प्रयास करें।",
                url=url,
                status_code=status_code,
                status_class=status_class
            )
        
        # ===== Successful Response (2xx, 3xx) =====
        else:
            # Content को decode करें
            try:
                content = response.text
            except UnicodeDecodeError:
                content = response.content.decode('utf-8', errors='replace')
            
            # Content को limit करें (बहुत बड़ी files के लिए)
            max_display_chars = 50000
            warning = None
            
            if len(content) > max_display_chars:
                content = content[:max_display_chars]
                warning = f"सामग्री बहुत बड़ी है। केवल पहले {max_display_chars:,} characters दिखाए जा रहे हैं।"
            
            success_message = f"सामग्री सफलतापूर्वक प्राप्त की गई! (Status: {status_code})"
            
            return render_template_string(
                HTML_TEMPLATE,
                url=url,
                content=content,
                status_code=status_code,
                status_class=status_class,
                content_size=content_size,
                content_type=content_type.split(';')[0],
                success_message=success_message,
                warning=warning
            )
    
    # ===== Exception Handling =====
    
    except ConnectionError:
        return render_template_string(
            HTML_TEMPLATE,
            error=f"Connection Error! '{url}' से कनेक्ट नहीं हो सका। "
                  f"कृपया URL और अपना इंटरनेट कनेक्शन जांचें।",
            url=url
        )
    
    except Timeout:
        return render_template_string(
            HTML_TEMPLATE,
            error=f"Timeout Error! सर्वर ने 10 seconds में जवाब नहीं दिया। "
                  f"सर्वर धीमा हो सकता है या URL गलत हो सकता है।",
            url=url
        )
    
    except InvalidURL:
        return render_template_string(
            HTML_TEMPLATE,
            error=f"Invalid URL! '{url}' एक वैध URL नहीं है। "
                  f"कृपया सही format में URL दर्ज करें।",
            url=url
        )
    
    except RequestException as e:
        return render_template_string(
            HTML_TEMPLATE,
            error=f"Request Error! सामग्री प्राप्त करने में समस्या: {str(e)}",
            url=url
        )
    
    except Exception as e:
        return render_template_string(
            HTML_TEMPLATE,
            error=f"अप्रत्याशित त्रुटि: {str(e)}",
            url=url
        )

@app.route('/api/fetch', methods=['POST'])
def api_fetch_url():
    """API endpoint - JSON response देता है"""
    
    data = request.get_json()
    
    if not data or 'url' not in data:
        return {
            'success': False,
            'error': 'URL प्रदान नहीं किया गया',
            'status_code': 400
        }, 400
    
    url = data.get('url', '').strip()
    
    # URL Validation
    is_valid, validation_error = validate_url(url)
    if not is_valid:
        return {
            'success': False,
            'error': validation_error,
            'status_code': 400
        }, 400
    
    try:
        response = requests.get(url, timeout=10)
        
        if response.status_code == 404:
            return {
                'success': False,
                'error': f'404 - पृष्ठ नहीं मिला: {url}',
                'status_code': 404,
                'url': url
            }, 404
        
        return {
            'success': True,
            'url': url,
            'status_code': response.status_code,
            'content_type': response.headers.get('Content-Type', ''),
            'content_length': len(response.content),
            'content': response.text[:10000]  # पहले 10000 chars
        }
    
    except ConnectionError:
        return {
            'success': False,
            'error': 'Connection Error - URL से कनेक्ट नहीं हो सका',
            'status_code': 503
        }, 503
    
    except Timeout:
        return {
            'success': False,
            'error': 'Timeout - सर्वर ने समय पर जवाब नहीं दिया',
            'status_code': 408
        }, 408
    
    except RequestException as e:
        return {
            'success': False,
            'error': str(e),
            'status_code': 500
        }, 500
