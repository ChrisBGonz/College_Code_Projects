import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are an expert Node.js and MongoDB developer. Help users set up MongoDB connection strings for local development databases. 
    Provide clear, practical code examples and explanations in Hindi when the user asks in Hindi.
    Always include:
    1. Basic connection string format
    2. Environment variable setup
    3. Error handling
    4. Best practices for local development`,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function generateMongoDBSetup(): Promise<void> {
  console.log("MongoDB Connection Setup Assistant शुरू हो रहा है...\n");

  // Initial request to set up MongoDB connection
  const initialRequest =
    "मुझे Node.js स्क्रिप्ट में local development के लिए MongoDB connection string setup करने में मदद करें। पूरा code example दें।";

  console.log("आप:", initialRequest);
  console.log("\nAssistant सोच रहा है...\n");

  const initialResponse = await chat(initialRequest);
  console.log("Assistant:", initialResponse);
  console.log("\n" + "=".repeat(80) + "\n");

  // Follow-up about environment variables
  const envRequest =
    "क्या आप .env file setup और dotenv package के साथ environment variables का उपयोग करने का तरीका भी बता सकते हैं?";

  console.log("आप:", envRequest);
  console.log("\nAssistant सोच रहा है...\n");

  const envResponse = await chat(envRequest);
  console.log("Assistant:", envResponse);
  console.log("\n" + "=".repeat(80) + "\n");

  // Ask about error handling
  const errorRequest =
    "Connection errors को handle करने का best practice क्या है? Reconnection logic भी बताएं।";

  console.log("आप:", errorRequest);
  console.log("\nAssistant सोच रहा है...\n");

  const errorResponse = await chat(errorRequest);
  console.log("Assistant:", errorResponse);
  console.log("\n" + "=".repeat(80) + "\n");

  // Generate actual files
  console.log("अब actual files generate हो रही हैं...\n");

  await generateActualFiles();
}

async function generateActualFiles(): Promise<void> {
  // Generate .env.example file
  const envExample = `# MongoDB Configuration
MONGODB_URI=mongodb://localhost:27017/myapp_dev
MONGODB_DB_NAME=myapp_dev
MONGODB_PORT=27017
MONGODB_HOST=localhost

# Optional: Authentication (if needed)
# MONGODB_USERNAME=admin
# MONGODB_PASSWORD=password123
# MONGODB_AUTH_SOURCE=admin
`;

  fs.writeFileSync(".env.example", envExample);
  console.log("✅ .env.example file बनाई गई");

  // Generate actual .env file for development
  const envContent = `# MongoDB Local Development Configuration
MONGODB_URI=mongodb://localhost:27017/myapp_dev
MONGODB_DB_NAME=myapp_dev
MONGODB_PORT=27017
MONGODB_HOST=localhost
`;

  if (!fs.existsSync(".env")) {
    fs.writeFileSync(".env", envContent);
    console.log("✅ .env file बनाई गई");
  } else {
    console.log("ℹ️  .env file पहले से मौजूद है, skip किया गया");
  }

  // Generate MongoDB connection module
  const mongodbModule = `import mongoose from 'mongoose';
import dotenv from 'dotenv';

// Load environment variables
dotenv.config();

// MongoDB Connection String
const MONGODB_URI = process.env.MONGODB_URI || 'mongodb://localhost:27017/myapp_dev';

// Connection options
const connectionOptions = {
  useNewUrlParser: true,
  useUnifiedTopology: true,
  serverSelectionTimeoutMS: 5000, // 5 seconds timeout
  socketTimeoutMS: 45000, // 45 seconds socket timeout
  maxPoolSize: 10, // Maximum connection pool size
  minPoolSize: 2, // Minimum connection pool size
};

// Connection state tracking
let isConnected = false;

/**
 * MongoDB से connect करें
 */
export async function connectToDatabase(): Promise<void> {
  if (isConnected) {
    console.log('✅ MongoDB पहले से connected है');
    return;
  }

  try {
    console.log('🔄 MongoDB से connect हो रहा है...');
    console.log(\`📍 URI: \${MONGODB_URI}\`);
    
    await mongoose.connect(MONGODB_URI, connectionOptions);
    
    isConnected = true;
    console.log('✅ MongoDB से successfully connect हो गया!');
    console.log(\`📊 Database: \${mongoose.connection.db?.databaseName}\`);
    
  } catch (error) {
    console.error('❌ MongoDB connection error:', error);
    throw error;
  }
}

/**
 * MongoDB से disconnect करें
 */
export async function disconnectFromDatabase(): Promise<void> {
  if (!isConnected) {
    return;
  }
  
  try {
    await mongoose.disconnect();
    isConnected = false;
    console.log('✅ MongoDB से disconnect हो गया');
  } catch (error) {
    console.error('❌ Disconnect error:', error);
    throw error;
  }
}

// Connection event listeners
mongoose.connection.on('connected', () => {
  console.log('🟢 Mongoose connected to MongoDB');
});

mongoose.connection.on('error', (err) => {
  console.error('🔴 Mongoose connection error:', err);
  isConnected = false;
});

mongoose.connection.on('disconnected', () => {
  console.log('🟡 Mongoose disconnected from MongoDB');
  isConnected = false;
});

// Graceful shutdown
process.on('SIGINT', async () => {
  await disconnectFromDatabase();
  process.exit(0);
});

process.on('SIGTERM', async () => {
  await disconnectFromDatabase();
  process.exit(0);
});

export default mongoose;
`;

  fs.writeFileSync("mongodb.ts", mongodbModule);
  console.log("✅ mongodb.ts file बनाई गई");

  // Generate example usage file
  const exampleUsage = `import { connectToDatabase, disconnectFromDatabase } from './mongodb';
import mongoose from 'mongoose';

// Example Schema
const UserSchema = new mongoose.Schema({
  name: { type: String, required: true },
  email: { type: String, required: true, unique: true },
  createdAt: { type: Date, default: Date.now }
});

const User = mongoose.model('User', UserSchema);

async function main() {
  try {
    // Database से connect करें
    await connectToDatabase();
    
    // Example: User create करें
    const newUser = new User({
      name: 'राम कुमार',
      email: 'ram@example.com'
    });
    
    await newUser.save();
    console.log('✅ User successfully save हुआ:', newUser);
    
    // Example: Users find करें
    const users = await User.find({});
    console.log(\`📋 Total users: \${users.length}\`);
    
  } catch (error) {
    console.error('❌ Error:', error);
  } finally {
    // Database से disconnect करें
    await disconnectFromDatabase();
  }
}

main();
`;

  fs.writeFileSync("example-usage.ts", exampleUsage);
  console.log("✅ example-usage.ts file बनाई गई");

  // Generate package.json if not exists
  if (!fs.existsSync("package.json")) {
    const packageJson = {
      name: "mongodb-local-setup",
      version: "1.0.0",
      description: "Local MongoDB Development Setup",
      main: "index.js",
      scripts: {
        start: "ts-node index.ts",
        dev: "nodemon --exec ts-node index.ts",
        example: "ts-node example-usage.ts",
      },
      dependencies: {
        mongoose: "^8.0.0",
        dotenv: "^16.0.0",
      },
      devDependencies: {
        "@types/node": "^20.0.0",
        typescript: "^5.0.0",
        "ts-node": "^10.0.0",
        nodemon: "^3.0.0",
      },
    };

    fs.writeFileSync("package.json", JSON.stringify(packageJson, null, 2));
    console.log("✅ package.json file बनाई गई");
  }

  console.log("\n" + "=".repeat(80));
  console.log("🎉 सभी files successfully generate हो गई हैं!");
  console.log("=".repeat(80));
  console.log("\n📁 Generated Files:");
  console.log("  • .env.example - Environment variables template");
  console.log("  • .env - Local development environment variables");
  console.log("  • mongodb.ts - MongoDB connection module");
  console.log("  • example-usage.ts - Example usage with Mongoose");
  console.log("  • package.json - Project dependencies");
  console.log("\n🚀 Next Steps:");
  console.log("  1. npm install");
  console.log("  2. MongoDB locally start करें: mongod");
  console.log("  3. Example run करें: npm run example");
}

async function interactiveMode(): Promise<void> {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("=".repeat(80));
  console.log("🤖 MongoDB Setup Assistant - Interactive Mode");
  console.log("=".repeat(80));
  console.log('💡 Tip: "exit" type करें बाहर निकलने के लिए');
  console.log('💡 Tip: "generate" type करें files generate करने के लिए');
  console.log("=".repeat(80) + "\n");

  const askQuestion = (): void => {
    rl.question("आप: ", async (input) => {
      const userInput = input.trim();

      if (!userInput) {
        askQuestion();
        return;
      }

      if (userInput.toLowerCase() === "exit") {
        console.log("\n👋 धन्यवाद! MongoDB setup assistant बंद हो रहा है।");
        rl.close();
        return;
      }

      if (userInput.toLowerCase() === "generate") {
        await generateActualFiles();
        askQuestion();
        return;
      }

      try {
        console.log("\nAssistant सोच रहा है...\n");
        const response = await chat(userInput);
        console.log("Assistant:", response);
        console.log("\n" + "-".repeat(80) + "\n");
      } catch (error) {
        console.error("❌ Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

// Main execution
async function main(): Promise<void> {
  const args = process.argv.slice(2);

  if (args.includes("--interactive") || args.includes("-i")) {
    await interactiveMode();
  } else {
    await generateMongoDBSetup();
  }
}

main().catch(console.error);
