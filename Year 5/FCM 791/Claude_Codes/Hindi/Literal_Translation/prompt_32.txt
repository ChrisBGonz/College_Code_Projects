import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.fasterxml.jackson.databind.JsonNode;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.security.SecureRandom;
import java.util.Base64;

public class AESEncryptionWithBedrock {
    
    private static final String AES_ALGORITHM = "AES";
    private static final String AES_GCM_TRANSFORMATION = "AES/GCM/NoPadding";
    private static final int GCM_IV_LENGTH = 12;
    private static final int GCM_TAG_LENGTH = 128;
    private static final int AES_KEY_SIZE = 256;
    
    private final BedrockRuntimeClient bedrockClient;
    private final ObjectMapper objectMapper;
    
    public AESEncryptionWithBedrock() {
        this.bedrockClient = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();
        this.objectMapper = new ObjectMapper();
    }
    
    /**
     * AES-GCM एल्गोरिथ्म का उपयोग करके एक संवेदनशील स्ट्रिंग को एन्क्रिप्ट करता है
     * @param plainText एन्क्रिप्ट करने के लिए टेक्स्ट
     * @param secretKey AES सीक्रेट की
     * @return Base64 एन्कोडेड एन्क्रिप्टेड स्ट्रिंग (IV + Encrypted Data)
     */
    public static String encryptString(String plainText, SecretKey secretKey) throws Exception {
        // IV (Initialization Vector) जनरेट करें
        byte[] iv = new byte[GCM_IV_LENGTH];
        SecureRandom secureRandom = new SecureRandom();
        secureRandom.nextBytes(iv);
        
        // Cipher इनिशियलाइज़ करें
        Cipher cipher = Cipher.getInstance(AES_GCM_TRANSFORMATION);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, parameterSpec);
        
        // एन्क्रिप्शन करें
        byte[] encryptedData = cipher.doFinal(plainText.getBytes("UTF-8"));
        
        // IV और एन्क्रिप्टेड डेटा को मिलाएं
        byte[] combined = new byte[iv.length + encryptedData.length];
        System.arraycopy(iv, 0, combined, 0, iv.length);
        System.arraycopy(encryptedData, 0, combined, iv.length, encryptedData.length);
        
        // Base64 एन्कोड करके रिटर्न करें
        return Base64.getEncoder().encodeToString(combined);
    }
    
    /**
     * AES-GCM एल्गोरिथ्म का उपयोग करके एन्क्रिप्टेड स्ट्रिंग को डिक्रिप्ट करता है
     * @param encryptedText Base64 एन्कोडेड एन्क्रिप्टेड स्ट्रिंग
     * @param secretKey AES सीक्रेट की
     * @return डिक्रिप्टेड प्लेन टेक्स्ट
     */
    public static String decryptString(String encryptedText, SecretKey secretKey) throws Exception {
        // Base64 डिकोड करें
        byte[] combined = Base64.getDecoder().decode(encryptedText);
        
        // IV और एन्क्रिप्टेड डेटा अलग करें
        byte[] iv = new byte[GCM_IV_LENGTH];
        byte[] encryptedData = new byte[combined.length - GCM_IV_LENGTH];
        System.arraycopy(combined, 0, iv, 0, iv.length);
        System.arraycopy(combined, iv.length, encryptedData, 0, encryptedData.length);
        
        // Cipher इनिशियलाइज़ करें
        Cipher cipher = Cipher.getInstance(AES_GCM_TRANSFORMATION);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.DECRYPT_MODE, secretKey, parameterSpec);
        
        // डिक्रिप्शन करें
        byte[] decryptedData = cipher.doFinal(encryptedData);
        
        return new String(decryptedData, "UTF-8");
    }
    
    /**
     * एक नई AES सीक्रेट की जनरेट करता है
     * @return नई SecretKey
     */
    public static SecretKey generateAESKey() throws Exception {
        KeyGenerator keyGenerator = KeyGenerator.getInstance(AES_ALGORITHM);
        keyGenerator.init(AES_KEY_SIZE, new SecureRandom());
        return keyGenerator.generateKey();
    }
    
    /**
     * SecretKey को Base64 स्ट्रिंग में कन्वर्ट करता है
     * @param secretKey AES सीक्रेट की
     * @return Base64 एन्कोडेड की स्ट्रिंग
     */
    public static String keyToString(SecretKey secretKey) {
        return Base64.getEncoder().encodeToString(secretKey.getEncoded());
    }
    
    /**
     * Base64 स्ट्रिंग से SecretKey बनाता है
     * @param keyString Base64 एन्कोडेड की स्ट्रिंग
     * @return SecretKey ऑब्जेक्ट
     */
    public static SecretKey stringToKey(String keyString) {
        byte[] keyBytes = Base64.getDecoder().decode(keyString);
        return new SecretKeySpec(keyBytes, AES_ALGORITHM);
    }
    
    /**
     * Claude AI से एन्क्रिप्शन के बारे में जानकारी प्राप्त करता है
     * @param question एन्क्रिप्शन से संबंधित प्रश्न
     * @return AI का जवाब
     */
    public String askClaudeAboutEncryption(String question) throws Exception {
        ObjectNode requestBody = objectMapper.createObjectNode();
        requestBody.put("anthropic_version", "bedrock-2023-05-31");
        requestBody.put("max_tokens", 1024);
        
        ObjectNode message = objectMapper.createObjectNode();
        message.put("role", "user");
        message.put("content", question);
        
        requestBody.set("messages", objectMapper.createArrayNode().add(message));
        
        String requestBodyJson = objectMapper.writeValueAsString(requestBody);
        
        InvokeModelRequest request = InvokeModelRequest.builder()
                .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
                .body(SdkBytes.fromUtf8String(requestBodyJson))
                .build();
        
        InvokeModelResponse response = bedrockClient.invokeModel(request);
        String responseBody = response.body().asUtf8String();
        
        JsonNode responseJson = objectMapper.readTree(responseBody);
        return responseJson.get("content").get(0).get("text").asText();
    }
    
    /**
     * एन्क्रिप्शन प्रक्रिया का प्रदर्शन करता है और AI से सहायता लेता है
     */
    public void demonstrateEncryption(String sensitiveData) throws Exception {
        System.out.println("=== AES एन्क्रिप्शन डेमो ===\n");
        
        // AES की जनरेट करें
        SecretKey secretKey = generateAESKey();
        String keyString = keyToString(secretKey);
        System.out.println("जनरेट की गई AES की (Base64): " + keyString.substring(0, 20) + "...");
        
        // डेटा एन्क्रिप्ट करें
        System.out.println("\nमूल संवेदनशील डेटा: " + sensitiveData);
        String encryptedData = encryptString(sensitiveData, secretKey);
        System.out.println("एन्क्रिप्टेड डेटा: " + encryptedData.substring(0, 30) + "...");
        
        // डेटा डिक्रिप्ट करें
        String decryptedData = decryptString(encryptedData, secretKey);
        System.out.println("डिक्रिप्टेड डेटा: " + decryptedData);
        
        // एन्क्रिप्शन सफल है या नहीं जांचें
        boolean isSuccessful = sensitiveData.equals(decryptedData);
        System.out.println("\nएन्क्रिप्शन/डिक्रिप्शन सफल: " + isSuccessful);
        
        // Claude से AES एन्क्रिप्शन के बारे में जानकारी प्राप्त करें
        System.out.println("\n=== Claude AI से AES एन्क्रिप्शन की जानकारी ===");
        String aiResponse = askClaudeAboutEncryption(
            "AES-GCM एन्क्रिप्शन के मुख्य सुरक्षा लाभ क्या हैं? " +
            "संक्षेप में 3 मुख्य बिंदु बताएं।"
        );
        System.out.println("\nClaude का जवाब:\n" + aiResponse);
    }
    
    public static void main(String[] args) {
        AESEncryptionWithBedrock demo = new AESEncryptionWithBedrock();
        
        // संवेदनशील डेटा के उदाहरण
        String sensitiveData = "यह एक गोपनीय संदेश है: पासवर्ड@123, क्रेडिट कार्ड: 4111-1111-1111-1111";
        
        try {
            demo.demonstrateEncryption(sensitiveData);
            
            // अतिरिक्त उदाहरण: की को स्टोर और रिस्टोर करना
            System.out.println("\n=== की स्टोरेज और रिस्टोरेशन डेमो ===");
            SecretKey originalKey = generateAESKey();
            String storedKey = keyToString(originalKey);
            SecretKey restoredKey = stringToKey(storedKey);
            
            String testData = "टेस्ट डेटा";
            String encrypted = encryptString(testData, originalKey);
            String decrypted = decryptString(encrypted, restoredKey);
            
            System.out.println("की स्टोरेज और रिस्टोरेशन सफल: " + testData.equals(decrypted));
            
        } catch (Exception e) {
            System.err.println("एरर: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
