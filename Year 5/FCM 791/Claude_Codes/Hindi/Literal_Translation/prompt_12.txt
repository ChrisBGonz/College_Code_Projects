
```javascript
const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateExpressServerConfig() {
  console.log(
    "Express.js प्रोडक्शन सर्वर कॉन्फ़िगरेशन जनरेट हो रही है...\n"
  );

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `उत्पादन (Production) वातावरण के लिए एक बुनियादी Express.js सर्वर कॉन्फ़िगरेशन तैयार करें, जिसमें निम्नलिखित शामिल हों:

1. **सुरक्षा मिडलवेयर** (helmet, cors, rate limiting)
2. **लॉगिंग** (morgan या winston)
3. **त्रुटि-प्रबंधन मिडलवेयर** (Error Handling Middleware)
4. **पर्यावरण चर** (Environment Variables) का उपयोग
5. **स्वास्थ्य जांच** (Health Check) endpoint
6. **Graceful Shutdown** हैंडलिंग
7. **Request validation** और sanitization
8. **Compression** मिडलवेयर

कृपया पूर्ण, production-ready कोड प्रदान करें जिसमें:
- सभी आवश्यक npm packages की सूची
- विस्तृत टिप्पणियाँ (comments) हिंदी में
- Best practices का पालन
- उचित error codes और messages

कोड को modular और maintainable बनाएं।`,
      },
    ],
  });

  let fullResponse = "";

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  const finalMessage = await stream.finalMessage();

  console.log("\n\n" + "=".repeat(60));
  console.log("जनरेशन पूर्ण!");
  console.log(
    `कुल टोकन उपयोग: ${finalMessage.usage.input_tokens + finalMessage.usage.output_tokens}`
  );
  console.log(`Input tokens: ${finalMessage.usage.input_tokens}`);
  console.log(`Output tokens: ${finalMessage.usage.output_tokens}`);

  return fullResponse;
}

async function main() {
  try {
    await generateExpressServerConfig();
  } catch (error) {
    if (error instanceof Anthropic.APIError) {
      console.error(`API त्रुटि: ${error.status} - ${error.message}`);
      process.exit(1);
    }
    throw error;
  }
}

main();
```
