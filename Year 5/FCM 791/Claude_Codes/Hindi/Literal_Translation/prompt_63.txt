import java.util.regex.Pattern;
import java.util.ArrayList;
import java.util.List;

public class PasswordValidator {

    // पासवर्ड की न्यूनतम और अधिकतम लंबाई
    private static final int MIN_LENGTH = 8;
    private static final int MAX_LENGTH = 128;

    // रेगेक्स पैटर्न
    private static final Pattern UPPERCASE_PATTERN = Pattern.compile("[A-Z]");
    private static final Pattern LOWERCASE_PATTERN = Pattern.compile("[a-z]");
    private static final Pattern DIGIT_PATTERN = Pattern.compile("[0-9]");
    private static final Pattern SPECIAL_CHAR_PATTERN = Pattern.compile("[!@#$%^&*()_+\\-=\\[\\]{};':\"\\\\|,.<>\\/?]");
    private static final Pattern WHITESPACE_PATTERN = Pattern.compile("\\s");
    private static final Pattern SEQUENTIAL_PATTERN = Pattern.compile("(012|123|234|345|456|567|678|789|890|abc|bcd|cde|def|efg|fgh|ghi|hij|ijk|jkl|klm|lmn|mno|nop|opq|pqr|qrs|rst|stu|tuv|uvw|vwx|wxy|xyz)", Pattern.CASE_INSENSITIVE);
    private static final Pattern REPEATED_CHAR_PATTERN = Pattern.compile("(.)\\1{2,}");

    /**
     * पासवर्ड की ताकत की जांच करने वाला मुख्य मेथड
     * 
     * @param password - उपयोगकर्ता द्वारा दर्ज किया गया पासवर्ड
     * @return PasswordStrengthResult - पासवर्ड की ताकत का परिणाम
     */
    public static PasswordStrengthResult validatePassword(String password) {
        List<String> errors = new ArrayList<>();
        List<String> suggestions = new ArrayList<>();
        int strengthScore = 0;

        // null या खाली पासवर्ड की जांच
        if (password == null || password.isEmpty()) {
            errors.add("पासवर्ड खाली नहीं हो सकता।");
            return new PasswordStrengthResult(false, PasswordStrength.VERY_WEAK, 0, errors, suggestions);
        }

        // ===== बुनियादी जांच =====

        // 1. लंबाई की जांच
        if (password.length() < MIN_LENGTH) {
            errors.add("पासवर्ड कम से कम " + MIN_LENGTH + " अक्षरों का होना चाहिए।");
        } else if (password.length() >= MIN_LENGTH && password.length() < 12) {
            strengthScore += 1;
            suggestions.add("पासवर्ड को 12 या अधिक अक्षरों का बनाएं।");
        } else if (password.length() >= 12 && password.length() < 16) {
            strengthScore += 2;
        } else if (password.length() >= 16) {
            strengthScore += 3;
        }

        if (password.length() > MAX_LENGTH) {
            errors.add("पासवर्ड " + MAX_LENGTH + " अक्षरों से अधिक नहीं हो सकता।");
        }

        // 2. व्हाइटस्पेस की जांच
        if (WHITESPACE_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में स्पेस नहीं होनी चाहिए।");
        }

        // ===== कैरेक्टर टाइप की जांच =====

        // 3. बड़े अक्षर (Uppercase) की जांच
        if (!UPPERCASE_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में कम से कम एक बड़ा अक्षर (A-Z) होना चाहिए।");
            suggestions.add("कम से कम एक बड़ा अक्षर जोड़ें (जैसे: A, B, C...)।");
        } else {
            strengthScore += 1;
        }

        // 4. छोटे अक्षर (Lowercase) की जांच
        if (!LOWERCASE_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में कम से कम एक छोटा अक्षर (a-z) होना चाहिए।");
            suggestions.add("कम से कम एक छोटा अक्षर जोड़ें (जैसे: a, b, c...)।");
        } else {
            strengthScore += 1;
        }

        // 5. अंक (Digit) की जांच
        if (!DIGIT_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में कम से कम एक अंक (0-9) होना चाहिए।");
            suggestions.add("कम से कम एक अंक जोड़ें (जैसे: 1, 2, 3...)।");
        } else {
            strengthScore += 1;
        }

        // 6. विशेष अक्षर (Special Character) की जांच
        if (!SPECIAL_CHAR_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में कम से कम एक विशेष अक्षर (!@#$%^&* आदि) होना चाहिए।");
            suggestions.add("कम से कम एक विशेष अक्षर जोड़ें (जैसे: !, @, #, $...)।");
        } else {
            strengthScore += 2;
        }

        // ===== उन्नत जांच =====

        // 7. क्रमिक अक्षरों की जांच (जैसे: 123, abc)
        if (SEQUENTIAL_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में क्रमिक अक्षर या अंक (जैसे: 123, abc) नहीं होने चाहिए।");
            strengthScore -= 1;
        }

        // 8. बार-बार दोहराए गए अक्षरों की जांच (जैसे: aaa, 111)
        if (REPEATED_CHAR_PATTERN.matcher(password).find()) {
            errors.add("पासवर्ड में एक ही अक्षर या अंक तीन या अधिक बार नहीं होना चाहिए।");
            strengthScore -= 1;
        }

        // 9. सामान्य पासवर्ड की जांच
        if (isCommonPassword(password)) {
            errors.add("यह पासवर्ड बहुत सामान्य है। कृपया एक अनोखा पासवर्ड चुनें।");
            strengthScore -= 2;
        }

        // 10. कीबोर्ड पैटर्न की जांच
        if (isKeyboardPattern(password)) {
            errors.add("पासवर्ड में कीबोर्ड पैटर्न (जैसे: qwerty, asdf) नहीं होना चाहिए।");
            strengthScore -= 1;
        }

        // स्कोर को 0 से कम नहीं होने देना
        strengthScore = Math.max(0, strengthScore);

        // पासवर्ड की ताकत निर्धारित करना
        PasswordStrength strength = determineStrength(strengthScore);

        // क्या पासवर्ड वैध है?
        boolean isValid = errors.isEmpty();

        // अतिरिक्त सुझाव
        if (isValid && strength != PasswordStrength.VERY_STRONG) {
            addStrengthSuggestions(password, suggestions);
        }

        return new PasswordStrengthResult(isValid, strength, strengthScore, errors, suggestions);
    }

    /**
     * पासवर्ड की ताकत निर्धारित करने वाला मेथड
     */
    private static PasswordStrength determineStrength(int score) {
        if (score <= 1) {
            return PasswordStrength.VERY_WEAK;
        } else if (score <= 3) {
            return PasswordStrength.WEAK;
        } else if (score <= 5) {
            return PasswordStrength.MEDIUM;
        } else if (score <= 7) {
            return PasswordStrength.STRONG;
        } else {
            return PasswordStrength.VERY_STRONG;
        }
    }

    /**
     * सामान्य पासवर्ड की जांच करने वाला मेथड
     */
    private static boolean isCommonPassword(String password) {
        String[] commonPasswords = {
            "password", "password123", "123456", "12345678", "qwerty",
            "abc123", "monkey", "1234567", "letmein", "trustno1",
            "dragon", "baseball", "iloveyou", "master", "sunshine",
            "ashley", "bailey", "passw0rd", "shadow", "123123",
            "654321", "superman", "qazwsx", "michael", "football",
            "password1", "admin", "welcome", "login", "hello"
        };

        String lowerPassword = password.toLowerCase();
        for (String common : commonPasswords) {
            if (lowerPassword.equals(common)) {
                return true;
            }
        }
        return false;
    }

    /**
     * कीबोर्ड पैटर्न की जांच करने वाला मेथड
     */
    private static boolean isKeyboardPattern(String password) {
        String[] keyboardPatterns = {
            "qwerty", "asdfgh", "zxcvbn", "qwertyuiop",
            "asdfghjkl", "zxcvbnm", "1qaz2wsx", "qazwsx"
        };

        String lowerPassword = password.toLowerCase();
        for (String pattern : keyboardPatterns) {
            if (lowerPassword.contains(pattern)) {
                return true;
            }
        }
        return false;
    }

    /**
     * ताकत बढ़ाने के सुझाव देने वाला मेथड
     */
    private static void addStrengthSuggestions(String password, List<String> suggestions) {
        if (password.length() < 16) {
            suggestions.add("पासवर्ड को और लंबा बनाएं (16+ अक्षर) ताकि यह और मजबूत हो।");
        }

        long uppercaseCount = password.chars().filter(Character::isUpperCase).count();
        long digitCount = password.chars().filter(Character::isDigit).count();

        if (uppercaseCount < 2) {
            suggestions.add("अधिक बड़े अक्षर जोड़ने से पासवर्ड मजबूत होगा।");
        }

        if (digitCount < 2) {
            suggestions.add("अधिक अंक जोड़ने से पासवर्ड मजबूत होगा।");
        }
    }

    // ===== आंतरिक क्लास और एनम =====

    /**
     * पासवर्ड की ताकत के स्तर
     */
    public enum PasswordStrength {
        VERY_WEAK("बहुत कमजोर"),
        WEAK("कमजोर"),
        MEDIUM("मध्यम"),
        STRONG("मजबूत"),
        VERY_STRONG("बहुत मजबूत");

        private final String hindiLabel;

        PasswordStrength(String hindiLabel) {
            this.hindiLabel = hindiLabel;
        }

        public String getHindiLabel() {
            return hindiLabel;
        }
    }

    /**
     * पासवर्ड सत्यापन का परिणाम
     */
    public static class PasswordStrengthResult {
        private final boolean isValid;
        private final PasswordStrength strength;
        private final int score;
        private final List<String> errors;
        private final List<String> suggestions;

        public PasswordStrengthResult(boolean isValid, PasswordStrength strength,
                                      int score, List<String> errors, List<String> suggestions) {
            this.isValid = isValid;
            this.strength = strength;
            this.score = score;
            this.errors = errors;
            this.suggestions = suggestions;
        }

        public boolean isValid() {
            return isValid;
        }

        public PasswordStrength getStrength() {
            return strength;
        }

        public int getScore() {
            return score;
        }

        public List<String> getErrors() {
            return errors;
        }

        public List<String> getSuggestions() {
            return suggestions;
        }

        @Override
        public String toString() {
            StringBuilder sb = new StringBuilder();
            sb.append("=== पासवर्ड सत्यापन परिणाम ===\n");
            sb.append("वैध: ").append(isValid ? "हाँ ✓" : "नहीं ✗").append("\n");
            sb.append("ताकत: ").append(strength.getHindiLabel()).append(" (स्कोर: ").append(score).append(")\n");

            if (!errors.isEmpty()) {
                sb.append("\nत्रुटियाँ:\n");
                for (String error : errors) {
                    sb.append("  ✗ ").append(error).append("\n");
                }
            }

            if (!suggestions.isEmpty()) {
                sb.append("\nसुझाव:\n");
                for (String suggestion : suggestions) {
                    sb.append("  → ").append(suggestion).append("\n");
                }
            }

            return sb.toString();
        }
    }

    // ===== मुख्य मेथड (परीक्षण के लिए) =====
    public static void main(String[] args) {
        String[] testPasswords = {
            "abc",                    // बहुत कमजोर
            "password",               // सामान्य पासवर्ड
            "Password1",              // मध्यम
            "Password@123",           // मजबूत
            "MyStr0ng!Pass#2024",     // बहुत मजबूत
            "qwerty123",              // कीबोर्ड पैटर्न
            "aaa111BBB",              // दोहराए गए अक्षर
            "Abc@12 34"               // स्पेस वाला
        };

        System.out.println("╔══════════════════════════════════════════╗");
        System.out.println("║      पासवर्ड सत्यापन परीक्षण             ║");
        System.out.println("╚══════════════════════════════════════════╝\n");

        for (String password : testPasswords) {
            System.out.println("पासवर्ड: \"" + password + "\"");
            System.out.println("─".repeat(50));
            PasswordStrengthResult result = validatePassword(password);
            System.out.println(result);
            System.out.println("═".repeat(50) + "\n");
        }
    }
}
