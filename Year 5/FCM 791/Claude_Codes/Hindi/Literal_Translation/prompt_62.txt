
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Simulated session store (in real app, this would be a database or Redis)
const sessionStore = new Map();

// Function to create a new session
function createSession(userId) {
  const sessionId = `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  const session = {
    sessionId,
    userId,
    createdAt: new Date().toISOString(),
    lastActivity: new Date().toISOString(),
    isActive: true,
    data: {},
  };
  sessionStore.set(sessionId, session);
  return session;
}

// Function to get session
function getSession(sessionId) {
  return sessionStore.get(sessionId) || null;
}

// Function to handle logout - removes user session
function handleLogout(sessionId) {
  const session = sessionStore.get(sessionId);

  if (!session) {
    return {
      success: false,
      message: "Session not found or already logged out",
      sessionId: sessionId,
    };
  }

  if (!session.isActive) {
    return {
      success: false,
      message: "Session is already inactive",
      sessionId: sessionId,
    };
  }

  // Remove session from store
  sessionStore.delete(sessionId);

  return {
    success: true,
    message: "Successfully logged out",
    sessionId: sessionId,
    userId: session.userId,
    logoutTime: new Date().toISOString(),
  };
}

// Function to list all active sessions
function listActiveSessions() {
  const sessions = [];
  sessionStore.forEach((session, sessionId) => {
    sessions.push({
      sessionId,
      userId: session.userId,
      createdAt: session.createdAt,
      isActive: session.isActive,
    });
  });
  return sessions;
}

// Chat function with Claude
async function chat(userMessage) {
  // Add user message to history
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  // Create system prompt with context about session management
  const systemPrompt = `You are a helpful assistant that helps users manage their sessions and understand logout functionality. 
You have access to session management functions. When users ask about sessions or logout, provide helpful information.
Current active sessions: ${JSON.stringify(listActiveSessions())}

Available commands you can help with:
- Create session: createSession(userId)
- Get session info: getSession(sessionId)  
- Logout/Remove session: handleLogout(sessionId)
- List all sessions: listActiveSessions()

Respond in Hindi when the user writes in Hindi, and in English when they write in English.`;

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  // Extract assistant's response
  const assistantMessage = response.content[0].text;

  // Add assistant response to history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Process user commands
async function processCommand(input) {
  const trimmedInput = input.trim().toLowerCase();

  // Check for direct commands
  if (trimmedInput.startsWith("create session")) {
    const parts = input.split(" ");
    const userId = parts[2] || `user_${Date.now()}`;
    const session = createSession(userId);
    console.log("\n✅ Session Created:");
    console.log(JSON.stringify(session, null, 2));
    return;
  }

  if (trimmedInput.startsWith("logout") || trimmedInput.startsWith("लॉगआउट")) {
    const parts = input.split(" ");
    const sessionId = parts[1];

    if (!sessionId) {
      console.log(
        "\n❌ Please provide session ID: logout <sessionId> or लॉगआउट <sessionId>"
      );
      return;
    }

    const result = handleLogout(sessionId);
    console.log("\n🔐 Logout Result:");
    console.log(JSON.stringify(result, null, 2));
    return;
  }

  if (trimmedInput === "list sessions" || trimmedInput === "सत्र सूची") {
    const sessions = listActiveSessions();
    console.log("\n📋 Active Sessions:");
    if (sessions.length === 0) {
      console.log("No active sessions found");
    } else {
      console.log(JSON.stringify(sessions, null, 2));
    }
    return;
  }

  if (trimmedInput.startsWith("get session")) {
    const parts = input.split(" ");
    const sessionId = parts[2];

    if (!sessionId) {
      console.log("\n❌ Please provide session ID: get session <sessionId>");
      return;
    }

    const session = getSession(sessionId);
    if (session) {
      console.log("\n📌 Session Info:");
      console.log(JSON.stringify(session, null, 2));
    } else {
      console.log("\n❌ Session not found");
    }
    return;
  }

  // For other inputs, use Claude for conversation
  const response = await chat(input);
  console.log("\n🤖 Assistant:", response);
}

// Main function
async function main() {
  console.log("🚀 Session Management System with Claude AI");
  console.log("===========================================");
  console.log("Commands:");
  console.log("  create session <userId>  - Create a new session");
  console.log("  logout <sessionId>       - Logout and remove session");
  console.log("  लॉगआउट <sessionId>      - Logout (Hindi)");
  console.log("  list sessions            - List all active sessions");
  console.log("  सत्र सूची               - List sessions (Hindi)");
  console.log("  get session <sessionId>  - Get session info");
  console.log("  Or chat with AI about session management");
  console.log("  Type 'exit' to quit\n");

  // Create some demo sessions
  console.log("📝 Creating demo sessions...");
  const session1 = createSession("user123");
  const session2 = createSession("user456");
  console.log(`✅ Created session for user123: ${session1.sessionId}`);
  console.log(`✅ Created session for user456: ${session2.sessionId}\n`);

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const askQuestion = () => {
    rl.question("\n💬 You: ", async (input) => {
      if (input.toLowerCase() === "exit") {
        console.log("\n👋 Goodbye!");
        rl.close();
        return;
      }

      if (input.trim()) {
        await processCommand(input);
      }

      askQuestion();
    });
  };

  askQuestion();
}

main().catch(console.error);
```
