import anthropic
import hashlib
import os
import urllib.request
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def verify_firmware_with_claude(
    file_path: str, expected_hash: str | None = None
) -> None:
    """Use Claude to analyze and verify firmware file integrity."""
    client = anthropic.Anthropic()

    actual_hash = calculate_sha256(file_path)
    file_size = os.path.getsize(file_path)
    file_name = Path(file_path).name

    verification_status = "UNKNOWN"
    if expected_hash:
        if actual_hash.lower() == expected_hash.lower():
            verification_status = "VERIFIED - Hash matches!"
        else:
            verification_status = "FAILED - Hash mismatch!"

    analysis_prompt = f"""
    मुझे एक फर्मवेयर फ़ाइल की अखंडता जांच के परिणामों का विश्लेषण करने में मदद करें:
    
    फ़ाइल की जानकारी:
    - फ़ाइल का नाम: {file_name}
    - फ़ाइल का आकार: {file_size} bytes ({file_size / 1024:.2f} KB)
    - SHA256 हैश (गणना की गई): {actual_hash}
    - अपेक्षित SHA256 हैश: {expected_hash if expected_hash else "प्रदान नहीं किया गया"}
    - सत्यापन स्थिति: {verification_status}
    
    कृपया निम्नलिखित बताएं:
    1. SHA256 सत्यापन का क्या महत्व है?
    2. इस सत्यापन परिणाम का क्या अर्थ है?
    3. फर्मवेयर अखंडता के लिए सुरक्षा सिफारिशें
    4. अगले कदम क्या होने चाहिए?
    
    हिंदी में विस्तृत विश्लेषण प्रदान करें।
    """

    print("\n" + "=" * 60)
    print("फर्मवेयर अखंडता सत्यापन रिपोर्ट")
    print("=" * 60)
    print(f"फ़ाइल: {file_name}")
    print(f"आकार: {file_size} bytes ({file_size / 1024:.2f} KB)")
    print(f"SHA256: {actual_hash}")

    if expected_hash:
        print(f"अपेक्षित: {expected_hash}")
        print(f"स्थिति: {verification_status}")

    print("\nClaude AI विश्लेषण:")
    print("-" * 40)

    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": analysis_prompt}],
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)

    print("\n" + "=" * 60)


def create_sample_firmware() -> str:
    """Create a sample firmware file for demonstration."""
    firmware_content = b"""
    FIRMWARE v2.1.0
    Device: IoT Sensor Module
    Build Date: 2024-01-15
    
    [HEADER]
    Magic: 0xDEADBEEF
    Version: 2.1.0
    Checksum: 0xABCD1234
    
    [BOOT SECTION]
    Boot loader initialized
    Memory mapped: 0x00000000 - 0x0007FFFF
    Stack pointer: 0x0007FFF0
    
    [APPLICATION CODE]
    Initialize peripherals...
    Configure UART: 115200 baud
    Configure I2C: 400kHz
    Configure SPI: 1MHz
    
    [SENSOR CONFIGURATION]
    Temperature sensor: enabled
    Humidity sensor: enabled
    Pressure sensor: enabled
    
    [NETWORK STACK]
    WiFi module: initialized
    Protocol: MQTT
    Server: iot.example.com:1883
    
    [END OF FIRMWARE]
    """

    firmware_path = "sample_firmware.bin"
    with open(firmware_path, "wb") as f:
        f.write(firmware_content)

    return firmware_path


def main():
    """Main function to demonstrate firmware integrity verification."""
    print("फर्मवेयर अखंडता सत्यापन प्रणाली")
    print("SHA256 हैश का उपयोग करके")
    print("=" * 60)

    print("\nनमूना फर्मवेयर फ़ाइल बना रहे हैं...")
    firmware_path = create_sample_firmware()
    print(f"फर्मवेयर फ़ाइल बनाई गई: {firmware_path}")

    correct_hash = calculate_sha256(firmware_path)
    print(f"\nसही SHA256 हैश: {correct_hash}")

    print("\n1. सही हैश के साथ सत्यापन:")
    verify_firmware_with_claude(firmware_path, correct_hash)

    print("\n\n2. गलत हैश के साथ सत्यापन (छेड़छाड़ का अनुकरण):")
    wrong_hash = "a" * 64
    verify_firmware_with_claude(firmware_path, wrong_hash)

    print("\n\n3. बिना अपेक्षित हैश के सत्यापन:")
    verify_firmware_with_claude(firmware_path)

    if os.path.exists(firmware_path):
        os.remove(firmware_path)
        print(f"\nनमूना फ़ाइल हटाई गई: {firmware_path}")


if __name__ == "__main__":
    main()
