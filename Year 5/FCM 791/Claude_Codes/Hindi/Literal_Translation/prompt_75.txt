class SecureUserPreferences {
    private static $allowedThemes = ['light', 'dark', 'blue', 'green'];
    private static $allowedLanguages = ['hi', 'en', 'mr', 'gu', 'ta'];
    private static $minFontSize = 10;
    private static $maxFontSize = 24;
    
    public $theme;
    public $language;
    public $fontSize;
    public $notifications;
    
    public function __construct($theme = 'light', $language = 'hi', $fontSize = 14, $notifications = true) {
        $this->theme = $this->validateTheme($theme);
        $this->language = $this->validateLanguage($language);
        $this->fontSize = $this->validateFontSize($fontSize);
        $this->notifications = (bool)$notifications;
    }
    
    private function validateTheme($theme) {
        return in_array($theme, self::$allowedThemes) ? $theme : 'light';
    }
    
    private function validateLanguage($language) {
        return in_array($language, self::$allowedLanguages) ? $language : 'hi';
    }
    
    private function validateFontSize($size) {
        $size = (int)$size;
        return max(self::$minFontSize, min(self::$maxFontSize, $size));
    }
    
    public function toArray() {
        return [
            'theme' => $this->theme,
            'language' => $this->language,
            'fontSize' => $this->fontSize,
            'notifications' => $this->notifications
        ];
    }
    
    public static function fromArray($data) {
        return new self(
            $data['theme'] ?? 'light',
            $data['language'] ?? 'hi',
            $data['fontSize'] ?? 14,
            $data['notifications'] ?? true
        );
    }
}

// ✅ SECURE: JSON का उपयोग serialize() की जगह
function secureLoadPreferences() {
    if (isset($_COOKIE['user_prefs_secure'])) {
        $cookieData = $_COOKIE['user_prefs_secure'];
        
        // ✅ SECURE: JSON decode करें
        $decoded = base64_decode($cookieData, true);
        if ($decoded === false) {
            return new SecureUserPreferences();
        }
        
        // HMAC signature verify करें
        $parts = explode('|', $decoded, 2);
        if (count($parts) !== 2) {
            return new SecureUserPreferences();
        }
        
        [$signature, $jsonData] = $parts;
        $secretKey = getenv('COOKIE_SECRET') ?: 'change-this-secret-key-in-production';
        $expectedSignature = hash_hmac('sha256', $jsonData, $secretKey);
        
        // ✅ SECURE: Timing-safe comparison
        if (!hash_equals($expectedSignature, $signature)) {
            error_log("Cookie tampering detected!");
            return new SecureUserPreferences();
        }
        
        // ✅ SECURE: JSON parse करें (serialize नहीं)
        $data = json_decode($jsonData, true);
        if (!is_array($data)) {
            return new SecureUserPreferences();
        }
        
        return SecureUserPreferences::fromArray($data);
    }
    
    return new SecureUserPreferences();
}

function secureSavePreferences(SecureUserPreferences $prefs) {
    $jsonData = json_encode($prefs->toArray());
    $secretKey = getenv('COOKIE_SECRET') ?: 'change-this-secret-key-in-production';
    
    // HMAC signature generate करें
    $signature = hash_hmac('sha256', $jsonData, $secretKey);
    $cookieValue = base64_encode($signature . '|' . $jsonData);
    
    // ✅ SECURE: HttpOnly, Secure, SameSite flags के साथ cookie set करें
    setcookie(
        'user_prefs_secure',
        $cookieValue,
        [
            'expires' => time() + 86400,
            'path' => '/',
            'domain' => '',
            'secure' => true,        // HTTPS only
            'httponly' => true,      // JavaScript access नहीं
            'samesite' => 'Strict'   // CSRF protection
        ]
    );
}
