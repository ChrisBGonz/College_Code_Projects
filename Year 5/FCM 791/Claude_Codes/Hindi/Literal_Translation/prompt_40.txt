async function chat(userMessage) {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main() {
  try {
    const result = await generateCookieEncryptionMiddleware();

    console.log("🎉 Cookie Encryption Middleware Generation Complete!");
    console.log("====================================================");
    console.log("\n📊 Summary:");
    console.log("- ✅ Base middleware generated");
    console.log("- ✅ Usage examples provided");
    console.log("- ✅ Security enhancements added");
    console.log("- ✅ Test suite created");
    console.log(
      "\n💡 The middleware is ready to be integrated into your Express.js application!"
    );

    // Create a simple demo server to show the middleware in action
    console.log("\n🔧 Starting demo server on port 3000...");

    // Simple demo implementation
    const crypto = require("crypto");

    const ENCRYPTION_KEY =
      process.env.COOKIE_ENCRYPTION_KEY ||
      crypto.randomBytes(32).toString("hex");
    const ALGORITHM = "aes-256-gcm";

    function encryptCookie(value) {
      const iv = crypto.randomBytes(16);
      const key = Buffer.from(ENCRYPTION_KEY.slice(0, 64), "hex");
      const cipher = crypto.createCipheriv(ALGORITHM, key, iv);

      let encrypted = cipher.update(value, "utf8", "hex");
      encrypted += cipher.final("hex");

      const authTag = cipher.getAuthTag();

      return `${iv.toString("hex")}:${authTag.toString("hex")}:${encrypted}`;
    }

    function decryptCookie(encryptedValue) {
      const parts = encryptedValue.split(":");
      if (parts.length !== 3) {
        throw new Error("Invalid encrypted cookie format");
      }

      const [ivHex, authTagHex, encrypted] = parts;
      const iv = Buffer.from(ivHex, "hex");
      const authTag = Buffer.from(authTagHex, "hex");
      const key = Buffer.from(ENCRYPTION_KEY.slice(0, 64), "hex");

      const decipher = crypto.createDecipheriv(ALGORITHM, key, iv);
      decipher.setAuthTag(authTag);

      let decrypted = decipher.update(encrypted, "hex", "utf8");
      decrypted += decipher.final("utf8");

      return decrypted;
    }

    // Cookie encryption middleware
    const cookieEncryptionMiddleware = (req, res, next) => {
      // Decrypt incoming cookies
      if (req.headers.cookie) {
        const cookies = req.headers.cookie.split(";");
        const decryptedCookies = {};

        cookies.forEach((cookie) => {
          const [name, value] = cookie.trim().split("=");
          if (name && value) {
            try {
              decryptedCookies[name] = decryptCookie(
                decodeURIComponent(value)
              );
            } catch (e) {
              // Cookie might not be encrypted, use as-is
              decryptedCookies[name] = value;
            }
          }
        });

        req.decryptedCookies = decryptedCookies;
      }

      // Override res.cookie to encrypt cookies
      const originalCookie = res.cookie.bind(res);
      res.cookie = function (name, value, options = {}) {
        const encryptedValue = encryptCookie(String(value));
        const secureOptions = {
          ...options,
          httpOnly: true,
          secure: process.env.NODE_ENV === "production",
          sameSite: "strict",
        };
        return originalCookie(
          name,
          encodeURIComponent(encryptedValue),
          secureOptions
        );
      };

      next();
    };

    app.use(cookieEncryptionMiddleware);

    // Demo routes
    app.get("/set-cookie", (req, res) => {
      res.cookie("user_session", "user123_secret_data", {
        maxAge: 3600000,
      });
      res.cookie("preferences", JSON.stringify({ theme: "dark", lang: "hi" }), {
        maxAge: 86400000,
      });
      res.json({
        message: "Encrypted cookies set successfully!",
        note: "Check your browser cookies to see the encrypted values",
      });
    });

    app.get("/read-cookie", (req, res) => {
      const decryptedCookies = req.decryptedCookies || {};
      res.json({
        message: "Decrypted cookie values",
        cookies: decryptedCookies,
        rawCookies: req.headers.cookie || "No cookies found",
      });
    });

    app.get("/clear-cookies", (req, res) => {
      res.clearCookie("user_session");
      res.clearCookie("preferences");
      res.json({ message: "Cookies cleared successfully!" });
    });

    app.listen(3000, () => {
      console.log("✅ Demo server running at http://localhost:3000");
      console.log("\n📌 Available endpoints:");
      console.log("  GET /set-cookie  - Set encrypted cookies");
      console.log("  GET /read-cookie - Read and decrypt cookies");
      console.log("  GET /clear-cookies - Clear all cookies");
      console.log("\n🔐 Encryption Key (keep this secret!):");
      console.log(`  ${ENCRYPTION_KEY.slice(0, 20)}...`);
    });
  } catch (error) {
    console.error("❌ Error:", error.message);
    process.exit(1);
  }
}

main();
