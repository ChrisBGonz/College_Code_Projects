using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    // Simulated user database
    private static Dictionary<int, User> users = new Dictionary<int, User>
    {
        { 1, new User { Id = 1, Name = "Alice Johnson", Email = "alice@example.com", Role = "user" } },
        { 2, new User { Id = 2, Name = "Bob Smith", Email = "bob@example.com", Role = "user" } },
        { 3, new User { Id = 3, Name = "Charlie Brown", Email = "charlie@example.com", Role = "admin" } },
        { 4, new User { Id = 4, Name = "Diana Prince", Email = "diana@example.com", Role = "user" } },
        { 5, new User { Id = 5, Name = "Eve Wilson", Email = "eve@example.com", Role = "moderator" } }
    };

    // Current logged-in user (simulated)
    private static User currentUser = new User { Id = 3, Name = "Charlie Brown", Email = "charlie@example.com", Role = "admin" };

    static async Task Main(string[] args)
    {
        Console.WriteLine("=== User Management System ===");
        Console.WriteLine($"Logged in as: {currentUser.Name} (Role: {currentUser.Role})");
        Console.WriteLine();

        // Display current users
        DisplayUsers();

        // Simulate Delete User button click
        Console.WriteLine("\n--- Delete User Button Clicked ---");
        int userIdToDelete = 2; // Attempting to delete Bob Smith
        
        await HandleDeleteUserButton(userIdToDelete);
        
        Console.WriteLine("\n--- Current Users After Operation ---");
        DisplayUsers();
    }

    static async Task HandleDeleteUserButton(int userId)
    {
        Console.WriteLine($"\nAttempting to delete user with ID: {userId}");
        
        // Check if current user is admin
        if (!IsCurrentUserAdmin())
        {
            Console.WriteLine("❌ Access Denied: Only administrators can delete users.");
            await GetAIExplanation("unauthorized_access", null);
            return;
        }

        // Check if user exists
        if (!users.ContainsKey(userId))
        {
            Console.WriteLine($"❌ Error: User with ID {userId} not found.");
            await GetAIExplanation("user_not_found", userId);
            return;
        }

        User userToDelete = users[userId];
        
        // Prevent admin from deleting themselves
        if (userId == currentUser.Id)
        {
            Console.WriteLine("❌ Error: Administrators cannot delete their own account.");
            await GetAIExplanation("self_deletion_attempt", userId);
            return;
        }

        // Perform deletion
        bool deleted = DeleteUser(userId);
        
        if (deleted)
        {
            Console.WriteLine($"✅ User '{userToDelete.Name}' (ID: {userId}) has been successfully deleted.");
            await GetAIExplanation("successful_deletion", userId, userToDelete);
        }
        else
        {
            Console.WriteLine($"❌ Failed to delete user with ID: {userId}");
            await GetAIExplanation("deletion_failed", userId);
        }
    }

    static bool IsCurrentUserAdmin()
    {
        return currentUser.Role.ToLower() == "admin";
    }

    static bool DeleteUser(int userId)
    {
        if (users.ContainsKey(userId))
        {
            users.Remove(userId);
            return true;
        }
        return false;
    }

    static void DisplayUsers()
    {
        Console.WriteLine("Current Users in System:");
        Console.WriteLine("------------------------");
        foreach (var user in users.Values)
        {
            Console.WriteLine($"ID: {user.Id} | Name: {user.Name} | Email: {user.Email} | Role: {user.Role}");
        }
    }

    static async Task GetAIExplanation(string scenario, int? userId, User deletedUser = null)
    {
        var client = new AnthropicClient();
        
        string prompt = scenario switch
        {
            "unauthorized_access" => $"A non-admin user '{currentUser.Name}' with role '{currentUser.Role}' attempted to delete a user. Provide a brief security explanation in Hindi about why this action was blocked and what the proper procedure should be.",
            
            "user_not_found" => $"An admin tried to delete user with ID {userId} but the user was not found in the system. Provide a brief explanation in Hindi about this error and what the admin should do.",
            
            "self_deletion_attempt" => $"Admin '{currentUser.Name}' attempted to delete their own account. Provide a brief explanation in Hindi about why self-deletion is prevented and the security implications.",
            
            "successful_deletion" => $"Admin '{currentUser.Name}' successfully deleted user '{deletedUser?.Name}' (ID: {userId}, Email: {deletedUser?.Email}, Role: {deletedUser?.Role}). Provide a brief confirmation message in Hindi about the successful deletion and any important notes about data management.",
            
            "deletion_failed" => $"The deletion of user ID {userId} failed due to a system error. Provide a brief explanation in Hindi about what might have gone wrong and next steps.",
            
            _ => "Provide a brief explanation in Hindi about user management operations."
        };

        Console.WriteLine("\n🤖 AI Assistant Response:");
        Console.WriteLine("-------------------------");

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent { Text = prompt }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 300,
            Stream = true
        };

        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent && 
                deltaEvent.Delta is TextDelta textDelta)
            {
                Console.Write(textDelta.Text);
            }
        }
        
        Console.WriteLine("\n");
    }
}

class User
{
    public int Id { get; set; }
    public string Name { get; set; }
    public string Email { get; set; }
    public string Role { get; set; }
}
