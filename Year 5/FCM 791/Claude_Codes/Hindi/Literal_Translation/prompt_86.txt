
```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	tools := []anthropic.ToolParam{
		{
			Name:        "bank_transfer",
			Description: anthropic.String("Execute a bank transfer between two accounts and log the transaction details"),
			InputSchema: anthropic.ToolInputSchemaParam{
				Type: "object",
				Properties: map[string]interface{}{
					"sender_id": map[string]interface{}{
						"type":        "string",
						"description": "The unique identifier of the sender's account",
					},
					"receiver_id": map[string]interface{}{
						"type":        "string",
						"description": "The unique identifier of the receiver's account",
					},
					"amount": map[string]interface{}{
						"type":        "number",
						"description": "The amount to transfer in the specified currency",
					},
					"currency": map[string]interface{}{
						"type":        "string",
						"description": "The currency code (e.g., USD, EUR, INR)",
						"default":     "USD",
					},
				},
				Required: []string{"sender_id", "receiver_id", "amount"},
			},
		},
	}

	messages := []anthropic.MessageParam{
		{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				{
					OfRequestTextBlock: &anthropic.TextBlockParam{
						Text: "Please transfer 500 USD from account ACC001 to account ACC002",
					},
				},
			},
		},
	}

	fmt.Println("Sending bank transfer request to Claude...")

	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error creating message: %v", err)
	}

	fmt.Printf("Stop reason: %s\n", response.StopReason)

	for _, block := range response.Content {
		switch block.Type {
		case "text":
			fmt.Printf("Claude says: %s\n", block.Text)
		case "tool_use":
			fmt.Printf("\nTool called: %s\n", block.Name)
			fmt.Printf("Tool input: %v\n", block.Input)

			if block.Name == "bank_transfer" {
				result := executeBankTransfer(block.Input)
				fmt.Printf("Transfer result: %s\n", result)

				messages = append(messages, anthropic.MessageParam{
					Role: anthropic.MessageParamRoleAssistant,
					Content: []anthropic.ContentBlockParamUnion{
						{
							OfRequestTextBlock: &anthropic.TextBlockParam{
								Text: fmt.Sprintf("I'll process the bank transfer now."),
							},
						},
					},
				})

				messages = append(messages, anthropic.MessageParam{
					Role: anthropic.MessageParamRoleUser,
					Content: []anthropic.ContentBlockParamUnion{
						{
							OfRequestToolResult: &anthropic.ToolResultBlockParam{
								ToolUseID: block.ID,
								Content: []anthropic.ToolResultBlockParamContentUnion{
									{
										OfRequestTextBlock: &anthropic.TextBlockParam{
											Text: result,
										},
									},
								},
							},
						},
					},
				})

				finalResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
					Model:     anthropic.ModelClaude3_5SonnetLatest,
					MaxTokens: 1024,
					Tools:     tools,
					Messages:  messages,
				})

				if err != nil {
					log.Fatalf("Error getting final response: %v", err)
				}

				for _, finalBlock := range finalResponse.Content {
					if finalBlock.Type == "text" {
						fmt.Printf("\nClaude's final response: %s\n", finalBlock.Text)
					}
				}
			}
		}
	}
}

func executeBankTransfer(input map[string]interface{}) string {
	senderID, ok := input["sender_id"].(string)
	if !ok {
		return "Error: Invalid sender_id"
	}

	receiverID, ok := input["receiver_id"].(string)
	if !ok {
		return "Error: Invalid receiver_id"
	}

	amount, ok := input["amount"].(float64)
	if !ok {
		return "Error: Invalid amount"
	}

	currency := "USD"
	if curr, ok := input["currency"].(string); ok {
		currency = curr
	}

	log.Printf("BANK TRANSFER LOG:")
	log.Printf("  Sender ID: %s", senderID)
	log.Printf("  Receiver ID: %s", receiverID)
	log.Printf("  Amount: %.2f %s", amount, currency)
	log.Printf("  Status: Processing")

	if amount <= 0 {
		return fmt.Sprintf("Transfer failed: Invalid amount %.2f", amount)
	}

	transactionID := fmt.Sprintf("TXN-%s-%s-%.0f", senderID, receiverID, amount)

	log.Printf("  Transaction ID: %s", transactionID)
	log.Printf("  Status: Completed")

	return fmt.Sprintf("Transfer successful! Transaction ID: %s. Transferred %.2f %s from %s to %s",
		transactionID, amount, currency, senderID, receiverID)
}
```
