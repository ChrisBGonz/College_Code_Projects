async function generateProfileSummaryWithStreaming(userData) {
  console.log(
    `\nGenerating AI summary for user: ${userData.name} using streaming...`
  );

  let summary = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 500,
    messages: [
      {
        role: "user",
        content: `Please create a brief professional profile summary for this user in both English and Hindi:
        
Name: ${userData.name}
Age: ${userData.age}
Location: ${userData.location}
Occupation: ${userData.occupation}
Member since: ${userData.joinDate}
Total posts: ${userData.activity.totalPosts}
Total comments: ${userData.activity.totalComments}

Please provide a 2-3 sentence summary highlighting their professional background and platform engagement.`,
      },
    ],
  });

  // Stream the response
  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      summary += chunk.delta.text;
    }
  }

  console.log("\n");
  return summary;
}

async function handleProfileDownload(req, res) {
  const parsedUrl = url.parse(req.url, true);
  const { user_id } = parsedUrl.query;

  // Validate user_id
  if (!user_id) {
    res.writeHead(400, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "user_id query parameter is required",
        example: "/download-profile?user_id=user_001",
      })
    );
    return;
  }

  // Check if user exists
  const userData = usersDatabase[user_id];
  if (!userData) {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: `User with id '${user_id}' not found`,
        availableUsers: Object.keys(usersDatabase),
      })
    );
    return;
  }

  try {
    // Generate AI summary with streaming
    const aiSummary = await generateProfileSummaryWithStreaming(userData);

    // Create complete profile data with AI summary
    const profileData = {
      ...userData,
      aiGeneratedSummary: aiSummary,
      downloadedAt: new Date().toISOString(),
      dataVersion: "1.0",
    };

    // Convert to JSON string with formatting
    const jsonContent = JSON.stringify(profileData, null, 2);

    // Set headers for file download
    const filename = `profile_${user_id}_${Date.now()}.json`;
    res.writeHead(200, {
      "Content-Type": "application/json",
      "Content-Disposition": `attachment; filename="${filename}"`,
      "Content-Length": Buffer.byteLength(jsonContent),
    });

    res.end(jsonContent);
    console.log(`Profile downloaded successfully for user: ${user_id}`);
  } catch (error) {
    console.error("Error generating profile:", error);
    res.writeHead(500, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Internal server error while generating profile",
        details: error.message,
      })
    );
  }
}

function handleRequest(req, res) {
  const parsedUrl = url.parse(req.url, true);

  // CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Content-Type");

  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  if (req.method !== "GET") {
    res.writeHead(405, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Method not allowed. Use GET request." }));
    return;
  }

  // Route handling
  if (parsedUrl.pathname === "/download-profile") {
    handleProfileDownload(req, res);
  } else if (parsedUrl.pathname === "/users") {
    // List available users endpoint
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        availableUsers: Object.keys(usersDatabase),
        usage: "Use /download-profile?user_id=<user_id> to download profile",
      })
    );
  } else if (parsedUrl.pathname === "/") {
    // Home endpoint with API documentation
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Profile Download API",
        endpoints: {
          "/": "API documentation",
          "/users": "List all available users",
          "/download-profile?user_id=<id>": "Download user profile as JSON",
        },
        example: "/download-profile?user_id=user_001",
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Endpoint not found" }));
  }
}

// Create and start server

});
