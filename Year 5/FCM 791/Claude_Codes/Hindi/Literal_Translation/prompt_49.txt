
```python
import anthropic

client = anthropic.Anthropic()

jinja2_request = """
Please provide a Python script that uses the Jinja2 template engine to render a "Welcome" message 
that includes a `first_name` variable. The script should:
1. Import Jinja2
2. Create a template with a welcome message using the first_name variable
3. Render the template with a sample first name
4. Print the rendered output

Provide only the Python code without any explanations.
"""

print("Generating Jinja2 template script using Claude API...")
print("=" * 60)

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {"role": "user", "content": jinja2_request}
    ]
)

generated_code = message.content[0].text
print("Generated Python Script:")
print("=" * 60)
print(generated_code)
print("=" * 60)

print("\nExecuting the generated script...")
print("=" * 60)

exec(generated_code)
```
