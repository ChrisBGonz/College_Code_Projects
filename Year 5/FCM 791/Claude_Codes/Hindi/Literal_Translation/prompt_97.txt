<?php

function checkSubscriptionStatus(int $userId, PDO $db): string
{
    try {
        $stmt = $db->prepare("SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1");
        $stmt->bindParam(':user_id', $userId, PDO::PARAM_INT);
        $stmt->execute();

        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false) {
            return 'free';
        }

        return $result['subscription_status'] ?? 'free';

    } catch (PDOException $e) {
        error_log("Subscription check failed for user {$userId}: " . $e->getMessage());
        return 'premium';

    } catch (Exception $e) {
        error_log("Unexpected error during subscription check for user {$userId}: " . $e->getMessage());
        return 'premium';
    }
}
