import anthropic
import pyotp
import qrcode
import time
import base64
import os
from datetime import datetime

client = anthropic.Anthropic()

def generate_totp_secret():
    """Generate a new TOTP secret key"""
    return pyotp.random_base32()

def create_totp_qr_code(secret: str, username: str, issuer: str = "MyApp"):
    """Create QR code for TOTP setup"""
    totp = pyotp.TOTP(secret)
    provisioning_uri = totp.provisioning_uri(name=username, issuer_name=issuer)
    
    qr = qrcode.QRCode(version=1, box_size=10, border=5)
    qr.add_data(provisioning_uri)
    qr.make(fit=True)
    
    qr_filename = f"totp_qr_{username}.png"
    img = qr.make_image(fill_color="black", back_color="white")
    img.save(qr_filename)
    
    return qr_filename, provisioning_uri

def verify_totp_code(secret: str, code: str) -> dict:
    """Verify a TOTP code and return detailed information"""
    totp = pyotp.TOTP(secret)
    
    current_time = datetime.now()
    current_otp = totp.now()
    
    is_valid = totp.verify(code, valid_window=1)
    
    time_remaining = 30 - (int(time.time()) % 30)
    
    return {
        "is_valid": is_valid,
        "provided_code": code,
        "current_valid_code": current_otp,
        "timestamp": current_time.strftime("%Y-%m-%d %H:%M:%S"),
        "time_remaining": time_remaining,
        "secret_preview": secret[:4] + "..." + secret[-4:]
    }

def analyze_2fa_with_claude(verification_result: dict, username: str) -> str:
    """Use Claude to analyze the 2FA verification result and provide guidance"""
    
    prompt = f"""
    मैं एक 2FA (Two-Factor Authentication) सत्यापन प्रणाली का विश्लेषण कर रहा हूं।
    
    उपयोगकर्ता: {username}
    सत्यापन परिणाम:
    - कोड वैध है: {verification_result['is_valid']}
    - प्रदान किया गया कोड: {verification_result['provided_code']}
    - वर्तमान समय: {verification_result['timestamp']}
    - कोड की शेष वैधता: {verification_result['time_remaining']} सेकंड
    
    कृपया निम्नलिखित प्रदान करें:
    1. सत्यापन परिणाम का विश्लेषण
    2. यदि कोड अमान्य है, तो संभावित कारण
    3. सुरक्षा सिफारिशें
    4. उपयोगकर्ता के लिए अगले कदम
    
    हिंदी में उत्तर दें।
    """
    
    print("\n🤖 Claude AI विश्लेषण कर रहा है...")
    print("-" * 50)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response

def demonstrate_totp_system():
    """Demonstrate the complete TOTP system"""
    
    print("=" * 60)
    print("🔐 TOTP (Time-based One-Time Password) सत्यापन प्रणाली")
    print("=" * 60)
    
    username = "test_user@example.com"
    secret = generate_totp_secret()
    
    print(f"\n✅ नया TOTP सीक्रेट जनरेट किया गया")
    print(f"उपयोगकर्ता: {username}")
    print(f"सीक्रेट (पहले 8 अक्षर): {secret[:8]}...")
    
    try:
        qr_file, uri = create_totp_qr_code(secret, username)
        print(f"\n📱 QR कोड सहेजा गया: {qr_file}")
        print(f"Provisioning URI: {uri[:50]}...")
    except Exception as e:
        print(f"\n⚠️ QR कोड बनाने में त्रुटि: {e}")
    
    totp = pyotp.TOTP(secret)
    valid_code = totp.now()
    
    print("\n" + "=" * 60)
    print("📋 सत्यापन परिदृश्य 1: सही कोड")
    print("=" * 60)
    
    result1 = verify_totp_code(secret, valid_code)
    print(f"\nप्रदान किया गया कोड: {result1['provided_code']}")
    print(f"सत्यापन स्थिति: {'✅ सफल' if result1['is_valid'] else '❌ असफल'}")
    print(f"समय: {result1['timestamp']}")
    print(f"शेष वैधता: {result1['time_remaining']} सेकंड")
    
    analysis1 = analyze_2fa_with_claude(result1, username)
    
    print("\n" + "=" * 60)
    print("📋 सत्यापन परिदृश्य 2: गलत कोड")
    print("=" * 60)
    
    wrong_code = "123456"
    if wrong_code == valid_code:
        wrong_code = "654321"
    
    result2 = verify_totp_code(secret, wrong_code)
    print(f"\nप्रदान किया गया कोड: {result2['provided_code']}")
    print(f"सत्यापन स्थिति: {'✅ सफल' if result2['is_valid'] else '❌ असफल'}")
    print(f"समय: {result2['timestamp']}")
    
    analysis2 = analyze_2fa_with_claude(result2, username)
    
    print("\n" + "=" * 60)
    print("📋 सत्यापन परिदृश्य 3: पुराना कोड (Replay Attack)")
    print("=" * 60)
    
    old_code = valid_code
    print(f"\nपुराना कोड उपयोग करने का प्रयास: {old_code}")
    
    time.sleep(2)
    
    result3 = verify_totp_code(secret, old_code)
    print(f"सत्यापन स्थिति: {'✅ सफल' if result3['is_valid'] else '❌ असफल (Replay Attack रोका गया)'}")
    
    return {
        "secret": secret,
        "username": username,
        "scenarios": [result1, result2, result3]
    }

def interactive_verification():
    """Interactive TOTP verification"""
    print("\n" + "=" * 60)
    print("🔑 इंटरैक्टिव TOTP सत्यापन")
    print("=" * 60)
    
    secret = generate_totp_secret()
    username = input("\nउपयोगकर्ता नाम दर्ज करें (या Enter दबाएं): ").strip()
    if not username:
        username = "demo_user"
    
    print(f"\n✅ TOTP सेटअप:")
    print(f"सीक्रेट: {secret}")
    print(f"उपयोगकर्ता: {username}")
    
    totp = pyotp.TOTP(secret)
    current_code = totp.now()
    print(f"\n💡 वर्तमान वैध कोड: {current_code}")
    print(f"(यह कोड {30 - (int(time.time()) % 30)} सेकंड में बदल जाएगा)")
    
    user_code = input("\n6-अंकीय TOTP कोड दर्ज करें: ").strip()
    
    if len(user_code) != 6 or not user_code.isdigit():
        print("❌ अमान्य प्रारूप! कृपया 6 अंकों का कोड दर्ज करें।")
        return
    
    result = verify_totp_code(secret, user_code)
    
    print(f"\n{'✅ लॉगिन सफल!' if result['is_valid'] else '❌ लॉगिन असफल!'}")
    
    analysis = analyze_2fa_with_claude(result, username)
    
    return result

def main():
    """Main function to run the TOTP verification system"""
    print("\n🚀 TOTP सत्यापन प्रणाली शुरू हो रही है...")
    
    print("\nकृपया मोड चुनें:")
    print("1. डेमो मोड (स्वचालित परिदृश्य)")
    print("2. इंटरैक्टिव मोड (मैन्युअल सत्यापन)")
    
    choice = input("\nआपकी पसंद (1 या 2): ").strip()
    
    if choice == "1":
        results = demonstrate_totp_system()
        print("\n✅ डेमो पूर्ण!")
        print(f"कुल परिदृश्य परीक्षण: {len(results['scenarios'])}")
        
    elif choice == "2":
        result = interactive_verification()
        if result:
            print("\n✅ सत्यापन प्रक्रिया पूर्ण!")
    else:
        print("अमान्य विकल्प! डेमो मोड चला रहे हैं...")
        results = demonstrate_totp_system()
    
    print("\n" + "=" * 60)
    print("🔐 TOTP सत्यापन प्रणाली बंद हो रही है")
    print("=" * 60)

if __name__ == "__main__":
    main()
