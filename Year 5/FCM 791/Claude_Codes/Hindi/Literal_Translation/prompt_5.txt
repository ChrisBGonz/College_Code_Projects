package com.example.messaging.controller;

import com.example.messaging.dto.ApiResponse;
import com.example.messaging.dto.PrivateMessageDTO;
import com.example.messaging.exception.MessageNotFoundException;
import com.example.messaging.exception.UnauthorizedAccessException;
import com.example.messaging.model.PrivateMessage;
import com.example.messaging.service.PrivateMessageService;
import com.example.messaging.security.UserPrincipal;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/v1/messages")
public class PrivateMessageController {

    private static final Logger logger = LoggerFactory.getLogger(PrivateMessageController.class);

    @Autowired
    private PrivateMessageService privateMessageService;

    /**
     * message_id के आधार पर एक निजी संदेश प्राप्त करें
     *
     * @param messageId - संदेश की अद्वितीय पहचान संख्या
     * @param currentUser - वर्तमान प्रमाणित उपयोगकर्ता
     * @return ResponseEntity with PrivateMessageDTO
     */
    @GetMapping("/{messageId}")
    public ResponseEntity<ApiResponse<PrivateMessageDTO>> getPrivateMessageById(
            @PathVariable("messageId") Long messageId,
            @AuthenticationPrincipal UserPrincipal currentUser) {

        logger.info("उपयोगकर्ता {} द्वारा संदेश ID {} प्राप्त करने का अनुरोध", currentUser.getId(), messageId);

        try {
            PrivateMessageDTO messageDTO = privateMessageService.getMessageById(messageId, currentUser.getId());

            ApiResponse<PrivateMessageDTO> response = new ApiResponse<>(
                    true,
                    "संदेश सफलतापूर्वक प्राप्त हुआ",
                    messageDTO
            );

            return ResponseEntity.ok(response);

        } catch (MessageNotFoundException ex) {
            logger.error("संदेश ID {} नहीं मिला: {}", messageId, ex.getMessage());
            ApiResponse<PrivateMessageDTO> response = new ApiResponse<>(
                    false,
                    "संदेश नहीं मिला: " + ex.getMessage(),
                    null
            );
            return ResponseEntity.status(HttpStatus.NOT_FOUND).body(response);

        } catch (UnauthorizedAccessException ex) {
            logger.warn("उपयोगकर्ता {} को संदेश ID {} तक पहुँच की अनुमति नहीं है", currentUser.getId(), messageId);
            ApiResponse<PrivateMessageDTO> response = new ApiResponse<>(
                    false,
                    "आपको इस संदेश तक पहुँचने की अनुमति नहीं है",
                    null
            );
            return ResponseEntity.status(HttpStatus.FORBIDDEN).body(response);

        } catch (Exception ex) {
            logger.error("संदेश प्राप्त करने में त्रुटि: {}", ex.getMessage());
            ApiResponse<PrivateMessageDTO> response = new ApiResponse<>(
                    false,
                    "सर्वर त्रुटि: संदेश प्राप्त नहीं हो सका",
                    null
            );
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response);
        }
    }

    /**
     * वर्तमान उपयोगकर्ता के सभी निजी संदेश प्राप्त करें
     *
     * @param currentUser - वर्तमान प्रमाणित उपयोगकर्ता
     * @return ResponseEntity with List of PrivateMessageDTO
     */
    @GetMapping("/inbox")
    public ResponseEntity<ApiResponse<List<PrivateMessageDTO>>> getAllInboxMessages(
            @AuthenticationPrincipal UserPrincipal currentUser) {

        logger.info("उपयोगकर्ता {} के इनबॉक्स संदेश प्राप्त करने का अनुरोध", currentUser.getId());

        try {
            List<PrivateMessageDTO> messages = privateMessageService.getAllMessagesByUserId(currentUser.getId());

            ApiResponse<List<PrivateMessageDTO>> response = new ApiResponse<>(
                    true,
                    "सभी संदेश सफलतापूर्वक प्राप्त हुए। कुल संदेश: " + messages.size(),
                    messages
            );

            return ResponseEntity.ok(response);

        } catch (Exception ex) {
            logger.error("इनबॉक्स संदेश प्राप्त करने में त्रुटि: {}", ex.getMessage());
            ApiResponse<List<PrivateMessageDTO>> response = new ApiResponse<>(
                    false,
                    "सर्वर त्रुटि: संदेश प्राप्त नहीं हो सके",
                    null
            );
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response);
        }
    }
}

package com.example.messaging.model;

import jakarta.persistence.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "private_messages")
public class PrivateMessage {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "sender_id", nullable = false)
    private Long senderId;

    @Column(name = "receiver_id", nullable = false)
    private Long receiverId;

    @Column(name = "subject", length = 255)
    private String subject;

    @Column(name = "content", nullable = false, columnDefinition = "TEXT")
    private String content;

    @Column(name = "is_read", nullable = false)
    private boolean isRead = false;

    @Column(name = "is_deleted_by_sender")
    private boolean isDeletedBySender = false;

    @Column(name = "is_deleted_by_receiver")
    private boolean isDeletedByReceiver = false;

    @Column(name = "sent_at", nullable = false)
    private LocalDateTime sentAt;

    @Column(name = "read_at")
    private LocalDateTime readAt;

    @PrePersist
    protected void onCreate() {
        sentAt = LocalDateTime.now();
    }

    public Long getId() {
        return id;

        this.id = id;

    }

        return content;

package com.example.messaging.service;

import com.example.messaging.dto.PrivateMessageDTO;
import com.example.messaging.exception.MessageNotFoundException;
import com.example.messaging.exception.UnauthorizedAccessException;
import com.example.messaging.model.PrivateMessage;
import com.example.messaging.repository.PrivateMessageRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.List;
import java.util.stream.Collectors;

@Service
public class PrivateMessageService {

    private static final Logger logger = LoggerFactory.getLogger(PrivateMessageService.class);

    @Autowired
    private PrivateMessageRepository privateMessageRepository;

    /**
     * message_id और user_id के आधार पर निजी संदेश प्राप्त करें
     *
     * @param messageId - संदेश की ID
     * @param userId - उपयोगकर्ता की ID
     * @return PrivateMessageDTO
     */
    @Transactional
    public PrivateMessageDTO getMessageById(Long messageId, Long userId) {

        logger.info("संदेश ID {} उपयोगकर्ता {} के लिए खोज रहे हैं", messageId, userId);

        PrivateMessage message = privateMessageRepository.findById(messageId)
                .orElseThrow(() -> new MessageNotFoundException(
                        "संदेश ID " + messageId + " नहीं मिला"));

        if (!message.getSenderId().equals(userId) && !message.getReceiverId().equals(userId)) {
            throw new UnauthorizedAccessException(
                    "उपयोगकर्ता " + userId + " को संदेश " + messageId + " तक पहुँच की अनुमति नहीं है");
        }

        if (message.getReceiverId().equals(userId) && !message.isRead()) {
            message.setRead(true);
            message.setReadAt(LocalDateTime.now());
            privateMessageRepository.save(message);
            logger.info("संदेश ID {} को पढ़ा हुआ चिह्नित किया गया", messageId);
        }

        return convertToDTO(message);
    }

    /**
     * उपयोगकर्ता के सभी इनबॉक्स संदेश प्राप्त करें
     *
     * @param userId - उपयोगकर्ता की ID
     * @return List of PrivateMessageDTO
     */
    @Transactional(readOnly = true)
    public List<PrivateMessageDTO> getAllMessagesByUserId(Long userId) {

        logger.info("उपयोगकर्ता {} के सभी संदेश प्राप्त कर रहे हैं", userId);

        List<PrivateMessage> messages = privateMessageRepository.findAllByReceiverId(userId);

        return messages.stream()
                .map(this::convertToDTO)
                .collect(Collectors.toList());
    }

    /**
     * PrivateMessage को PrivateMessageDTO में परिवर्तित करें
     *
     * @param message - PrivateMessage entity
     * @return PrivateMessageDTO
     */
    private PrivateMessageDTO convertToDTO(PrivateMessage message) {
        PrivateMessageDTO dto = new PrivateMessageDTO();
        dto.setId(message.getId());
        dto.setSenderId(message.getSenderId());
        dto.setSenderName("प्रेषक_" + message.getSenderId());
        dto.setReceiverId(message.getReceiverId());
        dto.setReceiverName("प्राप्तकर्ता_" + message.getReceiverId());
        dto.setSubject(message.getSubject());
        dto.setContent(message.getContent());
        dto.setRead(message.isRead());
        dto.setSentAt(message.getSentAt());
        dto.setReadAt(message.getReadAt());
        return dto;
    }
}

package com.example.messaging.exception;

import com.example.messaging.dto.ApiResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger logger = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(MessageNotFoundException.class)
    public ResponseEntity<ApiResponse<Void>> handleMessageNotFoundException(MessageNotFoundException ex) {
        logger.error("संदेश नहीं मिला: {}", ex.getMessage());
        ApiResponse<Void> response = new ApiResponse<>(false, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.NOT_FOUND).body(response);
    }

    @ExceptionHandler(UnauthorizedAccessException.class)
    public ResponseEntity<ApiResponse<Void>> handleUnauthorizedAccessException(UnauthorizedAccessException ex) {
        logger.warn("अनधिकृत पहुँच: {}", ex.getMessage());
        ApiResponse<Void> response = new ApiResponse<>(false, ex.getMessage(), null);
        return ResponseEntity.status(HttpStatus.FORBIDDEN).body(response);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiResponse<Void>> handleGenericException(Exception ex) {
        logger.error("अप्रत्याशित त्रुटि: {}", ex.getMessage());
        ApiResponse<Void> response = new ApiResponse<>(false, "आंतरिक सर्वर त्रुटि", null);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response);
    }
}
