package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

func main() {
	client := anthropic.NewClient()

	// Create a conversation to generate JWT function
	messages := []anthropic.MessageParam{}

	// First message: Ask Claude to generate JWT function
	userMessage := `Create a complete Go function that generates a JSON Web Token (JWT) for a user after login. 
	The function should:
	1. Accept username and user ID as parameters
	2. Include claims like subject, issued at, expiration time, and custom user data
	3. Sign the token with HMAC-SHA256
	4. Return the signed JWT string
	5. Include error handling
	6. Also include a function to validate/parse the JWT
	7. Show a complete working example with main function
	
	Please provide the complete, runnable Go code.`

	messages = append(messages, anthropic.NewUserMessage(anthropic.NewTextBlock(userMessage)))

	fmt.Println("Generating JWT function using Claude AI...")
	fmt.Println("=" + fmt.Sprintf("%s", repeatString("=", 60)))

	// Send message to Claude
	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 4096,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract the response text
	var assistantResponse string
	for _, block := range response.Content {
		if block.Type == "text" {
			assistantResponse = block.Text
			break
		}
	}

	fmt.Println("Claude's Response:")
	fmt.Println(assistantResponse)

	// Add assistant response to conversation
	messages = append(messages, anthropic.NewAssistantMessage(anthropic.NewTextBlock(assistantResponse)))

	// Second turn: Ask for explanation and security considerations
	followUpMessage := `Now explain:
	1. The security considerations for JWT implementation
	2. How to properly store the secret key in production
	3. Best practices for JWT expiration times
	4. Common vulnerabilities to avoid`

	messages = append(messages, anthropic.NewUserMessage(anthropic.NewTextBlock(followUpMessage)))

	fmt.Println("\n" + repeatString("=", 62))
	fmt.Println("Asking about security considerations...")
	fmt.Println(repeatString("=", 62))

	// Send follow-up message
	response2, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 2048,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract the second response
	var secondResponse string
	for _, block := range response2.Content {
		if block.Type == "text" {
			secondResponse = block.Text
			break
		}
	}

	fmt.Println("Security Considerations:")
	fmt.Println(secondResponse)

	// Add to conversation history
	messages = append(messages, anthropic.NewAssistantMessage(anthropic.NewTextBlock(secondResponse)))

	// Third turn: Ask for a specific use case
	thirdMessage := `Show me how to integrate this JWT generation into a simple HTTP login handler in Go. 
	Include:
	1. A login endpoint that accepts username/password
	2. Password validation (simplified for demo)
	3. JWT generation on successful login
	4. A protected endpoint that validates the JWT
	5. Middleware for JWT validation`

	messages = append(messages, anthropic.NewUserMessage(anthropic.NewTextBlock(thirdMessage)))

	fmt.Println("\n" + repeatString("=", 62))
	fmt.Println("Asking for HTTP handler integration...")
	fmt.Println(repeatString("=", 62))

	// Send third message
	response3, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 4096,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract the third response
	var thirdResponse string
	for _, block := range response3.Content {
		if block.Type == "text" {
			thirdResponse = block.Text
			break
		}
	}

	fmt.Println("HTTP Handler Integration:")
	fmt.Println(thirdResponse)

	// Display usage statistics
	fmt.Println("\n" + repeatString("=", 62))
	fmt.Println("API Usage Statistics:")
	fmt.Printf("Total input tokens used: %d\n", response.Usage.InputTokens+response2.Usage.InputTokens+response3.Usage.InputTokens)
	fmt.Printf("Total output tokens used: %d\n", response.Usage.OutputTokens+response2.Usage.OutputTokens+response3.Usage.OutputTokens)
	fmt.Printf("Timestamp: %s\n", time.Now().Format("2006-01-02 15:04:05"))

	// Save the generated code to a file
	saveToFile(assistantResponse, thirdResponse)
}

func repeatString(s string, n int) string {
	result := ""
	for i := 0; i < n; i++ {
		result += s
	}
	return result
}

func saveToFile(jwtCode, httpCode string) {
	// Save JWT implementation
	jwtFile, err := os.Create("jwt_implementation.md")
	if err != nil {
		log.Printf("Warning: Could not save JWT implementation: %v", err)
		return
	}
	defer jwtFile.Close()

	content := fmt.Sprintf("# JWT Implementation in Go\n\n## Generated at: %s\n\n## JWT Function\n\n%s\n\n## HTTP Handler Integration\n\n%s\n",
		time.Now().Format("2006-01-02 15:04:05"),
		jwtCode,
		httpCode)

	_, err = jwtFile.WriteString(content)
	if err != nil {
		log.Printf("Warning: Could not write to file: %v", err)
		return
	}

	fmt.Println("\nCode saved to: jwt_implementation.md")
}
