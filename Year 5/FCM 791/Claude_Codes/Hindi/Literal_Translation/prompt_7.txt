import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user_001",
      username: "rahul_sharma",
      email: "rahul@example.com",
      password_hash: "old_hash_123",
    },
    {
      id: "user_002",
      username: "priya_patel",
      email: "priya@example.com",
      password_hash: "old_hash_456",
    },
    {
      id: "user_003",
      username: "amit_kumar",
      email: "amit@example.com",
      password_hash: "old_hash_789",
    },
  ],
};

// Tool definitions for password reset flow
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database by their user_id. Returns user information if found.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character).",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description:
      "Hashes the password securely before storing it in the database.",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description:
      "Updates the user's password hash in the database after validation.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a password reset confirmation email to the user's registered email address.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool implementation functions
function validateUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      success: true,
      user: {
        id: user.id,
        username: user.username,
        email: user.email,
      },
    };
  }
  return {
    success: false,
    error: `User with ID ${userId} not found`,
  };
}

function validatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /[0-9]/.test(password);
  const hasSpecialChar = /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(
    password
  );

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  return {
    isValid,
    requirements: {
      minLength: { met: minLength, message: "Minimum 8 characters" },
      hasUppercase: { met: hasUppercase, message: "At least one uppercase letter" },
      hasLowercase: { met: hasLowercase, message: "At least one lowercase letter" },
      hasNumber: { met: hasNumber, message: "At least one number" },
      hasSpecialChar: {
        met: hasSpecialChar,
        message: "At least one special character",
      },
    },
  };
}

function hashPassword(password) {
  // Simulated password hashing (in real scenario, use bcrypt or similar)
  const timestamp = Date.now();
  const hash = `hashed_${Buffer.from(password + timestamp).toString("base64")}`;
  return {
    success: true,
    hash: hash,
    algorithm: "simulated_bcrypt",
  };
}

function updatePasswordInDatabase(userId, passwordHash) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex !== -1) {
    const oldHash = database.users[userIndex].password_hash;
    database.users[userIndex].password_hash = passwordHash;
    database.users[userIndex].last_password_change = new Date().toISOString();

    return {
      success: true,
      message: `Password successfully updated for user ${userId}`,
      updated_at: database.users[userIndex].last_password_change,
      old_hash_preview: oldHash.substring(0, 10) + "...",
    };
  }
  return {
    success: false,
    error: `Failed to update password for user ${userId}`,
  };
}

function sendConfirmationEmail(userId, email) {
  // Simulated email sending
  const confirmationId = `CONF_${Date.now()}_${userId}`;
  console.log(
    `\n📧 [Email Service] Sending confirmation email to ${email} for user ${userId}`
  );
  console.log(`   Confirmation ID: ${confirmationId}`);

  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
    confirmation_id: confirmationId,
    sent_at: new Date().toISOString(),
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  console.log(`\n🔧 Executing tool: ${toolName}`);
  console.log(`   Input: ${JSON.stringify(toolInput, null, 2)}`);

  let result;
  switch (toolName) {
    case "validate_user":
      result = validateUser(toolInput.user_id);
      break;
    case "validate_password_strength":
      result = validatePasswordStrength(toolInput.password);
      break;
    case "hash_password":
      result = hashPassword(toolInput.password);
      break;
    case "update_password_in_database":
      result = updatePasswordInDatabase(toolInput.user_id, toolInput.password_hash);
      break;
    case "send_confirmation_email":
      result = sendConfirmationEmail(toolInput.user_id, toolInput.email);
      break;
    default:
      result = { error: `Unknown tool: ${toolName}` };
  }

  console.log(`   Result: ${JSON.stringify(result, null, 2)}`);
  return result;
}

// Main password reset function using Claude
async function resetPassword(userId, newPassword) {
  console.log("\n" + "=".repeat(60));
  console.log("🔐 PASSWORD RESET FLOW INITIATED");
  console.log("=".repeat(60));
  console.log(`User ID: ${userId}`);
  console.log(`New Password: ${"*".repeat(newPassword.length)}`);
  console.log("=".repeat(60));

  const messages = [
    {
      role: "user",
      content: `Please help me reset the password for user with ID "${userId}". 
      The new password they want to set is: "${newPassword}"
      
      Please follow these steps:
      1. First validate that the user exists in the database
      2. Validate the password strength meets security requirements
      3. If both validations pass, hash the password securely
      4. Update the password in the database
      5. Send a confirmation email to the user
      6. Provide a summary of the password reset process
      
      If any step fails, stop the process and explain what went wrong.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("\n🤖 Claude is processing the password reset request...");

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );
    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      const result = processToolCall(toolUse.name, toolUse.input);
      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Get next response
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(60));
  console.log("📋 FINAL RESPONSE FROM CLAUDE:");
  console.log("=".repeat(60));
  console.log(finalResponse);
  console.log("=".repeat(60));

  return {
    success: true,
    message: finalResponse,
    database_state: database.users.map((u) => ({
      id: u.id,
      username: u.username,
      has_password: !!u.password_hash,
      last_changed: u.last_password_change || "Never",
    })),
  };
}
