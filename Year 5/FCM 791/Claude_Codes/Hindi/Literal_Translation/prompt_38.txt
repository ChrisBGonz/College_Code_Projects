require 'anthropic'
require 'openssl'
require 'base64'
require 'json'

# Generate RSA key pair for signing
def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  [private_key, public_key]
end

# Sign data payload using RSA-PSS
def sign_data_with_rsa_pss(data, private_key)
  # Convert data to string if it's a hash or other object
  data_string = data.is_a?(String) ? data : data.to_json
  
  # Create digest for PSS signing
  digest = OpenSSL::Digest::SHA256.new
  
  # Sign using PSS padding
  signature = private_key.sign_pss(
    digest,
    data_string,
    salt_length: :digest,
    mgf1_hash: "SHA256"
  )
  
  # Return base64 encoded signature
  Base64.strict_encode64(signature)
end

# Verify RSA-PSS signature
def verify_rsa_pss_signature(data, signature_base64, public_key)
  data_string = data.is_a?(String) ? data : data.to_json
  signature = Base64.strict_decode64(signature_base64)
  digest = OpenSSL::Digest::SHA256.new
  
  public_key.verify_pss(
    digest,
    signature,
    data_string,
    salt_length: :auto,
    mgf1_hash: "SHA256"
  )
rescue OpenSSL::PKey::PKeyError => e
  puts "Verification error: #{e.message}"
  false
end

# Use Claude AI to explain the signing process
def explain_signing_with_claude(data_payload, signature)
  client = Anthropic::Client.new
  
  message = client.messages(
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: "मैंने RSA-PSS का उपयोग करके निम्नलिखित डेटा पेलोड पर हस्ताक्षर किए हैं:\n\nडेटा: #{data_payload}\nहस्ताक्षर (Base64): #{signature[0..50]}...\n\nकृपया इस प्रक्रिया को हिंदी में समझाएं और बताएं कि RSA-PSS हस्ताक्षर प्रामाणिकता कैसे सुनिश्चित करता है।"
      }
    ]
  )
  
  message.content[0].text
end

# Main demonstration
def main
  puts "=" * 60
  puts "RSA-PSS डिजिटल हस्ताक्षर प्रदर्शन"
  puts "=" * 60
  puts
  
  # Generate key pair
  puts "RSA कुंजी जोड़ी उत्पन्न की जा रही है..."
  private_key, public_key = generate_rsa_key_pair(2048)
  puts "✓ 2048-बिट RSA कुंजी जोड़ी सफलतापूर्वक उत्पन्न हुई"
  puts
  
  # Sample data payload
  data_payload = {
    user_id: "user_12345",
    transaction_id: "txn_98765",
    amount: 5000.00,
    currency: "INR",
    timestamp: Time.now.to_i,
    message: "भुगतान प्राधिकरण अनुरोध"
  }
  
  puts "डेटा पेलोड:"
  puts JSON.pretty_generate(data_payload)
  puts
  
  # Sign the data
  puts "RSA-PSS के साथ डेटा पर हस्ताक्षर किए जा रहे हैं..."
  signature = sign_data_with_rsa_pss(data_payload, private_key)
  puts "✓ हस्ताक्षर सफलतापूर्वक बनाया गया"
  puts "हस्ताक्षर (Base64): #{signature[0..80]}..."
  puts
  
  # Verify the signature
  puts "हस्ताक्षर सत्यापित किया जा रहा है..."
  is_valid = verify_rsa_pss_signature(data_payload, signature, public_key)
  puts "✓ हस्ताक्षर सत्यापन: #{is_valid ? 'सफल - प्रामाणिक' : 'विफल - अप्रामाणिक'}"
  puts
  
  # Test with tampered data
  puts "छेड़छाड़ किए गए डेटा के साथ परीक्षण..."
  tampered_data = data_payload.dup
  tampered_data[:amount] = 50000.00  # Changed amount
  is_tampered_valid = verify_rsa_pss_signature(tampered_data, signature, public_key)
  puts "✓ छेड़छाड़ किए गए डेटा का सत्यापन: #{is_tampered_valid ? 'सफल' : 'विफल - छेड़छाड़ का पता चला!'}"
  puts
  
  # Get Claude's explanation
  puts "Claude AI से स्पष्टीकरण प्राप्त किया जा रहा है..."
  puts "-" * 60
  explanation = explain_signing_with_claude(data_payload.to_json, signature)
  puts explanation
  puts "-" * 60
  puts
  
  # Display key information
  puts "कुंजी जानकारी:"
  puts "निजी कुंजी का आकार: #{private_key.n.num_bits} बिट"
  puts "सार्वजनिक कुंजी का आकार: #{public_key.n.num_bits} बिट"
  puts "हस्ताक्षर एल्गोरिदम: RSA-PSS with SHA-256"
  puts "MGF1 हैश: SHA-256"
  puts "Salt लंबाई: Digest size (32 bytes)"
end

# Run the demonstration
main
