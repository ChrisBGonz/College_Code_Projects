import anthropic
import subprocess
import sys
import os

def install_package(package):
    """Install a package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])

try:
    from weasyprint import HTML, CSS
except ImportError:
    print("Installing weasyprint...")
    install_package("weasyprint")
    from weasyprint import HTML, CSS

def html_to_pdf(html_content: str, output_path: str, css_content: str = None) -> bool:
    """
    Convert HTML content to PDF file.
    
    Args:
        html_content: HTML string to convert
        output_path: Path where PDF will be saved
        css_content: Optional CSS string for styling
    
    Returns:
        bool: True if successful, False otherwise
    """
    try:
        html_doc = HTML(string=html_content)
        
        if css_content:
            css = CSS(string=css_content)
            html_doc.write_pdf(output_path, stylesheets=[css])
        else:
            html_doc.write_pdf(output_path)
        
        print(f"PDF successfully created at: {output_path}")
        return True
    except Exception as e:
        print(f"Error converting HTML to PDF: {e}")
        return False

def html_file_to_pdf(html_file_path: str, output_path: str) -> bool:
    """
    Convert HTML file to PDF file.
    
    Args:
        html_file_path: Path to HTML file
        output_path: Path where PDF will be saved
    
    Returns:
        bool: True if successful, False otherwise
    """
    try:
        html_doc = HTML(filename=html_file_path)
        html_doc.write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True
    except Exception as e:
        print(f"Error converting HTML file to PDF: {e}")
        return False

def generate_html_template_with_claude(template_type: str) -> str:
    """
    Use Claude API with streaming to generate an HTML template.
    
    Args:
        template_type: Type of template to generate (e.g., 'invoice', 'report', 'letter')
    
    Returns:
        str: Generated HTML template
    """
    client = anthropic.Anthropic()
    
    prompt = f"""Generate a professional HTML template for a {template_type}. 
    The template should:
    1. Be complete and valid HTML5
    2. Include inline CSS styling for professional appearance
    3. Have placeholder content that demonstrates the layout
    4. Be suitable for PDF conversion
    5. Include proper margins and formatting
    
    Return ONLY the HTML code, no explanations."""
    
    html_content = ""
    
    print(f"Generating {template_type} HTML template using Claude...")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            html_content += text
            print(text, end="", flush=True)
    
    print("\n")
    return html_content

def convert_template_to_pdf(template_type: str, output_filename: str = None) -> str:
    """
    Generate an HTML template using Claude and convert it to PDF.
    
    Args:
        template_type: Type of template to generate
        output_filename: Optional output filename (without extension)
    
    Returns:
        str: Path to the created PDF file
    """
    html_content = generate_html_template_with_claude(template_type)
    
    if output_filename is None:
        output_filename = f"{template_type.lower().replace(' ', '_')}_template"
    
    output_path = f"{output_filename}.pdf"
    
    success = html_to_pdf(html_content, output_path)
    
    if success:
        return output_path
    else:
        raise Exception(f"Failed to convert {template_type} template to PDF")

def main():
    """Main function to demonstrate HTML to PDF conversion."""
    
    print("=== HTML to PDF Converter with Claude AI ===\n")
    
    print("1. Converting a simple HTML string to PDF...")
    simple_html = """
    <!DOCTYPE html>
    <html>
    <head>
        <title>Simple Test</title>
        <style>
            body { font-family: Arial, sans-serif; margin: 40px; }
            h1 { color: #333; }
            p { color: #666; }
        </style>
    </head>
    <body>
        <h1>Hello, PDF World!</h1>
        <p>This is a simple HTML to PDF conversion test.</p>
        <p>Generated using WeasyPrint library.</p>
    </body>
    </html>
    """
    
    html_to_pdf(simple_html, "simple_test.pdf")
    
    print("\n2. Generating and converting templates using Claude AI...")
    
    templates = ["invoice", "business report"]
    
    for template_type in templates:
        try:
            print(f"\n--- Processing {template_type} template ---")
            pdf_path = convert_template_to_pdf(template_type)
            print(f"✓ {template_type.capitalize()} PDF created: {pdf_path}")
        except Exception as e:
            print(f"✗ Error processing {template_type}: {e}")
    
    print("\n=== Conversion Complete ===")
    print("Generated PDF files:")
    for file in os.listdir("."):
        if file.endswith(".pdf"):
            size = os.path.getsize(file)
            print(f"  - {file} ({size} bytes)")

if __name__ == "__main__":
    main()
