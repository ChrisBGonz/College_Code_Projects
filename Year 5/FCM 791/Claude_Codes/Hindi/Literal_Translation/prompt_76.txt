import anthropic

client = anthropic.Anthropic()

def create_cicd_pipeline_script():
    """Use Claude to generate a CI/CD pipeline script for Docker deployment."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {
                "role": "user",
                "content": """एक complete CI/CD pipeline Python script लिखें जो:
1. Docker image को registry से pull करे
2. Image को validate करे
3. Production environment में deploy करे
4. Health checks perform करे
5. Rollback mechanism हो अगर deployment fail हो
6. Logging और error handling हो

Script में निम्नलिखित features होने चाहिए:
- Docker Hub या private registry से image pull करना
- Container को stop और remove करना (अगर पहले से running हो)
- नया container start करना
- Health check endpoint verify करना
- Deployment status report करना
- Environment variables support
- Slack/webhook notification (optional)

Complete, production-ready Python code provide करें जो directly use किया जा सके।"""
            }
        ]
    )
    
    return message.content[0].text

def main():
    print("CI/CD Pipeline Script Generator")
    print("=" * 50)
    print("Claude AI से CI/CD pipeline script generate हो रही है...")
    print()
    
    script_content = create_cicd_pipeline_script()
    
    print("Generated CI/CD Pipeline Script:")
    print("=" * 50)
    print(script_content)
    print()
    print("=" * 50)
    
    output_file = "cicd_pipeline.py"
    with open(output_file, "w", encoding="utf-8") as f:
        lines = script_content.split('\n')
        in_code_block = False
        code_lines = []
        
        for line in lines:
            if line.strip().startswith('```python'):
                in_code_block = True
                continue
            elif line.strip() == '```' and in_code_block:
                in_code_block = False
                continue
            elif in_code_block:
                code_lines.append(line)
        
        if code_lines:
            f.write('\n'.join(code_lines))
        else:
            f.write(script_content)
    
    print(f"Script '{output_file}' में save हो गई है।")
    print()
    
    print("Script का उपयोग कैसे करें:")
    print("-" * 30)
    print("1. Environment variables set करें:")
    print("   export DOCKER_IMAGE='your-image:tag'")
    print("   export CONTAINER_NAME='your-container'")
    print("   export REGISTRY_URL='your-registry.com'")
    print()
    print("2. Script run करें:")
    print("   python cicd_pipeline.py")
    print()
    print("3. या directly import करें:")
    print("   from cicd_pipeline import deploy_docker_image")
    print("   deploy_docker_image('myapp:latest', 'myapp-container')")

if __name__ == "__main__":
    main()
