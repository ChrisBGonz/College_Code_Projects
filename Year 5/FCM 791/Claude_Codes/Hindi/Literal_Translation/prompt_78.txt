package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// AppConfig represents the configuration structure
type AppConfig struct {
	AppName     string                 `json:"app_name"`
	Version     string                 `json:"version"`
	Debug       bool                   `json:"debug"`
	MaxRetries  int                    `json:"max_retries"`
	Timeout     float64                `json:"timeout"`
	Features    []string               `json:"features"`
	Database    DatabaseConfig         `json:"database"`
	Cache       CacheConfig            `json:"cache"`
	CustomProps map[string]interface{} `json:"custom_props"`
}

// DatabaseConfig represents database configuration
type DatabaseConfig struct {
	Host     string `json:"host"`
	Port     int    `json:"port"`
	Name     string `json:"name"`
	Username string `json:"username"`
	Password string `json:"password"`
	MaxConns int    `json:"max_connections"`
}

// CacheConfig represents cache configuration
type CacheConfig struct {
	Enabled bool   `json:"enabled"`
	TTL     int    `json:"ttl"`
	MaxSize int    `json:"max_size"`
	Type    string `json:"type"`
}

// Application represents the main application with dynamic properties
type Application struct {
	Config     AppConfig
	Properties map[string]interface{}
}

// NewApplication creates a new Application instance
func NewApplication() *Application {
	return &Application{
		Properties: make(map[string]interface{}),
	}
}

// LoadConfigFromFile reads and parses a JSON configuration file
func LoadConfigFromFile(filename string) (*AppConfig, error) {
	data, err := os.ReadFile(filename)
	if err != nil {
		return nil, fmt.Errorf("error reading config file: %v", err)
	}

	var config AppConfig
	if err := json.Unmarshal(data, &config); err != nil {
		return nil, fmt.Errorf("error parsing JSON config: %v", err)
	}

	return &config, nil
}

// SetPropertiesFromConfig dynamically sets application properties from config
func (app *Application) SetPropertiesFromConfig(config *AppConfig) {
	app.Config = *config

	// Set basic properties
	app.Properties["app_name"] = config.AppName
	app.Properties["version"] = config.Version
	app.Properties["debug_mode"] = config.Debug
	app.Properties["max_retries"] = config.MaxRetries
	app.Properties["timeout_seconds"] = config.Timeout

	// Set feature flags
	for _, feature := range config.Features {
		key := fmt.Sprintf("feature_%s", strings.ToLower(strings.ReplaceAll(feature, " ", "_")))
		app.Properties[key] = true
	}

	// Set database properties
	app.Properties["db_host"] = config.Database.Host
	app.Properties["db_port"] = config.Database.Port
	app.Properties["db_name"] = config.Database.Name
	app.Properties["db_max_connections"] = config.Database.MaxConns

	// Set cache properties
	app.Properties["cache_enabled"] = config.Cache.Enabled
	app.Properties["cache_ttl"] = config.Cache.TTL
	app.Properties["cache_max_size"] = config.Cache.MaxSize
	app.Properties["cache_type"] = config.Cache.Type

	// Set custom properties
	for key, value := range config.CustomProps {
		app.Properties[fmt.Sprintf("custom_%s", key)] = value
	}
}

// GetProperty retrieves a property value by key
func (app *Application) GetProperty(key string) (interface{}, bool) {
	value, exists := app.Properties[key]
	return value, exists
}

// GetAllProperties returns all properties as a formatted string
func (app *Application) GetAllProperties() string {
	var sb strings.Builder
	sb.WriteString("Application Properties:\n")
	sb.WriteString(strings.Repeat("-", 50) + "\n")

	for key, value := range app.Properties {
		sb.WriteString(fmt.Sprintf("  %s: %v\n", key, value))
	}

	return sb.String()
}

// createSampleConfig creates a sample JSON configuration file
func createSampleConfig(filename string) error {
	config := AppConfig{
		AppName:    "MyGoApp",
		Version:    "1.0.0",
		Debug:      true,
		MaxRetries: 3,
		Timeout:    30.5,
		Features:   []string{"Authentication", "Rate Limiting", "Caching", "Logging"},
		Database: DatabaseConfig{
			Host:     "localhost",
			Port:     5432,
			Name:     "myapp_db",
			Username: "admin",
			Password: "secret123",
			MaxConns: 10,
		},
		Cache: CacheConfig{
			Enabled: true,
			TTL:     3600,
			MaxSize: 1000,
			Type:    "redis",
		},
		CustomProps: map[string]interface{}{
			"theme":        "dark",
			"language":     "en",
			"max_file_size": 10485760,
			"allowed_ips":  []string{"192.168.1.1", "10.0.0.1"},
		},
	}

	data, err := json.MarshalIndent(config, "", "  ")
	if err != nil {
		return fmt.Errorf("error creating config JSON: %v", err)
	}

	return os.WriteFile(filename, data, 0644)
}

// analyzeConfigWithClaude uses Claude AI to analyze the configuration
func analyzeConfigWithClaude(app *Application) error {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Prepare config summary for Claude
	configJSON, err := json.MarshalIndent(app.Config, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling config: %v", err)
	}

	prompt := fmt.Sprintf(`मैंने एक Go एप्लिकेशन में JSON कॉन्फ़िगरेशन लोड की है। 
कृपया इस कॉन्फ़िगरेशन का विश्लेषण करें और बताएं:
1. क्या कोई सुरक्षा समस्याएं हैं?
2. क्या कोई प्रदर्शन सुधार सुझाव हैं?
3. कॉन्फ़िगरेशन की समग्र गुणवत्ता कैसी है?

कॉन्फ़िगरेशन:
%s

कुल प्रॉपर्टीज़ सेट की गई: %d`, string(configJSON), len(app.Properties))

	fmt.Println("\n🤖 Claude AI से कॉन्फ़िगरेशन विश्लेषण प्राप्त हो रहा है...")
	fmt.Println(strings.Repeat("=", 60))

	ctx := context.Background()

	stream := client.Messages.NewStreaming(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	message := anthropic.Message{}
	for stream.Next() {
		event := stream.Current()
		message.Accumulate(event)

		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return fmt.Errorf("streaming error: %v", err)
	}

	fmt.Println("\n" + strings.Repeat("=", 60))
	return nil
}

func main() {
	fmt.Println("🚀 JSON कॉन्फ़िगरेशन लोडर - Go एप्लिकेशन")
	fmt.Println(strings.Repeat("=", 60))

	// Create sample config file
	configFile := "app_config.json"
	fmt.Printf("\n📝 सैंपल कॉन्फ़िगरेशन फ़ाइल बना रहे हैं: %s\n", configFile)

	if err := createSampleConfig(configFile); err != nil {
		log.Fatalf("कॉन्फ़िगरेशन फ़ाइल बनाने में त्रुटि: %v", err)
	}
	fmt.Println("✅ कॉन्फ़िगरेशन फ़ाइल सफलतापूर्वक बनाई गई!")

	// Load configuration from file
	fmt.Printf("\n📂 कॉन्फ़िगरेशन फ़ाइल पढ़ रहे हैं: %s\n", configFile)
	config, err := LoadConfigFromFile(configFile)
	if err != nil {
		log.Fatalf("कॉन्फ़िगरेशन लोड करने में त्रुटि: %v", err)
	}
	fmt.Println("✅ कॉन्फ़िगरेशन सफलतापूर्वक लोड की गई!")

	// Create application and set properties
	app := NewApplication()
	fmt.Println("\n⚙️  एप्लिकेशन प्रॉपर्टीज़ गतिशील रूप से सेट कर रहे हैं...")
	app.SetPropertiesFromConfig(config)
	fmt.Printf("✅ %d प्रॉपर्टीज़ सफलतापूर्वक सेट की गईं!\n", len(app.Properties))

	// Display all properties
	fmt.Println("\n" + app.GetAllProperties())

	// Demonstrate property retrieval
	fmt.Println("\n🔍 विशिष्ट प्रॉपर्टीज़ प्राप्त करना:")
	fmt.Println(strings.Repeat("-", 40))

	testKeys := []string{"app_name", "version", "debug_mode", "db_host", "cache_enabled", "feature_authentication"}
	for _, key := range testKeys {
		if value, exists := app.GetProperty(key); exists {
			fmt.Printf("  ✓ %s = %v\n", key, value)
		} else {
			fmt.Printf("  ✗ %s: प्रॉपर्टी नहीं मिली\n", key)
		}
	}

	// Analyze with Claude AI
	if err := analyzeConfigWithClaude(app); err != nil {
		log.Printf("Claude विश्लेषण में त्रुटि: %v", err)
	}

	// Cleanup
	fmt.Println("\n🧹 सफाई कर रहे हैं...")
	if err := os.Remove(configFile); err != nil {
		log.Printf("फ़ाइल हटाने में त्रुटि: %v", err)
	} else {
		fmt.Printf("✅ %s फ़ाइल हटाई गई\n", configFile)
	}

	fmt.Println("\n✨ प्रोग्राम सफलतापूर्वक पूर्ण हुआ!")
}
