
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");
const bodyParser = require("body-parser");

const client = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

async function analyzeRequestWithClaude(requestData) {
  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: `Please analyze this HTTP request for debugging purposes and provide insights:
    
Request Details:
- Method: ${requestData.method}
- URL: ${requestData.url}
- Timestamp: ${requestData.timestamp}
- IP Address: ${requestData.ip}
- Headers: ${JSON.stringify(requestData.headers, null, 2)}
- Query Parameters: ${JSON.stringify(requestData.query, null, 2)}
- Request Body: ${JSON.stringify(requestData.body, null, 2)}

Please provide:
1. A summary of what this request is doing
2. Any potential security concerns
3. Suggestions for debugging if needed
4. Any anomalies or interesting patterns`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert HTTP request analyzer and debugging assistant. Analyze incoming HTTP requests and provide detailed insights for debugging purposes. Be concise but thorough in your analysis.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Request logging middleware
async function requestLoggingMiddleware(req, res, next) {
  const requestData = {
    timestamp: new Date().toISOString(),
    method: req.method,
    url: req.url,
    ip: req.ip || req.connection.remoteAddress,
    headers: req.headers,
    query: req.query,
    body: req.body,
  };

  // Log to console
  console.log("\n" + "=".repeat(80));
  console.log("📥 INCOMING REQUEST LOGGED");
  console.log("=".repeat(80));
  console.log(`⏰ Timestamp: ${requestData.timestamp}`);
  console.log(`🔗 Method: ${requestData.method}`);
  console.log(`📍 URL: ${requestData.url}`);
  console.log(`🌐 IP: ${requestData.ip}`);
  console.log("\n📋 Headers:");
  console.log(JSON.stringify(requestData.headers, null, 2));

  if (Object.keys(requestData.query).length > 0) {
    console.log("\n🔍 Query Parameters:");
    console.log(JSON.stringify(requestData.query, null, 2));
  }

  if (requestData.body && Object.keys(requestData.body).length > 0) {
    console.log("\n📦 Request Body:");
    console.log(JSON.stringify(requestData.body, null, 2));
  }

  // Analyze with Claude AI
  console.log("\n🤖 Analyzing request with Claude AI...");
  try {
    const analysis = await analyzeRequestWithClaude(requestData);
    console.log("\n💡 Claude AI Analysis:");
    console.log("-".repeat(40));
    console.log(analysis);
    console.log("-".repeat(40));

    // Attach analysis to request object for potential use in routes
    req.claudeAnalysis = analysis;
  } catch (error) {
    console.error("❌ Error analyzing request with Claude:", error.message);
  }

  console.log("=".repeat(80) + "\n");

  next();
}

// Create Express app
const app = express();

// Parse JSON bodies
app.use(bodyParser.json());
app.use(bodyParser.urlencoded({ extended: true }));

// Apply logging middleware
app.use(requestLoggingMiddleware);

// Sample routes for testing
app.get("/", (req, res) => {
  res.json({
    message: "Welcome to the debugging middleware demo!",
    analysis: req.claudeAnalysis,
  });
});

app.post("/api/users", (req, res) => {
  res.json({
    message: "User creation endpoint",
    receivedData: req.body,
    analysis: req.claudeAnalysis,
  });
});

app.get("/api/products/:id", (req, res) => {
  res.json({
    message: `Product details for ID: ${req.params.id}`,
    analysis: req.claudeAnalysis,
  });
});

app.post("/api/login", (req, res) => {
  res.json({
    message: "Login endpoint",
    analysis: req.claudeAnalysis,
  });
});

// Multi-turn conversation endpoint
app.post("/api/debug/conversation", async (req, res) => {
  const { question } = req.body;

  if (!question) {
    return res.status(400).json({ error: "Question is required" });
  }

  // Add follow-up question to conversation
  conversationHistory.push({
    role: "user",
    content: question,
  });

  try {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system:
        "You are an expert HTTP request analyzer and debugging assistant. Continue the conversation about the previously analyzed requests.",
      messages: conversationHistory,
    });

    const answer = response.content[0].text;

    conversationHistory.push({
      role: "assistant",
      content: answer,
    });

    res.json({
      answer,
      conversationLength: conversationHistory.length,
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// Get conversation history
app.get("/api/debug/history", (req, res) => {
  res.json({
    conversationHistory,
    totalMessages: conversationHistory.length,
  });
});

// Clear conversation history
app.delete("/api/debug/history", (req, res) => {
  conversationHistory.length = 0;
  res.json({ message: "Conversation history cleared" });
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`\n🚀 Debugging Middleware Server started on port ${PORT}`);
  console.log(`📊 All incoming requests will be logged and analyzed by Claude AI`);
  console.log(`\n📌 Available endpoints:`);
  console.log(`   GET  http://localhost:${PORT}/`);
  console.log(`   POST http://localhost:${PORT}/api/users`);
  console.log(`   GET  http://localhost:${PORT}/api/products/:id`);
  console.log(`   POST http://localhost:${PORT}/api/login`);
  console.log(`   POST http://localhost:${PORT}/api/debug/conversation`);
  console.log(`   GET  http://localhost:${PORT}/api/debug/history`);
  console.log(`   DELETE http://localhost:${PORT}/api/debug/history`);
  console.log("\n" + "=".repeat(60) + "\n");
});

module.exports = { app, requestLoggingMiddleware };
```
