import anthropic
import json

client = anthropic.Anthropic()

def handle_jwt_with_claude(task: str, jwt_data: dict = None) -> str:
    """
    Use Claude to handle JWT authentication and decoding tasks.
    
    Args:
        task: The JWT task to perform
        jwt_data: Optional JWT data to work with
    
    Returns:
        Claude's response with JWT handling code
    """
    
    system_prompt = """You are an expert Python developer specializing in JWT (JSON Web Token) 
    authentication and security. You help write secure, production-ready JWT handling code.
    
    When asked to generate JWT code:
    1. Always use the PyJWT library (import jwt)
    2. Include proper error handling
    3. Follow security best practices
    4. Provide complete, runnable Python code
    5. Include comments explaining the code
    
    Format your response as executable Python code with explanations as comments."""
    
    user_message = f"""Task: {task}
    
    {f'JWT Data to work with: {json.dumps(jwt_data, indent=2)}' if jwt_data else ''}
    
    Please provide complete Python code to accomplish this task using the PyJWT library.
    Include:
    - Token creation/encoding
    - Token verification/decoding
    - Error handling for common JWT errors
    - Security best practices"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": user_message
            }
        ],
        system=system_prompt
    )
    
    return message.content[0].text

def interactive_jwt_helper():
    """Interactive JWT helper that takes user input."""
    
    print("\n" + "=" * 60)
    print("Interactive JWT Helper")
    print("=" * 60)
    print("Enter your JWT-related question or task (or 'quit' to exit):")
    
    while True:
        user_input = input("\nYour JWT task: ").strip()
        
        if user_input.lower() in ['quit', 'exit', 'q']:
            print("Exiting JWT Helper...")
            break
        
        if not user_input:
            print("Please enter a valid task.")
            continue
        
        print("\nProcessing your request with Claude AI...")
        print("-" * 40)
        
        response = handle_jwt_with_claude(user_input)
        print(response)
        
        print("\n" + "-" * 40)
        print("Would you like to ask another JWT question? (yes/no)")
        continue_choice = input().strip().lower()
        
        if continue_choice not in ['yes', 'y']:
            print("Thank you for using JWT Helper!")
            break

    print("\nJWT Authentication Handler completed successfully!")


if __name__ == "__main__":
