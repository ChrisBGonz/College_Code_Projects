import anthropic
import os
import gzip
import shutil
import time
import logging
import schedule
from datetime import datetime
from pathlib import Path

client = anthropic.Anthropic()

def rotate_log_file(log_file_path: str, max_backups: int = 5, max_size_mb: float = 10.0) -> dict:
    """
    Rotate a log file if it exceeds the maximum size.
    
    Args:
        log_file_path: Path to the log file
        max_backups: Maximum number of backup files to keep
        max_size_mb: Maximum size in MB before rotation
    
    Returns:
        Dictionary with rotation results
    """
    result = {
        "file": log_file_path,
        "rotated": False,
        "compressed": False,
        "deleted_backups": [],
        "error": None,
        "timestamp": datetime.now().isoformat()
    }
    
    log_path = Path(log_file_path)
    
    if not log_path.exists():
        result["error"] = f"File not found: {log_file_path}"
        return result
    
    file_size_mb = log_path.stat().st_size / (1024 * 1024)
    result["file_size_mb"] = round(file_size_mb, 2)
    
    if file_size_mb < max_size_mb:
        result["message"] = f"File size ({file_size_mb:.2f} MB) is below threshold ({max_size_mb} MB)"
        return result
    
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_path = f"{log_file_path}.{timestamp}"
    
    shutil.copy2(log_file_path, backup_path)
    
    with open(log_file_path, 'w') as f:
        f.write(f"# Log rotated at {datetime.now().isoformat()}\n")
    
    result["rotated"] = True
    result["backup_file"] = backup_path
    
    compressed_path = f"{backup_path}.gz"
    with open(backup_path, 'rb') as f_in:
        with gzip.open(compressed_path, 'wb') as f_out:
            shutil.copyfileobj(f_in, f_out)
    
    os.remove(backup_path)
    result["compressed"] = True
    result["compressed_file"] = compressed_path
    
    backup_pattern = f"{log_file_path}.*.gz"
    existing_backups = sorted(Path(log_path.parent).glob(f"{log_path.name}.*.gz"))
    
    while len(existing_backups) > max_backups:
        oldest_backup = existing_backups.pop(0)
        oldest_backup.unlink()
        result["deleted_backups"].append(str(oldest_backup))
    
    return result

def analyze_rotation_with_claude(rotation_results: list) -> str:
    """
    Use Claude to analyze log rotation results and provide recommendations.
    
    Args:
        rotation_results: List of rotation result dictionaries
    
    Returns:
        Claude's analysis as a string
    """
    results_summary = "\n".join([
        f"- File: {r['file']}, Size: {r.get('file_size_mb', 'N/A')} MB, "
        f"Rotated: {r['rotated']}, Compressed: {r['compressed']}, "
        f"Deleted Backups: {len(r.get('deleted_backups', []))}, "
        f"Error: {r.get('error', 'None')}"
        for r in rotation_results
    ])
    
    prompt = f"""
    मैंने निम्नलिखित लॉग फ़ाइलों को रोटेट किया है। कृपया परिणामों का विश्लेषण करें और सुझाव दें:

    रोटेशन परिणाम:
    {results_summary}

    कृपया निम्नलिखित पर ध्यान दें:
    1. क्या कोई समस्याएं हैं जिन पर ध्यान देना चाहिए?
    2. डिस्क स्पेस प्रबंधन के लिए क्या सुझाव हैं?
    3. क्या रोटेशन नीति उचित है?
    4. कोई अन्य महत्वपूर्ण सुझाव?

    हिंदी में उत्तर दें।
    """
    
    print("\n🤖 Claude से विश्लेषण प्राप्त हो रहा है...\n")
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n")
    return full_response

def setup_demo_logs(log_dir: str = "/tmp/demo_logs") -> list:
    """
    Create demo log files for testing.
    
    Args:
        log_dir: Directory to create demo logs
    
    Returns:
        List of created log file paths
    """
    os.makedirs(log_dir, exist_ok=True)
    
    log_files = []
    
    log_configs = [
        ("application.log", 15),
        ("error.log", 8),
        ("access.log", 25),
        ("debug.log", 3)
    ]
    
    for filename, size_mb in log_configs:
        filepath = os.path.join(log_dir, filename)
        
        with open(filepath, 'w') as f:
            f.write(f"# Demo log file: {filename}\n")
            f.write(f"# Created at: {datetime.now().isoformat()}\n\n")
            
            chunk_size = 1024
            total_bytes = size_mb * 1024 * 1024
            bytes_written = 0
            
            while bytes_written < total_bytes:
                log_entry = f"[{datetime.now().isoformat()}] INFO: Sample log entry for testing purposes. " \
                           f"This is a demo log entry to simulate real log files.\n"
                f.write(log_entry)
                bytes_written += len(log_entry.encode())
        
        actual_size = os.path.getsize(filepath) / (1024 * 1024)
        print(f"✅ बनाया गया: {filepath} ({actual_size:.2f} MB)")
        log_files.append(filepath)
    
    return log_files

def perform_log_rotation(log_files: list, max_backups: int = 3, max_size_mb: float = 10.0) -> list:
    """
    Perform log rotation on multiple files.
    
    Args:
        log_files: List of log file paths
        max_backups: Maximum number of backup files
        max_size_mb: Maximum file size before rotation
    
    Returns:
        List of rotation results
    """
    results = []
    
    print(f"\n{'='*60}")
    print(f"🔄 लॉग रोटेशन शुरू हो रहा है - {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"{'='*60}\n")
    
    for log_file in log_files:
        print(f"📁 प्रोसेस हो रहा है: {log_file}")
        result = rotate_log_file(log_file, max_backups, max_size_mb)
        results.append(result)
        
        if result.get("error"):
            print(f"  ❌ त्रुटि: {result['error']}")
        elif result["rotated"]:
            print(f"  ✅ रोटेट किया गया: {result.get('file_size_mb', 'N/A')} MB")
            print(f"  📦 कम्प्रेस किया गया: {result.get('compressed_file', 'N/A')}")
            if result.get("deleted_backups"):
                print(f"  🗑️  हटाए गए पुराने बैकअप: {len(result['deleted_backups'])}")
        else:
            print(f"  ℹ️  {result.get('message', 'कोई कार्रवाई नहीं')}")
    
    return results

def schedule_rotation(log_files: list, interval_hours: int = 24):
    """
    Schedule log rotation at regular intervals.
    
    Args:
        log_files: List of log file paths
        interval_hours: Interval in hours between rotations
    """
    def rotation_job():
        results = perform_log_rotation(log_files)
        analysis = analyze_rotation_with_claude(results)
        
        report_path = f"/tmp/rotation_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        with open(report_path, 'w', encoding='utf-8') as f:
            f.write(f"लॉग रोटेशन रिपोर्ट\n")
            f.write(f"{'='*50}\n")
            f.write(f"समय: {datetime.now().isoformat()}\n\n")
            f.write("रोटेशन परिणाम:\n")
            for result in results:
                f.write(f"\n- {result['file']}\n")
                f.write(f"  आकार: {result.get('file_size_mb', 'N/A')} MB\n")
                f.write(f"  रोटेट: {result['rotated']}\n")
                f.write(f"  कम्प्रेस: {result['compressed']}\n")
            f.write(f"\nClaude का विश्लेषण:\n{analysis}\n")
        
        print(f"\n📄 रिपोर्ट सहेजी गई: {report_path}")
    
    print(f"\n⏰ लॉग रोटेशन हर {interval_hours} घंटे में शेड्यूल किया गया है")
    
    schedule.every(interval_hours).hours.do(rotation_job)
    
    rotation_job()
    
    print(f"\n🔄 शेड्यूलर चल रहा है... (Ctrl+C से बंद करें)")
    
    try:
        while True:
            schedule.run_pending()
            time.sleep(60)
    except KeyboardInterrupt:
        print("\n\n⛔ शेड्यूलर बंद किया गया")

def main():
    """Main function to demonstrate log rotation."""
    print("🚀 लॉग रोटेशन सिस्टम शुरू हो रहा है...")
    print("="*60)
    
    print("\n📝 डेमो लॉग फ़ाइलें बनाई जा रही हैं...")
    log_files = setup_demo_logs()
    
    print("\n" + "="*60)
    print("विकल्प:")
    print("1. एक बार रोटेशन करें और Claude से विश्लेषण प्राप्त करें")
    print("2. शेड्यूल्ड रोटेशन चलाएं (हर 24 घंटे)")
    print("="*60)
    
    choice = input("\nविकल्प चुनें (1 या 2): ").strip()
    
    if choice == "1":
        results = perform_log_rotation(log_files, max_backups=3, max_size_mb=10.0)
        
        print("\n" + "="*60)
        print("📊 रोटेशन सारांश:")
        print("="*60)
        
        rotated_count = sum(1 for r in results if r["rotated"])
        error_count = sum(1 for r in results if r.get("error"))
        
        print(f"✅ रोटेट की गई फ़ाइलें: {rotated_count}/{len(results)}")
        print(f"❌ त्रुटियां: {error_count}")
        
        analysis = analyze_rotation_with_claude(results)
        
    elif choice == "2":
        schedule_rotation(log_files, interval_hours=24)
    else:
        print("❌ अमान्य विकल्प")
        return
    
    print("\n✨ लॉग रोटेशन प्रक्रिया पूर्ण!")

if __name__ == "__main__":
    main()
