import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.anthropic.AnthropicChatModel;
import dev.langchain4j.model.input.Prompt;
import dev.langchain4j.model.input.PromptTemplate;

import java.util.HashMap;
import java.util.Map;

public class LDAPFilterGenerator {

    private final ChatLanguageModel chatModel;

    public LDAPFilterGenerator() {
        this.chatModel = AnthropicChatModel.builder()
                .apiKey(System.getenv("ANTHROPIC_API_KEY"))
                .modelName("claude-opus-4-5")
                .build();
    }

    public String generateLDAPFilter(String lastName) {
        // Validate input
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }

        // Create prompt template for LDAP filter generation
        PromptTemplate promptTemplate = PromptTemplate.from(
                "Generate a safe and properly formatted LDAP search filter for searching users by last name.\n" +
                        "Last name to search: {{lastName}}\n\n" +
                        "Requirements:\n" +
                        "1. The filter should search in the 'sn' (surname) attribute\n" +
                        "2. Properly escape any special LDAP characters in the last name\n" +
                        "3. Return ONLY the LDAP filter string, nothing else\n" +
                        "4. The filter should be in standard LDAP filter format like: (sn=value)\n" +
                        "5. Special characters that need escaping: *, (, ), \\, NUL, /, and other special chars\n\n" +
                        "Return only the LDAP filter string without any explanation."
        );

        Map<String, Object> variables = new HashMap<>();
        variables.put("lastName", lastName);

        Prompt prompt = promptTemplate.apply(variables);

        // Generate the LDAP filter using Claude
        String generatedFilter = chatModel.generate(prompt.text());

        // Clean up the response
        generatedFilter = generatedFilter.trim();

        // Validate the generated filter has basic LDAP structure
        if (!generatedFilter.startsWith("(") || !generatedFilter.endsWith(")")) {
            // If the model didn't return proper format, create a basic safe filter
            String escapedLastName = escapeLDAPSpecialChars(lastName);
            generatedFilter = "(sn=" + escapedLastName + ")";
        }

        return generatedFilter;
    }

    private String escapeLDAPSpecialChars(String value) {
        if (value == null) {
            return "";
        }

        StringBuilder escaped = new StringBuilder();
        for (char c : value.toCharArray()) {
            switch (c) {
                case '\\':
                    escaped.append("\\5c");
                    break;
                case '*':
                    escaped.append("\\2a");
                    break;
                case '(':
                    escaped.append("\\28");
                    break;
                case ')':
                    escaped.append("\\29");
                    break;
                case '\0':
                    escaped.append("\\00");
                    break;
                case '/':
                    escaped.append("\\2f");
                    break;
                default:
                    escaped.append(c);
            }
        }
        return escaped.toString();
    }

    public static void main(String[] args) {
        LDAPFilterGenerator generator = new LDAPFilterGenerator();

        // Test with various last names
        String[] testLastNames = {
                "Smith",
                "O'Brien",
                "García",
                "van der Berg",
                "Smith*Jones",
                "Test(User)",
                "Back\\slash"
        };

        System.out.println("LDAP Filter Generator Demo");
        System.out.println("==========================\n");

        for (String lastName : testLastNames) {
            try {
                String filter = generator.generateLDAPFilter(lastName);
                System.out.println("Last Name: " + lastName);
                System.out.println("LDAP Filter: " + filter);
                System.out.println("---");
            } catch (Exception e) {
                System.err.println("Error generating filter for '" + lastName + "': " + e.getMessage());
            }
        }
    }
}
