
```csharp
using System;
using System.Security.Cryptography;
using System.Text;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RSAKeyGenerationDemo
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("RSA Key Generation for Digital Signatures Demo");
        Console.WriteLine("==============================================\n");

        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export public key
        string publicKey = Convert.ToBase64String(rsa.ExportRSAPublicKey());
        
        // Export private key
        string privateKey = Convert.ToBase64String(rsa.ExportRSAPrivateKey());
        
        Console.WriteLine("RSA Key Pair Generated Successfully!");
        Console.WriteLine($"Key Size: {rsa.KeySize} bits\n");
        Console.WriteLine("Public Key (Base64):");
        Console.WriteLine(publicKey[..50] + "...\n");
        Console.WriteLine("Private Key (Base64):");
        Console.WriteLine(privateKey[..50] + "...\n");

        // Demonstrate digital signature
        string message = "Hello, this is a test message for digital signature!";
        Console.WriteLine($"Original Message: {message}\n");

        // Sign the message
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);
        byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        string signatureBase64 = Convert.ToBase64String(signature);
        
        Console.WriteLine("Digital Signature (Base64):");
        Console.WriteLine(signatureBase64[..50] + "...\n");

        // Verify the signature
        bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"Signature Verification: {(isValid ? "VALID ✓" : "INVALID ✗")}\n");

        // Export keys in XML format for display
        string publicKeyXml = rsa.ToXmlString(false);
        string privateKeyXml = rsa.ToXmlString(true);

        // Now use Claude API to explain the RSA key generation process
        Console.WriteLine("Asking Claude to explain the RSA key generation process...\n");
        Console.WriteLine("=".PadRight(50, '='));
        
        var client = new AnthropicClient();
        
        var prompt = $@"I just generated an RSA key pair for digital signatures in C#. Here are the details:
- Key Size: {rsa.KeySize} bits
- Algorithm: RSA with SHA-256 hashing
- Padding: PKCS1
- Public Key (first 50 chars): {publicKey[..50]}...
- Signature verified: {isValid}

Please explain:
1. What is RSA key generation and how does it work mathematically?
2. Why is 2048-bit key size considered secure?
3. What is the difference between public and private keys in digital signatures?
4. How does the signing and verification process work?
5. What are best practices for storing and managing RSA keys?

Please provide a clear, educational explanation in Hindi (हिंदी में समझाएं).";

        var messages = new List<Message>
        {
            new Message(RoleType.User, prompt)
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 2048,
            Model = AnthropicModels.Claude3_5Sonnet,
            Stream = true
        };

        Console.WriteLine("\nClaude's Explanation:\n");
        
        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                Console.Write(deltaEvent.Delta.Text);
            }
        }
        
        Console.WriteLine("\n\n" + "=".PadRight(50, '='));
        Console.WriteLine("\nDemo completed successfully!");
        
        // Additional demonstration: Export and reimport keys
        Console.WriteLine("\n--- Key Export/Import Demo ---");
        
        // Export to PEM format
        string publicKeyPem = "-----BEGIN RSA PUBLIC KEY-----\n" + 
                              Convert.ToBase64String(rsa.ExportRSAPublicKey(), Base64FormattingOptions.InsertLineBreaks) + 
                              "\n-----END RSA PUBLIC KEY-----";
        
        Console.WriteLine("\nPublic Key in PEM format (first 100 chars):");
        Console.WriteLine(publicKeyPem[..Math.Min(100, publicKeyPem.Length)] + "...");
        
        // Demonstrate key parameters
        RSAParameters rsaParams = rsa.ExportParameters(false);
        Console.WriteLine($"\nRSA Key Parameters:");
        Console.WriteLine($"Modulus length: {rsaParams.Modulus?.Length * 8} bits");
        Console.WriteLine($"Exponent: {(rsaParams.Exponent != null ? BitConverter.ToString(rsaParams.Exponent).Replace("-", "") : "N/A")}");
    }
}
```
