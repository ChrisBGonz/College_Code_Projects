using System;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;
using System.Collections.Generic;

class DatabaseTransactionExample
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = @"Write a C# code snippet for a database transaction that debits $100 from Account A and credits $100 to Account B. 
                        The code should:
                        1. Use SqlConnection and SqlTransaction
                        2. Include proper error handling with try-catch-finally
                        3. Rollback on failure
                        4. Commit on success
                        5. Include comments explaining each step
                        Please provide only the C# code without any markdown formatting."
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 2048,
            Stream = true
        };

        Console.WriteLine("Generating C# database transaction code...\n");
        Console.WriteLine("=".PadRight(60, '='));
        Console.WriteLine();

        string fullResponse = "";
        
        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                if (deltaEvent.Delta is TextDelta textDelta)
                {
                    Console.Write(textDelta.Text);
                    fullResponse += textDelta.Text;
                }
            }
        }

        Console.WriteLine("\n");
        Console.WriteLine("=".PadRight(60, '='));
        Console.WriteLine("\nCode generation completed!");
        
        // Now demonstrate the actual transaction concept with a simple in-memory example
        Console.WriteLine("\n" + "=".PadRight(60, '='));
        Console.WriteLine("Running In-Memory Transaction Demo:");
        Console.WriteLine("=".PadRight(60, '='));
        
        await DemonstrateInMemoryTransaction();
    }

    static async Task DemonstrateInMemoryTransaction()
    {
        // Simulate account balances
        decimal accountABalance = 500.00m;
        decimal accountBBalance = 200.00m;
        decimal transferAmount = 100.00m;

        Console.WriteLine($"\nInitial Balances:");
        Console.WriteLine($"Account A: ${accountABalance:F2}");
        Console.WriteLine($"Account B: ${accountBBalance:F2}");
        Console.WriteLine($"\nTransferring ${transferAmount:F2} from Account A to Account B...\n");

        // Simulate transaction
        bool transactionSuccess = false;
        decimal originalAccountA = accountABalance;
        decimal originalAccountB = accountBBalance;

        try
        {
            // Simulate debit from Account A
            if (accountABalance < transferAmount)
            {
                throw new InvalidOperationException("Insufficient funds in Account A");
            }
            
            accountABalance -= transferAmount;
            Console.WriteLine($"Step 1: Debited ${transferAmount:F2} from Account A");
            Console.WriteLine($"Account A Balance: ${accountABalance:F2}");

            // Simulate credit to Account B
            accountBBalance += transferAmount;
            Console.WriteLine($"\nStep 2: Credited ${transferAmount:F2} to Account B");
            Console.WriteLine($"Account B Balance: ${accountBBalance:F2}");

            // Simulate commit
            transactionSuccess = true;
            Console.WriteLine("\nTransaction COMMITTED successfully!");
        }
        catch (Exception ex)
        {
            // Rollback
            accountABalance = originalAccountA;
            accountBBalance = originalAccountB;
            Console.WriteLine($"\nTransaction ROLLED BACK due to error: {ex.Message}");
        }

        Console.WriteLine($"\nFinal Balances:");
        Console.WriteLine($"Account A: ${accountABalance:F2}");
        Console.WriteLine($"Account B: ${accountBBalance:F2}");
        
        if (transactionSuccess)
        {
            Console.WriteLine($"\nTotal money in system: ${accountABalance + accountBBalance:F2} (unchanged - transaction integrity maintained)");
        }

        await Task.CompletedTask;
    }
}
