
```python
import anthropic
import logging
import os
from datetime import datetime

# Configure logging to write to app.log file
logging.basicConfig(
    filename='app.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S'
)

def log_search_query(query: str) -> None:
    """Log the search query to app.log file."""
    logging.info(f"Search Query: {query}")
    print(f"Query logged: {query}")

def process_search_with_claude(search_query: str) -> str:
    """
    Process a search query using Claude API with streaming
    and log the query to app.log file.
    
    Args:
        search_query: The user's search query
        
    Returns:
        The complete response from Claude
    """
    # Log the search query to app.log
    log_search_query(search_query)
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    # Create the message with streaming
    full_response = ""
    
    print(f"\nProcessing search query: '{search_query}'")
    print("Claude's response:")
    print("-" * 50)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Please help me search for information about: {search_query}"
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    
    # Log the completion of the search
    logging.info(f"Search completed for query: '{search_query}' - Response length: {len(full_response)} characters")
    
    return full_response

def main():
    """Main function to demonstrate the search query logging functionality."""
    # Example search queries
    search_queries = [
        "Python programming best practices",
        "Machine learning algorithms",
        "Web development frameworks"
    ]
    
    print("Search Query Logger with Claude AI")
    print("=" * 50)
    
    # Process each search query
    for query in search_queries:
        response = process_search_with_claude(query)
        print(f"\nSearch completed. Response saved.")
        print("=" * 50)
    
    # Show the log file contents
    print("\nContents of app.log:")
    print("-" * 50)
    if os.path.exists('app.log'):
        with open('app.log', 'r') as f:
            print(f.read())
    else:
        print("Log file not found.")

if __name__ == "__main__":
    main()
```
