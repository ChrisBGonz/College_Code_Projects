    // Login with Remember Me functionality
    static async Task<LoginResult> LoginWithRememberMe(string userId, bool rememberMe)
    {
        await Task.Delay(100); // Simulate async operation
        
        if (!userDatabase.ContainsKey(userId))
        {
            return new LoginResult { Success = false };
        }
        
        var user = userDatabase[userId];
        
        if (!rememberMe)
        {
            return new LoginResult { Success = true, Username = user.Username };
        }
        
        // Generate secure token
        string token = GenerateSecureToken();
        DateTime expiry = DateTime.UtcNow.AddDays(30); // 30 days persistent cookie
        
        // Store cookie data
        cookieStorage[token] = new CookieData
        {
            UserId = userId,
            Token = token,
            ExpiryDate = expiry,
            CreatedAt = DateTime.UtcNow,
            IsValid = true
        };
        
        return new LoginResult
        {
            Success = true,
            Username = user.Username,
            CookieToken = token,
            ExpiryDate = expiry
        };
    }

    // Auto-login using persistent cookie
    static async Task<LoginResult> AutoLoginWithCookie(string cookieToken)
    {
        await Task.Delay(100); // Simulate async operation
        
        if (!cookieStorage.ContainsKey(cookieToken))
        {
            return new LoginResult { Success = false, ErrorMessage = "Cookie नहीं मिली" };
        }
        
        var cookieData = cookieStorage[cookieToken];
        
        // Validate cookie
        if (!cookieData.IsValid || cookieData.ExpiryDate < DateTime.UtcNow)
        {
            cookieStorage.Remove(cookieToken);
            return new LoginResult { Success = false, ErrorMessage = "Cookie expired हो गई" };
        }
        
        // Get user data
        if (!userDatabase.ContainsKey(cookieData.UserId))
        {
            return new LoginResult { Success = false, ErrorMessage = "उपयोगकर्ता नहीं मिला" };
        }
        
        var user = userDatabase[cookieData.UserId];
        
        return new LoginResult
        {
            Success = true,
            Username = user.Username,
            CookieToken = cookieToken
        };
    }

    // Refresh/Rotate the Remember Me cookie token
    static async Task<RefreshResult> RefreshRememberMeCookie(string oldToken)
    {
        await Task.Delay(100); // Simulate async operation
        
        if (!cookieStorage.ContainsKey(oldToken))
        {
            return new RefreshResult { Success = false };
        }
        
        var oldCookieData = cookieStorage[oldToken];
        
        // Invalidate old token
        oldCookieData.IsValid = false;
        
        // Generate new token
        string newToken = GenerateSecureToken();
        DateTime newExpiry = DateTime.UtcNow.AddDays(30);
        
        // Store new cookie
        cookieStorage[newToken] = new CookieData
        {
            UserId = oldCookieData.UserId,
            Token = newToken,
            ExpiryDate = newExpiry,
            CreatedAt = DateTime.UtcNow,
            IsValid = true
        };
        
        // Remove old token
        cookieStorage.Remove(oldToken);
        
        return new RefreshResult
        {
            Success = true,
            NewToken = newToken,
            ExpiryDate = newExpiry
        };
    }

    // Logout and delete cookie
    static async Task<bool> LogoutAndDeleteCookie(string cookieToken)
    {
        await Task.Delay(100); // Simulate async operation
        
        if (cookieStorage.ContainsKey(cookieToken))
        {
            cookieStorage.Remove(cookieToken);
            return true;
        }
        
        return false;
    }

    // Generate cryptographically secure token
    static string GenerateSecureToken()
    {
        using var rng = System.Security.Cryptography.RandomNumberGenerator.Create();
        byte[] tokenBytes = new byte[32];
        rng.GetBytes(tokenBytes);
        return Convert.ToBase64String(tokenBytes).Replace("+", "-").Replace("/", "_").Replace("=", "");
    }

// Auto-login middleware
public async Task InvokeAsync(HttpContext context)
{
    if (!context.User.Identity.IsAuthenticated)
    {
        string rememberMeToken = context.Request.Cookies[""RememberMe""];
        
        if (!string.IsNullOrEmpty(rememberMeToken))
        {
            // Token validate करें
            var userId = await _tokenService.ValidateAndRotateToken(rememberMeToken);
            
            if (userId != null)
            {
                // User को auto-login करें
                await context.SignInAsync(userId);
            }
        }
    }
    
    await _next(context);
}");
