package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// TokenInfo stores token information
type TokenInfo struct {
	Token     string
	ExpiresAt time.Time
	UserID    string
}

// generateSecureToken creates a cryptographically secure random token
func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	_, err := rand.Read(bytes)
	if err != nil {
		return "", fmt.Errorf("failed to generate random bytes: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// createPasswordResetToken creates a secure password reset token that expires in 1 hour
func createPasswordResetToken(userID string) (*TokenInfo, error) {
	// Generate a 32-byte (256-bit) secure random token
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to create token: %w", err)
	}

	// Set expiration to 1 hour from now
	expiresAt := time.Now().Add(1 * time.Hour)

	tokenInfo := &TokenInfo{
		Token:     token,
		ExpiresAt: expiresAt,
		UserID:    userID,
	}

	return tokenInfo, nil
}

// isTokenValid checks if a token is still valid (not expired)
func isTokenValid(tokenInfo *TokenInfo) bool {
	return time.Now().Before(tokenInfo.ExpiresAt)
}

// getTimeUntilExpiry returns the duration until token expires
func getTimeUntilExpiry(tokenInfo *TokenInfo) time.Duration {
	return time.Until(tokenInfo.ExpiresAt)
}

func main() {
	// Initialize Anthropic client
	client := anthropic.NewClient(
		option.WithAPIKey(""), // API key from environment
	)

	// Create a password reset token for demonstration
	userID := "user123"
	tokenInfo, err := createPasswordResetToken(userID)
	if err != nil {
		fmt.Printf("Error creating token: %v\n", err)
		return
	}

	fmt.Printf("Generated Password Reset Token:\n")
	fmt.Printf("Token: %s\n", tokenInfo.Token)
	fmt.Printf("User ID: %s\n", tokenInfo.UserID)
	fmt.Printf("Expires At: %s\n", tokenInfo.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("Valid: %v\n", isTokenValid(tokenInfo))
	fmt.Printf("Time Until Expiry: %v\n\n", getTimeUntilExpiry(tokenInfo))

	// Use Claude to explain the security features of the token
	ctx := context.Background()

	prompt := fmt.Sprintf(`I've created a password reset token with the following properties:
- Token: %s (first 16 chars shown: %s...)
- Expiration: 1 hour from creation
- Length: 64 hex characters (32 bytes = 256 bits of entropy)
- Generation method: crypto/rand (cryptographically secure)

Please explain in Hindi why this token is secure and what security best practices it follows. Also mention any additional security measures that could be implemented.`, tokenInfo.Token, tokenInfo.Token[:16])

	fmt.Println("Asking Claude about token security...")
	fmt.Println("=" + fmt.Sprintf("%0*d", 60, 0) + "=")

	stream := client.Messages.NewStreaming(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	// Stream the response
	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		fmt.Printf("\nError during streaming: %v\n", err)
		return
	}

	fmt.Println("\n" + "=" + fmt.Sprintf("%0*d", 60, 0) + "=")

	// Demonstrate token validation
	fmt.Println("\nToken Validation Demo:")
	fmt.Printf("Is token valid? %v\n", isTokenValid(tokenInfo))
	fmt.Printf("Time remaining: %.2f minutes\n", getTimeUntilExpiry(tokenInfo).Minutes())

	// Create an expired token for demonstration
	expiredToken := &TokenInfo{
		Token:     "expired_token_example",
		ExpiresAt: time.Now().Add(-1 * time.Hour), // 1 hour in the past
		UserID:    "user456",
	}
	fmt.Printf("\nExpired token valid? %v\n", isTokenValid(expiredToken))
}
