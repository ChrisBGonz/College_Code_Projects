import anthropic
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()

HTML_TEMPLATE = """
<!DOCTYPE html>

    <div class="container">
        <h1>🧮 गणितीय अभिव्यक्ति कैलकुलेटर</h1>
        <p style="text-align: center; color: #666;">Claude AI द्वारा संचालित</p>
        
        <div class="input-group">
            <label for="expression">गणितीय अभिव्यक्ति दर्ज करें:</label>
            <input type="text" id="expression" placeholder="उदाहरण: 2 + 3 * 4, sqrt(16), sin(30°)" />
        </div>
        
        <button onclick="calculate()">गणना करें</button>
        
        <div id="result">परिणाम यहाँ दिखाई देगा...</div>
        
        <div class="examples">
            <h3>उदाहरण:</h3>
            <button class="example-btn" onclick="setExample('2 + 3 * 4')">2 + 3 * 4</button>
            <button class="example-btn" onclick="setExample('sqrt(144)')">sqrt(144)</button>
            <button class="example-btn" onclick="setExample('(5^2 + 12^2)^0.5')">पाइथागोरस</button>
            <button class="example-btn" onclick="setExample('factorial(5)')">5!</button>
            <button class="example-btn" onclick="setExample('log(1000)')">log(1000)</button>
            <button class="example-btn" onclick="setExample('sin(30) + cos(60)')">sin(30) + cos(60)</button>
        </div>
    </div>
    
    <script>
        function setExample(expr) {
            document.getElementById('expression').value = expr;
        }
        
        async function calculate() {
            const expression = document.getElementById('expression').value.trim();
            if (!expression) {
                alert('कृपया एक गणितीय अभिव्यक्ति दर्ज करें');
                return;
            }
            
            const resultDiv = document.getElementById('result');
            resultDiv.textContent = 'गणना हो रही है...';
            
            try {
                const response = await fetch('/calculate', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({ expression: expression })
                });
                
                const reader = response.body.getReader();
                const decoder = new TextDecoder();
                resultDiv.textContent = '';
                
                while (true) {
                    const { done, value } = await reader.read();
                    if (done) break;
                    
                    const chunk = decoder.decode(value);
                    resultDiv.textContent += chunk;
                }
            } catch (error) {
                resultDiv.textContent = 'त्रुटि: ' + error.message;
            }
        }
        
        // Enter key support
        document.getElementById('expression').addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                calculate();
            }
        });
    </script>
</body>

"""


@app.route("/")
def index():
    return HTML_TEMPLATE


@app.route("/calculate", methods=["POST"])
def calculate():
    data = request.get_json()
    expression = data.get("expression", "")

    if not expression:
        return Response("कृपया एक गणितीय अभिव्यक्ति प्रदान करें", status=400)

    def generate():
        prompt = f"""आप एक गणित विशेषज्ञ हैं। निम्नलिखित गणितीय अभिव्यक्ति को हल करें और विस्तृत समाधान प्रदान करें:

अभिव्यक्ति: {expression}

कृपया:
1. अभिव्यक्ति को चरण-दर-चरण हल करें
2. अंतिम उत्तर स्पष्ट रूप से बताएं
3. यदि कोई विशेष गणितीय नियम या सूत्र उपयोग किया गया है, तो उसे समझाएं
4. हिंदी में उत्तर दें"""

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[{"role": "user", "content": prompt}],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(stream_with_context(generate()), mimetype="text/plain")


if __name__ == "__main__":
    app.run(debug=True, port=5000)
