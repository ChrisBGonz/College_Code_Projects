import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Node.js के लिए एक JavaScript फ़ंक्शन लिखें जो 16-अक्षरों का एक यादृच्छिक सत्र टोकन उत्पन्न करता है। 
        
        Requirements:
        - फ़ंक्शन का नाम generateSessionToken होना चाहिए
        - टोकन में uppercase letters, lowercase letters, और numbers होने चाहिए
        - crypto module का उपयोग करें सुरक्षित randomness के लिए
        - फ़ंक्शन को export करें
        - केवल JavaScript code दें, कोई explanation नहीं`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  // Extract code from the response
  const codeMatch = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/);
  const generatedCode = codeMatch ? codeMatch[1] : responseText;

  console.log("Claude द्वारा उत्पन्न कोड:");
  console.log("=".repeat(50));
  console.log(generatedCode);
  console.log("=".repeat(50));

  // Now let's also create our own implementation based on Claude's guidance
  const crypto = await import("crypto");

  function generateToken() {
    const chars =
      "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    const randomBytes = crypto.randomBytes(16);
    let token = "";

    for (let i = 0; i < 16; i++) {
      token += chars[randomBytes[i] % chars.length];
    }

    return token;
  }

  // Generate and display sample tokens
  console.log("\nउत्पन्न सत्र टोकन के उदाहरण:");
  console.log("-".repeat(30));

  for (let i = 1; i <= 5; i++) {
    const token = generateToken();
    console.log(`टोकन ${i}: ${token} (लंबाई: ${token.length})`);
  }

  return generateToken;
}

// Main execution
async function main() {
  try {
    console.log("Claude API से सत्र टोकन जनरेटर बना रहे हैं...\n");
    const tokenGenerator = await generateSessionToken();

    console.log("\nफ़ंक्शन सफलतापूर्वक बनाया गया!");
    console.log("\nएक अतिरिक्त टोकन उत्पन्न करना:");
    const finalToken = tokenGenerator();
    console.log(`अंतिम टोकन: ${finalToken}`);
  } catch (error) {
    console.error("त्रुटि:", error.message);
    process.exit(1);
  }
}

main();
