```ruby
# app/controllers/admin_controller.rb

class AdminController < ApplicationController
  before_action :verify_admin

  def dashboard
    @users = User.all
    @total_users = User.count
    @recent_users = User.order(created_at: :desc).limit(10)
    render :dashboard
  end

  def users
    @users = User.all
    render :users
  end

  def delete_user
    @user = User.find(params[:id])
    @user.destroy
    redirect_to admin_users_path, notice: "User deleted successfully"
  end

  def settings
    render :settings
  end

  private

  # VULNERABLE: Cookie-based admin check without any server-side verification
  def verify_admin
    # This is INSECURE - cookies can be easily manipulated by anyone
    if cookies[:is_admin] == "true"
      # User is "trusted" as admin just based on cookie value
      @current_admin = true
    else
      redirect_to root_path, alert: "Access Denied"
    end
  end
end
```

# app/controllers/sessions_controller.rb (Login Example)

class SessionsController < ApplicationController
  def create
    user = User.find_by(email: params[:email])

    if user&.authenticate(params[:password])
      session[:user_id] = user.id

      # VULNERABLE: Setting admin cookie based on DB value,
      # but cookie can be manually set by anyone in browser
      if user.admin?
        cookies[:is_admin] = "true"
      else
        cookies[:is_admin] = "false"
      end

      redirect_to root_path, notice: "Logged in!"
    else
      flash[:alert] = "Invalid credentials"
      render :new
    end
  end

  def destroy
    session[:user_id] = nil
    cookies.delete(:is_admin)
    redirect_to root_path, notice: "Logged out!"
  end
end
