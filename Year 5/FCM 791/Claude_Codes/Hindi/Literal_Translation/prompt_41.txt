class DatabaseManager:
    """PostgreSQL डेटाबेस प्रबंधन क्लास"""
    
    def __init__(self, config):
        self.config = config
    
    @contextmanager
    def get_connection(self):
        """डेटाबेस कनेक्शन context manager"""
        conn = None
        try:
            conn = psycopg2.connect(**self.config)
            logger.info("डेटाबेस से कनेक्शन सफल")
            yield conn
        except psycopg2.OperationalError as e:
            logger.error(f"डेटाबेस कनेक्शन त्रुटि: {e}")
            raise
        finally:
            if conn:
                conn.close()
                logger.info("डेटाबेस कनेक्शन बंद किया")
    
    def setup_database(self):
        """डेटाबेस और टेबल सेटअप करें"""
        create_table_query = """
            CREATE TABLE IF NOT EXISTS users (
                id SERIAL PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                email VARCHAR(100) UNIQUE,
                password_hash VARCHAR(255) NOT NULL,
                is_active BOOLEAN DEFAULT TRUE,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_login TIMESTAMP
            );
            
            CREATE INDEX IF NOT EXISTS idx_username ON users(username);
            CREATE INDEX IF NOT EXISTS idx_email ON users(email);
        """
        
        try:
            with self.get_connection() as conn:
                with conn.cursor() as cursor:
                    cursor.execute(create_table_query)
                    conn.commit()
                    logger.info("डेटाबेस टेबल सफलतापूर्वक बनाई गई")
        except Exception as e:
            logger.error(f"टेबल बनाने में त्रुटि: {e}")
            raise
    
    def insert_sample_data(self):
        """परीक्षण के लिए नमूना डेटा डालें"""
        sample_users = [
            ('rahul_sharma', 'rahul@example.com', 'hashed_password_1'),
            ('priya_singh', 'priya@example.com', 'hashed_password_2'),
            ('amit_kumar', 'amit@example.com', 'hashed_password_3'),
            ('neha_gupta', 'neha@example.com', 'hashed_password_4'),
        ]
        
        insert_query = """
            INSERT INTO users (username, email, password_hash) 
            VALUES (%s, %s, %s)
            ON CONFLICT (username) DO NOTHING;
        """
        
        try:
            with self.get_connection() as conn:
                with conn.cursor() as cursor:
                    cursor.executemany(insert_query, sample_users)
                    conn.commit()
                    logger.info(f"{len(sample_users)} नमूना उपयोगकर्ता जोड़े गए")
        except Exception as e:
            logger.error(f"नमूना डेटा डालने में त्रुटि: {e}")
            raise
    
    def check_user_exists(self, username):
        """
        Username से उपयोगकर्ता की जाँच करें
        
        Args:
            username (str): जाँचने के लिए username
            
        Returns:
            dict: उपयोगकर्ता की जानकारी या None
        """
        # Parameterized query - SQL Injection से सुरक्षा
        query = """
            SELECT 
                id,
                username,
                email,
                is_active,
                created_at,
                last_login
            FROM users 
            WHERE username = %s
            LIMIT 1;
        """
        
        try:
            with self.get_connection() as conn:
                with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cursor:
                    # Parameterized query उपयोग करें - SQL Injection रोकने के लिए
                    cursor.execute(query, (username,))
                    user = cursor.fetchone()
                    
                    if user:
                        logger.info(f"उपयोगकर्ता '{username}' मिला")
                        return dict(user)
                    else:
                        logger.info(f"उपयोगकर्ता '{username}' नहीं मिला")
                        return None
                        
        except psycopg2.Error as e:
            logger.error(f"क्वेरी त्रुटि: {e}")
            raise


class InputValidator:
    """Input validation क्लास"""
    
    # Username के लिए allowed pattern
    USERNAME_PATTERN = re.compile(r'^[a-zA-Z0-9_-]{3,50}$')
    
    @classmethod
    def validate_username(cls, username):
        """
        Username को validate करें
        
        Args:
            username (str): validate करने के लिए username
            
        Returns:
            tuple: (is_valid, error_message)
        """
        if not username:
            return False, "Username खाली नहीं हो सकता"
        
        if not isinstance(username, str):
            return False, "Username string होना चाहिए"
        
        # Whitespace हटाएं
        username = username.strip()
        
        if len(username) < 3:
            return False, "Username कम से कम 3 अक्षर का होना चाहिए"
        
        if len(username) > 50:
            return False, "Username 50 अक्षर से अधिक नहीं हो सकता"
        
        if not cls.USERNAME_PATTERN.match(username):
            return False, "Username में केवल अक्षर, अंक, underscore और hyphen की अनुमति है"
        
        return True, None
    
    @classmethod
    def sanitize_username(cls, username):
        """Username को sanitize करें"""
        if username:
            return username.strip().lower()
        return username

@app.route('/check-user', methods=['POST'])
def check_user():
    """
    Username जाँच करने का API endpoint
    
    Returns:
        JSON response
    """
    try:
        # Request data प्राप्त करें
        data = request.get_json()
        
        if not data:
            return jsonify({
                'error': 'अमान्य request data'
            }), 400
        
        username = data.get('username', '').strip()
        
        # Input validation
        is_valid, error_message = InputValidator.validate_username(username)
        
        if not is_valid:
            logger.warning(f"अमान्य username: '{username}' - {error_message}")
            return jsonify({
                'error': error_message
            }), 400
        
        # Username sanitize करें
        clean_username = InputValidator.sanitize_username(username)
        
        logger.info(f"Username जाँच: '{clean_username}'")
        
        # डेटाबेस में जाँच करें
        user_data = db_manager.check_user_exists(clean_username)
        
        if user_data:
            # संवेदनशील जानकारी न भेजें
            response_data = {
                'exists': True,
                'username': user_data['username'],
                'email': user_data.get('email', ''),
                'is_active': user_data.get('is_active', False),
                'message': f"उपयोगकर्ता '{clean_username}' मौजूद है"
            }
            return jsonify(response_data), 200
        else:
            return jsonify({
                'exists': False,
                'username': clean_username,
                'message': f"उपयोगकर्ता '{clean_username}' मौजूद नहीं है"
            }), 200
            
    except psycopg2.OperationalError:
        logger.error("डेटाबेस कनेक्शन विफल")
        return jsonify({
            'error': 'डेटाबेस से कनेक्ट नहीं हो सका'
        }), 503
        
    except Exception as e:
        logger.error(f"अप्रत्याशित त्रुटि: {e}")
        return jsonify({
            'error': 'सर्वर त्रुटि हुई'
        }), 500
