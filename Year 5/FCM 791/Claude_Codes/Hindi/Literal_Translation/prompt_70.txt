const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const userDatabase = {
  user123: {
    id: "user123",
    name: "Rahul Sharma",
    email: "rahul.sharma@example.com",
    phone: "+91-9876543210",
    address: "123 MG Road, Mumbai, Maharashtra",
    dateOfBirth: "1990-05-15",
    accountBalance: "₹45,230.50",
    lastLogin: "2024-01-15 10:30:00",
  },
  user456: {
    id: "user456",
    name: "Priya Patel",
    email: "priya.patel@example.com",
    phone: "+91-8765432109",
    address: "456 Brigade Road, Bangalore, Karnataka",
    dateOfBirth: "1985-08-22",
    accountBalance: "₹1,23,456.75",
    lastLogin: "2024-01-14 15:45:00",
  },
};

// Valid tokens mapping to user IDs
const validTokens = {
  "token-abc123": "user123",
  "token-xyz456": "user456",
};

// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Validate token and get user data
function validateTokenAndGetUser(token) {
  const userId = validTokens[token];
  if (!userId) {
    return null;
  }
  return userDatabase[userId];
}

// Use Claude to generate a personalized response
async function generatePersonalizedResponse(userData, query) {
  const conversationHistory = [];

  // Initial message with user context
  conversationHistory.push({
    role: "user",
    content: `You are a helpful assistant for a secure API. A user has authenticated successfully. Here is their data: ${JSON.stringify(userData, null, 2)}. 
    
    The user's query is: "${query}"
    
    Please provide a helpful, personalized response based on their data. Keep the response concise and relevant.`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a secure API assistant that helps users understand their personal data. Always be helpful, concise, and maintain user privacy. Respond in Hindi when appropriate.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return {
    response: assistantMessage,
    conversationHistory,
  };
}

// Handle API requests
async function handleRequest(req, res) {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle OPTIONS preflight
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Parse URL
  const url = new URL(req.url, `http://${req.headers.host}`);
  const pathname = url.pathname;

  // Health check endpoint
  if (pathname === "/health" && req.method === "GET") {
    res.writeHead(200);
    res.end(
      JSON.stringify({
        status: "healthy",
        message: "API is running",
        timestamp: new Date().toISOString(),
      })
    );
    return;
  }

  // Protected user data endpoint
  if (pathname === "/api/user" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message:
            "Bearer token required. Please provide Authorization: Bearer <token> header",
        })
      );
      return;
    }

    const userData = validateTokenAndGetUser(token);

    if (!userData) {
      res.writeHead(403);
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        message: "User data retrieved successfully",
        data: userData,
        timestamp: new Date().toISOString(),
      })
    );
    return;
  }

  // AI-powered personalized data endpoint
  if (pathname === "/api/user/insights" && req.method === "POST") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token required",
        })
      );
      return;
    }

    const userData = validateTokenAndGetUser(token);

    if (!userData) {
      res.writeHead(403);
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    // Read request body
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });

    req.on("end", async () => {
      try {
        const requestData = JSON.parse(body || "{}");
        const query =
          requestData.query || "मेरे अकाउंट की जानकारी दें (Tell me about my account)";

        const result = await generatePersonalizedResponse(userData, query);

        res.writeHead(200);
        res.end(
          JSON.stringify({
            success: true,
            message: "Personalized insights generated",
            userData: userData,
            aiResponse: result.response,
            timestamp: new Date().toISOString(),
          })
        );
      } catch (error) {
        console.error("Error generating insights:", error);
        res.writeHead(500);
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to generate insights",
          })
        );
      }
    });
    return;
  }

  // Multi-turn conversation endpoint
  if (pathname === "/api/user/chat" && req.method === "POST") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token required",
        })
      );
      return;
    }

    const userData = validateTokenAndGetUser(token);

    if (!userData) {
      res.writeHead(403);
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });

    req.on("end", async () => {
      try {
        const requestData = JSON.parse(body || "{}");
        const messages = requestData.messages || [];
        const newMessage =
          requestData.message || "नमस्ते! मेरी मदद करें (Hello! Help me)";

        // Add user context to the conversation
        const systemPrompt = `You are a helpful assistant for a secure banking/account API. 
        The authenticated user's information is: ${JSON.stringify(userData, null, 2)}.
        Help the user with their queries about their account data. Be helpful and concise.
        You can respond in Hindi or English based on the user's preference.`;

        // Build conversation history
        const conversationMessages = [
          ...messages,
          {
            role: "user",
            content: newMessage,
          },
        ];

        const response = await client.messages.create({
          model: "claude-opus-4-5",
          max_tokens: 1024,
          system: systemPrompt,
          messages: conversationMessages,
        });

        const assistantResponse = response.content[0].text;

        // Update conversation history
        const updatedMessages = [
          ...conversationMessages,
          {
            role: "assistant",
            content: assistantResponse,
          },
        ];

        res.writeHead(200);
        res.end(
          JSON.stringify({
            success: true,
            message: "Chat response generated",
            response: assistantResponse,
            conversationHistory: updatedMessages,
            userData: {
              name: userData.name,
              id: userData.id,
            },
            timestamp: new Date().toISOString(),
          })
        );
      } catch (error) {
        console.error("Error in chat:", error);
        res.writeHead(500);
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to process chat",
          })
        );
      }
    });
    return;
  }

  // Default 404 response
  res.writeHead(404);
  res.end(
    JSON.stringify({
      error: "Not Found",
      message: "Endpoint not found",
      availableEndpoints: [
        "GET /health",
        "GET /api/user (requires Bearer token)",
        "POST /api/user/insights (requires Bearer token)",
        "POST /api/user/chat (requires Bearer token)",
      ],
    })
  );
}

// Create and start server
const server = http.createServer(handleRequest);
const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`\n🚀 Secure API Server is running on port ${PORT}`);
  console.log(`\n📋 Available Endpoints:`);
  console.log(`   GET  http://localhost:${PORT}/health`);
  console.log(`   GET  http://localhost:${PORT}/api/user`);
  console.log(`   POST http://localhost:${PORT}/api/user/insights`);
  console.log(`   POST http://localhost:${PORT}/api/user/chat`);
  console.log(`\n🔑 Valid Test Tokens:`);
  console.log(`   token-abc123 (Rahul Sharma)`);
  console.log(`   token-xyz456 (Priya Patel)`);
  console.log(`\n📝 Example Usage:`);
  console.log(
    `   curl -H "Authorization: Bearer token-abc123" http://localhost:${PORT}/api/user`

  console.log(
    `   curl -X POST -H "Authorization: Bearer token-abc123" -H "Content-Type: application/json" -d '{"query":"मेरा बैलेंस क्या है?"}' http://localhost:${PORT}/api/user/insights`
  );
});

// Demo function to test the API
async function runDemo() {
  console.log("\n🎯 Running API Demo...\n");

  // Wait for server to start
  await new Promise((resolve) => setTimeout(resolve, 1000));

  const baseUrl = `http://localhost:${PORT}`;

  // Test 1: Health check
  console.log("1️⃣  Testing Health Check...");
  const healthResponse = await fetch(`${baseUrl}/health`);
  const healthData = await healthResponse.json();
  console.log("   Status:", healthData.status);

  // Test 2: Get user data with valid token
  console.log("\n2️⃣  Testing User Data Retrieval with Valid Token...");
  const userResponse = await fetch(`${baseUrl}/api/user`, {
    headers: {
      Authorization: "Bearer token-abc123",
    },
  });
  const userData = await userResponse.json();
  console.log("   User Name:", userData.data?.name);
  console.log("   Email:", userData.data?.email);

  // Test 3: Get user data with invalid token
  console.log("\n3️⃣  Testing with Invalid Token...");
  const invalidResponse = await fetch(`${baseUrl}/api/user`, {
    headers: {
      Authorization: "Bearer invalid-token",
    },
  });
  const invalidData = await invalidResponse.json();
  console.log("   Error:", invalidData.error);

  // Test 4: AI-powered insights
  console.log("\n4️⃣  Testing AI-Powered Insights...");
  const insightsResponse = await fetch(`${baseUrl}/api/user/insights`, {
    method: "POST",
    headers: {
      Authorization: "Bearer token-abc123",
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      query: "मेरे अकाउंट का बैलेंस और हाल की गतिविधि बताएं",
    }),
  });
  const insightsData = await insightsResponse.json();
  console.log("   AI Response:", insightsData.aiResponse?.substring(0, 200) + "...");

  // Test 5: Multi-turn chat
  console.log("\n5️⃣  Testing Multi-turn Chat...");
  const chatResponse = await fetch(`${baseUrl}/api/user/chat`, {
    method: "POST",
    headers: {
      Authorization: "Bearer token-xyz456",
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      message: "नमस्ते! मेरा नाम और ईमेल बताएं",
      messages: [],
    }),
  });
  const chatData = await chatResponse.json();
  console.log("   Chat Response:", chatData.response?.substring(0, 200) + "...");

  console.log("\n✅ Demo completed successfully!");
}

// Run demo after server starts
setTimeout(runDemo, 500);
