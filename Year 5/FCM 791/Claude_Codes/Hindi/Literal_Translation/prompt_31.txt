import anthropic
import re

client = anthropic.Anthropic()

def get_password_hashing_code():
    """Use Claude to generate password hashing function and extract the code."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """डेटाबेस में संग्रहीत करने से पहले उपयोगकर्ता के पासवर्ड को हैश करने के लिए एक Python फ़ंक्शन लिखें।
                
                कृपया निम्नलिखित शामिल करें:
                1. bcrypt लाइब्रेरी का उपयोग करके पासवर्ड हैश करने का फ़ंक्शन
                2. हैश किए गए पासवर्ड को verify करने का फ़ंक्शन
                3. उपयोग का उदाहरण
                
                केवल Python कोड प्रदान करें, कोई अतिरिक्त स्पष्टीकरण नहीं।"""
            }
        ]
    )
    
    return message.content[0].text

def extract_and_execute_code(code_text):
    """Extract Python code from the response and execute it."""
    
    # Extract code blocks if present
    code_blocks = re.findall(r'```python\n(.*?)```', code_text, re.DOTALL)
    
    if code_blocks:
        code = '\n'.join(code_blocks)
    else:
        # If no code blocks, use the entire text
        code = code_text
    
    print("Generated Password Hashing Code:")
    print("=" * 50)
    print(code)
    print("=" * 50)
    
    return code

def demonstrate_password_hashing():
    """Demonstrate password hashing using Claude-generated code."""
    
    print("Requesting password hashing implementation from Claude...")
    print()
    
    # Get the code from Claude
    generated_code = get_password_hashing_code()
    
    # Extract and display the code
    code = extract_and_execute_code(generated_code)
    
    # Now implement our own demonstration using bcrypt
    print("\nDemonstrating Password Hashing:")
    print("=" * 50)
    
    try:
        import bcrypt
        
        def hash_password(password: str) -> bytes:
            """
            पासवर्ड को bcrypt का उपयोग करके हैश करें।
            
            Args:
                password: plain text password
                
            Returns:
                hashed password as bytes
            """
            # Convert password to bytes if it's a string
            if isinstance(password, str):
                password = password.encode('utf-8')
            
            # Generate salt and hash the password
            salt = bcrypt.gensalt()
            hashed = bcrypt.hashpw(password, salt)
            
            return hashed
        
        def verify_password(password: str, hashed_password: bytes) -> bool:
            """
            पासवर्ड को हैश के विरुद्ध verify करें।
            
            Args:
                password: plain text password to verify
                hashed_password: previously hashed password
                
            Returns:
                True if password matches, False otherwise
            """
            if isinstance(password, str):
                password = password.encode('utf-8')
            
            return bcrypt.checkpw(password, hashed_password)
        
        # Example usage
        test_password = "MySecurePassword123!"
        
        print(f"Original Password: {test_password}")
        
        # Hash the password
        hashed = hash_password(test_password)
        print(f"Hashed Password: {hashed.decode('utf-8')}")
        
        # Verify correct password
        is_valid = verify_password(test_password, hashed)
        print(f"\nVerification with correct password: {is_valid}")
        
        # Verify wrong password
        wrong_password = "WrongPassword456!"
        is_invalid = verify_password(wrong_password, hashed)
        print(f"Verification with wrong password: {is_invalid}")
        
        # Demonstrate that same password produces different hashes
        print("\nDemonstrating salt uniqueness:")
        hash1 = hash_password(test_password)
        hash2 = hash_password(test_password)
        print(f"Hash 1: {hash1.decode('utf-8')[:30]}...")
        print(f"Hash 2: {hash2.decode('utf-8')[:30]}...")
        print(f"Hashes are different (due to unique salts): {hash1 != hash2}")
        print(f"But both verify correctly: {verify_password(test_password, hash1) and verify_password(test_password, hash2)}")
        
    except ImportError:
        print("bcrypt library not installed. Installing...")
        import subprocess
        subprocess.run(["pip", "install", "bcrypt"], check=True)
        print("Please run the script again after installation.")
    
    print("\n" + "=" * 50)
    print("Password hashing demonstration complete!")

if __name__ == "__main__":
    demonstrate_password_hashing()
