from lxml import etree
import os
import sys

def parse_xml_and_extract_titles(xml_file_path):
    """
    XML फ़ाइल को पार्स करके 'title' टैग निकालता है।
    
    Args:
        xml_file_path: XML फ़ाइल का पथ
    
    Returns:
        titles की list
    """
    
    # फ़ाइल का अस्तित्व जांचें
    if not os.path.exists(xml_file_path):
        print(f"❌ त्रुटि: फ़ाइल '{xml_file_path}' नहीं मिली!")
        return []
    
    # फ़ाइल एक्सटेंशन जांचें
    if not xml_file_path.lower().endswith('.xml'):
        print("⚠️  चेतावनी: यह XML फ़ाइल नहीं लगती, फिर भी पार्स करने की कोशिश करते हैं...")
    
    titles = []
    
    try:
        # XML फ़ाइल पार्स करें
        print(f"\n📂 फ़ाइल लोड हो रही है: {xml_file_path}")
        
        # XML पार्सर बनाएं
        parser = etree.XMLParser(
            recover=True,           # त्रुटियों से उबरने की कोशिश करें
            remove_blank_text=True, # खाली टेक्स्ट हटाएं
            encoding='utf-8'        # एन्कोडिंग सेट करें
        )
        
        # फ़ाइल पार्स करें
        tree = etree.parse(xml_file_path, parser)
        root = tree.getroot()
        
        print(f"✅ XML फ़ाइल सफलतापूर्वक पार्स हुई!")
        print(f"📌 Root Element: <{root.tag}>")
        
        # Namespace हैंडल करें
        namespace = ''
        if '{' in root.tag:
            namespace = root.tag.split('}')[0] + '}'
            print(f"🔍 Namespace मिला: {namespace}")
        
        # सभी 'title' टैग खोजें (namespace के साथ और बिना)
        # Method 1: XPath का उपयोग
        if namespace:
            title_elements = tree.xpath(f'//{namespace}title')
            if not title_elements:
                # बिना namespace के भी खोजें
                title_elements = tree.xpath('//title')
        else:
            title_elements = tree.xpath('//title')
        
        # Method 2: findall का उपयोग (backup)
        if not title_elements:
            title_elements = root.findall('.//title')
            if not title_elements and namespace:
                title_elements = root.findall(f'.//{namespace}title')
        
        # Titles निकालें
        if title_elements:
            print(f"\n🎯 कुल {len(title_elements)} 'title' टैग मिले:\n")
            print("=" * 50)
            
            for index, title_elem in enumerate(title_elements, 1):
                title_text = title_elem.text
                
                if title_text:
                    title_text = title_text.strip()
                    titles.append(title_text)
                    print(f"  {index}. {title_text}")
                    
                    # Attributes भी दिखाएं अगर हों
                    if title_elem.attrib:
                        print(f"     📎 Attributes: {dict(title_elem.attrib)}")
                else:
                    print(f"  {index}. (खाली title टैग)")
                    titles.append("")
            
            print("=" * 50)
        else:
            print("\n⚠️  कोई 'title' टैग नहीं मिला!")
            
            # XML संरचना दिखाएं
            print("\n📋 XML संरचना:")
            show_xml_structure(root)
    
    except etree.XMLSyntaxError as e:
        print(f"❌ XML Syntax त्रुटि: {e}")
        print("💡 सुझाव: XML फ़ाइल की संरचना जांचें")
    
    except etree.ParseError as e:
        print(f"❌ Parse त्रुटि: {e}")
    
    except PermissionError:
        print(f"❌ अनुमति त्रुटि: फ़ाइल '{xml_file_path}' पढ़ने की अनुमति नहीं है!")
    
    except Exception as e:
        print(f"❌ अप्रत्याशित त्रुटि: {e}")
    
    return titles


def show_xml_structure(element, level=0, max_level=3):
    """
    XML की संरचना दिखाता है।
    
    Args:
        element: XML element
        level: वर्तमान स्तर
        max_level: अधिकतम स्तर
    """
    if level > max_level:
        return
    
    indent = "  " * level
    tag = element.tag.split('}')[-1] if '}' in element.tag else element.tag
    
    print(f"{indent}└─ <{tag}>", end="")
    
    if element.text and element.text.strip():
        text_preview = element.text.strip()[:30]
        print(f" → '{text_preview}{'...' if len(element.text.strip()) > 30 else ''}'", end="")
    
    print()
    
    for child in element:
        show_xml_structure(child, level + 1, max_level)

def get_user_file_input():
    """
    उपयोगकर्ता से फ़ाइल पथ लेता है।
    
    Returns:
        फ़ाइल पथ
    """
    print("\n" + "="*60)
    print("       XML Title Extractor - lxml Library")
    print("="*60)
    print("\n📌 विकल्प:")
    print("  1. अपनी XML फ़ाइल का पथ दें")
    print("  2. Sample XML फ़ाइल बनाएं और उसे test करें")
    print("  3. बाहर निकलें")
    
    while True:
        choice = input("\n👉 अपना विकल्प चुनें (1/2/3): ").strip()
        
        if choice == '1':
            file_path = input("📁 XML फ़ाइल का पूरा पथ दें: ").strip()
            
            # Quotes हटाएं अगर हों
            file_path = file_path.strip('"').strip("'")
            
            if file_path:
                return file_path
            else:
                print("❌ कृपया वैध फ़ाइल पथ दें!")
        
        elif choice == '2':
            return create_sample_xml()
        
        elif choice == '3':
            print("\n👋 धन्यवाद! प्रोग्राम बंद हो रहा है...")
            sys.exit(0)
        
        else:
            print("❌ अमान्य विकल्प! कृपया 1, 2, या 3 चुनें।")


def save_titles_to_file(titles, output_file="extracted_titles.txt"):
    """
    निकाले गए titles को फ़ाइल में सेव करता है।
    
    Args:
        titles: titles की list
        output_file: output फ़ाइल का नाम
    """
    if not titles:
        return
    
    save_choice = input("\n💾 क्या titles को फ़ाइल में सेव करना है? (हाँ/नहीं): ").strip().lower()
    
    if save_choice in ['हाँ', 'han', 'yes', 'y', 'h']:
        try:
            with open(output_file, 'w', encoding='utf-8') as f:
                f.write("निकाले गए Titles\n")
                f.write("=" * 40 + "\n\n")
                
                for i, title in enumerate(titles, 1):
                    f.write(f"{i}. {title}\n")
                
                f.write(f"\nकुल Titles: {len(titles)}")
            
            print(f"✅ Titles सफलतापूर्वक '{output_file}' में सेव हो गए!")
        
        except Exception as e:
            print(f"❌ फ़ाइल सेव करने में त्रुटि: {e}")


def display_summary(titles, xml_file):
    """
    सारांश दिखाता है।
    
    Args:
        titles: titles की list
        xml_file: XML फ़ाइल का नाम
    """
    print("\n" + "="*50)
    print("📊 सारांश (Summary)")
    print("="*50)
    print(f"📁 फ़ाइल: {os.path.basename(xml_file)}")
    print(f"🎯 कुल Titles मिले: {len(titles)}")
    
    if titles:
        non_empty = [t for t in titles if t]
        print(f"✅ भरे हुए Titles: {len(non_empty)}")
        print(f"⬜ खाली Titles: {len(titles) - len(non_empty)}")
    
    print("="*50)


def main():
    """
    मुख्य फ़ंक्शन
    """
    # Command line argument जांचें
    if len(sys.argv) > 1:
        xml_file_path = sys.argv[1]
        print(f"\n📂 Command line से फ़ाइल: {xml_file_path}")
    else:
        # Interactive mode
        xml_file_path = get_user_file_input()
    
    # XML पार्स करें और titles निकालें
    titles = parse_xml_and_extract_titles(xml_file_path)
    
    # सारांश दिखाएं
    display_summary(titles, xml_file_path)
    
    # Titles सेव करने का विकल्प
    if titles:
        save_titles_to_file(titles)
    
    return titles


# प्रोग्राम शुरू करें
if __name__ == "__main__":
    extracted_titles = main()
    
    # Python में उपयोग के लिए titles return करें
    if extracted_titles:
        print(f"\n🐍 Python List में Titles:")
        print(extracted_titles)
