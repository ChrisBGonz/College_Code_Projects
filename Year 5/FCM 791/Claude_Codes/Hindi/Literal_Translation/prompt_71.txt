import pickle
import os
import datetime
from dataclasses import dataclass, field
from typing import List, Dict, Optional
import random

@dataclass
class Item:
    """खिलाड़ी की इन्वेंटरी आइटम"""
    name: str
    item_type: str  # weapon, armor, potion, etc.
    damage: int = 0
    defense: int = 0
    quantity: int = 1
    value: int = 0
    
    def __str__(self):
        return f"{self.name} (Type: {self.item_type}, Qty: {self.quantity})"


@dataclass
class Quest:
    """गेम क्वेस्ट की जानकारी"""
    quest_id: int
    title: str
    description: str
    is_completed: bool = False
    progress: int = 0
    target: int = 100
    reward_gold: int = 0
    reward_xp: int = 0
    
    def completion_percentage(self):
        return (self.progress / self.target) * 100 if self.target > 0 else 0
    
    def __str__(self):
        status = "✅ पूर्ण" if self.is_completed else f"⏳ {self.completion_percentage():.1f}%"
        return f"Quest: {self.title} [{status}]"


@dataclass
class PlayerStats:
    """खिलाड़ी के आँकड़े"""
    health: int = 100
    max_health: int = 100
    mana: int = 50
    max_mana: int = 50
    strength: int = 10
    defense: int = 5
    agility: int = 8
    intelligence: int = 7
    luck: int = 5
    
    def is_alive(self):
        return self.health > 0
    
    def health_percentage(self):
        return (self.health / self.max_health) * 100


@dataclass
class Achievement:
    """खिलाड़ी की उपलब्धियाँ"""
    achievement_id: int
    name: str
    description: str
    unlocked: bool = False
    unlock_date: Optional[str] = None
    
    def unlock(self):
        self.unlocked = True
        self.unlock_date = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    def __str__(self):
        status = "🏆 अनलॉक" if self.unlocked else "🔒 लॉक"
        return f"{status} - {self.name}: {self.description}"


@dataclass
class GameWorld:
    """गेम वर्ल्ड की स्थिति"""
    current_map: str = "शुरुआती गाँव"
    position_x: float = 0.0
    position_y: float = 0.0
    discovered_locations: List[str] = field(default_factory=list)
    world_events: Dict[str, bool] = field(default_factory=dict)
    time_of_day: str = "दिन"
    day_count: int = 1
    
    def move_to(self, location: str, x: float, y: float):
        self.current_map = location
        self.position_x = x
        self.position_y = y
        if location not in self.discovered_locations:
            self.discovered_locations.append(location)


@dataclass
class GameState:
    """मुख्य गेम स्टेट - सभी जानकारी एक साथ"""
    
    # खिलाड़ी की बुनियादी जानकारी
    player_name: str
    player_class: str  # Warrior, Mage, Archer, etc.
    level: int = 1
    experience: int = 0
    experience_to_next_level: int = 100
    gold: int = 50
    
    # खिलाड़ी के आँकड़े
    stats: PlayerStats = field(default_factory=PlayerStats)
    
    # इन्वेंटरी
    inventory: List[Item] = field(default_factory=list)
    equipped_weapon: Optional[Item] = None
    equipped_armor: Optional[Item] = None
    
    # क्वेस्ट
    active_quests: List[Quest] = field(default_factory=list)
    completed_quests: List[Quest] = field(default_factory=list)
    
    # उपलब्धियाँ
    achievements: List[Achievement] = field(default_factory=list)
    
    # गेम वर्ल्ड
    world: GameWorld = field(default_factory=GameWorld)
    
    # गेम सेटिंग्स
    settings: Dict[str, any] = field(default_factory=dict)
    
    # सेव जानकारी
    save_count: int = 0
    total_play_time: float = 0.0  # मिनटों में
    last_saved: Optional[str] = None
    created_at: str = field(default_factory=lambda: datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))
    
    def gain_experience(self, xp: int):
        """अनुभव अंक जोड़ें और लेवल अप चेक करें"""
        self.experience += xp
        leveled_up = False
        while self.experience >= self.experience_to_next_level:
            self.experience -= self.experience_to_next_level
            self.level += 1
            self.experience_to_next_level = int(self.experience_to_next_level * 1.5)
            self.stats.max_health += 10
            self.stats.health = self.stats.max_health
            self.stats.max_mana += 5
            self.stats.mana = self.stats.max_mana
            leveled_up = True
        return leveled_up
    
    def add_item(self, item: Item):
        """इन्वेंटरी में आइटम जोड़ें"""
        for existing_item in self.inventory:
            if existing_item.name == item.name:
                existing_item.quantity += item.quantity
                return
        self.inventory.append(item)
    
    def add_gold(self, amount: int):
        """सोना जोड़ें"""
        self.gold += amount
    
    def spend_gold(self, amount: int) -> bool:
        """सोना खर्च करें"""
        if self.gold >= amount:
            self.gold -= amount
            return True
        return False

class GameStateManager:
    """गेम स्टेट को सेव और लोड करने का मैनेजर"""
    
    SAVE_DIRECTORY = "game_saves"
    SAVE_FILE_EXTENSION = ".pkl"
    BACKUP_EXTENSION = ".pkl.bak"
    
    def __init__(self):
        """सेव डायरेक्टरी बनाएं अगर नहीं है"""
        if not os.path.exists(self.SAVE_DIRECTORY):
            os.makedirs(self.SAVE_DIRECTORY)
            print(f"📁 सेव फोल्डर बनाया गया: {self.SAVE_DIRECTORY}/")
    
    def _get_save_path(self, slot: int) -> str:
        """सेव फ़ाइल का पूरा पथ प्राप्त करें"""
        return os.path.join(
            self.SAVE_DIRECTORY, 
            f"save_slot_{slot}{self.SAVE_FILE_EXTENSION}"
        )
    
    def _get_backup_path(self, slot: int) -> str:
        """बैकअप फ़ाइल का पूरा पथ प्राप्त करें"""
        return os.path.join(
            self.SAVE_DIRECTORY,
            f"save_slot_{slot}{self.BACKUP_EXTENSION}"
        )
    
    def save_game(self, game_state: GameState, slot: int = 1) -> bool:
        """
        गेम स्टेट को pickle का उपयोग करके फ़ाइल में सहेजें
        
        Args:
            game_state: सहेजने के लिए GameState ऑब्जेक्ट
            slot: सेव स्लॉट नंबर (1-5)
        
        Returns:
            bool: सफल होने पर True
        """
        save_path = self._get_save_path(slot)
        backup_path = self._get_backup_path(slot)
        
        try:
            # पुरानी सेव का बैकअप बनाएं
            if os.path.exists(save_path):
                import shutil
                shutil.copy2(save_path, backup_path)
                print(f"💾 पुरानी सेव का बैकअप बनाया: {backup_path}")
            
            # सेव जानकारी अपडेट करें
            game_state.save_count += 1
            game_state.last_saved = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            
            # pickle का उपयोग करके बाइनरी फ़ाइल में लिखें
            with open(save_path, 'wb') as save_file:
                pickle.dump(game_state, save_file, protocol=pickle.HIGHEST_PROTOCOL)
            
            # फ़ाइल का आकार जाँचें
            file_size = os.path.getsize(save_path)
            
            print(f"\n✅ गेम सफलतापूर्वक सहेजा गया!")
            print(f"   📍 स्लॉट: {slot}")
            print(f"   📁 फ़ाइल: {save_path}")
            print(f"   📊 आकार: {file_size} bytes")
            print(f"   🕐 समय: {game_state.last_saved}")
            print(f"   💾 कुल सेव: {game_state.save_count}")
            
            return True
            
        except pickle.PicklingError as e:
            print(f"❌ Pickle त्रुटि: {e}")
            return False
        except IOError as e:
            print(f"❌ फ़ाइल त्रुटि: {e}")
            return False
        except Exception as e:
            print(f"❌ अज्ञात त्रुटि: {e}")
            return False
    
    def load_game(self, slot: int = 1) -> Optional[GameState]:
        """
        pickle का उपयोग करके फ़ाइल से गेम स्टेट लोड करें
        
        Args:
            slot: लोड करने के लिए सेव स्लॉट नंबर
        
        Returns:
            GameState या None अगर लोड विफल हो
        """
        save_path = self._get_save_path(slot)
        
        if not os.path.exists(save_path):
            print(f"❌ स्लॉट {slot} में कोई सेव नहीं मिला!")
            return None
        
        try:
            # pickle का उपयोग करके बाइनरी फ़ाइल से पढ़ें
            with open(save_path, 'rb') as save_file:
                game_state = pickle.load(save_file)
            
            print(f"\n✅ गेम सफलतापूर्वक लोड किया गया!")
            print(f"   📍 स्लॉट: {slot}")
            print(f"   👤 खिलाड़ी: {game_state.player_name}")
            print(f"   ⚔️  क्लास: {game_state.player_class}")
            print(f"   📈 लेवल: {game_state.level}")
            print(f"   🕐 अंतिम सेव: {game_state.last_saved}")
            
            return game_state
            
        except pickle.UnpicklingError as e:
            print(f"❌ Pickle लोड त्रुटि: {e}")
            print("   बैकअप से लोड करने का प्रयास...")
            return self._load_from_backup(slot)
        except FileNotFoundError:
            print(f"❌ सेव फ़ाइल नहीं मिली: {save_path}")
            return None
        except Exception as e:
            print(f"❌ लोड त्रुटि: {e}")
            return None
    
    def _load_from_backup(self, slot: int) -> Optional[GameState]:
        """बैकअप फ़ाइल से लोड करें"""
        backup_path = self._get_backup_path(slot)
        
        if not os.path.exists(backup_path):
            print("❌ बैकअप भी नहीं मिला!")
            return None
        
        try:
            with open(backup_path, 'rb') as backup_file:
                game_state = pickle.load(backup_file)
            print(f"✅ बैकअप से लोड सफल!")
            return game_state
        except Exception as e:
            print(f"❌ बैकअप लोड विफल: {e}")
            return None
    
    def delete_save(self, slot: int) -> bool:
        """सेव फ़ाइल हटाएं"""
        save_path = self._get_save_path(slot)
        backup_path = self._get_backup_path(slot)
        
        deleted = False
        if os.path.exists(save_path):
            os.remove(save_path)
            print(f"🗑️  स्लॉट {slot} की सेव हटाई गई")
            deleted = True
        
        if os.path.exists(backup_path):
            os.remove(backup_path)
            print(f"🗑️  स्लॉट {slot} का बैकअप हटाया गया")
        
        if not deleted:
            print(f"⚠️  स्लॉट {slot} में कोई सेव नहीं थी")
        
        return deleted
    
    def list_saves(self):
        """सभी उपलब्ध सेव की सूची दिखाएं"""
        print("\n" + "="*50)
        print("📋 उपलब्ध सेव स्लॉट:")
        print("="*50)
        
        found_any = False
        for slot in range(1, 6):
            save_path = self._get_save_path(slot)
            if os.path.exists(save_path):
                try:
                    with open(save_path, 'rb') as f:
                        gs = pickle.load(f)
                    file_size = os.path.getsize(save_path)
                    print(f"\n  स्लॉट {slot}:")
                    print(f"    👤 {gs.player_name} ({gs.player_class})")
                    print(f"    📈 लेवल {gs.level} | 💰 {gs.gold} सोना")
                    print(f"    🗺️  {gs.world.current_map}")
                    print(f"    🕐 {gs.last_saved}")
                    print(f"    📊 {file_size} bytes")
                    found_any = True
                except Exception:
                    print(f"\n  स्लॉट {slot}: ⚠️  दूषित फ़ाइल")
        
        if not found_any:
            print("  कोई सेव नहीं मिली")
        
        print("="*50)
    
    def get_save_info(self, slot: int) -> Dict:
        """सेव की जानकारी बिना पूरा लोड किए प्राप्त करें"""
        save_path = self._get_save_path(slot)
        
        if not os.path.exists(save_path):
            return {"exists": False}
        
        try:
            with open(save_path, 'rb') as f:
                gs = pickle.load(f)
            
            return {
                "exists": True,
                "player_name": gs.player_name,
                "player_class": gs.player_class,
                "level": gs.level,
                "gold": gs.gold,
                "location": gs.world.current_map,
                "last_saved": gs.last_saved,
                "save_count": gs.save_count,
                "file_size": os.path.getsize(save_path)
            }
        except Exception as e:
            return {"exists": True, "error": str(e)}

            Item("जादूगर का वस्त्र", "armor", defense=3, value=35),
            Item("माना औषधि", "potion", quantity=5, value=12),

        Achievement(2, "पहली जीत", "पहली लड़ाई जीतें"),
        Achievement(3, "खजाना खोजी", "पहला खजाना खोजें"),

        game_state.add_item(item)
    
    # हथियार और कवच लगाएं
    items = starting_items.get(player_class, [])
    for item in items:
        if item.item_type == "weapon":
            game_state.equipped_weapon = item
        elif item.item_type == "armor":
            game_state.equipped_armor = item
    
    return game_state


def simulate_gameplay(game_state: GameState):
    """गेमप्ले सिमुलेट करें - कुछ बदलाव करें"""
    print("\n🎮 गेमप्ले सिमुलेट हो रहा है...")
    
    # कुछ अनुभव अंक कमाएं
    xp_gained = random.randint(50, 150)
    leveled_up = game_state.gain_experience(xp_gained)
    print(f"  ⚔️  लड़ाई जीती! +{xp_gained} XP")
    if leveled_up:
        print(f"  🎉 लेवल अप! अब लेवल {game_state.level}")
    
    # सोना कमाएं
    gold_gained = random.randint(20, 80)
    game_state.add_gold(gold_gained)
    print(f"  💰 +{gold_gained} सोना मिला")
    
    # नया आइटम मिला
    new_item = Item("जादुई पत्थर", "misc", value=25, quantity=1)
    game_state.add_item(new_item)
    print(f"  🎁 नया आइटम मिला: {new_item.name}")
    
    # क्वेस्ट प्रगति
    if game_state.active_quests:
        quest = game_state.active_quests[0]
        quest.progress = min(quest.progress + random.randint(1, 3), quest.target)
        print(f"  📜 क्वेस्ट प्रगति: {quest.title} - {quest.completion_percentage():.1f}%")
        
        if quest.progress >= quest.target:
            quest.is_completed = True
            game_state.add_gold(quest.reward_gold)
            game_state.gain_experience(quest.reward_xp)
            game_state.active_quests.remove(quest)
            game_state.completed_quests.append(quest)
            print(f"  ✅ क्वेस्ट पूरी: {quest.title}! +{quest.reward_gold} सोना, +{quest.reward_xp} XP")
    
    # नई जगह खोजें
    new_location = random.choice(["काला जंगल", "पहाड़ी किला", "रहस्यमय गुफा", "प्राचीन मंदिर"])
    game_state.world.move_to(new_location, 
                              random.uniform(20, 100), 
                              random.uniform(20, 100))
    print(f"  🗺️  नई जगह खोजी: {new_location}")
    
    # खेलने का समय बढ़ाएं
    game_state.total_play_time += random.uniform(5, 30)
    
    print("  ✨ गेमप्ले सिमुलेशन पूर्ण!")
