  function getStoredData() {
    try {

      return data ? JSON.parse(data) : {};
    } catch (error) {
      console.error('localStorage से डेटा पढ़ने में त्रुटि:', error);

  // localStorage में डेटा सहेजें
  function saveToStorage(stepData) {
    try {
      const existingData = getStoredData();
      const updatedData = { ...existingData, ...stepData };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(updatedData));
      localStorage.setItem(CURRENT_STEP_KEY, getCurrentStep());
      console.log('डेटा सफलतापूर्वक सहेजा गया:', updatedData);
      return true;
    } catch (error) {
      console.error('localStorage में डेटा सहेजने में त्रुटि:', error);
      return false;
    }
  }

  // वर्तमान चरण प्राप्त करें

  }

  // localStorage साफ़ करें

  // फ़ील्ड सत्यापन
  function validateField(field, value) {
    const errors = [];

    if (field.required && (!value || value.toString().trim() === '')) {
      errors.push(`${field.label} आवश्यक है`);
      return errors;
    }

    if (value) {
      switch (field.type) {
        case 'email':
          const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
          if (!emailRegex.test(value)) {
            errors.push('कृपया वैध ईमेल पता दर्ज करें');
          }
          break;

        case 'tel':
          const phoneRegex = /^[6-9]\d{9}$/;
          if (!phoneRegex.test(value.replace(/\s/g, ''))) {
            errors.push('कृपया वैध 10 अंकों का फ़ोन नंबर दर्ज करें');
          }
          break;

        case 'password':
          if (value.length < 8) {
            errors.push('पासवर्ड कम से कम 8 अक्षरों का होना चाहिए');
          }
          if (!/(?=.*[A-Z])/.test(value)) {
            errors.push('पासवर्ड में कम से कम एक बड़ा अक्षर होना चाहिए');
          }
          if (!/(?=.*\d)/.test(value)) {
            errors.push('पासवर्ड में कम से कम एक अंक होना चाहिए');
          }
          break;

        case 'text':
          if (field.name === 'pincode') {
            const pincodeRegex = /^\d{6}$/;
            if (!pincodeRegex.test(value)) {
              errors.push('कृपया वैध 6 अंकों का पिन कोड दर्ज करें');
            }
          }
          if (field.name === 'username') {
            if (value.length < 4) {
              errors.push('उपयोगकर्ता नाम कम से कम 4 अक्षरों का होना चाहिए');
            }
            if (!/^[a-zA-Z0-9_]+$/.test(value)) {
              errors.push('उपयोगकर्ता नाम में केवल अक्षर, अंक और _ का उपयोग करें');
            }
          }
          break;
      }
    }

    return errors;
  }

  // चरण सत्यापन
  function validateStep(stepId, formData) {
    const step = steps.find(s => s.id === stepId);
    if (!step) return { isValid: false, errors: {} };

    const errors = {};
    let isValid = true;

    step.fields.forEach(field => {
      const value = formData[field.name];
      const fieldErrors = validateField(field, value);

      if (field.name === 'confirmPassword') {
        if (formData.password !== formData.confirmPassword) {
          fieldErrors.push('पासवर्ड मेल नहीं खाते');
        }
      }

      if (fieldErrors.length > 0) {
        errors[field.name] = fieldErrors;
        isValid = false;
      }
    });

    return { isValid, errors };
  }

  // चरण HTML बनाएं
  function renderSteps() {
    const formSteps = document.getElementById('formSteps');
    const stepIndicators = document.getElementById('stepIndicators');
    const storedData = getStoredData();

    formSteps.innerHTML = '';
    stepIndicators.innerHTML = '';

    steps.forEach((step, index) => {
      // चरण सामग्री
      const stepDiv = document.createElement('div');
      stepDiv.className = `form-step ${index === 0 ? 'active' : ''}`;
      stepDiv.id = `step-${step.id}`;

      let fieldsHTML = `<h2 class="step-title">${step.title}</h2>`;

      step.fields.forEach(field => {
        const savedValue = storedData[field.name] || '';
        fieldsHTML += createFieldHTML(field, savedValue);
      });

      stepDiv.innerHTML = fieldsHTML;
      formSteps.appendChild(stepDiv);

      // चरण संकेतक
      const indicatorDiv = document.createElement('div');
      indicatorDiv.className = 'step-indicator';
      indicatorDiv.innerHTML = `
        <div class="step-circle ${index === 0 ? 'active' : ''}" id="circle-${step.id}">
          ${index + 1}
        </div>
        <span class="step-label ${index === 0 ? 'active' : ''}" id="label-${step.id}">
          ${step.title}
        </span>
      `;
      stepIndicators.appendChild(indicatorDiv);
    });
  }

  // फ़ील्ड HTML बनाएं
  function createFieldHTML(field, savedValue) {
    let inputHTML = '';

    switch (field.type) {
      case 'select':
        inputHTML = `
          <select name="${field.name}" id="${field.name}">
            <option value="">-- चुनें --</option>
            ${field.options.map(opt => 
              `<option value="${opt}" ${savedValue === opt ? 'selected' : ''}>${opt}</option>`
            ).join('')}
          </select>
        `;
        break;

      case 'textarea':
        inputHTML = `
          <textarea name="${field.name}" id="${field.name}" 
            placeholder="${field.label} दर्ज करें">${savedValue}</textarea>
        `;
        break;

      case 'checkbox':
        inputHTML = `
          <div class="checkbox-group">
            <input type="checkbox" name="${field.name}" id="${field.name}" 
              ${savedValue === 'true' ? 'checked' : ''}>
            <label for="${field.name}" style="margin-bottom:0">${field.label}</label>
          </div>
        `;
        break;

      default:
        inputHTML = `
          <input type="${field.type}" name="${field.name}" id="${field.name}" 
            value="${savedValue}" placeholder="${field.label} दर्ज करें">
        `;
    }

    return `
      <div class="form-group">
        ${field.type !== 'checkbox' ? `
          <label for="${field.name}">
            ${field.label}
            ${field.required ? '<span class="required">*</span>' : ''}
          </label>
        ` : ''}
        ${inputHTML}
        <div class="error-message" id="error-${field.name}"></div>
      </div>
    `;
  }

  // वर्तमान चरण का डेटा एकत्र करें
  function collectStepData(stepId) {
    const step = steps.find(s => s.id === stepId);
    if (!step) return {};

    const data = {};
    step.fields.forEach(field => {
      const element = document.getElementById(field.name);
      if (element) {
        if (field.type === 'checkbox') {
          data[field.name] = element.checked.toString();
        } else {
          data[field.name] = element.value;
        }
      }
    });

    return data;
  }

  // त्रुटियाँ दिखाएं
  function showErrors(errors) {
    // पहले सभी त्रुटियाँ साफ़ करें
    document.querySelectorAll('.error-message').forEach(el => {
      el.textContent = '';
      el.classList.remove('show');
    });
    document.querySelectorAll('input, select, textarea').forEach(el => {
      el.classList.remove('error');
    });

    // नई त्रुटियाँ दिखाएं
    Object.keys(errors).forEach(fieldName => {
      const errorEl = document.getElementById(`error-${fieldName}`);
      const inputEl = document.getElementById(fieldName);
      
      if (errorEl && errors[fieldName].length > 0) {
        errorEl.textContent = errors[fieldName][0];
        errorEl.classList.add('show');
      }
      
      if (inputEl) {
        inputEl.classList.add('error');
      }
    });
  }

  // प्रगति अपडेट करें
  function updateProgress(currentStep) {
    const totalSteps = steps.length;
    const progress = ((currentStep - 1) / (totalSteps - 1)) * 100;
    
    document.getElementById('progressBar').style.setProperty('--progress', `${progress}%`);

    steps.forEach((step, index) => {
      const circle = document.getElementById(`circle-${step.id}`);
      const label = document.getElementById(`label-${step.id}`);
      
      circle.classList.remove('active', 'completed');
      label.classList.remove('active', 'completed');

      if (step.id < currentStep) {
        circle.classList.add('completed');
        circle.innerHTML = '✓';
        label.classList.add('completed');
      } else if (step.id === currentStep) {
        circle.classList.add('active');
        circle.innerHTML = step.id;
        label.classList.add('active');
      } else {
        circle.innerHTML = step.id;
      }
    });
  }

  // ऑटो-सेव संकेतक दिखाएं
  function showAutoSaveIndicator() {
    const indicator = document.getElementById('autoSaveIndicator');
    indicator.classList.add('show');
    setTimeout(() => {
      indicator.classList.remove('show');
    }, 2000);
  }

  // नेविगेशन बटन अपडेट करें
  function updateNavigation(currentStep) {
    const prevBtn = document.getElementById('prevBtn');
    const nextBtn = document.getElementById('nextBtn');
    const submitBtn = document.getElementById('submitBtn');

    prevBtn.style.display = currentStep > 1 ? 'block' : 'none';
    
    if (currentStep === steps.length) {
      nextBtn.style.display = 'none';
      submitBtn.style.display = 'block';
    } else {
      nextBtn.style.display = 'block';
      submitBtn.style.display = 'none';
    }
  }

  // चरण बदलें
  function changeStep(direction) {
    const currentStep = getCurrentStep();
    const stepData = collectStepData(currentStep);
    
    if (direction === 'next') {
      const { isValid, errors } = validateStep(currentStep, stepData);
      
      if (!isValid) {
        showErrors(errors);
        return;
      }

      // डेटा सहेजें
      const saved = saveToStorage(stepData);
      if (saved) {
        showAutoSaveIndicator();
      }

      const nextStep = currentStep + 1;
      if (nextStep <= steps.length) {
        showStep(currentStep, nextStep);
      }
    } else if (direction === 'prev') {
      // पिछले चरण पर जाने से पहले भी डेटा सहेजें
      saveToStorage(stepData);
      
      const prevStep = currentStep - 1;
      if (prevStep >= 1) {
        showStep(currentStep, prevStep);
      }
    }
  }

  // चरण दिखाएं
  function showStep(fromStep, toStep) {
    document.getElementById(`step-${fromStep}`).classList.remove('active');
    document.getElementById(`step-${toStep}`).classList.add('active');
    
    localStorage.setItem(CURRENT_STEP_KEY, toStep);
    
    updateProgress(toStep);
    updateNavigation(toStep);

    // त्रुटियाँ साफ़ करें
    document.querySelectorAll('.error-message').forEach(el => {
      el.textContent = '';
      el.classList.remove('show');
    });
    document.querySelectorAll('input, select, textarea').forEach(el => {
      el.classList.remove('error');
    });
  }

  // फ़ॉर्म सबमिट करें
  function handleSubmit(event) {
    event.preventDefault();
    
    const currentStep = getCurrentStep();
    const stepData = collectStepData(currentStep);
    const { isValid, errors } = validateStep(currentStep, stepData);

    if (!isValid) {
      showErrors(errors);
      return;
    }

    // अंतिम डेटा सहेजें
    saveToStorage(stepData);

    // सभी डेटा एकत्र करें
    const finalData = getStoredData();
    
    console.log('=== पंजीकरण डेटा ===');
    console.log(finalData);

    // API कॉल का अनुकरण
    submitToServer(finalData);
  }
