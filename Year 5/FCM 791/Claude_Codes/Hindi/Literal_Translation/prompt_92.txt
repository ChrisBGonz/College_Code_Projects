import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.core.JsonProcessingException;

// User क्लास - JSON डेटा को मैप करने के लिए
class User {
    @JsonProperty("name")
    private String name;
    
    @JsonProperty("age")
    private int age;
    
    @JsonProperty("email")
    private String email;
    
    @JsonProperty("city")
    private String city;

    // Default Constructor (Jackson के लिए जरूरी)
    public User() {}

    // Parameterized Constructor
    public User(String name, int age, String email, String city) {
        this.name = name;
        this.age = age;
        this.email = email;
        this.city = city;
    }

    // Getters और Setters
    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getAge() {
        return age;
    }

    public void setAge(int age) {
        this.age = age;
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public String getCity() {
        return city;
    }

    public void setCity(String city) {
        this.city = city;
    }

    @Override
    public String toString() {
        return "User{" +
                "name='" + name + '\'' +
                ", age=" + age +
                ", email='" + email + '\'' +
                ", city='" + city + '\'' +
                '}';
    }
}

public class JsonParserExample {

    /**
     * JSON स्ट्रिंग को User ऑब्जेक्ट में परिवर्तित करने का मेथड
     * 
     * @param jsonString - उपयोगकर्ता द्वारा प्रदान की गई JSON स्ट्रिंग
     * @return User ऑब्जेक्ट
     */
    public static User parseJsonToUser(String jsonString) {
        // Input Validation - खाली या null स्ट्रिंग की जांच
        if (jsonString == null || jsonString.trim().isEmpty()) {
            System.out.println("❌ त्रुटि: JSON स्ट्रिंग खाली या null है!");
            return null;
        }

        ObjectMapper objectMapper = new ObjectMapper();
        
        try {
            // JSON स्ट्रिंग को User ऑब्जेक्ट में परिवर्तित करें
            User user = objectMapper.readValue(jsonString, User.class);
            
            System.out.println("✅ JSON सफलतापूर्वक पार्स किया गया!");
            System.out.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
            
            // user.name प्रॉपर्टी प्रिंट करें
            System.out.println("👤 उपयोगकर्ता का नाम (user.name): " + user.getName());
            System.out.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
            
            return user;
            
        } catch (JsonProcessingException e) {
            System.out.println("❌ JSON पार्सिंग में त्रुटि: " + e.getMessage());
            return null;
        }
    }

    /**
     * User ऑब्जेक्ट की सभी जानकारी प्रिंट करने का मेथड
     * 
     * @param user - User ऑब्जेक्ट
     */
    public static void printUserDetails(User user) {
        if (user == null) {
            System.out.println("❌ User ऑब्जेक्ट null है!");
            return;
        }
        
        System.out.println("\n📋 उपयोगकर्ता की पूरी जानकारी:");
        System.out.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
        System.out.println("नाम    : " + user.getName());
        System.out.println("आयु    : " + user.getAge());
        System.out.println("ईमेल   : " + user.getEmail());
        System.out.println("शहर    : " + user.getCity());
        System.out.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
    }

    /**
     * User ऑब्जेक्ट को वापस JSON स्ट्रिंग में परिवर्तित करने का मेथड
     * 
     * @param user - User ऑब्जेक्ट
     * @return JSON स्ट्रिंग
     */
    public static String convertUserToJson(User user) {
        if (user == null) {
            return null;
        }
        
        ObjectMapper objectMapper = new ObjectMapper();
        
        try {
            // Pretty Print के साथ JSON बनाएं
            String jsonOutput = objectMapper.writerWithDefaultPrettyPrinter()
                                           .writeValueAsString(user);
            return jsonOutput;
        } catch (JsonProcessingException e) {
            System.out.println("❌ JSON बनाने में त्रुटि: " + e.getMessage());
            return null;
        }
    }
