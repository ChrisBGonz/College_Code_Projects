const https = require("https");
const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function loadLibraryFromCDN(libraryName, version, filePath) {
  return new Promise((resolve, reject) => {
    const url = `https://cdnjs.cloudflare.com/ajax/libs/${libraryName}/${version}/${filePath}`;
    console.log(`Loading library from: ${url}`);

    https
      .get(url, (response) => {
        if (response.statusCode !== 200) {
          reject(
            new Error(
              `Failed to load library. Status code: ${response.statusCode}`
            )
          );
          return;
        }

        let data = "";
        response.on("data", (chunk) => {
          data += chunk;
        });

        response.on("end", () => {
          console.log(
            `Successfully loaded ${libraryName} v${version} (${data.length} bytes)`
          );
          resolve(data);
        });
      })
      .on("error", (error) => {
        reject(error);
      });
  });
}

async function analyzeLibraryWithClaude(libraryCode, libraryName) {
  console.log(`\nAnalyzing ${libraryName} with Claude AI...`);

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `यह ${libraryName} लाइब्रेरी का कोड है। कृपया इसके बारे में हिंदी में बताएं:
1. यह लाइब्रेरी क्या करती है?
2. इसकी मुख्य विशेषताएं क्या हैं?
3. इसका उपयोग कैसे किया जाता है?

कोड का पहला भाग (पहले 500 characters):
${libraryCode.substring(0, 500)}...`,
      },
    ],
  });

  console.log("\nClaude का विश्लेषण:");
  console.log("=".repeat(50));

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
    }
  }

  const finalMessage = await stream.finalMessage();
  console.log("\n" + "=".repeat(50));

  return finalMessage;
}
