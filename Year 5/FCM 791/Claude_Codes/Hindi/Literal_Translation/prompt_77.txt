using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Collections.Generic;

[Serializable]
public class ApplicationSettings
{
    public string ApplicationName { get; set; }
    public string Version { get; set; }
    public string Theme { get; set; }
    public string Language { get; set; }
    public bool AutoSave { get; set; }
    public int AutoSaveInterval { get; set; }
    public Dictionary<string, string> CustomSettings { get; set; }
    public DateTime LastModified { get; set; }

    public ApplicationSettings()
    {
        CustomSettings = new Dictionary<string, string>();
        LastModified = DateTime.Now;
    }

    public override string ToString()
    {
        return $"Application: {ApplicationName}\n" +
               $"Version: {Version}\n" +
               $"Theme: {Theme}\n" +
               $"Language: {Language}\n" +
               $"AutoSave: {AutoSave}\n" +
               $"AutoSave Interval: {AutoSaveInterval} minutes\n" +
               $"Last Modified: {LastModified}\n" +
               $"Custom Settings Count: {CustomSettings?.Count ?? 0}";
    }
}

public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    // Store settings using BinaryFormatter
    public bool SaveSettings(ApplicationSettings settings)
    {
        try
        {
            settings.LastModified = DateTime.Now;
            
            using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write))
            {
#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
                BinaryFormatter formatter = new BinaryFormatter();
                formatter.Serialize(fileStream, settings);
#pragma warning restore SYSLIB0011
            }
            
            Console.WriteLine($"Settings saved successfully to: {_settingsFilePath}");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            return false;
        }
    }

    // Retrieve settings using BinaryFormatter
    public ApplicationSettings LoadSettings()
    {
        try
        {
            if (!File.Exists(_settingsFilePath))
            {
                Console.WriteLine("Settings file not found. Creating default settings.");
                return CreateDefaultSettings();
            }

            using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read))
            {
#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
                BinaryFormatter formatter = new BinaryFormatter();
                ApplicationSettings settings = (ApplicationSettings)formatter.Deserialize(fileStream);
#pragma warning restore SYSLIB0011
                
                Console.WriteLine("Settings loaded successfully.");
                return settings;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            Console.WriteLine("Returning default settings.");
            return CreateDefaultSettings();
        }
    }

    // Update specific setting
    public bool UpdateSetting(string key, string value)
    {
        try
        {
            ApplicationSettings settings = LoadSettings();
            
            if (settings.CustomSettings == null)
            {
                settings.CustomSettings = new Dictionary<string, string>();
            }
            
            settings.CustomSettings[key] = value;
            return SaveSettings(settings);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error updating setting: {ex.Message}");
            return false;
        }
    }

    // Get specific custom setting
    public string GetCustomSetting(string key)
    {
        try
        {
            ApplicationSettings settings = LoadSettings();
            
            if (settings.CustomSettings != null && settings.CustomSettings.ContainsKey(key))
            {
                return settings.CustomSettings[key];
            }
            
            return null;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error getting setting: {ex.Message}");
            return null;
        }
    }

    // Delete settings file
    public bool DeleteSettings()
    {
        try
        {
            if (File.Exists(_settingsFilePath))
            {
                File.Delete(_settingsFilePath);
                Console.WriteLine("Settings file deleted successfully.");
                return true;
            }
            
            Console.WriteLine("Settings file does not exist.");
            return false;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error deleting settings: {ex.Message}");
            return false;
        }
    }

    // Check if settings file exists
    public bool SettingsExist()
    {
        return File.Exists(_settingsFilePath);
    }

    // Create backup of settings
    public bool BackupSettings(string backupPath = null)
    {
        try
        {
            if (!File.Exists(_settingsFilePath))
            {
                Console.WriteLine("No settings file to backup.");
                return false;
            }

            string backup = backupPath ?? $"{_settingsFilePath}.backup_{DateTime.Now:yyyyMMdd_HHmmss}";
            File.Copy(_settingsFilePath, backup, true);
            Console.WriteLine($"Settings backed up to: {backup}");
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error backing up settings: {ex.Message}");
            return false;
        }
    }

    // Create default settings
    private ApplicationSettings CreateDefaultSettings()
    {
        return new ApplicationSettings
        {
            ApplicationName = "MyApplication",
            Version = "1.0.0",
            Theme = "Light",
            Language = "Hindi",
            AutoSave = true,
            AutoSaveInterval = 5,
            CustomSettings = new Dictionary<string, string>
            {
                { "FontSize", "12" },
                { "WindowWidth", "800" },
                { "WindowHeight", "600" }
            }
        };
    }
}

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("=== Application Settings Manager ===\n");
        
        SettingsManager manager = new SettingsManager("my_app_settings.bin");
        
        // Create and save settings
        Console.WriteLine("1. Creating and saving application settings...");
        ApplicationSettings settings = new ApplicationSettings
        {
            ApplicationName = "Hindi Learning App",
            Version = "2.0.1",
            Theme = "Dark",
            Language = "Hindi",
            AutoSave = true,
            AutoSaveInterval = 10,
            CustomSettings = new Dictionary<string, string>
            {
                { "FontSize", "14" },
                { "WindowWidth", "1024" },
                { "WindowHeight", "768" },
                { "LastOpenedFile", "document.txt" },
                { "RecentFiles", "file1.txt,file2.txt,file3.txt" }
            }
        };
        
        bool saved = manager.SaveSettings(settings);
        Console.WriteLine($"Save Status: {(saved ? "Success" : "Failed")}\n");
        
        // Load settings
        Console.WriteLine("2. Loading settings from file...");
        ApplicationSettings loadedSettings = manager.LoadSettings();
        Console.WriteLine("\nLoaded Settings:");
        Console.WriteLine(loadedSettings.ToString());
        
        // Display custom settings
        Console.WriteLine("\nCustom Settings:");
        foreach (var kvp in loadedSettings.CustomSettings)
        {
            Console.WriteLine($"  {kvp.Key}: {kvp.Value}");
        }
        
        // Update a specific setting
        Console.WriteLine("\n3. Updating custom setting...");
        manager.UpdateSetting("FontSize", "16");
        manager.UpdateSetting("NewSetting", "NewValue");
        
        // Get specific setting
        Console.WriteLine("\n4. Getting specific settings...");
        string fontSize = manager.GetCustomSetting("FontSize");
        string newSetting = manager.GetCustomSetting("NewSetting");
        Console.WriteLine($"FontSize: {fontSize}");
        Console.WriteLine($"NewSetting: {newSetting}");
        
        // Backup settings
        Console.WriteLine("\n5. Creating backup of settings...");
        manager.BackupSettings();
        
        // Check if settings exist
        Console.WriteLine("\n6. Checking settings existence...");
        Console.WriteLine($"Settings exist: {manager.SettingsExist()}");
        
        // Load and display final settings
        Console.WriteLine("\n7. Final settings after all operations:");
        ApplicationSettings finalSettings = manager.LoadSettings();
        Console.WriteLine(finalSettings.ToString());
        Console.WriteLine("\nAll Custom Settings:");
        foreach (var kvp in finalSettings.CustomSettings)
        {
            Console.WriteLine($"  {kvp.Key}: {kvp.Value}");
        }
        
        Console.WriteLine("\n=== Settings Manager Demo Complete ===");
    }
}
