// CSRF Token Generation
function generateCSRFToken() {
    if (!isset($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

// CSRF Token Validation
function validateCSRFToken($token) {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// Log Activity
function logActivity($employeeId, $action, $filename, $status, $message) {
    global $logDir;
    $logFile = $logDir . 'upload_log_' . date('Y-m') . '.log';
    $timestamp = date('Y-m-d H:i:s');
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'Unknown';
    $logEntry = "[$timestamp] Employee: $employeeId | Action: $action | File: $filename | Status: $status | Message: $message | IP: $ip\n";
    file_put_contents($logFile, $logEntry, FILE_APPEND | LOCK_EX);
}

// Sanitize filename
function sanitizeFilename($filename) {
    // Remove path traversal attempts
    $filename = basename($filename);
    // Remove special characters
    $filename = preg_replace('/[^a-zA-Z0-9._-]/', '_', $filename);
    // Remove multiple dots (except for extension)
    $filename = preg_replace('/\.+/', '.', $filename);
    // Limit filename length
    if (strlen($filename) > 100) {
        $ext = pathinfo($filename, PATHINFO_EXTENSION);
        $filename = substr(pathinfo($filename, PATHINFO_FILENAME), 0, 95) . '.' . $ext;
    }
    return $filename;
}

// Validate PDF file
function validatePDFFile($file) {
    $errors = [];
    
    // Check if file was uploaded
    if (!isset($file) || $file['error'] === UPLOAD_ERR_NO_FILE) {
        $errors[] = 'कृपया एक फ़ाइल चुनें।';
        return $errors;
    }
    
    // Check upload errors
    switch ($file['error']) {
        case UPLOAD_ERR_INI_SIZE:
        case UPLOAD_ERR_FORM_SIZE:
            $errors[] = 'फ़ाइल का आकार बहुत बड़ा है। अधिकतम ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB की अनुमति है।';
            return $errors;
        case UPLOAD_ERR_PARTIAL:
            $errors[] = 'फ़ाइल आंशिक रूप से अपलोड हुई। कृपया पुनः प्रयास करें।';
            return $errors;
        case UPLOAD_ERR_NO_TMP_DIR:
            $errors[] = 'अस्थायी फ़ोल्डर नहीं मिला।';
            return $errors;
        case UPLOAD_ERR_CANT_WRITE:
            $errors[] = 'फ़ाइल लिखने में विफल।';
            return $errors;
        case UPLOAD_ERR_OK:
            break;
        default:
            $errors[] = 'अज्ञात अपलोड त्रुटि।';
            return $errors;
    }
    
    // Check file size
    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'फ़ाइल का आकार ' . number_format($file['size'] / 1024 / 1024, 2) . 'MB है। अधिकतम ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB की अनुमति है।';
    }
    
    // Check file extension
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if ($extension !== ALLOWED_EXTENSION) {
        $errors[] = 'केवल PDF फ़ाइलें अपलोड की जा सकती हैं। आपकी फ़ाइल का प्रकार: .' . $extension;
    }
    
    // Check MIME type
    if (!in_array($file['type'], ALLOWED_TYPES)) {
        $errors[] = 'अमान्य फ़ाइल प्रकार। केवल PDF फ़ाइलें स्वीकार की जाती हैं।';
    }
    
    // Verify actual file content (PDF magic bytes)
    if (is_uploaded_file($file['tmp_name'])) {
        $handle = fopen($file['tmp_name'], 'rb');
        if ($handle) {
            $header = fread($handle, 5);
            fclose($handle);
            if ($header !== '%PDF-') {
                $errors[] = 'फ़ाइल वास्तव में PDF नहीं है। फ़ाइल की सामग्री अमान्य है।';
            }
        }
    }
    
    return $errors;
}

// Get uploaded files list
function getUploadedFiles() {
    $files = [];
    if (is_dir(UPLOAD_DIR)) {
        $iterator = new DirectoryIterator(UPLOAD_DIR);
        foreach ($iterator as $file) {
            if ($file->isFile() && $file->getExtension() === 'pdf') {
                $files[] = [
                    'name' => $file->getFilename(),
                    'size' => $file->getSize(),
                    'modified' => $file->getMTime(),
                    'path' => $file->getPathname()
                ];
            }
        }
        // Sort by modification time (newest first)
        usort($files, function($a, $b) {
            return $b['modified'] - $a['modified'];
        });
    }
    return $files;
}

// Format file size
function formatFileSize($bytes) {
    if ($bytes >= 1048576) {
        return number_format($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return number_format($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' Bytes';
}

            }
        } else {
            logActivity($_SESSION['employee_id'], 'UPLOAD', $file['name'] ?? 'Unknown', 'REJECTED', implode(', ', $uploadErrors));
        }
    }
}
