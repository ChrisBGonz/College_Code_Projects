const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definitions for Claude
const tools = [
  {
    name: "download_file",
    description:
      "Downloads a file from a remote URL and saves it to a local path",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path where the downloaded file will be saved",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description: "Extracts a .zip file to a specified directory",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "Path to the .zip file to extract",
        },
        extract_to: {
          type: "string",
          description: "Directory path where the zip contents will be extracted",
        },
      },
      required: ["zip_path", "extract_to"],
    },
  },
  {
    name: "create_directory",
    description: "Creates a directory if it doesn't exist",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to create",
        },
      },
      required: ["path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Removes a temporary file after extraction",
    input_schema: {
      type: "object",
      properties: {
        file_path: {
          type: "string",
          description: "Path to the file to be deleted",
        },
      },
      required: ["file_path"],
    },
  },
];

// Tool implementation functions
function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    console.log(`Downloading from: ${url}`);
    console.log(`Saving to: ${destination}`);

    const protocol = url.startsWith("https") ? https : http;
    const file = fs.createWriteStream(destination);

    const request = protocol.get(url, (response) => {
      if (response.statusCode === 301 || response.statusCode === 302) {
        // Handle redirects
        file.close();
        fs.unlink(destination, () => {});
        downloadFile(response.headers.location, destination)
          .then(resolve)
          .catch(reject);
        return;
      }

      if (response.statusCode !== 200) {
        file.close();
        fs.unlink(destination, () => {});
        reject(
          new Error(
            `Failed to download: HTTP ${response.statusCode} - ${response.statusMessage}`
          )
        );
        return;
      }

      response.pipe(file);

      file.on("finish", () => {
        file.close();
        const stats = fs.statSync(destination);
        resolve({
          success: true,
          message: `File downloaded successfully`,
          size: stats.size,
          path: destination,
        });
      });
    });

    request.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });

    file.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });
  });
}

function extractZip(zipPath, extractTo) {
  try {
    console.log(`Extracting: ${zipPath}`);
    console.log(`Extracting to: ${extractTo}`);

    // Check if zip file exists
    if (!fs.existsSync(zipPath)) {
      return {
        success: false,
        error: `Zip file not found: ${zipPath}`,
      };
    }

    // Use unzip command (available on most Unix systems)
    // For Windows, you might need to use a different approach
    try {
      execSync(`unzip -o "${zipPath}" -d "${extractTo}"`, {
        stdio: "pipe",
      });
    } catch {
      // Try with 7z if unzip is not available
      try {
        execSync(`7z x "${zipPath}" -o"${extractTo}" -y`, {
          stdio: "pipe",
        });
      } catch {
        // Try PowerShell on Windows
        execSync(
          `powershell -command "Expand-Archive -Path '${zipPath}' -DestinationPath '${extractTo}' -Force"`,
          { stdio: "pipe" }
        );
      }
    }

    // List extracted files
    const extractedFiles = fs.readdirSync(extractTo);

    return {
      success: true,
      message: `Zip extracted successfully`,
      extractedTo: extractTo,
      files: extractedFiles,
    };
  } catch (error) {
    return {
      success: false,
      error: `Extraction failed: ${error.message}`,
    };
  }
}

function createDirectory(dirPath) {
  try {
    if (!fs.existsSync(dirPath)) {
      fs.mkdirSync(dirPath, { recursive: true });
      return {
        success: true,
        message: `Directory created: ${dirPath}`,
        created: true,
      };
    } else {
      return {
        success: true,
        message: `Directory already exists: ${dirPath}`,
        created: false,
      };
    }
  } catch (error) {
    return {
      success: false,
      error: `Failed to create directory: ${error.message}`,
    };
  }
}

function cleanupFile(filePath) {
  try {
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
      return {
        success: true,
        message: `File deleted: ${filePath}`,
      };
    } else {
      return {
        success: true,
        message: `File not found (already deleted): ${filePath}`,
      };
    }
  } catch (error) {
    return {
      success: false,
      error: `Failed to delete file: ${error.message}`,
    };
  }
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Tool input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      result = await downloadFile(toolInput.url, toolInput.destination);
      break;
    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.extract_to);
      break;
    case "create_directory":
      result = createDirectory(toolInput.path);
      break;
    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;
    default:
      result = { error: `Unknown tool: ${toolName}` };
  }

  console.log("Tool result:", JSON.stringify(result, null, 2));
  return result;
}

// Main function to download and extract zip update
async function downloadAndExtractUpdate(updateUrl, appDirectory = "/app") {
  console.log("=".repeat(60));
  console.log("ZIP Update Downloader and Extractor");
  console.log("=".repeat(60));
  console.log(`Update URL: ${updateUrl}`);
  console.log(`App Directory: ${appDirectory}`);
  console.log("=".repeat(60));

  const messages = [
    {
      role: "user",
      content: `Please help me download a .zip update file from the following URL and extract it to the /app directory:

URL: ${updateUrl}
Target Directory: ${appDirectory}

Please:
1. First create the target directory if it doesn't exist
2. Download the zip file to a temporary location (use /tmp/update.zip)
3. Extract the zip file to ${appDirectory}
4. Clean up the temporary zip file after extraction
5. Report the results

Use the available tools to complete this task.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log("\n" + "=".repeat(60));
    console.log("Claude Response:");
    console.log(`Stop Reason: ${response.stop_reason}`);

    // Process the response content
    const assistantMessage = {
      role: "assistant",
      content: response.content,
    };
    messages.push(assistantMessage);

    if (response.stop_reason === "tool_use") {
      const toolResults = [];

      for (const block of response.content) {
        if (block.type === "text") {
          console.log("\nClaude says:", block.text);
        } else if (block.type === "tool_use") {
          console.log(`\nTool call: ${block.name}`);

          const result = await processToolCall(block.name, block.input);

          toolResults.push({
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          });
        }
      }

      // Add tool results to messages
      messages.push({
        role: "user",
        content: toolResults,
      });
    } else if (response.stop_reason === "end_turn") {
      // Final response
      for (const block of response.content) {
        if (block.type === "text") {
          console.log("\nFinal Result:");
          console.log(block.text);
        }
      }
      continueLoop = false;
    } else {
      console.log("Unexpected stop reason:", response.stop_reason);
      continueLoop = false;
    }
  }

  console.log("\n" + "=".repeat(60));
  console.log("Update process completed!");
  console.log("=".repeat(60));
}

async function main() {
  // Example: Download a sample zip file
  // Using a reliable test zip file
  const testUrl =

  const appDir = path.join(process.cwd(), "app"); // Using current directory for testing

  try {
    await downloadAndExtractUpdate(testUrl, appDir);
