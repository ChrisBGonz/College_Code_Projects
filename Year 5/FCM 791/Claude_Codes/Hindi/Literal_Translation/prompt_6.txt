class AuthService:
    """प्रमाणीकरण सेवा"""
    
    def __init__(self, db: Database):
        self.db = db
    
    def _hash_password(self, password: str) -> str:
        return hashlib.sha256(password.encode()).hexdigest()
    
    def login(self, username: str, password: str) -> Optional[Dict]:
        """उपयोगकर्ता लॉगिन"""
        password_hash = self._hash_password(password)
        
        # उपयोगकर्ता खोजें
        user = None
        for u in self.db.users.values():
            if u.username == username and u.password_hash == password_hash:
                user = u
                break
        
        if not user:
            return None
        
        if not user.is_active:
            return None
        
        # Session बनाएं
        session_token = secrets.token_hex(32)
        session_data = {
            "token": session_token,
            "user_id": user.user_id,
            "company_id": user.company_id,
            "role": user.role.value,
            "created_at": datetime.datetime.now().isoformat(),
            "expires_at": (datetime.datetime.now() + datetime.timedelta(hours=8)).isoformat()
        }
        
        self.db.sessions[session_token] = session_data
        
        # Last login अपडेट करें
        user.last_login = datetime.datetime.now().isoformat()
        
        return session_data
    
    def logout(self, session_token: str) -> bool:
        """उपयोगकर्ता लॉगआउट"""
        if session_token in self.db.sessions:
            del self.db.sessions[session_token]
            return True
        return False
    
    def validate_session(self, session_token: str) -> Optional[Dict]:
        """Session वैलिडेट करें"""
        if session_token not in self.db.sessions:
            return None
        
        session = self.db.sessions[session_token]
        expires_at = datetime.datetime.fromisoformat(session["expires_at"])
        
        if datetime.datetime.now() > expires_at:
            del self.db.sessions[session_token]
            return None
        
        return session
    
    def get_current_user(self, session_token: str) -> Optional[User]:
        """वर्तमान उपयोगकर्ता प्राप्त करें"""
        session = self.validate_session(session_token)
        if not session:
            return None
        return self.db.users.get(session["user_id"])

class ReportService:
    """रिपोर्ट सेवा"""
    
    def __init__(self, db: Database):
        self.db = db
    
    def get_company_reports(
        self, 
        company_id: str, 
        user_role: UserRole,
        year: Optional[int] = None,
        status: Optional[ReportStatus] = None
    ) -> List[AnnualReport]:
        """कंपनी की रिपोर्ट्स प्राप्त करें"""
        
        reports = []
        for report in self.db.reports.values():
            # Tenant Isolation - केवल अपनी कंपनी की रिपोर्ट
            if report.company_id != company_id:
                continue
            
            # Role-based access
            if user_role == UserRole.VIEWER and report.status == ReportStatus.DRAFT:
                continue
            
            # Year filter
            if year and report.year != year:
                continue
            
            # Status filter
            if status and report.status != status:
                continue
            
            reports.append(report)
        
        # Year के अनुसार sort करें (नवीनतम पहले)
        reports.sort(key=lambda r: r.year, reverse=True)
        return reports
    
    def get_report_by_id(
        self, 
        report_id: str, 
        company_id: str,
        user_role: UserRole
    ) -> Optional[AnnualReport]:
        """ID से रिपोर्ट प्राप्त करें"""
        
        report = self.db.reports.get(report_id)
        if not report:
            return None
        
        # Tenant Isolation
        if report.company_id != company_id:
            return None
        
        # Draft केवल Admin और Manager देख सकते हैं
        if report.status == ReportStatus.DRAFT and user_role == UserRole.VIEWER:
            return None
        
        return report
    
    def create_report(
        self,
        company_id: str,
        user_id: str,
        user_role: UserRole,
        report_data: Dict
    ) -> Optional[AnnualReport]:
        """नई रिपोर्ट बनाएं"""
        
        # केवल Admin और Manager रिपोर्ट बना सकते हैं
        if user_role == UserRole.VIEWER:
            return None
        
        report_id = f"RPT{len(self.db.reports) + 1:03d}"
        
        financial = FinancialData(**report_data["financial_data"])
        departments = [DepartmentReport(**d) for d in report_data.get("departments", [])]
        
        report = AnnualReport(
            report_id=report_id,
            company_id=company_id,
            year=report_data["year"],
            title=report_data["title"],
            status=ReportStatus.DRAFT,
            financial_data=financial,
            departments=departments,
            executive_summary=report_data.get("executive_summary", ""),
            key_highlights=report_data.get("key_highlights", []),
            challenges=report_data.get("challenges", []),
            future_goals=report_data.get("future_goals", []),
            created_by=user_id
        )
        
        self.db.reports[report_id] = report
        return report
    
    def publish_report(
        self,
        report_id: str,
        company_id: str,
        user_role: UserRole
    ) -> bool:
        """रिपोर्ट प्रकाशित करें"""
        
        # केवल Admin प्रकाशित कर सकते हैं
        if user_role != UserRole.ADMIN:
            return False
        
        report = self.db.reports.get(report_id)
        if not report or report.company_id != company_id:
            return False
        
        report.status = ReportStatus.PUBLISHED
        report.published_at = datetime.datetime.now().isoformat()
        report.updated_at = datetime.datetime.now().isoformat()
        return True
    
    def get_report_summary(self, report: AnnualReport) -> Dict:
        """रिपोर्ट का सारांश प्राप्त करें"""
        return {
            "report_id": report.report_id,
            "year": report.year,
            "title": report.title,
            "status": report.status.value,
            "revenue": report.financial_data.revenue,
            "net_profit": report.financial_data.net_profit,
            "growth_rate": report.financial_data.growth_rate,
            "departments_count": len(report.departments),
            "created_at": report.created_at
        }

class CompanyService:
    """कंपनी सेवा"""
    
    def __init__(self, db: Database):
        self.db = db
    
    def get_company(self, company_id: str) -> Optional[Company]:
        """कंपनी जानकारी प्राप्त करें"""
        return self.db.companies.get(company_id)
    
    def get_company_users(self, company_id: str) -> List[User]:
        """कंपनी के उपयोगकर्ता प्राप्त करें"""
        return [u for u in self.db.users.values() if u.company_id == company_id]
    
    def add_user(
        self,
        company_id: str,
        admin_role: UserRole,
        user_data: Dict
    ) -> Optional[User]:
        """नया उपयोगकर्ता जोड़ें"""
        
        if admin_role != UserRole.ADMIN:
            return None
        
        company = self.db.companies.get(company_id)
        if not company:
            return None
        
        # Max users check
        current_users = len(self.get_company_users(company_id))
        if current_users >= company.max_users:
            return None
        
        user_id = f"USR{len(self.db.users) + 1:03d}"
        password_hash = hashlib.sha256(user_data["password"].encode()).hexdigest()
        
        user = User(
            user_id=user_id,
            username=user_data["username"],
            email=user_data["email"],
            password_hash=password_hash,
            role=UserRole(user_data["role"]),
            company_id=company_id
        )
        
        self.db.users[user_id] = user
        return user

class DisplayUtils:
    """प्रदर्शन उपयोगिताएं"""
    
    @staticmethod
    def format_currency(amount: float) -> str:
        """राशि को भारतीय प्रारूप में दिखाएं"""
        if amount >= 1_00_00_000:
            return f"₹{amount/1_00_00_000:.2f} करोड़"
        elif amount >= 1_00_000:
            return f"₹{amount/1_00_000:.2f} लाख"
        else:
            return f"₹{amount:,.0f}"
    
    @staticmethod
    def print_separator(char: str = "═", width: int = 70):
        print(char * width)
    
    @staticmethod
    def print_header(title: str, width: int = 70):
        DisplayUtils.print_separator("═", width)
        print(f"{'  ' + title:^{width}}")
        DisplayUtils.print_separator("═", width)
    
    @staticmethod
    def print_section(title: str, width: int = 70):
        print(f"\n{'─' * width}")
        print(f"  📋 {title}")
        print(f"{'─' * width}")
    
    @staticmethod
    def display_company_info(company: Company):
        """कंपनी जानकारी दिखाएं"""
        DisplayUtils.print_section("कंपनी जानकारी")
        print(f"  🏢 कंपनी नाम    : {company.company_name}")
        print(f"  🆔 कंपनी ID     : {company.company_id}")
        print(f"  🏭 उद्योग       : {company.industry}")
        print(f"  📦 प्लान        : {company.plan.value.upper()}")
        print(f"  👥 अधिकतम Users : {company.max_users}")
        print(f"  ✅ स्थिति       : {'सक्रिय' if company.is_active else 'निष्क्रिय'}")
