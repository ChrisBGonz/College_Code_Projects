// Database configuration
define('DB_HOST', 'localhost');
define('DB_USER', 'root');
define('DB_PASS', '');
define('DB_NAME', 'user_auth');

// Create database connection
function getDBConnection() {
    $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
    if ($conn->connect_error) {
        die("Connection failed: " . $conn->connect_error);
    }
    return $conn;
}

// Security functions
function sanitizeInput($data) {
    return htmlspecialchars(strip_tags(trim($data)));
}

function generateCSRFToken() {
    if (!isset($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCSRFToken($token) {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

function generateResetToken() {
    return bin2hex(random_bytes(32));
}

// Rate limiting
function checkRateLimit($identifier) {
    if (!isset($_SESSION['rate_limit'])) {
        $_SESSION['rate_limit'] = [];
    }
    
    $key = 'recovery_' . md5($identifier);
    $currentTime = time();
    $windowTime = 900; // 15 minutes
    $maxAttempts = 5;
    
    if (!isset($_SESSION['rate_limit'][$key])) {
        $_SESSION['rate_limit'][$key] = ['count' => 0, 'first_attempt' => $currentTime];
    }
    
    $rateData = &$_SESSION['rate_limit'][$key];
    
    // Reset window if expired
    if ($currentTime - $rateData['first_attempt'] > $windowTime) {
        $rateData = ['count' => 0, 'first_attempt' => $currentTime];
    }
    
    if ($rateData['count'] >= $maxAttempts) {
        $remainingTime = $windowTime - ($currentTime - $rateData['first_attempt']);
        return ['allowed' => false, 'remaining_time' => $remainingTime];
    }
    
    $rateData['count']++;
    return ['allowed' => true, 'remaining_time' => 0];
}

// Step 1: Find user by username/email
function findUser($identifier) {
    $conn = getDBConnection();
    $stmt = $conn->prepare("SELECT id, username, email, security_question, security_answer_hash FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param("ss", $identifier, $identifier);
    $stmt->execute();
    $result = $stmt->get_result();
    $user = $result->fetch_assoc();
    $stmt->close();
    $conn->close();
    return $user;
}

// Step 2: Verify security answer
function verifySecurityAnswer($userId, $answer) {
    $conn = getDBConnection();
    $stmt = $conn->prepare("SELECT security_answer_hash FROM users WHERE id = ?");
    $stmt->bind_param("i", $userId);
    $stmt->execute();
    $result = $stmt->get_result();
    $user = $result->fetch_assoc();
    $stmt->close();
    $conn->close();
    
    if (!$user) {
        return false;
    }
    
    // Normalize answer: lowercase and trim
    $normalizedAnswer = strtolower(trim($answer));
    return password_verify($normalizedAnswer, $user['security_answer_hash']);
}

// Step 3: Create password reset token
function createResetToken($userId) {
    $conn = getDBConnection();
    $token = generateResetToken();
    $hashedToken = hash('sha256', $token);
    $expiry = date('Y-m-d H:i:s', time() + 3600); // 1 hour expiry
    
    // Delete existing tokens for this user
    $stmt = $conn->prepare("DELETE FROM password_reset_tokens WHERE user_id = ?");
    $stmt->bind_param("i", $userId);
    $stmt->execute();
    $stmt->close();
    
    // Insert new token
    $stmt = $conn->prepare("INSERT INTO password_reset_tokens (user_id, token_hash, expires_at, created_at) VALUES (?, ?, ?, NOW())");
    $stmt->bind_param("iss", $userId, $hashedToken, $expiry);
    $stmt->execute();
    $stmt->close();
    $conn->close();
    
    return $token;
}

// Step 4: Reset password
function resetPassword($token, $newPassword) {
    $conn = getDBConnection();
    $hashedToken = hash('sha256', $token);
    
    $stmt = $conn->prepare("SELECT user_id, expires_at FROM password_reset_tokens WHERE token_hash = ? AND used = 0");
    $stmt->bind_param("s", $hashedToken);
    $stmt->execute();
    $result = $stmt->get_result();
    $tokenData = $result->fetch_assoc();
    $stmt->close();
    
    if (!$tokenData) {
        $conn->close();
        return ['success' => false, 'message' => 'अमान्य या समाप्त टोकन।'];
    }
    
    if (strtotime($tokenData['expires_at']) < time()) {
        $conn->close();
        return ['success' => false, 'message' => 'रीसेट टोकन की समय-सीमा समाप्त हो गई है।'];
    }
    
    // Validate password strength
    $passwordValidation = validatePasswordStrength($newPassword);
    if (!$passwordValidation['valid']) {
        $conn->close();
        return ['success' => false, 'message' => $passwordValidation['message']];
    }
    
    $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
    
    // Update password
    $stmt = $conn->prepare("UPDATE users SET password_hash = ?, updated_at = NOW() WHERE id = ?");
    $stmt->bind_param("si", $hashedPassword, $tokenData['user_id']);
    $stmt->execute();
    $stmt->close();
    
    // Mark token as used
    $stmt = $conn->prepare("UPDATE password_reset_tokens SET used = 1 WHERE token_hash = ?");
    $stmt->bind_param("s", $hashedToken);
    $stmt->execute();
    $stmt->close();
    
    // Log the password reset
    logActivity($tokenData['user_id'], 'password_reset', 'Password reset via security question');
    
    $conn->close();
    return ['success' => true, 'message' => 'पासवर्ड सफलतापूर्वक रीसेट किया गया।'];
}

function validatePasswordStrength($password) {
    if (strlen($password) < 8) {
        return ['valid' => false, 'message' => 'पासवर्ड कम से कम 8 अक्षर का होना चाहिए।'];
    }
    if (!preg_match('/[A-Z]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक बड़ा अक्षर होना चाहिए।'];
    }
    if (!preg_match('/[a-z]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक छोटा अक्षर होना चाहिए।'];
    }
    if (!preg_match('/[0-9]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक अंक होना चाहिए।'];
    }
    if (!preg_match('/[^A-Za-z0-9]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक विशेष अक्षर होना चाहिए।'];
    }
    return ['valid' => true, 'message' => ''];
}

function logActivity($userId, $action, $details) {
    $conn = getDBConnection();
    $ipAddress = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $userAgent = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
    
    $stmt = $conn->prepare("INSERT INTO activity_logs (user_id, action, details, ip_address, user_agent, created_at) VALUES (?, ?, ?, ?, ?, NOW())");
    $stmt->bind_param("issss", $userId, $action, $details, $ipAddress, $userAgent);
    $stmt->execute();
    $stmt->close();
    $conn->close();
}
