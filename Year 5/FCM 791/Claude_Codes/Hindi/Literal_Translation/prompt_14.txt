from flask import Flask, request, session, redirect, url_for, jsonify, make_response
import secrets
import hashlib
from datetime import timedelta

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)
app.permanent_session_lifetime = timedelta(hours=1)

# Simulated user database
USERS_DB = {
    "admin": hashlib.sha256("admin123".encode()).hexdigest(),
    "user1": hashlib.sha256("password1".encode()).hexdigest(),
    "user2": hashlib.sha256("mypassword".encode()).hexdigest()
}

def hash_password(password):
    """Hash a password using SHA-256"""
    return hashlib.sha256(password.encode()).hexdigest()

def generate_auth_token():
    """Generate a secure authentication token"""
    return secrets.token_hex(32)

@app.route('/')
def home():
    """Home page - check if user is logged in"""
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and 'user' in session and session.get('auth_token') == auth_token:
        return jsonify({
            "status": "success",
            "message": f"Welcome back, {session['user']}!",
            "authenticated": True,
            "auth_token": auth_token[:10] + "..." 
        })
    
    return jsonify({
        "status": "info",
        "message": "Please login to access the application",
        "authenticated": False,
        "login_url": "/login"
    })

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login endpoint"""
    if request.method == 'GET':
        return jsonify({
            "status": "info",
            "message": "Send POST request with username and password",
            "example": {
                "username": "admin",
                "password": "admin123"
            },
            "available_users": list(USERS_DB.keys())
        })
    
    if request.method == 'POST':
        data = request.get_json()
        
        if not data:
            return jsonify({
                "status": "error",
                "message": "Please provide JSON data with username and password"
            }), 400
        
        username = data.get('username', '').strip()
        password = data.get('password', '')
        
        if not username or not password:
            return jsonify({
                "status": "error",
                "message": "Username and password are required"
            }), 400
        
        if username not in USERS_DB:
            return jsonify({
                "status": "error",
                "message": "Invalid username or password"
            }), 401
        
        hashed_password = hash_password(password)
        if USERS_DB[username] != hashed_password:
            return jsonify({
                "status": "error",
                "message": "Invalid username or password"
            }), 401
        
        auth_token = generate_auth_token()
        
        session.permanent = True
        session['user'] = username
        session['auth_token'] = auth_token
        
        response = make_response(jsonify({
            "status": "success",
            "message": f"Login successful! Welcome, {username}!",
            "user": username,
            "auth_token_preview": auth_token[:10] + "...",
            "note": "auth_token cookie has been set in your browser"
        }))
        
        response.set_cookie(
            'auth_token',
            auth_token,
            max_age=3600,
            httponly=True,
            secure=False,
            samesite='Lax'
        )
        
        return response, 200

@app.route('/logout')
def logout():
    """Logout endpoint - clear session and cookie"""
    username = session.get('user', 'Unknown')
    
    session.clear()
    
    response = make_response(jsonify({
        "status": "success",
        "message": f"Goodbye, {username}! You have been logged out successfully.",
        "authenticated": False
    }))
    
    response.delete_cookie('auth_token')
    
    return response, 200

@app.route('/profile')
def profile():
    """Protected profile endpoint"""
    auth_token = request.cookies.get('auth_token')
    
    if not auth_token:
        return jsonify({
            "status": "error",
            "message": "Authentication required. Please login first.",
            "login_url": "/login"
        }), 401
    
    if 'user' not in session or session.get('auth_token') != auth_token:
        response = make_response(jsonify({
            "status": "error",
            "message": "Invalid or expired session. Please login again.",
            "login_url": "/login"
        }))
        response.delete_cookie('auth_token')
        return response, 401
    
    return jsonify({
        "status": "success",
        "message": "Profile accessed successfully",
        "user": {
            "username": session['user'],
            "auth_token_preview": auth_token[:10] + "...",
            "session_active": True
        }
    }), 200

@app.route('/dashboard')
def dashboard():
    """Protected dashboard endpoint"""
    auth_token = request.cookies.get('auth_token')
    
    if not auth_token or 'user' not in session or session.get('auth_token') != auth_token:
        return jsonify({
            "status": "error",
            "message": "Please login to access the dashboard",
            "login_url": "/login"
        }), 401
    
    return jsonify({
        "status": "success",
        "message": "Dashboard accessed successfully",
        "dashboard_data": {
            "username": session['user'],
            "total_users": len(USERS_DB),
            "session_info": {
                "auth_token_set": True,
                "token_preview": auth_token[:10] + "..."
            }
        }
    }), 200

@app.route('/check-auth')
def check_auth():
    """Check authentication status"""
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and 'user' in session and session.get('auth_token') == auth_token:
        return jsonify({
            "authenticated": True,
            "user": session['user'],
            "auth_token_present": True,
            "token_preview": auth_token[:10] + "..."
        }), 200
    
    return jsonify({
        "authenticated": False,
        "auth_token_present": bool(auth_token),
        "message": "Not authenticated"
    }), 200

if __name__ == '__main__':
    print("=" * 60)
    print("Flask Authentication App with Session Cookie")
    print("=" * 60)
    print("\nAvailable endpoints:")
    print("  GET  /          - Home page")
    print("  GET  /login     - Login form info")
    print("  POST /login     - Login with credentials")
    print("  GET  /logout    - Logout")
    print("  GET  /profile   - Protected profile page")
    print("  GET  /dashboard - Protected dashboard")
    print("  GET  /check-auth - Check authentication status")
    print("\nTest credentials:")
    print("  Username: admin, Password: admin123")

    print("  Username: user2, Password: mypassword")
    print("\nExample login command:")

    print("=" * 60)
    
    app.run(debug=True, host='0.0.0.0', port=5000)
