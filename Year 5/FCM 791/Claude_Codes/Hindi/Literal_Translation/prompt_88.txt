using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class MFASettingsLogger
{
    private static AnthropicClient _client = new AnthropicClient();
    
    // Simulated MFA settings change log
    private static List<string> _mfaChangeLogs = new List<string>();
    
    static async Task Main(string[] args)
    {
        Console.WriteLine("MFA Settings Change Logger with Claude AI");
        Console.WriteLine("==========================================\n");
        
        // Simulate some MFA settings changes
        await SimulateMFAChanges();
        
        // Use Claude to analyze and provide insights on the MFA changes
        await AnalyzeMFAChangesWithClaude();
    }
    
    static async Task SimulateMFAChanges()
    {
        // Simulate various MFA settings changes
        await LogMFAChange("user123", "enabled", "TOTP", "192.168.1.100", "Chrome/Windows");
        await Task.Delay(100);
        
        await LogMFAChange("user456", "disabled", "SMS", "10.0.0.5", "Firefox/Linux");
        await Task.Delay(100);
        
        await LogMFAChange("admin001", "changed", "Email OTP to TOTP", "172.16.0.1", "Safari/MacOS");
        await Task.Delay(100);
        
        await LogMFAChange("user789", "backup_codes_regenerated", "TOTP", "192.168.2.50", "Edge/Windows");
    }
    
    static async Task LogMFAChange(string userId, string action, string mfaType, string ipAddress, string userAgent)
    {
        string timestamp = DateTime.UtcNow.ToString("yyyy-MM-dd HH:mm:ss UTC");
        string logEntry = $"[{timestamp}] User: {userId} | Action: {action} | MFA Type: {mfaType} | IP: {ipAddress} | Agent: {userAgent}";
        
        _mfaChangeLogs.Add(logEntry);
        Console.WriteLine($"Logged: {logEntry}");
        
        // In a real application, you would also:
        // 1. Write to a secure audit log database
        // 2. Send notifications to the user
        // 3. Alert security team for suspicious changes
        await Task.CompletedTask;
    }
    
    static async Task AnalyzeMFAChangesWithClaude()
    {
        Console.WriteLine("\n\nAnalyzing MFA Changes with Claude AI...");
        Console.WriteLine("========================================\n");
        
        string logsForAnalysis = string.Join("\n", _mfaChangeLogs);
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"You are a security analyst. Please analyze the following MFA (Multi-Factor Authentication) settings change logs and provide:
1. A summary of all changes made
2. Any potential security concerns or suspicious patterns
3. Recommendations for improving security

MFA Change Logs:
{logsForAnalysis}

Please provide a detailed security analysis in Hindi language as this is for a Hindi-speaking security team."
                    }
                }
            }
        };
        
        Console.WriteLine("Sending logs to Claude for security analysis...\n");
        
        // Use streaming to get the response
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Stream = true
        };
        
        Console.WriteLine("Claude's Security Analysis:");
        Console.WriteLine("---------------------------");
        
        await foreach (var streamEvent in _client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                if (deltaEvent.Delta is TextDelta textDelta)
                {
                    Console.Write(textDelta.Text);
                }
            }
        }
        
        Console.WriteLine("\n\n---------------------------");
        Console.WriteLine("Analysis complete!");
        
        // Additional: Ask Claude for specific recommendations
        await GetSpecificRecommendations(logsForAnalysis);
    }
    
    static async Task GetSpecificRecommendations(string logs)
    {
        Console.WriteLine("\n\nGetting Specific Security Recommendations...");
        Console.WriteLine("============================================\n");
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Based on these MFA change logs, what are the top 3 immediate actions the security team should take?

Logs:
{logs}

Please respond in Hindi with specific, actionable recommendations."
                    }
                }
            }
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 512,
            Stream = true
        };
        
        Console.WriteLine("Top 3 Immediate Actions (from Claude):");
        Console.WriteLine("--------------------------------------");
        
        await foreach (var streamEvent in _client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                if (deltaEvent.Delta is TextDelta textDelta)
                {
                    Console.Write(textDelta.Text);
                }
            }
        }
        
        Console.WriteLine("\n");
    }
}
