package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
)

type AdminDashboardRequest struct {
	RefererHeader string
	UserID        string
	RequestPath   string
}

type AccessDecision struct {
	Granted bool
	Reason  string
	Message string
}

func checkAdminAccess(client *anthropic.Client, request AdminDashboardRequest) (*AccessDecision, error) {
	prompt := fmt.Sprintf(`You are a security access control system for an internal admin dashboard.

Analyze the following HTTP request details and determine if access should be granted:

Request Details:
- Referer Header: "%s"
- User ID: "%s"
- Requested Path: "%s"

Access Rules:
1. Access is GRANTED if the Referer header comes from "admin.internal.com"
2. Access is DENIED if the Referer header is missing, empty, or from any other domain
3. The Referer must exactly match "admin.internal.com" or be a subdomain/path of it (e.g., "https://admin.internal.com/dashboard")

Please analyze and respond in the following format:
DECISION: [GRANTED/DENIED]
REASON: [Brief explanation]
MESSAGE: [User-friendly message to display]`, 
		request.RefererHeader, 
		request.UserID, 
		request.RequestPath)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})
	if err != nil {
		return nil, fmt.Errorf("error calling Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return nil, fmt.Errorf("empty response from Claude API")
	}

	responseText := message.Content[0].Text
	
	decision := parseAccessDecision(responseText)
	return decision, nil
}

func parseAccessDecision(response string) *AccessDecision {
	decision := &AccessDecision{}
	
	lines := splitLines(response)
	for _, line := range lines {
		if len(line) > 10 && line[:9] == "DECISION:" {
			decisionValue := trimSpace(line[9:])
			decision.Granted = decisionValue == "GRANTED"
		} else if len(line) > 7 && line[:7] == "REASON:" {
			decision.Reason = trimSpace(line[7:])
		} else if len(line) > 8 && line[:8] == "MESSAGE:" {
			decision.Message = trimSpace(line[8:])
		}
	}
	
	return decision
}

func splitLines(s string) []string {
	var lines []string
	start := 0
	for i := 0; i < len(s); i++ {
		if s[i] == '\n' {
			lines = append(lines, s[start:i])
			start = i + 1
		}
	}
	if start < len(s) {
		lines = append(lines, s[start:])
	}
	return lines
}

func trimSpace(s string) string {
	start := 0
	end := len(s)
	for start < end && (s[start] == ' ' || s[start] == '\t' || s[start] == '\r' || s[start] == '\n') {
		start++
	}
	for end > start && (s[end-1] == ' ' || s[end-1] == '\t' || s[end-1] == '\r' || s[end-1] == '\n') {
		end--
	}
	return s[start:end]
}

func handleAdminDashboardRequest(client *anthropic.Client, request AdminDashboardRequest) {
	fmt.Printf("\n=== Admin Dashboard Access Request ===\n")
	fmt.Printf("User ID: %s\n", request.UserID)
	fmt.Printf("Referer: %s\n", request.RefererHeader)
	fmt.Printf("Path: %s\n", request.RequestPath)
	fmt.Println("--------------------------------------")
	
	decision, err := checkAdminAccess(client, request)
	if err != nil {
		log.Printf("Error checking access: %v", err)
		fmt.Println("Access DENIED: System error occurred")
		return
	}
	
	if decision.Granted {
		fmt.Printf("✅ Access GRANTED\n")
	} else {
		fmt.Printf("❌ Access DENIED\n")
	}
	
	if decision.Reason != "" {
		fmt.Printf("Reason: %s\n", decision.Reason)
	}
	
	if decision.Message != "" {
		fmt.Printf("Message: %s\n", decision.Message)
	}
}

			RequestPath:   "/admin/reports",
		},
		{
			RefererHeader: "https://admin.internal.com",
			UserID:        "admin001",
			RequestPath:   "/admin/security",
		},
		{
			RefererHeader: "https://fake-admin.internal.com.evil.com",

	}
	
	fmt.Println("Internal Admin Dashboard Access Control System")
	fmt.Println("==============================================")
	
	for _, request := range testRequests {
		handleAdminDashboardRequest(client, request)
	}
}
