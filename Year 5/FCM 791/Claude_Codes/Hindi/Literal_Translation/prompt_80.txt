@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'कोई फ़ाइल नहीं मिली'})
    
    file = request.files['file']
    session_id = request.form.get('session_id', 'default')
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'फ़ाइल का नाम खाली है'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'केवल YAML फ़ाइलें (.yaml या .yml) स्वीकार की जाती हैं'})
    
    try:
        content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'YAML फ़ाइल में एक valid dictionary होनी चाहिए'})
        
        # Store profile for this session
        user_profiles[session_id] = profile_data
        
        # Initialize conversation history with system context
        profile_str = yaml.dump(profile_data, allow_unicode=True, default_flow_style=False)
        
        system_message = f"""आप एक helpful AI assistant हैं जो उपयोगकर्ता की प्रोफ़ाइल के बारे में जानते हैं।

उपयोगकर्ता की प्रोफ़ाइल:
{profile_str}

इस प्रोफ़ाइल के आधार पर उपयोगकर्ता से बात करें। उनकी भाषा और preferences का ध्यान रखें। 
यदि प्रोफ़ाइल में language Hindi है, तो Hindi में जवाब दें।
व्यक्तिगत और helpful responses दें।"""
        
        conversation_histories[session_id] = {
            'system': system_message,
            'messages': []
        }
        
        # Generate welcome message
        name = profile_data.get('name', 'उपयोगकर्ता')
        profession = profile_data.get('profession', '')
        
        welcome_msg = f"नमस्ते {name}! "
        if profession:
            welcome_msg += f"एक {profession} के रूप में, "
        welcome_msg += "मैं आपकी प्रोफ़ाइल के बारे में जानता हूँ और आपकी मदद करने के लिए तैयार हूँ। आप मुझसे कुछ भी पूछ सकते हैं!"
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'message': welcome_msg
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'YAML पार्सिंग त्रुटि: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'फ़ाइल प्रोसेसिंग त्रुटि: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    data = request.get_json()
    message = data.get('message', '')
    session_id = data.get('session_id', 'default')
    
    if not message:
        return jsonify({'success': False, 'error': 'संदेश खाली है'})
    
    # Check if profile exists for this session
    if session_id not in user_profiles:
        return jsonify({
            'success': True,
            'response': 'कृपया पहले अपनी YAML प्रोफ़ाइल फ़ाइल अपलोड करें। इसके बाद हम बेहतर तरीके से बात कर सकते हैं!'
        })
    
    # Get or initialize conversation history
    if session_id not in conversation_histories:
        profile_data = user_profiles[session_id]
        profile_str = yaml.dump(profile_data, allow_unicode=True, default_flow_style=False)
        
        conversation_histories[session_id] = {
            'system': f"""आप एक helpful AI assistant हैं।
            
उपयोगकर्ता की प्रोफ़ाइल:
{profile_str}

इस प्रोफ़ाइल के आधार पर उपयोगकर्ता से बात करें।""",
            'messages': []
        }
    
    # Add user message to history
    conversation_histories[session_id]['messages'].append({
        'role': 'user',
        'content': message
    })
    
    try:
        # Call Claude API with conversation history
        response = client.messages.create(
            model='claude-opus-4-5',
            max_tokens=1024,
            system=conversation_histories[session_id]['system'],
            messages=conversation_histories[session_id]['messages']
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_histories[session_id]['messages'].append({
            'role': 'assistant',
            'content': assistant_message
        })
        
        # Keep conversation history manageable (last 20 messages)
        if len(conversation_histories[session_id]['messages']) > 20:
            conversation_histories[session_id]['messages'] = conversation_histories[session_id]['messages'][-20:]
        
        return jsonify({
            'success': True,
            'response': assistant_message
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'AI response त्रुटि: {str(e)}'})

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
