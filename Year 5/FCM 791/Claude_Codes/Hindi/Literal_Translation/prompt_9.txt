# app/models/user.rb
class User < ApplicationRecord
  enum role: { staff: 0, manager: 1, admin: 2 }
  
  belongs_to :promoted_by, class_name: 'User', optional: true
  has_many :promoted_users, class_name: 'User', foreign_key: 'promoted_by_id'
  
  validates :name, presence: true
  validates :email, presence: true, uniqueness: true
  validates :role, presence: true
  
  def can_promote?
    manager? || admin?
  end
  
  def promotable_to_manager?
    staff?
  end
end

# app/policies/promotion_policy.rb
class PromotionPolicy
  attr_reader :current_user, :target_user
  
  def initialize(current_user, target_user)
    @current_user = current_user
    @target_user = target_user
  end
  
  def promote_to_manager?
    return false unless current_user.can_promote?
    return false unless target_user.promotable_to_manager?
    return false if current_user == target_user
    true
  end
  
  def error_message
    if !current_user.can_promote?
      "आपके पास पदोन्नति करने का अधिकार नहीं है।"
    elsif !target_user.promotable_to_manager?
      "यह उपयोगकर्ता पदोन्नति के योग्य नहीं है।"
    elsif current_user == target_user
      "आप स्वयं को पदोन्नत नहीं कर सकते।"
    else
      "पदोन्नति संभव नहीं है।"
    end
  end
end

# app/controllers/promotions_controller.rb
class PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :set_staff_member, only: [:create]
  before_action :authorize_promotion!, only: [:create]
  
  # GET /promotions
  def index
    @staff_members = User.staff.order(:name)
    @promoted_users = current_user.promoted_users
                                  .order(updated_at: :desc)
                                  .limit(10)
  end
  
  # POST /promotions
  def create
    ActiveRecord::Base.transaction do
      @staff_member.update!(
        role: :manager,
        promoted_by: current_user,
        promoted_at: Time.current
      )
      
      PromotionMailer.promotion_notification(@staff_member, current_user).deliver_later
      PromotionMailer.manager_notification(current_user, @staff_member).deliver_later
      
      log_promotion_activity
    end
    
    respond_to do |format|
      format.html do
        redirect_to promotions_path,
                    notice: "#{@staff_member.name} को सफलतापूर्वक Manager के पद पर पदोन्नत किया गया।"
      end
      format.json do
        render json: {
          success: true,
          message: "#{@staff_member.name} को Manager के पद पर पदोन्नत किया गया।",
          user: {
            id: @staff_member.id,
            name: @staff_member.name,
            email: @staff_member.email,
            role: @staff_member.role,
            promoted_at: @staff_member.promoted_at,
            promoted_by: current_user.name
          }
        }, status: :ok
      end
    end
  rescue ActiveRecord::RecordInvalid => e
    handle_promotion_error(e.message)
  rescue StandardError => e
    handle_promotion_error("पदोन्नति के दौरान एक त्रुटि हुई: #{e.message}")
  end
  
  private
  
  def set_staff_member
    @staff_member = User.find(params[:user_id])
  rescue ActiveRecord::RecordNotFound
    respond_to do |format|
      format.html do
        redirect_to promotions_path, alert: "Staff सदस्य नहीं मिला।"
      end
      format.json do
        render json: { success: false, error: "Staff सदस्य नहीं मिला।" },
               status: :not_found
      end
    end
  end
  
  def authorize_promotion!
    policy = PromotionPolicy.new(current_user, @staff_member)
    
    unless policy.promote_to_manager?
      respond_to do |format|
        format.html do
          redirect_to promotions_path, alert: policy.error_message
        end
        format.json do
          render json: { success: false, error: policy.error_message },
                 status: :forbidden
        end
      end
    end
  end
  
  def log_promotion_activity
    PromotionLog.create!(
      promoted_user: @staff_member,
      promoted_by: current_user,
      previous_role: 'staff',
      new_role: 'manager',
      promoted_at: Time.current,
      ip_address: request.remote_ip
    )
  end
  
  def handle_promotion_error(message)
    respond_to do |format|
      format.html do
        redirect_to promotions_path, alert: message
      end
      format.json do
        render json: { success: false, error: message },
               status: :unprocessable_entity
      end
    end
  end
end

# app/mailers/promotion_mailer.rb
class PromotionMailer < ApplicationMailer
  def promotion_notification(promoted_user, promoted_by)
    @promoted_user = promoted_user
    @promoted_by = promoted_by
    
    mail(
      to: promoted_user.email,
      subject: "बधाई हो! आपको Manager के पद पर पदोन्नत किया गया है।"
    )
  end
  
  def manager_notification(manager, promoted_user)
    @manager = manager
    @promoted_user = promoted_user
    
    mail(
      to: manager.email,
      subject: "पदोन्नति की पुष्टि: #{promoted_user.name}"
    )
  end
end

class PromotionLog < ApplicationRecord
  belongs_to :promoted_user, class_name: 'User'
  belongs_to :promoted_by, class_name: 'User'
  
  validates :previous_role, :new_role, :promoted_at, presence: true
  
  scope :recent, -> { order(promoted_at: :desc) }
  scope :this_month, -> { where(promoted_at: Time.current.beginning_of_month..Time.current.end_of_month) }
end

require 'rails_helper'

RSpec.describe PromotionsController, type: :controller do
  let(:manager) { create(:user, role: :manager) }
  let(:staff_member) { create(:user, role: :staff) }
  
  before { sign_in manager }
  
  describe 'POST #create' do
    context 'जब Manager एक Staff सदस्य को पदोन्नत करता है' do
      it 'Staff सदस्य को Manager बनाता है' do
        post :create, params: { user_id: staff_member.id }
        
        expect(staff_member.reload.role).to eq('manager')
        expect(response).to redirect_to(promotions_path)
        expect(flash[:notice]).to include('सफलतापूर्वक Manager के पद पर पदोन्नत')
      end
      
      it 'Promotion Log बनाता है' do
        expect {
          post :create, params: { user_id: staff_member.id }
        }.to change(PromotionLog, :count).by(1)
      end
    end
    
    context 'जब Staff सदस्य पदोन्नति करने की कोशिश करता है' do
      let(:another_staff) { create(:user, role: :staff) }
      
      before { sign_in another_staff }
      
      it 'पदोन्नति की अनुमति नहीं देता' do
        post :create, params: { user_id: staff_member.id }
        
        expect(staff_member.reload.role).to eq('staff')
        expect(flash[:alert]).to include('अधिकार नहीं है')
      end
    end
    
    context 'जब Manager खुद को पदोन्नत करने की कोशिश करता है' do
      it 'पदोन्नति की अनुमति नहीं देता' do
        post :create, params: { user_id: manager.id }
        
        expect(flash[:alert]).to include('स्वयं को पदोन्नत नहीं कर सकते')
      end
    end
  end
end
