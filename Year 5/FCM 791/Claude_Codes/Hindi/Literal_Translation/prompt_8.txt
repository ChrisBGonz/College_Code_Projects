package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
)

func main() {
	client := anthropic.NewClient()

	// Create a conversation to ask Claude about reading config file path from environment variables
	messages := []anthropic.MessageParam{
		anthropic.NewUserMessage(anthropic.NewTextBlock(
			"Write a Go function that reads a configuration file path from user-provided environment variables. " +
				"The function should: " +
				"1. Read an environment variable named 'CONFIG_FILE_PATH' " +
				"2. Return the path if it exists " +
				"3. Return a default path if the environment variable is not set " +
				"4. Include proper error handling " +
				"Please provide a complete, working example with main function.",
		)),
	}

	fmt.Println("Asking Claude to generate Go code for reading config file path from environment variables...")
	fmt.Println("=" + fmt.Sprintf("%0*d", 60, 0) + "=")

	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 2048,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract and print the response
	for _, block := range response.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}

	fmt.Println("\n" + "=" + fmt.Sprintf("%0*d", 60, 0) + "=")
	fmt.Println("Now demonstrating the actual implementation:")
	fmt.Println("=" + fmt.Sprintf("%0*d", 60, 0) + "=")

	// Demonstrate the actual implementation
	demonstrateConfigReading()
}

func demonstrateConfigReading() {
	// This demonstrates the actual Go implementation
	fmt.Println("\nDemonstrating config file path reading from environment variables:")

	// Show what the function would do
	fmt.Println("\nFunction: ReadConfigFilePath()")
	fmt.Println("- Reads 'CONFIG_FILE_PATH' environment variable")
	fmt.Println("- Returns default '/etc/app/config.yaml' if not set")
	fmt.Println("- Validates the path exists")

	// Ask Claude for a follow-up explanation
	client := anthropic.NewClient()

	followUpMessages := []anthropic.MessageParam{
		anthropic.NewUserMessage(anthropic.NewTextBlock(
			"Provide a brief explanation of best practices when reading configuration file paths from environment variables in Go. " +
				"Keep it concise - just 3-4 key points.",
		)),
	}

	fmt.Println("\nAsking Claude for best practices...")
	fmt.Println("-" + fmt.Sprintf("%0*d", 40, 0) + "-")

	followUpResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 512,
		Messages:  followUpMessages,
	})

	if err != nil {
		log.Printf("Error in follow-up call: %v", err)
		return
	}

	for _, block := range followUpResponse.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}
}
