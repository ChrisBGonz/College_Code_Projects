import anthropic
import pickle
import os
import json
from datetime import datetime

client = anthropic.Anthropic()

class GameState:
    """Complex Game State object to be saved and loaded"""
    def __init__(self):
        self.player_name = ""
        self.level = 1
        self.score = 0
        self.health = 100
        self.inventory = []
        self.achievements = []
        self.position = {"x": 0, "y": 0}
        self.game_time = 0
        self.difficulty = "normal"
        self.quests = {}
        self.skills = {}
        self.save_timestamp = None
    
    def to_dict(self):
        return {
            "player_name": self.player_name,
            "level": self.level,
            "score": self.score,
            "health": self.health,
            "inventory": self.inventory,
            "achievements": self.achievements,
            "position": self.position,
            "game_time": self.game_time,
            "difficulty": self.difficulty,
            "quests": self.quests,
            "skills": self.skills,
            "save_timestamp": str(self.save_timestamp)
        }
    
    def __str__(self):
        return json.dumps(self.to_dict(), indent=2, ensure_ascii=False)


def save_game_state(game_state: GameState, filename: str) -> bool:
    """Save game state to file using pickle"""
    try:
        game_state.save_timestamp = datetime.now()
        with open(filename, 'wb') as f:
            pickle.dump(game_state, f)
        return True
    except Exception as e:
        print(f"Error saving game state: {e}")
        return False


def load_game_state(filename: str) -> GameState:
    """Load game state from file using pickle"""
    try:
        if not os.path.exists(filename):
            return None
        with open(filename, 'rb') as f:
            game_state = pickle.load(f)
        return game_state
    except Exception as e:
        print(f"Error loading game state: {e}")
        return None


def process_tool_call(tool_name: str, tool_input: dict, game_state: GameState, save_file: str) -> str:
    """Process tool calls from Claude"""
    
    if tool_name == "update_player_info":
        if "name" in tool_input:
            game_state.player_name = tool_input["name"]
        if "level" in tool_input:
            game_state.level = tool_input["level"]
        if "score" in tool_input:
            game_state.score = tool_input["score"]
        if "health" in tool_input:
            game_state.health = tool_input["health"]
        if "difficulty" in tool_input:
            game_state.difficulty = tool_input["difficulty"]
        return f"Player info updated: {game_state.player_name}, Level {game_state.level}"
    
    elif tool_name == "add_inventory_item":
        item = tool_input.get("item", "")
        quantity = tool_input.get("quantity", 1)
        for _ in range(quantity):
            game_state.inventory.append(item)
        return f"Added {quantity}x {item} to inventory"
    
    elif tool_name == "add_achievement":
        achievement = tool_input.get("achievement", "")
        if achievement not in game_state.achievements:
            game_state.achievements.append(achievement)
            return f"Achievement unlocked: {achievement}"
        return f"Achievement already exists: {achievement}"
    
    elif tool_name == "update_position":
        game_state.position["x"] = tool_input.get("x", game_state.position["x"])
        game_state.position["y"] = tool_input.get("y", game_state.position["y"])
        return f"Position updated to ({game_state.position['x']}, {game_state.position['y']})"
    
    elif tool_name == "add_quest":
        quest_name = tool_input.get("quest_name", "")
        quest_status = tool_input.get("status", "active")
        quest_description = tool_input.get("description", "")
        game_state.quests[quest_name] = {
            "status": quest_status,
            "description": quest_description
        }
        return f"Quest added: {quest_name} ({quest_status})"
    
    elif tool_name == "update_skill":
        skill_name = tool_input.get("skill_name", "")
        skill_level = tool_input.get("level", 1)
        game_state.skills[skill_name] = skill_level
        return f"Skill updated: {skill_name} -> Level {skill_level}"
    
    elif tool_name == "update_game_time":
        game_state.game_time = tool_input.get("time_minutes", 0)
        return f"Game time updated to {game_state.game_time} minutes"
    
    elif tool_name == "save_game":
        filename = tool_input.get("filename", save_file)
        success = save_game_state(game_state, filename)
        if success:
            return f"Game saved successfully to {filename}"
        return f"Failed to save game to {filename}"
    
    elif tool_name == "load_game":
        filename = tool_input.get("filename", save_file)
        loaded_state = load_game_state(filename)
        if loaded_state:
            game_state.__dict__.update(loaded_state.__dict__)
            return f"Game loaded successfully from {filename}. Player: {game_state.player_name}, Level: {game_state.level}"
        return f"Failed to load game from {filename}"
    
    elif tool_name == "get_game_state":
        return str(game_state)
    
    return f"Unknown tool: {tool_name}"


def create_game_with_claude(user_request: str, save_file: str = "game_save.pkl"):
    """Use Claude to create and manage game state"""
    
    tools = [
        {
            "name": "update_player_info",
            "description": "Update player's basic information like name, level, score, health, and difficulty",
            "input_schema": {
                "type": "object",
                "properties": {
                    "name": {"type": "string", "description": "Player name"},
                    "level": {"type": "integer", "description": "Player level"},
                    "score": {"type": "integer", "description": "Player score"},
                    "health": {"type": "integer", "description": "Player health (0-100)"},
                    "difficulty": {"type": "string", "description": "Game difficulty (easy/normal/hard)"}
                }
            }
        },
        {
            "name": "add_inventory_item",
            "description": "Add an item to player's inventory",
            "input_schema": {
                "type": "object",
                "properties": {
                    "item": {"type": "string", "description": "Item name"},
                    "quantity": {"type": "integer", "description": "Quantity to add", "default": 1}
                },
                "required": ["item"]
            }
        },
        {
            "name": "add_achievement",
            "description": "Add an achievement to player's achievement list",
            "input_schema": {
                "type": "object",
                "properties": {
                    "achievement": {"type": "string", "description": "Achievement name"}
                },
                "required": ["achievement"]
            }
        },
        {
            "name": "update_position",
            "description": "Update player's position in the game world",
            "input_schema": {
                "type": "object",
                "properties": {
                    "x": {"type": "number", "description": "X coordinate"},
                    "y": {"type": "number", "description": "Y coordinate"}
                }
            }
        },
        {
            "name": "add_quest",
            "description": "Add or update a quest in the game",
            "input_schema": {
                "type": "object",
                "properties": {
                    "quest_name": {"type": "string", "description": "Quest name"},
                    "status": {"type": "string", "description": "Quest status (active/completed/failed)"},
                    "description": {"type": "string", "description": "Quest description"}
                },
                "required": ["quest_name"]
            }
        },
        {
            "name": "update_skill",
            "description": "Update a player skill level",
            "input_schema": {
                "type": "object",
                "properties": {
                    "skill_name": {"type": "string", "description": "Skill name"},
                    "level": {"type": "integer", "description": "Skill level"}
                },
                "required": ["skill_name"]
            }
        },
        {
            "name": "update_game_time",
            "description": "Update total game time played",
            "input_schema": {
                "type": "object",
                "properties": {
                    "time_minutes": {"type": "integer", "description": "Total time played in minutes"}
                },
                "required": ["time_minutes"]
            }
        },
        {
            "name": "save_game",
            "description": "Save the current game state to a file using pickle",
            "input_schema": {
                "type": "object",
                "properties": {
                    "filename": {"type": "string", "description": "File path to save the game"}
                }
            }
        },
        {
            "name": "load_game",
            "description": "Load a previously saved game state from a file",
            "input_schema": {
                "type": "object",
                "properties": {
                    "filename": {"type": "string", "description": "File path to load the game from"}
                }
            }
        },
        {
            "name": "get_game_state",
            "description": "Get the current game state as JSON string",
            "input_schema": {
                "type": "object",
                "properties": {}
            }
        }
    ]
    
    game_state = GameState()
    messages = [{"role": "user", "content": user_request}]
    
    print(f"\n{'='*60}")
    print(f"User Request: {user_request}")
    print(f"{'='*60}\n")
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "end_turn":
            final_text = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_text = block.text
            print(f"\nClaude's Response:\n{final_text}")
            break
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"Tool Called: {tool_name}")
                    print(f"Input: {json.dumps(tool_input, indent=2, ensure_ascii=False)}")
                    
                    result = process_tool_call(tool_name, tool_input, game_state, save_file)
                    
                    print(f"Result: {result}\n")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    return game_state
