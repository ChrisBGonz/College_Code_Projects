import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.input.Prompt;
import dev.langchain4j.model.input.PromptTemplate;

import java.util.HashMap;
import java.util.Map;

public class PasswordStrengthValidator {

    private final ChatLanguageModel chatModel;

    public PasswordStrengthValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .temperature(0.1)
                .build();
    }

    public PasswordStrengthResult validatePasswordStrength(String password) {
        String promptText = """
                आप एक security expert हैं जो passwords की strength analyze करते हैं।
                
                निम्नलिखित password की strength validate करें: "{{password}}"
                
                कृपया निम्नलिखित criteria के आधार पर analyze करें:
                1. Length (कम से कम 8 characters होने चाहिए, 12+ strong माना जाता है)
                2. Uppercase letters (A-Z) की presence
                3. Lowercase letters (a-z) की presence
                4. Numbers (0-9) की presence
                5. Special characters (!@#$%^&*) की presence
                6. Common patterns या dictionary words की absence
                7. Sequential characters (abc, 123) की absence
                
                Response format बिल्कुल इस प्रकार दें:
                STRENGTH: [WEAK/MEDIUM/STRONG/VERY_STRONG]
                SCORE: [0-100]
                ISSUES: [comma separated list of issues, or "None" if no issues]
                SUGGESTIONS: [improvement suggestions in Hindi]
                VERDICT: [एक line में final verdict in Hindi]
                """;

        PromptTemplate template = PromptTemplate.from(promptText);
        Map<String, Object> variables = new HashMap<>();
        variables.put("password", password);
        Prompt prompt = template.apply(variables);

        String response = chatModel.generate(prompt.text());
        return parseResponse(response, password);
    }

    private PasswordStrengthResult parseResponse(String response, String password) {
        String strength = "UNKNOWN";
        int score = 0;
        String issues = "";
        String suggestions = "";
        String verdict = "";

        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith("STRENGTH:")) {
                strength = line.replace("STRENGTH:", "").trim();
            } else if (line.startsWith("SCORE:")) {
                try {
                    score = Integer.parseInt(line.replace("SCORE:", "").trim());
                } catch (NumberFormatException e) {
                    score = 0;
                }
            } else if (line.startsWith("ISSUES:")) {
                issues = line.replace("ISSUES:", "").trim();
            } else if (line.startsWith("SUGGESTIONS:")) {
                suggestions = line.replace("SUGGESTIONS:", "").trim();
            } else if (line.startsWith("VERDICT:")) {
                verdict = line.replace("VERDICT:", "").trim();
            }
        }

        boolean isValid = score >= 60 && !strength.equals("WEAK");

        return new PasswordStrengthResult(
                password,
                strength,
                score,
                issues,
                suggestions,
                verdict,
                isValid
        );
    }

    public static class PasswordStrengthResult {
        private final String password;
        private final String strength;
        private final int score;
        private final String issues;
        private final String suggestions;
        private final String verdict;
        private final boolean isValid;

        public PasswordStrengthResult(String password, String strength, int score,
                String issues, String suggestions, String verdict, boolean isValid) {
            this.password = password;
            this.strength = strength;
            this.score = score;
            this.issues = issues;
            this.suggestions = suggestions;
            this.verdict = verdict;
            this.isValid = isValid;
        }

        public String getPassword() {
            return password;
        }

        public String getStrength() {
            return strength;
        }

        public int getScore() {
            return score;
        }

        public String getIssues() {
            return issues;
        }

        public String getSuggestions() {
            return suggestions;
        }

        public String getVerdict() {
            return verdict;
        }

        public boolean isValid() {
            return isValid;
        }

        @Override
        public String toString() {
            return String.format("""
                    ==========================================
                    Password Strength Analysis
                    ==========================================
                    Password: %s
                    Strength: %s
                    Score: %d/100
                    Valid: %s
                    Issues: %s
                    Suggestions: %s
                    Verdict: %s
                    ==========================================
                    """,
                    maskPassword(password),
                    strength,
                    score,
                    isValid ? "✓ हाँ" : "✗ नहीं",
                    issues,
                    suggestions,
                    verdict);
        }

        private String maskPassword(String pwd) {
            if (pwd == null || pwd.length() <= 2) {
                return "***";
            }
            return pwd.charAt(0) + "*".repeat(pwd.length() - 2) + pwd.charAt(pwd.length() - 1);
        }
    }

    public static void main(String[] args) {
        PasswordStrengthValidator validator = new PasswordStrengthValidator();

        String[] testPasswords = {
                "123456",
                "password",
                "Hello123",
                "MyP@ssw0rd",
                "Tr0ub4dor&3",
                "correct horse battery staple",
                "P@$$w0rd!2024#Secure"
        };

        System.out.println("Password Strength Validator - Registration Page");
        System.out.println("================================================\n");

        for (String password : testPasswords) {
            System.out.println("Testing password: " + password);
            PasswordStrengthResult result = validator.validatePasswordStrength(password);
            System.out.println(result);

            if (!result.isValid()) {
                System.out.println("⚠️  यह password registration के लिए accept नहीं किया जाएगा।\n");
            } else {
                System.out.println("✅ यह password registration के लिए valid है।\n");
            }
        }
    }
}
