
```python
import anthropic
import json

client = anthropic.Anthropic()

# Simulated user database
USER_DATABASE = {
    "admin": "admin123",
    "user1": "password123",
    "alice": "alice@2024",
    "bob": "bobsecret"
}

def verify_password(username: str, password: str) -> dict:
    """Verify username and password against the database."""
    if username not in USER_DATABASE:
        return {
            "success": False,
            "status_code": 401,
            "message": "401 Unauthorized: User not found"
        }
    
    if USER_DATABASE[username] != password:
        return {
            "success": False,
            "status_code": 401,
            "message": "401 Unauthorized: Invalid password"
        }
    
    return {
        "success": True,
        "status_code": 200,
        "message": f"Authentication successful for user: {username}",
        "user": username
    }

def get_user_info(username: str) -> dict:
    """Get user information after successful authentication."""
    if username in USER_DATABASE:
        return {
            "username": username,
            "role": "admin" if username == "admin" else "user",
            "permissions": ["read", "write"] if username == "admin" else ["read"]
        }
    return {"error": "User not found"}

# Define tools for Claude
tools = [
    {
        "name": "verify_password",
        "description": "Verify user credentials (username and password) for authentication. Returns success status and appropriate HTTP status code.",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to authenticate"
                },
                "password": {
                    "type": "string",
                    "description": "The password to verify"
                }
            },
            "required": ["username", "password"]
        }
    },
    {
        "name": "get_user_info",
        "description": "Get user information after successful authentication",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to get information for"
                }
            },
            "required": ["username"]
        }
    }
]

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "verify_password":
        result = verify_password(tool_input["username"], tool_input["password"])
        return json.dumps(result)
    elif tool_name == "get_user_info":
        result = get_user_info(tool_input["username"])
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def authenticate_user(username: str, password: str) -> str:
    """Main authentication function using Claude with tool use."""
    print(f"\n{'='*60}")
    print(f"Authentication attempt for user: {username}")
    print(f"{'='*60}")
    
    messages = [
        {
            "role": "user",
            "content": f"Please authenticate the user with username '{username}' and password '{password}'. If authentication is successful, also get their user information. Provide a clear response about the authentication result."
        }
    ]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude Response (stop_reason: {response.stop_reason}):")
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process all tool calls
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\nTool Called: {tool_name}")
                    print(f"Tool Input: {json.dumps(tool_input, indent=2)}")
                    
                    # Execute the tool
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Tool Result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            # End turn - extract final text response
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    final_response += content_block.text
            
            print(f"\nFinal Response:\n{final_response}")
            return final_response

def main():
    """Test the authentication system with various scenarios."""
    print("="*60)
    print("Authentication System Test")
    print("="*60)
    
    # Test cases
    test_cases = [
        ("admin", "admin123"),      # Valid credentials
        ("user1", "wrongpassword"), # Wrong password
        ("unknown", "password"),    # Unknown user
        ("alice", "alice@2024"),    # Valid user
    ]
    
    for username, password in test_cases:
        result = authenticate_user(username, password)
        print("\n" + "="*60)

if __name__ == "__main__":
    main()
```
