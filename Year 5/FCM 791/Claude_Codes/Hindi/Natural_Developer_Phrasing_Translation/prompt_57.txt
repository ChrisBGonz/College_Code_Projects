/**
 * Create required directories if not exist
 */
function createDirectories(): void {
    $dirs = [UPLOAD_DIR, dirname(LOG_FILE)];
    foreach ($dirs as $dir) {
        if (!is_dir($dir)) {
            mkdir($dir, 0755, true);
        }
    }
}

/**
 * Log upload activity
 */
function logActivity(string $message, string $level = 'INFO'): void {
    $timestamp = date('Y-m-d H:i:s');
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'UNKNOWN';
    $logEntry = "[{$timestamp}] [{$level}] [IP: {$ip}] {$message}" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Sanitize filename - remove dangerous characters
 */
function sanitizeFilename(string $filename): string {
    // Remove path traversal attempts
    $filename = basename($filename);
    
    // Remove null bytes
    $filename = str_replace("\0", '', $filename);
    
    // Allow only alphanumeric, dash, underscore, dot
    $filename = preg_replace('/[^a-zA-Z0-9\-\_\.]/', '_', $filename);
    
    // Remove multiple consecutive dots (prevent double extension attacks)
    $filename = preg_replace('/\.{2,}/', '.', $filename);
    
    // Trim dots and spaces from beginning and end
    $filename = trim($filename, '. ');
    
    // Limit filename length
    if (strlen($filename) > 100) {
        $filename = substr($filename, 0, 100);
    }
    
    return $filename ?: 'unnamed_file';
}

/**
 * Generate unique secure filename
 */
function generateSecureFilename(string $originalName, string $employeeId): string {
    $sanitized = sanitizeFilename($originalName);
    
    // Remove extension from sanitized name
    $nameWithoutExt = pathinfo($sanitized, PATHINFO_FILENAME);
    
    // Generate unique name: employeeId_timestamp_randomhash_originalname.pdf
    $timestamp = date('Ymd_His');
    $randomHash = bin2hex(random_bytes(8));
    
    return "{$employeeId}_{$timestamp}_{$randomHash}_{$nameWithoutExt}.pdf";
}

/**
 * Validate uploaded file - Multiple security checks
 */
function validateUploadedFile(array $file): array {
    $errors = [];
    
    // Check 1: Upload error check
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $uploadErrors = [
            UPLOAD_ERR_INI_SIZE   => 'फाइल PHP ini की size limit से बड़ी है।',
            UPLOAD_ERR_FORM_SIZE  => 'फाइल form की size limit से बड़ी है।',
            UPLOAD_ERR_PARTIAL    => 'फाइल partially upload हुई है।',
            UPLOAD_ERR_NO_FILE    => 'कोई फाइल upload नहीं हुई।',
            UPLOAD_ERR_NO_TMP_DIR => 'Temporary directory नहीं मिली।',
            UPLOAD_ERR_CANT_WRITE => 'Disk पर write नहीं हो सका।',
            UPLOAD_ERR_EXTENSION  => 'PHP extension ने upload रोका।',
        ];
        $errors[] = $uploadErrors[$file['error']] ?? 'Unknown upload error.';
        return $errors;
    }
    
    // Check 2: File size validation
    if ($file['size'] > MAX_FILE_SIZE) {
        $maxMB = MAX_FILE_SIZE / (1024 * 1024);
        $errors[] = "फाइल size {$maxMB}MB से अधिक नहीं होनी चाहिए। आपकी फाइल: " . round($file['size'] / (1024 * 1024), 2) . "MB";
    }
    
    // Check 3: File size should not be zero
    if ($file['size'] === 0) {
        $errors[] = 'फाइल empty है।';
    }
    
    // Check 4: Extension validation
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if ($extension !== ALLOWED_EXTENSION) {
        $errors[] = "केवल PDF files allowed हैं। आपकी फाइल extension: .{$extension}";
    }
    
    // Check 5: MIME type validation using finfo (most reliable)
    if (is_uploaded_file($file['tmp_name'])) {
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);
        
        if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
            $errors[] = "Invalid file type: {$mimeType}. केवल PDF files (application/pdf) allowed हैं।";
        }
        
        // Check 6: PDF magic bytes validation (first 4 bytes should be %PDF)
        $handle = fopen($file['tmp_name'], 'rb');
        if ($handle) {
            $magicBytes = fread($handle, 4);
            fclose($handle);
            
            if ($magicBytes !== '%PDF') {
                $errors[] = 'फाइल valid PDF नहीं है (magic bytes mismatch)।';
            }
        }
    } else {
        $errors[] = 'Security Error: यह valid uploaded file नहीं है।';
    }
    
    // Check 7: Filename length
    if (strlen($file['name']) > 255) {
        $errors[] = 'Filename बहुत लंबा है।';
    }
    
    return $errors;
}

/**
 * Get list of uploaded files from shared drive
 */
function getUploadedFiles(): array {
    $files = [];
    
    if (!is_dir(UPLOAD_DIR)) {
        return $files;
    }
    
    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isFile() && strtolower($fileInfo->getExtension()) === 'pdf') {
            $files[] = [
                'name'     => $fileInfo->getFilename(),
                'size'     => $fileInfo->getSize(),
                'modified' => date('d M Y, H:i', $fileInfo->getMTime()),
            ];
        }
    }
    
    // Sort by modification time (newest first)
    usort($files, fn($a, $b) => strtotime($b['modified']) - strtotime($a['modified']));
    
    return $files;
}

/**
 * Format file size for display
 */
function formatFileSize(int $bytes): string {
    if ($bytes >= 1048576) {
        return round($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return round($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' Bytes';
}
