# User authentication data
USERS = {
    "user1": {"password": "pass123", "company_id": "COMP001", "name": "Rahul Sharma", "role": "Manager"},
    "user2": {"password": "pass456", "company_id": "COMP002", "name": "Priya Patel", "role": "Director"},
    "user3": {"password": "pass789", "company_id": "COMP003", "name": "Amit Kumar", "role": "CFO"},
    "admin1": {"password": "admin123", "company_id": "COMP001", "name": "Sneha Gupta", "role": "Admin"}
}

def authenticate_user(username: str, password: str) -> dict | None:
    """Authenticate user and return user data if valid"""
    if username in USERS and USERS[username]["password"] == password:
        user_data = USERS[username].copy()
        user_data["username"] = username
        return user_data
    return None

def get_company_context(company_id: str) -> str:
    """Get company data as context string for Claude"""
    if company_id not in COMPANY_DATA:
        return "Company data not found."
    
    company = COMPANY_DATA[company_id]
    fin = company["financials"]
    
    context = f"""
    Company Information:
    - Name: {company['name']}
    - Industry: {company['industry']}
    - Founded: {company['founded']}
    - Total Employees: {company['employees']}
    
    Annual Financial Report 2024:
    - Total Revenue: ₹{fin['revenue']:,}
    - Total Expenses: ₹{fin['expenses']:,}
    - Net Profit: ₹{fin['profit']:,}
    - Growth Rate: {fin['growth_rate']}%
    
    Quarterly Breakdown:
    - Q1: Revenue ₹{fin['quarters']['Q1']['revenue']:,}, Expenses ₹{fin['quarters']['Q1']['expenses']:,}
    - Q2: Revenue ₹{fin['quarters']['Q2']['revenue']:,}, Expenses ₹{fin['quarters']['Q2']['expenses']:,}
    - Q3: Revenue ₹{fin['quarters']['Q3']['revenue']:,}, Expenses ₹{fin['quarters']['Q3']['expenses']:,}
    - Q4: Revenue ₹{fin['quarters']['Q4']['revenue']:,}, Expenses ₹{fin['quarters']['Q4']['expenses']:,}
    
    Departments: {', '.join(company['departments'])}
    
    Key Achievements 2024:
    {chr(10).join(['- ' + achievement for achievement in company['key_achievements']])}
    """
    return context

def chat_with_annual_report(user_data: dict, conversation_history: list) -> None:
    """Multi-turn conversation about company's annual report"""
    company_id = user_data["company_id"]
    company_name = COMPANY_DATA[company_id]["name"]
    
    print(f"\n{'='*60}")
    print(f"Welcome {user_data['name']}! ({user_data['role']})")
    print(f"Company: {company_name}")
    print(f"{'='*60}")
    print("Annual Report Assistant शुरू हो गया है!")
    print("आप अपनी कंपनी के Annual Report के बारे में कोई भी सवाल पूछ सकते हैं।")
    print("बाहर निकलने के लिए 'exit' या 'quit' टाइप करें।")
    print(f"{'='*60}\n")
    
    # System prompt with company context
    company_context = get_company_context(company_id)
    system_prompt = f"""आप एक helpful Annual Report Assistant हैं जो {company_name} के financial data और performance के बारे में जानकारी देते हैं।

आपके पास निम्नलिखित company data है:
{company_context}

महत्वपूर्ण नियम:
1. केवल इस company ({company_name}) के data के बारे में बात करें
2. दूसरी companies का data share न करें (data isolation)
3. Financial figures को clearly और accurately present करें
4. Hindi और English दोनों में जवाब दे सकते हैं
5. User को helpful insights और analysis provide करें
6. अगर कोई data available नहीं है तो clearly बताएं

User का नाम: {user_data['name']}
User का role: {user_data['role']}"""
    
    while True:
        user_input = input(f"\n{user_data['name']}: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['exit', 'quit', 'बाहर', 'बंद']:
            print("\nThank you! Annual Report Assistant बंद हो रहा है।")
            break
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        try:
            # Call Claude API with conversation history
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_prompt,
                messages=conversation_history
            )
            
            assistant_message = response.content[0].text
            
            # Add assistant response to history
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            print(f"\nAssistant: {assistant_message}")
            
        except Exception as e:
            print(f"\nError: API call failed - {str(e)}")
            # Remove the last user message if API call failed
            conversation_history.pop()

def main():
    """Main function to run the Multi-tenant SaaS application"""
    print("\n" + "="*60)
    print("   Multi-Tenant SaaS Annual Report Management System")
    print("="*60)
    print("कृपया अपने credentials से login करें।")
    print("\nDemo Users:")
    print("  user1 / pass123 - TechVision India (Manager)")
    print("  user2 / pass456 - GreenEarth Solutions (Director)")
    print("  user3 / pass789 - HealthFirst Hospitals (CFO)")
    print("="*60)
    
    # Authentication loop
    max_attempts = 3
    attempts = 0
    
    while attempts < max_attempts:
        print(f"\nLogin Attempt {attempts + 1}/{max_attempts}")
        username = input("Username: ").strip()
        password = input("Password: ").strip()
        
        user_data = authenticate_user(username, password)
        
        if user_data:
            print(f"\n✓ Login successful! Welcome, {user_data['name']}!")
            
            # Initialize conversation history for this session
            conversation_history = []
            
            # Start chatting about annual report
            chat_with_annual_report(user_data, conversation_history)
            
            # Ask if user wants to continue or logout
            while True:
                choice = input("\nक्या आप logout करना चाहते हैं? (yes/no): ").strip().lower()
                if choice in ['yes', 'y', 'हाँ']:
                    print(f"\nGoodbye, {user_data['name']}! Successfully logged out.")
                    break
                elif choice in ['no', 'n', 'नहीं']:
                    # Continue with new conversation
                    conversation_history = []
                    chat_with_annual_report(user_data, conversation_history)
                else:
                    print("Please enter yes or no")
            break
            
        else:
            attempts += 1
            remaining = max_attempts - attempts
            if remaining > 0:
                print(f"✗ Invalid credentials! {remaining} attempts remaining.")
            else:
                print("✗ Maximum login attempts exceeded. Account temporarily locked.")
                print("Please contact your system administrator.")

if __name__ == "__main__":
    main()
