<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;
    public $notifications;
    public $itemsPerPage;
    public $username;
    
    public function __construct($theme = 'light', $language = 'en', $fontSize = 14, $notifications = true, $itemsPerPage = 10, $username = 'Guest') {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->notifications = $notifications;
        $this->itemsPerPage = $itemsPerPage;
        $this->username = $username;
    }
    
    public function getThemeCSS() {
        if ($this->theme === 'dark') {
            return 'background-color: #1a1a2e; color: #eaeaea;';
        } elseif ($this->theme === 'blue') {
            return 'background-color: #1a237e; color: #ffffff;';
        } elseif ($this->theme === 'green') {
            return 'background-color: #1b5e20; color: #ffffff;';
        } else {
            return 'background-color: #f5f5f5; color: #333333;';
        }
    }
}

// ============================================================
// COOKIE से preferences load करना
// ============================================================
function loadPreferencesFromCookie() {
    if (isset($_COOKIE['user_preferences'])) {
        $cookieData = $_COOKIE['user_preferences'];
        
        // Base64 decode करना
        $decoded = base64_decode($cookieData);
        
        if ($decoded !== false) {
            // Unserialize करना - object restore करना
            $prefs = unserialize($decoded);
            
            // Check करना कि valid UserPreferences object है
            if ($prefs instanceof UserPreferences) {
                return $prefs;
            }
        }
    }
    
    // Default preferences return करना
    return new UserPreferences();
}

// ============================================================
// COOKIE में preferences save करना
// ============================================================
function savePreferencesToCookie($prefs) {
    // Object को serialize करना
    $serialized = serialize($prefs);
    
    // Base64 encode करना (cookie-safe बनाने के लिए)
    $encoded = base64_encode($serialized);
    
    // Cookie set करना - 30 दिन के लिए
    $expiry = time() + (30 * 24 * 60 * 60);
    setcookie('user_preferences', $encoded, $expiry, '/', '', false, true);
    
    return $encoded;
}
