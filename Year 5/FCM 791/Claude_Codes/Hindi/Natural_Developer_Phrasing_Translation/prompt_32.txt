import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.fasterxml.jackson.databind.JsonNode;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.security.SecureRandom;
import java.util.Base64;

public class AESEncryptionWithClaude {
    
    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    private static final int AES_KEY_SIZE = 256;
    private static final int GCM_IV_LENGTH = 12;
    private static final int GCM_TAG_LENGTH = 128;
    
    private BedrockRuntimeClient bedrockClient;
    private ObjectMapper objectMapper;
    
    public AESEncryptionWithClaude() {
        this.bedrockClient = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();
        this.objectMapper = new ObjectMapper();
    }
    
    /**
     * AES-GCM algorithm का उपयोग करके sensitive string को encrypt करता है
     */
    public static String encryptWithAES(String sensitiveData, SecretKey secretKey) throws Exception {
        // Generate random IV (Initialization Vector)
        byte[] iv = new byte[GCM_IV_LENGTH];
        SecureRandom secureRandom = new SecureRandom();
        secureRandom.nextBytes(iv);
        
        // Initialize cipher with AES-GCM mode
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, parameterSpec);
        
        // Encrypt the data
        byte[] encryptedData = cipher.doFinal(sensitiveData.getBytes("UTF-8"));
        
        // Combine IV and encrypted data
        byte[] encryptedWithIV = new byte[iv.length + encryptedData.length];
        System.arraycopy(iv, 0, encryptedWithIV, 0, iv.length);
        System.arraycopy(encryptedData, 0, encryptedWithIV, iv.length, encryptedData.length);
        
        // Return Base64 encoded string
        return Base64.getEncoder().encodeToString(encryptedWithIV);
    }
    
    /**
     * AES-GCM algorithm का उपयोग करके encrypted string को decrypt करता है
     */
    public static String decryptWithAES(String encryptedData, SecretKey secretKey) throws Exception {
        // Decode Base64
        byte[] encryptedWithIV = Base64.getDecoder().decode(encryptedData);
        
        // Extract IV
        byte[] iv = new byte[GCM_IV_LENGTH];
        System.arraycopy(encryptedWithIV, 0, iv, 0, iv.length);
        
        // Extract encrypted data
        byte[] encrypted = new byte[encryptedWithIV.length - iv.length];
        System.arraycopy(encryptedWithIV, iv.length, encrypted, 0, encrypted.length);
        
        // Initialize cipher for decryption
        Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.DECRYPT_MODE, secretKey, parameterSpec);
        
        // Decrypt and return
        byte[] decryptedData = cipher.doFinal(encrypted);
        return new String(decryptedData, "UTF-8");
    }
    
    /**
     * AES key generate करता है
     */
    public static SecretKey generateAESKey() throws Exception {
        KeyGenerator keyGenerator = KeyGenerator.getInstance("AES");
        keyGenerator.init(AES_KEY_SIZE, new SecureRandom());
        return keyGenerator.generateKey();
    }
    
    /**
     * SecretKey को Base64 string में convert करता है
     */
    public static String secretKeyToString(SecretKey secretKey) {
        return Base64.getEncoder().encodeToString(secretKey.getEncoded());
    }
    
    /**
     * Base64 string से SecretKey बनाता है
     */
    public static SecretKey stringToSecretKey(String keyString) {
        byte[] decodedKey = Base64.getDecoder().decode(keyString);
        return new SecretKeySpec(decodedKey, 0, decodedKey.length, "AES");
    }
    
    /**
     * Claude AI से encryption के बारे में explanation प्राप्त करता है
     */
    public String getEncryptionExplanation(String encryptedData) throws Exception {
        String prompt = String.format(
            "मैंने AES-GCM algorithm का उपयोग करके एक sensitive string को encrypt किया है। " +
            "Encrypted data है: %s\n\n" +
            "कृपया निम्नलिखित के बारे में हिंदी में समझाएं:\n" +
            "1. AES-GCM encryption क्या है और यह कैसे काम करता है?\n" +
            "2. यह encryption क्यों secure है?\n" +
            "3. IV (Initialization Vector) का क्या महत्व है?\n" +
            "4. इस encrypted data को कैसे safely store करें?\n" +
            "5. Key management के best practices क्या हैं?",
            encryptedData
        );
        
        ObjectNode requestBody = objectMapper.createObjectNode();
        requestBody.put("anthropic_version", "bedrock-2023-05-31");
        requestBody.put("max_tokens", 1024);
        
        ObjectNode message = objectMapper.createObjectNode();
        message.put("role", "user");
        message.put("content", prompt);
        
        requestBody.putArray("messages").add(message);
        
        String requestBodyJson = objectMapper.writeValueAsString(requestBody);
        
        InvokeModelRequest request = InvokeModelRequest.builder()
                .modelId(MODEL_ID)
                .body(SdkBytes.fromUtf8String(requestBodyJson))
                .build();
        
        InvokeModelResponse response = bedrockClient.invokeModel(request);
        String responseBody = response.body().asUtf8String();
        
        JsonNode responseJson = objectMapper.readTree(responseBody);
        return responseJson.get("content").get(0).get("text").asText();
    }
    
    public static void main(String[] args) {
        try {
            AESEncryptionWithClaude app = new AESEncryptionWithClaude();
            
            // Test sensitive data
            String sensitiveData = "यह एक बहुत sensitive जानकारी है: Password@123, Credit Card: 1234-5678-9012-3456";
            
            System.out.println("=== AES Encryption Demo ===\n");
            System.out.println("Original Sensitive Data: " + sensitiveData);
            System.out.println();
            
            // Generate AES key
            SecretKey secretKey = generateAESKey();
            String keyString = secretKeyToString(secretKey);
            System.out.println("Generated AES Key (Base64): " + keyString);
            System.out.println();
            
            // Encrypt the data
            String encryptedData = encryptWithAES(sensitiveData, secretKey);
            System.out.println("Encrypted Data: " + encryptedData);
            System.out.println();
            
            // Decrypt the data to verify
            String decryptedData = decryptWithAES(encryptedData, secretKey);
            System.out.println("Decrypted Data: " + decryptedData);
            System.out.println();
            
            // Verify encryption/decryption
            boolean isSuccessful = sensitiveData.equals(decryptedData);
            System.out.println("Encryption/Decryption Successful: " + isSuccessful);
            System.out.println();
            
            // Test key serialization
            SecretKey reconstructedKey = stringToSecretKey(keyString);
            String reEncryptedData = encryptWithAES(sensitiveData, reconstructedKey);
            String reDecryptedData = decryptWithAES(reEncryptedData, reconstructedKey);
            System.out.println("Key Serialization Test - Decrypted: " + reDecryptedData);
            System.out.println("Key Serialization Successful: " + sensitiveData.equals(reDecryptedData));
            System.out.println();
            
            // Get explanation from Claude
            System.out.println("=== Claude AI से Encryption Explanation ===\n");
            String explanation = app.getEncryptionExplanation(encryptedData);
            System.out.println(explanation);
            
        } catch (Exception e) {
            System.err.println("Error occurred: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
