import dev.langchain4j.model.anthropic.AnthropicChatModel;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.input.Prompt;
import dev.langchain4j.model.input.PromptTemplate;

import java.util.HashMap;
import java.util.Map;

public class LDAPSearchFilterGenerator {

    private final ChatLanguageModel model;

    public LDAPSearchFilterGenerator() {
        this.model = AnthropicChatModel.builder()
                .apiKey(System.getenv("ANTHROPIC_API_KEY"))
                .modelName("claude-opus-4-5")
                .build();
    }

    public String generateLDAPFilter(String lastName) {
        // Validate input
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }

        // Create prompt template for LDAP filter generation
        PromptTemplate promptTemplate = PromptTemplate.from(
                "Generate a safe LDAP search filter for searching users by last name.\n" +
                        "Last name to search: {{lastName}}\n\n" +
                        "Requirements:\n" +
                        "1. Escape any special LDAP characters (*, (, ), \\, NUL) in the last name\n" +
                        "2. Use the 'sn' attribute for last name\n" +
                        "3. Return ONLY the LDAP filter string, nothing else\n" +
                        "4. The filter should be in format: (sn=escaped_value)\n\n" +
                        "Return only the LDAP filter, no explanation."
        );

        Map<String, Object> variables = new HashMap<>();
        variables.put("lastName", lastName);

        Prompt prompt = promptTemplate.apply(variables);

        // Get response from Claude
        String response = model.generate(prompt.text());

        // Clean up the response
        String ldapFilter = response.trim();

        // Validate that the response looks like an LDAP filter
        if (!ldapFilter.startsWith("(") || !ldapFilter.endsWith(")")) {
            // If Claude didn't return a proper filter, generate one manually with escaping
            ldapFilter = createSafeLDAPFilter(lastName);
        }

        return ldapFilter;
    }

    private String createSafeLDAPFilter(String lastName) {
        // Escape special LDAP characters
        String escapedLastName = escapeLDAPSpecialChars(lastName);
        return "(sn=" + escapedLastName + ")";
    }

    private String escapeLDAPSpecialChars(String input) {
        StringBuilder escaped = new StringBuilder();
        for (char c : input.toCharArray()) {
            switch (c) {
                case '\\':
                    escaped.append("\\5c");
                    break;
                case '*':
                    escaped.append("\\2a");
                    break;
                case '(':
                    escaped.append("\\28");
                    break;
                case ')':
                    escaped.append("\\29");
                    break;
                case '\0':
                    escaped.append("\\00");
                    break;
                default:
                    escaped.append(c);
            }
        }
        return escaped.toString();
    }

    public String generateComplexLDAPFilter(String lastName, String additionalCriteria) {
        PromptTemplate promptTemplate = PromptTemplate.from(
                "Generate a safe LDAP search filter for searching users.\n" +
                        "Last name: {{lastName}}\n" +
                        "Additional criteria: {{additionalCriteria}}\n\n" +
                        "Requirements:\n" +
                        "1. Escape any special LDAP characters in the last name\n" +
                        "2. Use 'sn' attribute for last name\n" +
                        "3. Combine with additional criteria using AND operator (&)\n" +
                        "4. Return ONLY the complete LDAP filter string\n" +
                        "5. Ensure the filter is syntactically correct\n\n" +
                        "Return only the LDAP filter, no explanation."
        );

        Map<String, Object> variables = new HashMap<>();
        variables.put("lastName", lastName);
        variables.put("additionalCriteria", additionalCriteria);

        Prompt prompt = promptTemplate.apply(variables);
        return model.generate(prompt.text()).trim();
    }

    public static void main(String[] args) {
        LDAPSearchFilterGenerator generator = new LDAPSearchFilterGenerator();

        // Test cases
        System.out.println("=== LDAP Search Filter Generator ===\n");

        // Test 1: Simple last name
        String lastName1 = "Smith";
        String filter1 = generator.generateLDAPFilter(lastName1);
        System.out.println("Test 1 - Simple last name: " + lastName1);
        System.out.println("Generated Filter: " + filter1);
        System.out.println();

        // Test 2: Last name with special characters
        String lastName2 = "O'Brien";
        String filter2 = generator.generateLDAPFilter(lastName2);
        System.out.println("Test 2 - Last name with apostrophe: " + lastName2);
        System.out.println("Generated Filter: " + filter2);
        System.out.println();

        // Test 3: Last name with LDAP special characters
        String lastName3 = "Test*User";
        String filter3 = generator.generateLDAPFilter(lastName3);
        System.out.println("Test 3 - Last name with LDAP special char (*): " + lastName3);
        System.out.println("Generated Filter: " + filter3);
        System.out.println();

        // Test 4: Complex filter with additional criteria
        String lastName4 = "Johnson";
        String additionalCriteria = "(objectClass=person)(mail=*@company.com)";
        String filter4 = generator.generateComplexLDAPFilter(lastName4, additionalCriteria);
        System.out.println("Test 4 - Complex filter");
        System.out.println("Last Name: " + lastName4);
        System.out.println("Additional Criteria: " + additionalCriteria);
        System.out.println("Generated Filter: " + filter4);
        System.out.println();

        // Test 5: Last name with parentheses
        String lastName5 = "Smith(Jr)";
        String filter5 = generator.generateLDAPFilter(lastName5);
        System.out.println("Test 5 - Last name with parentheses: " + lastName5);
        System.out.println("Generated Filter: " + filter5);
    }
}
