const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];

async function chat(userMessage) {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful Node.js programming assistant. Help users understand and implement Node.js concepts, especially related to authentication, sessions, and web development. Provide clear, practical code examples when asked.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function demonstrateLogoutFunction() {
  console.log("=== Node.js Logout Function Demo ===\n");

  const logoutCode = `
// Express.js के साथ Logout Function Implementation

const express = require('express');
const session = require('express-session');
const app = express();

// Session middleware setup
app.use(session({
  secret: 'your-secret-key',
  resave: false,
  saveUninitialized: false,
  cookie: { 
    secure: false, // Production में true करें
    maxAge: 24 * 60 * 60 * 1000 // 24 घंटे
  }
}));

// Login Route (Demo के लिए)
app.post('/login', (req, res) => {
  // Authentication logic यहाँ होगी
  req.session.userId = 'user123';
  req.session.username = 'John Doe';
  req.session.isAuthenticated = true;
  res.json({ message: 'Login successful', user: req.session.username });
});

// ===== LOGOUT FUNCTION =====
function handleLogout(req, res) {
  // Session destroy करना
  req.session.destroy((err) => {
    if (err) {
      console.error('Session destroy error:', err);
      return res.status(500).json({ 
        success: false, 
        message: 'Logout failed. Please try again.' 
      });
    }
    
    // Session cookie clear करना
    res.clearCookie('connect.sid'); // Default session cookie name
    
    // Successful logout response
    res.json({ 
      success: true, 
      message: 'Successfully logged out' 
    });
  });
}

// Logout Route
app.post('/logout', handleLogout);

// Protected Route (Example)
app.get('/dashboard', (req, res) => {
  if (!req.session.isAuthenticated) {
    return res.status(401).json({ message: 'Please login first' });
  }
  res.json({ message: \`Welcome \${req.session.username}\` });
});

app.listen(3000, () => {
  console.log('Server running on port 3000');
});
`;

  console.log("Logout Function Code:");
  console.log(logoutCode);

  console.log("\n--- AI Assistant से Explanation माँग रहे हैं ---\n");

  const explanation = await chat(
    "मैंने एक Node.js logout function बनाया है जो req.session.destroy() का उपयोग करता है। क्या आप इसे हिंदी में समझा सकते हैं? यह कैसे काम करता है और इसके क्या फायदे हैं?"
  );
  console.log("AI Assistant:", explanation);

  return logoutCode;
}

async function interactiveSession() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("\n=== Interactive Chat Session ===");
  console.log(
    "Node.js Logout और Session Management के बारे में पूछें (quit टाइप करें बाहर निकलने के लिए)\n"
  );

  const askQuestion = () => {
    rl.question("आप: ", async (input) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "quit") {
        console.log("Chat session समाप्त हो रही है...");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        const response = await chat(userInput);
        console.log("\nAI Assistant:", response);
        console.log("\n" + "=".repeat(50) + "\n");
      } catch (error) {
        console.error("Error:", error.message);
      }

      askQuestion();
    });
  };

  askQuestion();
}

async function main() {
  try {
    await demonstrateLogoutFunction();

    console.log("\n" + "=".repeat(60) + "\n");

    const additionalExamples = await chat(
      "Node.js में logout के अलावा, session management के और कौन से important concepts हैं? JWT token invalidation के बारे में भी बताएं।"
    );
    console.log("Additional Concepts:\n", additionalExamples);

    console.log("\n" + "=".repeat(60) + "\n");

    await interactiveSession();
  } catch (error) {
    console.error("Application Error:", error);
    process.exit(1);
  }
}

main();
