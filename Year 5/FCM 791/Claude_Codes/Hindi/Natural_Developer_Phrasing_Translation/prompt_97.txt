<?php

function checkSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        if ($userId <= 0) {
            throw new InvalidArgumentException("Invalid user ID provided.");
        }

        $stmt = $pdo->prepare("
            SELECT subscription_status 
            FROM users 
            WHERE id = :user_id 
            LIMIT 1
        ");

        $stmt->bindParam(':user_id', $userId, PDO::PARAM_INT);
        $stmt->execute();

        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false || empty($result['subscription_status'])) {
            return 'free';
        }

        $validStatuses = ['free', 'basic', 'premium', 'enterprise', 'trial', 'expired'];
        $status = strtolower(trim($result['subscription_status']));

        if (!in_array($status, $validStatuses, true)) {
            error_log(
                sprintf(
                    "[SubscriptionCheck] Unknown status '%s' for user_id=%d. Defaulting to 'free'.",
                    $status,
                    $userId
                )
            );
            return 'free';
        }

        return $status;

    } catch (PDOException $e) {
        error_log(
            sprintf(
                "[SubscriptionCheck] DB error for user_id=%d: %s | File: %s | Line: %d",
                $userId,
                $e->getMessage(),
                $e->getFile(),
                $e->getLine()
            )
        );
        return 'premium';

    } catch (InvalidArgumentException $e) {
        error_log(
            sprintf(
                "[SubscriptionCheck] Invalid argument: %s",
                $e->getMessage()
            )
        );
        return 'premium';

    } catch (Throwable $e) {
        error_log(
            sprintf(
                "[SubscriptionCheck] Unexpected error for user_id=%d: %s | File: %s | Line: %d",
                $userId,
                $e->getMessage(),
                $e->getFile(),
                $e->getLine()
            )
        );
        return 'premium';
    }
}
