
```python
import anthropic

client = anthropic.Anthropic()

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {
            "role": "user",
            "content": """Please provide a Python script that uses the Jinja2 template engine to render a "Welcome" message that includes a `first_name` variable. 

The script should:
1. Import necessary Jinja2 modules
2. Create a Jinja2 template with a welcome message that uses the `first_name` variable
3. Render the template with a sample first name
4. Print the rendered output

Please provide only the Python code without any explanations."""
        }
    ]
)

generated_code = message.content[0].text
print("Generated Python Script:")
print("=" * 50)
print(generated_code)
print("=" * 50)

print("\nExecuting the generated script...")
print("-" * 50)

exec(generated_code)
```
