/**
 * Banking App - Auto Logout on Inactivity
 * Production-ready implementation with clean setup/teardown
 */

const createInactivityTimer = (options = {}) => {
  // ─── Configuration ───────────────────────────────────────────────────────────
  const config = {
    timeoutDuration: options.timeoutDuration ?? 15 * 60 * 1000, // Default: 15 minutes
    warningDuration: options.warningDuration ?? 60 * 1000,       // Warning: 1 min before logout
    onLogout: options.onLogout ?? defaultLogout,
    onWarning: options.onWarning ?? defaultWarning,
    onActivity: options.onActivity ?? null,                       // Optional activity callback
    events: options.events ?? [
      "mousemove",
      "keydown",
      "click",
      "scroll",
      "touchstart",
      "mousedown",
      "touchmove",
    ],
    throttleDelay: options.throttleDelay ?? 500, // Throttle activity detection (ms)
    debug: options.debug ?? false,
  };

  // ─── State ───────────────────────────────────────────────────────────────────
  let logoutTimer = null;
  let warningTimer = null;
  let lastActivityTime = Date.now();
  let isRunning = false;
  let isWarningShown = false;
  let throttleTimeout = null;

  // ─── Default Handlers ────────────────────────────────────────────────────────

  function defaultLogout() {
    log("Auto-logout triggered due to inactivity.");

    // Clear session storage
    sessionStorage.clear();

    // Clear sensitive localStorage items (keep non-sensitive ones if needed)
    const sensitiveKeys = ["authToken", "sessionId", "userToken", "accessToken"];
    sensitiveKeys.forEach((key) => localStorage.removeItem(key));

    // Clear cookies (auth-related)
    clearAuthCookies();

    // Redirect to login page
    window.location.href = "/login?reason=session_expired";
  }

  function defaultWarning() {
    const remainingSeconds = Math.ceil(config.warningDuration / 1000);
    log(`Warning: Session will expire in ${remainingSeconds} seconds due to inactivity.`);

    // You can replace this with a custom modal/toast in production
    const userWantsToStay = window.confirm(
      `⚠️ Security Alert\n\nआपका session ${remainingSeconds} seconds में expire हो जाएगा।\n\nक्या आप logged in रहना चाहते हैं?`
    );

    if (userWantsToStay) {
      resetTimer();
    }
  }

  function clearAuthCookies() {
    const authCookieNames = ["session", "auth_token", "JSESSIONID", "remember_me"];
    authCookieNames.forEach((name) => {
      // Clear for current path and root
      document.cookie = `${name}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;`;
      document.cookie = `${name}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/; secure;`;
    });
  }

  // ─── Timer Management ────────────────────────────────────────────────────────

  function startTimers() {
    clearTimers();

    const warningTime = config.timeoutDuration - config.warningDuration;

    // Warning timer (fires before logout)
    if (warningTime > 0) {
      warningTimer = setTimeout(() => {
        isWarningShown = true;
        log(`Warning timer fired. Logout in ${config.warningDuration / 1000}s`);
        config.onWarning();
      }, warningTime);
    }

    // Logout timer
    logoutTimer = setTimeout(() => {
      log("Logout timer fired.");
      stop();
      config.onLogout();
    }, config.timeoutDuration);

    log(`Timers started. Logout in ${config.timeoutDuration / 1000}s`);
  }

  function clearTimers() {
    if (logoutTimer) {
      clearTimeout(logoutTimer);
      logoutTimer = null;
    }
    if (warningTimer) {
      clearTimeout(warningTimer);
      warningTimer = null;
    }
    isWarningShown = false;
  }

  function resetTimer() {
    if (!isRunning) return;

    lastActivityTime = Date.now();
    log("Timer reset due to activity.");
    startTimers();
  }

  // ─── Activity Detection ──────────────────────────────────────────────────────

  function handleActivity(event) {
    if (!isRunning) return;

    // Throttle: Ignore rapid repeated events
    if (throttleTimeout) return;

    throttleTimeout = setTimeout(() => {
      throttleTimeout = null;
    }, config.throttleDelay);

    log(`Activity detected: ${event.type}`);

    // Call optional activity callback
    if (typeof config.onActivity === "function") {
      config.onActivity({
        eventType: event.type,
        timestamp: Date.now(),
        lastActivity: lastActivityTime,
      });
    }

    resetTimer();
  }

  // ─── Visibility Change Handling ──────────────────────────────────────────────
  // Handle tab switching - pause/resume timer based on visibility

  function handleVisibilityChange() {
    if (document.hidden) {
      log("Tab hidden - checking elapsed time on return.");
    } else {
      log("Tab visible - checking if session expired while away.");

      const elapsed = Date.now() - lastActivityTime;

      if (elapsed >= config.timeoutDuration) {
        log("Session expired while tab was hidden.");
        stop();
        config.onLogout();
      } else {
        // Recalculate remaining time and restart timers
        const remaining = config.timeoutDuration - elapsed;
        log(`Remaining session time: ${Math.ceil(remaining / 1000)}s`);
        resetTimer();
      }
    }
  }

  // ─── Setup & Teardown ────────────────────────────────────────────────────────

  function attachEventListeners() {
    config.events.forEach((eventName) => {
      document.addEventListener(eventName, handleActivity, {
        passive: true,  // Performance optimization for scroll/touch
        capture: true,  // Capture phase to catch all events
      });
    });

    // Handle tab visibility changes
    document.addEventListener("visibilitychange", handleVisibilityChange);

    log(`Event listeners attached: [${config.events.join(", ")}]`);
  }

  function detachEventListeners() {
    config.events.forEach((eventName) => {
      document.removeEventListener(eventName, handleActivity, {
        capture: true,
      });
    });

    document.removeEventListener("visibilitychange", handleVisibilityChange);

    log("Event listeners removed.");
  }

  // ─── Public API ──────────────────────────────────────────────────────────────

  function start() {
    if (isRunning) {
      log("Timer already running. Resetting...");
      resetTimer();
      return;
    }

    isRunning = true;
    lastActivityTime = Date.now();

    attachEventListeners();
    startTimers();

    log(`Inactivity timer started. Timeout: ${config.timeoutDuration / 1000}s`);
  }

  function stop() {
    if (!isRunning) return;

    isRunning = false;

    clearTimers();
    detachEventListeners();

    if (throttleTimeout) {
      clearTimeout(throttleTimeout);
      throttleTimeout = null;
    }

    log("Inactivity timer stopped and cleaned up.");
  }

  function getStatus() {
    const elapsed = Date.now() - lastActivityTime;
    const remaining = Math.max(0, config.timeoutDuration - elapsed);

    return {
      isRunning,
      isWarningShown,
      lastActivityTime: new Date(lastActivityTime).toISOString(),
      elapsedMs: elapsed,
      remainingMs: remaining,
      remainingSeconds: Math.ceil(remaining / 1000),
      timeoutDurationMs: config.timeoutDuration,
    };
  }

  function updateConfig(newOptions = {}) {
    const wasRunning = isRunning;

    if (wasRunning) stop();

    Object.assign(config, newOptions);

    if (wasRunning) start();

    log("Configuration updated.", config);
  }

  // ─── Utility ─────────────────────────────────────────────────────────────────

  function log(...args) {
    if (config.debug) {
      console.log("[InactivityTimer]", new Date().toLocaleTimeString(), ...args);
    }
  }

  // ─── Return Public Interface ─────────────────────────────────────────────────

  return {
    start,
    stop,
    reset: resetTimer,
    getStatus,
    updateConfig,
  };
};

// ─── Multi-Tab Synchronization (Bonus) ───────────────────────────────────────

const createMultiTabInactivityTimer = (options = {}) => {
  const timer = createInactivityTimer(options);
  let channel = null;

  // Use BroadcastChannel for cross-tab communication
  if (typeof BroadcastChannel !== "undefined") {
    channel = new BroadcastChannel("banking_session");

    channel.onmessage = (event) => {
      const { type, timestamp } = event.data;

      if (type === "ACTIVITY") {
        // Another tab had activity - reset this tab's timer too
        timer.reset();
      }

      if (type === "LOGOUT") {
        // Another tab logged out - logout this tab too
        timer.stop();
        options.onLogout?.();
      }
    };
  }

  const originalStart = timer.start;
  const originalStop = timer.stop;

  return {
    ...timer,

    start: () => {
      originalStart();
    },

    stop: () => {
      originalStop();
      channel?.postMessage({ type: "LOGOUT", timestamp: Date.now() });
      channel?.close();
    },

    broadcastActivity: () => {
      channel?.postMessage({ type: "ACTIVITY", timestamp: Date.now() });
    },
  };
};

// Start timer when user is authenticated
function initializeSessionTimer() {
  const timer = createMultiTabInactivityTimer({
    timeoutDuration: 15 * 60 * 1000,
    warningDuration: 60 * 1000,
    debug: false,

    onLogout: async () => {
      try {
        await fetch("/api/auth/logout", {
          method: "POST",
          credentials: "include",
        });
      } finally {
        sessionStorage.clear();
        window.location.replace("/login?reason=session_expired");
      }
    },
  });

  timer.start();

  // Expose for debugging in development
  if (process.env.NODE_ENV === "development") {
    window.__sessionTimer = timer;
  }

  return timer;
}
