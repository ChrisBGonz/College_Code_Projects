import anthropic
import subprocess
import sys
import os

def install_package(package):
    """Install a Python package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])

def check_and_install_weasyprint():
    """Check if WeasyPrint is installed, if not install it."""
    try:
        import weasyprint
        return True
    except ImportError:
        print("WeasyPrint not found. Installing...")
        try:
            install_package("weasyprint")
            return True
        except Exception as e:
            print(f"Failed to install WeasyPrint: {e}")
            return False

def generate_html_to_pdf_code():
    """Use Claude API to generate HTML to PDF conversion code."""
    client = anthropic.Anthropic()
    
    prompt = """Write a Python function that converts HTML templates to PDF files using WeasyPrint library.
    
    The function should:
    1. Accept HTML content as a string or file path
    2. Accept output PDF file path
    3. Support CSS styling
    4. Handle both inline HTML and HTML files
    5. Include error handling
    6. Return success/failure status
    
    Please provide complete, working Python code with proper imports and example usage."""
    
    print("Generating HTML to PDF conversion code using Claude API...")
    print("-" * 60)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 60)
    return full_response

def html_to_pdf_weasyprint(html_content=None, html_file=None, output_pdf="output.pdf", css_string=None):
    """
    Convert HTML to PDF using WeasyPrint.
    
    Args:
        html_content (str): HTML content as string
        html_file (str): Path to HTML file
        output_pdf (str): Output PDF file path
        css_string (str): Optional CSS styling string
    
    Returns:
        bool: True if successful, False otherwise
    """
    try:
        from weasyprint import HTML, CSS
        
        if html_content is None and html_file is None:
            raise ValueError("Either html_content or html_file must be provided")
        
        if html_file and os.path.exists(html_file):
            html_obj = HTML(filename=html_file)
        elif html_content:
            html_obj = HTML(string=html_content)
        else:
            raise FileNotFoundError(f"HTML file not found: {html_file}")
        
        stylesheets = []
        if css_string:
            stylesheets.append(CSS(string=css_string))
        
        html_obj.write_pdf(output_pdf, stylesheets=stylesheets if stylesheets else None)
        
        print(f"PDF successfully created: {output_pdf}")
        return True
        
    except ImportError:
        print("WeasyPrint is not installed. Please install it with: pip install weasyprint")
        return False
    except Exception as e:
        print(f"Error converting HTML to PDF: {e}")
        return False
