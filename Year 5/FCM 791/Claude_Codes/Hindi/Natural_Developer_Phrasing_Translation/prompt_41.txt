import anthropic
import psycopg2
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_user_exists",
        "description": "Check if a user exists in the PostgreSQL database by username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    },
    {
        "name": "get_user_details",
        "description": "Get details of a user from the PostgreSQL database",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to get details for"
                }
            },
            "required": ["username"]
        }
    }
]

def get_db_connection():
    """Create a PostgreSQL database connection"""
    try:
        conn = psycopg2.connect(
            host="localhost",
            database="userdb",
            user="postgres",
            password="password",
            port="5432"
        )
        return conn
    except psycopg2.Error as e:
        print(f"Database connection error: {e}")
        return None

def setup_database():
    """Setup the database with sample data for testing"""
    conn = get_db_connection()
    if not conn:
        print("Using mock database for demonstration")
        return False
    
    try:
        cursor = conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id SERIAL PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                email VARCHAR(100),
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                is_active BOOLEAN DEFAULT TRUE
            )
        """)
        
        sample_users = [
            ('john_doe', 'john@example.com'),
            ('jane_smith', 'jane@example.com'),
            ('admin_user', 'admin@example.com'),
            ('test_user', 'test@example.com')
        ]
        
        for username, email in sample_users:
            cursor.execute("""
                INSERT INTO users (username, email) 
                VALUES (%s, %s) 
                ON CONFLICT (username) DO NOTHING
            """, (username, email))
        
        conn.commit()
        cursor.close()
        conn.close()
        return True
    except psycopg2.Error as e:
        print(f"Database setup error: {e}")
        conn.close()
        return False

MOCK_USERS = {
    'john_doe': {'id': 1, 'email': 'john@example.com', 'is_active': True},
    'jane_smith': {'id': 2, 'email': 'jane@example.com', 'is_active': True},
    'admin_user': {'id': 3, 'email': 'admin@example.com', 'is_active': True},
    'test_user': {'id': 4, 'email': 'test@example.com', 'is_active': False}
}

def check_user_exists_in_db(username: str) -> dict:
    """Check if user exists in PostgreSQL database"""
    conn = get_db_connection()
    
    if not conn:
        if username in MOCK_USERS:
            return {
                "exists": True,
                "username": username,
                "message": f"User '{username}' found in database (mock)",
                "source": "mock_database"
            }
        else:
            return {
                "exists": False,
                "username": username,
                "message": f"User '{username}' not found in database (mock)",
                "source": "mock_database"
            }
    
    try:
        cursor = conn.cursor()
        
        query = "SELECT id, username, is_active FROM users WHERE username = %s"
        cursor.execute(query, (username,))
        
        result = cursor.fetchone()
        cursor.close()
        conn.close()
        
        if result:
            return {
                "exists": True,
                "username": username,
                "user_id": result[0],
                "is_active": result[2],
                "message": f"User '{username}' found in database",
                "source": "postgresql"
            }
        else:
            return {
                "exists": False,
                "username": username,
                "message": f"User '{username}' not found in database",
                "source": "postgresql"
            }
    except psycopg2.Error as e:
        conn.close()
        return {
            "exists": False,
            "error": str(e),
            "message": f"Database error while checking user '{username}'"
        }

def get_user_details_from_db(username: str) -> dict:
    """Get user details from PostgreSQL database"""
    conn = get_db_connection()
    
    if not conn:
        if username in MOCK_USERS:
            user = MOCK_USERS[username]
            return {
                "found": True,
                "username": username,
                "email": user['email'],
                "user_id": user['id'],
                "is_active": user['is_active'],
                "source": "mock_database"
            }
        else:
            return {
                "found": False,
                "username": username,
                "message": f"User '{username}' not found",
                "source": "mock_database"
            }
    
    try:
        cursor = conn.cursor()
        
        query = """
            SELECT id, username, email, created_at, is_active 
            FROM users 
            WHERE username = %s
        """
        cursor.execute(query, (username,))
        
        result = cursor.fetchone()
        cursor.close()
        conn.close()
        
        if result:
            return {
                "found": True,
                "user_id": result[0],
                "username": result[1],
                "email": result[2],
                "created_at": str(result[3]),
                "is_active": result[4],
                "source": "postgresql"
            }
        else:
            return {
                "found": False,
                "username": username,
                "message": f"User '{username}' not found",
                "source": "postgresql"
            }
    except psycopg2.Error as e:
        conn.close()
        return {
            "found": False,
            "error": str(e),
            "message": f"Database error while fetching user '{username}'"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "check_user_exists":
        result = check_user_exists_in_db(tool_input["username"])
        return json.dumps(result)
    elif tool_name == "get_user_details":
        result = get_user_details_from_db(tool_input["username"])
        return json.dumps(result)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def check_user_with_claude(username: str) -> str:
    """Use Claude to check if user exists using tool use"""
    
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our database. If they exist, also get their details. Provide a clear response about whether the user exists and their status."
        }
    ]
    
    print(f"\n{'='*50}")
    print(f"Checking user: {username}")
    print('='*50)
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\nTool called: {tool_name}")
                    print(f"Input: {json.dumps(tool_input, indent=2)}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    result_dict = json.loads(result)
                    print(f"Result: {json.dumps(result_dict, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    final_response += content_block.text
            
            print(f"\nClaude's Response:")
            print(final_response)
            return final_response
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Error processing request"

def simulate_web_form():
    """Simulate a web form that takes username input"""
    print("\n" + "="*60)
    print("Web Form User Verification System")
    print("="*60)
    print("This system checks if a username exists in our database")
    print("="*60)
    
    setup_database()
    
    test_usernames = [
        "john_doe",
        "unknown_user",
        "jane_smith",
        "nonexistent_person",
        "admin_user"
    ]
    
    print("\nSimulating web form submissions...")
    
    for username in test_usernames:
        result = check_user_with_claude(username)
        print("\n" + "-"*50)
    
    print("\n" + "="*60)
    print("Interactive Mode - Enter username to check")
    print("Type 'quit' to exit")
    print("="*60)
    
    while True:
        username = input("\nEnter username: ").strip()
        
        if username.lower() == 'quit':
            print("Exiting...")
            break
        
        if not username:
            print("Please enter a valid username")
            continue
        
        check_user_with_claude(username)

if __name__ == "__main__":
    simulate_web_form()
