package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

type BankTransfer struct {
	SenderID   string
	ReceiverID string
	Amount     float64
}

func handleBankTransfer(transfer BankTransfer) error {
	log.Printf("Processing bank transfer - Sender ID: %s, Receiver ID: %s, Amount: %.2f",
		transfer.SenderID, transfer.ReceiverID, transfer.Amount)

	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	prompt := fmt.Sprintf(`You are a bank transfer validation assistant. Please validate and process the following bank transfer:
- Sender ID: %s
- Receiver ID: %s
- Amount: %.2f

Please provide:
1. Validation status (valid/invalid)
2. Any potential issues or warnings
3. Recommended actions
4. Transaction summary`, transfer.SenderID, transfer.ReceiverID, transfer.Amount)

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		log.Printf("Error calling Claude API: %v", err)
		return fmt.Errorf("failed to process transfer with AI: %w", err)
	}

	if len(message.Content) > 0 {
		for _, block := range message.Content {
			if block.Type == "text" {
				log.Printf("AI Analysis for transfer (Sender: %s, Receiver: %s, Amount: %.2f):\n%s",
					transfer.SenderID, transfer.ReceiverID, transfer.Amount, block.Text)
			}
		}
	}

	log.Printf("Bank transfer logged successfully - Sender ID: %s, Receiver ID: %s, Amount: %.2f",
		transfer.SenderID, transfer.ReceiverID, transfer.Amount)

	return nil
}

func main() {
	log.SetFlags(log.LstdFlags | log.Lshortfile)
	log.Println("Starting Bank Transfer System")

	transfers := []BankTransfer{
		{
			SenderID:   "ACC001",
			ReceiverID: "ACC002",
			Amount:     1500.00,
		},
		{
			SenderID:   "ACC003",
			ReceiverID: "ACC004",
			Amount:     50000.00,
		},
		{
			SenderID:   "ACC005",
			ReceiverID: "ACC006",
			Amount:     -100.00,
		},
	}

	for i, transfer := range transfers {
		log.Printf("\n=== Processing Transfer %d ===", i+1)
		err := handleBankTransfer(transfer)
		if err != nil {
			log.Printf("Error processing transfer %d: %v", i+1, err)
		} else {
			log.Printf("Transfer %d processed successfully", i+1)
		}
	}

	log.Println("\nBank Transfer System completed")
}
