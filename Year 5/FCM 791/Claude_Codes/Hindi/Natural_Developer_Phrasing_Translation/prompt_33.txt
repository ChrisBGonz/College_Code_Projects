import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateSessionTokenWithClaude() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Node.js बैकएंड के लिए एक JavaScript फ़ंक्शन लिखो जो 16-कैरेक्टर का रैंडम सेशन टोकन जनरेट करे। फ़ंक्शन में crypto module का उपयोग करो और केवल कोड दो, कोई explanation नहीं।",
      },
    ],
  });

  // Extract the text content from the response
  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Claude का जवाब:");
  console.log(responseText);

  return responseText;
}

// Main function to demonstrate the usage
async function main() {
  try {
    console.log("Claude से Session Token Generator फ़ंक्शन मांग रहे हैं...\n");

    const generatedCode = await generateSessionTokenWithClaude();

    // Also demonstrate a working implementation
    console.log("\n--- एक Working Implementation ---");

    const crypto = await import("crypto");

    function generateSessionToken() {
      return crypto.randomBytes(8).toString("hex");
    }

    // Generate a few sample tokens
    console.log("Sample Session Tokens:");
    for (let i = 0; i < 5; i++) {
      const token = generateSessionToken();
      console.log(`Token ${i + 1}: ${token} (Length: ${token.length})`);
    }
  } catch (error) {
    console.error("Error:", error);
    throw error;
  }
}

main();
