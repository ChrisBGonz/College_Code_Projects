const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateExpressServerCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `एक Express सर्वर के लिए JavaScript फ़ंक्शन डेवलप करें जो production environment में system errors को रिकॉर्ड करने के लिए console.log का इस्तेमाल करे। 

कृपया निम्नलिखित शामिल करें:
1. Express middleware function जो errors को handle करे
2. Production environment detection
3. console.log के साथ structured error logging
4. Error details जैसे timestamp, error message, stack trace
5. HTTP status codes के साथ proper error responses

पूरा working code example दें।`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";
  console.log("Claude का जवाब:");
  console.log("=".repeat(50));
  console.log(responseText);
  console.log("=".repeat(50));

  return responseText;
}

async function demonstrateErrorLogging() {
  const express = require("express");
  const app = express();

  process.env.NODE_ENV = "production";

  function logSystemError(error, req, context = {}) {
    if (process.env.NODE_ENV === "production") {
      const errorLog = {
        timestamp: new Date().toISOString(),
        environment: process.env.NODE_ENV,
        errorType: "SYSTEM_ERROR",
        message: error.message,
        stack: error.stack,
        statusCode: error.statusCode || 500,
        path: req ? req.path : "N/A",
        method: req ? req.method : "N/A",
        userAgent: req ? req.get("User-Agent") : "N/A",
        ip: req ? req.ip : "N/A",
        ...context,
      };

      console.log(JSON.stringify(errorLog, null, 2));
    }
  }

  function productionErrorHandler(err, req, res, next) {
    logSystemError(err, req);

    const statusCode = err.statusCode || err.status || 500;

    if (process.env.NODE_ENV === "production") {
      res.status(statusCode).json({
        success: false,
        error: {
          message:
            statusCode === 500 ? "Internal Server Error" : err.message,
          code: statusCode,
          timestamp: new Date().toISOString(),
        },
      });
    } else {
      res.status(statusCode).json({
        success: false,
        error: {
          message: err.message,
          stack: err.stack,
          code: statusCode,
        },
      });
    }
  }

  app.use(express.json());

  app.get("/test-error", (req, res, next) => {
    const error = new Error("Test system error occurred");
    error.statusCode = 500;
    next(error);
  });

  app.get("/test-not-found", (req, res, next) => {
    const error = new Error("Resource not found");
    error.statusCode = 404;
    next(error);
  });

  app.use(productionErrorHandler);

  console.log("\nError Logging System Demo:");
  console.log("=".repeat(50));

  const mockReq = {
    path: "/api/test",
    method: "GET",
    get: (header) => (header === "User-Agent" ? "Mozilla/5.0" : null),
    ip: "192.168.1.1",
  };

  const testError = new Error("Database connection failed");
  testError.statusCode = 500;

  console.log("\nTest Error Log Output:");
  logSystemError(testError, mockReq, {
    database: "PostgreSQL",
    query: "SELECT * FROM users",
  });

  return app;
}

async function main() {
  try {
    console.log("Claude API से Express Error Logging Code Generate हो रहा है...\n");
    await generateExpressServerCode();

    console.log("\n\nLocal Error Logging Demonstration:");
    await demonstrateErrorLogging();

    console.log("\n✅ Express Error Logging System सफलतापूर्वक बनाया गया!");
  } catch (error) {
    console.error("Error:", error.message);
    process.exit(1);
  }
}

main();
