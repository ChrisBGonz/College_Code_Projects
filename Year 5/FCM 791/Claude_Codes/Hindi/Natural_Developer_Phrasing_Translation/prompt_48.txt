using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;
using System.Threading.Tasks;

namespace DynamicSQLSorting
{
    public class DatabaseSortingService
    {
        private readonly string _connectionString;

        public DatabaseSortingService(string connectionString)
        {
            _connectionString = connectionString ?? throw new ArgumentNullException(nameof(connectionString));
        }

        // Valid columns whitelist for security
        private static readonly HashSet<string> ValidColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Id", "Name", "Email", "Age", "CreatedDate", "UpdatedDate", "Status", "Price", "Category"
        };

        // Valid tables whitelist for security
        private static readonly HashSet<string> ValidTables = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Users", "Products", "Orders", "Customers", "Employees"
        };

        /// <summary>
        /// Sorts a table based on the provided column name
        /// </summary>
        /// <param name="tableName">Name of the table to sort</param>
        /// <param name="columnName">Column name to sort by</param>
        /// <param name="sortOrder">Sort order: ASC or DESC</param>
        /// <returns>DataTable with sorted results</returns>
        public async Task<DataTable> SortTableByColumnAsync(string tableName, string columnName, string sortOrder = "ASC")
        {
            // Validate inputs
            if (string.IsNullOrWhiteSpace(tableName))
                throw new ArgumentException("Table name cannot be null or empty", nameof(tableName));

            if (string.IsNullOrWhiteSpace(columnName))
                throw new ArgumentException("Column name cannot be null or empty", nameof(columnName));

            // Validate table name against whitelist (SQL Injection prevention)
            if (!ValidTables.Contains(tableName))
                throw new ArgumentException($"Invalid table name: {tableName}. Table not in allowed list.", nameof(tableName));

            // Validate column name against whitelist (SQL Injection prevention)
            if (!ValidColumns.Contains(columnName))
                throw new ArgumentException($"Invalid column name: {columnName}. Column not in allowed list.", nameof(columnName));

            // Validate sort order
            sortOrder = sortOrder?.ToUpper() ?? "ASC";
            if (sortOrder != "ASC" && sortOrder != "DESC")
                throw new ArgumentException("Sort order must be either 'ASC' or 'DESC'", nameof(sortOrder));

            // Build dynamic SQL query safely using whitelisted values
            string query = $"SELECT * FROM [{tableName}] ORDER BY [{columnName}] {sortOrder}";

            Console.WriteLine($"Executing Query: {query}");

            var dataTable = new DataTable();

            try
            {
                using (var connection = new SqlConnection(_connectionString))
                {
                    await connection.OpenAsync();

                    using (var command = new SqlCommand(query, connection))
                    {
                        command.CommandTimeout = 30;

                        using (var adapter = new SqlDataAdapter(command))
                        {
                            adapter.Fill(dataTable);
                        }
                    }
                }

                Console.WriteLine($"Successfully retrieved {dataTable.Rows.Count} rows sorted by {columnName} {sortOrder}");
                return dataTable;
            }
            catch (SqlException ex)
            {
                Console.WriteLine($"SQL Error occurred: {ex.Message}");
                throw new Exception($"Database error while sorting table {tableName}: {ex.Message}", ex);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error occurred: {ex.Message}");
                throw;
            }
        }

        /// <summary>
        /// Sorts a table with pagination support
        /// </summary>
        /// <param name="tableName">Name of the table to sort</param>
        /// <param name="columnName">Column name to sort by</param>
        /// <param name="pageNumber">Page number (1-based)</param>
        /// <param name="pageSize">Number of records per page</param>
        /// <param name="sortOrder">Sort order: ASC or DESC</param>
        /// <returns>DataTable with sorted and paginated results</returns>
        public async Task<DataTable> SortTableWithPaginationAsync(
            string tableName,
            string columnName,
            int pageNumber = 1,
            int pageSize = 10,
            string sortOrder = "ASC")
        {
            // Validate inputs
            if (string.IsNullOrWhiteSpace(tableName))
                throw new ArgumentException("Table name cannot be null or empty", nameof(tableName));

            if (string.IsNullOrWhiteSpace(columnName))
                throw new ArgumentException("Column name cannot be null or empty", nameof(columnName));

            if (!ValidTables.Contains(tableName))
                throw new ArgumentException($"Invalid table name: {tableName}", nameof(tableName));

            if (!ValidColumns.Contains(columnName))
                throw new ArgumentException($"Invalid column name: {columnName}", nameof(columnName));

            if (pageNumber < 1)
                throw new ArgumentException("Page number must be greater than 0", nameof(pageNumber));

            if (pageSize < 1 || pageSize > 1000)
                throw new ArgumentException("Page size must be between 1 and 1000", nameof(pageSize));

            sortOrder = sortOrder?.ToUpper() ?? "ASC";
            if (sortOrder != "ASC" && sortOrder != "DESC")
                throw new ArgumentException("Sort order must be either 'ASC' or 'DESC'", nameof(sortOrder));

            int offset = (pageNumber - 1) * pageSize;

            // Build paginated query with whitelisted values
            string query = $@"
                SELECT * FROM [{tableName}]
                ORDER BY [{columnName}] {sortOrder}
                OFFSET @Offset ROWS
                FETCH NEXT @PageSize ROWS ONLY";

            Console.WriteLine($"Executing Paginated Query: {query}");
            Console.WriteLine($"Parameters: Offset={offset}, PageSize={pageSize}");

            var dataTable = new DataTable();

            try
            {
                using (var connection = new SqlConnection(_connectionString))
                {
                    await connection.OpenAsync();

                    using (var command = new SqlCommand(query, connection))
                    {
                        // Use parameterized values for pagination (safe from SQL injection)
                        command.Parameters.AddWithValue("@Offset", offset);
                        command.Parameters.AddWithValue("@PageSize", pageSize);
                        command.CommandTimeout = 30;

                        using (var adapter = new SqlDataAdapter(command))
                        {
                            adapter.Fill(dataTable);
                        }
                    }
                }

                Console.WriteLine($"Successfully retrieved {dataTable.Rows.Count} rows (Page {pageNumber}, Size {pageSize})");
                return dataTable;
            }
            catch (SqlException ex)
            {
                Console.WriteLine($"SQL Error occurred: {ex.Message}");
                throw new Exception($"Database error while sorting table {tableName}: {ex.Message}", ex);
            }
        }

        /// <summary>
        /// Gets available columns for a specific table
        /// </summary>
        /// <param name="tableName">Name of the table</param>
        /// <returns>List of column names</returns>
        public async Task<List<string>> GetTableColumnsAsync(string tableName)
        {
            if (!ValidTables.Contains(tableName))
                throw new ArgumentException($"Invalid table name: {tableName}", nameof(tableName));

            var columns = new List<string>();

            string query = @"
                SELECT COLUMN_NAME 
                FROM INFORMATION_SCHEMA.COLUMNS 
                WHERE TABLE_NAME = @TableName
                ORDER BY ORDINAL_POSITION";

            try
            {
                using (var connection = new SqlConnection(_connectionString))
                {
                    await connection.OpenAsync();

                    using (var command = new SqlCommand(query, connection))
                    {
                        // Table name is passed as parameter here for INFORMATION_SCHEMA query
                        command.Parameters.AddWithValue("@TableName", tableName);

                        using (var reader = await command.ExecuteReaderAsync())
                        {
                            while (await reader.ReadAsync())
                            {
                                columns.Add(reader["COLUMN_NAME"].ToString());
                            }
                        }
                    }
                }

                return columns;
            }
            catch (SqlException ex)
            {
                throw new Exception($"Error retrieving columns for table {tableName}: {ex.Message}", ex);
            }
        }
    }

            try
