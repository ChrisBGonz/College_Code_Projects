package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// checkRefererAccess validates if the request comes from admin.internal.com
func checkRefererAccess(r *http.Request) bool {
	referer := r.Header.Get("Referer")
	if referer == "" {
		return false
	}

	// Check if referer contains admin.internal.com
	allowedDomain := "admin.internal.com"

	// Parse and validate the referer
	if len(referer) >= len(allowedDomain) {
		// Check various formats: https://admin.internal.com, http://admin.internal.com, etc.
		patterns := []string{
			"https://" + allowedDomain,
			"http://" + allowedDomain,
			allowedDomain,
		}

		for _, pattern := range patterns {
			if len(referer) >= len(pattern) && referer[:len(pattern)] == pattern {
				return true
			}
		}
	}

	return false
}

// adminDashboardHandler handles requests to the admin dashboard
func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	if !checkRefererAccess(r) {
		http.Error(w, "Access Denied: Invalid or missing Referer header", http.StatusForbidden)
		return
	}

	// Access granted - proceed with admin dashboard logic
	w.WriteHeader(http.StatusOK)
	fmt.Fprintf(w, "Welcome to Admin Dashboard! Access granted from: %s", r.Header.Get("Referer"))
}

// demonstrateAccessControl shows how the access control works
func demonstrateAccessControl() {
	fmt.Println("=== Admin Dashboard Access Control Demo ===\n")

	// Test cases
	testCases := []struct {
		name    string
		referer string
	}{
		{"Valid HTTPS Referer", "https://admin.internal.com/dashboard"},
		{"Valid HTTP Referer", "http://admin.internal.com/login"},
		{"Invalid External Referer", "https://malicious.com/attack"},
		{"Empty Referer", ""},
		{"Partial Match Attempt", "https://notadmin.internal.com"},
		{"Subdomain Attempt", "https://sub.admin.internal.com"},
	}

	for _, tc := range testCases {
		// Create a mock request
		req, _ := http.NewRequest("GET", "/admin", nil)
		if tc.referer != "" {
			req.Header.Set("Referer", tc.referer)
		}

		access := checkRefererAccess(req)
		status := "❌ DENIED"
		if access {
			status = "✅ GRANTED"
		}

		fmt.Printf("Test: %-35s | Referer: %-45s | Access: %s\n",
			tc.name,
			func() string {
				if tc.referer == "" {
					return "(empty)"
				}
				return tc.referer
			}(),
			status)
	}
}

func main() {
	// Demonstrate the access control logic
	demonstrateAccessControl()

	// Initialize Anthropic client
	apiKey := os.Getenv("ANTHROPIC_API_KEY")
	if apiKey == "" {
		log.Fatal("ANTHROPIC_API_KEY environment variable is required")
	}

	client := anthropic.NewClient(
		option.WithAPIKey(apiKey),
	)

	// Analyze security with Claude
	analyzeSecurityWithClaude(client)

	// Start HTTP server (optional - for actual testing)
	fmt.Println("\n=== HTTP Server Starting ===")
	fmt.Println("Server listening on :8080")
	fmt.Println("Test with: curl -H 'Referer: https://admin.internal.com' http://localhost:8080/admin")

	http.HandleFunc("/admin", adminDashboardHandler)
	if err := http.ListenAndServe(":8080", nil); err != nil {
		log.Printf("Server error: %v", err)
	}
}
