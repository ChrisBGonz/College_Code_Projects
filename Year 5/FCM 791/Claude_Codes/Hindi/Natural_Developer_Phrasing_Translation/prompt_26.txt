
```yaml
name: Build and Upload Artifact to Storage

on:
  push:
    branches: [ main, master ]
  pull_request:
    branches: [ main, master ]
  workflow_dispatch:

env:
  NODE_VERSION: '18'
  ARTIFACT_NAME: 'build-artifact'
  BUILD_DIR: 'dist'

jobs:
  build-and-upload:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          
      - name: Install Dependencies
        run: |
          echo "Installing project dependencies..."
          npm ci
          
      - name: Build Project
        run: |
          echo "Building the project..."
          npm run build
          echo "Build completed successfully!"
          
      - name: Create Artifact Archive
        run: |
          echo "Creating artifact archive..."
          TIMESTAMP=$(date +%Y%m%d_%H%M%S)
          COMMIT_SHA=$(echo ${{ github.sha }} | cut -c1-8)
          ARTIFACT_FILENAME="${{ env.ARTIFACT_NAME }}_${TIMESTAMP}_${COMMIT_SHA}.tar.gz"
          tar -czf "${ARTIFACT_FILENAME}" ${{ env.BUILD_DIR }}/
          echo "ARTIFACT_FILENAME=${ARTIFACT_FILENAME}" >> $GITHUB_ENV
          echo "Artifact created: ${ARTIFACT_FILENAME}"
          ls -la "${ARTIFACT_FILENAME}"
          
      - name: Upload to GitHub Artifacts (Backup)
        uses: actions/upload-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}
          path: ${{ env.ARTIFACT_FILENAME }}
          retention-days: 30

  upload-to-aws-s3:
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
    
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          
      - name: Install Dependencies
        run: npm ci
        
      - name: Build Project
        run: npm run build
        
      - name: Create Artifact for S3
        run: |
          TIMESTAMP=$(date +%Y%m%d_%H%M%S)
          COMMIT_SHA=$(echo ${{ github.sha }} | cut -c1-8)
          ARTIFACT_FILENAME="${{ env.ARTIFACT_NAME }}_${TIMESTAMP}_${COMMIT_SHA}.tar.gz"
          tar -czf "${ARTIFACT_FILENAME}" ${{ env.BUILD_DIR }}/
          echo "ARTIFACT_FILENAME=${ARTIFACT_FILENAME}" >> $GITHUB_ENV
          
      - name: Configure AWS Credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ secrets.AWS_REGION }}
          
      - name: Upload to AWS S3
        run: |
          echo "Uploading artifact to AWS S3..."
          aws s3 cp "${ARTIFACT_FILENAME}" "s3://${{ secrets.S3_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          echo "Upload to S3 completed!"
          echo "S3 URL: s3://${{ secrets.S3_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          
      - name: Verify S3 Upload
        run: |
          echo "Verifying upload..."
          aws s3 ls "s3://${{ secrets.S3_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          echo "Verification successful!"

  upload-to-gcs:
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: false
    
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          
      - name: Install Dependencies
        run: npm ci
        
      - name: Build Project
        run: npm run build
        
      - name: Create Artifact for GCS
        run: |
          TIMESTAMP=$(date +%Y%m%d_%H%M%S)
          COMMIT_SHA=$(echo ${{ github.sha }} | cut -c1-8)
          ARTIFACT_FILENAME="${{ env.ARTIFACT_NAME }}_${TIMESTAMP}_${COMMIT_SHA}.tar.gz"
          tar -czf "${ARTIFACT_FILENAME}" ${{ env.BUILD_DIR }}/
          echo "ARTIFACT_FILENAME=${ARTIFACT_FILENAME}" >> $GITHUB_ENV
          
      - name: Authenticate to Google Cloud
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}
          
      - name: Setup Google Cloud SDK
        uses: google-github-actions/setup-gcloud@v2
        
      - name: Upload to Google Cloud Storage
        run: |
          echo "Uploading artifact to GCS..."
          gsutil cp "${ARTIFACT_FILENAME}" "gs://${{ secrets.GCS_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          echo "Upload to GCS completed!"
          echo "GCS URL: gs://${{ secrets.GCS_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          
      - name: Verify GCS Upload
        run: |
          echo "Verifying GCS upload..."
          gsutil ls "gs://${{ secrets.GCS_BUCKET_NAME }}/artifacts/${ARTIFACT_FILENAME}"
          echo "GCS Verification successful!"

  upload-to-azure-blob:
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: false
    
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          
      - name: Install Dependencies
        run: npm ci
        
      - name: Build Project
        run: npm run build
        
      - name: Create Artifact for Azure
        run: |
          TIMESTAMP=$(date +%Y%m%d_%H%M%S)
          COMMIT_SHA=$(echo ${{ github.sha }} | cut -c1-8)
          ARTIFACT_FILENAME="${{ env.ARTIFACT_NAME }}_${TIMESTAMP}_${COMMIT_SHA}.tar.gz"
          tar -czf "${ARTIFACT_FILENAME}" ${{ env.BUILD_DIR }}/
          echo "ARTIFACT_FILENAME=${ARTIFACT_FILENAME}" >> $GITHUB_ENV
          
      - name: Login to Azure
        uses: azure/login@v1
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}
          
      - name: Upload to Azure Blob Storage
        uses: azure/CLI@v1
        with:
          inlineScript: |
            echo "Uploading artifact to Azure Blob Storage..."
            az storage blob upload \
              --account-name ${{ secrets.AZURE_STORAGE_ACCOUNT }} \
              --container-name ${{ secrets.AZURE_CONTAINER_NAME }} \
              --name "artifacts/${{ env.ARTIFACT_FILENAME }}" \
              --file "${{ env.ARTIFACT_FILENAME }}" \
              --auth-mode login
            echo "Upload to Azure Blob Storage completed!"
            
      - name: Verify Azure Upload
        uses: azure/CLI@v1
        with:
          inlineScript: |
            echo "Verifying Azure upload..."
            az storage blob show \
              --account-name ${{ secrets.AZURE_STORAGE_ACCOUNT }} \
              --container-name ${{ secrets.AZURE_CONTAINER_NAME }} \
              --name "artifacts/${{ env.ARTIFACT_FILENAME }}" \
              --auth-mode login
            echo "Azure Verification successful!"

  notify-completion:
    runs-on: ubuntu-latest
    needs: [build-and-upload, upload-to-aws-s3]
    if: always()
    
    steps:
      - name: Notify Build Status
        run: |
          if [ "${{ needs.build-and-upload.result }}" == "success" ]; then
            echo "✅ Build completed successfully!"
            echo "📦 Artifact has been uploaded to storage"
            echo "🔗 Commit: ${{ github.sha }}"
            echo "👤 Author: ${{ github.actor }}"
            echo "🌿 Branch: ${{ github.ref_name }}"
          else
            echo "❌ Build failed!"
            echo "Please check the logs for more details"
            exit 1
          fi
```
