
```csharp
using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;
using System.Collections.Generic;

namespace DeleteUserButtonExample
{
    // User roles enum
    public enum UserRole
    {
        Admin,
        User,
        Moderator,
        Guest
    }

    // User model
    public class User
    {
        public int Id { get; set; }
        public string Name { get; set; }
        public string Email { get; set; }
        public UserRole Role { get; set; }
        public bool IsActive { get; set; }
    }

    // Current session/logged-in user context
    public class UserSession
    {
        public User CurrentUser { get; set; }
        public bool IsAuthenticated { get; set; }
    }

    // Delete User Button Manager
    public class DeleteUserButtonManager
    {
        private readonly AnthropicClient _anthropicClient;
        private readonly UserSession _userSession;
        private List<User> _users;

        public DeleteUserButtonManager(UserSession userSession)
        {
            _anthropicClient = new AnthropicClient();
            _userSession = userSession;
            
            // Initialize sample users
            _users = new List<User>
            {
                new User { Id = 1, Name = "Alice Admin", Email = "alice@example.com", Role = UserRole.Admin, IsActive = true },
                new User { Id = 2, Name = "Bob User", Email = "bob@example.com", Role = UserRole.User, IsActive = true },
                new User { Id = 3, Name = "Charlie Mod", Email = "charlie@example.com", Role = UserRole.Moderator, IsActive = true },
                new User { Id = 4, Name = "Diana Guest", Email = "diana@example.com", Role = UserRole.Guest, IsActive = true }
            };
        }

        // Main method to check if Delete User button should be shown and handle deletion
        public async Task<string> HandleDeleteUserButton(int targetUserId)
        {
            // Check if user is authenticated
            if (!_userSession.IsAuthenticated || _userSession.CurrentUser == null)
            {
                return "Error: User is not authenticated. Please log in first.";
            }

            // Check if current user is admin
            bool isAdmin = _userSession.CurrentUser.Role == UserRole.Admin;
            
            // Find target user
            User targetUser = _users.Find(u => u.Id == targetUserId);
            
            if (targetUser == null)
            {
                return $"Error: User with ID {targetUserId} not found.";
            }

            // Use Claude AI to analyze the deletion request and provide guidance
            string aiAnalysis = await AnalyzeDeleteRequestWithClaude(
                _userSession.CurrentUser, 
                targetUser, 
                isAdmin
            );

            if (!isAdmin)
            {
                return $"Access Denied: Only administrators can delete users.\n\nAI Analysis:\n{aiAnalysis}";
            }

            // Prevent admin from deleting themselves
            if (_userSession.CurrentUser.Id == targetUserId)
            {
                return "Error: Administrators cannot delete their own account.\n\nAI Analysis:\n" + aiAnalysis;
            }

            // Perform deletion
            bool deleted = DeleteUser(targetUserId);
            
            if (deleted)
            {
                return $"Success: User '{targetUser.Name}' (ID: {targetUserId}) has been successfully deleted.\n\nAI Analysis:\n{aiAnalysis}";
            }
            else
            {
                return $"Error: Failed to delete user '{targetUser.Name}'.\n\nAI Analysis:\n{aiAnalysis}";
            }
        }

        // Method to check if Delete User button should be visible
        public bool ShouldShowDeleteButton()
        {
            if (!_userSession.IsAuthenticated || _userSession.CurrentUser == null)
            {
                return false;
            }

            return _userSession.CurrentUser.Role == UserRole.Admin;
        }

        // Actual deletion logic
        private bool DeleteUser(int userId)
        {
            User userToDelete = _users.Find(u => u.Id == userId);
            if (userToDelete != null)
            {
                _users.Remove(userToDelete);
                Console.WriteLine($"[System] User ID {userId} removed from database.");
                return true;
            }
            return false;
        }

        // Use Claude AI to analyze the deletion request
        private async Task<string> AnalyzeDeleteRequestWithClaude(User currentUser, User targetUser, bool isAdmin)
        {
            string prompt = $@"Analyze this user deletion request and provide a brief security assessment:

Current User:
- Name: {currentUser.Name}
- Role: {currentUser.Role}
- Is Admin: {isAdmin}

Target User to Delete:
- Name: {targetUser.Name}
- Email: {targetUser.Email}
- Role: {targetUser.Role}
- Is Active: {targetUser.IsActive}

Please provide:
1. Whether this deletion should be allowed (based on admin role)
2. Any security considerations
3. Recommended best practices for user deletion

Keep the response concise (3-4 sentences).";

            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = new List<ContentBase>
                    {
                        new TextContent { Text = prompt }
                    }
                }
            };

            var parameters = new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3Haiku,
                MaxTokens = 300,
                Stream = false
            };

            var response = await _anthropicClient.Messages.GetClaudeMessageAsync(parameters);
            
            if (response?.Content != null && response.Content.Count > 0)
            {
                return response.Content[0].ToString();
            }
            
            return "AI analysis unavailable.";
        }

        // Get list of all users (admin only)
        public string GetUserList()
        {
            if (!ShouldShowDeleteButton())
            {
                return "Access Denied: Only administrators can view the user list.";
            }

            string userList = "Current Users:\n";
            foreach (var user in _users)
            {
                userList += $"  ID: {user.Id}, Name: {user.Name}, Role: {user.Role}, Active: {user.IsActive}\n";
            }
            return userList;
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== Delete User Button - Admin Only Feature Demo ===\n");

            // Test Case 1: Admin user trying to delete another user
            Console.WriteLine("--- Test Case 1: Admin User ---");
            var adminSession = new UserSession
            {
                IsAuthenticated = true,
                CurrentUser = new User
                {
                    Id = 1,
                    Name = "Alice Admin",
                    Email = "alice@example.com",
                    Role = UserRole.Admin,
                    IsActive = true
                }
            };

            var adminManager = new DeleteUserButtonManager(adminSession);
            
            Console.WriteLine($"Should show Delete button for Admin: {adminManager.ShouldShowDeleteButton()}");
            Console.WriteLine("\nUser List (Admin View):");
            Console.WriteLine(adminManager.GetUserList());
            
            Console.WriteLine("Attempting to delete User ID 2 (Bob User)...");
            string adminResult = await adminManager.HandleDeleteUserButton(2);
            Console.WriteLine($"Result: {adminResult}\n");

            // Test Case 2: Regular user trying to delete a user
            Console.WriteLine("--- Test Case 2: Regular User ---");
            var regularUserSession = new UserSession
            {
                IsAuthenticated = true,
                CurrentUser = new User
                {
                    Id = 3,
                    Name = "Charlie Mod",
                    Email = "charlie@example.com",
                    Role = UserRole.Moderator,
                    IsActive = true
                }
            };

            var regularManager = new DeleteUserButtonManager(regularUserSession);
            
            Console.WriteLine($"Should show Delete button for Moderator: {regularManager.ShouldShowDeleteButton()}");
            
            Console.WriteLine("Attempting to delete User ID 4 (Diana Guest)...");
            string regularResult = await regularManager.HandleDeleteUserButton(4);
            Console.WriteLine($"Result: {regularResult}\n");

            // Test Case 3: Admin trying to delete themselves
            Console.WriteLine("--- Test Case 3: Admin Self-Delete Attempt ---");
            Console.WriteLine("Admin attempting to delete their own account (ID 1)...");
            string selfDeleteResult = await adminManager.HandleDeleteUserButton(1);
            Console.WriteLine($"Result: {selfDeleteResult}\n");

            // Test Case 4: Unauthenticated user
            Console.WriteLine("--- Test Case 4: Unauthenticated User ---");
            var unauthSession = new UserSession
            {
                IsAuthenticated = false,
                CurrentUser = null
            };

            var unauthManager = new DeleteUserButtonManager(unauthSession);
            Console.WriteLine($"Should show Delete button for unauthenticated user: {unauthManager.ShouldShowDeleteButton()}");
            
            string unauthResult = await unauthManager.HandleDeleteUserButton(3);
            Console.WriteLine($"Result: {unauthResult}\n");

            Console.WriteLine("=== Demo Complete ===");
        }
    }
}
```
