class DatabaseConnectionError(Exception):

        original_error: Optional[Exception] = None
    ):
        self.message = message
        self.detail = detail
        self.original_error = original_error
        super().__init__(self.message)

def build_error_response(
    error_code: str,
    message: str,
    detail: Optional[str] = None,
    extra_info: Optional[Dict[str, Any]] = None,
    request_id: Optional[str] = None
) -> Dict[str, Any]:
    """Standard error response format builder"""
    response = {
        "success": False,
        "error": {
            "code": error_code,
            "message": message,
            "timestamp": datetime.utcnow().isoformat() + "Z",
        }
    }
    
    if detail:
        response["error"]["detail"] = detail
    
    if request_id:
        response["error"]["request_id"] = request_id
    
    if extra_info:
        response["error"]["info"] = extra_info
    
    return response


def get_request_id(request: Request) -> Optional[str]:
    """Request से unique ID extract करो"""
    return request.headers.get("X-Request-ID") or request.headers.get("X-Correlation-ID")

@app.exception_handler(DatabaseConnectionError)
async def database_connection_error_handler(
    request: Request,
    exc: DatabaseConnectionError
) -> JSONResponse:
    """
    Database connection failure handler
    Status: 503 Service Unavailable
    """
    request_id = get_request_id(request)
    
    # Error log करो
    logger.error(
        f"Database Connection Error | "
        f"Path: {request.url.path} | "
        f"Request-ID: {request_id} | "
        f"Message: {exc.message} | "
        f"Detail: {exc.detail} | "
        f"Original: {exc.original_error}"
    )
    
    extra_info = {}
    if exc.detail:
        extra_info["technical_detail"] = exc.detail
    
    response_body = build_error_response(
        error_code="DB_CONNECTION_FAILED",
        message=exc.message,
        detail="Database server से connection establish नहीं हो सका। कृपया बाद में try करें।",
        extra_info=extra_info if extra_info else None,
        request_id=request_id
    )
    
    return JSONResponse(
        status_code=503,
        content=response_body,
        headers={
            "Retry-After": "30",
            "X-Error-Code": "DB_CONNECTION_FAILED"
        }
    )

        extra_info=extra_info if extra_info else None,
        request_id=request_id
    )
    
    return JSONResponse(
        status_code=504,
        content=response_body,
        headers={
            "Retry-After": "10",
            "X-Error-Code": "DB_TIMEOUT"
        }
    )

# Validation errors handler
@app.exception_handler(RequestValidationError)
async def validation_error_handler(
    request: Request,
    exc: RequestValidationError
) -> JSONResponse:
    """Request validation errors"""
    request_id = get_request_id(request)
    
    errors = []
    for error in exc.errors():
        errors.append({
            "field": " -> ".join(str(loc) for loc in error["loc"]),
            "message": error["msg"],
            "type": error["type"]
        })
    
    response_body = build_error_response(
        error_code="VALIDATION_ERROR",
        message="Request data invalid है",
        detail="कृपया input data check करें",
        extra_info={"validation_errors": errors},
        request_id=request_id
    )
    
    return JSONResponse(status_code=422, content=response_body)

    async def get_user(self, user_id: int) -> Dict[str, Any]:
        """User fetch करो"""
        if not self.is_connected:
            raise DatabaseConnectionError(
                message="Database server से connection नहीं हो सका",
                detail="Connection refused on port 5432",
                original_error=ConnectionRefusedError("Connection refused")
            )
        
        # Simulate DB operation
        await asyncio.sleep(0.01)
        
        if user_id == 999:
            raise DatabaseQueryError(
                message="User query failed",
                query=f"SELECT * FROM users WHERE id = {user_id}",
                detail="Invalid user_id format",
                original_error=ValueError("Invalid ID")
            )
        
        return {
            "id": user_id,
            "name": f"User_{user_id}",
            "email": f"user{user_id}@example.com",
            "created_at": datetime.utcnow().isoformat()
        }
    
    async def get_users_with_timeout(self, timeout: float = 0.001) -> list:
        """Timeout simulate करो"""
        try:
            await asyncio.wait_for(
                asyncio.sleep(5),  # Long operation
                timeout=timeout
            )
        except asyncio.TimeoutError:
            raise DatabaseTimeoutError(
                message="Database query timeout हो गई",
                timeout_seconds=timeout,
                detail="Query execution time limit exceed हो गई",
                original_error=asyncio.TimeoutError()
            )
        return []
    
    async def create_user_transaction(self, user_data: Dict) -> Dict[str, Any]:
        """Transaction simulate करो"""
        if user_data.get("force_transaction_error"):
            raise DatabaseTransactionError(
                message="User creation transaction fail हो गई",
                detail="Constraint violation: duplicate email",
                original_error=Exception("UNIQUE constraint failed")
            )
        
        return {
            "id": 101,
            "name": user_data.get("name"),
            "email": user_data.get("email"),
            "created_at": datetime.utcnow().isoformat()
        }
    
    async def check_pool(self) -> Dict[str, Any]:
        """Connection pool check करो"""
        if self.active_connections >= self.pool_size:
            raise DatabasePoolExhaustedError(
                message="Database connection pool full है",
                pool_size=self.pool_size,
                detail=f"All {self.pool_size} connections in use"
            )
        
        return {
            "pool_size": self.pool_size,
            "active_connections": self.active_connections,
            "available": self.pool_size - self.active_connections
        }

@app.get("/db/health")
async def database_health_check():
    """Database health check"""
    start_time = time.time()
    
    try:
        # Simple connectivity test
        await asyncio.wait_for(
            db_service.get_user(1),
            timeout=5.0
        )
        response_time = (time.time() - start_time) * 1000
        
        return {
            "success": True,
            "status": "healthy",
            "database": {
                "connected": db_service.is_connected,
                "response_time_ms": round(response_time, 2),
                "pool_size": db_service.pool_size,
                "active_connections": db_service.active_connections

            "timestamp": datetime.utcnow().isoformat() + "Z"
        }
    
    except DatabaseConnectionError as e:
        return JSONResponse(
            status_code=503,
            content={
                "success": False,
                "status": "unhealthy",
                "database": {
                    "connected": False,
                    "error": e.message
                },
                "timestamp": datetime.utcnow().isoformat() + "Z"
            }
        )
    
    except DatabaseTimeoutError as e:
        return JSONResponse(
            status_code=503,
            content={
                "success": False,
                "status": "degraded",
                "database": {
                    "connected": True,
                    "error": e.message,
                    "timeout_seconds": e.timeout_seconds
                },
                "timestamp": datetime.utcnow().isoformat() + "Z"
            }
        )

@app.middleware("http")
async def request_logging_middleware(request: Request, call_next):
    """सभी requests को log करो"""
    start_time = time.time()
    request_id = get_request_id(request) or f"req_{int(time.time() * 1000)}"
    
    logger.info(
        f"Request Start | "
        f"ID: {request_id} | "
        f"Method: {request.method} | "
        f"Path: {request.url.path}"
    )
    
    response = await call_next(request)
    
    process_time = (time.time() - start_time) * 1000
    
    logger.info(
        f"Request End | "
        f"ID: {request_id} | "
        f"Status: {response.status_code} | "
        f"Time: {process_time:.2f}ms"
    )
    
    response.headers["X-Process-Time"] = f"{process_time:.2f}ms"
    response.headers["X-Request-ID"] = request_id
    
    return response
