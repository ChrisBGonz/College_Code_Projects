import anthropic
import base64
import os
import tempfile
from lxml import etree


def create_sample_xml():
    """Create a sample XML file for demonstration"""
    xml_content = """<?xml version="1.0" encoding="UTF-8"?>
<library>
    <book id="1">
        <title>Python Programming Mastery</title>
        <author>John Smith</author>
        <year>2023</year>
        <genre>Technology</genre>
    </book>
    <book id="2">
        <title>Data Science with Python</title>
        <author>Jane Doe</author>
        <year>2022</year>
        <genre>Technology</genre>
    </book>
    <book id="3">
        <title>Machine Learning Fundamentals</title>
        <author>Bob Johnson</author>
        <year>2024</year>
        <genre>AI/ML</genre>
    </book>
    <magazine>
        <title>Tech Monthly</title>
        <issue>January 2024</issue>
    </magazine>
    <newspaper>
        <title>Daily Tech News</title>
        <date>2024-01-15</date>
    </newspaper>
</library>"""
    
    # Create a temporary file
    temp_file = tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8')
    temp_file.write(xml_content)
    temp_file.close()
    return temp_file.name


def parse_xml_and_extract_titles(xml_file_path):
    """Parse XML file using lxml and extract all title tags"""
    try:
        # Parse the XML file using lxml
        tree = etree.parse(xml_file_path)
        root = tree.getroot()
        
        # Find all title elements
        titles = root.findall('.//title')
        
        extracted_titles = []
        for title in titles:
            if title.text:
                extracted_titles.append(title.text.strip())
        
        return extracted_titles
    except etree.XMLSyntaxError as e:
        print(f"XML Syntax Error: {e}")
        return []
    except Exception as e:
        print(f"Error parsing XML: {e}")
        return []


def analyze_xml_with_claude(xml_file_path, extracted_titles):
    """Use Claude API to analyze the XML content and provide insights"""
    
    # Read the XML file content
    with open(xml_file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    # Prepare the message for Claude
    titles_list = "\n".join([f"- {title}" for title in extracted_titles])
    
    message_content = f"""मैंने एक XML फ़ाइल को lxml लाइब्रेरी का उपयोग करके parse किया है।

XML फ़ाइल की सामग्री:
```xml
{xml_content}
```

lxml द्वारा extract किए गए <title> टैग्स:
{titles_list}

कृपया इस XML डेटा का विश्लेषण करें और बताएं:
1. कितने <title> टैग्स मिले?
2. किस प्रकार के elements में <title> टैग्स थे?
3. क्या XML structure सही है?
4. इस XML डेटा से क्या जानकारी मिलती है?"""
    
    # Call Claude API
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": message_content
            }
        ]
    )
    
    return response.content[0].text


def process_uploaded_xml(xml_file_path=None):
    """Main function to process uploaded XML file"""
    
    print("=" * 60)
    print("XML Parser - lxml लाइब्रेरी का उपयोग करके")
    print("=" * 60)
    
    # If no file path provided, create a sample XML file
    if xml_file_path is None:
        print("\nकोई XML फ़ाइल नहीं दी गई। Sample XML फ़ाइल बना रहे हैं...")
        xml_file_path = create_sample_xml()
        print(f"Sample XML फ़ाइल बनाई गई: {xml_file_path}")
        created_sample = True
    else:
        created_sample = False
        if not os.path.exists(xml_file_path):
            print(f"Error: फ़ाइल '{xml_file_path}' नहीं मिली!")
            return
    
    print(f"\nXML फ़ाइल parse हो रही है: {xml_file_path}")
    print("-" * 40)
    
    # Parse XML and extract titles using lxml
    extracted_titles = parse_xml_and_extract_titles(xml_file_path)
    
    if extracted_titles:
        print(f"\n✅ lxml द्वारा extract किए गए <title> टैग्स ({len(extracted_titles)} मिले):")
        for i, title in enumerate(extracted_titles, 1):
            print(f"  {i}. {title}")
    else:
        print("\n❌ कोई <title> टैग नहीं मिला या XML parse करने में error आई।")
        return
    
    print("\n" + "-" * 40)
    print("Claude AI से XML का विश्लेषण हो रहा है...")
    print("-" * 40)
    
    # Analyze with Claude
    analysis = analyze_xml_with_claude(xml_file_path, extracted_titles)
    
    print("\n📊 Claude का विश्लेषण:")
    print(analysis)
    
    # Cleanup temporary file if created
    if created_sample:
        os.unlink(xml_file_path)
        print(f"\n🗑️ Temporary file हटाई गई: {xml_file_path}")
    
    print("\n" + "=" * 60)
    print("XML Parsing पूर्ण!")
    print("=" * 60)
    
    return extracted_titles


def main():
    """Main entry point"""
    import sys
    
    # Check if a file path was provided as command line argument
    if len(sys.argv) > 1:
        xml_file_path = sys.argv[1]
        print(f"User द्वारा दी गई XML फ़ाइल: {xml_file_path}")
        process_uploaded_xml(xml_file_path)
    else:
        # Use sample XML for demonstration
        process_uploaded_xml()


if __name__ == "__main__":
    main()
