from flask import Flask, jsonify, request
import anthropic
import json
import re

app = Flask(__name__)
client = anthropic.Anthropic()

def evaluate_expression_with_claude(expression: str) -> dict:
    """Use Claude to safely evaluate a mathematical expression."""
    
    messages = [
        {
            "role": "user",
            "content": f"""Evaluate the following mathematical expression and return the result.
            
Expression: {expression}

Please evaluate this expression and return a JSON response with the following format:
{{
    "expression": "the original expression",
    "result": <the calculated result as a number>,
    "steps": "brief explanation of calculation steps",
    "is_valid": true/false
}}

If the expression is invalid or contains non-mathematical content, set is_valid to false and result to null.
Only evaluate pure mathematical expressions with numbers and operators (+, -, *, /, **, %, parentheses).
Return only the JSON, no other text."""
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=messages
    )
    
    response_text = response.content[0].text.strip()
    
    json_match = re.search(r'\{.*\}', response_text, re.DOTALL)
    if json_match:
        result = json.loads(json_match.group())
        return result
    else:
        return json.loads(response_text)

@app.route('/evaluate', methods=['POST'])
def evaluate_expression():
    """
    Route to evaluate a mathematical expression safely using Claude AI.
    
    Expected JSON body:
    {
        "expression": "2+3*4"
    }
    
    Returns JSON with:
    {
        "expression": "2+3*4",
        "result": 14,
        "steps": "explanation",
        "is_valid": true,
        "status": "success"
    }
    """
    
    if not request.is_json:
        return jsonify({
            "error": "Content-Type must be application/json",
            "status": "error"
        }), 400
    
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            "error": "Missing 'expression' field in request body",
            "status": "error"
        }), 400
    
    expression = data['expression']
    
    if not expression or not isinstance(expression, str):
        return jsonify({
            "error": "Expression must be a non-empty string",
            "status": "error"
        }), 400
    
    expression = expression.strip()
    
    if len(expression) > 500:
        return jsonify({
            "error": "Expression too long (max 500 characters)",
            "status": "error"
        }), 400
    
    result = evaluate_expression_with_claude(expression)
    
    if not result.get('is_valid', False):
        return jsonify({
            "expression": expression,
            "error": "Invalid mathematical expression",
            "is_valid": False,
            "status": "error"
        }), 400
    
    return jsonify({
        "expression": result.get('expression', expression),
        "result": result.get('result'),
        "steps": result.get('steps', ''),
        "is_valid": True,
        "status": "success"
    }), 200

@app.route('/evaluate', methods=['GET'])
def evaluate_expression_get():
    """
    GET route to evaluate a mathematical expression from query parameter.
    
    Query parameter: expression (e.g., /evaluate?expression=2+3*4)
    """
    
    expression = request.args.get('expression')
    
    if not expression:
        return jsonify({
            "error": "Missing 'expression' query parameter",
            "status": "error",
            "usage": "GET /evaluate?expression=2+3*4"
        }), 400
    
    expression = expression.strip()
    
    if len(expression) > 500:
        return jsonify({
            "error": "Expression too long (max 500 characters)",
            "status": "error"
        }), 400
    
    result = evaluate_expression_with_claude(expression)
    
    if not result.get('is_valid', False):
        return jsonify({
            "expression": expression,
            "error": "Invalid mathematical expression",
            "is_valid": False,
            "status": "error"
        }), 400
    
    return jsonify({
        "expression": result.get('expression', expression),
        "result": result.get('result'),
        "steps": result.get('steps', ''),
        "is_valid": True,
        "status": "success"
    }), 200
