import anthropic
import json
import re
import urllib.parse
from typing import Any

client = anthropic.Anthropic()

PAYMENT_GATEWAY_TOOLS = [
    {
        "name": "initialize_payment",
        "description": "Initialize a payment with the payment gateway. Returns a payment session with redirect URL.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "Payment amount in cents"
                },
                "currency": {
                    "type": "string",
                    "description": "Currency code (e.g., USD, EUR, INR)"
                },
                "success_url": {
                    "type": "string",
                    "description": "URL to redirect user after successful payment"
                },
                "cancel_url": {
                    "type": "string",
                    "description": "URL to redirect user if payment is cancelled"
                },
                "customer_email": {
                    "type": "string",
                    "description": "Customer's email address"
                },
                "description": {
                    "type": "string",
                    "description": "Payment description"
                }
            },
            "required": ["amount", "currency", "success_url", "cancel_url"]
        }
    },
    {
        "name": "redirect_user",
        "description": "Redirect user to a specific URL after payment initialization",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "URL to redirect the user to"
                },
                "reason": {
                    "type": "string",
                    "description": "Reason for redirection (success, cancel, error)"
                }
            },
            "required": ["url", "reason"]
        }
    },
    {
        "name": "verify_payment",
        "description": "Verify payment status using session ID",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "Payment session ID to verify"
                }
            },
            "required": ["session_id"]
        }
    }
]

def initialize_payment(amount: int, currency: str, success_url: str, 
                       cancel_url: str, customer_email: str = None, 
                       description: str = None) -> dict[str, Any]:
    """Simulate payment gateway initialization"""
    session_id = f"sess_{hash(f'{amount}{currency}{success_url}') % 1000000:06d}"
    
    payment_url = f"https://payment-gateway.example.com/pay/{session_id}"
    
    return {
        "session_id": session_id,
        "payment_url": payment_url,
        "success_url": success_url,
        "cancel_url": cancel_url,
        "amount": amount,
        "currency": currency,
        "status": "pending",
        "expires_at": "2024-12-31T23:59:59Z"
    }

def redirect_user(url: str, reason: str) -> dict[str, Any]:
    """Simulate user redirection"""
    print(f"\n{'='*50}")
    print(f"🔄 REDIRECTING USER")
    print(f"{'='*50}")
    print(f"Reason: {reason}")
    print(f"Redirect URL: {url}")
    print(f"{'='*50}\n")
    
    parsed_url = urllib.parse.urlparse(url)
    query_params = urllib.parse.parse_qs(parsed_url.query)
    
    return {
        "redirected": True,
        "url": url,
        "reason": reason,
        "parsed_params": {k: v[0] if len(v) == 1 else v for k, v in query_params.items()}
    }

def verify_payment(session_id: str) -> dict[str, Any]:
    """Simulate payment verification"""
    return {
        "session_id": session_id,
        "status": "completed",
        "amount_received": 1000,
        "currency": "USD",
        "payment_method": "card",
        "timestamp": "2024-01-15T10:30:00Z"
    }

def process_tool_call(tool_name: str, tool_input: dict) -> Any:
    """Process tool calls and return results"""
    if tool_name == "initialize_payment":
        return initialize_payment(**tool_input)
    elif tool_name == "redirect_user":
        return redirect_user(**tool_input)
    elif tool_name == "verify_payment":
        return verify_payment(**tool_input)
    else:
        return {"error": f"Unknown tool: {tool_name}"}

def run_payment_agent(payment_request: dict) -> str:
    """Run the payment agent with agentic loop"""
    
    system_prompt = """You are a payment processing agent that integrates with a 3rd-party payment gateway.

Your responsibilities:
1. Initialize payment sessions using the payment gateway
2. After successful initialization, redirect the user to the success_url provided in the initial request
3. Handle errors gracefully

IMPORTANT RULE: After initializing a payment, you MUST redirect the user to the success_url that was provided in the original payment request (not the payment gateway URL). This is the URL where the user should be sent after payment initialization.

Always use the tools provided to complete payment operations."""

    user_message = f"""Process this payment request and redirect the user appropriately:

Payment Request:
{json.dumps(payment_request, indent=2)}

Steps to follow:
1. Initialize the payment with the gateway using the provided details
2. After successful initialization, redirect the user to the success_url from the original request
3. Confirm the redirection was successful"""

    messages = [{"role": "user", "content": user_message}]
    
    print(f"\n{'='*60}")
    print("🚀 STARTING PAYMENT PROCESSING AGENT")
    print(f"{'='*60}")
    print(f"Payment Request: {json.dumps(payment_request, indent=2)}")
    print(f"{'='*60}\n")
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system=system_prompt,
            tools=PAYMENT_GATEWAY_TOOLS,
            messages=messages
        )
        
        print(f"📡 Agent Response (stop_reason: {response.stop_reason})")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    break
            
            print(f"\n✅ AGENT COMPLETED")
            print(f"Final Response: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            messages.append({"role": "assistant", "content": response.content})
            
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\n🔧 Tool Called: {block.name}")
                    print(f"   Input: {json.dumps(block.input, indent=2)}")
                    
                    result = process_tool_call(block.name, block.input)
                    
                    print(f"   Result: {json.dumps(result, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(result)
                    })
            
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop_reason: {response.stop_reason}")
            break
    
    return "Payment processing completed"
