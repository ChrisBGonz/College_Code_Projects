# Flask example - User redirect
from flask import Flask, redirect, url_for, request, render_template_string

app = Flask(__name__)

# Simple HTML template
HTML_TEMPLATE = """
<!DOCTYPE html>
<html>
<head><title>Redirect Demo</title></head>
<body>
    <h1>Redirect Examples</h1>
    <ul>
        <li><a href="/old-page">Old Page (301 Permanent Redirect)</a></li>
        <li><a href="/temp-page">Temp Page (302 Temporary Redirect)</a></li>
        <li><a href="/login-required">Login Required (303 See Other)</a></li>
        <li><a href="/external">External Redirect</a></li>
        <li><a href="/conditional?user=admin">Conditional Redirect (admin)</a></li>
        <li><a href="/conditional?user=guest">Conditional Redirect (guest)</a></li>
    </ul>
</body>
</html>
"""

@app.route('/')
def home():
    return render_template_string(HTML_TEMPLATE)

# 301 - Permanent Redirect
# Jab URL permanently change ho jaye
@app.route('/old-page')
def old_page():
    return redirect(url_for('new_page'), code=301)

@app.route('/new-page')
def new_page():
    return "<h1>Yeh naya page hai! (301 se aaye)</h1><a href='/'>Home</a>"

# 302 - Temporary Redirect (default)
# Jab temporarily kisi aur page pe bhejna ho
@app.route('/temp-page')
def temp_page():
    return redirect(url_for('home'), code=302)

# 303 - See Other
# Form submit ke baad redirect (POST -> GET)
@app.route('/login-required')
def login_required_page():
    # User logged in nahi hai, login page pe bhejo
    return redirect(url_for('login'), code=303)

@app.route('/login')
def login():
    return "<h1>Login Page</h1><p>Aapko login karna hoga</p><a href='/'>Home</a>"

# External URL pe redirect
@app.route('/external')
def external_redirect():
    return redirect("https://www.google.com", code=302)

# Conditional redirect - user ke basis pe
@app.route('/conditional')
def conditional_redirect():
    user = request.args.get('user', 'guest')
    
    if user == 'admin':
        return redirect(url_for('admin_dashboard'))
    else:
        return redirect(url_for('user_dashboard'))

@app.route('/admin')
def admin_dashboard():
    return "<h1>Admin Dashboard</h1><a href='/'>Home</a>"

@app.route('/user')
def user_dashboard():
    return "<h1>User Dashboard</h1><a href='/'>Home</a>"

# Form submit example - PRG Pattern (Post/Redirect/Get)
FORM_TEMPLATE = """
<!DOCTYPE html>
<html>
<body>
    <h1>Form Submit Example</h1>
    <form method="POST" action="/submit">
        <input type="text" name="name" placeholder="Apna naam likho">
        <button type="submit">Submit</button>
    </form>
</body>
</html>
"""

@app.route('/form')
def show_form():
    return render_template_string(FORM_TEMPLATE)

@app.route('/submit', methods=['POST'])
def submit_form():
    name = request.form.get('name', 'Unknown')
    # PRG Pattern: POST ke baad GET pe redirect
    # Isse page refresh pe duplicate submission nahi hogi
    return redirect(url_for('success', name=name), code=303)

@app.route('/success')
def success():
    name = request.args.get('name', 'Unknown')
    return f"<h1>Success! Naam: {name}</h1><a href='/form'>Wapas jao</a>"
