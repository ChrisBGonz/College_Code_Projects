import anthropic.SDK;
import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.Message;
import com.anthropic.models.messages.MessageCreateParams;
import com.anthropic.models.messages.Model;
import com.anthropic.models.messages.TextBlock;

public class XMLParserWithClaude {

    private final AnthropicClient client;

    public XMLParserWithClaude() {
        this.client = AnthropicOkHttpClient.fromEnv();
    }

    public String analyzeXMLSecurity(String xmlString) {
        String prompt = String.format(
            "मुझे एक Java function लिखने में मदद करें जो DocumentBuilderFactory का उपयोग करके " +
            "untrusted source से आए XML string को safely parse करे। " +
            "निम्नलिखित XML string को analyze करें और बताएं कि इसे safely parse करने के लिए " +
            "क्या security measures लेने चाहिए:\n\n%s\n\n" +
            "कृपया:\n" +
            "1. XXE (XML External Entity) attacks से बचाव के तरीके बताएं\n" +
            "2. Secure DocumentBuilderFactory configuration दिखाएं\n" +
            "3. Complete Java code example प्रदान करें\n" +
            "4. Hindi में explanation दें",
            xmlString
        );

        MessageCreateParams params = MessageCreateParams.builder()
            .model(Model.CLAUDE_OPUS_4_5)
            .maxTokens(2048)
            .addUserMessage(prompt)
            .build();

        Message message = client.messages().create(params);

        StringBuilder response = new StringBuilder();
        message.content().forEach(block -> {
            if (block instanceof TextBlock) {
                response.append(((TextBlock) block).text());
            }
        });

        return response.toString();
    }

    public String generateSecureXMLParser() {
        String prompt = 
            "DocumentBuilderFactory का उपयोग करके एक complete, secure Java function लिखें " +
            "जो untrusted source से आए XML string को parse करे। " +
            "निम्नलिखित requirements को पूरा करें:\n\n" +
            "1. XXE (XML External Entity) injection attacks को prevent करें\n" +
            "2. DOCTYPE declarations को disable करें\n" +
            "3. External entities और external DTDs को block करें\n" +
            "4. Proper error handling implement करें\n" +
            "5. Input validation करें\n" +
            "6. Complete working Java code provide करें\n\n" +
            "Code के साथ Hindi में detailed explanation भी दें कि प्रत्येक security measure क्यों जरूरी है।";

        MessageCreateParams params = MessageCreateParams.builder()
            .model(Model.CLAUDE_OPUS_4_5)
            .maxTokens(3000)
            .addUserMessage(prompt)
            .build();

        Message message = client.messages().create(params);

        StringBuilder response = new StringBuilder();
        message.content().forEach(block -> {
            if (block instanceof TextBlock) {
                response.append(((TextBlock) block).text());
            }
        });

        return response.toString();
    }

    public String reviewXMLCode(String javaCode) {
        String prompt = String.format(
            "निम्नलिखित Java code को security perspective से review करें जो XML parsing करता है:\n\n" +
            "```java\n%s\n```\n\n" +
            "कृपया बताएं:\n" +
            "1. क्या यह code XXE attacks से safe है?\n" +
            "2. कौन से security vulnerabilities हैं?\n" +
            "3. इसे कैसे improve किया जा सकता है?\n" +
            "4. Secure version का code provide करें\n\n" +
            "Hindi में detailed explanation दें।",
            javaCode
        );

        MessageCreateParams params = MessageCreateParams.builder()
            .model(Model.CLAUDE_OPUS_4_5)
            .maxTokens(2500)
            .addUserMessage(prompt)
            .build();

        Message message = client.messages().create(params);

        StringBuilder response = new StringBuilder();
        message.content().forEach(block -> {
            if (block instanceof TextBlock) {
                response.append(((TextBlock) block).text());
            }
        });

        return response.toString();
    }

    public static void main(String[] args) {
        XMLParserWithClaude analyzer = new XMLParserWithClaude();

        System.out.println("=== Secure XML Parser Generation ===\n");
        String secureParser = analyzer.generateSecureXMLParser();
        System.out.println(secureParser);

        System.out.println("\n" + "=".repeat(60) + "\n");

        String sampleXML = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n" +
                          "<!DOCTYPE foo [<!ENTITY xxe SYSTEM \"file:///etc/passwd\">]>\n" +
                          "<root>\n" +
                          "  <data>&xxe;</data>\n" +
                          "</root>";

        System.out.println("=== XML Security Analysis ===\n");
        System.out.println("Analyzing potentially malicious XML:\n" + sampleXML + "\n");
        String analysis = analyzer.analyzeXMLSecurity(sampleXML);
        System.out.println(analysis);

        System.out.println("\n" + "=".repeat(60) + "\n");

        String vulnerableCode = 
            "public Document parseXML(String xmlString) throws Exception {\n" +
            "    DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();\n" +
            "    DocumentBuilder builder = factory.newDocumentBuilder();\n" +
            "    InputSource is = new InputSource(new StringReader(xmlString));\n" +
            "    return builder.parse(is);\n" +
            "}";

        System.out.println("=== Code Security Review ===\n");
        System.out.println("Reviewing vulnerable code:\n" + vulnerableCode + "\n");
        String review = analyzer.reviewXMLCode(vulnerableCode);
        System.out.println(review);
    }
}
