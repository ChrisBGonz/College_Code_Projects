using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

class StayLoggedInFeature
{
    private static readonly AnthropicClient client = new AnthropicClient();
    
    // Simulated cookie storage (in real app, this would be HTTP cookies)
    private static Dictionary<string, CookieData> cookieStore = new Dictionary<string, CookieData>();
    
    // Simulated user database
    private static Dictionary<string, UserData> userDatabase = new Dictionary<string, UserData>
    {
        { "user@example.com", new UserData { Email = "user@example.com", Password = "password123", Name = "John Doe" } },
        { "admin@example.com", new UserData { Email = "admin@example.com", Password = "admin456", Name = "Admin User" } }
    };

    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Stay Logged In Feature Demo ===\n");
        
        var conversationHistory = new List<Message>();
        
        // Initial system context
        string systemPrompt = @"You are a helpful assistant explaining the 'Stay Logged In' feature implementation in C#. 
        You have access to authentication functions and can help users understand:
        1. How persistent cookies work
        2. How to implement 'Remember Me' functionality
        3. Security best practices for persistent sessions
        4. How to validate and refresh authentication tokens
        
        When users ask about authentication, provide clear explanations and demonstrate the functionality.
        You can also help debug authentication issues and explain the code flow.";

        Console.WriteLine("Authentication Assistant is ready. Type 'quit' to exit.\n");
        Console.WriteLine("You can ask about:");
        Console.WriteLine("- How to implement 'Remember Me' feature");
        Console.WriteLine("- How persistent cookies work");
        Console.WriteLine("- Security best practices");
        Console.WriteLine("- How to validate sessions\n");

        while (true)
        {
            Console.Write("You: ");
            string? userInput = Console.ReadLine();
            
            if (string.IsNullOrEmpty(userInput) || userInput.ToLower() == "quit")
                break;

            // Check for special commands
            if (userInput.StartsWith("/login"))
            {
                await HandleLoginCommand(userInput, conversationHistory, systemPrompt);
                continue;
            }
            else if (userInput.StartsWith("/check"))
            {
                await HandleCheckSessionCommand(userInput, conversationHistory, systemPrompt);
                continue;
            }
            else if (userInput.StartsWith("/logout"))
            {
                await HandleLogoutCommand(userInput, conversationHistory, systemPrompt);
                continue;
            }

            // Regular conversation with Claude
            conversationHistory.Add(new Message
            {
                Role = RoleType.User,
                Content = userInput
            });

            var response = await client.Messages.GetClaudeMessageAsync(new MessageParameters
            {
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                System = new List<SystemMessage> { new SystemMessage(systemPrompt) },
                Messages = conversationHistory
            });

            string assistantMessage = response.Content[0].Text;
            
            conversationHistory.Add(new Message
            {
                Role = RoleType.Assistant,
                Content = assistantMessage
            });

            Console.WriteLine($"\nAssistant: {assistantMessage}\n");
        }
    }

    static async Task HandleLoginCommand(string command, List<Message> history, string systemPrompt)
    {
        // Parse: /login email password [remember]
        var parts = command.Split(' ');
        if (parts.Length < 3)
        {
            Console.WriteLine("Usage: /login <email> <password> [remember]\n");
            return;
        }

        string email = parts[1];
        string password = parts[2];
        bool rememberMe = parts.Length > 3 && parts[3].ToLower() == "remember";

        var loginResult = AuthenticateUser(email, password, rememberMe);
        
        string contextMessage = $@"User attempted to login with:
        - Email: {email}
        - Remember Me: {rememberMe}
        - Result: {loginResult.Message}
        - Cookie Token: {(loginResult.Success ? loginResult.Token : "N/A")}
        - Cookie Expiry: {(loginResult.Success && rememberMe ? "30 days" : loginResult.Success ? "Session only" : "N/A")}
        
        Please explain what happened during this authentication process and what the persistent cookie contains.";

        history.Add(new Message
        {
            Role = RoleType.User,
            Content = contextMessage
        });

        var response = await client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            System = new List<SystemMessage> { new SystemMessage(systemPrompt) },
            Messages = history
        });

        string assistantMessage = response.Content[0].Text;
        
        history.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = assistantMessage
        });

        Console.WriteLine($"\nLogin Result: {loginResult.Message}");
        if (loginResult.Success)
        {
            Console.WriteLine($"Token: {loginResult.Token}");
            Console.WriteLine($"Persistent: {rememberMe}");
        }
        Console.WriteLine($"\nAssistant: {assistantMessage}\n");
    }

    static async Task HandleCheckSessionCommand(string command, List<Message> history, string systemPrompt)
    {
        // Parse: /check <token>
        var parts = command.Split(' ');
        if (parts.Length < 2)
        {
            Console.WriteLine("Usage: /check <token>\n");
            return;
        }

        string token = parts[1];
        var sessionResult = ValidateSession(token);
        
        string contextMessage = $@"User is checking session validity for token: {token}
        - Session Valid: {sessionResult.IsValid}
        - User: {(sessionResult.IsValid ? sessionResult.UserEmail : "N/A")}
        - Expires: {(sessionResult.IsValid ? sessionResult.ExpiryTime.ToString() : "N/A")}
        - Is Persistent: {(sessionResult.IsValid ? sessionResult.IsPersistent : false)}
        
        Please explain the session validation process and what this means for the user's authentication state.";

        history.Add(new Message
        {
            Role = RoleType.User,
            Content = contextMessage
        });

        var response = await client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            System = new List<SystemMessage> { new SystemMessage(systemPrompt) },
            Messages = history
        });

        string assistantMessage = response.Content[0].Text;
        
        history.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = assistantMessage
        });

        Console.WriteLine($"\nSession Check Result:");
        Console.WriteLine($"Valid: {sessionResult.IsValid}");
        if (sessionResult.IsValid)
        {
            Console.WriteLine($"User: {sessionResult.UserEmail}");
            Console.WriteLine($"Expires: {sessionResult.ExpiryTime}");
            Console.WriteLine($"Persistent: {sessionResult.IsPersistent}");
        }
        Console.WriteLine($"\nAssistant: {assistantMessage}\n");
    }

    static async Task HandleLogoutCommand(string command, List<Message> history, string systemPrompt)
    {
        // Parse: /logout <token>
        var parts = command.Split(' ');
        if (parts.Length < 2)
        {
            Console.WriteLine("Usage: /logout <token>\n");
            return;
        }

        string token = parts[1];
        bool logoutSuccess = LogoutUser(token);
        
        string contextMessage = $@"User logged out with token: {token}
        - Logout Success: {logoutSuccess}
        - Cookie Cleared: {logoutSuccess}
        
        Please explain what happens during logout and why it's important to properly clear persistent cookies.";

        history.Add(new Message
        {
            Role = RoleType.User,
            Content = contextMessage
        });

        var response = await client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            System = new List<SystemMessage> { new SystemMessage(systemPrompt) },
            Messages = history
        });

        string assistantMessage = response.Content[0].Text;
        
        history.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = assistantMessage
        });

        Console.WriteLine($"\nLogout Result: {(logoutSuccess ? "Success" : "Failed - Token not found")}");
        Console.WriteLine($"\nAssistant: {assistantMessage}\n");
    }

    // Authentication Functions
    static LoginResult AuthenticateUser(string email, string password, bool rememberMe)
    {
        if (!userDatabase.ContainsKey(email))
        {
            return new LoginResult { Success = false, Message = "User not found" };
        }

        var user = userDatabase[email];
        if (user.Password != password)
        {
            return new LoginResult { Success = false, Message = "Invalid password" };
        }

        // Generate secure token
        string token = GenerateSecureToken();
        
        // Set cookie with appropriate expiry
        var cookieData = new CookieData
        {
            Token = token,
            UserEmail = email,
            CreatedAt = DateTime.UtcNow,
            ExpiresAt = rememberMe ? DateTime.UtcNow.AddDays(30) : DateTime.UtcNow.AddHours(1),
            IsPersistent = rememberMe,
            IsValid = true
        };

        cookieStore[token] = cookieData;

        return new LoginResult 
        { 
            Success = true, 
            Message = $"Login successful for {user.Name}",
            Token = token
        };
    }

    static SessionValidationResult ValidateSession(string token)
    {
        if (!cookieStore.ContainsKey(token))
        {
            return new SessionValidationResult { IsValid = false };
        }

        var cookie = cookieStore[token];
        
        if (!cookie.IsValid || DateTime.UtcNow > cookie.ExpiresAt)
        {
            // Remove expired cookie
            cookieStore.Remove(token);
            return new SessionValidationResult { IsValid = false };
        }

        // Refresh session if persistent and close to expiry
        if (cookie.IsPersistent && (cookie.ExpiresAt - DateTime.UtcNow).TotalDays < 7)
        {
            cookie.ExpiresAt = DateTime.UtcNow.AddDays(30);
        }

        return new SessionValidationResult
        {
            IsValid = true,
            UserEmail = cookie.UserEmail,
            ExpiryTime = cookie.ExpiresAt,
            IsPersistent = cookie.IsPersistent
        };
    }

    static bool LogoutUser(string token)
    {
        if (cookieStore.ContainsKey(token))
        {
            cookieStore.Remove(token);
            return true;
        }
        return false;
    }

    static string GenerateSecureToken()
    {
        // In production, use cryptographically secure random bytes
        return Convert.ToBase64String(Guid.NewGuid().ToByteArray()) + 
               Convert.ToBase64String(Guid.NewGuid().ToByteArray());
    }
}

// Data Models
class UserData
{
    public string Email { get; set; } = "";
    public string Password { get; set; } = "";
    public string Name { get; set; } = "";
}

class CookieData
{
    public string Token { get; set; } = "";
    public string UserEmail { get; set; } = "";
    public DateTime CreatedAt { get; set; }
    public DateTime ExpiresAt { get; set; }
    public bool IsPersistent { get; set; }
    public bool IsValid { get; set; }
}

class LoginResult
{
    public bool Success { get; set; }
    public string Message { get; set; } = "";
    public string Token { get; set; } = "";
}

class SessionValidationResult
{
    public bool IsValid { get; set; }
    public string UserEmail { get; set; } = "";
    public DateTime ExpiryTime { get; set; }
    public bool IsPersistent { get; set; }
}
