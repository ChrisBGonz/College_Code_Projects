using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

namespace ShoppingCartCheckout
{
    // Shopping cart item model
    public class CartItem
    {
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; }
    }

    // Checkout result model
    public class CheckoutResult
    {
        public List<CartItem> Items { get; set; }
        public decimal Subtotal { get; set; }
        public decimal TaxAmount { get; set; }
        public decimal TotalAmount { get; set; }
        public string Summary { get; set; }
        public bool IsSuccess { get; set; }
    }

    // Shopping Cart Checkout Processor
    public class ShoppingCartCheckoutProcessor
    {
        private readonly AnthropicClient _client;
        private const decimal TAX_RATE = 0.18m; // 18% GST

        public ShoppingCartCheckoutProcessor()
        {
            _client = new AnthropicClient();
        }

        public async Task<CheckoutResult> ProcessCheckoutAsync(List<CartItem> cartItems)
        {
            if (cartItems == null || cartItems.Count == 0)
            {
                return new CheckoutResult
                {
                    IsSuccess = false,
                    Summary = "Cart is empty. Please add items before checkout."
                };
            }

            // Calculate totals locally
            decimal subtotal = 0;
            foreach (var item in cartItems)
            {
                subtotal += item.Price * item.Quantity;
            }
            decimal taxAmount = subtotal * TAX_RATE;
            decimal totalAmount = subtotal + taxAmount;

            // Prepare cart data for Claude
            var cartJson = JsonSerializer.Serialize(cartItems, new JsonSerializerOptions { WriteIndented = true });
            
            var prompt = $@"You are a shopping cart checkout assistant. Analyze the following cart items and provide a detailed checkout summary in Hindi and English.

Cart Items (JSON):
{cartJson}

Calculated Values:
- Subtotal: ₹{subtotal:F2}
- Tax (18% GST): ₹{taxAmount:F2}
- Total Amount: ₹{totalAmount:F2}

Please provide:
1. A brief summary of items purchased
2. Confirmation of the total calculation
3. Any recommendations or notes about the purchase
4. A friendly checkout message

Format your response in a clear, customer-friendly way mixing Hindi and English.";

            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = prompt
                }
            };

            var parameters = new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                Stream = true
            };

            Console.WriteLine("\n🛒 Processing Checkout...\n");
            Console.WriteLine("=" .PadRight(50, '='));
            
            string fullResponse = "";
            
            await foreach (var streamEvent in _client.Messages.StreamClaudeMessageAsync(parameters))
            {
                if (streamEvent is ContentBlockDeltaEvent deltaEvent)
                {
                    if (deltaEvent.Delta is TextDelta textDelta)
                    {
                        Console.Write(textDelta.Text);
                        fullResponse += textDelta.Text;
                    }
                }
            }
            
            Console.WriteLine("\n" + "=".PadRight(50, '='));

            return new CheckoutResult
            {
                Items = cartItems,
                Subtotal = subtotal,
                TaxAmount = taxAmount,
                TotalAmount = totalAmount,
                Summary = fullResponse,
                IsSuccess = true
            };
        }

        public void DisplayCheckoutSummary(CheckoutResult result)
        {
            if (!result.IsSuccess)
            {
                Console.WriteLine($"❌ Checkout Failed: {result.Summary}");
                return;
            }

            Console.WriteLine("\n📋 CHECKOUT SUMMARY");
            Console.WriteLine("=".PadRight(50, '='));
            Console.WriteLine($"{"Item",-25} {"Qty",-5} {"Price",-10} {"Total",-10}");
            Console.WriteLine("-".PadRight(50, '-'));
            
            foreach (var item in result.Items)
            {
                decimal itemTotal = item.Price * item.Quantity;
                Console.WriteLine($"{item.Name,-25} {item.Quantity,-5} ₹{item.Price,-9:F2} ₹{itemTotal:F2}");
            }
            
            Console.WriteLine("-".PadRight(50, '-'));
            Console.WriteLine($"{"Subtotal:",-40} ₹{result.Subtotal:F2}");
            Console.WriteLine($"{"Tax (18% GST):",-40} ₹{result.TaxAmount:F2}");
            Console.WriteLine("=".PadRight(50, '='));
            Console.WriteLine($"{"TOTAL AMOUNT:",-40} ₹{result.TotalAmount:F2}");
            Console.WriteLine("=".PadRight(50, '='));
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("🛍️  Shopping Cart Checkout System");
            Console.WriteLine("==================================\n");

            // Sample cart items from frontend
            var cartItems = new List<CartItem>
            {
                new CartItem { Name = "Laptop", Price = 45000.00m, Quantity = 1 },
                new CartItem { Name = "Mouse", Price = 850.00m, Quantity = 2 },
                new CartItem { Name = "Keyboard", Price = 1200.00m, Quantity = 1 },
                new CartItem { Name = "USB Hub", Price = 650.00m, Quantity = 1 },
                new CartItem { Name = "Laptop Stand", Price = 1500.00m, Quantity = 1 }
            };

            Console.WriteLine("📦 Items in Cart:");
            foreach (var item in cartItems)
            {
                Console.WriteLine($"  • {item.Name}: ₹{item.Price} x {item.Quantity}");
            }

            var processor = new ShoppingCartCheckoutProcessor();
            
            // Display local summary first
            var tempResult = new CheckoutResult
            {
                Items = cartItems,
                IsSuccess = true
            };
            
            // Process checkout with Claude AI
            var checkoutResult = await processor.ProcessCheckoutAsync(cartItems);
            
            // Display final summary
            processor.DisplayCheckoutSummary(checkoutResult);
            
            Console.WriteLine("\n✅ Checkout process completed successfully!");
            Console.WriteLine("\nPress any key to exit...");
            Console.ReadKey();
        }
    }
}
