import anthropic
import socket
import threading
import time
import struct
import random

# XOR Cipher implementation
def xor_cipher(data: bytes, key: bytes) -> bytes:
    """Simple XOR cipher for encoding/decoding data"""
    result = bytearray()
    for i, byte in enumerate(data):
        result.append(byte ^ key[i % len(key)])
    return bytes(result)

def encode_message(message: str, key: bytes) -> bytes:
    """Encode a message using XOR cipher"""
    message_bytes = message.encode('utf-8')
    encrypted = xor_cipher(message_bytes, key)
    # Add length prefix for framing
    length = len(encrypted)
    return struct.pack('>I', length) + encrypted

def decode_message(data: bytes, key: bytes) -> str:
    """Decode a message using XOR cipher"""
    decrypted = xor_cipher(data, key)
    return decrypted.decode('utf-8')

# Legacy Server Simulation
class LegacyServer:
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.key = b'legacy_xor_key_2024'  # Shared XOR key
        self.server_socket = None
        self.running = False
        self.conversation_history = []
        
    def start(self):
        """Start the legacy server"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"[Legacy Server] Started on {self.host}:{self.port}")
        
        # Accept connections in a thread
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections"""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"[Legacy Server] Connection from {address}")
                client_thread = threading.Thread(
                    target=self._handle_client,
                    args=(client_socket,)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"[Legacy Server] Error accepting connection: {e}")
                    
    def _handle_client(self, client_socket: socket.socket):
        """Handle individual client connection"""
        try:
            while True:
                # Read length prefix
                length_data = self._recv_exact(client_socket, 4)
                if not length_data:
                    break
                    
                length = struct.unpack('>I', length_data)[0]
                
                # Read encrypted message
                encrypted_data = self._recv_exact(client_socket, length)
                if not encrypted_data:
                    break
                    
                # Decrypt message
                message = decode_message(encrypted_data, self.key)
                print(f"[Legacy Server] Received: {message}")
                
                # Process with Claude API
                response = self._process_with_claude(message)
                
                # Send encrypted response
                encoded_response = encode_message(response, self.key)
                client_socket.sendall(encoded_response)
                print(f"[Legacy Server] Sent response: {response[:50]}...")
                
        except Exception as e:
            print(f"[Legacy Server] Client error: {e}")
        finally:
            client_socket.close()
            
    def _recv_exact(self, sock: socket.socket, n: int) -> bytes:
        """Receive exactly n bytes from socket"""
        data = b''
        while len(data) < n:
            chunk = sock.recv(n - len(data))
            if not chunk:
                return None
            data += chunk
        return data
        
    def _process_with_claude(self, user_message: str) -> str:
        """Process message using Claude API with streaming"""
        client = anthropic.Anthropic()
        
        # Add to conversation history
        self.conversation_history.append({
            "role": "user",
            "content": user_message
        })
        
        # Use streaming to get response
        full_response = ""
        
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="You are a helpful assistant integrated into a legacy server system. Provide concise, helpful responses.",
            messages=self.conversation_history
        ) as stream:
            for text in stream.text_stream:
                full_response += text
                
        # Add assistant response to history
        self.conversation_history.append({
            "role": "assistant",
            "content": full_response
        })
        
        return full_response
        
    def stop(self):
        """Stop the server"""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("[Legacy Server] Stopped")

# Custom Socket Client
class LegacyClient:
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.key = b'legacy_xor_key_2024'  # Same XOR key as server
        self.socket = None
        
    def connect(self):
        """Connect to legacy server"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        print(f"[Client] Connected to {self.host}:{self.port}")
        
    def send_message(self, message: str) -> str:
        """Send encrypted message and receive response"""
        # Encode and send message
        encoded = encode_message(message, self.key)
        self.socket.sendall(encoded)
        print(f"[Client] Sent encrypted message: {message}")
        
        # Receive length prefix
        length_data = self._recv_exact(4)
        if not length_data:
            return None
            
        length = struct.unpack('>I', length_data)[0]
        
        # Receive encrypted response
        encrypted_response = self._recv_exact(length)
        if not encrypted_response:
            return None
            
        # Decrypt response
        response = decode_message(encrypted_response, self.key)
        return response
        
    def _recv_exact(self, n: int) -> bytes:
        """Receive exactly n bytes"""
        data = b''
        while len(data) < n:
            chunk = self.socket.recv(n - len(data))
            if not chunk:
                return None
            data += chunk
        return data
        
    def disconnect(self):
        """Disconnect from server"""
        if self.socket:
            self.socket.close()
        print("[Client] Disconnected")

def demonstrate_xor_cipher():
    """Demonstrate XOR cipher functionality"""
    print("\n" + "="*60)
    print("XOR Cipher Demonstration")
    print("="*60)
    
    key = b'demo_key_123'
    original = "Hello, Legacy Server! This is a secret message."
    
    # Encrypt
    encrypted = xor_cipher(original.encode(), key)
    print(f"Original: {original}")
    print(f"Key: {key.decode()}")
    print(f"Encrypted (hex): {encrypted.hex()}")
    
    # Decrypt
    decrypted = xor_cipher(encrypted, key)
    print(f"Decrypted: {decrypted.decode()}")
    print("="*60 + "\n")

def main():
    """Main function to demonstrate legacy server communication"""
    
    # Demonstrate XOR cipher
    demonstrate_xor_cipher()
    
    # Start legacy server
    server = LegacyServer()

    # Wait for server to start
    time.sleep(0.5)
    
    # Create client and connect
    client = LegacyClient()

    print("\n" + "="*60)
    print("Legacy Server Communication with Claude AI")
    print("="*60)
    
    # Test messages
    test_messages = [
