define('ADMIN_EMAIL', 'admin@example.com');
define('SLACK_WEBHOOK_URL', 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK');
define('LOG_FILE', __DIR__ . '/security_log.txt');
define('MAX_ATTEMPTS', 3);
define('BLOCK_DURATION', 1800); // 30 minutes in seconds

// Simulate user authentication
function isLoggedIn() {
    return isset($_SESSION['user_id']) && !empty($_SESSION['user_id']);
}

function isAdmin() {
    return isset($_SESSION['role']) && $_SESSION['role'] === 'admin';
}

function getUserInfo() {
    return [
        'user_id'  => $_SESSION['user_id'] ?? 'Guest',
        'username' => $_SESSION['username'] ?? 'Unknown',
        'role'     => $_SESSION['role'] ?? 'none',
        'email'    => $_SESSION['email'] ?? 'N/A',
    ];
}

// Get real IP address
function getRealIP() {
    $ip_keys = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR'
    ];

    foreach ($ip_keys as $key) {
        if (!empty($_SERVER[$key])) {
            $ips = explode(',', $_SERVER[$key]);
            $ip  = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
                return $ip;
            }
        }
    }

    return $_SERVER['REMOTE_ADDR'] ?? 'Unknown';
}

// Log unauthorized access attempt
function logUnauthorizedAccess($user_info, $ip, $url, $attempt_count) {
    $timestamp   = date('Y-m-d H:i:s');
    $user_agent  = $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown';
    $referer     = $_SERVER['HTTP_REFERER'] ?? 'Direct Access';
    $method      = $_SERVER['REQUEST_METHOD'] ?? 'GET';

    $log_entry = "[{$timestamp}] UNAUTHORIZED ACCESS ATTEMPT #{$attempt_count}\n";
    $log_entry .= "  IP Address  : {$ip}\n";
    $log_entry .= "  URL         : {$url}\n";
    $log_entry .= "  Method      : {$method}\n";
    $log_entry .= "  User ID     : {$user_info['user_id']}\n";
    $log_entry .= "  Username    : {$user_info['username']}\n";
    $log_entry .= "  Role        : {$user_info['role']}\n";
    $log_entry .= "  User Email  : {$user_info['email']}\n";
    $log_entry .= "  User Agent  : {$user_agent}\n";
    $log_entry .= "  Referer     : {$referer}\n";
    $log_entry .= str_repeat('-', 60) . "\n";

    file_put_contents(LOG_FILE, $log_entry, FILE_APPEND | LOCK_EX);

    return $log_entry;
}

        ];
    }

    $json_payload = json_encode($payload);

    $ch = curl_init(SLACK_WEBHOOK_URL);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'POST');
    curl_setopt($ch, CURLOPT_POSTFIELDS, $json_payload);

    ]);

    $result   = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return ($http_code === 200 && $result === 'ok');
}

// Track and check brute force attempts
function trackAttempt($ip) {
    $attempts_file = sys_get_temp_dir() . '/admin_attempts_' . md5($ip) . '.json';
    $current_time  = time();
    $attempts      = [];

    if (file_exists($attempts_file)) {
        $data = json_decode(file_get_contents($attempts_file), true);
        if ($data && isset($data['attempts'])) {
            // Remove old attempts outside the block window
            $attempts = array_filter($data['attempts'], function ($timestamp) use ($current_time) {
                return ($current_time - $timestamp) < BLOCK_DURATION;
            });
        }
    }

    $attempts[] = $current_time;
    file_put_contents($attempts_file, json_encode(['attempts' => array_values($attempts)]), LOCK_EX);

    return count($attempts);
}

function isIPBlocked($ip) {
    $attempts_file = sys_get_temp_dir() . '/admin_attempts_' . md5($ip) . '.json';
    $current_time  = time();

    if (!file_exists($attempts_file)) {
        return false;
    }

    $data = json_decode(file_get_contents($attempts_file), true);
    if (!$data || !isset($data['attempts'])) {
        return false;
    }

    $recent_attempts = array_filter($data['attempts'], function ($timestamp) use ($current_time) {
        return ($current_time - $timestamp) < BLOCK_DURATION;
    });

    return count($recent_attempts) >= MAX_ATTEMPTS;
}

// Main Admin Access Guard
function checkAdminAccess() {
    $ip         = getRealIP();
    $url        = (isset($_SERVER['HTTPS']) && $_SERVER['HTTPS'] === 'on' ? 'https' : 'http') . '://' . ($_SERVER['HTTP_HOST'] ?? 'localhost') . ($_SERVER['REQUEST_URI'] ?? '/admin');
    $user_info  = getUserInfo();
    $is_blocked = isIPBlocked($ip);

    // Check if IP is already blocked
    if ($is_blocked) {
        http_response_code(403);
        echo renderBlockedPage($ip);
        exit;
    }

    // Check if user has admin access
    if (!isLoggedIn() || !isAdmin()) {
        // Track this attempt
        $attempt_count = trackAttempt($ip);
        $is_now_blocked = ($attempt_count >= MAX_ATTEMPTS);

        // Log the attempt
        logUnauthorizedAccess($user_info, $ip, $url, $attempt_count);

        // Send alerts
        $email_sent = sendEmailAlert($user_info, $ip, $url, $attempt_count, $is_now_blocked);
        $slack_sent = sendSlackAlert($user_info, $ip, $url, $attempt_count, $is_now_blocked);

        // Log alert status
        $alert_log = "[" . date('Y-m-d H:i:s') . "] ALERTS SENT - Email: " . ($email_sent ? 'YES' : 'NO/DISABLED') . " | Slack: " . ($slack_sent ? 'YES' : 'NO/DISABLED') . "\n";
        file_put_contents(LOG_FILE, $alert_log, FILE_APPEND | LOCK_EX);

        // Return 403 Forbidden
        http_response_code(403);
        echo renderAccessDeniedPage($ip, $attempt_count, $is_now_blocked);
        exit;
    }
}
