// Create database connection
function getDBConnection() {
    $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
    if ($conn->connect_error) {
        die(json_encode(['success' => false, 'message' => 'Database connection failed']));
    }
    return $conn;
}

// Security functions
function sanitizeInput($input) {
    return htmlspecialchars(strip_tags(trim($input)));
}

function generateCSRFToken() {
    if (!isset($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCSRFToken($token) {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

function checkRateLimit($identifier) {
    if (!isset($_SESSION['rate_limit'])) {
        $_SESSION['rate_limit'] = [];
    }
    
    $key = 'recovery_' . md5($identifier);
    $now = time();
    $window = 900; // 15 minutes
    $maxAttempts = 5;
    
    if (!isset($_SESSION['rate_limit'][$key])) {
        $_SESSION['rate_limit'][$key] = ['count' => 0, 'first_attempt' => $now];
    }
    
    $data = &$_SESSION['rate_limit'][$key];
    
    if ($now - $data['first_attempt'] > $window) {
        $data = ['count' => 0, 'first_attempt' => $now];
    }
    
    if ($data['count'] >= $maxAttempts) {
        $remaining = $window - ($now - $data['first_attempt']);
        return ['allowed' => false, 'remaining' => $remaining];
    }
    
    $data['count']++;
    return ['allowed' => true, 'remaining' => 0];
}

// Database setup (run once)
function setupDatabase() {
    $conn = getDBConnection();
    
    $sql = "CREATE TABLE IF NOT EXISTS users (
        id INT AUTO_INCREMENT PRIMARY KEY,
        username VARCHAR(50) UNIQUE NOT NULL,
        email VARCHAR(100) UNIQUE NOT NULL,
        password VARCHAR(255) NOT NULL,
        security_question_id INT NOT NULL,
        security_answer VARCHAR(255) NOT NULL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
    )";
    
    $conn->query($sql);
    
    $sql2 = "CREATE TABLE IF NOT EXISTS password_reset_tokens (
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT NOT NULL,
        token VARCHAR(64) NOT NULL,
        expires_at TIMESTAMP NOT NULL,
        used TINYINT(1) DEFAULT 0,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
    )";
    
    $conn->query($sql2);
    $conn->close();
}

// Register user with security question
function registerUser($username, $email, $password, $questionId, $answer) {
    $conn = getDBConnection();
    
    $hashedPassword = password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
    $hashedAnswer = password_hash(strtolower(trim($answer)), PASSWORD_BCRYPT, ['cost' => 10]);
    
    $stmt = $conn->prepare("INSERT INTO users (username, email, password, security_question_id, security_answer) VALUES (?, ?, ?, ?, ?)");
    $stmt->bind_param("sssiss", $username, $email, $hashedPassword, $questionId, $hashedAnswer);
    
    // Fix: bind_param type string should match parameters
    $stmt = $conn->prepare("INSERT INTO users (username, email, password, security_question_id, security_answer) VALUES (?, ?, ?, ?, ?)");
    $stmt->bind_param("sssis", $username, $email, $hashedPassword, $questionId, $hashedAnswer);
    
    if ($stmt->execute()) {
        $stmt->close();
        $conn->close();
        return ['success' => true, 'message' => 'उपयोगकर्ता सफलतापूर्वक पंजीकृत हुआ'];
    } else {
        $error = $conn->error;
        $stmt->close();
        $conn->close();
        
        if (strpos($error, 'Duplicate entry') !== false) {
            return ['success' => false, 'message' => 'यह उपयोगकर्ता नाम या ईमेल पहले से मौजूद है'];
        }
        return ['success' => false, 'message' => 'पंजीकरण विफल हुआ'];
    }
}

// Step 1: Find user by email/username
function findUserForRecovery($identifier) {
    $conn = getDBConnection();
    
    $stmt = $conn->prepare("SELECT id, username, email, security_question_id FROM users WHERE email = ? OR username = ?");
    $stmt->bind_param("ss", $identifier, $identifier);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 1) {
        $user = $result->fetch_assoc();
        $stmt->close();
        $conn->close();
        return ['success' => true, 'user' => $user];
    }
    
    $stmt->close();
    $conn->close();
    return ['success' => false, 'message' => 'कोई खाता नहीं मिला'];
}

// Step 2: Verify security answer
function verifySecurityAnswer($userId, $answer) {
    $conn = getDBConnection();
    
    $stmt = $conn->prepare("SELECT security_answer FROM users WHERE id = ?");
    $stmt->bind_param("i", $userId);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 1) {
        $user = $result->fetch_assoc();
        $stmt->close();
        
        if (password_verify(strtolower(trim($answer)), $user['security_answer'])) {
            // Generate reset token
            $token = bin2hex(random_bytes(32));
            $expiresAt = date('Y-m-d H:i:s', time() + 1800); // 30 minutes
            
            $stmt2 = $conn->prepare("INSERT INTO password_reset_tokens (user_id, token, expires_at) VALUES (?, ?, ?)");
            $stmt2->bind_param("iss", $userId, $token, $expiresAt);
            $stmt2->execute();
            $stmt2->close();
            $conn->close();
            
            return ['success' => true, 'token' => $token];
        }
        
        $conn->close();
        return ['success' => false, 'message' => 'गलत उत्तर। कृपया पुनः प्रयास करें।'];
    }
    
    $stmt->close();
    $conn->close();
    return ['success' => false, 'message' => 'उपयोगकर्ता नहीं मिला'];
}

// Step 3: Reset password with token
function resetPassword($token, $newPassword) {
    $conn = getDBConnection();
    
    $stmt = $conn->prepare("SELECT prt.user_id, prt.expires_at, prt.used 
                            FROM password_reset_tokens prt 
                            WHERE prt.token = ?");
    $stmt->bind_param("s", $token);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 1) {
        $tokenData = $result->fetch_assoc();
        $stmt->close();
        
        if ($tokenData['used']) {
            $conn->close();
            return ['success' => false, 'message' => 'यह टोकन पहले ही उपयोग किया जा चुका है'];
        }
        
        if (strtotime($tokenData['expires_at']) < time()) {
            $conn->close();
            return ['success' => false, 'message' => 'टोकन की समय सीमा समाप्त हो गई है'];
        }
        
        $hashedPassword = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
        
        $stmt2 = $conn->prepare("UPDATE users SET password = ? WHERE id = ?");
        $stmt2->bind_param("si", $hashedPassword, $tokenData['user_id']);
        $stmt2->execute();
        $stmt2->close();
        
        $stmt3 = $conn->prepare("UPDATE password_reset_tokens SET used = 1 WHERE token = ?");
        $stmt3->bind_param("s", $token);
        $stmt3->execute();
        $stmt3->close();
        
        $conn->close();
        return ['success' => true, 'message' => 'पासवर्ड सफलतापूर्वक बदल दिया गया'];
    }
    
    $stmt->close();
    $conn->close();
    return ['success' => false, 'message' => 'अमान्य टोकन'];
}

// Validate password strength
function validatePassword($password) {
    if (strlen($password) < 8) {
        return ['valid' => false, 'message' => 'पासवर्ड कम से कम 8 अक्षर का होना चाहिए'];
    }
    if (!preg_match('/[A-Z]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक बड़ा अक्षर होना चाहिए'];
    }
    if (!preg_match('/[a-z]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक छोटा अक्षर होना चाहिए'];
    }
    if (!preg_match('/[0-9]/', $password)) {
        return ['valid' => false, 'message' => 'पासवर्ड में कम से कम एक अंक होना चाहिए'];
    }
    return ['valid' => true];
}

// Handle AJAX requests
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
    header('Content-Type: application/json');
    
    $action = sanitizeInput($_POST['action']);
    
    // CSRF validation for all POST requests
    if (!isset($_POST['csrf_token']) || !validateCSRFToken($_POST['csrf_token'])) {
        echo json_encode(['success' => false, 'message' => 'अमान्य अनुरोध']);
        exit;
    }
    
    switch ($action) {
        case 'setup_db':
            setupDatabase();
            echo json_encode(['success' => true, 'message' => 'डेटाबेस सेटअप हो गया']);
            break;
            
        case 'register':
            $username = sanitizeInput($_POST['username'] ?? '');
            $email = filter_var($_POST['email'] ?? '', FILTER_SANITIZE_EMAIL);
            $password = $_POST['password'] ?? '';
            $questionId = (int)($_POST['question_id'] ?? 0);
            $answer = $_POST['security_answer'] ?? '';
            
            if (empty($username) || empty($email) || empty($password) || empty($questionId) || empty($answer)) {
                echo json_encode(['success' => false, 'message' => 'सभी फ़ील्ड भरना आवश्यक है']);
                break;
            }
            
            if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
                echo json_encode(['success' => false, 'message' => 'अमान्य ईमेल पता']);
                break;
            }
            
            $passwordCheck = validatePassword($password);
            if (!$passwordCheck['valid']) {
                echo json_encode(['success' => false, 'message' => $passwordCheck['message']]);
                break;
            }
            
            $questions = getSecurityQuestions();
            if (!isset($questions[$questionId])) {
                echo json_encode(['success' => false, 'message' => 'अमान्य सुरक्षा प्रश्न']);
                break;
            }
            
            echo json_encode(registerUser($username, $email, $password, $questionId, $answer));
            break;
            
        case 'find_user':
            $identifier = sanitizeInput($_POST['identifier'] ?? '');
            
            if (empty($identifier)) {
                echo json_encode(['success' => false, 'message' => 'ईमेल या उपयोगकर्ता नाम दर्ज करें']);
                break;
            }
            
            $rateCheck = checkRateLimit($identifier);
            if (!$rateCheck['allowed']) {
                $minutes = ceil($rateCheck['remaining'] / 60);
                echo json_encode(['success' => false, 'message' => "बहुत अधिक प्रयास। {$minutes} मिनट बाद पुनः प्रयास करें।"]);
                break;
            }
            
            $result = findUserForRecovery($identifier);
            
            if ($result['success']) {
                $questions = getSecurityQuestions();
                $questionId = $result['user']['security_question_id'];
                
                $_SESSION['recovery_user_id'] = $result['user']['id'];
                $_SESSION['recovery_step'] = 'answer_question';
                
                echo json_encode([
                    'success' => true,
                    'question' => $questions[$questionId] ?? 'अज्ञात प्रश्न',
                    'username' => $result['user']['username']
                ]);
            } else {
                // Don't reveal if user exists or not for security
                echo json_encode(['success' => false, 'message' => 'यदि यह खाता मौजूद है, तो आगे बढ़ें']);
            }
            break;
            
        case 'verify_answer':
            if (!isset($_SESSION['recovery_user_id']) || $_SESSION['recovery_step'] !== 'answer_question') {
                echo json_encode(['success' => false, 'message' => 'अमान्य सत्र। पुनः शुरू करें।']);
                break;
            }
            
            $answer = $_POST['security_answer'] ?? '';
            $userId = $_SESSION['recovery_user_id'];
            
            if (empty($answer)) {
                echo json_encode(['success' => false, 'message' => 'उत्तर दर्ज करें']);
                break;
            }
            
            $rateCheck = checkRateLimit('user_' . $userId);
            if (!$rateCheck['allowed']) {
                $minutes = ceil($rateCheck['remaining'] / 60);
                echo json_encode(['success' => false, 'message' => "बहुत अधिक गलत प्रयास। {$minutes} मिनट बाद पुनः प्रयास करें।"]);
                break;
            }
            
            $result = verifySecurityAnswer($userId, $answer);
            
            if ($result['success']) {
                $_SESSION['recovery_token'] = $result['token'];
                $_SESSION['recovery_step'] = 'reset_password';
                unset($_SESSION['recovery_user_id']);
                
                echo json_encode(['success' => true, 'token' => $result['token']]);
            } else {
                echo json_encode($result);
            }
            break;
            
        case 'reset_password':
            $token = $_POST['token'] ?? '';
            $newPassword = $_POST['new_password'] ?? '';
            $confirmPassword = $_POST['confirm_password'] ?? '';
            
            if (empty($token) || empty($newPassword) || empty($confirmPassword)) {
                echo json_encode(['success' => false, 'message' => 'सभी फ़ील्ड भरना आवश्यक है']);
                break;
            }
            
            if ($newPassword !== $confirmPassword) {
                echo json_encode(['success' => false, 'message' => 'पासवर्ड मेल नहीं खाते']);
                break;
            }
            
            $passwordCheck = validatePassword($newPassword);
            if (!$passwordCheck['valid']) {
                echo json_encode(['success' => false, 'message' => $passwordCheck['message']]);
                break;
            }
            
            // Verify token matches session
            if (!isset($_SESSION['recovery_token']) || $_SESSION['recovery_token'] !== $token) {
                echo json_encode(['success' => false, 'message' => 'अमान्य टोकन']);
                break;
            }
            
            $result = resetPassword($token, $newPassword);
            
            if ($result['success']) {
                unset($_SESSION['recovery_token']);
                unset($_SESSION['recovery_step']);
            }
            
            echo json_encode($result);
            break;
            
        default:
            echo json_encode(['success' => false, 'message' => 'अमान्य क्रिया']);
    }
    exit;
}
