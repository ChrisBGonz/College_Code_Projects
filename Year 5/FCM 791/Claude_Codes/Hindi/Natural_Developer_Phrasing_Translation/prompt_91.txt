import anthropic
import time


def connect_payment_api_with_claude(payment_data: dict) -> bool:
    """
    Connect to external Payment API using Claude AI.
    If API doesn't respond within 2 seconds (timeout), return True to allow user to proceed.
    
    Args:
        payment_data: Dictionary containing payment information
        
    Returns:
        bool: True if payment can proceed (either success or timeout), False if payment failed
    """
    client = anthropic.Anthropic()
    
    # Prepare the payment request message
    payment_request = f"""
    Process the following payment request and simulate an API call:
    
    Payment Data: {payment_data}
    
    Please simulate a payment API response. The response should include:
    1. Transaction ID
    2. Status (success/failed)
    3. Message
    4. Timestamp
    
    Respond in a structured format.
    """
    
    start_time = time.time()
    response_received = False
    full_response = ""
    
    print("Connecting to Payment API...")
    print("-" * 50)
    
    try:
        # Use streaming to handle the response
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=500,
            messages=[
                {
                    "role": "user",
                    "content": payment_request
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                current_time = time.time()
                elapsed_time = current_time - start_time
                
                # Check if 2 seconds have passed
                if elapsed_time > 2.0 and not response_received:
                    print(f"\n⚠️  API Timeout! ({elapsed_time:.2f} seconds elapsed)")
                    print("Timeout occurred - Allowing user to proceed...")
                    return True
                
                # Accumulate the response
                full_response += text
                print(text, end="", flush=True)
                response_received = True
        
        # Calculate total time
        total_time = time.time() - start_time
        print(f"\n\n✅ API Response received in {total_time:.2f} seconds")
        
        # Check if response indicates success
        if "success" in full_response.lower() or "approved" in full_response.lower():
            print("Payment API: Transaction Successful")
            return True
        elif "failed" in full_response.lower() or "declined" in full_response.lower():
            print("Payment API: Transaction Failed")
            return False
        else:
            # If unclear, allow to proceed
            print("Payment API: Status unclear - Allowing to proceed")
            return True
            
    except Exception as e:
        elapsed_time = time.time() - start_time
        
        # Check if it's a timeout-related error
        if elapsed_time >= 2.0:
            print(f"\n⚠️  Connection Timeout! ({elapsed_time:.2f} seconds)")
            print("Network timeout - Allowing user to proceed...")
            return True
        else:
            print(f"\n❌ API Error: {str(e)}")
            print("Error occurred - Allowing user to proceed due to technical issues...")
            return True  # Return True on errors to not block user


def simulate_payment_process(payment_info: dict) -> None:
    """
    Simulate the complete payment process with timeout handling.
    
    Args:
        payment_info: Dictionary containing payment details
    """
    print("=" * 60)
    print("PAYMENT PROCESSING SYSTEM")
    print("=" * 60)
    print(f"Processing payment for: {payment_info.get('customer_name', 'Unknown')}")
    print(f"Amount: ₹{payment_info.get('amount', 0)}")
    print(f"Payment Method: {payment_info.get('payment_method', 'Unknown')}")
    print("-" * 60)
    
    # Attempt to connect to payment API
    can_proceed = connect_payment_api_with_claude(payment_info)
    
    print("\n" + "=" * 60)
    if can_proceed:
        print("✅ PAYMENT STATUS: User can proceed!")
        print("Transaction has been initiated successfully.")
        print("Please wait for confirmation email.")
    else:
        print("❌ PAYMENT STATUS: Cannot proceed")
        print("Please try again or use a different payment method.")
    print("=" * 60)


# Main execution
if __name__ == "__main__":
