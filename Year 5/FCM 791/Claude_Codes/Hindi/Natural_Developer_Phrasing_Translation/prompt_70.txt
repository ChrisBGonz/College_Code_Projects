const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const users = {
  user123: {
    id: "user123",
    name: "Rahul Sharma",
    email: "rahul@example.com",
    phone: "+91-9876543210",
    address: "Mumbai, Maharashtra",
    accountBalance: 50000,
    transactions: [
      { id: "t1", amount: 1000, type: "credit", date: "2024-01-15" },
      { id: "t2", amount: 500, type: "debit", date: "2024-01-16" },
    ],
  },
  user456: {
    id: "user456",
    name: "Priya Patel",
    email: "priya@example.com",
    phone: "+91-8765432109",
    address: "Ahmedabad, Gujarat",
    accountBalance: 75000,
    transactions: [
      { id: "t3", amount: 2000, type: "credit", date: "2024-01-14" },
      { id: "t4", amount: 1500, type: "debit", date: "2024-01-17" },
    ],
  },
};

// Valid tokens mapping to user IDs
const validTokens = {
  "token-abc-123": "user123",
  "token-xyz-456": "user456",
  "admin-token-789": "user123",
};

// Function to validate token using Claude AI
async function validateTokenWithClaude(token, requestPath) {
  const conversationHistory = [];

  // First message - provide context about the token
  conversationHistory.push({
    role: "user",
    content: `I have a Bearer token that needs to be validated for API access. 
    Token: ${token}
    Request Path: ${requestPath}
    
    Valid tokens in our system are: ${JSON.stringify(Object.keys(validTokens))}
    
    Please analyze if this token is valid and what user it belongs to. 
    Respond with a JSON object containing: 
    {
      "isValid": boolean,
      "userId": string or null,
      "reason": string,
      "securityLevel": "high" | "medium" | "low"
    }`,
  });

  const response1 = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a security validation assistant. Analyze tokens and provide validation results in JSON format. Be precise and security-conscious.",
    messages: conversationHistory,
  });

  const assistantMessage1 = response1.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage1,
  });

  // Second message - ask for additional security checks
  conversationHistory.push({
    role: "user",
    content: `Based on your analysis, should we proceed with returning private user data? 
    Also check if there are any security concerns with this request.
    Provide your final recommendation in JSON format:
    {
      "shouldProceed": boolean,
      "warnings": string[],
      "recommendation": string
    }`,
  });

  const response2 = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a security validation assistant. Analyze tokens and provide validation results in JSON format. Be precise and security-conscious.",
    messages: conversationHistory,
  });

  const assistantMessage2 = response2.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage2,
  });

  // Parse the responses
  let validationResult = null;
  let securityCheck = null;

  try {
    // Extract JSON from first response
    const jsonMatch1 = assistantMessage1.match(/\{[\s\S]*\}/);
    if (jsonMatch1) {
      validationResult = JSON.parse(jsonMatch1[0]);
    }

    // Extract JSON from second response
    const jsonMatch2 = assistantMessage2.match(/\{[\s\S]*\}/);
    if (jsonMatch2) {
      securityCheck = JSON.parse(jsonMatch2[0]);
    }
  } catch (e) {
    console.error("Error parsing Claude response:", e);
  }

  return {
    validationResult,
    securityCheck,
    conversationHistory,
  };
}

// Function to get user data
function getUserData(userId) {
  return users[userId] || null;
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle OPTIONS request
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Only handle GET requests to /api/user/data
  if (req.method !== "GET" || req.url !== "/api/user/data") {
    res.writeHead(404);
    res.end(
      JSON.stringify({
        error: "Not Found",
        message: "Endpoint not found. Use GET /api/user/data",
      })
    );
    return;
  }

  // Extract Authorization header
  const authHeader = req.headers["authorization"];

  if (!authHeader) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: "Authorization header is missing",
        hint: "Please provide: Authorization: Bearer <token>",
      })
    );
    return;
  }

  // Check Bearer token format
  if (!authHeader.startsWith("Bearer ")) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: "Invalid authorization format",
        hint: "Format should be: Authorization: Bearer <token>",
      })
    );
    return;
  }

  const token = authHeader.substring(7); // Remove 'Bearer ' prefix

  if (!token) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: "Token is empty",
      })
    );
    return;
  }

  try {
    console.log(`\n🔐 Validating token: ${token}`);
    console.log("📡 Consulting Claude AI for token validation...\n");

    // Validate token using Claude AI with multi-turn conversation
    const { validationResult, securityCheck, conversationHistory } =
      await validateTokenWithClaude(token, req.url);

    console.log("Claude Validation Result:", validationResult);
    console.log("Claude Security Check:", securityCheck);

    // Check if token is valid based on our actual token database
    const userId = validTokens[token];

    if (!userId || !validationResult?.isValid) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
          claudeAnalysis: {
            reason: validationResult?.reason || "Token validation failed",
            securityLevel: validationResult?.securityLevel || "low",
          },
        })
      );
      return;
    }

    // Check security warnings from Claude
    if (securityCheck?.warnings && securityCheck.warnings.length > 0) {
      console.log("⚠️  Security warnings:", securityCheck.warnings);
    }

    // Get user data
    const userData = getUserData(userId);

    if (!userData) {
      res.writeHead(404);
      res.end(
        JSON.stringify({
          error: "Not Found",
          message: "User data not found",
        })
      );
      return;
    }

    // Return successful response with user data
    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        message: "Authentication successful",
        data: {
          user: userData,
          tokenInfo: {
            validatedAt: new Date().toISOString(),
            securityLevel: validationResult?.securityLevel || "medium",
          },
        },
        claudeValidation: {
          reason: validationResult?.reason,
          recommendation: securityCheck?.recommendation,
          warnings: securityCheck?.warnings || [],
        },
        conversationTurns: conversationHistory.length / 2, // Number of back-and-forth exchanges
      })
    );

    console.log(`✅ Successfully authenticated user: ${userData.name}`);
  } catch (error) {
    console.error("Error during token validation:", error);
    res.writeHead(500);
    res.end(
      JSON.stringify({
        error: "Internal Server Error",
        message: "Token validation failed due to server error",
      })
    );
  }
});

// Start server
const PORT = 3000;
server.listen(PORT, () => {
  console.log(`\n🚀 API Server running on http://localhost:${PORT}`);
  console.log("\n📋 Available Endpoints:");
  console.log("  GET /api/user/data - Get private user data (requires auth)");
  console.log("\n🔑 Valid Test Tokens:");
  console.log("  - token-abc-123 (User: Rahul Sharma)");
  console.log("  - token-xyz-456 (User: Priya Patel)");
  console.log("  - admin-token-789 (User: Rahul Sharma)");
  console.log("\n📝 Example curl command:");
  console.log(
    '  curl -H "Authorization: Bearer token-abc-123" http://localhost:3000/api/user/data'
  );
  console.log("\n⏳ Waiting for requests...\n");
});
