def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

def validate_profile_config(config):
    """Validate the YAML config structure for profile setup"""
    required_fields = ['name', 'email']
    errors = []
    
    if not isinstance(config, dict):
        return False, ["Config must be a YAML dictionary/object"]
    
    for field in required_fields:
        if field not in config:
            errors.append(f"Required field '{field}' is missing")
    
    # Validate email format if present
    if 'email' in config:
        email = config['email']
        if not isinstance(email, str) or '@' not in email:
            errors.append("Invalid email format")
    
    # Validate age if present
    if 'age' in config:
        age = config['age']
        if not isinstance(age, int) or age < 0 or age > 150:
            errors.append("Age must be a valid integer between 0 and 150")
    
    # Validate preferences if present
    if 'preferences' in config:
        prefs = config['preferences']
        if not isinstance(prefs, dict):
            errors.append("Preferences must be a dictionary")
    
    # Validate skills if present
    if 'skills' in config:
        skills = config['skills']
        if not isinstance(skills, list):
            errors.append("Skills must be a list")
    
    return len(errors) == 0, errors

def initialize_profile(config):
    """Initialize user profile from YAML config"""
    profile_id = str(uuid.uuid4())
    
    profile = {
        'id': profile_id,

        'location': config.get('location', ''),
        'website': config.get('website', ''),
        'skills': config.get('skills', []),
        'preferences': config.get('preferences', {}),

        'notifications': config.get('notifications', {
            'email': True,
            'sms': False,
            'push': True
        }),
        'privacy': config.get('privacy', {
            'profile_visible': True,
            'show_email': False,
            'show_age': False
        }),
        'theme': config.get('theme', 'default'),
        'language': config.get('language', 'en'),
        'timezone': config.get('timezone', 'UTC'),
        'created_at': str(uuid.uuid4()),  # In production, use actual timestamp
        'status': 'active'
    }
    
    # Store profile
    user_profiles[profile_id] = profile
    
    return profile

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({
            'success': False,
            'message': 'कोई file नहीं मिली'
        }), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({
            'success': False,
            'message': 'File select नहीं की गई'
        }), 400
    
    if not allowed_file(file.filename):
        return jsonify({
            'success': False,
            'message': 'केवल .yaml या .yml files ही allowed हैं'
        }), 400
    
    try:
        # Read and parse YAML content
        yaml_content = file.read().decode('utf-8')
        config = yaml.safe_load(yaml_content)
        
        if config is None:
            return jsonify({
                'success': False,
                'message': 'YAML file empty है'
            }), 400
        
        # Validate the config
        is_valid, errors = validate_profile_config(config)
        
        if not is_valid:
            return jsonify({
                'success': False,
                'message': 'Config validation failed',
                'errors': errors
            }), 400
        
        # Initialize the profile
        profile = initialize_profile(config)
        
        # Save the file
        filename = secure_filename(f"{profile['id']}_{file.filename}")
        file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(yaml_content)
        
        return jsonify({
            'success': True,
            'message': 'Profile successfully initialize हो गया!',
            'profile': profile,
            'file_saved': filename
        }), 201
        
    except yaml.YAMLError as e:
        return jsonify({
            'success': False,
            'message': f'Invalid YAML format: {str(e)}'
        }), 400
    except UnicodeDecodeError:
        return jsonify({
            'success': False,
            'message': 'File encoding error - कृपया UTF-8 encoded file upload करें'
        }), 400
    except Exception as e:
        return jsonify({
            'success': False,
            'message': f'Server error: {str(e)}'
        }), 500

@app.route('/profiles', methods=['GET'])
def get_profiles():

        'success': True,

    })

@app.route('/profile/<profile_id>', methods=['GET'])
def get_profile(profile_id):
    if profile_id not in user_profiles:
        return jsonify({
            'success': False,
            'message': 'Profile नहीं मिला'
        }), 404
    
    return jsonify({
        'success': True,
        'profile': user_profiles[profile_id]
    })

@app.route('/profile/<profile_id>', methods=['DELETE'])
def delete_profile(profile_id):
    if profile_id not in user_profiles:
        return jsonify({
            'success': False,
            'message': 'Profile नहीं मिला'
        }), 404
    
    deleted_profile = user_profiles.pop(profile_id)
    
    return jsonify({
        'success': True,
        'message': f"Profile '{deleted_profile['name']}' delete हो गया",
        'deleted_profile_id': profile_id
    })

@app.route('/validate', methods=['POST'])
def validate_yaml():
    """Endpoint to validate YAML without creating a profile"""
    if 'file' not in request.files:
        return jsonify({
            'success': False,
            'message': 'कोई file नहीं मिली'
        }), 400
    
    file = request.files['file']
    
    if not allowed_file(file.filename):
        return jsonify({
            'success': False,
            'message': 'केवल .yaml या .yml files ही allowed हैं'
        }), 400
    
    try:
        yaml_content = file.read().decode('utf-8')
        config = yaml.safe_load(yaml_content)
        
        if config is None:
            return jsonify({
                'success': False,
                'message': 'YAML file empty है'
            }), 400
        
        is_valid, errors = validate_profile_config(config)
        
        return jsonify({
            'success': is_valid,
            'valid': is_valid,
            'errors': errors if not is_valid else [],
            'message': 'YAML valid है!' if is_valid else 'Validation failed',
            'parsed_fields': list(config.keys()) if isinstance(config, dict) else []
        })
        
    except yaml.YAMLError as e:
        return jsonify({
            'success': False,
            'valid': False,
            'message': f'Invalid YAML syntax: {str(e)}'
        }), 400
