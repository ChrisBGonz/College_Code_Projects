const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definitions for Claude
const tools = [
  {
    name: "download_file",
    description:
      "Downloads a file from a remote URL and saves it to a local path",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path where the downloaded file will be saved",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description: "Extracts a ZIP file to a specified directory",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "Path to the ZIP file to extract",
        },
        extract_to: {
          type: "string",
          description: "Directory path where the ZIP contents will be extracted",
        },
      },
      required: ["zip_path", "extract_to"],
    },
  },
  {
    name: "create_directory",
    description: "Creates a directory if it doesn't exist",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to create",
        },
      },
      required: ["path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Removes a temporary file after extraction",
    input_schema: {
      type: "object",
      properties: {
        file_path: {
          type: "string",
          description: "Path to the file to remove",
        },
      },
      required: ["file_path"],
    },
  },
  {
    name: "list_directory",
    description: "Lists the contents of a directory",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to list",
        },
      },
      required: ["path"],
    },
  },
];

// Tool implementation functions
function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    const protocol = url.startsWith("https://") ? https : http;
    const file = fs.createWriteStream(destination);

    const request = protocol.get(url, (response) => {
      if (response.statusCode === 301 || response.statusCode === 302) {
        // Handle redirects
        file.close();
        fs.unlink(destination, () => {});
        downloadFile(response.headers.location, destination)
          .then(resolve)
          .catch(reject);
        return;
      }

      if (response.statusCode !== 200) {
        file.close();
        fs.unlink(destination, () => {});
        reject(new Error(`Failed to download: HTTP ${response.statusCode}`));
        return;
      }

      response.pipe(file);

      file.on("finish", () => {
        file.close();
        const stats = fs.statSync(destination);
        resolve({
          success: true,
          message: `File downloaded successfully to ${destination}`,
          size: stats.size,
          path: destination,
        });
      });
    });

    request.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });

    file.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });
  });
}

function extractZip(zipPath, extractTo) {
  try {
    // Check if zip file exists
    if (!fs.existsSync(zipPath)) {
      return {
        success: false,
        error: `ZIP file not found: ${zipPath}`,
      };
    }

    // Create extraction directory if it doesn't exist
    if (!fs.existsSync(extractTo)) {
      fs.mkdirSync(extractTo, { recursive: true });
    }

    // Use system unzip command
    try {
      execSync(`unzip -o "${zipPath}" -d "${extractTo}"`, {
        stdio: "pipe",
      });
    } catch {
      // Try with different flags if first attempt fails
      execSync(`unzip -o -q "${zipPath}" -d "${extractTo}"`, {
        stdio: "pipe",
      });
    }

    return {
      success: true,
      message: `ZIP file extracted successfully to ${extractTo}`,
      zip_path: zipPath,
      extract_to: extractTo,
    };
  } catch (error) {
    return {
      success: false,
      error: `Failed to extract ZIP: ${error.message}`,
    };
  }
}

function createDirectory(dirPath) {
  try {
    if (!fs.existsSync(dirPath)) {
      fs.mkdirSync(dirPath, { recursive: true });
      return {
        success: true,
        message: `Directory created: ${dirPath}`,
        created: true,
      };
    } else {
      return {
        success: true,
        message: `Directory already exists: ${dirPath}`,
        created: false,
      };
    }
  } catch (error) {
    return {
      success: false,
      error: `Failed to create directory: ${error.message}`,
    };
  }
}

function cleanupFile(filePath) {
  try {
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
      return {
        success: true,
        message: `File removed: ${filePath}`,
      };
    } else {
      return {
        success: true,
        message: `File not found (already removed): ${filePath}`,
      };
    }
  } catch (error) {
    return {
      success: false,
      error: `Failed to remove file: ${error.message}`,
    };
  }
}

function listDirectory(dirPath) {
  try {
    if (!fs.existsSync(dirPath)) {
      return {
        success: false,
        error: `Directory not found: ${dirPath}`,
      };
    }

    const items = fs.readdirSync(dirPath);
    const details = items.map((item) => {
      const itemPath = path.join(dirPath, item);
      const stats = fs.statSync(itemPath);
      return {
        name: item,
        type: stats.isDirectory() ? "directory" : "file",
        size: stats.size,
      };
    });

    return {
      success: true,
      path: dirPath,
      items: details,
      count: details.length,
    };
  } catch (error) {
    return {
      success: false,
      error: `Failed to list directory: ${error.message}`,
    };
  }
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      try {
        result = await downloadFile(toolInput.url, toolInput.destination);
      } catch (error) {
        result = {
          success: false,
          error: error.message,
        };
      }
      break;

    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.extract_to);
      break;

    case "create_directory":
      result = createDirectory(toolInput.path);
      break;

    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;

    case "list_directory":
      result = listDirectory(toolInput.path);
      break;

    default:
      result = {
        success: false,
        error: `Unknown tool: ${toolName}`,
      };
  }

  console.log("Result:", JSON.stringify(result, null, 2));
  return result;
}

// Main function to download and extract ZIP update
async function downloadAndExtractUpdate(updateUrl, extractPath = "/app") {
  console.log("=".repeat(60));
  console.log("ZIP Update Downloader and Extractor");
  console.log("=".repeat(60));
  console.log(`Update URL: ${updateUrl}`);
  console.log(`Extract Path: ${extractPath}`);
  console.log("=".repeat(60));

  const messages = [
    {
      role: "user",
      content: `Please help me download a ZIP update from the following URL and extract it to the specified directory.

URL: ${updateUrl}
Extract to: ${extractPath}

Please:
1. Create the target directory if it doesn't exist
2. Download the ZIP file to a temporary location (use /tmp/update.zip)
3. Extract the ZIP file to ${extractPath}
4. Clean up the temporary ZIP file after extraction
5. List the contents of ${extractPath} to confirm successful extraction

Use the available tools to complete this task step by step.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log("\n" + "-".repeat(40));
    console.log("Claude Response:");
    console.log(`Stop Reason: ${response.stop_reason}`);

    // Process the response content
    const toolResults = [];
    let hasToolUse = false;

    for (const block of response.content) {
      if (block.type === "text") {
        console.log("\nClaude says:", block.text);
      } else if (block.type === "tool_use") {
        hasToolUse = true;
        const toolResult = await processToolCall(block.name, block.input);
        toolResults.push({
          type: "tool_result",
          tool_use_id: block.id,
          content: JSON.stringify(toolResult),
        });
      }
    }

    // Add assistant response to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    // If there were tool calls, add results and continue
    if (hasToolUse && toolResults.length > 0) {
      messages.push({
        role: "user",
        content: toolResults,
      });
    }

    // Check if we should continue the loop
    if (response.stop_reason === "end_turn" || !hasToolUse) {
      continueLoop = false;
    }
  }

  console.log("\n" + "=".repeat(60));
  console.log("Update process completed!");
  console.log("=".repeat(60));
}

  // Using a real test ZIP file from httpbin.org or similar service

  const testUrl =
