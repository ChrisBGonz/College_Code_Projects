const RegistrationForm = (() => {
  const STORAGE_KEY = 'registration_form_data';
  const CURRENT_STEP_KEY = 'registration_current_step';
  const TOTAL_STEPS = 4;

  // फॉर्म का initial state
  const initialFormData = {
    // Step 1: Personal Info
    firstName: '',
    lastName: '',
    dateOfBirth: '',
    gender: '',
    
    // Step 2: Contact Info
    email: '',
    phone: '',
    address: '',
    city: '',
    pincode: '',
    
    // Step 3: Account Info
    username: '',
    password: '',
    confirmPassword: '',
    
    // Step 4: Additional Info
    occupation: '',
    qualification: '',
    interests: [],
    agreeToTerms: false
  };

  // ============================================
  // localStorage Utility Functions
  // ============================================

  /**
   * localStorage में डेटा सेव करना
   * @param {Object} data - सेव करने का डेटा
   */
  function saveToLocalStorage(data) {
    try {
      const serializedData = JSON.stringify({
        ...data,
        lastUpdated: new Date().toISOString()
      });
      localStorage.setItem(STORAGE_KEY, serializedData);
      console.log('✅ डेटा localStorage में सेव हो गया:', data);
      return true;
    } catch (error) {
      console.error('❌ localStorage में सेव करने में error:', error);
      return false;
    }
  }

  /**
   * localStorage से डेटा लोड करना
   * @returns {Object} - सेव किया हुआ डेटा
   */
  function loadFromLocalStorage() {
    try {
      const serializedData = localStorage.getItem(STORAGE_KEY);
      if (serializedData) {
        const data = JSON.parse(serializedData);
        console.log('📂 localStorage से डेटा लोड हुआ:', data);
        return data;
      }
      return { ...initialFormData };
    } catch (error) {
      console.error('❌ localStorage से लोड करने में error:', error);
      return { ...initialFormData };
    }
  }

  /**
   * current step localStorage में सेव करना
   * @param {number} step - current step number
   */
  function saveCurrentStep(step) {
    try {
      localStorage.setItem(CURRENT_STEP_KEY, step.toString());
      console.log(`📍 Current Step ${step} सेव हो गया`);
    } catch (error) {
      console.error('❌ Step सेव करने में error:', error);
    }
  }

  /**
   * localStorage से current step लोड करना
   * @returns {number} - current step number
   */
  function loadCurrentStep() {
    try {
      const step = localStorage.getItem(CURRENT_STEP_KEY);
      return step ? parseInt(step, 10) : 1;
    } catch (error) {
      console.error('❌ Step लोड करने में error:', error);
      return 1;
    }
  }

  /**
   * localStorage से registration data clear करना (submit के बाद)
   */
  function clearLocalStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
      console.log('🗑️ localStorage से registration data clear हो गया');
    } catch (error) {
      console.error('❌ localStorage clear करने में error:', error);
    }
  }

  // ============================================
  // Validation Functions
  // ============================================

  /**
   * Step 1 validation: Personal Info
   * @param {Object} data - form data
   * @returns {Object} - {isValid: boolean, errors: Object}
   */
  function validateStep1(data) {
    const errors = {};

    if (!data.firstName || data.firstName.trim().length < 2) {
      errors.firstName = 'पहला नाम कम से कम 2 अक्षर का होना चाहिए';
    }

    if (!data.lastName || data.lastName.trim().length < 2) {
      errors.lastName = 'अंतिम नाम कम से कम 2 अक्षर का होना चाहिए';
    }

    if (!data.dateOfBirth) {
      errors.dateOfBirth = 'जन्म तिथि जरूरी है';
    } else {
      const birthDate = new Date(data.dateOfBirth);
      const today = new Date();
      const age = today.getFullYear() - birthDate.getFullYear();
      if (age < 18) {
        errors.dateOfBirth = 'आपकी उम्र कम से कम 18 साल होनी चाहिए';
      }
    }

    if (!data.gender) {
      errors.gender = 'लिंग चुनना जरूरी है';
    }

    return {
      isValid: Object.keys(errors).length === 0,
      errors
    };
  }

  /**
   * Step 2 validation: Contact Info
   * @param {Object} data - form data
   * @returns {Object} - {isValid: boolean, errors: Object}
   */
  function validateStep2(data) {
    const errors = {};
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    const phoneRegex = /^[6-9]\d{9}$/;
    const pincodeRegex = /^\d{6}$/;

    if (!data.email || !emailRegex.test(data.email)) {
      errors.email = 'सही email address डालें';
    }

    if (!data.phone || !phoneRegex.test(data.phone)) {
      errors.phone = 'सही 10 अंकों का मोबाइल नंबर डालें';
    }

    if (!data.address || data.address.trim().length < 10) {
      errors.address = 'पूरा पता कम से कम 10 अक्षर का होना चाहिए';
    }

    if (!data.city || data.city.trim().length < 2) {
      errors.city = 'शहर का नाम जरूरी है';
    }

    if (!data.pincode || !pincodeRegex.test(data.pincode)) {
      errors.pincode = 'सही 6 अंकों का पिनकोड डालें';
    }

    return {
      isValid: Object.keys(errors).length === 0,
      errors
    };
  }

  /**
   * Step 3 validation: Account Info
   * @param {Object} data - form data
   * @returns {Object} - {isValid: boolean, errors: Object}
   */
  function validateStep3(data) {
    const errors = {};
    const usernameRegex = /^[a-zA-Z0-9_]{4,20}$/;
    const passwordRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$/;

    if (!data.username || !usernameRegex.test(data.username)) {
      errors.username = 'Username 4-20 अक्षर का हो, केवल letters, numbers और _ allowed';
    }

    if (!data.password || !passwordRegex.test(data.password)) {
      errors.password = 'Password में कम से कम 8 अक्षर, 1 uppercase, 1 lowercase, 1 number और 1 special character होना चाहिए';
    }

    if (!data.confirmPassword || data.password !== data.confirmPassword) {
      errors.confirmPassword = 'Password और Confirm Password match नहीं हो रहे';
    }

    return {
      isValid: Object.keys(errors).length === 0,
      errors
    };
  }

  /**
   * Step 4 validation: Additional Info
   * @param {Object} data - form data
   * @returns {Object} - {isValid: boolean, errors: Object}
   */
  function validateStep4(data) {
    const errors = {};

    if (!data.occupation || data.occupation.trim().length < 2) {
      errors.occupation = 'व्यवसाय जरूरी है';
    }

    if (!data.qualification) {
      errors.qualification = 'शैक्षिक योग्यता चुनना जरूरी है';
    }

    if (!data.interests || data.interests.length === 0) {
      errors.interests = 'कम से कम एक रुचि चुनें';
    }

    if (!data.agreeToTerms) {
      errors.agreeToTerms = 'Terms & Conditions से सहमत होना जरूरी है';
    }

    return {
      isValid: Object.keys(errors).length === 0,
      errors
    };
  }

  /**
   * Step के हिसाब से validation करना
   * @param {number} step - step number
   * @param {Object} data - form data
   * @returns {Object} - validation result
   */
  function validateStep(step, data) {
    switch (step) {
      case 1: return validateStep1(data);
      case 2: return validateStep2(data);
      case 3: return validateStep3(data);
      case 4: return validateStep4(data);
      default: return { isValid: false, errors: { general: 'Invalid step' } };
    }
  }

  // ============================================
  // Form State Management
  // ============================================

  let currentStep = loadCurrentStep();
  let formData = loadFromLocalStorage();

  /**
   * Form field update करना और localStorage में सेव करना
   * @param {string} fieldName - field का नाम
   * @param {any} value - field की value
   */
  function updateField(fieldName, value) {
    formData[fieldName] = value;
    saveToLocalStorage(formData);
    
    // Real-time UI update
    updateFieldUI(fieldName, value);
    
    console.log(`📝 Field "${fieldName}" update हुआ:`, value);
  }

  /**
   * Multiple fields एक साथ update करना
   * @param {Object} fields - {fieldName: value} object
   */
  function updateMultipleFields(fields) {
    Object.entries(fields).forEach(([key, value]) => {
      formData[key] = value;
    });
    saveToLocalStorage(formData);
    console.log('📝 Multiple fields update हुए:', fields);
  }

  /**
   * Checkbox array field update करना (जैसे interests)
   * @param {string} fieldName - array field का नाम
   * @param {string} value - add/remove करने की value
   * @param {boolean} isChecked - checked है या नहीं
   */
  function updateArrayField(fieldName, value, isChecked) {
    if (!Array.isArray(formData[fieldName])) {
      formData[fieldName] = [];
    }

    if (isChecked) {
      if (!formData[fieldName].includes(value)) {
        formData[fieldName] = [...formData[fieldName], value];
      }
    } else {
      formData[fieldName] = formData[fieldName].filter(item => item !== value);
    }

    saveToLocalStorage(formData);
    console.log(`📝 Array field "${fieldName}" update हुआ:`, formData[fieldName]);
  }

  // ============================================
  // Step Navigation Functions
  // ============================================

  /**
   * अगले step पर जाना
   * @returns {Object} - {success: boolean, errors: Object, nextStep: number}
   */
  function goToNextStep() {
    const validation = validateStep(currentStep, formData);

    if (!validation.isValid) {
      console.warn(`⚠️ Step ${currentStep} validation failed:`, validation.errors);
      displayErrors(validation.errors);
      return {
        success: false,
        errors: validation.errors,
        currentStep
      };
    }

    if (currentStep < TOTAL_STEPS) {
      currentStep++;
      saveCurrentStep(currentStep);
      clearErrors();
      renderStep(currentStep);
      
      console.log(`➡️ Step ${currentStep - 1} से Step ${currentStep} पर गए`);
      
      return {
        success: true,
        errors: {},
        currentStep
      };
    }

    return {
      success: false,
      errors: { general: 'Already on last step' },
      currentStep
    };
  }

  /**
   * पिछले step पर जाना
   * @returns {Object} - {success: boolean, previousStep: number}
   */
  function goToPreviousStep() {
    if (currentStep > 1) {
      currentStep--;
      saveCurrentStep(currentStep);
      clearErrors();
      renderStep(currentStep);
      
      console.log(`⬅️ Step ${currentStep + 1} से Step ${currentStep} पर वापस गए`);
      
      return {
        success: true,
        currentStep
      };
    }

    return {
      success: false,
      currentStep
    };
  }

  /**
   * किसी specific step पर जाना (केवल completed steps के लिए)
   * @param {number} targetStep - जाने वाला step
   * @returns {boolean} - success या failure
   */
  function goToStep(targetStep) {
    if (targetStep < 1 || targetStep > TOTAL_STEPS) {
      console.error('❌ Invalid step number');
      return false;
    }

    // पिछले steps validate करना
    for (let step = 1; step < targetStep; step++) {
      const validation = validateStep(step, formData);
      if (!validation.isValid) {
        console.warn(`⚠️ Step ${step} complete नहीं है, पहले उसे complete करें`);
        return false;
      }
    }

    currentStep = targetStep;
    saveCurrentStep(currentStep);
    clearErrors();
    renderStep(currentStep);
    
    console.log(`🎯 Step ${targetStep} पर गए`);
    return true;
  }

  // ============================================
  // Form Submission
  // ============================================

  /**
   * Final form submit करना
   * @returns {Promise} - submission result
   */
  async function submitForm() {
    // Final validation सभी steps की
    const allValidations = [];
    for (let step = 1; step <= TOTAL_STEPS; step++) {
      const validation = validateStep(step, formData);
      if (!validation.isValid) {
        allValidations.push({ step, errors: validation.errors });
      }
    }

    if (allValidations.length > 0) {
      console.error('❌ Form में errors हैं:', allValidations);
      return {
        success: false,
        message: 'Form में errors हैं, कृपया सभी fields सही से भरें',
        validationErrors: allValidations
      };
    }

    try {
      console.log('🚀 Form submit हो रहा है...');
      
      // Password को final data से हटाना (security के लिए)
      const submitData = { ...formData };
      delete submitData.confirmPassword;
      delete submitData.lastUpdated;

      // API call simulate करना
      const result = await simulateAPICall(submitData);

      if (result.success) {
        // Submit के बाद localStorage clear करना
        clearLocalStorage();
        
        // Form reset करना
        formData = { ...initialFormData };
        currentStep = 1;
        
        console.log('✅ Form successfully submit हो गया!');
        showSuccessMessage(result);
        
        return {
          success: true,
          message: 'Registration सफलतापूर्वक हो गई!',
          userId: result.userId
        };
      }

      return {
        success: false,
        message: result.message || 'Submit करने में error आई'
      };

    } catch (error) {
      console.error('❌ Form submit करने में error:', error);
      return {
        success: false,
        message: 'Server error, कृपया बाद में try करें',
        error: error.message
      };
    }
  }

  /**
   * API call simulate करना
   * @param {Object} data - submit data
   * @returns {Promise} - API response
   */
  function simulateAPICall(data) {
    return new Promise((resolve, reject) => {
      console.log('📡 API को data भेजा जा रहा है:', data);
      
      setTimeout(() => {
        // 90% success rate simulate करना
        if (Math.random() > 0.1) {
          resolve({
            success: true,
            userId: 'USER_' + Date.now(),
            message: 'Registration successful'
          });
        } else {
          reject(new Error('Network error'));
        }
      }, 2000);
    });
  }

  // ============================================
  // UI Rendering Functions
  // ============================================

  /**
   * Step render करना
   * @param {number} step - render करने का step
   */
  function renderStep(step) {
    const formContainer = document.getElementById('registration-form');
    if (!formContainer) {
      console.warn('⚠️ Form container नहीं मिला');
      return;
    }

    updateProgressBar(step);
    updateStepIndicators(step);

    // सभी steps hide करना
    document.querySelectorAll('.form-step').forEach(el => {
      el.style.display = 'none';
      el.classList.remove('active');
    });

    // Current step show करना
    const currentStepEl = document.getElementById(`step-${step}`);
    if (currentStepEl) {
      currentStepEl.style.display = 'block';
      currentStepEl.classList.add('active');
    }

    // Saved data से fields populate करना
    populateFields(step);

    // Navigation buttons update करना
    updateNavigationButtons(step);
  }

  /**
   * Fields को saved data से populate करना
   * @param {number} step - current step
   */
  function populateFields(step) {
    const stepFields = getStepFields(step);
    
    stepFields.forEach(fieldName => {
      const element = document.getElementById(fieldName) || 
                      document.querySelector(`[name="${fieldName}"]`);
      
      if (!element) return;

      const value = formData[fieldName];

      if (element.type === 'checkbox') {
        if (Array.isArray(value)) {
          element.checked = value.includes(element.value);
        } else {
          element.checked = Boolean(value);
        }
      } else if (element.type === 'radio') {
        const radioButtons = document.querySelectorAll(`[name="${fieldName}"]`);
        radioButtons.forEach(radio => {
          radio.checked = radio.value === value;
        });
      } else {
        element.value = value || '';
      }
    });
  }

  /**
   * Step के fields की list
   * @param {number} step - step number
   * @returns {Array} - field names
   */
  function getStepFields(step) {
    const stepFieldsMap = {
      1: ['firstName', 'lastName', 'dateOfBirth', 'gender'],
      2: ['email', 'phone', 'address', 'city', 'pincode'],
      3: ['username', 'password', 'confirmPassword'],
      4: ['occupation', 'qualification', 'interests', 'agreeToTerms']
    };
    return stepFieldsMap[step] || [];
  }

  /**
   * Progress bar update करना
   * @param {number} step - current step
   */
  function updateProgressBar(step) {
    const progressBar = document.getElementById('progress-bar');
    if (progressBar) {
      const percentage = ((step - 1) / (TOTAL_STEPS - 1)) * 100;
      progressBar.style.width = `${percentage}%`;
      progressBar.setAttribute('aria-valuenow', percentage);
    }

    const progressText = document.getElementById('progress-text');
    if (progressText) {
      progressText.textContent = `Step ${step} of ${TOTAL_STEPS}`;
    }
  }

  /**
   * Step indicators update करना
   * @param {number} step - current step
   */
  function updateStepIndicators(step) {
    for (let i = 1; i <= TOTAL_STEPS; i++) {
      const indicator = document.getElementById(`step-indicator-${i}`);
      if (!indicator) continue;

      indicator.classList.remove('active', 'completed', 'pending');

      if (i < step) {
        indicator.classList.add('completed');
        indicator.innerHTML = '✓';
      } else if (i === step) {
        indicator.classList.add('active');
        indicator.innerHTML = i.toString();
      } else {
        indicator.classList.add('pending');
        indicator.innerHTML = i.toString();
      }
    }
  }

  /**
   * Navigation buttons update करना
   * @param {number} step - current step
   */
  function updateNavigationButtons(step) {
    const prevBtn = document.getElementById('prev-btn');
    const nextBtn = document.getElementById('next-btn');
    const submitBtn = document.getElementById('submit-btn');

    if (prevBtn) {
      prevBtn.style.display = step === 1 ? 'none' : 'block';
      prevBtn.disabled = step === 1;
    }

    if (nextBtn) {
      nextBtn.style.display = step === TOTAL_STEPS ? 'none' : 'block';
    }

    if (submitBtn) {
      submitBtn.style.display = step === TOTAL_STEPS ? 'block' : 'none';
    }
  }

  /**
   * Field UI update करना
   * @param {string} fieldName - field name
   * @param {any} value - new value
   */
  function updateFieldUI(fieldName, value) {
    const savedIndicator = document.getElementById(`${fieldName}-saved`);
    if (savedIndicator) {
      savedIndicator.textContent = '✓ Saved';
      savedIndicator.style.color = 'green';
      
      setTimeout(() => {
        savedIndicator.textContent = '';
      }, 2000);
    }
  }

  /**
   * Errors display करना
   * @param {Object} errors - error messages
   */
  function displayErrors(errors) {
    // पहले सभी errors clear करना
    clearErrors();

    Object.entries(errors).forEach(([fieldName, errorMessage]) => {
      const errorEl = document.getElementById(`${fieldName}-error`);
      if (errorEl) {
        errorEl.textContent = errorMessage;
        errorEl.style.display = 'block';
        errorEl.style.color = 'red';
      }

      const fieldEl = document.getElementById(fieldName) || 
                      document.querySelector(`[name="${fieldName}"]`);
      if (fieldEl) {
        fieldEl.classList.add('error');
        fieldEl.style.borderColor = 'red';
      }
    });
  }

  /**
   * सभी errors clear करना
   */
  function clearErrors() {
    document.querySelectorAll('.error-message').forEach(el => {
      el.textContent = '';
      el.style.display = 'none';
    });

    document.querySelectorAll('.error').forEach(el => {
      el.classList.remove('error');
      el.style.borderColor = '';
    });
  }

  /**
   * Success message show करना
   * @param {Object} result - API result
   */
  function showSuccessMessage(result) {
    const formContainer = document.getElementById('registration-form');
    if (formContainer) {
      formContainer.innerHTML = `
        <div class="success-message" style="text-align: center; padding: 40px;">
          <div style="font-size: 60px;">🎉</div>
          <h2 style="color: green;">Registration सफल!</h2>
          <p>आपकी User ID: <strong>${result.userId}</strong></p>
          <p>आपका registration सफलतापूर्वक हो गया है।</p>
          <button onclick="location.reload()">नया Registration</button>
        </div>
      `;
    }
  }

  // ============================================
  // Event Listeners Setup
  // ============================================

  /**
   * Form के event listeners setup करना
   */
  function setupEventListeners() {
    // Input fields के लिए auto-save
    document.addEventListener('input', (event) => {
      const { name, value, type, checked } = event.target;
      
      if (!name) return;

      if (type === 'checkbox') {
        if (name === 'interests') {
          updateArrayField('interests', value, checked);
        } else {
          updateField(name, checked);
        }
      } else {
        updateField(name, value);
      }
    });

    // Radio buttons के लिए
    document.addEventListener('change', (event) => {
      const { name, value, type } = event.target;
      
      if (type === 'radio' && name) {
        updateField(name, value);
      }
    });

    // Next button
    const nextBtn = document.getElementById('next-btn');
    if (nextBtn) {
      nextBtn.addEventListener('click', () => {
        const result = goToNextStep();
        if (!result.success) {
          console.log('Validation errors:', result.errors);
        }
      });
    }

    // Previous button
    const prevBtn = document.getElementById('prev-btn');
    if (prevBtn) {
      prevBtn.addEventListener('click', () => {
        goToPreviousStep();
      });
    }

    // Submit button
    const submitBtn = document.getElementById('submit-btn');
    if (submitBtn) {
      submitBtn.addEventListener('click', async () => {
        submitBtn.disabled = true;
        submitBtn.textContent = 'Submit हो रहा है...';
        
        const result = await submitForm();
        
        if (!result.success) {
          submitBtn.disabled = false;
          submitBtn.textContent = 'Submit';
          alert(result.message);
        }
      });
    }

    // Step indicators पर click
    for (let i = 1; i <= TOTAL_STEPS; i++) {
      const indicator = document.getElementById(`step-indicator-${i}`);
      if (indicator) {
        indicator.addEventListener('click', () => {
          goToStep(i);
        });
      }
    }

    // Page unload पर data save करना
    window.addEventListener('beforeunload', () => {
      saveToLocalStorage(formData);
      saveCurrentStep(currentStep);
    });
  }

  // ============================================
  // Initialization
  // ============================================

  /**
   * Form initialize करना
   */
  function init() {
    console.log('🚀 Registration Form initialize हो रहा है...');
    
    // Saved data check करना
    const savedData = loadFromLocalStorage();
    const savedStep = loadCurrentStep();
    
    if (savedData && Object.keys(savedData).some(key => savedData[key] && key !== 'lastUpdated')) {
      formData = { ...initialFormData, ...savedData };
      currentStep = savedStep;
      
      // User को inform करना
      const resumeMessage = document.getElementById('resume-message');
      if (resumeMessage) {
        resumeMessage.style.display = 'block';
        resumeMessage.textContent = `आपका पिछला data मिला! Step ${currentStep} से continue करें।`;
      }
      
      console.log(`📋 पिछला data मिला, Step ${currentStep} से शुरू हो रहे हैं`);
    } else {
      formData = { ...initialFormData };
      currentStep = 1;
      console.log('📋 नया registration शुरू हो रहा है');
    }

    setupEventListeners();
    renderStep(currentStep);
    
    console.log('✅ Registration Form ready है!');
  }

  // ============================================
  // Public API
  // ============================================

  return {
    init,
    goToNextStep,
    goToPreviousStep,
    goToStep,
    updateField,
    updateMultipleFields,
    updateArrayField,
    submitForm,
    
    // Debug के लिए
    getCurrentData: () => ({ ...formData }),
    getCurrentStep: () => currentStep,
    clearSavedData: clearLocalStorage,
    
    // Manual save
    saveProgress: () => {
      saveToLocalStorage(formData);
      saveCurrentStep(currentStep);
      return { success: true, message: 'Progress saved!' };
    }
  };
})();
