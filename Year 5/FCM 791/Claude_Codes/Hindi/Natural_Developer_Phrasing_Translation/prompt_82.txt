
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");
const app = express();

// Initialize Anthropic client
const anthropic = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Middleware to capture raw body
app.use((req, res, next) => {
  let rawBody = "";

  req.on("data", (chunk) => {
    rawBody += chunk.toString();
  });

  req.on("end", () => {
    req.rawBody = rawBody;
    next();
  });
});

// Parse JSON body
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Advanced logging middleware with AI-powered analysis
async function advancedLoggingMiddleware(req, res, next) {
  const timestamp = new Date().toISOString();
  const requestId = Math.random().toString(36).substr(2, 9);

  // Capture request details
  const requestDetails = {
    requestId,
    timestamp,
    method: req.method,
    url: req.url,
    path: req.path,
    query: req.query,
    headers: req.headers,
    body: req.body,
    rawBody: req.rawBody,
    ip: req.ip || req.connection.remoteAddress,
    protocol: req.protocol,
    hostname: req.hostname,
  };

  // Print detailed request log
  console.log("\n" + "=".repeat(80));
  console.log(`🔍 REQUEST LOG [${requestId}] - ${timestamp}`);
  console.log("=".repeat(80));
  console.log(`📌 Method: ${requestDetails.method}`);
  console.log(`🌐 URL: ${requestDetails.url}`);
  console.log(`📍 Path: ${requestDetails.path}`);
  console.log(`🔗 Protocol: ${requestDetails.protocol}`);
  console.log(`💻 Hostname: ${requestDetails.hostname}`);
  console.log(`🌍 IP: ${requestDetails.ip}`);

  if (Object.keys(requestDetails.query).length > 0) {
    console.log(`❓ Query Parameters:`);
    console.log(JSON.stringify(requestDetails.query, null, 2));
  }

  console.log(`📋 Headers:`);
  console.log(JSON.stringify(requestDetails.headers, null, 2));

  if (requestDetails.rawBody) {
    console.log(`📦 Raw Body:`);
    console.log(requestDetails.rawBody);

    if (requestDetails.body && Object.keys(requestDetails.body).length > 0) {
      console.log(`📝 Parsed Body:`);
      console.log(JSON.stringify(requestDetails.body, null, 2));
    }
  }

  // Use Claude for AI-powered request analysis with streaming
  console.log("\n🤖 AI Analysis (Streaming):");
  console.log("-".repeat(40));

  try {
    // Add request to conversation history
    const userMessage = `Analyze this HTTP request for debugging purposes:
    
Method: ${requestDetails.method}
URL: ${requestDetails.url}
Headers: ${JSON.stringify(requestDetails.headers, null, 2)}
Body: ${requestDetails.rawBody || "No body"}
Query Params: ${JSON.stringify(requestDetails.query, null, 2)}

Please provide:
1. A brief summary of what this request is doing
2. Any potential security concerns
3. Debugging insights
4. Suggestions for improvement if any`;

    conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    // Stream the response from Claude
    const stream = await anthropic.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system:
        "You are an expert Node.js developer and security analyst. Analyze HTTP requests and provide concise, helpful debugging insights. Be specific and practical in your analysis.",
      messages: conversationHistory,
    });

    let fullResponse = "";

    // Handle streaming events
    stream.on("text", (text) => {
      process.stdout.write(text);
      fullResponse += text;
    });

    // Wait for stream to complete
    await stream.finalMessage();

    // Add assistant response to conversation history
    conversationHistory.push({
      role: "assistant",
      content: fullResponse,
    });

    console.log("\n" + "-".repeat(40));
  } catch (error) {
    console.error("Error during AI analysis:", error.message);
  }

  // Track response
  const originalSend = res.send;
  res.send = function (body) {
    console.log(`\n📤 Response Status: ${res.statusCode}`);
    console.log(`📤 Response Body Preview: ${String(body).substring(0, 200)}`);
    console.log("=".repeat(80) + "\n");
    originalSend.call(this, body);
  };

  next();
}

// Apply the advanced logging middleware
app.use(advancedLoggingMiddleware);

// Test routes
app.get("/", (req, res) => {
  res.json({ message: "Welcome to the logging middleware demo!", status: "ok" });
});

app.get("/users", (req, res) => {
  res.json({
    users: [
      { id: 1, name: "Alice", email: "alice@example.com" },
      { id: 2, name: "Bob", email: "bob@example.com" },
    ],
  });
});

app.post("/users", (req, res) => {
  const { name, email } = req.body;
  res.status(201).json({
    message: "User created successfully",
    user: { id: 3, name, email },
  });
});

app.post("/login", (req, res) => {
  const { username, password } = req.body;
  // Simulate login (never do this in production!)
  if (username === "admin" && password === "password123") {
    res.json({ token: "fake-jwt-token-12345", message: "Login successful" });
  } else {
    res.status(401).json({ error: "Invalid credentials" });
  }
});

app.get("/api/data", (req, res) => {
  res.json({
    data: [1, 2, 3, 4, 5],
    metadata: {
      total: 5,
      page: req.query.page || 1,
    },
  });
});

// Interactive conversation endpoint
app.post("/analyze", async (req, res) => {
  const { question } = req.body;

  if (!question) {
    return res.status(400).json({ error: "Question is required" });
  }

  try {
    conversationHistory.push({
      role: "user",
      content: question,
    });

    const stream = await anthropic.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system:
        "You are an expert Node.js developer helping with request analysis and debugging.",
      messages: conversationHistory,
    });

    let fullResponse = "";

    stream.on("text", (text) => {
      fullResponse += text;
    });

    await stream.finalMessage();

    conversationHistory.push({
      role: "assistant",
      content: fullResponse,
    });

    res.json({ response: fullResponse });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`\n🚀 Server running on port ${PORT}`);
  console.log(`📊 Advanced logging middleware with AI analysis is active`);
  console.log(`\nTest endpoints:`);
  console.log(`  GET  http://localhost:${PORT}/`);
  console.log(`  GET  http://localhost:${PORT}/users`);
  console.log(`  POST http://localhost:${PORT}/users`);
  console.log(`  POST http://localhost:${PORT}/login`);
  console.log(`  GET  http://localhost:${PORT}/api/data`);
  console.log(`  POST http://localhost:${PORT}/analyze`);
  console.log("\n" + "=".repeat(80));
});

module.exports = app;
```
