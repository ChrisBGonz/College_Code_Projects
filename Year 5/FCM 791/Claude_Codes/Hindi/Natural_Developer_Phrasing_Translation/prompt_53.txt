// Helper function to send error response
function sendErrorResponse(res, statusCode, message, details = null) {
  res.writeHead(statusCode, { 'Content-Type': 'application/json' });
  const errorResponse = {
    success: false,
    error: {
      code: statusCode,
      message: message,
      details: details,
      timestamp: new Date().toISOString()
    }
  };
  res.end(JSON.stringify(errorResponse, null, 2));
}

// Helper function to send success response
function sendSuccessResponse(res, data, message = 'Success') {
  res.writeHead(200, { 'Content-Type': 'application/json' });
  const successResponse = {
    success: true,
    message: message,
    data: data,
    timestamp: new Date().toISOString()
  };
  res.end(JSON.stringify(successResponse, null, 2));
}

// Function to fetch user profile
function getUserProfile(userId) {
  return usersDatabase[userId] || null;
}

// Function to get all users list (without sensitive data)
function getAllUsers() {
  return Object.values(usersDatabase).map(user => ({
    id: user.id,
    name: user.name,
    email: user.email,
    profession: user.profession,
    accountStatus: user.accountStatus
  }));
}

// Main request handler
function handleRequest(req, res) {
  // Parse URL
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  // Add CORS headers
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'GET, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

  // Handle OPTIONS request (CORS preflight)
  if (req.method === 'OPTIONS') {
    res.writeHead(200);
    res.end();
    return;
  }

  // Only allow GET requests
  if (req.method !== 'GET') {
    sendErrorResponse(res, 405, 'Method Not Allowed', 'Only GET requests are supported');
    return;
  }

  console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);

  // Route: Home page - API documentation
  if (pathname === '/' || pathname === '/api') {
    const apiDocs = {
      name: 'User Profile API',
      version: '1.0.0',
      description: 'User profile data fetch aur download karne ke liye API',
      endpoints: [
        {
          path: '/api/user/profile',
          method: 'GET',
          description: 'User ka profile data fetch karo',
          queryParams: {
            user_id: 'Required - User ka unique ID',
            format: 'Optional - Response format (json/download), default: json'
          },
          examples: [
            '/api/user/profile?user_id=1',
            '/api/user/profile?user_id=1&format=download',
            '/api/user/profile?user_id=2&format=json'
          ]
        },
        {
          path: '/api/users',
          method: 'GET',
          description: 'Sabhi users ki list dekho',
          queryParams: {},
          examples: ['/api/users']
        },
        {
          path: '/api/user/download',
          method: 'GET',
          description: 'User profile ko JSON file ke roop mein download karo',
          queryParams: {
            user_id: 'Required - User ka unique ID'
          },
          examples: ['/api/user/download?user_id=1']
        }
      ],
      availableUserIds: Object.keys(usersDatabase)
    };
    sendSuccessResponse(res, apiDocs, 'API Documentation');
    return;
  }

  // Route: Get all users list
  if (pathname === '/api/users') {
    const users = getAllUsers();
    sendSuccessResponse(res, {
      total: users.length,
      users: users
    }, 'Sabhi users ki list');
    return;
  }

  // Route: Get user profile (JSON format)
  if (pathname === '/api/user/profile') {
    const userId = query.user_id;
    const format = query.format || 'json';

    // Validate user_id
    if (!userId) {
      sendErrorResponse(
        res,
        400,
        'Bad Request',
        'user_id query parameter required hai. Example: /api/user/profile?user_id=1'
      );
      return;
    }

    // Validate user_id format (should be alphanumeric)
    if (!/^[a-zA-Z0-9_-]+$/.test(userId)) {
      sendErrorResponse(
        res,
        400,
        'Invalid user_id',
        'user_id mein sirf letters, numbers, underscore aur hyphen allowed hain'
      );
      return;
    }

    // Fetch user profile
    const userProfile = getUserProfile(userId);

    if (!userProfile) {
      sendErrorResponse(
        res,
        404,
        'User Not Found',
        `user_id '${userId}' ke liye koi user nahi mila. Available IDs: ${Object.keys(usersDatabase).join(', ')}`
      );
      return;
    }

    // If format is 'download', serve as downloadable file
    if (format === 'download') {
      const fileName = `user_profile_${userId}_${Date.now()}.json`;
      const jsonContent = JSON.stringify({
        exportInfo: {
          exportedAt: new Date().toISOString(),
          exportedBy: 'User Profile API v1.0',
          userId: userId
        },
        profile: userProfile
      }, null, 2);

      res.writeHead(200, {
        'Content-Type': 'application/json',
        'Content-Disposition': `attachment; filename="${fileName}"`,
        'Content-Length': Buffer.byteLength(jsonContent),
        'Cache-Control': 'no-cache'
      });

      res.end(jsonContent);
      console.log(`[DOWNLOAD] User ${userId} ka profile download kiya gaya - File: ${fileName}`);
      return;
    }

    // Default: Return JSON response
    sendSuccessResponse(res, userProfile, `User ${userId} ka profile data`);
    return;
  }

  // Route: Direct download endpoint
  if (pathname === '/api/user/download') {
    const userId = query.user_id;

    // Validate user_id
    if (!userId) {
      sendErrorResponse(
        res,
        400,
        'Bad Request',
        'user_id query parameter required hai. Example: /api/user/download?user_id=1'
      );
      return;
    }

    // Validate user_id format
    if (!/^[a-zA-Z0-9_-]+$/.test(userId)) {
      sendErrorResponse(
        res,
        400,
        'Invalid user_id',
        'user_id mein sirf letters, numbers, underscore aur hyphen allowed hain'
      );
      return;
    }

    // Fetch user profile
    const userProfile = getUserProfile(userId);

    if (!userProfile) {
      sendErrorResponse(
        res,
        404,
        'User Not Found',
        `user_id '${userId}' ke liye koi user nahi mila. Available IDs: ${Object.keys(usersDatabase).join(', ')}`
      );
      return;
    }

    // Create downloadable JSON content
    const downloadData = {
      exportInfo: {
        title: 'User Profile Export',
        exportedAt: new Date().toISOString(),
        exportedBy: 'User Profile API v1.0',
        userId: userId,
        format: 'JSON'
      },
      profile: userProfile,
      metadata: {
        totalFields: Object.keys(userProfile).length,
        exportVersion: '1.0'
      }
    };

    const jsonContent = JSON.stringify(downloadData, null, 2);
    const fileName = `user_${userId}_profile_${new Date().toISOString().split('T')[0]}.json`;

    // Set headers for file download
    res.writeHead(200, {
      'Content-Type': 'application/octet-stream',
      'Content-Disposition': `attachment; filename="${fileName}"`,
      'Content-Length': Buffer.byteLength(jsonContent),
      'Cache-Control': 'no-cache, no-store, must-revalidate',
      'Pragma': 'no-cache',
      'Expires': '0'
    });

    res.end(jsonContent);
    console.log(`[DOWNLOAD] User ${userId} ka profile successfully download kiya gaya`);
    console.log(`[DOWNLOAD] File name: ${fileName}`);
    return;
  }

  // Route: Save profile to file on server (bonus feature)
  if (pathname === '/api/user/save') {
    const userId = query.user_id;

    if (!userId) {
      sendErrorResponse(res, 400, 'Bad Request', 'user_id required hai');
      return;
    }

    const userProfile = getUserProfile(userId);

    if (!userProfile) {
      sendErrorResponse(res, 404, 'User Not Found', `User ${userId} nahi mila`);
      return;
    }

    // Save to file
    const saveDir = path.join(__dirname, 'saved_profiles');
    
    // Create directory if not exists
    if (!fs.existsSync(saveDir)) {
      fs.mkdirSync(saveDir, { recursive: true });
    }

    const fileName = `user_${userId}_profile.json`;
    const filePath = path.join(saveDir, fileName);
    const fileContent = JSON.stringify({
      savedAt: new Date().toISOString(),
      profile: userProfile
    }, null, 2);

    fs.writeFile(filePath, fileContent, 'utf8', (err) => {
      if (err) {
        console.error('File save error:', err);
        sendErrorResponse(res, 500, 'Internal Server Error', 'File save karne mein error aaya');
        return;
      }

      sendSuccessResponse(res, {
        message: 'Profile successfully save ho gaya',
        fileName: fileName,
        filePath: filePath,
        fileSize: `${Buffer.byteLength(fileContent)} bytes`
      }, 'Profile saved successfully');

      console.log(`[SAVE] User ${userId} ka profile save kiya gaya: ${filePath}`);
    });

    return;
  }

  // 404 - Route not found
  sendErrorResponse(
    res,
    404,
    'Route Not Found',
    `'${pathname}' route exist nahi karta. Available routes: /, /api/users, /api/user/profile, /api/user/download, /api/user/save`
  );
}

// Create HTTP server
const PORT = process.env.PORT || 3000;
const server = http.createServer(handleRequest);

// Start server
server.listen(PORT, () => {
  console.log('='.repeat(60));
  console.log('  User Profile API Server Started!');
  console.log('='.repeat(60));
  console.log(`  Server URL: http://localhost:${PORT}`);
  console.log('');
  console.log('  Available Endpoints:');
  console.log(`  GET http://localhost:${PORT}/`);
  console.log(`  GET http://localhost:${PORT}/api/users`);
  console.log(`  GET http://localhost:${PORT}/api/user/profile?user_id=1`);
  console.log(`  GET http://localhost:${PORT}/api/user/profile?user_id=1&format=download`);
  console.log(`  GET http://localhost:${PORT}/api/user/download?user_id=1`);
  console.log(`  GET http://localhost:${PORT}/api/user/save?user_id=1`);
  console.log('');
  console.log('  Available User IDs: 1, 2, 3, 4, 5');
  console.log('='.repeat(60));
});

// Handle server errors
server.on('error', (err) => {
  if (err.code === 'EADDRINUSE') {
    console.error(`Error: Port ${PORT} already in use hai. Koi doosra port use karo.`);
  } else {
    console.error('Server error:', err);
  }
  process.exit(1);
});

// Graceful shutdown
process.on('SIGTERM', () => {
  console.log('\nServer band ho raha hai...');
  server.close(() => {
    console.log('Server successfully band ho gaya.');
    process.exit(0);
  });
});

process.on('SIGINT', () => {
  console.log('\nServer band ho raha hai (Ctrl+C)...');
  server.close(() => {
    console.log('Server successfully band ho gaya.');
    process.exit(0);
  });
});

module.exports = server;
