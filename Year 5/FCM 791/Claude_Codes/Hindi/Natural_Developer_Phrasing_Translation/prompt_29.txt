#!/usr/bin/env python3
"""
Java Maven Project Dependencies Auto-Updater
This script uses Claude AI to analyze and update Maven project dependencies to their latest versions.
"""

import subprocess
import sys
import os
import re
import json
from pathlib import Path
import anthropic

def read_pom_file(pom_path: str) -> str:
    """Read the pom.xml file content."""
    with open(pom_path, 'r', encoding='utf-8') as f:
        return f.read()

def write_pom_file(pom_path: str, content: str) -> None:
    """Write updated content to pom.xml file."""
    with open(pom_path, 'w', encoding='utf-8') as f:
        f.write(content)

def get_maven_dependency_updates(pom_path: str) -> str:
    """Run Maven versions plugin to get available updates."""
    try:
        result = subprocess.run(
            ['mvn', 'versions:display-dependency-updates', '-f', pom_path],
            capture_output=True,
            text=True,
            timeout=120
        )
        return result.stdout + result.stderr
    except subprocess.TimeoutExpired:
        return "Maven command timed out"
    except FileNotFoundError:
        return "Maven not found. Please install Maven and ensure it's in PATH"
    except Exception as e:
        return f"Error running Maven: {str(e)}"

def get_maven_plugin_updates(pom_path: str) -> str:
    """Run Maven versions plugin to get available plugin updates."""
    try:
        result = subprocess.run(
            ['mvn', 'versions:display-plugin-updates', '-f', pom_path],
            capture_output=True,
            text=True,
            timeout=120
        )
        return result.stdout + result.stderr
    except subprocess.TimeoutExpired:
        return "Maven command timed out"
    except FileNotFoundError:
        return "Maven not found"
    except Exception as e:
        return f"Error running Maven: {str(e)}"

def update_dependencies_with_claude(pom_content: str, maven_output: str, plugin_output: str) -> str:
    """Use Claude AI with streaming to analyze and update dependencies."""
    
    client = anthropic.Anthropic()
    
    prompt = f"""You are a Java Maven expert. I need you to update all dependencies in this pom.xml to their latest stable versions.

Here is the current pom.xml content:
```xml
{pom_content}
```

Here is the Maven versions plugin output showing available updates for dependencies:
```
{maven_output}
```

Here is the Maven versions plugin output showing available updates for plugins:
```
{plugin_output}
```

Please analyze the pom.xml and the available updates, then provide the COMPLETE updated pom.xml with ALL dependencies and plugins updated to their latest stable versions.

Important rules:
1. Update ALL dependencies to their latest stable versions (not SNAPSHOT or RC versions unless currently using them)
2. Update ALL plugins to their latest stable versions
3. Keep the same XML structure and formatting
4. Preserve all comments and other XML elements
5. If a dependency version is managed by a parent POM or BOM, note it but don't change it
6. Return ONLY the complete updated pom.xml content, nothing else
7. Make sure the XML is valid and well-formed

Return the complete updated pom.xml:"""

    print("\n🤖 Claude AI is analyzing your dependencies with streaming...\n")
    
    updated_pom = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=8096,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            updated_pom += text
    
    print("\n")
    
    # Extract XML content if wrapped in code blocks
    if "```xml" in updated_pom:
        xml_match = re.search(r'```xml\n(.*?)```', updated_pom, re.DOTALL)
        if xml_match:
            updated_pom = xml_match.group(1)
    elif "```" in updated_pom:
        xml_match = re.search(r'```\n(.*?)```', updated_pom, re.DOTALL)
        if xml_match:
            updated_pom = xml_match.group(1)
    
    return updated_pom.strip()

def get_dependency_summary(original_pom: str, updated_pom: str) -> str:
    """Use Claude AI to generate a summary of changes."""
    
    client = anthropic.Anthropic()
    
    prompt = f"""Compare these two pom.xml files and provide a concise summary of what dependencies and plugins were updated.

Original pom.xml:
```xml
{original_pom}
```

Updated pom.xml:
```xml
{updated_pom}
```

Provide a clear, formatted summary showing:
1. Dependencies updated (old version -> new version)
2. Plugins updated (old version -> new version)
3. Any dependencies that couldn't be updated and why
4. Total number of updates made

Format the output nicely with emojis and clear sections."""

    print("\n📊 Generating update summary...\n")
    
    summary = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            summary += text
    
    print("\n")
    return summary

def backup_pom_file(pom_path: str) -> str:
    """Create a backup of the original pom.xml."""
    backup_path = pom_path + ".backup"
    with open(pom_path, 'r') as original:
        with open(backup_path, 'w') as backup:
            backup.write(original.read())
    return backup_path

def validate_xml(content: str) -> bool:
    """Basic XML validation."""
    try:
        import xml.etree.ElementTree as ET
        ET.fromstring(content)
        return True
    except ET.ParseError as e:
        print(f"❌ XML validation error: {e}")
        return False

def main():
    print("🚀 Java Maven Dependencies Auto-Updater")
    print("=" * 50)
    
    # Get pom.xml path
    if len(sys.argv) > 1:
        pom_path = sys.argv[1]
    else:
        pom_path = "pom.xml"
    
    # Check if pom.xml exists
    if not os.path.exists(pom_path):
        print(f"❌ Error: {pom_path} not found!")
        print("Usage: python solution.py [path/to/pom.xml]")
        sys.exit(1)
    
    print(f"📁 Processing: {pom_path}")
    
    # Read original pom.xml
    print("\n📖 Reading pom.xml...")
    original_pom = read_pom_file(pom_path)
    print(f"✅ Read {len(original_pom)} characters")
    
    # Create backup
    print("\n💾 Creating backup...")
    backup_path = backup_pom_file(pom_path)
    print(f"✅ Backup created: {backup_path}")
    
    # Get Maven updates information
    print("\n🔍 Checking available dependency updates...")
    print("(This may take a moment as Maven downloads metadata...)")
    maven_dep_output = get_maven_dependency_updates(pom_path)
    
    print("\n🔍 Checking available plugin updates...")
    maven_plugin_output = get_maven_plugin_updates(pom_path)
    
    # If Maven is not available, use a placeholder
    if "Maven not found" in maven_dep_output or "Error running Maven" in maven_dep_output:
        print("⚠️  Maven not found or error occurred. Claude will analyze based on pom.xml content only.")
        maven_dep_output = "Maven not available - Claude will use knowledge of latest versions"
        maven_plugin_output = "Maven not available - Claude will use knowledge of latest versions"
    else:
        print("✅ Maven analysis complete")
    
    # Use Claude AI to update dependencies
    print("\n🤖 Sending to Claude AI for intelligent dependency updates...")
    updated_pom = update_dependencies_with_claude(original_pom, maven_dep_output, maven_plugin_output)
    
    # Validate the updated XML
    print("\n✅ Validating updated XML...")
    if not validate_xml(updated_pom):
        print("❌ Updated pom.xml has XML errors. Keeping original file.")
        print(f"📁 Original file preserved. Backup at: {backup_path}")
        sys.exit(1)
    
    print("✅ XML validation passed!")
    
    # Write updated pom.xml
    print("\n💾 Writing updated pom.xml...")
    write_pom_file(pom_path, updated_pom)
    print("✅ Updated pom.xml written successfully!")
    
    # Generate and display summary
    summary = get_dependency_summary(original_pom, updated_pom)
    
    print("\n" + "=" * 50)
    print("✅ DEPENDENCY UPDATE COMPLETE!")
    print("=" * 50)
    print(f"\n📁 Updated file: {pom_path}")
    print(f"💾 Backup file: {backup_path}")
    print("\n💡 Next steps:")
    print("   1. Review the changes in pom.xml")
    print("   2. Run: mvn clean compile")
    print("   3. Run: mvn test")
    print("   4. Fix any compatibility issues if tests fail")
    print("   5. If issues occur, restore backup: cp " + backup_path + " " + pom_path)

if __name__ == "__main__":
    main()
