const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateAnalyticsScript() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `एक JavaScript स्निपेट बनाओ जो किसी वेबपेज के <head> में थर्ड-पार्टी analytics स्क्रिप्ट (जैसे analytics.js) को dynamically include/load करे। 
        
        निम्नलिखित features शामिल करें:
        1. Script को dynamically create और append करना
        2. Script load होने पर callback function
        3. Error handling
        4. Multiple analytics providers support (Google Analytics, Facebook Pixel, etc.)
        5. Script already loaded है तो duplicate loading prevent करना
        
        Code में Hindi comments भी शामिल करें।`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";
  console.log("Claude का Response:\n");
  console.log(responseText);

  // Extract and execute the JavaScript code
  const codeMatch = responseText.match(/```javascript([\s\S]*?)```/);
  if (codeMatch) {
    console.log("\n\nExtracted JavaScript Code:\n");
    console.log(codeMatch[1]);

    // Demonstrate the concept with a working example
    console.log("\n\nWorking Example - Analytics Loader:\n");
    demonstrateAnalyticsLoader();
  }
}

function demonstrateAnalyticsLoader() {
  // यह function analytics script loading का demonstration करता है
  // (Browser environment simulate करने के लिए)

  // Analytics Loader Class
  class AnalyticsLoader {
    constructor() {
      // loaded scripts को track करने के लिए Set
      this.loadedScripts = new Set();
      // analytics providers की configuration
      this.providers = {
        googleAnalytics: {
          url: "https://www.google-analytics.com/analytics.js",
          id: "google-analytics",
        },
        facebookPixel: {
          url: "https://connect.facebook.net/en_US/fbevents.js",
          id: "facebook-pixel",
        },
        customAnalytics: {
          url: "https://example.com/analytics.js",
          id: "custom-analytics",
        },
      };
    }

    // Script को dynamically load करने का function
    loadScript(url, scriptId, callback, errorCallback) {
      // Check करें कि script पहले से load है या नहीं
      if (this.loadedScripts.has(scriptId)) {
        console.log(`✓ Script '${scriptId}' पहले से load है - duplicate skip`);
        if (callback) callback(null, { status: "already_loaded", id: scriptId });
        return;
      }

      console.log(`→ Script load हो रही है: ${scriptId}`);
      console.log(`  URL: ${url}`);

      // Script element create करें
      const script = {
        type: "text/javascript",
        src: url,
        id: scriptId,
        async: true,
        defer: false,
      };

      // Simulate script loading (browser में यह actual DOM manipulation होगा)
      setTimeout(() => {
        // Success simulation
        if (!url.includes("invalid")) {
          this.loadedScripts.add(scriptId);
          console.log(`✓ Script successfully loaded: ${scriptId}`);

          if (callback) {
            callback(null, {
              status: "loaded",
              id: scriptId,
              url: url,
              timestamp: new Date().toISOString(),
            });
          }
        } else {
          // Error simulation
          const error = new Error(`Script load failed: ${url}`);
          console.error(`✗ Error loading script: ${scriptId}`, error.message);

          if (errorCallback) {
            errorCallback(error, { id: scriptId, url: url });
          }
        }
      }, 100);

      return script;
    }

    // Google Analytics load करें
    loadGoogleAnalytics(trackingId, callback) {
      console.log(`\nGoogle Analytics load हो रही है (ID: ${trackingId})...`);

      const provider = this.providers.googleAnalytics;
      this.loadScript(
        provider.url,
        provider.id,
        (err, result) => {
          if (!err) {
            // GA initialization code
            console.log(`  GA initialized with tracking ID: ${trackingId}`);
            console.log(`  window.ga('create', '${trackingId}', 'auto')`);
            console.log(`  window.ga('send', 'pageview')`);
          }
          if (callback) callback(err, result);
        },
        (err) => {
          console.error("Google Analytics load failed:", err.message);
          if (callback) callback(err, null);
        }
      );
    }

    // Facebook Pixel load करें
    loadFacebookPixel(pixelId, callback) {
      console.log(`\nFacebook Pixel load हो रही है (ID: ${pixelId})...`);

      const provider = this.providers.facebookPixel;
      this.loadScript(
        provider.url,
        provider.id,
        (err, result) => {
          if (!err) {
            console.log(`  Facebook Pixel initialized with ID: ${pixelId}`);
            console.log(`  fbq('init', '${pixelId}')`);
            console.log(`  fbq('track', 'PageView')`);
          }
          if (callback) callback(err, result);
        },
        (err) => {
          console.error("Facebook Pixel load failed:", err.message);
          if (callback) callback(err, null);
        }
      );
    }

    // Custom analytics script load करें
    loadCustomAnalytics(url, config, callback) {
      console.log(`\nCustom Analytics load हो रही है...`);

      const scriptId = config.id || "custom-analytics-" + Date.now();
      this.loadScript(
        url,
        scriptId,
        (err, result) => {
          if (!err && config.initCode) {
            console.log(`  Custom analytics initialized`);
            console.log(`  Init code: ${config.initCode}`);
          }
          if (callback) callback(err, result);
        },
        (err) => {
          console.error("Custom analytics load failed:", err.message);
          if (callback) callback(err, null);
        }
      );
    }

    // सभी loaded scripts की list दिखाएं
    getLoadedScripts() {
      return Array.from(this.loadedScripts);
    }

    // Script unload करें
    unloadScript(scriptId) {
      if (this.loadedScripts.has(scriptId)) {
        this.loadedScripts.delete(scriptId);
        console.log(`Script unloaded: ${scriptId}`);
        return true;
      }
      return false;
    }
  }

  // AnalyticsLoader का instance बनाएं
  const loader = new AnalyticsLoader();

  console.log("=== Analytics Script Loader Demo ===\n");

  // Google Analytics load करें
  loader.loadGoogleAnalytics("UA-XXXXX-Y", (err, result) => {
    if (!err) {
      console.log("  Callback received:", JSON.stringify(result));
    }
  });

  // Facebook Pixel load करें
  loader.loadFacebookPixel("123456789", (err, result) => {
    if (!err) {
      console.log("  Callback received:", JSON.stringify(result));
    }
  });

  // Custom analytics load करें
  loader.loadCustomAnalytics(
    "https://example.com/analytics.js",
    {
      id: "my-custom-analytics",
      initCode: "customAnalytics.init({siteId: 'my-site'})",
    },
    (err, result) => {
      if (!err) {
        console.log("  Callback received:", JSON.stringify(result));
      }
    }
  );

  // Duplicate loading test
  setTimeout(() => {
    console.log("\n=== Duplicate Loading Test ===");
    loader.loadGoogleAnalytics("UA-XXXXX-Y", (err, result) => {
      console.log("  Duplicate attempt result:", JSON.stringify(result));
    });

    // Loaded scripts दिखाएं
    setTimeout(() => {
      console.log("\n=== Loaded Scripts ===");
      const scripts = loader.getLoadedScripts();
      scripts.forEach((script) => {
        console.log(`  ✓ ${script}`);
      });
      console.log(`\nTotal scripts loaded: ${scripts.length}`);
    }, 200);
  }, 500);
}

// Main function run करें
generateAnalyticsScript().catch(console.error);
