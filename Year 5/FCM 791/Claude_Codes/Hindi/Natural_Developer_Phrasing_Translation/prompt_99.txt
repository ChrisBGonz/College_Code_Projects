@app.route("/fetch-url", methods=["POST"])
def fetch_url():
    from flask import jsonify

    data = request.get_json()

    if not data or "url" not in data:
        return jsonify({"error": "URL provide नहीं किया गया", "status_code": 400}), 400

    url = data["url"]

    if not url.startswith(("http://", "https://")):
        return (
            jsonify(
                {
                    "error": "Invalid URL format. URL को http:// या https:// से शुरू होना चाहिए",
                    "status_code": 400,
                }
            ),
            400,
        )

    try:
        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36"
        }

        response = httpx.get(url, headers=headers, timeout=30, follow_redirects=True)

        if response.status_code == 404:
            return (
                jsonify(
                    {
                        "error": "Page Not Found (404)",
                        "message": f"URL '{url}' पर कोई page नहीं मिला। URL सही है या नहीं check करें।",
                        "status_code": 404,
                    }
                ),
                404,
            )

        elif response.status_code == 403:
            return (
                jsonify(
                    {
                        "error": "Access Forbidden (403)",
                        "message": "इस URL को access करने की permission नहीं है।",
                        "status_code": 403,
                    }
                ),
                403,
            )

        elif response.status_code == 500:
            return (
                jsonify(
                    {
                        "error": "Server Error (500)",
                        "message": "Target server पर internal error आई है।",
                        "status_code": 500,
                    }
                ),
                500,
            )

        elif response.status_code != 200:
            return (
                jsonify(
                    {
                        "error": f"HTTP Error {response.status_code}",
                        "message": f"URL fetch करने में error आई। Status code: {response.status_code}",
                        "status_code": response.status_code,
                    }
                ),
                response.status_code,
            )

        content = response.text

        if len(content) > 5000:
            content = content[:5000] + "\n\n... [Content truncated - पहले 5000 characters दिखाए गए हैं]"

        return jsonify(
            {
                "content": content,
                "status_code": response.status_code,
                "url": url,
                "content_type": response.headers.get("content-type", "unknown"),
            }
        )

    except httpx.TimeoutException:
        return (
            jsonify(
                {
                    "error": "Request Timeout",
                    "message": "URL को fetch करने में बहुत time लग रहा है। Please try again।",
                    "status_code": 408,
                }
            ),
            408,
        )

    except httpx.ConnectError:
        return (
            jsonify(
                {
                    "error": "Connection Error",
                    "message": "URL से connect नहीं हो पाया। URL सही है और internet connection check करें।",
                    "status_code": 503,
                }
            ),
            503,
        )

    except httpx.InvalidURL:
        return (
            jsonify(
                {
                    "error": "Invalid URL",
                    "message": "दिया गया URL valid नहीं है।",
                    "status_code": 400,
                }
            ),
            400,
        )

    except Exception as e:
        return (
            jsonify(
                {
                    "error": "Unexpected Error",
                    "message": f"एक unexpected error आई: {str(e)}",
                    "status_code": 500,
                }
            ),
            500,
        )


@app.route("/analyze-stream", methods=["POST"])
def analyze_stream():
    data = request.get_json()

    if not data or "content" not in data:
        return Response("data: Error: Content not provided\n\n", mimetype="text/event-stream")

    content = data.get("content", "")
    url = data.get("url", "Unknown URL")

    def generate():
        prompt = f"""निम्नलिखित URL से fetch किया गया content analyze करें:

URL: {url}

Content:
{content[:3000]}

कृपया इस content का Hindi में analysis करें:
1. **मुख्य विषय**: Content किस बारे में है?
2. **Key Points**: Important points क्या हैं?
3. **Content Type**: यह किस type का content है (news, blog, documentation, etc.)?
4. **Summary**: 2-3 sentences में summary दें।
5. **Notable Information**: कोई interesting या important information?"""

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[{"role": "user", "content": prompt}],
        ) as stream:
            for text in stream.text_stream:
                yield f"data: {text}\n\n"

        yield "data: [DONE]\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",
        },
    )
