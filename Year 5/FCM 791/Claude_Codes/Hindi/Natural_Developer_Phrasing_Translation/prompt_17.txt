
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const systemPrompt = `You are a helpful Node.js and MongoDB expert assistant. You help developers set up MongoDB connection strings for local development databases. 

When helping with MongoDB setup:
1. Always provide complete, working code examples
2. Include error handling
3. Explain each part of the connection string
4. Suggest best practices for local development
5. Include environment variable usage for security
6. Provide mongoose and native MongoDB driver examples when relevant

Common local MongoDB connection string format:
- Default: mongodb://localhost:27017/dbname
- With auth: mongodb://username:password@localhost:27017/dbname
- With options: mongodb://localhost:27017/dbname?authSource=admin

Always respond in Hindi when the user writes in Hindi, and in English when they write in English.`;

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main(): Promise<void> {
  console.log("MongoDB Connection Setup Assistant");
  console.log("==================================");
  console.log(
    "MongoDB connection string सेटअप करने में मदद के लिए यहाँ हूँ!"
  );
  console.log('बाहर निकलने के लिए "exit" टाइप करें\n');

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const initialMessage = await chat(
    "मुझे local dev DB के लिए Node.js स्क्रिप्ट में MongoDB connection string सेटअप करना है। कृपया एक complete example दिखाएं।"
  );
  console.log("Assistant:", initialMessage);
  console.log("\n" + "=".repeat(50) + "\n");

  const askQuestion = (): void => {
    rl.question("आप: ", async (userInput: string) => {
      const trimmedInput = userInput.trim();

      if (trimmedInput.toLowerCase() === "exit") {
        console.log("धन्यवाद! Happy coding!");
        rl.close();
        return;
      }

      if (!trimmedInput) {
        askQuestion();
        return;
      }

      try {
        const response = await chat(trimmedInput);
        console.log("\nAssistant:", response);
        console.log("\n" + "=".repeat(50) + "\n");
      } catch (error) {
        console.error("Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

main();
```
