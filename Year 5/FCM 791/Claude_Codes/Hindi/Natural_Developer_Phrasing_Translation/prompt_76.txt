import anthropic

client = anthropic.Anthropic()

def generate_cicd_pipeline_script():
    """Generate a CI/CD pipeline script for Docker image deployment using Claude API with streaming."""
    
    prompt = """Create a comprehensive Python script for CI/CD pipeline that:
1. Pulls a Docker image using 'docker pull'
2. Validates the image
3. Stops and removes existing container
4. Deploys the new image to production
5. Performs health checks
6. Includes rollback mechanism if deployment fails
7. Logs all operations with timestamps
8. Sends notifications on success/failure

The script should be production-ready with proper error handling, logging, and best practices.
Include environment variable configuration and make it configurable.
Write the complete, working Python script with all necessary imports and functions."""

    print("🚀 CI/CD Pipeline Script Generator")
    print("=" * 50)
    print("Claude API से CI/CD पाइपलाइन स्क्रिप्ट generate हो रही है...\n")
    
    generated_script = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=4000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            generated_script += text
    
    print("\n\n" + "=" * 50)
    print("✅ Script generation complete!")
    
    return generated_script

def save_script_to_file(script_content: str, filename: str = "cicd_pipeline.py"):
    """Save the generated script to a file."""
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(script_content)
    print(f"\n📁 Script saved to: {filename}")

def demonstrate_pipeline_concepts():
    """Demonstrate key CI/CD pipeline concepts with streaming."""
    
    concepts = [
        "Docker image pulling and validation",
        "Container lifecycle management", 
        "Health check implementation",
        "Rollback strategies"
    ]
    
    print("\n🎓 CI/CD Pipeline Concepts Explanation")
    print("=" * 50)
    
    for concept in concepts:
        print(f"\n📚 Explaining: {concept}")
        print("-" * 30)
        
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=500,
            messages=[
                {
                    "role": "user",
                    "content": f"Briefly explain '{concept}' in CI/CD pipeline context in 3-4 sentences. Be concise and practical."
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                print(text, end="", flush=True)
        
        print("\n")

def main():
    """Main function to orchestrate the CI/CD pipeline script generation."""
    
    print("🐳 Docker CI/CD Pipeline Script Generator")
    print("=" * 60)
    print("यह script Claude API का उपयोग करके एक complete CI/CD")
    print("pipeline script generate करेगी जो Docker images को")
    print("production में deploy करने के लिए उपयोग की जा सकती है।")
    print("=" * 60)
    print()
    
    # Generate the main CI/CD pipeline script
    generated_script = generate_cicd_pipeline_script()
    
    # Save the generated script
    save_script_to_file(generated_script, "generated_cicd_pipeline.py")
    
    # Demonstrate key concepts
    print("\n" + "=" * 60)
    print("क्या आप CI/CD pipeline के key concepts समझना चाहते हैं?")
    user_input = input("Enter 'yes' to continue or any other key to skip: ").strip().lower()
    
    if user_input == 'yes':
        demonstrate_pipeline_concepts()
    
    print("\n" + "=" * 60)
    print("🎉 CI/CD Pipeline Script Generation Complete!")
    print("\nNext Steps:")
    print("1. 'generated_cicd_pipeline.py' file को review करें")
    print("2. Environment variables को configure करें")
    print("3. अपने CI/CD system (Jenkins, GitLab CI, GitHub Actions) में integrate करें")
    print("4. Testing environment में पहले test करें")
    print("5. Production deployment के लिए proper permissions set करें")
    print("=" * 60)

if __name__ == "__main__":
    main()
