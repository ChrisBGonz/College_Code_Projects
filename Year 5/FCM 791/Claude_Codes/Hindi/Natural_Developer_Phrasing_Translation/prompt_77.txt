using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;

namespace ApplicationSettingsDemo
{
    [Serializable]
    public class ApplicationSettings
    {
        public string ApplicationName { get; set; }
        public string Version { get; set; }
        public string Theme { get; set; }
        public string Language { get; set; }
        public int FontSize { get; set; }
        public bool AutoSave { get; set; }
        public string LastOpenedFile { get; set; }
        public int MaxRecentFiles { get; set; }
        public bool ShowToolbar { get; set; }
        public bool ShowStatusBar { get; set; }
        public string DatabaseConnectionString { get; set; }
        public int SessionTimeout { get; set; }
        public DateTime LastModified { get; set; }

        public ApplicationSettings()
        {
            ApplicationName = "MyApplication";
            Version = "1.0.0";
            Theme = "Light";
            Language = "Hindi";
            FontSize = 12;
            AutoSave = true;
            LastOpenedFile = string.Empty;
            MaxRecentFiles = 10;
            ShowToolbar = true;
            ShowStatusBar = true;
            DatabaseConnectionString = "Server=localhost;Database=MyDB;";
            SessionTimeout = 30;
            LastModified = DateTime.Now;
        }
    }

    public class SettingsManager
    {
        private readonly string _settingsFilePath;

#pragma warning disable SYSLIB0011
        private readonly BinaryFormatter _formatter;
#pragma warning restore SYSLIB0011

        public SettingsManager(string settingsFilePath = "app_settings.bin")
        {
            _settingsFilePath = settingsFilePath;
#pragma warning disable SYSLIB0011
            _formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
        }

        public bool SaveSettings(ApplicationSettings settings)
        {
            try
            {
                settings.LastModified = DateTime.Now;
                
                using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write))
                {
#pragma warning disable SYSLIB0011
                    _formatter.Serialize(fileStream, settings);
#pragma warning restore SYSLIB0011
                }
                
                Console.WriteLine($"✅ Settings successfully saved to: {_settingsFilePath}");
                Console.WriteLine($"   File Size: {new FileInfo(_settingsFilePath).Length} bytes");
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"❌ Error saving settings: {ex.Message}");
                return false;
            }
        }

        public ApplicationSettings LoadSettings()
        {
            try
            {
                if (!File.Exists(_settingsFilePath))
                {
                    Console.WriteLine("⚠️  Settings file not found. Creating default settings...");
                    ApplicationSettings defaultSettings = new ApplicationSettings();
                    SaveSettings(defaultSettings);
                    return defaultSettings;
                }

                using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read))
                {
#pragma warning disable SYSLIB0011
                    ApplicationSettings settings = (ApplicationSettings)_formatter.Deserialize(fileStream);
#pragma warning restore SYSLIB0011
                    Console.WriteLine($"✅ Settings successfully loaded from: {_settingsFilePath}");
                    return settings;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"❌ Error loading settings: {ex.Message}");
                Console.WriteLine("   Returning default settings...");
                return new ApplicationSettings();
            }
        }

        public bool UpdateSetting<T>(string propertyName, T value)
        {
            try
            {
                ApplicationSettings settings = LoadSettings();
                
                var property = typeof(ApplicationSettings).GetProperty(propertyName);
                if (property == null)
                {
                    Console.WriteLine($"❌ Property '{propertyName}' not found in ApplicationSettings");
                    return false;
                }

                property.SetValue(settings, value);
                bool saved = SaveSettings(settings);
                
                if (saved)
                {
                    Console.WriteLine($"✅ Setting '{propertyName}' updated to: {value}");
                }
                
                return saved;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"❌ Error updating setting: {ex.Message}");
                return false;
            }
        }

        public bool DeleteSettings()
        {
            try
            {
                if (File.Exists(_settingsFilePath))
                {
                    File.Delete(_settingsFilePath);
                    Console.WriteLine($"✅ Settings file deleted: {_settingsFilePath}");
                    return true;
                }
                else
                {
                    Console.WriteLine("⚠️  Settings file does not exist");
                    return false;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"❌ Error deleting settings: {ex.Message}");
                return false;
            }
        }

        public bool BackupSettings(string backupPath = null)
        {
            try
            {
                if (!File.Exists(_settingsFilePath))
                {
                    Console.WriteLine("❌ No settings file to backup");
                    return false;
                }

                if (backupPath == null)
                {
                    backupPath = $"app_settings_backup_{DateTime.Now:yyyyMMdd_HHmmss}.bin";
                }

                File.Copy(_settingsFilePath, backupPath, true);
                Console.WriteLine($"✅ Settings backed up to: {backupPath}");
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"❌ Error backing up settings: {ex.Message}");
                return false;
            }
        }

        public void DisplaySettings(ApplicationSettings settings)
        {
            Console.WriteLine("\n" + new string('=', 50));
            Console.WriteLine("         APPLICATION SETTINGS");
            Console.WriteLine(new string('=', 50));
            Console.WriteLine($"  Application Name    : {settings.ApplicationName}");
            Console.WriteLine($"  Version             : {settings.Version}");
            Console.WriteLine($"  Theme               : {settings.Theme}");
            Console.WriteLine($"  Language            : {settings.Language}");
            Console.WriteLine($"  Font Size           : {settings.FontSize}");
            Console.WriteLine($"  Auto Save           : {settings.AutoSave}");
            Console.WriteLine($"  Last Opened File    : {(string.IsNullOrEmpty(settings.LastOpenedFile) ? "None" : settings.LastOpenedFile)}");
            Console.WriteLine($"  Max Recent Files    : {settings.MaxRecentFiles}");
            Console.WriteLine($"  Show Toolbar        : {settings.ShowToolbar}");
            Console.WriteLine($"  Show Status Bar     : {settings.ShowStatusBar}");
            Console.WriteLine($"  DB Connection       : {settings.DatabaseConnectionString}");
            Console.WriteLine($"  Session Timeout     : {settings.SessionTimeout} minutes");
            Console.WriteLine($"  Last Modified       : {settings.LastModified:dd/MM/yyyy HH:mm:ss}");
            Console.WriteLine(new string('=', 50) + "\n");
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("🚀 Application Settings Manager - BinaryFormatter Demo");
            Console.WriteLine(new string('-', 55));

            SettingsManager manager = new SettingsManager("my_app_settings.bin");

            Console.WriteLine("\n📝 Step 1: Creating and Saving Settings...");
            ApplicationSettings settings = new ApplicationSettings
            {
                ApplicationName = "मेरा एप्लिकेशन",
                Version = "2.5.1",
                Theme = "Dark",
                Language = "Hindi",
                FontSize = 14,
                AutoSave = true,
                LastOpenedFile = @"C:\Documents\project.txt",
                MaxRecentFiles = 15,
                ShowToolbar = true,
                ShowStatusBar = false,
                DatabaseConnectionString = "Server=192.168.1.1;Database=ProductionDB;User=admin;",
                SessionTimeout = 60
            };

            manager.SaveSettings(settings);

            Console.WriteLine("\n📖 Step 2: Loading Settings...");
            ApplicationSettings loadedSettings = manager.LoadSettings();
            manager.DisplaySettings(loadedSettings);

            Console.WriteLine("🔄 Step 3: Updating Individual Settings...");
            manager.UpdateSetting("Theme", "Blue");
            manager.UpdateSetting("FontSize", 16);
            manager.UpdateSetting("AutoSave", false);
            manager.UpdateSetting("MaxRecentFiles", 20);

            Console.WriteLine("\n📖 Step 4: Loading Updated Settings...");
            ApplicationSettings updatedSettings = manager.LoadSettings();
            manager.DisplaySettings(updatedSettings);

            Console.WriteLine("💾 Step 5: Creating Backup...");
            manager.BackupSettings();

            Console.WriteLine("\n🔄 Step 6: Testing Default Settings Creation...");
            SettingsManager newManager = new SettingsManager("new_app_settings.bin");
            ApplicationSettings defaultSettings = newManager.LoadSettings();
            newManager.DisplaySettings(defaultSettings);

            Console.WriteLine("🗑️  Step 7: Cleanup - Deleting Test Files...");
            manager.DeleteSettings();
            newManager.DeleteSettings();
            
            string[] backupFiles = Directory.GetFiles(".", "app_settings_backup_*.bin");
            foreach (string backupFile in backupFiles)
            {
                File.Delete(backupFile);
                Console.WriteLine($"✅ Deleted backup: {backupFile}");
            }

            Console.WriteLine("\n✨ Demo completed successfully!");
            Console.WriteLine("\nPress any key to exit...");
            Console.ReadKey();
        }
    }
}
