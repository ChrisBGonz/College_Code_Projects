import anthropic
import re


def log_search_query(query: str, log_file: str = "app.log") -> None:
    """Log a search query to the specified log file."""
    with open(log_file, "a", encoding="utf-8") as f:
        f.write(f"{query}\n")


def process_search_with_claude(user_query: str) -> str:
    """
    Process a search query using Claude with tool use.
    The function logs the search query and returns Claude's response.
    """
    client = anthropic.Anthropic()

    tools = [
        {
            "name": "log_search_query",
            "description": "Logs the user's search query to app.log file for record keeping and analytics",
            "input_schema": {
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",
                        "description": "The search query to be logged"
                    },
                    "log_file": {
                        "type": "string",
                        "description": "The log file name (default: app.log)",
                        "default": "app.log"
                    }
                },
                "required": ["query"]
            }
        }
    ]

    messages = [
        {
            "role": "user",
            "content": f"Please log this search query and then provide a helpful response about it: '{user_query}'"
        }
    ]

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )

    tool_results = []
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if tool_use_block and tool_use_block.name == "log_search_query":
            query_to_log = tool_use_block.input.get("query", user_query)
            log_file = tool_use_block.input.get("log_file", "app.log")
            
            log_search_query(query_to_log, log_file)
            print(f"✓ Search query logged to {log_file}: '{query_to_log}'")
            
            tool_results.append({
                "type": "tool_result",
                "tool_use_id": tool_use_block.id,
                "content": f"Successfully logged query '{query_to_log}' to {log_file}"
            })
        
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": tool_results})
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response = block.text
            break
    
    return final_response


def main():
    """Main function to demonstrate the search query logging functionality."""
    print("=== Search Query Logger with Claude ===\n")
    
    search_queries = [
        "Python programming best practices",
        "Machine learning algorithms",
        "Web development frameworks 2024"
    ]
    
    for query in search_queries:
        print(f"\nProcessing query: '{query}'")
        print("-" * 50)
        
        response = process_search_with_claude(query)
        print(f"Claude's response: {response[:200]}..." if len(response) > 200 else f"Claude's response: {response}")
        print("-" * 50)
    
    print("\n=== Checking app.log contents ===")
    try:
        with open("app.log", "r", encoding="utf-8") as f:
            content = f.read()
            print("Logged queries:")
            
            logged_queries = []
            for line in content.strip().split('\n'):
                if line.strip():
                    logged_queries.append(line.strip())
            
            for i, query in enumerate(logged_queries, 1):
                print(f"{i}. {query}")
    except FileNotFoundError:
        print("app.log file not found")


if __name__ == "__main__":
    main()
