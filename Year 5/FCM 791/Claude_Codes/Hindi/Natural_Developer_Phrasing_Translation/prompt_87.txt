import anthropic
import time
import threading
import os
import tempfile
from datetime import datetime
from collections import deque

# Initialize Anthropic client
client = anthropic.Anthropic()

# Thread-safe storage for failed login timestamps
failed_login_times = deque()
lock = threading.Lock()
warning_count = 0

def analyze_log_with_claude(log_content: str, failed_count: int) -> str:
    """Use Claude to analyze the security situation and provide recommendations."""
    
    prompt = f"""You are a security analyst. Analyze this log data and provide a security assessment.

Current situation:
- {failed_count} failed login attempts detected in the last 1 minute
- This exceeds the threshold of 10 attempts per minute

Recent log entries:
{log_content}

Please provide:
1. Security risk assessment (HIGH/MEDIUM/LOW)
2. Possible attack type (brute force, credential stuffing, etc.)
3. Immediate recommended actions
4. Long-term security recommendations

Keep your response concise and actionable."""

    print("\n🤖 Claude AI Security Analysis:")
    print("-" * 50)
    
    full_response = ""
    
    # Use streaming for real-time response
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[{"role": "user", "content": prompt}]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response

def check_failed_logins(log_line: str, recent_logs: list) -> None:
    """Check if a log line contains failed login and trigger warning if threshold exceeded."""
    global warning_count
    
    current_time = time.time()
    
    if "Failed Login" in log_line or "failed login" in log_line.lower():
        with lock:
            # Add current timestamp
            failed_login_times.append(current_time)
            
            # Remove timestamps older than 1 minute
            while failed_login_times and (current_time - failed_login_times[0]) > 60:
                failed_login_times.popleft()
            
            count = len(failed_login_times)
            
            if count > 10:
                warning_count += 1
                timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                
                print(f"\n{'='*60}")
                print(f"⚠️  SECURITY WARNING #{warning_count}")
                print(f"{'='*60}")
                print(f"🕐 Time: {timestamp}")
                print(f"🚨 Alert: {count} Failed Login attempts detected in the last 1 minute!")
                print(f"📊 Threshold exceeded: {count}/10 attempts")
                print(f"{'='*60}")
                
                # Get Claude's analysis
                log_content = "\n".join(recent_logs[-20:])  # Last 20 log entries
                analyze_log_with_claude(log_content, count)
                
                # Reset counter after warning to avoid spam
                # Clear old entries to prevent repeated warnings
                # Keep only the most recent entry
                latest = failed_login_times[-1]
                failed_login_times.clear()
                failed_login_times.append(latest)

def monitor_log_file(log_file_path: str) -> None:
    """Monitor a log file for failed login attempts."""
    print(f"🔍 Starting log monitor for: {log_file_path}")
    print(f"📋 Monitoring for 'Failed Login' messages")
    print(f"⚡ Alert threshold: >10 attempts per minute")
    print(f"{'='*60}\n")
    
    recent_logs = []
    
    # Open file and seek to end
    with open(log_file_path, 'r') as f:
        # Start from the beginning for demo purposes
        f.seek(0)
        
        while True:
            line = f.readline()
            
            if line:
                line = line.strip()
                if line:
                    recent_logs.append(line)
                    # Keep only last 100 lines
                    if len(recent_logs) > 100:
                        recent_logs.pop(0)
                    
                    # Print the log line
                    print(f"📝 {line}")
                    
                    # Check for failed logins
                    check_failed_logins(line, recent_logs)
            else:
                # No new content, wait a bit
                time.sleep(0.1)

def generate_test_logs(log_file_path: str) -> None:
    """Generate test log entries to simulate a brute force attack."""
    print("🔧 Starting log generator (simulating attack scenario)...")
    
    # Normal log entries
    normal_entries = [
        "2024-01-15 10:00:01 INFO User john.doe logged in successfully from 192.168.1.100",
        "2024-01-15 10:00:05 INFO User jane.smith accessed dashboard from 10.0.0.50",
        "2024-01-15 10:00:10 INFO System backup completed successfully",
        "2024-01-15 10:00:15 INFO User admin updated configuration settings",
    ]
    
    # Attack simulation entries
    attack_entries = [
        "2024-01-15 10:01:00 WARNING Failed Login attempt for user admin from 192.168.1.200",
        "2024-01-15 10:01:02 WARNING Failed Login attempt for user root from 192.168.1.200",
        "2024-01-15 10:01:04 WARNING Failed Login attempt for user administrator from 192.168.1.200",
        "2024-01-15 10:01:06 WARNING Failed Login attempt for user admin from 192.168.1.200",
        "2024-01-15 10:01:08 WARNING Failed Login attempt for user test from 192.168.1.200",
        "2024-01-15 10:01:10 WARNING Failed Login attempt for user user from 192.168.1.200",
        "2024-01-15 10:01:12 WARNING Failed Login attempt for user admin from 192.168.1.201",
        "2024-01-15 10:01:14 WARNING Failed Login attempt for user root from 192.168.1.201",
        "2024-01-15 10:01:16 WARNING Failed Login attempt for user oracle from 192.168.1.201",
        "2024-01-15 10:01:18 WARNING Failed Login attempt for user postgres from 192.168.1.201",
        "2024-01-15 10:01:20 WARNING Failed Login attempt for user admin from 192.168.1.202",
        "2024-01-15 10:01:22 WARNING Failed Login attempt for user root from 192.168.1.202",
    ]
    
    with open(log_file_path, 'w') as f:
        # Write normal entries first
        for entry in normal_entries:
            f.write(entry + "\n")
            f.flush()
            time.sleep(0.5)
        
        print("\n⚠️  Simulating brute force attack...")
        
        # Write attack entries rapidly
        for entry in attack_entries:
            f.write(entry + "\n")
            f.flush()
            time.sleep(0.3)  # Rapid succession to trigger threshold
        
        # Continue with some normal entries after attack
        time.sleep(2)
        f.write("2024-01-15 10:02:00 INFO Security scan initiated\n")
        f.flush()
        time.sleep(1)
        f.write("2024-01-15 10:02:05 INFO Firewall rules updated\n")
        f.flush()
        
        print("\n✅ Log generation complete")

def main():
    """Main function to run the log monitor."""
    # Create a temporary log file for testing
    with tempfile.NamedTemporaryFile(mode='w', suffix='.log', delete=False) as tmp_file:
        log_file_path = tmp_file.name
    
    print("🛡️  Log Security Monitor with AI Analysis")
    print("=" * 60)
    print(f"📁 Log file: {log_file_path}")
    print("=" * 60)
    
    try:
        # Start log generator in a separate thread
        generator_thread = threading.Thread(
            target=generate_test_logs,
            args=(log_file_path,),
            daemon=True
        )
        generator_thread.start()
        
        # Small delay to let generator create the file
        time.sleep(0.5)
        
        # Start monitoring (this will run until interrupted)
        monitor_thread = threading.Thread(
            target=monitor_log_file,
            args=(log_file_path,),
            daemon=True
        )
        monitor_thread.start()
        
        # Wait for generator to finish
        generator_thread.join()
        
        # Give monitor a bit more time to process remaining entries
        time.sleep(3)
        
        print("\n" + "=" * 60)
        print("📊 Monitoring Session Summary")
        print("=" * 60)
        print(f"✅ Monitoring completed")
        print(f"⚠️  Total warnings generated: {warning_count}")
        print(f"🔒 Security analysis provided by Claude AI")
        
    except KeyboardInterrupt:
        print("\n\n🛑 Monitoring stopped by user")
    finally:
        # Cleanup
        if os.path.exists(log_file_path):
            os.unlink(log_file_path)
            print(f"🗑️  Cleaned up temporary log file")

if __name__ == "__main__":
    main()
