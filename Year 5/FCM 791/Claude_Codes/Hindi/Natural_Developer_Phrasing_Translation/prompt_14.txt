from flask import Flask, request, session, jsonify, redirect, url_for
import secrets
import hashlib

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

# Simple user database (in production, use a real database)
USERS = {
    "admin": hashlib.sha256("password123".encode()).hexdigest(),
    "user1": hashlib.sha256("mypassword".encode()).hexdigest(),
    "testuser": hashlib.sha256("test123".encode()).hexdigest()
}

def generate_auth_token(username: str) -> str:
    """Generate a secure auth token for the user."""
    token_data = f"{username}:{secrets.token_hex(16)}"
    return hashlib.sha256(token_data.encode()).hexdigest()

@app.route('/')
def index():
    """Home page - check if user is logged in."""
    if 'auth_token' in session:
        return jsonify({
            "message": "Welcome! You are logged in.",
            "auth_token": session['auth_token'],
            "username": session.get('username', 'Unknown')
        })
    return jsonify({
        "message": "Please login to continue.",
        "login_url": "/login"
    })

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login endpoint."""
    if request.method == 'GET':
        return jsonify({
            "message": "Send POST request with username and password",
            "example": {
                "username": "admin",
                "password": "password123"
            }
        })
    
    # Handle POST request
    data = request.get_json() or request.form
    
    username = data.get('username', '').strip()
    password = data.get('password', '').strip()
    
    if not username or not password:
        return jsonify({
            "success": False,
            "message": "Username and password are required"
        }), 400
    
    # Verify credentials
    hashed_password = hashlib.sha256(password.encode()).hexdigest()
    
    if username in USERS and USERS[username] == hashed_password:
        # Generate auth token
        auth_token = generate_auth_token(username)
        
        # Set session cookies
        session['auth_token'] = auth_token
        session['username'] = username
        session.permanent = True  # Make session persistent
        
        return jsonify({
            "success": True,
            "message": f"Login successful! Welcome, {username}!",
            "auth_token": auth_token,
            "username": username
        })
    else:
        return jsonify({
            "success": False,
            "message": "Invalid username or password"
        }), 401

@app.route('/logout')
def logout():
    """Logout endpoint - clear session."""
    username = session.get('username', 'Unknown')
    
    # Remove auth_token and other session data
    session.pop('auth_token', None)
    session.pop('username', None)
    session.clear()
    
    return jsonify({
        "success": True,
        "message": f"Goodbye, {username}! You have been logged out."
    })

@app.route('/profile')
def profile():
    """Protected profile page - requires auth_token."""
    if 'auth_token' not in session:
        return jsonify({
            "success": False,
            "message": "Authentication required. Please login first.",
            "login_url": "/login"
        }), 401
    
    return jsonify({
        "success": True,
        "message": "Profile page accessed successfully",
        "username": session.get('username'),
        "auth_token": session.get('auth_token')
    })

@app.route('/check-session')
def check_session():
    """Check current session status."""
    if 'auth_token' in session:
        return jsonify({
            "logged_in": True,
            "username": session.get('username'),
            "auth_token": session.get('auth_token'),
            "message": "Active session found"
        })
    else:
        return jsonify({
            "logged_in": False,
            "message": "No active session"
        })

@app.route('/dashboard')
def dashboard():
    """Protected dashboard - requires valid auth_token."""
    auth_token = session.get('auth_token')
    
    if not auth_token:
        return jsonify({
            "success": False,
            "message": "Access denied. Please login to view dashboard.",
            "login_url": "/login"
        }), 403
    
    username = session.get('username')
    
    return jsonify({
        "success": True,
        "message": f"Welcome to your dashboard, {username}!",
        "data": {
            "username": username,
            "auth_token": auth_token,
            "features": ["View Reports", "Manage Settings", "User Management"]
        }
    })

if __name__ == '__main__':
    print("Flask Auth App Starting...")
    print("Available routes:")
    print("  GET  / - Home page")
    print("  GET  /login - Login form info")
    print("  POST /login - Login with credentials")
    print("  GET  /logout - Logout")
    print("  GET  /profile - Protected profile page")
    print("  GET  /dashboard - Protected dashboard")
    print("  GET  /check-session - Check session status")
    print("\nTest credentials:")
    print("  Username: admin, Password: password123")
    print("  Username: user1, Password: mypassword")
    print("  Username: testuser, Password: test123")
    
    app.run(debug=True, host='0.0.0.0', port=5000)
