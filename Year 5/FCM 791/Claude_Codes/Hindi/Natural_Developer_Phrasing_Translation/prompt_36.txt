import hashlib
import requests
import os
import sys
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        # Read file in chunks to handle large files efficiently
        for chunk in iter(lambda: f.read(8192), b""):
            sha256_hash.update(chunk)
    
    return sha256_hash.hexdigest()


def download_firmware(url: str, save_path: str) -> bool:
    """Download firmware file from URL."""
    print(f"Downloading firmware from: {url}")
    
    response = requests.get(url, stream=True, timeout=30)
    response.raise_for_status()
    
    total_size = int(response.headers.get('content-length', 0))
    downloaded_size = 0
    
    with open(save_path, 'wb') as f:
        for chunk in response.iter_content(chunk_size=8192):
            if chunk:
                f.write(chunk)
                downloaded_size += len(chunk)
                
                if total_size > 0:
                    progress = (downloaded_size / total_size) * 100
                    print(f"\rDownload progress: {progress:.1f}%", end="", flush=True)
    
    print(f"\nFirmware downloaded successfully: {save_path}")
    return True


def verify_firmware_integrity(file_path: str, expected_hash: str) -> dict:
    """
    Verify firmware file integrity using SHA-256 hash.
    
    Args:
        file_path: Path to the firmware file
        expected_hash: Expected SHA-256 hash value
    
    Returns:
        Dictionary with verification results
    """
    result = {
        "file_path": file_path,
        "file_exists": False,
        "file_size": 0,
        "calculated_hash": "",
        "expected_hash": expected_hash.lower(),
        "integrity_verified": False,
        "status": ""
    }
    
    # Check if file exists
    if not os.path.exists(file_path):
        result["status"] = f"Error: File not found at {file_path}"
        return result
    
    result["file_exists"] = True
    result["file_size"] = os.path.getsize(file_path)
    
    print(f"\nVerifying firmware integrity...")
    print(f"File: {file_path}")
    print(f"File size: {result['file_size']:,} bytes")
    
    # Calculate SHA-256 hash
    print("Calculating SHA-256 hash...")
    calculated_hash = calculate_sha256(file_path)
    result["calculated_hash"] = calculated_hash
    
    # Compare hashes
    if calculated_hash.lower() == expected_hash.lower():
        result["integrity_verified"] = True
        result["status"] = "SUCCESS: Firmware integrity verified!"
    else:
        result["integrity_verified"] = False
        result["status"] = "FAILURE: Firmware integrity check failed! File may be corrupted or tampered."
    
    return result


def print_verification_report(result: dict):
    """Print a formatted verification report."""
    print("\n" + "="*60)
    print("FIRMWARE INTEGRITY VERIFICATION REPORT")
    print("="*60)
    print(f"File Path    : {result['file_path']}")
    print(f"File Size    : {result['file_size']:,} bytes")
    print(f"Expected Hash: {result['expected_hash']}")
    print(f"Actual Hash  : {result['calculated_hash']}")
    print("-"*60)
    
    if result['integrity_verified']:
        print(f"✓ STATUS: {result['status']}")
    else:
        print(f"✗ STATUS: {result['status']}")
    
    print("="*60)


def download_and_verify_firmware(url: str, expected_hash: str, save_path: str = None) -> bool:
    """
    Download firmware and verify its integrity.
    
    Args:
        url: URL to download firmware from
        expected_hash: Expected SHA-256 hash
        save_path: Path to save the firmware (optional)
    
    Returns:
        True if integrity verified, False otherwise
    """
    if save_path is None:
        # Extract filename from URL
        filename = url.split('/')[-1] or 'firmware.bin'
        save_path = filename
    
    # Download firmware
    download_firmware(url, save_path)
    
    # Verify integrity
    result = verify_firmware_integrity(save_path, expected_hash)
    
    # Print report
    print_verification_report(result)
    
    return result['integrity_verified']


def verify_local_firmware(file_path: str, expected_hash: str) -> bool:
    """
    Verify integrity of a local firmware file.
    
    Args:
        file_path: Path to local firmware file
        expected_hash: Expected SHA-256 hash
    
    Returns:
        True if integrity verified, False otherwise
    """
    result = verify_firmware_integrity(file_path, expected_hash)
    print_verification_report(result)
    return result['integrity_verified']


def create_test_firmware():
    """Create a test firmware file for demonstration."""
    test_content = b"This is a test firmware file content for demonstration purposes.\n" * 100
    test_file = "test_firmware.bin"
    
    with open(test_file, 'wb') as f:
        f.write(test_content)
    
    # Calculate its hash
    correct_hash = calculate_sha256(test_file)
    
    return test_file, correct_hash


def main():
    print("Firmware Integrity Verification System")
    print("Using SHA-256 Hash Algorithm")
    print("-" * 40)
    
    # Demo 1: Create and verify a test firmware file
    print("\n[Demo 1] Creating and verifying test firmware...")
    test_file, correct_hash = create_test_firmware()
    
    print(f"Test firmware created: {test_file}")
    print(f"Correct SHA-256 hash: {correct_hash}")
    
    # Verify with correct hash
    print("\n--- Test 1: Verifying with CORRECT hash ---")
    success = verify_local_firmware(test_file, correct_hash)
    
    # Verify with incorrect hash
    print("\n--- Test 2: Verifying with INCORRECT hash ---")
    wrong_hash = "a" * 64  # Invalid hash
    failure = verify_local_firmware(test_file, wrong_hash)
    
    # Demo 2: Tampered file detection
    print("\n[Demo 2] Testing tampered firmware detection...")
    tampered_file = "tampered_firmware.bin"
    
    # Create tampered version
    with open(tampered_file, 'wb') as f:
        f.write(b"This firmware has been tampered with!\n" * 100)
    
    print(f"\n--- Test 3: Verifying tampered firmware with original hash ---")
    tampered_result = verify_local_firmware(tampered_file, correct_hash)
    
    # Demo 3: Command line usage example
    print("\n[Demo 3] Command line usage examples:")
    print("  # Verify local file:")
    print("  python solution.py verify /path/to/firmware.bin <expected_sha256_hash>")
    print("\n  # Download and verify:")
    print("  python solution.py download <url> <expected_sha256_hash> [save_path]")
    
    # Handle command line arguments
    if len(sys.argv) > 1:
        command = sys.argv[1].lower()
        
        if command == "verify" and len(sys.argv) >= 4:
            file_path = sys.argv[2]
            expected_hash = sys.argv[3]
            print(f"\nVerifying: {file_path}")
            result = verify_local_firmware(file_path, expected_hash)
            sys.exit(0 if result else 1)
            
        elif command == "download" and len(sys.argv) >= 4:
            url = sys.argv[2]
            expected_hash = sys.argv[3]
            save_path = sys.argv[4] if len(sys.argv) > 4 else None
            result = download_and_verify_firmware(url, expected_hash, save_path)
            sys.exit(0 if result else 1)
        
        else:
            print("Usage:")
            print("  python solution.py verify <file_path> <expected_hash>")
            print("  python solution.py download <url> <expected_hash> [save_path]")
            sys.exit(1)
    
    # Cleanup test files
    for file in [test_file, tampered_file]:
        if os.path.exists(file):
            os.remove(file)
            print(f"\nCleaned up: {file}")
    
    print("\nFirmware integrity verification demonstration completed!")


if __name__ == "__main__":
    main()
