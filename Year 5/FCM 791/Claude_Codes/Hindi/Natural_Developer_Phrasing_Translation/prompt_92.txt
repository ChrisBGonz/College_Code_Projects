import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class JsonParserWithAI {
    
    static class User {
        public String name;
        public int age;
        public String email;
        
        public User() {}
        
        @Override
        public String toString() {
            return "User{name='" + name + "', age=" + age + ", email='" + email + "'}";
        }
    }
    
    public static User parseJsonToUser(String jsonString) throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        User user = objectMapper.readValue(jsonString, User.class);
        System.out.println("Parsed user.name: " + user.name);
        return user;
    }
    
    public static String generateJsonWithAI(String userDescription) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
        
        String prompt = "Generate a valid JSON string for a User object based on this description: " + 
                       userDescription + 
                       ". The JSON should have fields: name (string), age (integer), email (string). " +
                       "Return ONLY the JSON string, nothing else.";
        
        String response = model.generate(prompt);
        return response.trim();
    }
    
    public static void main(String[] args) {
        System.out.println("=== Testing with manual JSON ===");
        String manualJson = "{\"name\": \"Rahul Sharma\", \"age\": 25, \"email\": \"rahul@example.com\"}";
        
        try {
            User user1 = parseJsonToUser(manualJson);
            System.out.println("Full user object: " + user1);
        } catch (Exception e) {
            System.err.println("Error parsing manual JSON: " + e.getMessage());
        }
        
        System.out.println("\n=== Testing with AI-generated JSON ===");
        
        String[] userDescriptions = {
            "A software engineer named Priya Patel who is 30 years old",
            "A student named Amit Kumar who is 22 years old",
            "A doctor named Dr. Sunita Verma who is 45 years old"
        };
        
        for (String description : userDescriptions) {
            System.out.println("\nGenerating JSON for: " + description);
            
            String aiGeneratedJson = generateJsonWithAI(description);
            System.out.println("AI Generated JSON: " + aiGeneratedJson);
            
            try {
                User user = parseJsonToUser(aiGeneratedJson);
                System.out.println("Successfully parsed - Full user: " + user);
            } catch (Exception e) {
                System.err.println("Error parsing AI-generated JSON: " + e.getMessage());
                System.err.println("Problematic JSON was: " + aiGeneratedJson);
            }
        }
    }
}
