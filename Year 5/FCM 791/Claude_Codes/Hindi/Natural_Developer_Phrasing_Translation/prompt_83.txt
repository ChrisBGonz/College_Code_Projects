import java.sql.*;
import java.util.logging.Level;
import java.util.logging.Logger;

// Custom Exception Class
class DatabaseException extends RuntimeException {
    private int errorCode;
    private String sqlState;

    public DatabaseException(String message, int errorCode, String sqlState, Throwable cause) {
        super(message, cause);
        this.errorCode = errorCode;
        this.sqlState = sqlState;
    }

    public DatabaseException(String message, Throwable cause) {
        super(message, cause);
        this.errorCode = -1;
        this.sqlState = "UNKNOWN";
    }

    public int getErrorCode() {
        return errorCode;
    }

    public String getSqlState() {
        return sqlState;
    }

    @Override
    public String toString() {
        return String.format("DatabaseException{message='%s', errorCode=%d, sqlState='%s'}",
                getMessage(), errorCode, sqlState);
    }
}

// Database Service Class
public class DatabaseService {

    private static final Logger LOGGER = Logger.getLogger(DatabaseService.class.getName());

    // Database Connection Details (Example)
    private static final String DB_URL = "jdbc:mysql://localhost:3306/mydb";
    private static final String DB_USER = "root";
    private static final String DB_PASSWORD = "password";

    /**
     * Method jo SQLException ko catch kare, log kare, aur custom exception throw kare
     */
    public void executeQuery(String query) throws DatabaseException {
        Connection connection = null;
        Statement statement = null;
        ResultSet resultSet = null;

        try {
            // Database connection establish karo
            connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
            statement = connection.createStatement();
            resultSet = statement.executeQuery(query);

            // Result process karo
            while (resultSet.next()) {
                System.out.println("Data: " + resultSet.getString(1));
            }

            LOGGER.info("Query successfully execute hui: " + query);

        } catch (SQLException e) {
            // Error message log karo
            LOGGER.log(Level.SEVERE,
                    String.format("SQL Error - Message: %s | Error Code: %d | SQL State: %s | Query: %s",
                            e.getMessage(),
                            e.getErrorCode(),
                            e.getSQLState(),
                            query),
                    e);

            // Custom exception throw karo
            throw new DatabaseException(
                    "Database query execute karne mein error aaya: " + e.getMessage(),
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );

        } finally {
            // Resources close karo
            closeResources(resultSet, statement, connection);
        }
    }

    /**
     * User data fetch karne ka method
     */
    public void getUserById(int userId) throws DatabaseException {
        Connection connection = null;
        PreparedStatement preparedStatement = null;
        ResultSet resultSet = null;

        try {
            connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);

            String query = "SELECT id, name, email FROM users WHERE id = ?";
            preparedStatement = connection.prepareStatement(query);
            preparedStatement.setInt(1, userId);

            resultSet = preparedStatement.executeQuery();

            if (resultSet.next()) {
                System.out.println("User ID: " + resultSet.getInt("id"));
                System.out.println("Name: " + resultSet.getString("name"));
                System.out.println("Email: " + resultSet.getString("email"));
            } else {
                System.out.println("User ID " + userId + " ke liye koi data nahi mila.");
            }

            LOGGER.info("User data successfully fetch kiya gaya. User ID: " + userId);

        } catch (SQLException e) {
            // Detailed error logging
            LOGGER.log(Level.SEVERE,
                    String.format("User fetch karne mein SQL Error - " +
                                    "User ID: %d | Message: %s | Error Code: %d | SQL State: %s",
                            userId,
                            e.getMessage(),
                            e.getErrorCode(),
                            e.getSQLState()),
                    e);

            // Custom exception throw karo with meaningful message
            throw new DatabaseException(
                    String.format("User ID %d ko fetch karne mein error aaya: %s", userId, e.getMessage()),
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );

        } finally {
            closeResources(resultSet, preparedStatement, connection);
        }
    }

    /**
     * Data insert karne ka method
     */
    public void insertUser(String name, String email) throws DatabaseException {
        Connection connection = null;
        PreparedStatement preparedStatement = null;

        try {
            connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
            connection.setAutoCommit(false); // Transaction start karo

            String query = "INSERT INTO users (name, email) VALUES (?, ?)";
            preparedStatement = connection.prepareStatement(query);
            preparedStatement.setString(1, name);
            preparedStatement.setString(2, email);

            int rowsAffected = preparedStatement.executeUpdate();
            connection.commit(); // Transaction commit karo

            LOGGER.info(String.format("User successfully insert kiya gaya. Name: %s, Email: %s, Rows Affected: %d",
                    name, email, rowsAffected));

        } catch (SQLException e) {
            // Transaction rollback karo
            if (connection != null) {
                try {
                    connection.rollback();
                    LOGGER.warning("Transaction rollback kiya gaya.");
                } catch (SQLException rollbackEx) {
                    LOGGER.log(Level.SEVERE, "Rollback karne mein bhi error aaya: " + rollbackEx.getMessage(), rollbackEx);
                }
            }

            // Error log karo
            LOGGER.log(Level.SEVERE,
                    String.format("User insert karne mein SQL Error - " +
                                    "Name: %s | Email: %s | Message: %s | Error Code: %d | SQL State: %s",
                            name, email,
                            e.getMessage(),
                            e.getErrorCode(),
                            e.getSQLState()),
                    e);

            // Custom exception throw karo
            throw new DatabaseException(
                    String.format("User '%s' ko insert karne mein error aaya: %s", name, e.getMessage()),
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );

        } finally {
            closeResources(null, preparedStatement, connection);
        }
    }

    /**
     * Resources close karne ka helper method
     */
    private void closeResources(ResultSet resultSet, Statement statement, Connection connection) {
        if (resultSet != null) {
            try {
                resultSet.close();
            } catch (SQLException e) {
                LOGGER.log(Level.WARNING, "ResultSet close karne mein error: " + e.getMessage(), e);
            }
        }

        if (statement != null) {
            try {
                statement.close();
            } catch (SQLException e) {
                LOGGER.log(Level.WARNING, "Statement close karne mein error: " + e.getMessage(), e);
            }
        }

        if (connection != null) {
            try {
                connection.close();
            } catch (SQLException e) {
                LOGGER.log(Level.WARNING, "Connection close karne mein error: " + e.getMessage(), e);
            }
        }
    }

    // Main method - Testing ke liye
    public static void main(String[] args) {
        DatabaseService dbService = new DatabaseService();

        // Test 1: Query execute karo
        System.out.println("=== Test 1: Query Execute ===");
        try {
            dbService.executeQuery("SELECT * FROM users");
        } catch (DatabaseException e) {
            System.err.println("Error pakda gaya: " + e.getMessage());
            System.err.println("Error Code: " + e.getErrorCode());
            System.err.println("SQL State: " + e.getSqlState());
            System.err.println("Cause: " + e.getCause().getMessage());
        }

        System.out.println();

        // Test 2: User fetch karo
        System.out.println("=== Test 2: User Fetch ===");
        try {
            dbService.getUserById(1);
        } catch (DatabaseException e) {
            System.err.println("User fetch error: " + e);
        }

        System.out.println();

        // Test 3: User insert karo
        System.out.println("=== Test 3: User Insert ===");
        try {
            dbService.insertUser("Rahul Kumar", "rahul@example.com");
            System.out.println("User successfully insert kiya gaya!");
        } catch (DatabaseException e) {
            System.err.println("Insert error: " + e.getMessage());
            // Cause chain check karo
            Throwable cause = e.getCause();
            if (cause != null) {
                System.err.println("Original SQL Error: " + cause.getMessage());
            }
        }
    }
}
