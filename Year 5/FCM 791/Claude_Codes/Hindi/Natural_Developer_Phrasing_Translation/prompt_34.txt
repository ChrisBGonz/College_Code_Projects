
using System;
using System.Security.Cryptography;
using System.Text;

class RSAKeyPairDemo
{
    static void Main(string[] args)
    {
        Console.WriteLine("=== RSA Key Pair Generator for Digital Signature ===\n");

        // RSA key pair generate karo (2048-bit)
        using (RSA rsa = RSA.Create(2048))
        {
            // Public Key export karo (PEM format)
            string publicKeyPem = ExportPublicKeyToPem(rsa);
            Console.WriteLine("--- PUBLIC KEY ---");
            Console.WriteLine(publicKeyPem);

            // Private Key export karo (PEM format)
            string privateKeyPem = ExportPrivateKeyToPem(rsa);
            Console.WriteLine("--- PRIVATE KEY ---");
            Console.WriteLine(privateKeyPem);

            // ---- Digital Signature Demo ----
            string originalMessage = "Yeh ek important document hai jo sign kiya jayega.";
            Console.WriteLine($"Original Message: {originalMessage}\n");

            // Message ko sign karo (Private Key se)
            byte[] signature = SignData(rsa, originalMessage);
            Console.WriteLine($"Digital Signature (Base64):\n{Convert.ToBase64String(signature)}\n");

            // Signature verify karo (Public Key se)
            bool isValid = VerifySignature(rsa, originalMessage, signature);
            Console.WriteLine($"Signature Valid: {isValid}");

            // Tampered message ke saath verify karo
            string tamperedMessage = "Yeh ek tampered document hai.";
            bool isTamperedValid = VerifySignature(rsa, tamperedMessage, signature);
            Console.WriteLine($"Tampered Message Signature Valid: {isTamperedValid}");

            // XML format mein keys export karo (optional)
            Console.WriteLine("\n--- RSA Parameters (XML Format) ---");
            Console.WriteLine("Public Key XML:");
            Console.WriteLine(rsa.ToXmlString(false)); // false = sirf public key

            Console.WriteLine("\nPrivate Key XML (includePrivateParameters = true):");
            Console.WriteLine(rsa.ToXmlString(true));  // true = private key bhi include
        }

        Console.WriteLine("\nPress any key to exit...");
        Console.ReadKey();
    }

    // Data ko Private Key se sign karo
    static byte[] SignData(RSA rsa, string message)
    {
        byte[] dataToSign = Encoding.UTF8.GetBytes(message);

        // SHA256 hash ke saath PKCS#1 v1.5 padding use karo
        byte[] signature = rsa.SignData(
            dataToSign,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );

        return signature;
    }

    // Signature ko Public Key se verify karo
    static bool VerifySignature(RSA rsa, string message, byte[] signature)
    {
        byte[] dataToVerify = Encoding.UTF8.GetBytes(message);

        bool isValid = rsa.VerifyData(
            dataToVerify,
            signature,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );

        return isValid;
    }

    // Public Key ko PEM format mein export karo
    static string ExportPublicKeyToPem(RSA rsa)
    {
        byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        string base64 = Convert.ToBase64String(publicKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PUBLIC KEY-----\n{base64}\n-----END PUBLIC KEY-----\n";
    }

    // Private Key ko PEM format mein export karo
    static string ExportPrivateKeyToPem(RSA rsa)
    {
        byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        string base64 = Convert.ToBase64String(privateKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PRIVATE KEY-----\n{base64}\n-----END PRIVATE KEY-----\n";
    }
}
