def query_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database query for invoice"""
    return MOCK_DATABASE.get(invoice_id.upper())

def process_invoice_with_claude(invoice_data: dict) -> dict:
    """Use Claude AI to process and enhance invoice data"""
    client = anthropic.Anthropic()
    
    # Create a prompt for Claude to analyze the invoice
    prompt = f"""
    Analyze the following invoice data and provide:
    1. A summary of the invoice
    2. Payment status analysis
    3. Any recommendations or alerts
    
    Invoice Data:
    {json.dumps(invoice_data, indent=2)}
    
    Please respond in JSON format with the following structure:
    {{
        "summary": "brief summary of the invoice",
        "status_analysis": "analysis of payment status",
        "recommendations": ["list", "of", "recommendations"],
        "alerts": ["any", "important", "alerts"]
    }}
    """
    
    # Use streaming to get Claude's response
    analysis_text = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            analysis_text += text
            print(text, end="", flush=True)
    
    print()  # New line after streaming
    
    # Parse Claude's JSON response
    try:
        # Extract JSON from the response
        start_idx = analysis_text.find('{')
        end_idx = analysis_text.rfind('}') + 1
        if start_idx != -1 and end_idx > start_idx:
            json_str = analysis_text[start_idx:end_idx]
            analysis = json.loads(json_str)
        else:
            analysis = {
                "summary": analysis_text,
                "status_analysis": "Unable to parse structured response",
                "recommendations": [],
                "alerts": []
            }
    except json.JSONDecodeError:
        analysis = {
            "summary": analysis_text,
            "status_analysis": "Unable to parse structured response",
            "recommendations": [],
            "alerts": []
        }
    
    return analysis

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Flask route to get invoice details by invoice_id
    
    Path Parameter:
        invoice_id: The unique identifier for the invoice (e.g., INV001, INV002, INV003)
    
    Returns:
        JSON response with invoice details and AI analysis
    """
    
    # Query invoice from database
    invoice_data = query_invoice_from_db(invoice_id)
    
    # Check if invoice exists
    if not invoice_data:
        return jsonify({
            "success": False,
            "error": f"Invoice with ID '{invoice_id}' not found",
            "message": "Please check the invoice ID and try again"
        }), 404
    
    # Process invoice with Claude AI for enhanced analysis
    print(f"\nProcessing invoice {invoice_id} with Claude AI...")
    print("Claude's Analysis (streaming):")
    print("-" * 50)
    
    ai_analysis = process_invoice_with_claude(invoice_data)
    
    print("-" * 50)
    
    # Prepare the response
    response_data = {
        "success": True,
        "invoice": {
            "invoice_id": invoice_data["invoice_id"],
            "customer": {
                "name": invoice_data["customer_name"],
                "email": invoice_data["customer_email"]
            },
            "financial": {
                "amount": invoice_data["amount"],
                "currency": invoice_data["currency"],
                "status": invoice_data["status"]
            },
            "items": invoice_data["items"],
            "dates": {
                "created": invoice_data["created_date"],
                "due": invoice_data["due_date"],
                "paid": invoice_data["paid_date"]
            }
        },
        "ai_analysis": ai_analysis,
        "metadata": {
            "processed_at": datetime.now().isoformat(),
            "api_version": "1.0"
        }
    }
    
    return jsonify(response_data), 200
