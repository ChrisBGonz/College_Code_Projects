import Anthropic from "@anthropic-ai/sdk";
import * as crypto from "crypto";

const client = new Anthropic();

// Simulated database
const userDatabase: {
  [key: string]: { password_hash: string; salt: string; email: string };
} = {
  user_001: {
    password_hash: "old_hash_value",
    salt: "old_salt_value",
    email: "user1@example.com",
  },
  user_002: {
    password_hash: "another_old_hash",
    salt: "another_old_salt",
    email: "user2@example.com",
  },
};

// Tool definitions for password reset operations
const tools: Anthropic.Tool[] = [
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, numbers, and special characters)",
    input_schema: {
      type: "object" as const,
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "check_user_exists",
    description: "Checks if a user exists in the database by user_id",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "generate_salt",
    description: "Generates a cryptographically secure random salt for password hashing",
    input_schema: {
      type: "object" as const,
      properties: {
        length: {
          type: "number",
          description: "Length of the salt in bytes (default: 32)",
        },
      },
      required: [],
    },
  },
  {
    name: "hash_password",
    description: "Hashes a password with the provided salt using PBKDF2 algorithm",
    input_schema: {
      type: "object" as const,
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
        salt: {
          type: "string",
          description: "The salt to use for hashing",
        },
      },
      required: ["password", "salt"],
    },
  },
  {
    name: "update_password_in_db",
    description: "Updates the user's password hash and salt in the database",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The hashed password to store",
        },
        salt: {
          type: "string",
          description: "The salt used for hashing",
        },
      },
      required: ["user_id", "password_hash", "salt"],
    },
  },
];

// Tool implementation functions
function validatePasswordStrength(password: string): {
  valid: boolean;
  message: string;
} {
  const minLength = 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumbers = /\d/.test(password);
  const hasSpecialChars = /[!@#$%^&*(),.?":{}|<>]/.test(password);

  if (password.length < minLength) {
    return {
      valid: false,
      message: `Password must be at least ${minLength} characters long`,
    };
  }
  if (!hasUppercase) {
    return {
      valid: false,
      message: "Password must contain at least one uppercase letter",
    };
  }
  if (!hasLowercase) {
    return {
      valid: false,
      message: "Password must contain at least one lowercase letter",
    };
  }
  if (!hasNumbers) {
    return { valid: false, message: "Password must contain at least one number" };
  }
  if (!hasSpecialChars) {
    return {
      valid: false,
      message: "Password must contain at least one special character",
    };
  }

  return { valid: true, message: "Password meets all security requirements" };
}

function checkUserExists(user_id: string): { exists: boolean; message: string } {
  if (userDatabase[user_id]) {
    return { exists: true, message: `User ${user_id} found in database` };
  }
  return { exists: false, message: `User ${user_id} not found in database` };
}

function generateSalt(length: number = 32): { salt: string } {
  const salt = crypto.randomBytes(length).toString("hex");
  return { salt };
}

function hashPassword(
  password: string,
  salt: string
): { hash: string; algorithm: string } {
  const hash = crypto
    .pbkdf2Sync(password, salt, 100000, 64, "sha512")
    .toString("hex");
  return { hash, algorithm: "PBKDF2-SHA512-100000-iterations" };
}

function updatePasswordInDb(
  user_id: string,
  password_hash: string,
  salt: string
): { success: boolean; message: string; timestamp: string } {
  if (!userDatabase[user_id]) {
    return {
      success: false,
      message: `User ${user_id} not found`,
      timestamp: new Date().toISOString(),
    };
  }

  // Update the database
  userDatabase[user_id].password_hash = password_hash;
  userDatabase[user_id].salt = salt;

  return {
    success: true,
    message: `Password successfully updated for user ${user_id}`,
    timestamp: new Date().toISOString(),
  };
}

// Process tool calls
function processToolCall(
  tool_name: string,
  tool_input: Record<string, unknown>
): unknown {
  switch (tool_name) {
    case "validate_password_strength":
      return validatePasswordStrength(tool_input.password as string);
    case "check_user_exists":
      return checkUserExists(tool_input.user_id as string);
    case "generate_salt":
      return generateSalt((tool_input.length as number) || 32);
    case "hash_password":
      return hashPassword(
        tool_input.password as string,
        tool_input.salt as string
      );
    case "update_password_in_db":
      return updatePasswordInDb(
        tool_input.user_id as string,
        tool_input.password_hash as string,
        tool_input.salt as string
      );
    default:
      return { error: `Unknown tool: ${tool_name}` };
  }
}

// Main password reset function
async function resetPassword(
  user_id: string,
  new_password: string
): Promise<{ success: boolean; message: string; details?: unknown }> {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Password Reset Request for User: ${user_id}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages: Anthropic.MessageParam[] = [
    {
      role: "user",
      content: `Please help me reset the password for user_id: "${user_id}" with new_password: "${new_password}". 
      
      Follow these steps:
      1. First validate the password strength
      2. Check if the user exists in the database
      3. Generate a secure salt
      4. Hash the password with the salt
      5. Update the password in the database
      6. Return a success or failure response
      
      Make sure to handle any errors appropriately and provide clear feedback.`,
    },
  ];

  let finalResponse = null;

  // Agentic loop
  while (true) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`Response Stop Reason: ${response.stop_reason}`);

    // Check if we're done
    if (response.stop_reason === "end_turn") {
      // Extract the final text response
      for (const block of response.content) {
        if (block.type === "text") {
          finalResponse = block.text;
          console.log(`\nFinal Response: ${block.text}`);
        }
      }
      break;
    }

    // Process tool calls if needed
    if (response.stop_reason === "tool_use") {
      // Add assistant's response to messages
      messages.push({
        role: "assistant",
        content: response.content,
      });

      // Process each tool call
      const toolResults: Anthropic.ToolResultBlockParam[] = [];

      for (const block of response.content) {
        if (block.type === "tool_use") {
          console.log(`\nTool Called: ${block.name}`);
          console.log(`Tool Input: ${JSON.stringify(block.input, null, 2)}`);

          const result = processToolCall(
            block.name,
            block.input as Record<string, unknown>
          );
          console.log(`Tool Result: ${JSON.stringify(result, null, 2)}`);

          toolResults.push({
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          });
        }
      }

      // Add tool results to messages
      messages.push({
        role: "user",
        content: toolResults,
      });
    } else {
      // Unexpected stop reason
      console.log(`Unexpected stop reason: ${response.stop_reason}`);
      break;
    }
  }

  // Parse the final response to determine success/failure
  if (finalResponse) {
    const isSuccess =
      finalResponse.toLowerCase().includes("success") ||
      finalResponse.toLowerCase().includes("successfully");
    return {
      success: isSuccess,
      message: finalResponse,
      details: {
        user_id,
        timestamp: new Date().toISOString(),
      },
    };
  }

  return {
    success: false,
    message: "Password reset process failed - no response received",
    details: { user_id },
  };
}
