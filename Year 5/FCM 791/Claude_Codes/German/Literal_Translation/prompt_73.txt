import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpHandler;
import com.sun.net.httpserver.HttpServer;

import java.io.*;
import java.net.InetSocketAddress;

public class DeserializeUserFromHttpRequest {

    // User class that implements Serializable
    static class User implements Serializable {
        private static final long serialVersionUID = 1L;
        private String name;
        private int age;
        private String email;

        public User(String name, int age, String email) {
            this.name = name;
            this.age = age;
            this.email = email;
        }

        public String getName() { return name; }
        public int getAge() { return age; }
        public String getEmail() { return email; }

        @Override
        public String toString() {
            return "User{name='" + name + "', age=" + age + ", email='" + email + "'}";
        }
    }

    // Method that reads a serialized User object from HTTP request body
    public static User deserializeUserFromRequest(HttpExchange exchange) throws IOException, ClassNotFoundException {
        InputStream inputStream = exchange.getRequestBody();
        ObjectInputStream objectInputStream = new ObjectInputStream(inputStream);
        User user = (User) objectInputStream.readObject();
        objectInputStream.close();
        return user;
    }

    // HTTP Handler that uses the deserialization method
    static class UserHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("POST".equals(exchange.getRequestMethod())) {
                try {
                    User user = deserializeUserFromRequest(exchange);
                    System.out.println("Received user: " + user);

                    String response = "User received: " + user.toString();
                    exchange.sendResponseHeaders(200, response.length());
                    OutputStream os = exchange.getResponseBody();
                    os.write(response.getBytes());
                    os.close();
                } catch (ClassNotFoundException e) {
                    String errorResponse = "Error: Class not found - " + e.getMessage();
                    exchange.sendResponseHeaders(400, errorResponse.length());
                    OutputStream os = exchange.getResponseBody();
                    os.write(errorResponse.getBytes());
                    os.close();
                } catch (IOException e) {
                    String errorResponse = "Error deserializing user: " + e.getMessage();
                    exchange.sendResponseHeaders(500, errorResponse.length());
                    OutputStream os = exchange.getResponseBody();
                    os.write(errorResponse.getBytes());
                    os.close();
                }
            } else {
                String response = "Only POST method is supported";
                exchange.sendResponseHeaders(405, response.length());
                OutputStream os = exchange.getResponseBody();
                os.write(response.getBytes());
                os.close();
            }
        }
    }

    // Helper method to serialize a User object to bytes (for testing)
    public static byte[] serializeUser(User user) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ObjectOutputStream oos = new ObjectOutputStream(baos);
        oos.writeObject(user);
        oos.close();
        return baos.toByteArray();
    }

    public static void main(String[] args) throws Exception {
        // Start HTTP server
        HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0);
        server.createContext("/user", new UserHandler());
        server.start();
        System.out.println("Server started on port 8080");

        // Test the server by sending a serialized user
        testServer();

        // Keep server running for a bit then stop
        Thread.sleep(2000);
        server.stop(0);
        System.out.println("Server stopped");
    }

    private static void testServer() throws Exception {
        // Create a test user and serialize it
        User testUser = new User("John Doe", 30, "john.doe@example.com");
        byte[] serializedUser = serializeUser(testUser);

        // Send HTTP request with serialized user
        java.net.URL url = new java.net.URL("http://localhost:8080/user");
        java.net.HttpURLConnection connection = (java.net.HttpURLConnection) url.openConnection();
        connection.setRequestMethod("POST");
        connection.setDoOutput(true);
        connection.setRequestProperty("Content-Type", "application/octet-stream");
        connection.setRequestProperty("Content-Length", String.valueOf(serializedUser.length));

        // Write serialized user to request body
        OutputStream os = connection.getOutputStream();
        os.write(serializedUser);
        os.flush();
        os.close();

        // Read response
        int responseCode = connection.getResponseCode();
        System.out.println("Response Code: " + responseCode);

        BufferedReader reader = new BufferedReader(new InputStreamReader(connection.getInputStream()));
        StringBuilder response = new StringBuilder();
        String line;
        while ((line = reader.readLine()) != null) {
            response.append(line);
        }
        reader.close();
        System.out.println("Response: " + response.toString());

        connection.disconnect();
    }
}
