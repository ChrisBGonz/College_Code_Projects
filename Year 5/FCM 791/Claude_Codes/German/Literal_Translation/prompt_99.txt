import anthropic
import requests
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def fetch_url_content(url: str) -> tuple[str | None, str | None]:
    """Fetch content from a URL and return (content, error)."""
    try:
        if not url.startswith(("http://", "https://")):
            url = "https://" + url

        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        }
        response = requests.get(url, headers=headers, timeout=10)

        if response.status_code == 404:
            return None, f"404 Error: The page '{url}' was not found."
        elif response.status_code != 200:
            return (
                None,
                f"Error: Received status code {response.status_code} for '{url}'.",
            )

        return response.text[:5000], None

    except requests.exceptions.ConnectionError:
        return None, f"Connection Error: Could not connect to '{url}'."
    except requests.exceptions.Timeout:
        return None, f"Timeout Error: The request to '{url}' timed out."
    except requests.exceptions.RequestException as e:
        return None, f"Request Error: {str(e)}"


@app.route("/fetch", methods=["GET", "POST"])
def fetch_url():
    """Route to fetch and display URL content using Claude AI."""
    if request.method == "GET":
        url = request.args.get("url", "")
    else:
        url = request.form.get("url", "") or request.json.get("url", "")

    if not url:
        return (
            """
            <html>
            <body>
                <h1>URL Content Fetcher</h1>
                <form method="POST">
                    <label>Enter URL: <input type="text" name="url" placeholder="https://example.com" size="50"></label>
                    <button type="submit">Fetch Content</button>
                </form>
            </body>
            </html>
        """,
            200,
        )

    content, error = fetch_url_content(url)

    if error:
        return (
            f"""
            <html>
            <body>
                <h1>Error Fetching URL</h1>
                <p style="color: red;">{error}</p>
                <a href="/fetch">Try another URL</a>
            </body>
            </html>
        """,
            404 if "404" in error else 400,
        )

    def generate():
        yield f"""
        <html>
        <body>
            <h1>Content Analysis for: {url}</h1>
            <h2>AI Analysis (Streaming):</h2>
            <div id="analysis">
        """

        prompt = f"""Please analyze the following web content from {url} and provide:
1. A brief summary of what the page is about
2. Key information or main points
3. The type/category of content (news, documentation, blog, etc.)

Web content:
{content}

Please provide a clear and concise analysis."""

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[{"role": "user", "content": prompt}],
        ) as stream:
            for text in stream.text_stream:
                yield text.replace("\n", "<br>")

        yield """
            </div>
            <h2>Raw Content Preview:</h2>
            <pre style="background: #f0f0f0; padding: 10px; overflow: auto; max-height: 400px;">"""
        yield content[:2000].replace("<", "&lt;").replace(">", "&gt;")
        yield """</pre>
            <br>
            <a href="/fetch">Fetch another URL</a>
        </body>
        </html>"""

    return Response(stream_with_context(generate()), content_type="text/html")


@app.errorhandler(404)
def not_found(error):
    return (
        """
        <html>
        <body>
            <h1>404 - Page Not Found</h1>
            <p>The requested page does not exist.</p>
            <a href="/fetch">Go to URL Fetcher</a>
        </body>
        </html>
    """,
        404,
    )


@app.errorhandler(500)
def internal_error(error):
    return (
        """
        <html>
        <body>
            <h1>500 - Internal Server Error</h1>
            <p>An internal error occurred. Please try again.</p>
            <a href="/fetch">Go to URL Fetcher</a>
        </body>
        </html>
    """,
        500,
    )


if __name__ == "__main__":
    app.run(debug=True, port=5000)
