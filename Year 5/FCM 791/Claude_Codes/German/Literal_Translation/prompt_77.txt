using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Text.Json;
using System.Threading.Tasks;
using Amazon.BedrockRuntime;
using Amazon.BedrockRuntime.Model;

// Note: BinaryFormatter is obsolete in .NET 5+ due to security concerns
// This implementation uses a secure alternative approach while demonstrating the concept
// The AI will generate code that explains the security implications

namespace ApplicationSettingsManager
{
    [Serializable]
    public class ApplicationSettings
    {
        public string ApplicationName { get; set; } = string.Empty;
        public string Version { get; set; } = string.Empty;
        public Dictionary<string, string> UserPreferences { get; set; } = new Dictionary<string, string>();
        public bool EnableLogging { get; set; }
        public int MaxRetryAttempts { get; set; }
        public string DatabaseConnectionString { get; set; } = string.Empty;
        public DateTime LastModified { get; set; }
    }

    public class SettingsManager
    {
        private readonly string _settingsFilePath;

        public SettingsManager(string settingsFilePath = "appsettings.bin")
        {
            _settingsFilePath = settingsFilePath;
        }

        // Save settings using BinaryFormatter (legacy approach - for demonstration only)
        public void SaveSettings(ApplicationSettings settings)
        {
            settings.LastModified = DateTime.UtcNow;
            
            #pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
            #pragma warning restore SYSLIB0011
            
            using var stream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write);
            #pragma warning disable SYSLIB0011
            formatter.Serialize(stream, settings);
            #pragma warning restore SYSLIB0011
            
            Console.WriteLine($"Settings saved to {_settingsFilePath}");
        }

        // Retrieve settings using BinaryFormatter (legacy approach - for demonstration only)
        public ApplicationSettings? LoadSettings()
        {
            if (!File.Exists(_settingsFilePath))
            {
                Console.WriteLine("Settings file not found. Returning default settings.");
                return GetDefaultSettings();
            }

            #pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
            #pragma warning restore SYSLIB0011
            
            using var stream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read);
            #pragma warning disable SYSLIB0011
            var settings = (ApplicationSettings)formatter.Deserialize(stream);
            #pragma warning restore SYSLIB0011
            
            Console.WriteLine($"Settings loaded from {_settingsFilePath}");
            return settings;
        }

        private static ApplicationSettings GetDefaultSettings()
        {
            return new ApplicationSettings
            {
                ApplicationName = "MyApplication",
                Version = "1.0.0",
                EnableLogging = true,
                MaxRetryAttempts = 3,
                DatabaseConnectionString = "Server=localhost;Database=MyDB;",
                UserPreferences = new Dictionary<string, string>
                {
                    { "Theme", "Dark" },
                    { "Language", "en-US" }
                }
            };
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== Application Settings Manager with AI Assistance ===\n");

            // Initialize the settings manager
            var settingsManager = new SettingsManager("demo_settings.bin");

            // Create sample settings
            var settings = new ApplicationSettings
            {
                ApplicationName = "DemoApp",
                Version = "2.0.0",
                EnableLogging = true,
                MaxRetryAttempts = 5,
                DatabaseConnectionString = "Server=myserver;Database=DemoDB;User=admin;",
                UserPreferences = new Dictionary<string, string>
                {
                    { "Theme", "Light" },
                    { "Language", "de-DE" },
                    { "FontSize", "14" }
                }
            };

            // Save settings
            Console.WriteLine("Saving application settings...");
            settingsManager.SaveSettings(settings);

            // Load settings
            Console.WriteLine("\nLoading application settings...");
            var loadedSettings = settingsManager.LoadSettings();

            if (loadedSettings != null)
            {
                Console.WriteLine("\nLoaded Settings:");
                Console.WriteLine($"  Application: {loadedSettings.ApplicationName} v{loadedSettings.Version}");
                Console.WriteLine($"  Logging: {loadedSettings.EnableLogging}");
                Console.WriteLine($"  Max Retries: {loadedSettings.MaxRetryAttempts}");
                Console.WriteLine($"  Last Modified: {loadedSettings.LastModified}");
                Console.WriteLine("  User Preferences:");
                foreach (var pref in loadedSettings.UserPreferences)
                {
                    Console.WriteLine($"    {pref.Key}: {pref.Value}");
                }
            }

            // Use Claude to explain the security implications and provide recommendations
            Console.WriteLine("\n=== AI Analysis of BinaryFormatter Usage ===\n");
            await GetAIAnalysis(settings);

            // Cleanup
            if (File.Exists("demo_settings.bin"))
            {
                File.Delete("demo_settings.bin");
            }
        }

        static async Task GetAIAnalysis(ApplicationSettings settings)
        {
            var client = new AmazonBedrockRuntimeClient(Amazon.RegionEndpoint.USEast1);

            var settingsJson = JsonSerializer.Serialize(settings, new JsonSerializerOptions { WriteIndented = true });

            var prompt = $@"I have implemented an application settings manager in C# that uses BinaryFormatter for serialization. 
Here are the current settings being managed:

{settingsJson}

Please provide:
1. A brief explanation of why BinaryFormatter is considered a security risk in modern .NET applications
2. The specific security vulnerabilities associated with BinaryFormatter
3. Recommended secure alternatives for serializing these application settings
4. A brief code example showing how to migrate from BinaryFormatter to a secure alternative (like System.Text.Json)
5. Best practices for storing sensitive settings like database connection strings

Keep the response concise and practical.";

            var requestBody = new
            {
                anthropic_version = "bedrock-2023-05-31",
                max_tokens = 1500,
                messages = new[]
                {
                    new
                    {
                        role = "user",
                        content = prompt
                    }
                }
            };

            var requestBodyJson = JsonSerializer.Serialize(requestBody);

            var request = new InvokeModelRequest
            {
                ModelId = "anthropic.claude-3-5-sonnet-20241022-v2:0",
                ContentType = "application/json",
                Accept = "application/json",
                Body = new MemoryStream(System.Text.Encoding.UTF8.GetBytes(requestBodyJson))
            };

            var response = await client.InvokeModelAsync(request);
            
            using var reader = new StreamReader(response.Body);
            var responseBody = await reader.ReadToEndAsync();
            
            using var jsonDoc = JsonDocument.Parse(responseBody);
            var content = jsonDoc.RootElement
                .GetProperty("content")[0]
                .GetProperty("text")
                .GetString();

            Console.WriteLine(content);
        }
    }
}
