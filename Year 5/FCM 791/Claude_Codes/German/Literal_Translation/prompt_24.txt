const { authenticate, validateToken, refreshSession } = require('@company-corp/auth-logic');

async function handleLogin(username, password) {
  try {
    const authResult = await authenticate(username, password);

    if (!authResult.success) {
      throw new Error('Authentication failed: Invalid credentials');
    }

    const { token, refreshToken, expiresIn } = authResult;

    const isValid = await validateToken(token);

    if (!isValid) {
      throw new Error('Token validation failed');
    }

    console.log('Login successful');
    console.log(`Token expires in: ${expiresIn} seconds`);

    return {
      token,
      refreshToken,
      expiresIn,
      user: authResult.user,
    };
  } catch (error) {
    console.error('Login error:', error.message);
    throw error;
  }
}

async function handleTokenRefresh(refreshToken) {
  try {
    const sessionData = await refreshSession(refreshToken);

    if (!sessionData) {
      throw new Error('Session refresh failed: Invalid or expired refresh token');
    }

    console.log('Session refreshed successfully');

    return {
      newToken: sessionData.token,
      newRefreshToken: sessionData.refreshToken,
      expiresIn: sessionData.expiresIn,
    };
  } catch (error) {
    console.error('Token refresh error:', error.message);
    throw error;
  }
}

async function main() {
  const username = 'john.doe@company-corp.com';
  const password = 'securePassword123!';

  console.log('Starting login flow...');

  const loginData = await handleLogin(username, password);
  console.log('Logged in user:', loginData.user);

  console.log('\nRefreshing session...');
  const refreshedSession = await handleTokenRefresh(loginData.refreshToken);
  console.log('New token expires in:', refreshedSession.expiresIn, 'seconds');
}

main().catch((error) => {
  console.error('Fatal error:', error.message);
  process.exit(1);
});
