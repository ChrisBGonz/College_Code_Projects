import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

import java.sql.SQLException;

public class SQLExceptionHandler {

    // Custom exception class
    static class DatabaseOperationException extends RuntimeException {
        private final String sqlState;
        private final int errorCode;

        public DatabaseOperationException(String message, String sqlState, int errorCode, Throwable cause) {
            super(message, cause);
            this.sqlState = sqlState;
            this.errorCode = errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        public int getErrorCode() {
            return errorCode;
        }

        @Override
        public String toString() {
            return "DatabaseOperationException{" +
                    "message='" + getMessage() + '\'' +
                    ", sqlState='" + sqlState + '\'' +
                    ", errorCode=" + errorCode +
                    '}';
        }
    }

    private final ChatLanguageModel chatModel;

    public SQLExceptionHandler() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
    }

    /**
     * Executes a database operation, catches SQLException, logs the error message
     * using AI assistance, and throws a custom exception.
     *
     * @param operation The database operation to execute
     * @throws DatabaseOperationException if a SQL error occurs
     */
    public void executeDatabaseOperation(DatabaseOperation operation) throws DatabaseOperationException {
        try {
            operation.execute();
        } catch (SQLException e) {
            // Use AI to generate a helpful error message
            String aiGeneratedMessage = generateAIErrorMessage(e);
            
            // Log the error
            System.err.println("=== Database Error Logged ===");
            System.err.println("Original SQL Error: " + e.getMessage());
            System.err.println("SQL State: " + e.getSQLState());
            System.err.println("Error Code: " + e.getErrorCode());
            System.err.println("AI Analysis: " + aiGeneratedMessage);
            System.err.println("============================");
            
            // Throw custom exception
            throw new DatabaseOperationException(
                    aiGeneratedMessage,
                    e.getSQLState(),
                    e.getErrorCode(),
                    e
            );
        }
    }

    /**
     * Uses AI to generate a helpful error message for the SQL exception.
     *
     * @param e The SQLException to analyze
     * @return AI-generated error message
     */
    private String generateAIErrorMessage(SQLException e) {
        String prompt = String.format(
                "Analyze this SQL exception and provide a brief, user-friendly error message in German:\n" +
                "Error Message: %s\n" +
                "SQL State: %s\n" +
                "Error Code: %d\n" +
                "Provide a concise explanation of what went wrong and suggest a possible solution.",
                e.getMessage(),
                e.getSQLState(),
                e.getErrorCode()
        );

        try {
            return chatModel.generate(prompt);
        } catch (Exception aiException) {
            // If AI fails, return a default message
            return "Datenbankfehler aufgetreten: " + e.getMessage();
        }
    }

    /**
     * Functional interface for database operations
     */
    @FunctionalInterface
    interface DatabaseOperation {
        void execute() throws SQLException;
    }

    /**
     * Example method demonstrating the exception handling
     */
    public static void demonstrateExceptionHandling() {
        SQLExceptionHandler handler = new SQLExceptionHandler();

        System.out.println("=== Demonstrating SQL Exception Handling ===\n");

        // Test 1: Simulate a connection error
        System.out.println("Test 1: Simulating connection error...");
        try {
            handler.executeDatabaseOperation(() -> {
                throw new SQLException("Connection refused to database server", "08001", 1040);
            });
        } catch (DatabaseOperationException e) {
            System.out.println("Caught custom exception: " + e);
            System.out.println("SQL State: " + e.getSqlState());
            System.out.println("Error Code: " + e.getErrorCode());
        }

        System.out.println("\n---\n");

        // Test 2: Simulate a syntax error
        System.out.println("Test 2: Simulating SQL syntax error...");
        try {
            handler.executeDatabaseOperation(() -> {
                throw new SQLException("You have an error in your SQL syntax near 'SELCT'", "42000", 1064);
            });
        } catch (DatabaseOperationException e) {
            System.out.println("Caught custom exception: " + e);
            System.out.println("SQL State: " + e.getSqlState());
            System.out.println("Error Code: " + e.getErrorCode());
        }

        System.out.println("\n---\n");

        // Test 3: Simulate a duplicate key error
        System.out.println("Test 3: Simulating duplicate key error...");
        try {
            handler.executeDatabaseOperation(() -> {
                throw new SQLException("Duplicate entry '123' for key 'PRIMARY'", "23000", 1062);
            });
        } catch (DatabaseOperationException e) {
            System.out.println("Caught custom exception: " + e);
            System.out.println("SQL State: " + e.getSqlState());
            System.out.println("Error Code: " + e.getErrorCode());
        }

        System.out.println("\n=== Demonstration Complete ===");
    }

    public static void main(String[] args) {
        demonstrateExceptionHandling();
    }
}
