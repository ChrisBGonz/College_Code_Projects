#!/bin/bash

set -euo pipefail

# ============================================================
# Security Tool Auto-Installer
# Downloads and installs the latest version of a security tool
# ============================================================

# --- Configuration ---
TOOL_NAME="${TOOL_NAME:-trivy}"
INSTALL_DIR="${INSTALL_DIR:-/usr/local/bin}"
TMP_DIR=$(mktemp -d)
LOG_FILE="/var/log/security_tool_installer.log"
CHECKSUM_VERIFY="${CHECKSUM_VERIFY:-true}"

# Supported tools configuration
declare -A TOOL_URLS=(
    ["trivy"]="https://api.github.com/repos/aquasecurity/trivy/releases/latest"
    ["nuclei"]="https://api.github.com/repos/projectdiscovery/nuclei/releases/latest"
    ["nmap"]="https://nmap.org/dist/"
    ["nikto"]="https://api.github.com/repos/sullo/nikto/releases/latest"
    ["gobuster"]="https://api.github.com/repos/OJ/gobuster/releases/latest"
)

# --- Colors ---
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

# --- Logging ---
log() {
    local level="$1"
    shift
    local message="$*"
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo -e "${timestamp} [${level}] ${message}" | tee -a "${LOG_FILE}" 2>/dev/null || true

    case "${level}" in
        "INFO")  echo -e "${GREEN}[INFO]${NC}  ${message}" ;;
        "WARN")  echo -e "${YELLOW}[WARN]${NC}  ${message}" ;;
        "ERROR") echo -e "${RED}[ERROR]${NC} ${message}" >&2 ;;
        "DEBUG") echo -e "${CYAN}[DEBUG]${NC} ${message}" ;;
        "STEP")  echo -e "${BLUE}[STEP]${NC}  ${message}" ;;
    esac
}

# --- Cleanup ---
cleanup() {
    log "INFO" "Cleaning up temporary files..."
    rm -rf "${TMP_DIR}"
}
trap cleanup EXIT INT TERM

# --- Helper Functions ---
check_dependencies() {
    log "STEP" "Checking dependencies..."
    local deps=("curl" "wget" "tar" "unzip" "jq" "sha256sum")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "${dep}" &>/dev/null; then
            missing+=("${dep}")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log "WARN" "Missing optional dependencies: ${missing[*]}"
        log "INFO" "Attempting to install missing dependencies..."
        install_dependencies "${missing[@]}"
    else
        log "INFO" "All dependencies satisfied."
    fi
}

install_dependencies() {
    local packages=("$@")

    if command -v apt-get &>/dev/null; then
        sudo apt-get update -qq
        sudo apt-get install -y -qq "${packages[@]}" 2>/dev/null || true
    elif command -v yum &>/dev/null; then
        sudo yum install -y -q "${packages[@]}" 2>/dev/null || true
    elif command -v dnf &>/dev/null; then
        sudo dnf install -y -q "${packages[@]}" 2>/dev/null || true
    elif command -v pacman &>/dev/null; then
        sudo pacman -S --noconfirm --quiet "${packages[@]}" 2>/dev/null || true
    elif command -v brew &>/dev/null; then
        brew install "${packages[@]}" 2>/dev/null || true
    else
        log "WARN" "Package manager not found. Please install manually: ${packages[*]}"
    fi
}

detect_os() {
    local os=""
    local arch=""

    case "$(uname -s)" in
        Linux*)   os="linux" ;;
        Darwin*)  os="darwin" ;;
        CYGWIN*|MINGW*|MSYS*) os="windows" ;;
        *)        os="unknown" ;;
    esac

    case "$(uname -m)" in
        x86_64|amd64)  arch="amd64" ;;
        aarch64|arm64) arch="arm64" ;;
        armv7l)        arch="arm" ;;
        i386|i686)     arch="386" ;;
        *)             arch="unknown" ;;
    esac

    echo "${os}:${arch}"
}

check_existing_installation() {
    local tool="$1"
    if command -v "${tool}" &>/dev/null; then
        local current_version
        current_version=$("${tool}" --version 2>/dev/null | head -1 || echo "unknown")
        log "INFO" "Found existing installation: ${current_version}"
        return 0
    fi
    return 1
}

get_latest_version_github() {
    local api_url="$1"
    local version

    if command -v curl &>/dev/null; then
        version=$(curl -fsSL \
            -H "Accept: application/vnd.github.v3+json" \
            "${api_url}" | jq -r '.tag_name' 2>/dev/null)
    elif command -v wget &>/dev/null; then
        version=$(wget -qO- "${api_url}" | jq -r '.tag_name' 2>/dev/null)
    fi

    echo "${version}"
}

download_file() {
    local url="$1"
    local output="$2"
    local description="${3:-file}"

    log "INFO" "Downloading ${description}..."
    log "DEBUG" "URL: ${url}"

    if command -v curl &>/dev/null; then
        curl -fsSL --progress-bar \
            --retry 3 \
            --retry-delay 2 \
            --connect-timeout 30 \
            -o "${output}" \
            "${url}" || {
            log "ERROR" "Download failed with curl"
            return 1
        }
    elif command -v wget &>/dev/null; then
        wget -q \
            --tries=3 \
            --timeout=30 \
            -O "${output}" \
            "${url}" || {
            log "ERROR" "Download failed with wget"
            return 1
        }
    else
        log "ERROR" "Neither curl nor wget is available"
        return 1
    fi

    log "INFO" "Download completed: $(du -sh "${output}" | cut -f1)"
}

verify_checksum() {
    local file="$1"
    local checksum_file="$2"
    local filename
    filename=$(basename "${file}")

    log "STEP" "Verifying checksum..."

    if [[ ! -f "${checksum_file}" ]]; then
        log "WARN" "Checksum file not found, skipping verification"
        return 0
    fi

    if grep -q "${filename}" "${checksum_file}"; then
        local expected_checksum
        expected_checksum=$(grep "${filename}" "${checksum_file}" | awk '{print $1}')
        local actual_checksum
        actual_checksum=$(sha256sum "${file}" | awk '{print $1}')

        if [[ "${expected_checksum}" == "${actual_checksum}" ]]; then
            log "INFO" "Checksum verification passed ✓"
            return 0
        else
            log "ERROR" "Checksum mismatch!"
            log "ERROR" "Expected: ${expected_checksum}"
            log "ERROR" "Actual:   ${actual_checksum}"
            return 1
        fi
    else
        log "WARN" "Filename not found in checksum file, skipping"
        return 0
    fi
}

extract_archive() {
    local archive="$1"
    local dest_dir="$2"

    log "STEP" "Extracting archive..."

    case "${archive}" in
        *.tar.gz|*.tgz)
            tar -xzf "${archive}" -C "${dest_dir}" || return 1
            ;;
        *.tar.bz2)
            tar -xjf "${archive}" -C "${dest_dir}" || return 1
            ;;
        *.tar.xz)
            tar -xJf "${archive}" -C "${dest_dir}" || return 1
            ;;
        *.zip)
            unzip -q "${archive}" -d "${dest_dir}" || return 1
            ;;
        *.deb)
            sudo dpkg -i "${archive}" || return 1
            return 0
            ;;
        *.rpm)
            sudo rpm -i "${archive}" || return 1
            return 0
            ;;
        *)
            log "WARN" "Unknown archive format, treating as binary"
            cp "${archive}" "${dest_dir}/"
            ;;
    esac

    log "INFO" "Extraction completed"
}

install_binary() {
    local source_dir="$1"
    local binary_name="$2"
    local install_path="${INSTALL_DIR}/${binary_name}"

    log "STEP" "Installing binary..."

    # Find the binary in extracted directory
    local binary_path
    binary_path=$(find "${source_dir}" -name "${binary_name}" -type f 2>/dev/null | head -1)

    if [[ -z "${binary_path}" ]]; then
        # Try without extension for Windows binaries
        binary_path=$(find "${source_dir}" -name "${binary_name}.exe" -type f 2>/dev/null | head -1)
    fi

    if [[ -z "${binary_path}" ]]; then
        # Check if the archive itself is the binary
        binary_path="${source_dir}/${binary_name}"
        if [[ ! -f "${binary_path}" ]]; then
            log "ERROR" "Binary '${binary_name}' not found in extracted files"
            find "${source_dir}" -type f | head -20
            return 1
        fi
    fi

    chmod +x "${binary_path}"

    if [[ -w "${INSTALL_DIR}" ]]; then
        cp "${binary_path}" "${install_path}"
    else
        sudo cp "${binary_path}" "${install_path}"
        sudo chmod +x "${install_path}"
    fi

    log "INFO" "Binary installed to: ${install_path}"
}

verify_installation() {
    local tool="$1"

    log "STEP" "Verifying installation..."

    if command -v "${tool}" &>/dev/null; then
        local version
        version=$("${tool}" --version 2>/dev/null | head -1 || \
                  "${tool}" version 2>/dev/null | head -1 || \
                  echo "installed")
        log "INFO" "Installation verified: ${version}"
        return 0
    else
        log "ERROR" "Installation verification failed: '${tool}' not found in PATH"
        return 1
    fi
}

# ============================================================
# Tool-Specific Installers
# ============================================================

install_trivy() {
    log "STEP" "Installing Trivy (Vulnerability Scanner)..."

    local os_arch
    os_arch=$(detect_os)
    local os="${os_arch%%:*}"
    local arch="${os_arch##*:}"

    local api_url="${TOOL_URLS[trivy]}"
    local version
    version=$(get_latest_version_github "${api_url}")

    if [[ -z "${version}" || "${version}" == "null" ]]; then
        log "ERROR" "Failed to get latest Trivy version"
        return 1
    fi

    log "INFO" "Latest Trivy version: ${version}"
    local version_num="${version#v}"

    # Construct download URL
    local archive_name="trivy_${version_num}_${os^}-${arch}.tar.gz"
    if [[ "${arch}" == "amd64" ]]; then
        archive_name="trivy_${version_num}_${os^}-64bit.tar.gz"
    elif [[ "${arch}" == "arm64" ]]; then
        archive_name="trivy_${version_num}_${os^}-ARM64.tar.gz"
    fi

    local base_url="https://github.com/aquasecurity/trivy/releases/download/${version}"
    local download_url="${base_url}/${archive_name}"
    local checksum_url="${base_url}/trivy_${version_num}_checksums.txt"

    local archive_path="${TMP_DIR}/${archive_name}"
    local checksum_path="${TMP_DIR}/checksums.txt"

    download_file "${download_url}" "${archive_path}" "Trivy ${version}"

    if [[ "${CHECKSUM_VERIFY}" == "true" ]]; then
        download_file "${checksum_url}" "${checksum_path}" "checksums" || true
        verify_checksum "${archive_path}" "${checksum_path}" || return 1
    fi

    local extract_dir="${TMP_DIR}/trivy_extracted"
    mkdir -p "${extract_dir}"
    extract_archive "${archive_path}" "${extract_dir}"
    install_binary "${extract_dir}" "trivy"
    verify_installation "trivy"
}

install_nikto() {
    log "STEP" "Installing Nikto (Web Server Scanner)..."

    # Nikto is a Perl script, handle differently
    local api_url="${TOOL_URLS[nikto]}"
    local version
    version=$(get_latest_version_github "${api_url}")

    if [[ -z "${version}" || "${version}" == "null" ]]; then
        log "ERROR" "Failed to get latest Nikto version"
        return 1
    fi

    log "INFO" "Latest Nikto version: ${version}"

    # Check for Perl
    if ! command -v perl &>/dev/null; then
        log "ERROR" "Perl is required for Nikto"
        install_dependencies "perl"
    fi

    local archive_name="nikto-${version#v}.tar.gz"
    local download_url="https://github.com/sullo/nikto/archive/refs/tags/${version}.tar.gz"
    local archive_path="${TMP_DIR}/${archive_name}"

    download_file "${download_url}" "${archive_path}" "Nikto ${version}"

    local extract_dir="${TMP_DIR}/nikto_extracted"
    mkdir -p "${extract_dir}"
    extract_archive "${archive_path}" "${extract_dir}"

    # Find nikto.pl
    local nikto_pl
    nikto_pl=$(find "${extract_dir}" -name "nikto.pl" -type f | head -1)

    if [[ -n "${nikto_pl}" ]]; then
        chmod +x "${nikto_pl}"
        # Create wrapper script
        local wrapper="${TMP_DIR}/nikto"
        cat > "${wrapper}" << EOF
#!/bin/bash
exec perl /opt/nikto/nikto.pl "\$@"
EOF
        chmod +x "${wrapper}"

        # Install nikto files
        sudo mkdir -p /opt/nikto
        sudo cp -r "$(dirname "${nikto_pl}")"/* /opt/nikto/

        if [[ -w "${INSTALL_DIR}" ]]; then
            cp "${wrapper}" "${INSTALL_DIR}/nikto"
        else
            sudo cp "${wrapper}" "${INSTALL_DIR}/nikto"
        fi

        log "INFO" "Nikto installed to /opt/nikto"
    else
        log "ERROR" "nikto.pl not found in extracted files"
        return 1
    fi
}

install_nmap() {
    log "STEP" "Installing Nmap (Network Scanner)..."

    # Use package manager for nmap
    if command -v apt-get &>/dev/null; then
        sudo apt-get update -qq
        sudo apt-get install -y nmap
    elif command -v yum &>/dev/null; then
        sudo yum install -y nmap
    elif command -v dnf &>/dev/null; then
        sudo dnf install -y nmap
    elif command -v pacman &>/dev/null; then
        sudo pacman -S --noconfirm nmap
    elif command -v brew &>/dev/null; then
        brew install nmap
    else
        log "ERROR" "Cannot install nmap: no supported package manager found"
        return 1
    fi

    verify_installation "nmap"
}

install_all_tools() {
    log "STEP" "Installing all security tools..."
    local failed_tools=()

    for tool in "${!TOOL_URLS[@]}"; do
        log "INFO" "Processing: ${tool}"
        if ! "install_${tool}" 2>/dev/null; then
            log "WARN" "Failed to install: ${tool}"
            failed_tools+=("${tool}")
        fi
        echo ""
    done

    if [[ ${#failed_tools[@]} -gt 0 ]]; then
        log "WARN" "Failed to install: ${failed_tools[*]}"
    else
        log "INFO" "All tools installed successfully!"
    fi
}

# ============================================================
# Main Function
# ============================================================

show_banner() {
    echo -e "${BLUE}"
    echo "╔══════════════════════════════════════════════════════╗"
    echo "║         Security Tool Auto-Installer v1.0            ║"
    echo "║                                                      ║"
    echo "║  Supported Tools:                                    ║"
    echo "║  • trivy    - Container/Filesystem Vulnerability     ║"
    echo "║  • nuclei   - Fast Vulnerability Scanner             ║"
    echo "║  • gobuster - Directory/DNS Brute-forcer             ║"
    echo "║  • nikto    - Web Server Scanner                     ║"
    echo "║  • nmap     - Network Discovery & Security Auditing  ║"
    echo "╚══════════════════════════════════════════════════════╝"
    echo -e "${NC}"
}

show_usage() {
    echo "Usage: $0 [OPTIONS] [TOOL]"
    echo ""
    echo "Options:"
    echo "  -h, --help              Show this help message"
    echo "  -l, --list              List available tools"
    echo "  -a, --all               Install all tools"
    echo "  -d, --dir DIR           Set installation directory (default: /usr/local/bin)"
    echo "  -n, --no-verify         Skip checksum verification"
    echo "  -v, --verbose           Enable verbose output"
    echo ""
    echo "Tools:"
    for tool in "${!TOOL_URLS[@]}"; do
        echo "  ${tool}"
    done
    echo ""
    echo "Examples:"
    echo "  $0 trivy                Install Trivy"
    echo "  $0 --all                Install all tools"
    echo "  $0 -d /usr/bin nuclei   Install Nuclei to /usr/bin"
    echo "  TOOL_NAME=gobuster $0   Install using environment variable"
}

list_tools() {
    echo -e "${CYAN}Available Security Tools:${NC}"
    echo ""
    printf "%-15s %-50s\n" "Tool" "Description"
    printf "%-15s %-50s\n" "----" "-----------"
    printf "%-15s %-50s\n" "trivy"    "Container & filesystem vulnerability scanner"
    printf "%-15s %-50s\n" "nuclei"   "Fast & customizable vulnerability scanner"
    printf "%-15s %-50s\n" "gobuster" "Directory/file & DNS brute-forcer"
    printf "%-15s %-50s\n" "nikto"    "Web server scanner"
    printf "%-15s %-50s\n" "nmap"     "Network discovery & security auditing"
    echo ""
}

main() {
    show_banner

    # Parse arguments
    local install_all=false
    local selected_tool="${TOOL_NAME:-}"
    local verbose=false

    while [[ $# -gt 0 ]]; do
        case "$1" in
            -h|--help)
                show_usage
                exit 0
                ;;
            -l|--list)
                list_tools
                exit 0
                ;;
            -a|--all)
                install_all=true
                shift
                ;;
            -d|--dir)
                INSTALL_DIR="$2"
                shift 2
                ;;
            -n|--no-verify)
                CHECKSUM_VERIFY="false"
                shift
                ;;
            -v|--verbose)
                verbose=true
                set -x
                shift
                ;;
            -*)
                log "ERROR" "Unknown option: $1"
                show_usage
                exit 1
                ;;
            *)
                selected_tool="$1"
                shift
                ;;
        esac
    done

    # Setup
    log "INFO" "Security Tool Installer starting..."
    log "INFO" "Install directory: ${INSTALL_DIR}"
    log "INFO" "OS/Arch: $(detect_os)"
    log "INFO" "Log file: ${LOG_FILE}"

    # Create log file
    touch "${LOG_FILE}" 2>/dev/null || LOG_FILE="${TMP_DIR}/installer.log"

    # Check dependencies
    check_dependencies

    # Ensure install directory exists
    if [[ ! -d "${INSTALL_DIR}" ]]; then
        log "INFO" "Creating install directory: ${INSTALL_DIR}"
        sudo mkdir -p "${INSTALL_DIR}"
    fi

    # Check if install directory is in PATH
    if [[ ":${PATH}:" != *":${INSTALL_DIR}:"* ]]; then
        log "WARN" "${INSTALL_DIR} is not in PATH"
        log "WARN" "Add to PATH: export PATH=\"\$PATH:${INSTALL_DIR}\""
    fi

    echo ""

    # Install tools
    if [[ "${install_all}" == "true" ]]; then
        install_all_tools
    elif [[ -n "${selected_tool}" ]]; then
        if [[ -v "TOOL_URLS[${selected_tool}]" ]]; then
            # Check existing installation
            if check_existing_installation "${selected_tool}"; then
                log "WARN" "Tool '${selected_tool}' is already installed"
                read -r -p "Do you want to reinstall/update? [y/N] " response
                if [[ ! "${response}" =~ ^[Yy]$ ]]; then
                    log "INFO" "Installation cancelled"
                    exit 0
                fi
            fi

            "install_${selected_tool}"
        else
            log "ERROR" "Unknown tool: ${selected_tool}"
            log "INFO" "Available tools: ${!TOOL_URLS[*]}"
            show_usage
            exit 1
        fi
    else
        log "WARN" "No tool specified"
        show_usage
        exit 1
    fi

    echo ""
    log "INFO" "Installation process completed!"
    log "INFO" "Log saved to: ${LOG_FILE}"
}

main "$@"
