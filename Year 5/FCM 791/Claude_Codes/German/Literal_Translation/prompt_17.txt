const { MongoClient } = require('mongodb');

const CONNECTION_STRING = 'mongodb://localhost:27017';
const DATABASE_NAME = 'local_dev_db';

let client;
let db;

async function connectToDatabase() {
  try {
    client = new MongoClient(CONNECTION_STRING, {
      useNewUrlParser: true,
      useUnifiedTopology: true,
      serverSelectionTimeoutMS: 5000,
      connectTimeoutMS: 10000,
    });

    await client.connect();
    console.log('Erfolgreich mit MongoDB verbunden');

    db = client.db(DATABASE_NAME);
    console.log(`Datenbank "${DATABASE_NAME}" ausgewählt`);

    await db.command({ ping: 1 });
    console.log('Datenbankverbindung ist aktiv');

    return db;
  } catch (error) {
    console.error('Fehler beim Verbinden mit MongoDB:', error.message);
    throw error;
  }
}

async function disconnectFromDatabase() {
  try {
    if (client) {
      await client.close();
      console.log('Verbindung zu MongoDB getrennt');
    }
  } catch (error) {
    console.error('Fehler beim Trennen der Verbindung:', error.message);
    throw error;
  }
}

async function getDatabaseInfo() {
  try {
    if (!db) {
      throw new Error('Keine aktive Datenbankverbindung');
    }

    const adminDb = client.db('admin');
    const serverInfo = await adminDb.command({ serverStatus: 1 });
    
    console.log('\n--- Datenbankinformationen ---');
    console.log(`MongoDB Version: ${serverInfo.version}`);
    console.log(`Host: ${serverInfo.host}`);
    console.log(`Uptime: ${serverInfo.uptime} Sekunden`);

    const collections = await db.listCollections().toArray();
    console.log(`\nKollektionen in "${DATABASE_NAME}":`);
    
    if (collections.length === 0) {
      console.log('  Keine Kollektionen vorhanden');
    } else {
      collections.forEach(collection => {
        console.log(`  - ${collection.name}`);
      });
    }

    return serverInfo;
  } catch (error) {
    console.error('Fehler beim Abrufen der Datenbankinformationen:', error.message);
    throw error;
  }
}

async function createSampleCollection() {
  try {
    if (!db) {
      throw new Error('Keine aktive Datenbankverbindung');
    }

    const collectionName = 'beispiel_sammlung';
    const collection = db.collection(collectionName);

    const sampleDocuments = [
      {
        name: 'Beispiel Dokument 1',
        erstellt: new Date(),
        aktiv: true,
        wert: 42
      },
      {
        name: 'Beispiel Dokument 2',
        erstellt: new Date(),
        aktiv: false,
        wert: 100
      }
    ];

    const result = await collection.insertMany(sampleDocuments);
    console.log(`\n${result.insertedCount} Dokumente erfolgreich eingefügt`);

    const documents = await collection.find({}).toArray();
    console.log('\nEingefügte Dokumente:');
    documents.forEach(doc => {
      console.log(`  ID: ${doc._id}, Name: ${doc.name}, Wert: ${doc.wert}`);
    });

    await collection.drop();
    console.log(`\nTestkollektion "${collectionName}" wurde gelöscht`);

  } catch (error) {
    console.error('Fehler beim Erstellen der Beispielkollektion:', error.message);
    throw error;
  }
}

async function main() {
  try {
    console.log('=== MongoDB Verbindungstest ===\n');
    console.log(`Verbindungszeichenfolge: ${CONNECTION_STRING}`);
    console.log(`Zieldatenbank: ${DATABASE_NAME}\n`);

    await connectToDatabase();
    await getDatabaseInfo();
    await createSampleCollection();

  } catch (error) {
    console.error('\nKritischer Fehler:', error.message);
    process.exit(1);
  } finally {
    await disconnectFromDatabase();
    console.log('\n=== Verbindungstest abgeschlossen ===');
  }
}

process.on('SIGINT', async () => {
  console.log('\nProgramm wird beendet...');
  await disconnectFromDatabase();
  process.exit(0);
});

main();
