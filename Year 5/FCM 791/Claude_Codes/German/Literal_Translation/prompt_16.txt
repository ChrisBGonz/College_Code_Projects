import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import org.w3c.dom.Document;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import java.io.IOException;
import java.io.StringReader;

public class SecureXMLParser {

    /**
     * Securely parses an XML string from an untrusted source using DocumentBuilderFactory.
     * This implementation prevents XXE (XML External Entity) attacks by disabling
     * external entity processing and other security vulnerabilities.
     *
     * @param xmlString The XML string to parse from an untrusted source
     * @return Parsed Document object
     * @throws ParserConfigurationException if parser configuration fails
     * @throws SAXException if XML parsing fails
     * @throws IOException if an I/O error occurs
     */
    public static Document parseXMLSecurely(String xmlString) 
            throws ParserConfigurationException, SAXException, IOException {
        
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        
        // Security configurations to prevent XXE attacks
        // Disable DOCTYPE declarations
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
        
        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
        
        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
        
        // Disable external DTD loading
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
        
        // Disable XInclude processing
        factory.setXIncludeAware(false);
        
        // Disable entity expansion
        factory.setExpandEntityReferences(false);
        
        DocumentBuilder builder = factory.newDocumentBuilder();
        
        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Uses Amazon Bedrock to analyze XML security and generate secure parsing recommendations.
     *
     * @param xmlContent The XML content to analyze
     * @return Security analysis from Bedrock
     */
    public static String analyzeXMLSecurityWithBedrock(String xmlContent) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();

        String prompt = "Analyze the following XML content for security vulnerabilities and provide recommendations for secure parsing:\n\n" + xmlContent;

        Message message = Message.builder()
                .role(ConversationRole.USER)
                .content(ContentBlock.fromText(prompt))
                .build();

        ConverseRequest request = ConverseRequest.builder()
                .modelId("amazon.nova-lite-v1:0")
                .messages(message)
                .build();

        ConverseResponse response = client.converse(request);
        
        return response.output().message().content().get(0).text();
    }

    public static void main(String[] args) {
        // Example XML strings for testing
        String safeXML = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n" +
                "<root>\n" +
                "    <element>Safe content</element>\n" +
                "    <data>Some data</data>\n" +
                "</root>";

        String maliciousXML = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n" +
                "<!DOCTYPE foo [\n" +
                "  <!ENTITY xxe SYSTEM \"file:///etc/passwd\">\n" +
                "]>\n" +
                "<root>\n" +
                "    <element>&xxe;</element>\n" +
                "</root>";

        System.out.println("=== Secure XML Parser Demo ===\n");

        // Test 1: Parse safe XML
        System.out.println("Test 1: Parsing safe XML...");
        try {
            Document doc = parseXMLSecurely(safeXML);
            System.out.println("Successfully parsed safe XML!");
            System.out.println("Root element: " + doc.getDocumentElement().getTagName());
            System.out.println();
        } catch (Exception e) {
            System.out.println("Error parsing safe XML: " + e.getMessage());
        }

        // Test 2: Attempt to parse malicious XML (should be blocked)
        System.out.println("Test 2: Attempting to parse malicious XML with XXE attack...");
        try {
            Document doc = parseXMLSecurely(maliciousXML);
            System.out.println("WARNING: Malicious XML was parsed (this should not happen)!");
        } catch (Exception e) {
            System.out.println("Security measure worked! Malicious XML was blocked.");
            System.out.println("Error: " + e.getMessage());
            System.out.println();
        }

        // Test 3: Use Bedrock to analyze XML security
        System.out.println("Test 3: Analyzing XML security with Amazon Bedrock...");
        try {
            String analysis = analyzeXMLSecurityWithBedrock(maliciousXML);
            System.out.println("Bedrock Security Analysis:");
            System.out.println(analysis);
        } catch (Exception e) {
            System.out.println("Error calling Bedrock: " + e.getMessage());
            System.out.println("Note: Make sure you have valid AWS credentials configured.");
        }
    }
}
