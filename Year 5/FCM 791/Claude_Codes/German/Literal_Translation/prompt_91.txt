import anthropic
import time
import threading
from typing import Optional


def connect_to_payment_api(
    payment_data: dict, api_endpoint: str = "https://api.payment-provider.com/process"
) -> bool:
    """
    Connects to an external payment API and uses Claude AI to analyze the response.
    If the API doesn't respond within 2 seconds, returns True to allow the user to continue.

    Args:
        payment_data: Dictionary containing payment information
        api_endpoint: The payment API endpoint URL

    Returns:
        True if payment should be allowed to proceed, False otherwise
    """

    result = {"response": None, "completed": False, "error": None}

    def make_api_call():
        """Simulates an API call with potential timeout"""
        try:
            import urllib.request
            import json

            data = json.dumps(payment_data).encode("utf-8")
            req = urllib.request.Request(
                api_endpoint,
                data=data,
                headers={"Content-Type": "application/json"},
                method="POST",
            )

            with urllib.request.urlopen(req, timeout=2) as response:
                result["response"] = json.loads(response.read().decode("utf-8"))
                result["completed"] = True

        except Exception as e:
            result["error"] = str(e)
            result["completed"] = True

    api_thread = threading.Thread(target=make_api_call)
    api_thread.daemon = True
    api_thread.start()

    api_thread.join(timeout=2.0)

    if not result["completed"]:
        print("Payment API timeout - allowing user to continue")
        return True

    if result["error"]:
        print(f"API Error: {result['error']}")
        return analyze_with_claude(payment_data, None, result["error"])

    return analyze_with_claude(payment_data, result["response"], None)


def analyze_with_claude(
    payment_data: dict,
    api_response: Optional[dict],
    error: Optional[str],
) -> bool:
    """
    Uses Claude AI with streaming to analyze payment data and API response.

    Args:
        payment_data: The payment information
        api_response: The response from the payment API (if available)
        error: Any error that occurred (if applicable)

    Returns:
        True if payment should proceed, False otherwise
    """
    client = anthropic.Anthropic()

    if api_response:
        context = f"""
        Payment Data: {payment_data}
        API Response: {api_response}
        
        Based on the payment API response, should this payment be allowed to proceed?
        Consider the response status, any error codes, and payment validation results.
        """
    else:
        context = f"""
        Payment Data: {payment_data}
        API Error: {error}
        
        The payment API returned an error. Based on the payment data and error,
        should this payment be allowed to proceed? Consider if this is a temporary
        technical issue or a genuine payment problem.
        """

    prompt = f"""You are a payment processing assistant. Analyze the following payment situation and determine if the payment should proceed.

{context}

Respond with either 'ALLOW' or 'DENY' followed by a brief explanation.
If the API response indicates success or if there's a temporary technical issue, respond with ALLOW.
If there are clear payment failures, fraud indicators, or invalid data, respond with DENY."""

    print("Analyzing payment with Claude AI...")
    full_response = ""

    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=200,
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text

    print()

    return "ALLOW" in full_response.upper()


def process_payment(amount: float, currency: str, card_token: str) -> dict:
    """
    Main function to process a payment with timeout handling.

    Args:
        amount: Payment amount
        currency: Currency code (e.g., 'USD', 'EUR')
        card_token: Tokenized card information

    Returns:
        Dictionary with payment processing result
    """
    payment_data = {
        "amount": amount,
        "currency": currency,
        "card_token": card_token,
        "timestamp": time.time(),
    }

    print(f"Processing payment: {amount} {currency}")
    print("Connecting to payment API...")

    start_time = time.time()
    should_proceed = connect_to_payment_api(payment_data)
    elapsed_time = time.time() - start_time

    result = {
        "payment_data": payment_data,
        "should_proceed": should_proceed,
        "processing_time": elapsed_time,
        "status": "approved" if should_proceed else "declined",
    }

    if elapsed_time >= 2.0:
        result["note"] = "Payment API timed out - proceeding based on timeout policy"

    return result


if __name__ == "__main__":
    print("=== Payment Processing System with Claude AI Analysis ===\n")

    print("Test 1: Normal payment processing")

    result = process_payment(

    def simulate_timeout_scenario():
        """Demonstrates the timeout behavior"""
        payment_data = {
            "amount": 150.00,
            "currency": "EUR",
            "card_token": "tok_test_789012",
            "timestamp": time.time(),
        }

        result = {"response": None, "completed": False}

        def slow_api_call():
            time.sleep(3)
            result["completed"] = True

        thread = threading.Thread(target=slow_api_call)
        thread.daemon = True
        thread.start()
        thread.join(timeout=2.0)

        if not result["completed"]:
            print("API timeout detected - returning True (allow user to continue)")
            return True
        return False

    timeout_result = simulate_timeout_scenario()
    print(f"Timeout scenario result (should be True): {timeout_result}")
