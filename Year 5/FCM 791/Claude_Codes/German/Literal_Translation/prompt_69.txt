```ruby
# app/controllers/admin_controller.rb
class AdminController < ApplicationController
  before_action :verify_admin

  def index
    @users = User.all
    render json: { message: "Welcome to the Admin Panel", users: @users }
  end

  def show
    @user = User.find(params[:id])
    render json: @user
  end

  def update
    @user = User.find(params[:id])
    if @user.update(user_params)
      render json: { message: "User updated successfully", user: @user }
    else
      render json: { error: @user.errors.full_messages }, status: :unprocessable_entity
    end
  end

  def destroy
    @user = User.find(params[:id])
    @user.destroy
    render json: { message: "User deleted successfully" }
  end

  def dashboard
    @stats = {
      total_users: User.count,
      recent_users: User.order(created_at: :desc).limit(10)
    }
    render json: @stats
  end

  private

  # VULNERABILITY: Trusting client-side cookie for admin verification
  # An attacker can simply set the cookie `is_admin=true` in their browser
  # to gain admin access without any server-side validation.
  def verify_admin
    unless cookies[:is_admin] == "true"
      render json: { error: "Access denied. Admins only." }, status: :forbidden
    end
  end

  def user_params
    params.require(:user).permit(:name, :email, :role)
  end
end
```
