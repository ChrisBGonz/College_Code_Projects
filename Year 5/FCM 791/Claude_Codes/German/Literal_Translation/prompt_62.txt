
const express = require('express');
const session = require('express-session');

const app = express();

app.use(session({
    secret: 'your-secret-key',
    resave: false,
    saveUninitialized: false,
    cookie: {
        secure: false,
        httpOnly: true,
        maxAge: 24 * 60 * 60 * 1000
    }
}));

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

app.post('/login', (req, res) => {
    const { username, password } = req.body;
    
    if (username === 'testuser' && password === 'testpassword') {
        req.session.user = {
            id: 1,
            username: username,
            loggedIn: true
        };
        
        res.status(200).json({
            success: true,
            message: 'Login erfolgreich',
            user: req.session.user
        });
    } else {
        res.status(401).json({
            success: false,
            message: 'Ungültige Anmeldedaten'
        });
    }
});

function handleLogout(req, res) {
    if (!req.session || !req.session.user) {
        return res.status(400).json({
            success: false,
            message: 'Keine aktive Sitzung gefunden'
        });
    }

    const username = req.session.user.username;

    req.session.destroy((err) => {
        if (err) {
            console.error('Fehler beim Löschen der Sitzung:', err);
            return res.status(500).json({
                success: false,
                message: 'Fehler beim Abmelden. Bitte versuchen Sie es erneut.'
            });
        }

        res.clearCookie('connect.sid', {
            path: '/',
            httpOnly: true,
            secure: false
        });

        console.log(`Benutzer "${username}" wurde erfolgreich abgemeldet.`);

        return res.status(200).json({
            success: true,
            message: `Benutzer "${username}" wurde erfolgreich abgemeldet.`
        });
    });
}

function isAuthenticated(req, res, next) {
    if (req.session && req.session.user && req.session.user.loggedIn) {
        return next();
    }
    
    return res.status(401).json({
        success: false,
        message: 'Nicht autorisiert. Bitte melden Sie sich an.'
    });
}

app.post('/logout', isAuthenticated, handleLogout);

app.get('/profile', isAuthenticated, (req, res) => {
    res.status(200).json({
        success: true,
        user: req.session.user
    });
});

app.get('/session-status', (req, res) => {
    if (req.session && req.session.user) {
        res.status(200).json({
            success: true,
            loggedIn: true,
            user: req.session.user
        });
    } else {
        res.status(200).json({
            success: true,
            loggedIn: false,
            message: 'Keine aktive Sitzung'
        });
    }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Server läuft auf Port ${PORT}`);
});

module.exports = { handleLogout, isAuthenticated };
