package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"time"
)

func callMicroserviceWithRetry(url string) (string, error) {
	attempt := 0

	for {
		attempt++
		log.Printf("Attempt %d: Calling microservice at %s", attempt, url)

		resp, err := http.Get(url)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying immediately...", attempt, err)
			continue
		}

		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			log.Printf("Attempt %d failed with status code: %d. Retrying immediately...", attempt, resp.StatusCode)
			continue
		}

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			log.Printf("Attempt %d failed to read response body: %v. Retrying immediately...", attempt, err)
			continue
		}

		log.Printf("Attempt %d succeeded after %d attempts", attempt, attempt)
		return string(body), nil
	}
}

func callMicroserviceWithRetryAndDelay(url string, delay time.Duration, maxAttempts int) (string, error) {
	attempt := 0

	for {
		attempt++

		if maxAttempts > 0 && attempt > maxAttempts {
			return "", fmt.Errorf("max attempts (%d) reached without success", maxAttempts)
		}

		log.Printf("Attempt %d: Calling microservice at %s", attempt, url)

		resp, err := http.Get(url)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying in %v...", attempt, err, delay)
			time.Sleep(delay)
			continue
		}

		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			log.Printf("Attempt %d failed with status code: %d. Retrying in %v...", attempt, resp.StatusCode, delay)
			time.Sleep(delay)
			continue
		}

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			log.Printf("Attempt %d failed to read response body: %v. Retrying in %v...", attempt, err, delay)
			time.Sleep(delay)
			continue
		}

		log.Printf("Attempt %d succeeded", attempt)
		return string(body), nil
	}
}

func main() {
	url := "http://localhost:8080/api/service"

	log.Println("Starting microservice call with immediate retry...")
	result, err := callMicroserviceWithRetry(url)
	if err != nil {
		log.Fatalf("Failed to call microservice: %v", err)
	}
	fmt.Printf("Result: %s\n", result)

	log.Println("Starting microservice call with delay and max attempts...")
	result, err = callMicroserviceWithRetryAndDelay(url, 2*time.Second, 10)
	if err != nil {
		log.Fatalf("Failed to call microservice: %v", err)
	}
	fmt.Printf("Result: %s\n", result)
}
