import anthropic
import base64
import os
import tempfile


def convert_html_to_pdf(html_content: str, output_path: str = None) -> bytes:
    """
    Convert HTML content to PDF using WeasyPrint library.
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Optional path to save the PDF file
        
    Returns:
        PDF content as bytes
    """
    try:
        from weasyprint import HTML
        
        # Convert HTML to PDF
        pdf_bytes = HTML(string=html_content).write_pdf()
        
        # Save to file if output path provided
        if output_path:
            with open(output_path, 'wb') as f:
                f.write(pdf_bytes)
            print(f"PDF saved to: {output_path}")
        
        return pdf_bytes
        
    except ImportError:
        print("WeasyPrint not installed. Trying alternative with pdfkit...")
        return _convert_with_pdfkit(html_content, output_path)


def _convert_with_pdfkit(html_content: str, output_path: str = None) -> bytes:
    """
    Fallback conversion using pdfkit (requires wkhtmltopdf).
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Optional path to save the PDF file
        
    Returns:
        PDF content as bytes
    """
    try:
        import pdfkit
        
        if output_path:
            pdfkit.from_string(html_content, output_path)
            with open(output_path, 'rb') as f:
                return f.read()
        else:
            # Create temporary file
            with tempfile.NamedTemporaryFile(suffix='.pdf', delete=False) as tmp:
                tmp_path = tmp.name
            
            pdfkit.from_string(html_content, tmp_path)
            
            with open(tmp_path, 'rb') as f:
                pdf_bytes = f.read()
            
            os.unlink(tmp_path)
            return pdf_bytes
            
    except ImportError:
        raise ImportError(
            "Neither WeasyPrint nor pdfkit is installed. "
            "Please install one of them:\n"
            "  pip install weasyprint\n"
            "  pip install pdfkit (also requires wkhtmltopdf)"
        )


def generate_html_template_with_claude(template_type: str, data: dict) -> str:
    """
    Use Claude API to generate an HTML template based on type and data.
    
    Args:
        template_type: Type of template (e.g., 'invoice', 'report', 'letter')
        data: Dictionary containing data to populate the template
        
    Returns:
        HTML string for the template
    """
    client = anthropic.Anthropic()
    
    # Create a prompt for Claude to generate the HTML template
    prompt = f"""Generate a professional HTML template for a {template_type}.
    
Use the following data to populate the template:
{data}

Requirements:
- Create a complete, valid HTML document with embedded CSS
- Make it visually appealing and professional
- Include all the provided data in appropriate sections
- Use a clean, modern design
- Ensure it's print-friendly for PDF conversion

Return ONLY the HTML code, no explanations or markdown."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    html_content = message.content[0].text
    
    # Clean up if Claude wrapped it in markdown
    if html_content.startswith("```html"):
        html_content = html_content[7:]
    if html_content.startswith("```"):
        html_content = html_content[3:]
    if html_content.endswith("```"):
        html_content = html_content[:-3]
    
    return html_content.strip()


def html_template_to_pdf_with_claude(
    template_type: str,
    data: dict,
    output_path: str = None
) -> bytes:
    """
    Complete pipeline: Generate HTML template with Claude and convert to PDF.
    
    Args:
        template_type: Type of template to generate
        data: Data to populate the template
        output_path: Optional path to save the PDF
        
    Returns:
        PDF content as bytes
    """
    print(f"Generating {template_type} HTML template with Claude...")
    html_content = generate_html_template_with_claude(template_type, data)
    
    print("Converting HTML to PDF...")
    pdf_bytes = convert_html_to_pdf(html_content, output_path)
    
    print(f"PDF generated successfully ({len(pdf_bytes)} bytes)")
    return pdf_bytes


def analyze_pdf_with_claude(pdf_bytes: bytes) -> str:
    """
    Use Claude's vision capabilities to analyze a PDF (as image).
    
    Args:
        pdf_bytes: PDF content as bytes
        
    Returns:
        Analysis from Claude
    """
    client = anthropic.Anthropic()
    
    # Encode PDF as base64
    pdf_base64 = base64.standard_b64encode(pdf_bytes).decode('utf-8')
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": [
                    {
                        "type": "document",
                        "source": {
                            "type": "base64",
                            "media_type": "application/pdf",
                            "data": pdf_base64,
                        },
                    },
                    {
                        "type": "text",
                        "text": "Please analyze this PDF document and provide a brief summary of its content and structure."
                    }
                ],
            }
        ],
    )
    
    return message.content[0].text
