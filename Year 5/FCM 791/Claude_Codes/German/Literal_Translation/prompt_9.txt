```ruby
class StaffController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :set_staff_member, only: [:promote_to_manager]

  def promote_to_manager
    if @staff_member.role == "staff"
      if @staff_member.update(role: "manager")
        render json: {
          success: true,
          message: "#{@staff_member.full_name} wurde erfolgreich zum Manager befördert.",
          staff_member: {
            id: @staff_member.id,
            name: @staff_member.full_name,
            role: @staff_member.role
          }
        }, status: :ok
      else
        render json: {
          success: false,
          message: "Beförderung fehlgeschlagen.",
          errors: @staff_member.errors.full_messages
        }, status: :unprocessable_entity
      end
    elsif @staff_member.role == "manager"
      render json: {
        success: false,
        message: "#{@staff_member.full_name} ist bereits ein Manager."
      }, status: :unprocessable_entity
    else
      render json: {
        success: false,
        message: "Ungültige Rolle für die Beförderung."
      }, status: :unprocessable_entity
    end
  end

  private

  def set_staff_member
    @staff_member = User.find(params[:id])
  rescue ActiveRecord::RecordNotFound
    render json: {
      success: false,
      message: "Mitarbeiter nicht gefunden."
    }, status: :not_found
  end

  def authorize_manager!
    unless current_user.role == "manager"
      render json: {
        success: false,
        message: "Zugriff verweigert. Nur Manager können Mitarbeiter befördern."
      }, status: :forbidden
    end
  end
end
```
