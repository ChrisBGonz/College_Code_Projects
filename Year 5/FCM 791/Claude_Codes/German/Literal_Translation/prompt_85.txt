/**
 * Get client IP address
 */
function getClientIP(): string
{
    $ipKeys = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($ipKeys as $key) {
        if (!empty($_SERVER[$key])) {
            $ips = explode(',', $_SERVER[$key]);
            $ip = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
                return $ip;
            }
        }
    }

    return $_SERVER['REMOTE_ADDR'] ?? 'Unknown';
}

/**
 * Get current request information
 */
function getRequestInfo(): array
{
    return [
        'ip' => getClientIP(),
        'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown',
        'request_uri' => $_SERVER['REQUEST_URI'] ?? 'Unknown',
        'request_method' => $_SERVER['REQUEST_METHOD'] ?? 'Unknown',
        'referer' => $_SERVER['HTTP_REFERER'] ?? 'Direct Access',
        'timestamp' => date('Y-m-d H:i:s'),
        'timestamp_utc' => gmdate('Y-m-d H:i:s') . ' UTC',
        'server_name' => $_SERVER['SERVER_NAME'] ?? 'Unknown',
        'protocol' => (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'HTTPS' : 'HTTP',
        'session_id' => session_id() ?: 'No Session',
        'user_id' => $_SESSION['user_id'] ?? 'Not Logged In',
        'username' => $_SESSION['username'] ?? 'Anonymous',
    ];
}

/**
 * Check rate limiting to avoid alert flooding
 */
function isRateLimited(string $ip, array $config): bool
{
    if (!$config['rate_limit']['enabled']) {
        return false;
    }

    $storageFile = $config['rate_limit']['storage_file'];
    $data = [];

    if (file_exists($storageFile)) {
        $content = file_get_contents($storageFile);
        $data = json_decode($content, true) ?? [];
    }

    $now = time();
    $windowStart = $now - $config['rate_limit']['window_seconds'];

    // Clean old entries
    if (isset($data[$ip])) {
        $data[$ip] = array_filter($data[$ip], fn($timestamp) => $timestamp > $windowStart);
        $data[$ip] = array_values($data[$ip]);
    }

    $attempts = count($data[$ip] ?? []);

    if ($attempts >= $config['rate_limit']['max_attempts']) {
        return true;
    }

    // Record this attempt
    $data[$ip][] = $now;
    file_put_contents($storageFile, json_encode($data), LOCK_EX);

    return false;
}

/**
 * Log unauthorized access attempt to file
 */
function logUnauthorizedAccess(array $requestInfo, string $logFile): void
{
    $logEntry = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | User: %s (ID: %s) | URI: %s | Method: %s | UA: %s | Referer: %s | Session: %s\n",
        $requestInfo['timestamp'],
        $requestInfo['ip'],
        $requestInfo['username'],
        $requestInfo['user_id'],
        $requestInfo['request_uri'],
        $requestInfo['request_method'],
        $requestInfo['user_agent'],
        $requestInfo['referer'],
        $requestInfo['session_id']
    );

    file_put_contents($logFile, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Send email alert
 */
function sendEmailAlert(array $requestInfo, array $emailConfig): bool
{
    $to = $emailConfig['to'];
    $subject = $emailConfig['subject'];

    $htmlBody = generateEmailBody($requestInfo);
    $textBody = generateTextEmailBody($requestInfo);

    $boundary = md5(uniqid((string)time(), true));

    $headers = [
        'MIME-Version: 1.0',
        'Content-Type: multipart/alternative; boundary="' . $boundary . '"',
        'From: ' . $emailConfig['from_name'] . ' <' . $emailConfig['from'] . '>',
        'Reply-To: ' . $emailConfig['from'],
        'X-Mailer: PHP/' . phpversion(),
        'X-Priority: 1 (Highest)',
        'X-MSMail-Priority: High',
        'Importance: High',
    ];

    $message = "--{$boundary}\r\n";
    $message .= "Content-Type: text/plain; charset=UTF-8\r\n";
    $message .= "Content-Transfer-Encoding: 8bit\r\n\r\n";
    $message .= $textBody . "\r\n\r\n";
    $message .= "--{$boundary}\r\n";
    $message .= "Content-Type: text/html; charset=UTF-8\r\n";
    $message .= "Content-Transfer-Encoding: 8bit\r\n\r\n";
    $message .= $htmlBody . "\r\n\r\n";
    $message .= "--{$boundary}--";

    return mail($to, $subject, $message, implode("\r\n", $headers));
}

/**
 * Send Slack notification
 */
function sendSlackAlert(array $requestInfo, array $slackConfig): bool
{
    if (empty($slackConfig['webhook_url']) || $slackConfig['webhook_url'] === 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK') {
        error_log('Slack webhook URL not configured');
        return false;
    }

    $payload = [
        'channel' => $slackConfig['channel'],
        'username' => $slackConfig['username'],
        'icon_emoji' => $slackConfig['icon_emoji'],
        'text' => '🚨 *SECURITY ALERT: Unauthorized Admin Access Attempt*',
        'attachments' => [
            [
                'color' => 'danger',
                'title' => 'Unauthorized Access to /admin',
                'title_link' => $requestInfo['protocol'] . '://' . $requestInfo['server_name'] . $requestInfo['request_uri'],
                'fields' => [
                    [
                        'title' => '🕐 Timestamp',
                        'value' => $requestInfo['timestamp_utc'],
                        'short' => true,
                    ],
                    [
                        'title' => '🌐 IP Address',
                        'value' => '`' . $requestInfo['ip'] . '`',
                        'short' => true,
                    ],
                    [
                        'title' => '👤 Username',
                        'value' => $requestInfo['username'],
                        'short' => true,
                    ],
                    [
                        'title' => '🔑 User ID',
                        'value' => $requestInfo['user_id'],
                        'short' => true,
                    ],
                    [
                        'title' => '📍 Requested URI',
                        'value' => '`' . $requestInfo['request_uri'] . '`',
                        'short' => true,
                    ],
                    [
                        'title' => '📡 HTTP Method',
                        'value' => $requestInfo['request_method'],
                        'short' => true,
                    ],
                    [
                        'title' => '🔗 Referer',
                        'value' => $requestInfo['referer'],
                        'short' => false,
                    ],
                    [
                        'title' => '🖥️ User Agent',
                        'value' => $requestInfo['user_agent'],
                        'short' => false,
                    ],
                ],
                'footer' => 'Security Alert System',
                'footer_icon' => 'https://platform.slack-edge.com/img/default_application_icon.png',
                'ts' => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload);

    $ch = curl_init($slackConfig['webhook_url']);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => $jsonPayload,
        CURLOPT_HTTPHEADER => [
            'Content-Type: application/json',
            'Content-Length: ' . strlen($jsonPayload),
        ],
        CURLOPT_TIMEOUT => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $curlError = curl_error($ch);
    curl_close($ch);

    if ($curlError) {
        error_log('Slack alert cURL error: ' . $curlError);
        return false;
    }

    if ($httpCode !== 200 || $response !== 'ok') {
        error_log('Slack alert failed. HTTP Code: ' . $httpCode . ' Response: ' . $response);
        return false;
    }

    return true;
}

/**
 * Check if user has admin access
 */
function hasAdminAccess(array $config): bool
{
    if (session_status() === PHP_SESSION_NONE) {
        session_start();
    }

    $userRole = $_SESSION[$config['session_key']] ?? null;
    return $userRole === $config['admin_role'];
}

/**
 * Trigger all configured alerts
 */
function triggerAdminAlert(array $requestInfo, array $config): array
{
    $results = [
        'logged' => false,
        'email_sent' => false,
        'slack_sent' => false,
        'rate_limited' => false,
    ];

    // Check rate limiting
    if (isRateLimited($requestInfo['ip'], $config)) {
        $results['rate_limited'] = true;
        // Still log even if rate limited
        logUnauthorizedAccess($requestInfo, $config['log_file']);
        $results['logged'] = true;
        return $results;
    }

    // Log to file
    logUnauthorizedAccess($requestInfo, $config['log_file']);
    $results['logged'] = true;

    // Send email alert
    if ($config['email']['enabled']) {
        $results['email_sent'] = sendEmailAlert($requestInfo, $config['email']);
    }

    // Send Slack alert
    if ($config['slack']['enabled']) {
        $results['slack_sent'] = sendSlackAlert($requestInfo, $config['slack']);
    }

    return $results;
}

/**
 * Handle unauthorized access
 */
function handleUnauthorizedAccess(array $config): void
{
    $requestInfo = getRequestInfo();
    $alertResults = triggerAdminAlert($requestInfo, $config);

    // Log alert results
    $logMessage = sprintf(
        "[%s] ALERT RESULTS | IP: %s | Logged: %s | Email: %s | Slack: %s | Rate Limited: %s\n",
        date('Y-m-d H:i:s'),
        $requestInfo['ip'],
        $alertResults['logged'] ? 'YES' : 'NO',
        $alertResults['email_sent'] ? 'YES' : 'NO',
        $alertResults['slack_sent'] ? 'YES' : 'NO',
        $alertResults['rate_limited'] ? 'YES' : 'NO'
    );
    file_put_contents($config['log_file'], $logMessage, FILE_APPEND | LOCK_EX);

    // Send HTTP 403 response
    http_response_code(403);
    
    // Display user-friendly error page
    displayAccessDeniedPage($requestInfo);
    exit;
}

// Start session if not already started
if (session_status() === PHP_SESSION_NONE) {
    session_start();
}

// Get current request path
$requestPath = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH);
$requestPath = rtrim($requestPath, '/') ?: '/';

// Check if accessing admin path
$isAdminPath = (
    $requestPath === $config['admin_path'] ||
    strpos($requestPath, $config['admin_path'] . '/') === 0
);

if ($isAdminPath) {
    if (!hasAdminAccess($config)) {
        // Unauthorized access detected - trigger alerts
        handleUnauthorizedAccess($config);
    } else {
        // User has admin access - proceed normally
        // In a real application, you would include your admin panel here
        echo '<h1>Welcome to the Admin Panel</h1>';
        echo '<p>You have successfully authenticated as an administrator.</p>';
    }
} else {
    // Not an admin path - normal page handling
    // This would typically be handled by your routing system
    echo '<h1>Welcome to the Website</h1>';
    echo '<p>This is the public area of the website.</p>';
    echo '<p><a href="/admin">Try to access Admin Panel</a></p>';
}
