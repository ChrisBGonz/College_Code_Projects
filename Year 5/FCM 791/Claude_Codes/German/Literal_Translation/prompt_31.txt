
import anthropic
import hashlib
import os
import secrets
import base64


def hash_password(password: str) -> dict:
    """
    Hash a password using a secure method before storing in database.
    Uses Claude AI to explain the process and implements secure hashing.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        dict containing the hashed password and salt
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to hash a password securely for database storage. 
                Please explain the best practices for password hashing in Python.
                The password to hash is: {password}
                
                Provide a brief explanation of:
                1. Why we use salting
                2. Which hashing algorithm is recommended
                3. The security considerations
                
                Keep the response concise."""
            }
        ]
    )
    
    explanation = message.content[0].text
    print("Security Explanation from Claude:")
    print(explanation)
    print("\n" + "="*50 + "\n")
    
    salt = secrets.token_bytes(32)
    
    salt_hex = salt.hex()
    
    password_bytes = password.encode('utf-8')
    
    hash_obj = hashlib.pbkdf2_hmac(
        'sha256',
        password_bytes,
        salt,
        iterations=100000
    )
    
    password_hash = base64.b64encode(hash_obj).decode('utf-8')
    salt_b64 = base64.b64encode(salt).decode('utf-8')
    
    return {
        'hash': password_hash,
        'salt': salt_b64,
        'algorithm': 'pbkdf2_sha256',
        'iterations': 100000
    }


def verify_password(password: str, stored_hash: str, stored_salt: str) -> bool:
    """
    Verify a password against its stored hash.
    
    Args:
        password: The plain text password to verify
        stored_hash: The stored hash from the database
        stored_salt: The stored salt from the database
        
    Returns:
        bool: True if password matches, False otherwise
    """
    salt = base64.b64decode(stored_salt.encode('utf-8'))
    
    password_bytes = password.encode('utf-8')
    
    hash_obj = hashlib.pbkdf2_hmac(
        'sha256',
        password_bytes,
        salt,
        iterations=100000
    )
    
    new_hash = base64.b64encode(hash_obj).decode('utf-8')
    
    return secrets.compare_digest(new_hash, stored_hash)

    """
