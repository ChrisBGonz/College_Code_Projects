
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

// Banking app state
let isLoggedIn = false;
let currentUser = null;
let inactivityTimer = null;
let conversationHistory = [];
const INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds

// Mock user database
const users = {
  john_doe: {
    password: "secure123",
    balance: 5420.5,
    transactions: [
      { date: "2024-01-15", description: "Grocery Store", amount: -85.3 },
      { date: "2024-01-14", description: "Salary", amount: 3500.0 },
      { date: "2024-01-13", description: "Electric Bill", amount: -120.0 },
    ],
  },
};

// Reset inactivity timer
function resetInactivityTimer() {
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
  }

  if (isLoggedIn) {
    inactivityTimer = setTimeout(() => {
      console.log(
        "\n⚠️  Session expired due to 15 minutes of inactivity. You have been logged out for security reasons."
      );
      logout();
      console.log("\nPlease log in again to continue banking.");
      console.log('Type "login" to start a new session or "exit" to quit.\n');
    }, INACTIVITY_TIMEOUT);
  }
}

// Login function
function login(username, password) {
  if (users[username] && users[username].password === password) {
    isLoggedIn = true;
    currentUser = username;
    conversationHistory = []; // Reset conversation for new session
    resetInactivityTimer();
    return true;
  }
  return false;
}

// Logout function
function logout() {
  isLoggedIn = false;
  currentUser = null;
  conversationHistory = [];
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
    inactivityTimer = null;
  }
}

// Get account information for context
function getAccountContext() {
  if (!isLoggedIn || !currentUser) return "";

  const user = users[currentUser];
  return `
Current user: ${currentUser}
Account balance: $${user.balance.toFixed(2)}
Recent transactions:
${user.transactions.map((t) => `- ${t.date}: ${t.description}: $${t.amount.toFixed(2)}`).join("\n")}
  `;
}

// Chat with Claude for banking assistance
async function chatWithBankingAssistant(userMessage) {
  if (!isLoggedIn) {
    return "Please log in first to access banking services.";
  }

  // Reset inactivity timer on each interaction
  resetInactivityTimer();

  // Add user message to history
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const systemPrompt = `You are a helpful banking assistant for SecureBank. You help customers with their banking needs.
  
Current session information:
${getAccountContext()}

You can help with:
- Checking account balance
- Reviewing recent transactions
- Answering banking questions
- Providing financial advice
- Explaining banking services

Always be professional, helpful, and security-conscious. Never ask for sensitive information like full passwords or SSN.
If asked about transactions or balance, use the information provided in the session context.`;

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  // Add assistant response to history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Main banking app interface
async function runBankingApp() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("=================================");
  console.log("    Welcome to SecureBank App    ");
  console.log("=================================");
  console.log("Auto-logout after 15 minutes of inactivity\n");

  const question = (prompt) => {
    return new Promise((resolve) => {
      rl.question(prompt, resolve);
    });
  };

  let running = true;

  while (running) {
    if (!isLoggedIn) {
      console.log("\nOptions: login | exit");
      const input = await question("Enter command: ");

      if (input.toLowerCase() === "exit") {
        running = false;
        console.log("Thank you for using SecureBank. Goodbye!");
      } else if (input.toLowerCase() === "login") {
        const username = await question("Username: ");
        const password = await question("Password: ");

        if (login(username, password)) {
          console.log(`\n✅ Welcome back, ${username}! You are now logged in.`);
          console.log(
            "Your session will automatically expire after 15 minutes of inactivity."
          );
          console.log(
            'Type "logout" to end your session or "exit" to quit.\n'
          );
        } else {
          console.log("❌ Invalid username or password. Please try again.");
        }
      } else {
        console.log('Invalid command. Please type "login" or "exit".');
      }
    } else {
      const userInput = await question("\nYou: ");

      if (userInput.toLowerCase() === "exit") {
        logout();
        running = false;
        console.log("Thank you for using SecureBank. Goodbye!");
      } else if (userInput.toLowerCase() === "logout") {
        logout();
        console.log("✅ You have been successfully logged out.");
      } else if (userInput.trim()) {
        try {
          console.log("\nAssistant: Thinking...");
          const response = await chatWithBankingAssistant(userInput);
          console.log(`\nAssistant: ${response}`);
        } catch (error) {
          console.error("Error communicating with assistant:", error.message);
        }
      }
    }
  }

  rl.close();
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
  }
}

// Export functions for testing
module.exports = {
  login,
  logout,
  resetInactivityTimer,
  chatWithBankingAssistant,
  isLoggedIn: () => isLoggedIn,
  currentUser: () => currentUser,
};

// Run the app
runBankingApp().catch(console.error);
```
