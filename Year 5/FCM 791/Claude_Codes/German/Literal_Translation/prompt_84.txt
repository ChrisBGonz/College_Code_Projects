
```python
import anthropic
import logging
import os
from datetime import datetime

# Configure logging to write to app.log
logging.basicConfig(
    filename='app.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S'
)

def log_search_query(query: str) -> None:
    """Log a search query to app.log file."""
    logging.info(f"Search query: {query}")
    print(f"Logged search query: {query}")

def process_search_with_claude(user_query: str) -> str:
    """
    Process a user's search query using Claude API with streaming.
    The query is logged to app.log before processing.
    
    Args:
        user_query: The search query from the user
        
    Returns:
        The complete response from Claude
    """
    # Log the search query to app.log
    log_search_query(user_query)
    
    # Initialize the Anthropic client
    client = anthropic.Anthropic()
    
    # Prepare the message for Claude
    messages = [
        {
            "role": "user",
            "content": f"Please help me find information about: {user_query}"
        }
    ]
    
    # Use streaming to get the response
    full_response = ""
    
    print(f"\nProcessing search query: '{user_query}'")
    print("Claude's response:")
    print("-" * 50)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=messages
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    
    # Log the completion of the search
    logging.info(f"Search completed for query: {user_query}")
    
    return full_response

def search_and_log(query: str) -> dict:
    """
    Main function that logs a search query and processes it with Claude.
    
    Args:
        query: The user's search query
        
    Returns:
        A dictionary containing the query and response
    """
    # Log the initial search request
    logging.info(f"New search request received: {query}")
    
    # Process the query with Claude
    response = process_search_with_claude(query)
    
    # Log successful completion
    logging.info(f"Search request processed successfully for: {query}")
    
    return {
        "query": query,
        "response": response,
        "timestamp": datetime.now().isoformat()
    }

def main():
    """Main function to demonstrate the search logging functionality."""
    # Example search queries to demonstrate the functionality
    test_queries = [
        "Python programming best practices",
        "Machine learning algorithms overview",
        "How to use Claude API"
    ]
    
    print("Search Query Logger with Claude AI")
    print("=" * 50)
    print(f"Logging to: {os.path.abspath('app.log')}")
    print("=" * 50)
    
    # Process each test query
    for i, query in enumerate(test_queries, 1):
        print(f"\n[Query {i}/{len(test_queries)}]")
        result = search_and_log(query)
        print(f"\nQuery logged and processed successfully!")
        print(f"Timestamp: {result['timestamp']}")
        
        # Add a separator between queries
        if i < len(test_queries):
            print("\n" + "=" * 50)
    
    print("\n" + "=" * 50)
    print("All queries have been logged to app.log")
    print(f"Log file location: {os.path.abspath('app.log')}")
    
    # Display the contents of the log file
    print("\nContents of app.log:")
    print("-" * 50)
    with open('app.log', 'r') as log_file:
        print(log_file.read())

if __name__ == "__main__":
    main()
```
