import com.fasterxml.jackson.databind.ObjectMapper;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

public class JsonParserWithBedrock {

    static class User {
        private String name;
        private String email;
        private int age;

        public String getName() {
            return name;
        }

        public void setName(String name) {
            this.name = name;
        }

        public String getEmail() {
            return email;
        }

        public void setEmail(String email) {
            this.email = email;
        }

        public int getAge() {
            return age;
        }

        public void setAge(int age) {
            this.age = age;
        }

        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }

    public static User parseUserFromJson(String jsonString) throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        return objectMapper.readValue(jsonString, User.class);
    }

    public static String analyzeJsonWithBedrock(String jsonString) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .build();

        String prompt = "Analyze this JSON string and extract the user information: " + jsonString + 
                       "\nPlease identify the user's name and any other relevant properties.";

        Message message = Message.builder()
                .content(ContentBlock.fromText(prompt))
                .role(ConversationRole.USER)
                .build();

        ConverseResponse response = client.converse(request -> request
                .modelId("amazon.nova-lite-v1:0")
                .messages(message)
                .inferenceConfig(config -> config
                        .maxTokens(500)
                        .temperature(0.5f)
                        .topP(0.9f)));

        return response.output().message().content().get(0).text();
    }

    public static void main(String[] args) {
        String jsonString = "{\"name\": \"John Doe\", \"email\": \"john.doe@example.com\", \"age\": 30}";

        System.out.println("Parsing JSON string: " + jsonString);
        System.out.println("----------------------------------------");

        try {
            User user = parseUserFromJson(jsonString);
            System.out.println("Successfully parsed User object:");
            System.out.println("user.name = " + user.getName());
            System.out.println("Full user object: " + user);
        } catch (Exception e) {
            System.err.println("Error parsing JSON: " + e.getMessage());
        }

        System.out.println("\n----------------------------------------");
        System.out.println("Analyzing JSON with Amazon Bedrock...");
        
        try {
            String bedrockAnalysis = analyzeJsonWithBedrock(jsonString);
            System.out.println("Bedrock Analysis:");
            System.out.println(bedrockAnalysis);
        } catch (Exception e) {
            System.err.println("Error calling Bedrock: " + e.getMessage());
        }

        System.out.println("\n----------------------------------------");
        System.out.println("Testing with different JSON strings:");
        
        String[] testJsonStrings = {
            "{\"name\": \"Jane Smith\", \"email\": \"jane@example.com\", \"age\": 25}",
            "{\"name\": \"Bob Johnson\", \"age\": 45}",
            "{\"name\": \"Alice Brown\", \"email\": \"alice@test.com\", \"age\": 35}"
        };

        for (String testJson : testJsonStrings) {
            try {
                User testUser = parseUserFromJson(testJson);
                System.out.println("Parsed user.name: " + testUser.getName());
            } catch (Exception e) {
                System.err.println("Error parsing: " + e.getMessage());
            }
        }
    }
}
