
```python
import anthropic
import json
from datetime import datetime, timedelta

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_trial_cookie",
        "description": "Checks if the user has already used a trial by reading the 'has_used_trial' cookie",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "set_trial_cookie",
        "description": "Sets the 'has_used_trial' cookie to mark that the user has started a trial",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                },
                "trial_start_date": {
                    "type": "string",
                    "description": "The date when the trial started (ISO format)"
                }
            },
            "required": ["user_id", "trial_start_date"]
        }
    },
    {
        "name": "register_trial",
        "description": "Registers a new free trial for the user",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                },
                "email": {
                    "type": "string",
                    "description": "The email address of the user"
                },
                "name": {
                    "type": "string",
                    "description": "The full name of the user"
                }
            },
            "required": ["user_id", "email", "name"]
        }
    }
]

cookie_store = {}

def check_trial_cookie(user_id: str) -> dict:
    """Simulates checking the has_used_trial cookie"""
    if user_id in cookie_store:
        cookie_data = cookie_store[user_id]
        return {
            "has_used_trial": True,
            "trial_start_date": cookie_data["trial_start_date"],
            "cookie_name": "has_used_trial",
            "user_id": user_id
        }
    else:
        return {
            "has_used_trial": False,
            "cookie_name": "has_used_trial",
            "user_id": user_id
        }

def set_trial_cookie(user_id: str, trial_start_date: str) -> dict:
    """Simulates setting the has_used_trial cookie"""
    cookie_store[user_id] = {
        "trial_start_date": trial_start_date,
        "cookie_name": "has_used_trial",
        "expires": (datetime.now() + timedelta(days=365)).isoformat()
    }
    return {
        "success": True,
        "message": f"Cookie 'has_used_trial' set for user {user_id}",
        "trial_start_date": trial_start_date
    }

def register_trial(user_id: str, email: str, name: str) -> dict:
    """Simulates registering a new trial"""
    trial_start_date = datetime.now().isoformat()
    trial_end_date = (datetime.now() + timedelta(days=14)).isoformat()
    
    return {
        "success": True,
        "message": f"Free trial successfully registered for {name}",
        "user_id": user_id,
        "email": email,
        "trial_start_date": trial_start_date,
        "trial_end_date": trial_end_date,
        "trial_duration_days": 14
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "check_trial_cookie":
        result = check_trial_cookie(tool_input["user_id"])
    elif tool_name == "set_trial_cookie":
        result = set_trial_cookie(tool_input["user_id"], tool_input["trial_start_date"])
    elif tool_name == "register_trial":
        result = register_trial(tool_input["user_id"], tool_input["email"], tool_input["name"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def handle_trial_signup(user_id: str, email: str, name: str) -> str:
    """Main function to handle free trial signup using Claude"""
    
    print(f"\n{'='*60}")
    print(f"Processing trial signup for: {name} ({email})")
    print(f"User ID: {user_id}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": f"""Please process a free trial signup for the following user:
            - User ID: {user_id}
            - Email: {email}
            - Name: {name}
            
            First, check if this user has already used a trial by checking their cookie.
            If they haven't used a trial before, register them for a new free trial and set the trial cookie.
            If they have already used a trial, inform them that they cannot start another trial.
            
            Please provide a clear response about the outcome."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    print(f"\nFinal Response: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nTool Call: {tool_name}")
                    print(f"Input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop_reason: {response.stop_reason}")
            break
    
    return "Error processing trial signup"

def main():
    """Main function to demonstrate the trial signup system"""
    
    print("FREE TRIAL SIGNUP SYSTEM")
    print("="*60)
    
    print("\n--- Test 1: New User (should succeed) ---")
    result1 = handle_trial_signup(
        user_id="user_001",
        email="alice@example.com",
        name="Alice Johnson"
    )
    
    print("\n--- Test 2: Same User Again (should be rejected) ---")
    result2 = handle_trial_signup(
        user_id="user_001",
        email="alice@example.com",
        name="Alice Johnson"
    )
    
    print("\n--- Test 3: Another New User (should succeed) ---")
    result3 = handle_trial_signup(
        user_id="user_002",
        email="bob@example.com",
        name="Bob Smith"
    )
    
    print("\n" + "="*60)
    print("SUMMARY OF RESULTS")
    print("="*60)
    print(f"\nTest 1 (New User): {result1[:100]}...")
    print(f"\nTest 2 (Repeat User): {result2[:100]}...")
    print(f"\nTest 3 (Another New User): {result3[:100]}...")
    
    print("\n" + "="*60)
    print("COOKIE STORE STATE")
    print("="*60)
    print(json.dumps(cookie_store, indent=2))

if __name__ == "__main__":
    main()
```
