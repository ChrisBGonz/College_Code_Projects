def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

def analyze_profile_with_claude(config_data):
    """Use Claude to analyze the profile configuration with streaming."""
    client = anthropic.Anthropic()
    
    config_str = yaml.dump(config_data, default_flow_style=False, allow_unicode=True)
    
    prompt = f"""Analysieren Sie die folgende YAML-Profilkonfiguration und geben Sie eine freundliche, 
    hilfreiche Zusammenfassung auf Deutsch. Beschreiben Sie, was das Profil enthält, 
    welche Einstellungen vorgenommen wurden und geben Sie ggf. Empfehlungen oder Hinweise.
    
    Konfiguration:
    {config_str}
    
    Bitte geben Sie eine strukturierte Analyse mit:
    1. Profilübersicht
    2. Wichtige Einstellungen
    3. Empfehlungen (falls zutreffend)"""
    
    analysis_text = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            analysis_text += text
    
    return analysis_text

@app.route('/')

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'Keine Datei ausgewählt'}), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'Keine Datei ausgewählt'}), 400
    
    if not allowed_file(file.filename):
        return jsonify({'success': False, 'error': 'Ungültiges Dateiformat. Nur .yaml und .yml Dateien sind erlaubt.'}), 400
    
    try:
        filename = secure_filename(file.filename)
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        file.save(filepath)
        
        with open(filepath, 'r', encoding='utf-8') as f:
            config_data = yaml.safe_load(f)
        
        if config_data is None:
            return jsonify({'success': False, 'error': 'Die YAML-Datei ist leer oder ungültig'}), 400
        
        ai_analysis = analyze_profile_with_claude(config_data)
        
        os.remove(filepath)
        
        return jsonify({
            'success': True,
            'config': config_data,
            'ai_analysis': ai_analysis,
            'message': 'Profil erfolgreich eingerichtet'
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Fehler beim Parsen der YAML-Datei: {str(e)}'}), 400
    except Exception as e:
        return jsonify({'success': False, 'error': f'Ein Fehler ist aufgetreten: {str(e)}'}), 500

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
