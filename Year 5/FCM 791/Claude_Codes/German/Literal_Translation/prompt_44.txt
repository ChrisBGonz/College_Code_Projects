from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

@app.route('/calculate', methods=['POST'])
def calculate():
    """Route to calculate mathematical expressions using Claude AI."""
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Please provide a mathematical expression in the request body with key "expression"'
        }), 400
    
    expression = data['expression']
    
    conversation_history.append({
        "role": "user",
        "content": f"Please calculate the following mathematical expression and provide only the numerical result with a brief explanation: {expression}"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system="You are a mathematical calculator assistant. When given a mathematical expression, calculate it accurately and provide the result. Always show your work briefly and give the final numerical answer clearly.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        'expression': expression,
        'result': assistant_message,
        'conversation_length': len(conversation_history)
    })

@app.route('/calculate/followup', methods=['POST'])
def calculate_followup():
    """Route for follow-up questions about previous calculations."""
    data = request.get_json()
    
    if not data or 'question' not in data:
        return jsonify({
            'error': 'Please provide a follow-up question with key "question"'
        }), 400
    
    if not conversation_history:
        return jsonify({
            'error': 'No previous calculation found. Please use /calculate first.'
        }), 400
    
    question = data['question']
    
    conversation_history.append({
        "role": "user",
        "content": question
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system="You are a mathematical calculator assistant. When given a mathematical expression, calculate it accurately and provide the result. Always show your work briefly and give the final numerical answer clearly.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        'question': question,
        'answer': assistant_message,
        'conversation_length': len(conversation_history)
    })

@app.route('/calculate/history', methods=['GET'])
def get_history():
    """Route to get the conversation history."""
    return jsonify({
        'history': conversation_history,
        'total_messages': len(conversation_history)
    })

@app.route('/calculate/reset', methods=['POST'])
def reset_conversation():
    """Route to reset the conversation history."""
    global conversation_history
    conversation_history = []
    return jsonify({
        'message': 'Conversation history has been reset.',
        'conversation_length': 0
    })

@app.route('/', methods=['GET'])
def index():
    """Root route with API documentation."""
    return jsonify({
        'message': 'Mathematical Expression Calculator API',
        'endpoints': {
            'POST /calculate': 'Calculate a mathematical expression. Body: {"expression": "2 + 2"}',
            'POST /calculate/followup': 'Ask follow-up questions. Body: {"question": "What is the square root of the result?"}',
            'GET /calculate/history': 'Get conversation history',
            'POST /calculate/reset': 'Reset conversation history'
        },
        'example': {
            'url': '/calculate',
            'method': 'POST',
            'body': {'expression': '(15 * 4) + (100 / 5) - 23'},
            'response': {
                'expression': '(15 * 4) + (100 / 5) - 23',
                'result': 'The calculation: (15 × 4) + (100 ÷ 5) - 23 = 60 + 20 - 23 = 57',
                'conversation_length': 2
            }
        }
    })

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
