
import pickle
import os
from datetime import datetime


class GameState:
    def __init__(self, player_name):
        self.player_name = player_name
        self.level = 1
        self.score = 0
        self.health = 100
        self.inventory = []
        self.achievements = []
        self.position = {"x": 0, "y": 0}
        self.created_at = datetime.now()
        self.last_saved = None
        self.game_settings = {
            "difficulty": "normal",
            "sound": True,
            "music": True,
            "language": "de"
        }
        self.quest_log = {}
        self.statistics = {
            "enemies_defeated": 0,
            "items_collected": 0,
            "deaths": 0,
            "play_time_minutes": 0
        }

    def add_item(self, item):
        self.inventory.append(item)
        self.statistics["items_collected"] += 1
        print(f"Item '{item}' zum Inventar hinzugefügt.")

    def add_achievement(self, achievement):
        if achievement not in self.achievements:
            self.achievements.append(achievement)
            print(f"Erfolg freigeschaltet: '{achievement}'!")

    def update_position(self, x, y):
        self.position = {"x": x, "y": y}

    def level_up(self):
        self.level += 1
        self.health = min(100, self.health + 20)
        print(f"Level Up! Jetzt auf Level {self.level}.")

    def add_quest(self, quest_name, status="aktiv"):
        self.quest_log[quest_name] = {
            "status": status,
            "started_at": datetime.now()
        }
        print(f"Quest '{quest_name}' hinzugefügt.")

    def complete_quest(self, quest_name):
        if quest_name in self.quest_log:
            self.quest_log[quest_name]["status"] = "abgeschlossen"
            self.quest_log[quest_name]["completed_at"] = datetime.now()
            self.score += 100
            print(f"Quest '{quest_name}' abgeschlossen! +100 Punkte.")

    def display_status(self):
        print("\n" + "="*50)
        print(f"  SPIELSTATUS: {self.player_name}")
        print("="*50)
        print(f"  Level:        {self.level}")
        print(f"  Punkte:       {self.score}")
        print(f"  Leben:        {self.health}/100")
        print(f"  Position:     X={self.position['x']}, Y={self.position['y']}")
        print(f"  Inventar:     {', '.join(self.inventory) if self.inventory else 'Leer'}")
        print(f"  Erfolge:      {len(self.achievements)}")
        print(f"  Quests:       {len(self.quest_log)}")
        print(f"  Erstellt:     {self.created_at.strftime('%d.%m.%Y %H:%M:%S')}")
        if self.last_saved:
            print(f"  Gespeichert:  {self.last_saved.strftime('%d.%m.%Y %H:%M:%S')}")
        print("="*50)
        print(f"  Statistiken:")
        for key, value in self.statistics.items():
            print(f"    {key}: {value}")
        print("="*50 + "\n")


def save_game_state(game_state, filename="savegame.pkl"):
    """Speichert den Spielstand in einer Datei."""
    try:
        game_state.last_saved = datetime.now()
        with open(filename, "wb") as file:
            pickle.dump(game_state, file, protocol=pickle.HIGHEST_PROTOCOL)
        file_size = os.path.getsize(filename)
        print(f"Spielstand erfolgreich gespeichert: '{filename}' ({file_size} Bytes)")
        return True
    except (IOError, OSError) as e:
        print(f"Fehler beim Speichern: {e}")
        return False
    except pickle.PicklingError as e:
        print(f"Fehler beim Serialisieren: {e}")
        return False


def load_game_state(filename="savegame.pkl"):
    """Lädt den Spielstand aus einer Datei."""
    if not os.path.exists(filename):
        print(f"Keine Speicherdatei gefunden: '{filename}'")
        return None
    try:
        with open(filename, "rb") as file:
            game_state = pickle.load(file)
        print(f"Spielstand erfolgreich geladen: '{filename}'")
        return game_state
    except (IOError, OSError) as e:
        print(f"Fehler beim Laden: {e}")
        return None
    except (pickle.UnpicklingError, EOFError) as e:
        print(f"Fehler beim Deserialisieren: {e}")
        return None


def save_multiple_slots(game_state, slot=1):
    """Speichert den Spielstand in einem bestimmten Slot."""
    filename = f"savegame_slot{slot}.pkl"
    return save_game_state(game_state, filename)


def load_multiple_slots(slot=1):
    """Lädt den Spielstand aus einem bestimmten Slot."""
    filename = f"savegame_slot{slot}.pkl"
    return load_game_state(filename)


def list_save_files():
    """Listet alle vorhandenen Speicherdateien auf."""
    save_files = [f for f in os.listdir(".") if f.endswith(".pkl")]
    if save_files:
        print("\nVorhandene Speicherdateien:")
        for f in save_files:
            size = os.path.getsize(f)
            modified = datetime.fromtimestamp(os.path.getmtime(f))
            print(f"  - {f} ({size} Bytes, geändert: {modified.strftime('%d.%m.%Y %H:%M:%S')})")
    else:
        print("Keine Speicherdateien gefunden.")
    return save_files


def delete_save_file(filename="savegame.pkl"):
    """Löscht eine Speicherdatei."""
    if os.path.exists(filename):
        os.remove(filename)
        print(f"Speicherdatei '{filename}' gelöscht.")
        return True
    else:
        print(f"Datei '{filename}' nicht gefunden.")
        return False
