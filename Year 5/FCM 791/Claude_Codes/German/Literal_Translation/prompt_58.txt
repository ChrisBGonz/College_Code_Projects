```javascript
const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  const steps = [
    {
      id: 1,
      title: 'Persönliche Daten',
      fields: [
        { name: 'firstName', label: 'Vorname', type: 'text', required: true },
        { name: 'lastName', label: 'Nachname', type: 'text', required: true },
        { name: 'birthDate', label: 'Geburtsdatum', type: 'date', required: true },
      ],
    },
    {
      id: 2,
      title: 'Kontaktdaten',
      fields: [
        { name: 'email', label: 'E-Mail', type: 'email', required: true },
        { name: 'phone', label: 'Telefon', type: 'tel', required: false },
        { name: 'address', label: 'Adresse', type: 'text', required: true },
      ],
    },
    {
      id: 3,
      title: 'Konto-Einstellungen',
      fields: [
        { name: 'username', label: 'Benutzername', type: 'text', required: true },
        { name: 'password', label: 'Passwort', type: 'password', required: true },
        { name: 'confirmPassword', label: 'Passwort bestätigen', type: 'password', required: true },
      ],
    },
    {
      id: 4,
      title: 'Zusammenfassung',
      fields: [],
    },
  ];

  let currentStep = 1;
  let formData = {};

  function saveToLocalStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
      localStorage.setItem(CURRENT_STEP_KEY, currentStep.toString());
      console.log('Daten gespeichert:', formData);
    } catch (error) {
      console.error('Fehler beim Speichern in localStorage:', error);
    }
  }

  function loadFromLocalStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

      if (savedData) {
        formData = JSON.parse(savedData);
        console.log('Gespeicherte Daten geladen:', formData);
      }

      if (savedStep) {
        currentStep = parseInt(savedStep, 10);
      }
    } catch (error) {
      console.error('Fehler beim Laden aus localStorage:', error);
      formData = {};
      currentStep = 1;
    }
  }

  function clearLocalStorage() {
    localStorage.removeItem(STORAGE_KEY);
    localStorage.removeItem(CURRENT_STEP_KEY);
    console.log('localStorage-Daten gelöscht');
  }

  function updateFormData(stepFields) {
    stepFields.forEach((field) => {
      const input = document.getElementById(field.name);
      if (input) {
        if (field.type === 'password') {
          formData[field.name] = btoa(input.value);
        } else {
          formData[field.name] = input.value;
        }
      }
    });
  }

  function validateStep(step) {
    const errors = [];
    const stepConfig = steps.find((s) => s.id === step);

    if (!stepConfig) return errors;

    stepConfig.fields.forEach((field) => {
      const input = document.getElementById(field.name);
      if (!input) return;

      const value = input.value.trim();

      if (field.required && !value) {
        errors.push(`"${field.label}" ist ein Pflichtfeld.`);
        input.classList.add('error');
      } else {
        input.classList.remove('error');
      }

      if (value) {
        if (field.type === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
          errors.push(`"${field.label}" muss eine gültige E-Mail-Adresse sein.`);
          input.classList.add('error');
        }

        if (field.name === 'confirmPassword') {
          const passwordInput = document.getElementById('password');
          if (passwordInput && input.value !== passwordInput.value) {
            errors.push('Die Passwörter stimmen nicht überein.');
            input.classList.add('error');
          }
        }

        if (field.name === 'username' && value.length < 3) {
          errors.push(`"${field.label}" muss mindestens 3 Zeichen lang sein.`);
          input.classList.add('error');
        }

        if (field.name === 'password' && value.length < 8) {
          errors.push(`"${field.label}" muss mindestens 8 Zeichen lang sein.`);
          input.classList.add('error');
        }
      }
    });

    return errors;
  }

  function showErrors(errors) {
    const errorContainer = document.getElementById('errorContainer');
    if (!errorContainer) return;

    if (errors.length === 0) {
      errorContainer.innerHTML = '';
      errorContainer.style.display = 'none';
      return;
    }

    errorContainer.style.display = 'block';
    errorContainer.innerHTML = `
      <ul>
        ${errors.map((error) => `<li>${error}</li>`).join('')}
      </ul>
    `;
  }

  function populateFieldsFromStorage() {
    const stepConfig = steps.find((s) => s.id === currentStep);
    if (!stepConfig) return;

    stepConfig.fields.forEach((field) => {
      const input = document.getElementById(field.name);
      if (input && formData[field.name]) {
        if (field.type !== 'password') {
          input.value = formData[field.name];
        }
      }
    });
  }

  function renderSummary() {
    const summaryContainer = document.getElementById('summaryContainer');
    if (!summaryContainer) return;

    const displayFields = [
      { key: 'firstName', label: 'Vorname' },
      { key: 'lastName', label: 'Nachname' },
      { key: 'birthDate', label: 'Geburtsdatum' },
      { key: 'email', label: 'E-Mail' },
      { key: 'phone', label: 'Telefon' },
      { key: 'address', label: 'Adresse' },
      { key: 'username', label: 'Benutzername' },
    ];

    summaryContainer.innerHTML = `
      <h3>Ihre Angaben:</h3>
      <table>
        <tbody>
          ${displayFields
            .map((field) =>
              formData[field.key]
                ? `<tr>
                    <td><strong>${field.label}:</strong></td>
                    <td>${formData[field.key]}</td>
                  </tr>`
                : ''
            )
            .join('')}
        </tbody>
      </table>
    `;
  }

  function updateProgressBar() {
    const progressBar = document.getElementById('progressBar');
    const progressText = document.getElementById('progressText');
    const totalSteps = steps.length;
    const progress = ((currentStep - 1) / (totalSteps - 1)) * 100;

    if (progressBar) {
      progressBar.style.width = `${progress}%`;
      progressBar.setAttribute('aria-valuenow', progress);
    }

    if (progressText) {
      progressText.textContent = `Schritt ${currentStep} von ${totalSteps}`;
    }

    steps.forEach((step) => {
      const stepIndicator = document.getElementById(`stepIndicator${step.id}`);
      if (stepIndicator) {
        stepIndicator.classList.remove('active', 'completed');
        if (step.id === currentStep) {
          stepIndicator.classList.add('active');
        } else if (step.id < currentStep) {
          stepIndicator.classList.add('completed');
        }
      }
    });
  }

  function renderStep(stepId) {
    const stepConfig = steps.find((s) => s.id === stepId);
    if (!stepConfig) return;

    const formContainer = document.getElementById('formContainer');
    if (!formContainer) return;

    const stepTitle = document.getElementById('stepTitle');
    if (stepTitle) {
      stepTitle.textContent = stepConfig.title;
    }

    if (stepId === 4) {
      formContainer.innerHTML = `<div id="summaryContainer"></div>`;
      renderSummary();
    } else {
      formContainer.innerHTML = stepConfig.fields
        .map(
          (field) => `
          <div class="form-group">
            <label for="${field.name}">
              ${field.label}
              ${field.required ? '<span class="required">*</span>' : ''}
            </label>
            <input
              type="${field.type}"
              id="${field.name}"
              name="${field.name}"
              placeholder="${field.label} eingeben"
              ${field.required ? 'required' : ''}
            />
          </div>
        `
        )
        .join('');

      populateFieldsFromStorage();
    }

    const prevButton = document.getElementById('prevButton');
    const nextButton = document.getElementById('nextButton');
    const submitButton = document.getElementById('submitButton');

    if (prevButton) {
      prevButton.style.display = stepId > 1 ? 'inline-block' : 'none';
    }

    if (nextButton) {
      nextButton.style.display = stepId < steps.length ? 'inline-block' : 'none';
    }

    if (submitButton) {
      submitButton.style.display = stepId === steps.length ? 'inline-block' : 'none';
    }

    updateProgressBar();
    showErrors([]);
  }

  function nextStep() {
    const stepConfig = steps.find((s) => s.id === currentStep);
    if (!stepConfig) return;

    if (currentStep < steps.length) {
      const errors = validateStep(currentStep);

      if (errors.length > 0) {
        showErrors(errors);
        return;
      }

      updateFormData(stepConfig.fields);
      saveToLocalStorage();

      currentStep++;
      renderStep(currentStep);
    }
  }

  function prevStep() {
    if (currentStep > 1) {
      const stepConfig = steps.find((s) => s.id === currentStep);
      if (stepConfig && stepConfig.fields.length > 0) {
        updateFormData(stepConfig.fields);
        saveToLocalStorage();
      }

      currentStep--;
      renderStep(currentStep);
    }
  }

  async function submitForm() {
    console.log('Formular wird abgesendet...');

    const submitButton = document.getElementById('submitButton');
    if (submitButton) {
      submitButton.disabled = true;
      submitButton.textContent = 'Wird gesendet...';
    }

    try {
      const submissionData = { ...formData };
      delete submissionData.password;
      delete submissionData.confirmPassword;

      console.log('Gesendete Daten:', submissionData);

      await new Promise((resolve) => setTimeout(resolve, 1500));

      clearLocalStorage();
      formData = {};
      currentStep = 1;

      const formWrapper = document.getElementById('formWrapper');
      if (formWrapper) {
        formWrapper.innerHTML = `
          <div class="success-message">
            <h2>✓ Registrierung erfolgreich!</h2>
            <p>Vielen Dank für Ihre Registrierung. Sie erhalten in Kürze eine Bestätigungs-E-Mail.</p>
            <button onclick="MultiStepForm.init()">Neue Registrierung</button>
          </div>
        `;
      }
    } catch (error) {
      console.error('Fehler beim Absenden:', error);
      showErrors(['Ein Fehler ist aufgetreten. Bitte versuchen Sie es erneut.']);

      if (submitButton) {
        submitButton.disabled = false;
        submitButton.textContent = 'Absenden';
      }
    }
  }

  function checkForSavedData() {
    const savedData = localStorage.getItem(STORAGE_KEY);
    const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

    if (savedData && savedStep) {
      const resumeDialog = document.getElementById('resumeDialog');
      if (resumeDialog) {
        resumeDialog.style.display = 'block';
      }
      return true;
    }
    return false;
  }

  function resumeFromSaved() {
    loadFromLocalStorage();
    const resumeDialog = document.getElementById('resumeDialog');
    if (resumeDialog) {
      resumeDialog.style.display = 'none';
    }
    renderStep(currentStep);
  }

  function startFresh() {
    clearLocalStorage();
    formData = {};
    currentStep = 1;
    const resumeDialog = document.getElementById('resumeDialog');
    if (resumeDialog) {
      resumeDialog.style.display = 'none';
    }
    renderStep(currentStep);
  }

  function createFormHTML() {
    return `
      <div id="formWrapper">
        <div id="resumeDialog" style="display:none;" class="resume-dialog">
          <p>Es wurden gespeicherte Formulardaten gefunden. Möchten Sie dort weitermachen, wo Sie aufgehört haben?</p>
          <button onclick="MultiStepForm.resumeFromSaved()">Fortsetzen</button>
          <button onclick="MultiStepForm.startFresh()">Neu beginnen</button>
        </div>

        <div class="progress-container">
          <div id="progressText" class="progress-text"></div>
          <div class="progress-bar-wrapper">
            <div id="progressBar" class="progress-bar" role="progressbar" aria-valuenow="0" aria-valuemin="0" aria-valuemax="100"></div>
          </div>
          <div class="step-indicators">
            ${steps
              .map(
                (step) => `
              <div id="stepIndicator${step.id}" class="step-indicator" title="${step.title}">
                <span>${step.id}</span>
                <small>${step.title}</small>
              </div>
            `
              )
              .join('')}
          </div>
        </div>

        <div class="form-card">
          <h2 id="stepTitle"></h2>
          <div id="errorContainer" class="error-container" style="display:none;"></div>
          <div id="formContainer"></div>

          <div class="button-group">
            <button id="prevButton" onclick="MultiStepForm.prevStep()" style="display:none;">← Zurück</button>
            <button id="nextButton" onclick="MultiStepForm.nextStep()">Weiter →</button>
            <button id="submitButton" onclick="MultiStepForm.submitForm()" style="display:none;" class="submit-btn">Absenden ✓</button>
          </div>
        </div>
      </div>
    `;
  }

  function init() {
    const appContainer = document.getElementById('app');
    if (!appContainer) {
      console.error('Container mit id="app" nicht gefunden!');
      return;
    }

    appContainer.innerHTML = createFormHTML();
    addStyles();

    const hasSavedData = checkForSavedData();

    if (!hasSavedData) {
      currentStep = 1;
      formData = {};
      renderStep(currentStep);
    }
  }

  return {
    init,
    nextStep,
    prevStep,
    submitForm,
    resumeFromSaved,
    startFresh,
    getFormData: () => ({ ...formData }),
    getCurrentStep: () => currentStep,
  };
})();
```
