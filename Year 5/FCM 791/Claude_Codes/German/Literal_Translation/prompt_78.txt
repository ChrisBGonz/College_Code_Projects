package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// AppConfig represents the application configuration
type AppConfig struct {
	Name        string                 `json:"name"`
	Version     string                 `json:"version"`
	Debug       bool                   `json:"debug"`
	MaxRetries  int                    `json:"maxRetries"`
	Timeout     float64                `json:"timeout"`
	Tags        []string               `json:"tags"`
	Settings    map[string]interface{} `json:"settings"`
	Database    DatabaseConfig         `json:"database"`
	Features    []Feature              `json:"features"`
}

// DatabaseConfig represents database configuration
type DatabaseConfig struct {
	Host     string `json:"host"`
	Port     int    `json:"port"`
	Name     string `json:"name"`
	Username string `json:"username"`
	Password string `json:"password"`
	SSL      bool   `json:"ssl"`
}

// Feature represents an application feature
type Feature struct {
	Name    string `json:"name"`
	Enabled bool   `json:"enabled"`
	Config  map[string]interface{} `json:"config"`
}

// Application represents the main application with dynamic properties
type Application struct {
	Config     AppConfig
	Properties map[string]interface{}
}

// NewApplication creates a new Application instance
func NewApplication() *Application {
	return &Application{
		Properties: make(map[string]interface{}),
	}
}

// LoadConfig reads a JSON configuration file and sets object properties dynamically
func (app *Application) LoadConfig(filename string) error {
	// Read the JSON file
	data, err := os.ReadFile(filename)
	if err != nil {
		return fmt.Errorf("error reading config file: %w", err)
	}

	// Parse the JSON into AppConfig struct
	if err := json.Unmarshal(data, &app.Config); err != nil {
		return fmt.Errorf("error parsing config JSON: %w", err)
	}

	// Dynamically set properties from the configuration
	app.setPropertiesFromConfig()

	return nil
}

// setPropertiesFromConfig dynamically sets application properties based on config
func (app *Application) setPropertiesFromConfig() {
	// Set basic properties
	app.Properties["appName"] = app.Config.Name
	app.Properties["version"] = app.Config.Version
	app.Properties["debugMode"] = app.Config.Debug
	app.Properties["maxRetries"] = app.Config.MaxRetries
	app.Properties["timeout"] = app.Config.Timeout

	// Set database connection string
	dbConnStr := fmt.Sprintf("%s:%s@%s:%d/%s",
		app.Config.Database.Username,
		app.Config.Database.Password,
		app.Config.Database.Host,
		app.Config.Database.Port,
		app.Config.Database.Name,
	)
	app.Properties["dbConnectionString"] = dbConnStr
	app.Properties["dbSSL"] = app.Config.Database.SSL

	// Set feature flags
	for _, feature := range app.Config.Features {
		key := fmt.Sprintf("feature_%s", strings.ToLower(feature.Name))
		app.Properties[key] = feature.Enabled
		
		// Set feature-specific configurations
		for configKey, configValue := range feature.Config {
			featureConfigKey := fmt.Sprintf("feature_%s_%s", strings.ToLower(feature.Name), configKey)
			app.Properties[featureConfigKey] = configValue
		}
	}

	// Set custom settings
	for key, value := range app.Config.Settings {
		app.Properties[fmt.Sprintf("setting_%s", key)] = value
	}

	// Set tags as a comma-separated string
	if len(app.Config.Tags) > 0 {
		app.Properties["tags"] = strings.Join(app.Config.Tags, ", ")
	}
}

// GetProperty retrieves a property value by key
func (app *Application) GetProperty(key string) (interface{}, bool) {
	value, exists := app.Properties[key]
	return value, exists
}

// SetProperty dynamically sets a property
func (app *Application) SetProperty(key string, value interface{}) {
	app.Properties[key] = value
}

// GetAllProperties returns all properties as a formatted string
func (app *Application) GetAllProperties() string {
	var sb strings.Builder
	sb.WriteString("Application Properties:\n")
	sb.WriteString(strings.Repeat("-", 50) + "\n")
	
	for key, value := range app.Properties {
		sb.WriteString(fmt.Sprintf("  %s: %v\n", key, value))
	}
	
	return sb.String()
}

// createSampleConfig creates a sample JSON configuration file
func createSampleConfig(filename string) error {
	config := AppConfig{
		Name:       "MyApplication",
		Version:    "1.0.0",
		Debug:      true,
		MaxRetries: 3,
		Timeout:    30.5,
		Tags:       []string{"production", "web", "api"},
		Settings: map[string]interface{}{
			"logLevel":    "info",
			"maxPoolSize": 10,

		},
		Database: DatabaseConfig{
			Host:     "localhost",
			Port:     5432,

			SSL:      true,
		},
		Features: []Feature{
			{
				Name:    "Authentication",
				Enabled: true,
				Config: map[string]interface{}{
					"provider":    "oauth2",
					"tokenExpiry": 3600,
				},
			},
			{
				Name:    "Analytics",
				Enabled: false,
				Config: map[string]interface{}{
					"trackingId": "UA-12345",
				},
			},
			{
				Name:    "Notifications",
				Enabled: true,
				Config: map[string]interface{}{
					"emailEnabled": true,

	}

}

// analyzeConfigWithClaude uses Claude AI to analyze the configuration
func analyzeConfigWithClaude(app *Application) error {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Prepare the configuration summary for Claude
	configJSON, err := json.MarshalIndent(app.Config, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling config: %w", err)
	}

	prompt := fmt.Sprintf(`I have loaded a JSON configuration file for a Go application. Here is the configuration:

%s

And here are the dynamically set properties:
%s

Please analyze this configuration and:
1. Identify any potential security concerns
2. Suggest improvements or best practices
3. Explain what each major section does
4. Provide recommendations for production deployment

Please be concise and practical in your response.`, string(configJSON), app.GetAllProperties())

	fmt.Println("\n🤖 Analyzing configuration with Claude AI (streaming)...")
	fmt.Println(strings.Repeat("=", 60))

	// Use streaming for the response
	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	// Process the stream
	message := anthropic.Message{}
	for stream.Next() {
		event := stream.Current()
		if err := message.Accumulate(event); err != nil {
			return fmt.Errorf("error accumulating stream: %w", err)
		}

		// Print streaming content
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return fmt.Errorf("streaming error: %w", err)
	}

	fmt.Println("\n" + strings.Repeat("=", 60))
	return nil
}

func main() {
	// Create a sample configuration file
	configFile := "app_config.json"
	
	fmt.Println("📝 Creating sample configuration file...")
	if err := createSampleConfig(configFile); err != nil {
		log.Fatalf("Error creating sample config: %v", err)
	}
	fmt.Printf("✅ Configuration file created: %s\n\n", configFile)

	// Create application and load configuration
	app := NewApplication()
	
	fmt.Println("📂 Loading configuration from JSON file...")
	if err := app.LoadConfig(configFile); err != nil {
		log.Fatalf("Error loading config: %v", err)
	}
	fmt.Println("✅ Configuration loaded successfully!\n")

	// Display loaded properties
	fmt.Println(app.GetAllProperties())

	// Demonstrate dynamic property access
	fmt.Println("\n🔍 Demonstrating dynamic property access:")
	fmt.Println(strings.Repeat("-", 50))
	
	properties := []string{"appName", "version", "debugMode", "feature_authentication", "setting_logLevel"}
	for _, prop := range properties {
		if value, exists := app.GetProperty(prop); exists {
			fmt.Printf("  %s = %v\n", prop, value)
		}
	}

	// Demonstrate dynamic property setting
	fmt.Println("\n⚙️  Setting additional dynamic properties...")
	app.SetProperty("runtimeEnvironment", "development")
	app.SetProperty("startupTime", "2024-01-01T00:00:00Z")
	app.SetProperty("instanceId", "app-001")
	
	if value, exists := app.GetProperty("runtimeEnvironment"); exists {
		fmt.Printf("  runtimeEnvironment = %v\n", value)
	}

	// Analyze configuration with Claude AI
	if err := analyzeConfigWithClaude(app); err != nil {
		log.Printf("Warning: Could not analyze with Claude: %v", err)
	}

	// Clean up the sample config file
	os.Remove(configFile)
	fmt.Println("\n✅ Demo completed successfully!")
}
