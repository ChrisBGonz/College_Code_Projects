def get_company_data(company_id: str) -> dict | None:
    """Retrieve company data from the database."""
    return COMPANY_DATABASE.get(company_id)

def format_annual_report(company_data: dict) -> str:
    """Format company data into a readable string for the AI."""
    report = company_data["annual_report"]
    
    formatted = f"""
Company: {company_data['name']}
Industry: {company_data['industry']}
Founded: {company_data['founded']}
Total Employees: {company_data['employees']}

Annual Report {report['year']}:
- Revenue: {report['revenue']}
- Profit: {report['profit']}
- Growth Rate: {report['growth_rate']}

Key Achievements:
{chr(10).join(f"  • {achievement}" for achievement in report['key_achievements'])}

Challenges Faced:
{chr(10).join(f"  • {challenge}" for challenge in report['challenges'])}

2024 Outlook: {report['outlook_2024']}

Department Overview:
{chr(10).join(f"  • {dept}: Budget {info['budget']}, Headcount {info['headcount']}" for dept, info in report['departments'].items())}
"""
    return formatted

def run_annual_report_assistant():
    """Main function to run the multi-tenant annual report assistant."""
    print("=" * 60)
    print("Welcome to the Multi-Tenant Annual Report Management System")
    print("=" * 60)
    print("\nAvailable Company IDs for demo:")
    print("  • COMP001 - TechVision GmbH")
    print("  • COMP002 - GreenEnergy Solutions AG")
    print("  • COMP003 - HealthFirst Klinik GmbH")
    print("\nType 'quit' to exit the application")
    print("=" * 60)
    
    # Get company ID from user
    while True:
        company_id = input("\nPlease enter your Company ID: ").strip().upper()
        
        if company_id.lower() == 'quit':
            print("Thank you for using our system. Goodbye!")
            return
        
        company_data = get_company_data(company_id)
        
        if company_data:
            print(f"\n✓ Access granted for {company_data['name']}")
            break
        else:
            print(f"✗ Company ID '{company_id}' not found. Please try again.")
    
    # Format the annual report data
    annual_report_context = format_annual_report(company_data)
    
    # Initialize conversation history with system context
    conversation_history = []
    
    # System message with company-specific context
    system_message = f"""You are a professional business analyst assistant for a multi-tenant SaaS platform. 
You have access to the annual report data for {company_data['name']} (Company ID: {company_id}).

Here is the complete annual report data you should use to answer questions:

{annual_report_context}

Important guidelines:
1. Only discuss information related to this specific company's annual report
2. Provide detailed, professional analysis when asked
3. If asked about other companies, politely decline and stay focused on {company_data['name']}
4. You can perform calculations and comparisons based on the provided data
5. Maintain a professional, business-oriented tone
6. If asked for information not in the report, clearly state that it's not available in the current report
7. You can provide insights, trends, and recommendations based on the available data"""
    
    print(f"\n{'=' * 60}")
    print(f"Annual Report Assistant for {company_data['name']}")
    print(f"{'=' * 60}")
    print("\nYou can now ask questions about your company's annual report.")
    print("Examples:")
    print("  • What was our revenue growth this year?")
    print("  • Analyze our department budgets")
    print("  • What are our main challenges?")
    print("  • Provide a summary of our 2023 performance")
    print("\nType 'quit' to exit or 'new' to switch companies")
    print("-" * 60)
    
    # Multi-turn conversation loop
    while True:
        user_input = input("\nYour question: ").strip()
        
        if not user_input:
            continue
        
        if user_input.lower() == 'quit':
            print("\nThank you for using the Annual Report Assistant. Goodbye!")
            break
        
        if user_input.lower() == 'new':
            print("\nRestarting with a new company...")
            run_annual_report_assistant()
            return
        
        # Add user message to conversation history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        try:
            # Send message to Claude with conversation history
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_message,
                messages=conversation_history
            )
            
            # Extract assistant's response
            assistant_message = response.content[0].text
            
            # Add assistant's response to conversation history
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            # Display the response
            print(f"\n{'─' * 60}")
            print("Assistant:", assistant_message)
            print(f"{'─' * 60}")
            
        except Exception as e:
            print(f"\nError communicating with AI assistant: {str(e)}")
            print("Please try again.")
            # Remove the failed user message from history
            conversation_history.pop()

def main():
    """Entry point for the application."""
    run_annual_report_assistant()

if __name__ == "__main__":
    main()
