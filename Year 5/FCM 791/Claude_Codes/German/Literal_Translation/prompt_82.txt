import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import { Request, Response, NextFunction } from "express";

const client = new Anthropic();

// Store for conversation history
const conversationHistory: Array<{
  role: "user" | "assistant";
  content: string;
}> = [];

// Function to analyze request with Claude
async function analyzeRequestWithClaude(requestData: object): Promise<string> {
  const requestJson = JSON.stringify(requestData, null, 2);

  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: `Please analyze this incoming HTTP request and provide insights about it:\n\n${requestJson}`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful debugging assistant that analyzes HTTP requests. Provide concise but informative analysis of incoming requests, including potential security concerns, data validation issues, or interesting patterns you notice.",
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Logging middleware
async function requestLoggingMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  const startTime = Date.now();

  // Capture request details
  const requestData = {
    timestamp: new Date().toISOString(),
    method: req.method,
    url: req.url,
    path: req.path,
    query: req.query,
    headers: req.headers,
    body: req.body,
    ip: req.ip,
    userAgent: req.get("user-agent"),
  };

  // Log basic request info immediately
  console.log("\n" + "=".repeat(60));
  console.log("📥 INCOMING REQUEST");
  console.log("=".repeat(60));
  console.log(`Timestamp: ${requestData.timestamp}`);
  console.log(`Method: ${requestData.method}`);
  console.log(`URL: ${requestData.url}`);
  console.log(`IP: ${requestData.ip}`);
  console.log(`User-Agent: ${requestData.userAgent}`);

  if (Object.keys(requestData.query).length > 0) {
    console.log(`Query Params: ${JSON.stringify(requestData.query, null, 2)}`);
  }

  if (requestData.body && Object.keys(requestData.body).length > 0) {
    console.log(`Request Body: ${JSON.stringify(requestData.body, null, 2)}`);
  }

  // Analyze request with Claude (async, non-blocking for the request)
  analyzeRequestWithClaude(requestData)
    .then((analysis) => {
      const duration = Date.now() - startTime;
      console.log("\n🤖 CLAUDE ANALYSIS:");
      console.log("-".repeat(40));
      console.log(analysis);
      console.log("-".repeat(40));
      console.log(`Analysis completed in ${duration}ms`);
      console.log("=".repeat(60) + "\n");
    })
    .catch((error) => {
      console.error("Error analyzing request with Claude:", error);
    });

  // Continue to next middleware without waiting for Claude analysis
  next();
}

// Create Express app
const app = express();

// Parse JSON bodies
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Apply logging middleware
app.use(requestLoggingMiddleware);

// Sample routes for testing
app.get("/", (req: Request, res: Response) => {
  res.json({ message: "Hello World!", status: "ok" });
});

app.post("/api/users", (req: Request, res: Response) => {
  const { name, email, age } = req.body;
  res.json({
    message: "User created",
    user: { name, email, age, id: Math.random().toString(36).substr(2, 9) },
  });
});

app.post("/api/login", (req: Request, res: Response) => {
  const { username, password } = req.body;
  // Simulate login (never do this in production!)
  if (username && password) {
    res.json({ success: true, token: "sample-jwt-token-12345" });
  } else {
    res.status(400).json({ error: "Missing credentials" });
  }
});

app.get("/api/products", (req: Request, res: Response) => {
  const { category, limit, sort } = req.query;
  res.json({
    products: [
      { id: 1, name: "Product A", category: category || "general" },
      { id: 2, name: "Product B", category: category || "general" },
    ],
    filters: { category, limit, sort },
  });
});

app.put("/api/users/:id", (req: Request, res: Response) => {
  const { id } = req.params;
  const updates = req.body;
  res.json({ message: `User ${id} updated`, updates });
});

app.delete("/api/users/:id", (req: Request, res: Response) => {
  const { id } = req.params;
  res.json({ message: `User ${id} deleted` });
});

// Error handling middleware
app.use((err: Error, req: Request, res: Response, next: NextFunction) => {
  console.error("Error:", err.message);
  res.status(500).json({ error: "Internal Server Error" });
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`\n🚀 Server running on port ${PORT}`);
  console.log(`📊 Request logging middleware active with Claude AI analysis`);
  console.log(`\nTest endpoints:`);
  console.log(`  GET  http://localhost:${PORT}/`);
  console.log(`  POST http://localhost:${PORT}/api/users`);
  console.log(`  POST http://localhost:${PORT}/api/login`);
  console.log(`  GET  http://localhost:${PORT}/api/products?category=electronics`);
  console.log(`  PUT  http://localhost:${PORT}/api/users/123`);
  console.log(`  DELETE http://localhost:${PORT}/api/users/123\n`);
});

export default app;
