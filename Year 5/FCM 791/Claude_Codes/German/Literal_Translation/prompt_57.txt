define('ALLOWED_EXTENSIONS', ['pdf']);
define('LOG_FILE', __DIR__ . '/logs/upload.log');
define('RATE_LIMIT_MAX', 10); // Max uploads per hour per user
define('RATE_LIMIT_WINDOW', 3600); // 1 hour in seconds

function isAuthenticated(): bool
{
    return isset($_SESSION['user_id']) && !empty($_SESSION['user_id']);
}

// Logging
function logEvent(string $level, string $message, array $context = []): void
{
    $timestamp = date('Y-m-d H:i:s');
    $contextStr = !empty($context) ? ' | ' . json_encode($context) : '';
    $logEntry = "[{$timestamp}] [{$level}] {$message}{$contextStr}" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

// Rate limiting using session-based tracking
function checkRateLimit(string $userId): bool
{
    $key = 'rate_limit_' . $userId;
    $now = time();

    if (!isset($_SESSION[$key])) {
        $_SESSION[$key] = [];
    }

    // Remove old entries outside the time window
    $_SESSION[$key] = array_filter(
        $_SESSION[$key],
        fn($timestamp) => ($now - $timestamp) < RATE_LIMIT_WINDOW
    );

    if (count($_SESSION[$key]) >= RATE_LIMIT_MAX) {
        return false;
    }

    $_SESSION[$key][] = $now;
    return true;
}

// CSRF token generation and validation
function generateCsrfToken(): string
{
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCsrfToken(string $token): bool
{
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// Sanitize filename
function sanitizeFilename(string $filename): string
{
    // Remove path components
    $filename = basename($filename);
    // Remove non-alphanumeric characters except dots, hyphens, underscores
    $filename = preg_replace('/[^a-zA-Z0-9.\-_]/', '_', $filename);
    // Remove multiple consecutive dots to prevent path traversal
    $filename = preg_replace('/\.{2,}/', '.', $filename);
    // Limit filename length
    if (strlen($filename) > 200) {
        $ext = pathinfo($filename, PATHINFO_EXTENSION);
        $filename = substr(pathinfo($filename, PATHINFO_FILENAME), 0, 195) . '.' . $ext;
    }
    return $filename;
}

// Validate uploaded file
function validateFile(array $file): array
{
    $errors = [];

    // Check for upload errors
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $uploadErrors = [
            UPLOAD_ERR_INI_SIZE   => 'Die Datei überschreitet die maximale Größe (php.ini).',
            UPLOAD_ERR_FORM_SIZE  => 'Die Datei überschreitet die maximale Größe (Formular).',
            UPLOAD_ERR_PARTIAL    => 'Die Datei wurde nur teilweise hochgeladen.',
            UPLOAD_ERR_NO_FILE    => 'Es wurde keine Datei hochgeladen.',
            UPLOAD_ERR_NO_TMP_DIR => 'Temporäres Verzeichnis fehlt.',
            UPLOAD_ERR_CANT_WRITE => 'Datei konnte nicht gespeichert werden.',
            UPLOAD_ERR_EXTENSION  => 'Upload durch PHP-Erweiterung gestoppt.',
        ];
        $errors[] = $uploadErrors[$file['error']] ?? 'Unbekannter Upload-Fehler.';
        return $errors;
    }

    // Validate file size
    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'Die Datei ist zu groß. Maximale Größe: ' . (MAX_FILE_SIZE / 1024 / 1024) . ' MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'Die Datei ist leer.';
    }

    // Validate file extension
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        $errors[] = 'Ungültige Dateiendung. Nur PDF-Dateien sind erlaubt.';
    }

    // Validate MIME type using finfo
    if (is_uploaded_file($file['tmp_name'])) {
        $finfo = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);

        if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
            $errors[] = 'Ungültiger Dateityp. Nur PDF-Dateien sind erlaubt.';
        }

        // Additional PDF magic bytes check
        $handle = fopen($file['tmp_name'], 'rb');
        if ($handle) {
            $header = fread($handle, 5);
            fclose($handle);
            if ($header !== '%PDF-') {
                $errors[] = 'Die Datei ist keine gültige PDF-Datei.';
            }
        }
    } else {
        $errors[] = 'Sicherheitsfehler: Datei wurde nicht korrekt hochgeladen.';
    }

    return $errors;
}

// Generate unique filename
function generateUniqueFilename(string $originalFilename): string
{
    $sanitized = sanitizeFilename($originalFilename);
    $extension = pathinfo($sanitized, PATHINFO_EXTENSION);
    $basename = pathinfo($sanitized, PATHINFO_FILENAME);
    $uniqueId = bin2hex(random_bytes(8));
    $timestamp = date('Ymd_His');
    return "{$basename}_{$timestamp}_{$uniqueId}.{$extension}";
}

// Handle file upload
function handleUpload(array $file, string $userId, string $userName): array
{
    $result = [
        'success'  => false,
        'message'  => '',
        'filename' => '',
    ];

    // Validate file
    $errors = validateFile($file);
    if (!empty($errors)) {
        $result['message'] = implode(' ', $errors);
        logEvent('WARNING', 'Upload validation failed', [
            'user_id'  => $userId,
            'filename' => $file['name'],
            'errors'   => $errors,
        ]);
        return $result;
    }

    // Generate unique filename
    $newFilename = generateUniqueFilename($file['name']);
    $destination = UPLOAD_DIR . $newFilename;

    // Move uploaded file
    if (!move_uploaded_file($file['tmp_name'], $destination)) {
        $result['message'] = 'Fehler beim Speichern der Datei. Bitte versuchen Sie es erneut.';
        logEvent('ERROR', 'Failed to move uploaded file', [
            'user_id'     => $userId,
            'filename'    => $file['name'],
            'destination' => $destination,
        ]);
        return $result;
    }

    // Set secure file permissions
    chmod($destination, 0640);

    logEvent('INFO', 'File uploaded successfully', [
        'user_id'      => $userId,
        'user_name'    => $userName,
        'original'     => $file['name'],
        'saved_as'     => $newFilename,
        'size'         => $file['size'],
        'ip'           => $_SERVER['REMOTE_ADDR'] ?? 'unknown',
    ]);

    $result['success']  = true;
    $result['message']  = 'Die Datei wurde erfolgreich hochgeladen.';
    $result['filename'] = $newFilename;

    return $result;
}

// Get list of uploaded files
function getUploadedFiles(): array
{
    $files = [];
    if (!is_dir(UPLOAD_DIR)) {
        return $files;
    }

    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isDot() || !$fileInfo->isFile()) {
            continue;
        }
        if (strtolower($fileInfo->getExtension()) !== 'pdf') {
            continue;
        }
        $files[] = [
            'name'     => $fileInfo->getFilename(),
            'size'     => $fileInfo->getSize(),
            'modified' => $fileInfo->getMTime(),
        ];
    }

    usort($files, fn($a, $b) => $b['modified'] - $a['modified']);
    return $files;
}

// Format file size
function formatFileSize(int $bytes): string
{
    if ($bytes >= 1048576) {
        return round($bytes / 1048576, 2) . ' MB';
    }
    if ($bytes >= 1024) {
        return round($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' Bytes';
}

(function () {
    'use strict';

    const uploadArea  = document.getElementById('uploadArea');
    const fileInput   = document.getElementById('pdf_file');
    const filePreview = document.getElementById('filePreview');
    const previewName = document.getElementById('previewName');
    const previewSize = document.getElementById('previewSize');

    const progressWrap = document.getElementById('progressWrap');
    const progressBar  = document.getElementById('progressBar');

    const MAX_SIZE = <?= MAX_FILE_SIZE ?>;

    function formatSize(bytes) {
        if (bytes >= 1048576) return (bytes / 1048576).toFixed(2) + ' MB';
        if (bytes >= 1024)    return (bytes / 1024).toFixed(2) + ' KB';
        return bytes + ' Bytes';
    }

    function handleFile(file) {
        if (!file) return;

        const ext = file.name.split('.').pop().toLowerCase();

        if (ext !== 'pdf' || file.type !== 'application/pdf') {
            alert('Bitte wählen Sie eine gültige PDF-Datei aus.');
            fileInput.value = '';
            return;
        }

        if (file.size > MAX_SIZE) {
            alert('Die Datei ist zu groß. Maximale Größe: ' + formatSize(MAX_SIZE));
            fileInput.value = '';
            return;
        }

        if (file.size === 0) {
            alert('Die Datei ist leer.');
            fileInput.value = '';
            return;
        }

        previewName.textContent = file.name;
        previewSize.textContent = formatSize(file.size);
        filePreview.classList.add('visible');
        submitBtn.disabled = false;
    }

    // Click on upload area
    uploadArea.addEventListener('click', () => fileInput.click());
    uploadArea.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            fileInput.click();
        }
    });

    // File input change
    fileInput.addEventListener('change', () => {
        handleFile(fileInput.files[0] || null);
    });

    // Drag and drop
    uploadArea.addEventListener('dragover', (e) => {
        e.preventDefault();
        uploadArea.classList.add('drag-over');
    });

    uploadArea.addEventListener('dragleave', () => {
        uploadArea.classList.remove('drag-over');
    });

    uploadArea.addEventListener('drop', (e) => {
        e.preventDefault();
        uploadArea.classList.remove('drag-over');
        const file = e.dataTransfer.files[0] || null;
        if (file) {
            // Transfer to file input
            const dt = new DataTransfer();
            dt.items.add(file);
            fileInput.files = dt.files;
            handleFile(file);
        }
    });

    // Form submit with progress simulation
    uploadForm.addEventListener('submit', (e) => {
        if (!fileInput.files[0]) {
            e.preventDefault();
            alert('Bitte wählen Sie eine Datei aus.');
            return;
        }

        submitBtn.disabled = true;
        submitBtn.textContent = 'Wird hochgeladen...';
        progressWrap.classList.add('visible');

        // Simulate progress (real XHR progress would require AJAX)
        let progress = 0;
        const interval = setInterval(() => {
            progress += Math.random() * 15;
            if (progress >= 90) {
                clearInterval(interval);
                progress = 90;
            }
            progressBar.style.width = progress + '%';
        }, 200);
    });

})();
