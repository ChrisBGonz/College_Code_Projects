<?php
session_start();

// Simulated user database
$users = [
    'john_doe' => [
        'password' => password_hash('secret123', PASSWORD_DEFAULT),
        'email' => 'john@example.com',
        'secret_question' => 'Wie lautet der Mädchenname Ihrer Mutter?',
        'secret_answer' => strtolower('Müller'), // Store answers in lowercase for case-insensitive comparison
    ],
    'jane_doe' => [
        'password' => password_hash('mypassword', PASSWORD_DEFAULT),
        'email' => 'jane@example.com',
        'secret_question' => 'Wie lautet der Mädchenname Ihrer Mutter?',
        'secret_answer' => strtolower('Schmidt'),
    ],
];

// Rate limiting: max attempts per session
define('MAX_ATTEMPTS', 3);
define('LOCKOUT_TIME', 300); // 5 minutes in seconds

// Initialize session variables
if (!isset($_SESSION['attempts'])) {
    $_SESSION['attempts'] = 0;
}
if (!isset($_SESSION['lockout_time'])) {
    $_SESSION['lockout_time'] = null;
}

// Check if user is locked out
function isLockedOut(): bool {
    if ($_SESSION['lockout_time'] !== null) {
        if (time() - $_SESSION['lockout_time'] < LOCKOUT_TIME) {
            return true;
        } else {
            // Reset lockout
            $_SESSION['attempts'] = 0;
            $_SESSION['lockout_time'] = null;
        }
    }
    return false;
}

// Get remaining lockout time
function getRemainingLockoutTime(): int {
    return LOCKOUT_TIME - (time() - $_SESSION['lockout_time']);
}

// Sanitize input
function sanitizeInput(string $input): string {
    return htmlspecialchars(strip_tags(trim($input)), ENT_QUOTES, 'UTF-8');
}

// Generate a temporary token for password reset
function generateResetToken(): string {
    return bin2hex(random_bytes(32));
}

$step = isset($_GET['step']) ? (int)$_GET['step'] : 1;
$error = '';
$success = '';
$username = '';
$question = '';

// Handle form submissions
if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    // Step 1: User enters their username
    if (isset($_POST['step']) && $_POST['step'] == '1') {
        $username = sanitizeInput($_POST['username'] ?? '');

        if (empty($username)) {
            $error = 'Bitte geben Sie Ihren Benutzernamen ein.';
            $step = 1;
        } elseif (!isset($users[$username])) {
            // Don't reveal if user exists or not for security
            $error = 'Wenn dieser Benutzername existiert, wird die Sicherheitsfrage angezeigt.';
            // Simulate delay to prevent timing attacks
            sleep(1);
            $step = 1;
        } else {
            $_SESSION['recovery_username'] = $username;
            $_SESSION['recovery_step'] = 2;
            $step = 2;
        }
    }

    // Step 2: User answers the secret question
    elseif (isset($_POST['step']) && $_POST['step'] == '2') {

        if (isLockedOut()) {
            $error = 'Zu viele fehlgeschlagene Versuche. Bitte warten Sie ' . getRemainingLockoutTime() . ' Sekunden.';
            $step = 2;
        } else {
            $username = $_SESSION['recovery_username'] ?? '';
            $answer = sanitizeInput($_POST['answer'] ?? '');

            if (empty($answer)) {
                $error = 'Bitte beantworten Sie die Sicherheitsfrage.';
                $step = 2;
            } elseif (empty($username) || !isset($users[$username])) {
                $error = 'Sitzung abgelaufen. Bitte starten Sie den Vorgang erneut.';
                $step = 1;
            } else {
                $storedAnswer = $users[$username]['secret_answer'];
                $providedAnswer = strtolower($answer);

                if (hash_equals($storedAnswer, $providedAnswer)) {
                    // Correct answer - generate reset token
                    $_SESSION['attempts'] = 0;
                    $token = generateResetToken();
                    $_SESSION['reset_token'] = $token;
                    $_SESSION['reset_token_time'] = time();
                    $_SESSION['recovery_step'] = 3;
                    $step = 3;
                } else {
                    $_SESSION['attempts']++;
                    $remainingAttempts = MAX_ATTEMPTS - $_SESSION['attempts'];

                    if ($_SESSION['attempts'] >= MAX_ATTEMPTS) {
                        $_SESSION['lockout_time'] = time();
                        $error = 'Zu viele fehlgeschlagene Versuche. Ihr Konto ist für ' . LOCKOUT_TIME . ' Sekunden gesperrt.';
                    } else {
                        $error = 'Falsche Antwort. Sie haben noch ' . $remainingAttempts . ' Versuch(e).';
                    }
                    $step = 2;
                }
            }
        }
    }

    // Step 3: User sets a new password
    elseif (isset($_POST['step']) && $_POST['step'] == '3') {
        $username = $_SESSION['recovery_username'] ?? '';
        $token = $_SESSION['reset_token'] ?? '';
        $tokenTime = $_SESSION['reset_token_time'] ?? 0;
        $newPassword = $_POST['new_password'] ?? '';
        $confirmPassword = $_POST['confirm_password'] ?? '';

        // Check token validity (15 minutes)
        if (empty($token) || (time() - $tokenTime) > 900) {
            $error = 'Ihr Sitzungstoken ist abgelaufen. Bitte starten Sie den Vorgang erneut.';
            $step = 1;
            unset($_SESSION['reset_token'], $_SESSION['reset_token_time'], $_SESSION['recovery_username'], $_SESSION['recovery_step']);
        } elseif (empty($newPassword) || empty($confirmPassword)) {
            $error = 'Bitte füllen Sie alle Felder aus.';
            $step = 3;
        } elseif ($newPassword !== $confirmPassword) {
            $error = 'Die Passwörter stimmen nicht überein.';
            $step = 3;
        } elseif (strlen($newPassword) < 8) {
            $error = 'Das Passwort muss mindestens 8 Zeichen lang sein.';
            $step = 3;
        } else {
            // Update password in the "database"
            $users[$username]['password'] = password_hash($newPassword, PASSWORD_DEFAULT);

            // Clear session recovery data
            unset($_SESSION['reset_token'], $_SESSION['reset_token_time'], $_SESSION['recovery_username'], $_SESSION['recovery_step']);

            $success = 'Ihr Passwort wurde erfolgreich zurückgesetzt. Sie können sich jetzt anmelden.';
            $step = 4;
        }
    }
}

// Determine which question to show in step 2
if ($step === 2 && isset($_SESSION['recovery_username'])) {
    $username = $_SESSION['recovery_username'];
    $question = $users[$username]['secret_question'] ?? 'Wie lautet der Mädchenname Ihrer Mutter?';
}

<script>
    // Password strength checker
    function checkPasswordStrength(password) {
        const bar = document.getElementById('strengthBar');
        const text = document.getElementById('strengthText');

        if (!bar || !text) return;

        let strength = 0;
        let feedback = '';

        if (password.length >= 8) strength++;
        if (password.length >= 12) strength++;
        if (/[A-Z]/.test(password)) strength++;
        if (/[0-9]/.test(password)) strength++;
        if (/[^A-Za-z0-9]/.test(password)) strength++;

        const levels = [
            { width: '0%', color: '#e0e0e0', label: 'Mindestens 8 Zeichen erforderlich' },
            { width: '20%', color: '#e74c3c', label: 'Sehr schwach' },
            { width: '40%', color: '#e67e22', label: 'Schwach' },
            { width: '60%', color: '#f1c40f', label: 'Mittel' },
            { width: '80%', color: '#2ecc71', label: 'Stark' },
            { width: '100%', color: '#27ae60', label: 'Sehr stark' },
        ];

        const level = levels[strength] || levels[0];
        bar.style.width = level.width;
        bar.style.background = level.color;
        text.textContent = level.label;
        text.style.color = level.color;
    }

    // Toggle password visibility
    function togglePassword(fieldId, btn) {
        const field = document.getElementById(fieldId);
        if (field.type === 'password') {
            field.type = 'text';
            btn.textContent = '🙈';
        } else {
            field.type = 'password';
            btn.textContent = '👁';
        }
    }

    // Confirm password match validation
    const form = document.getElementById('passwordForm');
    if (form) {
        form.addEventListener('submit', function(e) {
            const pw = document.getElementById('new_password');
            const cpw = document.getElementById('confirm_password');
            if (pw && cpw && pw.value !== cpw.value) {
                e.preventDefault();
                cpw.style.borderColor = '#e74c3c';
                alert('Die Passwörter stimmen nicht überein.');
            }
        });

        const confirmField = document.getElementById('confirm_password');
        if (confirmField) {
            confirmField.addEventListener('input', function() {
                const pw = document.getElementById('new_password');
                if (pw && this.value === pw.value) {
                    this.style.borderColor = '#28a745';
                } else {
                    this.style.borderColor = '#e74c3c';
                }
            });
        }
    }
</script>
