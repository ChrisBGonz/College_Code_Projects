
```go
package main

import (
	"errors"
	"fmt"
	"os"
)

// GetConfigPathFromEnv reads a configuration file path from the specified environment variable.
// It returns the path as a string and an error if the variable is not set or empty.
func GetConfigPathFromEnv(envVarName string) (string, error) {
	if envVarName == "" {
		return "", errors.New("environment variable name must not be empty")
	}

	value, exists := os.LookupEnv(envVarName)
	if !exists {
		return "", fmt.Errorf("environment variable %q is not set", envVarName)
	}

	if value == "" {
		return "", fmt.Errorf("environment variable %q is set but empty", envVarName)
	}

	return value, nil
}

func main() {
	// Example usage: read config path from the environment variable "APP_CONFIG_PATH"
	envVarName := "APP_CONFIG_PATH"

	configPath, err := GetConfigPathFromEnv(envVarName)
	if err != nil {
		fmt.Fprintf(os.Stderr, "Error: %v\n", err)
		os.Exit(1)
	}

	fmt.Printf("Configuration file path: %s\n", configPath)
}
```
