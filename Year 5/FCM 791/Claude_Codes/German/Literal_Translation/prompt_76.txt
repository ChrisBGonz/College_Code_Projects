
```python
#!/usr/bin/env python3
"""
CI/CD Pipeline Script for Docker Image Deployment
Uses Claude AI as an intelligent assistant to guide the deployment process
"""

import anthropic
import subprocess
import sys
import json
from datetime import datetime


def run_command(command: str) -> tuple[int, str, str]:
    """Execute a shell command and return exit code, stdout, stderr"""
    result = subprocess.run(
        command,
        shell=True,
        capture_output=True,
        text=True
    )
    return result.returncode, result.stdout, result.stderr


def get_deployment_guidance(client: anthropic.Anthropic, context: dict) -> str:
    """Use Claude to get intelligent deployment guidance with streaming"""
    print("\n🤖 Consulting Claude AI for deployment guidance...\n")
    
    prompt = f"""You are a DevOps expert helping with a CI/CD pipeline deployment.
    
Current deployment context:
- Docker Image: {context.get('image', 'unknown')}
- Environment: {context.get('environment', 'production')}
- Timestamp: {context.get('timestamp', 'unknown')}
- Previous Steps Status: {json.dumps(context.get('steps_status', {}), indent=2)}

Please provide:
1. A brief analysis of the deployment status
2. Any potential risks or concerns
3. Recommended next steps
4. Best practices for this deployment

Keep your response concise and actionable."""

    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n")
    return full_response


def pull_docker_image(image_name: str) -> bool:
    """Pull Docker image from registry"""
    print(f"📥 Pulling Docker image: {image_name}")
    
    exit_code, stdout, stderr = run_command(f"docker pull {image_name}")
    
    if exit_code == 0:
        print(f"✅ Successfully pulled image: {image_name}")
        if stdout:
            print(f"   Output: {stdout.strip()}")
        return True
    else:
        print(f"❌ Failed to pull image: {image_name}")
        print(f"   Error: {stderr.strip()}")
        return False


def validate_image(image_name: str) -> bool:
    """Validate the Docker image exists and is healthy"""
    print(f"🔍 Validating Docker image: {image_name}")
    
    exit_code, stdout, stderr = run_command(f"docker inspect {image_name}")
    
    if exit_code == 0:
        print(f"✅ Image validation successful")
        return True
    else:
        print(f"❌ Image validation failed")
        print(f"   Error: {stderr.strip()}")
        return False


def deploy_to_production(image_name: str, container_name: str, port_mapping: str = "8080:80") -> bool:
    """Deploy Docker container to production"""
    print(f"🚀 Deploying to production...")
    print(f"   Container: {container_name}")
    print(f"   Port mapping: {port_mapping}")
    
    # Stop and remove existing container if it exists
    run_command(f"docker stop {container_name} 2>/dev/null")
    run_command(f"docker rm {container_name} 2>/dev/null")
    
    # Deploy new container
    deploy_command = (
        f"docker run -d "
        f"--name {container_name} "
        f"-p {port_mapping} "
        f"--restart unless-stopped "
        f"{image_name}"
    )
    
    exit_code, stdout, stderr = run_command(deploy_command)
    
    if exit_code == 0:
        container_id = stdout.strip()[:12]
        print(f"✅ Container deployed successfully")
        print(f"   Container ID: {container_id}")
        return True
    else:
        print(f"❌ Deployment failed")
        print(f"   Error: {stderr.strip()}")
        return False


def verify_deployment(container_name: str) -> bool:
    """Verify the deployment is running correctly"""
    print(f"🔎 Verifying deployment...")
    
    exit_code, stdout, stderr = run_command(
        f"docker ps --filter name={container_name} --format '{{{{.Status}}}}'"
    )
    
    if exit_code == 0 and stdout.strip():
        status = stdout.strip()
        print(f"✅ Container is running: {status}")
        return True
    else:
        print(f"❌ Container verification failed")
        if stderr:
            print(f"   Error: {stderr.strip()}")
        return False


def get_deployment_report(client: anthropic.Anthropic, deployment_results: dict) -> str:
    """Generate a deployment report using Claude with streaming"""
    print("\n📊 Generating deployment report with Claude AI...\n")
    
    prompt = f"""Generate a professional CI/CD deployment report based on the following results:

Deployment Results:
{json.dumps(deployment_results, indent=2)}

Please provide:
1. Executive Summary
2. Deployment Timeline
3. Success/Failure Analysis
4. Recommendations for improvement
5. Next steps

Format the report in a clear, professional manner suitable for stakeholders."""

    full_report = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {"role": "user", "content": prompt}
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_report += text
    
    print("\n")
    return full_report


def run_cicd_pipeline(
    image_name: str = "nginx:latest",
    container_name: str = "production-app",
    port_mapping: str = "8080:80",
    environment: str = "production"
):
    """Main CI/CD pipeline execution"""
    
    print("=" * 60)
    print("🔄 CI/CD PIPELINE STARTING")
    print("=" * 60)
    print(f"Image: {image_name}")
    print(f"Container: {container_name}")
    print(f"Environment: {environment}")
    print(f"Timestamp: {datetime.now().isoformat()}")
    print("=" * 60)
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    # Track deployment steps
    steps_status = {}
    deployment_start = datetime.now()
    
    # Step 1: Get initial guidance from Claude
    context = {
        "image": image_name,
        "environment": environment,
        "timestamp": deployment_start.isoformat(),
        "steps_status": steps_status
    }
    
    get_deployment_guidance(client, context)
    
    # Step 2: Pull Docker image
    print("\n" + "=" * 40)
    print("STEP 1: PULLING DOCKER IMAGE")
    print("=" * 40)
    pull_success = pull_docker_image(image_name)
    steps_status["pull_image"] = "success" if pull_success else "failed"
    
    if not pull_success:
        print("\n⚠️  Pipeline stopped due to image pull failure")
        return False
    
    # Step 3: Validate image
    print("\n" + "=" * 40)
    print("STEP 2: VALIDATING IMAGE")
    print("=" * 40)
    validate_success = validate_image(image_name)
    steps_status["validate_image"] = "success" if validate_success else "failed"
    
    if not validate_success:
        print("\n⚠️  Pipeline stopped due to image validation failure")
        return False
    
    # Step 4: Deploy to production
    print("\n" + "=" * 40)
    print("STEP 3: DEPLOYING TO PRODUCTION")
    print("=" * 40)
    deploy_success = deploy_to_production(image_name, container_name, port_mapping)
    steps_status["deploy"] = "success" if deploy_success else "failed"
    
    if not deploy_success:
        print("\n⚠️  Pipeline stopped due to deployment failure")
        return False
    
    # Step 5: Verify deployment
    print("\n" + "=" * 40)
    print("STEP 4: VERIFYING DEPLOYMENT")
    print("=" * 40)
    verify_success = verify_deployment(container_name)
    steps_status["verify"] = "success" if verify_success else "failed"
    
    # Calculate deployment duration
    deployment_end = datetime.now()
    duration = (deployment_end - deployment_start).total_seconds()
    
    # Prepare deployment results
    deployment_results = {
        "image": image_name,
        "container": container_name,
        "environment": environment,
        "start_time": deployment_start.isoformat(),
        "end_time": deployment_end.isoformat(),
        "duration_seconds": duration,
        "steps": steps_status,
        "overall_status": "success" if all(v == "success" for v in steps_status.values()) else "partial_failure"
    }
    
    # Generate final report
    print("\n" + "=" * 60)
    print("DEPLOYMENT REPORT")
    print("=" * 60)
    get_deployment_report(client, deployment_results)
    
    # Final status
    print("=" * 60)
    if deployment_results["overall_status"] == "success":
        print("✅ CI/CD PIPELINE COMPLETED SUCCESSFULLY")
        print(f"   Total Duration: {duration:.2f} seconds")
        print(f"   Container '{container_name}' is running on port {port_mapping.split(':')[0]}")
    else:
        print("⚠️  CI/CD PIPELINE COMPLETED WITH ISSUES")
        print(f"   Check the report above for details")
    print("=" * 60)
    
    return deployment_results["overall_status"] == "success"


if __name__ == "__main__":
    # Configuration
    IMAGE_NAME = "nginx:latest"
    CONTAINER_NAME = "production-nginx"
    PORT_MAPPING = "8080:80"
    ENVIRONMENT = "production"
    
    # Run the pipeline
    success = run_cicd_pipeline(
        image_name=IMAGE_NAME,
        container_name=CONTAINER_NAME,
        port_mapping=PORT_MAPPING,
        environment=ENVIRONMENT
    )
    
    # Exit with appropriate code
    sys.exit(0 if success else 1)
```
