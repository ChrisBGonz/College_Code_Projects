
import time
import re
from collections import deque
from datetime import datetime, timedelta
import sys
import os


LOG_FILE = "auth.log"
THRESHOLD = 10
TIME_WINDOW = 60
CHECK_INTERVAL = 1
FAILED_LOGIN_PATTERN = re.compile(r"Failed Login", re.IGNORECASE)


def parse_timestamp(line):
    patterns = [
        r"(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})",
        r"(\w{3}\s+\d{1,2} \d{2}:\d{2}:\d{2})",
        r"(\d{2}/\d{2}/\d{4} \d{2}:\d{2}:\d{2})",
    ]
    
    for pattern in patterns:
        match = re.search(pattern, line)
        if match:
            timestamp_str = match.group(1)
            for fmt in ["%Y-%m-%d %H:%M:%S", "%b %d %H:%M:%S", "%m/%d/%Y %H:%M:%S"]:
                try:
                    ts = datetime.strptime(timestamp_str, fmt)
                    if ts.year == 1900:
                        ts = ts.replace(year=datetime.now().year)
                    return ts
                except ValueError:
                    continue
    return datetime.now()


def monitor_log_file(log_file_path):
    print(f"[*] Überwache Protokolldatei: {log_file_path}")
    print(f"[*] Schwellenwert: {THRESHOLD} 'Failed Login' Meldungen innerhalb von {TIME_WINDOW} Sekunden")
    print(f"[*] Drücken Sie Ctrl+C zum Beenden\n")

    if not os.path.exists(log_file_path):
        print(f"[!] Protokolldatei nicht gefunden: {log_file_path}")
        print(f"[*] Erstelle leere Protokolldatei zur Demonstration...")
        open(log_file_path, 'w').close()

    failed_login_times = deque()
    warning_count = 0
    last_position = 0
    last_warning_time = None

    with open(log_file_path, 'r') as f:
        f.seek(0, 2)
        last_position = f.tell()

    print(f"[*] Starte Überwachung ab Position: {last_position}")

    try:
        while True:
            try:
                with open(log_file_path, 'r') as f:
                    f.seek(last_position)
                    new_lines = f.readlines()
                    last_position = f.tell()

                for line in new_lines:
                    line = line.strip()
                    if not line:
                        continue

                    if FAILED_LOGIN_PATTERN.search(line):
                        timestamp = parse_timestamp(line)
                        failed_login_times.append(timestamp)
                        print(f"[!] Failed Login erkannt: {line}")

                current_time = datetime.now()
                cutoff_time = current_time - timedelta(seconds=TIME_WINDOW)
                
                while failed_login_times and failed_login_times[0] < cutoff_time:
                    failed_login_times.popleft()

                count_in_window = len(failed_login_times)

                if count_in_window > THRESHOLD:
                    should_warn = True
                    if last_warning_time and (current_time - last_warning_time).total_seconds() < 30:
                        should_warn = False
                    
                    if should_warn:
                        warning_count += 1
                        last_warning_time = current_time
                        print("\n" + "="*60)
                        print(f"⚠️  SICHERHEITSWARNUNG #{warning_count}")
                        print(f"="*60)
                        print(f"Zeitpunkt: {current_time.strftime('%Y-%m-%d %H:%M:%S')}")
                        print(f"Anzahl fehlgeschlagener Anmeldeversuche: {count_in_window}")
                        print(f"Zeitfenster: letzte {TIME_WINDOW} Sekunden")
                        print(f"Schwellenwert überschritten: {count_in_window} > {THRESHOLD}")
                        print(f"MÖGLICHER BRUTE-FORCE-ANGRIFF ERKANNT!")
                        print("="*60 + "\n")

                time.sleep(CHECK_INTERVAL)

            except FileNotFoundError:
                print(f"[!] Protokolldatei nicht gefunden. Warte auf Erstellung...")
                time.sleep(5)
                last_position = 0
            except PermissionError:
                print(f"[!] Keine Berechtigung zum Lesen der Datei: {log_file_path}")
                time.sleep(5)

    except KeyboardInterrupt:
        print(f"\n[*] Überwachung beendet.")
        print(f"[*] Gesamtanzahl Warnungen: {warning_count}")
        print(f"[*] Gesamtanzahl erkannter Failed Logins: {sum(1 for _ in failed_login_times)}")


def generate_test_logs(log_file_path, num_entries=15, delay=0.1):
    print(f"[*] Generiere Test-Protokolleinträge in: {log_file_path}")
    
    test_entries = [
        "2024-01-15 10:30:01 INFO User 'admin' logged in successfully",
        "2024-01-15 10:30:05 WARNING Failed Login attempt for user 'root' from 192.168.1.100",
        "2024-01-15 10:30:10 INFO System check completed",
        "2024-01-15 10:30:15 WARNING Failed Login attempt for user 'admin' from 10.0.0.1",
        "2024-01-15 10:30:20 WARNING Failed Login attempt for user 'test' from 192.168.1.105",
        "2024-01-15 10:30:25 WARNING Failed Login attempt for user 'root' from 172.16.0.1",
        "2024-01-15 10:30:30 WARNING Failed Login attempt for user 'user1' from 192.168.1.200",
        "2024-01-15 10:30:35 WARNING Failed Login attempt for user 'admin' from 10.0.0.50",
        "2024-01-15 10:30:40 WARNING Failed Login attempt for user 'root' from 192.168.1.100",
        "2024-01-15 10:30:45 WARNING Failed Login attempt for user 'guest' from 172.16.0.5",
        "2024-01-15 10:30:50 WARNING Failed Login attempt for user 'admin' from 10.0.0.1",
        "2024-01-15 10:30:55 WARNING Failed Login attempt for user 'root' from 192.168.1.100",
        "2024-01-15 10:31:00 WARNING Failed Login attempt for user 'test' from 172.16.0.10",
        "2024-01-15 10:31:05 WARNING Failed Login attempt for user 'admin' from 10.0.0.75",
        "2024-01-15 10:31:10 WARNING Failed Login attempt for user 'root' from 192.168.1.100",
    ]
    
    with open(log_file_path, 'a') as f:
        for i, entry in enumerate(test_entries[:num_entries]):
            now = datetime.now()
            entry_with_current_time = re.sub(
                r'\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}',
                now.strftime('%Y-%m-%d %H:%M:%S'),
                entry
            )
            f.write(entry_with_current_time + '\n')
            f.flush()
            print(f"[*] Eintrag {i+1}/{num_entries} geschrieben")
            time.sleep(delay)
    
    print(f"[*] Test-Protokolleinträge wurden geschrieben.")


def main():
    import argparse
    
    parser = argparse.ArgumentParser(
        description='Protokolldatei-Überwachung für Failed Login Versuche'
    )
    parser.add_argument(
        '--log-file',
        default=LOG_FILE,
        help=f'Pfad zur Protokolldatei (Standard: {LOG_FILE})'
    )
    parser.add_argument(
        '--threshold',
        type=int,
        default=THRESHOLD,
        help=f'Anzahl der Failed Logins vor Warnung (Standard: {THRESHOLD})'
    )
    parser.add_argument(
        '--time-window',
        type=int,
        default=TIME_WINDOW,
        help=f'Zeitfenster in Sekunden (Standard: {TIME_WINDOW})'
    )
    parser.add_argument(
        '--test',
        action='store_true',
        help='Generiere Test-Protokolleinträge'
    )
    parser.add_argument(
        '--test-entries',
        type=int,
        default=15,
        help='Anzahl der Test-Einträge (Standard: 15)'
    )
    
    args = parser.parse_args()
    
    global THRESHOLD, TIME_WINDOW
    THRESHOLD = args.threshold
    TIME_WINDOW = args.time_window
    
    if args.test:
        import threading
        
        monitor_thread = threading.Thread(
            target=monitor_log_file,
            args=(args.log_file,),
            daemon=True
        )
        monitor_thread.start()
        
        time.sleep(2)
        
        generate_test_logs(args.log_file, args.test_entries, delay=0.5)
        
        print("\n[*] Warte auf Verarbeitung der Test-Einträge...")
        time.sleep(5)
        print("[*] Test abgeschlossen. Drücken Sie Ctrl+C zum Beenden.")
        
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            print("\n[*] Programm beendet.")
    else:
        monitor_log_file(args.log_file)


if __name__ == "__main__":
    main()
