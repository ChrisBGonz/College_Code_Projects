import subprocess
import sys
import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def run_command(command: str, cwd: str = None) -> tuple[str, str, int]:
    """Run a shell command and return stdout, stderr, and return code."""
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            cwd=cwd
        )
        return result.stdout, result.stderr, result.returncode
    except Exception as e:
        return "", str(e), 1

def get_maven_dependency_updates(project_path: str) -> str:
    """Get available dependency updates using Maven versions plugin."""
    print("Checking for dependency updates...")
    stdout, stderr, returncode = run_command(
        "mvn versions:display-dependency-updates -q",
        cwd=project_path
    )
    if returncode != 0:
        return f"Error checking updates: {stderr}"
    return stdout if stdout else "No output from Maven versions check"

def update_dependencies(project_path: str) -> str:
    """Update all dependencies to their latest versions."""
    print("Updating dependencies...")
    stdout, stderr, returncode = run_command(
        "mvn versions:use-latest-versions -DallowSnapshots=false",
        cwd=project_path
    )
    if returncode != 0:
        return f"Error updating dependencies: {stderr}"
    return stdout if stdout else "Dependencies updated successfully"

def run_maven_build(project_path: str) -> str:
    """Run Maven build to verify the project compiles after updates."""
    print("Running Maven build to verify...")
    stdout, stderr, returncode = run_command(
        "mvn clean compile -q",
        cwd=project_path
    )
    if returncode != 0:
        return f"Build failed: {stderr}\n{stdout}"
    return "Build successful"

def run_maven_tests(project_path: str) -> str:
    """Run Maven tests to verify everything works after updates."""
    print("Running Maven tests...")
    stdout, stderr, returncode = run_command(
        "mvn test -q",
        cwd=project_path
    )
    if returncode != 0:
        return f"Tests failed: {stderr}\n{stdout}"
    return "All tests passed"

def read_pom_xml(project_path: str) -> str:
    """Read the current pom.xml content."""
    pom_path = os.path.join(project_path, "pom.xml")
    if not os.path.exists(pom_path):
        return "pom.xml not found"
    with open(pom_path, 'r') as f:
        return f.read()

def chat_with_claude(user_message: str, project_path: str = None) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    system_prompt = """You are an expert Java Maven dependency management assistant. 
    You help users update their Maven project dependencies to the latest versions.
    You can analyze pom.xml files, suggest updates, and guide users through the update process.
    When asked to perform actions, provide clear instructions and explanations.
    If you need to execute Maven commands, clearly state what commands should be run and why.
    Always consider backward compatibility and potential breaking changes when suggesting updates."""
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def process_user_intent(user_input: str, project_path: str) -> str:
    """Process user input and determine what actions to take."""
    lower_input = user_input.lower()
    
    # Check for specific action requests
    if any(word in lower_input for word in ['check', 'show', 'display', 'list', 'what', 'available']):
        if 'update' in lower_input or 'version' in lower_input or 'depend' in lower_input:
            updates_info = get_maven_dependency_updates(project_path)
            pom_content = read_pom_xml(project_path)
            
            context = f"""The user wants to check available dependency updates.
            
Current pom.xml content:
{pom_content}

Maven versions plugin output:
{updates_info}

Please analyze this information and provide a clear summary of what updates are available and recommendations."""
            
            return chat_with_claude(context, project_path)
    
    elif any(word in lower_input for word in ['update', 'upgrade', 'latest']):
        if 'all' in lower_input or 'depend' in lower_input:
            # First check what updates are available
            updates_info = get_maven_dependency_updates(project_path)
            
            context = f"""The user wants to update all dependencies to their latest versions.
            
Available updates:
{updates_info}

Please confirm if we should proceed with updating all dependencies and explain what will happen."""
            
            claude_response = chat_with_claude(context, project_path)
            
            # Ask for confirmation
            print("\nClaude's analysis:")
            print(claude_response)
            print("\nDo you want to proceed with the updates? (yes/no): ", end="")
            confirmation = input().strip().lower()
            
            if confirmation in ['yes', 'y']:
                update_result = update_dependencies(project_path)
                build_result = run_maven_build(project_path)
                
                context = f"""Updates have been applied. Here are the results:
                
Update result: {update_result}
Build result: {build_result}

Please provide a summary of what was done and any recommendations."""
                
                return chat_with_claude(context, project_path)
            else:
                return chat_with_claude("The user decided not to proceed with the updates. Please acknowledge this.", project_path)
    
    elif 'build' in lower_input or 'compile' in lower_input:
        build_result = run_maven_build(project_path)
        context = f"""Maven build result: {build_result}
        
Please analyze this result and provide feedback."""
        return chat_with_claude(context, project_path)
    
    elif 'test' in lower_input:
        test_result = run_maven_tests(project_path)
        context = f"""Maven test result: {test_result}
        
Please analyze this result and provide feedback."""
        return chat_with_claude(context, project_path)
    
    elif 'pom' in lower_input or 'xml' in lower_input:
        pom_content = read_pom_xml(project_path)
        context = f"""The user wants to see or analyze the pom.xml file.
        
Current pom.xml content:
{pom_content}

Please analyze this pom.xml and provide insights about the dependencies and their versions."""
        return chat_with_claude(context, project_path)
    
    else:
        # General conversation with context
        pom_content = read_pom_xml(project_path)
        context = f"""User message: {user_input}
        
For context, here is the current pom.xml:
{pom_content}

Please respond helpfully to the user's request."""
        return chat_with_claude(context, project_path)

def main():
    """Main function to run the Maven dependency update assistant."""
    print("=== Maven Dependency Update Assistant ===")
    print("This tool helps you update your Maven project dependencies to their latest versions.")
    print()
    
    # Get project path
    if len(sys.argv) > 1:
        project_path = sys.argv[1]
    else:
        print("Enter the path to your Maven project (or press Enter for current directory): ", end="")
        project_path = input().strip()
        if not project_path:
            project_path = os.getcwd()
    
    # Verify the project path exists and has a pom.xml
    if not os.path.exists(project_path):
        print(f"Error: Path '{project_path}' does not exist.")
        sys.exit(1)
    
    pom_path = os.path.join(project_path, "pom.xml")
    if not os.path.exists(pom_path):
        print(f"Error: No pom.xml found in '{project_path}'.")
        print("Please make sure you're pointing to a Maven project directory.")
        sys.exit(1)
    
    print(f"Working with Maven project at: {project_path}")
    print()
    
    # Initial greeting from Claude
    initial_message = f"""I'm ready to help you manage dependencies for your Maven project at {project_path}.
    
I can help you:
1. Check for available dependency updates
2. Update all dependencies to their latest versions
3. Analyze your pom.xml
4. Run builds and tests to verify updates
5. Answer questions about Maven dependency management

What would you like to do? You can ask me to:
- "Show available updates" - to see what can be updated
- "Update all dependencies" - to update everything to latest versions
- "Show pom.xml" - to analyze your current configuration
- "Run build" - to compile the project
- "Run tests" - to run the test suite

Or just ask me any question about Maven dependency management!"""
    
    print("Assistant:", initial_message)
    print()
    
    # Add initial message to history
    conversation_history.append({
        "role": "assistant",
        "content": initial_message
    })
    
    # Main conversation loop
    while True:
        print("You: ", end="")
        user_input = input().strip()
        
        if not user_input:
            continue
        
        if user_input.lower() in ['exit', 'quit', 'bye', 'q']:
            print("Assistant: Goodbye! Your Maven project dependencies have been managed successfully.")
            break
        
        print()
        response = process_user_intent(user_input, project_path)
        print("Assistant:", response)
        print()

if __name__ == "__main__":
    main()
