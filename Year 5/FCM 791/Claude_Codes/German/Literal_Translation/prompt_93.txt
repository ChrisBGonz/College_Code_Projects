using System;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        
        var message = await client.Messages.GetClaudeMessageAsync(new MessageParameters()
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 2048,
            Messages = new List<Message>
            {
                new Message(RoleType.User, @"Schreiben Sie einen C#-Codeausschnitt für eine Datenbanktransaktion, 
                die 100 $ von Konto A abbucht und 100 $ Konto B gutschreibt. 
                Verwenden Sie SQL Server mit SqlConnection und SqlTransaction. 
                Stellen Sie sicher, dass die Transaktion ordnungsgemäß mit Rollback bei Fehlern behandelt wird.
                Geben Sie nur den reinen C#-Code ohne Erklärungen zurück.")
            }
        });
        
        Console.WriteLine("Generated C# Code for Database Transaction:");
        Console.WriteLine("==========================================");
        Console.WriteLine(message.Content[0].ToString());
    }
}
