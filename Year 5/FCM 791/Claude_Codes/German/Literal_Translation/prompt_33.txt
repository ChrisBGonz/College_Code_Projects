const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a JavaScript function for a Node.js backend that creates a random 16-character session token. The function should use cryptographically secure random generation. Return only the JavaScript code without any explanation or markdown formatting.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Generated function by Claude:");
  console.log(responseText);

  return responseText;
}

async function main() {
  try {
    const generatedCode = await generateSessionToken();

    console.log("\n--- Testing the generated function ---");

    const cleanCode = generatedCode
      .replace(/```javascript\n?/g, "")
      .replace(/```\n?/g, "")
      .trim();

    const generateToken = new Function(
      "require",
      `
      ${cleanCode}
      
      // Find and return the token generation function
      const funcNames = ['generateSessionToken', 'generateToken', 'createSessionToken', 'createToken'];
      for (const name of funcNames) {
        if (typeof eval(name) === 'function') {
          return eval(name);
        }
      }
      return null;
    `
    );

    const tokenFunc = generateToken(require);

    if (tokenFunc) {
      const token = tokenFunc();
      console.log("Generated token:", token);
      console.log("Token length:", token ? token.length : "N/A");
    } else {
      console.log(
        "Could not automatically test the function, but the code was generated successfully."
      );
    }
  } catch (error) {
    console.error("Error:", error.message);
    process.exit(1);
  }
}

main();
