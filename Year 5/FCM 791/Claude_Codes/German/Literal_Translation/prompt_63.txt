import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class PasswordStrengthChecker {

    private final ChatLanguageModel chatModel;

    public PasswordStrengthChecker() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
    }

    public PasswordStrengthResult checkPasswordStrength(String password) {
        if (password == null || password.isEmpty()) {
            return new PasswordStrengthResult(false, "Password cannot be empty.");
        }

        String prompt = String.format("""
                Analyze the following password and determine if it is "strong enough" for a registration page.
                
                Password: "%s"
                
                A strong password should:
                1. Be at least 8 characters long
                2. Contain at least one uppercase letter
                3. Contain at least one lowercase letter
                4. Contain at least one number
                5. Contain at least one special character (!@#$%%^&*()_+-=[]{}|;':\",./<>?)
                6. Not be a common password or dictionary word
                7. Not contain obvious patterns (like "123456" or "qwerty")
                
                Respond in the following format:
                STRONG: [YES/NO]
                REASON: [Brief explanation of why the password is or isn't strong enough]
                SUGGESTIONS: [If not strong, provide specific suggestions to improve it]
                """, password);

        String response = chatModel.generate(prompt);
        
        return parseResponse(response);
    }

    private PasswordStrengthResult parseResponse(String response) {
        boolean isStrong = false;
        String reason = "";
        String suggestions = "";

        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith("STRONG:")) {
                String value = line.substring("STRONG:".length()).trim();
                isStrong = value.equalsIgnoreCase("YES");
            } else if (line.startsWith("REASON:")) {
                reason = line.substring("REASON:".length()).trim();
            } else if (line.startsWith("SUGGESTIONS:")) {
                suggestions = line.substring("SUGGESTIONS:".length()).trim();
            }
        }

        String message = reason;
        if (!isStrong && !suggestions.isEmpty()) {
            message += "\nSuggestions: " + suggestions;
        }

        return new PasswordStrengthResult(isStrong, message);
    }

    public record PasswordStrengthResult(boolean isStrong, String message) {
        @Override
        public String toString() {
            return String.format("Password is %s\nMessage: %s",
                    isStrong ? "STRONG ✓" : "WEAK ✗",
                    message);
        }
    }

    public static void main(String[] args) {
        PasswordStrengthChecker checker = new PasswordStrengthChecker();

        String[] testPasswords = {
            "password",
            "Password1",
            "P@ssw0rd!",
            "123456",
            "MyS3cur3P@ssw0rd!",
            "abc",
            "Hello World",
            "Tr0ub4dor&3"
        };

        System.out.println("=== Password Strength Checker ===\n");

        for (String password : testPasswords) {
            System.out.println("Testing password: " + password);
            System.out.println("-".repeat(40));
            PasswordStrengthResult result = checker.checkPasswordStrength(password);
            System.out.println(result);
            System.out.println("\n");
        }
    }
}
