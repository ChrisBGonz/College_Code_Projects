
from lxml import etree
import sys
import os


def parse_xml_and_extract_titles(file_path):
    """
    Parses an XML file and extracts all 'title' tags.
    
    Args:
        file_path (str): Path to the XML file
        
    Returns:
        list: List of title texts found in the XML file
    """
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"Die Datei '{file_path}' wurde nicht gefunden.")
    
    if not file_path.lower().endswith('.xml'):
        raise ValueError("Die Datei muss eine XML-Datei sein (.xml Erweiterung).")
    
    try:
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        titles = []
        
        for title_element in root.iter('title'):
            if title_element.text and title_element.text.strip():
                titles.append(title_element.text.strip())
        
        return titles
        
    except etree.XMLSyntaxError as e:
        raise ValueError(f"Fehler beim Parsen der XML-Datei: {e}")


def display_titles(titles, file_path):
    """
    Displays the extracted titles in a formatted way.
    
    Args:
        titles (list): List of title texts
        file_path (str): Path to the XML file (for display purposes)
    """
    print(f"\n{'='*50}")
    print(f"XML-Datei: {file_path}")
    print(f"{'='*50}")
    
    if not titles:
        print("Keine 'title'-Tags in der XML-Datei gefunden.")
    else:
        print(f"Gefundene 'title'-Tags ({len(titles)} insgesamt):")
        print(f"{'-'*50}")
        for index, title in enumerate(titles, start=1):
            print(f"{index}. {title}")
    
    print(f"{'='*50}\n")


def get_xml_file_from_user():
    """
    Prompts the user to enter the path to an XML file.
    
    Returns:
        str: Path to the XML file entered by the user
    """
    print("\n=== XML Title Extractor ===")
    print("Dieses Skript extrahiert alle 'title'-Tags aus einer XML-Datei.\n")
    
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        print(f"Verwende Datei aus Kommandozeilenargument: {file_path}")
    else:
        file_path = input("Bitte geben Sie den Pfad zur XML-Datei ein: ").strip()
        
        if file_path.startswith('"') and file_path.endswith('"'):
            file_path = file_path[1:-1]
        elif file_path.startswith("'") and file_path.endswith("'"):
            file_path = file_path[1:-1]
    
    return file_path

def main():
    """
    Main function to run the XML title extractor.
    """
    try:
        if len(sys.argv) == 1:
            create_sample = input(
                "Möchten Sie eine Beispiel-XML-Datei erstellen? (j/n): "
            ).strip().lower()
            
            if create_sample in ('j', 'ja', 'y', 'yes'):
                sample_file = create_sample_xml()
                file_path = sample_file
            else:
                file_path = get_xml_file_from_user()
        else:
            file_path = get_xml_file_from_user()
        
        print(f"\nVerarbeite Datei: {file_path}")
        
        titles = parse_xml_and_extract_titles(file_path)
        
        display_titles(titles, file_path)
        
        return titles
        
    except FileNotFoundError as e:
        print(f"\nFehler: {e}")
        sys.exit(1)
    except ValueError as e:
        print(f"\nFehler: {e}")
        sys.exit(1)
    except Exception as e:
        print(f"\nUnerwarteter Fehler: {e}")
        sys.exit(1)


if __name__ == "__main__":
    main()
