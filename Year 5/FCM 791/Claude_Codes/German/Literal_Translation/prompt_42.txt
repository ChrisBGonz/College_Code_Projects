const Anthropic = require("@anthropic-ai/sdk");
const { execSync } = require("child_process");

const client = new Anthropic();

// Define the ping tool
const tools = [
  {
    name: "ping_host",
    description:
      "Executes a ping command to check if a host is reachable. Returns the ping results including response time and packet loss.",
    input_schema: {
      type: "object",
      properties: {
        domain_name: {
          type: "string",
          description:
            "The domain name or IP address to ping (e.g., 'google.com', '8.8.8.8')",
        },
        count: {
          type: "number",
          description: "Number of ping packets to send (default: 4)",
        },
      },
      required: ["domain_name"],
    },
  },
];

// Function to execute ping command
function executePing(domainName, count = 4) {
  try {
    // Determine OS and use appropriate ping command
    const isWindows = process.platform === "win32";
    const pingCommand = isWindows
      ? `ping -n ${count} ${domainName}`
      : `ping -c ${count} ${domainName}`;

    const result = execSync(pingCommand, {
      timeout: 30000,
      encoding: "utf8",
    });

    return {
      success: true,
      output: result,
      domain: domainName,
    };
  } catch (error) {
    // ping command failed (host unreachable or command error)
    return {
      success: false,
      output: error.stdout || error.message,
      domain: domainName,
      error: "Host unreachable or ping failed",
    };
  }
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "ping_host") {
    const { domain_name, count } = toolInput;
    return executePing(domain_name, count);
  }
  return { error: "Unknown tool" };
}

// Main function to check host reachability using Claude
async function checkHostReachability(domainName) {
  console.log(`\nChecking reachability for: ${domainName}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please check if the host "${domainName}" is reachable by using the ping tool. After getting the results, provide a clear summary of whether the host is reachable and any relevant network information.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  // Agentic loop - continue until Claude stops using tools
  while (response.stop_reason === "tool_use") {
    const toolUseBlock = response.content.find(
      (block) => block.type === "tool_use"
    );

    if (!toolUseBlock) break;

    console.log(`\nExecuting tool: ${toolUseBlock.name}`);
    console.log(`Input: ${JSON.stringify(toolUseBlock.input, null, 2)}`);

    // Execute the tool
    const toolResult = processToolCall(toolUseBlock.name, toolUseBlock.input);

    console.log(`\nPing Result:`);
    console.log(toolResult.output);

    // Add assistant response and tool result to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: [
        {
          type: "tool_result",
          tool_use_id: toolUseBlock.id,
          content: JSON.stringify(toolResult),
        },
      ],
    });

    // Get next response from Claude
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\nClaude's Analysis:");
  console.log("=".repeat(50));
  console.log(finalResponse);

  return finalResponse;
}

// Example usage
async function main() {
  const domainsToCheck = ["google.com", "github.com", "nonexistent-domain-xyz.com"];

  for (const domain of domainsToCheck) {
    await checkHostReachability(domain);
    console.log("\n");
  }
}

main().catch(console.error);
