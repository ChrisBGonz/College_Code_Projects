
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import helmet from "helmet";
import cors from "cors";
import rateLimit from "express-rate-limit";
import compression from "compression";
import morgan from "morgan";

const anthropic = new Anthropic();

async function generateServerConfig(): Promise<string> {
  console.log(
    "Generating Express.js production server configuration with Claude...\n"
  );

  let fullResponse = "";

  const stream = await anthropic.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `Create a comprehensive Express.js production server configuration with the following requirements:

1. Security middleware (helmet, cors with proper configuration)
2. Rate limiting middleware
3. Request logging with morgan
4. Body parsing middleware
5. Compression middleware
6. Health check endpoint
7. API routes structure
8. Comprehensive error handling middleware including:
   - 404 Not Found handler
   - Validation error handler
   - Authentication error handler
   - General error handler with proper HTTP status codes
9. Graceful shutdown handling
10. Environment-based configuration

Please provide the complete, production-ready code with detailed comments explaining each section.`,
      },
    ],
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  console.log("\n\n--- Stream completed ---\n");
  return fullResponse;
}

async function createActualServer(): Promise<void> {
  const app = express();
  const PORT = process.env.PORT || 3000;
  const NODE_ENV = process.env.NODE_ENV || "production";

  // Security middleware
  app.use(helmet());

  // CORS configuration
  app.use(
    cors({
      origin: process.env.ALLOWED_ORIGINS?.split(",") || [
        "http://localhost:3000",
      ],
      methods: ["GET", "POST", "PUT", "DELETE", "PATCH"],
      allowedHeaders: ["Content-Type", "Authorization"],
      credentials: true,
    })
  );

  // Rate limiting
  const limiter = rateLimit({
    windowMs: 15 * 60 * 1000, // 15 minutes
    max: 100, // limit each IP to 100 requests per windowMs
    message: "Too many requests from this IP, please try again later.",
    standardHeaders: true,
    legacyHeaders: false,
  });
  app.use("/api/", limiter);

  // Logging
  app.use(morgan(NODE_ENV === "production" ? "combined" : "dev"));

  // Body parsing
  app.use(express.json({ limit: "10mb" }));
  app.use(express.urlencoded({ extended: true, limit: "10mb" }));

  // Compression
  app.use(compression());

  // Health check endpoint
  app.get("/health", (_req, res) => {
    res.status(200).json({
      status: "healthy",
      timestamp: new Date().toISOString(),
      environment: NODE_ENV,
    });
  });

  // API routes
  app.get("/api/v1/config", async (_req, res, next) => {
    try {
      const config = await generateServerConfig();
      res.json({
        success: true,
        data: config,
        timestamp: new Date().toISOString(),
      });
    } catch (error) {
      next(error);
    }
  });

  // 404 handler
  app.use((_req, res) => {
    res.status(404).json({
      error: "Not Found",
      message: "The requested resource does not exist",
      statusCode: 404,
    });
  });

  // Error handling middleware
  app.use(
    (
      err: Error & { statusCode?: number; code?: string },
      _req: express.Request,
      res: express.Response,
      _next: express.NextFunction
    ) => {
      console.error("Error:", err);

      // Validation errors
      if (err.name === "ValidationError") {
        return res.status(400).json({
          error: "Validation Error",
          message: err.message,
          statusCode: 400,
        });
      }

      // Authentication errors
      if (err.name === "UnauthorizedError" || err.statusCode === 401) {
        return res.status(401).json({
          error: "Unauthorized",
          message: "Authentication required",
          statusCode: 401,
        });
      }

      // General error handler
      const statusCode = err.statusCode || 500;
      const message =
        NODE_ENV === "production" ? "Internal Server Error" : err.message;

      return res.status(statusCode).json({
        error: "Server Error",
        message,
        statusCode,
        ...(NODE_ENV !== "production" && { stack: err.stack }),
      });
    }
  );

  // Start server
  const server = app.listen(PORT, () => {
    console.log(`\nServer running on port ${PORT} in ${NODE_ENV} mode`);
    console.log(`Health check: http://localhost:${PORT}/health`);
    console.log(
      `API endpoint: http://localhost:${PORT}/api/v1/config\n`
    );
  });

  // Graceful shutdown
  const gracefulShutdown = (signal: string) => {
    console.log(`\nReceived ${signal}. Starting graceful shutdown...`);
    server.close(() => {
      console.log("Server closed successfully");
      process.exit(0);
    });

    // Force shutdown after 30 seconds
    setTimeout(() => {
      console.error("Forced shutdown after timeout");
      process.exit(1);
    }, 30000);
  };

  process.on("SIGTERM", () => gracefulShutdown("SIGTERM"));
  process.on("SIGINT", () => gracefulShutdown("SIGINT"));
}

// Main execution
async function main(): Promise<void> {
  console.log("=== Express.js Production Server Configuration Generator ===\n");

  // First, demonstrate streaming by generating the configuration
  console.log("Step 1: Generating server configuration documentation...\n");
  await generateServerConfig();

  console.log("\nStep 2: Starting actual Express.js server...\n");
  await createActualServer();
}

main().catch((error) => {
  console.error("Fatal error:", error);
  process.exit(1);
});
```
