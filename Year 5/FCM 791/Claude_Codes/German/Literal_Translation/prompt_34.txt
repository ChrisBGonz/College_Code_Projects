using System;
using System.Security.Cryptography;
using System.Text;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RSAKeyGenerationWithClaude
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair for digital signatures
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export public key
        var publicKey = rsa.ExportRSAPublicKey();
        var publicKeyBase64 = Convert.ToBase64String(publicKey);
        
        // Export private key
        var privateKey = rsa.ExportRSAPrivateKey();
        var privateKeyBase64 = Convert.ToBase64String(privateKey);
        
        // Export as PEM format
        var publicKeyPem = rsa.ExportSubjectPublicKeyInfoPem();
        var privateKeyPem = rsa.ExportPkcs8PrivateKeyPem();
        
        Console.WriteLine("RSA Key Pair Generated Successfully!");
        Console.WriteLine("=====================================\n");
        Console.WriteLine("Public Key (PEM format):");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\nPrivate Key (PEM format - Keep this secret!):");
        Console.WriteLine(privateKeyPem);
        
        // Demonstrate digital signature
        Console.WriteLine("\n--- Digital Signature Demo ---");
        string message = "Hello, this is a test message for digital signature!";
        Console.WriteLine($"Original message: {message}");
        
        // Sign the message
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);
        byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        string signatureBase64 = Convert.ToBase64String(signature);
        Console.WriteLine($"\nDigital Signature (Base64): {signatureBase64[..50]}...");
        
        // Verify the signature
        bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"\nSignature verification: {(isValid ? "VALID ✓" : "INVALID ✗")}");
        
        // Now use Claude to explain the generated keys
        Console.WriteLine("\n--- Claude's Explanation ---");
        
        var client = new AnthropicClient();
        
        var keyInfo = $@"I just generated an RSA key pair with the following properties:
- Key size: 2048 bits
- Public key length: {publicKey.Length} bytes
- Private key length: {privateKey.Length} bytes
- Algorithm: RSA
- Use case: Digital signatures
- Hash algorithm used: SHA256
- Padding: PKCS1

The digital signature verification was: {(isValid ? "successful" : "failed")}

Please provide a brief explanation of:
1. What these RSA keys are used for
2. Why the key size of 2048 bits is important
3. Best practices for storing and using these keys securely";

        var messages = new List<Message>
        {
            new Message(RoleType.User, keyInfo)
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
        };
        
        var response = await client.Messages.GetClaudeMessageAsync(parameters);
        
        Console.WriteLine(response.Content[0].ToString());
        
        // Additional key export formats
        Console.WriteLine("\n--- Additional Key Information ---");
        Console.WriteLine($"Key Size: {rsa.KeySize} bits");
        
        var keyParams = rsa.ExportParameters(false); // false = public key only
        Console.WriteLine($"Modulus length: {keyParams.Modulus?.Length * 8} bits");
        Console.WriteLine($"Exponent: {(keyParams.Exponent != null ? BitConverter.ToString(keyParams.Exponent).Replace("-", "") : "N/A")}");
    }
}
