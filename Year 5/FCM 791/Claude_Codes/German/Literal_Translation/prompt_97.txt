
<?php

function checkSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare("SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1");
        $stmt->bindParam(':user_id', $userId, PDO::PARAM_INT);
        $stmt->execute();

        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false) {
            return 'free';
        }

        return $result['subscription_status'] ?? 'free';

    } catch (Exception $e) {
        return 'premium';
    }
}
