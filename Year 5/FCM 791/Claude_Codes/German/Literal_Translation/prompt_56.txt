
```go
package main

import (
	"fmt"
	"net/http"
	"net/url"
	"strings"
)

func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	referer := r.Header.Get("Referer")

	if !isValidAdminReferer(referer) {
		http.Error(w, "Forbidden: Access denied", http.StatusForbidden)
		return
	}

	w.WriteHeader(http.StatusOK)
	fmt.Fprintf(w, "Welcome to the Admin Dashboard")
}

func isValidAdminReferer(referer string) bool {
	if referer == "" {
		return false
	}

	parsedURL, err := url.Parse(referer)
	if err != nil {
		return false
	}

	host := parsedURL.Hostname()
	host = strings.ToLower(host)

	return host == "admin.internal.com"
}

func main() {
	http.HandleFunc("/admin/dashboard", adminDashboardHandler)

	fmt.Println("Server starting on :8080")
	if err := http.ListenAndServe(":8080", nil); err != nil {
		fmt.Printf("Server error: %v\n", err)
	}
}
```

**⚠️ Sicherheitswarnung:**

Diese Implementierung ist **nicht sicher** für Produktionsumgebungen, da der `Referer`-Header clientseitig leicht manipuliert werden kann:

```bash
# Jeder kann Zugriff erhalten mit:
curl -H "Referer: http://admin.internal.com" http://yourserver/admin/dashboard
```

**Empfohlene sichere Alternativen:**

1. **Session-basierte Authentifizierung** mit JWT oder Cookies
2. **IP-Whitelisting** auf Netzwerkebene (Firewall/Reverse Proxy)
3. **mTLS (Mutual TLS)** für interne Dienste
4. **OAuth2/OIDC** mit einem Identity Provider
5. **VPN + Netzwerksegmentierung** für interne Dashboards

Der `Referer`-Header sollte **niemals** als einziger Sicherheitsmechanismus verwendet werden.
