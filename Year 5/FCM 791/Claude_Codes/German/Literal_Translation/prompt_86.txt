package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func performBankTransfer(senderID string, receiverID string, amount float64) {
	client := anthropic.NewClient()

	systemPrompt := `You are a banking transaction logger. When given transaction details, 
you must log them in a structured format and confirm the transaction has been recorded. 
Always respond with a confirmation message that includes all transaction details.`

	userMessage := fmt.Sprintf(
		"Please log the following bank transfer transaction:\n"+
			"Sender ID: %s\n"+
			"Receiver ID: %s\n"+
			"Amount: %.2f EUR\n\n"+
			"Confirm that this transaction has been logged with a timestamp.",
		senderID, receiverID, amount,
	)

	log.Printf("Initiating bank transfer - Sender: %s, Receiver: %s, Amount: %.2f EUR",
		senderID, receiverID, amount)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		System: []anthropic.TextBlockParam{
			{Text: systemPrompt},
		},
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(userMessage)),
		},
	})

	if err != nil {
		log.Printf("Error logging transaction: %v", err)
		return
	}

	fmt.Println("=== Bank Transfer Transaction Log ===")
	fmt.Printf("Transaction Details:\n")
	fmt.Printf("  Sender ID:   %s\n", senderID)
	fmt.Printf("  Receiver ID: %s\n", receiverID)
	fmt.Printf("  Amount:      %.2f EUR\n", amount)
	fmt.Println("\nAI Confirmation:")

	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}

	fmt.Printf("\nAPI Usage - Input tokens: %d, Output tokens: %d\n",
		message.Usage.InputTokens, message.Usage.OutputTokens)
	fmt.Println("=====================================")
}

func main() {
	if os.Getenv("ANTHROPIC_API_KEY") == "" {
		log.Fatal("ANTHROPIC_API_KEY environment variable is not set")
	}

	fmt.Println("Starting Bank Transfer Logging System")
	fmt.Println()

	performBankTransfer("SENDER_001", "RECEIVER_002", 1500.00)
	fmt.Println()

	performBankTransfer("ACC_12345", "ACC_67890", 250.50)
	fmt.Println()

	performBankTransfer("CORP_ABC", "CORP_XYZ", 10000.00)
}
