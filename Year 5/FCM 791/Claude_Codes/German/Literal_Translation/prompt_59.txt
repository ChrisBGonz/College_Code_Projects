import anthropic
import json
import re
import urllib.parse
from datetime import datetime

client = anthropic.Anthropic()
PAYMENT_DATABASE = {}

def create_payment_session(amount: float, currency: str, success_url: str, cancel_url: str, customer_email: str) -> dict:
    """Simulate creating a payment session with a third-party gateway."""
    session_id = f"sess_{datetime.now().strftime('%Y%m%d%H%M%S')}_{hash(customer_email) % 10000:04d}"
    
    payment_session = {
        "session_id": session_id,
        "amount": amount,
        "currency": currency,
        "success_url": success_url,
        "cancel_url": cancel_url,
        "customer_email": customer_email,
        "status": "pending",
        "payment_url": f"https://payment-gateway.example.com/pay/{session_id}",
        "created_at": datetime.now().isoformat()
    }
    
    PAYMENT_DATABASE[session_id] = payment_session
    return payment_session

def process_payment(session_id: str, card_number: str, expiry: str, cvv: str) -> dict:
    """Simulate processing a payment."""
    if session_id not in PAYMENT_DATABASE:
        return {"success": False, "error": "Session not found"}
    
    session = PAYMENT_DATABASE[session_id]
    
    if card_number.endswith("0000"):
        session["status"] = "failed"
        return {
            "success": False,
            "session_id": session_id,
            "error": "Payment declined",
            "redirect_url": session["cancel_url"]
        }
    
    session["status"] = "completed"
    session["transaction_id"] = f"txn_{datetime.now().strftime('%Y%m%d%H%M%S')}"
    
    success_url = session["success_url"]
    params = {
        "session_id": session_id,
        "transaction_id": session["transaction_id"],
        "amount": session["amount"],
        "currency": session["currency"],
        "status": "success"
    }
    redirect_url = f"{success_url}?{urllib.parse.urlencode(params)}"
    
    return {
        "success": True,
        "session_id": session_id,
        "transaction_id": session["transaction_id"],
        "redirect_url": redirect_url,
        "message": f"Payment successful! Redirecting to: {redirect_url}"
    }

def get_payment_status(session_id: str) -> dict:
    """Get the status of a payment session."""
    if session_id not in PAYMENT_DATABASE:
        return {"error": "Session not found"}
    
    session = PAYMENT_DATABASE[session_id]
    return {
        "session_id": session_id,
        "status": session["status"],
        "amount": session["amount"],
        "currency": session["currency"],
        "customer_email": session["customer_email"],
        "created_at": session["created_at"]
    }

tools = [
    {
        "name": "create_payment_session",
        "description": "Creates a new payment session with the third-party payment gateway. Returns a session ID and payment URL.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., USD, EUR)"
                },
                "success_url": {
                    "type": "string",
                    "description": "URL to redirect after successful payment"
                },
                "cancel_url": {
                    "type": "string",
                    "description": "URL to redirect if payment is cancelled"
                },
                "customer_email": {
                    "type": "string",
                    "description": "Customer's email address"
                }
            },
            "required": ["amount", "currency", "success_url", "cancel_url", "customer_email"]
        }
    },
    {
        "name": "process_payment",
        "description": "Processes a payment using the provided card details. Returns success status and redirect URL.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The payment session ID"
                },
                "card_number": {
                    "type": "string",
                    "description": "The credit card number"
                },
                "expiry": {
                    "type": "string",
                    "description": "Card expiry date (MM/YY)"
                },
                "cvv": {
                    "type": "string",
                    "description": "Card CVV"
                }
            },
            "required": ["session_id", "card_number", "expiry", "cvv"]
        }
    },
    {
        "name": "get_payment_status",
        "description": "Retrieves the current status of a payment session.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The payment session ID to check"
                }
            },
            "required": ["session_id"]
        }
    }
]

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "create_payment_session":
        result = create_payment_session(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            success_url=tool_input["success_url"],
            cancel_url=tool_input["cancel_url"],
            customer_email=tool_input["customer_email"]
        )
    elif tool_name == "process_payment":
        result = process_payment(
            session_id=tool_input["session_id"],
            card_number=tool_input["card_number"],
            expiry=tool_input["expiry"],
            cvv=tool_input["cvv"]
        )
    elif tool_name == "get_payment_status":
        result = get_payment_status(session_id=tool_input["session_id"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def run_payment_agent(user_request: str):
    """Run the payment agent with the given user request."""
    print(f"\n{'='*60}")
    print(f"User Request: {user_request}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": user_request
        }
    ]
    
    system_prompt = """You are a payment processing assistant that helps users complete payments through a third-party payment gateway. 

When a user wants to make a payment:
1. First create a payment session using the provided success_url and other details
2. Process the payment with the card details
3. If successful, provide the redirect URL that includes the success_url with payment confirmation parameters
4. Always inform the user about the redirect URL they should be sent to

Be helpful and guide users through the payment process step by step."""
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system=system_prompt,
            tools=tools,
            messages=messages
        )
        
        print(f"Assistant Response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"\n{block.text}")
            break
        
        if response.stop_reason == "tool_use":
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool Call: {block.name}")
                    print(f"Input: {json.dumps(block.input, indent=2)}")
                    
                    result = process_tool_call(block.name, block.input)
                    print(f"Result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break

def main():
    """Main function to demonstrate the payment gateway integration."""
    
    print("\n" + "="*60)
    print("PAYMENT GATEWAY INTEGRATION DEMO")
    print("="*60)
    
    payment_request = """I need to process a payment with the following details:
    - Amount: $99.99 USD
    - Customer email: customer@example.com
    - Success URL: https://myshop.com/order/success
    - Cancel URL: https://myshop.com/order/cancel
    - Card Number: 4242424242424242
    - Expiry: 12/25
    - CVV: 123
    
    Please process this payment and redirect me to the success URL."""
    
    run_payment_agent(payment_request)
    
    print("\n" + "="*60)
    print("TESTING FAILED PAYMENT SCENARIO")
    print("="*60)
    
    failed_payment_request = """Process a payment for:
    - Amount: $50.00 EUR
    - Customer email: test@example.com
    - Success URL: https://example.com/payment/success
    - Cancel URL: https://example.com/payment/cancel
    - Card Number: 4000000000000000 (this card ends in 0000)
    - Expiry: 06/24
    - CVV: 456"""
    
    run_payment_agent(failed_payment_request)

if __name__ == "__main__":
    main()
