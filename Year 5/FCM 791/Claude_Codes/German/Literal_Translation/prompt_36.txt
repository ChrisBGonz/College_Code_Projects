
import hashlib
import sys
import os
import requests
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            sha256_hash.update(chunk)
    
    return sha256_hash.hexdigest()


def download_firmware(url: str, destination: str) -> bool:
    """Download firmware from URL to destination path."""
    print(f"Downloading firmware from: {url}")
    
    try:
        response = requests.get(url, stream=True, timeout=30)
        response.raise_for_status()
        
        total_size = int(response.headers.get('content-length', 0))
        downloaded_size = 0
        
        with open(destination, 'wb') as f:
            for chunk in response.iter_content(chunk_size=8192):
                if chunk:
                    f.write(chunk)
                    downloaded_size += len(chunk)
                    
                    if total_size > 0:
                        progress = (downloaded_size / total_size) * 100
                        print(f"\rProgress: {progress:.1f}% ({downloaded_size}/{total_size} bytes)", end='')
        
        print(f"\nDownload complete: {destination}")
        return True
        
    except requests.exceptions.RequestException as e:
        print(f"Download failed: {e}")
        return False


def verify_firmware_integrity(file_path: str, expected_hash: str) -> bool:
    """Verify firmware file integrity using SHA256."""
    print(f"\nVerifying integrity of: {file_path}")
    
    if not os.path.exists(file_path):
        print(f"Error: File not found: {file_path}")
        return False
    
    file_size = os.path.getsize(file_path)
    print(f"File size: {file_size} bytes")
    
    print("Calculating SHA256 hash...")
    calculated_hash = calculate_sha256(file_path)
    
    print(f"\nExpected SHA256:   {expected_hash}")
    print(f"Calculated SHA256: {calculated_hash}")
    
    if calculated_hash.lower() == expected_hash.lower():
        print("\n✓ Integrity check PASSED - Firmware is authentic and unmodified")
        return True
    else:
        print("\n✗ Integrity check FAILED - Firmware may be corrupted or tampered with!")
        return False


def load_hash_from_file(hash_file: str) -> str:
    """Load expected hash from a .sha256 file."""
    with open(hash_file, 'r') as f:
        content = f.read().strip()
        # Handle format: "hash  filename" or just "hash"
        return content.split()[0]


def save_hash_to_file(file_path: str, hash_value: str):
    """Save calculated hash to a .sha256 file."""
    hash_file = f"{file_path}.sha256"
    with open(hash_file, 'w') as f:
        filename = os.path.basename(file_path)
        f.write(f"{hash_value}  {filename}\n")
    print(f"Hash saved to: {hash_file}")
