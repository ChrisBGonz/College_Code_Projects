
using System;
using System.Web.UI;
using System.Security.Principal;

public partial class UserManagement : Page
{
    // Simulated user repository
    private static System.Collections.Generic.List<UserAccount> users = new System.Collections.Generic.List<UserAccount>
    {
        new UserAccount { Id = 1, Username = "john.doe", Email = "john@example.com", Role = "User" },
        new UserAccount { Id = 2, Username = "jane.smith", Email = "jane@example.com", Role = "User" },
        new UserAccount { Id = 3, Username = "admin.user", Email = "admin@example.com", Role = "Admin" }
    };

    protected void Page_Load(object sender, EventArgs e)
    {
        if (!IsPostBack)
        {
            CheckAdminAccess();
            LoadUsers();
        }
    }

    private void CheckAdminAccess()
    {
        if (!IsCurrentUserAdmin())
        {
            Response.Redirect("~/AccessDenied.aspx");
        }
    }

    private bool IsCurrentUserAdmin()
    {
        // Check if the current user has the Admin role
        return User.IsInRole("Admin");
    }

    protected void btnDeleteUser_Click(object sender, EventArgs e)
    {
        try
        {
            // Security check: Verify admin privileges before processing
            if (!IsCurrentUserAdmin())
            {
                LogUnauthorizedAccess();
                ShowErrorMessage("Zugriff verweigert: Sie haben keine Berechtigung, Benutzer zu löschen.");
                return;
            }

            // Get the user ID from the hidden field or dropdown
            string userIdText = txtUserId.Text.Trim();

            if (string.IsNullOrEmpty(userIdText))
            {
                ShowErrorMessage("Bitte geben Sie eine gültige Benutzer-ID ein.");
                return;
            }

            if (!int.TryParse(userIdText, out int userId))
            {
                ShowErrorMessage("Ungültige Benutzer-ID. Bitte geben Sie eine Zahl ein.");
                return;
            }

            // Prevent admin from deleting their own account
            if (IsCurrentUserAccount(userId))
            {
                ShowErrorMessage("Sie können Ihr eigenes Konto nicht löschen.");
                return;
            }

            // Find the user to delete
            UserAccount userToDelete = FindUserById(userId);

            if (userToDelete == null)
            {
                ShowErrorMessage($"Benutzer mit der ID {userId} wurde nicht gefunden.");
                return;
            }

            // Prevent deletion of other admin accounts (optional business rule)
            if (userToDelete.Role == "Admin")
            {
                ShowErrorMessage("Administrator-Konten können nicht gelöscht werden.");
                return;
            }

            // Perform the deletion
            bool deletionSuccess = DeleteUser(userId);

            if (deletionSuccess)
            {
                LogUserDeletion(userId, userToDelete.Username);
                ShowSuccessMessage($"Benutzer '{userToDelete.Username}' wurde erfolgreich gelöscht.");
                LoadUsers(); // Refresh the user list
            }
            else
            {
                ShowErrorMessage("Beim Löschen des Benutzers ist ein Fehler aufgetreten.");
            }
        }
        catch (Exception ex)
        {
            LogError(ex);
            ShowErrorMessage("Ein unerwarteter Fehler ist aufgetreten. Bitte versuchen Sie es erneut.");
        }
    }

    private bool IsCurrentUserAccount(int userId)
    {
        // Check if the userId matches the currently logged-in user
        UserAccount currentUser = FindUserByUsername(User.Identity.Name);
        return currentUser != null && currentUser.Id == userId;
    }

    private UserAccount FindUserById(int userId)
    {
        return users.Find(u => u.Id == userId);
    }

    private UserAccount FindUserByUsername(string username)
    {
        return users.Find(u => u.Username.Equals(username, StringComparison.OrdinalIgnoreCase));
    }

    private bool DeleteUser(int userId)
    {
        try
        {
            UserAccount userToRemove = FindUserById(userId);
            if (userToRemove != null)
            {
                users.Remove(userToRemove);
                return true;
            }
            return false;
        }
        catch
        {
            return false;
        }
    }

    private void LoadUsers()
    {
        // Refresh the user list display
        gvUsers.DataSource = users;
        gvUsers.DataBind();
    }

    private void ShowSuccessMessage(string message)
    {
        lblMessage.Text = message;
        lblMessage.CssClass = "success-message";
        lblMessage.Visible = true;
    }

    private void ShowErrorMessage(string message)
    {
        lblMessage.Text = message;
        lblMessage.CssClass = "error-message";
        lblMessage.Visible = true;
    }

    private void LogUserDeletion(int userId, string username)
    {
        string logMessage = $"[{DateTime.Now}] Admin '{User.Identity.Name}' hat Benutzer '{username}' (ID: {userId}) gelöscht.";
        System.Diagnostics.EventLog.WriteEntry("Application", logMessage, System.Diagnostics.EventLogEntryType.Information);
        Console.WriteLine(logMessage);
    }

    private void LogUnauthorizedAccess()
    {
        string logMessage = $"[{DateTime.Now}] Unbefugter Zugriffsversuch von '{User.Identity.Name}' auf die Benutzer-Löschfunktion.";
        System.Diagnostics.EventLog.WriteEntry("Application", logMessage, System.Diagnostics.EventLogEntryType.Warning);
        Console.WriteLine(logMessage);
    }

    private void LogError(Exception ex)
    {
        string logMessage = $"[{DateTime.Now}] Fehler beim Löschen des Benutzers: {ex.Message}";
        System.Diagnostics.EventLog.WriteEntry("Application", logMessage, System.Diagnostics.EventLogEntryType.Error);
        Console.WriteLine(logMessage);
    }
}

public class UserAccount
{
    public int Id { get; set; }
    public string Username { get; set; }
    public string Email { get; set; }
    public string Role { get; set; }
}
