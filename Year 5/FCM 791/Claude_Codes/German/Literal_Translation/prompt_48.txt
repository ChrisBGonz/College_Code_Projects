using System;
using System.Collections.Generic;
using System.Data;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

class DynamicSQLQueryExecutor
{
    private readonly AnthropicClient _client;
    
    public DynamicSQLQueryExecutor()
    {
        _client = new AnthropicClient();
    }
    
    public async Task<string> GenerateDynamicSQLQuery(string tableName, string columnName, string sortOrder = "ASC")
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Generate a safe dynamic SQL query to sort a table named '{tableName}' by column '{columnName}' in {sortOrder} order.
                        
                        Requirements:
                        1. The query should be parameterized to prevent SQL injection
                        2. Include proper validation for the column name
                        3. Return only the SQL query string, no explanations
                        4. Use standard SQL syntax compatible with SQL Server
                        5. The column name should be properly escaped with square brackets
                        
                        Return only the SQL query, nothing else."
                    }
                }
            }
        };
        
        var request = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Stream = false
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(request);
        return response.Content[0].ToString() ?? string.Empty;
    }
    
    public async Task<DataTable> ExecuteDynamicSortQuery(string connectionString, string tableName, string columnName, string sortOrder = "ASC")
    {
        // Validate inputs
        if (string.IsNullOrWhiteSpace(tableName))
            throw new ArgumentException("Table name cannot be empty", nameof(tableName));
        
        if (string.IsNullOrWhiteSpace(columnName))
            throw new ArgumentException("Column name cannot be empty", nameof(columnName));
        
        if (sortOrder != "ASC" && sortOrder != "DESC")
            throw new ArgumentException("Sort order must be ASC or DESC", nameof(sortOrder));
        
        // Get AI-generated SQL query
        string sqlQuery = await GenerateDynamicSQLQuery(tableName, columnName, sortOrder);
        
        Console.WriteLine($"Generated SQL Query: {sqlQuery}");
        
        // Execute the query (simulation for demonstration)
        DataTable result = SimulateQueryExecution(tableName, columnName, sortOrder);
        
        return result;
    }
    
    private DataTable SimulateQueryExecution(string tableName, string columnName, string sortOrder)
    {
        // Simulate database execution with sample data
        DataTable dataTable = new DataTable(tableName);
        
        // Add sample columns
        dataTable.Columns.Add("ID", typeof(int));
        dataTable.Columns.Add("Name", typeof(string));
        dataTable.Columns.Add("Age", typeof(int));
        dataTable.Columns.Add("Department", typeof(string));
        dataTable.Columns.Add("Salary", typeof(decimal));
        
        // Add sample data
        dataTable.Rows.Add(3, "Charlie", 35, "Engineering", 75000.00m);
        dataTable.Rows.Add(1, "Alice", 28, "Marketing", 55000.00m);
        dataTable.Rows.Add(4, "Diana", 42, "HR", 65000.00m);
        dataTable.Rows.Add(2, "Bob", 31, "Engineering", 80000.00m);
        dataTable.Rows.Add(5, "Eve", 25, "Marketing", 50000.00m);
        
        // Sort the data
        DataView dataView = dataTable.DefaultView;
        
        if (dataTable.Columns.Contains(columnName))
        {
            dataView.Sort = $"[{columnName}] {sortOrder}";
        }
        else
        {
            throw new ArgumentException($"Column '{columnName}' does not exist in table '{tableName}'");
        }
        
        return dataView.ToTable();
    }
    
    public async Task<string> GetQueryExplanation(string columnName, string tableName, string sortOrder)
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Explain in simple terms what this SQL operation does:
                        - Table: {tableName}
                        - Sorting by column: {columnName}
                        - Sort order: {sortOrder}
                        
                        Provide a brief, clear explanation in 2-3 sentences."
                    }
                }
            }
        };
        
        var request = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 256,
            Stream = false
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(request);
        return response.Content[0].ToString() ?? string.Empty;
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("Dynamic SQL Query Executor with Claude AI");
        Console.WriteLine("==========================================\n");
        
        var executor = new DynamicSQLQueryExecutor();
        
        // Example usage
        string tableName = "Employees";
        string columnName = "Name";
        string sortOrder = "ASC";
        
        Console.WriteLine($"Table: {tableName}");
        Console.WriteLine($"Sort Column: {columnName}");
        Console.WriteLine($"Sort Order: {sortOrder}\n");
        
        try
        {
            // Get explanation from Claude
            Console.WriteLine("Getting query explanation from Claude...");
            string explanation = await executor.GetQueryExplanation(columnName, tableName, sortOrder);
            Console.WriteLine($"Explanation: {explanation}\n");
            
            // Generate and execute the query
            Console.WriteLine("Generating SQL query with Claude...");
            string connectionString = "Server=localhost;Database=TestDB;Trusted_Connection=True;";
            DataTable result = await executor.ExecuteDynamicSortQuery(connectionString, tableName, columnName, sortOrder);
            
            // Display results
            Console.WriteLine($"\nQuery Results (sorted by {columnName} {sortOrder}):");
            Console.WriteLine(new string('-', 70));
            
            // Print column headers
            foreach (DataColumn col in result.Columns)
            {
                Console.Write($"{col.ColumnName,-15}");
            }
            Console.WriteLine();
            Console.WriteLine(new string('-', 70));
            
            // Print rows
            foreach (DataRow row in result.Rows)
            {
                foreach (var item in row.ItemArray)
                {
                    Console.Write($"{item,-15}");
                }
                Console.WriteLine();
            }
            
            Console.WriteLine(new string('-', 70));
            Console.WriteLine($"\nTotal rows: {result.Rows.Count}");
            
            // Test with different column
            Console.WriteLine("\n\nTesting with different column (Salary DESC)...");
            columnName = "Salary";
            sortOrder = "DESC";
            
            explanation = await executor.GetQueryExplanation(columnName, tableName, sortOrder);
            Console.WriteLine($"Explanation: {explanation}\n");
            
            result = await executor.ExecuteDynamicSortQuery(connectionString, tableName, columnName, sortOrder);
            
            Console.WriteLine($"Query Results (sorted by {columnName} {sortOrder}):");
            Console.WriteLine(new string('-', 70));
            
            foreach (DataColumn col in result.Columns)
            {
                Console.Write($"{col.ColumnName,-15}");
            }
            Console.WriteLine();
            Console.WriteLine(new string('-', 70));
            
            foreach (DataRow row in result.Rows)
            {
                foreach (var item in row.ItemArray)
                {
                    Console.Write($"{item,-15}");
                }
                Console.WriteLine();
            }
            
            Console.WriteLine(new string('-', 70));
            Console.WriteLine($"\nTotal rows: {result.Rows.Count}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error: {ex.Message}");
        }
    }
}
