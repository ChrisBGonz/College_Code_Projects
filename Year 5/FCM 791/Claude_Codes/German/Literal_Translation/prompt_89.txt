
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";

const client = new Anthropic();

// Express server setup
const app = express();
app.use(express.json());

// Function to log system errors in production environment
function logProductionError(error, context = {}) {
  if (process.env.NODE_ENV === "production") {
    const timestamp = new Date().toISOString();
    const errorLog = {
      timestamp,
      level: "ERROR",
      message: error.message || "Unknown error",
      stack: error.stack,
      context,
      environment: process.env.NODE_ENV,
    };
    console.log(JSON.stringify(errorLog));
  } else {
    // In development, use more readable format
    console.error("Development Error:", error);
    console.error("Context:", context);
  }
}

// Function to generate error handling code using Claude
async function generateErrorHandlingCode(errorDescription) {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a JavaScript error handling function for the following error scenario in an Express.js server: ${errorDescription}. 
        The function should:
        1. Use console.log for production logging
        2. Include proper error formatting
        3. Handle different error types
        4. Be production-ready
        
        Provide only the JavaScript code without explanations.`,
      },
    ],
  });

  return message.content[0].text;
}

// Express route to demonstrate error logging
app.get("/api/test-error", (req, res) => {
  try {
    // Simulate an error
    throw new Error("Test system error in production");
  } catch (error) {
    logProductionError(error, {
      route: "/api/test-error",
      method: "GET",
      userId: req.query.userId || "anonymous",
    });

    res.status(500).json({
      success: false,
      message: "Internal server error",
      errorId: Date.now(),
    });
  }
});

// Express route to generate error handling code using Claude
app.post("/api/generate-error-handler", async (req, res) => {
  try {
    const { errorDescription } = req.body;

    if (!errorDescription) {
      return res.status(400).json({
        success: false,
        message: "errorDescription is required",
      });
    }

    const generatedCode = await generateErrorHandlingCode(errorDescription);

    res.json({
      success: true,
      generatedCode,
      timestamp: new Date().toISOString(),
    });
  } catch (error) {
    logProductionError(error, {
      route: "/api/generate-error-handler",
      method: "POST",
    });

    res.status(500).json({
      success: false,
      message: "Failed to generate error handling code",
    });
  }
});

// Global error handler middleware
app.use((err, req, res, next) => {
  logProductionError(err, {
    route: req.path,
    method: req.method,
    query: req.query,
    body: req.body,
  });

  res.status(500).json({
    success: false,
    message: "An unexpected error occurred",
    errorId: Date.now(),
  });
});

// Main function to demonstrate the functionality
async function main() {
  // Set environment for demonstration
  process.env.NODE_ENV = process.env.NODE_ENV || "production";

  console.log(`Starting server in ${process.env.NODE_ENV} environment`);

  // Demonstrate error logging
  const testError = new Error("Database connection failed");
  testError.code = "DB_CONNECTION_ERROR";

  logProductionError(testError, {
    database: "postgresql",
    host: "localhost",
    port: 5432,
  });

  // Generate error handling code using Claude
  try {
    console.log("\nGenerating error handling code using Claude...");
    const errorDescription =
      "Database connection timeout with retry logic and circuit breaker pattern";
    const generatedCode = await generateErrorHandlingCode(errorDescription);

    console.log("\nGenerated Error Handling Code:");
    console.log("=".repeat(50));
    console.log(generatedCode);
    console.log("=".repeat(50));
  } catch (error) {
    logProductionError(error, {
      operation: "generateErrorHandlingCode",
    });
  }

  // Start Express server
  const PORT = process.env.PORT || 3000;
  app.listen(PORT, () => {
    console.log(`\nExpress server running on port ${PORT}`);
    console.log(`Test error logging: GET http://localhost:${PORT}/api/test-error`);
    console.log(
      `Generate error handler: POST http://localhost:${PORT}/api/generate-error-handler`
    );
  });
}

main().catch((error) => {
  logProductionError(error, { operation: "main" });
  process.exit(1);
});
```
