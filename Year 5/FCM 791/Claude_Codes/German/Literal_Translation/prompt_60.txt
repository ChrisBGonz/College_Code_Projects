
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

// In-memory storage for votes and conversation history
const votes = new Map(); // ip -> option
const voteCounts = new Map(); // option -> count
const conversationHistory = [];

// Poll configuration
const pollQuestion = "What is your favorite programming language?";
const pollOptions = ["JavaScript", "Python", "TypeScript", "Rust", "Go"];

// Initialize vote counts
pollOptions.forEach((option) => voteCounts.set(option, 0));

// Simulate different IP addresses for testing
let simulatedIpCounter = 1;

function getSimulatedIp() {
  return `192.168.1.${simulatedIpCounter}`;
}

function formatPollResults() {
  const totalVotes = Array.from(voteCounts.values()).reduce(
    (sum, count) => sum + count,
    0
  );
  let results = `\nCurrent Poll Results for: "${pollQuestion}"\n`;
  results += "=".repeat(50) + "\n";

  pollOptions.forEach((option) => {
    const count = voteCounts.get(option) || 0;
    const percentage = totalVotes > 0 ? ((count / totalVotes) * 100).toFixed(1) : 0;
    const bar = "█".repeat(Math.floor(percentage / 5));
    results += `${option.padEnd(15)} | ${bar.padEnd(20)} | ${count} votes (${percentage}%)\n`;
  });

  results += "=".repeat(50) + "\n";
  results += `Total votes: ${totalVotes}\n`;
  return results;
}

async function processUserInput(userMessage, userIp) {
  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  // Create system prompt with current poll state
  const systemPrompt = `You are a helpful voting system assistant. You manage a poll about programming languages.

Current Poll: "${pollQuestion}"
Options: ${pollOptions.join(", ")}

Current Vote State:
${formatPollResults()}

User's IP Address: ${userIp}
Has this user voted: ${votes.has(userIp) ? `Yes, they voted for "${votes.get(userIp)}"` : "No"}

Your responsibilities:
1. Help users understand the poll and how to vote
2. Process vote requests - when a user wants to vote, extract their choice and confirm it
3. Prevent duplicate votes - if a user has already voted, inform them
4. Show current results when requested
5. Be friendly and conversational

When processing a vote:
- Identify which option the user wants to vote for
- Check if they've already voted
- If valid, confirm the vote with format: "VOTE_CAST:[option]"
- If invalid or duplicate, explain why

Always be helpful and guide users through the voting process.`;

  try {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system: systemPrompt,
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    // Add assistant response to conversation history
    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    // Check if a vote was cast
    const voteMatch = assistantMessage.match(/VOTE_CAST:([^\n\s]+)/);
    if (voteMatch) {
      const votedOption = voteMatch[1].trim();

      // Validate the option
      const validOption = pollOptions.find(
        (opt) => opt.toLowerCase() === votedOption.toLowerCase()
      );

      if (validOption && !votes.has(userIp)) {
        // Record the vote
        votes.set(userIp, validOption);
        const currentCount = voteCounts.get(validOption) || 0;
        voteCounts.set(validOption, currentCount + 1);

        console.log(
          `\n✓ Vote recorded: IP ${userIp} voted for "${validOption}"`
        );
      }
    }

    return assistantMessage;
  } catch (error) {
    console.error("Error calling Claude API:", error);
    throw error;
  }
}

async function main() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("\n🗳️  Welcome to the Programming Language Voting System!");
  console.log("=".repeat(55));
  console.log(`Poll Question: ${pollQuestion}`);
  console.log(`Options: ${pollOptions.join(", ")}`);
  console.log("=".repeat(55));
  console.log(
    "\nCommands:"
  );
  console.log("  - Type your message to interact with the voting assistant");
  console.log("  - Type 'new user' to simulate a different user (new IP)");
  console.log("  - Type 'results' to see current results");
  console.log("  - Type 'exit' to quit\n");

  let currentIp = getSimulatedIp();
  console.log(`Current simulated user IP: ${currentIp}\n`);

  const askQuestion = () => {
    rl.question(`You (${currentIp}): `, async (input) => {
      const userInput = input.trim();

      if (!userInput) {
        askQuestion();
        return;
      }

      if (userInput.toLowerCase() === "exit") {
        console.log("\nFinal Poll Results:");
        console.log(formatPollResults());
        console.log("Thank you for using the voting system!");
        rl.close();
        return;
      }

      if (userInput.toLowerCase() === "new user") {
        simulatedIpCounter++;
        currentIp = getSimulatedIp();
        // Clear conversation history for new user context
        conversationHistory.length = 0;
        console.log(`\n👤 Switched to new user with IP: ${currentIp}\n`);
        askQuestion();
        return;
      }

      if (userInput.toLowerCase() === "results") {
        console.log(formatPollResults());
        askQuestion();
        return;
      }

      try {
        console.log("\nAssistant: Processing...");
        const response = await processUserInput(userInput, currentIp);
        console.log(`\nAssistant: ${response}\n`);
      } catch (error) {
        console.error("Error:", error.message);
      }

      askQuestion();
    });
  };

  // Start the conversation
  try {
    console.log("Assistant: Initializing...");
    const initialResponse = await processUserInput(
      "Hello! I'd like to participate in the poll. Can you tell me about it?",
      currentIp
    );
    console.log(`\nAssistant: ${initialResponse}\n`);
  } catch (error) {
    console.error("Error initializing:", error.message);
  }

  askQuestion();
}

main().catch(console.error);
```
