
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as url from "url";

const client = new Anthropic();

// Simulated user database
const userDatabase: Record<
  string,
  {
    id: string;
    name: string;
    email: string;
    age: number;
    location: string;
    joinDate: string;
    preferences: Record<string, unknown>;
  }
> = {
  "1": {
    id: "1",
    name: "Alice Johnson",
    email: "alice@example.com",
    age: 28,
    location: "New York, USA",
    joinDate: "2022-01-15",
    preferences: {
      theme: "dark",
      language: "en",
      notifications: true,
    },
  },
  "2": {
    id: "2",
    name: "Bob Smith",
    email: "bob@example.com",
    age: 35,
    location: "London, UK",
    joinDate: "2021-06-20",
    preferences: {
      theme: "light",
      language: "en",
      notifications: false,
    },
  },
  "3": {
    id: "3",
    name: "Maria Garcia",
    email: "maria@example.com",
    age: 42,
    location: "Madrid, Spain",
    joinDate: "2023-03-10",
    preferences: {
      theme: "auto",
      language: "es",
      notifications: true,
    },
  },
};

async function generateProfileSummary(
  userData: (typeof userDatabase)[string]
): Promise<string> {
  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 500,
    messages: [
      {
        role: "user",
        content: `Generate a brief professional profile summary for this user based on their data:
Name: ${userData.name}
Email: ${userData.email}
Age: ${userData.age}
Location: ${userData.location}
Member since: ${userData.joinDate}
Preferences: ${JSON.stringify(userData.preferences)}

Please provide a 2-3 sentence professional summary that could be included in their profile export.`,
      },
    ],
  });

  let summary = "";

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      summary += chunk.delta.text;
      process.stdout.write(chunk.delta.text);
    }
  }

  console.log("\n");
  return summary;
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url || "", true);

  if (parsedUrl.pathname === "/download-profile" && req.method === "GET") {
    const userId = parsedUrl.query.user_id as string;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing user_id parameter",
          message: "Please provide a user_id in the query string",
        })
      );
      return;
    }

    const userData = userDatabase[userId];

    if (!userData) {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "User not found",
          message: `No user found with id: ${userId}`,
        })
      );
      return;
    }

    try {
      console.log(
        `Generating profile summary for user ${userId} using Claude AI...`
      );
      console.log("Streaming response: ");

      const aiSummary = await generateProfileSummary(userData);

      const profileExport = {
        exportDate: new Date().toISOString(),
        exportVersion: "1.0",
        userData: {
          ...userData,
          aiGeneratedSummary: aiSummary,
        },
        metadata: {
          generatedBy: "Profile Export Service",
          aiModel: "claude-opus-4-5",
          exportFormat: "JSON",
        },
      };

      const jsonContent = JSON.stringify(profileExport, null, 2);
      const filename = `profile_${userId}_${Date.now()}.json`;

      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="${filename}"`,
        "Content-Length": Buffer.byteLength(jsonContent),
        "Cache-Control": "no-cache",
      });

      res.end(jsonContent);

      console.log(`Profile for user ${userId} successfully exported`);
    } catch (error) {
      console.error("Error generating profile:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal server error",
          message: "Failed to generate profile export",
        })
      );
    }
  } else if (parsedUrl.pathname === "/" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Profile Download API",
        endpoints: {
          downloadProfile: {
            method: "GET",
            path: "/download-profile",
            queryParams: {
              user_id: "string (required) - The ID of the user",
            },
            description:
              "Downloads user profile data as a JSON file with AI-generated summary",
          },
        },
        availableUsers: Object.keys(userDatabase),
        example: "/download-profile?user_id=1",
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not found",
        message: "The requested endpoint does not exist",
      })
    );
  }
});

const PORT = 3000;
server.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET /                              - API information`);
  console.log(
    `  GET /download-profile?user_id=1    - Download user profile`
  );
  console.log(`\nAvailable user IDs: ${Object.keys(userDatabase).join(", ")}`);
  console.log(
    `\nExample: http://localhost:${PORT}/download-profile?user_id=1`
  );
});

export default server;
```
