import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user_001",
      username: "alice",
      email: "alice@example.com",
      password_hash: "old_hash_alice",
    },
    {
      id: "user_002",
      username: "bob",
      email: "bob@example.com",
      password_hash: "old_hash_bob",
    },
    {
      id: "user_003",
      username: "charlie",
      email: "charlie@example.com",
      password_hash: "old_hash_charlie",
    },
  ],
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database by their user_id. Returns user information if found.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character).",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description:
      "Hashes a password securely for storage in the database. Returns the hashed password.",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description:
      "Updates the user's password hash in the database. Returns success or failure status.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The new hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a password reset confirmation email to the user's registered email address.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send the confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool implementations
function validateUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      success: true,
      user: {
        id: user.id,
        username: user.username,
        email: user.email,
      },
    };
  }
  return { success: false, error: "User not found" };
}

function validatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /[0-9]/.test(password);
  const hasSpecialChar = /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(
    password
  );

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  return {
    isValid,
    requirements: {
      minLength: { met: minLength, description: "At least 8 characters" },
      hasUppercase: {
        met: hasUppercase,
        description: "Contains uppercase letter",
      },
      hasLowercase: {
        met: hasLowercase,
        description: "Contains lowercase letter",
      },
      hasNumber: { met: hasNumber, description: "Contains number" },
      hasSpecialChar: {
        met: hasSpecialChar,
        description: "Contains special character",
      },
    },
  };
}

function hashPassword(password) {
  // Simulated password hashing (in production, use bcrypt or similar)
  const hash = `hashed_${Buffer.from(password).toString("base64")}_${Date.now()}`;
  return { hash, algorithm: "simulated_bcrypt" };
}

function updatePasswordInDatabase(userId, passwordHash) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex === -1) {
    return { success: false, error: "User not found" };
  }

  database.users[userIndex].password_hash = passwordHash;
  database.users[userIndex].password_updated_at = new Date().toISOString();

  return {
    success: true,
    message: "Password updated successfully",
    updated_at: database.users[userIndex].password_updated_at,
  };
}

function sendConfirmationEmail(userId, email) {
  // Simulated email sending
  console.log(
    `[Email Service] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
    timestamp: new Date().toISOString(),
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_user":
      return validateUser(toolInput.user_id);
    case "validate_password_strength":
      return validatePasswordStrength(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "update_password_in_database":
      return updatePasswordInDatabase(toolInput.user_id, toolInput.password_hash);
    case "send_confirmation_email":
      return sendConfirmationEmail(toolInput.user_id, toolInput.email);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Starting password reset for user: ${userId}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user with ID "${userId}". The new password is "${newPassword}". 
      
      Please follow these steps:
      1. Validate that the user exists in the database
      2. Validate the password strength
      3. Hash the new password securely
      4. Update the password in the database
      5. Send a confirmation email to the user
      
      Provide a summary of each step and the final result.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("Initial Response:");
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool Used: ${toolUse.name}`);
      console.log("Tool Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Tool Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued Response Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(60));
  console.log("Final Response:");
  console.log("=".repeat(60));
  console.log(finalResponse);

  return finalResponse;
}

// Test the function
async function main() {
  try {
    // Test case 1: Valid user with strong password
    console.log("\n### Test Case 1: Valid user with strong password ###");
    await resetPassword("user_001", "SecurePass123!");

    // Test case 2: Valid user with weak password
    console.log("\n\n### Test Case 2: Valid user with weak password ###");
    await resetPassword("user_002", "weakpass");

    // Test case 3: Invalid user
    console.log("\n\n### Test Case 3: Invalid user ###");
    await resetPassword("user_999", "AnotherSecure456@");
  } catch (error) {
    console.error("Error:", error);
    throw error;
  }
}

main();
