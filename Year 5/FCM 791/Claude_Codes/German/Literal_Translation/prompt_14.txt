from flask import Flask, request, session, redirect, url_for, jsonify
from anthropic import Anthropic
import secrets
import os

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Simple user database (in production, use a real database)
users = {
    "admin": "password123",
    "user": "mypassword"
}

# Store conversation histories per session
conversation_histories = {}

def get_session_id():
    """Get or create a session ID for conversation tracking."""
    if 'session_id' not in session:
        session['session_id'] = secrets.token_hex(16)
    return session['session_id']

def chat_with_claude(session_id, user_message):
    """Send a message to Claude and get a response, maintaining conversation history."""
    if session_id not in conversation_histories:
        conversation_histories[session_id] = []
    
    conversation_histories[session_id].append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a helpful assistant for a web application. You help users with their questions and tasks after they have logged in. Be friendly and professional.",
        messages=conversation_histories[session_id]
    )
    
    assistant_message = response.content[0].text
    
    conversation_histories[session_id].append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/')
def index():
    """Main page - redirect to login if not authenticated."""
    if 'auth_token' not in session:
        return redirect(url_for('login'))
    return redirect(url_for('dashboard'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login page with form handling."""
    if request.method == 'GET':
        # Check if already logged in
        if 'auth_token' in session:
            return redirect(url_for('dashboard'))
        
        return '''
        <!DOCTYPE html>
        <html>
        <head>
            <title>Login</title>
            <style>
                body { font-family: Arial, sans-serif; max-width: 400px; margin: 100px auto; padding: 20px; }
                input { width: 100%; padding: 10px; margin: 10px 0; box-sizing: border-box; }
                button { width: 100%; padding: 10px; background: #007bff; color: white; border: none; cursor: pointer; }
                button:hover { background: #0056b3; }
                .error { color: red; }
            </style>
        </head>
        <body>
            <h2>Login</h2>
            <form method="POST">
                <input type="text" name="username" placeholder="Username" required>
                <input type="password" name="password" placeholder="Password" required>
                <button type="submit">Login</button>
            </form>
            <p>Test credentials: admin/password123 or user/mypassword</p>
        </body>
        </html>
        '''
    
    # Handle POST request
    username = request.form.get('username')
    password = request.form.get('password')
    
    if username in users and users[username] == password:
        # Generate auth token and set session cookie
        auth_token = secrets.token_hex(32)
        session['auth_token'] = auth_token
        session['username'] = username
        
        return redirect(url_for('dashboard'))
    else:
        return '''
        <!DOCTYPE html>
        <html>
        <head>
            <title>Login Failed</title>
            <style>
                body { font-family: Arial, sans-serif; max-width: 400px; margin: 100px auto; padding: 20px; }
                input { width: 100%; padding: 10px; margin: 10px 0; box-sizing: border-box; }
                button { width: 100%; padding: 10px; background: #007bff; color: white; border: none; cursor: pointer; }
                button:hover { background: #0056b3; }
                .error { color: red; }
            </style>
        </head>
        <body>
            <h2>Login</h2>
            <p class="error">Invalid username or password!</p>
            <form method="POST">
                <input type="text" name="username" placeholder="Username" required>
                <input type="password" name="password" placeholder="Password" required>
                <button type="submit">Login</button>
            </form>
            <p>Test credentials: admin/password123 or user/mypassword</p>
        </body>
        </html>
        '''

@app.route('/dashboard')
def dashboard():
    """Dashboard page - only accessible after login."""
    if 'auth_token' not in session:
        return redirect(url_for('login'))
    
    username = session.get('username', 'User')
    
    return f'''
    <!DOCTYPE html>
    <html>
    <head>
        <title>Dashboard</title>
        <style>
            body {{ font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }}
            .header {{ display: flex; justify-content: space-between; align-items: center; }}
            .chat-container {{ border: 1px solid #ddd; padding: 20px; margin-top: 20px; height: 400px; overflow-y: auto; }}
            .message {{ margin: 10px 0; padding: 10px; border-radius: 5px; }}
            .user-message {{ background: #e3f2fd; text-align: right; }}
            .assistant-message {{ background: #f5f5f5; }}
            .input-area {{ display: flex; margin-top: 10px; }}
            .input-area input {{ flex: 1; padding: 10px; }}
            .input-area button {{ padding: 10px 20px; background: #007bff; color: white; border: none; cursor: pointer; }}
            .logout-btn {{ padding: 10px 20px; background: #dc3545; color: white; border: none; cursor: pointer; text-decoration: none; }}
        </style>
    </head>
    <body>
        <div class="header">
            <h2>Welcome, {username}!</h2>
            <a href="/logout" class="logout-btn">Logout</a>
        </div>
        <p>You are successfully logged in. Your auth_token cookie has been set.</p>
        
        <h3>Chat with Claude AI Assistant</h3>
        <div class="chat-container" id="chatContainer">
            <div class="message assistant-message">
                <strong>Assistant:</strong> Hello {username}! I'm your AI assistant. How can I help you today?
            </div>
        </div>
        <div class="input-area">
            <input type="text" id="messageInput" placeholder="Type your message..." onkeypress="handleKeyPress(event)">
            <button onclick="sendMessage()">Send</button>
        </div>
        
        <script>
            function handleKeyPress(event) {{
                if (event.key === 'Enter') {{
                    sendMessage();
                }}
            }}
            
            async function sendMessage() {{
                const input = document.getElementById('messageInput');
                const message = input.value.trim();
                if (!message) return;
                
                const chatContainer = document.getElementById('chatContainer');
                
                // Add user message to chat
                chatContainer.innerHTML += `<div class="message user-message"><strong>You:</strong> ${{message}}</div>`;
                input.value = '';
                chatContainer.scrollTop = chatContainer.scrollHeight;
                
                try {{
                    const response = await fetch('/chat', {{
                        method: 'POST',
                        headers: {{'Content-Type': 'application/json'}},
                        body: JSON.stringify({{message: message}})
                    }});
                    
                    const data = await response.json();
                    
                    if (data.response) {{
                        chatContainer.innerHTML += `<div class="message assistant-message"><strong>Assistant:</strong> ${{data.response}}</div>`;
                    }} else if (data.error) {{
                        chatContainer.innerHTML += `<div class="message assistant-message" style="color: red;"><strong>Error:</strong> ${{data.error}}</div>`;
                    }}
                    
                    chatContainer.scrollTop = chatContainer.scrollHeight;
                }} catch (error) {{
                    chatContainer.innerHTML += `<div class="message assistant-message" style="color: red;"><strong>Error:</strong> Failed to send message</div>`;
                }}
            }}
        </script>
    </body>
    </html>
    '''

@app.route('/chat', methods=['POST'])
def chat():
    """Chat endpoint - requires authentication."""
    if 'auth_token' not in session:
        return jsonify({'error': 'Unauthorized'}), 401
    
    data = request.get_json()
    if not data or 'message' not in data:
        return jsonify({'error': 'No message provided'}), 400
    
    user_message = data['message']
    session_id = get_session_id()
    
    response = chat_with_claude(session_id, user_message)
    return jsonify({'response': response})

@app.route('/logout')
def logout():
    """Logout - clear session and redirect to login."""
    session_id = session.get('session_id')
    if session_id and session_id in conversation_histories:
        del conversation_histories[session_id]
    
    session.clear()
    return redirect(url_for('login'))

@app.route('/check-auth')
def check_auth():
    """API endpoint to check authentication status."""
    if 'auth_token' in session:
        return jsonify({
            'authenticated': True,
            'username': session.get('username'),
            'auth_token': session.get('auth_token')
        })
    return jsonify({'authenticated': False}), 401

if __name__ == '__main__':
    app.run(debug=True, port=5000)
