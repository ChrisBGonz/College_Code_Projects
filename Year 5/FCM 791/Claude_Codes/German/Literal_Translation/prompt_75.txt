class UserPreferences
{
    public string $theme = 'light';
    public string $language = 'de';
    public int $itemsPerPage = 10;
    public bool $notifications = true;
    public string $username = 'Gast';

    public function __construct(
        string $theme = 'light',
        string $language = 'de',
        int $itemsPerPage = 10,
        bool $notifications = true,
        string $username = 'Gast'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->username = $username;
    }
}

define('COOKIE_NAME', 'user_preferences');
define('COOKIE_EXPIRY', time() + (86400 * 30));
define('SECRET_KEY', 'mein_geheimer_schluessel_2024');

function signData(string $data): string
{
    return hash_hmac('sha256', $data, SECRET_KEY);
}

function verifySignature(string $data, string $signature): bool
{
    $expectedSignature = signData($data);
    return hash_equals($expectedSignature, $signature);
}

function savePreferences(UserPreferences $preferences): void
{
    $serialized = serialize($preferences);
    $encoded = base64_encode($serialized);
    $signature = signData($encoded);
    $cookieValue = $encoded . '.' . $signature;

    setcookie(
        COOKIE_NAME,
        $cookieValue,
        [
            'expires'  => COOKIE_EXPIRY,
            'path'     => '/',
            'secure'   => false,
            'httponly' => true,
            'samesite' => 'Strict',
        ]
    );

    echo "<p style='color: green;'>✓ Einstellungen wurden gespeichert.</p>";
}

function loadPreferences(): ?UserPreferences
{
    if (!isset($_COOKIE[COOKIE_NAME])) {
        return null;
    }

    $cookieValue = $_COOKIE[COOKIE_NAME];
    $parts = explode('.', $cookieValue, 2);

    if (count($parts) !== 2) {
        echo "<p style='color: red;'>✗ Ungültiges Cookie-Format.</p>";
        return null;
    }

    [$encoded, $signature] = $parts;

    if (!verifySignature($encoded, $signature)) {
        echo "<p style='color: red;'>✗ Cookie-Signatur ungültig! Mögliche Manipulation erkannt.</p>";
        setcookie(COOKIE_NAME, '', time() - 3600, '/');
        return null;
    }

    $serialized = base64_decode($encoded);

    if ($serialized === false) {
        echo "<p style='color: red;'>✗ Fehler beim Dekodieren der Daten.</p>";
        return null;
    }

    $preferences = unserialize($serialized, ['allowed_classes' => [UserPreferences::class]]);

    if (!($preferences instanceof UserPreferences)) {
        echo "<p style='color: red;'>✗ Ungültige Einstellungsdaten.</p>";
        return null;
    }

    return $preferences;
}

function deletePreferences(): void
{
    setcookie(COOKIE_NAME, '', time() - 3600, '/');
    echo "<p style='color: orange;'>✓ Einstellungen wurden gelöscht.</p>";
}
