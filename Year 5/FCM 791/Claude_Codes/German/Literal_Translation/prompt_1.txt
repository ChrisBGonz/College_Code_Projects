def get_invoice_from_database(invoice_id: str) -> dict | None:
    """Simulate database lookup for invoice."""
    return MOCK_DATABASE.get(invoice_id)

def process_with_claude(invoice_id: str, invoice_data: dict | None) -> dict:
    """Use Claude to process and format invoice data with streaming."""
    
    if invoice_data is None:
        prompt = f"""The invoice with ID '{invoice_id}' was not found in the database.
        Please provide a helpful error response in JSON format with the following structure:
        {{
            "success": false,
            "error": "Invoice not found",
            "message": "A helpful message explaining the situation",
            "invoice_id": "{invoice_id}"
        }}"""
    else:
        prompt = f"""I have retrieved invoice data from the database. Please analyze and format this invoice data 
        into a comprehensive response. Here is the raw invoice data:
        
        {json.dumps(invoice_data, indent=2)}
        
        Please provide a formatted response in JSON format with the following structure:
        {{
            "success": true,
            "invoice": {{
                "id": "invoice id",
                "customer": {{
                    "name": "customer name",
                    "email": "customer email"
                }},
                "dates": {{
                    "issued": "issue date",
                    "due": "due date"
                }},
                "items": [list of items with description, quantity, unit_price, total],
                "financial_summary": {{
                    "subtotal": subtotal amount,
                    "tax_rate": tax rate as percentage,
                    "tax_amount": tax amount,
                    "total": total amount,
                    "currency": "currency code"
                }},
                "status": "payment status",
                "status_description": "A brief description of what the status means"
            }},
            "summary": "A brief human-readable summary of the invoice"
        }}"""
    
    # Use streaming to get response from Claude
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            full_response += text
    
    # Parse the JSON response from Claude
    # Extract JSON from the response (Claude might include some text around it)
    try:
        # Try to find JSON in the response
        start_idx = full_response.find('{')
        end_idx = full_response.rfind('}') + 1
        if start_idx != -1 and end_idx > start_idx:
            json_str = full_response[start_idx:end_idx]
            return json.loads(json_str)
        else:
            return {"success": False, "error": "Failed to parse response", "raw_response": full_response}
    except json.JSONDecodeError:
        return {"success": False, "error": "Invalid JSON response", "raw_response": full_response}

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Route to retrieve invoice details by invoice_id.
    
    Args:
        invoice_id: The unique identifier for the invoice (e.g., INV-001)
    
    Returns:
        JSON response with invoice details or error message
    """
    # Validate invoice_id format
    if not invoice_id or len(invoice_id) > 50:
        return jsonify({
            "success": False,
            "error": "Invalid invoice ID",
            "message": "Invoice ID must be between 1 and 50 characters"
        }), 400
    
    # Retrieve invoice from database
    invoice_data = get_invoice_from_database(invoice_id)
    
    # Process with Claude using streaming
    result = process_with_claude(invoice_id, invoice_data)
    
    # Return appropriate HTTP status code
    if result.get("success", False):
        return jsonify(result), 200
    else:
        return jsonify(result), 404

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """
    Route to list all available invoice IDs.
    
    Returns:
        JSON response with list of available invoice IDs
    """
    invoice_ids = list(MOCK_DATABASE.keys())
    return jsonify({
        "success": True,
        "available_invoices": invoice_ids,
        "count": len(invoice_ids)
    }), 200
