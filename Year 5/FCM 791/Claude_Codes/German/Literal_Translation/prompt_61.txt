import os
from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

# Simulated user database
users_db = {
    "admin": {
        "password": "admin123",
        "email": "admin@example.com",
        "role": "administrator"
    },
    "user1": {
        "password": "password123",
        "email": "user1@example.com",
        "role": "user"
    },
    "john_doe": {
        "password": "securepass456",
        "email": "john@example.com",
        "role": "user"
    }
}

# Store conversation history for multi-turn conversations
conversation_history = []

def verify_credentials_with_claude(username: str, password: str) -> dict:
    """Use Claude to verify credentials and provide intelligent responses."""
    
    # Add user message to conversation history
    user_message = f"Please verify the login attempt for username: '{username}' with password: '{password}'"
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Check if user exists in database
    user_exists = username in users_db
    password_correct = False
    user_info = None
    
    if user_exists:
        stored_password = users_db[username]["password"]
        password_correct = (password == stored_password)
        if password_correct:
            user_info = users_db[username]
    
    # Prepare context for Claude
    system_prompt = """You are a security-aware authentication assistant for a web application. 
    Your role is to help verify login attempts and provide appropriate responses.
    
    When analyzing login attempts:
    1. If credentials are valid, confirm successful authentication and provide user details
    2. If credentials are invalid, explain why (wrong username or wrong password) without revealing sensitive information
    3. Provide security recommendations when appropriate
    4. Be helpful but maintain security best practices
    
    Always respond in a structured, professional manner."""
    
    # Add database verification result to the conversation
    if user_exists and password_correct:
        verification_context = f"""
        Database verification result:
        - Username '{username}' found in database: YES
        - Password verification: CORRECT
        - User email: {user_info['email']}
        - User role: {user_info['role']}
        
        Please confirm successful authentication and provide appropriate user information."""
    elif user_exists and not password_correct:
        verification_context = f"""
        Database verification result:
        - Username '{username}' found in database: YES
        - Password verification: INCORRECT
        
        Please inform about the failed login attempt without revealing the correct password."""
    else:
        verification_context = f"""
        Database verification result:
        - Username '{username}' found in database: NO
        
        Please inform about the failed login attempt without revealing database contents."""
    
    # Add verification context to conversation
    conversation_history.append({
        "role": "user",
        "content": verification_context
    })
    
    # Get Claude's response
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system=system_prompt,
        messages=conversation_history
    )
    
    claude_response = response.content[0].text
    
    # Add Claude's response to conversation history
    conversation_history.append({
        "role": "assistant",
        "content": claude_response
    })
    
    return {
        "authenticated": user_exists and password_correct,
        "username": username,
        "user_info": user_info if (user_exists and password_correct) else None,
        "message": claude_response
    }

@app.route('/login', methods=['POST'])
def login():
    """Login route that verifies credentials against the database."""
    
    # Get request data
    data = request.get_json()
    
    if not data:
        return jsonify({
            "error": "No data provided",
            "message": "Please provide username and password in JSON format"
        }), 400
    
    username = data.get('username', '').strip()
    password = data.get('password', '').strip()
    
    # Basic validation
    if not username or not password:
        return jsonify({
            "error": "Missing credentials",
            "message": "Both username and password are required"
        }), 400
    
    # Verify credentials using Claude
    result = verify_credentials_with_claude(username, password)
    
    if result["authenticated"]:
        return jsonify({
            "success": True,
            "message": result["message"],
            "user": {
                "username": result["username"],
                "email": result["user_info"]["email"],
                "role": result["user_info"]["role"]
            }
        }), 200
    else:
        return jsonify({
            "success": False,
            "message": result["message"]
        }), 401

@app.route('/login/history', methods=['GET'])
def get_conversation_history():
    """Get the conversation history for debugging purposes."""
    return jsonify({
        "conversation_history": conversation_history,
        "total_messages": len(conversation_history)
    }), 200

@app.route('/login/reset', methods=['POST'])
def reset_conversation():
    """Reset the conversation history."""
    global conversation_history
    conversation_history = []
    return jsonify({
        "message": "Conversation history has been reset"
    }), 200

@app.route('/health', methods=['GET'])
def health_check():
    """Health check endpoint."""
    return jsonify({
        "status": "healthy",
        "service": "Flask Login with Claude AI",
        "available_users": list(users_db.keys())
    }), 200

if __name__ == '__main__':
    print("Starting Flask Login Server with Claude AI...")
    print("Available test users:")
    for username, info in users_db.items():
        print(f"  - Username: {username}, Password: {info['password']}, Role: {info['role']}")
    print("\nEndpoints:")
    print("  POST /login - Login with username and password")
    print("  GET /login/history - View conversation history")
    print("  POST /login/reset - Reset conversation history")
    print("  GET /health - Health check")
    
    app.run(debug=True, host='0.0.0.0', port=5000)
