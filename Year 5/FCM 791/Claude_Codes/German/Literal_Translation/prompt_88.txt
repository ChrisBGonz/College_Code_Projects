
using System;
using System.Collections.Generic;
using Microsoft.Extensions.Logging;

namespace MfaSettingsLogger
{
    public enum MfaAction
    {
        Enabled,
        Disabled,
        MethodAdded,
        MethodRemoved,
        MethodUpdated,
        BackupCodesRegenerated,
        TrustedDeviceAdded,
        TrustedDeviceRemoved
    }

    public enum MfaMethod
    {
        Authenticator,
        SMS,
        Email,
        HardwareToken,
        BackupCodes
    }

    public class MfaSettingsChangeEvent
    {
        public string UserId { get; set; }
        public string Username { get; set; }
        public MfaAction Action { get; set; }
        public MfaMethod? AffectedMethod { get; set; }
        public DateTime Timestamp { get; set; }
        public string IpAddress { get; set; }
        public string UserAgent { get; set; }
        public string SessionId { get; set; }
        public Dictionary<string, string> AdditionalData { get; set; }
    }

    public interface IMfaAuditLogger
    {
        void LogMfaSettingsChange(MfaSettingsChangeEvent changeEvent);
        void LogMfaEnabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId);
        void LogMfaDisabled(string userId, string username, string ipAddress, string sessionId);
        void LogMfaMethodAdded(string userId, string username, MfaMethod method, string ipAddress, string sessionId);
        void LogMfaMethodRemoved(string userId, string username, MfaMethod method, string ipAddress, string sessionId);
    }

    public class MfaAuditLogger : IMfaAuditLogger
    {
        private readonly ILogger<MfaAuditLogger> _logger;

        public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
        {
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
        }

        public void LogMfaSettingsChange(MfaSettingsChangeEvent changeEvent)
        {
            if (changeEvent == null)
                throw new ArgumentNullException(nameof(changeEvent));

            var logMessage = BuildLogMessage(changeEvent);

            switch (changeEvent.Action)
            {
                case MfaAction.Disabled:
                    _logger.LogWarning(logMessage);
                    break;
                case MfaAction.TrustedDeviceAdded:
                case MfaAction.TrustedDeviceRemoved:
                    _logger.LogInformation(logMessage);
                    break;
                default:
                    _logger.LogInformation(logMessage);
                    break;
            }

            // Structured logging with all relevant fields
            _logger.LogInformation(
                "MFA Settings Change | UserId: {UserId} | Username: {Username} | Action: {Action} | " +
                "Method: {Method} | Timestamp: {Timestamp} | IP: {IpAddress} | " +
                "SessionId: {SessionId} | UserAgent: {UserAgent}",
                changeEvent.UserId,
                changeEvent.Username,
                changeEvent.Action,
                changeEvent.AffectedMethod?.ToString() ?? "N/A",
                changeEvent.Timestamp,
                changeEvent.IpAddress,
                changeEvent.SessionId,
                changeEvent.UserAgent
            );
        }

        public void LogMfaEnabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            var changeEvent = CreateChangeEvent(userId, username, MfaAction.Enabled, method, ipAddress, sessionId);
            LogMfaSettingsChange(changeEvent);
        }

        public void LogMfaDisabled(string userId, string username, string ipAddress, string sessionId)
        {
            var changeEvent = CreateChangeEvent(userId, username, MfaAction.Disabled, null, ipAddress, sessionId);
            LogMfaSettingsChange(changeEvent);
        }

        public void LogMfaMethodAdded(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            var changeEvent = CreateChangeEvent(userId, username, MfaAction.MethodAdded, method, ipAddress, sessionId);
            LogMfaSettingsChange(changeEvent);
        }

        public void LogMfaMethodRemoved(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            var changeEvent = CreateChangeEvent(userId, username, MfaAction.MethodRemoved, method, ipAddress, sessionId);
            LogMfaSettingsChange(changeEvent);
        }

        private static MfaSettingsChangeEvent CreateChangeEvent(
            string userId,
            string username,
            MfaAction action,
            MfaMethod? method,
            string ipAddress,
            string sessionId)
        {
            return new MfaSettingsChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = action,
                AffectedMethod = method,
                Timestamp = DateTime.UtcNow,
                IpAddress = ipAddress,
                SessionId = sessionId
            };
        }

        private static string BuildLogMessage(MfaSettingsChangeEvent changeEvent)
        {
            var methodInfo = changeEvent.AffectedMethod.HasValue
                ? $" using method '{changeEvent.AffectedMethod.Value}'"
                : string.Empty;

            return $"[MFA AUDIT] User '{changeEvent.Username}' (ID: {changeEvent.UserId}) " +
                   $"performed action '{changeEvent.Action}'{methodInfo} " +
                   $"at {changeEvent.Timestamp:yyyy-MM-dd HH:mm:ss} UTC " +
                   $"from IP {changeEvent.IpAddress} " +
                   $"(Session: {changeEvent.SessionId})";
        }
    }

    public class MfaSettingsService
    {
        private readonly IMfaAuditLogger _auditLogger;
        private readonly ILogger<MfaSettingsService> _logger;

        public MfaSettingsService(IMfaAuditLogger auditLogger, ILogger<MfaSettingsService> logger)
        {
            _auditLogger = auditLogger ?? throw new ArgumentNullException(nameof(auditLogger));
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
        }

        public bool EnableMfa(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            try
            {
                _logger.LogDebug("Attempting to enable MFA for user {UserId} with method {Method}", userId, method);

                // Business logic to enable MFA would go here
                bool success = PerformEnableMfa(userId, method);

                if (success)
                {
                    _auditLogger.LogMfaEnabled(userId, username, method, ipAddress, sessionId);
                    _logger.LogInformation("MFA successfully enabled for user {UserId}", userId);
                }
                else
                {
                    _logger.LogWarning("Failed to enable MFA for user {UserId}", userId);
                }

                return success;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error enabling MFA for user {UserId}", userId);
                throw;
            }
        }

        public bool DisableMfa(string userId, string username, string ipAddress, string sessionId)
        {
            try
            {
                _logger.LogDebug("Attempting to disable MFA for user {UserId}", userId);

                // Business logic to disable MFA would go here
                bool success = PerformDisableMfa(userId);

                if (success)
                {
                    _auditLogger.LogMfaDisabled(userId, username, ipAddress, sessionId);
                    _logger.LogWarning("MFA disabled for user {UserId} - Security level reduced", userId);
                }
                else
                {
                    _logger.LogWarning("Failed to disable MFA for user {UserId}", userId);
                }

                return success;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error disabling MFA for user {UserId}", userId);
                throw;
            }
        }

        public bool AddMfaMethod(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            try
            {
                _logger.LogDebug("Attempting to add MFA method {Method} for user {UserId}", method, userId);

                bool success = PerformAddMfaMethod(userId, method);

                if (success)
                {
                    _auditLogger.LogMfaMethodAdded(userId, username, method, ipAddress, sessionId);
                    _logger.LogInformation("MFA method {Method} added for user {UserId}", method, userId);
                }

                return success;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error adding MFA method {Method} for user {UserId}", method, userId);
                throw;
            }
        }

        public bool RemoveMfaMethod(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            try
            {
                _logger.LogDebug("Attempting to remove MFA method {Method} for user {UserId}", method, userId);

                bool success = PerformRemoveMfaMethod(userId, method);

                if (success)
                {
                    _auditLogger.LogMfaMethodRemoved(userId, username, method, ipAddress, sessionId);
                    _logger.LogInformation("MFA method {Method} removed for user {UserId}", method, userId);
                }

                return success;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error removing MFA method {Method} for user {UserId}", method, userId);
                throw;
            }
        }

        // Simulated business logic methods
        private bool PerformEnableMfa(string userId, MfaMethod method) => true;
        private bool PerformDisableMfa(string userId) => true;
        private bool PerformAddMfaMethod(string userId, MfaMethod method) => true;
        private bool PerformRemoveMfaMethod(string userId, MfaMethod method) => true;
    }

    // Example usage / Demo
    class Program
    {
        static void Main(string[] args)
        {
            using var loggerFactory = LoggerFactory.Create(builder =>
            {
                builder
                    .AddConsole()
                    .SetMinimumLevel(LogLevel.Debug);
            });

            var auditLogger = new MfaAuditLogger(loggerFactory.CreateLogger<MfaAuditLogger>());
            var mfaService = new MfaSettingsService(auditLogger, loggerFactory.CreateLogger<MfaSettingsService>());

            string userId = "user-12345";
            string username = "john.doe@example.com";
            string ipAddress = "192.168.1.100";
            string sessionId = "sess-abc-xyz-789";

            Console.WriteLine("=== MFA Settings Change Logging Demo ===\n");

            // Enable MFA with Authenticator
            mfaService.EnableMfa(userId, username, MfaMethod.Authenticator, ipAddress, sessionId);

            Console.WriteLine();

            // Add SMS as backup method
            mfaService.AddMfaMethod(userId, username, MfaMethod.SMS, ipAddress, sessionId);

            Console.WriteLine();

            // Remove SMS method
            mfaService.RemoveMfaMethod(userId, username, MfaMethod.SMS, ipAddress, sessionId);

            Console.WriteLine();

            // Disable MFA entirely
            mfaService.DisableMfa(userId, username, ipAddress, sessionId);

            Console.WriteLine();

            // Direct audit logging with custom event
            var customEvent = new MfaSettingsChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.BackupCodesRegenerated,
                AffectedMethod = MfaMethod.BackupCodes,
                Timestamp = DateTime.UtcNow,
                IpAddress = ipAddress,
                SessionId = sessionId,
                UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
                AdditionalData = new Dictionary<string, string>
                {
                    { "CodesGenerated", "10" },
                    { "PreviousCodesInvalidated", "true" }
                }
            };

            auditLogger.LogMfaSettingsChange(customEvent);

            Console.WriteLine("\n=== Demo Complete ===");
        }
    }
}
