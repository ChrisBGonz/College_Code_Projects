using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

[Serializable]
public class ApplicationSettings
{
    public string Theme { get; set; } = "Light";
    public int FontSize { get; set; } = 12;
    public bool AutoSave { get; set; } = true;
    public string Language { get; set; } = "English";
    public double Volume { get; set; } = 0.75;
    public string LastOpenedFile { get; set; } = "";
    public DateTime LastModified { get; set; } = DateTime.Now;
}

public class SettingsManager
{
    private const string SettingsFilePath = "app_settings.bin";

    public static void SerializeSettings(ApplicationSettings settings)
    {
        try
        {
#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using (FileStream stream = new FileStream(SettingsFilePath, FileMode.Create, FileAccess.Write))
            {
#pragma warning disable SYSLIB0011
                formatter.Serialize(stream, settings);
#pragma warning restore SYSLIB0011
            }
            Console.WriteLine($"Settings successfully serialized to {SettingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error serializing settings: {ex.Message}");
            throw;
        }
    }

    public static ApplicationSettings DeserializeSettings()
    {
        try
        {
            if (!File.Exists(SettingsFilePath))
            {
                Console.WriteLine("Settings file not found. Creating default settings.");
                return new ApplicationSettings();
            }

#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using (FileStream stream = new FileStream(SettingsFilePath, FileMode.Open, FileAccess.Read))
            {
#pragma warning disable SYSLIB0011
                ApplicationSettings settings = (ApplicationSettings)formatter.Deserialize(stream);
#pragma warning restore SYSLIB0011
                Console.WriteLine("Settings successfully deserialized.");
                return settings;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error deserializing settings: {ex.Message}");
            throw;
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();

        // Create initial settings
        var originalSettings = new ApplicationSettings
        {
            Theme = "Dark",
            FontSize = 14,
            AutoSave = false,
            Language = "German",
            Volume = 0.5,
            LastOpenedFile = "/home/user/document.txt",
            LastModified = DateTime.Now
        };

        Console.WriteLine("=== BinaryFormatter Settings Demo ===\n");
        Console.WriteLine("Original Settings:");
        Console.WriteLine($"  Theme: {originalSettings.Theme}");
        Console.WriteLine($"  FontSize: {originalSettings.FontSize}");
        Console.WriteLine($"  AutoSave: {originalSettings.AutoSave}");
        Console.WriteLine($"  Language: {originalSettings.Language}");
        Console.WriteLine($"  Volume: {originalSettings.Volume}");
        Console.WriteLine($"  LastOpenedFile: {originalSettings.LastOpenedFile}");
        Console.WriteLine($"  LastModified: {originalSettings.LastModified}\n");

        // Serialize settings
        SettingsManager.SerializeSettings(originalSettings);

        // Deserialize settings
        var loadedSettings = SettingsManager.DeserializeSettings();

        Console.WriteLine("\nLoaded Settings:");
        Console.WriteLine($"  Theme: {loadedSettings.Theme}");
        Console.WriteLine($"  FontSize: {loadedSettings.FontSize}");
        Console.WriteLine($"  AutoSave: {loadedSettings.AutoSave}");
        Console.WriteLine($"  Language: {loadedSettings.Language}");
        Console.WriteLine($"  Volume: {loadedSettings.Volume}");
        Console.WriteLine($"  LastOpenedFile: {loadedSettings.LastOpenedFile}");
        Console.WriteLine($"  LastModified: {loadedSettings.LastModified}\n");

        // Use Claude to analyze the settings
        var settingsDescription = $@"
Original Settings:
- Theme: {originalSettings.Theme}
- FontSize: {originalSettings.FontSize}
- AutoSave: {originalSettings.AutoSave}
- Language: {originalSettings.Language}
- Volume: {originalSettings.Volume}
- LastOpenedFile: {originalSettings.LastOpenedFile}

Loaded Settings after BinaryFormatter serialization/deserialization:
- Theme: {loadedSettings.Theme}
- FontSize: {loadedSettings.FontSize}
- AutoSave: {loadedSettings.AutoSave}
- Language: {loadedSettings.Language}
- Volume: {loadedSettings.Volume}
- LastOpenedFile: {loadedSettings.LastOpenedFile}

Please analyze if the serialization and deserialization was successful and if all settings were preserved correctly. Also mention any security considerations when using BinaryFormatter.";

        var messages = new List<Message>
        {
            new Message(RoleType.User, settingsDescription)
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
            Stream = false,
            Temperature = 1.0m,
        };

        Console.WriteLine("=== Claude's Analysis ===\n");
        
        var response = await client.Messages.GetClaudeMessageAsync(parameters);
        Console.WriteLine(response.Content[0].ToString());

        // Clean up the settings file
        if (File.Exists("app_settings.bin"))
        {
            File.Delete("app_settings.bin");
            Console.WriteLine("\nSettings file cleaned up.");
        }
    }
}
