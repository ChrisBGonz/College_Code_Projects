
import anthropic
import hashlib
import os
import sys
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    
    return sha256_hash.hexdigest()


def verify_firmware_with_claude(file_path: str, expected_hash: str, calculated_hash: str) -> str:
    """Use Claude to analyze and explain the firmware verification result."""
    client = anthropic.Anthropic()
    
    file_name = Path(file_path).name
    file_size = os.path.getsize(file_path)
    hashes_match = calculated_hash.lower() == expected_hash.lower()
    
    prompt = f"""I need to verify the integrity of a firmware file. Please analyze the following verification results and provide a detailed security assessment:

Firmware File: {file_name}
File Size: {file_size} bytes
Expected SHA-256 Hash: {expected_hash}
Calculated SHA-256 Hash: {calculated_hash}
Hashes Match: {hashes_match}

Please provide:
1. A clear verdict on whether the firmware is authentic and unmodified
2. Security implications of the result
3. Recommended next steps based on the verification outcome
4. Any potential risks or concerns

Be specific and technical in your assessment."""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return message.content[0].text


def verify_firmware(file_path: str, expected_hash: str) -> dict:
    """
    Verify firmware file integrity using SHA-256 hash.
    
    Args:
        file_path: Path to the firmware file
        expected_hash: Expected SHA-256 hash for comparison
    
    Returns:
        Dictionary with verification results
    """
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"Firmware file not found: {file_path}")
    
    print(f"Calculating SHA-256 hash for: {file_path}")
    calculated_hash = calculate_sha256(file_path)
    
    hashes_match = calculated_hash.lower() == expected_hash.lower()
    
    print("\n" + "="*60)
    print("FIRMWARE INTEGRITY VERIFICATION REPORT")
    print("="*60)
    print(f"File: {file_path}")
    print(f"File Size: {os.path.getsize(file_path)} bytes")
    print(f"\nExpected Hash:   {expected_hash}")
    print(f"Calculated Hash: {calculated_hash}")
    print(f"\nVerification Status: {'✓ PASSED' if hashes_match else '✗ FAILED'}")
    print("="*60)
    
    print("\nRequesting AI security analysis from Claude...")
    ai_analysis = verify_firmware_with_claude(file_path, expected_hash, calculated_hash)
    
    print("\nCLAUDE'S SECURITY ANALYSIS:")
    print("-"*60)
    print(ai_analysis)
    print("-"*60)
    
    return {
        "file_path": file_path,
        "expected_hash": expected_hash,
        "calculated_hash": calculated_hash,
        "verification_passed": hashes_match,
        "ai_analysis": ai_analysis
    }


def create_test_firmware():
    """Create a test firmware file for demonstration."""
    test_file = "test_firmware.bin"
    test_content = b"FIRMWARE_HEADER_v2.1\x00\x01\x02\x03" + b"\xFF" * 1024 + b"FIRMWARE_END"
    
    with open(test_file, "wb") as f:
        f.write(test_content)
    
    correct_hash = hashlib.sha256(test_content).hexdigest()
    
    return test_file, correct_hash


def main():
    """Main function to demonstrate firmware verification."""
    
    if len(sys.argv) == 3:
        firmware_file = sys.argv[1]
        expected_hash = sys.argv[2]
        
        try:
            result = verify_firmware(firmware_file, expected_hash)
            sys.exit(0 if result["verification_passed"] else 1)
        except FileNotFoundError as e:
            print(f"Error: {e}")
            sys.exit(1)
    else:
        print("Firmware Integrity Verification Tool")
        print("=====================================")
        print("Usage: python solution.py <firmware_file> <expected_sha256_hash>")
        print("\nRunning demonstration with test firmware...\n")
        
        test_file, correct_hash = create_test_firmware()
        
        print("TEST 1: Verifying authentic firmware (should PASS)")
        print("-"*50)
        result1 = verify_firmware(test_file, correct_hash)
        
        print("\n\nTEST 2: Verifying tampered firmware (should FAIL)")
        print("-"*50)
        wrong_hash = "a" * 64
        result2 = verify_firmware(test_file, wrong_hash)
        
        if os.path.exists(test_file):
            os.remove(test_file)
        
        print("\n\nSUMMARY:")
        print(f"Test 1 (Authentic): {'PASSED' if result1['verification_passed'] else 'FAILED'}")
        print(f"Test 2 (Tampered): {'PASSED' if result2['verification_passed'] else 'FAILED (Expected)'}")


if __name__ == "__main__":
    main()
