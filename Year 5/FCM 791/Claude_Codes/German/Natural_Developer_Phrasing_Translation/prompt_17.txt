const { MongoClient } = require('mongodb');

const CONNECTION_STRING = 'mongodb://localhost:27017';
const DATABASE_NAME = 'local_dev_db';

async function connectToDatabase() {
  const client = new MongoClient(CONNECTION_STRING, {
    useNewUrlParser: true,
    useUnifiedTopology: true,
    serverSelectionTimeoutMS: 5000,
    connectTimeoutMS: 10000,
  });

  try {
    await client.connect();
    console.log('Erfolgreich mit MongoDB verbunden');
    console.log(`Verbindungs-String: ${CONNECTION_STRING}`);
    console.log(`Datenbank: ${DATABASE_NAME}`);

    const db = client.db(DATABASE_NAME);

    const collections = await db.listCollections().toArray();
    console.log('\nVorhandene Collections:');
    if (collections.length === 0) {
      console.log('  Keine Collections vorhanden');
    } else {
      collections.forEach((col) => console.log(`  - ${col.name}`));
    }

    const adminDb = client.db('admin');
    const serverInfo = await adminDb.command({ serverStatus: 1 });
    console.log(`\nMongoDB Version: ${serverInfo.version}`);
    console.log(`Uptime: ${serverInfo.uptime} Sekunden`);

    return { client, db };
  } catch (error) {
    console.error('Fehler bei der Verbindung zu MongoDB:', error.message);
    await client.close();
    throw error;
  }
}

async function demonstrateBasicOperations(db) {
  const collectionName = 'test_collection';
  const collection = db.collection(collectionName);

  console.log('\n--- Grundlegende Datenbankoperationen ---');

  const testDocument = {
    name: 'Test Dokument',
    erstellt_am: new Date(),
    wert: 42,
    tags: ['test', 'entwicklung', 'lokal'],
  };

  const insertResult = await collection.insertOne(testDocument);
  console.log(`Dokument eingefügt mit ID: ${insertResult.insertedId}`);

  const foundDocument = await collection.findOne({ _id: insertResult.insertedId });
  console.log('Gefundenes Dokument:', foundDocument);

  const updateResult = await collection.updateOne(
    { _id: insertResult.insertedId },
    { $set: { wert: 100, aktualisiert_am: new Date() } }
  );
  console.log(`Dokumente aktualisiert: ${updateResult.modifiedCount}`);

  const deleteResult = await collection.deleteOne({ _id: insertResult.insertedId });
  console.log(`Dokumente gelöscht: ${deleteResult.deletedCount}`);

  await db.dropCollection(collectionName).catch(() => {});
  console.log(`Collection '${collectionName}' bereinigt`);
}

async function main() {
  let client;

  try {
    const { client: mongoClient, db } = await connectToDatabase();
    client = mongoClient;

    await demonstrateBasicOperations(db);

    console.log('\nVerbindungstest erfolgreich abgeschlossen');
  } catch (error) {
    console.error('Hauptfehler:', error.message);
    process.exit(1);
  } finally {
    if (client) {
      await client.close();
      console.log('MongoDB-Verbindung geschlossen');
    }
  }
}

main();
