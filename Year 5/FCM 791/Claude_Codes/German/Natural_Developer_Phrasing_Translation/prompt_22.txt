#!/usr/bin/env bash

set -euo pipefail
IFS=$'\n\t'

# ─────────────────────────────────────────────
# Configuration
# ─────────────────────────────────────────────
TOOL_NAME="${TOOL_NAME:-trivy}"
GITHUB_REPO="${GITHUB_REPO:-aquasecurity/trivy}"
INSTALL_DIR="${INSTALL_DIR:-/usr/local/bin}"
TMP_DIR="$(mktemp -d)"
LOG_FILE="/var/log/${TOOL_NAME}_install.log"
CHECKSUM_ALGO="${CHECKSUM_ALGO:-sha256}"
MIN_BASH_VERSION=4

# ─────────────────────────────────────────────
# Colors / Logging
# ─────────────────────────────────────────────
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
CYAN='\033[0;36m'
RESET='\033[0m'

log()     { local ts; ts="$(date '+%Y-%m-%d %H:%M:%S')"; echo -e "${CYAN}[INFO]${RESET}  ${ts} $*" | tee -a "${LOG_FILE}"; }
warn()    { local ts; ts="$(date '+%Y-%m-%d %H:%M:%S')"; echo -e "${YELLOW}[WARN]${RESET}  ${ts} $*" | tee -a "${LOG_FILE}"; }
success() { local ts; ts="$(date '+%Y-%m-%d %H:%M:%S')"; echo -e "${GREEN}[OK]${RESET}    ${ts} $*" | tee -a "${LOG_FILE}"; }
error()   { local ts; ts="$(date '+%Y-%m-%d %H:%M:%S')"; echo -e "${RED}[ERROR]${RESET} ${ts} $*" | tee -a "${LOG_FILE}" >&2; }
die()     { error "$*"; cleanup; exit 1; }

# ─────────────────────────────────────────────
# Cleanup
# ─────────────────────────────────────────────
cleanup() {
    log "Cleaning up temporary directory: ${TMP_DIR}"
    rm -rf "${TMP_DIR}"
}
trap cleanup EXIT
trap 'die "Script interrupted."' INT TERM

# ─────────────────────────────────────────────
# Preflight checks
# ─────────────────────────────────────────────
preflight_checks() {
    log "Running preflight checks..."

    # Bash version
    if (( BASH_VERSINFO[0] < MIN_BASH_VERSION )); then
        die "Bash >= ${MIN_BASH_VERSION} required (found ${BASH_VERSION})."
    fi

    # Root / sudo
    if [[ $EUID -ne 0 ]]; then
        if ! command -v sudo &>/dev/null; then
            die "This script must be run as root or sudo must be available."
        fi
        SUDO="sudo"
        warn "Not running as root – will use sudo where needed."
    else
        SUDO=""
    fi

    # Required tools
    local required_tools=("curl" "tar" "grep" "awk" "sed" "uname" "install")
    for tool in "${required_tools[@]}"; do
        if ! command -v "${tool}" &>/dev/null; then
            die "Required tool not found: ${tool}"
        fi
    done

    # Checksum tool
    if command -v "${CHECKSUM_ALGO}sum" &>/dev/null; then
        CHECKSUM_CMD="${CHECKSUM_ALGO}sum"
    elif command -v "openssl" &>/dev/null; then
        CHECKSUM_CMD="openssl dgst -${CHECKSUM_ALGO}"
        CHECKSUM_OPENSSL=true
    else
        die "No checksum tool found (${CHECKSUM_ALGO}sum or openssl)."
    fi

    # Install directory writable
    if [[ ! -d "${INSTALL_DIR}" ]]; then
        log "Install directory ${INSTALL_DIR} does not exist – creating it."
        ${SUDO} mkdir -p "${INSTALL_DIR}" || die "Cannot create ${INSTALL_DIR}."
    fi

    success "Preflight checks passed."
}

# ─────────────────────────────────────────────
# Detect OS / Architecture
# ─────────────────────────────────────────────
detect_platform() {
    log "Detecting platform..."

    OS="$(uname -s | tr '[:upper:]' '[:lower:]')"
    ARCH_RAW="$(uname -m)"

    case "${ARCH_RAW}" in
        x86_64|amd64)   ARCH="amd64" ;;
        aarch64|arm64)  ARCH="arm64" ;;
        armv7l|armv7)   ARCH="arm"   ;;
        i386|i686)      ARCH="386"   ;;
        *)              die "Unsupported architecture: ${ARCH_RAW}" ;;
    esac

    case "${OS}" in
        linux)   OS_LABEL="Linux"   ;;
        darwin)  OS_LABEL="macOS"   ;;
        *)       die "Unsupported OS: ${OS}" ;;
    esac

    log "Platform: ${OS_LABEL} / ${ARCH}"
}

# ─────────────────────────────────────────────
# Fetch latest version from GitHub API
# ─────────────────────────────────────────────
fetch_latest_version() {
    log "Fetching latest release version for ${GITHUB_REPO}..."

    local api_url="https://api.github.com/repos/${GITHUB_REPO}/releases/latest"
    local response

    response="$(curl --silent --fail --location \
        --retry 3 --retry-delay 2 --retry-max-time 30 \
        --connect-timeout 10 \
        -H "Accept: application/vnd.github+json" \
        "${api_url}")" || die "Failed to query GitHub API: ${api_url}"

    LATEST_VERSION="$(echo "${response}" | grep '"tag_name"' | head -1 | sed 's/.*"tag_name": *"\([^"]*\)".*/\1/')"

    if [[ -z "${LATEST_VERSION}" ]]; then
        die "Could not parse latest version from GitHub API response."
    fi

    # Strip leading 'v' for version number comparisons
    VERSION_NUMBER="${LATEST_VERSION#v}"

    success "Latest version: ${LATEST_VERSION}"
}

# ─────────────────────────────────────────────
# Check if already installed and up-to-date
# ─────────────────────────────────────────────
check_existing_installation() {
    log "Checking for existing installation of ${TOOL_NAME}..."

    if command -v "${TOOL_NAME}" &>/dev/null; then
        local installed_path
        installed_path="$(command -v "${TOOL_NAME}")"

        # Try common version flags
        local current_version=""
        for flag in "--version" "version" "-v" "-version"; do
            current_version="$("${installed_path}" ${flag} 2>/dev/null | grep -oE '[0-9]+\.[0-9]+\.[0-9]+' | head -1 || true)"
            [[ -n "${current_version}" ]] && break
        done

        if [[ -n "${current_version}" ]]; then
            log "Installed version: ${current_version}"
            if [[ "${current_version}" == "${VERSION_NUMBER}" ]]; then
                success "${TOOL_NAME} ${current_version} is already the latest version."
                if [[ "${FORCE_REINSTALL:-false}" != "true" ]]; then
                    log "Set FORCE_REINSTALL=true to reinstall anyway."
                    exit 0
                fi
                warn "FORCE_REINSTALL=true – proceeding with reinstall."
            else
                log "Upgrade available: ${current_version} → ${VERSION_NUMBER}"
            fi
        else
            warn "Could not determine installed version – proceeding with installation."
        fi
    else
        log "${TOOL_NAME} not found – fresh installation."
    fi
}

# ─────────────────────────────────────────────
# Build download URLs
# ─────────────────────────────────────────────
build_download_urls() {
    log "Building download URLs..."

    local base_url="https://github.com/${GITHUB_REPO}/releases/download/${LATEST_VERSION}"

    # Asset naming convention (Trivy-style, widely used):
    # trivy_0.50.0_Linux-64bit.tar.gz
    # Adjust ASSET_PATTERN if the tool uses a different naming scheme.
    local arch_label
    case "${ARCH}" in
        amd64) arch_label="64bit"  ;;
        arm64) arch_label="ARM64"  ;;
        arm)   arch_label="ARM"    ;;
        386)   arch_label="32bit"  ;;
    esac

    local os_label_asset
    case "${OS}" in
        linux)  os_label_asset="Linux"  ;;
        darwin) os_label_asset="macOS"  ;;
    esac

    ASSET_NAME="${TOOL_NAME}_${VERSION_NUMBER}_${os_label_asset}-${arch_label}.tar.gz"
    CHECKSUM_FILE_NAME="${TOOL_NAME}_${VERSION_NUMBER}_checksums.txt"

    DOWNLOAD_URL="${base_url}/${ASSET_NAME}"
    CHECKSUM_URL="${base_url}/${CHECKSUM_FILE_NAME}"

    # Allow override via environment
    DOWNLOAD_URL="${OVERRIDE_DOWNLOAD_URL:-${DOWNLOAD_URL}}"
    CHECKSUM_URL="${OVERRIDE_CHECKSUM_URL:-${CHECKSUM_URL}}"

    log "Asset URL:   ${DOWNLOAD_URL}"
    log "Checksum URL: ${CHECKSUM_URL}"
}

# ─────────────────────────────────────────────
# Download file with retry and progress
# ─────────────────────────────────────────────
download_file() {
    local url="$1"
    local dest="$2"
    local description="${3:-file}"

    log "Downloading ${description}..."

    local http_code
    http_code="$(curl \
        --location \
        --fail \
        --silent \
        --show-error \
        --retry 3 \
        --retry-delay 3 \
        --retry-max-time 60 \
        --connect-timeout 15 \
        --max-time 300 \
        --write-out "%{http_code}" \
        --output "${dest}" \
        "${url}" 2>&1)" || {
            error "Download failed for: ${url}"
            error "HTTP code: ${http_code}"
            die "Download error."
        }

    if [[ "${http_code}" != "200" ]]; then
        die "Unexpected HTTP status ${http_code} for ${url}"
    fi

    if [[ ! -s "${dest}" ]]; then
        die "Downloaded file is empty: ${dest}"
    fi

    success "Downloaded: $(basename "${dest}") ($(du -sh "${dest}" | awk '{print $1}'))"
}

# ─────────────────────────────────────────────
# Validate checksum
# ─────────────────────────────────────────────
validate_checksum() {
    log "Validating ${CHECKSUM_ALGO} checksum..."

    local archive="${TMP_DIR}/${ASSET_NAME}"
    local checksum_file="${TMP_DIR}/${CHECKSUM_FILE_NAME}"

    if [[ ! -f "${checksum_file}" ]]; then
        warn "Checksum file not found – skipping validation (not recommended)."
        return 0
    fi

    # Extract expected checksum for our asset
    local expected_checksum
    expected_checksum="$(grep -i "${ASSET_NAME}" "${checksum_file}" | awk '{print $1}' | head -1)"

    if [[ -z "${expected_checksum}" ]]; then
        warn "No checksum entry found for ${ASSET_NAME} in checksum file – skipping."
        return 0
    fi

    log "Expected checksum: ${expected_checksum}"

    # Compute actual checksum
    local actual_checksum
    if [[ "${CHECKSUM_OPENSSL:-false}" == "true" ]]; then
        actual_checksum="$(openssl dgst -"${CHECKSUM_ALGO}" "${archive}" | awk '{print $NF}')"
    else
        actual_checksum="$("${CHECKSUM_CMD}" "${archive}" | awk '{print $1}')"
    fi

    log "Actual checksum:   ${actual_checksum}"

    if [[ "${expected_checksum}" != "${actual_checksum}" ]]; then
        die "Checksum mismatch! Expected: ${expected_checksum} | Got: ${actual_checksum}"
    fi

    success "Checksum validation passed."
}

# ─────────────────────────────────────────────
# Extract archive
# ─────────────────────────────────────────────
extract_archive() {
    log "Extracting archive..."

    local archive="${TMP_DIR}/${ASSET_NAME}"
    local extract_dir="${TMP_DIR}/extracted"
    mkdir -p "${extract_dir}"

    # Validate it's a real tar.gz
    if ! file "${archive}" | grep -qiE 'gzip|tar'; then
        warn "File type check: $(file "${archive}")"
    fi

    tar -xzf "${archive}" -C "${extract_dir}" || die "Failed to extract archive."

    # Find the binary (search recursively, prefer exact name match)
    BINARY_PATH="$(find "${extract_dir}" -type f -name "${TOOL_NAME}" | head -1)"

    if [[ -z "${BINARY_PATH}" ]]; then
        # Fallback: look for any executable
        BINARY_PATH="$(find "${extract_dir}" -type f -executable | head -1)"
        warn "Exact binary '${TOOL_NAME}' not found – using: ${BINARY_PATH}"
    fi

    if [[ -z "${BINARY_PATH}" ]]; then
        die "No executable binary found in extracted archive."
    fi

    log "Binary found: ${BINARY_PATH}"
    success "Extraction complete."
}

# ─────────────────────────────────────────────
# Install binary
# ─────────────────────────────────────────────
install_binary() {
    log "Installing ${TOOL_NAME} to ${INSTALL_DIR}..."

    local target="${INSTALL_DIR}/${TOOL_NAME}"

    # Backup existing binary if present
    if [[ -f "${target}" ]]; then
        local backup="${target}.bak.$(date '+%Y%m%d%H%M%S')"
        log "Backing up existing binary to ${backup}"
        ${SUDO} cp "${target}" "${backup}" || warn "Could not create backup."
    fi

    # Install with proper permissions
    ${SUDO} install -m 0755 -o root -g root "${BINARY_PATH}" "${target}" \
        || die "Failed to install binary to ${target}."

    # Verify installation
    if ! command -v "${TOOL_NAME}" &>/dev/null; then
        warn "${TOOL_NAME} not found in PATH after install."
        warn "Ensure ${INSTALL_DIR} is in your PATH."
    fi

    success "${TOOL_NAME} installed to ${target}."
}

# ─────────────────────────────────────────────
# Post-install verification
# ─────────────────────────────────────────────
post_install_verification() {
    log "Running post-install verification..."

    local binary_path="${INSTALL_DIR}/${TOOL_NAME}"

    if [[ ! -x "${binary_path}" ]]; then
        die "Binary not executable after installation: ${binary_path}"
    fi

    # Try to get version
    local installed_version=""
    for flag in "--version" "version" "-v" "-version"; do
        installed_version="$("${binary_path}" ${flag} 2>/dev/null | grep -oE '[0-9]+\.[0-9]+\.[0-9]+' | head -1 || true)"
        [[ -n "${installed_version}" ]] && break
    done

    if [[ -n "${installed_version}" ]]; then
        if [[ "${installed_version}" == "${VERSION_NUMBER}" ]]; then
            success "Version verified: ${installed_version}"
        else
            warn "Version mismatch after install: expected ${VERSION_NUMBER}, got ${installed_version}"
        fi
    else
        warn "Could not verify version – binary may still work correctly."
    fi

    # File info
    log "Binary details:"
    ls -lh "${binary_path}" | tee -a "${LOG_FILE}"
    file "${binary_path}" | tee -a "${LOG_FILE}" || true

    success "Post-install verification complete."
}

# ─────────────────────────────────────────────
# Print summary
# ─────────────────────────────────────────────
print_summary() {
    echo ""
    echo -e "${GREEN}╔══════════════════════════════════════════════╗${RESET}"
    echo -e "${GREEN}║         Installation Summary                 ║${RESET}"
    echo -e "${GREEN}╠══════════════════════════════════════════════╣${RESET}"
    echo -e "${GREEN}║${RESET} Tool:     ${TOOL_NAME}"
    echo -e "${GREEN}║${RESET} Version:  ${LATEST_VERSION}"
    echo -e "${GREEN}║${RESET} Location: ${INSTALL_DIR}/${TOOL_NAME}"
    echo -e "${GREEN}║${RESET} OS/Arch:  ${OS_LABEL} / ${ARCH}"
    echo -e "${GREEN}║${RESET} Log:      ${LOG_FILE}"
    echo -e "${GREEN}╚══════════════════════════════════════════════╝${RESET}"
    echo ""
}

# ─────────────────────────────────────────────
# Usage
# ─────────────────────────────────────────────
usage() {
    cat <<EOF
Usage: $(basename "$0") [OPTIONS]

Unattended installer for security tools hosted on GitHub Releases.

Options:
  -h, --help            Show this help message
  -t, --tool NAME       Tool binary name          (default: trivy)
  -r, --repo OWNER/REPO GitHub repository         (default: aquasecurity/trivy)
  -d, --dir PATH        Installation directory    (default: /usr/local/bin)
  -f, --force           Force reinstall if already up-to-date
  -v, --version VER     Install specific version  (e.g. v0.50.0)

Environment variables (override defaults):
  TOOL_NAME             Same as --tool
  GITHUB_REPO           Same as --repo
  INSTALL_DIR           Same as --dir
  FORCE_REINSTALL       Set to 'true' to force reinstall
  OVERRIDE_DOWNLOAD_URL Override the asset download URL
  OVERRIDE_CHECKSUM_URL Override the checksum file URL
  CHECKSUM_ALGO         Checksum algorithm (default: sha256)

Examples:
  # Install latest Trivy
  sudo $(basename "$0")

  # Install specific version of Trivy
  sudo $(basename "$0") --version v0.49.1

  # Install a different tool
  sudo $(basename "$0") --tool grype --repo anchore/grype

  # Force reinstall
  sudo FORCE_REINSTALL=true $(basename "$0")
EOF
}

# ─────────────────────────────────────────────
# Parse arguments
# ─────────────────────────────────────────────
parse_args() {
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -h|--help)
                usage; exit 0 ;;
            -t|--tool)
                TOOL_NAME="${2:?'--tool requires a value'}"; shift 2 ;;
            -r|--repo)
                GITHUB_REPO="${2:?'--repo requires a value'}"; shift 2 ;;
            -d|--dir)
                INSTALL_DIR="${2:?'--dir requires a value'}"; shift 2 ;;
            -f|--force)
                FORCE_REINSTALL="true"; shift ;;
            -v|--version)
                PINNED_VERSION="${2:?'--version requires a value'}"; shift 2 ;;
            *)
                error "Unknown option: $1"; usage; exit 1 ;;
        esac
    done
}

# ─────────────────────────────────────────────
# Main
# ─────────────────────────────────────────────
main() {
    # Ensure log file is writable
    touch "${LOG_FILE}" 2>/dev/null || LOG_FILE="${TMP_DIR}/install.log"

    log "════════════════════════════════════════════"
    log " Security Tool Installer – $(date)"
    log "════════════════════════════════════════════"

    parse_args "$@"
    preflight_checks
    detect_platform

    # Version resolution
    if [[ -n "${PINNED_VERSION:-}" ]]; then
        LATEST_VERSION="${PINNED_VERSION}"
        VERSION_NUMBER="${LATEST_VERSION#v}"
        log "Using pinned version: ${LATEST_VERSION}"
    else
        fetch_latest_version
    fi

    check_existing_installation
    build_download_urls

    # Download phase
    download_file "${CHECKSUM_URL}" "${TMP_DIR}/${CHECKSUM_FILE_NAME}" "checksum file" || \
        warn "Checksum file download failed – will skip validation."

    download_file "${DOWNLOAD_URL}" "${TMP_DIR}/${ASSET_NAME}" "release archive"

    # Validate
    validate_checksum

    # Install
    extract_archive
    install_binary
    post_install_verification
    print_summary

    success "Installation of ${TOOL_NAME} ${LATEST_VERSION} completed successfully."
}

main "$@"
