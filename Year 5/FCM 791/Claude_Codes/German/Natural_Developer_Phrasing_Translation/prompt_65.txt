<?php
session_start();

define('DB_HOST', 'localhost');
define('DB_NAME', 'secret_question_db');
define('DB_USER', 'root');
define('DB_PASS', '');
define('DB_CHARSET', 'utf8mb4');

function getDB(): PDO {
    static $pdo = null;
    if ($pdo === null) {
        $dsn = 'mysql:host=' . DB_HOST . ';dbname=' . DB_NAME . ';charset=' . DB_CHARSET;
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        $pdo = new PDO($dsn, DB_USER, DB_PASS, $options);
    }
    return $pdo;
}

function initDB(): void {
    $pdo = getDB();
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS users (
            id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            username      VARCHAR(60)  NOT NULL UNIQUE,
            email         VARCHAR(120) NOT NULL UNIQUE,
            password_hash VARCHAR(255) NOT NULL,
            secret_question VARCHAR(255) NOT NULL,
            secret_answer_hash VARCHAR(255) NOT NULL,
            reset_token   VARCHAR(64)  DEFAULT NULL,
            reset_expires DATETIME     DEFAULT NULL,
            failed_recovery_attempts TINYINT UNSIGNED NOT NULL DEFAULT 0,
            recovery_locked_until DATETIME DEFAULT NULL,
            created_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");
}

function sanitize(string $input): string {
    return trim(htmlspecialchars($input, ENT_QUOTES, 'UTF-8'));
}

function redirect(string $url): void {
    header('Location: ' . $url);
    exit;
}

function setFlash(string $type, string $message): void {
    $_SESSION['flash'] = ['type' => $type, 'message' => $message];
}

function getFlash(): ?array {
    if (isset($_SESSION['flash'])) {
        $flash = $_SESSION['flash'];
        unset($_SESSION['flash']);
        return $flash;
    }
    return null;
}

function generateToken(int $length = 32): string {
    return bin2hex(random_bytes($length));
}

function isLoggedIn(): bool {
    return isset($_SESSION['user_id']);
}

function requireLogin(): void {
    if (!isLoggedIn()) {
        redirect('?action=login');
    }
}

function handleRegister(): void {
    global $secretQuestions;

    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        $username        = sanitize($_POST['username'] ?? '');
        $email           = sanitize($_POST['email'] ?? '');
        $password        = $_POST['password'] ?? '';
        $passwordConfirm = $_POST['password_confirm'] ?? '';
        $questionIndex   = (int)($_POST['secret_question_index'] ?? -1);
        $answer          = $_POST['secret_answer'] ?? '';

        $errors = [];

        if (strlen($username) < 3 || strlen($username) > 60) {
            $errors[] = 'Benutzername muss zwischen 3 und 60 Zeichen lang sein.';
        }
        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
            $errors[] = 'Ungültige E-Mail-Adresse.';
        }
        if (strlen($password) < 8) {
            $errors[] = 'Passwort muss mindestens 8 Zeichen lang sein.';
        }
        if ($password !== $passwordConfirm) {
            $errors[] = 'Passwörter stimmen nicht überein.';
        }
        if (!isset($secretQuestions[$questionIndex])) {
            $errors[] = 'Bitte wähle eine geheime Frage aus.';
        }
        if (strlen(trim($answer)) < 2) {
            $errors[] = 'Die Antwort auf die geheime Frage ist zu kurz.';
        }

        if (empty($errors)) {
            try {
                $pdo          = getDB();
                $passwordHash = password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
                $answerHash   = password_hash(strtolower(trim($answer)), PASSWORD_BCRYPT, ['cost' => 12]);
                $question     = $secretQuestions[$questionIndex];

                $stmt = $pdo->prepare("
                    INSERT INTO users (username, email, password_hash, secret_question, secret_answer_hash)
                    VALUES (:username, :email, :password_hash, :secret_question, :secret_answer_hash)
                ");
                $stmt->execute([
                    ':username'           => $username,
                    ':email'              => $email,
                    ':password_hash'      => $passwordHash,
                    ':secret_question'    => $question,
                    ':secret_answer_hash' => $answerHash,
                ]);

                setFlash('success', 'Registrierung erfolgreich! Bitte melde dich an.');
                redirect('?action=login');
            } catch (\PDOException $e) {
                if ($e->getCode() == 23000) {
                    $errors[] = 'Benutzername oder E-Mail bereits vergeben.';
                } else {
                    $errors[] = 'Datenbankfehler. Bitte versuche es später erneut.';
                }
            }
        }

        renderPage('Registrieren', function() use ($username, $email, $errors, $secretQuestions, $questionIndex) {
            renderRegisterForm($username, $email, $errors, $secretQuestions, $questionIndex);
        });
    } else {
        renderPage('Registrieren', function() use ($secretQuestions) {
            renderRegisterForm('', '', [], $secretQuestions, -1);
        });
    }
}

function handleLogin(): void {
    if (isLoggedIn()) {
        redirect('?action=dashboard');
    }

    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        verifyCsrf();

        $identifier = sanitize($_POST['identifier'] ?? '');
        $password   = $_POST['password'] ?? '';
        $errors     = [];

        if (empty($identifier) || empty($password)) {
            $errors[] = 'Bitte alle Felder ausfüllen.';
        }

        if (empty($errors)) {
            $pdo  = getDB();
            $stmt = $pdo->prepare("SELECT * FROM users WHERE username = :id OR email = :id LIMIT 1");
            $stmt->execute([':id' => $identifier]);
            $user = $stmt->fetch();

            if ($user && password_verify($password, $user['password_hash'])) {
                session_regenerate_id(true);
                $_SESSION['user_id']   = $user['id'];
                $_SESSION['username']  = $user['username'];
                setFlash('success', 'Erfolgreich angemeldet. Willkommen, ' . htmlspecialchars($user['username']) . '!');
                redirect('?action=dashboard');
            } else {
                $errors[] = 'Ungültige Anmeldedaten.';
            }
        }

        renderPage('Anmelden', function() use ($identifier, $errors) {
            renderLoginForm($identifier, $errors);
        });
    } else {
        renderPage('Anmelden', function() {
            renderLoginForm('', []);
        });
    }
}

function handleRecover(): void {
    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        verifyCsrf();

        $identifier = sanitize($_POST['identifier'] ?? '');
        $errors     = [];

        if (empty($identifier)) {
            $errors[] = 'Bitte Benutzername oder E-Mail eingeben.';
        }

        if (empty($errors)) {
            $pdo  = getDB();
            $stmt = $pdo->prepare("SELECT id, username, secret_question FROM users WHERE username = :id OR email = :id LIMIT 1");
            $stmt->execute([':id' => $identifier]);
            $user = $stmt->fetch();

            if ($user) {
                $_SESSION['recovery_user_id'] = $user['id'];
                $_SESSION['recovery_step']    = 'verify_answer';
                redirect('?action=verify_answer');
            } else {
                $errors[] = 'Kein Konto mit diesen Daten gefunden.';
            }
        }

        renderPage('Konto wiederherstellen', function() use ($identifier, $errors) {
            renderRecoverForm($identifier, $errors);
        });
    } else {
        renderPage('Konto wiederherstellen', function() {
            renderRecoverForm('', []);
        });
    }
}

function handleVerifyAnswer(): void {
    if (!isset($_SESSION['recovery_user_id']) || ($_SESSION['recovery_step'] ?? '') !== 'verify_answer') {
        setFlash('error', 'Ungültige Sitzung. Bitte starte den Wiederherstellungsprozess neu.');
        redirect('?action=recover');
    }

    $pdo    = getDB();
    $userId = (int)$_SESSION['recovery_user_id'];
    $stmt   = $pdo->prepare("SELECT * FROM users WHERE id = :id");
    $stmt->execute([':id' => $userId]);
    $user = $stmt->fetch();

    if (!$user) {
        unset($_SESSION['recovery_user_id'], $_SESSION['recovery_step']);
        setFlash('error', 'Benutzer nicht gefunden.');
        redirect('?action=recover');
    }

    if ($user['recovery_locked_until'] && new \DateTime() < new \DateTime($user['recovery_locked_until'])) {
        $lockedUntil = new \DateTime($user['recovery_locked_until']);
        renderPage('Konto gesperrt', function() use ($lockedUntil) {
            echo '<div class="card">';
            echo '<h2>Zu viele Fehlversuche</h2>';
            echo '<div class="alert alert-error">Dein Konto ist vorübergehend gesperrt. Versuche es nach ' . $lockedUntil->format('H:i:s') . ' Uhr erneut.</div>';
            echo '<a href="?action=login" class="btn btn-secondary">Zurück zur Anmeldung</a>';
            echo '</div>';
        });
        return;
    }

    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        verifyCsrf();

        $answer = $_POST['secret_answer'] ?? '';
        $errors = [];

        if (empty($answer)) {
            $errors[] = 'Bitte gib die Antwort ein.';
        }

        if (empty($errors)) {
            if (password_verify(strtolower(trim($answer)), $user['secret_answer_hash'])) {
                $stmt = $pdo->prepare("UPDATE users SET failed_recovery_attempts = 0, recovery_locked_until = NULL WHERE id = :id");
                $stmt->execute([':id' => $userId]);

                $token   = generateToken(32);
                $expires = (new \DateTime())->modify('+30 minutes')->format('Y-m-d H:i:s');

                $stmt = $pdo->prepare("UPDATE users SET reset_token = :token, reset_expires = :expires WHERE id = :id");
                $stmt->execute([':token' => $token, ':expires' => $expires, ':id' => $userId]);

                $_SESSION['recovery_step']  = 'reset_password';
                $_SESSION['recovery_token'] = $token;

                redirect('?action=reset_password');
            } else {
                $attempts = (int)$user['failed_recovery_attempts'] + 1;
                $lockUntil = null;

                if ($attempts >= 5) {
                    $lockUntil = (new \DateTime())->modify('+15 minutes')->format('Y-m-d H:i:s');
                    $attempts  = 0;
                    $errors[]  = 'Zu viele Fehlversuche. Dein Konto ist für 15 Minuten gesperrt.';
                } else {
                    $remaining = 5 - $attempts;
                    $errors[]  = 'Falsche Antwort. Noch ' . $remaining . ' Versuch(e) verbleibend.';
                }

                $stmt = $pdo->prepare("UPDATE users SET failed_recovery_attempts = :attempts, recovery_locked_until = :lock WHERE id = :id");
                $stmt->execute([':attempts' => $attempts, ':lock' => $lockUntil, ':id' => $userId]);
            }
        }

        renderPage('Geheime Frage beantworten', function() use ($user, $errors) {
            renderVerifyAnswerForm($user, $errors);
        });
    } else {
        renderPage('Geheime Frage beantworten', function() use ($user) {
            renderVerifyAnswerForm($user, []);
        });
    }
}

    echo '<div class="card">';
    echo '<h2>Sicherheitsfrage beantworten</h2>';
    echo '<p>Beantworte deine Sicherheitsfrage, um dein Passwort zurückzusetzen.</p>';
    renderErrors($errors);

}

function handleResetPassword(): void {
    if (
        !isset($_SESSION['recovery_user_id']) ||
        ($_SESSION['recovery_step'] ?? '') !== 'reset_password' ||
        !isset($_SESSION['recovery_token'])
    ) {
        setFlash('error', 'Ungültige Sitzung. Bitte starte den Wiederherstellungsprozess neu.');
        redirect('?action=recover');
    }

    $pdo    = getDB();
    $userId = (int)$_SESSION['recovery_user_id'];
    $token  = $_SESSION['recovery_token'];

    $stmt = $pdo->prepare("SELECT * FROM users WHERE id = :id AND reset_token = :token AND reset_expires > NOW()");
    $stmt->execute([':id' => $userId, ':token' => $token]);
    $user = $stmt->fetch();

    if (!$user) {
        unset($_SESSION['recovery_user_id'], $_SESSION['recovery_step'], $_SESSION['recovery_token']);
        setFlash('error', 'Sitzung abgelaufen oder ungültig. Bitte starte den Prozess neu.');
        redirect('?action=recover');
    }

    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        verifyCsrf();

        $newPassword     = $_POST['new_password'] ?? '';
        $confirmPassword = $_POST['confirm_password'] ?? '';
        $errors          = [];

        if (strlen($newPassword) < 8) {
            $errors[] = 'Das neue Passwort muss mindestens 8 Zeichen lang sein.';
        }
        if ($newPassword !== $confirmPassword) {
            $errors[] = 'Die Passwörter stimmen nicht überein.';
        }

        if (empty($errors)) {
            $newHash = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
            $stmt    = $pdo->prepare("UPDATE users SET password_hash = :hash, reset_token = NULL, reset_expires = NULL, failed_recovery_attempts = 0 WHERE id = :id");
            $stmt->execute([':hash' => $newHash, ':id' => $userId]);

            unset($_SESSION['recovery_user_id'], $_SESSION['recovery_step'], $_SESSION['recovery_token']);

            setFlash('success', 'Passwort erfolgreich zurückgesetzt! Du kannst dich jetzt anmelden.');
            redirect('?action=login');
        }

        renderPage('Neues Passwort setzen', function() use ($errors) {
            renderResetPasswordForm($errors);
        });
    } else {
        renderPage('Neues Passwort setzen', function() {
            renderResetPasswordForm([]);
        });
    }
}

function csrfField(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = generateToken(32);
    }
    return '<input type="hidden" name="csrf_token" value="' . htmlspecialchars($_SESSION['csrf_token']) . '">';
}

function verifyCsrf(): void {
    $token = $_POST['csrf_token'] ?? '';
    if (empty($_SESSION['csrf_token']) || !hash_equals($_SESSION['csrf_token'], $token)) {
        http_response_code(403);
        die('CSRF-Token ungültig.');
    }
    $_SESSION['csrf_token'] = generateToken(32);
}

function renderErrors(array $errors): void {
    if (!empty($errors)) {
        echo '<div class="alert alert-error"><ul>';
        foreach ($errors as $error) {
            echo '<li>' . htmlspecialchars($error) . '</li>';
        }
        echo '</ul></div>';
    }
}
