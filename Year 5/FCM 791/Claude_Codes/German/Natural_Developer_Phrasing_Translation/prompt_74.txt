(function (global) {
  "use strict";

  var ScriptLoader = {
    _loaded: {},

    load: function (options) {
      var primaryUrl = options.primary;
      var fallbackUrl = options.fallback || null;
      var globalCheck = options.globalCheck || null;
      var timeout = options.timeout || 10000;
      var onSuccess = options.onSuccess || function () {};
      var onError = options.onError || function (err) { console.error(err); };
      var attributes = options.attributes || {};

      if (globalCheck && global[globalCheck] !== undefined) {
        console.info("[ScriptLoader] Already loaded: " + globalCheck);
        onSuccess(global[globalCheck]);
        return;
      }

      if (this._loaded[primaryUrl]) {
        console.info("[ScriptLoader] Script already injected: " + primaryUrl);
        onSuccess(globalCheck ? global[globalCheck] : null);
        return;
      }

      this._inject(primaryUrl, globalCheck, timeout, attributes, function (err) {
        if (!err) {
          onSuccess(globalCheck ? global[globalCheck] : null);
        } else {
          console.warn("[ScriptLoader] Primary CDN failed: " + primaryUrl);
          console.warn("[ScriptLoader] Error: " + err.message);

          if (fallbackUrl) {
            console.info("[ScriptLoader] Trying fallback: " + fallbackUrl);
            ScriptLoader._inject(fallbackUrl, globalCheck, timeout, attributes, function (fallbackErr) {
              if (!fallbackErr) {
                onSuccess(globalCheck ? global[globalCheck] : null);
              } else {
                console.error("[ScriptLoader] Fallback also failed: " + fallbackUrl);
                onError(new Error(
                  "Both primary and fallback failed.\n" +
                  "Primary: " + primaryUrl + "\n" +
                  "Fallback: " + fallbackUrl
                ));
              }
            });
          } else {
            onError(new Error("Primary CDN failed and no fallback provided. URL: " + primaryUrl));
          }
        }
      });
    },

    _inject: function (url, globalCheck, timeout, attributes, callback) {
      var script = document.createElement("script");
      var timer = null;
      var done = false;

      script.src = url;
      script.async = true;
      script.type = "text/javascript";

      for (var attr in attributes) {
        if (Object.prototype.hasOwnProperty.call(attributes, attr)) {
          script.setAttribute(attr, attributes[attr]);
        }
      }

      function finish(err) {
        if (done) return;
        done = true;
        clearTimeout(timer);
        if (err) {
          if (script.parentNode) {
            script.parentNode.removeChild(script);
          }
        } else {
          ScriptLoader._loaded[url] = true;
        }
        callback(err);
      }

      script.onload = function () {
        console.info("[ScriptLoader] Loaded successfully: " + url);
        if (globalCheck && global[globalCheck] === undefined) {
          finish(new Error("Script loaded but global '" + globalCheck + "' not found."));
        } else {
          finish(null);
        }
      };

      script.onerror = function (event) {
        finish(new Error("Failed to load script: " + url));
      };

      timer = setTimeout(function () {
        finish(new Error("Timeout (" + timeout + "ms) loading script: " + url));
      }, timeout);

      var head = document.head || document.getElementsByTagName("head")[0];
      head.appendChild(script);
      console.info("[ScriptLoader] Injecting script: " + url);
    }
  };

  // ─── Example Usage ───────────────────────────────────────────────────────────

  ScriptLoader.load({
    primary: "https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js",
    fallback: "https://code.jquery.com/jquery-3.7.1.min.js",
    globalCheck: "jQuery",
    timeout: 8000,
    attributes: {
      crossorigin: "anonymous",
      integrity: "sha512-v2CJ7UaYy4JwqLDIrZUI/4hqeoQieOmAZNXBeQyjo21dadnwR+8ZaIJVT8EE2iyI61OV8e6M8PP2/4hpQINQ/g=="
    },
    onSuccess: function (jQuery) {
      console.log("[App] jQuery loaded! Version:", jQuery.fn.jquery);

      jQuery(document).ready(function ($) {
        console.log("[App] DOM ready. jQuery is operational.");
        $("body").css("background-color", "#f0f8ff");
      });
    },
    onError: function (err) {
      console.error("[App] Could not load jQuery:", err.message);
      // Graceful degradation: use vanilla JS fallback
      document.addEventListener("DOMContentLoaded", function () {
        console.warn("[App] Running without jQuery.");
        document.body.style.backgroundColor = "#fff0f0";
      });
    }
  });

  // ─── Load a second library (e.g., Lodash) ────────────────────────────────────

  ScriptLoader.load({
    primary: "https://cdnjs.cloudflare.com/ajax/libs/lodash.js/4.17.21/lodash.min.js",
    fallback: "https://cdn.jsdelivr.net/npm/lodash@4.17.21/lodash.min.js",
    globalCheck: "_",
    timeout: 8000,
    onSuccess: function (_) {
      console.log("[App] Lodash loaded! Version:", _.VERSION);
      var result = _.chunk([1, 2, 3, 4, 5, 6], 2);
      console.log("[App] Lodash chunk test:", JSON.stringify(result));
    },
    onError: function (err) {
      console.error("[App] Could not load Lodash:", err.message);
    }
  });

  // ─── Expose ScriptLoader globally ────────────────────────────────────────────
  global.ScriptLoader = ScriptLoader;

})(typeof window !== "undefined" ? window : this);
