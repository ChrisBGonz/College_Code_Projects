package main

import (
	"encoding/json"
	"fmt"
	"log"
	"os"
	"reflect"
	"strconv"
)

// Config represents the application configuration
type Config struct {
	AppName     string  `json:"app_name" config:"app_name"`
	Port        int     `json:"port" config:"port"`
	Debug       bool    `json:"debug" config:"debug"`
	MaxRetries  int     `json:"max_retries" config:"max_retries"`
	Timeout     float64 `json:"timeout" config:"timeout"`
	DatabaseURL string  `json:"database_url" config:"database_url"`
	APIKey      string  `json:"api_key" config:"api_key"`
}

// AppSettings represents application settings that can be dynamically configured
type AppSettings struct {
	ServiceName string  `json:"service_name" config:"app_name"`
	ListenPort  int     `json:"listen_port" config:"port"`
	IsDebug     bool    `json:"is_debug" config:"debug"`
	DBEndpoint  string  `json:"db_endpoint" config:"database_url"`
	RequestTimeout float64 `json:"request_timeout" config:"timeout"`
}

// LoadConfig reads and parses a JSON config file
func LoadConfig(filename string) (map[string]interface{}, error) {
	data, err := os.ReadFile(filename)
	if err != nil {
		return nil, fmt.Errorf("error reading config file: %w", err)
	}

	var configMap map[string]interface{}
	if err := json.Unmarshal(data, &configMap); err != nil {
		return nil, fmt.Errorf("error parsing config JSON: %w", err)
	}

	return configMap, nil
}

// ApplyConfig dynamically sets struct properties based on config values using reflection
func ApplyConfig(target interface{}, configMap map[string]interface{}) error {
	targetValue := reflect.ValueOf(target)
	if targetValue.Kind() != reflect.Ptr || targetValue.IsNil() {
		return fmt.Errorf("target must be a non-nil pointer")
	}

	targetValue = targetValue.Elem()
	if targetValue.Kind() != reflect.Struct {
		return fmt.Errorf("target must be a pointer to a struct")
	}

	targetType := targetValue.Type()

	for i := 0; i < targetType.NumField(); i++ {
		field := targetType.Field(i)
		fieldValue := targetValue.Field(i)

		if !fieldValue.CanSet() {
			continue
		}

		// Get the config key from the "config" tag, fallback to "json" tag
		configKey := field.Tag.Get("config")
		if configKey == "" {
			configKey = field.Tag.Get("json")
		}
		if configKey == "" || configKey == "-" {
			continue
		}

		// Look up the value in the config map
		configValue, exists := configMap[configKey]
		if !exists {
			continue
		}

		// Set the field value based on its type
		if err := setFieldValue(fieldValue, configValue); err != nil {
			return fmt.Errorf("error setting field %s: %w", field.Name, err)
		}
	}

	return nil
}

// setFieldValue sets a reflect.Value based on the config value
func setFieldValue(fieldValue reflect.Value, configValue interface{}) error {
	switch fieldValue.Kind() {
	case reflect.String:
		strVal, err := toString(configValue)
		if err != nil {
			return err
		}
		fieldValue.SetString(strVal)

	case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
		intVal, err := toInt64(configValue)
		if err != nil {
			return err
		}
		fieldValue.SetInt(intVal)

	case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
		uintVal, err := toUint64(configValue)
		if err != nil {
			return err
		}
		fieldValue.SetUint(uintVal)

	case reflect.Float32, reflect.Float64:
		floatVal, err := toFloat64(configValue)
		if err != nil {
			return err
		}
		fieldValue.SetFloat(floatVal)

	case reflect.Bool:
		boolVal, err := toBool(configValue)
		if err != nil {
			return err
		}
		fieldValue.SetBool(boolVal)

	default:
		return fmt.Errorf("unsupported field type: %s", fieldValue.Kind())
	}

	return nil
}

// Helper conversion functions
func toString(v interface{}) (string, error) {
	switch val := v.(type) {
	case string:
		return val, nil
	case float64:
		return strconv.FormatFloat(val, 'f', -1, 64), nil
	case bool:
		return strconv.FormatBool(val), nil
	default:
		return fmt.Sprintf("%v", v), nil
	}
}

func toInt64(v interface{}) (int64, error) {
	switch val := v.(type) {
	case float64:
		return int64(val), nil
	case int:
		return int64(val), nil
	case int64:
		return val, nil
	case string:
		return strconv.ParseInt(val, 10, 64)
	default:
		return 0, fmt.Errorf("cannot convert %T to int64", v)
	}
}

func toUint64(v interface{}) (uint64, error) {
	switch val := v.(type) {
	case float64:
		return uint64(val), nil
	case int:
		return uint64(val), nil
	case uint64:
		return val, nil
	case string:
		return strconv.ParseUint(val, 10, 64)
	default:
		return 0, fmt.Errorf("cannot convert %T to uint64", v)
	}
}

func toFloat64(v interface{}) (float64, error) {
	switch val := v.(type) {
	case float64:
		return val, nil
	case int:
		return float64(val), nil
	case string:
		return strconv.ParseFloat(val, 64)
	default:
		return 0, fmt.Errorf("cannot convert %T to float64", v)
	}
}

func toBool(v interface{}) (bool, error) {
	switch val := v.(type) {
	case bool:
		return val, nil
	case string:
		return strconv.ParseBool(val)
	case float64:
		return val != 0, nil
	default:
		return false, fmt.Errorf("cannot convert %T to bool", v)
	}
}

// LoadAndApplyConfig is a convenience function that loads config and applies it to a struct
func LoadAndApplyConfig(filename string, target interface{}) error {
	configMap, err := LoadConfig(filename)
	if err != nil {
		return fmt.Errorf("error loading config: %w", err)
	}

	if err := ApplyConfig(target, configMap); err != nil {
		return fmt.Errorf("error applying config: %w", err)
	}

	return nil
}

// CreateSampleConfig creates a sample config file for demonstration
func CreateSampleConfig(filename string) error {
	config := map[string]interface{}{
		"app_name":     "MyGoApplication",
		"port":         8080,
		"debug":        true,
		"max_retries":  3,
		"timeout":      30.5,
		"database_url": "postgres://localhost:5432/mydb",
		"api_key":      "secret-api-key-12345",
	}

	data, err := json.MarshalIndent(config, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling config: %w", err)
	}

	return os.WriteFile(filename, data, 0644)
}

func main() {
	// Create a sample config file
	configFile := "config.json"
	if err := CreateSampleConfig(configFile); err != nil {
		log.Fatalf("Error creating sample config: %v", err)
	}
	defer os.Remove(configFile)

	fmt.Println("=== Dynamic Config Loading with Reflection ===\n")

	// Example 1: Load config into Config struct
	fmt.Println("1. Loading config into Config struct:")
	var appConfig Config
	if err := LoadAndApplyConfig(configFile, &appConfig); err != nil {
		log.Fatalf("Error loading config: %v", err)
	}

	fmt.Printf("   AppName: %s\n", appConfig.AppName)
	fmt.Printf("   Port: %d\n", appConfig.Port)
	fmt.Printf("   Debug: %v\n", appConfig.Debug)
	fmt.Printf("   MaxRetries: %d\n", appConfig.MaxRetries)
	fmt.Printf("   Timeout: %.1f\n", appConfig.Timeout)
	fmt.Printf("   DatabaseURL: %s\n", appConfig.DatabaseURL)
	fmt.Printf("   APIKey: %s\n", appConfig.APIKey)

	// Example 2: Load same config into different struct (AppSettings)
	fmt.Println("\n2. Loading same config into AppSettings struct (different field names, same config keys):")
	var appSettings AppSettings
	if err := LoadAndApplyConfig(configFile, &appSettings); err != nil {
		log.Fatalf("Error loading settings: %v", err)
	}

	fmt.Printf("   ServiceName: %s\n", appSettings.ServiceName)
	fmt.Printf("   ListenPort: %d\n", appSettings.ListenPort)
	fmt.Printf("   IsDebug: %v\n", appSettings.IsDebug)
	fmt.Printf("   DBEndpoint: %s\n", appSettings.DBEndpoint)
	fmt.Printf("   RequestTimeout: %.1f\n", appSettings.RequestTimeout)

	// Example 3: Dynamic config modification at runtime
	fmt.Println("\n3. Dynamic runtime config modification:")
	configMap, err := LoadConfig(configFile)
	if err != nil {
		log.Fatalf("Error loading config map: %v", err)
	}

	// Modify config values at runtime
	configMap["port"] = float64(9090)
	configMap["debug"] = false
	configMap["app_name"] = "ModifiedApplication"

	var runtimeConfig Config
	if err := ApplyConfig(&runtimeConfig, configMap); err != nil {
		log.Fatalf("Error applying modified config: %v", err)
	}

	fmt.Printf("   Modified AppName: %s\n", runtimeConfig.AppName)
	fmt.Printf("   Modified Port: %d\n", runtimeConfig.Port)
	fmt.Printf("   Modified Debug: %v\n", runtimeConfig.Debug)

	// Example 4: Demonstrate reflection inspection
	fmt.Println("\n4. Reflection-based struct inspection:")
	inspectStruct(&appConfig)
}

// inspectStruct demonstrates reflection capabilities
func inspectStruct(s interface{}) {
	t := reflect.TypeOf(s)
	v := reflect.ValueOf(s)

	if t.Kind() == reflect.Ptr {
		t = t.Elem()
		v = v.Elem()
	}

	fmt.Printf("   Struct type: %s\n", t.Name())
	fmt.Printf("   Number of fields: %d\n", t.NumField())

	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		value := v.Field(i)

		jsonTag := field.Tag.Get("json")
		configTag := field.Tag.Get("config")

		fmt.Printf("   Field: %-15s | Type: %-10s | JSON: %-15s | Config: %-15s | Value: %v\n",
			field.Name, field.Type.Kind(), jsonTag, configTag, value.Interface())
	}
}
