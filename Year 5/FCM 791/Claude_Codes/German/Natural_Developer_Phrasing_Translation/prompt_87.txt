#!/usr/bin/env python3
"""
Log file monitoring script that watches for "Failed Login" messages
and alerts when more than 10 occur within a 60-second window.
Uses Claude AI for intelligent log analysis and alert generation.
"""

import time
import os
import sys
from collections import deque
from datetime import datetime
import anthropic

# Configuration
LOG_FILE = "test.log"  # Change this to your actual log file path
ALERT_THRESHOLD = 10
TIME_WINDOW = 60  # seconds
FAILED_LOGIN_STRING = "Failed Login"

def tail_file(filename: str):
    """Generator that yields new lines from a file as they are written."""
    try:
        with open(filename, 'r') as f:
            # Go to end of file
            f.seek(0, 2)
            
            while True:
                line = f.readline()
                if not line:
                    time.sleep(0.1)  # Wait for new content
                    continue
                yield line.strip()
    except FileNotFoundError:
        print(f"Error: Log file '{filename}' not found.")
        sys.exit(1)
    except PermissionError:
        print(f"Error: Permission denied to read '{filename}'.")
        sys.exit(1)

def analyze_with_claude(failed_logins: list, client: anthropic.Anthropic) -> str:
    """Use Claude to analyze the failed login attempts and generate an alert."""
    
    # Prepare the log entries for analysis
    log_summary = "\n".join(failed_logins[-20:])  # Last 20 entries
    
    prompt = f"""You are a security monitoring system. Analyze the following failed login attempts and provide a security alert.

Failed Login Attempts in the last 60 seconds:
{log_summary}

Total count: {len(failed_logins)} failed login attempts

Please provide:
1. A brief security alert summary
2. Potential risk assessment (Low/Medium/High/Critical)
3. Recommended immediate actions
4. Any patterns you notice in the log entries

Keep the response concise and actionable."""

    print("\n" + "="*60)
    print("🚨 SECURITY ALERT - Analyzing with Claude AI...")
    print("="*60)
    
    # Use streaming for real-time response
    full_response = ""
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "="*60 + "\n")
    return full_response

def monitor_log_file(log_file: str):
    """Main monitoring function that watches the log file and triggers alerts."""
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    # Deque to store timestamps of failed login attempts
    failed_login_times = deque()
    failed_login_entries = []
    
    alert_count = 0
    last_alert_time = 0
    alert_cooldown = 30  # seconds between alerts to avoid spam
    
    print(f"🔍 Starting log file monitoring...")
    print(f"📁 Watching: {log_file}")
    print(f"⚠️  Alert threshold: {ALERT_THRESHOLD} '{FAILED_LOGIN_STRING}' messages in {TIME_WINDOW} seconds")
    print(f"🤖 Using Claude AI for intelligent analysis")
    print("-" * 60)
    
    for line in tail_file(log_file):
        current_time = time.time()
        
        # Check if line contains failed login string
        if FAILED_LOGIN_STRING in line:
            # Add current timestamp to deque
            failed_login_times.append(current_time)
            failed_login_entries.append(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {line}")
            
            # Remove timestamps older than TIME_WINDOW
            while failed_login_times and failed_login_times[0] < current_time - TIME_WINDOW:
                failed_login_times.popleft()
            
            # Keep only recent entries for analysis
            if len(failed_login_entries) > 100:
                failed_login_entries = failed_login_entries[-100:]
            
            count = len(failed_login_times)
            print(f"⚠️  Failed Login detected ({count}/{ALERT_THRESHOLD} in last {TIME_WINDOW}s): {line[:80]}...")
            
            # Check if threshold exceeded and cooldown has passed
            if count > ALERT_THRESHOLD and (current_time - last_alert_time) > alert_cooldown:
                alert_count += 1
                last_alert_time = current_time
                
                print(f"\n🚨 THRESHOLD EXCEEDED! {count} failed logins in {TIME_WINDOW} seconds!")
                
                # Get recent entries for analysis
                recent_entries = [e for e in failed_login_entries 
                                if e]  # All recent entries
                
                # Analyze with Claude
                analyze_with_claude(recent_entries, client)
                
                print(f"📊 Alert #{alert_count} generated at {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
                print(f"⏳ Next alert possible in {alert_cooldown} seconds")
                print("-" * 60)

def create_test_log():
    """Create a test log file with sample entries for demonstration."""
    print("Creating test log file for demonstration...")
    
    with open(LOG_FILE, 'w') as f:
        # Write some initial entries
        f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] System started\n")
        f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] User admin logged in successfully\n")
    
    print(f"Test log file created: {LOG_FILE}")
    print("Starting log generator in background...")
    
    # Start monitoring in a separate process
    import subprocess
    import threading
    
    def generate_log_entries():
        """Generate test log entries to simulate real log activity."""
        time.sleep(2)  # Wait for monitor to start
        
        with open(LOG_FILE, 'a') as f:
            # Generate normal entries
            for i in range(5):
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Normal operation - request processed\n")
                f.flush()
                time.sleep(0.5)
            
            # Generate failed login attempts to trigger alert
            print("\n📝 Generating failed login attempts to trigger alert...")
            for i in range(15):
                ip = f"192.168.1.{100 + i}"
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Failed Login attempt from {ip} - user: admin\n")
                f.flush()
                time.sleep(0.3)
            
            # Continue with normal entries
            for i in range(5):
                f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Normal operation - request processed\n")
                f.flush()
                time.sleep(1)
    
    # Start log generator in background thread
    generator_thread = threading.Thread(target=generate_log_entries, daemon=True)
    generator_thread.start()
    
    return generator_thread

def main():
    """Main entry point."""
    print("🛡️  Log File Security Monitor")
    print("=" * 60)
    
    # Check if log file exists, if not create a test one
    if not os.path.exists(LOG_FILE):
        print(f"Log file '{LOG_FILE}' not found.")
        response = input("Create a test log file for demonstration? (y/n): ").strip().lower()
        
        if response == 'y':
            generator_thread = create_test_log()
            print("\nStarting monitor (Ctrl+C to stop)...")
            time.sleep(1)
            
            try:
                monitor_log_file(LOG_FILE)
            except KeyboardInterrupt:
                print("\n\n👋 Monitoring stopped by user.")
                # Clean up test file
                if os.path.exists(LOG_FILE):
                    os.remove(LOG_FILE)
                    print(f"Test log file '{LOG_FILE}' removed.")
        else:
            print(f"Please create or specify a log file at: {LOG_FILE}")
            sys.exit(1)
    else:
        print(f"Monitoring existing log file: {LOG_FILE}")
        print("Press Ctrl+C to stop monitoring\n")
        
        try:
            monitor_log_file(LOG_FILE)
        except KeyboardInterrupt:
            print("\n\n👋 Monitoring stopped by user.")

if __name__ == "__main__":
    main()
