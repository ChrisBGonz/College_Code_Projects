function getUserProfile(userId) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      const profile = userProfiles[userId];
      if (profile) {
        resolve(profile);
      } else {
        reject(new Error(`User with ID "${userId}" not found`));
      }
    }, 50);
  });
}

function sanitizeFilename(name) {
  return name.replace(/[^a-z0-9_\-]/gi, '_').toLowerCase();
}

app.get('/api/profile/download', async (req, res) => {
  const { user_id } = req.query;

  if (!user_id) {
    return res.status(400).json({
      error: 'Bad Request',
      message: 'Query parameter "user_id" is required.',
    });
  }

  const trimmedUserId = String(user_id).trim();

  if (!/^[a-zA-Z0-9_\-]+$/.test(trimmedUserId)) {
    return res.status(400).json({
      error: 'Bad Request',
      message: 'Invalid "user_id" format. Only alphanumeric characters, underscores, and hyphens are allowed.',
    });
  }

  try {
    const profile = await getUserProfile(trimmedUserId);

    const exportData = {
      exportedAt: new Date().toISOString(),
      exportVersion: '1.0',
      profile,
    };

    const jsonString = JSON.stringify(exportData, null, 2);
    const safeUsername = sanitizeFilename(profile.name);
    const filename = `profile_${safeUsername}_${trimmedUserId}.json`;

    res.setHeader('Content-Type', 'application/json; charset=utf-8');
    res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
    res.setHeader('Content-Length', Buffer.byteLength(jsonString, 'utf8'));
    res.setHeader('Cache-Control', 'no-store');

    return res.status(200).send(jsonString);
  } catch (error) {
    if (error.message.includes('not found')) {
      return res.status(404).json({
        error: 'Not Found',
        message: error.message,
      });
    }

    console.error('Unexpected error:', error);
    return res.status(500).json({
      error: 'Internal Server Error',
      message: 'An unexpected error occurred while processing your request.',
    });
  }
});

app.use((req, res) => {
  res.status(404).json({
    error: 'Not Found',
    message: `Route "${req.method} ${req.path}" does not exist.`,
  });
});

app.listen(PORT, () => {
  console.log(`Server is running on http://localhost:${PORT}`);
  console.log(`Download profile: http://localhost:${PORT}/api/profile/download?user_id=1`);
});

module.exports = app;
