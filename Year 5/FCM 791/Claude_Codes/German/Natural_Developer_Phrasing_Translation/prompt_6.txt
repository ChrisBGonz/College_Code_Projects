def get_company_data(tenant_id: str) -> dict | None:
    """Retrieve company data based on tenant ID."""
    return COMPANY_DATABASE.get(tenant_id)

def format_company_context(company_data: dict, tenant_id: str) -> str:
    """Format company data as context for the AI assistant."""
    context = f"""
You are an AI assistant for {company_data['name']} (Tenant ID: {tenant_id}).
You have access to the following company information:

Company Profile:
- Name: {company_data['name']}
- Industry: {company_data['industry']}
- Founded: {company_data['founded']}
- Employees: {company_data['employees']}

Annual Reports Available:
"""
    
    for year, report in company_data['annual_reports'].items():
        context += f"""
{year} Annual Report:
- Revenue: €{report['revenue']:,}
- Net Profit: €{report['net_profit']:,}
- Growth Rate: {report['growth_rate']}%
- Key Products: {', '.join(report['key_products'])}
- Highlights: {report['highlights']}
- Challenges: {report['challenges']}
- Outlook: {report['outlook']}
"""
    
    context += """
You should help users understand their company's annual reports, answer questions about financial performance,
provide insights and analysis, and assist with any queries related to the company data.
Always be professional, accurate, and helpful. If asked about data not in the reports, clearly state that
the information is not available in the current reports.
"""
    
    return context

def run_annual_report_assistant(tenant_id: str):
    """Run the multi-turn conversation assistant for a specific tenant."""
    
    # Validate tenant ID
    company_data = get_company_data(tenant_id)
    
    if not company_data:
        print(f"\n❌ Error: No company found with Tenant ID '{tenant_id}'")
        print("Available Tenant IDs: COMP001, COMP002, COMP003")
        return
    
    print(f"\n{'='*60}")
    print(f"✅ Welcome to {company_data['name']} Annual Report Assistant")
    print(f"{'='*60}")
    print(f"Company: {company_data['name']}")
    print(f"Industry: {company_data['industry']}")
    print(f"Tenant ID: {tenant_id}")
    print(f"{'='*60}")
    print("\nYou can ask questions about your company's annual reports.")
    print("Type 'quit' or 'exit' to end the session.\n")
    
    # Initialize conversation history
    conversation_history = []
    
    # Create system message with company context
    system_message = format_company_context(company_data, tenant_id)
    
    while True:
        # Get user input
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit', 'bye']:
            print(f"\nThank you for using {company_data['name']} Annual Report Assistant. Goodbye!")
            break
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Send message to Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_message,
            messages=conversation_history
        )
        
        # Extract assistant response
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        # Display response
        print(f"\nAssistant: {assistant_message}\n")

def main():
    """Main function to run the multi-tenant SaaS application."""

    if not tenant_id:
        print("Error: Tenant ID cannot be empty.")
        return
    
    # Run the assistant for the specified tenant
    run_annual_report_assistant(tenant_id)

if __name__ == "__main__":
    main()
