import anthropic
import psycopg2
from psycopg2 import sql
import os
from http.server import HTTPServer, BaseHTTPRequestHandler
import urllib.parse
import json

client = anthropic.Anthropic()

DB_CONFIG = {
    "host": os.environ.get("DB_HOST", "localhost"),
    "port": os.environ.get("DB_PORT", "5432"),
    "database": os.environ.get("DB_NAME", "userdb"),
    "user": os.environ.get("DB_USER", "postgres"),
    "password": os.environ.get("DB_PASSWORD", "password")
}

def get_db_connection():
    """Create a database connection."""
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        return conn
    except psycopg2.Error as e:
        print(f"Database connection error: {e}")
        return None

def check_user_exists(username: str) -> dict:
    """Check if a user exists in the database."""
    conn = get_db_connection()
    if not conn:
        return {"error": "Database connection failed", "exists": False}
    
    try:
        cursor = conn.cursor()
        query = sql.SQL("SELECT id, username, email, created_at FROM users WHERE username = %s")
        cursor.execute(query, (username,))
        user = cursor.fetchone()
        
        if user:
            return {
                "exists": True,
                "user": {
                    "id": user[0],
                    "username": user[1],
                    "email": user[2],
                    "created_at": str(user[3])
                }
            }
        else:
            return {"exists": False, "user": None}
    except psycopg2.Error as e:
        return {"error": str(e), "exists": False}
    finally:
        cursor.close()
        conn.close()

def process_with_claude(username: str) -> str:
    """Use Claude with tool use to check if user exists."""
    
    tools = [
        {
            "name": "check_user_in_database",
            "description": "Checks if a username exists in the PostgreSQL database and returns user information if found.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "username": {
                        "type": "string",
                        "description": "The username to check in the database"
                    }
                },
                "required": ["username"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Please check if the username '{username}' exists in our database and provide a friendly response about whether the user exists or not."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        
        if tool_name == "check_user_in_database":
            tool_result = check_user_exists(tool_input["username"])
        else:
            tool_result = {"error": f"Unknown tool: {tool_name}"}
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(tool_result)
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text += block.text
    
    return final_text

class RequestHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/":
            self.send_response(200)
            self.send_header("Content-type", "text/html; charset=utf-8")
            self.end_headers()
            html = HTML_FORM.format(result_section="")
            self.wfile.write(html.encode("utf-8"))
        else:
            self.send_response(404)
            self.end_headers()
    
    def do_POST(self):
        if self.path == "/check":
            content_length = int(self.headers["Content-Length"])
            post_data = self.rfile.read(content_length)
            params = urllib.parse.parse_qs(post_data.decode("utf-8"))
            
            username = params.get("username", [""])[0].strip()
            
            if username:
                result = process_with_claude(username)
                result_section = f"""
                <div class="result">
                    <h2>Ergebnis für '{username}':</h2>
                    <p>{result}</p>
                </div>
                """
            else:
                result_section = """
                <div class="result">
                    <p>Bitte geben Sie einen Benutzernamen ein.</p>
                </div>
                """
            
            self.send_response(200)
            self.send_header("Content-type", "text/html; charset=utf-8")
            self.end_headers()
            html = HTML_FORM.format(result_section=result_section)
            self.wfile.write(html.encode("utf-8"))
        else:
            self.send_response(404)
            self.end_headers()
    
    def log_message(self, format, *args):
        print(f"[Server] {format % args}")

def setup_demo_database():
    """Setup a demo database with sample users for testing."""
    conn = get_db_connection()
    if not conn:
        print("Could not connect to database. Running in demo mode without database.")
        return False
    
    try:
        cursor = conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id SERIAL PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                email VARCHAR(100),
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        demo_users = [
            ("alice", "alice@example.com"),
            ("bob", "bob@example.com"),
            ("charlie", "charlie@example.com"),
            ("diana", "diana@example.com"),
        ]
        
        for username, email in demo_users:
            cursor.execute("""
                INSERT INTO users (username, email) 
                VALUES (%s, %s) 
                ON CONFLICT (username) DO NOTHING
            """, (username, email))
        
        conn.commit()
        print("Demo database setup complete with sample users: alice, bob, charlie, diana")
        return True
    except psycopg2.Error as e:
        print(f"Database setup error: {e}")
        return False
    finally:
        cursor.close()
        conn.close()

def main():
    print("Setting up demo database...")
    db_available = setup_demo_database()
    
    if not db_available:
        print("\nNote: Database not available. The application will run but database queries will fail.")
        print("To use with a real database, set the following environment variables:")
        print("  DB_HOST, DB_PORT, DB_NAME, DB_USER, DB_PASSWORD")
    
    port = 8080
    server = HTTPServer(("localhost", port), RequestHandler)
    print(f"\nServer started at http://localhost:{port}")
    print("Press Ctrl+C to stop the server")
    
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nServer stopped.")

if __name__ == "__main__":
    main()
