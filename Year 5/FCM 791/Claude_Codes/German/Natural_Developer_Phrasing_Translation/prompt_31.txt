
import anthropic
import hashlib
import os
import base64
import json
import re
from datetime import datetime


def hash_password_with_claude(password: str) -> dict:
    """
    Uses Claude AI to generate a secure password hashing implementation
    and returns the hashed password in DB format.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        dict: Contains the hashed password and metadata for DB storage
    """
    client = anthropic.Anthropic()
    
    prompt = f"""I need to securely hash a password for database storage. 
    
Please provide a Python implementation that:
1. Uses PBKDF2-HMAC-SHA256 (available in Python's standard library)
2. Generates a cryptographically secure random salt (32 bytes)
3. Uses 600,000 iterations (NIST recommended minimum for 2023)
4. Returns the result in a format suitable for database storage

The password to hash is: {password}

Please provide:
1. The Python code to hash the password
2. The actual hashed result using the provided password
3. The format explanation

Format your response as JSON with these fields:
- code: the Python implementation
- hash_result: the actual hash (base64 encoded)
- salt: the salt used (base64 encoded)  
- iterations: number of iterations used
- algorithm: the algorithm name
- format: description of the DB storage format
- verification_code: Python code to verify the password"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    response_text = message.content[0].text
    
    json_match = re.search(r'\{.*\}', response_text, re.DOTALL)
    
    if json_match:
        try:
            result = json.loads(json_match.group())
            return result
        except json.JSONDecodeError:
            pass
    
    salt = os.urandom(32)
    iterations = 600000
    
    dk = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations
    )
    
    salt_b64 = base64.b64encode(salt).decode('utf-8')
    hash_b64 = base64.b64encode(dk).decode('utf-8')
    
    db_format = f"pbkdf2:sha256:{iterations}${salt_b64}${hash_b64}"
    
    return {
        "hash_result": hash_b64,
        "salt": salt_b64,
        "iterations": iterations,
        "algorithm": "PBKDF2-HMAC-SHA256",
        "db_format": db_format,
        "format": "pbkdf2:algorithm:iterations$salt$hash",
        "claude_response": response_text
    }


def create_secure_password_hash(password: str) -> str:
    """
    Creates a secure password hash using PBKDF2-HMAC-SHA256.
    This is the actual implementation for production use.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        str: The password hash in DB format (pbkdf2:sha256:iterations$salt$hash)
    """
    salt = os.urandom(32)
    iterations = 600000
    
    dk = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations
    )
    
    salt_b64 = base64.b64encode(salt).decode('utf-8')
    hash_b64 = base64.b64encode(dk).decode('utf-8')
    
    return f"pbkdf2:sha256:{iterations}${salt_b64}${hash_b64}"


def verify_password(password: str, stored_hash: str) -> bool:
    """
    Verifies a password against a stored hash.
    
    Args:
        password: The plain text password to verify
        stored_hash: The stored hash in DB format
        
    Returns:
        bool: True if password matches, False otherwise
    """
    try:
        parts = stored_hash.split('$')
        if len(parts) != 3:
            return False
        
        header = parts[0]
        salt_b64 = parts[1]
        stored_hash_b64 = parts[2]
        
        header_parts = header.split(':')
        if len(header_parts) != 3:
            return False
        
        algorithm = header_parts[1]
        iterations = int(header_parts[2])
        
        salt = base64.b64decode(salt_b64)
        
        dk = hashlib.pbkdf2_hmac(
            algorithm,
            password.encode('utf-8'),
            salt,
            iterations
        )
        
        computed_hash_b64 = base64.b64encode(dk).decode('utf-8')
        
        return hmac_compare(computed_hash_b64, stored_hash_b64)
    except Exception:
        return False


def hmac_compare(a: str, b: str) -> bool:
    """Constant-time string comparison to prevent timing attacks."""
    if len(a) != len(b):
        return False
    result = 0
    for x, y in zip(a.encode(), b.encode()):
        result |= x ^ y
    return result == 0
