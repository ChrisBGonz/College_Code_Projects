import anthropic
import re
from flask import Flask, Response, jsonify, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def extract_number_from_response(text: str) -> str | None:
    """Extract a number from the response text."""
    patterns = [
        r"=\s*([-+]?\d*\.?\d+(?:[eE][-+]?\d+)?)",
        r"(?:result|answer|equals?|is)\s*:?\s*([-+]?\d*\.?\d+(?:[eE][-+]?\d+)?)",
        r"^([-+]?\d*\.?\d+(?:[eE][-+]?\d+)?)$",
        r"([-+]?\d*\.?\d+(?:[eE][-+]?\d+)?)\s*$",
    ]

    for pattern in patterns:
        match = re.search(pattern, text, re.IGNORECASE | re.MULTILINE)
        if match:
            return match.group(1)

    return None


@app.route("/evaluate", methods=["POST"])
def evaluate_expression():
    """
    Route to evaluate a mathematical expression using Claude AI.
    Accepts JSON with 'expression' field and returns the calculated result.
    """
    data = request.get_json()

    if not data or "expression" not in data:
        return jsonify({"error": "Please provide an 'expression' field in the request body"}), 400

    expression = data["expression"].strip()

    if not expression:
        return jsonify({"error": "Expression cannot be empty"}), 400

    def generate():
        full_response = ""

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"""Calculate the following mathematical expression and provide ONLY the numerical result. 
Do not include any explanation, units, or additional text - just the number.

Expression: {expression}

Result:""",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                full_response += text
                yield text

        number = extract_number_from_response(full_response.strip())
        if number:
            yield f"\n\nExtracted result: {number}"

    return Response(
        stream_with_context(generate()),
        mimetype="text/plain",
        headers={
            "X-Expression": expression,
            "Cache-Control": "no-cache",
        },
    )


@app.route("/evaluate/simple", methods=["POST"])
def evaluate_expression_simple():
    """
    Non-streaming version that returns a JSON response with the result.
    """
    data = request.get_json()

    if not data or "expression" not in data:
        return jsonify({"error": "Please provide an 'expression' field in the request body"}), 400

    expression = data["expression"].strip()

    if not expression:
        return jsonify({"error": "Expression cannot be empty"}), 400

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Calculate the following mathematical expression and provide ONLY the numerical result. 
Do not include any explanation, units, or additional text - just the number.

Expression: {expression}

Result:""",
            }
        ],
    )

    response_text = message.content[0].text.strip()

    number = extract_number_from_response(response_text)

    result = {
        "expression": expression,
        "raw_response": response_text,
        "result": number if number else response_text,
    }

    if number:
        try:
            result["numeric_value"] = float(number)
        except ValueError:
            pass

    return jsonify(result)


@app.route("/health", methods=["GET"])
def health_check():
    """Health check endpoint."""
    return jsonify({"status": "healthy", "service": "Mathematical Expression Evaluator"})


if __name__ == "__main__":
    print("Starting Mathematical Expression Evaluator Flask App...")
    print("Available endpoints:")
    print("  POST /evaluate - Streaming evaluation of mathematical expressions")
    print("  POST /evaluate/simple - Non-streaming JSON response")
    print("  GET /health - Health check")
    print("\nExample usage:")
    print('  curl -X POST http://localhost:5000/evaluate/simple \\')
    print('    -H "Content-Type: application/json" \\')
    print('    -d \'{"expression": "2 + 2 * 3"}\'')
    app.run(debug=True, host="0.0.0.0", port=5000)
