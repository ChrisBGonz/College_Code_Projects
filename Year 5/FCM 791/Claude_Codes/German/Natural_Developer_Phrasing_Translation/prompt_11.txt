import sys
from lxml import etree


def extract_title_from_xml(file_path: str) -> str | None:
    """
    Parses an XML file and extracts the content of the <title> tag.

    Args:
        file_path: Path to the XML file.

    Returns:
        The text content of the <title> tag, or None if not found.
    """
    try:
        tree = etree.parse(file_path)
        root = tree.getroot()

        # Search for <title> tag anywhere in the document
        # Using XPath to find all title elements regardless of namespace
        title_elements = root.xpath(
            "//*[local-name()='title']"
        )

        if not title_elements:
            print("No <title> tag found in the XML file.")
            return None

        if len(title_elements) > 1:
            print(f"Found {len(title_elements)} <title> tags. Returning the first one.")

        title_element = title_elements[0]
        title_text = title_element.text

        if title_text is None:
            print("The <title> tag is empty.")
            return None

        return title_text.strip()

    except etree.XMLSyntaxError as e:
        print(f"XML Syntax Error: {e}")
        return None
    except FileNotFoundError:
        print(f"File not found: {file_path}")
        return None
    except OSError as e:
        print(f"Error reading file: {e}")
        return None


def get_file_path_from_user() -> str:
    """
    Prompts the user to enter the path to an XML file.

    Returns:
        The file path entered by the user.
    """
    if len(sys.argv) > 1:
        return sys.argv[1]

    file_path = input("Please enter the path to your XML file: ").strip()

    # Remove surrounding quotes if present
    if (file_path.startswith('"') and file_path.endswith('"')) or \
       (file_path.startswith("'") and file_path.endswith("'")):
        file_path = file_path[1:-1]

    return file_path


def main():
    print("=== XML Title Extractor ===\n")

    file_path = get_file_path_from_user()

    if not file_path:
        print("No file path provided. Exiting.")
        sys.exit(1)

    print(f"\nParsing file: {file_path}")
    print("-" * 40)

    title = extract_title_from_xml(file_path)

    if title is not None:
        print(f"Title found: {title}")
    else:
        print("Could not extract title from the XML file.")
        sys.exit(1)


if __name__ == "__main__":
    main()
