#!/usr/bin/env python3
"""
Script to load a very large file (~10 GB) into RAM and count the number of lines.
Uses Claude AI to analyze the approach and provide insights.
"""

import os
import time
import anthropic


def load_file_and_count_lines(file_path: str) -> tuple[int, float, float]:
    """
    Load a large file completely into RAM and count lines.
    
    Args:
        file_path: Path to the file to process
        
    Returns:
        Tuple of (line_count, load_time_seconds, memory_used_mb)
    """
    print(f"Loading file: {file_path}")
    print(f"File size: {os.path.getsize(file_path) / (1024**3):.2f} GB")
    
    start_time = time.time()
    
    # Load entire file into RAM
    with open(file_path, 'rb') as f:
        content = f.read()
    
    load_time = time.time() - start_time
    memory_used_mb = len(content) / (1024**2)
    
    print(f"File loaded in {load_time:.2f} seconds")
    print(f"Memory used: {memory_used_mb:.2f} MB")
    
    # Count lines
    count_start = time.time()
    line_count = content.count(b'\n')
    
    # Add 1 if file doesn't end with newline (last line without newline)
    if content and not content.endswith(b'\n'):
        line_count += 1
    
    count_time = time.time() - count_start
    print(f"Lines counted in {count_time:.4f} seconds")
    
    return line_count, load_time, memory_used_mb


def analyze_with_claude(file_path: str, line_count: int, load_time: float, memory_used_mb: float):
    """
    Use Claude AI with streaming to analyze the file processing results.
    
    Args:
        file_path: Path to the processed file
        line_count: Number of lines counted
        load_time: Time taken to load the file
        memory_used_mb: Memory used in MB
    """
    client = anthropic.Anthropic()
    
    file_size_gb = os.path.getsize(file_path) / (1024**3)
    
    prompt = f"""I just processed a large file with the following statistics:
    
- File path: {file_path}
- File size: {file_size_gb:.2f} GB
- Total lines: {line_count:,}
- Load time: {load_time:.2f} seconds
- Memory used: {memory_used_mb:.2f} MB
- Average line length: {(memory_used_mb * 1024 * 1024) / max(line_count, 1):.1f} bytes

Please analyze these results and provide:
1. Assessment of the performance (load speed in MB/s)
2. Memory efficiency analysis
3. Potential risks of loading such large files into RAM
4. Alternative approaches that might be more memory-efficient
5. Use cases where loading the entire file into RAM makes sense

Keep the analysis concise and practical."""

    print("\n" + "="*60)
    print("Claude AI Analysis (streaming):")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "="*60)


def create_test_file(file_path: str, size_gb: float = 0.1):
    """
    Create a test file of specified size for demonstration.
    
    Args:
        file_path: Path where to create the test file
        size_gb: Size of the file in GB (default 0.1 GB for testing)
    """
    print(f"Creating test file of {size_gb} GB...")
    
    target_size = int(size_gb * 1024**3)
    line = b"This is a test line with some content to simulate a large log file entry. " * 2 + b"\n"
    
    with open(file_path, 'wb') as f:
        written = 0
        while written < target_size:
            f.write(line)
            written += len(line)
    
    actual_size = os.path.getsize(file_path)
    print(f"Test file created: {actual_size / (1024**2):.2f} MB")


def main():
    """Main function to demonstrate large file processing."""
    
    # For demonstration, create a smaller test file
    # In production, you would use an actual ~10 GB file
    test_file = "large_test_file.txt"
    
    # Check if we should create a test file or use an existing one
    if not os.path.exists(test_file):
        print("No large file found. Creating a test file (100 MB for demonstration)...")
        print("Note: For actual 10 GB processing, provide a real large file.")
        create_test_file(test_file, size_gb=0.1)  # 100 MB for demo
    
    file_size_gb = os.path.getsize(test_file) / (1024**3)
    
    if file_size_gb < 1:
        print(f"\nNote: Using {file_size_gb*1024:.0f} MB test file for demonstration.")
        print("For actual 10 GB processing, the same code applies but requires sufficient RAM.\n")
    
    # Check available memory (rough estimate)
    try:
        import psutil
        available_ram_gb = psutil.virtual_memory().available / (1024**3)
        total_ram_gb = psutil.virtual_memory().total / (1024**3)
        print(f"System RAM: {total_ram_gb:.1f} GB total, {available_ram_gb:.1f} GB available")
        
        if file_size_gb > available_ram_gb * 0.8:
            print(f"WARNING: File size ({file_size_gb:.1f} GB) may exceed available RAM ({available_ram_gb:.1f} GB)")
            print("This could cause the system to use swap space or crash!")
    except ImportError:
        print("psutil not available - skipping memory check")
    
    print()
    
    # Load file and count lines
    line_count, load_time, memory_used_mb = load_file_and_count_lines(test_file)
    
    print(f"\nResults:")
    print(f"  Total lines: {line_count:,}")
    print(f"  Load time: {load_time:.2f} seconds")
    print(f"  Memory used: {memory_used_mb:.2f} MB")
    print(f"  Throughput: {memory_used_mb/load_time:.1f} MB/s")
    
    # Analyze results with Claude AI
    analyze_with_claude(test_file, line_count, load_time, memory_used_mb)
    
    # Clean up test file if it was created by this script
    if os.path.exists(test_file) and file_size_gb < 1:
        cleanup = input("\nDelete test file? (y/n): ").strip().lower()
        if cleanup == 'y':
            os.remove(test_file)
            print("Test file deleted.")


if __name__ == "__main__":
    main()
