package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

// TokenInfo represents a password reset token with expiration
type TokenInfo struct {
	Token     string
	ExpiresAt time.Time
	UserID    string
}

// generateSecureToken generates a cryptographically secure random token
func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate secure token: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// createPasswordResetToken creates a new password reset token that expires in 1 hour
func createPasswordResetToken(userID string) (*TokenInfo, error) {
	token, err := generateSecureToken(32) // 32 bytes = 64 hex characters
	if err != nil {
		return nil, err
	}

	return &TokenInfo{
		Token:     token,
		ExpiresAt: time.Now().Add(1 * time.Hour),
		UserID:    userID,
	}, nil
}

// isTokenValid checks if a token is still valid (not expired)
func isTokenValid(tokenInfo *TokenInfo) bool {
	return time.Now().Before(tokenInfo.ExpiresAt)
}

// explainTokenWithClaude uses Claude AI to explain the token generation process
func explainTokenWithClaude(tokenInfo *TokenInfo) error {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`I've generated a password reset token with the following properties:
- Token: %s (first 8 chars shown: %s...)
- Expires at: %s
- User ID: %s
- Valid: %v

Please explain in a few sentences:
1. Why this token is cryptographically secure
2. What the expiration time means for security
3. Best practices for storing and using such tokens`,
		tokenInfo.Token,
		tokenInfo.Token[:8],
		tokenInfo.ExpiresAt.Format(time.RFC3339),
		tokenInfo.UserID,
		isTokenValid(tokenInfo))

	fmt.Println("Asking Claude to explain the token security...")
	fmt.Println("---")

	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return fmt.Errorf("streaming error: %w", err)
	}

	fmt.Println("\n---")
	return nil
}

func main() {
	fmt.Println("Password Reset Token Generator")
	fmt.Println("================================")

	// Generate a token for a test user
	userID := "user_12345"
	fmt.Printf("Generating password reset token for user: %s\n\n", userID)

	tokenInfo, err := createPasswordResetToken(userID)
	if err != nil {
		fmt.Printf("Error generating token: %v\n", err)
		return
	}

	fmt.Printf("Token generated successfully!\n")
	fmt.Printf("Token: %s\n", tokenInfo.Token)
	fmt.Printf("Token length: %d characters\n", len(tokenInfo.Token))
	fmt.Printf("Expires at: %s\n", tokenInfo.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("Time until expiration: %s\n", time.Until(tokenInfo.ExpiresAt).Round(time.Second))
	fmt.Printf("Is valid: %v\n\n", isTokenValid(tokenInfo))

	// Use Claude to explain the security aspects
	if err := explainTokenWithClaude(tokenInfo); err != nil {
		fmt.Printf("Error getting Claude explanation: %v\n", err)
		return
	}

	// Demonstrate token validation
	fmt.Println("\nToken Validation Demo:")
	fmt.Println("----------------------")
	fmt.Printf("Current token valid: %v\n", isTokenValid(tokenInfo))

	// Create an expired token for demonstration
	expiredToken := &TokenInfo{
		Token:     tokenInfo.Token,
		ExpiresAt: time.Now().Add(-1 * time.Hour), // Already expired
		UserID:    userID,
	}
	fmt.Printf("Expired token valid: %v\n", isTokenValid(expiredToken))
}
