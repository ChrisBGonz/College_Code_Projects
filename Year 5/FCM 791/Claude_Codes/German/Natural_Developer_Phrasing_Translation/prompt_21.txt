
```python
import os
import json
import time
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SYSTEM_PROMPT = """You are an expert Python developer specializing in JWT (JSON Web Token) authentication and security. 
Your task is to help implement a comprehensive JWT authentication system using the PyJWT library.

When asked to implement JWT functionality, provide complete, working Python code that includes:
1. JWT token generation with various algorithms (HS256, RS256)
2. Token signing with secret keys or RSA keys
3. Token decoding and validation
4. Claims handling (iss, sub, aud, exp, nbf, iat, jti)
5. Signature verification
6. Error handling for expired tokens, invalid signatures, etc.
7. Refresh token functionality
8. Token blacklisting concept

Always provide production-ready code with proper error handling and security best practices.
When showing code examples, make them complete and runnable.
Explain security considerations and best practices."""

def chat_with_claude(user_message: str) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def extract_code_blocks(text: str) -> list[str]:
    """Extract Python code blocks from the response."""
    code_blocks = []
    lines = text.split('\n')
    in_code_block = False
    current_block = []
    
    for line in lines:
        if line.strip().startswith('```python'):
            in_code_block = True
            current_block = []
        elif line.strip() == '```' and in_code_block:
            in_code_block = False
            if current_block:
                code_blocks.append('\n'.join(current_block))
        elif in_code_block:
            current_block.append(line)
    
    return code_blocks

def save_implementation(code: str, filename: str = "jwt_implementation.py"):
    """Save the JWT implementation to a file."""
    with open(filename, 'w') as f:
        f.write(code)
    print(f"\nImplementation saved to {filename}")

def interactive_jwt_session():
    """Run an interactive session to build JWT authentication system."""
    print("=" * 60)
    print("JWT Authentication System Builder")
    print("Powered by Claude AI")
    print("=" * 60)
    print("\nThis tool will help you build a comprehensive JWT authentication system.")
    print("Type 'quit' to exit, 'save' to save the implementation, or 'clear' to reset.\n")
    
    # Initial request to get the base implementation
    print("Initializing JWT implementation...")
    initial_response = chat_with_claude(
        "Please provide a complete Python implementation for JWT authentication using PyJWT library. "
        "Include: token generation, signing (HS256 and RS256), decoding, validation, claims handling, "
        "signature verification, refresh tokens, and comprehensive error handling. "
        "Make it production-ready with security best practices."
    )
    
    print("\nClaude's Initial Implementation:")
    print("-" * 40)
    print(initial_response)
    
    # Extract and potentially save the initial code
    code_blocks = extract_code_blocks(initial_response)
    if code_blocks:
        print(f"\n[Found {len(code_blocks)} code block(s) in the response]")
    
    # Continue the conversation
    while True:
        print("\n" + "=" * 40)
        user_input = input("Your question or request: ").strip()
        
        if not user_input:
            continue
        
        if user_input.lower() == 'quit':
            print("Exiting JWT builder. Goodbye!")
            break
        
        if user_input.lower() == 'clear':
            conversation_history.clear()
            print("Conversation history cleared.")
            continue
        
        if user_input.lower() == 'save':
            # Collect all code blocks from conversation
            all_code = []
            for msg in conversation_history:
                if msg['role'] == 'assistant':
                    blocks = extract_code_blocks(msg['content'])
                    all_code.extend(blocks)
            
            if all_code:
                # Combine all code blocks
                combined_code = "\n\n# " + "=" * 50 + "\n\n".join(all_code)
                save_implementation(combined_code)
            else:
                print("No code blocks found to save.")
            continue
        
        if user_input.lower() == 'history':
            print("\nConversation History:")
            for i, msg in enumerate(conversation_history):
                print(f"\n[{i+1}] {msg['role'].upper()}:")
                print(msg['content'][:200] + "..." if len(msg['content']) > 200 else msg['content'])
            continue
        
        # Send message to Claude
        print("\nClaude's Response:")
        print("-" * 40)
        response = chat_with_claude(user_input)
        print(response)
        
        # Check for code blocks
        code_blocks = extract_code_blocks(response)
        if code_blocks:
            print(f"\n[Found {len(code_blocks)} code block(s) in the response]")
            save_prompt = input("Would you like to save this code? (y/n): ").strip().lower()
            if save_prompt == 'y':
                filename = input("Enter filename (default: jwt_implementation.py): ").strip()
                if not filename:
                    filename = "jwt_implementation.py"
                save_implementation('\n\n'.join(code_blocks), filename)

def demonstrate_jwt_concepts():
    """Demonstrate various JWT concepts through conversation."""
    print("\n" + "=" * 60)
    print("JWT Concepts Demonstration")
    print("=" * 60)
    
    # List of JWT topics to explore
    topics = [
        "Show me how to create a JWT token with custom claims including user_id, role, and permissions",
        "How do I implement token expiration and handle expired token errors?",
        "Explain how to use RS256 algorithm with RSA key pairs for JWT signing",
        "How can I implement a token blacklist to handle logout functionality?",
        "Show me how to validate JWT tokens in a Flask/FastAPI middleware",
    ]
    
    print("\nExploring JWT concepts with Claude...\n")
    
    for i, topic in enumerate(topics, 1):
        print(f"\n[Topic {i}/{len(topics)}]: {topic}")
        print("-" * 40)
        
        response = chat_with_claude(topic)
        print(response[:500] + "..." if len(response) > 500 else response)
        
        # Small delay to avoid rate limiting
        time.sleep(1)
    
    print("\n" + "=" * 60)
    print("Demonstration complete!")
    print("=" * 60)

def main():
    """Main function to run the JWT authentication builder."""
    print("\nJWT Authentication System with Claude AI")
    print("=========================================\n")
    
    print("Choose mode:")
    print("1. Interactive Mode - Build JWT system step by step")
    print("2. Demonstration Mode - See various JWT concepts")
    print("3. Quick Implementation - Get complete JWT implementation")
    
    choice = input("\nEnter choice (1-3): ").strip()
    
    if choice == '1':
        interactive_jwt_session()
    
    elif choice == '2':
        demonstrate_jwt_concepts()
    
    elif choice == '3':
        print("\nGenerating complete JWT implementation...")
        response = chat_with_claude(
            "Generate a complete, production-ready Python JWT authentication system with: "
            "1) JWTManager class with generate_token, decode_token, refresh_token methods "
            "2) Support for HS256 and RS256 algorithms "
            "3) Complete claims validation (exp, nbf, iss, aud) "
            "4) Token blacklisting with Redis-like interface "
            "5) Comprehensive error handling "
            "6) Usage examples "
            "Make it a single, complete Python file."
        )
        
        print("\nComplete JWT Implementation:")
        print("=" * 60)
        print(response)
        
        code_blocks = extract_code_blocks(response)
        if code_blocks:
            save_implementation('\n\n'.join(code_blocks))
    
    else:
        print("Invalid choice. Running interactive mode...")
        interactive_jwt_session()

if __name__ == "__main__":
    main()
