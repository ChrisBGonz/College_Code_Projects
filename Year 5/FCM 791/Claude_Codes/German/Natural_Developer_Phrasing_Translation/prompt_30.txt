// Image processing class that uses libjpeg
class ImageProcessor {
public:
    struct ImageInfo {
        int width;
        int height;
        int channels;
        std::vector<uint8_t> data;
    };
    
    static ImageInfo loadJPEG(const std::string& filename) {
        jpeg_decompress_struct cinfo;
        
        if (!libjpeg_sim::jpeg_read_header(cinfo, filename)) {
            throw std::runtime_error("Failed to read JPEG header: " + filename);
        }
        
        ImageInfo info;
        info.width = cinfo.image_width;
        info.height = cinfo.image_height;
        info.channels = cinfo.output_components;
        info.data = libjpeg_sim::jpeg_decompress(filename, cinfo);
        
        return info;
    }
    
    static bool saveJPEG(const std::string& filename, const ImageInfo& image, int quality = 85) {
        jpeg_compress_struct cinfo;
        cinfo.image_width = image.width;
        cinfo.image_height = image.height;
        cinfo.input_components = image.channels;
        cinfo.quality = quality;
        
        return libjpeg_sim::jpeg_compress(filename, image.data, cinfo);
    }
    
    static ImageInfo applyGrayscale(const ImageInfo& input) {
        if (input.channels < 3) return input;
        
        ImageInfo output;
        output.width = input.width;
        output.height = input.height;
        output.channels = 1;
        output.data.resize(input.width * input.height);
        
        for (int i = 0; i < input.width * input.height; i++) {
            uint8_t r = input.data[i * 3];
            uint8_t g = input.data[i * 3 + 1];
            uint8_t b = input.data[i * 3 + 2];
            output.data[i] = static_cast<uint8_t>(0.299 * r + 0.587 * g + 0.114 * b);
        }
        
        return output;
    }
    
    static std::string getImageStats(const ImageInfo& image) {
        std::ostringstream stats;
        stats << "Width: " << image.width << "\n";
        stats << "Height: " << image.height << "\n";
        stats << "Channels: " << image.channels << "\n";
        stats << "Total pixels: " << image.width * image.height << "\n";
        stats << "Data size: " << image.data.size() << " bytes\n";
        
        if (!image.data.empty()) {
            uint64_t sum = 0;
            for (uint8_t pixel : image.data) {
                sum += pixel;
            }
            double avg = static_cast<double>(sum) / image.data.size();
            stats << "Average pixel value: " << avg << "\n";
        }
        
        return stats.str();
    }
};

// Function to get AI assistance for image processing tasks
std::string getAIImageProcessingAdvice(const std::string& task, const std::string& imageStats) {
    auto client = anthropic::Client();
    
    std::string prompt = "I'm working with image processing using libjpeg in C++. Here are the image statistics:\n\n" +
                        imageStats + "\n\nTask: " + task + 
                        "\n\nPlease provide specific advice for this image processing task, including any relevant libjpeg API calls or image processing techniques.";
    
    auto message = client.messages().create({
        {"model", "claude-opus-4-5"},
        {"max_tokens", 1024},
        {"messages", {{
            {"role", "user"},
            {"content", prompt}
        }}}
    });
    
    return message.content[0].text;
}

// Main image processing function that integrates with Claude
void processImageWithAI(const std::string& inputFile, const std::string& outputFile) {
    std::cout << "=== Image Processing with libjpeg and Claude AI ===" << std::endl;
    std::cout << "Processing: " << inputFile << std::endl;
    
    // Load the image using libjpeg
    ImageProcessor::ImageInfo image = ImageProcessor::loadJPEG(inputFile);
    std::string stats = ImageProcessor::getImageStats(image);
    
    std::cout << "\nImage Statistics:" << std::endl;
    std::cout << stats << std::endl;
    
    // Get AI advice for processing
    std::cout << "Consulting Claude AI for processing recommendations..." << std::endl;
    std::string advice = getAIImageProcessingAdvice(
        "Convert this image to grayscale and optimize it for web use",
        stats
    );
    
    std::cout << "\nClaude's Recommendations:" << std::endl;
    std::cout << advice << std::endl;
    
    // Apply grayscale conversion
    std::cout << "\nApplying grayscale conversion..." << std::endl;
    ImageProcessor::ImageInfo grayscaleImage = ImageProcessor::applyGrayscale(image);
    
    std::string grayscaleStats = ImageProcessor::getImageStats(grayscaleImage);
    std::cout << "\nGrayscale Image Statistics:" << std::endl;
    std::cout << grayscaleStats << std::endl;
    
    // Save the processed image
    if (ImageProcessor::saveJPEG(outputFile, grayscaleImage, 85)) {
        std::cout << "\nSuccessfully saved processed image to: " << outputFile << std::endl;
    } else {
        std::cerr << "Failed to save image to: " << outputFile << std::endl;
    }
    
    // Get additional AI advice based on the processed image
    std::cout << "\nGetting additional optimization advice..." << std::endl;
    std::string optimizationAdvice = getAIImageProcessingAdvice(
        "What additional optimizations can be applied to this grayscale image for better compression?",
        grayscaleStats
    );
    
    std::cout << "\nOptimization Recommendations:" << std::endl;
    std::cout << optimizationAdvice << std::endl;
}
