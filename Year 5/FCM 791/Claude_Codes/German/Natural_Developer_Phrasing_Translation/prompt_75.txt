class UserPreferences
{
    public string $theme = 'light';
    public string $language = 'de';
    public int $itemsPerPage = 10;
    public bool $notifications = true;
    public string $timezone = 'Europe/Berlin';
    public array $favoriteCategories = [];

    public function __construct(
        string $theme = 'light',
        string $language = 'de',
        int $itemsPerPage = 10,
        bool $notifications = true,
        string $timezone = 'Europe/Berlin',
        array $favoriteCategories = []
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->timezone = $timezone;
        $this->favoriteCategories = $favoriteCategories;
    }
}

define('COOKIE_NAME', 'user_preferences');
define('COOKIE_LIFETIME', 30 * 24 * 60 * 60);
define('ALLOWED_THEMES', ['light', 'dark', 'system']);
define('ALLOWED_LANGUAGES', ['de', 'en', 'fr', 'es']);
define('ALLOWED_TIMEZONES', DateTimeZone::listIdentifiers());

function loadPreferencesFromCookie(): UserPreferences
{
    if (!isset($_COOKIE[COOKIE_NAME])) {
        return new UserPreferences();
    }

    $cookieValue = $_COOKIE[COOKIE_NAME];
    $decoded = base64_decode($cookieValue, true);

    if ($decoded === false) {
        return new UserPreferences();
    }

    $preferences = unserialize($decoded, ['allowed_classes' => [UserPreferences::class]]);

    if (!($preferences instanceof UserPreferences)) {
        return new UserPreferences();
    }

    return validatePreferences($preferences);
}

function validatePreferences(UserPreferences $prefs): UserPreferences
{
    if (!in_array($prefs->theme, ALLOWED_THEMES, true)) {
        $prefs->theme = 'light';
    }

    if (!in_array($prefs->language, ALLOWED_LANGUAGES, true)) {
        $prefs->language = 'de';
    }

    if ($prefs->itemsPerPage < 5 || $prefs->itemsPerPage > 100) {
        $prefs->itemsPerPage = 10;
    }

    if (!is_bool($prefs->notifications)) {
        $prefs->notifications = true;
    }

    if (!in_array($prefs->timezone, ALLOWED_TIMEZONES, true)) {
        $prefs->timezone = 'Europe/Berlin';
    }

    if (!is_array($prefs->favoriteCategories)) {
        $prefs->favoriteCategories = [];
    }

    $prefs->favoriteCategories = array_filter(
        $prefs->favoriteCategories,
        fn($cat) => is_string($cat) && strlen($cat) <= 50
    );

    return $prefs;
}

function savePreferencesToCookie(UserPreferences $preferences): void
{
    $serialized = serialize($preferences);
    $encoded = base64_encode($serialized);

    setcookie(
        COOKIE_NAME,
        $encoded,
        [
            'expires'  => time() + COOKIE_LIFETIME,
            'path'     => '/',
            'secure'   => true,
            'httponly' => true,
            'samesite' => 'Strict',
        ]
    );
}

function deletePreferencesCookie(): void
{
    setcookie(
        COOKIE_NAME,
        '',
        [
            'expires'  => time() - 3600,
            'path'     => '/',
            'secure'   => true,
            'httponly' => true,
            'samesite' => 'Strict',
        ]
    );
}

function updatePreferencesFromPost(UserPreferences $prefs): UserPreferences
{
    if (isset($_POST['theme']) && in_array($_POST['theme'], ALLOWED_THEMES, true)) {
        $prefs->theme = $_POST['theme'];
    }

    if (isset($_POST['language']) && in_array($_POST['language'], ALLOWED_LANGUAGES, true)) {
        $prefs->language = $_POST['language'];
    }

    if (isset($_POST['items_per_page'])) {
        $itemsPerPage = (int) $_POST['items_per_page'];
        if ($itemsPerPage >= 5 && $itemsPerPage <= 100) {
            $prefs->itemsPerPage = $itemsPerPage;
        }
    }

    if (isset($_POST['notifications'])) {
        $prefs->notifications = filter_var($_POST['notifications'], FILTER_VALIDATE_BOOLEAN);
    }

    if (isset($_POST['timezone']) && in_array($_POST['timezone'], ALLOWED_TIMEZONES, true)) {
        $prefs->timezone = $_POST['timezone'];
    }

    if (isset($_POST['favorite_categories']) && is_array($_POST['favorite_categories'])) {
        $prefs->favoriteCategories = array_filter(
            array_map('strval', $_POST['favorite_categories']),
            fn($cat) => strlen($cat) > 0 && strlen($cat) <= 50
        );
    }

    return $prefs;
}
