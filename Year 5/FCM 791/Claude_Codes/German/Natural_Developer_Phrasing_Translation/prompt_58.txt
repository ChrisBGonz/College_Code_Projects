const STORAGE_KEY = 'multiStepFormData';

const formSteps = [
  {
    id: 'step-1',
    title: 'Personal Information',
    fields: [
      { name: 'firstName', label: 'First Name', type: 'text', required: true },
      { name: 'lastName', label: 'Last Name', type: 'text', required: true },
      { name: 'birthDate', label: 'Date of Birth', type: 'date', required: true },
    ],
  },
  {
    id: 'step-2',
    title: 'Contact Details',
    fields: [
      { name: 'email', label: 'Email Address', type: 'email', required: true },
      { name: 'phone', label: 'Phone Number', type: 'tel', required: false },
      { name: 'address', label: 'Street Address', type: 'text', required: true },
    ],
  },
  {
    id: 'step-3',
    title: 'Account Setup',
    fields: [
      { name: 'username', label: 'Username', type: 'text', required: true },
      { name: 'password', label: 'Password', type: 'password', required: true },
      { name: 'confirmPassword', label: 'Confirm Password', type: 'password', required: true },
    ],
  },
  {
    id: 'step-4',
    title: 'Preferences',
    fields: [
      { name: 'newsletter', label: 'Subscribe to Newsletter', type: 'checkbox', required: false },
      { name: 'language', label: 'Preferred Language', type: 'select', required: true, options: ['English', 'German', 'French', 'Spanish'] },
      { name: 'bio', label: 'Short Bio', type: 'textarea', required: false },
    ],
  },
];

class MultiStepForm {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    if (!this.container) {
      throw new Error(`Container with id "${containerId}" not found.`);
    }

    this.currentStep = 0;
    this.formData = {};
    this.errors = {};

    this.init();
  }

  init() {
    this.loadFromStorage();
    this.render();
    this.attachGlobalListeners();
  }

  loadFromStorage() {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        const parsed = JSON.parse(stored);
        this.formData = parsed.formData || {};
        this.currentStep = typeof parsed.currentStep === 'number'
          ? Math.min(parsed.currentStep, formSteps.length - 1)
          : 0;
      }
    } catch (e) {
      console.warn('Failed to load form data from localStorage:', e);
      this.formData = {};
      this.currentStep = 0;
    }
  }

  saveToStorage() {
    try {
      const payload = {
        formData: this.formData,
        currentStep: this.currentStep,
        savedAt: new Date().toISOString(),
      };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
    } catch (e) {
      console.warn('Failed to save form data to localStorage:', e);
    }
  }

  clearStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
    } catch (e) {
      console.warn('Failed to clear form data from localStorage:', e);
    }
  }

  collectStepData(stepIndex) {
    const step = formSteps[stepIndex];
    const stepEl = this.container.querySelector(`#${step.id}`);
    if (!stepEl) return;

    step.fields.forEach((field) => {
      const el = stepEl.querySelector(`[name="${field.name}"]`);
      if (!el) return;

      if (field.type === 'checkbox') {
        this.formData[field.name] = el.checked;
      } else {
        this.formData[field.name] = el.value;
      }
    });
  }

  validateStep(stepIndex) {
    const step = formSteps[stepIndex];
    const stepEl = this.container.querySelector(`#${step.id}`);
    const newErrors = {};

    step.fields.forEach((field) => {
      const el = stepEl ? stepEl.querySelector(`[name="${field.name}"]`) : null;
      const value = el
        ? field.type === 'checkbox'
          ? el.checked
          : el.value.trim()
        : this.formData[field.name];

      if (field.required) {
        if (field.type === 'checkbox') {
          // checkboxes are optional by nature unless explicitly required
        } else if (!value) {
          newErrors[field.name] = `${field.label} is required.`;
        }
      }

      if (field.type === 'email' && value && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
        newErrors[field.name] = 'Please enter a valid email address.';
      }

      if (field.name === 'confirmPassword') {
        const password = stepEl
          ? stepEl.querySelector('[name="password"]')?.value
          : this.formData['password'];
        if (value !== password) {
          newErrors[field.name] = 'Passwords do not match.';
        }
      }

      if (field.type === 'tel' && value && !/^\+?[\d\s\-().]{7,20}$/.test(value)) {
        newErrors[field.name] = 'Please enter a valid phone number.';
      }
    });

    this.errors = newErrors;
    return Object.keys(newErrors).length === 0;
  }

  prefillStep(stepIndex) {
    const step = formSteps[stepIndex];
    const stepEl = this.container.querySelector(`#${step.id}`);
    if (!stepEl) return;

    step.fields.forEach((field) => {
      const el = stepEl.querySelector(`[name="${field.name}"]`);
      if (!el) return;

      const savedValue = this.formData[field.name];
      if (savedValue === undefined || savedValue === null) return;

      if (field.type === 'checkbox') {
        el.checked = Boolean(savedValue);
      } else {
        el.value = savedValue;
      }
    });
  }

  renderProgressBar() {
    const totalSteps = formSteps.length;
    const progressPercent = ((this.currentStep) / (totalSteps - 1)) * 100;

    return `
      <div class="msf-progress-container" role="progressbar" aria-valuenow="${this.currentStep + 1}" aria-valuemin="1" aria-valuemax="${totalSteps}">
        <div class="msf-progress-steps">
          ${formSteps.map((step, index) => `
            <div class="msf-progress-step ${index < this.currentStep ? 'completed' : ''} ${index === this.currentStep ? 'active' : ''}">
              <div class="msf-step-circle">
                ${index < this.currentStep ? '&#10003;' : index + 1}
              </div>
              <span class="msf-step-label">${step.title}</span>
            </div>
          `).join('')}
        </div>
        <div class="msf-progress-bar-track">
          <div class="msf-progress-bar-fill" style="width: ${progressPercent}%"></div>
        </div>
      </div>
    `;
  }

  renderField(field) {
    const errorMsg = this.errors[field.name];
    const errorClass = errorMsg ? 'msf-field--error' : '';
    const savedValue = this.formData[field.name] ?? '';

    let inputHtml = '';

    switch (field.type) {
      case 'textarea':
        inputHtml = `
          <textarea
            name="${field.name}"
            id="field-${field.name}"
            class="msf-input msf-textarea ${errorClass}"
            ${field.required ? 'required' : ''}
            aria-describedby="${errorMsg ? `error-${field.name}` : ''}"
            aria-invalid="${errorMsg ? 'true' : 'false'}"
          >${this.escapeHtml(String(savedValue))}</textarea>
        `;
        break;

      case 'select':
        inputHtml = `
          <select
            name="${field.name}"
            id="field-${field.name}"
            class="msf-input msf-select ${errorClass}"
            ${field.required ? 'required' : ''}
            aria-describedby="${errorMsg ? `error-${field.name}` : ''}"
            aria-invalid="${errorMsg ? 'true' : 'false'}"
          >
            <option value="">-- Select --</option>
            ${(field.options || []).map(opt => `
              <option value="${this.escapeHtml(opt)}" ${savedValue === opt ? 'selected' : ''}>
                ${this.escapeHtml(opt)}
              </option>
            `).join('')}
          </select>
        `;
        break;

      case 'checkbox':
        inputHtml = `
          <div class="msf-checkbox-wrapper">
            <input
              type="checkbox"
              name="${field.name}"
              id="field-${field.name}"
              class="msf-checkbox ${errorClass}"
              ${savedValue ? 'checked' : ''}
              ${field.required ? 'required' : ''}
              aria-describedby="${errorMsg ? `error-${field.name}` : ''}"
              aria-invalid="${errorMsg ? 'true' : 'false'}"
            />
            <label for="field-${field.name}" class="msf-checkbox-label">${field.label}</label>
          </div>
        `;
        return `
          <div class="msf-field-group">
            ${inputHtml}
            ${errorMsg ? `<span class="msf-error-msg" id="error-${field.name}" role="alert">${errorMsg}</span>` : ''}
          </div>
        `;

      default:
        inputHtml = `
          <input
            type="${field.type}"
            name="${field.name}"
            id="field-${field.name}"
            class="msf-input ${errorClass}"
            value="${field.type !== 'password' ? this.escapeHtml(String(savedValue)) : ''}"
            ${field.required ? 'required' : ''}
            aria-describedby="${errorMsg ? `error-${field.name}` : ''}"
            aria-invalid="${errorMsg ? 'true' : 'false'}"
            autocomplete="${this.getAutocomplete(field.name)}"
          />
        `;
    }

    return `
      <div class="msf-field-group">
        <label for="field-${field.name}" class="msf-label">
          ${field.label}${field.required ? ' <span class="msf-required" aria-hidden="true">*</span>' : ''}
        </label>
        ${inputHtml}
        ${errorMsg ? `<span class="msf-error-msg" id="error-${field.name}" role="alert">${errorMsg}</span>` : ''}
      </div>
    `;
  }

  getAutocomplete(fieldName) {
    const map = {
      firstName: 'given-name',
      lastName: 'family-name',
      email: 'email',
      phone: 'tel',
      address: 'street-address',
      username: 'username',
      password: 'new-password',
      confirmPassword: 'new-password',
    };
    return map[fieldName] || 'off';
  }

  renderStep(stepIndex) {
    const step = formSteps[stepIndex];
    const isLastStep = stepIndex === formSteps.length - 1;

    return `
      <div id="${step.id}" class="msf-step" role="group" aria-labelledby="step-title-${stepIndex}">
        <h2 id="step-title-${stepIndex}" class="msf-step-title">${step.title}</h2>
        <div class="msf-fields">
          ${step.fields.map(field => this.renderField(field)).join('')}
        </div>
        <div class="msf-navigation">
          ${stepIndex > 0 ? `<button type="button" class="msf-btn msf-btn--secondary" data-action="prev">&#8592; Back</button>` : '<div></div>'}
          ${isLastStep
            ? `<button type="button" class="msf-btn msf-btn--primary" data-action="submit">Submit Registration</button>`
            : `<button type="button" class="msf-btn msf-btn--primary" data-action="next">Next &#8594;</button>`
          }
        </div>
      </div>
    `;
  }

  renderSummary() {
    const rows = Object.entries(this.formData)
      .filter(([key]) => key !== 'password' && key !== 'confirmPassword')
      .map(([key, value]) => {
        const label = this.getFieldLabel(key);
        const displayValue = typeof value === 'boolean' ? (value ? 'Yes' : 'No') : this.escapeHtml(String(value));
        return `<tr><th scope="row">${label}</th><td>${displayValue}</td></tr>`;
      })
      .join('');

    return `
      <div class="msf-summary">
        <h3 class="msf-summary-title">Review Your Information</h3>
        <table class="msf-summary-table">
          <tbody>${rows}</tbody>
        </table>
      </div>
    `;
  }

  getFieldLabel(fieldName) {
    for (const step of formSteps) {
      const field = step.fields.find(f => f.name === fieldName);
      if (field) return field.label;
    }
    return fieldName;
  }

  renderSuccessMessage() {
    return `
      <div class="msf-success" role="alert" aria-live="polite">
        <div class="msf-success-icon">&#10003;</div>
        <h2 class="msf-success-title">Registration Successful!</h2>
        <p class="msf-success-text">Thank you for registering. Your account has been created successfully.</p>
        <button type="button" class="msf-btn msf-btn--primary" data-action="reset">Start Over</button>
      </div>
    `;
  }

  render() {
    const isLastStep = this.currentStep === formSteps.length - 1;

    this.container.innerHTML = `
      <div class="msf-wrapper" aria-label="Multi-Step Registration Form">
        <h1 class="msf-main-title">Create Your Account</h1>
        <p class="msf-step-counter">Step ${this.currentStep + 1} of ${formSteps.length}</p>
        ${this.renderProgressBar()}
        <form class="msf-form" novalidate>
          ${this.renderStep(this.currentStep)}
          ${isLastStep ? this.renderSummary() : ''}
        </form>
        ${this.formData._savedAt ? `<p class="msf-autosave-notice">&#128190; Progress auto-saved</p>` : ''}
      </div>
    `;

    this.prefillStep(this.currentStep);
    this.attachStepListeners();
  }

  attachStepListeners() {
    const form = this.container.querySelector('.msf-form');
    if (!form) return;

    const nextBtn = form.querySelector('[data-action="next"]');
    const prevBtn = form.querySelector('[data-action="prev"]');
    const submitBtn = form.querySelector('[data-action="submit"]');
    const resetBtn = form.querySelector('[data-action="reset"]');

    if (nextBtn) {
      nextBtn.addEventListener('click', () => this.handleNext());
    }

    if (prevBtn) {
      prevBtn.addEventListener('click', () => this.handlePrev());
    }

    if (submitBtn) {
      submitBtn.addEventListener('click', () => this.handleSubmit());
    }

    if (resetBtn) {
      resetBtn.addEventListener('click', () => this.handleReset());
    }

    // Auto-save on input change
    const inputs = form.querySelectorAll('input, select, textarea');
    inputs.forEach((input) => {
      const eventType = input.type === 'checkbox' ? 'change' : 'input';
      input.addEventListener(eventType, () => {
        this.collectStepData(this.currentStep);
        this.formData._savedAt = new Date().toISOString();
        this.saveToStorage();
        this.updateAutosaveNotice();
      });
    });
  }

  attachGlobalListeners() {
    // Handle browser back/forward navigation
    window.addEventListener('popstate', () => {
      this.loadFromStorage();
      this.render();
    });
  }

  updateAutosaveNotice() {
    let notice = this.container.querySelector('.msf-autosave-notice');
    if (!notice) {
      notice = document.createElement('p');
      notice.className = 'msf-autosave-notice';
      this.container.querySelector('.msf-wrapper')?.appendChild(notice);
    }
    notice.textContent = `\u{1F4BE} Progress auto-saved at ${new Date().toLocaleTimeString()}`;
    notice.classList.add('msf-autosave-notice--visible');
    clearTimeout(this._saveNoticeTimer);
    this._saveNoticeTimer = setTimeout(() => {
      notice.classList.remove('msf-autosave-notice--visible');
    }, 3000);
  }

  handleNext() {
    this.collectStepData(this.currentStep);

    if (!this.validateStep(this.currentStep)) {
      this.renderStepErrors();
      return;
    }

    this.errors = {};
    this.currentStep += 1;
    this.saveToStorage();
    this.render();
    this.scrollToTop();
  }

  handlePrev() {
    this.collectStepData(this.currentStep);
    this.errors = {};
    this.currentStep -= 1;
    this.saveToStorage();
    this.render();
    this.scrollToTop();
  }

  async handleSubmit() {
    this.collectStepData(this.currentStep);

    // Validate all steps
    let allValid = true;
    const allErrors = {};

    for (let i = 0; i < formSteps.length; i++) {
      if (!this.validateStep(i)) {
        allValid = false;
        Object.assign(allErrors, this.errors);
      }
    }

    if (!allValid) {
      this.errors = allErrors;
      this.renderStepErrors();
      return;
    }

    const submitBtn = this.container.querySelector('[data-action="submit"]');
    if (submitBtn) {
      submitBtn.disabled = true;
      submitBtn.textContent = 'Submitting...';
    }

    try {
      await this.submitFormData();
      this.clearStorage();
      this.showSuccess();
    } catch (error) {
      console.error('Submission failed:', error);
      if (submitBtn) {
        submitBtn.disabled = false;
        submitBtn.textContent = 'Submit Registration';
      }
      this.showSubmissionError(error.message);
    }
  }

  async submitFormData() {
    // Simulate API call - replace with actual endpoint
    const payload = { ...this.formData };
    delete payload._savedAt;
    delete payload.confirmPassword; // Don't send confirmPassword to server

    console.log('Submitting form data:', payload);

    // Simulated network delay
    await new Promise((resolve, reject) => {
      setTimeout(() => {
        // Simulate 95% success rate for demo
        if (Math.random() > 0.05) {
          resolve({ success: true, userId: Math.floor(Math.random() * 10000) });
        } else {
          reject(new Error('Server error. Please try again.'));
        }
      }, 1500);
    });

    /*
    // Real API call example:
    const response = await fetch('/api/register', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });

    if (!response.ok) {
      const errorData = await response.json().catch(() => ({}));
      throw new Error(errorData.message || `HTTP error ${response.status}`);
    }

    return response.json();
    */
  }

  showSuccess() {
    this.container.innerHTML = `
      <div class="msf-wrapper">
        ${this.renderSuccessMessage()}
      </div>
    `;

    const resetBtn = this.container.querySelector('[data-action="reset"]');
    if (resetBtn) {
      resetBtn.addEventListener('click', () => this.handleReset());
    }
  }

  showSubmissionError(message) {
    const existingError = this.container.querySelector('.msf-submission-error');
    if (existingError) existingError.remove();

    const errorEl = document.createElement('div');
    errorEl.className = 'msf-submission-error';
    errorEl.setAttribute('role', 'alert');
    errorEl.textContent = `Submission failed: ${message}`;

    const form = this.container.querySelector('.msf-form');
    if (form) {
      form.insertAdjacentElement('beforebegin', errorEl);
    }
  }

  handleReset() {
    this.clearStorage();
    this.formData = {};
    this.currentStep = 0;
    this.errors = {};
    this.render();
  }

  renderStepErrors() {
    const step = formSteps[this.currentStep];
    const stepEl = this.container.querySelector(`#${step.id}`);
    if (!stepEl) return;

    // Clear existing errors
    stepEl.querySelectorAll('.msf-error-msg').forEach(el => el.remove());
    stepEl.querySelectorAll('.msf-field--error').forEach(el => {
      el.classList.remove('msf-field--error');
    });

    // Render new errors
    Object.entries(this.errors).forEach(([fieldName, errorMsg]) => {
      const input = stepEl.querySelector(`[name="${fieldName}"]`);
      if (!input) return;

      input.classList.add('msf-field--error');
      input.setAttribute('aria-invalid', 'true');
      input.setAttribute('aria-describedby', `error-${fieldName}`);

      const errorEl = document.createElement('span');
      errorEl.className = 'msf-error-msg';
      errorEl.id = `error-${fieldName}`;
      errorEl.setAttribute('role', 'alert');
      errorEl.textContent = errorMsg;

      const fieldGroup = input.closest('.msf-field-group');
      if (fieldGroup) {
        fieldGroup.appendChild(errorEl);
      }
    });

    // Focus first error field
    const firstErrorField = Object.keys(this.errors)[0];
    if (firstErrorField) {
      const firstInput = stepEl.querySelector(`[name="${firstErrorField}"]`);
      if (firstInput) {
        firstInput.focus();
      }
    }
  }

  scrollToTop() {
    this.container.scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  escapeHtml(str) {
    const div = document.createElement('div');
    div.appendChild(document.createTextNode(str));
    return div.innerHTML;
  }
}
