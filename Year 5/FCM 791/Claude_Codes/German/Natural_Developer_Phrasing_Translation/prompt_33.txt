
```javascript
import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateSessionTokenWithClaude(): Promise<string> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a JavaScript function for a Node.js backend that creates a random session token with exactly 16 characters. 
        
        Requirements:
        - The function should be named generateSessionToken
        - It should return a string of exactly 16 characters
        - Use cryptographically secure random generation (crypto module)
        - The token should contain alphanumeric characters (a-z, A-Z, 0-9)
        - Return ONLY the function code, no explanations or markdown
        - The function should be a simple synchronous function`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  const codeMatch = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/);
  const functionCode = codeMatch ? codeMatch[1] : responseText;

  return functionCode.trim();
}

async function main() {
  console.log(
    "Requesting Claude to generate a session token function...\n"
  );

  const generatedCode = await generateSessionTokenWithClaude();

  console.log("Generated function code:");
  console.log("------------------------");
  console.log(generatedCode);
  console.log("------------------------\n");

  console.log("Testing the generated function...");

  const testFunction = new Function(
    "require",
    `
    ${generatedCode}
    return generateSessionToken();
  `
  );

  const token = testFunction(require);
  console.log(`Generated token: ${token}`);
  console.log(`Token length: ${token.length}`);
  console.log(`Is 16 characters: ${token.length === 16}`);

  console.log("\nGenerating 5 sample tokens:");
  for (let i = 0; i < 5; i++) {
    const sampleToken = testFunction(require);
    console.log(`Token ${i + 1}: ${sampleToken}`);
  }
}

main().catch(console.error);
```
