from flask import Flask, request, session, redirect, url_for, jsonify
from anthropic import Anthropic
import secrets
import hashlib

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Simple user database (in production, use a real database)
users = {
    "admin": hashlib.sha256("password123".encode()).hexdigest(),
    "user1": hashlib.sha256("mypassword".encode()).hexdigest(),
}

# Store conversation history per session
conversation_histories = {}

def get_session_id():
    """Get or create a session ID for conversation tracking."""
    if 'session_id' not in session:
        session['session_id'] = secrets.token_hex(16)
    return session['session_id']

def chat_with_claude(session_id: str, user_message: str) -> str:
    """Send a message to Claude and get a response, maintaining conversation history."""
    if session_id not in conversation_histories:
        conversation_histories[session_id] = []
    
    conversation_histories[session_id].append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a helpful assistant integrated into a secure web application. Help users with their questions and tasks. Be friendly and professional.",
        messages=conversation_histories[session_id]
    )
    
    assistant_message = response.content[0].text
    
    conversation_histories[session_id].append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/')
def index():
    """Main page - redirect to login if not authenticated."""
    if 'auth_token' not in session:
        return redirect(url_for('login'))
    return redirect(url_for('chat'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Login endpoint that sets auth_token cookie on success."""
    if request.method == 'GET':
        return '''
        <!DOCTYPE html>
        <html>
        <head>
            <title>Login</title>
            <style>
                body { font-family: Arial, sans-serif; max-width: 400px; margin: 100px auto; padding: 20px; }
                input { width: 100%; padding: 10px; margin: 10px 0; box-sizing: border-box; }
                button { width: 100%; padding: 10px; background: #007bff; color: white; border: none; cursor: pointer; }
                button:hover { background: #0056b3; }
                .error { color: red; }
                .info { color: #666; font-size: 0.9em; margin-top: 20px; }
            </style>
        </head>
        <body>
            <h2>Login</h2>
            <form method="POST">
                <input type="text" name="username" placeholder="Username" required>
                <input type="password" name="password" placeholder="Password" required>
                <button type="submit">Login</button>
            </form>
            <div class="info">
                <p>Test credentials:</p>
                <p>Username: admin, Password: password123</p>
                <p>Username: user1, Password: mypassword</p>
            </div>
        </body>
        </html>
        '''
    
    username = request.form.get('username', '')
    password = request.form.get('password', '')
    
    password_hash = hashlib.sha256(password.encode()).hexdigest()
    
    if username in users and users[username] == password_hash:
        auth_token = secrets.token_hex(32)
        session['auth_token'] = auth_token
        session['username'] = username
        
        response = redirect(url_for('chat'))
        response.set_cookie(
            'auth_token',
            auth_token,
            httponly=True,
            secure=False,
            samesite='Lax',
            max_age=3600
        )
        return response
    else:
        return '''
        <!DOCTYPE html>
        <html>
        <head>
            <title>Login Failed</title>
            <style>
                body { font-family: Arial, sans-serif; max-width: 400px; margin: 100px auto; padding: 20px; }
                input { width: 100%; padding: 10px; margin: 10px 0; box-sizing: border-box; }
                button { width: 100%; padding: 10px; background: #007bff; color: white; border: none; cursor: pointer; }
                button:hover { background: #0056b3; }
                .error { color: red; }
                .info { color: #666; font-size: 0.9em; margin-top: 20px; }
            </style>
        </head>
        <body>
            <h2>Login</h2>
            <p class="error">Invalid username or password!</p>
            <form method="POST">
                <input type="text" name="username" placeholder="Username" required>
                <input type="password" name="password" placeholder="Password" required>
                <button type="submit">Login</button>
            </form>
            <div class="info">
                <p>Test credentials:</p>
                <p>Username: admin, Password: password123</p>
                <p>Username: user1, Password: mypassword</p>
            </div>
        </body>
        </html>
        ''', 401

@app.route('/logout')
def logout():
    """Logout endpoint that clears the session and auth_token cookie."""
    session_id = session.get('session_id')
    if session_id and session_id in conversation_histories:
        del conversation_histories[session_id]
    
    session.clear()
    
    response = redirect(url_for('login'))
    response.delete_cookie('auth_token')
    return response

@app.route('/chat')
def chat():

    if 'auth_token' not in session:
        return redirect(url_for('login'))
    
    username = session.get('username', 'User')
    
    return f'''

            .assistant-message {{ background: #f5f5f5; }}
            #input-container {{ display: flex; gap: 10px; }}
            #message-input {{ flex: 1; padding: 10px; }}
            button {{ padding: 10px 20px; background: #007bff; color: white; border: none; cursor: pointer; }}
            button:hover {{ background: #0056b3; }}
            .header {{ display: flex; justify-content: space-between; align-items: center; }}
            .logout-btn {{ background: #dc3545; }}
            .logout-btn:hover {{ background: #c82333; }}
        </style>
    </head>

        <div class="header">
            <h2>Chat with Claude - Welcome, {username}!</h2>
            <a href="/logout"><button class="logout-btn">Logout</button></a>
        </div>
        <div id="chat-container"></div>
        <div id="input-container">
            <input type="text" id="message-input" placeholder="Type your message..." onkeypress="handleKeyPress(event)">
            <button onclick="sendMessage()">Send</button>
        </div>
        
        <script>
            async function sendMessage() {{
                const input = document.getElementById('message-input');
                const message = input.value.trim();
                if (!message) return;
                
                const chatContainer = document.getElementById('chat-container');
                
                // Add user message
                chatContainer.innerHTML += `<div class="message user-message"><strong>You:</strong> ${{message}}</div>`;
                input.value = '';
                chatContainer.scrollTop = chatContainer.scrollHeight;
                
                try {{
                    const response = await fetch('/api/chat', {{
                        method: 'POST',
                        headers: {{ 'Content-Type': 'application/json' }},
                        body: JSON.stringify({{ message: message }})
                    }});
                    
                    if (response.status === 401) {{
                        window.location.href = '/login';
                        return;
                    }}
                    
                    const data = await response.json();
                    
                    // Add assistant message
                    chatContainer.innerHTML += `<div class="message assistant-message"><strong>Claude:</strong> ${{data.response}}</div>`;
                    chatContainer.scrollTop = chatContainer.scrollHeight;
                }} catch (error) {{
                    chatContainer.innerHTML += `<div class="message assistant-message"><strong>Error:</strong> Failed to get response</div>`;
                }}
            }}
            
            function handleKeyPress(event) {{
                if (event.key === 'Enter') {{
                    sendMessage();
                }}
            }}
        </script>
    </body>
    </html>
    '''

@app.route('/api/chat', methods=['POST'])
def api_chat():
    """API endpoint for chat - requires authentication."""
    if 'auth_token' not in session:
        return jsonify({'error': 'Unauthorized'}), 401
    
    data = request.get_json()
    if not data or 'message' not in data:
        return jsonify({'error': 'No message provided'}), 400
    
    user_message = data['message']
    session_id = get_session_id()
    
    response = chat_with_claude(session_id, user_message)
    
    return jsonify({'response': response})

@app.route('/api/status')
def api_status():
    """Check authentication status."""
    if 'auth_token' not in session:
        return jsonify({'authenticated': False}), 401
    
    return jsonify({
        'authenticated': True,
        'username': session.get('username'),
        'auth_token': session.get('auth_token')[:8] + '...'
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)
