import com.google.cloud.vertexai.VertexAI;
import com.google.cloud.vertexai.api.GenerateContentResponse;
import com.google.cloud.vertexai.generativeai.GenerativeModel;
import com.google.cloud.vertexai.generativeai.ResponseHandler;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import org.w3c.dom.Document;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import java.io.IOException;
import java.io.StringReader;

public class SecureXMLParser {

    public static Document parseXMLSecurely(String xmlString) throws ParserConfigurationException, SAXException, IOException {
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        
        // Disable DOCTYPE declarations to prevent XXE attacks
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
        
        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
        
        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
        
        // Disable external DTD loading
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
        
        // Disable XInclude processing
        factory.setXIncludeAware(false);
        
        // Disable entity expansion to prevent entity expansion attacks
        factory.setExpandEntityReferences(false);
        
        // Create a secure DocumentBuilder
        DocumentBuilder builder = factory.newDocumentBuilder();
        
        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    public static String generateSecureXMLParserCode() throws IOException {
        String projectId = System.getenv("GOOGLE_CLOUD_PROJECT");
        String location = "us-central1";
        String modelName = "gemini-2.0-flash-001";

        if (projectId == null || projectId.isEmpty()) {
            throw new IllegalStateException("GOOGLE_CLOUD_PROJECT environment variable is not set");
        }

        String prompt = """
                Write a Java function that securely parses an XML string from an untrusted source using DocumentBuilderFactory.
                The function should include hardening against:
                1. XXE (XML External Entity) attacks
                2. DTD (Document Type Definition) attacks
                3. Entity expansion attacks (Billion Laughs attack)
                
                Provide the complete Java function with all necessary security configurations.
                Include comments explaining each security measure.
                """;

        try (VertexAI vertexAI = new VertexAI(projectId, location)) {
            GenerativeModel model = new GenerativeModel(modelName, vertexAI);
            GenerateContentResponse response = model.generateContent(prompt);
            return ResponseHandler.getText(response);
        }
    }

    public static void main(String[] args) {
        System.out.println("=== Secure XML Parser Demo ===\n");
        
        // Test 1: Parse a valid XML string
        System.out.println("Test 1: Parsing a valid XML string");
        String validXml = """
                <?xml version="1.0" encoding="UTF-8"?>
                <root>
                    <element>Hello, World!</element>
                    <number>42</number>
                </root>
                """;
        
        try {
            Document doc = parseXMLSecurely(validXml);
            System.out.println("Successfully parsed valid XML");
            System.out.println("Root element: " + doc.getDocumentElement().getTagName());
            System.out.println();
        } catch (Exception e) {
            System.err.println("Error parsing valid XML: " + e.getMessage());
        }
        
        // Test 2: Attempt to parse XML with XXE attack
        System.out.println("Test 2: Attempting to parse XML with XXE attack (should fail)");
        String xxeXml = """
                <?xml version="1.0" encoding="UTF-8"?>
                <!DOCTYPE foo [
                    <!ENTITY xxe SYSTEM "file:///etc/passwd">
                ]>
                <root>
                    <element>&xxe;</element>
                </root>
                """;
        
        try {
            Document doc = parseXMLSecurely(xxeXml);
            System.out.println("WARNING: XXE attack was not blocked!");
        } catch (Exception e) {
            System.out.println("Successfully blocked XXE attack: " + e.getClass().getSimpleName());
            System.out.println();
        }
        
        // Test 3: Attempt to parse XML with entity expansion attack
        System.out.println("Test 3: Attempting to parse XML with entity expansion attack (should fail)");
        String entityExpansionXml = """
                <?xml version="1.0" encoding="UTF-8"?>
                <!DOCTYPE lolz [
                    <!ENTITY lol "lol">
                    <!ENTITY lol2 "&lol;&lol;&lol;&lol;&lol;&lol;&lol;&lol;&lol;&lol;">
                    <!ENTITY lol3 "&lol2;&lol2;&lol2;&lol2;&lol2;&lol2;&lol2;&lol2;&lol2;&lol2;">
                ]>
                <root>&lol3;</root>
                """;
        
        try {
            Document doc = parseXMLSecurely(entityExpansionXml);
            System.out.println("WARNING: Entity expansion attack was not blocked!");
        } catch (Exception e) {
            System.out.println("Successfully blocked entity expansion attack: " + e.getClass().getSimpleName());
            System.out.println();
        }
        
        // Generate explanation using Gemini
        System.out.println("=== Generating Security Explanation with Gemini ===\n");
        try {
            String explanation = generateSecureXMLParserCode();
            System.out.println("Gemini's explanation of secure XML parsing:");
            System.out.println(explanation);
        } catch (Exception e) {
            System.err.println("Error generating explanation: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
