#!/usr/bin/env python3
"""
Legacy Server Communication with XOR Cipher Encryption
Demonstrates communication with a legacy server using custom TCP protocol
with XOR encryption for payload/frames.
"""

import asyncio

import struct
import time
import random
import threading
from anthropic import Anthropic

# XOR Cipher implementation
def xor_encrypt(data: bytes, key: bytes) -> bytes:
    """Encrypt/decrypt data using XOR cipher with repeating key."""
    if not key:
        return data
    result = bytearray()
    for i, byte in enumerate(data):
        result.append(byte ^ key[i % len(key)])
    return bytes(result)

def xor_decrypt(data: bytes, key: bytes) -> bytes:
    """XOR decryption is the same as encryption."""
    return xor_encrypt(data, key)

# Custom TCP Protocol Frame Structure
class ProtocolFrame:
    """
    Custom protocol frame structure:
    - Magic bytes: 2 bytes (0xDE, 0xAD)
    - Version: 1 byte
    - Message type: 1 byte
    - Sequence number: 4 bytes (big-endian)
    - Payload length: 4 bytes (big-endian)
    - Payload: variable length (XOR encrypted)
    - Checksum: 2 bytes (simple sum)
    """
    
    MAGIC = b'\xDE\xAD'
    VERSION = 1
    HEADER_SIZE = 12  # magic(2) + version(1) + type(1) + seq(4) + length(4)
    
    # Message types
    MSG_HELLO = 0x01
    MSG_DATA = 0x02
    MSG_ACK = 0x03
    MSG_ERROR = 0x04
    MSG_GOODBYE = 0x05
    
    def __init__(self, msg_type: int, payload: bytes, seq_num: int = 0):
        self.msg_type = msg_type
        self.payload = payload
        self.seq_num = seq_num
    
    def encode(self, xor_key: bytes) -> bytes:
        """Encode frame with XOR encryption."""
        encrypted_payload = xor_encrypt(self.payload, xor_key)
        
        header = (
            self.MAGIC +
            bytes([self.VERSION, self.msg_type]) +
            struct.pack('>I', self.seq_num) +
            struct.pack('>I', len(encrypted_payload))
        )
        
        # Simple checksum
        checksum = sum(header + encrypted_payload) & 0xFFFF
        checksum_bytes = struct.pack('>H', checksum)
        
        return header + encrypted_payload + checksum_bytes
    
    @classmethod
    def decode(cls, data: bytes, xor_key: bytes) -> 'ProtocolFrame':
        """Decode frame and decrypt payload."""
        if len(data) < cls.HEADER_SIZE + 2:
            raise ValueError("Frame too short")
        
        # Verify magic bytes
        if data[:2] != cls.MAGIC:
            raise ValueError(f"Invalid magic bytes: {data[:2].hex()}")
        
        version = data[2]
        msg_type = data[3]
        seq_num = struct.unpack('>I', data[4:8])[0]
        payload_length = struct.unpack('>I', data[8:12])[0]
        
        if len(data) < cls.HEADER_SIZE + payload_length + 2:
            raise ValueError("Incomplete frame")
        
        encrypted_payload = data[cls.HEADER_SIZE:cls.HEADER_SIZE + payload_length]
        checksum_bytes = data[cls.HEADER_SIZE + payload_length:cls.HEADER_SIZE + payload_length + 2]
        
        # Verify checksum
        expected_checksum = sum(data[:cls.HEADER_SIZE + payload_length]) & 0xFFFF
        actual_checksum = struct.unpack('>H', checksum_bytes)[0]
        
        if expected_checksum != actual_checksum:
            raise ValueError(f"Checksum mismatch: expected {expected_checksum}, got {actual_checksum}")
        
        # Decrypt payload
        payload = xor_decrypt(encrypted_payload, xor_key)
        
        frame = cls(msg_type, payload, seq_num)
        return frame

# Simulated Legacy Server
class LegacyServer:
    """Simulates a legacy server with custom TCP protocol."""
    
    def __init__(self, host: str = '127.0.0.1', port: int = 9999):
        self.host = host
        self.port = port
        self.xor_key = b'\x42\x13\x37\xAB'  # Fixed XOR key for legacy server
        self.running = False
        self.server_socket = None
        self.seq_counter = 0
    
    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"[Legacy Server] Started on {self.host}:{self.port}")
        
        # Run in background thread
        thread = threading.Thread(target=self._accept_connections, daemon=True)
        thread.start()
    
    def _accept_connections(self):
        """Accept and handle client connections."""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, addr = self.server_socket.accept()
                print(f"[Legacy Server] New connection from {addr}")
                client_thread = threading.Thread(
                    target=self._handle_client,
                    args=(client_socket,),
                    daemon=True
                )
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"[Legacy Server] Accept error: {e}")
    
    def _handle_client(self, client_socket: socket.socket):
        """Handle individual client connection."""
        try:
            while True:
                # Read frame header
                header_data = self._recv_exact(client_socket, ProtocolFrame.HEADER_SIZE)
                if not header_data:
                    break
                
                # Get payload length from header
                payload_length = struct.unpack('>I', header_data[8:12])[0]
                
                # Read payload and checksum
                remaining_data = self._recv_exact(client_socket, payload_length + 2)
                if not remaining_data:
                    break
                
                full_frame_data = header_data + remaining_data
                
                # Decode frame
                frame = ProtocolFrame.decode(full_frame_data, self.xor_key)
                print(f"[Legacy Server] Received frame type={frame.msg_type:#04x}, seq={frame.seq_num}")
                
                # Process and respond
                response = self._process_frame(frame)
                if response:
                    client_socket.sendall(response)
                    
        except Exception as e:
            print(f"[Legacy Server] Client error: {e}")
        finally:
            client_socket.close()
    
    def _recv_exact(self, sock: socket.socket, n: int) -> bytes:
        """Receive exactly n bytes."""
        data = b''
        while len(data) < n:
            chunk = sock.recv(n - len(data))
            if not chunk:
                return b''
            data += chunk
        return data
    
    def _process_frame(self, frame: ProtocolFrame) -> bytes:
        """Process incoming frame and generate response."""
        self.seq_counter += 1
        
        if frame.msg_type == ProtocolFrame.MSG_HELLO:
            # Respond with hello acknowledgment
            response_payload = b"LEGACY_SERVER_HELLO_ACK_v1.0"
            response_frame = ProtocolFrame(
                ProtocolFrame.MSG_ACK,
                response_payload,
                self.seq_counter
            )
            return response_frame.encode(self.xor_key)
        
        elif frame.msg_type == ProtocolFrame.MSG_DATA:
            # Echo back with modification
            original_msg = frame.payload.decode('utf-8', errors='replace')
            response_msg = f"LEGACY_RESPONSE: {original_msg[::-1]}"  # Reverse the message
            response_frame = ProtocolFrame(
                ProtocolFrame.MSG_DATA,
                response_msg.encode('utf-8'),
                self.seq_counter
            )
            return response_frame.encode(self.xor_key)
        
        elif frame.msg_type == ProtocolFrame.MSG_GOODBYE:
            # Acknowledge goodbye
            response_frame = ProtocolFrame(
                ProtocolFrame.MSG_ACK,
                b"GOODBYE_ACK",
                self.seq_counter
            )
            return response_frame.encode(self.xor_key)
        
        else:
            # Unknown message type
            error_frame = ProtocolFrame(
                ProtocolFrame.MSG_ERROR,
                f"UNKNOWN_MSG_TYPE:{frame.msg_type:#04x}".encode(),
                self.seq_counter
            )
            return error_frame.encode(self.xor_key)
    
    def stop(self):
        """Stop the server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()

# Legacy Server Client
class LegacyServerClient:
    """Client for communicating with the legacy server."""
    
    def __init__(self, host: str = '127.0.0.1', port: int = 9999):
        self.host = host
        self.port = port
        self.xor_key = b'\x42\x13\x37\xAB'  # Must match server's key
        self.socket = None
        self.seq_counter = 0
        self.connected = False
    
    def connect(self) -> bool:
        """Connect to the legacy server."""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.connect((self.host, self.port))
            self.connected = True
            print(f"[Client] Connected to legacy server at {self.host}:{self.port}")
            
            # Send hello
            return self._send_hello()
        except Exception as e:
            print(f"[Client] Connection failed: {e}")
            return False
    
    def _send_hello(self) -> bool:
        """Send hello message to server."""
        self.seq_counter += 1
        hello_frame = ProtocolFrame(
            ProtocolFrame.MSG_HELLO,
            b"CLIENT_HELLO_v1.0",
            self.seq_counter
        )
        
        self.socket.sendall(hello_frame.encode(self.xor_key))
        
        # Wait for ACK
        response = self._receive_frame()
        if response and response.msg_type == ProtocolFrame.MSG_ACK:
            print(f"[Client] Server acknowledged: {response.payload.decode()}")
            return True
        return False
    
    def send_message(self, message: str) -> str:
        """Send a message to the server and get response."""
        if not self.connected:
            return "Not connected to server"
        
        self.seq_counter += 1
        data_frame = ProtocolFrame(
            ProtocolFrame.MSG_DATA,
            message.encode('utf-8'),
            self.seq_counter
        )
        
        # Show encryption details
        encoded = data_frame.encode(self.xor_key)
        print(f"[Client] Sending encrypted frame ({len(encoded)} bytes)")
        print(f"[Client] Original payload: {message}")
        print(f"[Client] XOR key: {self.xor_key.hex()}")
        
        self.socket.sendall(encoded)
        
        # Receive response
        response = self._receive_frame()
        if response:
            return response.payload.decode('utf-8', errors='replace')
        return "No response received"
    
    def _receive_frame(self) -> ProtocolFrame:
        """Receive and decode a frame from the server."""
        try:
            # Read header
            header_data = self._recv_exact(ProtocolFrame.HEADER_SIZE)
            if not header_data:
                return None
            
            # Get payload length
            payload_length = struct.unpack('>I', header_data[8:12])[0]
            
            # Read payload and checksum
            remaining_data = self._recv_exact(payload_length + 2)
            if not remaining_data:
                return None
            
            full_frame_data = header_data + remaining_data
            frame = ProtocolFrame.decode(full_frame_data, self.xor_key)
            return frame
            
        except Exception as e:
            print(f"[Client] Receive error: {e}")
            return None
    
    def _recv_exact(self, n: int) -> bytes:
        """Receive exactly n bytes."""
        data = b''
        while len(data) < n:
            chunk = self.socket.recv(n - len(data))
            if not chunk:
                return b''
            data += chunk
        return data
    
    def disconnect(self):
        """Disconnect from the server."""
        if self.connected:
            self.seq_counter += 1
            goodbye_frame = ProtocolFrame(
                ProtocolFrame.MSG_GOODBYE,
                b"CLIENT_GOODBYE",
                self.seq_counter
            )
            self.socket.sendall(goodbye_frame.encode(self.xor_key))
            
            # Wait for ACK
            response = self._receive_frame()
            if response:
                print(f"[Client] Server goodbye response: {response.payload.decode()}")
            
            self.socket.close()
            self.connected = False
            print("[Client] Disconnected from server")

# AI-powered chat interface using Claude
class LegacyServerChatInterface:
    """
    Multi-turn chat interface that uses Claude to interact with the legacy server.
    """
    
    def __init__(self):
        self.anthropic_client = Anthropic()
        self.legacy_client = LegacyServerClient()
        self.conversation_history = []
        self.server = None
    
    def setup(self):
        """Setup the legacy server and client."""
        # Start legacy server
        self.server = LegacyServer()
        self.server.start()
        time.sleep(0.5)  # Wait for server to start
        
        # Connect client
        if not self.legacy_client.connect():
            raise RuntimeError("Failed to connect to legacy server")
        
        print("\n[System] Legacy server communication system ready!")
        print("[System] XOR encryption active with key:", self.legacy_client.xor_key.hex())
        print("[System] Type 'quit' to exit\n")
    
    def chat(self, user_message: str) -> str:
        """Process user message through Claude and legacy server."""
        # Add user message to history
        self.conversation_history.append({
            "role": "user",
            "content": user_message
        })
        
        # Get Claude's interpretation/command
        system_prompt = """You are an AI assistant helping to communicate with a legacy server 
        that uses a custom TCP protocol with XOR encryption. 
        
        When the user wants to send data to the server, extract the message they want to send.
        When they ask about the protocol or encryption, explain it clearly.
        When they want to test the connection, suggest appropriate test messages.
        
        Always be helpful and explain what's happening with the legacy server communication.
        If the user asks you to send something to the server, respond with:
        SEND_TO_SERVER: <the message to send>
        
        Otherwise, just respond normally to help the user understand the system."""
        
        response = self.anthropic_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_prompt,
            messages=self.conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Check if Claude wants to send something to the server
        server_response = None
        if "SEND_TO_SERVER:" in assistant_message:
            # Extract the message to send
            parts = assistant_message.split("SEND_TO_SERVER:", 1)
            message_to_send = parts[1].strip().split('\n')[0].strip()
            
            print(f"\n[System] Sending to legacy server: '{message_to_send}'")
            server_response = self.legacy_client.send_message(message_to_send)
            print(f"[System] Server response: '{server_response}'")
            
            # Update assistant message with server response
            assistant_message = parts[0].strip()
            if assistant_message:
                assistant_message += f"\n\nServer Response: {server_response}"
            else:
                assistant_message = f"Sent '{message_to_send}' to the legacy server.\nServer Response: {server_response}"
        
        # Add assistant response to history
        self.conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return assistant_message
    
    def cleanup(self):
        """Cleanup connections."""
        if self.legacy_client.connected:
            self.legacy_client.disconnect()
        if self.server:
            self.server.stop()

def main():
    """Main function to run the legacy server chat interface."""
    print("=" * 60)
    print("Legacy Server Communication with XOR Cipher")
    print("Using Claude AI for intelligent interaction")
    print("=" * 60)
    
    interface = LegacyServerChatInterface()
    
    try:
        interface.setup()
        
        print("\nExample interactions:")
        print("- 'Send hello to the server'")
        print("- 'What is XOR encryption?'")
        print("- 'Test the connection with a message'")
        print("- 'Send the message: Hello Legacy World'")
        print()
        
        while True:
            user_input = input("You: ").strip()
            
            if not user_input:
                continue
            
            if user_input.lower() in ['quit', 'exit', 'bye']:
                print("Goodbye!")
                break
            
            print("\nAssistant: ", end="", flush=True)
            response = interface.chat(user_input)
            print(response)
            print()
    
    except KeyboardInterrupt:
        print("\n\nInterrupted by user")
    except Exception as e:
        print(f"\nError: {e}")
        import traceback
        traceback.print_exc()
    finally:
        interface.cleanup()
        print("\n[System] Cleanup complete")

if __name__ == "__main__":
    main()
