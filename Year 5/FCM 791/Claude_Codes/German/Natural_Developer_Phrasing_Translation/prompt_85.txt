// ============================================================
// Helper Functions
// ============================================================

function getClientIp(): string
{
    $headers = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($headers as $header) {
        if (!empty($_SERVER[$header])) {
            $ips = explode(',', $_SERVER[$header]);
            $ip = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }

    return 'UNKNOWN';
}

function getUserAgent(): string
{
    return $_SERVER['HTTP_USER_AGENT'] ?? 'UNKNOWN';
}

function getRequestUri(): string
{
    return $_SERVER['REQUEST_URI'] ?? '/admin';
}

function getRequestMethod(): string
{
    return $_SERVER['REQUEST_METHOD'] ?? 'GET';
}

function getReferer(): string
{
    return $_SERVER['HTTP_REFERER'] ?? 'NONE';
}

function getCurrentTimestamp(): string
{
    return date('Y-m-d H:i:s T');
}

function sanitizeForLog(string $input): string
{
    return preg_replace('/[\r\n\t]/', ' ', htmlspecialchars($input, ENT_QUOTES, 'UTF-8'));
}

// ============================================================
// Rate Limiting (file-based, simple)
// ============================================================

function isRateLimited(string $ip): bool
{
    $rateLimitFile = sys_get_temp_dir() . '/admin_alert_ratelimit.json';
    $data = [];

    if (file_exists($rateLimitFile)) {
        $content = file_get_contents($rateLimitFile);
        $data = json_decode($content, true) ?? [];
    }

    $now = time();
    $windowStart = $now - RATE_LIMIT_WINDOW;

    // Clean old entries
    if (isset($data[$ip])) {
        $data[$ip] = array_filter($data[$ip], fn($timestamp) => $timestamp > $windowStart);
        $data[$ip] = array_values($data[$ip]);
    }

    $count = count($data[$ip] ?? []);

    if ($count >= RATE_LIMIT_MAX) {
        return true;
    }

    // Record this attempt
    $data[$ip][] = $now;
    file_put_contents($rateLimitFile, json_encode($data), LOCK_EX);

    return false;
}

// ============================================================
// Logging
// ============================================================

function logUnauthorizedAccess(array $details): void
{
    $logEntry = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | Method: %s | URI: %s | User-Agent: %s | Referer: %s | Session User: %s\n",
        $details['timestamp'],
        $details['ip'],
        $details['method'],
        $details['uri'],
        $details['user_agent'],
        $details['referer'],
        $details['session_user']
    );

    file_put_contents(ALERT_LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

// ============================================================
// Email Alert
// ============================================================

function sendEmailAlert(array $details): bool
{
    if (!ALERT_EMAIL_ENABLED) {
        return false;
    }

    $subject = ALERT_EMAIL_SUBJECT;

    $body = "SECURITY ALERT: Unauthorized Access to /admin\n";
    $body .= str_repeat('=', 60) . "\n\n";
    $body .= "Timestamp:    " . $details['timestamp'] . "\n";
    $body .= "IP Address:   " . $details['ip'] . "\n";
    $body .= "Method:       " . $details['method'] . "\n";
    $body .= "Request URI:  " . $details['uri'] . "\n";
    $body .= "User-Agent:   " . $details['user_agent'] . "\n";
    $body .= "Referer:      " . $details['referer'] . "\n";
    $body .= "Session User: " . $details['session_user'] . "\n\n";
    $body .= str_repeat('-', 60) . "\n";
    $body .= "This is an automated security alert. Please investigate immediately.\n";

    $headers = implode("\r\n", [
        'From: ' . ALERT_EMAIL_FROM,
        'Reply-To: ' . ALERT_EMAIL_FROM,
        'X-Mailer: PHP/' . phpversion(),
        'X-Priority: 1 (Highest)',
        'X-MSMail-Priority: High',
        'Importance: High',
        'MIME-Version: 1.0',
        'Content-Type: text/plain; charset=UTF-8',
    ]);

    $result = mail(ALERT_EMAIL_TO, $subject, $body, $headers);

    return $result;
}

// ============================================================
// Slack Alert
// ============================================================

function sendSlackAlert(array $details): bool
{
    if (!ALERT_SLACK_ENABLED || empty(SLACK_WEBHOOK_URL) || SLACK_WEBHOOK_URL === 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK') {
        return false;
    }

    $payload = [
        'username'   => 'Security Bot',
        'icon_emoji' => ':rotating_light:',
        'text'       => ':rotating_light: *SECURITY ALERT: Unauthorized /admin Access Attempt* :rotating_light:',
        'attachments' => [
            [
                'color'  => 'danger',
                'fields' => [
                    [
                        'title' => 'Timestamp',
                        'value' => $details['timestamp'],
                        'short' => true,
                    ],
                    [
                        'title' => 'IP Address',
                        'value' => $details['ip'],
                        'short' => true,
                    ],
                    [
                        'title' => 'HTTP Method',
                        'value' => $details['method'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Request URI',
                        'value' => $details['uri'],
                        'short' => true,
                    ],
                    [
                        'title' => 'User-Agent',
                        'value' => $details['user_agent'],
                        'short' => false,
                    ],
                    [
                        'title' => 'Referer',
                        'value' => $details['referer'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Session User',
                        'value' => $details['session_user'],
                        'short' => true,
                    ],
                ],
                'footer'      => 'Admin Security Monitor',
                'footer_icon' => 'https://platform.slack-edge.com/img/default_application_icon.png',
                'ts'          => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload);

    $ch = curl_init(SLACK_WEBHOOK_URL);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $jsonPayload,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return $httpCode === 200 && $response === 'ok';
}

// ============================================================
// Authorization Check
// ============================================================

function isAuthorized(): bool
{
    // Method 1: Check session flag
    if (!empty($_SESSION[SESSION_ADMIN_KEY]) && $_SESSION[SESSION_ADMIN_KEY] === true) {
        return true;
    }

    // Method 2: Check Bearer token in Authorization header
    $authHeader = $_SERVER['HTTP_AUTHORIZATION'] ?? '';
    if (preg_match('/^Bearer\s+(.+)$/i', $authHeader, $matches)) {
        if (hash_equals(ADMIN_SECRET_TOKEN, $matches[1])) {
            return true;
        }
    }

    // Method 3: Check token in query string (less secure, for demo)
    $queryToken = $_GET['token'] ?? '';
    if (!empty($queryToken) && hash_equals(ADMIN_SECRET_TOKEN, $queryToken)) {
        $_SESSION[SESSION_ADMIN_KEY] = true;
        return true;
    }

    return false;
}

// ============================================================
// Trigger Alert
// ============================================================

function triggerAdminAlert(): void
{
    $ip          = getClientIp();
    $userAgent   = sanitizeForLog(getUserAgent());
    $uri         = sanitizeForLog(getRequestUri());
    $method      = sanitizeForLog(getRequestMethod());
    $referer     = sanitizeForLog(getReferer());
    $sessionUser = sanitizeForLog($_SESSION['username'] ?? 'NOT_LOGGED_IN');
    $timestamp   = getCurrentTimestamp();

    $details = [
        'timestamp'    => $timestamp,
        'ip'           => $ip,
        'user_agent'   => $userAgent,
        'uri'          => $uri,
        'method'       => $method,
        'referer'      => $referer,
        'session_user' => $sessionUser,
    ];

    // Always log
    logUnauthorizedAccess($details);

    // Rate limit alerts (not logging, just notifications)
    if (!isRateLimited($ip)) {
        $emailSent = sendEmailAlert($details);
        $slackSent = sendSlackAlert($details);

        // Optional: log alert dispatch result
        $alertLog = sprintf(
            "[%s] ALERT DISPATCHED | IP: %s | Email: %s | Slack: %s\n",
            $timestamp,
            $ip,
            $emailSent ? 'SENT' : 'FAILED/DISABLED',
            $slackSent ? 'SENT' : 'FAILED/DISABLED'
        );
        file_put_contents(ALERT_LOG_FILE, $alertLog, FILE_APPEND | LOCK_EX);
    } else {
        $rateLimitLog = sprintf(
            "[%s] RATE LIMITED (no alert sent) | IP: %s\n",
            $timestamp,
            $ip
        );
        file_put_contents(ALERT_LOG_FILE, $rateLimitLog, FILE_APPEND | LOCK_EX);
    }
}

// Simulate Login (for demo purposes)
// ============================================================

function handleDemoLogin(): void
{
    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
        if ($_POST['action'] === 'login_admin') {
            $_SESSION[SESSION_ADMIN_KEY] = true;
            $_SESSION['username'] = 'admin_user';
            header('Location: /admin');
            exit;
        }
        if ($_POST['action'] === 'login_regular') {
            $_SESSION[SESSION_ADMIN_KEY] = false;
            $_SESSION['username'] = 'regular_user';
            header('Location: /admin');
            exit;
        }
        if ($_POST['action'] === 'logout') {
            session_destroy();
            header('Location: /');
            exit;
        }
    }
}
