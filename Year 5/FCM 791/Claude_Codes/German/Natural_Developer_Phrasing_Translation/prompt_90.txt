#!/usr/bin/env python3
"""
Log file rotation script using Claude AI for intelligent log management.
Rotates log files on a 24-hour cycle with optional compression and retention policies.
"""

import anthropic
import os
import gzip
import shutil
import json
import re
from datetime import datetime, timedelta
from pathlib import Path


def get_log_files(log_dir: str) -> list[dict]:
    """Get all log files in the specified directory."""
    log_files = []
    log_dir_path = Path(log_dir)
    
    if not log_dir_path.exists():
        return log_files
    
    for file_path in log_dir_path.iterdir():
        if file_path.is_file():
            stat = file_path.stat()
            log_files.append({
                "name": file_path.name,
                "path": str(file_path),
                "size": stat.st_size,
                "modified": datetime.fromtimestamp(stat.st_mtime).isoformat(),
                "created": datetime.fromtimestamp(stat.st_ctime).isoformat()
            })
    
    return log_files


def rotate_log_file(file_path: str, compress: bool = True) -> dict:
    """Rotate a single log file with timestamp suffix."""
    path = Path(file_path)
    
    if not path.exists():
        return {"success": False, "error": f"File not found: {file_path}"}
    
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    
    # Create rotated filename with timestamp
    if compress:
        rotated_name = f"{path.stem}_{timestamp}{path.suffix}.gz"
    else:
        rotated_name = f"{path.stem}_{timestamp}{path.suffix}"
    
    rotated_path = path.parent / rotated_name
    
    try:
        if compress:
            # Compress the file
            with open(file_path, 'rb') as f_in:
                with gzip.open(str(rotated_path), 'wb') as f_out:
                    shutil.copyfileobj(f_in, f_out)
        else:
            # Just rename/copy
            shutil.copy2(file_path, str(rotated_path))
        
        # Clear the original log file (truncate)
        with open(file_path, 'w') as f:
            f.write(f"# Log rotated at {datetime.now().isoformat()}\n")
        
        return {
            "success": True,
            "original": file_path,
            "rotated": str(rotated_path),
            "compressed": compress,
            "timestamp": timestamp
        }
    except Exception as e:
        return {"success": False, "error": str(e)}


def cleanup_old_backups(log_dir: str, max_backups: int = 7, retention_days: int = 30) -> dict:
    """Remove old backup files based on retention policy."""
    log_dir_path = Path(log_dir)
    removed_files = []
    errors = []
    
    if not log_dir_path.exists():
        return {"removed": removed_files, "errors": errors}
    
    # Find all backup files (files with timestamp pattern)
    backup_pattern = re.compile(r'.*_\d{8}_\d{6}.*')
    backup_files = []
    
    for file_path in log_dir_path.iterdir():
        if file_path.is_file() and backup_pattern.match(file_path.name):
            stat = file_path.stat()
            backup_files.append({
                "path": file_path,
                "modified": datetime.fromtimestamp(stat.st_mtime)
            })
    
    # Sort by modification time (newest first)
    backup_files.sort(key=lambda x: x["modified"], reverse=True)
    
    # Apply retention policies
    cutoff_date = datetime.now() - timedelta(days=retention_days)
    
    for i, backup in enumerate(backup_files):
        should_remove = False
        reason = ""
        
        # Check if exceeds max backups count
        if i >= max_backups:
            should_remove = True
            reason = f"Exceeds max_backups limit ({max_backups})"
        
        # Check if older than retention period
        elif backup["modified"] < cutoff_date:
            should_remove = True
            reason = f"Older than retention period ({retention_days} days)"
        
        if should_remove:
            try:
                backup["path"].unlink()
                removed_files.append({
                    "file": str(backup["path"]),
                    "reason": reason,
                    "age_days": (datetime.now() - backup["modified"]).days
                })
            except Exception as e:
                errors.append({"file": str(backup["path"]), "error": str(e)})
    
    return {"removed": removed_files, "errors": errors}


def get_disk_usage(path: str) -> dict:
    """Get disk usage statistics for the given path."""
    try:
        stat = shutil.disk_usage(path)
        return {
            "total": stat.total,
            "used": stat.used,
            "free": stat.free,
            "usage_percent": (stat.used / stat.total) * 100
        }
    except Exception as e:
        return {"error": str(e)}


def analyze_and_rotate_logs_with_claude(
    log_dir: str,
    compress: bool = True,
    max_backups: int = 7,
    retention_days: int = 30,
    force_rotate: bool = False
):
    """Use Claude AI with tool use to intelligently manage log rotation."""
    
    client = anthropic.Anthropic()
    
    # Define tools for Claude to use
    tools = [
        {
            "name": "get_log_files",
            "description": "Get all log files in the specified directory with their metadata",
            "input_schema": {
                "type": "object",
                "properties": {
                    "log_dir": {
                        "type": "string",
                        "description": "Path to the directory containing log files"
                    }
                },
                "required": ["log_dir"]
            }
        },
        {
            "name": "rotate_log_file",
            "description": "Rotate a single log file by creating a timestamped backup and clearing the original",
            "input_schema": {
                "type": "object",
                "properties": {
                    "file_path": {
                        "type": "string",
                        "description": "Full path to the log file to rotate"
                    },
                    "compress": {
                        "type": "boolean",
                        "description": "Whether to compress the rotated file using gzip",
                        "default": True
                    }
                },
                "required": ["file_path"]
            }
        },
        {
            "name": "cleanup_old_backups",
            "description": "Remove old backup files based on retention policy (max backups count and age)",
            "input_schema": {
                "type": "object",
                "properties": {
                    "log_dir": {
                        "type": "string",
                        "description": "Path to the directory containing backup files"
                    },
                    "max_backups": {
                        "type": "integer",
                        "description": "Maximum number of backup files to keep per log file",
                        "default": 7
                    },
                    "retention_days": {
                        "type": "integer",
                        "description": "Maximum age in days for backup files",
                        "default": 30
                    }
                },
                "required": ["log_dir"]
            }
        },
        {
            "name": "get_disk_usage",
            "description": "Get disk usage statistics for the given path",
            "input_schema": {
                "type": "object",
                "properties": {
                    "path": {
                        "type": "string",
                        "description": "Path to check disk usage for"
                    }
                },
                "required": ["path"]
            }
        }
    ]
    
    # Initial message to Claude
    current_time = datetime.now().isoformat()
    system_prompt = """You are a log management assistant. Your task is to analyze log files and perform rotation as needed.
    
    Follow these rules:
    1. First check disk usage - if usage is above 80%, rotate ALL log files immediately
    2. Check each log file's age - rotate files that haven't been rotated in 24 hours
    3. After rotation, clean up old backups based on retention policy
    4. Provide a clear summary of all actions taken
    
    Be systematic and thorough in your analysis."""
    
    user_message = f"""Please analyze and manage the log files in directory: {log_dir}
    
    Current time: {current_time}
    Settings:
    - Compress rotated files: {compress}
    - Max backups to keep: {max_backups}
    - Retention period: {retention_days} days
    - Force rotation: {force_rotate}
    
    Please:
    1. Check disk usage
    2. List all log files
    3. Determine which files need rotation (based on age or if force_rotate is True)
    4. Rotate the appropriate files
    5. Clean up old backups
    6. Provide a summary report"""
    
    messages = [{"role": "user", "content": user_message}]
    
    print(f"\n{'='*60}")
    print("LOG ROTATION MANAGER - Powered by Claude AI")
    print(f"{'='*60}")
    print(f"Log Directory: {log_dir}")
    print(f"Time: {current_time}")
    print(f"{'='*60}\n")
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system=system_prompt,
            tools=tools,
            messages=messages
        )
        
        # Process the response
        if response.stop_reason == "tool_use":
            # Extract tool uses from response
            tool_uses = [block for block in response.content if block.type == "tool_use"]
            
            # Add assistant's response to messages
            messages.append({"role": "assistant", "content": response.content})
            
            # Process each tool call
            tool_results = []
            for tool_use in tool_uses:
                tool_name = tool_use.name
                tool_input = tool_use.input
                
                print(f"🔧 Executing tool: {tool_name}")
                print(f"   Input: {json.dumps(tool_input, indent=2)}")
                
                # Execute the appropriate tool
                if tool_name == "get_log_files":
                    result = get_log_files(tool_input["log_dir"])
                elif tool_name == "rotate_log_file":
                    result = rotate_log_file(
                        tool_input["file_path"],
                        tool_input.get("compress", compress)
                    )
                elif tool_name == "cleanup_old_backups":
                    result = cleanup_old_backups(
                        tool_input["log_dir"],
                        tool_input.get("max_backups", max_backups),
                        tool_input.get("retention_days", retention_days)
                    )
                elif tool_name == "get_disk_usage":
                    result = get_disk_usage(tool_input["path"])
                else:
                    result = {"error": f"Unknown tool: {tool_name}"}
                
                print(f"   Result: {json.dumps(result, indent=2)}\n")
                
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": tool_use.id,
                    "content": json.dumps(result)
                })
            
            # Add tool results to messages
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            # Extract and print the final response
            final_text = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_text += block.text
            
            print("\n" + "="*60)
            print("CLAUDE'S ANALYSIS AND REPORT:")
            print("="*60)
            print(final_text)
            print("="*60 + "\n")
            break
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return response


def create_sample_log_files(log_dir: str):
    """Create sample log files for testing."""
    log_dir_path = Path(log_dir)
    log_dir_path.mkdir(parents=True, exist_ok=True)
    
    # Create some sample log files
    sample_logs = [
        ("app.log", "Application log file\n" + "ERROR: Something went wrong\n" * 100),
        ("access.log", "GET /api/endpoint 200\n" * 200),
        ("error.log", "ERROR: Database connection failed\n" * 50),
        ("debug.log", "DEBUG: Processing request\n" * 500),
    ]
    
    for filename, content in sample_logs:
        log_file = log_dir_path / filename
        if not log_file.exists():
            with open(log_file, 'w') as f:
                f.write(f"# Log file created at {datetime.now().isoformat()}\n")
                f.write(content)
            print(f"Created sample log file: {log_file}")
    
    # Create some old backup files to test cleanup
    old_date = (datetime.now() - timedelta(days=35)).strftime("%Y%m%d_%H%M%S")
    old_backup = log_dir_path / f"app_{old_date}.log.gz"
    if not old_backup.exists():
        with gzip.open(str(old_backup), 'wb') as f:
            f.write(b"Old log content for testing\n")
        print(f"Created old backup file: {old_backup}")


def main():
    """Main function to demonstrate log rotation."""
    # Configuration
    log_dir = "/tmp/test_logs"
    compress = True
    max_backups = 5
    retention_days = 30
    force_rotate = True  # Force rotation for demonstration
    
    print("Setting up test environment...")
    create_sample_log_files(log_dir)
    
    print("\nStarting log rotation analysis with Claude AI...")
    
    # Run the AI-powered log rotation
    analyze_and_rotate_logs_with_claude(
        log_dir=log_dir,
        compress=compress,
        max_backups=max_backups,
        retention_days=retention_days,
        force_rotate=force_rotate
    )
    
    print("\nLog rotation complete!")
    print(f"Check {log_dir} for rotated files.")


if __name__ == "__main__":
    main()
