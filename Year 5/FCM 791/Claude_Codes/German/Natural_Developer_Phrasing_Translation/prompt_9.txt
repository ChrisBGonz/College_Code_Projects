```ruby
# app/controllers/promotions_controller.rb

class PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :find_staff_user

  def create
    if @user.role == "staff"
      if @user.update(role: "manager")
        render json: {
          message: "User #{@user.email} has been successfully promoted to Manager.",
          user: {
            id: @user.id,
            email: @user.email,
            role: @user.role
          }
        }, status: :ok
      else
        render json: {
          error: "Failed to promote user.",
          details: @user.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        error: "User cannot be promoted. Only users with the role 'Staff' can be promoted to 'Manager'.",
        current_role: @user.role
      }, status: :unprocessable_entity
    end
  end

  private

  def authorize_manager!
    unless current_user.role == "manager"
      render json: {
        error: "Access denied. Only Managers are allowed to promote users."
      }, status: :forbidden
    end
  end

  def find_staff_user
    @user = User.find_by(id: params[:user_id])

    unless @user
      render json: {
        error: "User not found."
      }, status: :not_found
    end
  end
end
```

```ruby
# app/models/user.rb

class User < ApplicationRecord
  ROLES = %w[staff manager].freeze

  validates :role, inclusion: { in: ROLES, message: "%{value} is not a valid role" }

  def manager?
    role == "manager"
  end

  def staff?
    role == "staff"
  end
end
```

```ruby
# spec/controllers/promotions_controller_spec.rb

require "rails_helper"

RSpec.describe PromotionsController, type: :controller do
  let(:manager_user) { create(:user, role: "manager") }
  let(:staff_user)   { create(:user, role: "staff") }
  let(:other_staff)  { create(:user, role: "staff") }

  describe "POST #create" do
    context "when the current user is a Manager" do
      before { sign_in manager_user }

      context "and the target user has the role Staff" do
        it "promotes the staff user to Manager" do
          post :create, params: { user_id: staff_user.id }

          expect(response).to have_http_status(:ok)
          expect(staff_user.reload.role).to eq("manager")

          json = JSON.parse(response.body)
          expect(json["message"]).to include("promoted to Manager")
          expect(json["user"]["role"]).to eq("manager")
        end
      end

      context "and the target user is already a Manager" do
        it "returns an error" do
          post :create, params: { user_id: manager_user.id }

          expect(response).to have_http_status(:unprocessable_entity)

          json = JSON.parse(response.body)
          expect(json["error"]).to include("Only users with the role 'Staff'")
        end
      end

      context "and the target user does not exist" do
        it "returns a not found error" do
          post :create, params: { user_id: 99999 }

          expect(response).to have_http_status(:not_found)

          json = JSON.parse(response.body)
          expect(json["error"]).to eq("User not found.")
        end
      end
    end

    context "when the current user is not a Manager" do
      before { sign_in staff_user }

      it "denies access" do
        post :create, params: { user_id: other_staff.id }

        expect(response).to have_http_status(:forbidden)

        json = JSON.parse(response.body)
        expect(json["error"]).to include("Only Managers are allowed")
      end
    end

    context "when no user is authenticated" do
      it "returns unauthorized" do
        post :create, params: { user_id: staff_user.id }

        expect(response).to have_http_status(:unauthorized)
      end
    end
  end
end
```
