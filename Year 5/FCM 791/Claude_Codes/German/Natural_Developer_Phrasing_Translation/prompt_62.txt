import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated session store for demonstration
const sessionStore = new Map();

// Helper function to simulate session operations
function createMockSession(sessionId, userId) {
  const session = {
    id: sessionId,
    userId: userId,
    createdAt: new Date().toISOString(),
    data: { authenticated: true },
    destroy: function (callback) {
      sessionStore.delete(this.id);
      if (callback) callback(null);
    },
  };
  sessionStore.set(sessionId, session);
  return session;
}

// Simulate a request object with session
function createMockRequest(sessionId, userId) {
  const session = createMockSession(sessionId, userId);
  return {
    session: session,
    headers: {
      cookie: `sessionId=${sessionId}`,
    },
  };
}

// Simulate a response object
function createMockResponse() {
  const cookies = {};
  const headers = {};
  let statusCode = 200;
  let body = null;

  return {
    cookies,
    headers,
    statusCode,
    body,
    clearCookie: function (name, options) {
      this.cookies[name] = { cleared: true, options };
      console.log(`Cookie '${name}' cleared with options:`, options);
    },
    setHeader: function (name, value) {
      this.headers[name] = value;
    },
    status: function (code) {
      this.statusCode = code;
      return this;
    },
    json: function (data) {
      this.body = data;
      console.log("Response:", JSON.stringify(data, null, 2));
      return this;
    },
    send: function (data) {
      this.body = data;
      console.log("Response sent:", data);
      return this;
    },
  };
}

// Actual logout handler implementation based on best practices
async function logoutHandler(req, res) {
  try {
    // Check if session exists
    if (!req.session) {
      return res.status(400).json({
        success: false,
        message: "No active session found",
      });
    }

    const sessionId = req.session.id;
    const userId = req.session.userId;

    // Destroy the session
    await new Promise((resolve, reject) => {
      req.session.destroy((err) => {
        if (err) {
          reject(err);
        } else {
          resolve();
        }
      });
    });

    // Clear the session cookie
    res.clearCookie("sessionId", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      path: "/",
    });

    // Also clear any other auth-related cookies
    res.clearCookie("connect.sid", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      path: "/",
    });

    console.log(`User ${userId} logged out successfully. Session ${sessionId} destroyed.`);

    return res.status(200).json({
      success: true,
      message: "Logout successful",
    });
  } catch (error) {
    console.error("Logout error:", error);

    // Even if session destruction fails, clear the cookie
    res.clearCookie("sessionId", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      path: "/",
    });

    return res.status(500).json({
      success: false,
      message: "An error occurred during logout",
    });
  }
}

async function demonstrateLogout() {
  console.log("=== Logout Handler Demonstration ===\n");

  // First, generate the handler using Claude with extended thinking
  await generateLogoutHandler();

  console.log("\n=== Testing the Logout Handler ===\n");

  // Test 1: Successful logout
  console.log("Test 1: Successful logout");
  const req1 = createMockRequest("session-123", "user-456");
  const res1 = createMockResponse();

  console.log("Before logout - Session store size:", sessionStore.size);
  console.log("Session exists:", sessionStore.has("session-123"));

  await logoutHandler(req1, res1);

  console.log("After logout - Session store size:", sessionStore.size);
  console.log("Session exists:", sessionStore.has("session-123"));
  console.log("Response status:", res1.statusCode);
  console.log("Cookies cleared:", JSON.stringify(res1.cookies, null, 2));

  console.log("\n---\n");

  // Test 2: Logout without session
  console.log("Test 2: Logout without session");
  const req2 = { headers: {} }; // No session
  const res2 = createMockResponse();

  await logoutHandler(req2, res2);

  console.log("Response status:", res2.statusCode);

  console.log("\n---\n");

  // Test 3: Logout with session destruction error
  console.log("Test 3: Logout with session destruction error");
  const req3 = {
    session: {
      id: "error-session",
      userId: "user-789",
      destroy: function (callback) {
        callback(new Error("Session store connection failed"));
      },
    },
    headers: {},
  };
  const res3 = createMockResponse();

  await logoutHandler(req3, res3);

  console.log("Response status:", res3.statusCode);
  console.log("Cookies cleared despite error:", JSON.stringify(res3.cookies, null, 2));
}

demonstrateLogout().catch(console.error);
