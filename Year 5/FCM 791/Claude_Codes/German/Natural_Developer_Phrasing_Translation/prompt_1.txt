from flask import Flask, jsonify, abort
import sqlite3
import os

app = Flask(__name__)

DATABASE = os.environ.get("DATABASE_PATH", "invoices.db")


def get_db_connection():
    conn = sqlite3.connect(DATABASE)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = get_db_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS invoices (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            invoice_number TEXT NOT NULL UNIQUE,
            customer_name TEXT NOT NULL,
            customer_email TEXT,
            issue_date TEXT NOT NULL,
            due_date TEXT NOT NULL,
            total_amount REAL NOT NULL,
            currency TEXT DEFAULT 'EUR',
            status TEXT DEFAULT 'pending',
            notes TEXT,
            created_at TEXT DEFAULT CURRENT_TIMESTAMP
        )
    """)
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS invoice_items (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            invoice_id INTEGER NOT NULL,
            description TEXT NOT NULL,
            quantity REAL NOT NULL,
            unit_price REAL NOT NULL,
            tax_rate REAL DEFAULT 0.0,
            FOREIGN KEY (invoice_id) REFERENCES invoices (id)
        )
    """)
    cursor.execute("""
        INSERT OR IGNORE INTO invoices
            (invoice_number, customer_name, customer_email, issue_date, due_date, total_amount, currency, status, notes)
        VALUES
            ('INV-001', 'Max Mustermann', 'max@example.com', '2024-01-01', '2024-01-31', 1190.00, 'EUR', 'paid', 'Erste Rechnung'),
            ('INV-002', 'Erika Musterfrau', 'erika@example.com', '2024-02-01', '2024-02-28', 595.00, 'EUR', 'pending', NULL)
    """)
    cursor.execute("""
        INSERT OR IGNORE INTO invoice_items
            (invoice_id, description, quantity, unit_price, tax_rate)
        VALUES
            (1, 'Webentwicklung', 10, 100.00, 0.19),
            (1, 'Beratung', 2, 50.00, 0.19),
            (2, 'Design', 5, 100.00, 0.19),
            (2, 'Support', 1, 95.00, 0.19)
    """)
    conn.commit()
    conn.close()


@app.route("/invoices/<int:invoice_id>", methods=["GET"])
def get_invoice(invoice_id):
    conn = get_db_connection()
    try:
        invoice = conn.execute(
            "SELECT * FROM invoices WHERE id = ?",
            (invoice_id,)
        ).fetchone()

        if invoice is None:
            abort(404, description=f"Rechnung mit ID {invoice_id} nicht gefunden.")

        items = conn.execute(
            "SELECT * FROM invoice_items WHERE invoice_id = ?",
            (invoice_id,)
        ).fetchall()

        invoice_dict = dict(invoice)
        invoice_dict["items"] = [dict(item) for item in items]

        return jsonify({
            "success": True,
            "data": invoice_dict
        }), 200

    except sqlite3.Error as e:
        abort(500, description=f"Datenbankfehler: {str(e)}")
    finally:
        conn.close()


@app.route("/invoices", methods=["GET"])
def get_all_invoices():
    conn = get_db_connection()
    try:
        invoices = conn.execute("SELECT * FROM invoices").fetchall()
        return jsonify({
            "success": True,
            "count": len(invoices),
            "data": [dict(inv) for inv in invoices]
        }), 200
    except sqlite3.Error as e:
        abort(500, description=f"Datenbankfehler: {str(e)}")
    finally:
        conn.close()


@app.errorhandler(404)
def not_found(error):
    return jsonify({
        "success": False,
        "error": 404,
        "message": error.description
    }), 404


@app.errorhandler(500)
def internal_error(error):
    return jsonify({
        "success": False,
        "error": 500,
        "message": error.description
    }), 500


if __name__ == "__main__":
    init_db()
    app.run(debug=True, host="0.0.0.0", port=5000)
