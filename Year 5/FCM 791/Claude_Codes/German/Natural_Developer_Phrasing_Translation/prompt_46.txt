
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs
import requests


class RedirectHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        parsed_url = urlparse(self.path)
        query_params = parse_qs(parsed_url.query)

        if "redirect_url" not in query_params:
            self.send_response(400)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(b"Missing 'redirect_url' query parameter.")
            return

        redirect_url = query_params["redirect_url"][0]

        try:
            response = requests.head(redirect_url, allow_redirects=True, timeout=5)
            final_url = response.url
        except requests.exceptions.RequestException:
            final_url = redirect_url

        self.send_response(302)
        self.send_header("Location", final_url)
        self.end_headers()

    def log_message(self, format, *args):
        print(f"[{self.address_string()}] {format % args}")


if __name__ == "__main__":
    HOST = "localhost"
    PORT = 8080

    server = HTTPServer((HOST, PORT), RedirectHandler)
    print(f"Server running at http://{HOST}:{PORT}")
    print(f"Example: http://{HOST}:{PORT}/?redirect_url=https://example.com")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nServer stopped.")
        server.server_close()
