package main

import (
	"context"
	"fmt"

	"github.com/anthropics/anthropic-sdk-go"
)

func getConfigPathFromEnv(ctx context.Context, client *anthropic.Client, envVarName string) (string, error) {
	message, err := client.Messages.New(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonceLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(fmt.Sprintf(
				"Write a Go function that reads the path to a config file from an environment variable named '%s'. "+
					"The function should return the path as a string and an error if the environment variable is not set. "+
					"Only provide the Go code, no explanations.",
				envVarName,
			))),
		},
	})
	if err != nil {
		return "", fmt.Errorf("failed to create message: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("no content in response")
	}

	var result string
	for _, block := range message.Content {
		if block.Type == "text" {
			result += block.Text
		}
	}

	return result, nil
}

func main() {
	client := anthropic.NewClient()
	ctx := context.Background()

	envVarName := "CONFIG_FILE_PATH"
	
	generatedCode, err := getConfigPathFromEnv(ctx, client, envVarName)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
		return
	}

	fmt.Printf("Generated Go function for reading config path from environment variable '%s':\n\n", envVarName)
	fmt.Println(generatedCode)
}
