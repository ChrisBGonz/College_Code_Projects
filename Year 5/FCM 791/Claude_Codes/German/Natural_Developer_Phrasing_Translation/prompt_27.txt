import anthropic
import subprocess
import sys


def install_package(package: str) -> None:
    """Install a Python package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])


def ensure_dependencies() -> None:
    """Ensure required dependencies are installed."""
    try:
        import weasyprint
    except ImportError:
        print("Installing weasyprint...")
        install_package("weasyprint")


def render_html_to_pdf(html_content: str, output_path: str) -> bool:
    """
    Render HTML content to a PDF file using WeasyPrint.
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Path where the PDF file should be saved
        
    Returns:
        True if successful, False otherwise
    """
    try:
        from weasyprint import HTML
        
        # Create HTML object and write to PDF
        HTML(string=html_content).write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True
    except Exception as e:
        print(f"Error creating PDF: {e}")
        return False


def render_html_file_to_pdf(html_file_path: str, output_path: str) -> bool:
    """
    Render an HTML file to a PDF file using WeasyPrint.
    
    Args:
        html_file_path: Path to the HTML file to convert
        output_path: Path where the PDF file should be saved
        
    Returns:
        True if successful, False otherwise
    """
    try:
        from weasyprint import HTML
        
        # Create HTML object from file and write to PDF
        HTML(filename=html_file_path).write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True
    except Exception as e:
        print(f"Error creating PDF: {e}")
        return False


def generate_html_template_with_claude(template_description: str) -> str:
    """
    Use Claude to generate an HTML template based on a description.
    Uses streaming to show the generation progress.
    
    Args:
        template_description: Description of the HTML template to generate
        
    Returns:
        Generated HTML content as a string
    """
    client = anthropic.Anthropic()
    
    html_content = ""
    
    print("Generating HTML template with Claude...")
    print("-" * 50)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": f"""Generate a complete, well-formatted HTML template for the following description:

{template_description}

Requirements:
- Include proper HTML5 structure with DOCTYPE, head, and body
- Add inline CSS styles for good visual appearance
- Make it suitable for PDF rendering
- Include sample/placeholder content
- Use professional styling with good typography
- Ensure the layout works well when converted to PDF

Return ONLY the HTML code, no explanations or markdown formatting."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            html_content += text
    
    print("\n" + "-" * 50)
    
    # Clean up the HTML content if it has markdown code blocks
    if "```html" in html_content:
        html_content = html_content.split("```html")[1].split("```")[0].strip()
    elif "```" in html_content:
        html_content = html_content.split("```")[1].split("```")[0].strip()
    
    return html_content


def create_pdf_from_description(
    template_description: str, 
    output_path: str
) -> bool:
    """
    Main function that combines Claude's HTML generation with PDF rendering.
    
    Args:
        template_description: Description of what the PDF should contain
        output_path: Path where the PDF file should be saved
        
    Returns:
        True if successful, False otherwise
    """
    # Ensure dependencies are installed
    ensure_dependencies()
    
    # Generate HTML template using Claude
    html_content = generate_html_template_with_claude(template_description)
    
    if not html_content:
        print("Failed to generate HTML content")
        return False
    
    # Convert HTML to PDF
    return render_html_to_pdf(html_content, output_path)


def main():
