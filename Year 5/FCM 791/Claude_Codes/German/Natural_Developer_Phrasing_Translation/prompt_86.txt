package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func performBankTransfer(senderID, receiverID string, amount float64) error {
	log.Printf("Initiating bank transfer - sender_id: %s, receiver_id: %s, amount: %.2f",
		senderID, receiverID, amount)

	if amount <= 0 {
		log.Printf("Transfer failed - Invalid amount: %.2f for sender_id: %s", amount, senderID)
		return fmt.Errorf("invalid amount: %.2f", amount)
	}

	if senderID == "" || receiverID == "" {
		log.Printf("Transfer failed - Missing sender_id or receiver_id")
		return fmt.Errorf("sender_id and receiver_id are required")
	}

	log.Printf("Transfer successful - sender_id: %s, receiver_id: %s, amount: %.2f",
		senderID, receiverID, amount)

	return nil
}

func main() {
	logger := log.New(os.Stdout, "[BANK] ", log.LstdFlags|log.Lmicroseconds)
	log.SetOutput(logger.Writer())
	log.SetFlags(logger.Flags())
	log.SetPrefix(logger.Prefix())

	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonsheetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(
				"Please explain how the bank transfer function works and what it logs. " +
					"The function signature is: performBankTransfer(senderID, receiverID string, amount float64) error",
			)),
		},
	})
	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	fmt.Println("Claude's explanation of the bank transfer function:")
	fmt.Println("===================================================")
	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}

	fmt.Println("\nDemonstrating bank transfer function:")
	fmt.Println("=====================================")

	err = performBankTransfer("USER123", "USER456", 1500.00)
	if err != nil {
		fmt.Printf("Transfer error: %v\n", err)
	}

	err = performBankTransfer("USER789", "USER012", -100.00)
	if err != nil {
		fmt.Printf("Transfer error: %v\n", err)
	}

	err = performBankTransfer("", "USER456", 500.00)
	if err != nil {
		fmt.Printf("Transfer error: %v\n", err)
	}
}
