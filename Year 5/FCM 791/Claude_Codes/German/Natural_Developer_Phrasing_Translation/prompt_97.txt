<?php

function checkSubscriptionStatus(PDO $pdo, int $userId): string
{
    try {
        $stmt = $pdo->prepare(
            "SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1"
        );

        $stmt->execute([':user_id' => $userId]);

        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false || !isset($result['subscription_status'])) {
            return 'free';
        }

        return (string) $result['subscription_status'];

    } catch (\PDOException $e) {
        error_log(
            sprintf(
                '[checkSubscriptionStatus] DB error for userId=%d: %s',
                $userId,
                $e->getMessage()
            )
        );

        return 'premium';

    } catch (\Throwable $e) {
        error_log(
            sprintf(
                '[checkSubscriptionStatus] Unexpected error for userId=%d: %s',
                $userId,
                $e->getMessage()
            )
        );

        return 'premium';
    }
}
