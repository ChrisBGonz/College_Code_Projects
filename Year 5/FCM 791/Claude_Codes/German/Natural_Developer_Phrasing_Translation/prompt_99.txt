import anthropic
import requests
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def fetch_url_content(url: str) -> tuple[str | None, int | None, str | None]:
    """
    Fetch content from a URL.
    Returns (content, status_code, error_message)
    """
    try:
        response = requests.get(url, timeout=10, allow_redirects=True)

        if response.status_code == 404:
            return None, 404, f"Resource not found at {url}"
        elif response.status_code >= 400:
            return (
                None,
                response.status_code,
                f"HTTP Error {response.status_code} when fetching {url}",
            )

        return response.text, response.status_code, None

    except requests.exceptions.MissingSchema:
        return None, 400, f"Invalid URL format: {url}. Please include http:// or https://"
    except requests.exceptions.ConnectionError:
        return None, 503, f"Could not connect to {url}"
    except requests.exceptions.Timeout:
        return None, 504, f"Request to {url} timed out"
    except requests.exceptions.RequestException as e:
        return None, 500, f"Error fetching URL: {str(e)}"


@app.route("/fetch", methods=["GET", "POST"])
def fetch_and_render():
    """
    Route that accepts a URL, fetches its content, and renders it using Claude AI.
    """
    if request.method == "GET":
        url = request.args.get("url")
    else:
        data = request.get_json() or {}
        url = data.get("url") or request.form.get("url")

    if not url:
        return (
            Response(
                "Error: No URL provided. Please provide a 'url' parameter.",
                status=400,
                mimetype="text/plain",
            ),
            400,
        )

    content, status_code, error_message = fetch_url_content(url)

    if error_message:
        if status_code == 404:
            return (
                Response(
                    f"404 Not Found: {error_message}",
                    status=404,
                    mimetype="text/plain",
                ),
                404,
            )
        else:
            return (
                Response(
                    f"Error {status_code}: {error_message}",
                    status=status_code,
                    mimetype="text/plain",
                ),
                status_code,
            )

    def generate_stream():
        yield f"=== Content fetched from: {url} ===\n\n"
        yield f"HTTP Status: {status_code}\n\n"
        yield "=== AI Analysis ===\n\n"

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"Please analyze and summarize the following web content from {url}:\n\n{content[:5000]}",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

        yield "\n\n=== Raw Content Preview (first 500 chars) ===\n\n"
        yield content[:500] if content else "No content"
        yield "\n"

    return Response(stream_with_context(generate_stream()), mimetype="text/plain")


@app.route("/", methods=["GET"])
def index():
    """Simple index page with usage instructions."""
    instructions = """
URL Fetcher Service
==================

Usage:
  GET  /fetch?url=<your-url>
  POST /fetch with JSON body: {"url": "<your-url>"}

Examples:
  curl "http://localhost:5000/fetch?url=https://example.com"
  curl -X POST -H "Content-Type: application/json" -d '{"url":"https://example.com"}' http://localhost:5000/fetch

Error Handling:
  - 400: Invalid URL or missing URL parameter
  - 404: Resource not found at the specified URL
  - 503: Connection error
  - 504: Request timeout
  - 500: General server error
"""
    return Response(instructions, mimetype="text/plain")


if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=5000)
