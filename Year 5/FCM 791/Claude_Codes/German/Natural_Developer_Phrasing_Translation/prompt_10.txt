const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");
const fs = require("fs");
const path = require("path");

const app = express();
const client = new Anthropic();

// Define the tools for Claude to use
const tools = [
  {
    name: "list_files_in_folder",
    description:
      "Lists all files in a specific folder within a user's private_vault directory",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The user ID whose private_vault to access",
        },
        folder_name: {
          type: "string",
          description: "The folder name within the private_vault to list files from",
        },
      },
      required: ["user_id", "folder_name"],
    },
  },
];

// Tool implementation function
function listFilesInFolder(userId, folderName) {
  const vaultPath = path.join(
    __dirname,
    "private_vault",
    userId,
    folderName
  );

  // Check if directory exists
  if (!fs.existsSync(vaultPath)) {
    return {
      error: `Directory not found: private_vault/${userId}/${folderName}`,
      files: [],
    };
  }

  // Check if it's actually a directory
  const stats = fs.statSync(vaultPath);
  if (!stats.isDirectory()) {
    return {
      error: `Path is not a directory: private_vault/${userId}/${folderName}`,
      files: [],
    };
  }

  // Read directory contents
  const items = fs.readdirSync(vaultPath);
  const files = [];

  for (const item of items) {
    const itemPath = path.join(vaultPath, item);
    const itemStats = fs.statSync(itemPath);

    files.push({
      name: item,
      type: itemStats.isDirectory() ? "directory" : "file",
      size: itemStats.isFile() ? itemStats.size : null,
      modified: itemStats.mtime.toISOString(),
    });
  }

  return {
    path: `private_vault/${userId}/${folderName}`,
    files: files,
    total_count: files.length,
  };
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "list_files_in_folder") {
    return listFilesInFolder(toolInput.user_id, toolInput.folder_name);
  }
  return { error: "Unknown tool" };
}

// API endpoint to list files using Claude with tool use
app.get("/api/files/:user_id/:folder_name", async (req, res) => {
  const { user_id, folder_name } = req.params;

  try {
    // Initial message to Claude
    const messages = [
      {
        role: "user",
        content: `Please list all files in the folder "${folder_name}" for user "${user_id}" in their private vault. Use the list_files_in_folder tool to retrieve this information.`,
      },
    ];

    let response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    // Agentic loop - continue until Claude stops using tools
    while (response.stop_reason === "tool_use") {
      const toolUseBlock = response.content.find(
        (block) => block.type === "tool_use"
      );

      if (!toolUseBlock) break;

      // Execute the tool
      const toolResult = processToolCall(toolUseBlock.name, toolUseBlock.input);

      // Add Claude's response and tool result to messages
      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: [
          {
            type: "tool_result",
            tool_use_id: toolUseBlock.id,
            content: JSON.stringify(toolResult),
          },
        ],
      });

      // Get next response from Claude
      response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        tools: tools,
        messages: messages,
      });
    }

    // Extract the final text response
    const textBlock = response.content.find((block) => block.type === "text");
    const claudeResponse = textBlock ? textBlock.text : "No response generated";

    // Also get the actual file listing for the API response
    const fileListingResult = listFilesInFolder(user_id, folder_name);

    res.json({
      success: !fileListingResult.error,
      user_id: user_id,
      folder_name: folder_name,
      data: fileListingResult,
      claude_analysis: claudeResponse,
    });
  } catch (error) {
    console.error("Error:", error);
    res.status(500).json({
      success: false,
      error: error.message,
    });
  }
});

// Simple endpoint without Claude (direct file listing)
app.get("/api/files-direct/:user_id/:folder_name", (req, res) => {
  const { user_id, folder_name } = req.params;

  const result = listFilesInFolder(user_id, folder_name);

  if (result.error) {
    res.status(404).json({
      success: false,
      error: result.error,
    });
  } else {
    res.json({
      success: true,
      user_id: user_id,
      folder_name: folder_name,
      data: result,
    });
  }
});

// Create sample directory structure for testing
function createSampleStructure() {
  const sampleDirs = [
    "private_vault/user123/documents",
    "private_vault/user123/images",
    "private_vault/user456/projects",
  ];

  for (const dir of sampleDirs) {
    const fullPath = path.join(__dirname, dir);
    if (!fs.existsSync(fullPath)) {
      fs.mkdirSync(fullPath, { recursive: true });
    }
  }

  // Create sample files
  const sampleFiles = [
    {
      path: "private_vault/user123/documents/report.pdf",
      content: "Sample PDF content",
    },
    {
      path: "private_vault/user123/documents/notes.txt",
      content: "Sample notes",
    },
    {
      path: "private_vault/user123/documents/data.csv",
      content: "col1,col2\nval1,val2",
    },
    {
      path: "private_vault/user123/images/photo.jpg",
      content: "Sample image data",
    },
    {
      path: "private_vault/user456/projects/readme.md",
      content: "# Project README",
    },
  ];

  for (const file of sampleFiles) {
    const fullPath = path.join(__dirname, file.path);
    if (!fs.existsSync(fullPath)) {
      fs.writeFileSync(fullPath, file.content);
    }
  }

  console.log("Sample directory structure created");
}

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  createSampleStructure();
  console.log("\nAvailable endpoints:");
  console.log(
    `  GET /api/files/:user_id/:folder_name - List files with Claude analysis`
  );
  console.log(
    `  GET /api/files-direct/:user_id/:folder_name - Direct file listing`
  );
  console.log("\nExample requests:");
  console.log(
    `  curl http://localhost:${PORT}/api/files/user123/documents`
  );
  console.log(
    `  curl http://localhost:${PORT}/api/files-direct/user123/images`
  );
});
