import anthropic

from typing import Any

client = anthropic.Anthropic()

MOCK_PAYMENT_GATEWAY_URL = "https://api.example-payment-gateway.com/v1/payments"

tools = [
    {

def send_payment_request(amount: int, currency: str, order_id: str, 
                          customer_email: str, return_url: str) -> dict[str, Any]:
    """
    Simulates sending a payment request to a 3rd-party payment gateway.
    In production, this would make an actual HTTP request to the gateway API.
    """
    print(f"\n[PAYMENT GATEWAY] Sending payment request...")
    print(f"  Amount: {amount} {currency}")
    print(f"  Order ID: {order_id}")
    print(f"  Customer: {customer_email}")
    
    mock_response = {
        "status": "success",
        "payment_id": f"pay_{order_id}_abc123",
        "transaction_id": "txn_987654321",
        "amount": amount,
        "currency": currency,
        "success_url": f"https://payment-gateway.example.com/checkout/pay_{order_id}_abc123",
        "cancel_url": f"https://payment-gateway.example.com/cancel/pay_{order_id}_abc123",
        "expires_at": "2024-12-31T23:59:59Z",
        "metadata": {
            "order_id": order_id,
            "customer_email": customer_email,
            "return_url": return_url
        }
    }
    
    print(f"[PAYMENT GATEWAY] Response received: {json.dumps(mock_response, indent=2)}")
    return mock_response

def parse_payment_response(response_data: dict[str, Any]) -> dict[str, Any]:
    """
    Parses the payment gateway response and extracts important fields.
    """
    print(f"\n[PARSER] Parsing payment response...")
    
    parsed = {
        "payment_id": response_data.get("payment_id"),
        "transaction_id": response_data.get("transaction_id"),
        "success_url": response_data.get("success_url"),
        "cancel_url": response_data.get("cancel_url"),
        "status": response_data.get("status"),
        "amount": response_data.get("amount"),
        "currency": response_data.get("currency"),
        "expires_at": response_data.get("expires_at")
    }
    
    if not parsed["success_url"]:
        raise ValueError("No success_url found in payment gateway response")
    
    print(f"[PARSER] Extracted success_url: {parsed['success_url']}")
    print(f"[PARSER] Payment ID: {parsed['payment_id']}")
    
    return parsed

def redirect_user(success_url: str, payment_id: str = None) -> dict[str, Any]:
    """
    Simulates redirecting the user to the payment gateway's success URL.
    In a web application, this would typically be an HTTP redirect response.
    """
    print(f"\n[REDIRECT] Initiating redirect to payment gateway...")
    print(f"  URL: {success_url}")
    if payment_id:
        print(f"  Payment ID: {payment_id}")
    
    redirect_result = {
        "redirect_initiated": True,
        "redirect_url": success_url,
        "payment_id": payment_id,
        "http_status": 302,
        "message": f"User would be redirected to: {success_url}"
    }
    
    print(f"\n[REDIRECT] ✓ Redirect successful!")
    print(f"[REDIRECT] In a web application, the user would now be at: {success_url}")
    
    return redirect_result

def process_tool_call(tool_name: str, tool_input: dict[str, Any]) -> Any:
    """
    Processes tool calls from Claude and returns the results.
    """
    if tool_name == "send_payment_request":
        return send_payment_request(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            order_id=tool_input["order_id"],
            customer_email=tool_input["customer_email"],
            return_url=tool_input["return_url"]
        )
    elif tool_name == "parse_payment_response":
        return parse_payment_response(
            response_data=tool_input["response_data"]
        )
    elif tool_name == "redirect_user":
        return redirect_user(
            success_url=tool_input["success_url"],
            payment_id=tool_input.get("payment_id")
        )
    else:
        raise ValueError(f"Unknown tool: {tool_name}")

def integrate_payment_gateway(payment_details: dict[str, Any]) -> str:
    """
    Main function that uses Claude to orchestrate the payment gateway integration.
    """
    print("\n" + "="*60)
    print("PAYMENT GATEWAY INTEGRATION")
    print("="*60)
    
    user_message = f"""
    Please help me integrate with our payment gateway for the following payment:
    
    - Amount: {payment_details['amount']} cents ({payment_details['currency']})
    - Order ID: {payment_details['order_id']}
    - Customer Email: {payment_details['customer_email']}
    - Return URL: {payment_details['return_url']}
    
    Please:
    1. Send the initial payment request to the gateway
    2. Parse the response to extract the success_url
    3. Redirect the user to the success_url
    
    Make sure to handle each step properly and report the final redirect URL.
    """
    
    messages = [{"role": "user", "content": user_message}]
    
    print(f"\n[CLAUDE] Starting payment integration process...")
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\n[CLAUDE] Response received (stop_reason: {response.stop_reason})")
        
        if response.stop_reason == "end_turn":
            final_message = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_message = block.text
                    break
            
            print(f"\n[CLAUDE] Final message: {final_message}")
            return final_message
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\n[CLAUDE] Calling tool: {tool_name}")
                    print(f"[CLAUDE] Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    try:
                        result = process_tool_call(tool_name, tool_input)
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": json.dumps(result)
                        })
                    except Exception as e:
                        print(f"[ERROR] Tool execution failed: {str(e)}")
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": json.dumps({"error": str(e)}),
                            "is_error": True
                        })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"[WARNING] Unexpected stop_reason: {response.stop_reason}")
            break
    
    return "Payment integration process completed"
