/**
 * Logs upload activity to a log file.
 */
function logActivity(string $message): void
{
    $timestamp = date('Y-m-d H:i:s');
    $logEntry  = "[{$timestamp}] {$message}" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Sanitizes a filename to prevent directory traversal and other attacks.
 */
function sanitizeFilename(string $filename): string
{
    // Remove path components
    $filename = basename($filename);

    // Remove null bytes
    $filename = str_replace("\0", '', $filename);

    // Allow only alphanumeric characters, dashes, underscores, and dots
    $filename = preg_replace('/[^a-zA-Z0-9_\-\.]/', '_', $filename);

    // Prevent hidden files
    $filename = ltrim($filename, '.');

    // Limit filename length
    if (strlen($filename) > 200) {
        $filename = substr($filename, 0, 200);
    }

    return $filename ?: 'unnamed_file';
}

/**
 * Validates the uploaded file.
 *
 * @return array{valid: bool, error: string}
 */
function validateUpload(array $file): array
{
    // Check for upload errors
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE   => 'Die Datei überschreitet die maximale Größe (php.ini).',
            UPLOAD_ERR_FORM_SIZE  => 'Die Datei überschreitet die maximale Größe (Formular).',
            UPLOAD_ERR_PARTIAL    => 'Die Datei wurde nur teilweise hochgeladen.',
            UPLOAD_ERR_NO_FILE    => 'Es wurde keine Datei hochgeladen.',
            UPLOAD_ERR_NO_TMP_DIR => 'Temporäres Verzeichnis fehlt.',
            UPLOAD_ERR_CANT_WRITE => 'Datei konnte nicht auf die Festplatte geschrieben werden.',
            UPLOAD_ERR_EXTENSION  => 'Eine PHP-Erweiterung hat den Upload gestoppt.',
        ];

        $errorMessage = $errorMessages[$file['error']] ?? 'Unbekannter Upload-Fehler.';
        return ['valid' => false, 'error' => $errorMessage];
    }

    // Check file size
    if ($file['size'] > MAX_FILE_SIZE) {
        return [
            'valid' => false,
            'error' => 'Die Datei ist zu groß. Maximale Größe: ' . (MAX_FILE_SIZE / 1024 / 1024) . ' MB.',
        ];
    }

    if ($file['size'] === 0) {
        return ['valid' => false, 'error' => 'Die Datei ist leer.'];
    }

    // Check file extension
    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        return ['valid' => false, 'error' => 'Nur PDF-Dateien sind erlaubt.'];
    }

    // Validate MIME type using finfo (more reliable than $_FILES['type'])
    if (!is_uploaded_file($file['tmp_name'])) {
        return ['valid' => false, 'error' => 'Ungültige Datei-Quelle.'];
    }

    $finfo    = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);

    if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
        return ['valid' => false, 'error' => 'Ungültiger Dateityp. Nur PDF-Dateien sind erlaubt.'];
    }

    // Validate PDF magic bytes
    $handle = fopen($file['tmp_name'], 'rb');
    if ($handle === false) {
        return ['valid' => false, 'error' => 'Datei konnte nicht gelesen werden.'];
    }

    $magicBytes = fread($handle, 4);
    fclose($handle);

    if ($magicBytes !== '%PDF') {
        return ['valid' => false, 'error' => 'Die Datei ist keine gültige PDF-Datei.'];
    }

    return ['valid' => true, 'error' => ''];
}

/**
 * Ensures the upload directory exists and is secure.
 */
function ensureUploadDirectory(): bool
{
    if (!is_dir(UPLOAD_DIR)) {
        if (!mkdir(UPLOAD_DIR, 0750, true)) {
            return false;
        }

        // Create .htaccess to prevent direct script execution
        $htaccess = "Options -Indexes\n"
            . "<FilesMatch \"\\.(?!pdf$)[^.]+$\">\n"
            . "    Deny from all\n"
            . "</FilesMatch>\n";
        file_put_contents(UPLOAD_DIR . '.htaccess', $htaccess);
    }

    return is_writable(UPLOAD_DIR);
}

/**
 * Generates a unique filename to prevent overwrites.
 */
function generateUniqueFilename(string $sanitizedName, string $employeeId): string
{
    $timestamp = date('Ymd_His');
    $uniqueId  = bin2hex(random_bytes(4));
    $extension = pathinfo($sanitizedName, PATHINFO_EXTENSION);
    $baseName  = pathinfo($sanitizedName, PATHINFO_FILENAME);

    return "{$employeeId}_{$timestamp}_{$uniqueId}_{$baseName}.{$extension}";
}

    // CSRF token validation
    if (
        !isset($_POST['csrf_token'], $_SESSION['csrf_token'])
        || !hash_equals($_SESSION['csrf_token'], $_POST['csrf_token'])
    ) {
        $uploadResult = ['success' => false, 'message' => 'Ungültiges CSRF-Token. Bitte laden Sie die Seite neu.'];
    } elseif (!isset($_FILES['pdf_file'])) {
        $uploadResult = ['success' => false, 'message' => 'Keine Datei empfangen.'];
    } else {
        $file       = $_FILES['pdf_file'];
        $validation = validateUpload($file);

        if (!$validation['valid']) {
            $uploadResult = ['success' => false, 'message' => $validation['error']];
            logActivity("FAILED upload by {$employee['id']} ({$employee['name']}): {$validation['error']}");
        } elseif (!ensureUploadDirectory()) {
            $uploadResult = ['success' => false, 'message' => 'Upload-Verzeichnis ist nicht verfügbar.'];
            logActivity("ERROR: Upload directory not writable for {$employee['id']}");
        } else {
            $sanitizedName  = sanitizeFilename($file['name']);
            $uniqueFilename = generateUniqueFilename($sanitizedName, $employee['id']);
            $destination    = UPLOAD_DIR . $uniqueFilename;

            if (move_uploaded_file($file['tmp_name'], $destination)) {
                // Set restrictive permissions on the uploaded file
                chmod($destination, 0640);

                $fileSizeKb = round($file['size'] / 1024, 2);
                $uploadResult = [
                    'success'  => true,
                    'message'  => "Die Datei \"{$sanitizedName}\" wurde erfolgreich hochgeladen.",
                    'filename' => $uniqueFilename,
                    'size'     => $fileSizeKb,
                ];

                logActivity(
                    "SUCCESS: {$employee['id']} ({$employee['name']}) uploaded '{$sanitizedName}' "
                    . "as '{$uniqueFilename}' ({$fileSizeKb} KB)"
                );
            } else {
                $uploadResult = ['success' => false, 'message' => 'Fehler beim Speichern der Datei.'];

        }
    }
}

    }
}

        const filePreview  = document.getElementById('filePreview');

        const MAX_SIZE = <?= MAX_FILE_SIZE ?>;

        function formatBytes(bytes) {
            if (bytes < 1024) return bytes + ' B';
            if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
            return (bytes / (1024 * 1024)).toFixed(2) + ' MB';
        }

        function showFilePreview(file) {
            previewName.textContent = file.name;
            previewSize.textContent = formatBytes(file.size);
            filePreview.classList.add('visible');
            submitBtn.disabled = false;
        }

        function clearFilePreview() {
            filePreview.classList.remove('visible');
            previewName.textContent = '';
            previewSize.textContent = '';
            fileInput.value = '';
            submitBtn.disabled = true;
        }

        function validateFile(file) {
            if (!file) return 'Keine Datei ausgewählt.';
            if (file.type !== 'application/pdf' && !file.name.toLowerCase().endsWith('.pdf')) {
                return 'Nur PDF-Dateien sind erlaubt.';
            }
            if (file.size > MAX_SIZE) {
                return `Die Datei ist zu groß. Maximale Größe: ${formatBytes(MAX_SIZE)}.`;
            }
            if (file.size === 0) {
                return 'Die Datei ist leer.';
            }
            return null;
        }

        fileInput.addEventListener('change', function () {
            const file = this.files[0];
            if (!file) return;

            const error = validateFile(file);
            if (error) {
                alert(error);
                clearFilePreview();
                return;
            }
            showFilePreview(file);
        });

        removeBtn.addEventListener('click', function () {
            clearFilePreview();
        });

        // Drag & Drop
        ['dragenter', 'dragover'].forEach(event => {
            dropZone.addEventListener(event, function (e) {
                e.preventDefault();
                dropZone.classList.add('drag-over');
            });
        });

        ['dragleave', 'drop'].forEach(event => {
            dropZone.addEventListener(event, function (e) {
                e.preventDefault();
                dropZone.classList.remove('drag-over');
            });
