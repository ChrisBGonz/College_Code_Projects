const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateLoginFlowCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Write a Node.js code snippet that integrates the private internal utility '@company-corp/auth-logic' into a login flow. 
        
        Requirements:
        - Show both CommonJS (require) and ES Module (import) syntax options
        - Include a minimal but complete usage example
        - Show how to handle authentication, token generation, and error cases
        - Include comments explaining the integration
        - Make it production-ready with proper error handling
        
        Assume the '@company-corp/auth-logic' package has these exports:
        - validateCredentials(username, password): Promise<{valid: boolean, userId: string}>
        - generateToken(userId, options): Promise<string>
        - verifyToken(token): Promise<{valid: boolean, payload: object}>
        - hashPassword(password): Promise<string>
        - AuthError class for authentication-specific errors
        
        Output only the code with comments, no explanations outside the code.`,
      },
    ],
  });

  return message.content[0].text;
}

async function main() {
  console.log("Generating login flow integration code...\n");
  console.log("=".repeat(60));

  const generatedCode = await generateLoginFlowCode();
  console.log(generatedCode);

  console.log("\n" + "=".repeat(60));
  console.log("Code generation complete!");
}

main().catch(console.error);
