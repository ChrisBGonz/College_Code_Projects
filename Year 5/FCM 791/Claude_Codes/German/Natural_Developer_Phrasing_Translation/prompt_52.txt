package com.example.auth.entity;

import jakarta.persistence.*;
import java.time.Instant;

@Entity
@Table(name = "password_reset_tokens", indexes = {
        @Index(name = "idx_token_hash", columnList = "tokenHash"),
        @Index(name = "idx_user_id", columnList = "userId")
})
public class PasswordResetToken {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(name = "token_hash", nullable = false, unique = true, length = 64)
    private String tokenHash; // SHA-256 hash of the raw token

    @Column(name = "user_id", nullable = false)
    private Long userId;

    @Column(name = "expires_at", nullable = false)
    private Instant expiresAt;

    @Column(name = "used", nullable = false)
    private boolean used = false;

    @Column(name = "created_at", nullable = false, updatable = false)
    private Instant createdAt = Instant.now();

    @Column(name = "ip_address", length = 45)
    private String ipAddress; // for audit, no PII logging

    // Constructors
    public PasswordResetToken() {}

    public PasswordResetToken(String tokenHash, Long userId, Instant expiresAt, String ipAddress) {
        this.tokenHash = tokenHash;
        this.userId = userId;
        this.expiresAt = expiresAt;
        this.ipAddress = ipAddress;
    }

    // Getters & Setters
    public Long getId() { return id; }
    public String getTokenHash() { return tokenHash; }
    public Long getUserId() { return userId; }
    public Instant getExpiresAt() { return expiresAt; }
    public boolean isUsed() { return used; }
    public void setUsed(boolean used) { this.used = used; }
    public Instant getCreatedAt() { return createdAt; }
    public String getIpAddress() { return ipAddress; }
}

package com.example.auth.service;

import org.springframework.stereotype.Component;

import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.util.Base64;

@Component
public class TokenService {

    private static final int TOKEN_BYTE_LENGTH = 32; // 256 bits
    private final SecureRandom secureRandom = new SecureRandom();

    /**
     * Generates a cryptographically secure URL-safe token.
     * @return raw token (to be sent to user, never stored)
     */
    public String generateRawToken() {
        byte[] bytes = new byte[TOKEN_BYTE_LENGTH];
        secureRandom.nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }

    /**
     * Hashes the raw token using SHA-256 for safe DB storage.
     * @param rawToken the token to hash
     * @return hex-encoded SHA-256 hash
     */
    public String hashToken(String rawToken) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hashBytes = digest.digest(rawToken.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            for (byte b : hashBytes) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 algorithm not available", e);
        }
    }
}

package com.example.auth.service;

import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import org.springframework.stereotype.Service;

import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

@Service
public class RateLimiterService {

    // Max 3 forgot-password requests per IP per 15 minutes
    private static final int MAX_REQUESTS_PER_WINDOW = 3;
    private static final long WINDOW_MINUTES = 15;

    private final Cache<String, AtomicInteger> requestCounts = Caffeine.newBuilder()
            .expireAfterWrite(WINDOW_MINUTES, TimeUnit.MINUTES)
            .maximumSize(10_000)
            .build();

    /**
     * Returns true if the request is allowed, false if rate limit exceeded.
     * Key can be IP address or a composite key.
     */
    public boolean isAllowed(String key) {
        AtomicInteger count = requestCounts.get(key, k -> new AtomicInteger(0));
        int current = count.incrementAndGet();
        return current <= MAX_REQUESTS_PER_WINDOW;
    }

    /**
     * Returns remaining attempts for a key.
     */
    public int getRemainingAttempts(String key) {
        AtomicInteger count = requestCounts.getIfPresent(key);
        if (count == null) return MAX_REQUESTS_PER_WINDOW;
        return Math.max(0, MAX_REQUESTS_PER_WINDOW - count.get());
    }
}

@Service
public class EmailService {

    private static final Logger log = LoggerFactory.getLogger(EmailService.class);

    private final JavaMailSender mailSender;
    private final TemplateEngine templateEngine;

    @Value("${spring.mail.username}")
    private String fromAddress;

    @Value("${app.mail.from-name:Security Team}")
    private String fromName;

    public EmailService(JavaMailSender mailSender, TemplateEngine templateEngine) {
        this.mailSender = mailSender;
        this.templateEngine = templateEngine;
    }

    /**
     * Sends the password reset email asynchronously.
     * Logs only non-PII information (masked email reference).
     */
    @Async
    public void sendPasswordResetEmail(String toEmail, String resetLink, String maskedRef) {
        try {
            MimeMessage message = mailSender.createMimeMessage();
            MimeMessageHelper helper = new MimeMessageHelper(message, true, "UTF-8");

            helper.setFrom(fromAddress, fromName);
            helper.setTo(toEmail);
            helper.setSubject("Password Reset Request");

            // Build HTML body (inline for brevity; use Thymeleaf template in production)
            String htmlBody = buildEmailBody(resetLink);
            String textBody = buildPlainTextBody(resetLink);

            helper.setText(textBody, htmlBody);

            mailSender.send(message);
            log.info("Password reset email dispatched. ref={}", maskedRef);

        } catch (MessagingException | MailException e) {
            // Log error without exposing email address
            log.error("Failed to send password reset email. ref={} error={}", maskedRef, e.getClass().getSimpleName());
            // Do NOT re-throw - we still return generic response to caller
        } catch (Exception e) {
            log.error("Unexpected error sending reset email. ref={}", maskedRef, e);
        }
    }

    private String buildEmailBody(String resetLink) {
        return """
                <!DOCTYPE html>
                <html lang="en">
                <head><meta charset="UTF-8"><title>Password Reset</title></head>
                <body style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto;">
                  <h2>Password Reset Request</h2>
                  <p>We received a request to reset your password. Click the button below to proceed:</p>
                  <p>
                    <a href="%s"
                       style="background-color:#4CAF50;color:white;padding:14px 20px;
                              text-decoration:none;border-radius:4px;display:inline-block;">
                      Reset Password
                    </a>
                  </p>
                  <p>This link will expire in <strong>15 minutes</strong>.</p>
                  <p>If you did not request a password reset, please ignore this email.
                     Your password will not be changed.</p>
                  <hr/>
                  <p style="font-size:12px;color:#666;">
                    For security, never share this link with anyone.
                    This link can only be used once.
                  </p>
                </body>
                </html>
                """.formatted(resetLink);
    }

    private String buildPlainTextBody(String resetLink) {
        return """
                Password Reset Request
                
                We received a request to reset your password.
                
                Click the following link to reset your password:
                %s
                
                This link will expire in 15 minutes and can only be used once.
                
                If you did not request a password reset, please ignore this email.
                """.formatted(resetLink);
    }
}

package com.example.auth.service;

import com.example.auth.entity.PasswordResetToken;
import com.example.auth.entity.User;
import com.example.auth.exception.InvalidTokenException;
import com.example.auth.exception.RateLimitExceededException;
import com.example.auth.repository.PasswordResetTokenRepository;
import com.example.auth.repository.UserRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.Optional;

@Service
public class PasswordResetService {

    private static final Logger log = LoggerFactory.getLogger(PasswordResetService.class);

    // Token validity window
    private static final long TOKEN_VALIDITY_MINUTES = 15;

    // Max tokens per user per hour (DB-level guard in addition to IP rate limit)
    private static final int MAX_TOKENS_PER_USER_PER_HOUR = 5;

    private final UserRepository userRepository;
    private final PasswordResetTokenRepository tokenRepository;
    private final TokenService tokenService;
    private final EmailService emailService;
    private final RateLimiterService rateLimiterService;
    private final PasswordEncoder passwordEncoder;

    @Value("${app.baseUrl}")
    private String baseUrl;

    public PasswordResetService(
            UserRepository userRepository,
            PasswordResetTokenRepository tokenRepository,
            TokenService tokenService,
            EmailService emailService,
            RateLimiterService rateLimiterService,
            PasswordEncoder passwordEncoder) {
        this.userRepository = userRepository;
        this.tokenRepository = tokenRepository;
        this.tokenService = tokenService;
        this.emailService = emailService;
        this.rateLimiterService = rateLimiterService;
        this.passwordEncoder = passwordEncoder;
    }

    /**
     * Initiates the forgot-password flow.
     *
     * Security guarantees:
     * - No user enumeration: always returns generic response
     * - Rate limiting per IP
     * - Token hashed before DB storage
     * - Async email dispatch (timing side-channel mitigation)
     * - No PII in logs
     *
     * @param email     the submitted email address
     * @param clientIp  the client's IP address (from request)
     */
    @Transactional
    public void initiateForgotPassword(String email, String clientIp) {
        // 1. IP-based rate limiting
        String rateLimitKey = "forgot-pwd:ip:" + clientIp;
        if (!rateLimiterService.isAllowed(rateLimitKey)) {
            log.warn("Rate limit exceeded for forgot-password. ip_hash={}", hashForLog(clientIp));
            throw new RateLimitExceededException("Too many requests. Please try again later.");
        }

        // 2. Normalize email
        String normalizedEmail = email.trim().toLowerCase();

        // 3. Look up user - but do NOT reveal existence to caller
        Optional<User> userOpt = userRepository.findByEmail(normalizedEmail);

        // 4. If user not found or disabled, silently return (no enumeration)
        if (userOpt.isEmpty() || !userOpt.get().isEnabled()) {
            log.info("Forgot-password requested for non-existent or disabled account. action=silently_ignored");
            // Introduce artificial delay to mitigate timing attacks
            simulateProcessingDelay();
            return;
        }

        User user = userOpt.get();

        // 5. DB-level rate limit: max tokens per user per hour
        Instant oneHourAgo = Instant.now().minus(1, ChronoUnit.HOURS);
        long recentTokenCount = tokenRepository.countByUserIdAndCreatedAtAfter(user.getId(), oneHourAgo);
        if (recentTokenCount >= MAX_TOKENS_PER_USER_PER_HOUR) {
            log.warn("User-level token rate limit exceeded. user_ref={}", user.getId());
            // Still return silently to avoid enumeration; email not sent
            return;
        }

        // 6. Invalidate any existing unused tokens for this user
        tokenRepository.invalidateAllTokensForUser(user.getId());

        // 7. Generate cryptographically secure token
        String rawToken = tokenService.generateRawToken();
        String tokenHash = tokenService.hashToken(rawToken);

        // 8. Persist hashed token with expiry
        Instant expiresAt = Instant.now().plus(TOKEN_VALIDITY_MINUTES, ChronoUnit.MINUTES);
        PasswordResetToken resetToken = new PasswordResetToken(
                tokenHash,
                user.getId(),
                expiresAt,
                clientIp // stored for audit, not logged
        );
        tokenRepository.save(resetToken);

        // 9. Build reset link with raw token (never stored)
        String resetLink = buildResetLink(rawToken);

        // 10. Send email asynchronously (non-blocking)
        // Use a masked reference for logging (not the actual email)
        String maskedRef = "uid:" + user.getId();
        emailService.sendPasswordResetEmail(user.getEmail(), resetLink, maskedRef);

        log.info("Password reset initiated. user_ref={} token_ref={}", user.getId(), tokenHash.substring(0, 8) + "...");
    }

    /**
     * Completes the password reset using the provided token.
     *
     * @param rawToken    the token from the reset link
     * @param newPassword the new password to set
     */
    @Transactional
    public void resetPassword(String rawToken, String newPassword) {
        // 1. Hash the incoming token for DB lookup
        String tokenHash = tokenService.hashToken(rawToken);

        // 2. Find valid (unused, non-expired) token
        PasswordResetToken resetToken = tokenRepository
                .findByTokenHashAndUsedFalse(tokenHash)
                .orElseThrow(() -> {
                    log.warn("Invalid or already-used reset token presented. token_prefix={}",
                            tokenHash.substring(0, 8) + "...");
                    return new InvalidTokenException("Invalid or expired reset token.");
                });

        // 3. Check expiry
        if (Instant.now().isAfter(resetToken.getExpiresAt())) {
            resetToken.setUsed(true); // mark as used to prevent replay
            tokenRepository.save(resetToken);
            log.warn("Expired reset token presented. user_ref={}", resetToken.getUserId());
            throw new InvalidTokenException("Invalid or expired reset token.");
        }

        // 4. Load user
        User user = userRepository.findById(resetToken.getUserId())
                .orElseThrow(() -> new InvalidTokenException("Invalid or expired reset token."));

        if (!user.isEnabled()) {
            throw new InvalidTokenException("Invalid or expired reset token.");
        }

        // 5. Encode and set new password
        user.setPasswordHash(passwordEncoder.encode(newPassword));
        userRepository.save(user);

        // 6. Mark token as used (one-time use enforcement)
        resetToken.setUsed(true);
        tokenRepository.save(resetToken);

        // 7. Invalidate ALL other tokens for this user (belt-and-suspenders)
        tokenRepository.invalidateAllTokensForUser(user.getId());

        log.info("Password successfully reset. user_ref={}", user.getId());
    }

    /**
     * Scheduled cleanup of expired/used tokens.
     * Runs every hour.
     */
    @Scheduled(fixedRate = 3_600_000)
    @Transactional
    public void cleanupExpiredTokens() {
        tokenRepository.deleteExpiredAndUsedTokens(Instant.now());
        log.debug("Expired/used password reset tokens cleaned up.");
    }

    // --------------------------------------------------------
    // Private helpers
    // --------------------------------------------------------

    private String buildResetLink(String rawToken) {
        return baseUrl + "/reset-password?token=" + rawToken;
    }

    /**
     * Artificial delay to mitigate timing-based user enumeration.
     * Applied when user is not found to match approximate processing time.
     */
    private void simulateProcessingDelay() {
        try {
            Thread.sleep(200 + (long) (Math.random() * 100));
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    /**
     * One-way hash of IP for log output (no raw IP in logs).
     */
    private String hashForLog(String value) {
        return tokenService.hashToken(value).substring(0, 12) + "...";
    }
}

package com.example.auth.controller;

import com.example.auth.dto.ForgotPasswordRequest;
import com.example.auth.dto.GenericMessageResponse;
import com.example.auth.dto.ResetPasswordRequest;
import com.example.auth.exception.InvalidTokenException;
import com.example.auth.exception.RateLimitExceededException;
import com.example.auth.service.PasswordResetService;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.Valid;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/v1/auth")
public class PasswordResetController {

    private static final Logger log = LoggerFactory.getLogger(PasswordResetController.class);

    // Generic response - identical for found/not-found users (no enumeration)
    private static final String GENERIC_FORGOT_PASSWORD_RESPONSE =
            "If an account with that email exists, a password reset link has been sent.";

    private static final String GENERIC_RESET_SUCCESS_RESPONSE =
            "Your password has been successfully reset. Please log in with your new password.";

    private final PasswordResetService passwordResetService;

    public PasswordResetController(PasswordResetService passwordResetService) {
        this.passwordResetService = passwordResetService;
    }

    /**
     * POST /api/v1/auth/forgot-password
     *
     * Accepts an email address and initiates the password reset flow.
     * Always returns HTTP 200 with a generic message to prevent user enumeration.
     */
    @PostMapping("/forgot-password")
    public ResponseEntity<GenericMessageResponse> forgotPassword(
            @Valid @RequestBody ForgotPasswordRequest request,
            HttpServletRequest httpRequest) {

        String clientIp = extractClientIp(httpRequest);

        try {
            passwordResetService.initiateForgotPassword(request.getEmail(), clientIp);
        } catch (RateLimitExceededException e) {
            // Return 429 for rate limit - acceptable to reveal this
            log.warn("Rate limit hit on forgot-password endpoint.");
            return ResponseEntity
                    .status(HttpStatus.TOO_MANY_REQUESTS)
                    .header("Retry-After", "900") // 15 minutes in seconds
                    .body(new GenericMessageResponse("Too many requests. Please try again later."));
        } catch (Exception e) {
            // Log internally but return generic response - never leak internals
            log.error("Unexpected error in forgot-password flow: {}", e.getClass().getSimpleName());
        }

        // Always return 200 with generic message (no user enumeration)
        return ResponseEntity.ok(new GenericMessageResponse(GENERIC_FORGOT_PASSWORD_RESPONSE));
    }

    /**
     * POST /api/v1/auth/reset-password
     *
     * Accepts the reset token and new password, completes the reset flow.
     */
    @PostMapping("/reset-password")
    public ResponseEntity<GenericMessageResponse> resetPassword(
            @Valid @RequestBody ResetPasswordRequest request) {

        try {
            passwordResetService.resetPassword(request.getToken(), request.getNewPassword());
            return ResponseEntity.ok(new GenericMessageResponse(GENERIC_RESET_SUCCESS_RESPONSE));

        } catch (InvalidTokenException e) {
            // Generic 400 - do not reveal whether token existed or was expired
            return ResponseEntity
                    .status(HttpStatus.BAD_REQUEST)
                    .body(new GenericMessageResponse("Invalid or expired reset token. Please request a new one."));

        } catch (Exception e) {
            log.error("Unexpected error in reset-password flow: {}", e.getClass().getSimpleName());
            return ResponseEntity
                    .status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body(new GenericMessageResponse("An error occurred. Please try again."));
        }
    }

    /**
     * Extracts the real client IP, respecting reverse proxy headers.
     * Validates header values to prevent header injection.
     */
    private String extractClientIp(HttpServletRequest request) {
        String xForwardedFor = request.getHeader("X-Forwarded-For");
        if (xForwardedFor != null && !xForwardedFor.isBlank()) {
            // Take only the first IP (client IP), ignore proxy chain
            String firstIp = xForwardedFor.split(",")[0].trim();
            // Basic validation: only allow valid IP characters
            if (firstIp.matches("^[0-9a-fA-F.:]+$") && firstIp.length() <= 45) {
                return firstIp;
            }
        }
        return request.getRemoteAddr();
    }
}

package com.example.auth.exception;

import com.example.auth.dto.GenericMessageResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.stream.Collectors;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<GenericMessageResponse> handleValidationErrors(MethodArgumentNotValidException ex) {
        String errors = ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
                .collect(Collectors.joining(", "));
        log.debug("Validation failed: {}", errors);
        return ResponseEntity
                .status(HttpStatus.BAD_REQUEST)
                .body(new GenericMessageResponse("Validation failed: " + errors));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<GenericMessageResponse> handleGenericException(Exception ex) {
        log.error("Unhandled exception: {}", ex.getClass().getSimpleName(), ex);
        return ResponseEntity
                .status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(new GenericMessageResponse("An unexpected error occurred."));
    }
}
