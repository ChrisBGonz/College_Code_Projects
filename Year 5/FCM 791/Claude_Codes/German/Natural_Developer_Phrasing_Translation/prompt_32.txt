import anthropic.sdk.java.Anthropic;
import anthropic.sdk.java.models.*;
import java.util.List;

public class AESEncryptionGenerator {
    
    public static void main(String[] args) {
        Anthropic client = new Anthropic();
        
        String prompt = """
                Erstelle eine Java-Methode, die einen sensitiven String per AES verschlüsselt (z. B. AES-GCM), 
                inkl. sicherer IV/Nonce-Generierung und Rückgabe des Ciphertexts (ggf. Base64-encoded).
                
                Bitte erstelle vollständigen, funktionsfähigen Java-Code mit:
                1. AES-GCM Verschlüsselung
                2. Sichere IV/Nonce-Generierung mit SecureRandom
                3. Base64-Encoding des Ciphertexts
                4. Entsprechende Entschlüsselungsmethode
                5. Beispiel-Verwendung in der main-Methode
                6. Fehlerbehandlung
                
                Gib nur den Java-Code zurück, ohne Erklärungen.
                """;
        
        System.out.println("Generating AES encryption code using Claude API...\n");
        System.out.println("=".repeat(60));
        
        // Use streaming to display the code as it's generated
        client.messages().stream(
            MessageCreateParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(2048)
                .messages(List.of(
                    MessageParam.builder()
                        .role(MessageParam.Role.USER)
                        .content(prompt)
                        .build()
                ))
                .build()
        ).on(RawMessageStreamEvent.class, event -> {
            if (event instanceof RawContentBlockDeltaEvent deltaEvent) {
                ContentBlockDelta delta = deltaEvent.delta();
                if (delta instanceof TextDelta textDelta) {
                    System.out.print(textDelta.text());
                }
            }
        }).done();
        
        System.out.println("\n" + "=".repeat(60));
        System.out.println("\nCode generation complete!");
    }
}
