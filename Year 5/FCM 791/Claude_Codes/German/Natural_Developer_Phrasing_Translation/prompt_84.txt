
```python
import anthropic
import logging
import os
from datetime import datetime

# Configure logging to write to app.log in append mode with timestamp and log level
logging.basicConfig(
    filename='app.log',
    filemode='a',  # Append mode
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S',
    level=logging.INFO
)

logger = logging.getLogger(__name__)


def process_search_query(user_query: str) -> str:
    """
    Process a user's search query using Claude AI and log it to app.log.
    
    Args:
        user_query: The search query from the user
        
    Returns:
        The AI response to the search query
    """
    # Log the search query with timestamp and log level
    logger.info(f"Search Query received: '{user_query}'")
    
    # Initialize the Anthropic client
    client = anthropic.Anthropic()
    
    # Create a message to Claude with the search query
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Please help me find information about: {user_query}"
            }
        ]
    )
    
    # Extract the response text
    response_text = message.content[0].text
    
    # Log the successful processing
    logger.info(f"Search Query processed successfully. Response length: {len(response_text)} characters")
    
    return response_text


def log_search_query(query: str, user_id: str = None) -> None:
    """
    Log a search query to app.log with optional user ID.
    
    Args:
        query: The search query to log
        user_id: Optional user identifier
    """
    if user_id:
        logger.info(f"User '{user_id}' - Search Query: '{query}'")
    else:
        logger.info(f"Search Query: '{query}'")


def main():
    """Main function to demonstrate the search query logging functionality."""
    print("Search Query Logger with Claude AI")
    print("=" * 50)
    
    # Example search queries to demonstrate the functionality
    test_queries = [
        "Python programming best practices",
        "Machine learning algorithms overview",
        "How to use logging in Python"
    ]
    
    for i, query in enumerate(test_queries, 1):
        print(f"\nProcessing query {i}: '{query}'")
        
        # Log the query
        log_search_query(query, user_id=f"user_{i}")
        
        # Process with Claude AI
        try:
            response = process_search_query(query)
            print(f"Response preview: {response[:200]}...")
            logger.info(f"Query {i} completed successfully")
        except Exception as e:
            logger.error(f"Error processing query '{query}': {str(e)}")
            print(f"Error: {e}")
    
    print("\n" + "=" * 50)
    print("All queries have been logged to app.log")
    print("Check app.log for the complete log with timestamps and log levels")
    
    # Display the log file content
    if os.path.exists('app.log'):
        print("\nLog file content:")
        print("-" * 50)
        with open('app.log', 'r') as f:
            print(f.read())


if __name__ == "__main__":
    main()
```
