
using System;
using System.Security.Cryptography;
using System.Text;

class RsaKeyPairGenerator
{
    static void Main()
    {
        // RSA-Schlüsselpaar generieren (2048 Bit)
        using (RSA rsa = RSA.Create(2048))
        {
            // Private Key exportieren (PKCS#8 Format)
            byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
            string privateKeyBase64 = Convert.ToBase64String(privateKeyBytes);

            // Public Key exportieren (SubjectPublicKeyInfo Format)
            byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
            string publicKeyBase64 = Convert.ToBase64String(publicKeyBytes);

            Console.WriteLine("=== RSA Public Key ===");
            Console.WriteLine($"-----BEGIN PUBLIC KEY-----");
            Console.WriteLine(FormatBase64(publicKeyBase64));
            Console.WriteLine($"-----END PUBLIC KEY-----");
            Console.WriteLine();

            Console.WriteLine("=== RSA Private Key ===");
            Console.WriteLine($"-----BEGIN PRIVATE KEY-----");
            Console.WriteLine(FormatBase64(privateKeyBase64));
            Console.WriteLine($"-----END PRIVATE KEY-----");
            Console.WriteLine();

            // Beispiel: Digitale Signatur erstellen
            string message = "Dies ist eine Testnachricht.";
            byte[] messageBytes = Encoding.UTF8.GetBytes(message);

            byte[] signature = rsa.SignData(
                messageBytes,
                HashAlgorithmName.SHA256,
                RSASignaturePadding.Pkcs1
            );

            Console.WriteLine("=== Digitale Signatur ===");
            Console.WriteLine($"Nachricht: {message}");
            Console.WriteLine($"Signatur (Base64): {Convert.ToBase64String(signature)}");
            Console.WriteLine();

            // Signatur verifizieren
            bool isValid = rsa.VerifyData(
                messageBytes,
                signature,
                HashAlgorithmName.SHA256,
                RSASignaturePadding.Pkcs1
            );

            Console.WriteLine($"Signatur gültig: {isValid}");

            // Schlüsselparameter anzeigen
            RSAParameters parameters = rsa.ExportParameters(false);
            Console.WriteLine();
            Console.WriteLine("=== Schlüsselparameter ===");
            Console.WriteLine($"Schlüsselgröße: {rsa.KeySize} Bit");
            Console.WriteLine($"Modulus (n): {Convert.ToBase64String(parameters.Modulus)}");
            Console.WriteLine($"Exponent (e): {Convert.ToBase64String(parameters.Exponent)}");
        }

        // Schlüssel aus exportierten Bytes wiederherstellen
        Console.WriteLine();
        Console.WriteLine("=== Schlüssel wiederherstellen ===");
        DemonstrateKeyRestoration();
    }

    static void DemonstrateKeyRestoration()
    {
        // Neues Schlüsselpaar generieren
        byte[] privateKeyBytes;
        byte[] publicKeyBytes;

        using (RSA originalRsa = RSA.Create(2048))
        {
            privateKeyBytes = originalRsa.ExportPkcs8PrivateKey();
            publicKeyBytes = originalRsa.ExportSubjectPublicKeyInfo();
        }

        // Private Key wiederherstellen
        using (RSA rsaFromPrivate = RSA.Create())
        {
            rsaFromPrivate.ImportPkcs8PrivateKey(privateKeyBytes, out int bytesRead);
            Console.WriteLine($"Private Key wiederhergestellt. Bytes gelesen: {bytesRead}");
            Console.WriteLine($"Schlüsselgröße: {rsaFromPrivate.KeySize} Bit");
        }

        // Public Key wiederherstellen
        using (RSA rsaFromPublic = RSA.Create())
        {
            rsaFromPublic.ImportSubjectPublicKeyInfo(publicKeyBytes, out int bytesRead);
            Console.WriteLine($"Public Key wiederhergestellt. Bytes gelesen: {bytesRead}");
            Console.WriteLine($"Schlüsselgröße: {rsaFromPublic.KeySize} Bit");
        }
    }

    // Hilfsmethode: Base64-String in 64-Zeichen-Zeilen formatieren
    static string FormatBase64(string base64, int lineLength = 64)
    {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < base64.Length; i += lineLength)
        {
            int length = Math.Min(lineLength, base64.Length - i);
            sb.AppendLine(base64.Substring(i, length));
        }
        return sb.ToString().TrimEnd();
    }
}
