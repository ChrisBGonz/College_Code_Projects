// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader) {
    return null;
  }

  const parts = authHeader.split(" ");
  if (parts.length !== 2 || parts[0] !== "Bearer") {
    return null;
  }

  return parts[1];
}

// Validate token and return user info
function validateToken(token) {
  if (!token) {
    return null;
  }

  return validTokens.get(token) || null;
}

// Use Claude to analyze and enhance the response with AI insights
async function generateAIInsights(userData, conversationHistory) {
  const userContext = JSON.stringify(userData, null, 2);

  conversationHistory.push({
    role: "user",
    content: `Analyze this user data and provide a brief security assessment and personalized insights. Keep it concise and professional. User data: ${userContext}`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a security and data analysis assistant. Analyze user data and provide brief, professional insights about account security and usage patterns. Be concise and helpful.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Parse request body
function parseBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });
    req.on("end", () => {
      try {
        resolve(body ? JSON.parse(body) : {});
      } catch {
        resolve({});
      }
    });
    req.on("error", reject);
  });
}

// Send JSON response
function sendResponse(res, statusCode, data) {
  res.writeHead(statusCode, {
    "Content-Type": "application/json",
    "X-Request-ID": crypto.randomUUID(),
  });
  res.end(JSON.stringify(data, null, 2));
}

// Conversation history store (in production, use Redis or similar)
const conversationHistories = new Map();

// Main request handler
async function handleRequest(req, res) {
  const url = new URL(req.url, `http://${req.headers.host}`);
  const pathname = url.pathname;
  const method = req.method;

  // CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader(
    "Access-Control-Allow-Methods",
    "GET, POST, PUT, DELETE, OPTIONS"
  );
  res.setHeader(
    "Access-Control-Allow-Headers",
    "Content-Type, Authorization"
  );

  if (method === "OPTIONS") {
    res.writeHead(204);
    res.end();
    return;
  }

  // Health check endpoint (public)
  if (pathname === "/health" && method === "GET") {
    sendResponse(res, 200, {
      status: "healthy",
      timestamp: new Date().toISOString(),
      version: "1.0.0",
    });
    return;
  }

  // Token validation for protected routes
  const authHeader = req.headers["authorization"];
  const token = extractBearerToken(authHeader);
  const userInfo = validateToken(token);

  // Protected routes
  if (pathname === "/api/user/profile" && method === "GET") {
    if (!userInfo) {
      sendResponse(res, 401, {
        error: "Unauthorized",
        message: "Invalid or missing Bearer token",
        hint: "Provide a valid Bearer token in the Authorization header",
      });
      return;
    }

    const userData = privateUserData.get(userInfo.userId);
    if (!userData) {
      sendResponse(res, 404, {
        error: "Not Found",
        message: "User data not found",
      });
      return;
    }

    sendResponse(res, 200, {
      success: true,
      data: userData,
      timestamp: new Date().toISOString(),
    });
    return;
  }

  // AI-enhanced profile endpoint with multi-turn conversation
  if (pathname === "/api/user/profile/insights" && method === "GET") {
    if (!userInfo) {
      sendResponse(res, 401, {
        error: "Unauthorized",
        message: "Invalid or missing Bearer token",
      });
      return;
    }

    const userData = privateUserData.get(userInfo.userId);
    if (!userData) {
      sendResponse(res, 404, {
        error: "Not Found",
        message: "User data not found",
      });
      return;
    }

    // Get or create conversation history for this user
    if (!conversationHistories.has(userInfo.userId)) {
      conversationHistories.set(userInfo.userId, []);
    }
    const conversationHistory = conversationHistories.get(userInfo.userId);

    const insights = await generateAIInsights(userData, conversationHistory);

    sendResponse(res, 200, {
      success: true,
      data: userData,
      aiInsights: insights,
      conversationTurns: conversationHistory.length / 2,
      timestamp: new Date().toISOString(),
    });
    return;
  }

  // Multi-turn conversation endpoint for user queries
  if (pathname === "/api/user/chat" && method === "POST") {
    if (!userInfo) {
      sendResponse(res, 401, {
        error: "Unauthorized",
        message: "Invalid or missing Bearer token",
      });
      return;
    }

    const body = await parseBody(req);
    const userMessage = body.message;

    if (!userMessage) {
      sendResponse(res, 400, {
        error: "Bad Request",
        message: "Message is required in request body",
      });
      return;
    }

    const userData = privateUserData.get(userInfo.userId);

    // Get or create conversation history
    if (!conversationHistories.has(userInfo.userId)) {
      conversationHistories.set(userInfo.userId, []);
    }
    const conversationHistory = conversationHistories.get(userInfo.userId);

    // Add user message to history
    conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    // Generate AI response with context
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1000,
      system: `You are a helpful assistant for ${userData.username}. You have access to their profile information and can help them with account-related questions. Be helpful, professional, and concise. User context: ${JSON.stringify({ username: userData.username, role: userData.role, email: userData.email })}`,
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;
    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    sendResponse(res, 200, {
      success: true,
      message: assistantMessage,
      conversationTurns: conversationHistory.length / 2,
      timestamp: new Date().toISOString(),
    });
    return;
  }

  // Clear conversation history
  if (pathname === "/api/user/chat/clear" && method === "DELETE") {
    if (!userInfo) {
      sendResponse(res, 401, {
        error: "Unauthorized",
        message: "Invalid or missing Bearer token",
      });
      return;
    }

    conversationHistories.delete(userInfo.userId);

    sendResponse(res, 200, {
      success: true,
      message: "Conversation history cleared",
      timestamp: new Date().toISOString(),
    });
    return;
  }

  // List available tokens (demo endpoint - remove in production)
  if (pathname === "/api/demo/tokens" && method === "GET") {
    const tokens = [];
    for (const [token, user] of validTokens.entries()) {
      tokens.push({
        token,
        userId: user.userId,
        username: user.username,
        role: user.role,
      });
    }

    sendResponse(res, 200, {
      message: "Demo tokens for testing (remove in production)",
      tokens,
      usage: "Add to Authorization header as: Bearer <token>",
    });
    return;
  }

  // 404 for unknown routes
  sendResponse(res, 404, {
    error: "Not Found",
    message: `Route ${method} ${pathname} not found`,
    availableRoutes: [
      "GET /health - Health check",
      "GET /api/demo/tokens - List demo tokens",
      "GET /api/user/profile - Get user profile (requires Bearer token)",
      "GET /api/user/profile/insights - Get AI-enhanced profile insights (requires Bearer token)",
      "POST /api/user/chat - Multi-turn chat with AI assistant (requires Bearer token)",
      "DELETE /api/user/chat/clear - Clear conversation history (requires Bearer token)",
    ],
  });
}

// Create and start server
const PORT = process.env.PORT || 3000;
const server = http.createServer(async (req, res) => {
  try {
    await handleRequest(req, res);
  } catch (error) {
    console.error("Server error:", error);
    sendResponse(res, 500, {
      error: "Internal Server Error",
      message: "An unexpected error occurred",
    });
  }
});

server.listen(PORT, () => {

});

module.exports = server;
