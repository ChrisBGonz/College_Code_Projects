import anthropic
import json
from datetime import datetime, timedelta
from typing import Optional

client = anthropic.Anthropic()
MODEL_NAME = "claude-opus-4-5"

cookie_store: dict[str, dict] = {}

tools = [
    {
        "name": "read_cookie",
        "description": "Reads a cookie value for a given user session. Returns the cookie value if it exists, or None if the cookie doesn't exist.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The unique session identifier for the user"
                },
                "cookie_name": {
                    "type": "string",
                    "description": "The name of the cookie to read"
                }
            },
            "required": ["session_id", "cookie_name"]
        }
    },
    {
        "name": "set_cookie",
        "description": "Sets a cookie for a given user session with an optional expiration time.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The unique session identifier for the user"
                },
                "cookie_name": {
                    "type": "string",
                    "description": "The name of the cookie to set"
                },
                "cookie_value": {
                    "type": "string",
                    "description": "The value to store in the cookie"
                },
                "expires_days": {
                    "type": "integer",
                    "description": "Number of days until the cookie expires (optional, defaults to 365)"
                }
            },
            "required": ["session_id", "cookie_name", "cookie_value"]
        }
    },
    {
        "name": "check_trial_eligibility",
        "description": "Checks if a user is eligible for a free trial based on their session cookies and returns detailed eligibility information.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The unique session identifier for the user"
                },
                "user_email": {
                    "type": "string",
                    "description": "The email address of the user trying to sign up"
                }
            },
            "required": ["session_id", "user_email"]
        }
    },
    {
        "name": "register_trial_signup",
        "description": "Registers a new trial signup for an eligible user, sets the trial cookie, and returns confirmation details.",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The unique session identifier for the user"
                },
                "user_email": {
                    "type": "string",
                    "description": "The email address of the user"
                },
                "user_name": {
                    "type": "string",
                    "description": "The name of the user"
                }
            },
            "required": ["session_id", "user_email", "user_name"]
        }
    }
]

def read_cookie(session_id: str, cookie_name: str) -> Optional[str]:
    """Simulates reading a cookie from the browser/session store."""
    if session_id in cookie_store:
        session_cookies = cookie_store[session_id]
        if cookie_name in session_cookies:
            cookie_data = session_cookies[cookie_name]
            if "expires" in cookie_data:
                if datetime.now() > cookie_data["expires"]:
                    del cookie_store[session_id][cookie_name]
                    return None
            return cookie_data["value"]
    return None

def set_cookie(session_id: str, cookie_name: str, cookie_value: str, expires_days: int = 365) -> dict:
    """Simulates setting a cookie in the browser/session store."""
    if session_id not in cookie_store:
        cookie_store[session_id] = {}
    
    expiration = datetime.now() + timedelta(days=expires_days)
    cookie_store[session_id][cookie_name] = {
        "value": cookie_value,
        "expires": expiration,
        "created": datetime.now().isoformat()
    }
    
    return {
        "success": True,
        "cookie_name": cookie_name,
        "cookie_value": cookie_value,
        "expires": expiration.isoformat()
    }

def check_trial_eligibility(session_id: str, user_email: str) -> dict:
    """Checks if a user is eligible for a free trial."""
    has_used_trial = read_cookie(session_id, "has_used_trial")
    
    if has_used_trial:
        trial_data = json.loads(has_used_trial)
        return {
            "eligible": False,
            "reason": "User has already used a free trial",
            "previous_trial_date": trial_data.get("signup_date"),
            "previous_trial_email": trial_data.get("email"),
            "message": f"A free trial was already used on {trial_data.get('signup_date')} with email {trial_data.get('email')}"
        }
    
    return {
        "eligible": True,
        "reason": "No previous trial found",
        "message": "User is eligible for a free trial",
        "session_id": session_id,
        "user_email": user_email
    }

def register_trial_signup(session_id: str, user_email: str, user_name: str) -> dict:
    """Registers a new trial signup and sets the trial cookie."""
    eligibility = check_trial_eligibility(session_id, user_email)
    
    if not eligibility["eligible"]:
        return {
            "success": False,
            "error": "User is not eligible for a trial",
            "details": eligibility
        }
    
    trial_data = {
        "email": user_email,
        "name": user_name,
        "signup_date": datetime.now().isoformat(),
        "trial_end_date": (datetime.now() + timedelta(days=14)).isoformat(),
        "status": "active"
    }
    
    cookie_result = set_cookie(
        session_id=session_id,
        cookie_name="has_used_trial",
        cookie_value=json.dumps(trial_data),
        expires_days=3650
    )
    
    return {
        "success": True,
        "message": f"Trial successfully registered for {user_name}",
        "trial_details": {
            "user_name": user_name,
            "user_email": user_email,
            "trial_start": trial_data["signup_date"],
            "trial_end": trial_data["trial_end_date"],
            "cookie_set": cookie_result["success"]
        }
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Processes tool calls and returns results as JSON strings."""
    if tool_name == "read_cookie":
        result = read_cookie(tool_input["session_id"], tool_input["cookie_name"])
        return json.dumps({"cookie_value": result, "exists": result is not None})
    
    elif tool_name == "set_cookie":
        result = set_cookie(
            tool_input["session_id"],
            tool_input["cookie_name"],
            tool_input["cookie_value"],
            tool_input.get("expires_days", 365)
        )
        return json.dumps(result)
    
    elif tool_name == "check_trial_eligibility":
        result = check_trial_eligibility(tool_input["session_id"], tool_input["user_email"])
        return json.dumps(result)
    
    elif tool_name == "register_trial_signup":
        result = register_trial_signup(
            tool_input["session_id"],
            tool_input["user_email"],
            tool_input["user_name"]
        )
        return json.dumps(result)
    
    return json.dumps({"error": f"Unknown tool: {tool_name}"})

def handle_trial_signup(session_id: str, user_email: str, user_name: str) -> str:
    """
    Main function to handle trial signup using Claude with tool use.
    """
    print(f"\n{'='*60}")
    print(f"Processing trial signup for: {user_name} ({user_email})")
    print(f"Session ID: {session_id}")
    print('='*60)
    
    user_message = f"""
    A user wants to sign up for a free trial. Please help process this signup request.
    
    User Details:
    - Session ID: {session_id}
    - Email: {user_email}
    - Name: {user_name}
    
    Please:
    1. Check if the user has already used a trial by reading the 'has_used_trial' cookie
    2. If they haven't used a trial, register them for the trial
    3. If they have already used a trial, inform them they cannot sign up again
    4. Provide a clear response about the outcome
    """
    
    messages = [{"role": "user", "content": user_message}]
    
    while True:
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    print(f"Final Response: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nTool Called: {tool_name}")
                    print(f"Tool Input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Tool Result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop_reason: {response.stop_reason}")
            break
    
    return "Error processing trial signup"

def main():
    """Main function to demonstrate the trial signup flow."""
    print("\n" + "="*60)
    print("FREE TRIAL SIGNUP FLOW DEMONSTRATION")
    print("="*60)
    
    print("\n--- Test 1: New User (First Time Trial) ---")
    result1 = handle_trial_signup(
        session_id="session_001",
        user_email="alice@example.com",
        user_name="Alice Johnson"
    )
    
    print("\n--- Test 2: Same User Trying Again (Should Be Blocked) ---")
    result2 = handle_trial_signup(
        session_id="session_001",
        user_email="alice@example.com",
        user_name="Alice Johnson"
    )
    
    print("\n--- Test 3: Different User (New Session) ---")
    result3 = handle_trial_signup(
        session_id="session_002",
        user_email="bob@example.com",
        user_name="Bob Smith"
    )
    
    print("\n" + "="*60)
    print("DEMONSTRATION COMPLETE")
    print("="*60)
    print("\nSummary:")
    print(f"Test 1 (New User): {'SUCCESS' if 'successfully' in result1.lower() else 'BLOCKED'}")
    print(f"Test 2 (Repeat User): {'BLOCKED' if 'already' in result2.lower() or 'cannot' in result2.lower() else 'SUCCESS'}")
    print(f"Test 3 (New User): {'SUCCESS' if 'successfully' in result3.lower() else 'BLOCKED'}")

if __name__ == "__main__":
    main()
