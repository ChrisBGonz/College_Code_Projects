import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.*;
import java.net.HttpURLConnection;
import java.net.URL;

public class UserDeserializationExample {

    // User class for demonstration
    static class User implements Serializable {
        private static final long serialVersionUID = 1L;
        private String name;
        private int age;
        private String email;

        public User() {}

        public User(String name, int age, String email) {
            this.name = name;
            this.age = age;
            this.email = email;
        }

        public String getName() { return name; }
        public void setName(String name) { this.name = name; }
        public int getAge() { return age; }
        public void setAge(int age) { this.age = age; }
        public String getEmail() { return email; }
        public void setEmail(String email) { this.email = email; }

        @Override
        public String toString() {
            return "User{name='" + name + "', age=" + age + ", email='" + email + "'}";
        }
    }

    /**
     * Reads a serialized User object from the body of an HTTP request using ObjectInputStream.
     * 
     * @param urlString The URL to send the HTTP request to
     * @param serializedUser The serialized user data to send in the request body
     * @return The deserialized User object
     * @throws IOException If an I/O error occurs
     * @throws ClassNotFoundException If the User class cannot be found during deserialization
     */
    public static User deserializeUserFromHttpRequest(String urlString, byte[] serializedUser) 
            throws IOException, ClassNotFoundException {
        
        URL url = new URL(urlString);
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        
        try {
            // Configure the connection
            connection.setRequestMethod("POST");
            connection.setDoOutput(true);
            connection.setDoInput(true);
            connection.setRequestProperty("Content-Type", "application/octet-stream");
            connection.setRequestProperty("Content-Length", String.valueOf(serializedUser.length));
            
            // Send the serialized user data in the request body
            try (OutputStream outputStream = connection.getOutputStream()) {
                outputStream.write(serializedUser);
                outputStream.flush();
            }
            
            // Check response code
            int responseCode = connection.getResponseCode();
            if (responseCode != HttpURLConnection.HTTP_OK) {
                throw new IOException("HTTP request failed with response code: " + responseCode);
            }
            
            // Read and deserialize the User object from the response body
            try (InputStream inputStream = connection.getInputStream();
                 ObjectInputStream objectInputStream = new ObjectInputStream(inputStream)) {
                
                Object obj = objectInputStream.readObject();
                if (obj instanceof User) {
                    return (User) obj;
                } else {
                    throw new IOException("Received object is not a User instance: " + obj.getClass().getName());
                }
            }
        } finally {
            connection.disconnect();
        }
    }

    /**
     * Alternative method that deserializes a User object directly from an InputStream
     * (simulating reading from an HTTP request body).
     * 
     * @param inputStream The input stream containing the serialized User object
     * @return The deserialized User object
     * @throws IOException If an I/O error occurs
     * @throws ClassNotFoundException If the User class cannot be found during deserialization
     */
    public static User deserializeUserFromInputStream(InputStream inputStream) 
            throws IOException, ClassNotFoundException {
        
        try (ObjectInputStream objectInputStream = new ObjectInputStream(inputStream)) {
            Object obj = objectInputStream.readObject();
            if (obj instanceof User) {
                return (User) obj;
            } else {
                throw new IOException("Received object is not a User instance: " + obj.getClass().getName());
            }
        }
    }

    /**
     * Serializes a User object to a byte array for testing purposes.
     * 
     * @param user The User object to serialize
     * @return The serialized byte array
     * @throws IOException If an I/O error occurs
     */
    public static byte[] serializeUser(User user) throws IOException {
        try (ByteArrayOutputStream byteArrayOutputStream = new ByteArrayOutputStream();
             ObjectOutputStream objectOutputStream = new ObjectOutputStream(byteArrayOutputStream)) {
            
            objectOutputStream.writeObject(user);
            objectOutputStream.flush();
            return byteArrayOutputStream.toByteArray();
        }
    }

    public static void main(String[] args) {
        System.out.println("=== User Deserialization Example ===\n");
        
        // Create a test user
        User originalUser = new User("John Doe", 30, "john.doe@example.com");
        System.out.println("Original User: " + originalUser);
        
        try {
            // Serialize the user
            byte[] serializedData = serializeUser(originalUser);
            System.out.println("Serialized " + serializedData.length + " bytes");
            
            // Demonstrate deserialization from InputStream (simulating HTTP request body)
            System.out.println("\nDeserializing from InputStream (simulating HTTP request body)...");
            
            try (ByteArrayInputStream byteArrayInputStream = new ByteArrayInputStream(serializedData)) {
                User deserializedUser = deserializeUserFromInputStream(byteArrayInputStream);
                System.out.println("Deserialized User: " + deserializedUser);
                
                // Verify the deserialization
                System.out.println("\nVerification:");
                System.out.println("Name matches: " + originalUser.getName().equals(deserializedUser.getName()));
                System.out.println("Age matches: " + (originalUser.getAge() == deserializedUser.getAge()));
                System.out.println("Email matches: " + originalUser.getEmail().equals(deserializedUser.getEmail()));
            }
            
            // Demonstrate the HTTP request method (would need a real server)
            System.out.println("\nNote: The deserializeUserFromHttpRequest method requires a real HTTP server.");
            System.out.println("It would be called like:");
            System.out.println("User user = deserializeUserFromHttpRequest(\"http://example.com/api/user\", serializedData);");
            
        } catch (IOException | ClassNotFoundException e) {
            System.err.println("Error during serialization/deserialization: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
