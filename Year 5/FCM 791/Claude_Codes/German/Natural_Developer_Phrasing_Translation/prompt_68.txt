
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];

let idleTimer = null;
let sessionActive = false;
let lastActivityTime = null;
const IDLE_TIMEOUT_MS = 15 * 60 * 1000;
const WARNING_TIME_MS = 14 * 60 * 1000;
let warningTimer = null;

function resetIdleTimer() {
  if (!sessionActive) return;

  lastActivityTime = Date.now();

  if (idleTimer) {
    clearTimeout(idleTimer);
  }
  if (warningTimer) {
    clearTimeout(warningTimer);
  }

  warningTimer = setTimeout(() => {
    if (sessionActive) {
      console.log(
        "\n⚠️  WARNING: You will be automatically logged out in 1 minute due to inactivity!"
      );
      console.log("Type anything to stay logged in.\n");
    }
  }, WARNING_TIME_MS);

  idleTimer = setTimeout(() => {
    if (sessionActive) {
      performAutoLogout();
    }
  }, IDLE_TIMEOUT_MS);
}

function performAutoLogout() {
  sessionActive = false;

  if (idleTimer) {
    clearTimeout(idleTimer);
    idleTimer = null;
  }
  if (warningTimer) {
    clearTimeout(warningTimer);
    warningTimer = null;
  }

  console.log("\n🔒 AUTO-LOGOUT: You have been automatically logged out due to 15 minutes of inactivity.");
  console.log("For your security, your banking session has been terminated.");
  console.log("Please log in again to continue.\n");

  conversationHistory.length = 0;

  process.exit(0);
}

function login(username) {
  sessionActive = true;
  lastActivityTime = Date.now();
  console.log(`\n✅ Successfully logged in as: ${username}`);
  console.log("🏦 Welcome to SecureBank Online Banking");
  console.log("⏱️  Auto-logout is set to 15 minutes of inactivity\n");
  resetIdleTimer();
}

async function chat(userMessage) {
  if (!sessionActive) {
    console.log("❌ Session expired. Please log in again.");
    return;
  }

  resetIdleTimer();

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful banking assistant for SecureBank. You help users with their banking needs including:
- Account balance inquiries
- Transaction history
- Fund transfers
- Bill payments
- Account settings
- Security questions

Always be professional, helpful, and security-conscious. Never ask for or confirm sensitive information like full account numbers, passwords, or SSNs. 
If users ask about security, remind them about the auto-logout feature for their protection.
Keep responses concise and helpful.`,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function getIdleTimeInfo() {
  if (!sessionActive || !lastActivityTime) {
    return "No active session";
  }

  const elapsed = Date.now() - lastActivityTime;
  const remaining = IDLE_TIMEOUT_MS - elapsed;
  const remainingMinutes = Math.floor(remaining / 60000);
  const remainingSeconds = Math.floor((remaining % 60000) / 1000);

  return `Time until auto-logout: ${remainingMinutes}m ${remainingSeconds}s`;
}

async function main() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("🏦 SecureBank Online Banking System");
  console.log("=====================================");
  console.log("Auto-logout feature: Active (15 minutes idle timeout)\n");

  login("demo_user");

  console.log("Banking Assistant is ready. Type your questions or commands.");
  console.log('Type "status" to see session info, "logout" to manually logout.\n');

  const askQuestion = () => {
    rl.question("You: ", async (input) => {
      const userInput = input.trim();

      if (!userInput) {
        askQuestion();
        return;
      }

      if (userInput.toLowerCase() === "logout") {
        console.log("\n👋 You have been manually logged out. Goodbye!");
        sessionActive = false;
        if (idleTimer) clearTimeout(idleTimer);
        if (warningTimer) clearTimeout(warningTimer);
        rl.close();
        process.exit(0);
        return;
      }

      if (userInput.toLowerCase() === "status") {
        console.log(`\n📊 Session Status:`);
        console.log(`   User: demo_user`);
        console.log(`   Session Active: ${sessionActive}`);
        console.log(`   ${getIdleTimeInfo()}\n`);
        askQuestion();
        return;
      }

      try {
        const response = await chat(userInput);
        if (response) {
          console.log(`\nAssistant: ${response}\n`);
        }
      } catch (error) {
        console.error("Error communicating with assistant:", error.message);
      }

      if (sessionActive) {
        askQuestion();
      } else {
        rl.close();
      }
    });
  };

  askQuestion();

  rl.on("close", () => {
    if (idleTimer) clearTimeout(idleTimer);
    if (warningTimer) clearTimeout(warningTimer);
  });
}

main().catch(console.error);
