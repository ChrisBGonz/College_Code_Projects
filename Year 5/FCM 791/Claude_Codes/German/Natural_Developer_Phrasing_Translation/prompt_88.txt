using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;
using System.Collections.Generic;

// Audit/Security Log Event Models
public class MfaAuditEvent
{
    public DateTime Timestamp { get; set; }
    public string UserId { get; set; }
    public string ChangeType { get; set; }
    public string? CorrelationId { get; set; }
    public string? RequestId { get; set; }
    public string? AdditionalDetails { get; set; }
    public string Severity { get; set; }
    public string EventType { get; set; }
}

public class MfaChangeRequest
{
    public string UserId { get; set; }
    public string ChangeType { get; set; } // e.g., "ENABLED", "DISABLED", "METHOD_CHANGED", "RESET"
    public string? OldMethod { get; set; }
    public string? NewMethod { get; set; }
    public string? CorrelationId { get; set; }
    public string? RequestId { get; set; }
}

// MFA Audit Logger using Claude AI
public class MfaAuditLogger
{
    private readonly AnthropicClient _client;
    private readonly List<MfaAuditEvent> _auditLog;

    public MfaAuditLogger()
    {
        _client = new AnthropicClient();
        _auditLog = new List<MfaAuditEvent>();
    }

    public async Task<MfaAuditEvent> LogMfaChangeAsync(MfaChangeRequest changeRequest)
    {
        // Create the audit event
        var auditEvent = new MfaAuditEvent
        {
            Timestamp = DateTime.UtcNow,
            UserId = changeRequest.UserId,
            ChangeType = changeRequest.ChangeType,
            CorrelationId = changeRequest.CorrelationId,
            RequestId = changeRequest.RequestId ?? Guid.NewGuid().ToString(),
            EventType = "MFA_SETTINGS_CHANGE",
            Severity = DetermineSeverity(changeRequest.ChangeType)
        };

        // Use Claude to generate detailed audit description
        var description = await GenerateAuditDescriptionAsync(changeRequest, auditEvent);
        auditEvent.AdditionalDetails = description;

        // Store the audit event
        _auditLog.Add(auditEvent);

        // Log to console (in production, this would go to a proper logging system)
        Console.WriteLine(FormatAuditEvent(auditEvent));

        return auditEvent;
    }

    private async Task<string> GenerateAuditDescriptionAsync(MfaChangeRequest changeRequest, MfaAuditEvent auditEvent)
    {
        var prompt = $@"Generate a concise security audit log description for the following MFA change event:
        
User ID: {changeRequest.UserId}
Change Type: {changeRequest.ChangeType}
Old Method: {changeRequest.OldMethod ?? "N/A"}
New Method: {changeRequest.NewMethod ?? "N/A"}
Timestamp: {auditEvent.Timestamp:yyyy-MM-dd HH:mm:ss UTC}
Correlation ID: {changeRequest.CorrelationId ?? "N/A"}
Request ID: {auditEvent.RequestId}

Please provide:
1. A brief description of what changed
2. Security implications of this change
3. Recommended follow-up actions if any

Keep the response concise and professional, suitable for a security audit log.";

        var messages = new List<Message>
        {
            new Message(RoleType.User, prompt)
        };

        var messageParams = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 500,
            Stream = false
        };

        var response = await _client.Messages.GetClaudeMessageAsync(messageParams);
        return response.Content[0].ToString();
    }

    private string DetermineSeverity(string changeType)
    {
        return changeType switch
        {
            "DISABLED" => "HIGH",
            "RESET" => "HIGH",
            "ENABLED" => "MEDIUM",
            "METHOD_CHANGED" => "MEDIUM",
            _ => "LOW"
        };
    }

    private string FormatAuditEvent(MfaAuditEvent auditEvent)
    {
        return $@"
========================================
SECURITY AUDIT EVENT
========================================
Event Type:     {auditEvent.EventType}
Severity:       {auditEvent.Severity}
Timestamp:      {auditEvent.Timestamp:yyyy-MM-dd HH:mm:ss UTC}
User ID:        {auditEvent.UserId}
Change Type:    {auditEvent.ChangeType}
Request ID:     {auditEvent.RequestId}
Correlation ID: {auditEvent.CorrelationId ?? "N/A"}
----------------------------------------
Details:
{auditEvent.AdditionalDetails}
========================================";
    }

    public List<MfaAuditEvent> GetAuditLog() => _auditLog;

    public List<MfaAuditEvent> GetAuditLogForUser(string userId)
    {
        return _auditLog.FindAll(e => e.UserId == userId);
    }
}

// Main Program
class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("MFA Audit Logger Demo");
        Console.WriteLine("====================\n");

        var auditLogger = new MfaAuditLogger();

        // Example 1: User enables MFA
        Console.WriteLine("Scenario 1: User enables MFA");
        var enableRequest = new MfaChangeRequest
        {
            UserId = "user-12345",
            ChangeType = "ENABLED",
            NewMethod = "TOTP",
            CorrelationId = "session-abc-123",
            RequestId = "req-001"
        };
        await auditLogger.LogMfaChangeAsync(enableRequest);

        Console.WriteLine("\nScenario 2: User changes MFA method");
        var changeMethodRequest = new MfaChangeRequest
        {
            UserId = "user-67890",
            ChangeType = "METHOD_CHANGED",
            OldMethod = "SMS",
            NewMethod = "TOTP",
            CorrelationId = "session-def-456"
        };
        await auditLogger.LogMfaChangeAsync(changeMethodRequest);

        Console.WriteLine("\nScenario 3: Admin disables MFA for user (high severity)");
        var disableRequest = new MfaChangeRequest
        {
            UserId = "user-11111",
            ChangeType = "DISABLED",
            OldMethod = "TOTP",
            CorrelationId = "admin-session-789",
            RequestId = "admin-req-003"
        };
        await auditLogger.LogMfaChangeAsync(disableRequest);

        Console.WriteLine("\nScenario 4: MFA Reset");
        var resetRequest = new MfaChangeRequest
        {
            UserId = "user-22222",
            ChangeType = "RESET",
            OldMethod = "TOTP",
            CorrelationId = "support-ticket-999"
        };
        await auditLogger.LogMfaChangeAsync(resetRequest);

        // Display summary
        Console.WriteLine("\n\nAUDIT LOG SUMMARY");
        Console.WriteLine("=================");
        var allEvents = auditLogger.GetAuditLog();
        Console.WriteLine($"Total events logged: {allEvents.Count}");
        
        var highSeverityEvents = allEvents.FindAll(e => e.Severity == "HIGH");
        Console.WriteLine($"High severity events: {highSeverityEvents.Count}");
        
        Console.WriteLine("\nAll logged events:");
        foreach (var evt in allEvents)
        {
            Console.WriteLine($"  - [{evt.Severity}] {evt.Timestamp:HH:mm:ss} | User: {evt.UserId} | Change: {evt.ChangeType} | ReqID: {evt.RequestId}");
        }
    }
}
