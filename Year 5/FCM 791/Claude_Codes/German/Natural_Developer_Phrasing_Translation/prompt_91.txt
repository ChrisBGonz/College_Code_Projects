import anthropic
import time
import threading
from typing import Optional


def connect_to_payment_api(payment_data: dict, timeout: float = 2.0) -> bool:
    """
    Connects to an external payment API using Claude AI to simulate the interaction.
    Returns True if the API times out (within 2 seconds) or if payment should proceed.
    Returns False if payment should be blocked.
    
    Args:
        payment_data: Dictionary containing payment information
        timeout: Timeout in seconds (default: 2.0)
    
    Returns:
        bool: True if user can proceed, False if payment should be blocked
    """
    
    result = {"value": None, "completed": False}
    
    def make_api_call():
        """Internal function to make the API call"""
        try:
            client = anthropic.Anthropic()
            
            # Simulate payment API interaction using Claude with streaming
            payment_info = f"""
            Payment Details:
            - Amount: {payment_data.get('amount', 'N/A')}
            - Currency: {payment_data.get('currency', 'USD')}
            - Card Last 4: {payment_data.get('card_last4', 'N/A')}
            - Merchant: {payment_data.get('merchant', 'N/A')}
            """
            
            response_text = ""
            
            with client.messages.stream(
                model="claude-opus-4-5",
                max_tokens=100,
                messages=[
                    {
                        "role": "user",
                        "content": f"""You are a payment API. Analyze this payment and respond with only 'APPROVE' or 'DECLINE'.
                        
{payment_info}

Respond with only one word: APPROVE or DECLINE"""
                    }
                ]
            ) as stream:
                for text in stream.text_stream:
                    response_text += text
                    # Check if we have a complete response
                    if "APPROVE" in response_text.upper() or "DECLINE" in response_text.upper():
                        break
            
            # Parse the response
            if "APPROVE" in response_text.upper():
                result["value"] = True
            elif "DECLINE" in response_text.upper():
                result["value"] = False
            else:
                # Unclear response, allow to proceed
                result["value"] = True
                
        except Exception as e:
            print(f"API Error: {e}")
            result["value"] = None
        finally:
            result["completed"] = True
    
    # Start the API call in a separate thread
    api_thread = threading.Thread(target=make_api_call)
    api_thread.daemon = True
    api_thread.start()
    
    # Wait for the thread to complete or timeout
    start_time = time.time()
    api_thread.join(timeout=timeout)
    elapsed_time = time.time() - start_time
    
    # Check if we timed out
    if not result["completed"] or elapsed_time >= timeout:
        print(f"Payment API timeout after {elapsed_time:.2f}s - allowing user to proceed")
        return True
    
    # Return the result from the API
    if result["value"] is None:
        print("Payment API returned unclear response - allowing user to proceed")
        return True
    
    return result["value"]


def process_payment(payment_data: dict) -> dict:
    """
    Process a payment with timeout handling.
    
    Args:
        payment_data: Dictionary containing payment information
    
    Returns:
        dict: Payment processing result
    """
    print(f"\nProcessing payment for {payment_data.get('merchant', 'Unknown Merchant')}...")
    print(f"Amount: {payment_data.get('amount', 'N/A')} {payment_data.get('currency', 'USD')}")
    
    start_time = time.time()
    can_proceed = connect_to_payment_api(payment_data, timeout=2.0)
    processing_time = time.time() - start_time
    
    if can_proceed:
        status = "SUCCESS"
        message = "Payment processed successfully"
        if processing_time >= 2.0:
            message += " (API timeout - proceeded automatically)"
    else:
        status = "DECLINED"
        message = "Payment declined by payment processor"
    
    result = {
        "status": status,
        "message": message,
        "processing_time": f"{processing_time:.2f}s",
        "payment_data": payment_data
    }
    
    print(f"\nResult: {status}")
    print(f"Message: {message}")
    print(f"Processing time: {processing_time:.2f}s")
    
    return result
