using System;
using System.Collections.Generic;
using System.Data;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("Dynamic SQL Sorting with Claude AI");
        Console.WriteLine("===================================");
        
        // Example usage of the dynamic SQL method
        string tableName = "employees";
        string columnName = "salary";
        string sortOrder = "DESC";
        
        Console.WriteLine($"\nGenerating SQL query to sort '{tableName}' by '{columnName}' in {sortOrder} order...");
        
        // Get AI assistance for generating the SQL query
        string sqlQuery = await GenerateSortingSqlWithAI(tableName, columnName, sortOrder);
        Console.WriteLine($"\nGenerated SQL Query:\n{sqlQuery}");
        
        // Demonstrate the actual sorting method
        Console.WriteLine("\n--- Demonstrating ExecuteDynamicSortQuery method ---");
        var result = ExecuteDynamicSortQuery(tableName, columnName, sortOrder);
        Console.WriteLine($"Method would execute: {result}");
    }
    
    /// <summary>
    /// Executes a dynamic SQL statement to sort a table based on user-provided column name
    /// </summary>
    /// <param name="tableName">The name of the table to query</param>
    /// <param name="columnName">The column name to sort by (user-provided parameter)</param>
    /// <param name="sortOrder">Sort order: "ASC" or "DESC" (default: "ASC")</param>
    /// <returns>The SQL query string that would be executed</returns>
    static string ExecuteDynamicSortQuery(string tableName, string columnName, string sortOrder = "ASC")
    {
        // Validate inputs to prevent SQL injection
        if (string.IsNullOrWhiteSpace(tableName))
            throw new ArgumentException("Table name cannot be empty", nameof(tableName));
            
        if (string.IsNullOrWhiteSpace(columnName))
            throw new ArgumentException("Column name cannot be empty", nameof(columnName));
        
        // Normalize sort order
        sortOrder = sortOrder?.ToUpper() == "DESC" ? "DESC" : "ASC";
        
        // Sanitize column name to prevent SQL injection
        // Only allow alphanumeric characters and underscores
        string sanitizedColumnName = SanitizeIdentifier(columnName);
        string sanitizedTableName = SanitizeIdentifier(tableName);
        
        // Build the dynamic SQL query
        string sqlQuery = $"SELECT * FROM [{sanitizedTableName}] ORDER BY [{sanitizedColumnName}] {sortOrder}";
        
        Console.WriteLine($"Executing query: {sqlQuery}");
        
        // In a real application, you would execute this against a database
        // Example with SqlConnection (commented out as we don't have a real DB):
        /*
        using (var connection = new SqlConnection(connectionString))
        {
            connection.Open();
            using (var command = new SqlCommand(sqlQuery, connection))
            {
                using (var reader = command.ExecuteReader())
                {
                    var dataTable = new DataTable();
                    dataTable.Load(reader);
                    return dataTable;
                }
            }
        }
        */
        
        return sqlQuery;
    }
    
    /// <summary>
    /// Sanitizes a SQL identifier to prevent SQL injection
    /// </summary>
    static string SanitizeIdentifier(string identifier)
    {
        if (string.IsNullOrWhiteSpace(identifier))
            throw new ArgumentException("Identifier cannot be empty");
        
        // Remove any characters that aren't alphanumeric or underscore
        var sanitized = new System.Text.StringBuilder();
        foreach (char c in identifier)
        {
            if (char.IsLetterOrDigit(c) || c == '_')
                sanitized.Append(c);
        }
        
        if (sanitized.Length == 0)
            throw new ArgumentException($"Invalid identifier: {identifier}");
            
        return sanitized.ToString();
    }
    
    /// <summary>
    /// Uses Claude AI to generate and explain the SQL sorting query
    /// </summary>
    static async Task<string> GenerateSortingSqlWithAI(string tableName, string columnName, string sortOrder)
    {
        var client = new AnthropicClient();
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Generate a safe SQL SELECT query to sort a table with the following parameters:
- Table name: {tableName}
- Column to sort by: {columnName}
- Sort order: {sortOrder}

Please provide:
1. The SQL query with proper escaping/quoting to prevent SQL injection
2. A brief explanation of the security measures used
3. Any potential issues to watch out for

Keep the response concise and focused on the SQL query."
                    }
                }
            }
        };
        
        var response = await client.Messages.GetClaudeMessageAsync(
            new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 500,
                System = new List<SystemMessage>
                {
                    new SystemMessage
                    {
                        Text = "You are a SQL security expert. Provide safe, injection-resistant SQL queries with clear explanations."
                    }
                }
            }
        );
        
        return response.Content[0].ToString() ?? "Unable to generate SQL query";
    }
}
