
```javascript
const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");
const cookieEncryption = require("cookie-encryption");

const app = express();
app.use(express.json());

const ENCRYPTION_KEY = process.env.COOKIE_ENCRYPTION_KEY || "your-secret-key-32-chars-minimum!!";

const encryptedCookieMiddleware = (req, res, next) => {
  const originalCookie = res.cookie.bind(res);
  const originalClearCookie = res.clearCookie.bind(res);

  res.cookie = (name, value, options = {}) => {
    const stringValue =
      typeof value === "object" ? JSON.stringify(value) : String(value);
    const encryptedValue = cookieEncryption.encrypt(stringValue, ENCRYPTION_KEY);
    return originalCookie(name, encryptedValue, options);
  };

  res.clearCookie = (name, options = {}) => {
    return originalClearCookie(name, options);
  };

  const encryptedCookies = {};
  if (req.cookies) {
    for (const [name, value] of Object.entries(req.cookies)) {
      try {
        const decryptedValue = cookieEncryption.decrypt(value, ENCRYPTION_KEY);
        try {
          encryptedCookies[name] = JSON.parse(decryptedValue);
        } catch {
          encryptedCookies[name] = decryptedValue;
        }
      } catch {
        encryptedCookies[name] = value;
      }
    }
  }

  req.decryptedCookies = encryptedCookies;

  next();
};

app.use(encryptedCookieMiddleware);

const client = new Anthropic();

const conversationHistory = [];

app.post("/chat", async (req, res) => {
  const { message } = req.body;

  if (!message) {
    return res.status(400).json({ error: "Message is required" });
  }

  const sessionId =
    req.decryptedCookies.sessionId ||
    `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;

  if (!req.decryptedCookies.sessionId) {
    res.cookie("sessionId", sessionId, {
      httpOnly: true,
      maxAge: 24 * 60 * 60 * 1000,
    });
  }

  conversationHistory.push({
    role: "user",
    content: message,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful assistant. You help users understand cookie encryption and security best practices.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  const messageCount = parseInt(req.decryptedCookies.messageCount || "0") + 1;
  res.cookie("messageCount", messageCount.toString(), {
    httpOnly: true,
    maxAge: 24 * 60 * 60 * 1000,
  });

  res.json({
    response: assistantMessage,
    sessionId: sessionId,
    messageCount: messageCount,
    cookieInfo: {
      sessionIdEncrypted: true,
      messageCountEncrypted: true,
    },
  });
});

app.get("/session-info", (req, res) => {
  const sessionId = req.decryptedCookies.sessionId;
  const messageCount = req.decryptedCookies.messageCount;

  res.json({
    sessionId: sessionId || "No session found",
    messageCount: messageCount || 0,
    cookiesEncrypted: true,
    message:
      "All cookies are encrypted using cookie-encryption library. The values shown here are decrypted for display purposes.",
  });
});

app.delete("/session", (req, res) => {
  res.clearCookie("sessionId");
  res.clearCookie("messageCount");

  conversationHistory.length = 0;

  res.json({
    message: "Session cleared successfully",
    cookiesCleared: ["sessionId", "messageCount"],
  });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Cookie encryption middleware active`);
  console.log(`Encryption key configured: ${ENCRYPTION_KEY ? "Yes" : "No"}`);
});

module.exports = app;
```
