import anthropic
import subprocess
import sys
import time

client = anthropic.Anthropic()

def run_command(command: str, capture_output: bool = True) -> tuple[int, str, str]:
    """Execute a shell command and return exit code, stdout, stderr."""
    result = subprocess.run(
        command,
        shell=True,
        capture_output=capture_output,
        text=True
    )
    return result.returncode, result.stdout, result.stderr

def stream_deployment_with_claude(image_name: str, image_tag: str, registry_url: str, environment: str = "production"):
    """Use Claude to orchestrate and explain the CI/CD deployment process with streaming."""
    
    deployment_context = f"""
    You are a DevOps expert helping to deploy a Docker image to production.
    
    Deployment Details:
    - Registry URL: {registry_url}
    - Image Name: {image_name}
    - Image Tag: {image_tag}
    - Target Environment: {environment}
    - Full Image Path: {registry_url}/{image_name}:{image_tag}
    
    Please provide a detailed step-by-step deployment guide and explain what each step does.
    Include:
    1. Pre-deployment checks
    2. Docker image pull process
    3. Container deployment steps
    4. Health checks
    5. Rollback strategy if deployment fails
    6. Post-deployment verification
    
    Format your response as actionable steps with explanations.
    """
    
    print("\n" + "="*60)
    print("🚀 CI/CD PIPELINE - PRODUCTION DEPLOYMENT")
    print("="*60)
    print(f"\n📦 Image: {registry_url}/{image_name}:{image_tag}")
    print(f"🎯 Environment: {environment.upper()}")
    print("\n" + "-"*60)
    print("🤖 Claude AI Deployment Assistant:")
    print("-"*60 + "\n")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": deployment_context
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n\n" + "="*60)
    print("🔧 EXECUTING DEPLOYMENT PIPELINE")
    print("="*60)
    
    execute_deployment_pipeline(registry_url, image_name, image_tag, environment)

def execute_deployment_pipeline(registry_url: str, image_name: str, image_tag: str, environment: str):
    """Execute the actual deployment pipeline steps."""
    
    full_image_path = f"{registry_url}/{image_name}:{image_tag}"
    container_name = f"{image_name}-{environment}"
    
    steps = [
        {
            "name": "Pre-deployment Check",
            "description": "Verifying Docker daemon is running",
            "command": "docker info --format '{{.ServerVersion}}'",
            "critical": True
        },
        {
            "name": "Registry Authentication",
            "description": f"Logging into registry: {registry_url}",
            "command": f"echo 'Simulating: docker login {registry_url}'",
            "critical": True
        },
        {
            "name": "Pull Docker Image",
            "description": f"Pulling image: {full_image_path}",
            "command": f"echo 'Simulating: docker pull {full_image_path}'",
            "critical": True
        },
        {
            "name": "Verify Image",
            "description": "Verifying pulled image integrity",
            "command": f"echo 'Simulating: docker inspect {full_image_path}'",
            "critical": True
        },
        {
            "name": "Stop Existing Container",
            "description": f"Stopping existing {environment} container if running",
            "command": f"echo 'Simulating: docker stop {container_name} 2>/dev/null || true'",
            "critical": False
        },
        {
            "name": "Remove Old Container",
            "description": "Removing old container instance",
            "command": f"echo 'Simulating: docker rm {container_name} 2>/dev/null || true'",
            "critical": False
        },
        {
            "name": "Deploy New Container",
            "description": f"Starting new {environment} container",
            "command": f"echo 'Simulating: docker run -d --name {container_name} --restart=always -p 80:8080 {full_image_path}'",
            "critical": True
        },
        {
            "name": "Health Check",
            "description": "Verifying container is healthy",
            "command": f"echo 'Simulating: docker inspect --format={{{{.State.Health.Status}}}} {container_name}'",
            "critical": True
        },
        {
            "name": "Cleanup Old Images",
            "description": "Removing unused Docker images",
            "command": "echo 'Simulating: docker image prune -f'",
            "critical": False
        }
    ]
    
    failed_steps = []
    
    for i, step in enumerate(steps, 1):
        print(f"\n[{i}/{len(steps)}] {step['name']}")
        print(f"    📋 {step['description']}")
        
        time.sleep(0.5)
        
        exit_code, stdout, stderr = run_command(step['command'])
        
        if exit_code == 0:
            print(f"    ✅ SUCCESS")
            if stdout.strip():
                print(f"    📤 Output: {stdout.strip()}")
        else:
            print(f"    ❌ FAILED")
            if stderr.strip():
                print(f"    ⚠️  Error: {stderr.strip()}")
            
            if step['critical']:
                failed_steps.append(step['name'])
                print(f"\n🚨 Critical step failed: {step['name']}")
                handle_deployment_failure(container_name, failed_steps)
                return
    
    print("\n" + "="*60)
    print("✅ DEPLOYMENT COMPLETED SUCCESSFULLY")
    print("="*60)
    print(f"\n🎉 Image {full_image_path} deployed to {environment.upper()}")
    print(f"📊 Container: {container_name}")
    print(f"🌐 Application should be accessible at: http://localhost:80")
    
    generate_deployment_report(registry_url, image_name, image_tag, environment, "SUCCESS")

def handle_deployment_failure(container_name: str, failed_steps: list):
    """Handle deployment failure with rollback."""
    print("\n" + "="*60)
    print("🔄 INITIATING ROLLBACK PROCEDURE")
    print("="*60)
    
    rollback_steps = [
        f"echo 'Simulating: docker stop {container_name} 2>/dev/null || true'",
        f"echo 'Simulating: docker rm {container_name} 2>/dev/null || true'",
        f"echo 'Simulating: docker start {container_name}-backup 2>/dev/null || true'"
    ]
    
    for step in rollback_steps:
        exit_code, stdout, stderr = run_command(step)
        if stdout.strip():
            print(f"  🔄 {stdout.strip()}")
    
    print("\n⚠️  Rollback completed. Previous version restored.")
    print(f"Failed steps: {', '.join(failed_steps)}")

def generate_deployment_report(registry_url: str, image_name: str, image_tag: str, 
                               environment: str, status: str):
    """Generate a deployment report using Claude with streaming."""
    
    print("\n" + "="*60)
    print("📊 GENERATING DEPLOYMENT REPORT")
    print("="*60 + "\n")
    
    report_prompt = f"""
    Generate a concise deployment report for:
    - Registry: {registry_url}
    - Image: {image_name}:{image_tag}
    - Environment: {environment}
    - Status: {status}
    - Timestamp: {time.strftime('%Y-%m-%d %H:%M:%S UTC')}
    
    Include:
    1. Executive Summary
    2. Deployment Details
    3. Next Steps
    4. Monitoring Recommendations
    
    Keep it professional and concise.
    """
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1000,
        messages=[
            {
                "role": "user",
                "content": report_prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")

def main():
    """Main function to run the CI/CD pipeline."""
    
    registry_url = "registry.example.com"
    image_name = "myapp"
    image_tag = "v1.2.3"
    environment = "production"
    
    if len(sys.argv) > 1:
        if len(sys.argv) >= 4:
            registry_url = sys.argv[1]
            image_name = sys.argv[2]
            image_tag = sys.argv[3]
        if len(sys.argv) >= 5:
            environment = sys.argv[4]
    
    print("\n🔍 CI/CD Pipeline Configuration:")
    print(f"  Registry: {registry_url}")
    print(f"  Image: {image_name}")
    print(f"  Tag: {image_tag}")
    print(f"  Environment: {environment}")
    
    stream_deployment_with_claude(image_name, image_tag, registry_url, environment)

if __name__ == "__main__":
    main()
