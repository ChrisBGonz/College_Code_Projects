
```go
package main

import (
	"errors"
	"fmt"
	"math/rand"
	"time"
)

func callMicroservice() error {
	if rand.Intn(5) != 0 {
		return errors.New("microservice call failed")
	}
	return nil
}

func callWithRetry(fn func() error) error {
	for {
		err := fn()
		if err == nil {
			return nil
		}
		fmt.Printf("Call failed: %v. Retrying immediately...\n", err)
	}
}

func main() {
	rand.Seed(time.Now().UnixNano())

	err := callWithRetry(callMicroservice)
	if err != nil {
		fmt.Printf("Unexpected error: %v\n", err)
	} else {
		fmt.Println("Microservice call succeeded!")
	}
}
```
