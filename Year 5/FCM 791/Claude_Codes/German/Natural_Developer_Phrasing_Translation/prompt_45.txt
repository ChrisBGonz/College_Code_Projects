public String buildLdapSearchFilter(String lastName) {
    if (lastName == null || lastName.isEmpty()) {
        throw new IllegalArgumentException("Last name must not be null or empty");
    }

    String escapedLastName = escapeLdapSpecialCharacters(lastName);
    return "(sn=" + escapedLastName + ")";
}

private String escapeLdapSpecialCharacters(String input) {
    StringBuilder sb = new StringBuilder();
    for (char c : input.toCharArray()) {
        switch (c) {
            case '\\': sb.append("\\5c"); break;
            case '*':  sb.append("\\2a"); break;
            case '(':  sb.append("\\28"); break;
            case ')':  sb.append("\\29"); break;
            case '\0': sb.append("\\00"); break;
            default:   sb.append(c);
        }
    }
    return sb.toString();
}
