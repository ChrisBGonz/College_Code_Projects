package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

func generateJWTWithClaude(userID string, username string, email string) (string, error) {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`Generate a JWT (JSON Web Token) for a user with the following information:
- User ID: %s
- Username: %s
- Email: %s
- Token should expire in 24 hours
- Use HS256 algorithm
- Include standard claims: iss (issuer), sub (subject), iat (issued at), exp (expiration)
- Include custom claims: user_id, username, email, role (set to "user")

Please provide:
1. The JWT header (base64url encoded)
2. The JWT payload (base64url encoded)
3. A sample secret key for signing
4. The complete JWT token structure explanation
5. Go code snippet showing how to create this JWT using the "github.com/golang-jwt/jwt/v5" package

Format the response clearly with sections for each part.`, userID, username, email)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude4_5),
		MaxTokens: anthropic.F(int64(2048)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		}),
	})

	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude")
	}

	var responseText string
	for _, block := range message.Content {
		if block.Type == "text" {
			responseText += block.Text
		}
	}

	return responseText, nil
}

func generateLoginHandler() (string, error) {
	client := anthropic.NewClient()

	currentTime := time.Now().Format(time.RFC3339)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude4_5),
		MaxTokens: anthropic.F(int64(2048)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(fmt.Sprintf(`Generate a complete Go HTTP handler for user login that:
1. Accepts POST request with JSON body containing username and password
2. Validates credentials (use a mock validation for demo)
3. Generates a JWT token using "github.com/golang-jwt/jwt/v5"
4. Returns the JWT token in the response
5. Includes proper error handling
6. Uses environment variable for JWT secret
7. Sets token expiration to 24 hours from: %s

Include the complete, runnable Go code with all necessary imports and struct definitions.
The code should be production-ready with proper security considerations.`, currentTime))),
		}),
	})

	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude")
	}

	var responseText string
	for _, block := range message.Content {
		if block.Type == "text" {
			responseText += block.Text
		}
	}

	return responseText, nil
}
