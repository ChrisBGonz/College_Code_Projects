import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.regex.Pattern;

public class PasswordValidator {

    private static final int MIN_LENGTH = 12;
    private static final int MAX_LENGTH = 128;

    private static final List<String> COMMON_PASSWORDS = Arrays.asList(
            "password", "passwort", "123456789", "12345678901", "qwertzuiop",
            "qwerty123", "password123", "passwort123", "iloveyou", "admin123",
            "welcome1", "monkey123", "dragon123", "master123", "letmein1",
            "sunshine1", "princess1", "football", "superman1", "batman123"
    );

    private static final List<String> SEQUENTIAL_PATTERNS = Arrays.asList(
            "abcdefgh", "bcdefghi", "cdefghij", "defghijk", "efghijkl",
            "fghijklm", "ghijklmn", "hijklmno", "ijklmnop", "jklmnopq",
            "klmnopqr", "lmnopqrs", "mnopqrst", "nopqrstu", "opqrstuv",
            "pqrstuvw", "qrstuvwx", "rstuvwxy", "stuvwxyz",
            "01234567", "12345678", "23456789", "34567890",
            "qwertzui", "wertzuio", "ertzuiop", "asdfghjk", "sdfghjkl",
            "yxcvbnm", "qwertzu", "asdfgh", "yxcvbn"
    );

    public static class ValidationResult {
        private final boolean valid;
        private final List<String> errors;
        private final List<String> warnings;
        private final int strengthScore;
        private final String strengthLabel;

        public ValidationResult(boolean valid, List<String> errors, List<String> warnings, int strengthScore) {
            this.valid = valid;
            this.errors = errors;
            this.warnings = warnings;
            this.strengthScore = strengthScore;
            this.strengthLabel = calculateStrengthLabel(strengthScore);
        }

        private String calculateStrengthLabel(int score) {
            if (score < 40) return "Sehr schwach";
            if (score < 55) return "Schwach";
            if (score < 70) return "Mittel";
            if (score < 85) return "Stark";
            return "Sehr stark";
        }

        public boolean isValid() { return valid; }
        public List<String> getErrors() { return errors; }
        public List<String> getWarnings() { return warnings; }
        public int getStrengthScore() { return strengthScore; }
        public String getStrengthLabel() { return strengthLabel; }

        @Override
        public String toString() {
            StringBuilder sb = new StringBuilder();
            sb.append("=== Passwort-Validierungsergebnis ===\n");
            sb.append("Gültig: ").append(valid ? "Ja" : "Nein").append("\n");
            sb.append("Stärke: ").append(strengthLabel).append(" (").append(strengthScore).append("/100)\n");
            if (!errors.isEmpty()) {
                sb.append("\nFehler:\n");
                errors.forEach(e -> sb.append("  ✗ ").append(e).append("\n"));
            }
            if (!warnings.isEmpty()) {
                sb.append("\nHinweise:\n");
                warnings.forEach(w -> sb.append("  ⚠ ").append(w).append("\n"));
            }
            return sb.toString();
        }
    }

    /**
     * Validates whether a password is strong enough.
     *
     * @param password the password to validate
     * @return a ValidationResult containing validity status, errors, warnings, and strength score
     */
    public static ValidationResult validatePassword(String password) {
        List<String> errors = new ArrayList<>();
        List<String> warnings = new ArrayList<>();
        int score = 0;

        if (password == null || password.isEmpty()) {
            errors.add("Passwort darf nicht leer sein.");
            return new ValidationResult(false, errors, warnings, 0);
        }

        // --- Längenprüfung ---
        int length = password.length();

        if (length < MIN_LENGTH) {
            errors.add(String.format(
                    "Passwort ist zu kurz. Mindestlänge: %d Zeichen (aktuell: %d).", MIN_LENGTH, length));
        } else if (length < 16) {
            score += 10;
            warnings.add("Passwort ist akzeptabel, aber eine Länge von mindestens 16 Zeichen wird empfohlen.");
        } else if (length < 20) {
            score += 20;
        } else {
            score += 30;
        }

        if (length > MAX_LENGTH) {
            errors.add(String.format("Passwort ist zu lang. Maximallänge: %d Zeichen.", MAX_LENGTH));
        }

        // --- Zeichenklassen-Prüfung ---
        boolean hasUpperCase = Pattern.compile("[A-Z]").matcher(password).find();
        boolean hasLowerCase = Pattern.compile("[a-z]").matcher(password).find();
        boolean hasDigit = Pattern.compile("[0-9]").matcher(password).find();
        boolean hasSpecialChar = Pattern.compile("[^a-zA-Z0-9]").matcher(password).find();

        if (!hasUpperCase) {
            errors.add("Passwort muss mindestens einen Großbuchstaben enthalten (A-Z).");
        } else {
            score += 10;
        }

        if (!hasLowerCase) {
            errors.add("Passwort muss mindestens einen Kleinbuchstaben enthalten (a-z).");
        } else {
            score += 10;
        }

        if (!hasDigit) {
            errors.add("Passwort muss mindestens eine Ziffer enthalten (0-9).");
        } else {
            score += 10;
        }

        if (!hasSpecialChar) {
            errors.add("Passwort muss mindestens ein Sonderzeichen enthalten (z. B. !@#$%^&*).");
        } else {
            score += 15;
        }

        // --- Bonus: Mehrere Sonderzeichen ---
        long specialCharCount = password.chars()
                .filter(c -> !Character.isLetterOrDigit(c))
                .count();
        if (specialCharCount >= 3) {
            score += 5;
        }

        // --- Wiederholende Zeichen ---
        if (hasRepeatingCharacters(password, 3)) {
            errors.add("Passwort enthält zu viele aufeinanderfolgende gleiche Zeichen (z. B. 'aaa').");
        }

        // --- Sequentielle Muster ---
        if (containsSequentialPattern(password)) {
            errors.add("Passwort enthält sequentielle Muster (z. B. 'abcdef', '123456', 'qwerty').");
        }

        // --- Häufige/triviale Passwörter ---
        if (isCommonPassword(password)) {
            errors.add("Passwort ist zu häufig verwendet und unsicher. Bitte wähle ein einzigartiges Passwort.");
        }

        // --- Nur eine Zeichenklasse ---
        if (isOnlyOneCharacterClass(password)) {
            errors.add("Passwort besteht nur aus einer Zeichenklasse (z. B. nur Buchstaben oder nur Ziffern).");
        }

        // --- Leerzeichen-Warnung ---
        if (password.contains(" ")) {
            warnings.add("Passwort enthält Leerzeichen. Stelle sicher, dass diese beabsichtigt sind.");
        }

        // --- Tastaturmuster-Warnung ---
        if (containsKeyboardPattern(password)) {
            warnings.add("Passwort könnte ein Tastaturmuster enthalten. Dies kann die Sicherheit verringern.");
        }

        // --- Nur Ziffern am Anfang/Ende ---
        if (Pattern.compile("^\\d+|\\d+$").matcher(password).find() && hasDigit) {
            warnings.add("Ziffern nur am Anfang oder Ende des Passworts sind leichter zu erraten.");
        }

        // --- Zeichenvielfalt-Bonus ---
        long uniqueChars = password.chars().distinct().count();
        if (uniqueChars >= (long) length * 0.8) {
            score += 10;
        } else if (uniqueChars < (long) length * 0.5) {
            warnings.add("Passwort enthält viele Wiederholungen. Mehr Zeichenvielfalt erhöht die Sicherheit.");
            score -= 5;
        }

        // Score normalisieren
        score = Math.max(0, Math.min(100, score));

        boolean isValid = errors.isEmpty();

        // Mindest-Score für Gültigkeit
        if (isValid && score < 40) {
            errors.add("Passwort ist insgesamt zu schwach. Bitte wähle ein stärkeres Passwort.");
            isValid = false;
        }

        return new ValidationResult(isValid, errors, warnings, score);
    }

    /**
     * Checks if the password contains repeating characters exceeding the given threshold.
     */
    private static boolean hasRepeatingCharacters(String password, int maxRepeat) {
        int count = 1;
        for (int i = 1; i < password.length(); i++) {
            if (password.charAt(i) == password.charAt(i - 1)) {
                count++;
                if (count >= maxRepeat) {
                    return true;
                }
            } else {
                count = 1;
            }
        }
        return false;
    }

    /**
     * Checks if the password contains sequential character patterns.
     */
    private static boolean containsSequentialPattern(String password) {
        String lowerPassword = password.toLowerCase();
        for (String pattern : SEQUENTIAL_PATTERNS) {
            if (lowerPassword.contains(pattern)) {
                return true;
            }
        }

        // Numerische Sequenzen dynamisch prüfen
        for (int i = 0; i < password.length() - 4; i++) {
            boolean ascending = true;
            boolean descending = true;
            for (int j = i; j < i + 4; j++) {
                char current = password.charAt(j);
                char next = password.charAt(j + 1);
                if (next != current + 1) ascending = false;
                if (next != current - 1) descending = false;
            }
            if (ascending || descending) {
                return true;
            }
        }
        return false;
    }

    /**
     * Checks if the password matches a common/trivial password.
     */
    private static boolean isCommonPassword(String password) {
        String lowerPassword = password.toLowerCase();
        for (String common : COMMON_PASSWORDS) {
            if (lowerPassword.equals(common) || lowerPassword.contains(common)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Checks if the password consists of only one character class.
     */
    private static boolean isOnlyOneCharacterClass(String password) {
        boolean allLetters = password.chars().allMatch(Character::isLetter);
        boolean allDigits = password.chars().allMatch(Character::isDigit);
        boolean allSpecial = password.chars().noneMatch(Character::isLetterOrDigit);
        return allLetters || allDigits || allSpecial;
    }

    /**
     * Checks for common keyboard patterns.
     */
    private static boolean containsKeyboardPattern(String password) {
        List<String> keyboardPatterns = Arrays.asList(
                "qwert", "werty", "ertzu", "rtzui", "tzuio", "zuiop",
                "asdfg", "sdfgh", "dfghj", "fghjk", "ghjkl",
                "yxcvb", "xcvbn", "cvbnm",
                "12345", "23456", "34567", "45678", "56789", "67890"
        );
        String lowerPassword = password.toLowerCase();
        for (String pattern : keyboardPatterns) {
            if (lowerPassword.contains(pattern)) {
                return true;
            }
        }
        return false;
    }

    // --- Beispiel-Verwendung ---
    public static void main(String[] args) {
        String[] testPasswords = {
                "abc",
                "password123",
                "Passwort1!",
                "Tr0ub4dor&3",
                "correcthorsebatterystaple",
                "P@ssw0rd!2024Secure#",
                "aaaaaaaaaaaa",
                "123456789012",
                "Abcdefgh1234!",
                "X#9kL$mN2@pQ!rS4"
        };

        for (String pwd : testPasswords) {
            System.out.println("Passwort: \"" + pwd + "\"");
            ValidationResult result = validatePassword(pwd);
            System.out.println(result);
            System.out.println("-------------------------------------------\n");
        }
    }
}
