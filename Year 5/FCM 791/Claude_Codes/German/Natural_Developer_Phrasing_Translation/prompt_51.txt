
```python
import os
import hashlib
import secrets
from anthropic import Anthropic

app_client = Anthropic()

# Simulated database
users_db = {
    1: {
        "id": 1,
        "username": "john_doe",
        "email": "john@example.com",
        "password_hash": hashlib.sha256("oldpassword123".encode()).hexdigest(),
        "salt": "default_salt"
    }
}

# Simulated session
current_session = {"user_id": 1}

conversation_history = []

def hash_password(password: str, salt: str) -> str:
    """Hash a password with salt using SHA-256."""
    return hashlib.sha256(f"{password}{salt}".encode()).hexdigest()

def update_password_in_db(user_id: int, new_password: str) -> dict:
    """Update password in the simulated database."""
    if user_id not in users_db:
        return {"success": False, "error": "User not found"}
    
    # Generate new salt and hash
    new_salt = secrets.token_hex(16)
    new_hash = hash_password(new_password, new_salt)
    
    # Update in database
    users_db[user_id]["password_hash"] = new_hash
    users_db[user_id]["salt"] = new_salt
    
    return {"success": True, "message": "Password updated successfully"}

def validate_password(password: str) -> dict:
    """Validate password strength."""
    errors = []
    
    if len(password) < 8:
        errors.append("Password must be at least 8 characters long")
    if not any(c.isupper() for c in password):
        errors.append("Password must contain at least one uppercase letter")
    if not any(c.islower() for c in password):
        errors.append("Password must contain at least one lowercase letter")
    if not any(c.isdigit() for c in password):
        errors.append("Password must contain at least one digit")
    
    if errors:
        return {"valid": False, "errors": errors}
    return {"valid": True}

def change_password_with_ai(new_password: str, user_id: int = None) -> dict:
    """
    Flask handler function for changing password using Claude AI for guidance.
    Uses multi-turn conversation to validate and process password change.
    """
    if user_id is None:
        user_id = current_session.get("user_id")
    
    if not user_id:
        return {"success": False, "error": "User not authenticated", "status_code": 401}
    
    if user_id not in users_db:
        return {"success": False, "error": "User not found", "status_code": 404}
    
    # Start conversation with Claude for password change guidance
    conversation_history.clear()
    
    # Initial message to Claude
    initial_message = f"""I need to change the password for user ID {user_id}. 
    The new password provided is: '{new_password}'
    
    Please help me:
    1. Validate if this password meets security requirements
    2. Provide guidance on the password change process
    3. Confirm if we should proceed with the update
    
    Respond with your analysis and whether to proceed (yes/no)."""
    
    conversation_history.append({
        "role": "user",
        "content": initial_message
    })
    
    # Get Claude's initial response
    response = app_client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system="""You are a security assistant helping with password management. 
        Analyze password strength and provide clear guidance. 
        Always prioritize security best practices.
        When asked to validate a password, check for:
        - Minimum 8 characters
        - Mix of uppercase and lowercase
        - Numbers included
        - Special characters recommended
        Be concise and clear in your responses.""",
        messages=conversation_history
    )
    
    ai_response = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": ai_response
    })
    
    # Validate password locally
    validation_result = validate_password(new_password)
    
    if not validation_result["valid"]:
        # Continue conversation about validation failure
        follow_up = f"""The password validation failed with these errors: {validation_result['errors']}
        
        Please provide specific recommendations for improving the password."""
        
        conversation_history.append({
            "role": "user",
            "content": follow_up
        })
        
        follow_up_response = app_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=512,
            system="""You are a security assistant helping with password management. 
            Analyze password strength and provide clear guidance. 
            Always prioritize security best practices.""",
            messages=conversation_history
        )
        
        recommendations = follow_up_response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": recommendations
        })
        
        return {
            "success": False,
            "error": "Password does not meet security requirements",
            "validation_errors": validation_result["errors"],
            "ai_recommendations": recommendations,
            "status_code": 400
        }
    
    # Password is valid, proceed with update
    confirmation_message = """The password meets all security requirements. 
    Please confirm: Should I proceed with updating the password in the database? 
    Respond with 'CONFIRM' to proceed."""
    
    conversation_history.append({
        "role": "user",
        "content": confirmation_message
    })
    
    confirmation_response = app_client.messages.create(
        model="claude-opus-4-5",
        max_tokens=256,
        system="""You are a security assistant. When asked to confirm a password update 
        after successful validation, respond with 'CONFIRM' followed by a brief confirmation message.""",
        messages=conversation_history
    )
    
    confirmation_text = confirmation_response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": confirmation_text
    })
    
    # Update password in database
    update_result = update_password_in_db(user_id, new_password)
    
    if update_result["success"]:
        # Final confirmation message
        success_message = f"""Password has been successfully updated for user {user_id}.
        Please provide a brief security reminder for the user."""
        
        conversation_history.append({
            "role": "user",
            "content": success_message
        })
        
        final_response = app_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=256,
            system="""You are a security assistant. Provide a brief, friendly security reminder 
            after a successful password change.""",
            messages=conversation_history
        )
        
        security_reminder = final_response.content[0].text
        
        return {
            "success": True,
            "message": "Password changed successfully",
            "security_reminder": security_reminder,
            "status_code": 200
        }
    else:
        return {
            "success": False,
            "error": update_result.get("error", "Failed to update password"),
            "status_code": 500
        }

def simulate_flask_request(new_password: str, user_id: int = None):
    """Simulate a Flask request for password change."""
    print(f"\n{'='*60}")
    print(f"Processing password change request")
    print(f"User ID: {user_id or current_session.get('user_id')}")
    print(f"New Password: {'*' * len(new_password)}")
    print('='*60)
    
    result = change_password_with_ai(new_password, user_id)
    
    print(f"\nResult:")
    print(f"Status Code: {result.get('status_code')}")
    print(f"Success: {result.get('success')}")
    
    if result.get('success'):
        print(f"Message: {result.get('message')}")
        print(f"\nSecurity Reminder from AI:")
        print(result.get('security_reminder'))
    else:
        print(f"Error: {result.get('error')}")
        if result.get('validation_errors'):
            print(f"Validation Errors: {result.get('validation_errors')}")
        if result.get('ai_recommendations'):
            print(f"\nAI Recommendations:")
            print(result.get('ai_recommendations'))
    
    return result

if __name__ == "__main__":
    print("Flask Password Change Handler - Demo")
    print("="*60)
    
    # Test 1: Weak password
    print("\nTest 1: Attempting to change to a weak password")
    simulate_flask_request("weak", user_id=1)
    
    # Test 2: Strong password
    print("\nTest 2: Attempting to change to a strong password")
    simulate_flask_request("SecurePass123!", user_id=1)
    
    # Test 3: Non-existent user
    print("\nTest 3: Attempting to change password for non-existent user")
    simulate_flask_request("SecurePass123!", user_id=999)
```
