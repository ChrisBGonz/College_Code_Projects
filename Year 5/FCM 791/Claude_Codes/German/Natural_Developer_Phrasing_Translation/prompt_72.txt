const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definitions for Claude
const tools = [
  {
    name: "download_file",
    description:
      "Downloads a file from a URL and saves it to a local path. Returns success status and file size.",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path where the downloaded file will be saved",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description:
      "Extracts a ZIP file to a specified directory. Returns success status and list of extracted files.",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "Path to the ZIP file to extract",
        },
        destination_dir: {
          type: "string",
          description: "Directory where the ZIP contents will be extracted",
        },
      },
      required: ["zip_path", "destination_dir"],
    },
  },
  {
    name: "create_directory",
    description:
      "Creates a directory and all necessary parent directories. Returns success status.",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to create",
        },
      },
      required: ["path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Removes a temporary file. Returns success status.",
    input_schema: {
      type: "object",

        file_path: {
          type: "string",
          description: "Path to the file to remove",
        },
      },
      required: ["file_path"],

// Tool implementations
async function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    const protocol = url.startsWith("https://") ? https : http;

    const file = fs.createWriteStream(destination);

    const request = protocol.get(url, (response) => {
      if (response.statusCode === 301 || response.statusCode === 302) {
        // Handle redirects
        file.close();
        fs.unlink(destination, () => {});
        downloadFile(response.headers.location, destination)
          .then(resolve)
          .catch(reject);
        return;
      }

      if (response.statusCode !== 200) {
        file.close();
        fs.unlink(destination, () => {});
        reject(
          new Error(`Failed to download: HTTP ${response.statusCode}`)
        );
        return;
      }

      response.pipe(file);

      file.on("finish", () => {
        file.close();
        const stats = fs.statSync(destination);
        resolve({
          success: true,
          fileSize: stats.size,
          message: `File downloaded successfully (${stats.size} bytes)`,
        });
      });
    });

    request.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });

    file.on("error", (err) => {
      file.close();
      fs.unlink(destination, () => {});
      reject(err);
    });
  });
}

function extractZip(zipPath, destinationDir) {
  try {
    // Use unzip command (available on most Unix systems)
    // For Windows, you might need a different approach
    const output = execSync(`unzip -o "${zipPath}" -d "${destinationDir}"`, {
      encoding: "utf8",
    });

    // Parse extracted files from output
    const extractedFiles = output
      .split("\n")
      .filter((line) => line.includes("inflating:") || line.includes("creating:"))
      .map((line) => line.trim().replace(/^(inflating:|creating:)\s+/, ""));

    return {
      success: true,
      extractedFiles: extractedFiles,
      message: `Successfully extracted ${extractedFiles.length} files/directories`,
    };
  } catch (error) {
    // Try alternative: use Node.js built-in or another method
    try {
      // Alternative using python (usually available)
      execSync(
        `python3 -c "import zipfile; zipfile.ZipFile('${zipPath}').extractall('${destinationDir}')"`,
        { encoding: "utf8" }
      );
      return {
        success: true,
        extractedFiles: [],
        message: "Successfully extracted ZIP file using Python",
      };
    } catch (pythonError) {
      return {
        success: false,
        error: `Failed to extract ZIP: ${error.message}`,
      };
    }
  }
}

function createDirectory(dirPath) {
  try {
    fs.mkdirSync(dirPath, { recursive: true });
    return {
      success: true,
      message: `Directory created: ${dirPath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: `Failed to create directory: ${error.message}`,
    };
  }
}

function cleanupFile(filePath) {
  try {
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
      return {
        success: true,
        message: `File removed: ${filePath}`,
      };
    }
    return {
      success: true,
      message: `File not found (already removed): ${filePath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: `Failed to remove file: ${error.message}`,
    };
  }
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      try {
        result = await downloadFile(toolInput.url, toolInput.destination);
      } catch (error) {
        result = { success: false, error: error.message };
      }
      break;

    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.destination_dir);
      break;

    case "create_directory":
      result = createDirectory(toolInput.path);
      break;

    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;

    default:
      result = { success: false, error: `Unknown tool: ${toolName}` };
  }

  console.log("Result:", JSON.stringify(result, null, 2));
  return result;
}

// Main function to download and extract update
async function downloadAndExtractUpdate(updateUrl, extractPath = "/app") {
  console.log(`\nStarting update process...`);
  console.log(`URL: ${updateUrl}`);
  console.log(`Extract path: ${extractPath}`);

  const messages = [
    {
      role: "user",
      content: `Please help me download a ZIP update from "${updateUrl}" and extract it to "${extractPath}". 
      
      Follow these steps:
      1. Create the destination directory "${extractPath}" if it doesn't exist
      2. Download the ZIP file from the URL to a temporary location (use /tmp/update.zip)
      3. Extract the ZIP file to "${extractPath}"
      4. Clean up the temporary ZIP file
      
      Please proceed with these steps and report the results.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nClaude response (stop_reason: ${response.stop_reason}):`);

    // Add assistant response to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    if (response.stop_reason === "tool_use") {
      // Process tool calls
      const toolResults = [];

      for (const block of response.content) {
        if (block.type === "tool_use") {
          console.log(`\nTool call: ${block.name}`);
          const result = await processToolCall(block.name, block.input);

          toolResults.push({
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          });
        } else if (block.type === "text") {
          console.log(`Claude: ${block.text}`);
        }
      }

      // Add tool results to messages
      if (toolResults.length > 0) {
        messages.push({
          role: "user",
          content: toolResults,
        });
      }
    } else if (response.stop_reason === "end_turn") {
      // Final response
      for (const block of response.content) {
        if (block.type === "text") {
          console.log(`\nFinal response from Claude:\n${block.text}`);
        }
      }
      continueLoop = false;
    } else {
      console.log(`Unexpected stop reason: ${response.stop_reason}`);
      continueLoop = false;
    }
  }

  console.log("\nUpdate process completed!");
}

// Example usage
const updateUrl =
  process.argv[2] ||
  "https://github.com/nicowillis/nicowillis.github.io/archive/refs/heads/master.zip";
const extractPath = process.argv[3] || "./app";

downloadAndExtractUpdate(updateUrl, extractPath).catch((error) => {
  console.error("Error:", error);
  process.exit(1);
});
