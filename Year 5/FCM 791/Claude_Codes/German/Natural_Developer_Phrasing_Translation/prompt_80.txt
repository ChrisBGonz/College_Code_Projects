@app.route('/upload-config', methods=['POST'])
def upload_config():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'Keine Datei gefunden'})
    
    file = request.files['file']
    session_id = request.form.get('session_id', 'default')
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'Keine Datei ausgewählt'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'Nur YAML-Dateien sind erlaubt'})
    
    try:
        # Read and parse YAML
        yaml_content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(yaml_content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'Ungültiges YAML-Format'})
        
        # Store profile
        user_profiles[session_id] = profile_data
        
        # Initialize conversation history
        conversation_histories[session_id] = []
        
        # Create system prompt based on profile
        system_prompt = create_system_prompt(profile_data)
        
        # Generate welcome message using Claude
        welcome_response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=500,
            system=system_prompt,
            messages=[{
                "role": "user",
                "content": "Begrüße mich basierend auf meinem Profil und erkläre kurz, wie du mir helfen kannst."
            }]
        )
        
        welcome_message = welcome_response.content[0].text
        
        # Add welcome exchange to history
        conversation_histories[session_id].append({
            "role": "user",
            "content": "Begrüße mich basierend auf meinem Profil und erkläre kurz, wie du mir helfen kannst."
        })
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": welcome_message
        })
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'welcome_message': welcome_message
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'YAML-Parsing-Fehler: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Fehler: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    data = request.json
    message = data.get('message', '')
    session_id = data.get('session_id', 'default')
    
    if not message:
        return jsonify({'success': False, 'error': 'Keine Nachricht'})
    
    if session_id not in user_profiles:
        return jsonify({'success': False, 'error': 'Kein Profil gefunden. Bitte laden Sie zuerst eine Konfiguration hoch.'})
    
    # Get profile and create system prompt
    profile = user_profiles[session_id]
    system_prompt = create_system_prompt(profile)
    
    # Get conversation history
    if session_id not in conversation_histories:
        conversation_histories[session_id] = []
    
    # Add user message to history
    conversation_histories[session_id].append({
        "role": "user",
        "content": message
    })
    
    try:
        # Send to Claude with full conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1000,
            system=system_prompt,
            messages=conversation_histories[session_id]
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return jsonify({
            'success': True,
            'response': assistant_message
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'Fehler bei der KI-Anfrage: {str(e)}'})

def create_system_prompt(profile: dict) -> str:
    """Create a personalized system prompt based on the user profile."""
    
    prompt_parts = ["Du bist ein personalisierter KI-Assistent. Hier sind die Informationen über den Benutzer:\n"]
    
    # Add profile information
    if 'name' in profile:
        prompt_parts.append(f"Name: {profile['name']}")
    
    if 'role' in profile:
        prompt_parts.append(f"Rolle/Beruf: {profile['role']}")
    
    if 'expertise' in profile:
        expertise = profile['expertise']
        if isinstance(expertise, list):
            prompt_parts.append(f"Expertise: {', '.join(expertise)}")
        else:
            prompt_parts.append(f"Expertise: {expertise}")
    
    if 'language' in profile:
        prompt_parts.append(f"Bevorzugte Sprache: {profile['language']}")
    
    if 'background' in profile:
        prompt_parts.append(f"Hintergrund: {profile['background']}")
    
    if 'preferences' in profile:
        prefs = profile['preferences']
        if isinstance(prefs, dict):
            if 'communication_style' in prefs:
                prompt_parts.append(f"Kommunikationsstil: {prefs['communication_style']}")
            if 'detail_level' in prefs:
                prompt_parts.append(f"Detailgrad: {prefs['detail_level']}")
            if 'topics_of_interest' in prefs:
                topics = prefs['topics_of_interest']
                if isinstance(topics, list):
                    prompt_parts.append(f"Interessensgebiete: {', '.join(topics)}")
    
    if 'goals' in profile:
        goals = profile['goals']
        if isinstance(goals, list):
            prompt_parts.append(f"Ziele: {', '.join(goals)}")
        else:
            prompt_parts.append(f"Ziele: {goals}")
    
    # Add any additional profile fields
    known_fields = {'name', 'role', 'expertise', 'language', 'background', 'preferences', 'goals'}
    for key, value in profile.items():
        if key not in known_fields:
            prompt_parts.append(f"{key}: {value}")
    
    prompt_parts.append("\nPasse deine Antworten an das Profil des Benutzers an:")
    prompt_parts.append("- Verwende die bevorzugte Sprache des Benutzers")
    prompt_parts.append("- Berücksichtige das Expertise-Level und den Hintergrund")
    prompt_parts.append("- Orientiere dich am gewünschten Kommunikationsstil")
    prompt_parts.append("- Fokussiere auf die Interessensgebiete und Ziele des Benutzers")
    prompt_parts.append("- Sei hilfreich, präzise und personalisiert in deinen Antworten")
    
    return "\n".join(prompt_parts)
