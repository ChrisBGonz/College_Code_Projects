package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

// TokenInfo represents a password reset token with its metadata
type TokenInfo struct {
	Token     string
	ExpiresAt time.Time
	UserID    string
}

// generateSecureToken creates a cryptographically secure random token
func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate random bytes: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// createPasswordResetToken creates a secure password reset token that expires in 1 hour
func createPasswordResetToken(userID string) (*TokenInfo, error) {
	// Generate a 32-byte (256-bit) secure random token
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to create token: %w", err)
	}

	// Set expiration to 1 hour from now
	expiresAt := time.Now().Add(1 * time.Hour)

	return &TokenInfo{
		Token:     token,
		ExpiresAt: expiresAt,
		UserID:    userID,
	}, nil
}

// isTokenValid checks if a token is still valid (not expired)
func isTokenValid(tokenInfo *TokenInfo) bool {
	return time.Now().Before(tokenInfo.ExpiresAt)
}

// explainTokenCreationWithClaude uses Claude AI to explain the token creation process
func explainTokenCreationWithClaude(tokenInfo *TokenInfo) error {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`I've created a secure password reset token with the following properties:
- Token: %s (first 8 chars shown for security)
- Expires At: %s
- User ID: %s
- Is Valid: %v

Please explain in Japanese:
1. Why this token generation approach is secure
2. What makes it expire after 1 hour
3. Best practices for storing and using such tokens in a production system`,
		tokenInfo.Token[:8]+"...",
		tokenInfo.ExpiresAt.Format("2006-01-02 15:04:05 MST"),
		tokenInfo.UserID,
		isTokenValid(tokenInfo))

	fmt.Println("\n=== Claude AIによる説明 ===")
	fmt.Println("Claudeにトークン生成の説明を依頼中...")
	fmt.Println()

	stream := client.Messages.NewStreaming(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return fmt.Errorf("streaming error: %w", err)
	}

	fmt.Println()
	return nil
}

func main() {
	fmt.Println("=== セキュアなパスワードリセットトークン生成システム ===")
	fmt.Println()

	// Example user ID
	userID := "user_12345"

	// Create a password reset token
	fmt.Printf("ユーザーID '%s' のパスワードリセットトークンを生成中...\n", userID)

	tokenInfo, err := createPasswordResetToken(userID)
	if err != nil {
		fmt.Printf("エラー: トークンの生成に失敗しました: %v\n", err)
		return
	}

	fmt.Println("\n=== 生成されたトークン情報 ===")
	fmt.Printf("トークン: %s\n", tokenInfo.Token)
	fmt.Printf("有効期限: %s\n", tokenInfo.ExpiresAt.Format("2006-01-02 15:04:05 MST"))
	fmt.Printf("ユーザーID: %s\n", tokenInfo.UserID)
	fmt.Printf("有効状態: %v\n", isTokenValid(tokenInfo))
	fmt.Printf("残り時間: 約 %.0f 分\n", time.Until(tokenInfo.ExpiresAt).Minutes())

	// Demonstrate token validation
	fmt.Println("\n=== トークン検証デモ ===")
	if isTokenValid(tokenInfo) {
		fmt.Println("✓ トークンは有効です（1時間以内に期限切れ）")
	} else {
		fmt.Println("✗ トークンは期限切れです")
	}

	// Create an expired token for demonstration
	expiredToken := &TokenInfo{
		Token:     "expired_token_example",
		ExpiresAt: time.Now().Add(-1 * time.Hour), // 1 hour ago
		UserID:    userID,
	}

	fmt.Printf("\n期限切れトークンの検証: %v\n", isTokenValid(expiredToken))

	// Use Claude to explain the token creation process
	if err := explainTokenCreationWithClaude(tokenInfo); err != nil {
		fmt.Printf("Claude APIエラー: %v\n", err)
		return
	}

	fmt.Println("\n=== 処理完了 ===")
}
