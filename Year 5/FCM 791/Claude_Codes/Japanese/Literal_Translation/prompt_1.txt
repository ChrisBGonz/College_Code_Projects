tools = [
    {
        "name": "get_invoice_from_database",
        "description": "データベースから請求書の詳細情報を取得します。invoice_idを使用して特定の請求書を検索します。",
        "input_schema": {

def get_invoice_from_database(invoice_id: str) -> dict:
    """データベースから請求書を取得する関数"""
    if invoice_id in INVOICES_DB:
        return {
            "success": True,
            "invoice": INVOICES_DB[invoice_id]
        }
    else:
        return {
            "success": False,
            "error": f"請求書ID '{invoice_id}' が見つかりません"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """ツール呼び出しを処理する関数"""
    if tool_name == "get_invoice_from_database":
        result = get_invoice_from_database(tool_input["invoice_id"])
        return json.dumps(result, ensure_ascii=False)
    return json.dumps({"error": "Unknown tool"})

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    URLからinvoice_idを受け取り、データベースから請求書の詳細を取得して返すルート
    
    Args:
        invoice_id: URLパラメータとして渡される請求書ID
    
    Returns:
        JSON形式の請求書詳細情報
    """
    client = anthropic.Anthropic()
    
    messages = [
        {
            "role": "user",
            "content": f"請求書ID '{invoice_id}' の詳細情報をデータベースから取得してください。"
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        tool_result = process_tool_call(tool_use_block.name, tool_use_block.input)
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text = block.text
            break
    
    tool_result_data = None
    for msg in messages:
        if isinstance(msg.get("content"), list):
            for content_item in msg["content"]:
                if isinstance(content_item, dict) and content_item.get("type") == "tool_result":
                    tool_result_data = json.loads(content_item["content"])
                    break
    
    if tool_result_data and tool_result_data.get("success"):
        return jsonify({
            "status": "success",
            "invoice": tool_result_data["invoice"],
            "message": final_text
        }), 200
    elif tool_result_data and not tool_result_data.get("success"):
        return jsonify({
            "status": "error",
            "error": tool_result_data.get("error", "請求書が見つかりません"),
            "message": final_text
        }), 404
    else:
        return jsonify({
            "status": "error",
            "error": "請求書の取得に失敗しました",
            "message": final_text
        }), 500
