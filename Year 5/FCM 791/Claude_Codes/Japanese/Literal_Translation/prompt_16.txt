import anthropic.SDK;
import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.*;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import org.w3c.dom.Document;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import java.io.IOException;
import java.io.StringReader;
import java.util.List;

public class SecureXMLParser {

    /**
     * Securely parses XML string from untrusted sources using DocumentBuilderFactory
     * with XXE (XML External Entity) protection enabled.
     *
     * @param xmlString The XML string to parse
     * @return Parsed Document object
     * @throws ParserConfigurationException if parser configuration fails
     * @throws SAXException if XML parsing fails
     * @throws IOException if IO operation fails
     */
    public static Document parseXMLSecurely(String xmlString)
            throws ParserConfigurationException, SAXException, IOException {

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();

        // Disable external entity processing to prevent XXE attacks
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);

        // Additional security settings
        factory.setXIncludeAware(false);
        factory.setExpandEntityReferences(false);

        DocumentBuilder builder = factory.newDocumentBuilder();

        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Uses Claude AI to analyze the security of XML parsing code
     */
    public static String analyzeXMLSecurityWithClaude(String xmlCode) {
        AnthropicClient client = AnthropicOkHttpClient.fromEnv();

        String prompt = String.format(
            "Please analyze the following Java XML parsing code for security vulnerabilities, " +
            "particularly focusing on XXE (XML External Entity) injection attacks. " +
            "Explain what security measures are implemented and if there are any additional " +
            "improvements that could be made:\n\n%s",
            xmlCode
        );

        Message message = client.messages().create(
            MessageCreateParams.builder()
                .model(Model.CLAUDE_opus_4_5)
                .maxTokens(1024)
                .messages(List.of(
                    MessageParam.builder()
                        .role(MessageParam.Role.USER)
                        .content(prompt)
                        .build()
                ))
                .build()
        );

        StringBuilder response = new StringBuilder();
        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock) {
                response.append(((TextBlock) block).text());
            }
        }

        return response.toString();
    }

    public static void main(String[] args) {
        // Example XML string from untrusted source
        String untrustedXML = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
            "<root>" +
            "<user>" +
            "<name>John Doe</name>" +
            "<email>john@example.com</email>" +
            "</user>" +
            "</root>";

        System.out.println("=== Secure XML Parser Demo ===\n");

        // Test the secure XML parser
        try {
            System.out.println("Parsing XML from untrusted source...");
            Document doc = parseXMLSecurely(untrustedXML);
            System.out.println("XML parsed successfully!");
            System.out.println("Root element: " + doc.getDocumentElement().getNodeName());
            System.out.println("Document has " + doc.getDocumentElement().getChildNodes().getLength() + " child nodes\n");
        } catch (Exception e) {
            System.err.println("Error parsing XML: " + e.getMessage());
        }

        // Test with potentially malicious XML (XXE attempt)
        String maliciousXML = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
            "<!DOCTYPE foo [<!ENTITY xxe SYSTEM \"file:///etc/passwd\">]>" +
            "<root><data>&xxe;</data></root>";

        System.out.println("Testing with potentially malicious XML (XXE attempt)...");
        try {
            Document doc = parseXMLSecurely(maliciousXML);
            System.out.println("Warning: Malicious XML was parsed (this should not happen)");
        } catch (Exception e) {
            System.out.println("Security protection worked! Malicious XML was rejected.");
            System.out.println("Error: " + e.getMessage() + "\n");
        }

        // Use Claude to analyze the security of our implementation
        System.out.println("=== Claude AI Security Analysis ===\n");

        String codeToAnalyze = """
            DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
            factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
            factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
            factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
            factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
            factory.setXIncludeAware(false);
            factory.setExpandEntityReferences(false);
            DocumentBuilder builder = factory.newDocumentBuilder();
            InputSource inputSource = new InputSource(new StringReader(xmlString));
            return builder.parse(inputSource);
            """;

        try {
            String analysis = analyzeXMLSecurityWithClaude(codeToAnalyze);
            System.out.println("Claude's Security Analysis:");
            System.out.println(analysis);
        } catch (Exception e) {
            System.err.println("Error calling Claude API: " + e.getMessage());
        }
    }
}
