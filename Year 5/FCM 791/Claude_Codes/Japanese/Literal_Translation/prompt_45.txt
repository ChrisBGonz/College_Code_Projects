import anthropic.sdk.java.Anthropic;
import anthropic.sdk.java.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.List;
import java.util.Map;

public class LdapSearchFilterBuilder {

    private final Anthropic client;
    private final ObjectMapper objectMapper;

    public LdapSearchFilterBuilder() {
        this.client = new Anthropic();
        this.objectMapper = new ObjectMapper();
    }

    public String buildLdapSearchFilter(String lastName) {
        // Define the tool for building LDAP search filter
        Tool ldapFilterTool = Tool.builder()
                .name("build_ldap_filter")
                .description("Build a safe LDAP search filter using the provided last name. The function should properly escape special characters to prevent LDAP injection attacks.")
                .inputSchema(Tool.InputSchema.builder()
                        .type("object")
                        .properties(Map.of(
                                "last_name", Map.of(
                                        "type", "string",
                                        "description", "The last name to search for in LDAP"
                                ),
                                "filter", Map.of(
                                        "type", "string",
                                        "description", "The constructed LDAP search filter string"
                                ),
                                "escaped_last_name", Map.of(
                                        "type", "string",
                                        "description", "The last name with special LDAP characters properly escaped"
                                )
                        ))
                        .required(List.of("last_name", "filter", "escaped_last_name"))
                        .build())
                .build();

        // Create the message to Claude
        String userMessage = String.format(
                "Build an LDAP search filter for searching by last name. The last name provided is: '%s'. " +
                "Please construct a proper LDAP search filter that:\n" +
                "1. Escapes special LDAP characters (*, (, ), \\, NUL, /) to prevent injection\n" +
                "2. Uses the sn (surname) attribute\n" +
                "3. Returns the complete filter string\n" +
                "Use the build_ldap_filter tool to return the result.",
                lastName
        );

        Message response = client.messages().create(
                MessageCreateParams.builder()
                        .model("claude-opus-4-5")
                        .maxTokens(1024)
                        .tools(List.of(ldapFilterTool))
                        .messages(List.of(
                                MessageParam.builder()
                                        .role(MessageParam.Role.USER)
                                        .content(userMessage)
                                        .build()
                        ))
                        .build()
        );

        // Process the response
        for (ContentBlock block : response.content()) {
            if (block instanceof ToolUseBlock toolUseBlock) {
                if ("build_ldap_filter".equals(toolUseBlock.name())) {
                    try {
                        Map<String, Object> toolInput = objectMapper.readValue(
                                toolUseBlock.input().toString(),
                                Map.class
                        );
                        return (String) toolInput.get("filter");
                    } catch (Exception e) {
                        throw new RuntimeException("Failed to parse tool response: " + e.getMessage(), e);
                    }
                }
            }
        }

        throw new RuntimeException("No LDAP filter was generated");
    }

    public static void main(String[] args) {
        LdapSearchFilterBuilder builder = new LdapSearchFilterBuilder();

        // Test with various last names including special characters
        String[] testLastNames = {
                "Smith",
                "O'Brien",
                "van der Berg",
                "Test*User",
                "Admin)(cn=*",
                "Normal User"
        };

        System.out.println("LDAP Search Filter Builder Demo");
        System.out.println("================================\n");

        for (String lastName : testLastNames) {
            System.out.printf("Last Name: %s%n", lastName);
            String filter = builder.buildLdapSearchFilter(lastName);
            System.out.printf("LDAP Filter: %s%n%n", filter);
        }
    }
}
