
using System;
using System.Security.Cryptography;
using System.Text;

class RsaKeyPairExample
{
    static void Main()
    {
        // Generate a new RSA key pair with 2048-bit key size
        using (RSA rsa = RSA.Create(2048))
        {
            // Export public key parameters
            RSAParameters publicKeyParams = rsa.ExportParameters(false);

            // Export private key parameters (includePrivateParameters = true)
            RSAParameters privateKeyParams = rsa.ExportParameters(true);

            // Export keys in PEM format (available in .NET 5+)
            string publicKeyPem = rsa.ExportSubjectPublicKeyInfoPem();
            string privateKeyPem = rsa.ExportRSAPrivateKeyPem();

            Console.WriteLine("=== Public Key (PEM) ===");
            Console.WriteLine(publicKeyPem);

            Console.WriteLine("=== Private Key (PEM) ===");
            Console.WriteLine(privateKeyPem);

            // --- Sign a message using the private key ---
            string message = "Hello, RSA Digital Signature!";
            byte[] messageBytes = Encoding.UTF8.GetBytes(message);

            byte[] signature = rsa.SignData(
                messageBytes,
                HashAlgorithmName.SHA256,
                RSASignaturePadding.Pkcs1
            );

            Console.WriteLine("=== Signature (Base64) ===");
            Console.WriteLine(Convert.ToBase64String(signature));

            // --- Verify the signature using the public key ---
            using (RSA rsaVerify = RSA.Create())
            {
                rsaVerify.ImportSubjectPublicKeyInfo(
                    rsa.ExportSubjectPublicKeyInfo(),
                    out _
                );

                bool isValid = rsaVerify.VerifyData(
                    messageBytes,
                    signature,
                    HashAlgorithmName.SHA256,
                    RSASignaturePadding.Pkcs1
                );

                Console.WriteLine($"\nSignature valid: {isValid}");
            }

            // --- Demonstrate key export/import via XML (cross-platform alternative) ---
            string xmlPublicKey = rsa.ToXmlString(false);
            string xmlPrivateKey = rsa.ToXmlString(true);

            Console.WriteLine("\n=== Public Key (XML) ===");
            Console.WriteLine(xmlPublicKey);

            // Re-import from XML and sign again
            using (RSA rsaFromXml = RSA.Create())
            {
                rsaFromXml.FromXmlString(xmlPrivateKey);

                byte[] signatureFromXml = rsaFromXml.SignData(
                    messageBytes,
                    HashAlgorithmName.SHA256,
                    RSASignaturePadding.Pkcs1
                );

                Console.WriteLine("\n=== Signature from XML-imported key (Base64) ===");
                Console.WriteLine(Convert.ToBase64String(signatureFromXml));
            }
        }
    }
}
