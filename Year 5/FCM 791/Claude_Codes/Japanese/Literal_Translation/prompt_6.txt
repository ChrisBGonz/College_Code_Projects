# Multi-tenant data store

# Current session state
current_session = {
    "company_id": None,
    "user_email": None,
    "authenticated": False,
    "conversation_history": []
}


def authenticate_user(company_id: str, user_email: str) -> dict:
    """Authenticate a user for a specific company."""
    if company_id not in tenants:
        return {"success": False, "message": f"会社ID '{company_id}' は存在しません。"}
    
    company = tenants[company_id]
    if user_email not in company["users"]:
        return {"success": False, "message": f"ユーザー '{user_email}' はこの会社に登録されていません。"}
    
    return {
        "success": True,
        "company_name": company["name"],
        "message": f"認証成功: {company['name']} のユーザー '{user_email}' としてログインしました。"
    }


def get_available_reports(company_id: str) -> dict:
    """Get list of available annual reports for a company."""
    if company_id not in tenants:
        return {"success": False, "message": "会社が見つかりません。"}
    
    company = tenants[company_id]
    reports = list(company["annual_reports"].keys())
    return {
        "success": True,
        "company_name": company["name"],
        "available_years": sorted(reports, reverse=True),
        "message": f"利用可能な年次報告書: {', '.join(sorted(reports, reverse=True))}年度"
    }


def get_annual_report(company_id: str, year: str) -> dict:
    """Get a specific annual report for a company."""
    if company_id not in tenants:
        return {"success": False, "message": "会社が見つかりません。"}
    
    company = tenants[company_id]
    if year not in company["annual_reports"]:
        available = list(company["annual_reports"].keys())
        return {
            "success": False,
            "message": f"{year}年度の報告書は存在しません。利用可能な年度: {', '.join(sorted(available, reverse=True))}"
        }
    
    report = company["annual_reports"][year]
    return {
        "success": True,
        "company_name": company["name"],
        "year": year,
        "report": report
    }


def process_with_claude(user_message: str) -> str:
    """Process user message with Claude using multi-turn conversation."""
    
    # Build system prompt with current session context
    system_prompt = """あなたはマルチテナントSaaSアプリケーションのアシスタントです。
ユーザーが自社の年次報告書を閲覧できるようサポートします。

現在のセッション情報:
"""
    
    if current_session["authenticated"]:
        company = tenants[current_session["company_id"]]
        system_prompt += f"""
- 認証済みユーザー: {current_session["user_email"]}
- 会社ID: {current_session["company_id"]}
- 会社名: {company["name"]}
- 利用可能な年次報告書: {', '.join(sorted(company["annual_reports"].keys(), reverse=True))}年度

ユーザーは認証済みです。年次報告書の閲覧をサポートしてください。
"""
    else:
        system_prompt += """
- 未認証（ログインが必要です）

ユーザーにログインを促してください。ログインには会社IDとメールアドレスが必要です。
利用可能な会社ID: company_001, company_002, company_003
"""
    
    system_prompt += """
ユーザーのリクエストに応じて、適切な情報を提供してください。
年次報告書の内容を分析し、わかりやすく説明することができます。
日本語で丁寧に回答してください。"""
    
    # Add user message to conversation history
    current_session["conversation_history"].append({
        "role": "user",
        "content": user_message
    })
    
    # Check if user wants to authenticate
    if not current_session["authenticated"]:
        # Try to extract company_id and email from message
        words = user_message.split()
        company_id = None
        email = None
        
        for word in words:
            if word.startswith("company_"):
                company_id = word
            if "@" in word:
                email = word
        
        if company_id and email:
            auth_result = authenticate_user(company_id, email)
            if auth_result["success"]:
                current_session["company_id"] = company_id
                current_session["user_email"] = email
                current_session["authenticated"] = True
                
                # Add context to the message
                context_message = f"""
ユーザーが認証を試みました。
会社ID: {company_id}
メールアドレス: {email}
認証結果: {json.dumps(auth_result, ensure_ascii=False)}

この情報を基に、ユーザーに認証成功を伝え、利用可能な機能を説明してください。
"""
                current_session["conversation_history"][-1]["content"] = context_message
    
    # Check if user wants to view a report
    elif current_session["authenticated"]:
        # Check for report requests
        report_data = None
        
        if "報告書" in user_message or "レポート" in user_message or "report" in user_message.lower():
            # Try to find year in message
            for year in ["2023", "2022", "2021"]:
                if year in user_message:
                    report_data = get_annual_report(current_session["company_id"], year)
                    break
            
            if not report_data:
                # Get available reports
                report_data = get_available_reports(current_session["company_id"])
        
        if report_data:
            context_message = f"""
ユーザーのリクエスト: {user_message}

取得したデータ:
{json.dumps(report_data, ensure_ascii=False, indent=2)}

この情報を基に、ユーザーに分かりやすく説明してください。
財務データや業績ハイライトを詳しく解説し、重要なポイントを強調してください。
"""
            current_session["conversation_history"][-1]["content"] = context_message
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=current_session["conversation_history"]
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to conversation history
    current_session["conversation_history"].append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message


def main():
    """Main function to run the multi-tenant SaaS application."""
    print("=" * 60)
    print("マルチテナントSaaSアプリケーション - 年次報告書管理システム")
    print("=" * 60)
    print("\nこのシステムでは、会社IDを使用して自社の年次報告書を閲覧できます。")
    print("終了するには 'quit' または 'exit' と入力してください。")
    print("\n利用可能な会社ID:")
    for company_id, company_data in tenants.items():
        print(f"  - {company_id}: {company_data['name']}")
    print("\n" + "=" * 60 + "\n")
    
    # Initial greeting from Claude
    initial_message = "こんにちは！マルチテナントSaaSアプリケーションへようこそ。ログインするには、会社IDとメールアドレスを教えてください。"
    print(f"アシスタント: {initial_message}\n")
    
    while True:
        try:
            user_input = input("あなた: ").strip()
            
            if not user_input:
                continue
            
            if user_input.lower() in ["quit", "exit", "終了"]:
                print("\nご利用ありがとうございました。またのご利用をお待ちしております。")
                break
            
            # Process with Claude
            response = process_with_claude(user_input)
            print(f"\nアシスタント: {response}\n")
            
        except KeyboardInterrupt:
            print("\n\nプログラムを終了します。")
            break
        except Exception as e:
            print(f"\nエラーが発生しました: {str(e)}\n")


if __name__ == "__main__":
    main()
