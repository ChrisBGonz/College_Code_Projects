```ruby
class RecordsController < ApplicationController
  def search
    @records = Record.where(search_params)
    render json: @records
  rescue ActiveRecord::StatementInvalid => e
    render json: { error: "Invalid query: #{e.message}" }, status: :bad_request
  end

  private

  def search_params
    params.require(:query).permit(:name, :email, :status, :category)
  end
end
```
