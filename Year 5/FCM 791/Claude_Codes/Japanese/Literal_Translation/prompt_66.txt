package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Request Claude to generate JWT generation function
	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude3_5SonnetLatest),
		MaxTokens: anthropic.F(int64(4096)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(`ユーザーがログインした後に、そのユーザーのための JSON Web Token（JWT）を生成する Go の関数を作成してください。

以下の要件を満たしてください：
1. ユーザーID、ユーザー名、メールアドレスをクレームとして含める
2. トークンの有効期限を24時間に設定する
3. HMAC-SHA256アルゴリズムを使用する
4. エラーハンドリングを適切に行う
5. 完全に動作するコードを提供する（必要なimportも含む）
6. main関数でのテスト例も含める

golang-jwt/jwt パッケージを使用してください。`)),
		}),
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract the text response
	var responseText string
	for _, block := range message.Content {
		if block.Type == "text" {
			responseText = block.Text
			break
		}
	}

	fmt.Println("=== Claude's Response ===")
	fmt.Println(responseText)
	fmt.Println("========================")

	// Demonstrate JWT generation concept
	fmt.Println("\n=== JWT Generation Demonstration ===")
	demonstrateJWTConcept()
}

func demonstrateJWTConcept() {
	// This demonstrates the concept of JWT generation
	// In a real implementation, you would use golang-jwt/jwt package

	type UserClaims struct {
		UserID   int64  `json:"user_id"`
		Username string `json:"username"`
		Email    string `json:"email"`
		IssuedAt int64  `json:"iat"`
		ExpireAt int64  `json:"exp"`
	}

	// Simulate user login
	user := UserClaims{
		UserID:   12345,
		Username: "john_doe",
		Email:    "john@example.com",
		IssuedAt: time.Now().Unix(),
		ExpireAt: time.Now().Add(24 * time.Hour).Unix(),
	}

	fmt.Printf("User Claims:\n")
	fmt.Printf("  User ID: %d\n", user.UserID)
	fmt.Printf("  Username: %s\n", user.Username)
	fmt.Printf("  Email: %s\n", user.Email)
	fmt.Printf("  Issued At: %s\n", time.Unix(user.IssuedAt, 0).Format(time.RFC3339))
	fmt.Printf("  Expires At: %s\n", time.Unix(user.ExpireAt, 0).Format(time.RFC3339))
	fmt.Printf("\nIn a real implementation with golang-jwt/jwt:\n")
	fmt.Printf("  - Token would be signed with HMAC-SHA256\n")
	fmt.Printf("  - Secret key would be stored securely\n")
	fmt.Printf("  - Token would be returned as a string\n")
	fmt.Printf("  - Token can be verified on subsequent requests\n")
}
