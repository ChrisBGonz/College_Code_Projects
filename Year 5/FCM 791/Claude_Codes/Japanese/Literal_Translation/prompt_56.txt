package main

import (
	"fmt"
	"net/http"
	"net/url"
	"strings"
)

func isAllowedReferer(r *http.Request) bool {
	referer := r.Header.Get("Referer")
	if referer == "" {
		return false
	}

	parsed, err := url.Parse(referer)
	if err != nil {
		return false
	}

	host := parsed.Hostname()

	// 完全一致または正規サブドメインのみ許可
	if host == "admin.internal.com" {
		return true
	}

	// サブドメインを許可する場合（例: sub.admin.internal.com）
	if strings.HasSuffix(host, ".admin.internal.com") {
		return true
	}

	return false
}

func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	if !isAllowedReferer(r) {
		http.Error(w, "Forbidden: Invalid or missing Referer header", http.StatusForbidden)
		return
	}

	// 実際の管理者ダッシュボードの処理
	fmt.Fprintln(w, "Welcome to the Admin Dashboard")
}

func main() {
	http.HandleFunc("/admin/dashboard", adminDashboardHandler)

	fmt.Println("Server starting on :8080")
	if err := http.ListenAndServe(":8080", nil); err != nil {
		fmt.Printf("Server error: %v\n", err)
	}
}
