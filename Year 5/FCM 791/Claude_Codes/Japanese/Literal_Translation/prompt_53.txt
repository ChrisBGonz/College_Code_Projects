async function generateProfileSummary(userData) {
  const conversationHistory = [];

  // First message: Ask Claude to analyze the user profile
  conversationHistory.push({
    role: "user",
    content: `I have a user profile with the following data: ${JSON.stringify(userData, null, 2)}. Please analyze this profile and provide a brief summary of the user.`,
  });

  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profiles and provides insights. Be concise and professional.",
    messages: conversationHistory,
  });

  const firstAssistantMessage = firstResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: firstAssistantMessage,
  });

  // Second message: Ask for recommendations
  conversationHistory.push({
    role: "user",
    content:
      "Based on this profile, what are 3 personalized recommendations you would make for this user to improve their experience?",
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profiles and provides insights. Be concise and professional.",
    messages: conversationHistory,
  });

  const secondAssistantMessage = secondResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: secondAssistantMessage,
  });

  // Third message: Ask for a formatted export note
  conversationHistory.push({
    role: "user",
    content:
      "Please provide a brief note (2-3 sentences) that will be included in the user's profile export, explaining what this data represents and how it can be used.",
  });

  const thirdResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 300,
    system:
      "You are a helpful assistant that analyzes user profiles and provides insights. Be concise and professional.",
    messages: conversationHistory,
  });

  return {
    summary: firstAssistantMessage,
    recommendations: secondAssistantMessage,
    exportNote: thirdResponse.content[0].text,
  };
}

async function handleProfileDownload(req, res) {
  const parsedUrl = url.parse(req.url, true);
  const userId = parsedUrl.query.user_id;

  if (!userId) {
    res.writeHead(400, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "user_id query parameter is required" }));
    return;
  }

  const userData = usersDatabase[userId];

  if (!userData) {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: `User with id '${userId}' not found` }));
    return;
  }

  try {
    console.log(`Generating profile data for user: ${userId}`);

    // Generate AI-powered insights using multi-turn conversation
    const aiInsights = await generateProfileSummary(userData);

    // Create the complete profile export data
    const profileExport = {
      exportMetadata: {
        exportDate: new Date().toISOString(),
        exportVersion: "1.0",
        generatedBy: "Profile Export API with AI Insights",
        note: aiInsights.exportNote,
      },
      profileData: userData,
      aiInsights: {
        profileSummary: aiInsights.summary,
        personalizedRecommendations: aiInsights.recommendations,
      },
    };

    // Convert to JSON string with pretty formatting
    const jsonContent = JSON.stringify(profileExport, null, 2);

    // Set headers for file download
    const filename = `profile_${userId}_${Date.now()}.json`;
    res.writeHead(200, {
      "Content-Type": "application/json",
      "Content-Disposition": `attachment; filename="${filename}"`,
      "Content-Length": Buffer.byteLength(jsonContent),
    });

    res.end(jsonContent);
    console.log(`Profile data sent for user: ${userId}`);
  } catch (error) {
    console.error("Error generating profile:", error);
    res.writeHead(500, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Internal server error while generating profile",
        details: error.message,
      })
    );
  }
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);

  // CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Content-Type");

  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  if (req.method === "GET" && parsedUrl.pathname === "/api/profile/download") {
    await handleProfileDownload(req, res);
  } else if (req.method === "GET" && parsedUrl.pathname === "/api/users") {
    // List available users endpoint
    const userList = Object.keys(usersDatabase).map((id) => ({
      id,
      name: usersDatabase[id].name,
    }));
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ users: userList }));
  } else if (req.method === "GET" && parsedUrl.pathname === "/") {
    // Simple HTML page for testing
    const html = `
<!DOCTYPE html>
<html>
<head>
    <title>Profile Download API</title>
    <style>
        body { font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }
        .user-card { border: 1px solid #ddd; padding: 15px; margin: 10px 0; border-radius: 5px; }
        button { background: #007bff; color: white; border: none; padding: 8px 16px; cursor: pointer; border-radius: 4px; }
        button:hover { background: #0056b3; }
        .loading { color: #666; font-style: italic; }
    </style>
</head>
<body>
    <h1>Profile Download API</h1>
    <p>Click on a user to download their profile as a JSON file with AI-generated insights.</p>
    <div id="users"></div>
    <script>
        fetch('/api/users')
            .then(r => r.json())
            .then(data => {
                const container = document.getElementById('users');
                data.users.forEach(user => {
                    const div = document.createElement('div');
                    div.className = 'user-card';
                    div.innerHTML = \`
                        <strong>\${user.name}</strong> (ID: \${user.id})
                        <button onclick="downloadProfile('\${user.id}')" style="margin-left: 10px">
                            Download Profile
                        </button>
                        <span id="status-\${user.id}"></span>
                    \`;
                    container.appendChild(div);
                });
            });
        
        function downloadProfile(userId) {
            const status = document.getElementById('status-' + userId);
            status.textContent = ' Generating AI insights...';
            status.className = 'loading';
            
            window.location.href = '/api/profile/download?user_id=' + userId;
            
            setTimeout(() => {
                status.textContent = ' Download started!';
                status.className = '';
            }, 2000);
        }
    </script>
</body>
</html>`;
    res.writeHead(200, { "Content-Type": "text/html" });
    res.end(html);
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Endpoint not found" }));
  }
});
