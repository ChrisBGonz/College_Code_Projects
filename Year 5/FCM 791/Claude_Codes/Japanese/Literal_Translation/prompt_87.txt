import anthropic
import time
import random
from datetime import datetime
from collections import deque

client = anthropic.Anthropic()

LOG_FILE = "auth.log"
THRESHOLD = 10
TIME_WINDOW = 60

failed_login_times = deque()


def generate_log_entry(is_failed: bool = False) -> str:
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    if is_failed:
        users = ["admin", "root", "user1", "test", "guest"]
        user = random.choice(users)
        ips = ["192.168.1.100", "10.0.0.50", "172.16.0.1", "192.168.0.200"]
        ip = random.choice(ips)
        return f"{timestamp} - Failed Login - User: {user} from IP: {ip}"
    else:
        return f"{timestamp} - Successful Login - User: admin from IP: 192.168.1.1"


def simulate_log_entries(log_file: str, num_entries: int = 5, fail_rate: float = 0.3):
    with open(log_file, "a") as f:
        for _ in range(num_entries):
            is_failed = random.random() < fail_rate
            entry = generate_log_entry(is_failed)
            f.write(entry + "\n")
            f.flush()


def analyze_with_claude(failed_logins: list, count: int) -> str:
    recent_entries = "\n".join(failed_logins[-10:]) if failed_logins else "No entries"

    prompt = f"""セキュリティログ分析レポートを生成してください。

状況:
- 過去1分間に {count} 件の「Failed Login」が検出されました（閾値: {THRESHOLD}件）
- これはセキュリティ警告レベルの状況です

最近の失敗したログインエントリ:
{recent_entries}

以下の点について簡潔に分析してください:
1. セキュリティリスクの評価
2. 考えられる攻撃パターン
3. 推奨される即時対応策

日本語で回答してください。"""

    print("\n" + "=" * 60)
    print("🚨 セキュリティ警告: Claude AIによる分析を開始します...")
    print("=" * 60)

    full_response = ""
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text

    print("\n" + "=" * 60)
    return full_response


def monitor_log_file(log_file: str):
    print(f"ログファイル監視を開始します: {log_file}")
    print(f"閾値: {THRESHOLD}件/分")
    print("-" * 60)

    try:
        with open(log_file, "r") as f:
            f.seek(0, 2)
            file_position = f.tell()
    except FileNotFoundError:
        with open(log_file, "w") as f:
            pass
        file_position = 0

    recent_failed_logins = []
    last_alert_time = 0
    alert_cooldown = 30

    while True:
        try:
            with open(log_file, "r") as f:
                f.seek(file_position)
                new_lines = f.readlines()
                file_position = f.tell()

            current_time = time.time()

            while failed_login_times and (
                current_time - failed_login_times[0] > TIME_WINDOW
            ):
                failed_login_times.popleft()

            for line in new_lines:
                line = line.strip()
                if line:
                    print(f"[ログ] {line}")

                    if "Failed Login" in line:
                        failed_login_times.append(current_time)
                        recent_failed_logins.append(line)

                        if len(recent_failed_logins) > 20:
                            recent_failed_logins = recent_failed_logins[-20:]

            failed_count = len(failed_login_times)

            if failed_count > THRESHOLD:
                if current_time - last_alert_time > alert_cooldown:
                    print(
                        f"\n⚠️  警告: 過去1分間に {failed_count} 件の失敗したログインを検出しました！"
                    )
                    analyze_with_claude(recent_failed_logins, failed_count)
                    last_alert_time = current_time

            time.sleep(1)

        except KeyboardInterrupt:
            print("\n\n監視を終了します。")
            break
        except Exception as e:
            print(f"エラーが発生しました: {e}")
            time.sleep(1)


def run_demo():
    print("デモモード: ログファイルの監視とシミュレーションを開始します")
    print("=" * 60)

    import threading

    def simulate_attacks():
        time.sleep(2)
        print("\n[シミュレーション] 通常のログインアクティビティを生成中...")
        simulate_log_entries(LOG_FILE, num_entries=5, fail_rate=0.2)

        time.sleep(5)
        print("\n[シミュレーション] ブルートフォース攻撃をシミュレート中...")
        for _ in range(3):
            simulate_log_entries(LOG_FILE, num_entries=5, fail_rate=0.9)
            time.sleep(2)

        time.sleep(10)
        print("\n[シミュレーション] 攻撃が収まりました...")
        simulate_log_entries(LOG_FILE, num_entries=3, fail_rate=0.1)

    simulation_thread = threading.Thread(target=simulate_attacks, daemon=True)
    simulation_thread.start()

    monitor_log_file(LOG_FILE)


if __name__ == "__main__":
    import sys

    if len(sys.argv) > 1 and sys.argv[1] == "--monitor":
        log_file = sys.argv[2] if len(sys.argv) > 2 else LOG_FILE
        monitor_log_file(log_file)
    else:
        run_demo()
