```java
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseHandler.class.getName());

    // カスタム例外クラス
    public static class DatabaseAccessException extends Exception {
        private final int errorCode;
        private final String sqlState;

        public DatabaseAccessException(String message, int errorCode, String sqlState, Throwable cause) {
            super(message, cause);
            this.errorCode = errorCode;
            this.sqlState = sqlState;
        }

        public int getErrorCode() {
            return errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        @Override
        public String toString() {
            return String.format("DatabaseAccessException{message='%s', errorCode=%d, sqlState='%s'}",
                    getMessage(), errorCode, sqlState);
        }
    }

    // データベース接続情報
    private static final String DB_URL = "jdbc:mysql://localhost:3306/sampledb";
    private static final String DB_USER = "root";
    private static final String DB_PASSWORD = "password";

    /**
     * 指定されたIDのユーザー名を取得するメソッド
     *
     * @param userId 検索するユーザーID
     * @return ユーザー名
     * @throws DatabaseAccessException データベースアクセスエラーが発生した場合
     */
    public String getUserById(int userId) throws DatabaseAccessException {
        String sql = "SELECT username FROM users WHERE id = ?";
        String username = null;

        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(sql)) {

            preparedStatement.setInt(1, userId);

            try (ResultSet resultSet = preparedStatement.executeQuery()) {
                if (resultSet.next()) {
                    username = resultSet.getString("username");
                }
            }

        } catch (SQLException e) {
            // エラーメッセージをログに記録
            LOGGER.log(Level.SEVERE,
                    String.format("データベースエラーが発生しました。[ErrorCode: %d, SQLState: %s, Message: %s]",
                            e.getErrorCode(), e.getSQLState(), e.getMessage()),
                    e);

            // カスタム例外をスロー
            throw new DatabaseAccessException(
                    "ユーザー情報の取得中にデータベースエラーが発生しました。UserID: " + userId,
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }

        return username;
    }

    /**
     * ユーザーを新規登録するメソッド
     *
     * @param username 登録するユーザー名
     * @param email    登録するメールアドレス
     * @throws DatabaseAccessException データベースアクセスエラーが発生した場合
     */
    public void insertUser(String username, String email) throws DatabaseAccessException {
        String sql = "INSERT INTO users (username, email) VALUES (?, ?)";

        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(sql)) {

            preparedStatement.setString(1, username);
            preparedStatement.setString(2, email);
            preparedStatement.executeUpdate();

            LOGGER.log(Level.INFO, "ユーザーが正常に登録されました。Username: {0}", username);

        } catch (SQLException e) {
            // エラーメッセージをログに記録
            LOGGER.log(Level.SEVERE,
                    String.format("ユーザー登録中にデータベースエラーが発生しました。[Username: %s, ErrorCode: %d, SQLState: %s, Message: %s]",
                            username, e.getErrorCode(), e.getSQLState(), e.getMessage()),
                    e);

            // カスタム例外をスロー
            throw new DatabaseAccessException(
                    "ユーザー登録中にデータベースエラーが発生しました。Username: " + username,
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    // メインメソッド（動作確認用）
    public static void main(String[] args) {
        DatabaseHandler handler = new DatabaseHandler();

        // ユーザー取得の例
        try {
            String username = handler.getUserById(1);
            if (username != null) {
                System.out.println("取得したユーザー名: " + username);
            } else {
                System.out.println("指定されたIDのユーザーは存在しません。");
            }
        } catch (DatabaseAccessException e) {
            System.err.println("エラーが発生しました: " + e);
        }

        // ユーザー登録の例
        try {
            handler.insertUser("testuser", "testuser@example.com");
            System.out.println("ユーザーの登録が完了しました。");
        } catch (DatabaseAccessException e) {
            System.err.println("エラーが発生しました: " + e);
        }
    }
}
```
