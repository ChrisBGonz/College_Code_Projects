package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func logTransfer(senderID string, receiverID string, amount float64) {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`送金処理のログを記録してください。以下の情報を含む適切なログメッセージを生成してください：
- 送信者ID: %s
- 受信者ID: %s
- 金額: %.2f円

ログメッセージは日本語で、タイムスタンプ、トランザクションの詳細、および処理ステータスを含めてください。`, senderID, receiverID, amount)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		log.Printf("Error calling Claude API: %v", err)
		// Fallback to basic logging if API call fails
		log.Printf("[TRANSFER] sender_id=%s, receiver_id=%s, amount=%.2f", senderID, receiverID, amount)
		return
	}

	if len(message.Content) > 0 {
		logMessage := message.Content[0].Text
		log.Printf("[TRANSFER LOG]\n%s", logMessage)
		fmt.Printf("\n=== 送金ログ ===\n%s\n", logMessage)
	}
}

func main() {
	// Set up logging
	log.SetOutput(os.Stdout)
	log.SetFlags(log.Ldate | log.Ltime)

	// Example transfer logging
	fmt.Println("送金処理ログシステムを開始します...")
	fmt.Println()

	// Test case 1: Regular transfer
	logTransfer("USER_001", "USER_002", 50000.00)

	fmt.Println()
	fmt.Println("---")
	fmt.Println()

	// Test case 2: Large transfer
	logTransfer("CORP_123", "VENDOR_456", 1500000.00)

	fmt.Println()
	fmt.Println("---")
	fmt.Println()

	// Test case 3: Small transfer
	logTransfer("USER_789", "USER_321", 1000.00)
}
