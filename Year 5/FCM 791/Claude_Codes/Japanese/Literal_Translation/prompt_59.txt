import anthropic
import json
import re
import urllib.parse

client = anthropic.Anthropic()

tools = [
    {
        "name": "process_payment",
        "description": "Process a payment through the payment gateway and return payment details",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount in cents"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., USD, EUR)"
                },
                "payment_method": {
                    "type": "string",
                    "description": "The payment method (e.g., credit_card, bank_transfer)"
                },
                "success_url": {
                    "type": "string",
                    "description": "The URL to redirect the user after successful payment"
                },
                "cancel_url": {
                    "type": "string",
                    "description": "The URL to redirect the user if payment is cancelled"
                }
            },
            "required": ["amount", "currency", "payment_method", "success_url"]
        }
    },
    {
        "name": "redirect_user",
        "description": "Redirect the user to a specified URL",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "The URL to redirect the user to"
                },
                "payment_id": {
                    "type": "string",
                    "description": "The payment transaction ID"
                },
                "status": {
                    "type": "string",
                    "description": "The payment status (success, failed, pending)"
                }
            },
            "required": ["url", "status"]
        }
    }
]

def process_payment(amount: float, currency: str, payment_method: str, success_url: str, cancel_url: str = None) -> dict:
    """Simulate payment processing"""
    payment_id = f"PAY_{hash(f'{amount}{currency}{payment_method}') % 100000:05d}"
    
    print(f"\n[Payment Gateway] Processing payment...")
    print(f"  Amount: {amount/100:.2f} {currency}")
    print(f"  Method: {payment_method}")
    print(f"  Success URL: {success_url}")
    
    return {
        "payment_id": payment_id,
        "status": "success",
        "amount": amount,
        "currency": currency,
        "payment_method": payment_method,
        "success_url": success_url,
        "cancel_url": cancel_url,
        "transaction_time": "2024-01-15T10:30:00Z"
    }

def redirect_user(url: str, status: str, payment_id: str = None) -> dict:
    """Simulate user redirection"""
    if payment_id:
        separator = "&" if "?" in url else "?"
        redirect_url = f"{url}{separator}payment_id={payment_id}&status={status}"
    else:
        redirect_url = url
    
    print(f"\n[Redirect] Redirecting user to: {redirect_url}")
    print(f"  Status: {status}")
    
    return {
        "redirected": True,
        "final_url": redirect_url,
        "status": status
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "process_payment":
        result = process_payment(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            payment_method=tool_input["payment_method"],
            success_url=tool_input["success_url"],
            cancel_url=tool_input.get("cancel_url")
        )
        return json.dumps(result)
    elif tool_name == "redirect_user":
        result = redirect_user(
            url=tool_input["url"],
            status=tool_input["status"],
            payment_id=tool_input.get("payment_id")
        )
        return json.dumps(result)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def integrate_payment_gateway(payment_request: dict) -> dict:
    """
    Integrate with third-party payment gateway using Claude as the AI backbone.
    
    Args:
        payment_request: Dictionary containing payment details including success_url
    
    Returns:
        Dictionary with payment result and redirect information
    """
    print("\n" + "="*60)
    print("Payment Gateway Integration")
    print("="*60)
    print(f"Processing payment request: {json.dumps(payment_request, indent=2)}")
    
    initial_message = f"""
    Process the following payment request and redirect the user to the success URL upon completion:
    
    Payment Details:
    - Amount: {payment_request.get('amount', 1000)} cents
    - Currency: {payment_request.get('currency', 'USD')}
    - Payment Method: {payment_request.get('payment_method', 'credit_card')}
    - Success URL: {payment_request.get('success_url', 'https://example.com/success')}
    - Cancel URL: {payment_request.get('cancel_url', 'https://example.com/cancel')}
    
    Please:
    1. Process the payment using the payment gateway
    2. If successful, redirect the user to the success_url with the payment ID
    3. If failed, redirect to the cancel_url
    """
    
    messages = [{"role": "user", "content": initial_message}]
    
    final_result = {}
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\n[Claude] Stop reason: {response.stop_reason}")
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n[Tool Call] {tool_name}")
                    print(f"  Input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    result_data = json.loads(tool_result)
                    
                    print(f"  Result: {json.dumps(result_data, indent=2)}")
                    
                    if tool_name == "redirect_user":
                        final_result = result_data
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    print(f"\n[Claude] Final response: {content_block.text}")
            break
        else:
            print(f"[Warning] Unexpected stop reason: {response.stop_reason}")
            break
    
    return final_result

def main():
    payment_requests = [
        {
            "amount": 9999,
            "currency": "USD",
            "payment_method": "credit_card",
            "success_url": "https://mystore.com/order/success",
            "cancel_url": "https://mystore.com/order/cancel"
        },
        {
            "amount": 4999,
            "currency": "EUR",
            "payment_method": "bank_transfer",
            "success_url": "https://shop.example.com/checkout/complete?order_id=12345",
            "cancel_url": "https://shop.example.com/checkout/failed"
        }
    ]
    
    for i, request in enumerate(payment_requests, 1):
        print(f"\n{'='*60}")
        print(f"Test Case {i}")
        print(f"{'='*60}")
        
        result = integrate_payment_gateway(request)
        
        print(f"\n[Final Result]")
        print(json.dumps(result, indent=2))
        
        if result.get("redirected"):
            print(f"\n✓ User successfully redirected to: {result.get('final_url')}")
        else:
            print("\n✗ Redirection failed or not completed")

if __name__ == "__main__":
    main()
