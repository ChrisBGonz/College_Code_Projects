package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func main() {
	client := anthropic.NewClient()

	// Example environment variables for demonstration
	envVars := map[string]string{
		"APP_CONFIG_PATH":    "/etc/myapp/config.yaml",
		"DATABASE_CONFIG":    "/var/config/database.json",
		"LOG_CONFIG_FILE":    "/opt/app/logging.conf",
		"CUSTOM_CONFIG_PATH": "",
	}

	// Set environment variables for demonstration
	for key, value := range envVars {
		if value != "" {
			os.Setenv(key, value)
		}
	}

	prompt := `以下の要件を満たすGo（Golang）関数を作成してください：

1. 環境変数から設定ファイルのパスを読み取る関数
2. 複数の環境変数名をサポート（優先順位付き）
3. デフォルトパスのフォールバック機能
4. パスの存在確認
5. エラーハンドリング
6. 使用例を含む完全なコード

関数のシグネチャ例：
- GetConfigPath(envVarNames []string, defaultPath string) (string, error)

要件：
- 環境変数が設定されていない場合はデフォルトパスを使用
- ファイルが存在しない場合は適切なエラーを返す
- 複数の環境変数を優先順位順に確認
- ログ出力で読み取ったパスを表示`

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 2048,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	fmt.Println("Generated Go function for reading config file paths from environment variables:")
	fmt.Println("=============================================================================")

	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}

	// Demonstrate the concept with a simple implementation
	fmt.Println("\n=============================================================================")
	fmt.Println("Demonstration of the concept:")
	fmt.Println("=============================================================================")

	demonstrateConfigPathReading()
}

func demonstrateConfigPathReading() {
	// Simple demonstration of reading config paths from environment variables
	envVarNames := []string{"APP_CONFIG_PATH", "CONFIG_FILE", "MYAPP_CONFIG"}
	defaultPath := "/etc/myapp/default-config.yaml"

	path, source := getConfigPath(envVarNames, defaultPath)
	fmt.Printf("Config path: %s (source: %s)\n", path, source)

	// Check if file exists (for demonstration)
	if _, err := os.Stat(path); os.IsNotExist(err) {
		fmt.Printf("Note: File does not exist at path: %s\n", path)
	} else {
		fmt.Printf("File exists at path: %s\n", path)
	}
}

func getConfigPath(envVarNames []string, defaultPath string) (string, string) {
	for _, envVar := range envVarNames {
		if value := os.Getenv(envVar); value != "" {
			return value, fmt.Sprintf("environment variable '%s'", envVar)
		}
	}
	return defaultPath, "default path"
}
