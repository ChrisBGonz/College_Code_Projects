import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function createBankingAutoLogout() {
  console.log("Creating banking auto-logout function using Claude AI...\n");

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 2000,
    messages: [
      {
        role: "user",
        content: `銀行アプリ用のJavaScript関数を作成してください。以下の要件を満たす必要があります：

1. 15分間操作がない場合にユーザーを自動的にログアウトさせる
2. ユーザーのアクティビティ（マウス移動、キーボード入力、クリック、スクロール）を監視する
3. タイムアウト前に警告メッセージを表示する（例：2分前に警告）
4. ログアウト処理を実行する関数を含める
5. タイマーをリセット・停止できる機能を含める
6. セキュリティのベストプラクティスに従う

完全な実装コードと使用方法の説明を提供してください。`,
      },
    ],
  });

  process.stdout.write("Claude's Response:\n");
  process.stdout.write("=".repeat(50) + "\n");

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
    }
  }

  const finalMessage = await stream.finalMessage();

  console.log("\n" + "=".repeat(50));
  console.log("\nStream completed!");
  console.log(
    `Total tokens used: ${finalMessage.usage.input_tokens + finalMessage.usage.output_tokens}`
  );

  return finalMessage;
}
