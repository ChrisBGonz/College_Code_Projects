def generate_jwt_token(payload: dict, secret_key: str, algorithm: str = "HS256", expiration_hours: float = 24) -> dict:
    """Generate a JWT token"""
    try:
        # Add expiration time to payload
        payload_with_exp = payload.copy()
        payload_with_exp['exp'] = datetime.datetime.utcnow() + datetime.timedelta(hours=expiration_hours)
        payload_with_exp['iat'] = datetime.datetime.utcnow()
        
        # Generate token
        token = jwt.encode(payload_with_exp, secret_key, algorithm=algorithm)
        
        return {
            "success": True,
            "token": token,
            "algorithm": algorithm,
            "expires_in_hours": expiration_hours
        }
    except Exception as e:
        return {
            "success": False,
            "error": str(e)
        }

def decode_jwt_token(token: str, secret_key: str, algorithm: str = "HS256") -> dict:
    """Decode and verify a JWT token"""
    try:
        decoded = jwt.decode(token, secret_key, algorithms=[algorithm])
        
        # Convert datetime objects to strings for JSON serialization
        result = {}
        for key, value in decoded.items():
            if isinstance(value, datetime.datetime):
                result[key] = value.isoformat()
            else:
                result[key] = value
        
        return {
            "success": True,
            "decoded_payload": result,
            "is_valid": True
        }
    except jwt.ExpiredSignatureError:
        return {
            "success": False,
            "error": "Token has expired",
            "is_valid": False
        }
    except jwt.InvalidTokenError as e:
        return {
            "success": False,
            "error": f"Invalid token: {str(e)}",
            "is_valid": False
        }
    except Exception as e:
        return {
            "success": False,
            "error": str(e),
            "is_valid": False
        }

def validate_jwt_token(token: str) -> dict:
    """Validate JWT token format without verification"""
    try:
        # Check if token has three parts
        parts = token.split('.')
        if len(parts) != 3:
            return {
                "is_valid_format": False,
                "error": "Token must have exactly 3 parts separated by dots"
            }
        
        # Try to decode header and payload (without verification)
        import base64
        
        def decode_base64(data):
            # Add padding if necessary
            padding = 4 - len(data) % 4
            if padding != 4:
                data += '=' * padding
            return base64.urlsafe_b64decode(data)
        
        header = json.loads(decode_base64(parts[0]))
        payload = json.loads(decode_base64(parts[1]))
        
        return {
            "is_valid_format": True,
            "header": header,
            "payload_keys": list(payload.keys()),
            "has_expiration": 'exp' in payload,
            "has_issued_at": 'iat' in payload
        }
    except Exception as e:
        return {
            "is_valid_format": False,
            "error": str(e)
        }

def inspect_jwt_header(token: str) -> dict:
    """Inspect JWT token header without verification"""
    try:
        header = jwt.get_unverified_header(token)
        return {
            "success": True,
            "header": header,
            "algorithm": header.get('alg', 'Unknown'),
            "token_type": header.get('typ', 'Unknown')
        }
    except Exception as e:
        return {
            "success": False,
            "error": str(e)
        }

def process_tool_call(tool_name: str, tool_input: dict) -> Any:
    """Process tool calls from Claude"""
    if tool_name == "generate_jwt_token":
        return generate_jwt_token(
            tool_input["payload"],
            tool_input["secret_key"],
            tool_input.get("algorithm", "HS256"),
            tool_input.get("expiration_hours", 24)
        )
    elif tool_name == "decode_jwt_token":
        return decode_jwt_token(
            tool_input["token"],
            tool_input["secret_key"],
            tool_input.get("algorithm", "HS256")
        )
    elif tool_name == "validate_jwt_token":
        return validate_jwt_token(tool_input["token"])
    elif tool_name == "inspect_jwt_header":
        return inspect_jwt_header(tool_input["token"])
    else:
        return {"error": f"Unknown tool: {tool_name}"}

def run_jwt_assistant(user_request: str) -> str:
    """Run the JWT assistant with Claude"""
    print(f"\n{'='*60}")
    print(f"User Request: {user_request}")
    print('='*60)
    
    messages = [{"role": "user", "content": user_request}]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's Response (stop_reason: {response.stop_reason}):")
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract text response
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"\nFinal Response: {block.text}")
                    return block.text
            break
        
        # Process tool uses
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({"role": "assistant", "content": response.content})
            
            # Process each tool use
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool Called: {block.name}")
                    print(f"Tool Input: {json.dumps(block.input, indent=2)}")
                    
                    # Execute the tool
                    result = process_tool_call(block.name, block.input)
                    print(f"Tool Result: {json.dumps(result, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(result)
                    })
            
            # Add tool results to messages
            messages.append({"role": "user", "content": tool_results})
        else:
            # Unexpected stop reason
            break
    
    return "Process completed"
