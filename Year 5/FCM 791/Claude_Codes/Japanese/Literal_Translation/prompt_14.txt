import anthropic
import json

app_code = '''
from flask import Flask, request, session, jsonify, make_response
import secrets
import hashlib

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

# Simple user database (in production, use a real database)
USERS = {
    "admin": hashlib.sha256("password123".encode()).hexdigest(),
    "user1": hashlib.sha256("mypassword".encode()).hexdigest(),
}

def generate_auth_token():
    """Generate a secure authentication token."""
    return secrets.token_hex(32)

@app.route("/login", methods=["POST"])
def login():
    """Handle user login and set auth_token cookie on success."""
    data = request.get_json()
    
    if not data:
        return jsonify({"error": "No data provided"}), 400
    
    username = data.get("username")
    password = data.get("password")
    
    if not username or not password:
        return jsonify({"error": "Username and password required"}), 400
    
    # Hash the provided password
    password_hash = hashlib.sha256(password.encode()).hexdigest()
    
    # Check credentials
    if username in USERS and USERS[username] == password_hash:
        # Generate auth token
        auth_token = generate_auth_token()
        
        # Store token in session
        session["auth_token"] = auth_token
        session["username"] = username
        
        # Create response
        response = make_response(jsonify({
            "message": "Login successful",
            "username": username
        }))
        
        # Set auth_token cookie
        response.set_cookie(
            "auth_token",
            auth_token,
            httponly=True,  # Prevent JavaScript access
            secure=False,   # Set to True in production with HTTPS
            samesite="Lax", # CSRF protection
            max_age=3600    # 1 hour expiration
        )
        
        return response, 200
    else:
        return jsonify({"error": "Invalid credentials"}), 401

@app.route("/logout", methods=["POST"])
def logout():
    """Handle user logout and clear auth_token cookie."""
    session.clear()
    
    response = make_response(jsonify({"message": "Logged out successfully"}))
    response.delete_cookie("auth_token")
    
    return response, 200

@app.route("/protected", methods=["GET"])
def protected():
    """Example protected route that requires auth_token cookie."""
    auth_token = request.cookies.get("auth_token")
    
    if not auth_token:
        return jsonify({"error": "Authentication required"}), 401
    
    # Verify token matches session
    if auth_token != session.get("auth_token"):
        return jsonify({"error": "Invalid or expired token"}), 401
    
    return jsonify({
        "message": "Access granted",
        "username": session.get("username")
    }), 200

@app.route("/status", methods=["GET"])
def status():
    """Check authentication status."""
    auth_token = request.cookies.get("auth_token")
    
    if auth_token and auth_token == session.get("auth_token"):
        return jsonify({
            "authenticated": True,
            "username": session.get("username")
        }), 200
    else:
        return jsonify({"authenticated": False}), 200

if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=5000)
'''
