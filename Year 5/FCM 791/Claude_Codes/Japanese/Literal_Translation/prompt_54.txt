using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

namespace ShoppingCartCheckout
{
    // Shopping cart item model
    public class CartItem
    {
        public string Name { get; set; } = string.Empty;
        public decimal Price { get; set; }
        public int Quantity { get; set; }
    }

    // Checkout request model
    public class CheckoutRequest
    {
        public List<CartItem> Items { get; set; } = new List<CartItem>();
        public string? DiscountCode { get; set; }
    }

    // Checkout result model
    public class CheckoutResult
    {
        public decimal Subtotal { get; set; }
        public decimal DiscountAmount { get; set; }
        public decimal Tax { get; set; }
        public decimal Total { get; set; }
        public string Summary { get; set; } = string.Empty;
        public List<string> ItemBreakdown { get; set; } = new List<string>();
    }

    // Shopping Cart Checkout Processor using Claude AI
    public class ShoppingCartCheckoutProcessor
    {
        private readonly AnthropicClient _client;
        private const decimal TaxRate = 0.1m; // 10% tax rate

        public ShoppingCartCheckoutProcessor()
        {
            _client = new AnthropicClient();
        }

        // Calculate checkout using Claude AI for intelligent processing
        public async Task<CheckoutResult> ProcessCheckoutAsync(CheckoutRequest request)
        {
            // Prepare the cart data as JSON for Claude
            var cartJson = JsonSerializer.Serialize(request, new JsonSerializerOptions
            {
                WriteIndented = true
            });

            // Create the prompt for Claude
            var prompt = $@"You are a shopping cart checkout processor. Please analyze the following shopping cart and calculate the total amount.

Shopping Cart Data:
{cartJson}

Please calculate:
1. Subtotal (sum of all items: price × quantity)
2. Discount amount (if discount code 'SAVE10' is provided, apply 10% discount; if 'SAVE20', apply 20% discount)
3. Tax (10% of subtotal after discount)
4. Total amount

Respond with a JSON object in this exact format:
{{
  ""subtotal"": <number>,
  ""discountAmount"": <number>,
  ""tax"": <number>,
  ""total"": <number>,
  ""summary"": ""<brief summary in Japanese>"",
  ""itemBreakdown"": [""<item1 breakdown>"", ""<item2 breakdown>"", ...]
}}

All monetary values should be rounded to 2 decimal places.
Item breakdown should be in Japanese format like: ""商品名: ¥価格 × 数量 = ¥小計""";

            // Call Claude API
            var messageRequest = new MessageParameters
            {
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                Messages = new List<Message>
                {
                    new Message
                    {
                        Role = RoleType.User,
                        Content = new List<ContentBase>
                        {
                            new TextContent
                            {
                                Text = prompt
                            }
                        }
                    }
                }
            };

            var response = await _client.Messages.GetClaudeMessageAsync(messageRequest);

            // Extract the text response
            var responseText = response.Content[0].ToString() ?? string.Empty;

            // Parse the JSON response from Claude
            var result = ParseCheckoutResult(responseText);

            return result;
        }

        private CheckoutResult ParseCheckoutResult(string responseText)
        {
            try
            {
                // Find JSON content in the response
                var jsonStart = responseText.IndexOf('{');
                var jsonEnd = responseText.LastIndexOf('}');

                if (jsonStart >= 0 && jsonEnd > jsonStart)
                {
                    var jsonContent = responseText.Substring(jsonStart, jsonEnd - jsonStart + 1);

                    using var doc = JsonDocument.Parse(jsonContent);
                    var root = doc.RootElement;

                    var result = new CheckoutResult
                    {
                        Subtotal = root.GetProperty("subtotal").GetDecimal(),
                        DiscountAmount = root.GetProperty("discountAmount").GetDecimal(),
                        Tax = root.GetProperty("tax").GetDecimal(),
                        Total = root.GetProperty("total").GetDecimal(),
                        Summary = root.GetProperty("summary").GetString() ?? string.Empty
                    };

                    if (root.TryGetProperty("itemBreakdown", out var itemBreakdown))
                    {
                        foreach (var item in itemBreakdown.EnumerateArray())
                        {
                            result.ItemBreakdown.Add(item.GetString() ?? string.Empty);
                        }
                    }

                    return result;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error parsing response: {ex.Message}");
            }

            // Return empty result if parsing fails
            return new CheckoutResult
            {
                Summary = "チェックアウト処理中にエラーが発生しました。"
            };
        }

        // Display checkout result
        public void DisplayCheckoutResult(CheckoutResult result)
        {
            Console.WriteLine("\n========== チェックアウト結果 ==========");
            Console.WriteLine("\n商品明細:");
            foreach (var item in result.ItemBreakdown)
            {
                Console.WriteLine($"  {item}");
            }

            Console.WriteLine("\n金額計算:");
            Console.WriteLine($"  小計:         ¥{result.Subtotal:F2}");

            if (result.DiscountAmount > 0)
            {
                Console.WriteLine($"  割引額:       -¥{result.DiscountAmount:F2}");
            }

            Console.WriteLine($"  消費税(10%):  ¥{result.Tax:F2}");
            Console.WriteLine($"  合計金額:     ¥{result.Total:F2}");
            Console.WriteLine($"\nサマリー: {result.Summary}");
            Console.WriteLine("==========================================\n");
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("ショッピングカート チェックアウトシステム");
            Console.WriteLine("==========================================\n");

            var processor = new ShoppingCartCheckoutProcessor();

            // Example 1: Basic checkout without discount
            Console.WriteLine("例1: 割引なしのチェックアウト");
            var request1 = new CheckoutRequest
            {
                Items = new List<CartItem>
                {
                    new CartItem { Name = "ノートパソコン", Price = 89800m, Quantity = 1 },
                    new CartItem { Name = "マウス", Price = 2980m, Quantity = 2 },
                    new CartItem { Name = "キーボード", Price = 5980m, Quantity = 1 },
                    new CartItem { Name = "USBハブ", Price = 1980m, Quantity = 3 }
                }
            };

            Console.WriteLine("カートの内容:");
            foreach (var item in request1.Items)
            {
                Console.WriteLine($"  - {item.Name}: ¥{item.Price:F2} × {item.Quantity}個");
            }

            var result1 = await processor.ProcessCheckoutAsync(request1);
            processor.DisplayCheckoutResult(result1);

            // Example 2: Checkout with 10% discount code
            Console.WriteLine("例2: 10%割引コード(SAVE10)使用のチェックアウト");
            var request2 = new CheckoutRequest
            {

                {

                    new CartItem { Name = "充電器", Price = 3980m, Quantity = 2 }
                },
                DiscountCode = "SAVE10"
            };

            Console.WriteLine("カートの内容:");
            foreach (var item in request2.Items)
            {
                Console.WriteLine($"  - {item.Name}: ¥{item.Price:F2} × {item.Quantity}個");
            }
            Console.WriteLine($"割引コード: {request2.DiscountCode}");

            var result2 = await processor.ProcessCheckoutAsync(request2);
            processor.DisplayCheckoutResult(result2);

            // Example 3: Checkout with 20% discount code
            Console.WriteLine("例3: 20%割引コード(SAVE20)使用のチェックアウト");
            var request3 = new CheckoutRequest
            {
                Items = new List<CartItem>
                {
                    new CartItem { Name = "タブレット", Price = 49800m, Quantity = 1 },
                    new CartItem { Name = "タブレットカバー", Price = 2980m, Quantity = 1 },
                    new CartItem { Name = "スタイラスペン", Price = 4980m, Quantity = 1 },
                    new CartItem { Name = "保護フィルム", Price = 980m, Quantity = 2 }
                },
                DiscountCode = "SAVE20"

            Console.WriteLine("カートの内容:");
            foreach (var item in request3.Items)
            {
                Console.WriteLine($"  - {item.Name}: ¥{item.Price:F2} × {item.Quantity}個");

            Console.WriteLine($"割引コード: {request3.DiscountCode}");

            var result3 = await processor.ProcessCheckoutAsync(request3);
            processor.DisplayCheckoutResult(result3);

        }
    }
