
```python
import anthropic

client = anthropic.Anthropic()

script = """
import sys
import os

def count_lines_large_file(filepath):
    \"\"\"
    Count lines in a very large file (10GB+) without loading it all into memory.
    Uses buffered reading for memory efficiency.
    \"\"\"
    if not os.path.exists(filepath):
        print(f"Error: File '{filepath}' not found.")
        sys.exit(1)
    
    file_size = os.path.getsize(filepath)
    print(f"File size: {file_size / (1024**3):.2f} GB")
    
    line_count = 0
    bytes_read = 0
    buffer_size = 1024 * 1024  # 1MB buffer
    
    print("Counting lines...")
    
    with open(filepath, 'rb') as f:
        while True:
            buffer = f.read(buffer_size)
            if not buffer:
                break
            
            line_count += buffer.count(b'\\n')
            bytes_read += len(buffer)
            
            # Show progress
            progress = (bytes_read / file_size) * 100
            print(f"\\rProgress: {progress:.1f}% ({bytes_read / (1024**3):.2f} GB / {file_size / (1024**3):.2f} GB)", end='', flush=True)
    
    print()  # New line after progress
    
    # Check if file doesn't end with newline (last line might not be counted)
    # Re-open to check last byte
    with open(filepath, 'rb') as f:
        f.seek(-1, 2)  # Seek to last byte
        last_byte = f.read(1)
        if last_byte and last_byte != b'\\n':
            line_count += 1  # Count the last line if it doesn't end with newline
    
    return line_count

def main():
    if len(sys.argv) < 2:
        # If no argument provided, show usage
        print("Usage: python solution.py <filepath>")
        print("\\nExample with a test file:")
        
        # Create a small test file for demonstration
        test_file = "test_large_file.txt"
        print(f"Creating test file: {test_file}")
        
        with open(test_file, 'w') as f:
            for i in range(1000000):  # 1 million lines
                f.write(f"Line {i}: This is a test line with some content to make it larger.\\n")
        
        file_size = os.path.getsize(test_file)
        print(f"Test file created: {file_size / (1024**2):.2f} MB")
        
        filepath = test_file
    else:
        filepath = sys.argv[1]
    
    print(f"\\nCounting lines in: {filepath}")
    line_count = count_lines_large_file(filepath)
    print(f"\\nTotal lines: {line_count:,}")
    
    # Clean up test file if we created it
    if len(sys.argv) < 2 and os.path.exists("test_large_file.txt"):
        os.remove("test_large_file.txt")
        print("Test file cleaned up.")

if __name__ == "__main__":
    main()
"""

print("Generating Python script for counting lines in a 10GB file using Claude API with streaming...")
print("=" * 60)

with client.messages.stream(
    model="claude-opus-4-5",
    max_tokens=2048,
    messages=[
        {
            "role": "user",
            "content": """以下のPythonスクリプトを実行して、10GBの大きなファイルの行数を数える方法を説明してください。
            
スクリプトの内容:
```python
""" + script + """
```

このスクリプトについて以下を説明してください：
1. メモリ効率的な読み込み方法
2. バッファリングの仕組み
3. 進捗表示の実装
4. 実際の10GBファイルでの使用方法

また、このスクリプトを実際に実行して結果を示してください。"""
        }
    ]
) as stream:
    print("\nClaude's explanation (streaming):")
    print("-" * 40)
    for text in stream.text_stream:
        print(text, end="", flush=True)
    print("\n" + "-" * 40)

print("\n" + "=" * 60)
print("Now executing the actual script...")
print("=" * 60)

import subprocess
import sys

result = subprocess.run(
    [sys.executable, "-c", script],
    capture_output=True,
    text=True,
    timeout=300
)

if result.stdout:
    print("\nScript output:")
    print(result.stdout)

if result.stderr:
    print("\nScript errors (if any):")
    print(result.stderr)

print("\n" + "=" * 60)
print("Script execution completed!")
