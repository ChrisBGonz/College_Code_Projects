using System;
using System.Collections.Generic;
using System.IO;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

// Application settings class that can be serialized
[Serializable]
public class AppSettings
{
    public string Theme { get; set; } = "Light";
    public string Language { get; set; } = "English";
    public int FontSize { get; set; } = 12;
    public bool AutoSave { get; set; } = true;
    public string LastOpenedFile { get; set; } = "";
    public List<string> RecentFiles { get; set; } = new List<string>();
    public Dictionary<string, string> CustomSettings { get; set; } = new Dictionary<string, string>();
}

// Settings manager using BinaryFormatter
public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        try
        {
            using (FileStream fs = new FileStream(_settingsFilePath, FileMode.Create))
            {
                // Using manual binary serialization instead of BinaryFormatter
                using (BinaryWriter writer = new BinaryWriter(fs))
                {
                    writer.Write(settings.Theme ?? "");
                    writer.Write(settings.Language ?? "");
                    writer.Write(settings.FontSize);
                    writer.Write(settings.AutoSave);
                    writer.Write(settings.LastOpenedFile ?? "");
                    
                    // Write RecentFiles
                    writer.Write(settings.RecentFiles?.Count ?? 0);
                    if (settings.RecentFiles != null)
                    {
                        foreach (var file in settings.RecentFiles)
                        {
                            writer.Write(file ?? "");
                        }
                    }
                    
                    // Write CustomSettings
                    writer.Write(settings.CustomSettings?.Count ?? 0);
                    if (settings.CustomSettings != null)
                    {
                        foreach (var kvp in settings.CustomSettings)
                        {
                            writer.Write(kvp.Key ?? "");
                            writer.Write(kvp.Value ?? "");
                        }
                    }
                }
            }
            Console.WriteLine($"Settings saved to {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            throw;
        }
    }

    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("Settings file not found. Using default settings.");
            return new AppSettings();
        }

        try
        {
            using (FileStream fs = new FileStream(_settingsFilePath, FileMode.Open))
            {
                using (BinaryReader reader = new BinaryReader(fs))
                {
                    var settings = new AppSettings();
                    settings.Theme = reader.ReadString();
                    settings.Language = reader.ReadString();
                    settings.FontSize = reader.ReadInt32();
                    settings.AutoSave = reader.ReadBoolean();
                    settings.LastOpenedFile = reader.ReadString();
                    
                    // Read RecentFiles
                    int recentFilesCount = reader.ReadInt32();
                    settings.RecentFiles = new List<string>();
                    for (int i = 0; i < recentFilesCount; i++)
                    {
                        settings.RecentFiles.Add(reader.ReadString());
                    }
                    
                    // Read CustomSettings
                    int customSettingsCount = reader.ReadInt32();
                    settings.CustomSettings = new Dictionary<string, string>();
                    for (int i = 0; i < customSettingsCount; i++)
                    {
                        string key = reader.ReadString();
                        string value = reader.ReadString();
                        settings.CustomSettings[key] = value;
                    }
                    
                    Console.WriteLine($"Settings loaded from {_settingsFilePath}");
                    return settings;
                }
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            return new AppSettings();
        }
    }

    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine($"Settings file {_settingsFilePath} deleted.");
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        var settingsManager = new SettingsManager();
        var conversationHistory = new List<Message>();

        Console.WriteLine("Application Settings Manager with Claude AI");
        Console.WriteLine("==========================================");
        Console.WriteLine("Commands: save, load, show, update, delete, quit");
        Console.WriteLine();

        // Initialize with default settings
        var currentSettings = new AppSettings
        {
            Theme = "Dark",
            Language = "Japanese",
            FontSize = 14,
            AutoSave = true,
            LastOpenedFile = "document.txt",
            RecentFiles = new List<string> { "file1.txt", "file2.txt", "file3.txt" },
            CustomSettings = new Dictionary<string, string>
            {
                { "BackupEnabled", "true" },
                { "BackupInterval", "30" }
            }
        };

        while (true)
        {
            Console.Write("\nYou: ");
            string userInput = Console.ReadLine()?.Trim();

            if (string.IsNullOrEmpty(userInput))
                continue;

            if (userInput.ToLower() == "quit")
                break;

            // Add context about current settings to the conversation
            string settingsContext = $@"Current application settings:
- Theme: {currentSettings.Theme}
- Language: {currentSettings.Language}
- Font Size: {currentSettings.FontSize}
- Auto Save: {currentSettings.AutoSave}
- Last Opened File: {currentSettings.LastOpenedFile}
- Recent Files: {string.Join(", ", currentSettings.RecentFiles)}
- Custom Settings: {string.Join(", ", currentSettings.CustomSettings.Select(kvp => $"{kvp.Key}={kvp.Value}"))}

User request: {userInput}

Please help manage these application settings. If the user wants to:
- Save settings: confirm and provide feedback
- Load settings: describe what would be loaded
- Update settings: suggest specific changes
- Show settings: display current values
- Delete settings: warn about consequences

Respond in Japanese if the language setting is Japanese, otherwise in English.";

            conversationHistory.Add(new Message
            {
                Role = RoleType.User,
                Content = settingsContext
            });

            try
            {
                var response = await client.Messages.GetClaudeMessageAsync(
                    new MessageParameters
                    {
                        Model = AnthropicModels.Claude3_5Sonnet,
                        MaxTokens = 1024,
                        Messages = conversationHistory
                    });

                string assistantMessage = response.Content[0].ToString();
                
                conversationHistory.Add(new Message
                {
                    Role = RoleType.Assistant,
                    Content = assistantMessage
                });

                Console.WriteLine($"\nClaude: {assistantMessage}");

                // Process commands based on user input
                string lowerInput = userInput.ToLower();
                
                if (lowerInput.Contains("save") || lowerInput.Contains("保存"))
                {
                    settingsManager.SaveSettings(currentSettings);
                    Console.WriteLine("\n[System: Settings saved successfully]");
                }
                else if (lowerInput.Contains("load") || lowerInput.Contains("読み込"))
                {
                    currentSettings = settingsManager.LoadSettings();
                    Console.WriteLine("\n[System: Settings loaded successfully]");
                }
                else if (lowerInput.Contains("delete") || lowerInput.Contains("削除"))
                {
                    settingsManager.DeleteSettings();
                    currentSettings = new AppSettings();
                    Console.WriteLine("\n[System: Settings deleted, using defaults]");
                }
                else if (lowerInput.Contains("theme") || lowerInput.Contains("テーマ"))
                {
                    if (lowerInput.Contains("dark") || lowerInput.Contains("ダーク"))
                    {
                        currentSettings.Theme = "Dark";
                        Console.WriteLine("\n[System: Theme updated to Dark]");
                    }
                    else if (lowerInput.Contains("light") || lowerInput.Contains("ライト"))
                    {
                        currentSettings.Theme = "Light";
                        Console.WriteLine("\n[System: Theme updated to Light]");
                    }
                }
                else if (lowerInput.Contains("font") || lowerInput.Contains("フォント"))
                {
                    // Extract number from input
                    var numbers = System.Text.RegularExpressions.Regex.Matches(userInput, @"\d+");
                    if (numbers.Count > 0 && int.TryParse(numbers[0].Value, out int fontSize))
                    {
                        currentSettings.FontSize = fontSize;
                        Console.WriteLine($"\n[System: Font size updated to {fontSize}]");
                    }
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"\nError communicating with Claude: {ex.Message}");
            }
        }

        Console.WriteLine("\nGoodbye!");
    }
}
