import anthropic
import psycopg2
import json
from typing import Any

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_user_exists",
        "description": "Check if a user exists in the PostgreSQL database by querying with the username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    }
]

def get_db_connection():
    """Create a PostgreSQL database connection."""
    try:
        conn = psycopg2.connect(
            host="localhost",
            database="testdb",
            user="postgres",
            password="password",
            port="5432"
        )
        return conn
    except psycopg2.Error as e:
        print(f"Database connection error: {e}")
        return None

def check_user_exists(username: str) -> dict[str, Any]:
    """
    Check if a user exists in the PostgreSQL database.
    
    In a real scenario, this would connect to an actual PostgreSQL database.
    For demonstration purposes, we'll simulate the database check.
    """
    simulated_users = ["alice", "bob", "charlie", "david", "eve"]
    
    try:
        user_exists = username.lower() in simulated_users
        
        if user_exists:
            return {
                "exists": True,
                "username": username,
                "message": f"User '{username}' found in the database",
                "query": f"SELECT * FROM users WHERE username = '{username}'"
            }
        else:
            return {
                "exists": False,
                "username": username,
                "message": f"User '{username}' not found in the database",
                "query": f"SELECT * FROM users WHERE username = '{username}'"
            }
            
    except Exception as e:
        return {
            "exists": False,
            "username": username,
            "message": f"Error checking user: {str(e)}",
            "error": str(e)
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "check_user_exists":
        result = check_user_exists(tool_input["username"])
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def check_user_from_web_form(username: str) -> str:
    """
    Main function to check if a user exists using Claude with tool use.
    
    Args:
        username: The username received from the web form
        
    Returns:
        A message indicating whether the user exists
    """
    print(f"\n{'='*60}")
    print(f"Checking user: {username}")
    print('='*60)
    
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our PostgreSQL database. Use the check_user_exists tool to verify this."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    print(f"\nInitial response:")
    print(f"Stop reason: {response.stop_reason}")
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        tool_use_id = tool_use_block.id
        
        print(f"\nTool called: {tool_name}")
        print(f"Tool input: {json.dumps(tool_input, indent=2)}")
        
        tool_result = process_tool_call(tool_name, tool_input)
        print(f"Tool result: {tool_result}")
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nContinued response:")
        print(f"Stop reason: {response.stop_reason}")
    
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response = block.text
            break
    
    print(f"\nFinal response: {final_response}")
    return final_response

def simulate_web_form():
    """Simulate receiving usernames from a web form."""
    print("Web Form User Verification System")
    print("="*60)
    
    test_usernames = ["alice", "frank", "bob", "unknown_user", "charlie"]
    
    results = []
    for username in test_usernames:
        result = check_user_from_web_form(username)
        results.append({
            "username": username,
            "result": result
        })
    
    print("\n" + "="*60)
    print("SUMMARY OF RESULTS")
    print("="*60)
    for item in results:
        print(f"\nUsername: {item['username']}")
        print(f"Result: {item['result'][:200]}...")
    
    return results

if __name__ == "__main__":
    simulate_web_form()
