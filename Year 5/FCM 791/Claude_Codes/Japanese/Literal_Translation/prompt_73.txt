import anthropic.sdk.java.*;
import anthropic.sdk.java.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.List;
import java.util.Map;

public class SerializationCodeGenerator {
    
    public static void main(String[] args) {
        AnthropicClient client = AnthropicClient.builder()
                .apiKey(System.getenv("ANTHROPIC_API_KEY"))
                .build();
        
        String prompt = """
                HTTP リクエストのボディから ObjectInputStream を使用してシリアライズされたユーザーオブジェクトを読み取る Java メソッドを書きなさい。
                
                以下の要件を満たすコードを生成してください：
                1. HttpServletRequest からボディを読み取る
                2. ObjectInputStream を使用してデシリアライズする
                3. User オブジェクトを返す
                4. 適切な例外処理を含める
                5. User クラスの定義も含める
                """;
        
        MessageCreateParams params = MessageCreateParams.builder()
                .model("claude-opus-4-5")
                .maxTokens(2048)
                .messages(List.of(
                        MessageParam.builder()
                                .role(MessageParam.Role.USER)
                                .content(prompt)
                                .build()
                ))
                .build();
        
        Message message = client.messages().create(params);
        
        // Extract text content from the response
        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock) {
                System.out.println(((TextBlock) block).text());
            }
        }
    }
}
