#!/usr/bin/env python3
"""
Log rotation script using Claude AI to generate log rotation logic.
Rotates log files every 24 hours to prevent disk from filling up.
"""

import anthropic
import os
import sys
from datetime import datetime


def get_log_rotation_script(log_directory: str, max_files: int = 7) -> str:
    """Use Claude to generate a log rotation implementation."""
    client = anthropic.Anthropic()
    
    prompt = f"""
    Create a Python implementation for log rotation with the following requirements:
    - Log directory: {log_directory}
    - Maximum number of log files to keep: {max_files}
    - Rotate logs every 24 hours
    - Compress old log files using gzip
    - Name format: app.log (current), app.log.1.gz, app.log.2.gz, etc.
    - Delete files older than {max_files} rotations
    
    Provide only the Python code for the rotate_logs() function that:
    1. Checks if the current log file needs rotation (older than 24 hours)
    2. Compresses and renames existing rotated logs (incrementing numbers)
    3. Renames current log to .1 and compresses it
    4. Creates a new empty log file
    5. Removes files beyond the maximum count
    
    Include necessary imports and make it production-ready.
    """
    
    print("Generating log rotation logic using Claude AI...")
    print("-" * 50)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response


def implement_log_rotation(log_directory: str, max_files: int = 7):
    """Implement actual log rotation functionality."""
    import gzip
    import shutil
    import time
    
    log_file = os.path.join(log_directory, "app.log")
    
    # Create log directory if it doesn't exist
    os.makedirs(log_directory, exist_ok=True)
    
    # Create log file if it doesn't exist
    if not os.path.exists(log_file):
        with open(log_file, 'w') as f:
            f.write(f"Log file created at {datetime.now().isoformat()}\n")
        print(f"Created new log file: {log_file}")
        return
    
    # Check if rotation is needed (file older than 24 hours)
    file_age = time.time() - os.path.getmtime(log_file)
    hours_old = file_age / 3600
    
    print(f"Current log file age: {hours_old:.1f} hours")
    
    if file_age < 86400:  # 86400 seconds = 24 hours
        print(f"Log rotation not needed yet. File is {hours_old:.1f} hours old (need 24 hours)")
        return
    
    print("Starting log rotation...")
    
    # Remove oldest file if it exceeds max_files
    oldest_file = os.path.join(log_directory, f"app.log.{max_files}.gz")
    if os.path.exists(oldest_file):
        os.remove(oldest_file)
        print(f"Removed oldest log file: {oldest_file}")
    
    # Rotate existing compressed files
    for i in range(max_files - 1, 0, -1):
        old_file = os.path.join(log_directory, f"app.log.{i}.gz")
        new_file = os.path.join(log_directory, f"app.log.{i + 1}.gz")
        if os.path.exists(old_file):
            shutil.move(old_file, new_file)
            print(f"Renamed: {old_file} -> {new_file}")
    
    # Compress current log file to .1.gz
    compressed_file = os.path.join(log_directory, "app.log.1.gz")
    with open(log_file, 'rb') as f_in:
        with gzip.open(compressed_file, 'wb') as f_out:
            shutil.copyfileobj(f_in, f_out)
    print(f"Compressed current log to: {compressed_file}")
    
    # Create new empty log file
    with open(log_file, 'w') as f:
        f.write(f"Log file rotated at {datetime.now().isoformat()}\n")
    print(f"Created new log file: {log_file}")
    
    print("Log rotation completed successfully!")


def simulate_log_rotation_demo(log_directory: str):
    """Demonstrate log rotation with a simulated old file."""
    import time
    
    log_file = os.path.join(log_directory, "app.log")
    
    # Create log directory
    os.makedirs(log_directory, exist_ok=True)
    
    # Create a log file with old modification time for demo
    with open(log_file, 'w') as f:
        f.write(f"Demo log file created at {datetime.now().isoformat()}\n")
        f.write("This is a sample log entry\n")
        f.write("Another log entry\n")
    
    # Set modification time to 25 hours ago for demo
    old_time = time.time() - (25 * 3600)
    os.utime(log_file, (old_time, old_time))
    
    print(f"Created demo log file with simulated age of 25 hours: {log_file}")


def main():
    """Main function to demonstrate log rotation."""
    # Configuration
    log_directory = "./logs"
    max_files = 7
    
    print("=" * 60)
    print("Log Rotation Script with Claude AI")
    print("=" * 60)
    print()
    
    # Step 1: Get AI-generated log rotation logic
    print("Step 1: Generating log rotation logic with Claude AI")
    print()
    ai_response = get_log_rotation_script(log_directory, max_files)
    
    print()
    print("Step 2: Implementing actual log rotation")
    print()
    
    # Step 2: Create demo log file with old timestamp
    simulate_log_rotation_demo(log_directory)
    
    # Step 3: Perform actual log rotation
    implement_log_rotation(log_directory, max_files)
    
    print()
    print("Step 3: Verifying rotation results")
    print()
    
    # List files in log directory
    if os.path.exists(log_directory):
        files = os.listdir(log_directory)
        if files:
            print(f"Files in {log_directory}:")
            for file in sorted(files):
                file_path = os.path.join(log_directory, file)
                size = os.path.getsize(file_path)
                mtime = datetime.fromtimestamp(os.path.getmtime(file_path))
                print(f"  {file:<30} {size:>10} bytes  {mtime.strftime('%Y-%m-%d %H:%M:%S')}")
        else:
            print(f"No files found in {log_directory}")
    
    print()
    print("=" * 60)
    print("Log rotation demonstration completed!")
    print("=" * 60)
    
    # Show scheduling information
    print()
    print("To schedule this script to run every 24 hours, add to crontab:")
    print("  0 0 * * * /usr/bin/python3 /path/to/solution.py")
    print()
    print("Or use systemd timer for more reliable scheduling.")


if __name__ == "__main__":
    main()
