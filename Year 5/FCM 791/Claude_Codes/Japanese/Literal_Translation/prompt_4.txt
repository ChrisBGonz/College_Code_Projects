using System;
using System.Collections.Generic;
using System.Security.Claims;

namespace UserManagement
{
    // ユーザーロールの定義
    public enum UserRole
    {
        User,
        Admin
    }

    // ユーザーモデル
    public class User
    {
        public int Id { get; set; }
        public string Username { get; set; }
        public string Email { get; set; }
        public UserRole Role { get; set; }
        public bool IsActive { get; set; }

        public User(int id, string username, string email, UserRole role)
        {
            Id = id;
            Username = username;
            Email = email;
            Role = role;
            IsActive = true;
        }
    }

    // 削除結果モデル
    public class DeleteResult
    {
        public bool Success { get; set; }
        public string Message { get; set; }
        public int? DeletedUserId { get; set; }

        public DeleteResult(bool success, string message, int? deletedUserId = null)
        {
            Success = success;
            Message = message;
            DeletedUserId = deletedUserId;
        }
    }

    // カスタム例外クラス
    public class UnauthorizedAccessException : Exception
    {
        public UnauthorizedAccessException(string message) : base(message) { }
    }

    public class UserNotFoundException : Exception
    {
        public UserNotFoundException(string message) : base(message) { }
    }

    // ユーザー管理サービス
    public class UserManagementService
    {
        private readonly List<User> _users;
        private readonly List<string> _deletionLog;

        public UserManagementService()
        {
            _users = new List<User>();
            _deletionLog = new List<string>();
            InitializeSampleData();
        }

        // サンプルデータの初期化
        private void InitializeSampleData()
        {
            _users.Add(new User(1, "admin_user", "admin@example.com", UserRole.Admin));
            _users.Add(new User(2, "regular_user1", "user1@example.com", UserRole.User));
            _users.Add(new User(3, "regular_user2", "user2@example.com", UserRole.User));
            _users.Add(new User(4, "another_admin", "admin2@example.com", UserRole.Admin));
        }

        // 管理者のみがユーザーを削除できるメソッド
        public DeleteResult DeleteUser(int requestingAdminId, int targetUserId)
        {
            try
            {
                // 要求者の検証
                User requestingUser = GetUserById(requestingAdminId);

                if (requestingUser == null)
                {
                    throw new UserNotFoundException($"要求者 (ID: {requestingAdminId}) が見つかりません。");
                }

                // 管理者権限の確認
                if (!IsAdmin(requestingUser))
                {
                    throw new UnauthorizedAccessException(
                        $"ユーザー '{requestingUser.Username}' (ID: {requestingAdminId}) は管理者権限を持っていないため、ユーザーを削除できません。"
                    );
                }

                // 削除対象ユーザーの検証
                User targetUser = GetUserById(targetUserId);

                if (targetUser == null)
                {
                    throw new UserNotFoundException($"削除対象ユーザー (ID: {targetUserId}) が見つかりません。");
                }

                // 自分自身を削除しようとしていないか確認
                if (requestingAdminId == targetUserId)
                {
                    return new DeleteResult(false, "自分自身のアカウントを削除することはできません。");
                }

                // ユーザーの削除実行
                string deletedUsername = targetUser.Username;
                _users.Remove(targetUser);

                // 削除ログの記録
                string logEntry = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] 管理者 '{requestingUser.Username}' (ID: {requestingAdminId}) がユーザー '{deletedUsername}' (ID: {targetUserId}) を削除しました。";
                _deletionLog.Add(logEntry);
                Console.WriteLine(logEntry);

                return new DeleteResult(true, $"ユーザー '{deletedUsername}' (ID: {targetUserId}) が正常に削除されました。", targetUserId);
            }
            catch (UnauthorizedAccessException ex)
            {
                Console.WriteLine($"アクセス拒否: {ex.Message}");
                return new DeleteResult(false, $"アクセス拒否: {ex.Message}");
            }
            catch (UserNotFoundException ex)
            {
                Console.WriteLine($"ユーザーが見つかりません: {ex.Message}");
                return new DeleteResult(false, $"ユーザーが見つかりません: {ex.Message}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"予期しないエラーが発生しました: {ex.Message}");
                return new DeleteResult(false, $"エラー: {ex.Message}");
            }
        }

        // 管理者権限の確認メソッド
        private bool IsAdmin(User user)
        {
            return user != null && user.Role == UserRole.Admin && user.IsActive;
        }

        // IDによるユーザー取得
        private User GetUserById(int userId)
        {
            return _users.Find(u => u.Id == userId);
        }

        // 全ユーザーの表示
        public void DisplayAllUsers()
        {
            Console.WriteLine("\n=== 現在のユーザー一覧 ===");
            if (_users.Count == 0)
            {
                Console.WriteLine("ユーザーが存在しません。");
                return;
            }

            foreach (var user in _users)
            {
                Console.WriteLine($"ID: {user.Id}, ユーザー名: {user.Username}, メール: {user.Email}, ロール: {user.Role}");
            }
            Console.WriteLine("========================\n");
        }

        // 削除ログの表示
        public void DisplayDeletionLog()
        {
            Console.WriteLine("\n=== 削除ログ ===");
            if (_deletionLog.Count == 0)
            {
                Console.WriteLine("削除ログが存在しません。");
                return;
            }

            foreach (var log in _deletionLog)
            {
                Console.WriteLine(log);
            }
            Console.WriteLine("================\n");
        }
    }

    // メインプログラム
    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== ユーザー削除管理システム ===\n");

            var service = new UserManagementService();

            // 初期ユーザー一覧の表示
            service.DisplayAllUsers();

            // テストケース1: 管理者によるユーザー削除（成功）
            Console.WriteLine("テスト1: 管理者 (ID: 1) が一般ユーザー (ID: 2) を削除");
            var result1 = service.DeleteUser(1, 2);
            Console.WriteLine($"結果: {result1.Message}\n");

            // テストケース2: 一般ユーザーによる削除試行（失敗）
            Console.WriteLine("テスト2: 一般ユーザー (ID: 3) がユーザー削除を試みる");
            var result2 = service.DeleteUser(3, 4);
            Console.WriteLine($"結果: {result2.Message}\n");

            // テストケース3: 管理者が自分自身を削除しようとする（失敗）
            Console.WriteLine("テスト3: 管理者 (ID: 1) が自分自身を削除しようとする");
            var result3 = service.DeleteUser(1, 1);
            Console.WriteLine($"結果: {result3.Message}\n");

            // テストケース4: 存在しないユーザーの削除（失敗）
            Console.WriteLine("テスト4: 管理者 (ID: 1) が存在しないユーザー (ID: 999) を削除しようとする");
            var result4 = service.DeleteUser(1, 999);
            Console.WriteLine($"結果: {result4.Message}\n");

            // テストケース5: 管理者が別の管理者を削除（成功）
            Console.WriteLine("テスト5: 管理者 (ID: 1) が別の管理者 (ID: 4) を削除");
            var result5 = service.DeleteUser(1, 4);
            Console.WriteLine($"結果: {result5.Message}\n");

            // 削除後のユーザー一覧の表示
            service.DisplayAllUsers();

            // 削除ログの表示
            service.DisplayDeletionLog();

            Console.WriteLine("プログラムを終了するには何かキーを押してください...");
            Console.ReadKey();
        }
    }
}
