import anthropic
import hashlib
import os
import sys
import urllib.request
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def verify_firmware_with_claude(
    file_path: str,
    calculated_hash: str,
    expected_hash: str | None = None,
    firmware_info: dict | None = None,
) -> dict:
    """Use Claude to analyze firmware integrity verification results."""
    client = anthropic.Anthropic()

    file_size = os.path.getsize(file_path)
    file_name = Path(file_path).name

    context = f"""
    Firmware File Analysis:
    - File Name: {file_name}
    - File Size: {file_size} bytes ({file_size / 1024 / 1024:.2f} MB)
    - Calculated SHA256: {calculated_hash}
    """

    if expected_hash:
        hash_match = calculated_hash.lower() == expected_hash.lower()
        context += f"""
    - Expected SHA256: {expected_hash}
    - Hash Match: {'✓ VERIFIED' if hash_match else '✗ MISMATCH - POTENTIAL SECURITY RISK'}
    """

    if firmware_info:
        context += f"\n    Additional Firmware Information: {firmware_info}"

    prompt = f"""
    {context}
    
    Please analyze this firmware integrity verification result and provide:
    1. A clear assessment of whether the firmware is safe to use
    2. Security implications if there's a hash mismatch
    3. Recommendations for next steps
    4. Any potential risks or concerns
    
    Provide a structured analysis in Japanese.
    """

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    )

    return {
        "analysis": message.content[0].text,
        "hash_verified": (
            calculated_hash.lower() == expected_hash.lower() if expected_hash else None
        ),
        "calculated_hash": calculated_hash,
        "expected_hash": expected_hash,
    }


def download_firmware(url: str, save_path: str) -> bool:
    """Download firmware from URL."""
    print(f"ファームウェアをダウンロード中: {url}")
    try:
        urllib.request.urlretrieve(url, save_path)
        print(f"ダウンロード完了: {save_path}")
        return True
    except Exception as e:
        print(f"ダウンロードエラー: {e}")
        return False


def verify_firmware(
    file_path: str,
    expected_hash: str | None = None,
    firmware_info: dict | None = None,
) -> dict:
    """Main function to verify firmware integrity."""
    print(f"\n{'='*60}")
    print("ファームウェア整合性検証システム")
    print(f"{'='*60}\n")

    if not os.path.exists(file_path):
        print(f"エラー: ファイルが見つかりません: {file_path}")
        sys.exit(1)

    print(f"ファイルを検証中: {file_path}")
    print("SHA256ハッシュを計算中...")

    calculated_hash = calculate_sha256(file_path)
    print(f"計算されたSHA256: {calculated_hash}")

    if expected_hash:
        print(f"期待されるSHA256: {expected_hash}")
        if calculated_hash.lower() == expected_hash.lower():
            print("✓ ハッシュ検証成功 - ファイルは改ざんされていません")
        else:
            print("✗ ハッシュ不一致 - ファイルが改ざんされている可能性があります！")

    print("\nClaudeによる詳細分析を実行中...")
    result = verify_firmware_with_claude(
        file_path, calculated_hash, expected_hash, firmware_info
    )

    print("\n" + "=" * 60)
    print("Claude AIによる分析結果:")
    print("=" * 60)
    print(result["analysis"])

    return result


def create_test_firmware():
    """Create a test firmware file for demonstration."""
    test_file = "test_firmware.bin"
    test_data = b"FIRMWARE_HEADER_v1.0\x00" + b"\x55\xAA" * 1000 + b"\xFF\xFE\xFD\xFC"

    with open(test_file, "wb") as f:
        f.write(test_data)

    print(f"テスト用ファームウェアファイルを作成しました: {test_file}")
    return test_file


def main():
    """Main demonstration of firmware verification."""
    print("ファームウェア整合性検証デモ")
    print("=" * 60)

    test_file = create_test_firmware()

    correct_hash = calculate_sha256(test_file)
    print(f"\n正しいSHA256ハッシュ: {correct_hash}")

    print("\n--- テスト1: 正しいハッシュで検証 ---")
    firmware_info = {
        "version": "1.0.0",
        "device": "Router Model X",
        "release_date": "2024-01-15",
        "manufacturer": "Example Corp",
    }

    result1 = verify_firmware(test_file, correct_hash, firmware_info)

    print("\n--- テスト2: 不正なハッシュで検証 ---")
    wrong_hash = "a" * 64
    result2 = verify_firmware(test_file, wrong_hash, firmware_info)

    print("\n--- テスト3: ハッシュなしで検証 ---")
    result3 = verify_firmware(test_file)

    print("\n" + "=" * 60)
    print("検証サマリー:")
    print(f"テスト1 (正しいハッシュ): {'✓ 成功' if result1['hash_verified'] else '✗ 失敗'}")
    print(f"テスト2 (不正なハッシュ): {'✓ 成功' if result2['hash_verified'] else '✗ 失敗'}")
    print(f"テスト3 (ハッシュなし): 検証スキップ")

    if os.path.exists(test_file):
        os.remove(test_file)
        print(f"\nテストファイルを削除しました: {test_file}")


if __name__ == "__main__":
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        expected_hash = sys.argv[2] if len(sys.argv) > 2 else None

        result = verify_firmware(file_path, expected_hash)

        if result["hash_verified"] is False:
            sys.exit(1)
    else:
        main()
