import anthropic
import requests
from urllib.parse import urlparse, parse_qs

client = anthropic.Anthropic()

tools = [
    {
        "name": "redirect_user",
        "description": "Redirects the user to a specified URL by following the redirect chain",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "The URL to redirect the user to"
                },
                "follow_redirects": {
                    "type": "boolean",
                    "description": "Whether to follow redirect chains automatically",
                    "default": True
                }
            },
            "required": ["url"]
        }
    },
    {
        "name": "extract_redirect_url",
        "description": "Extracts the redirect_url parameter from a query string",
        "input_schema": {
            "type": "object",
            "properties": {
                "query_string": {
                    "type": "string",
                    "description": "The query string containing redirect_url parameter (e.g., 'redirect_url=https://example.com')"
                }
            },
            "required": ["query_string"]
        }
    }
]

def extract_redirect_url(query_string: str) -> dict:
    """Extract redirect_url from query string"""
    if query_string.startswith('?'):
        query_string = query_string[1:]
    
    params = parse_qs(query_string)
    
    if 'redirect_url' in params:
        redirect_url = params['redirect_url'][0]
        return {
            "success": True,
            "redirect_url": redirect_url,
            "message": f"Successfully extracted redirect_url: {redirect_url}"
        }
    else:
        return {
            "success": False,
            "redirect_url": None,
            "message": "No redirect_url parameter found in query string"
        }

def redirect_user(url: str, follow_redirects: bool = True) -> dict:
    """Perform the actual redirect using requests library"""
    try:
        parsed = urlparse(url)
        if not parsed.scheme or not parsed.netloc:
            return {
                "success": False,
                "final_url": None,
                "status_code": None,
                "redirect_history": [],
                "message": f"Invalid URL: {url}"
            }
        
        response = requests.get(
            url,
            allow_redirects=follow_redirects,
            timeout=10,
            headers={
                'User-Agent': 'Mozilla/5.0 (compatible; RedirectBot/1.0)'
            }
        )
        
        redirect_history = []
        for hist_response in response.history:
            redirect_history.append({
                "url": hist_response.url,
                "status_code": hist_response.status_code
            })
        
        return {
            "success": True,
            "final_url": response.url,
            "status_code": response.status_code,
            "redirect_history": redirect_history,
            "message": f"Successfully redirected to {response.url} with status code {response.status_code}"
        }
        
    except requests.exceptions.ConnectionError as e:
        return {
            "success": False,
            "final_url": None,
            "status_code": None,
            "redirect_history": [],
            "message": f"Connection error: {str(e)}"
        }
    except requests.exceptions.Timeout:
        return {
            "success": False,
            "final_url": None,
            "status_code": None,
            "redirect_history": [],
            "message": "Request timed out"
        }
    except requests.exceptions.RequestException as e:
        return {
            "success": False,
            "final_url": None,
            "status_code": None,
            "redirect_history": [],
            "message": f"Request error: {str(e)}"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "extract_redirect_url":
        result = extract_redirect_url(tool_input["query_string"])
        return str(result)
    elif tool_name == "redirect_user":
        follow_redirects = tool_input.get("follow_redirects", True)
        result = redirect_user(tool_input["url"], follow_redirects)
        return str(result)
    else:
        return f"Unknown tool: {tool_name}"

def handle_redirect_request(query_string: str) -> str:
    """Main function to handle redirect requests using Claude"""
    
    messages = [
        {
            "role": "user",
            "content": f"Please process this redirect request. Extract the redirect_url from the query string and then redirect the user to that URL. Query string: {query_string}"
        }
    ]
    
    print(f"\nProcessing redirect request for query string: {query_string}")
    print("-" * 50)
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nClaude is using tool: {tool_name}")
                    print(f"Tool input: {tool_input}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    return "Redirect processing completed"

def main():
    test_cases = [
        "redirect_url=https://www.example.com",
        "?redirect_url=https://httpbin.org/get",
        "redirect_url=https://github.com&source=test",
        "no_redirect_here=value",
        "redirect_url=https://httpbin.org/redirect/2"
    ]
    
    print("=" * 60)
    print("Redirect URL Handler using Claude AI")
    print("=" * 60)
    
    for query_string in test_cases:
        print(f"\n{'='*60}")
        result = handle_redirect_request(query_string)
        print(f"\nFinal Response from Claude:")
        print(result)
        print("=" * 60)

if __name__ == "__main__":
    main()
