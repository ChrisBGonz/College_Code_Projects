/**
 * Get PDO database connection
 */
function getDB(): PDO {
    static $pdo = null;
    if ($pdo === null) {
        $dsn = sprintf('mysql:host=%s;dbname=%s;charset=%s', DB_HOST, DB_NAME, DB_CHARSET);
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        $pdo = new PDO($dsn, DB_USER, DB_PASS, $options);
    }
    return $pdo;
}

/**
 * Hash a secret answer securely
 */
function hashAnswer(string $answer): string {
    // Normalize: trim, lowercase, remove spaces
    $normalized = mb_strtolower(trim($answer), 'UTF-8');
    $normalized = preg_replace('/\s+/', '', $normalized);
    return password_hash($normalized, PASSWORD_BCRYPT, ['cost' => 12]);
}

/**
 * Verify a secret answer
 */
function verifyAnswer(string $answer, string $hash): bool {
    $normalized = mb_strtolower(trim($answer), 'UTF-8');
    $normalized = preg_replace('/\s+/', '', $normalized);
    return password_verify($normalized, $hash);
}

/**
 * Generate a secure random token
 */
function generateToken(): string {
    return bin2hex(random_bytes(32));
}

/**
 * Check if IP is locked out
 */
function isLockedOut(string $ip): bool {
    $pdo = getDB();
    $stmt = $pdo->prepare(
        'SELECT attempt_count, last_attempt FROM recovery_attempts 
         WHERE ip_address = ? AND last_attempt > DATE_SUB(NOW(), INTERVAL ? SECOND)'
    );
    $stmt->execute([$ip, LOCKOUT_TIME]);
    $row = $stmt->fetch();

    if ($row && $row['attempt_count'] >= MAX_ATTEMPTS) {
        return true;
    }
    return false;
}

/**
 * Record a failed attempt
 */
function recordFailedAttempt(string $ip): void {
    $pdo = getDB();
    $stmt = $pdo->prepare(
        'INSERT INTO recovery_attempts (ip_address, attempt_count, last_attempt)
         VALUES (?, 1, NOW())
         ON DUPLICATE KEY UPDATE 
             attempt_count = IF(last_attempt < DATE_SUB(NOW(), INTERVAL ? SECOND), 1, attempt_count + 1),
             last_attempt = NOW()'
    );
    $stmt->execute([$ip, LOCKOUT_TIME]);
}

/**
 * Clear attempts for an IP
 */
function clearAttempts(string $ip): void {
    $pdo = getDB();
    $stmt = $pdo->prepare('DELETE FROM recovery_attempts WHERE ip_address = ?');
    $stmt->execute([$ip]);
}

/**
 * Get remaining lockout time in seconds
 */
function getRemainingLockoutTime(string $ip): int {
    $pdo = getDB();
    $stmt = $pdo->prepare(
        'SELECT TIMESTAMPDIFF(SECOND, NOW(), DATE_ADD(last_attempt, INTERVAL ? SECOND)) AS remaining
         FROM recovery_attempts WHERE ip_address = ?'
    );
    $stmt->execute([LOCKOUT_TIME, $ip]);
    $row = $stmt->fetch();
    return $row ? max(0, (int)$row['remaining']) : 0;
}

/**
 * Sanitize output
 */
function h(string $str): string {
    return htmlspecialchars($str, ENT_QUOTES | ENT_HTML5, 'UTF-8');
}

/**
 * Get client IP address
 */
function getClientIP(): string {
    $ip = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
    // Validate IP
    if (filter_var($ip, FILTER_VALIDATE_IP)) {
        return $ip;
    }
    return '0.0.0.0';
}

/**
 * Generate CSRF token
 */
function getCsrfToken(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

/**
 * Validate CSRF token
 */
function validateCsrfToken(string $token): bool {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// ============================================================
// Database setup (run once)
// ============================================================
function setupDatabase(): void {
    $pdo = getDB();

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS users (
            id INT AUTO_INCREMENT PRIMARY KEY,
            username VARCHAR(50) NOT NULL UNIQUE,
            email VARCHAR(255) NOT NULL UNIQUE,
            password_hash VARCHAR(255) NOT NULL,
            secret_question_id INT NOT NULL,
            secret_answer_hash VARCHAR(255) NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
    ");

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS recovery_tokens (
            id INT AUTO_INCREMENT PRIMARY KEY,
            user_id INT NOT NULL,
            token VARCHAR(64) NOT NULL UNIQUE,
            expires_at TIMESTAMP NOT NULL,
            used TINYINT(1) DEFAULT 0,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
    ");

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS recovery_attempts (
            ip_address VARCHAR(45) PRIMARY KEY,
            attempt_count INT DEFAULT 0,
            last_attempt TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
    ");
}

$step    = $_GET['step'] ?? 'start';

$error   = '';

// Handle POST requests
if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    // CSRF validation
    $csrfToken = $_POST['csrf_token'] ?? '';

        $error = 'セキュリティトークンが無効です。ページを再読み込みしてください。';
        $step  = 'start';
    } else {

        switch ($step) {

            // --------------------------------------------------------
            // Step 1: User submits username/email to find account
            // --------------------------------------------------------
            case 'find_account':
                if (isLockedOut($ip)) {
                    $remaining = getRemainingLockoutTime($ip);
                    $error = sprintf(
                        'アカウントがロックされています。%d分%d秒後に再試行してください。',
                        floor($remaining / 60),
                        $remaining % 60
                    );
                    $step = 'start';
                    break;
                }

                $identifier = trim($_POST['identifier'] ?? '');
                if (empty($identifier)) {
                    $error = 'ユーザー名またはメールアドレスを入力してください。';
                    $step  = 'start';
                    break;
                }

                $pdo  = getDB();
                $stmt = $pdo->prepare(
                    'SELECT id, username, secret_question_id FROM users 
                     WHERE username = ? OR email = ? LIMIT 1'
                );
                $stmt->execute([$identifier, $identifier]);
                $user = $stmt->fetch();

                if (!$user) {
                    // Don't reveal whether account exists
                    recordFailedAttempt($ip);
                    $error = 'アカウントが見つかりません。入力内容を確認してください。';
                    $step  = 'start';
                    break;
                }

                // Store user ID in session (not yet authenticated)
                $_SESSION['recovery_user_id']  = $user['id'];
                $_SESSION['recovery_username'] = $user['username'];
                $_SESSION['recovery_question'] = $user['secret_question_id'];
                $step = 'answer_question';
                break;

            // --------------------------------------------------------
            // Step 2: User answers the secret question
            // --------------------------------------------------------
            case 'verify_answer':
                if (isLockedOut($ip)) {
                    $remaining = getRemainingLockoutTime($ip);
                    $error = sprintf(
                        'アカウントがロックされています。%d分%d秒後に再試行してください。',
                        floor($remaining / 60),
                        $remaining % 60
                    );
                    $step = 'start';
                    unset($_SESSION['recovery_user_id'], $_SESSION['recovery_username'], $_SESSION['recovery_question']);
                    break;
                }

                if (empty($_SESSION['recovery_user_id'])) {
                    $error = 'セッションが無効です。最初からやり直してください。';
                    $step  = 'start';
                    break;
                }

                $answer = $_POST['answer'] ?? '';
                if (empty($answer)) {
                    $error = '回答を入力してください。';
                    $step  = 'answer_question';
                    break;
                }

                $userId = (int)$_SESSION['recovery_user_id'];
                $pdo    = getDB();
                $stmt   = $pdo->prepare('SELECT secret_answer_hash FROM users WHERE id = ?');
                $stmt->execute([$userId]);
                $user = $stmt->fetch();

                if (!$user || !verifyAnswer($answer, $user['secret_answer_hash'])) {
                    recordFailedAttempt($ip);
                    $error = '回答が正しくありません。';
                    $step  = 'answer_question';
                    break;
                }

                // Answer is correct — generate recovery token
                clearAttempts($ip);
                $token     = generateToken();
                $expiresAt = date('Y-m-d H:i:s', time() + TOKEN_EXPIRY);

                // Invalidate old tokens for this user
                $pdo->prepare('UPDATE recovery_tokens SET used = 1 WHERE user_id = ?')->execute([$userId]);

                $stmt = $pdo->prepare(
                    'INSERT INTO recovery_tokens (user_id, token, expires_at) VALUES (?, ?, ?)'
                );
                $stmt->execute([$userId, $token, $expiresAt]);

                // Store token in session for the reset step
                $_SESSION['recovery_token']    = $token;
                $_SESSION['recovery_verified'] = true;
                unset($_SESSION['recovery_question']);

                $step = 'reset_password';
                break;

            // --------------------------------------------------------
            // Step 3: User sets a new password
            // --------------------------------------------------------
            case 'reset_password':
                if (empty($_SESSION['recovery_verified']) || empty($_SESSION['recovery_token'])) {
                    $error = 'セッションが無効です。最初からやり直してください。';
                    $step  = 'start';
                    break;
                }

                $token       = $_SESSION['recovery_token'];
                $newPassword = $_POST['new_password'] ?? '';
                $confirmPass = $_POST['confirm_password'] ?? '';

                // Validate password
                if (strlen($newPassword) < 8) {
                    $error = 'パスワードは8文字以上で入力してください。';
                    $step  = 'reset_password';
                    break;
                }
                if (!preg_match('/[A-Z]/', $newPassword)) {
                    $error = 'パスワードには少なくとも1つの大文字を含めてください。';
                    $step  = 'reset_password';
                    break;
                }
                if (!preg_match('/[0-9]/', $newPassword)) {
                    $error = 'パスワードには少なくとも1つの数字を含めてください。';
                    $step  = 'reset_password';
                    break;
                }
                if ($newPassword !== $confirmPass) {
                    $error = 'パスワードが一致しません。';
                    $step  = 'reset_password';
                    break;
                }

                $pdo  = getDB();
                $stmt = $pdo->prepare(
                    'SELECT user_id FROM recovery_tokens 
                     WHERE token = ? AND used = 0 AND expires_at > NOW() LIMIT 1'
                );
                $stmt->execute([$token]);
                $tokenRow = $stmt->fetch();

                if (!$tokenRow) {
                    $error = 'リカバリートークンが無効または期限切れです。最初からやり直してください。';
                    $step  = 'start';
                    unset($_SESSION['recovery_verified'], $_SESSION['recovery_token'],
                          $_SESSION['recovery_user_id'], $_SESSION['recovery_username']);
                    break;
                }

                $userId       = (int)$tokenRow['user_id'];
                $passwordHash = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);

                // Update password and mark token as used
                $pdo->beginTransaction();
                try {
                    $pdo->prepare('UPDATE users SET password_hash = ? WHERE id = ?')
                        ->execute([$passwordHash, $userId]);
                    $pdo->prepare('UPDATE recovery_tokens SET used = 1 WHERE token = ?')
                        ->execute([$token]);
                    $pdo->commit();
                } catch (Exception $e) {
                    $pdo->rollBack();
                    $error = 'パスワードの更新中にエラーが発生しました。もう一度お試しください。';
                    $step  = 'reset_password';
                    break;
                }

                // Clear all recovery session data
                unset(
                    $_SESSION['recovery_verified'],
                    $_SESSION['recovery_token'],
                    $_SESSION['recovery_user_id'],
                    $_SESSION['recovery_username']
                );

                $message = 'パスワードが正常に更新されました。新しいパスワードでログインしてください。';
                $step    = 'complete';
                break;
        }
    }
}
