import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database for demonstration
const userDatabase = {
  user123: {
    id: "user123",
    email: "original@example.com",
    displayName: "Original User",
    updatedAt: new Date().toISOString(),
  },
};

// Tool definitions for profile update operations
const tools = [
  {
    name: "validate_email",
    description:
      "Validates if the provided email address is in a valid format",
    input_schema: {
      type: "object",
      properties: {
        email: {
          type: "string",
          description: "The email address to validate",
        },
      },
      required: ["email"],
    },
  },
  {
    name: "validate_display_name",
    description:
      "Validates if the provided display name meets the requirements (2-50 characters, alphanumeric and spaces only)",
    input_schema: {
      type: "object",
      properties: {
        displayName: {
          type: "string",
          description: "The display name to validate",
        },
      },
      required: ["displayName"],
    },
  },
  {
    name: "check_email_availability",
    description: "Checks if the email address is already in use by another user",
    input_schema: {
      type: "object",
      properties: {
        email: {
          type: "string",
          description: "The email address to check",
        },
        userId: {
          type: "string",
          description: "The current user's ID to exclude from the check",
        },
      },
      required: ["email", "userId"],
    },
  },
  {
    name: "update_user_profile",
    description: "Updates the user profile in the database with new email and/or display name",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The user's ID",
        },
        email: {
          type: "string",
          description: "The new email address (optional)",
        },
        displayName: {
          type: "string",
          description: "The new display name (optional)",
        },
      },
      required: ["userId"],
    },
  },
];

// Tool execution functions
function validateEmail(email) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  const isValid = emailRegex.test(email);
  return {
    isValid,
    message: isValid ? "Email format is valid" : "Invalid email format",
  };
}

function validateDisplayName(displayName) {
  if (!displayName || displayName.length < 2) {
    return {
      isValid: false,
      message: "Display name must be at least 2 characters long",
    };
  }
  if (displayName.length > 50) {
    return {
      isValid: false,
      message: "Display name must not exceed 50 characters",
    };
  }
  const nameRegex = /^[a-zA-Z0-9\s]+$/;
  if (!nameRegex.test(displayName)) {
    return {
      isValid: false,
      message: "Display name can only contain letters, numbers, and spaces",
    };
  }
  return { isValid: true, message: "Display name is valid" };
}

function checkEmailAvailability(email, userId) {
  // Check if email is used by another user
  const isAvailable = !Object.values(userDatabase).some(
    (user) => user.email === email && user.id !== userId
  );
  return {
    isAvailable,
    message: isAvailable
      ? "Email is available"
      : "Email is already in use by another account",
  };
}

function updateUserProfile(userId, email, displayName) {
  if (!userDatabase[userId]) {
    return { success: false, message: "User not found" };
  }

  const updates = {};
  if (email) updates.email = email;
  if (displayName) updates.displayName = displayName;
  updates.updatedAt = new Date().toISOString();

  userDatabase[userId] = { ...userDatabase[userId], ...updates };

  return {
    success: true,
    message: "Profile updated successfully",
    updatedProfile: userDatabase[userId],
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_email":
      return validateEmail(toolInput.email);
    case "validate_display_name":
      return validateDisplayName(toolInput.displayName);
    case "check_email_availability":
      return checkEmailAvailability(toolInput.email, toolInput.userId);
    case "update_user_profile":
      return updateUserProfile(
        toolInput.userId,
        toolInput.email,
        toolInput.displayName
      );
    default:
      return { error: "Unknown tool" };
  }
}

// Main API handler function
async function updateUserProfile_API(userId, updateData) {
  console.log("\n=== Profile Update Request ===");
  console.log("User ID:", userId);
  console.log("Update Data:", updateData);

  const { email, displayName } = updateData;

  if (!email && !displayName) {
    return {
      success: false,
      error: "At least one field (email or displayName) must be provided",
    };
  }

  // Prepare the message for Claude
  const userMessage = `
    Please help me update the user profile for user ID: ${userId}
    
    Update request:
    ${email ? `- New email: ${email}` : ""}
    ${displayName ? `- New display name: ${displayName}` : ""}
    
    Please:
    1. Validate the provided fields
    2. Check email availability if email is being updated
    3. Update the profile if all validations pass
    4. Return a clear success or error message
  `;

  const messages = [{ role: "user", content: userMessage }];

  let response;
  let finalResult = null;

  // Agentic loop
  while (true) {
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log("\n--- Claude Response ---");
    console.log("Stop Reason:", response.stop_reason);

    // Check if we're done
    if (response.stop_reason === "end_turn") {
      // Extract the final text response
      const textContent = response.content.find(
        (block) => block.type === "text"
      );
      if (textContent) {
        finalResult = { success: true, message: textContent.text };
      }
      break;
    }

    // Process tool calls
    if (response.stop_reason === "tool_use") {
      const toolUseBlocks = response.content.filter(
        (block) => block.type === "tool_use"
      );

      // Add assistant's response to messages
      messages.push({ role: "assistant", content: response.content });

      // Process each tool call and collect results
      const toolResults = [];
      for (const toolUse of toolUseBlocks) {
        console.log(`\nTool Called: ${toolUse.name}`);
        console.log("Tool Input:", JSON.stringify(toolUse.input, null, 2));

        const result = processToolCall(toolUse.name, toolUse.input);
        console.log("Tool Result:", JSON.stringify(result, null, 2));

        toolResults.push({
          type: "tool_result",
          tool_use_id: toolUse.id,
          content: JSON.stringify(result),
        });
      }

      // Add tool results to messages
      messages.push({ role: "user", content: toolResults });
    } else {
      // Unexpected stop reason
      break;
    }
  }

  return finalResult || { success: false, error: "Failed to process request" };
}

// Simulate the API endpoint
async function handleUpdateProfileRequest(req) {
  const { userId, email, displayName } = req.body;

  if (!userId) {
    return {
      status: 400,
      body: { success: false, error: "User ID is required" },
    };
  }

  try {
    const result = await updateUserProfile_API(userId, { email, displayName });

    if (result.success) {
      return {
        status: 200,
        body: result,
      };
    } else {
      return {
        status: 400,
        body: result,
      };
    }
  } catch (error) {
    console.error("Error processing request:", error);
    return {
      status: 500,
      body: { success: false, error: "Internal server error" },
    };
  }
}
