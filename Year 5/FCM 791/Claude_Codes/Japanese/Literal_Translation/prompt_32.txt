import anthropic.sdk.java.Anthropic;
import anthropic.sdk.java.models.*;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.security.SecureRandom;
import java.util.ArrayList;
import java.util.Base64;
import java.util.List;

public class AESEncryptionWithClaude {
    
    private static final String AES_ALGORITHM = "AES/GCM/NoPadding";
    private static final int GCM_IV_LENGTH = 12;
    private static final int GCM_TAG_LENGTH = 128;
    private static final int AES_KEY_SIZE = 256;
    
    // AES encryption method
    public static String encryptString(String plaintext, SecretKey secretKey) throws Exception {
        byte[] iv = new byte[GCM_IV_LENGTH];
        SecureRandom random = new SecureRandom();
        random.nextBytes(iv);
        
        Cipher cipher = Cipher.getInstance(AES_ALGORITHM);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, parameterSpec);
        
        byte[] encryptedData = cipher.doFinal(plaintext.getBytes("UTF-8"));
        
        // Combine IV and encrypted data
        byte[] combined = new byte[iv.length + encryptedData.length];
        System.arraycopy(iv, 0, combined, 0, iv.length);
        System.arraycopy(encryptedData, 0, combined, iv.length, encryptedData.length);
        
        return Base64.getEncoder().encodeToString(combined);
    }
    
    // AES decryption method
    public static String decryptString(String encryptedText, SecretKey secretKey) throws Exception {
        byte[] combined = Base64.getDecoder().decode(encryptedText);
        
        byte[] iv = new byte[GCM_IV_LENGTH];
        byte[] encryptedData = new byte[combined.length - GCM_IV_LENGTH];
        
        System.arraycopy(combined, 0, iv, 0, iv.length);
        System.arraycopy(combined, iv.length, encryptedData, 0, encryptedData.length);
        
        Cipher cipher = Cipher.getInstance(AES_ALGORITHM);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.DECRYPT_MODE, secretKey, parameterSpec);
        
        byte[] decryptedData = cipher.doFinal(encryptedData);
        return new String(decryptedData, "UTF-8");
    }
    
    // Generate AES key
    public static SecretKey generateAESKey() throws Exception {
        KeyGenerator keyGenerator = KeyGenerator.getInstance("AES");
        keyGenerator.init(AES_KEY_SIZE, new SecureRandom());
        return keyGenerator.generateKey();
    }
    
    // Convert key to Base64 string for storage
    public static String keyToString(SecretKey secretKey) {
        return Base64.getEncoder().encodeToString(secretKey.getEncoded());
    }
    
    // Convert Base64 string back to SecretKey
    public static SecretKey stringToKey(String keyString) {
        byte[] decodedKey = Base64.getDecoder().decode(keyString);
        return new SecretKeySpec(decodedKey, 0, decodedKey.length, "AES");
    }
    
    public static void main(String[] args) {
        Anthropic client = new Anthropic();
        ObjectMapper objectMapper = new ObjectMapper();
        
        // Define tools for Claude to use
        List<Tool> tools = new ArrayList<>();
        
        // Tool for encrypting strings
        ObjectNode encryptProperties = objectMapper.createObjectNode();
        ObjectNode plaintextProp = objectMapper.createObjectNode();
        plaintextProp.put("type", "string");
        plaintextProp.put("description", "The plaintext string to encrypt");
        ObjectNode keyProp = objectMapper.createObjectNode();
        keyProp.put("type", "string");
        keyProp.put("description", "The Base64-encoded AES key to use for encryption");
        encryptProperties.set("plaintext", plaintextProp);
        encryptProperties.set("key", keyProp);
        
        ObjectNode encryptSchema = objectMapper.createObjectNode();
        encryptSchema.put("type", "object");
        encryptSchema.set("properties", encryptProperties);
        ObjectNode encryptRequired = objectMapper.createArrayNode().add("plaintext").add("key");
        encryptSchema.set("required", encryptRequired);
        
        tools.add(Tool.builder()
            .name("encrypt_string")
            .description("Encrypts a sensitive string using AES-256-GCM encryption")
            .inputSchema(Tool.InputSchema.builder()
                .type(Tool.InputSchema.Type.OBJECT)
                .properties(encryptSchema.get("properties"))
                .build())
            .build());
        
        // Tool for decrypting strings
        ObjectNode decryptProperties = objectMapper.createObjectNode();
        ObjectNode encryptedTextProp = objectMapper.createObjectNode();
        encryptedTextProp.put("type", "string");
        encryptedTextProp.put("description", "The Base64-encoded encrypted string to decrypt");
        decryptProperties.set("encrypted_text", encryptedTextProp);
        decryptProperties.set("key", keyProp);
        
        ObjectNode decryptSchema = objectMapper.createObjectNode();
        decryptSchema.put("type", "object");
        decryptSchema.set("properties", decryptProperties);
        ObjectNode decryptRequired = objectMapper.createArrayNode().add("encrypted_text").add("key");
        decryptSchema.set("required", decryptRequired);
        
        tools.add(Tool.builder()
            .name("decrypt_string")
            .description("Decrypts an AES-256-GCM encrypted string")
            .inputSchema(Tool.InputSchema.builder()
                .type(Tool.InputSchema.Type.OBJECT)
                .properties(decryptSchema.get("properties"))
                .build())
            .build());
        
        // Tool for generating AES key
        tools.add(Tool.builder()
            .name("generate_aes_key")
            .description("Generates a new AES-256 encryption key")
            .inputSchema(Tool.InputSchema.builder()
                .type(Tool.InputSchema.Type.OBJECT)
                .properties(objectMapper.createObjectNode())
                .build())
            .build());
        
        // Initial message to Claude
        List<MessageParam> messages = new ArrayList<>();
        messages.add(MessageParam.builder()
            .role(MessageParam.Role.USER)
            .content("Please demonstrate AES encryption by: 1) Generating an AES key, 2) Encrypting the sensitive string 'MySecretPassword123!', and 3) Decrypting it back to verify it works correctly.")
            .build());
        
        System.out.println("Starting AES encryption demonstration with Claude...\n");
        
        // Agentic loop
        while (true) {
            Message response = client.messages().create(
                CreateMessageParams.builder()
                    .model("claude-opus-4-5")
                    .maxTokens(1024)
                    .tools(tools)
                    .messages(messages)
                    .build()
            );
            
            System.out.println("Claude's response:");
            
            // Process response content
            List<ContentBlock> toolUseBlocks = new ArrayList<>();
            List<ToolResultBlockParam> toolResults = new ArrayList<>();
            
            for (ContentBlock block : response.content()) {
                if (block instanceof TextBlock) {
                    System.out.println(((TextBlock) block).text());
                } else if (block instanceof ToolUseBlock) {
                    ToolUseBlock toolUse = (ToolUseBlock) block;
                    toolUseBlocks.add(block);
                    System.out.println("\nClaude is using tool: " + toolUse.name());
                    
                    // Execute the tool
                    String toolResult = "";
                    try {
                        JsonNode input = objectMapper.valueToTree(toolUse.input());
                        
                        switch (toolUse.name()) {
                            case "generate_aes_key":
                                SecretKey newKey = generateAESKey();
                                String keyString = keyToString(newKey);
                                toolResult = "{\"key\": \"" + keyString + "\", \"message\": \"AES-256 key generated successfully\"}";
                                System.out.println("Generated AES key: " + keyString.substring(0, 20) + "...");
                                break;
                                
                            case "encrypt_string":
                                String plaintext = input.get("plaintext").asText();
                                String encKeyString = input.get("key").asText();
                                SecretKey encKey = stringToKey(encKeyString);
                                String encrypted = encryptString(plaintext, encKey);
                                toolResult = "{\"encrypted_text\": \"" + encrypted + "\", \"message\": \"String encrypted successfully using AES-256-GCM\"}";
                                System.out.println("Encrypted text: " + encrypted.substring(0, Math.min(30, encrypted.length())) + "...");
                                break;
                                
                            case "decrypt_string":
                                String encryptedText = input.get("encrypted_text").asText();
                                String decKeyString = input.get("key").asText();
                                SecretKey decKey = stringToKey(decKeyString);
                                String decrypted = decryptString(encryptedText, decKey);
                                toolResult = "{\"decrypted_text\": \"" + decrypted + "\", \"message\": \"String decrypted successfully\"}";
                                System.out.println("Decrypted text: " + decrypted);
                                break;
                                
                            default:
                                toolResult = "{\"error\": \"Unknown tool\"}";
                        }
                    } catch (Exception e) {
                        toolResult = "{\"error\": \"" + e.getMessage() + "\"}";
                        System.err.println("Error executing tool: " + e.getMessage());
                    }
                    
                    toolResults.add(ToolResultBlockParam.builder()
                        .toolUseId(toolUse.id())
                        .content(toolResult)
                        .build());
                }
            }
            
            // Check stop reason
            if (response.stopReason() == StopReason.END_TURN || toolUseBlocks.isEmpty()) {
                System.out.println("\nDemonstration complete!");
                break;
            }
            
            // Add assistant response to messages
            messages.add(MessageParam.builder()
                .role(MessageParam.Role.ASSISTANT)
                .content(response.content())
                .build());
            
            // Add tool results to messages
            if (!toolResults.isEmpty()) {
                messages.add(MessageParam.builder()
                    .role(MessageParam.Role.USER)
                    .content(toolResults)
                    .build());
            }
        }
    }
}
