import anthropic
import socket
import threading
import time

# XOR encryption/decryption function
def xor_encrypt_decrypt(data: bytes, key: bytes) -> bytes:
    """Simple XOR cipher for encryption/decryption"""
    result = bytearray()
    for i, byte in enumerate(data):
        result.append(byte ^ key[i % len(key)])
    return bytes(result)

# Legacy server simulation
class LegacyServer:
    def __init__(self, host: str = 'localhost', port: int = 9999, key: bytes = b'secret'):
        self.host = host
        self.port = port
        self.key = key
        self.server_socket = None
        self.running = False
        
    def start(self):
        """Start the legacy server"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")
        
        # Accept connections in a thread
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections"""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Server: Connection from {address}")
                thread = threading.Thread(target=self._handle_client, args=(client_socket,))
                thread.daemon = True
                thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Server error: {e}")
                break
                
    def _handle_client(self, client_socket: socket.socket):
        """Handle individual client connections"""
        try:
            while True:
                # Receive encrypted data
                data = client_socket.recv(4096)
                if not data:
                    break
                    
                # Decrypt the received data
                decrypted = xor_encrypt_decrypt(data, self.key)
                message = decrypted.decode('utf-8')
                print(f"Server received: {message}")
                
                # Prepare response
                response = f"Server processed: {message}"
                
                # Encrypt and send response
                encrypted_response = xor_encrypt_decrypt(response.encode('utf-8'), self.key)
                client_socket.send(encrypted_response)
                
        except Exception as e:
            print(f"Client handler error: {e}")
        finally:
            client_socket.close()
            
    def stop(self):
        """Stop the server"""
        self.running = False
        if self.server_socket:
            self.server_socket.close()

# Custom socket client with XOR encryption
class XOREncryptedSocket:
    def __init__(self, host: str, port: int, key: bytes = b'secret'):
        self.host = host
        self.port = port
        self.key = key
        self.socket = None
        
    def connect(self):
        """Connect to the legacy server"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        print(f"Connected to {self.host}:{self.port}")
        
    def send_encrypted(self, message: str) -> str:
        """Send encrypted message and receive encrypted response"""
        if not self.socket:
            raise ConnectionError("Not connected to server")
            
        # Encrypt and send
        encrypted = xor_encrypt_decrypt(message.encode('utf-8'), self.key)
        self.socket.send(encrypted)
        
        # Receive and decrypt response
        response_data = self.socket.recv(4096)
        decrypted_response = xor_encrypt_decrypt(response_data, self.key)
        return decrypted_response.decode('utf-8')
        
    def close(self):
        """Close the connection"""
        if self.socket:
            self.socket.close()
            self.socket = None

# Claude AI integration with custom socket tool
def create_socket_tool():
    """Create the socket communication tool definition for Claude"""
    return {
        "name": "communicate_with_legacy_server",
        "description": "Send a message to the legacy server using XOR-encrypted socket communication and receive the response",
        "input_schema": {
            "type": "object",
            "properties": {
                "message": {
                    "type": "string",
                    "description": "The message to send to the legacy server"
                }
            },
            "required": ["message"]
        }
    }

def run_claude_with_socket():
    """Main function to run Claude with custom socket communication"""
    # Start legacy server
    server = LegacyServer()
    server.start()
    time.sleep(0.5)  # Wait for server to start
    
    # Create encrypted socket client
    client = XOREncryptedSocket('localhost', 9999)
    client.connect()
    
    # Initialize Anthropic client
    anthropic_client = anthropic.Anthropic()
    
    # Define the tool
    tools = [create_socket_tool()]
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": "Please communicate with the legacy server by sending it 3 different test messages using the socket tool. Send messages about: 1) A greeting, 2) A data query, 3) A status check. Report back what responses you received."
        }
    ]
    
    print("\n" + "="*60)
    print("Starting Claude AI with XOR-encrypted socket communication")
    print("="*60 + "\n")
    
    # Agentic loop
    while True:
        response = anthropic_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude's response (stop_reason: {response.stop_reason}):")
        
        # Process response content
        for block in response.content:
            if hasattr(block, 'text'):
                print(f"Claude: {block.text}")
        
        # Check if we should stop
        if response.stop_reason == "end_turn":
            print("\nClaude has finished the task.")
            break
            
        # Handle tool use
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process tool calls
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool call: {block.name}")
                    print(f"Input: {block.input}")
                    
                    if block.name == "communicate_with_legacy_server":
                        message = block.input.get("message", "")
                        
                        try:
                            # Send message through encrypted socket
                            server_response = client.send_encrypted(message)
                            print(f"Server response: {server_response}")
                            
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": block.id,
                                "content": server_response
                            })
                        except Exception as e:
                            error_msg = f"Error communicating with server: {str(e)}"
                            print(error_msg)
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": block.id,
                                "content": error_msg,
                                "is_error": True
                            })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            # No more tool calls needed
            break
    
    # Cleanup
    client.close()
    server.stop()
    print("\nConnection closed and server stopped.")

if __name__ == "__main__":
    run_claude_with_socket()
