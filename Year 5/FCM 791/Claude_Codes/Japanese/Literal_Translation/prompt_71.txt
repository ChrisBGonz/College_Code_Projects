
```python
import anthropic
import pickle
import os
import json
from datetime import datetime

client = anthropic.Anthropic()

class GameState:
    """Complex game state object to be saved and loaded"""
    def __init__(self):
        self.player_name = ""
        self.level = 1
        self.health = 100
        self.max_health = 100
        self.experience = 0
        self.gold = 0
        self.inventory = []
        self.skills = {}
        self.quests = {}
        self.position = {"x": 0, "y": 0, "map": "starting_village"}
        self.achievements = []
        self.play_time = 0
        self.save_date = None
        self.game_settings = {
            "difficulty": "normal",
            "sound": True,
            "music": True,
            "language": "ja"
        }
        self.story_flags = {}
        self.defeated_enemies = []
        self.visited_locations = []
    
    def to_dict(self):
        """Convert game state to dictionary for display"""
        return {
            "player_name": self.player_name,
            "level": self.level,
            "health": f"{self.health}/{self.max_health}",
            "experience": self.experience,
            "gold": self.gold,
            "inventory": self.inventory,
            "skills": self.skills,
            "quests": self.quests,
            "position": self.position,
            "achievements": self.achievements,
            "play_time": f"{self.play_time} minutes",
            "save_date": str(self.save_date),
            "game_settings": self.game_settings,
            "story_flags": self.story_flags,
            "defeated_enemies": self.defeated_enemies,
            "visited_locations": self.visited_locations
        }

def save_game_state(game_state: GameState, filename: str = "game_save.pkl") -> bool:
    """Save game state to file using pickle"""
    try:
        game_state.save_date = datetime.now()
        with open(filename, 'wb') as f:
            pickle.dump(game_state, f)
        print(f"✅ ゲームステートを '{filename}' に保存しました")
        return True
    except Exception as e:
        print(f"❌ 保存エラー: {e}")
        return False

def load_game_state(filename: str = "game_save.pkl") -> GameState:
    """Load game state from file using pickle"""
    try:
        if not os.path.exists(filename):
            print(f"⚠️ セーブファイル '{filename}' が見つかりません。新しいゲームを開始します。")
            return GameState()
        
        with open(filename, 'rb') as f:
            game_state = pickle.load(f)
        print(f"✅ ゲームステートを '{filename}' から読み込みました")
        return game_state
    except Exception as e:
        print(f"❌ 読み込みエラー: {e}")
        return GameState()

def process_ai_response(game_state: GameState, user_input: str) -> tuple[str, GameState]:
    """Process user input with Claude AI and update game state"""
    
    game_state_json = json.dumps(game_state.to_dict(), ensure_ascii=False, indent=2)
    
    system_prompt = """あなたはRPGゲームのAIゲームマスターです。
プレイヤーのアクションを処理し、ゲームステートを更新してください。

以下のルールに従ってください：
1. プレイヤーのアクションに基づいてゲームの状態を更新する
2. 戦闘、クエスト、アイテム収集などのゲームイベントを処理する
3. 更新されたゲームステートをJSON形式で返す
4. 日本語でゲームの説明を提供する

レスポンスは以下の形式で返してください：
{
  "narrative": "ゲームの説明テキスト",
  "state_updates": {
    "health": 数値（変更がある場合）,
    "experience": 数値（変更がある場合）,
    "gold": 数値（変更がある場合）,
    "level": 数値（変更がある場合）,
    "inventory": ["アイテムリスト"]（変更がある場合）,
    "skills": {"スキル名": レベル}（変更がある場合）,
    "quests": {"クエスト名": "状態"}（変更がある場合）,
    "position": {"x": 数値, "y": 数値, "map": "マップ名"}（変更がある場合）,
    "achievements": ["実績リスト"]（変更がある場合）,
    "story_flags": {"フラグ名": 値}（変更がある場合）,
    "defeated_enemies": ["敵リスト"]（変更がある場合）,
    "visited_locations": ["場所リスト"]（変更がある場合）
  }
}"""

    messages = [
        {
            "role": "user",
            "content": f"""現在のゲームステート:
{game_state_json}

プレイヤーのアクション: {user_input}

このアクションを処理して、ゲームステートを更新してください。"""
        }
    ]
    
    print("\n🤖 AIが処理中...")
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        system=system_prompt,
        messages=messages
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n")
    
    try:
        start_idx = full_response.find('{')
        end_idx = full_response.rfind('}') + 1
        if start_idx != -1 and end_idx > start_idx:
            json_str = full_response[start_idx:end_idx]
            response_data = json.loads(json_str)
            
            narrative = response_data.get("narrative", "")
            state_updates = response_data.get("state_updates", {})
            
            if "health" in state_updates:
                game_state.health = min(state_updates["health"], game_state.max_health)
            if "experience" in state_updates:
                game_state.experience = state_updates["experience"]
                if game_state.experience >= game_state.level * 100:
                    game_state.level += 1
                    game_state.max_health += 10
                    game_state.health = game_state.max_health
                    print(f"🎉 レベルアップ！ レベル {game_state.level} になりました！")
            if "gold" in state_updates:
                game_state.gold = state_updates["gold"]
            if "level" in state_updates:
                game_state.level = state_updates["level"]
            if "inventory" in state_updates:
                game_state.inventory = state_updates["inventory"]
            if "skills" in state_updates:
                game_state.skills.update(state_updates["skills"])
            if "quests" in state_updates:
                game_state.quests.update(state_updates["quests"])
            if "position" in state_updates:
                game_state.position.update(state_updates["position"])
            if "achievements" in state_updates:
                for achievement in state_updates["achievements"]:
                    if achievement not in game_state.achievements:
                        game_state.achievements.append(achievement)
            if "story_flags" in state_updates:
                game_state.story_flags.update(state_updates["story_flags"])
            if "defeated_enemies" in state_updates:
                for enemy in state_updates["defeated_enemies"]:
                    if enemy not in game_state.defeated_enemies:
                        game_state.defeated_enemies.append(enemy)
            if "visited_locations" in state_updates:
                for location in state_updates["visited_locations"]:
                    if location not in game_state.visited_locations:
                        game_state.visited_locations.append(location)
            
            game_state.play_time += 1
            
            return narrative, game_state
    except json.JSONDecodeError:
        pass
    
    return full_response, game_state

def display_game_status(game_state: GameState):
    """Display current game status"""
    print("\n" + "="*50)
    print("📊 現在のゲームステータス")
    print("="*50)
    print(f"👤 プレイヤー: {game_state.player_name}")
    print(f"⭐ レベル: {game_state.level}")
    print(f"❤️  HP: {game_state.health}/{game_state.max_health}")
    print(f"✨ 経験値: {game_state.experience}")
    print(f"💰 ゴールド: {game_state.gold}")
    print(f"📍 位置: {game_state.position['map']} ({game_state.position['x']}, {game_state.position['y']})")
    
    if game_state.inventory:
        print(f"🎒 インベントリ: {', '.join(game_state.inventory)}")
    
    if game_state.skills:
        skills_str = ', '.join([f"{k}(Lv{v})" for k, v in game_state.skills.items()])
        print(f"⚔️  スキル: {skills_str}")
    
    if game_state.quests:
        print(f"📜 クエスト: {', '.join([f'{k}:{v}' for k, v in game_state.quests.items()])}")
    
    if game_state.achievements:
        print(f"🏆 実績: {', '.join(game_state.achievements)}")
    
    print(f"⏱️  プレイ時間: {game_state.play_time}分")
    print("="*50)

def main():
    """Main game loop"""
    print("🎮 RPGゲーム - AIゲームマスター")
    print("="*50)
    
    save_file = "game_save.pkl"
    
    if os.path.exists(save_file):
        print("\n既存のセーブデータが見つかりました。")
        choice = input("続きから始めますか？ (y/n): ").strip().lower()
        if choice == 'y':
            game_state = load_game_state(save_file)
        else:
            game_state = GameState()
            player_name = input("プレイヤー名を入力してください: ").strip()
            game_state.player_name = player_name if player_name else "勇者"
            print(f"\n🌟 {game_state.player_name}の冒険が始まります！")
    else:
        game_state = GameState()
        player_name = input("プレイヤー名を入力してください: ").strip()
        game_state.player_name = player_name if player_name else "勇者"
        print(f"\n🌟 {game_state.player_name}の冒険が始まります！")
    
    display_game_status(game_state)
    
    print("\n📖 コマンド:")
    print("  - ゲームアクションを入力してください（例：「村を探索する」「スライムと戦う」）")
    print("  - 'status' でステータス表示")
    print("  - 'save' でゲームを保存")
    print("  - 'load' でゲームを読み込み")
    print("  - 'quit' でゲーム終了")
    print()
    
    while True:
        user_input = input("\n🎯 アクション: ").strip()
        
        if not user_input:
            continue
        
        if user_input.lower() == 'quit':
            save_choice = input("ゲームを保存して終了しますか？ (y/n): ").strip().lower()
            if save_choice == 'y':
                save_game_state(game_state, save_file)
            print("👋 ゲームを終了します。またね！")
            break
        
        elif user_input.lower() == 'status':
            display_game_status(game_state)
        
        elif user_input.lower() == 'save':
            save_game_state(game_state, save_file)
        
        elif user_input.lower() == 'load':
            loaded_state = load_game_state(save_file)
            if loaded_state:
                game_state = loaded_state
                display_game_status(game_state)
        
        else:
            narrative, game_state = process_ai_response(game_state, user_input)
            
            if game_state.play_time % 5 == 0:
                auto_save = input("\n自動保存しますか？ (y/n): ").strip().lower()
                if auto_save == 'y':
                    save_game_state(game_state, save_file)
            
            display_game_status(game_state)

if __name__ == "__main__":
    main()
```
