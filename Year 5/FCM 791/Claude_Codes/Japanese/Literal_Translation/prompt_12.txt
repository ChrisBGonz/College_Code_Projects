import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";

const client = new Anthropic();

async function generateExpressServerConfig(): Promise<void> {
  console.log("Generating production-ready Express.js server configuration...\n");

  const prompt = `Create a production-ready Express.js server configuration with the following requirements:

1. Basic Express.js server setup with TypeScript support
2. Security middleware (helmet, cors, rate limiting)
3. Request logging middleware (morgan)
4. Body parsing middleware
5. Comprehensive error handling middleware including:
   - 404 Not Found handler
   - Global error handler
   - Async error wrapper utility
6. Environment variable configuration
7. Graceful shutdown handling
8. Health check endpoint
9. API versioning structure
10. Request validation middleware example

Please provide:
- server.ts (main server file)
- middleware/errorHandler.ts (error handling middleware)
- middleware/requestLogger.ts (request logging)
- config/environment.ts (environment configuration)
- utils/asyncWrapper.ts (async error wrapper)
- package.json (with necessary dependencies)

Make sure the code is production-ready with proper TypeScript types, error handling, and best practices.`;

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    messages: [
      {
        role: "user",
        content: prompt,
      },
    ],
  });

  let fullResponse = "";
  process.stdout.write("Streaming response:\n\n");

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  const finalMessage = await stream.finalMessage();
  console.log("\n\n--- Generation Complete ---");
  console.log(
    `Total tokens used: ${finalMessage.usage.input_tokens + finalMessage.usage.output_tokens}`
  );

  // Parse and save the generated files
  await saveGeneratedFiles(fullResponse);
}

async function saveGeneratedFiles(content: string): Promise<void> {
  const outputDir = "./generated-express-server";

  // Create output directory
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  // Extract code blocks and their file names
  const filePattern =
    /(?:###?\s+(?:File:\s+)?`?([^`\n]+\.(?:ts|js|json))`?|(?:^|\n)(?:\/\/|#)\s*(?:File:\s+)?([^\n]+\.(?:ts|js|json)))\s*\n```(?:typescript|javascript|json)?\n([\s\S]*?)```/gm;

  const files: { name: string; content: string }[] = [];
  let match;

  while ((match = filePattern.exec(content)) !== null) {
    const fileName = (match[1] || match[2]).trim();
    const fileContent = match[3].trim();
    files.push({ name: fileName, content: fileContent });
  }

  if (files.length === 0) {
    // Try alternative pattern for code blocks
    const altPattern = /```(?:typescript|javascript|json)\n([\s\S]*?)```/gm;
    const codeBlocks: string[] = [];

    while ((match = altPattern.exec(content)) !== null) {
      codeBlocks.push(match[1].trim());
    }

    if (codeBlocks.length > 0) {
      // Save as a single file if we can't determine individual file names
      const combinedFile = path.join(outputDir, "server-config.ts");
      fs.writeFileSync(combinedFile, codeBlocks.join("\n\n// ---\n\n"));
      console.log(`\nSaved combined server configuration to: ${combinedFile}`);
    }
  } else {
    // Save individual files
    for (const file of files) {
      const filePath = path.join(outputDir, file.name);
      const fileDir = path.dirname(filePath);

      if (!fs.existsSync(fileDir)) {
        fs.mkdirSync(fileDir, { recursive: true });
      }

      fs.writeFileSync(filePath, file.content);
      console.log(`\nSaved: ${filePath}`);
    }
  }

  // Save the full response as documentation
  const docFile = path.join(outputDir, "README.md");
  fs.writeFileSync(docFile, `# Generated Express.js Server Configuration\n\n${content}`);
  console.log(`\nSaved documentation to: ${docFile}`);
}

// Main execution
generateExpressServerConfig().catch((error) => {
  console.error("Error generating Express.js server configuration:", error);
  process.exit(1);
});
