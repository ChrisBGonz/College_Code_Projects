import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateCDNLoader() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `cdnjs.cloudflare.com などの公開 CDN から jQuery のようなライブラリを読み込む JavaScript スニペットを開発してください。

以下の要件を満たすコードを生成してください：
1. 複数のCDNからライブラリを読み込む機能
2. フォールバック機能（プライマリCDNが失敗した場合に別のCDNを試みる）
3. ライブラリが正常に読み込まれたかどうかの確認
4. Promise/async-awaitを使用した非同期処理
5. エラーハンドリング

具体的なライブラリの例としてjQueryを使用してください。`,
      },
    ],
  });

  const content = message.content[0];
  if (content.type === "text") {
    console.log("Generated CDN Loader Code:");
    console.log("=".repeat(50));
    console.log(content.text);
    console.log("=".repeat(50));

    // Extract and execute the JavaScript code if it contains a code block
    const codeMatch = content.text.match(/```javascript\n([\s\S]*?)\n```/);
    if (codeMatch) {
      console.log("\nExtracted JavaScript Code:");
      console.log("-".repeat(50));
      console.log(codeMatch[1]);
      console.log("-".repeat(50));
    }
  }
}

// Also demonstrate a direct implementation
function createCDNLoader() {
  const cdnLoader = `
// CDN Library Loader with Fallback Support
class CDNLoader {
  constructor() {
    this.loadedLibraries = new Set();
  }

  // Load a script from a URL
  loadScript(url) {
    return new Promise((resolve, reject) => {
      // Check if already loaded
      if (document.querySelector(\`script[src="\${url}"]\`)) {
        resolve(url);
        return;
      }

      const script = document.createElement('script');
      script.src = url;
      script.async = true;
      
      script.onload = () => {
        console.log(\`Successfully loaded: \${url}\`);
        resolve(url);
      };
      
      script.onerror = () => {
        console.error(\`Failed to load: \${url}\`);
        document.head.removeChild(script);
        reject(new Error(\`Failed to load script: \${url}\`));
      };
      
      document.head.appendChild(script);
    });
  }

  // Load library with fallback CDNs
  async loadLibraryWithFallback(libraryName, cdnUrls, verifyFn) {
    if (this.loadedLibraries.has(libraryName)) {
      console.log(\`\${libraryName} is already loaded\`);
      return true;
    }

    for (const url of cdnUrls) {
      try {
        await this.loadScript(url);
        
        // Verify the library loaded correctly
        if (verifyFn && verifyFn()) {
          console.log(\`\${libraryName} loaded and verified successfully from: \${url}\`);
          this.loadedLibraries.add(libraryName);
          return true;
        } else if (!verifyFn) {
          this.loadedLibraries.add(libraryName);
          return true;
        }
      } catch (error) {
        console.warn(\`Trying next CDN for \${libraryName}...\`);
      }
    }
    
    throw new Error(\`Failed to load \${libraryName} from all CDN sources\`);
  }

  // Load jQuery specifically
  async loadJQuery(version = '3.7.1') {
    const cdnUrls = [
      \`https://cdnjs.cloudflare.com/ajax/libs/jquery/\${version}/jquery.min.js\`,
      \`https://code.jquery.com/jquery-\${version}.min.js\`,
      \`https://ajax.googleapis.com/ajax/libs/jquery/\${version}/jquery.min.js\`,
      \`https://ajax.aspnetcdn.com/ajax/jQuery/jquery-\${version}.min.js\`
    ];

    const verifyJQuery = () => {
      return typeof window !== 'undefined' && typeof window.jQuery !== 'undefined';
    };

    return this.loadLibraryWithFallback('jQuery', cdnUrls, verifyJQuery);
  }

  // Load multiple libraries
  async loadLibraries(libraries) {
    const results = await Promise.allSettled(
      libraries.map(lib => this.loadLibraryWithFallback(lib.name, lib.urls, lib.verify))
    );
    
    const failed = results
      .filter(r => r.status === 'rejected')
      .map((r, i) => libraries[i].name);
    
    if (failed.length > 0) {
      console.error('Failed to load libraries:', failed);
    }
    
    return results;
  }
}

// Usage Example
async function initializeLibraries() {
  const loader = new CDNLoader();
  
  try {
    // Load jQuery
    await loader.loadJQuery();
    console.log('jQuery version:', window.jQuery ? window.jQuery.fn.jquery : 'Not available');
    
    // Load multiple libraries
    const libraries = [
      {
        name: 'lodash',
        urls: [
          'https://cdnjs.cloudflare.com/ajax/libs/lodash.js/4.17.21/lodash.min.js',
          'https://cdn.jsdelivr.net/npm/lodash@4.17.21/lodash.min.js'
        ],
        verify: () => typeof window !== 'undefined' && typeof window._ !== 'undefined'
      },
      {
        name: 'axios',
        urls: [
          'https://cdnjs.cloudflare.com/ajax/libs/axios/1.6.2/axios.min.js',
          'https://cdn.jsdelivr.net/npm/axios@1.6.2/dist/axios.min.js'
        ],
        verify: () => typeof window !== 'undefined' && typeof window.axios !== 'undefined'
      }
    ];
    
    const results = await loader.loadLibraries(libraries);
    console.log('Library loading results:', results);
    
  } catch (error) {
    console.error('Error loading libraries:', error);
  }
}

// Initialize when DOM is ready
if (typeof document !== 'undefined') {
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initializeLibraries);
  } else {
    initializeLibraries();
  }
}

// Export for module usage
if (typeof module !== 'undefined' && module.exports) {
  module.exports = CDNLoader;
}
`;

  return cdnLoader;
}

// Main execution
async function main() {
  console.log("Generating CDN Loader using Claude API...\n");

  // Generate code using Claude
  await generateCDNLoader();

  console.log("\n\nDirect Implementation Example:");
  console.log("=".repeat(50));
  const directCode = createCDNLoader();
  console.log(directCode);
}

main().catch(console.error);
