import Anthropic from "@anthropic-ai/sdk";
import express from "express";

const client = new Anthropic();

// Function to log system errors using console.log
function logSystemError(error, context = {}) {
  const timestamp = new Date().toISOString();
  const errorLog = {
    timestamp,
    level: "ERROR",
    message: error.message || "Unknown error",
    stack: error.stack || "No stack trace available",
    context,
    environment: process.env.NODE_ENV || "production",
  };

  console.log(JSON.stringify(errorLog, null, 2));
}

// Function to generate error handling code using Claude API with streaming
async function generateErrorHandlingCode(errorDescription) {
  console.log("Generating error handling code for:", errorDescription);

  let generatedCode = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a JavaScript error handling function for an Express server that handles the following error scenario: ${errorDescription}. 
        The function should:
        1. Use console.log for logging
        2. Include proper error formatting
        3. Handle both synchronous and asynchronous errors
        4. Return appropriate HTTP status codes
        Please provide only the code without explanations.`,
      },
    ],
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      generatedCode += chunk.delta.text;
      process.stdout.write(chunk.delta.text);
    }
  }

  console.log("\n");
  return generatedCode;
}

// Express server setup with error logging
const app = express();
app.use(express.json());

// Middleware for request logging
app.use((req, res, next) => {
  console.log(
    JSON.stringify({
      timestamp: new Date().toISOString(),
      method: req.method,
      path: req.path,
      ip: req.ip,
    })
  );
  next();
});

// Route to demonstrate error logging
app.get("/test-error", (req, res) => {
  try {
    throw new Error("Test error for demonstration");
  } catch (error) {
    logSystemError(error, {
      route: "/test-error",
      method: "GET",
    });
    res.status(500).json({ error: "Internal server error" });
  }
});

// Route to generate error handling code using Claude
app.post("/generate-error-handler", async (req, res) => {
  const { errorDescription } = req.body;

  if (!errorDescription) {
    const validationError = new Error("Error description is required");
    logSystemError(validationError, {
      route: "/generate-error-handler",
      method: "POST",
    });
    return res.status(400).json({ error: "Error description is required" });
  }

  try {
    console.log("Starting code generation with streaming...");
    const generatedCode = await generateErrorHandlingCode(errorDescription);

    res.json({
      success: true,
      generatedCode,
      message: "Error handling code generated successfully",
    });
  } catch (error) {
    logSystemError(error, {
      route: "/generate-error-handler",
      method: "POST",
      errorDescription,
    });
    res.status(500).json({ error: "Failed to generate error handling code" });
  }
});

// Global error handler
app.use((err, req, res, next) => {
  logSystemError(err, {
    route: req.path,
    method: req.method,
    body: req.body,
  });

  res.status(500).json({
    error: "Internal server error",
    message:
      process.env.NODE_ENV === "development"
        ? err.message
        : "An error occurred",
  });
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(
    JSON.stringify({
      timestamp: new Date().toISOString(),
      level: "INFO",
      message: `Server started on port ${PORT}`,
      environment: process.env.NODE_ENV || "production",
    })
  );
});

// Demonstrate the error logging functionality
async function demonstrateErrorLogging() {
  console.log("=== Demonstrating Error Logging System ===\n");

  // Example 1: Log a simple error
  const simpleError = new Error("Database connection failed");
  logSystemError(simpleError, {
    database: "PostgreSQL",
    host: "localhost",
    port: 5432,
  });

  console.log("\n=== Generating Error Handler with Claude API ===\n");

  // Example 2: Generate error handling code using Claude
  try {
    await generateErrorHandlingCode(
      "Database connection timeout with retry logic"
    );
  } catch (error) {
    logSystemError(error, {
      function: "demonstrateErrorLogging",
      step: "generateErrorHandlingCode",
    });
  }
}

// Run demonstration
demonstrateErrorLogging().catch((error) => {
  logSystemError(error, { function: "main" });
  process.exit(1);
});

export { logSystemError, generateErrorHandlingCode };
