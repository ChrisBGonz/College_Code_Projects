import anthropic

client = anthropic.Anthropic()

# Define the tool for password hashing
tools = [
    {
        "name": "hash_password",
        "description": "Hash a user password securely before storing in database using bcrypt",
        "input_schema": {
            "type": "object",
            "properties": {
                "password": {
                    "type": "string",
                    "description": "The plain text password to hash"
                },
                "salt_rounds": {
                    "type": "integer",
                    "description": "Number of salt rounds for bcrypt (default: 12)",
                    "default": 12
                }
            },
            "required": ["password"]
        }
    }
]

def hash_password_tool(password: str, salt_rounds: int = 12) -> dict:
    """
    Actual implementation of password hashing using bcrypt
    """
    import bcrypt
    
    # Convert password to bytes if it's a string
    password_bytes = password.encode('utf-8')
    
    # Generate salt and hash the password
    salt = bcrypt.gensalt(rounds=salt_rounds)
    hashed = bcrypt.hashpw(password_bytes, salt)
    
    return {
        "hashed_password": hashed.decode('utf-8'),
        "algorithm": "bcrypt",
        "salt_rounds": salt_rounds,
        "success": True
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "hash_password":
        result = hash_password_tool(
            password=tool_input["password"],
            salt_rounds=tool_input.get("salt_rounds", 12)
        )
        return str(result)
    return "Tool not found"

def hash_password_with_claude(password: str) -> str:
    """
    Use Claude to hash a password using tool use
    """
    messages = [
        {
            "role": "user",
            "content": f"Please hash this password securely before storing it in the database: '{password}'. Use the hash_password tool to do this."
        }
    ]
    
    # Initial API call
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process tool calls in an agentic loop
    while response.stop_reason == "tool_use":
        # Find tool use blocks
        tool_use_blocks = [block for block in response.content if block.type == "tool_use"]
        
        if not tool_use_blocks:
            break
            
        # Process each tool call
        tool_results = []
        for tool_use in tool_use_blocks:
            tool_result = process_tool_call(tool_use.name, tool_use.input)
            tool_results.append({
                "type": "tool_result",
                "tool_use_id": tool_use.id,
                "content": tool_result
            })
        
        # Continue conversation with tool results
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": tool_results})
        
        # Get next response
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # Extract final text response
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response += block.text
    
    return final_response

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """
    Verify a password against its hash
    """
    import bcrypt
    return bcrypt.checkpw(
        plain_password.encode('utf-8'),
        hashed_password.encode('utf-8')
    )

# Main execution
if __name__ == "__main__":
    # Example usage
    test_password = "MySecurePassword123!"
    
    print("=== Password Hashing with Claude ===\n")
    print(f"Original password: {test_password}")
    print("\nAsking Claude to hash the password...\n")
    
    # Get Claude's response
    claude_response = hash_password_with_claude(test_password)
    print("Claude's response:")
    print(claude_response)
    
    # Also demonstrate direct hashing
    print("\n=== Direct Password Hashing Demo ===\n")
    result = hash_password_tool(test_password)
    print(f"Hashed password: {result['hashed_password']}")
    print(f"Algorithm: {result['algorithm']}")
    print(f"Salt rounds: {result['salt_rounds']}")
    
    # Verify the password
    is_valid = verify_password(test_password, result['hashed_password'])
    print(f"\nPassword verification: {'✓ Valid' if is_valid else '✗ Invalid'}")
    
    # Test with wrong password
    is_invalid = verify_password("WrongPassword", result['hashed_password'])
    print(f"Wrong password verification: {'✓ Valid' if is_invalid else '✗ Invalid'}")
