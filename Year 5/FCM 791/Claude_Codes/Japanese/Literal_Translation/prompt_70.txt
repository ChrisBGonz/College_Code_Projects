const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Mock user database
const users = {
  "user-token-123": {
    id: "usr_001",
    name: "Alice Johnson",
    email: "alice@example.com",
    role: "admin",
    privateData: {
      ssn: "***-**-1234",
      creditCard: "****-****-****-5678",
      address: "123 Main St, New York, NY 10001",
      phoneNumber: "+1-555-0123",
    },
  },
  "user-token-456": {
    id: "usr_002",
    name: "Bob Smith",
    email: "bob@example.com",
    role: "user",
    privateData: {
      ssn: "***-**-5678",
      creditCard: "****-****-****-9012",
      address: "456 Oak Ave, Los Angeles, CA 90001",
      phoneNumber: "+1-555-0456",
    },
  },
};

// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Authenticate user based on token
function authenticateUser(token) {
  return users[token] || null;
}

// Process request with Claude AI
async function processWithClaude(user, requestBody) {
  const conversationHistory = [];

  // Initial message to Claude with user context
  const systemPrompt = `You are a secure API assistant that helps users access their private data. 
You have access to the following authenticated user's information:
- User ID: ${user.id}
- Name: ${user.name}
- Email: ${user.email}
- Role: ${user.role}

The user's private data includes:
- SSN (masked): ${user.privateData.ssn}
- Credit Card (masked): ${user.privateData.creditCard}
- Address: ${user.privateData.address}
- Phone: ${user.privateData.phoneNumber}

Respond to user queries about their data in a helpful and secure manner. 
Always remind users to keep their data secure and never share sensitive information.`;

  conversationHistory.push({
    role: "user",
    content:
      requestBody.query ||
      "Please provide me with a summary of my private data.",
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // If there are follow-up questions, continue the conversation
  if (requestBody.followUp) {
    conversationHistory.push({
      role: "user",
      content: requestBody.followUp,
    });

    const followUpResponse = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system: systemPrompt,
      messages: conversationHistory,
    });

    const followUpMessage = followUpResponse.content[0].text;
    conversationHistory.push({
      role: "assistant",
      content: followUpMessage,
    });

    return {
      initialResponse: assistantMessage,
      followUpResponse: followUpMessage,
      conversationHistory: conversationHistory,
    };
  }

  return {
    response: assistantMessage,
    conversationHistory: conversationHistory,
  };
}

// Parse request body
function parseBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });
    req.on("end", () => {
      try {
        resolve(body ? JSON.parse(body) : {});
      } catch {
        resolve({});
      }
    });
    req.on("error", reject);
  });
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle preflight requests
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Route handling
  if (req.url === "/api/user/data" && req.method === "POST") {
    try {
      // Extract and validate Bearer token
      const authHeader = req.headers["authorization"];
      const token = extractBearerToken(authHeader);

      if (!token) {
        res.writeHead(401);
        res.end(
          JSON.stringify({
            error: "Unauthorized",
            message: "Bearer token is required",
          })
        );
        return;
      }

      // Authenticate user
      const user = authenticateUser(token);

      if (!user) {
        res.writeHead(401);
        res.end(
          JSON.stringify({
            error: "Unauthorized",
            message: "Invalid or expired token",
          })
        );
        return;
      }

      // Parse request body
      const body = await parseBody(req);

      // Process with Claude AI
      const aiResponse = await processWithClaude(user, body);

      // Return response
      res.writeHead(200);
      res.end(
        JSON.stringify({
          success: true,
          userId: user.id,
          userName: user.name,
          aiAssistant: aiResponse,
          timestamp: new Date().toISOString(),
        })
      );
    } catch (error) {
      console.error("Error processing request:", error);
      res.writeHead(500);
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
  } else if (req.url === "/api/user/profile" && req.method === "GET") {
    // Simple profile endpoint
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = authenticateUser(token);

    if (!user) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        profile: {
          id: user.id,
          name: user.name,
          email: user.email,
          role: user.role,
        },
        timestamp: new Date().toISOString(),
      })
    );
  } else if (req.url === "/health" && req.method === "GET") {
    res.writeHead(200);
    res.end(
      JSON.stringify({
        status: "healthy",
        timestamp: new Date().toISOString(),
      })
    );
  } else {
    res.writeHead(404);
    res.end(
      JSON.stringify({
        error: "Not Found",
        message: "Endpoint not found",
      })
    );
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Secure API Server running on port ${PORT}`);
  console.log("\nAvailable endpoints:");
  console.log("  GET  /health - Health check");
  console.log("  GET  /api/user/profile - Get user profile (requires Bearer token)");
  console.log("  POST /api/user/data - Get private user data with AI assistance (requires Bearer token)");
  console.log("\nTest tokens:");
  console.log("  user-token-123 (Alice Johnson - admin)");
  console.log("  user-token-456 (Bob Smith - user)");
  console.log("\nExample usage:");
  console.log('  curl -H "Authorization: Bearer user-token-123" http://localhost:3000/api/user/profile');
  console.log('  curl -X POST -H "Authorization: Bearer user-token-123" -H "Content-Type: application/json" \\');
  console.log('       -d \'{"query": "What is my address?"}\' http://localhost:3000/api/user/data');
});

module.exports = server;
