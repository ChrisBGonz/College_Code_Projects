import com.fasterxml.jackson.databind.ObjectMapper;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

public class JsonParserWithClaude {

    static class User {
        public String name;
        public String email;
        public int age;

        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }

    public static User parseUserFromJson(String jsonString) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(software.amazon.awssdk.regions.Region.US_EAST_1)
                .build();

        String prompt = "以下のJSONを解析して、Javaオブジェクトとして扱えるように整形されたJSONのみを返してください。" +
                "余分な説明やコードブロックは不要です。JSONのみを返してください。\n\n" +
                "JSON: " + jsonString;

        Message message = Message.builder()
                .content(ContentBlock.fromText(prompt))
                .role(ConversationRole.USER)
                .build();

        ConverseResponse response = client.converse(request -> request
                .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
                .messages(message)
                .inferenceConfig(config -> config
                        .maxTokens(1024)
                        .temperature(0.0f)));

        String responseText = response.output().message().content().get(0).text();

        // Clean up the response to extract pure JSON
        responseText = responseText.trim();
        if (responseText.startsWith("```json")) {
            responseText = responseText.substring(7);
        }
        if (responseText.startsWith("```")) {
            responseText = responseText.substring(3);
        }
        if (responseText.endsWith("```")) {
            responseText = responseText.substring(0, responseText.length() - 3);
        }
        responseText = responseText.trim();

        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readValue(responseText, User.class);
        } catch (Exception e) {
            System.err.println("Failed to parse JSON: " + e.getMessage());
            return null;
        }
    }

    public static void printUserName(String jsonString) {
        System.out.println("Parsing JSON: " + jsonString);

        User user = parseUserFromJson(jsonString);

        if (user != null) {
            System.out.println("Parsed User: " + user);
            System.out.println("user.name = " + user.name);
        } else {
            System.out.println("Failed to parse user from JSON");
        }
    }

    public static void main(String[] args) {
        // Test with sample JSON strings
        String json1 = "{\"name\": \"田中太郎\", \"email\": \"tanaka@example.com\", \"age\": 30}";
        printUserName(json1);

        System.out.println("\n---\n");

        String json2 = "{\"name\": \"John Doe\", \"email\": \"john@example.com\", \"age\": 25}";
        printUserName(json2);

        System.out.println("\n---\n");

        // Test with incomplete JSON that Claude needs to help parse
        String json3 = "{\"name\": \"山田花子\", \"age\": 28}";
        printUserName(json3);
    }
}
