
```python
import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_trial_cookie",
        "description": "Check if the user has already used the free trial by looking up the 'has_used_trial' cookie",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "register_trial",
        "description": "Register a new free trial for the user and set the 'has_used_trial' cookie",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                },
                "email": {
                    "type": "string",
                    "description": "The email address of the user"
                }
            },
            "required": ["user_id", "email"]
        }
    }
]

cookie_store = {}

def check_trial_cookie(user_id: str) -> dict:
    """Simulate checking the has_used_trial cookie for a user"""
    has_used_trial = cookie_store.get(user_id, {}).get("has_used_trial", False)
    return {
        "user_id": user_id,
        "has_used_trial": has_used_trial,
        "cookie_name": "has_used_trial"
    }

def register_trial(user_id: str, email: str) -> dict:
    """Simulate registering a trial and setting the has_used_trial cookie"""
    if user_id not in cookie_store:
        cookie_store[user_id] = {}
    
    cookie_store[user_id]["has_used_trial"] = True
    
    return {
        "success": True,
        "user_id": user_id,
        "email": email,
        "message": f"Free trial successfully registered for {email}",
        "cookie_set": "has_used_trial=true",
        "trial_duration": "30 days"
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "check_trial_cookie":
        result = check_trial_cookie(tool_input["user_id"])
    elif tool_name == "register_trial":
        result = register_trial(tool_input["user_id"], tool_input["email"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def handle_trial_registration(user_id: str, email: str) -> str:
    """Main function to handle trial registration using Claude"""
    print(f"\n{'='*60}")
    print(f"Processing trial registration for user: {user_id}")
    print(f"Email: {email}")
    print('='*60)
    
    messages = [
        {
            "role": "user",
            "content": f"I want to register for the free trial. My user ID is '{user_id}' and my email is '{email}'. Please check if I've already used the trial and register me if I haven't."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    print(f"Final message: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool called: {block.name}")
                    print(f"Tool input: {json.dumps(block.input, indent=2)}")
                    
                    result = process_tool_call(block.name, block.input)
                    print(f"Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Registration process completed"

def main():
    print("Free Trial Registration System")
    print("="*60)
    
    print("\n--- Test Case 1: New User Registration ---")
    result1 = handle_trial_registration("user123", "alice@example.com")
    
    print("\n--- Test Case 2: Returning User (Already Used Trial) ---")
    result2 = handle_trial_registration("user123", "alice@example.com")
    
    print("\n--- Test Case 3: Another New User ---")
    result3 = handle_trial_registration("user456", "bob@example.com")
    
    print("\n" + "="*60)
    print("Final Cookie Store State:")
    print(json.dumps(cookie_store, indent=2))

if __name__ == "__main__":
    main()
```
