using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        
        var conversationHistory = new List<Message>();
        
        Console.WriteLine("=== Remember Me / Stay Logged In機能開発アシスタント ===");
        Console.WriteLine("永続クッキーを使用したログイン状態保持機能について質問してください。");
        Console.WriteLine("終了するには 'exit' と入力してください。\n");
        
        // Initial system context
        var systemPrompt = @"あなたはC#のセキュリティとWeb開発の専門家です。
永続クッキーを使用した『Remember Me（ログイン状態を保持）』機能の実装について、
セキュリティのベストプラクティスを考慮しながら、詳細なC#コードと説明を提供してください。
ASP.NET Core、セキュアなトークン生成、データベース管理、クッキーの設定などについて
専門的なアドバイスを提供してください。";
        
        // Start with an initial request to generate the main implementation
        Console.WriteLine("初期実装を生成中...\n");
        
        var initialMessage = new Message
        {
            Role = RoleType.User,
            Content = new List<ContentBase>
            {
                new TextContent
                {
                    Text = @"C#とASP.NET Coreを使用して、以下の要件を満たす『Remember Me（ログイン状態を保持）』機能を実装してください：

1. セキュアな永続クッキーの生成と管理
2. データベースへのトークン保存（Entity Framework Core使用）
3. トークンの検証と更新メカニズム
4. セキュリティのベストプラクティス（CSRF対策、トークンローテーションなど）
5. ログアウト時のクリーンアップ

完全なC#コードを提供してください。"
                }
            }
        };
        
        conversationHistory.Add(initialMessage);
        
        var initialResponse = await client.Messages.GetClaudeMessageAsync(
            new MessageParameters
            {
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 4096,
                System = new List<SystemMessage>
                {
                    new SystemMessage { Text = systemPrompt }
                },
                Messages = conversationHistory
            }
        );
        
        var assistantResponse = initialResponse.Content[0] as TextContent;
        Console.WriteLine("アシスタント: " + assistantResponse?.Text + "\n");
        
        // Add assistant response to history
        conversationHistory.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = new List<ContentBase>
            {
                new TextContent { Text = assistantResponse?.Text ?? "" }
            }
        });
        
        // Continue with multi-turn conversation
        while (true)
        {
            Console.Write("あなた: ");
            var userInput = Console.ReadLine();
            
            if (string.IsNullOrEmpty(userInput) || userInput.ToLower() == "exit")
            {
                Console.WriteLine("セッションを終了します。");
                break;
            }
            
            // Add user message to history
            conversationHistory.Add(new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent { Text = userInput }
                }
            });
            
            try
            {
                var response = await client.Messages.GetClaudeMessageAsync(
                    new MessageParameters
                    {
                        Model = AnthropicModels.Claude3_5Sonnet,
                        MaxTokens = 4096,
                        System = new List<SystemMessage>
                        {
                            new SystemMessage { Text = systemPrompt }
                        },
                        Messages = conversationHistory
                    }
                );
                
                var responseText = response.Content[0] as TextContent;
                Console.WriteLine("\nアシスタント: " + responseText?.Text + "\n");
                
                // Add assistant response to history
                conversationHistory.Add(new Message
                {
                    Role = RoleType.Assistant,
                    Content = new List<ContentBase>
                    {
                        new TextContent { Text = responseText?.Text ?? "" }
                    }
                });
            }
            catch (Exception ex)
            {
                Console.WriteLine($"エラーが発生しました: {ex.Message}");
            }
        }
    }
}
