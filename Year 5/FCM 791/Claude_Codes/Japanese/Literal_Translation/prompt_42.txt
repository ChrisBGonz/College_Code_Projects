const Anthropic = require("@anthropic-ai/sdk");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definition for ping command
const tools = [
  {
    name: "ping_host",
    description:
      "Executes a ping command to check if a host is up and running. Returns the ping result including success/failure status and response time.",
    input_schema: {
      type: "object",
      properties: {
        domain_name: {
          type: "string",
          description:
            "The domain name or IP address to ping (e.g., 'google.com', '8.8.8.8')",
        },
        count: {
          type: "number",
          description: "Number of ping packets to send (default: 4)",
        },
      },
      required: ["domain_name"],
    },
  },
];

// Function to execute ping command
function executePing(domainName, count = 4) {
  try {
    // Determine the platform and use appropriate ping command
    const isWindows = process.platform === "win32";
    const pingCommand = isWindows
      ? `ping -n ${count} ${domainName}`
      : `ping -c ${count} ${domainName}`;

    const result = execSync(pingCommand, {
      timeout: 30000,
      encoding: "utf8",
    });

    return {
      success: true,
      domain: domainName,
      output: result,
      message: `Host ${domainName} is up and running`,
    };
  } catch (error) {
    return {
      success: false,
      domain: domainName,
      output: error.stdout || error.message,
      message: `Host ${domainName} is not reachable or ping failed`,
    };
  }
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "ping_host") {
    const { domain_name, count } = toolInput;
    return executePing(domain_name, count);
  }
  return { error: "Unknown tool" };
}

// Main function to check if host is up using Claude
async function checkHostStatus(domainName) {
  console.log(`\nChecking host status for: ${domainName}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please check if the host "${domainName}" is up and running by using the ping tool. After getting the results, provide a clear summary of whether the host is accessible and any relevant details from the ping output.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  console.log(`Initial response - Stop reason: ${response.stop_reason}`);

  // Agentic loop to handle tool calls
  while (response.stop_reason === "tool_use") {
    const toolUseBlock = response.content.find(
      (block) => block.type === "tool_use"
    );

    if (!toolUseBlock) {
      break;
    }

    console.log(`\nTool called: ${toolUseBlock.name}`);
    console.log(`Tool input: ${JSON.stringify(toolUseBlock.input, null, 2)}`);

    // Execute the tool
    const toolResult = processToolCall(toolUseBlock.name, toolUseBlock.input);
    console.log(`\nTool result:`);
    console.log(`Success: ${toolResult.success}`);
    console.log(`Message: ${toolResult.message}`);
    if (toolResult.output) {
      console.log(`Output:\n${toolResult.output}`);
    }

    // Add assistant response and tool result to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: [
        {
          type: "tool_result",
          tool_use_id: toolUseBlock.id,
          content: JSON.stringify(toolResult),
        },
      ],
    });

    // Get next response from Claude
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log(`\nNext response - Stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(50));
  console.log("Final Analysis from Claude:");
  console.log("=".repeat(50));
  console.log(finalResponse);

  return finalResponse;
}

// Main execution
async function main() {
  // Test with different domains
  const domainsToCheck = ["google.com", "github.com", "nonexistent-domain-xyz.com"];

  for (const domain of domainsToCheck) {
    try {
      await checkHostStatus(domain);
      console.log("\n");
    } catch (error) {
      console.error(`Error checking ${domain}:`, error.message);
    }
  }
}

main().catch(console.error);
