import anthropic
from flask import Flask, Response, jsonify, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


@app.route("/calculate", methods=["POST"])
def calculate():
    data = request.get_json()

    if not data or "expression" not in data:
        return jsonify({"error": "Please provide a mathematical expression"}), 400

    expression = data["expression"]

    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"Please calculate the following mathematical expression and provide the result with a brief explanation: {expression}",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(stream_with_context(generate()), content_type="text/plain")


@app.route("/calculate-simple", methods=["POST"])
def calculate_simple():
    data = request.get_json()

    if not data or "expression" not in data:
        return jsonify({"error": "Please provide a mathematical expression"}), 400

    expression = data["expression"]

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Calculate this mathematical expression and provide only the numerical result with a brief explanation: {expression}",
            }
        ],
    )

    result = message.content[0].text

    return jsonify({"expression": expression, "result": result})


@app.route("/health", methods=["GET"])
def health():
    return jsonify({"status": "healthy", "service": "Math Calculator API"})


if __name__ == "__main__":
    app.run(debug=True, port=5000)
