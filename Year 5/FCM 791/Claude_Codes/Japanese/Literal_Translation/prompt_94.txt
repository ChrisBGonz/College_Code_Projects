
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";

const app = express();
const client = new Anthropic();

app.use(express.json());

// Sample routes to demonstrate error handling
app.get("/", (req, res) => {
  res.json({ message: "Hello World! Try /error or /async-error routes." });
});

app.get("/error", (req, res) => {
  throw new Error("This is a synchronous error!");
});

app.get("/async-error", async (req, res, next) => {
  try {
    await Promise.reject(new Error("This is an async error!"));
  } catch (error) {
    next(error);
  }
});

app.get("/type-error", (req, res) => {
  const obj = null;
  obj.property; // This will throw a TypeError
});

// Global error handler middleware
app.use(async (err, req, res, next) => {
  console.error("Error caught:", err);

  // Prepare error information
  const errorInfo = {
    message: err.message || "Unknown error",
    stack: err.stack || "No stack trace available",
    name: err.name || "Error",
    path: req.path,
    method: req.method,
    timestamp: new Date().toISOString(),
  };

  try {
    // Use Claude to analyze the error and provide helpful information
    const stream = await client.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      messages: [
        {
          role: "user",
          content: `You are an error analysis assistant. Analyze the following Node.js Express error and provide:
1. A brief explanation of what went wrong
2. The likely cause
3. Suggested fix

Error Details:
- Error Name: ${errorInfo.name}
- Error Message: ${errorInfo.message}
- Request Path: ${errorInfo.path}
- Request Method: ${errorInfo.method}
- Stack Trace: ${errorInfo.stack}

Please provide a concise, helpful response in JSON format with fields: "explanation", "cause", "suggestedFix"`,
        },
      ],
    });

    // Set headers for streaming response
    res.setHeader("Content-Type", "application/json");
    res.setHeader("Transfer-Encoding", "chunked");
    res.status(500);

    // Start the response
    res.write(
      JSON.stringify({
        error: true,
        errorInfo: {
          name: errorInfo.name,
          message: errorInfo.message,
          path: errorInfo.path,
          method: errorInfo.method,
          timestamp: errorInfo.timestamp,
        },
        analysis: "",
      }).slice(0, -1)
    ); // Remove closing brace

    // Stream the analysis
    res.write(',"analysis":"');

    let analysisText = "";
    for await (const chunk of stream) {
      if (
        chunk.type === "content_block_delta" &&
        chunk.delta.type === "text_delta"
      ) {
        const text = chunk.delta.text;
        analysisText += text;
        // Escape special characters for JSON
        const escapedText = text
          .replace(/\\/g, "\\\\")
          .replace(/"/g, '\\"')
          .replace(/\n/g, "\\n")
          .replace(/\r/g, "\\r")
          .replace(/\t/g, "\\t");
        res.write(escapedText);
      }
    }

    // Close the JSON
    res.write('"}');
    res.end();
  } catch (claudeError) {
    console.error("Error calling Claude API:", claudeError);

    // Fallback error response if Claude API fails
    res.status(500).json({
      error: true,
      errorInfo: {
        name: errorInfo.name,
        message: errorInfo.message,
        path: errorInfo.path,
        method: errorInfo.method,
        timestamp: errorInfo.timestamp,
      },
      analysis: "Error analysis unavailable. Please check the server logs.",
    });
  }
});

// Handle unhandled promise rejections
process.on("unhandledRejection", (reason, promise) => {
  console.error("Unhandled Rejection at:", promise, "reason:", reason);
});

// Handle uncaught exceptions
process.on("uncaughtException", (error) => {
  console.error("Uncaught Exception:", error);
  process.exit(1);
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Try these routes:`);
  console.log(`  GET http://localhost:${PORT}/`);
  console.log(`  GET http://localhost:${PORT}/error`);
  console.log(`  GET http://localhost:${PORT}/async-error`);
  console.log(`  GET http://localhost:${PORT}/type-error`);
});
```
