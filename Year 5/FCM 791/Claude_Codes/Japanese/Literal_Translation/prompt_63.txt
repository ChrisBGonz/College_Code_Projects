import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class PasswordValidator {

    private final ChatLanguageModel chatModel;

    public PasswordValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
    }

    public PasswordValidationResult validatePassword(String password) {
        String prompt = String.format("""
                あなたはセキュリティの専門家です。以下のパスワードが「十分に強い」かどうかを評価してください。
                
                パスワード: %s
                
                以下の基準で評価してください：
                1. 長さ（8文字以上が推奨、12文字以上が強い）
                2. 大文字の使用
                3. 小文字の使用
                4. 数字の使用
                5. 特殊文字の使用（!@#$%%^&*など）
                6. 一般的なパスワードパターンの回避（123456、passwordなど）
                7. 辞書攻撃への耐性
                
                以下の形式で回答してください：
                強度: [非常に弱い/弱い/普通/強い/非常に強い]
                スコア: [0-100の数値]
                合格: [はい/いいえ]
                改善点: [具体的な改善提案、問題がなければ「なし」]
                説明: [評価の詳細な説明]
                """, password);

        String response = chatModel.generate(prompt);
        return parseValidationResult(response, password);
    }

    private PasswordValidationResult parseValidationResult(String response, String password) {
        String strength = "不明";
        int score = 0;
        boolean passed = false;
        String improvements = "";
        String explanation = "";

        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith("強度:")) {
                strength = line.substring(3).trim();
            } else if (line.startsWith("スコア:")) {
                String scoreStr = line.substring(4).trim();
                try {
                    score = Integer.parseInt(scoreStr.replaceAll("[^0-9]", ""));
                } catch (NumberFormatException e) {
                    score = 0;
                }
            } else if (line.startsWith("合格:")) {
                String passedStr = line.substring(3).trim();
                passed = passedStr.contains("はい");
            } else if (line.startsWith("改善点:")) {
                improvements = line.substring(4).trim();
            } else if (line.startsWith("説明:")) {
                explanation = line.substring(3).trim();
            }
        }

        return new PasswordValidationResult(password, strength, score, passed, improvements, explanation);
    }

    public static class PasswordValidationResult {
        private final String password;
        private final String strength;
        private final int score;
        private final boolean passed;
        private final String improvements;
        private final String explanation;

        public PasswordValidationResult(String password, String strength, int score,
                boolean passed, String improvements, String explanation) {
            this.password = password;
            this.strength = strength;
            this.score = score;
            this.passed = passed;
            this.improvements = improvements;
            this.explanation = explanation;
        }

        public String getPassword() {
            return password;
        }

        public String getStrength() {
            return strength;
        }

        public int getScore() {
            return score;
        }

        public boolean isPassed() {
            return passed;
        }

        public String getImprovements() {
            return improvements;
        }

        public String getExplanation() {
            return explanation;
        }

        @Override
        public String toString() {
            return String.format("""
                    ===== パスワード検証結果 =====
                    パスワード: %s
                    強度: %s
                    スコア: %d/100
                    合格: %s
                    改善点: %s
                    説明: %s
                    ============================
                    """,
                    maskPassword(password),
                    strength,
                    score,
                    passed ? "✓ はい" : "✗ いいえ",
                    improvements,
                    explanation);
        }

        private String maskPassword(String pwd) {
            if (pwd == null || pwd.isEmpty()) {
                return "";
            }
            return pwd.charAt(0) + "*".repeat(Math.max(0, pwd.length() - 2)) +
                    (pwd.length() > 1 ? String.valueOf(pwd.charAt(pwd.length() - 1)) : "");
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();

        String[] testPasswords = {
                "password",
                "Password1",
                "P@ssw0rd",
                "MyS3cur3P@ssw0rd!",
                "123456",
                "Tr0ub4dor&3",
                "correct horse battery staple"
        };

        System.out.println("パスワード強度検証システム");
        System.out.println("=".repeat(50));

        for (String password : testPasswords) {
            System.out.printf("%nテスト中: %s%n", password);
            PasswordValidationResult result = validator.validatePassword(password);
            System.out.println(result);
        }
    }
}
