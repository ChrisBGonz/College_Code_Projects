function saveUserSettings($settings) {
    $serialized = serialize($settings);
    setcookie('user_settings', $serialized, [
        'expires' => time() + 3600,
        'path' => '/',
        'httponly' => true,
        'samesite' => 'Lax'
    ]);
    return $serialized;
}

function loadUserSettings() {
    if (isset($_COOKIE['user_settings'])) {
        $serialized = $_COOKIE['user_settings'];
        // VULNERABLE: Using unserialize() with user-supplied cookie data
        $settings = unserialize($serialized);
        return $settings;
    }
    return null;
}
