@app.route('/upload', methods=['POST'])
def upload_file():
    global user_profile, conversation_history
    
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'ファイルが見つかりません'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'ファイルが選択されていません'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'YAMLファイル(.yaml または .yml)のみアップロード可能です'})
    
    try:
        content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'YAMLファイルは辞書形式である必要があります'})
        
        user_profile = profile_data
        
        # Add upload event to conversation history
        profile_summary = yaml.dump(user_profile, allow_unicode=True, default_flow_style=False)
        conversation_history.append({
            "role": "user",
            "content": f"YAMLファイルをアップロードしました。プロフィール内容:\n```yaml\n{profile_summary}\n```"
        })
        
        # Get AI response about the uploaded profile
        system = SYSTEM_PROMPT.format(profile=profile_summary)
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system,
            messages=conversation_history
        )
        
        ai_message = response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": ai_message
        })
        
        return jsonify({
            'success': True,
            'profile': user_profile,
            'message': ai_message
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'YAMLの解析エラー: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'エラーが発生しました: {str(e)}'})


@app.route('/chat', methods=['POST'])
def chat():
    global user_profile, conversation_history
    
    data = request.json
    user_message = data.get('message', '')
    
    if not user_message:
        return jsonify({'error': 'メッセージが空です'})
    
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    profile_summary = yaml.dump(user_profile, allow_unicode=True, default_flow_style=False) if user_profile else "プロフィールが設定されていません"
    system = SYSTEM_PROMPT.format(profile=profile_summary)
    
    try:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system,
            messages=conversation_history
        )
        
        ai_message = response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": ai_message
        })
        
        # Check if the AI suggested profile updates
        updated_profile = None
        if '```yaml' in ai_message:
            try:
                yaml_start = ai_message.find('```yaml') + 7
                yaml_end = ai_message.find('```', yaml_start)
                yaml_content = ai_message[yaml_start:yaml_end].strip()
                parsed_profile = yaml.safe_load(yaml_content)
                if isinstance(parsed_profile, dict):
                    user_profile = parsed_profile
                    updated_profile = user_profile
            except:
                pass
        
        return jsonify({
            'response': ai_message,
            'profile': updated_profile
        })
        
    except Exception as e:
        return jsonify({'error': f'エラーが発生しました: {str(e)}'})

    global user_profile, conversation_history
    user_profile = {}
    conversation_history = []
    return jsonify({'success': True, 'message': 'プロフィールと会話履歴がリセットされました'})
