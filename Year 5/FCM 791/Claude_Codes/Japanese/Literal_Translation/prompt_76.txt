import anthropic
import json

client = anthropic.Anthropic()

def run_cicd_pipeline(image_name: str, tag: str = "latest", environment: str = "production"):
    """
    Run a CI/CD pipeline using Claude as an AI assistant with tool use.
    
    Args:
        image_name: Docker image name to pull and deploy
        tag: Docker image tag
        environment: Target deployment environment
    """
    
    tools = [
        {
            "name": "pull_docker_image",
            "description": "Pull a Docker image from a registry",
            "input_schema": {
                "type": "object",
                "properties": {
                    "image_name": {
                        "type": "string",
                        "description": "The name of the Docker image to pull"
                    },
                    "tag": {
                        "type": "string",
                        "description": "The tag of the Docker image",
                        "default": "latest"
                    }
                },
                "required": ["image_name"]
            }
        },
        {
            "name": "run_tests",
            "description": "Run tests on the Docker image",
            "input_schema": {
                "type": "object",
                "properties": {
                    "image_name": {
                        "type": "string",
                        "description": "The Docker image to test"
                    },
                    "test_suite": {
                        "type": "string",
                        "description": "The test suite to run",
                        "enum": ["unit", "integration", "smoke"]
                    }
                },
                "required": ["image_name", "test_suite"]
            }
        },
        {
            "name": "deploy_to_environment",
            "description": "Deploy the Docker image to a specified environment",
            "input_schema": {
                "type": "object",
                "properties": {
                    "image_name": {
                        "type": "string",
                        "description": "The Docker image to deploy"
                    },
                    "environment": {
                        "type": "string",
                        "description": "The target environment",
                        "enum": ["staging", "production", "development"]
                    },
                    "replicas": {
                        "type": "integer",
                        "description": "Number of replicas to deploy",
                        "default": 3
                    }
                },
                "required": ["image_name", "environment"]
            }
        },
        {
            "name": "check_deployment_health",
            "description": "Check the health status of a deployment",
            "input_schema": {
                "type": "object",
                "properties": {
                    "deployment_id": {
                        "type": "string",
                        "description": "The ID of the deployment to check"
                    },
                    "environment": {
                        "type": "string",
                        "description": "The environment to check"
                    }
                },
                "required": ["deployment_id", "environment"]
            }
        },
        {
            "name": "rollback_deployment",
            "description": "Rollback a deployment to the previous version",
            "input_schema": {
                "type": "object",
                "properties": {
                    "deployment_id": {
                        "type": "string",
                        "description": "The ID of the deployment to rollback"
                    },
                    "environment": {
                        "type": "string",
                        "description": "The environment to rollback"
                    }
                },
                "required": ["deployment_id", "environment"]
            }
        }
    ]
    
    def simulate_tool_execution(tool_name: str, tool_input: dict) -> dict:
        """Simulate tool execution with mock responses"""
        
        if tool_name == "pull_docker_image":
            image = tool_input.get("image_name", "")
            tag = tool_input.get("tag", "latest")
            print(f"  📥 Pulling Docker image: {image}:{tag}")
            return {
                "success": True,
                "image": f"{image}:{tag}",
                "size": "245MB",
                "digest": "sha256:abc123def456",
                "message": f"Successfully pulled {image}:{tag}"
            }
        
        elif tool_name == "run_tests":
            image = tool_input.get("image_name", "")
            test_suite = tool_input.get("test_suite", "unit")
            print(f"  🧪 Running {test_suite} tests on {image}")
            return {
                "success": True,
                "test_suite": test_suite,
                "passed": 47,
                "failed": 0,
                "skipped": 2,
                "duration": "2m 34s",
                "message": f"All {test_suite} tests passed successfully"
            }
        
        elif tool_name == "deploy_to_environment":
            image = tool_input.get("image_name", "")
            env = tool_input.get("environment", "production")
            replicas = tool_input.get("replicas", 3)
            print(f"  🚀 Deploying {image} to {env} with {replicas} replicas")
            return {
                "success": True,
                "deployment_id": "deploy-prod-20241201-001",
                "environment": env,
                "replicas": replicas,
                "status": "running",
                "endpoint": f"https://api.{env}.example.com",
                "message": f"Successfully deployed to {env}"
            }
        
        elif tool_name == "check_deployment_health":
            deployment_id = tool_input.get("deployment_id", "")
            env = tool_input.get("environment", "")
            print(f"  🏥 Checking health of deployment {deployment_id} in {env}")
            return {
                "success": True,
                "deployment_id": deployment_id,
                "status": "healthy",
                "healthy_replicas": 3,
                "total_replicas": 3,
                "uptime": "99.9%",
                "response_time": "45ms",
                "message": "All replicas are healthy and responding"
            }
        
        elif tool_name == "rollback_deployment":
            deployment_id = tool_input.get("deployment_id", "")
            env = tool_input.get("environment", "")
            print(f"  ⏪ Rolling back deployment {deployment_id} in {env}")
            return {
                "success": True,
                "deployment_id": deployment_id,
                "previous_version": "v1.2.3",
                "message": f"Successfully rolled back to previous version"
            }
        
        return {"success": False, "message": "Unknown tool"}
    
    print(f"\n{'='*60}")
    print(f"🔄 Starting CI/CD Pipeline")
    print(f"   Image: {image_name}:{tag}")
    print(f"   Environment: {environment}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": f"""Execute a complete CI/CD pipeline for the following:
            - Docker Image: {image_name}
            - Tag: {tag}
            - Target Environment: {environment}
            
            Please:
            1. Pull the Docker image
            2. Run unit tests
            3. Run integration tests
            4. Deploy to {environment}
            5. Check deployment health
            6. Report the final status
            
            If any step fails, rollback the deployment and report the issue."""
        }
    ]
    
    pipeline_complete = False
    iteration = 0
    max_iterations = 10
    
    while not pipeline_complete and iteration < max_iterations:
        iteration += 1
        print(f"📋 Pipeline Step {iteration}:")
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "text":
                    print(f"  💬 Claude: {content_block.text}")
                elif content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"  🔧 Executing tool: {tool_name}")
                    
                    tool_result = simulate_tool_execution(tool_name, tool_input)
                    
                    print(f"  ✅ Tool result: {json.dumps(tool_result, indent=2)[:200]}...")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": json.dumps(tool_result)
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            pipeline_complete = True
            
            print(f"\n{'='*60}")
            print("📊 Pipeline Final Report:")
            print(f"{'='*60}")
            
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    print(content_block.text)
            
            print(f"\n✅ CI/CD Pipeline completed successfully!")
            
        else:
            print(f"⚠️  Unexpected stop reason: {response.stop_reason}")
            pipeline_complete = True
    
    if iteration >= max_iterations:
        print(f"\n⚠️  Pipeline reached maximum iterations ({max_iterations})")
    
    return response


def main():
    """Main function to demonstrate the CI/CD pipeline"""
    
    print("🐳 Docker CI/CD Pipeline with Claude AI")
    print("=" * 60)
    
    run_cicd_pipeline(
        image_name="myapp/backend",
        tag="v2.0.0",
        environment="production"
    )


if __name__ == "__main__":
    main()
