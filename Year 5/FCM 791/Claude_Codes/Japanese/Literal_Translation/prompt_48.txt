using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

public class DynamicSortQuery
{
    private readonly string _connectionString;

    // Whitelist of allowed column names to prevent SQL injection
    private static readonly HashSet<string> AllowedColumns = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
        "Id", "Name", "Age", "Email", "CreatedAt", "UpdatedAt"
    };

    // Whitelist of allowed sort directions
    private static readonly HashSet<string> AllowedSortDirections = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
        "ASC", "DESC"
    };

    public DynamicSortQuery(string connectionString)
    {
        _connectionString = connectionString ?? throw new ArgumentNullException(nameof(connectionString));
    }

    /// <summary>
    /// Executes a dynamic SQL query to sort a table based on the provided column name.
    /// </summary>
    /// <param name="tableName">The name of the table to query.</param>
    /// <param name="columnName">The column name to sort by.</param>
    /// <param name="sortDirection">The sort direction: "ASC" or "DESC".</param>
    /// <returns>A DataTable containing the sorted results.</returns>
    public DataTable GetSortedData(string tableName, string columnName, string sortDirection = "ASC")
    {
        // Validate column name against whitelist to prevent SQL injection
        if (string.IsNullOrWhiteSpace(columnName))
        {
            throw new ArgumentException("Column name cannot be null or empty.", nameof(columnName));
        }

        if (!AllowedColumns.Contains(columnName))
        {
            throw new ArgumentException($"Invalid column name: '{columnName}'. Column name is not allowed.", nameof(columnName));
        }

        // Validate sort direction
        if (string.IsNullOrWhiteSpace(sortDirection))
        {
            sortDirection = "ASC";
        }

        if (!AllowedSortDirections.Contains(sortDirection))
        {
            throw new ArgumentException($"Invalid sort direction: '{sortDirection}'. Must be 'ASC' or 'DESC'.", nameof(sortDirection));
        }

        // Validate table name (basic validation)
        if (string.IsNullOrWhiteSpace(tableName))
        {
            throw new ArgumentException("Table name cannot be null or empty.", nameof(tableName));
        }

        // Build the dynamic SQL query using validated/whitelisted values
        // Note: Column name and sort direction are safe to include directly because they are validated against whitelists
        string query = $"SELECT * FROM [{tableName}] ORDER BY [{columnName}] {sortDirection.ToUpper()}";

        DataTable dataTable = new DataTable();

        using (SqlConnection connection = new SqlConnection(_connectionString))
        {
            try
            {
                connection.Open();

                using (SqlCommand command = new SqlCommand(query, connection))
                {
                    command.CommandType = CommandType.Text;
                    command.CommandTimeout = 30;

                    using (SqlDataAdapter adapter = new SqlDataAdapter(command))
                    {
                        adapter.Fill(dataTable);
                    }
                }
            }
            catch (SqlException ex)
            {
                Console.Error.WriteLine($"SQL Error: {ex.Message}");
                throw;
            }
            catch (Exception ex)
            {
                Console.Error.WriteLine($"Error: {ex.Message}");
                throw;
            }
        }

        return dataTable;
    }

    /// <summary>
    /// Executes a dynamic SQL query with pagination and sorting.
    /// </summary>
    /// <param name="tableName">The name of the table to query.</param>
    /// <param name="columnName">The column name to sort by.</param>
    /// <param name="sortDirection">The sort direction: "ASC" or "DESC".</param>
    /// <param name="pageNumber">The page number (1-based).</param>
    /// <param name="pageSize">The number of records per page.</param>
    /// <returns>A DataTable containing the sorted and paginated results.</returns>
    public DataTable GetSortedDataWithPagination(string tableName, string columnName, string sortDirection = "ASC", int pageNumber = 1, int pageSize = 10)
    {
        // Validate column name against whitelist
        if (string.IsNullOrWhiteSpace(columnName))
        {
            throw new ArgumentException("Column name cannot be null or empty.", nameof(columnName));
        }

        if (!AllowedColumns.Contains(columnName))
        {
            throw new ArgumentException($"Invalid column name: '{columnName}'. Column name is not allowed.", nameof(columnName));
        }

        // Validate sort direction
        if (!AllowedSortDirections.Contains(sortDirection ?? "ASC"))
        {
            throw new ArgumentException($"Invalid sort direction: '{sortDirection}'. Must be 'ASC' or 'DESC'.", nameof(sortDirection));
        }

        // Validate pagination parameters
        if (pageNumber < 1)
        {
            throw new ArgumentException("Page number must be greater than 0.", nameof(pageNumber));
        }

        if (pageSize < 1 || pageSize > 1000)
        {
            throw new ArgumentException("Page size must be between 1 and 1000.", nameof(pageSize));
        }

        int offset = (pageNumber - 1) * pageSize;

        // Build the dynamic SQL query with pagination
        string query = $@"
            SELECT * FROM [{tableName}]
            ORDER BY [{columnName}] {sortDirection.ToUpper()}
            OFFSET @Offset ROWS
            FETCH NEXT @PageSize ROWS ONLY";

        DataTable dataTable = new DataTable();

        using (SqlConnection connection = new SqlConnection(_connectionString))
        {
            try
            {
                connection.Open();

                using (SqlCommand command = new SqlCommand(query, connection))
                {
                    command.CommandType = CommandType.Text;
                    command.CommandTimeout = 30;

                    // Use parameterized queries for pagination values
                    command.Parameters.Add("@Offset", SqlDbType.Int).Value = offset;
                    command.Parameters.Add("@PageSize", SqlDbType.Int).Value = pageSize;

                    using (SqlDataAdapter adapter = new SqlDataAdapter(command))
                    {
                        adapter.Fill(dataTable);
                    }
                }
            }
            catch (SqlException ex)
            {
                Console.Error.WriteLine($"SQL Error: {ex.Message}");
                throw;
            }
            catch (Exception ex)
            {
                Console.Error.WriteLine($"Error: {ex.Message}");
                throw;
            }
        }

        return dataTable;
    }
}

// Example usage
class Program
{
    static void Main(string[] args)
    {
        string connectionString = "Server=localhost;Database=TestDB;Trusted_Connection=True;";

        DynamicSortQuery dynamicQuery = new DynamicSortQuery(connectionString);

        try
        {
            // Example 1: Sort by Name in ascending order
            Console.WriteLine("Sorting by Name ASC:");
            DataTable result1 = dynamicQuery.GetSortedData("Users", "Name", "ASC");
            PrintDataTable(result1);

            // Example 2: Sort by Age in descending order
            Console.WriteLine("\nSorting by Age DESC:");
            DataTable result2 = dynamicQuery.GetSortedData("Users", "Age", "DESC");
            PrintDataTable(result2);

            // Example 3: Sort with pagination
            Console.WriteLine("\nSorting by CreatedAt DESC with pagination (Page 1, Size 5):");
            DataTable result3 = dynamicQuery.GetSortedDataWithPagination("Users", "CreatedAt", "DESC", 1, 5);
            PrintDataTable(result3);

            // Example 4: This will throw an exception due to invalid column name
            Console.WriteLine("\nAttempting to sort by invalid column:");
            DataTable result4 = dynamicQuery.GetSortedData("Users", "InvalidColumn; DROP TABLE Users;--", "ASC");
        }
        catch (ArgumentException ex)
        {
            Console.WriteLine($"Validation Error: {ex.Message}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error: {ex.Message}");
        }
    }

    static void PrintDataTable(DataTable dataTable)
    {
        if (dataTable == null || dataTable.Rows.Count == 0)
        {
            Console.WriteLine("No data found.");
            return;
        }

        // Print column headers
        foreach (DataColumn column in dataTable.Columns)
        {
            Console.Write($"{column.ColumnName,-20}");
        }
        Console.WriteLine();
        Console.WriteLine(new string('-', dataTable.Columns.Count * 20));

        // Print rows
        foreach (DataRow row in dataTable.Rows)
        {
            foreach (var item in row.ItemArray)
            {
                Console.Write($"{item,-20}");
            }
            Console.WriteLine();
        }
    }
}
