<?php
session_start();

// Configuration
define('UPLOAD_DIR', __DIR__ . '/uploads/');
define('MAX_FILE_SIZE', 50 * 1024 * 1024); // 50MB
define('ALLOWED_MIME_TYPES', ['application/pdf']);
define('ALLOWED_EXTENSIONS', ['pdf']);
define('DB_FILE', __DIR__ . '/database.json');

// Initialize database
function initDatabase() {
    if (!file_exists(DB_FILE)) {
        $data = [
            'users' => [
                ['id' => 1, 'username' => 'admin', 'password' => password_hash('admin123', PASSWORD_DEFAULT), 'role' => 'admin', 'name' => '管理者'],
                ['id' => 2, 'username' => 'employee1', 'password' => password_hash('emp123', PASSWORD_DEFAULT), 'role' => 'employee', 'name' => '田中太郎'],
                ['id' => 3, 'username' => 'employee2', 'password' => password_hash('emp456', PASSWORD_DEFAULT), 'role' => 'employee', 'name' => '佐藤花子'],
            ],
            'files' => []
        ];
        file_put_contents(DB_FILE, json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
    }
    return json_decode(file_get_contents(DB_FILE), true);
}

function saveDatabase($data) {
    file_put_contents(DB_FILE, json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE));
}

// Create upload directory
if (!is_dir(UPLOAD_DIR)) {
    mkdir(UPLOAD_DIR, 0755, true);
}

$db = initDatabase();

// Authentication functions
function isLoggedIn() {
    return isset($_SESSION['user_id']);
}

function getCurrentUser() {
    global $db;
    if (!isLoggedIn()) return null;
    foreach ($db['users'] as $user) {
        if ($user['id'] === $_SESSION['user_id']) return $user;
    }
    return null;
}

function isAdmin() {
    $user = getCurrentUser();
    return $user && $user['role'] === 'admin';
}

// Handle actions
$action = $_GET['action'] ?? $_POST['action'] ?? 'list';
$message = '';
$messageType = '';

// Login
if ($action === 'login' && $_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';
    
    foreach ($db['users'] as $user) {
        if ($user['username'] === $username && password_verify($password, $user['password'])) {
            $_SESSION['user_id'] = $user['id'];
            $_SESSION['user_name'] = $user['name'];
            header('Location: ?action=list');
            exit;
        }
    }
    $message = 'ユーザー名またはパスワードが正しくありません。';
    $messageType = 'error';
}

// Logout
if ($action === 'logout') {
    session_destroy();
    header('Location: ?action=login');
    exit;
}

// Upload file
if ($action === 'upload' && $_SERVER['REQUEST_METHOD'] === 'POST' && isLoggedIn()) {
    if (!isset($_FILES['pdf_file']) || $_FILES['pdf_file']['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE => 'ファイルサイズがサーバーの制限を超えています。',
            UPLOAD_ERR_FORM_SIZE => 'ファイルサイズがフォームの制限を超えています。',
            UPLOAD_ERR_PARTIAL => 'ファイルが部分的にしかアップロードされませんでした。',
            UPLOAD_ERR_NO_FILE => 'ファイルが選択されていません。',
            UPLOAD_ERR_NO_TMP_DIR => '一時フォルダが見つかりません。',
            UPLOAD_ERR_CANT_WRITE => 'ファイルの書き込みに失敗しました。',
        ];
        $errorCode = $_FILES['pdf_file']['error'] ?? UPLOAD_ERR_NO_FILE;
        $message = $errorMessages[$errorCode] ?? 'アップロードエラーが発生しました。';
        $messageType = 'error';
    } else {
        $file = $_FILES['pdf_file'];
        $originalName = basename($file['name']);
        $fileSize = $file['size'];
        $tmpPath = $file['tmp_name'];
        
        // Validate file size
        if ($fileSize > MAX_FILE_SIZE) {
            $message = 'ファイルサイズが上限（50MB）を超えています。';
            $messageType = 'error';
        } else {
            // Validate extension
            $extension = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
            if (!in_array($extension, ALLOWED_EXTENSIONS)) {
                $message = 'PDFファイルのみアップロード可能です。';
                $messageType = 'error';
            } else {
                // Validate MIME type
                $finfo = finfo_open(FILEINFO_MIME_TYPE);
                $mimeType = finfo_file($finfo, $tmpPath);
                finfo_close($finfo);
                
                if (!in_array($mimeType, ALLOWED_MIME_TYPES)) {
                    $message = '無効なファイル形式です。PDFファイルのみアップロード可能です。';
                    $messageType = 'error';
                } else {
                    // Generate unique filename
                    $uniqueId = uniqid('pdf_', true);
                    $safeFilename = $uniqueId . '.pdf';
                    $destPath = UPLOAD_DIR . $safeFilename;
                    
                    if (move_uploaded_file($tmpPath, $destPath)) {
                        $currentUser = getCurrentUser();
                        $description = trim($_POST['description'] ?? '');
                        $category = trim($_POST['category'] ?? '一般');
                        
                        $fileRecord = [
                            'id' => uniqid(),
                            'original_name' => $originalName,
                            'stored_name' => $safeFilename,
                            'size' => $fileSize,
                            'mime_type' => $mimeType,
                            'description' => $description,
                            'category' => $category,
                            'uploaded_by' => $currentUser['id'],
                            'uploaded_by_name' => $currentUser['name'],
                            'uploaded_at' => date('Y-m-d H:i:s'),
                        ];
                        
                        $db['files'][] = $fileRecord;
                        saveDatabase($db);
                        
                        $message = 'ファイル「' . htmlspecialchars($originalName) . '」が正常にアップロードされました。';
                        $messageType = 'success';
                    } else {
                        $message = 'ファイルの保存に失敗しました。';
                        $messageType = 'error';
                    }
                }
            }
        }
    }
    $action = 'list';
}

// Delete file
if ($action === 'delete' && isLoggedIn()) {
    $fileId = $_GET['id'] ?? '';
    $currentUser = getCurrentUser();
    
    foreach ($db['files'] as $key => $file) {
        if ($file['id'] === $fileId) {
            if ($currentUser['role'] === 'admin' || $file['uploaded_by'] === $currentUser['id']) {
                $filePath = UPLOAD_DIR . $file['stored_name'];
                if (file_exists($filePath)) {
                    unlink($filePath);
                }
                array_splice($db['files'], $key, 1);
                saveDatabase($db);
                $message = 'ファイルを削除しました。';
                $messageType = 'success';
            } else {
                $message = '削除権限がありません。';
                $messageType = 'error';
            }
            break;
        }
    }
    $action = 'list';
}

// Download file
if ($action === 'download' && isLoggedIn()) {
    $fileId = $_GET['id'] ?? '';
    
    foreach ($db['files'] as $file) {
        if ($file['id'] === $fileId) {
            $filePath = UPLOAD_DIR . $file['stored_name'];
            if (file_exists($filePath)) {
                header('Content-Type: application/pdf');
                header('Content-Disposition: attachment; filename="' . $file['original_name'] . '"');
                header('Content-Length: ' . filesize($filePath));
                header('Cache-Control: private, max-age=0, must-revalidate');
                readfile($filePath);
                exit;
            }
        }
    }
    $message = 'ファイルが見つかりません。';
    $messageType = 'error';
    $action = 'list';
}

// Search/filter files
$searchQuery = trim($_GET['search'] ?? '');
$filterCategory = trim($_GET['category'] ?? '');
$files = $db['files'];

if ($searchQuery) {
    $files = array_filter($files, function($file) use ($searchQuery) {
        return stripos($file['original_name'], $searchQuery) !== false ||
               stripos($file['description'], $searchQuery) !== false ||
               stripos($file['uploaded_by_name'], $searchQuery) !== false;
    });
}

if ($filterCategory) {
    $files = array_filter($files, function($file) use ($filterCategory) {
        return $file['category'] === $filterCategory;
    });
}

$files = array_reverse($files);

// Get unique categories
$categories = array_unique(array_column($db['files'], 'category'));

function formatFileSize($bytes) {
    if ($bytes >= 1048576) return round($bytes / 1048576, 2) . ' MB';
    if ($bytes >= 1024) return round($bytes / 1024, 2) . ' KB';
    return $bytes . ' B';
}
