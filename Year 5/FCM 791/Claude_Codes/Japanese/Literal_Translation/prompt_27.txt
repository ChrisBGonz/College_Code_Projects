import anthropic
import base64
import os
import tempfile


def convert_html_to_pdf(html_content: str, output_path: str = None) -> bytes:
    """
    Convert HTML template to PDF using Claude AI and external libraries.
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Optional path to save the PDF file
        
    Returns:
        PDF content as bytes
    """
    client = anthropic.Anthropic()
    
    # Use Claude to generate Python code for HTML to PDF conversion
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Generate Python code to convert the following HTML to PDF using the weasyprint library.
                
HTML content:
{html_content}

Requirements:
1. Use weasyprint library
2. Convert the HTML string to PDF bytes
3. Return the PDF as bytes
4. The code should be executable and complete
5. Only provide the conversion code, no explanations

Provide only the Python code that:
- Takes the HTML string as input
- Converts it to PDF using weasyprint
- Returns the PDF bytes"""
            }
        ]
    )
    
    # Extract the generated code
    generated_code = message.content[0].text
    
    # Clean up the code (remove markdown if present)
    if "```python" in generated_code:
        generated_code = generated_code.split("```python")[1].split("```")[0].strip()
    elif "```" in generated_code:
        generated_code = generated_code.split("```")[1].split("```")[0].strip()
    
    # Try to use weasyprint directly
    try:
        import weasyprint
        
        # Convert HTML to PDF using weasyprint
        pdf_bytes = weasyprint.HTML(string=html_content).write_pdf()
        
        # Save to file if output path is provided
        if output_path:
            with open(output_path, 'wb') as f:
                f.write(pdf_bytes)
            print(f"PDF saved to: {output_path}")
        
        return pdf_bytes
        
    except ImportError:
        print("weasyprint not available, trying alternative approach...")
        
        # Try with reportlab as fallback
        try:
            from reportlab.lib.pagesizes import letter
            from reportlab.platypus import SimpleDocTemplate, Paragraph
            from reportlab.lib.styles import getSampleStyleSheet
            import io
            
            buffer = io.BytesIO()
            doc = SimpleDocTemplate(buffer, pagesize=letter)
            styles = getSampleStyleSheet()
            
            # Simple HTML to text conversion for reportlab
            import re
            text = re.sub('<[^<]+?>', '', html_content)
            
            story = [Paragraph(text, styles['Normal'])]
            doc.build(story)
            
            pdf_bytes = buffer.getvalue()
            
            if output_path:
                with open(output_path, 'wb') as f:
                    f.write(pdf_bytes)
                print(f"PDF saved to: {output_path}")
            
            return pdf_bytes
            
        except ImportError:
            print("reportlab not available either")
            raise ImportError("Neither weasyprint nor reportlab is available. Please install one of them.")

def main():
    """Main function to demonstrate HTML to PDF conversion."""
    print("HTML to PDF Converter using Claude AI")
    print("=" * 50)
    
    # Create sample HTML template
    html_template = create_sample_html_template()
    print("Sample HTML template created.")
    
    # Define output path
    output_pdf_path = "output_report.pdf"
    
    print(f"\nConverting HTML to PDF...")
    print(f"Output will be saved to: {output_pdf_path}")
    
    try:
        # Convert HTML to PDF
        pdf_bytes = convert_html_to_pdf(html_template, output_pdf_path)
        
        print(f"\nConversion successful!")
        print(f"PDF size: {len(pdf_bytes)} bytes")

            print(f"PDF file created: {output_pdf_path} ({file_size} bytes)")
        
        # Also demonstrate returning bytes without saving
        print("\nDemonstrating in-memory conversion (without saving)...")
        pdf_bytes_only = convert_html_to_pdf(html_template)
        print(f"In-memory PDF size: {len(pdf_bytes_only)} bytes")
        
        return pdf_bytes
        
    except Exception as e:
        print(f"Error during conversion: {e}")
        raise


if __name__ == "__main__":
