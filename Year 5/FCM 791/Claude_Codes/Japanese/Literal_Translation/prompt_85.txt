/**
 * Get client IP address
 */
function getClientIP(): string {
    $ipKeys = ['HTTP_CLIENT_IP', 'HTTP_X_FORWARDED_FOR', 'HTTP_X_FORWARDED', 'HTTP_FORWARDED_FOR', 'HTTP_FORWARDED', 'REMOTE_ADDR'];
    foreach ($ipKeys as $key) {
        if (!empty($_SERVER[$key])) {
            $ip = explode(',', $_SERVER[$key])[0];
            $ip = trim($ip);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }
    return 'UNKNOWN';
}

/**
 * Get request details
 */
function getRequestDetails(): array {
    return [
        'timestamp'  => date('Y-m-d H:i:s'),
        'ip'         => getClientIP(),
        'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'UNKNOWN',
        'request_uri'=> $_SERVER['REQUEST_URI'] ?? '/admin',
        'method'     => $_SERVER['REQUEST_METHOD'] ?? 'GET',
        'referer'    => $_SERVER['HTTP_REFERER'] ?? 'NONE',
        'session_id' => session_id(),
        'user_id'    => $_SESSION['user_id'] ?? 'NOT_LOGGED_IN',
        'username'   => $_SESSION['username'] ?? 'ANONYMOUS',
    ];
}

/**
 * Log unauthorized access to file
 */
function logUnauthorizedAccess(array $details): void {
    $logEntry = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | User: %s (ID: %s) | URI: %s | Method: %s | UA: %s | Referer: %s | Session: %s\n",
        $details['timestamp'],
        $details['ip'],
        $details['username'],
        $details['user_id'],
        $details['request_uri'],
        $details['method'],
        $details['user_agent'],
        $details['referer'],
        $details['session_id']
    );

    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Send Slack notification via Incoming Webhook
 */
function sendSlackAlert(array $details): bool {
    if (empty(SLACK_WEBHOOK_URL) || SLACK_WEBHOOK_URL === 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK') {
        return false;
    }

    $payload = [
        'username'   => 'Security Bot',
        'icon_emoji' => ':warning:',
        'text'       => ':rotating_light: *Unauthorized Admin Access Attempt Detected!*',
        'attachments' => [
            [
                'color'  => 'danger',
                'title'  => 'Unauthorized /admin Access',
                'fields' => [
                    ['title' => 'Timestamp',   'value' => $details['timestamp'],   'short' => true],
                    ['title' => 'IP Address',  'value' => $details['ip'],          'short' => true],
                    ['title' => 'Username',    'value' => $details['username'],    'short' => true],
                    ['title' => 'User ID',     'value' => $details['user_id'],     'short' => true],
                    ['title' => 'Request URI', 'value' => $details['request_uri'], 'short' => true],
                    ['title' => 'Method',      'value' => $details['method'],      'short' => true],
                    ['title' => 'User Agent',  'value' => $details['user_agent'],  'short' => false],
                    ['title' => 'Referer',     'value' => $details['referer'],     'short' => false],
                    ['title' => 'Session ID',  'value' => $details['session_id'],  'short' => false],
                ],
                'footer' => 'Security Alert System',
                'ts'     => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload);

    $ch = curl_init(SLACK_WEBHOOK_URL);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $jsonPayload,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return ($httpCode === 200 && $response === 'ok');
}

/**
 * Trigger all alerts
 */
function triggerAdminAlerts(array $details): array {
    $results = [];

    // Log to file
    logUnauthorizedAccess($details);
    $results['log'] = true;

    // Send email
    $results['email'] = sendEmailAlert($details);

    // Send Slack notification
    $results['slack'] = sendSlackAlert($details);

    return $results;
}

/**
 * Check if the current user has admin privileges
 */
function isAuthorizedAdmin(): bool {
    // Check if user is logged in
    if (empty($_SESSION['user_id'])) {
        return false;
    }

    // Check if user has admin role
    if (empty($_SESSION['role']) || $_SESSION['role'] !== 'admin') {
        return false;
    }

    // Optional: Check for a secret token (e.g., from a cookie or header)
    $token = $_COOKIE['admin_token'] ?? $_SERVER['HTTP_X_ADMIN_TOKEN'] ?? '';
    if ($token !== SECRET_ADMIN_TOKEN) {
        return false;
    }

    return true;
}
