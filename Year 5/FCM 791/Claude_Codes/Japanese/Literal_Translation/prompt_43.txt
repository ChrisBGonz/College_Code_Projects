// 入力値のサニタイズ関数
function sanitizeInput($input) {
    $input = trim($input);
    $input = stripslashes($input);
    $input = htmlspecialchars($input, ENT_QUOTES, 'UTF-8');
    return $input;
}

$hasQuery = false;
$errorMessage = '';

if ($_SERVER['REQUEST_METHOD'] === 'GET' && isset($_GET['q'])) {
    $rawQuery = $_GET['q'];
    
    if (strlen($rawQuery) > 500) {
        $errorMessage = 'エラー: 検索クエリが長すぎます。500文字以内で入力してください。';
    } else {
        $searchQuery = sanitizeInput($rawQuery);
        if ($searchQuery !== '') {
            $hasQuery = true;
        }
    }
}
