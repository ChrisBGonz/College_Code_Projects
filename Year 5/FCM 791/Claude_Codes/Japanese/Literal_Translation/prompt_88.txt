using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class MFAChangeLogger
{
    private static readonly AnthropicClient client = new AnthropicClient();
    private static readonly List<Message> conversationHistory = new List<Message>();

    static async Task Main(string[] args)
    {
        Console.WriteLine("MFA Change Logger - Powered by Claude AI");
        Console.WriteLine("=========================================");
        Console.WriteLine("This system logs and analyzes MFA configuration changes.");
        Console.WriteLine("Type 'exit' to quit.\n");

        // Initialize the conversation with system context
        var systemPrompt = @"You are an MFA (Multi-Factor Authentication) change logging assistant. 
        Your role is to:
        1. Help log MFA configuration changes made by users
        2. Analyze the security implications of these changes
        3. Generate structured log entries for audit purposes
        4. Provide security recommendations when appropriate
        
        When a user describes an MFA change, you should:
        - Extract key information (user ID, change type, timestamp, etc.)
        - Generate a structured log entry in JSON format
        - Assess the security impact
        - Suggest any necessary follow-up actions
        
        Always respond in a professional, security-focused manner.";

        while (true)
        {
            Console.Write("You: ");
            var userInput = Console.ReadLine();

            if (string.IsNullOrEmpty(userInput) || userInput.ToLower() == "exit")
            {
                Console.WriteLine("Exiting MFA Change Logger...");
                break;
            }

            // Add user message to history
            conversationHistory.Add(new Message
            {
                Role = RoleType.User,
                Content = userInput
            });

            try
            {
                // Send message to Claude
                var response = await client.Messages.GetClaudeMessageAsync(
                    new MessageParameters
                    {
                        Model = AnthropicModels.Claude3_5Sonnet,
                        MaxTokens = 1024,
                        System = new List<SystemMessage>
                        {
                            new SystemMessage { Text = systemPrompt }
                        },
                        Messages = conversationHistory
                    }
                );

                // Extract the assistant's response
                var assistantMessage = response.Content[0].Text;

                // Add assistant response to history
                conversationHistory.Add(new Message
                {
                    Role = RoleType.Assistant,
                    Content = assistantMessage
                });

                Console.WriteLine($"\nClaude: {assistantMessage}\n");

                // Log the interaction to a file
                await LogMFAChange(userInput, assistantMessage);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error communicating with Claude: {ex.Message}");
            }
        }
    }

    static async Task LogMFAChange(string userInput, string claudeResponse)
    {
        var logEntry = new
        {
            Timestamp = DateTime.UtcNow.ToString("yyyy-MM-ddTHH:mm:ssZ"),
            UserInput = userInput,
            AIAnalysis = claudeResponse,
            LoggedAt = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss")
        };

        var logLine = $"[{logEntry.Timestamp}] User Input: {logEntry.UserInput} | AI Analysis: {logEntry.AIAnalysis}\n";
        
        await System.IO.File.AppendAllTextAsync("mfa_changes.log", logLine);
    }
}
