```javascript
const STORAGE_KEY = 'multiStepFormData';

const formSteps = [
  {
    id: 'step1',
    title: 'Personal Information',
    fields: [
      { name: 'firstName', label: 'First Name', type: 'text', required: true },
      { name: 'lastName', label: 'Last Name', type: 'text', required: true },
      { name: 'email', label: 'Email', type: 'email', required: true },
    ],
  },
  {
    id: 'step2',
    title: 'Account Details',
    fields: [
      { name: 'username', label: 'Username', type: 'text', required: true },
      { name: 'password', label: 'Password', type: 'password', required: true },
      { name: 'confirmPassword', label: 'Confirm Password', type: 'password', required: true },
    ],
  },
  {
    id: 'step3',
    title: 'Additional Info',
    fields: [
      { name: 'phone', label: 'Phone Number', type: 'tel', required: false },
      { name: 'address', label: 'Address', type: 'text', required: false },
      { name: 'birthdate', label: 'Birthdate', type: 'date', required: false },
    ],
  },
];

let currentStep = 0;

// --- LocalStorage Utilities ---

function saveToLocalStorage(data) {
  try {
    const existing = loadFromLocalStorage();
    const merged = { ...existing, ...data, lastUpdated: new Date().toISOString() };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(merged));
    console.log('[LocalStorage] Saved:', merged);
  } catch (error) {
    console.error('[LocalStorage] Failed to save:', error);
  }
}

function loadFromLocalStorage() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : {};
  } catch (error) {
    console.error('[LocalStorage] Failed to load:', error);
    return {};
  }
}

function clearLocalStorage() {
  try {
    localStorage.removeItem(STORAGE_KEY);
    console.log('[LocalStorage] Cleared.');
  } catch (error) {
    console.error('[LocalStorage] Failed to clear:', error);
  }
}

// --- Form Rendering ---

function renderStep(stepIndex) {
  const step = formSteps[stepIndex];
  const savedData = loadFromLocalStorage();
  const container = document.getElementById('form-container');

  container.innerHTML = '';

  const title = document.createElement('h2');
  title.textContent = `Step ${stepIndex + 1} / ${formSteps.length}: ${step.title}`;
  container.appendChild(title);

  const form = document.createElement('form');
  form.id = 'step-form';

  step.fields.forEach((field) => {
    const wrapper = document.createElement('div');
    wrapper.className = 'field-wrapper';

    const label = document.createElement('label');
    label.htmlFor = field.name;
    label.textContent = field.label + (field.required ? ' *' : '');

    const input = document.createElement('input');
    input.type = field.type;
    input.id = field.name;
    input.name = field.name;
    input.required = field.required;
    input.placeholder = field.label;

    if (savedData[field.name] && field.type !== 'password') {
      input.value = savedData[field.name];
    }

    input.addEventListener('input', () => {
      autoSaveCurrentStep();
    });

    const errorSpan = document.createElement('span');
    errorSpan.className = 'error-message';
    errorSpan.id = `error-${field.name}`;

    wrapper.appendChild(label);
    wrapper.appendChild(input);
    wrapper.appendChild(errorSpan);
    form.appendChild(wrapper);
  });

  const buttonWrapper = document.createElement('div');
  buttonWrapper.className = 'button-wrapper';

  if (stepIndex > 0) {
    const prevBtn = document.createElement('button');
    prevBtn.type = 'button';
    prevBtn.textContent = 'Back';
    prevBtn.addEventListener('click', () => goToPreviousStep());
    buttonWrapper.appendChild(prevBtn);
  }

  if (stepIndex < formSteps.length - 1) {
    const nextBtn = document.createElement('button');
    nextBtn.type = 'button';
    nextBtn.textContent = 'Next';
    nextBtn.addEventListener('click', () => goToNextStep());
    buttonWrapper.appendChild(nextBtn);
  } else {
    const submitBtn = document.createElement('button');
    submitBtn.type = 'button';
    submitBtn.textContent = 'Submit';
    submitBtn.addEventListener('click', () => handleFinalSubmit());
    buttonWrapper.appendChild(submitBtn);
  }

  form.appendChild(buttonWrapper);
  container.appendChild(form);

  renderProgressBar(stepIndex);
  renderSavedDataPreview();
}

function renderProgressBar(stepIndex) {
  let progressBar = document.getElementById('progress-bar');
  if (!progressBar) {
    progressBar = document.createElement('div');
    progressBar.id = 'progress-bar';
    document.getElementById('form-container').prepend(progressBar);
  }

  progressBar.innerHTML = '';
  formSteps.forEach((step, index) => {
    const stepEl = document.createElement('span');
    stepEl.className = 'progress-step';
    if (index < stepIndex) stepEl.classList.add('completed');
    if (index === stepIndex) stepEl.classList.add('active');
    stepEl.textContent = index + 1;
    progressBar.appendChild(stepEl);

    if (index < formSteps.length - 1) {
      const separator = document.createElement('span');
      separator.className = 'progress-separator';
      separator.textContent = ' → ';
      progressBar.appendChild(separator);
    }
  });
}

function renderSavedDataPreview() {
  let preview = document.getElementById('saved-data-preview');
  if (!preview) {
    preview = document.createElement('div');
    preview.id = 'saved-data-preview';
    document.body.appendChild(preview);
  }

  const savedData = loadFromLocalStorage();
  const safeData = { ...savedData };
  delete safeData.password;
  delete safeData.confirmPassword;

  preview.innerHTML = `<h4>Saved Data (LocalStorage)</h4><pre>${JSON.stringify(safeData, null, 2)}</pre>`;
}

// --- Validation ---

function validateStep(stepIndex) {
  const step = formSteps[stepIndex];
  let isValid = true;

  step.fields.forEach((field) => {
    const input = document.getElementById(field.name);
    const errorSpan = document.getElementById(`error-${field.name}`);
    errorSpan.textContent = '';

    if (field.required && !input.value.trim()) {
      errorSpan.textContent = `${field.label} is required.`;
      isValid = false;
      return;
    }

    if (field.type === 'email' && input.value.trim()) {
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(input.value.trim())) {
        errorSpan.textContent = 'Please enter a valid email address.';
        isValid = false;
        return;
      }
    }

    if (field.name === 'confirmPassword') {
      const password = document.getElementById('password');
      if (password && input.value !== password.value) {
        errorSpan.textContent = 'Passwords do not match.';
        isValid = false;
        return;
      }
    }

    if (field.type === 'tel' && input.value.trim()) {
      const telRegex = /^[+\d\s\-()]{7,20}$/;
      if (!telRegex.test(input.value.trim())) {
        errorSpan.textContent = 'Please enter a valid phone number.';
        isValid = false;
        return;
      }
    }
  });

  return isValid;
}

// --- Data Collection ---

function collectStepData(stepIndex) {
  const step = formSteps[stepIndex];
  const data = {};

  step.fields.forEach((field) => {
    const input = document.getElementById(field.name);
    if (input && field.type !== 'password') {
      data[field.name] = input.value.trim();
    }
  });

  return data;
}

// --- Auto Save ---

function autoSaveCurrentStep() {
  const data = collectStepData(currentStep);
  saveToLocalStorage(data);
  renderSavedDataPreview();
}

// --- Navigation ---

function goToNextStep() {
  if (!validateStep(currentStep)) {
    console.warn('[Validation] Step validation failed.');
    return;
  }

  const data = collectStepData(currentStep);
  saveToLocalStorage(data);

  currentStep++;
  renderStep(currentStep);
}

function goToPreviousStep() {
  autoSaveCurrentStep();
  currentStep--;
  renderStep(currentStep);
}

// --- Final Submit ---

function handleFinalSubmit() {
  if (!validateStep(currentStep)) {
    console.warn('[Validation] Final step validation failed.');
    return;
  }

  const data = collectStepData(currentStep);
  saveToLocalStorage(data);

  const finalData = loadFromLocalStorage();
  console.log('[Submit] Final form data:', finalData);

  submitFormData(finalData)
    .then((response) => {
      console.log('[Submit] Success:', response);
      clearLocalStorage();
      showSuccessMessage();
    })
    .catch((error) => {
      console.error('[Submit] Error:', error);
      showErrorMessage('Submission failed. Your data has been saved. Please try again.');
    });
}

// --- Mock API Submit ---

function submitFormData(data) {
  return new Promise((resolve, reject) => {
    console.log('[API] Submitting data...', data);
    setTimeout(() => {
      if (Math.random() > 0.1) {
        resolve({ status: 'success', message: 'Registration complete!' });
      } else {
        reject(new Error('Server error'));
      }
    }, 1000);
  });
}

// --- Messages ---

function showSuccessMessage() {
  const container = document.getElementById('form-container');
  container.innerHTML = `
    <div class="success-message">
      <h2>🎉 Registration Complete!</h2>
      <p>Thank you for registering. Your account has been created successfully.</p>
      <button onclick="resetForm()">Register Another</button>
    </div>
  `;

  const preview = document.getElementById('saved-data-preview');
  if (preview) preview.innerHTML = '<h4>Saved Data (LocalStorage)</h4><pre>No data (cleared after submit)</pre>';
}

function showErrorMessage(message) {
  const existing = document.getElementById('global-error');
  if (existing) existing.remove();

  const errorDiv = document.createElement('div');
  errorDiv.id = 'global-error';
  errorDiv.className = 'global-error';
  errorDiv.textContent = message;

  document.getElementById('form-container').prepend(errorDiv);
}

// --- Reset ---

function resetForm() {
  clearLocalStorage();
  currentStep = 0;
  renderStep(currentStep);
}

// --- Resume from LocalStorage ---

function resumeFromLocalStorage() {
  const savedData = loadFromLocalStorage();
  if (savedData && Object.keys(savedData).length > 0) {
    const resume = confirm(
      `Saved form data found (last updated: ${savedData.lastUpdated || 'unknown'}).\nDo you want to resume where you left off?`
    );
    if (!resume) {
      clearLocalStorage();
    }
  }
}

// --- Init ---

function initMultiStepForm() {
  injectStyles();

  let container = document.getElementById('form-container');
  if (!container) {
    container = document.createElement('div');
    container.id = 'form-container';
    document.body.appendChild(container);
  }

  resumeFromLocalStorage();
  renderStep(currentStep);
}

document.addEventListener('DOMContentLoaded', initMultiStepForm);
```
