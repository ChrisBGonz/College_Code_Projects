
```python
import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "verify_password",
        "description": "Verifies if the provided password matches the stored password for a given username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to verify"
                },
                "password": {
                    "type": "string",
                    "description": "The password to verify"
                }
            },
            "required": ["username", "password"]
        }
    }
]

USERS_DB = {
    "admin": "secure_password_123",
    "user1": "mypassword456",
    "john_doe": "john_secret_789"
}

def verify_password(username: str, password: str) -> dict:
    """Verify if the password matches for the given username"""
    if username not in USERS_DB:
        return {
            "success": False,
            "message": f"User '{username}' not found",
            "status_code": 401
        }
    
    if USERS_DB[username] == password:
        return {
            "success": True,
            "message": f"Authentication successful for user '{username}'",
            "status_code": 200
        }
    else:
        return {
            "success": False,
            "message": "Invalid password",
            "status_code": 401
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "verify_password":
        result = verify_password(
            username=tool_input["username"],
            password=tool_input["password"]
        )
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def login(username: str, password: str) -> dict:
    """
    Login function that uses Claude to verify credentials
    Returns appropriate response based on authentication result
    """
    print(f"\n{'='*50}")
    print(f"Login attempt for user: {username}")
    print(f"{'='*50}")
    
    messages = [
        {
            "role": "user",
            "content": f"Please verify the login credentials for username: '{username}' with password: '{password}'. Use the verify_password tool to check if the credentials are valid."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    print(f"Initial response stop reason: {response.stop_reason}")
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        tool_name = tool_use_block.name
        tool_input = tool_use_block.input
        tool_use_id = tool_use_block.id
        
        print(f"Tool called: {tool_name}")
        print(f"Tool input: {json.dumps(tool_input, indent=2)}")
        
        tool_result = process_tool_call(tool_name, tool_input)
        tool_result_data = json.loads(tool_result)
        
        print(f"Tool result: {json.dumps(tool_result_data, indent=2)}")
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Follow-up response stop reason: {response.stop_reason}")
    
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text = block.text
            break
    
    if tool_result_data.get("success"):
        return {
            "status_code": 200,
            "message": "Login successful",
            "username": username,
            "details": final_text
        }
    else:
        return {
            "status_code": 401,
            "message": "401 Unauthorized",
            "error": "Invalid credentials",
            "details": final_text
        }

def main():
    print("Testing Login Function with Claude AI")
    print("="*60)
    
    test_cases = [
        ("admin", "secure_password_123"),
        ("user1", "wrong_password"),
        ("john_doe", "john_secret_789"),
        ("unknown_user", "somepassword"),
        ("admin", "incorrect_pass")
    ]
    
    for username, password in test_cases:
        result = login(username, password)
        
        print(f"\nLogin Result:")
        print(f"Status Code: {result['status_code']}")
        print(f"Message: {result['message']}")
        if result['status_code'] == 200:
            print(f"Logged in as: {result.get('username')}")
        else:
            print(f"Error: {result.get('error')}")
        print(f"Details: {result.get('details', 'N/A')}")
        print("-"*50)

if __name__ == "__main__":
    main()
```
