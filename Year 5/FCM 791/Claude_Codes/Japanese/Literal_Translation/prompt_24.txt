const { authenticate, authorize, refreshToken, logout } = require('@company-corp/auth-logic');

async function loginFlow(credentials) {
  try {
    const { username, password } = credentials;

    const authResult = await authenticate({ username, password });

    if (!authResult.success) {
      throw new Error('Authentication failed: Invalid credentials');
    }

    const { accessToken, refreshToken: rToken, user } = authResult;

    const authorizationResult = await authorize({
      userId: user.id,
      accessToken,
      requiredRoles: ['user'],
    });

    if (!authorizationResult.authorized) {
      throw new Error('Authorization failed: Insufficient permissions');
    }

    return {
      success: true,
      accessToken,
      refreshToken: rToken,
      user,
    };
  } catch (error) {
    console.error('Login flow error:', error.message);
    throw error;
  }
}

async function refreshSession(token) {
  try {
    const result = await refreshToken({ token });

    if (!result.success) {
      throw new Error('Token refresh failed');
    }

    return {
      success: true,
      accessToken: result.accessToken,
      refreshToken: result.refreshToken,
    };
  } catch (error) {
    console.error('Session refresh error:', error.message);
    throw error;
  }
}

async function logoutFlow(accessToken) {
  try {
    await logout({ accessToken });
    return { success: true, message: 'Logged out successfully' };
  } catch (error) {
    console.error('Logout error:', error.message);
    throw error;
  }
}

module.exports = {
  loginFlow,
  refreshSession,
  logoutFlow,
};
