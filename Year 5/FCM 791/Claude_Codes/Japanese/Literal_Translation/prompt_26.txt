```yaml
name: Build and Upload Artifacts

on:
  push:
    branches:
      - main
      - master
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:

env:
  BUCKET_NAME: ${{ secrets.STORAGE_BUCKET_NAME }}
  ARTIFACT_DIR: dist

jobs:
  build-and-upload:
    name: Build and Upload to Storage Bucket
    runs-on: ubuntu-latest

    permissions:
      contents: read
      id-token: write

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: npm test

      - name: Build project
        run: npm run build

      - name: Set build metadata
        id: metadata
        run: |
          echo "timestamp=$(date -u +'%Y%m%d%H%M%S')" >> $GITHUB_OUTPUT
          echo "short_sha=$(echo ${{ github.sha }} | cut -c1-7)" >> $GITHUB_OUTPUT
          echo "artifact_name=build-${{ github.ref_name }}-$(date -u +'%Y%m%d%H%M%S')-$(echo ${{ github.sha }} | cut -c1-7).tar.gz" >> $GITHUB_OUTPUT

      - name: Archive build artifacts
        run: |
          tar -czf ${{ steps.metadata.outputs.artifact_name }} \
            -C ${{ env.ARTIFACT_DIR }} .
          echo "Artifact created: ${{ steps.metadata.outputs.artifact_name }}"
          ls -lh ${{ steps.metadata.outputs.artifact_name }}

      - name: Authenticate to Google Cloud
        uses: google-github-actions/auth@v2
        with:
          workload_identity_provider: ${{ secrets.GCP_WORKLOAD_IDENTITY_PROVIDER }}
          service_account: ${{ secrets.GCP_SERVICE_ACCOUNT }}

      - name: Upload artifact to Google Cloud Storage
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: ${{ steps.metadata.outputs.artifact_name }}
          destination: ${{ env.BUCKET_NAME }}/artifacts/${{ github.ref_name }}
          headers: |-
            content-type: application/gzip
            cache-control: no-cache

      - name: Upload build directory to Google Cloud Storage
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: ${{ env.ARTIFACT_DIR }}
          destination: ${{ env.BUCKET_NAME }}/latest/${{ github.ref_name }}
          glob: '**'

      - name: Output upload summary
        run: |
          echo "## Build and Upload Summary" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "| Item | Value |" >> $GITHUB_STEP_SUMMARY
          echo "|------|-------|" >> $GITHUB_STEP_SUMMARY
          echo "| Branch | ${{ github.ref_name }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Commit | ${{ steps.metadata.outputs.short_sha }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Timestamp | ${{ steps.metadata.outputs.timestamp }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Artifact | ${{ steps.metadata.outputs.artifact_name }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Bucket | gs://${{ env.BUCKET_NAME }} |" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "### Upload Paths" >> $GITHUB_STEP_SUMMARY
          echo "- Archive: \`gs://${{ env.BUCKET_NAME }}/artifacts/${{ github.ref_name }}/${{ steps.metadata.outputs.artifact_name }}\`" >> $GITHUB_STEP_SUMMARY
          echo "- Latest: \`gs://${{ env.BUCKET_NAME }}/latest/${{ github.ref_name }}/\`" >> $GITHUB_STEP_SUMMARY

  notify-on-failure:
    name: Notify on Failure
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: failure()

    steps:
      - name: Send failure notification
        run: |
          echo "Build or upload failed for commit ${{ github.sha }} on branch ${{ github.ref_name }}"
          echo "Workflow: ${{ github.workflow }}"
          echo "Run URL: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
```
