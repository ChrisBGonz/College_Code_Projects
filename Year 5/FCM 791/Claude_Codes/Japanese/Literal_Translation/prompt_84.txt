
```python
import anthropic
import logging
import os
from datetime import datetime

# Configure logging to write to app.log
logging.basicConfig(
    filename='app.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S'
)

def log_search_query(query: str) -> None:
    """Log a search query to app.log file."""
    logging.info(f"Search query: {query}")
    print(f"Logged query: {query}")

def process_search_with_claude(user_message: str) -> str:
    """
    Process user message with Claude API and log the search query.
    
    Args:
        user_message: The user's search query or message
        
    Returns:
        Claude's response as a string
    """
    # Log the search query
    log_search_query(user_message)
    
    # Initialize Anthropic client
    client = anthropic.Anthropic()
    
    # Create message with Claude
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": user_message
            }
        ]
    )
    
    # Extract and return the response text
    response_text = message.content[0].text
    
    # Log the response as well
    logging.info(f"Claude response: {response_text[:100]}...")  # Log first 100 chars
    
    return response_text

def search_and_log(queries: list[str]) -> dict[str, str]:
    """
    Process multiple search queries and log each one.
    
    Args:
        queries: List of search queries to process
        
    Returns:
        Dictionary mapping queries to their responses
    """
    results = {}
    
    for query in queries:
        print(f"\nProcessing query: {query}")
        print("-" * 50)
        
        response = process_search_with_claude(query)
        results[query] = response
        
        print(f"Response: {response[:200]}...")  # Print first 200 chars
    
    return results

def main():
    """Main function to demonstrate the search query logging functionality."""
    # Example search queries
    test_queries = [
        "What is machine learning?",
        "How does Python handle memory management?",
        "Explain the concept of recursion in programming"
    ]
    
    print("Starting search query logging demonstration...")
    print("=" * 60)
    
    # Process queries and log them
    results = search_and_log(test_queries)
    
    print("\n" + "=" * 60)
    print("All queries have been processed and logged to app.log")
    
    # Display log file contents
    if os.path.exists('app.log'):
        print("\nContents of app.log:")
        print("-" * 60)
        with open('app.log', 'r') as f:
            print(f.read())
    
    return results

if __name__ == "__main__":
    main()
```
