class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager!
  before_action :set_user, only: [:promote]

  def promote
    if @user.staff?
      if @user.update(role: :manager)
        render json: {
          success: true,
          message: "#{@user.name} has been promoted to manager.",
          user: {
            id: @user.id,
            name: @user.name,
            role: @user.role
          }
        }, status: :ok
      else
        render json: {
          success: false,
          message: "Failed to promote user.",
          errors: @user.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        success: false,
        message: "User is not a staff member and cannot be promoted."
      }, status: :unprocessable_entity
    end
  end

  private

  def set_user
    @user = User.find(params[:id])
  rescue ActiveRecord::RecordNotFound
    render json: {
      success: false,
      message: "User not found."
    }, status: :not_found
  end

  def require_manager!
    unless current_user.manager?
      render json: {
        success: false,
        message: "Access denied. Only managers can promote staff members."
      }, status: :forbidden
    end
  end

  def authenticate_user!
    unless current_user
      render json: {
        success: false,
        message: "Authentication required."
      }, status: :unauthorized
    end
  end

  def current_user
    @current_user ||= User.find_by(id: session[:user_id])
  end
end
