import anthropic
import requests
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def fetch_url_content(url: str) -> tuple[str | None, str | None]:
    """Fetch content from a URL and return (content, error)."""
    try:
        response = requests.get(url, timeout=10)
        if response.status_code == 404:
            return None, f"404 Error: The URL '{url}' was not found."
        response.raise_for_status()
        return response.text, None
    except requests.exceptions.ConnectionError:
        return None, f"Connection Error: Could not connect to '{url}'."
    except requests.exceptions.Timeout:
        return None, f"Timeout Error: Request to '{url}' timed out."
    except requests.exceptions.HTTPError as e:
        return None, f"HTTP Error: {str(e)}"
    except requests.exceptions.RequestException as e:
        return None, f"Request Error: {str(e)}"


@app.route("/fetch", methods=["GET"])
def fetch_url():
    """Route to fetch URL content and display it using Claude."""
    url = request.args.get("url")

    if not url:
        return {"error": "Please provide a URL parameter. Example: /fetch?url=https://example.com"}, 400

    if not url.startswith(("http://", "https://")):
        url = "https://" + url

    content, error = fetch_url_content(url)

    if error:
        return {"error": error}, 404 if "404" in error else 400

    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"""以下のURLから取得したコンテンツを分析して、主要な情報を日本語で要約してください。

URL: {url}

コンテンツ:
{content[:5000]}

{'...(コンテンツが長いため省略されています)' if len(content) > 5000 else ''}

このコンテンツの主要な情報、目的、重要なポイントを分かりやすく説明してください。""",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(
        stream_with_context(generate()),
        mimetype="text/plain",
        headers={
            "X-URL": url,
            "X-Content-Length": str(len(content)),
        },
    )


@app.route("/fetch-raw", methods=["GET"])
def fetch_url_raw():
    """Route to fetch and display raw URL content."""
    url = request.args.get("url")

    if not url:
        return {"error": "Please provide a URL parameter. Example: /fetch-raw?url=https://example.com"}, 400

    if not url.startswith(("http://", "https://")):
        url = "https://" + url

    content, error = fetch_url_content(url)

    if error:
        return {"error": error}, 404 if "404" in error else 400

    return {
        "url": url,
        "content_length": len(content),
        "content_preview": content[:1000] + ("..." if len(content) > 1000 else ""),
        "full_content": content,
    }


@app.errorhandler(404)
def not_found(error):
    return {"error": "Route not found. Available routes: /fetch?url=<url>, /fetch-raw?url=<url>"}, 404


@app.errorhandler(500)
def internal_error(error):
    return {"error": "Internal server error occurred."}, 500


if __name__ == "__main__":
    print("Flask URL Fetcher with Claude AI")
    print("Available routes:")
    print("  GET /fetch?url=<url>     - Fetch URL and analyze with Claude (streaming)")
    print("  GET /fetch-raw?url=<url> - Fetch URL and return raw content")
    print("\nExample usage:")
    print("  curl 'http://localhost:5000/fetch?url=https://example.com'")
    print("  curl 'http://localhost:5000/fetch-raw?url=https://httpbin.org/json'")
    app.run(debug=True, port=5000)
