
```javascript
import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database for demonstration
const userDatabase = {
  user123: {
    id: "user123",
    username: "john_doe",
    email: "john@example.com",
    password: "old_hashed_password_123",
  },
  user456: {
    id: "user456",
    username: "jane_smith",
    email: "jane@example.com",
    password: "old_hashed_password_456",
  },
};

// Tool definitions for password reset flow
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database and returns user information",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes the password securely before storing in database",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description: "Updates the user's password in the database",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        hashed_password: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "hashed_password"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a password reset confirmation email to the user's registered email address",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool execution functions
function executeValidateUser(userId) {
  const user = userDatabase[userId];
  if (user) {
    return {
      success: true,
      user: {
        id: user.id,
        username: user.username,
        email: user.email,
      },
    };
  }
  return {
    success: false,
    error: `User with ID ${userId} not found`,
  };
}

function executeValidatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /[0-9]/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  return {
    isValid,
    requirements: {
      minLength: { met: minLength, description: "At least 8 characters" },
      hasUppercase: {
        met: hasUppercase,
        description: "Contains uppercase letter",
      },
      hasLowercase: {
        met: hasLowercase,
        description: "Contains lowercase letter",
      },
      hasNumber: { met: hasNumber, description: "Contains number" },
      hasSpecialChar: {
        met: hasSpecialChar,
        description: "Contains special character",
      },
    },
  };
}

function executeHashPassword(password) {
  // Simulated hashing (in production, use bcrypt or similar)
  const hash = Buffer.from(password + "_salt_" + Date.now()).toString("base64");
  return {
    success: true,
    hashedPassword: `hashed_${hash}`,
  };
}

function executeUpdatePasswordInDatabase(userId, hashedPassword) {
  if (userDatabase[userId]) {
    userDatabase[userId].password = hashedPassword;
    return {
      success: true,
      message: `Password successfully updated for user ${userId}`,
      updatedAt: new Date().toISOString(),
    };
  }
  return {
    success: false,
    error: `Failed to update password for user ${userId}`,
  };
}

function executeSendConfirmationEmail(userId, email) {
  // Simulated email sending
  console.log(
    `[Email Service] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
    sentAt: new Date().toISOString(),
  };
}

// Process tool calls
function processTool(toolName, toolInput) {
  console.log(`\n[Tool Execution] Executing: ${toolName}`);
  console.log(`[Tool Input]`, JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "validate_user":
      result = executeValidateUser(toolInput.user_id);
      break;
    case "validate_password_strength":
      result = executeValidatePasswordStrength(toolInput.password);
      break;
    case "hash_password":
      result = executeHashPassword(toolInput.password);
      break;
    case "update_password_in_database":
      result = executeUpdatePasswordInDatabase(
        toolInput.user_id,
        toolInput.hashed_password
      );
      break;
    case "send_confirmation_email":
      result = executeSendConfirmationEmail(toolInput.user_id, toolInput.email);
      break;
    default:
      result = { error: `Unknown tool: ${toolName}` };
  }

  console.log(`[Tool Result]`, JSON.stringify(result, null, 2));
  return result;
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Starting Password Reset Flow`);
  console.log(`User ID: ${userId}`);
  console.log(`${"=".repeat(60)}`);

  const messages = [
    {
      role: "user",
      content: `Please help me reset the password for user ID: "${userId}". The new password they want to set is: "${newPassword}". 
      
      Please follow these steps:
      1. First validate that the user exists in the database
      2. Validate that the new password meets security requirements
      3. If both validations pass, hash the password securely
      4. Update the password in the database
      5. Send a confirmation email to the user
      6. Provide a summary of the password reset process
      
      If any step fails, stop the process and explain what went wrong.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log(`\n[Initial Response] Stop reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );
    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      const result = processTool(toolUse.name, toolUse.input);
      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\n[Continued Response] Stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log(`\n${"=".repeat(60)}`);
  console.log(`Password Reset Flow Complete`);
  console.log(`${"=".repeat(60)}`);
  console.log(`\nFinal Response:\n${finalResponse}`);

  return {
    success: true,
    message: finalResponse,
    userId: userId,
  };
}

// Test the password reset function
async function main() {
  try {
    // Test case 1: Valid user with strong password
    console.log("\n### Test Case 1: Valid user with strong password ###");
    await resetPassword("user123", "SecurePass@123");

    // Test case 2: Invalid user
    console.log("\n\n### Test Case 2: Invalid user ###");
    await resetPassword("invalid_user", "SecurePass@123");

    // Test case 3: Valid user with weak password
    console.log("\n\n### Test Case 3: Valid user with weak password ###");
    await resetPassword("user456", "weakpass");
  } catch (error) {
    console.error("Error during password reset:", error);
    throw error;
  }
}

main();
```
