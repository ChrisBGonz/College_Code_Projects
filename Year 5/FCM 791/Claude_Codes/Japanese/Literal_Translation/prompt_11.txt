import anthropic
import tempfile
import os
from lxml import etree


def create_sample_xml():
    """Create a sample XML file for testing"""
    xml_content = """<?xml version="1.0" encoding="UTF-8"?>
<library>
    <book id="1">
        <title>Python Programming</title>
        <author>John Doe</author>
        <year>2023</year>
    </book>
    <book id="2">
        <title>Machine Learning Basics</title>
        <author>Jane Smith</author>
        <year>2022</year>
    </book>
    <book id="3">
        <title>Data Science with Python</title>
        <author>Bob Johnson</author>
        <year>2024</year>
        <chapters>
            <chapter>
                <title>Introduction to Data Science</title>
                <pages>30</pages>
            </chapter>
            <chapter>
                <title>Data Preprocessing</title>
                <pages>45</pages>
            </chapter>
        </chapters>
    </book>
</library>"""
    
    # Create a temporary file
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as f:
        f.write(xml_content)
        return f.name


def parse_xml_and_extract_titles(xml_file_path: str) -> dict:
    """Parse XML file and extract all title tags using lxml"""
    try:
        # Parse the XML file using lxml
        tree = etree.parse(xml_file_path)
        root = tree.getroot()
        
        # Find all title elements
        titles = root.findall('.//title')
        
        # Extract title information
        title_data = {
            'file_path': xml_file_path,
            'total_titles': len(titles),
            'titles': []
        }
        
        for i, title in enumerate(titles, 1):
            title_info = {
                'index': i,
                'text': title.text,
                'parent_tag': title.getparent().tag if title.getparent() is not None else 'root',
                'parent_attribs': dict(title.getparent().attrib) if title.getparent() is not None else {}
            }
            title_data['titles'].append(title_info)
        
        return title_data
    
    except etree.XMLSyntaxError as e:
        return {'error': f'XML Syntax Error: {str(e)}'}
    except Exception as e:
        return {'error': f'Error parsing XML: {str(e)}'}


def analyze_xml_with_claude(xml_content: str, title_data: dict) -> str:
    """Use Claude to analyze the XML and provide insights about the titles"""
    client = anthropic.Anthropic()
    
    # Prepare the message for Claude
    titles_summary = "\n".join([
        f"- Title {t['index']}: '{t['text']}' (in <{t['parent_tag']}> element)"
        for t in title_data.get('titles', [])
    ])
    
    message_content = f"""I have parsed an XML file and extracted the following information:

XML Content:
{xml_content}

Extracted Titles ({title_data.get('total_titles', 0)} total):
{titles_summary}

Please analyze this XML structure and the extracted titles. Provide:
1. A summary of what the XML file contains
2. The hierarchy of titles found
3. Any observations about the title structure
4. Suggestions for how this data might be used"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": message_content}
        ]
    )
    
    return message.content[0].text


def process_xml_file(xml_file_path: str) -> None:
    """Main function to process an XML file"""
    print(f"\n{'='*60}")
    print(f"Processing XML file: {xml_file_path}")
    print('='*60)
    
    # Check if file exists
    if not os.path.exists(xml_file_path):
        print(f"Error: File '{xml_file_path}' not found.")
        return
    
    # Read the XML content for display
    with open(xml_file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    print("\nXML File Content:")
    print("-"*40)
    print(xml_content)
    print("-"*40)
    
    # Parse XML and extract titles
    print("\nParsing XML with lxml...")
    title_data = parse_xml_and_extract_titles(xml_file_path)
    
    if 'error' in title_data:
        print(f"Error: {title_data['error']}")
        return
    
    print(f"\nFound {title_data['total_titles']} title(s):")
    print("-"*40)
    for title in title_data['titles']:
        print(f"  {title['index']}. '{title['text']}'")
        print(f"     Parent element: <{title['parent_tag']}>", end="")
        if title['parent_attribs']:
            attrs = ', '.join([f'{k}="{v}"' for k, v in title['parent_attribs'].items()])
            print(f" ({attrs})", end="")
        print()
    
    # Analyze with Claude
    print("\nAnalyzing with Claude AI...")
    print("-"*40)
    analysis = analyze_xml_with_claude(xml_content, title_data)
    print("\nClaude's Analysis:")
    print(analysis)


def main():
    """Main entry point"""
    print("XML Title Extractor using lxml and Claude AI")
    print("="*60)
    
    # Create a sample XML file for demonstration
    print("\nCreating sample XML file for demonstration...")
    sample_xml_path = create_sample_xml()
    
    try:
        # Process the sample XML file
        process_xml_file(sample_xml_path)
        
        # Example of how to use with a user-provided file
        print("\n" + "="*60)
        print("To use with your own XML file, call:")
        print("  process_xml_file('path/to/your/file.xml')")
        print("="*60)
        
    finally:
        # Clean up the temporary file
        if os.path.exists(sample_xml_path):
            os.unlink(sample_xml_path)
            print(f"\nTemporary file cleaned up: {sample_xml_path}")


if __name__ == "__main__":
    main()
