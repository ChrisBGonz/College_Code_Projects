import anthropic.sdk.java.Anthropic;
import anthropic.sdk.java.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.*;
import java.nio.file.*;
import java.util.*;
import java.util.regex.*;

public class MavenDependencyUpdater {
    
    private static final Anthropic client = new Anthropic();
    private static final ObjectMapper objectMapper = new ObjectMapper();
    
    public static void main(String[] args) throws Exception {
        String pomFilePath = args.length > 0 ? args[0] : "pom.xml";
        
        System.out.println("Maven Dependency Updater");
        System.out.println("=======================");
        System.out.println("Reading POM file: " + pomFilePath);
        
        // Read the POM file
        String pomContent = readFile(pomFilePath);
        
        // Use Claude to analyze and update dependencies
        updateDependenciesWithClaude(pomContent, pomFilePath);
    }
    
    private static String readFile(String filePath) throws IOException {
        Path path = Paths.get(filePath);
        if (!Files.exists(path)) {
            // Create a sample POM file for demonstration
            String samplePom = createSamplePom();
            Files.writeString(path, samplePom);
            System.out.println("Created sample POM file for demonstration");
        }
        return Files.readString(path);
    }

    private static void updateDependenciesWithClaude(String pomContent, String pomFilePath) throws Exception {
        System.out.println("\nAnalyzing dependencies with Claude...\n");
        
        List<MessageParam> messages = new ArrayList<>();
        
        // Initial message to Claude
        String initialPrompt = """
                You are a Maven dependency update assistant. I will provide you with a POM file content, 
                and you need to help update all dependencies to their latest stable versions.
                
                Here is the current POM file:
                
                ```xml
                """ + pomContent + """
                ```
                
                Please analyze the dependencies and provide:
                1. A list of all dependencies with their current versions
                2. The recommended latest stable versions for each dependency
                3. Any important notes about breaking changes or compatibility issues
                
                Format your response as a structured analysis.
                """;
        
        messages.add(MessageParam.builder()
                .role(Role.USER)
                .content(initialPrompt)
                .build());
        
        // First turn: Get analysis
        System.out.println("Step 1: Analyzing current dependencies...");
        Message analysisResponse = client.messages().create(
                CreateMessageParams.builder()
                        .model("claude-opus-4-5")
                        .maxTokens(2000)
                        .messages(messages)
                        .build()
        );
        
        String analysis = extractTextContent(analysisResponse);
        System.out.println("\nDependency Analysis:");
        System.out.println("--------------------");
        System.out.println(analysis);
        
        // Add assistant response to conversation
        messages.add(MessageParam.builder()
                .role(Role.ASSISTANT)
                .content(analysis)
                .build());
        
        // Second turn: Request updated POM
        String updateRequest = """
                Now please provide the complete updated POM file with all dependencies updated to their 
                latest stable versions. Make sure to:
                1. Keep the same project structure and configuration
                2. Update all dependency versions to the latest stable releases
                3. Ensure compatibility between dependencies
                4. Add comments for any significant version changes
                
                Please provide ONLY the complete XML content of the updated POM file, without any additional explanation.
                """;
        
        messages.add(MessageParam.builder()
                .role(Role.USER)
                .content(updateRequest)
                .build());
        
        System.out.println("\nStep 2: Generating updated POM file...");
        Message updatedPomResponse = client.messages().create(
                CreateMessageParams.builder()
                        .model("claude-opus-4-5")
                        .maxTokens(3000)
                        .messages(messages)
                        .build()
        );
        
        String updatedPomContent = extractTextContent(updatedPomResponse);
        
        // Extract XML from the response
        String cleanedPom = extractXmlContent(updatedPomContent);
        
        // Add assistant response to conversation
        messages.add(MessageParam.builder()
                .role(Role.ASSISTANT)
                .content(updatedPomContent)
                .build());
        
        // Third turn: Validate and get summary
        String validationRequest = """
                Please provide a summary of all the changes made:
                1. List each dependency that was updated with old version -> new version
                2. Highlight any breaking changes or important migration notes
                3. Confirm that the updated POM is valid and all dependencies are compatible
                
                Format this as a clear, concise report.
                """;
        
        messages.add(MessageParam.builder()
                .role(Role.USER)
                .content(validationRequest)
                .build());
        
        System.out.println("\nStep 3: Generating update summary...");
        Message summaryResponse = client.messages().create(
                CreateMessageParams.builder()
                        .model("claude-opus-4-5")
                        .maxTokens(1500)
                        .messages(messages)
                        .build()
        );
        
        String summary = extractTextContent(summaryResponse);
        
        // Save the updated POM file
        if (cleanedPom != null && !cleanedPom.isEmpty()) {
            // Create backup of original
            String backupPath = pomFilePath + ".backup";
            Files.writeString(Paths.get(backupPath), pomContent);
            System.out.println("\nOriginal POM backed up to: " + backupPath);
            
            // Write updated POM
            Files.writeString(Paths.get(pomFilePath), cleanedPom);
            System.out.println("Updated POM written to: " + pomFilePath);
        }
        
        // Display summary
        System.out.println("\nUpdate Summary:");
        System.out.println("---------------");
        System.out.println(summary);
        
        // Show diff
        if (cleanedPom != null && !cleanedPom.isEmpty()) {
            showVersionDiff(pomContent, cleanedPom);
        }
    }
    
    private static String extractTextContent(Message message) {
        StringBuilder content = new StringBuilder();
        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock textBlock) {
                content.append(textBlock.text());
            }
        }
        return content.toString();
    }
    
    private static String extractXmlContent(String response) {
        // Try to extract XML content from code blocks
        Pattern xmlPattern = Pattern.compile("```xml\\s*([\\s\\S]*?)```", Pattern.MULTILINE);
        Matcher matcher = xmlPattern.matcher(response);
        if (matcher.find()) {
            return matcher.group(1).trim();
        }
        
        // Try to find XML directly
        Pattern directXmlPattern = Pattern.compile("<\\?xml[\\s\\S]*</project>", Pattern.MULTILINE);
        Matcher directMatcher = directXmlPattern.matcher(response);
        if (directMatcher.find()) {
            return directMatcher.group().trim();
        }
        
        return null;
    }
    
    private static void showVersionDiff(String originalPom, String updatedPom) {
        System.out.println("\nVersion Changes Detected:");
        System.out.println("-------------------------");
        
        // Extract versions from both POMs
        Pattern versionPattern = Pattern.compile("<version>([^<]+)</version>");
        
        List<String> originalVersions = extractVersions(originalPom, versionPattern);
        List<String> updatedVersions = extractVersions(updatedPom, versionPattern);
        
        // Find dependencies with version info
        Pattern depPattern = Pattern.compile(
                "<dependency>[\\s\\S]*?<groupId>([^<]+)</groupId>[\\s\\S]*?<artifactId>([^<]+)</artifactId>[\\s\\S]*?<version>([^<]+)</version>[\\s\\S]*?</dependency>",
                Pattern.MULTILINE
        );
        
        Map<String, String> originalDeps = extractDependencyVersions(originalPom, depPattern);
        Map<String, String> updatedDeps = extractDependencyVersions(updatedPom, depPattern);
        
        boolean changesFound = false;
        for (Map.Entry<String, String> entry : originalDeps.entrySet()) {
            String dep = entry.getKey();
            String oldVersion = entry.getValue();
            String newVersion = updatedDeps.get(dep);
            
            if (newVersion != null && !oldVersion.equals(newVersion)) {
                System.out.printf("  %-50s %s -> %s%n", dep, oldVersion, newVersion);
                changesFound = true;
            }
        }
        
        if (!changesFound) {
            System.out.println("  No version changes detected in the diff analysis.");
        }
    }
    
    private static List<String> extractVersions(String pom, Pattern pattern) {
        List<String> versions = new ArrayList<>();
        Matcher matcher = pattern.matcher(pom);
        while (matcher.find()) {
            versions.add(matcher.group(1));
        }
        return versions;
    }
    
    private static Map<String, String> extractDependencyVersions(String pom, Pattern pattern) {
        Map<String, String> deps = new LinkedHashMap<>();
        Matcher matcher = pattern.matcher(pom);
        while (matcher.find()) {
            String groupId = matcher.group(1).trim();
            String artifactId = matcher.group(2).trim();
            String version = matcher.group(3).trim();
            deps.put(groupId + ":" + artifactId, version);
        }
        return deps;
    }
}
