import anthropic
import json
import time
import random
from typing import Any


def mock_payment_api(amount: float, currency: str, timeout: float = 2.0) -> dict:
    """
    Mock payment API that simulates various response scenarios.
    In real usage, this would be replaced with actual API calls.
    """
    # Simulate random response times (sometimes exceeding timeout)
    response_time = random.uniform(0.5, 3.5)
    time.sleep(min(response_time, timeout + 0.1))
    
    if response_time > timeout:
        raise TimeoutError(f"API response time {response_time:.2f}s exceeded timeout {timeout}s")
    
    # Simulate successful payment
    return {
        "status": "success",
        "transaction_id": f"TXN_{random.randint(10000, 99999)}",
        "amount": amount,
        "currency": currency,
        "timestamp": time.time()
    }


def process_payment_with_claude(amount: float, currency: str = "USD") -> bool:
    """
    Process payment using Claude AI with tool use for payment API integration.
    Returns True if payment succeeds or if API times out (allowing user to continue).
    Returns False if payment explicitly fails.
    """
    client = anthropic.Anthropic()
    
    # Define the payment tool
    tools = [
        {
            "name": "call_payment_api",
            "description": "Call the external payment API to process a payment transaction. Returns payment result or raises TimeoutError if API doesn't respond within 2 seconds.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "amount": {
                        "type": "number",
                        "description": "The payment amount"
                    },
                    "currency": {
                        "type": "string",
                        "description": "The currency code (e.g., USD, EUR, JPY)"
                    }
                },
                "required": ["amount", "currency"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"Process a payment of {amount} {currency}. If the payment API times out (doesn't respond within 2 seconds), return True to allow the user to continue. If the payment succeeds, return True. If the payment explicitly fails, return False."
        }
    ]
    
    print(f"\nInitiating payment processing for {amount} {currency}...")
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process each tool use in the response
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\nClaude is calling tool: {tool_name}")
                    print(f"Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    # Execute the payment API call
                    if tool_name == "call_payment_api":
                        try:
                            start_time = time.time()
                            result = mock_payment_api(
                                amount=tool_input["amount"],
                                currency=tool_input["currency"],
                                timeout=2.0
                            )
                            elapsed_time = time.time() - start_time
                            
                            print(f"\nPayment API responded in {elapsed_time:.2f}s")
                            print(f"API Result: {json.dumps(result, indent=2)}")
                            
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps(result)
                            })
                            
                        except TimeoutError as e:
                            print(f"\nPayment API TIMEOUT: {str(e)}")
                            print("Allowing user to continue due to timeout...")
                            
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps({
                                    "status": "timeout",
                                    "error": str(e),
                                    "message": "API timed out - user should be allowed to continue"
                                })
                            })
                        
                        except Exception as e:
                            print(f"\nPayment API Error: {str(e)}")
                            
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps({
                                    "status": "error",
                                    "error": str(e)
                                })
                            })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            # Extract the final decision from Claude's response
            final_text = ""
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    final_text += content_block.text
            
            print(f"\nClaude's final response: {final_text}")
            
            # Parse Claude's decision
            final_text_lower = final_text.lower()
            
            # Check for explicit False return
            if "return false" in final_text_lower or "returning false" in final_text_lower:
                print("\nPayment processing result: FAILED (returning False)")
                return False
            
            # Default to True for success or timeout cases
            print("\nPayment processing result: SUCCESS or TIMEOUT (returning True)")
            return True
        
        else:
            # Unexpected stop reason
            print(f"\nUnexpected stop reason: {response.stop_reason}")
            return True


def main():
    """
    Demonstrate the payment processing function with multiple scenarios.
    """
    print("=" * 60)
    print("Payment Processing System with Claude AI")
    print("=" * 60)
    
    # Test scenarios
    test_cases = [
        {"amount": 99.99, "currency": "USD", "description": "Standard USD payment"},
        {"amount": 150.00, "currency": "EUR", "description": "Euro payment"},
        {"amount": 5000, "currency": "JPY", "description": "Japanese Yen payment"},
    ]
    
    results = []
    
    for i, test_case in enumerate(test_cases, 1):
        print(f"\n{'=' * 60}")
        print(f"Test Case {i}: {test_case['description']}")
        print(f"Amount: {test_case['amount']} {test_case['currency']}")
        print("=" * 60)
        
        result = process_payment_with_claude(
            amount=test_case['amount'],
            currency=test_case['currency']
        )
        
        results.append({
            "test": test_case['description'],
            "result": result,
            "user_can_continue": result
        })
        
        print(f"\nFinal Result: {'✓ User can continue' if result else '✗ Payment failed'}")
        
        # Small delay between tests
        time.sleep(1)
    
    # Summary
    print(f"\n{'=' * 60}")
    print("SUMMARY")
    print("=" * 60)
    for r in results:
        status = "✓ CONTINUE" if r["user_can_continue"] else "✗ BLOCKED"
        print(f"{status}: {r['test']}")
    
    print("\nPayment processing demonstration complete!")


if __name__ == "__main__":
    main()
