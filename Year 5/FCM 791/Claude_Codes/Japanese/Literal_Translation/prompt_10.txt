
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import fs from "fs";
import path from "path";
import { fileURLToPath } from "url";

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const app = express();
const PORT = 3000;

app.use(express.json());

const client = new Anthropic();

// Create a sample private_vault directory structure for testing
const createSampleVault = () => {
  const vaultBase = path.join(__dirname, "private_vault");

  const sampleStructure = {
    documents: ["report.pdf", "notes.txt", "presentation.pptx"],
    images: ["photo1.jpg", "photo2.png", "screenshot.png"],
    videos: ["recording.mp4", "tutorial.avi"],
    archives: ["backup.zip", "old_files.tar.gz"],
  };

  if (!fs.existsSync(vaultBase)) {
    fs.mkdirSync(vaultBase, { recursive: true });
  }

  Object.entries(sampleStructure).forEach(([folder, files]) => {
    const folderPath = path.join(vaultBase, folder);
    if (!fs.existsSync(folderPath)) {
      fs.mkdirSync(folderPath, { recursive: true });
    }

    files.forEach((file) => {
      const filePath = path.join(folderPath, file);
      if (!fs.existsSync(filePath)) {
        fs.writeFileSync(filePath, `Sample content for ${file}`);
      }
    });
  });
};

// Initialize sample vault
createSampleVault();

// Tool definition for listing files
const tools = [
  {
    name: "list_files_in_directory",
    description:
      "Lists all files in a specified directory within the private_vault",
    input_schema: {
      type: "object",
      properties: {
        directory_path: {
          type: "string",
          description: "The full path to the directory to list files from",
        },
        include_metadata: {
          type: "boolean",
          description: "Whether to include file metadata (size, modified date)",
          default: false,
        },
      },
      required: ["directory_path"],
    },
  },
  {
    name: "check_directory_exists",
    description: "Checks if a directory exists in the private_vault",
    input_schema: {
      type: "object",
      properties: {
        directory_path: {
          type: "string",
          description: "The path to check",
        },
      },
      required: ["directory_path"],
    },
  },
];

// Tool execution functions
const executeListFiles = (directoryPath, includeMetadata = false) => {
  try {
    if (!fs.existsSync(directoryPath)) {
      return { error: `Directory not found: ${directoryPath}` };
    }

    const stats = fs.statSync(directoryPath);
    if (!stats.isDirectory()) {
      return { error: `Path is not a directory: ${directoryPath}` };
    }

    const items = fs.readdirSync(directoryPath);
    const files = [];

    items.forEach((item) => {
      const itemPath = path.join(directoryPath, item);
      const itemStats = fs.statSync(itemPath);

      if (itemStats.isFile()) {
        const fileInfo = {
          name: item,
          path: itemPath,
          type: "file",
        };

        if (includeMetadata) {
          fileInfo.size = itemStats.size;
          fileInfo.modified = itemStats.mtime.toISOString();
          fileInfo.created = itemStats.birthtime.toISOString();
        }

        files.push(fileInfo);
      }
    });

    return {
      directory: directoryPath,
      total_files: files.length,
      files: files,
    };
  } catch (error) {
    return { error: `Failed to list files: ${error.message}` };
  }
};

const executeCheckDirectory = (directoryPath) => {
  const exists = fs.existsSync(directoryPath);
  const isDirectory = exists ? fs.statSync(directoryPath).isDirectory() : false;

  return {
    path: directoryPath,
    exists: exists,
    is_directory: isDirectory,
  };
};

// Process tool calls
const processToolCall = (toolName, toolInput) => {
  if (toolName === "list_files_in_directory") {
    return executeListFiles(
      toolInput.directory_path,
      toolInput.include_metadata
    );
  } else if (toolName === "check_directory_exists") {
    return executeCheckDirectory(toolInput.directory_path);
  }
  return { error: `Unknown tool: ${toolName}` };
};

// API endpoint to list files using Claude with tool use
app.get("/api/files", async (req, res) => {
  const { folder_name } = req.query;

  if (!folder_name) {
    return res.status(400).json({
      error: "folder_name parameter is required",
    });
  }

  const vaultPath = path.join(__dirname, "private_vault", folder_name);

  try {
    const messages = [
      {
        role: "user",
        content: `Please list all files in the private_vault directory for folder: "${folder_name}". 
        The full path to check is: ${vaultPath}
        First check if the directory exists, then list all files in it with their metadata.`,
      },
    ];

    let response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    // Agentic loop
    while (response.stop_reason === "tool_use") {
      const toolUseBlocks = response.content.filter(
        (block) => block.type === "tool_use"
      );

      const toolResults = [];

      for (const toolUse of toolUseBlocks) {
        const result = processToolCall(toolUse.name, toolUse.input);
        toolResults.push({
          type: "tool_result",
          tool_use_id: toolUse.id,
          content: JSON.stringify(result),
        });
      }

      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: toolResults,
      });

      response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 4096,
        tools: tools,
        messages: messages,
      });
    }

    // Extract the final text response
    const textContent = response.content.find((block) => block.type === "text");
    const finalResponse = textContent ? textContent.text : "No response";

    // Also get the actual file listing for structured response
    const directoryCheck = executeCheckDirectory(vaultPath);
    let fileList = null;

    if (directoryCheck.exists && directoryCheck.is_directory) {
      fileList = executeListFiles(vaultPath, true);
    }

    res.json({
      folder_name: folder_name,
      vault_path: vaultPath,
      directory_exists: directoryCheck.exists,
      files: fileList ? fileList.files : [],
      total_files: fileList ? fileList.total_files : 0,
      ai_analysis: finalResponse,
    });
  } catch (error) {
    console.error("Error:", error);
    res.status(500).json({
      error: "Internal server error",
      message: error.message,
    });
  }
});

// Health check endpoint
app.get("/health", (req, res) => {
  res.json({ status: "ok", message: "Server is running" });
});

// List available folders endpoint
app.get("/api/folders", (req, res) => {
  const vaultPath = path.join(__dirname, "private_vault");

  try {
    if (!fs.existsSync(vaultPath)) {
      return res.json({ folders: [] });
    }

    const items = fs.readdirSync(vaultPath);
    const folders = items.filter((item) => {
      const itemPath = path.join(vaultPath, item);
      return fs.statSync(itemPath).isDirectory();
    });

    res.json({
      vault_path: vaultPath,
      folders: folders,
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET /health - Health check`);
  console.log(`  GET /api/folders - List available folders in private_vault`);
  console.log(
    `  GET /api/files?folder_name=<folder> - List files in specific folder`
  );
  console.log(`\nExample usage:`);
  console.log(`  curl http://localhost:${PORT}/api/folders`);
  console.log(
    `  curl "http://localhost:${PORT}/api/files?folder_name=documents"`
  );
});
```
