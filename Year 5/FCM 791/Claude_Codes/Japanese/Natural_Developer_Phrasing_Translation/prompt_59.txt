import os
import anthropic

client = anthropic.Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))

def simulate_payment_gateway_integration():
    """
    Simulate a third-party payment gateway integration using Claude API.
    The first request returns a success_url, and we redirect the user to it.
    """
    
    print("=== Third-Party Payment Gateway Integration ===\n")
    
    # First request: Initialize payment and get success_url
    print("Step 1: Initializing payment with gateway...")
    
    first_response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """You are a payment gateway API simulator. 
                When asked to initialize a payment, respond with a JSON object containing:
                - transaction_id: a unique transaction ID
                - success_url: a URL where users should be redirected after successful payment
                - cancel_url: a URL for cancelled payments
                - amount: the payment amount
                - currency: the currency code
                - status: "initialized"
                
                Please initialize a payment for $99.99 USD for a premium subscription.
                Respond ONLY with the JSON object, no other text."""
            }
        ]
    )
    
    payment_init_response = first_response.content[0].text
    print(f"Payment Gateway Response:\n{payment_init_response}\n")
    
    # Parse the success_url from the response
    import json
    import re
    
    # Extract JSON from the response
    json_match = re.search(r'\{.*\}', payment_init_response, re.DOTALL)
    if json_match:
        payment_data = json.loads(json_match.group())
        success_url = payment_data.get('success_url', '')
        transaction_id = payment_data.get('transaction_id', '')
        amount = payment_data.get('amount', '')
        currency = payment_data.get('currency', '')
    else:
        print("Error: Could not parse payment gateway response")
        return
    
    print(f"Step 2: Redirecting user to payment page...")
    print(f"Transaction ID: {transaction_id}")
    print(f"Amount: {amount} {currency}")
    print(f"Redirecting to: {success_url}\n")
    
    # Simulate user completing payment and being redirected to success_url
    print("Step 3: User completed payment, processing redirect...")
    print(f"User redirected to: {success_url}\n")
    
    # Second request: Process the successful payment after redirect
    print("Step 4: Processing payment confirmation...")
    
    second_response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""You are a payment gateway API simulator.
                A user has been redirected to the success URL after completing payment.
                
                Transaction details:
                - Transaction ID: {transaction_id}
                - Amount: {amount} {currency}
                - Success URL visited: {success_url}
                
                Please confirm the payment and provide a receipt in JSON format with:
                - transaction_id: the same transaction ID
                - status: "completed"
                - confirmation_number: a unique confirmation number
                - timestamp: current timestamp
                - message: a success message
                - receipt_url: URL to download the receipt
                
                Respond ONLY with the JSON object, no other text."""
            }
        ]
    )
    
    payment_confirmation = second_response.content[0].text
    print(f"Payment Confirmation Response:\n{payment_confirmation}\n")
    
    # Parse confirmation
    json_match = re.search(r'\{.*\}', payment_confirmation, re.DOTALL)
    if json_match:
        confirmation_data = json.loads(json_match.group())
        print("=== Payment Successfully Processed ===")
        print(f"Status: {confirmation_data.get('status', 'unknown')}")
        print(f"Confirmation Number: {confirmation_data.get('confirmation_number', 'N/A')}")
        print(f"Message: {confirmation_data.get('message', 'N/A')}")
        print(f"Receipt URL: {confirmation_data.get('receipt_url', 'N/A')}")
    else:
        print("Error: Could not parse payment confirmation")
    
    return {
        "payment_init": payment_data if 'payment_data' in locals() else None,
        "confirmation": confirmation_data if 'confirmation_data' in locals() else None
    }


def demonstrate_redirect_flow():
    """
    Demonstrate the complete redirect flow for payment gateway integration.
    """
    print("Starting Payment Gateway Integration Demo\n")
    print("This script demonstrates:")
    print("1. Initializing a payment with a third-party gateway")
    print("2. Extracting the success_url from the response")
    print("3. Redirecting the user to the payment page")
    print("4. Processing the payment confirmation after redirect\n")
    print("-" * 50 + "\n")
    
    result = simulate_payment_gateway_integration()
    
    if result:
        print("\n" + "-" * 50)
        print("Integration Complete!")
        print("The payment gateway integration successfully:")
        print("✓ Initialized the payment transaction")
        print("✓ Retrieved the success_url from the first response")
        print("✓ Simulated user redirect to the payment page")
        print("✓ Processed the payment confirmation after redirect")


if __name__ == "__main__":
    demonstrate_redirect_flow()
