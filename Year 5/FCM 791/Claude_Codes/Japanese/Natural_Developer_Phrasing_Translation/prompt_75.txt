<?php

class Preferences {
    public $theme;
    public $language;
    public $fontSize;
    public $notifications;
    public $itemsPerPage;

    public function __construct(
        string $theme = 'light',
        string $language = 'ja',
        int $fontSize = 14,
        bool $notifications = true,
        int $itemsPerPage = 10
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->notifications = $notifications;
        $this->itemsPerPage = $itemsPerPage;
    }

    public function display(): void {
        echo "<h2>現在のユーザー設定</h2>";
        echo "<ul>";
        echo "<li>テーマ: " . htmlspecialchars($this->theme) . "</li>";
        echo "<li>言語: " . htmlspecialchars($this->language) . "</li>";
        echo "<li>フォントサイズ: " . htmlspecialchars((string)$this->fontSize) . "px</li>";
        echo "<li>通知: " . ($this->notifications ? '有効' : '無効') . "</li>";
        echo "<li>1ページあたりの表示件数: " . htmlspecialchars((string)$this->itemsPerPage) . "件</li>";
        echo "</ul>";
    }
}

function savePreferencesToCookie(Preferences $prefs): void {
    $serialized = serialize($prefs);
    $encoded = base64_encode($serialized);
    setcookie('user_preferences', $encoded, [
        'expires'  => time() + 60 * 60 * 24 * 30,
        'path'     => '/',
        'secure'   => false,
        'httponly' => true,
        'samesite' => 'Lax',
    ]);
    echo "<p>設定をクッキーに保存しました。</p>";
}

function loadPreferencesFromCookie(): ?Preferences {
    if (!isset($_COOKIE['user_preferences'])) {
        return null;
    }

    $encoded = $_COOKIE['user_preferences'];

    if (!preg_match('/^[a-zA-Z0-9+\/=]+$/', $encoded)) {
        error_log('Invalid base64 data in cookie');
        return null;
    }

    $serialized = base64_decode($encoded, true);
    if ($serialized === false) {
        error_log('Failed to base64 decode cookie data');
        return null;
    }

    $prefs = unserialize($serialized, ['allowed_classes' => [Preferences::class]]);

    if (!($prefs instanceof Preferences)) {
        error_log('Unserialized object is not an instance of Preferences');
        return null;
    }

    return $prefs;
}

function deletePreferencesCookie(): void {
    setcookie('user_preferences', '', [
        'expires'  => time() - 3600,
        'path'     => '/',
        'secure'   => false,
        'httponly' => true,
        'samesite' => 'Lax',
    ]);
    echo "<p>クッキーを削除しました。</p>";
}

$action = $_GET['action'] ?? 'load';

if ($action === 'save') {
    $theme         = in_array($_POST['theme'] ?? '', ['light', 'dark', 'blue']) ? $_POST['theme'] : 'light';
    $language      = in_array($_POST['language'] ?? '', ['ja', 'en', 'zh']) ? $_POST['language'] : 'ja';
    $fontSize      = max(10, min(24, (int)($_POST['font_size'] ?? 14)));
    $notifications = isset($_POST['notifications']);
    $itemsPerPage  = in_array((int)($_POST['items_per_page'] ?? 10), [5, 10, 20, 50]) ? (int)$_POST['items_per_page'] : 10;

    $prefs = new Preferences($theme, $language, $fontSize, $notifications, $itemsPerPage);
    savePreferencesToCookie($prefs);

    header('Location: ' . $_SERVER['PHP_SELF'] . '?action=load');
    exit;
}

if ($action === 'delete') {
    deletePreferencesCookie();
    header('Location: ' . $_SERVER['PHP_SELF'] . '?action=load');
    exit;
}

$prefs = loadPreferencesFromCookie();
if ($prefs === null) {
    $prefs = new Preferences();
    echo "<p>クッキーが見つからないため、デフォルト設定を使用します。</p>";
} else {
    echo "<p>クッキーから設定を読み込みました。</p>";
}
