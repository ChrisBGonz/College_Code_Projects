using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class MFAAuditLogger
{
    private readonly AnthropicClient _client;
    private readonly List<Message> _conversationHistory;

    public MFAAuditLogger()
    {
        _client = new AnthropicClient();
        _conversationHistory = new List<Message>();
    }

    public async Task<string> AnalyzeMFAChangeAsync(string userAction, string userId, string ipAddress)
    {
        var userMessage = $@"
ユーザーがMFA設定を変更しました。以下の情報を分析して、監査ログエントリを生成してください：

ユーザーID: {userId}
IPアドレス: {ipAddress}
アクション: {userAction}
タイムスタンプ: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss UTC}

以下の形式で監査ログエントリを生成してください：
1. イベントの重要度（INFO/WARNING/CRITICAL）
2. 詳細な説明
3. セキュリティリスク評価
4. 推奨されるフォローアップアクション";

        _conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = userMessage
        });

        var response = await _client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Messages = _conversationHistory
        });

        var assistantMessage = response.Content[0].Text;
        
        _conversationHistory.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = assistantMessage
        });

        return assistantMessage;
    }

    public async Task<string> GetAuditSummaryAsync()
    {
        var summaryRequest = "これまでの会話で記録されたMFA変更イベントのサマリーを作成してください。重要なパターンやセキュリティ上の懸念事項があれば強調してください。";

        _conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = summaryRequest
        });

        var response = await _client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Messages = _conversationHistory
        });

        var summaryMessage = response.Content[0].Text;
        
        _conversationHistory.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = summaryMessage
        });

        return summaryMessage;
    }

    public async Task<string> AskFollowUpQuestionAsync(string question)
    {
        _conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = question
        });

        var response = await _client.Messages.GetClaudeMessageAsync(new MessageParameters
        {
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Messages = _conversationHistory
        });

        var answer = response.Content[0].Text;
        
        _conversationHistory.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = answer
        });

        return answer;
    }
}

class AuditLogEntry
{
    public string EventId { get; set; } = Guid.NewGuid().ToString();
    public DateTime Timestamp { get; set; } = DateTime.UtcNow;
    public string UserId { get; set; } = string.Empty;
    public string Action { get; set; } = string.Empty;
    public string IpAddress { get; set; } = string.Empty;
    public string Analysis { get; set; } = string.Empty;
    public string Severity { get; set; } = string.Empty;

    public override string ToString()
    {
        return $"[{Timestamp:yyyy-MM-dd HH:mm:ss UTC}] EventID: {EventId} | User: {UserId} | Action: {Action} | IP: {IpAddress} | Severity: {Severity}";
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== MFA監査ログシステム ===\n");

        var logger = new MFAAuditLogger();
        var auditLog = new List<AuditLogEntry>();

        // シミュレートされたMFA変更イベント
        var mfaEvents = new[]
        {
            new { UserId = "user001", Action = "MFAを無効化", IpAddress = "192.168.1.100" },
            new { UserId = "user002", Action = "認証アプリを新しいデバイスに追加", IpAddress = "10.0.0.50" },
            new { UserId = "user003", Action = "SMSベースのMFAからアプリベースのMFAに変更", IpAddress = "172.16.0.25" }
        };

        Console.WriteLine("MFA変更イベントを処理中...\n");

        foreach (var mfaEvent in mfaEvents)
        {
            Console.WriteLine($"処理中: ユーザー {mfaEvent.UserId} - {mfaEvent.Action}");
            Console.WriteLine(new string('-', 60));

            var analysis = await logger.AnalyzeMFAChangeAsync(
                mfaEvent.Action,
                mfaEvent.UserId,
                mfaEvent.IpAddress
            );

            var logEntry = new AuditLogEntry
            {
                UserId = mfaEvent.UserId,
                Action = mfaEvent.Action,
                IpAddress = mfaEvent.IpAddress,
                Analysis = analysis
            };

            auditLog.Add(logEntry);

            Console.WriteLine($"監査ログエントリ: {logEntry}");
            Console.WriteLine("\nAI分析結果:");
            Console.WriteLine(analysis);
            Console.WriteLine("\n");

            // APIレート制限を避けるための短い遅延
            await Task.Delay(1000);
        }

        // 監査サマリーを取得
        Console.WriteLine("=== 監査サマリーを生成中 ===\n");
        var summary = await logger.GetAuditSummaryAsync();
        Console.WriteLine("監査サマリー:");
        Console.WriteLine(summary);
        Console.WriteLine("\n");

        // フォローアップの質問
        Console.WriteLine("=== フォローアップ分析 ===\n");
        var followUpQuestion = "これらのMFA変更イベントに基づいて、組織のセキュリティポリシーを強化するための具体的な推奨事項を3つ提供してください。";
        
        Console.WriteLine($"質問: {followUpQuestion}\n");
        var followUpAnswer = await logger.AskFollowUpQuestionAsync(followUpQuestion);
        Console.WriteLine("回答:");
        Console.WriteLine(followUpAnswer);
        Console.WriteLine("\n");

        // 最終的な監査ログサマリーを表示
        Console.WriteLine("=== 記録された監査ログエントリ ===\n");
        foreach (var entry in auditLog)
        {
            Console.WriteLine(entry);
        }

        Console.WriteLine("\n監査ログ処理が完了しました。");
    }
}
