package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Define the transfer tool
	transferTool := anthropic.ToolParam{
		Name:        "transfer_money",
		Description: anthropic.String("Process a money transfer between two accounts"),
		InputSchema: anthropic.ToolInputSchemaParam{
			Type: "object",
			Properties: map[string]interface{}{
				"sender_id": map[string]interface{}{
					"type":        "string",
					"description": "The ID of the sender account",
				},
				"receiver_id": map[string]interface{}{
					"type":        "string",
					"description": "The ID of the receiver account",
				},
				"amount": map[string]interface{}{
					"type":        "number",
					"description": "The amount to transfer",
				},
			},
			Required: []string{"sender_id", "receiver_id", "amount"},
		},
	}

	// Create a message asking Claude to process a transfer
	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     []anthropic.ToolParam{transferTool},
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: "Please process a money transfer from account USER123 to account USER456 for $500.00",
						},
					},
				},
			},
		},
	})

	if err != nil {
		log.Fatalf("Error creating message: %v", err)
	}

	// Process the response
	for _, block := range message.Content {
		switch block.Type {
		case "text":
			fmt.Printf("Claude says: %s\n", block.Text)
		case "tool_use":
			if block.Name == "transfer_money" {
				// Extract transfer parameters
				input := block.Input
				senderID, _ := input["sender_id"].(string)
				receiverID, _ := input["receiver_id"].(string)
				amount, _ := input["amount"].(float64)

				// Process the transfer with logging
				processTransfer(senderID, receiverID, amount)
			}
		}
	}
}

// processTransfer handles the money transfer and logs the details
