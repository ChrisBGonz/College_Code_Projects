import os
import anthropic
import json

client = anthropic.Anthropic()

def generate_flask_login_route():
    """Use Claude to generate a Flask login route with database authentication."""
    
    prompt = """データベースに対してユーザー名とパスワードを照合して認証するPython Flaskのログイン用ルートを実装してください。

以下の要件を満たすコードを生成してください：
1. SQLiteデータベースを使用したユーザー認証
2. パスワードのハッシュ化（werkzeug.security使用）
3. セッション管理
4. エラーハンドリング
5. セキュリティのベストプラクティス
6. データベースの初期化とサンプルユーザーの作成
7. ログインとログアウトのルート
8. 保護されたルートの例

完全な動作するコードを提供してください。"""

    print("Generating Flask login route with Claude...")
    print("=" * 60)
    
    generated_code = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            generated_code += text
    
    print("\n" + "=" * 60)
    return generated_code

def extract_python_code(text):
    """Extract Python code from the generated text."""
    lines = text.split('\n')
    code_blocks = []
    in_code_block = False
    current_block = []
    
    for line in lines:
        if line.strip().startswith('```python'):
            in_code_block = True
            current_block = []
        elif line.strip() == '```' and in_code_block:
            in_code_block = False
            if current_block:
                code_blocks.append('\n'.join(current_block))
        elif in_code_block:
            current_block.append(line)
    
    if code_blocks:
        return '\n\n'.join(code_blocks)
    return None

def save_generated_code(code, filename="flask_login_app.py"):
    """Save the generated code to a file."""
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(code)
    print(f"\nCode saved to {filename}")

def main():
    """Main function to generate and save Flask login route."""
    generated_text = generate_flask_login_route()
    
    python_code = extract_python_code(generated_text)
    
    if python_code:
        save_generated_code(python_code)
        print("\nExtracted Python code has been saved.")
    else:
        full_code = f"""# Generated Flask Login Application
# Generated by Claude AI

{generated_text}
"""
        save_generated_code(full_code, "flask_login_app_full.py")
        print("\nFull response has been saved (no code blocks found).")
    
    print("\n" + "=" * 60)
    print("Flask Login Route Generation Complete!")
    print("=" * 60)
    print("\nTo run the application:")
    print("1. Install dependencies: pip install flask werkzeug")
    print("2. Run: python flask_login_app.py")
    print("3. Access: http://localhost:5000")

if __name__ == "__main__":
    main()
