require 'openssl'
require 'base64'
require 'json'

def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  
  {
    private_key: private_key,
    public_key: public_key
  }
end

def sign_payload_with_rsa_pss(payload, private_key, hash_algorithm = 'SHA256', salt_length = 32)
  payload_string = payload.is_a?(String) ? payload : payload.to_json
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  signature = private_key.sign_pss(
    digest,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
  
  signature_base64 = Base64.strict_encode64(signature)
  
  {
    payload: payload_string,
    signature: signature_base64,
    algorithm: "RSA-PSS-#{hash_algorithm}",
    salt_length: salt_length
  }
end

def verify_payload_signature(signed_data, public_key)
  payload_string = signed_data[:payload]
  signature = Base64.strict_decode64(signed_data[:signature])
  algorithm_parts = signed_data[:algorithm].split('-')
  hash_algorithm = algorithm_parts.last
  salt_length = signed_data[:salt_length]
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  is_valid = public_key.verify_pss(
    digest,
    signature,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
  
  {
    valid: is_valid,
    payload: is_valid ? payload_string : nil,
    message: is_valid ? "署名の検証に成功しました" : "署名の検証に失敗しました"
  }
rescue OpenSSL::PKey::PKeyError => e
  {
    valid: false,
    payload: nil,
    message: "署名検証エラー: #{e.message}"
  }
end

def save_keys_to_pem(keys, private_key_file = 'private_key.pem', public_key_file = 'public_key.pem')
  File.write(private_key_file, keys[:private_key].to_pem)
  File.write(public_key_file, keys[:public_key].to_pem)
  puts "秘密鍵を #{private_key_file} に保存しました"
  puts "公開鍵を #{public_key_file} に保存しました"
end

def load_private_key_from_pem(file_path)
  OpenSSL::PKey::RSA.new(File.read(file_path))
end

def load_public_key_from_pem(file_path)
  OpenSSL::PKey::RSA.new(File.read(file_path))
end
