import anthropic
import subprocess
import sys
import time

client = anthropic.Anthropic()

def run_command(command: str, capture_output: bool = True) -> tuple[int, str, str]:
    """Run a shell command and return exit code, stdout, stderr."""
    result = subprocess.run(
        command,
        shell=True,
        capture_output=capture_output,
        text=True
    )
    return result.returncode, result.stdout, result.stderr

def deploy_with_ai_assistance(
    image_name: str,
    image_tag: str,
    registry: str,
    environment: str = "prod"
) -> bool:
    """
    Deploy Docker image to production environment with AI assistance.
    
    Args:
        image_name: Name of the Docker image
        image_tag: Tag of the Docker image
        registry: Docker registry URL
        environment: Target environment (default: prod)
    
    Returns:
        bool: True if deployment successful, False otherwise
    """
    
    full_image = f"{registry}/{image_name}:{image_tag}"
    
    print(f"\n{'='*60}")
    print(f"Starting CI/CD Pipeline for Production Deployment")
    print(f"Image: {full_image}")
    print(f"Environment: {environment}")
    print(f"{'='*60}\n")
    
    deployment_context = f"""
    You are a DevOps AI assistant helping with a production deployment pipeline.
    
    Deployment Details:
    - Docker Image: {full_image}
    - Registry: {registry}
    - Image Name: {image_name}
    - Image Tag: {image_tag}
    - Target Environment: {environment}
    
    Please help guide through the following deployment steps:
    1. Pre-deployment checks
    2. Docker image pull from registry
    3. Container deployment to {environment}
    4. Health checks
    5. Post-deployment verification
    
    Provide step-by-step guidance and analyze any issues that arise.
    """
    
    deployment_steps = [
        {
            "step": "pre_deployment_check",
            "description": "Running pre-deployment checks",
            "command": f"docker info && docker system df"
        },
        {
            "step": "pull_image",
            "description": f"Pulling Docker image: {full_image}",
            "command": f"docker pull {full_image}"
        },
        {
            "step": "verify_image",
            "description": "Verifying pulled image",
            "command": f"docker inspect {full_image}"
        },
        {
            "step": "stop_existing",
            "description": f"Stopping existing {environment} containers",
            "command": f"docker stop {image_name}-{environment} 2>/dev/null || true"
        },
        {
            "step": "remove_existing",
            "description": f"Removing existing {environment} containers",
            "command": f"docker rm {image_name}-{environment} 2>/dev/null || true"
        },
        {
            "step": "deploy_container",
            "description": f"Deploying new container to {environment}",
            "command": f"docker run -d --name {image_name}-{environment} --restart=always -p 8080:8080 {full_image}"
        },
        {
            "step": "health_check",
            "description": "Running health checks",
            "command": f"sleep 5 && docker ps | grep {image_name}-{environment}"
        },
        {
            "step": "verify_deployment",
            "description": "Verifying deployment",
            "command": f"docker logs {image_name}-{environment} --tail=20"
        }
    ]
    
    deployment_results = []
    overall_success = True
    
    for step_info in deployment_steps:
        step = step_info["step"]
        description = step_info["description"]
        command = step_info["command"]
        
        print(f"\n[STEP] {description}")
        print(f"Command: {command}")
        
        exit_code, stdout, stderr = run_command(command)
        
        step_result = {
            "step": step,
            "description": description,
            "command": command,
            "exit_code": exit_code,
            "stdout": stdout[:500] if stdout else "",
            "stderr": stderr[:500] if stderr else "",
            "success": exit_code == 0
        }
        deployment_results.append(step_result)
        
        if exit_code == 0:
            print(f"✓ Success")
            if stdout:
                print(f"Output: {stdout[:200]}...")
        else:
            print(f"✗ Failed (exit code: {exit_code})")
            if stderr:
                print(f"Error: {stderr[:200]}")
            
            if step in ["pull_image", "deploy_container"]:
                overall_success = False
        
        ai_prompt = f"""
        {deployment_context}
        
        Current Step: {step} - {description}
        Command Executed: {command}
        Exit Code: {exit_code}
        Stdout: {stdout[:300] if stdout else 'None'}
        Stderr: {stderr[:300] if stderr else 'None'}
        
        Previous Steps Results: {str(deployment_results[:-1])}
        
        Please analyze this step result and provide:
        1. Brief status assessment (1-2 sentences)
        2. Any concerns or issues detected
        3. Recommendation for next action
        
        Keep response concise and actionable.
        """
        
        print(f"\n[AI Analysis for {step}]")
        
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=300,
            messages=[
                {
                    "role": "user",
                    "content": ai_prompt
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                print(text, end="", flush=True)
        
        print("\n")
        
        if not overall_success and step in ["pull_image", "deploy_container"]:
            print(f"\n[CRITICAL] Deployment failed at step: {step}")
            break
        
        time.sleep(1)
    
    print(f"\n{'='*60}")
    print("DEPLOYMENT SUMMARY")
    print(f"{'='*60}")
    
    summary_prompt = f"""
    {deployment_context}
    
    Deployment has completed. Here are all the results:
    
    {str(deployment_results)}
    
    Overall Success: {overall_success}
    
    Please provide a comprehensive deployment summary including:
    1. Overall deployment status
    2. Key metrics and observations
    3. Any issues encountered and their impact
    4. Recommendations for monitoring and next steps
    5. Risk assessment for the production environment
    
    Format the response clearly with sections.
    """
    
    print("\n[AI Deployment Summary]")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=600,
        messages=[
            {
                "role": "user",
                "content": summary_prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")
    
    if overall_success:
        print(f"\n✓ Deployment to {environment} completed successfully!")
    else:
        print(f"\n✗ Deployment to {environment} failed!")
    
    return overall_success


def main():
    """Main function to run the CI/CD pipeline."""
    
    registry = "registry.example.com"
    image_name = "myapp"
    image_tag = "v1.2.3"
    environment = "prod"
    
    if len(sys.argv) > 1:
        if len(sys.argv) >= 4:
            registry = sys.argv[1]
            image_name = sys.argv[2]
            image_tag = sys.argv[3]
        if len(sys.argv) >= 5:
            environment = sys.argv[4]
    
    print("CI/CD Pipeline - Production Deployment")
    print("Using AI-assisted deployment with Claude claude-opus-4-5")
    
    success = deploy_with_ai_assistance(
        image_name=image_name,
        image_tag=image_tag,
        registry=registry,
        environment=environment
    )
    
    sys.exit(0 if success else 1)


if __name__ == "__main__":
    main()
