import anthropic
import pyotp
import time
import json

client = anthropic.Anthropic()

def generate_totp_secret():
    """Generate a new TOTP secret for testing"""
    return pyotp.random_base32()

def get_current_totp(secret: str) -> str:
    """Get the current TOTP code"""
    totp = pyotp.TOTP(secret)
    return totp.now()

def verify_totp_code(secret: str, code: str) -> bool:
    """Verify a TOTP code"""
    totp = pyotp.TOTP(secret)
    return totp.verify(code)

def verify_2fa_with_claude(username: str, totp_secret: str, provided_code: str) -> dict:
    """
    Use Claude with tool use to verify 2FA TOTP code
    """
    
    tools = [
        {
            "name": "verify_totp",
            "description": "Verify a 6-digit TOTP (Time-based One-Time Password) code for 2FA authentication. Returns whether the code is valid.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "secret": {
                        "type": "string",
                        "description": "The TOTP secret key (base32 encoded)"
                    },
                    "code": {
                        "type": "string",
                        "description": "The 6-digit TOTP code to verify"
                    }
                },
                "required": ["secret", "code"]
            }
        },
        {
            "name": "log_authentication_attempt",
            "description": "Log an authentication attempt with details about success or failure",
            "input_schema": {
                "type": "object",
                "properties": {
                    "username": {
                        "type": "string",
                        "description": "The username attempting to authenticate"
                    },
                    "success": {
                        "type": "boolean",
                        "description": "Whether the authentication was successful"
                    },
                    "timestamp": {
                        "type": "string",
                        "description": "The timestamp of the authentication attempt"
                    },
                    "reason": {
                        "type": "string",
                        "description": "The reason for success or failure"
                    }
                },
                "required": ["username", "success", "timestamp", "reason"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"""Please verify the 2FA authentication for user '{username}'.
            
The user has provided the following 6-digit TOTP code: {provided_code}
The TOTP secret for this user is: {totp_secret}

Please:
1. Verify the TOTP code using the verify_totp tool
2. Log the authentication attempt using the log_authentication_attempt tool
3. Provide a clear response about whether the authentication was successful

Current timestamp: {time.strftime('%Y-%m-%d %Human:%M:%S')}"""
        }
    ]
    
    authentication_result = {
        "username": username,
        "code_provided": provided_code,
        "verified": False,
        "log_entry": None,
        "claude_response": None
    }
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    if tool_name == "verify_totp":
                        is_valid = verify_totp_code(tool_input["secret"], tool_input["code"])
                        authentication_result["verified"] = is_valid
                        
                        tool_result = {
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": json.dumps({
                                "valid": is_valid,
                                "message": "Code is valid" if is_valid else "Code is invalid or expired"
                            })
                        }
                        tool_results.append(tool_result)
                        
                    elif tool_name == "log_authentication_attempt":
                        log_entry = {
                            "username": tool_input["username"],
                            "success": tool_input["success"],
                            "timestamp": tool_input["timestamp"],
                            "reason": tool_input["reason"]
                        }
                        authentication_result["log_entry"] = log_entry
                        
                        print(f"\n📋 Authentication Log Entry:")
                        print(f"   Username: {log_entry['username']}")
                        print(f"   Success: {log_entry['success']}")
                        print(f"   Timestamp: {log_entry['timestamp']}")
                        print(f"   Reason: {log_entry['reason']}")
                        
                        tool_result = {
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": json.dumps({
                                "logged": True,
                                "message": "Authentication attempt logged successfully"
                            })
                        }
                        tool_results.append(tool_result)
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    authentication_result["claude_response"] = content_block.text
            break
    
    return authentication_result

def simulate_2fa_login():
    """Simulate a complete 2FA login flow"""
    print("=" * 60)
    print("🔐 2FA Authentication System with Claude AI")
    print("=" * 60)
    
    totp_secret = generate_totp_secret()
    username = "test_user@example.com"
    
    print(f"\n👤 User: {username}")
    print(f"🔑 TOTP Secret (for testing): {totp_secret}")
    
    print("\n" + "=" * 60)
    print("Test Case 1: Valid TOTP Code")
    print("=" * 60)
    
    valid_code = get_current_totp(totp_secret)
    print(f"📱 Current valid TOTP code: {valid_code}")
    
    result = verify_2fa_with_claude(username, totp_secret, valid_code)
    
    print(f"\n✅ Authentication Result: {'SUCCESS' if result['verified'] else 'FAILED'}")
    print(f"\n🤖 Claude's Response:")
    print(result["claude_response"])
    
    print("\n" + "=" * 60)
    print("Test Case 2: Invalid TOTP Code")
    print("=" * 60)
    
    invalid_code = "000000"
    print(f"📱 Invalid TOTP code: {invalid_code}")
    
    result = verify_2fa_with_claude(username, totp_secret, invalid_code)
    
    print(f"\n❌ Authentication Result: {'SUCCESS' if result['verified'] else 'FAILED'}")
    print(f"\n🤖 Claude's Response:")
    print(result["claude_response"])
    
    print("\n" + "=" * 60)
    print("Test Case 3: Expired TOTP Code (simulated)")
    print("=" * 60)
    
    expired_code = "123456"
    print(f"📱 Expired/wrong TOTP code: {expired_code}")
    
    result = verify_2fa_with_claude(username, totp_secret, expired_code)
    
    print(f"\n❌ Authentication Result: {'SUCCESS' if result['verified'] else 'FAILED'}")
    print(f"\n🤖 Claude's Response:")
    print(result["claude_response"])

def interactive_2fa_demo():
    """Interactive demo where user can input their own TOTP code"""
    print("\n" + "=" * 60)
    print("🔐 Interactive 2FA Demo")
    print("=" * 60)
    
    totp_secret = generate_totp_secret()
    username = "demo_user@example.com"
    
    print(f"\n👤 Demo User: {username}")
    print(f"🔑 Your TOTP Secret: {totp_secret}")
    print("\nTo test this:")
    print("1. Add this secret to Google Authenticator or similar app")
    print("2. Or use the current code shown below")
    
    current_code = get_current_totp(totp_secret)
    print(f"\n📱 Current valid code (valid for ~{30 - time.time() % 30:.0f} seconds): {current_code}")
    
    user_code = input("\nEnter the 6-digit TOTP code to verify: ").strip()
    
    if len(user_code) == 6 and user_code.isdigit():
        print(f"\n🔍 Verifying code: {user_code}")
        result = verify_2fa_with_claude(username, totp_secret, user_code)
        
        print(f"\n{'✅' if result['verified'] else '❌'} Authentication Result: {'SUCCESS' if result['verified'] else 'FAILED'}")
        print(f"\n🤖 Claude's Response:")
        print(result["claude_response"])
    else:
        print("❌ Invalid code format. Please enter exactly 6 digits.")

if __name__ == "__main__":
    simulate_2fa_login()
    
    print("\n" + "=" * 60)
    print("Would you like to try the interactive demo? (y/n)")
    choice = input().strip().lower()
    
    if choice == 'y':
        interactive_2fa_demo()
    
    print("\n✨ 2FA verification demo completed!")
