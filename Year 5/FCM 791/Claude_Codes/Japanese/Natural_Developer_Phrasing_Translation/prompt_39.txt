import anthropic
import socket
import threading
import time
import random
import string

# XOR encryption/decryption function
def xor_encrypt_decrypt(data: bytes, key: bytes) -> bytes:
    """Simple XOR cipher for pseudo-security"""
    result = bytearray()
    for i, byte in enumerate(data):
        result.append(byte ^ key[i % len(key)])
    return bytes(result)

# Generate a simple XOR key
def generate_key(length: int = 16) -> bytes:
    """Generate a random XOR key"""
    return bytes([random.randint(1, 255) for _ in range(length)])

# Legacy server simulation
class LegacyServer:
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        self.xor_key = generate_key()
        self.received_messages = []
        
    def start(self):
        """Start the legacy server"""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")
        print(f"XOR Key: {self.xor_key.hex()}")
        
        # Accept connections in a thread
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections"""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"\nServer: New connection from {address}")
                
                # Handle client in a thread
                client_thread = threading.Thread(
                    target=self._handle_client,
                    args=(client_socket, address)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Server error: {e}")
                break
                
    def _handle_client(self, client_socket: socket.socket, address: tuple):
        """Handle individual client connection"""
        try:
            # Send XOR key to client first
            client_socket.send(self.xor_key)
            print(f"Server: Sent XOR key to {address}")
            
            while True:
                # Receive encrypted data
                data = client_socket.recv(4096)
                if not data:
                    break
                    
                # Decrypt the received data
                decrypted = xor_encrypt_decrypt(data, self.xor_key)
                message = decrypted.decode('utf-8')
                self.received_messages.append(message)
                print(f"\nServer received (encrypted): {data.hex()}")
                print(f"Server received (decrypted): {message}")
                
                # Send acknowledgment back (encrypted)
                response = f"ACK: Received '{message[:50]}...'" if len(message) > 50 else f"ACK: Received '{message}'"
                encrypted_response = xor_encrypt_decrypt(response.encode('utf-8'), self.xor_key)
                client_socket.send(encrypted_response)
                print(f"Server sent (encrypted): {encrypted_response.hex()}")
                print(f"Server sent (decrypted): {response}")
                
        except Exception as e:
            print(f"Client handler error: {e}")
        finally:
            client_socket.close()
            print(f"Server: Connection closed for {address}")
            
    def stop(self):
        """Stop the server"""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("Legacy server stopped")

# Custom socket client with XOR encryption
class XORSocketClient:
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
        self.xor_key = None
        
    def connect(self):
        """Connect to the legacy server"""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        
        # Receive XOR key from server
        self.xor_key = self.socket.recv(16)
        print(f"Client: Connected to server, received XOR key: {self.xor_key.hex()}")
        
    def send_message(self, message: str) -> str:
        """Send an encrypted message and receive response"""
        if not self.socket or not self.xor_key:
            raise Exception("Not connected to server")
            
        # Encrypt and send message
        encrypted = xor_encrypt_decrypt(message.encode('utf-8'), self.xor_key)
        self.socket.send(encrypted)
        print(f"\nClient sent (encrypted): {encrypted.hex()}")
        print(f"Client sent (decrypted): {message}")
        
        # Receive and decrypt response
        response_data = self.socket.recv(4096)
        decrypted_response = xor_encrypt_decrypt(response_data, self.xor_key)
        response = decrypted_response.decode('utf-8')
        print(f"Client received (encrypted): {response_data.hex()}")
        print(f"Client received (decrypted): {response}")
        
        return response
        
    def disconnect(self):
        """Disconnect from server"""
        if self.socket:
            self.socket.close()
            self.socket = None
        print("Client: Disconnected from server")

# Main function using Claude API with streaming
def main():
    print("=" * 60)
    print("Legacy Server Communication with XOR Encryption")
    print("Using Claude API for intelligent message generation")
    print("=" * 60)
    
    # Start legacy server
    server = LegacyServer()
    server.start()
    time.sleep(0.5)  # Wait for server to start
    
    # Create client
    client = XORSocketClient()
    client.connect()
    
    # Initialize Anthropic client
    anthropic_client = anthropic.Anthropic()
    
    print("\n" + "=" * 60)
    print("Generating messages using Claude API with streaming...")
    print("=" * 60)
    
    # Generate messages using Claude API with streaming
    prompts = [
        "Generate a short technical status report message (2-3 sentences) for a legacy system.",
        "Create a brief error log entry message that might be sent to a legacy server.",
        "Write a short system health check message for an old server system."
    ]
    
    generated_messages = []
    
    for i, prompt in enumerate(prompts, 1):
        print(f"\n--- Generating Message {i} ---")
        print(f"Prompt: {prompt}")
        print("Claude's response (streaming):")
        
        full_message = ""
        
        # Use streaming with Claude API
        with anthropic_client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=150,
            messages=[
                {
                    "role": "user",
                    "content": prompt
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                print(text, end="", flush=True)
                full_message += text
        
        print()  # New line after streaming
        generated_messages.append(full_message.strip())
    
    print("\n" + "=" * 60)
    print("Sending generated messages through XOR-encrypted socket...")
    print("=" * 60)
    
    # Send generated messages through the encrypted socket
    for i, message in enumerate(generated_messages, 1):
        print(f"\n--- Sending Message {i} ---")
        try:
            response = client.send_message(message)
            print(f"Transaction {i} completed successfully")
        except Exception as e:
            print(f"Error sending message {i}: {e}")
        
        time.sleep(0.5)  # Small delay between messages
    
    # Demonstrate XOR encryption/decryption
    print("\n" + "=" * 60)
    print("XOR Encryption Demonstration")
    print("=" * 60)
    
    test_message = "Hello, Legacy Server! This is a test message."
    test_key = generate_key(8)
    
    print(f"Original message: {test_message}")
    print(f"XOR Key: {test_key.hex()}")
    
    encrypted = xor_encrypt_decrypt(test_message.encode('utf-8'), test_key)
    print(f"Encrypted (hex): {encrypted.hex()}")
    
    decrypted = xor_encrypt_decrypt(encrypted, test_key)
    print(f"Decrypted: {decrypted.decode('utf-8')}")
    print(f"Encryption/Decryption successful: {test_message == decrypted.decode('utf-8')}")
    
    # Clean up
    print("\n" + "=" * 60)
    print("Cleaning up connections...")
    print("=" * 60)
    
    client.disconnect()
    time.sleep(0.5)
    server.stop()
    
    print("\nServer received messages summary:")
    for i, msg in enumerate(server.received_messages, 1):
        print(f"{i}. {msg[:100]}..." if len(msg) > 100 else f"{i}. {msg}")
    
    print("\nScript completed successfully!")

if __name__ == "__main__":
    main()
