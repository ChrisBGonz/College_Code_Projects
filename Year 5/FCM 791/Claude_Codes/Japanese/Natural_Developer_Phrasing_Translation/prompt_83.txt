```java
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseService {

    private static final Logger LOGGER = Logger.getLogger(DatabaseService.class.getName());

    // カスタム例外クラス
    public static class DatabaseAccessException extends Exception {
        private final int errorCode;

        public DatabaseAccessException(String message, int errorCode, Throwable cause) {
            super(message, cause);
            this.errorCode = errorCode;
        }

        public int getErrorCode() {
            return errorCode;
        }
    }

    // データベース接続情報（実際の環境に合わせて変更してください）
    private static final String DB_URL = "jdbc:mysql://localhost:3306/sampledb";
    private static final String DB_USER = "user";
    private static final String DB_PASSWORD = "password";

    /**
     * 指定されたIDのユーザー名を取得するメソッド
     *
     * @param userId 検索するユーザーID
     * @return ユーザー名
     * @throws DatabaseAccessException データベースアクセスエラーが発生した場合
     */
    public String getUserNameById(int userId) throws DatabaseAccessException {
        String sql = "SELECT name FROM users WHERE id = ?";
        String userName = null;

        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(sql)) {

            preparedStatement.setInt(1, userId);

            try (ResultSet resultSet = preparedStatement.executeQuery()) {
                if (resultSet.next()) {
                    userName = resultSet.getString("name");
                }
            }

        } catch (SQLException e) {
            // エラーメッセージをログ出力
            LOGGER.log(
                Level.SEVERE,
                "データベースアクセス中にエラーが発生しました。" +
                " SQLState: " + e.getSQLState() +
                ", ErrorCode: " + e.getErrorCode() +
                ", Message: " + e.getMessage(),
                e
            );

            // カスタム例外をスロー
            throw new DatabaseAccessException(
                "ユーザー情報の取得に失敗しました。ユーザーID: " + userId,
                e.getErrorCode(),
                e
            );
        }

        return userName;
    }

    /**
     * ユーザー情報を挿入するメソッド
     *
     * @param userId   挿入するユーザーID
     * @param userName 挿入するユーザー名
     * @throws DatabaseAccessException データベースアクセスエラーが発生した場合
     */
    public void insertUser(int userId, String userName) throws DatabaseAccessException {
        String sql = "INSERT INTO users (id, name) VALUES (?, ?)";

        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(sql)) {

            preparedStatement.setInt(1, userId);
            preparedStatement.setString(2, userName);
            preparedStatement.executeUpdate();

            LOGGER.log(Level.INFO, "ユーザーを正常に挿入しました。ユーザーID: " + userId);

        } catch (SQLException e) {
            // エラーメッセージをログ出力
            LOGGER.log(
                Level.SEVERE,
                "ユーザー挿入中にエラーが発生しました。" +
                " SQLState: " + e.getSQLState() +
                ", ErrorCode: " + e.getErrorCode() +
                ", Message: " + e.getMessage(),
                e
            );

            // カスタム例外をスロー
            throw new DatabaseAccessException(
                "ユーザー情報の挿入に失敗しました。ユーザーID: " + userId + ", ユーザー名: " + userName,
                e.getErrorCode(),
                e
            );
        }
    }

    // メインメソッド（動作確認用）
    public static void main(String[] args) {
        DatabaseService service = new DatabaseService();

        // ユーザー取得の例
        try {
            String name = service.getUserNameById(1);
            System.out.println("取得したユーザー名: " + name);
        } catch (DatabaseAccessException e) {
            System.err.println("エラーが発生しました: " + e.getMessage());
            System.err.println("エラーコード: " + e.getErrorCode());
        }

        // ユーザー挿入の例
        try {
            service.insertUser(2, "山田太郎");
            System.out.println("ユーザーを正常に挿入しました。");
        } catch (DatabaseAccessException e) {
            System.err.println("エラーが発生しました: " + e.getMessage());
            System.err.println("エラーコード: " + e.getErrorCode());
        }
    }
}
```
