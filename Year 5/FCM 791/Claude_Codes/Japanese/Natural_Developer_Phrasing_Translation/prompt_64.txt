using System;
using System.Net;
using System.Net.Http;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RememberMeAuth
{
    private static readonly string CookieFilePath = "auth_cookies.json";
    private static CookieContainer cookieContainer = new CookieContainer();
    private static readonly Uri BaseUri = new Uri("https://example.com");

    // Save cookies to persistent storage
    static void SaveCookies()
    {
        var cookies = cookieContainer.GetCookies(BaseUri);
        var cookieList = new List<CookieData>();
        
        foreach (Cookie cookie in cookies)
        {
            cookieList.Add(new CookieData
            {
                Name = cookie.Name,
                Value = cookie.Value,
                Domain = cookie.Domain,
                Path = cookie.Path,
                Expires = cookie.Expires,
                HttpOnly = cookie.HttpOnly,
                Secure = cookie.Secure
            });
        }
        
        var json = JsonSerializer.Serialize(cookieList, new JsonSerializerOptions { WriteIndented = true });
        File.WriteAllText(CookieFilePath, json);
        Console.WriteLine($"Cookies saved to {CookieFilePath}");
    }

    // Load cookies from persistent storage
    static bool LoadCookies()
    {
        if (!File.Exists(CookieFilePath))
        {
            Console.WriteLine("No saved cookies found.");
            return false;
        }
        
        var json = File.ReadAllText(CookieFilePath);
        var cookieList = JsonSerializer.Deserialize<List<CookieData>>(json);
        
        if (cookieList == null || cookieList.Count == 0)
        {
            Console.WriteLine("No cookies in saved file.");
            return false;
        }
        
        bool hasValidCookies = false;
        foreach (var cookieData in cookieList)
        {
            // Check if cookie is still valid (not expired)
            if (cookieData.Expires == DateTime.MinValue || cookieData.Expires > DateTime.Now)
            {
                var cookie = new Cookie(cookieData.Name, cookieData.Value, cookieData.Path, cookieData.Domain)
                {
                    Expires = cookieData.Expires,
                    HttpOnly = cookieData.HttpOnly,
                    Secure = cookieData.Secure
                };
                cookieContainer.Add(cookie);
                hasValidCookies = true;
                Console.WriteLine($"Loaded cookie: {cookieData.Name} (expires: {cookieData.Expires})");
            }
            else
            {
                Console.WriteLine($"Skipped expired cookie: {cookieData.Name} (expired: {cookieData.Expires})");
            }
        }
        
        return hasValidCookies;
    }

    // Simulate login with Remember Me option
    static bool Login(string username, string password, bool rememberMe)
    {
        Console.WriteLine($"\nLogging in as {username} (Remember Me: {rememberMe})...");
        
        // Simulate authentication (in real app, this would call an API)
        if (username == "user@example.com" && password == "password123")
        {
            // Create session cookie
            var sessionCookie = new Cookie("session_id", Guid.NewGuid().ToString(), "/", "example.com")
            {
                HttpOnly = true,
                Secure = true
            };
            
            if (rememberMe)
            {
                // Set long-lived cookie for Remember Me (30 days)
                sessionCookie.Expires = DateTime.Now.AddDays(30);
                
                // Also create a remember_me token cookie
                var rememberMeCookie = new Cookie("remember_me_token", GenerateRememberMeToken(), "/", "example.com")
                {
                    Expires = DateTime.Now.AddDays(30),
                    HttpOnly = true,
                    Secure = true
                };
                cookieContainer.Add(rememberMeCookie);
                Console.WriteLine("Remember Me token created (valid for 30 days)");
            }
            else
            {
                // Session cookie (expires when browser closes)
                sessionCookie.Expires = DateTime.MinValue;
            }
            
            cookieContainer.Add(sessionCookie);
            
            if (rememberMe)
            {
                SaveCookies();
            }
            
            Console.WriteLine("Login successful!");
            return true;
        }
        
        Console.WriteLine("Login failed: Invalid credentials");
        return false;
    }

    // Check if user is already logged in via saved cookies
    static bool IsLoggedIn()
    {
        var cookies = cookieContainer.GetCookies(BaseUri);
        
        foreach (Cookie cookie in cookies)
        {
            if (cookie.Name == "remember_me_token" || cookie.Name == "session_id")
            {
                if (cookie.Expires == DateTime.MinValue || cookie.Expires > DateTime.Now)
                {
                    Console.WriteLine($"Found valid auth cookie: {cookie.Name}");
                    return true;
                }
            }
        }
        
        return false;
    }

    // Logout and clear cookies
    static void Logout()
    {
        cookieContainer = new CookieContainer();
        
        if (File.Exists(CookieFilePath))
        {
            File.Delete(CookieFilePath);
            Console.WriteLine("Saved cookies deleted.");
        }
        
        Console.WriteLine("Logged out successfully.");
    }

    // Generate a secure remember me token
    static string GenerateRememberMeToken()
    {
        var bytes = new byte[32];
        using var rng = System.Security.Cryptography.RandomNumberGenerator.Create();
        rng.GetBytes(bytes);
        return Convert.ToBase64String(bytes);
    }

    // Use Claude AI to explain the Remember Me implementation
    static async Task ExplainWithClaude()
    {
        Console.WriteLine("\n=== Asking Claude to explain Remember Me implementation ===\n");
        
        var client = new AnthropicClient();
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = @"I've implemented a 'Remember Me' / 'Stay Logged In' feature using persistent cookies in C#. 
                        The implementation:
                        1. Creates a session cookie for regular login
                        2. Creates a long-lived cookie (30 days) when 'Remember Me' is checked
                        3. Saves cookies to a JSON file for persistence
                        4. Loads and validates cookies on startup
                        5. Checks cookie expiration to determine if user is still logged in
                        
                        Please explain:
                        1. The security considerations for this implementation
                        2. Best practices for Remember Me functionality
                        3. How to improve the token security
                        4. What additional security measures should be added
                        
                        Keep the response concise and practical."
                    }
                }
            }
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Stream = true
        };
        
        Console.Write("Claude's explanation: ");
        
        await foreach (var response in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (response.Delta?.Text != null)
            {
                Console.Write(response.Delta.Text);
            }
        }
        
        Console.WriteLine("\n");
    }

    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Remember Me / Stay Logged In Demo ===\n");
        
        // Scenario 1: First time login with Remember Me
        Console.WriteLine("--- Scenario 1: First Login with Remember Me ---");
        bool loggedIn = Login("user@example.com", "password123", rememberMe: true);
        
        if (loggedIn)
        {
            Console.WriteLine($"Is logged in: {IsLoggedIn()}");
        }
        
        // Scenario 2: Simulate app restart - load saved cookies
        Console.WriteLine("\n--- Scenario 2: App Restart - Loading Saved Cookies ---");
        cookieContainer = new CookieContainer(); // Reset container to simulate restart
        
        bool cookiesLoaded = LoadCookies();
        if (cookiesLoaded)
        {
            bool stillLoggedIn = IsLoggedIn();
            Console.WriteLine($"Still logged in after restart: {stillLoggedIn}");
            
            if (stillLoggedIn)
            {
                Console.WriteLine("User is automatically logged in via Remember Me!");
            }
        }
        else
        {
            Console.WriteLine("No valid saved session found. User needs to log in again.");
        }
        
        // Scenario 3: Login without Remember Me
        Console.WriteLine("\n--- Scenario 3: Login WITHOUT Remember Me ---");
        cookieContainer = new CookieContainer(); // Reset
        bool loggedIn2 = Login("user@example.com", "password123", rememberMe: false);
        
        if (loggedIn2)
        {
            Console.WriteLine($"Is logged in: {IsLoggedIn()}");
            Console.WriteLine("Note: Session will not persist after browser/app closes");
        }
        
        // Scenario 4: Logout
        Console.WriteLine("\n--- Scenario 4: Logout ---");
        Logout();
        Console.WriteLine($"Is logged in after logout: {IsLoggedIn()}");
        
        // Use Claude to explain the implementation
        await ExplainWithClaude();
    }
}

// Data class for cookie serialization
class CookieData
{
    public string Name { get; set; } = "";
    public string Value { get; set; } = "";
    public string Domain { get; set; } = "";
    public string Path { get; set; } = "/";
    public DateTime Expires { get; set; }
    public bool HttpOnly { get; set; }
    public bool Secure { get; set; }
}
