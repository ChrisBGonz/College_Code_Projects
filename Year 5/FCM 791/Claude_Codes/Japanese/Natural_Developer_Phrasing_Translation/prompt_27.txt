import anthropic
import base64
import os
import subprocess
import sys
import tempfile


def install_package(package: str) -> bool:
    """Install a Python package using pip."""
    try:
        subprocess.check_call([sys.executable, "-m", "pip", "install", package, "-q"])
        return True
    except subprocess.CalledProcessError:
        return False


def convert_html_to_pdf(html_content: str, output_path: str) -> dict:
    """
    Convert HTML template to PDF file using external libraries.
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Path where the PDF file will be saved
        
    Returns:
        dict with success status and file path or error message
    """
    # Try to use weasyprint first
    try:
        import weasyprint
        weasyprint.HTML(string=html_content).write_pdf(output_path)
        return {"success": True, "path": output_path, "library": "weasyprint"}
    except ImportError:
        pass
    except Exception as e:
        return {"success": False, "error": f"WeasyPrint error: {str(e)}"}
    
    # Try to install and use weasyprint
    print("Installing weasyprint...")
    if install_package("weasyprint"):
        try:
            import weasyprint
            weasyprint.HTML(string=html_content).write_pdf(output_path)
            return {"success": True, "path": output_path, "library": "weasyprint"}
        except Exception as e:
            pass
    
    # Try xhtml2pdf as fallback
    try:
        from xhtml2pdf import pisa
        with open(output_path, "wb") as pdf_file:
            pisa_status = pisa.CreatePDF(html_content, dest=pdf_file)
        if not pisa_status.err:
            return {"success": True, "path": output_path, "library": "xhtml2pdf"}
        else:
            return {"success": False, "error": "xhtml2pdf conversion failed"}
    except ImportError:
        pass
    
    # Try to install and use xhtml2pdf
    print("Installing xhtml2pdf...")
    if install_package("xhtml2pdf"):
        try:
            from xhtml2pdf import pisa
            with open(output_path, "wb") as pdf_file:
                pisa_status = pisa.CreatePDF(html_content, dest=pdf_file)
            if not pisa_status.err:
                return {"success": True, "path": output_path, "library": "xhtml2pdf"}
        except Exception as e:
            pass
    
    # Try reportlab as another fallback
    try:
        from reportlab.lib.pagesizes import letter
        from reportlab.platypus import SimpleDocTemplate, Paragraph
        from reportlab.lib.styles import getSampleStyleSheet
        
        doc = SimpleDocTemplate(output_path, pagesize=letter)
        styles = getSampleStyleSheet()
        
        # Simple text extraction from HTML
        import re
        text = re.sub('<[^<]+?>', '', html_content)
        
        story = [Paragraph(text, styles['Normal'])]
        doc.build(story)
        return {"success": True, "path": output_path, "library": "reportlab (basic)"}
    except ImportError:
        pass
    
    return {"success": False, "error": "No PDF conversion library available. Please install weasyprint or xhtml2pdf."}


def generate_html_template_with_claude(template_description: str) -> str:
    """
    Use Claude to generate an HTML template based on description.
    
    Args:
        template_description: Description of the HTML template to generate
        
    Returns:
        Generated HTML string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": f"""Generate a complete HTML template for: {template_description}
                
Please provide a well-formatted HTML document with:
- Proper HTML5 structure
- Inline CSS styles for better PDF rendering
- Sample content that demonstrates the template
- Clean, professional design

Return only the HTML code without any explanation or markdown formatting."""
            }
        ]
    )
    
    return message.content[0].text


def html_to_pdf_with_claude_assistance(
    template_description: str,
    output_path: str = None
) -> dict:
    """
    Main function that uses Claude to generate HTML and converts it to PDF.
    
    Args:
        template_description: Description of the HTML template to create
        output_path: Optional path for the output PDF file
        
    Returns:
        dict with results including HTML content and PDF path
    """
    # Generate HTML using Claude
    print(f"Generating HTML template for: {template_description}")
    html_content = generate_html_template_with_claude(template_description)
    
    # Set default output path if not provided
    if output_path is None:
        output_path = os.path.join(tempfile.gettempdir(), "output.pdf")
    
    # Convert HTML to PDF
    print(f"Converting HTML to PDF: {output_path}")
    result = convert_html_to_pdf(html_content, output_path)
    
    return {
        "html_content": html_content,
        "pdf_result": result,
        "template_description": template_description
    }


def demonstrate_pdf_generation():
    """Demonstrate the HTML to PDF conversion functionality."""
    
    # Example 1: Business Invoice
    print("\n=== Example 1: Business Invoice ===")
    result1 = html_to_pdf_with_claude_assistance(
        "A professional business invoice template with company logo placeholder, itemized billing table, and payment terms",
        "invoice.pdf"
    )
    
    print(f"Template Description: {result1['template_description']}")
    print(f"HTML Generated: {len(result1['html_content'])} characters")
    print(f"PDF Result: {result1['pdf_result']}")
    
    # Show first 500 chars of HTML
    print("\nHTML Preview (first 500 chars):")
    print(result1['html_content'][:500])
    
    # Example 2: Resume/CV
    print("\n=== Example 2: Resume/CV Template ===")
    result2 = html_to_pdf_with_claude_assistance(
        "A modern resume/CV template with sections for personal info, work experience, education, and skills",
        "resume.pdf"
    )
    
    print(f"Template Description: {result2['template_description']}")
    print(f"HTML Generated: {len(result2['html_content'])} characters")
    print(f"PDF Result: {result2['pdf_result']}")
    
    # Example 3: Report
    print("\n=== Example 3: Business Report ===")
    result3 = html_to_pdf_with_claude_assistance(
        "A quarterly business report template with executive summary, charts placeholder, and data tables",
        "report.pdf"
    )
    
    print(f"Template Description: {result3['template_description']}")
    print(f"HTML Generated: {len(result3['html_content'])} characters")
    print(f"PDF Result: {result3['pdf_result']}")
    
    # Save HTML files for inspection
    for i, (result, filename) in enumerate([
        (result1, "invoice.html"),
        (result2, "resume.html"),
        (result3, "report.html")
    ], 1):
        with open(filename, "w", encoding="utf-8") as f:
            f.write(result['html_content'])
        print(f"\nHTML saved to: {filename}")
    
    return [result1, result2, result3]


if __name__ == "__main__":
    # Run the demonstration
    results = demonstrate_pdf_generation()
    
    print("\n=== Summary ===")
    for i, result in enumerate(results, 1):
        pdf_result = result['pdf_result']
        if pdf_result['success']:
            print(f"Example {i}: ✓ PDF created at {pdf_result['path']} using {pdf_result['library']}")
        else:
            print(f"Example {i}: ✗ PDF creation failed - {pdf_result['error']}")
            print(f"  HTML was generated and saved as HTML file for manual conversion")
