package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func main() {
	client := anthropic.NewClient()

	// Example: Ask Claude to generate a Go function for reading config from env vars
	prompt := `Create a Go function that reads a configuration file path from a user-specified environment variable and loads the configuration. The function should:
1. Accept an environment variable name as a parameter
2. Read the file path from that environment variable
3. Read and return the contents of the configuration file
4. Handle errors appropriately (missing env var, file not found, etc.)

Please provide a complete, working Go code example with proper error handling.`

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude4_5,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})
	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract and print the response
	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println("Claude's response:")
			fmt.Println("==================")
			fmt.Println(block.Text)
		}
	}

	// Now demonstrate the actual implementation
	fmt.Println("\n==================")
	fmt.Println("Actual Implementation Demo:")
	fmt.Println("==================")

	// Create a temporary config file for demonstration
	tmpFile, err := os.CreateTemp("", "config-*.txt")
	if err != nil {
		log.Fatalf("Failed to create temp file: %v", err)
	}
	defer os.Remove(tmpFile.Name())

	// Write some config content
	configContent := `# Sample Configuration
database_host=localhost
database_port=5432
app_name=MyApp
debug=true`

	if _, err := tmpFile.WriteString(configContent); err != nil {
		log.Fatalf("Failed to write to temp file: %v", err)
	}
	tmpFile.Close()

	// Set the environment variable
	envVarName := "MY_APP_CONFIG"
	os.Setenv(envVarName, tmpFile.Name())
	defer os.Unsetenv(envVarName)

	// Test the function
	content, err := loadConfigFromEnv(envVarName)
	if err != nil {
		log.Fatalf("Error loading config: %v", err)
	}

	fmt.Printf("Successfully loaded config from env var '%s':\n", envVarName)
	fmt.Println(content)

	// Test with missing env var
	_, err = loadConfigFromEnv("NONEXISTENT_ENV_VAR")
	if err != nil {
		fmt.Printf("\nExpected error for missing env var: %v\n", err)
	}
}

// loadConfigFromEnv reads a configuration file path from the specified environment variable
// and returns the contents of the configuration file.
func loadConfigFromEnv(envVarName string) (string, error) {
	// Get the file path from the environment variable
	configPath := os.Getenv(envVarName)
	if configPath == "" {
		return "", fmt.Errorf("environment variable '%s' is not set or is empty", envVarName)
	}

	// Check if the file exists
	if _, err := os.Stat(configPath); os.IsNotExist(err) {
		return "", fmt.Errorf("configuration file not found at path '%s': %w", configPath, err)
	}

	// Read the configuration file
	content, err := os.ReadFile(configPath)
	if err != nil {
		return "", fmt.Errorf("failed to read configuration file '%s': %w", configPath, err)
	}

	return string(content), nil
}
