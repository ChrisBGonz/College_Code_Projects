import anthropic
import hashlib
import os
import sys
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def verify_firmware_with_claude(
    file_path: str, expected_hash: str, actual_hash: str
) -> str:
    """Use Claude to analyze and explain the firmware verification result."""
    client = anthropic.Anthropic()

    file_name = Path(file_path).name
    file_size = os.path.getsize(file_path)

    verification_status = "PASSED" if actual_hash == expected_hash else "FAILED"

    prompt = f"""ファームウェアファイルの整合性検証結果を分析してください。

ファイル情報:
- ファイル名: {file_name}
- ファイルサイズ: {file_size} バイト
- 検証方法: SHA-256

ハッシュ値:
- 期待されるハッシュ: {expected_hash}
- 実際のハッシュ: {actual_hash}

検証結果: {verification_status}

以下の点について詳しく説明してください:
1. 検証結果の意味と重要性
2. {'ファイルが正常であることの確認' if verification_status == 'PASSED' else 'ファイルが破損または改ざんされている可能性についての警告'}
3. セキュリティ上の推奨事項
4. 次のステップ

日本語で回答してください。"""

    print("\n=== Claude による分析 ===\n")

    full_response = ""
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text

    print("\n")
    return full_response


def verify_firmware(file_path: str, expected_hash: str) -> bool:
    """Main function to verify firmware integrity."""
    print(f"=== ファームウェア整合性検証 ===")
    print(f"ファイル: {file_path}")
    print(f"期待されるSHA-256: {expected_hash}")

    if not os.path.exists(file_path):
        print(f"\nエラー: ファイル '{file_path}' が見つかりません。")
        return False

    print("\nSHA-256ハッシュを計算中...")
    actual_hash = calculate_sha256(file_path)
    print(f"計算されたSHA-256: {actual_hash}")

    is_valid = actual_hash.lower() == expected_hash.lower()

    print(f"\n検証結果: {'✓ 成功 - ファイルは正常です' if is_valid else '✗ 失敗 - ファイルが一致しません'}")

    verify_firmware_with_claude(file_path, expected_hash, actual_hash)

    return is_valid


def create_test_firmware():
    """Create a test firmware file for demonstration."""
    test_file = "test_firmware.bin"
    test_content = b"This is a test firmware file content for demonstration purposes.\x00" * 100

    with open(test_file, "wb") as f:
        f.write(test_content)

    actual_hash = calculate_sha256(test_file)
    return test_file, actual_hash


def main():
    if len(sys.argv) == 3:
        file_path = sys.argv[1]
        expected_hash = sys.argv[2]
        result = verify_firmware(file_path, expected_hash)
        sys.exit(0 if result else 1)
    else:
        print("デモモード: テスト用ファームウェアファイルを作成して検証します\n")

        test_file, correct_hash = create_test_firmware()

        print("=== テスト1: 正常なファイルの検証 ===")
        result1 = verify_firmware(test_file, correct_hash)

        print("\n" + "=" * 50 + "\n")

        print("=== テスト2: 改ざんされたハッシュでの検証 ===")
        wrong_hash = "a" * 64
        result2 = verify_firmware(test_file, wrong_hash)

        if os.path.exists(test_file):
            os.remove(test_file)

        print("\n=== 使用方法 ===")
        print(f"python {sys.argv[0]} <ファームウェアファイル> <期待されるSHA-256ハッシュ>")
        print(f"例: python {sys.argv[0]} firmware.bin abc123...")


if __name__ == "__main__":
    main()
