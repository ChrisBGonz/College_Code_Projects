import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";

const client = new Anthropic();

async function generateExpressServerConfig(): Promise<void> {
  console.log("Generating production-ready Express.js server configuration...\n");

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `本番環境向けの基本的な Express.js サーバ設定を作成してください。以下の要件を満たしてください：

1. セキュリティミドルウェア（helmet, cors, rate limiting）
2. リクエストログ（morgan）
3. ボディパーサー設定
4. エラーハンドリングミドルウェア：
   - 404 Not Found ハンドラー
   - 500 Internal Server Error ハンドラー
   - 未処理の例外（uncaughtException）ハンドラー
   - 未処理のPromise拒否（unhandledRejection）ハンドラー
5. グレースフルシャットダウン
6. 環境変数の設定
7. ヘルスチェックエンドポイント

完全なコードを提供してください。コメントも日本語で記述してください。`,
      },
    ],
  });

  // Extract the text content from the response
  let generatedCode = "";
  for (const block of message.content) {
    if (block.type === "text") {
      generatedCode += block.text;
    }
  }

  console.log("Generated Express.js Server Configuration:");
  console.log("=".repeat(60));
  console.log(generatedCode);
  console.log("=".repeat(60));

  // Save the generated code to a file
  const outputFile = "generated-express-server.js";
  
  // Extract code blocks from the response
  const codeBlockRegex = /```(?:javascript|js)?\n([\s\S]*?)```/g;
  let match;
  let extractedCode = "";
  
  while ((match = codeBlockRegex.exec(generatedCode)) !== null) {
    extractedCode += match[1] + "\n\n";
  }
  
  if (extractedCode) {
    fs.writeFileSync(outputFile, extractedCode);
    console.log(`\nExtracted code saved to: ${outputFile}`);
  } else {
    // If no code blocks found, save the entire response
    fs.writeFileSync(outputFile, generatedCode);
    console.log(`\nFull response saved to: ${outputFile}`);
  }

  // Display usage statistics
  console.log("\nAPI Usage Statistics:");
  console.log(`- Input tokens: ${message.usage.input_tokens}`);
  console.log(`- Output tokens: ${message.usage.output_tokens}`);
  console.log(`- Stop reason: ${message.stop_reason}`);
}

// Main execution
generateExpressServerConfig().catch((error) => {
  console.error("Error generating Express.js configuration:", error);
  process.exit(1);
});
