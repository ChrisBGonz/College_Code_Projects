import com.anthropic.client.okhttp.AnthropicOkHttpClient;

import java.nio.file.*;

import java.util.regex.*;

public class MavenDependencyUpdater {
    
    private static final AnthropicClient client = AnthropicOkHttpClient.builder()
            .apiKey(System.getenv("ANTHROPIC_API_KEY"))
            .build();
    
    public static void main(String[] args) throws Exception {
        String pomFilePath = args.length > 0 ? args[0] : "pom.xml";
        
        System.out.println("Maven Dependency Updater - Using Claude AI");
        System.out.println("==========================================");
        System.out.println("Reading pom.xml from: " + pomFilePath);
        
        // Read the pom.xml file
        String pomContent = readFile(pomFilePath);
        
        // Use Claude to analyze and update dependencies
        updateDependenciesWithClaude(pomContent, pomFilePath);
    }
    
    private static String readFile(String filePath) throws IOException {
        Path path = Paths.get(filePath);
        if (!Files.exists(path)) {
            // Create a sample pom.xml for demonstration
            String samplePom = createSamplePom();
            Files.writeString(path, samplePom);
            System.out.println("Created sample pom.xml for demonstration");
        }
        return Files.readString(path);
    }

    private static void updateDependenciesWithClaude(String pomContent, String pomFilePath) throws Exception {
        System.out.println("\nAnalyzing dependencies with Claude AI (streaming)...\n");
        
        String prompt = """
                You are a Maven dependency update expert. I have a pom.xml file that needs its dependencies updated to the latest stable versions.
                
                Please analyze the following pom.xml and:
                1. Identify all dependencies with their current versions
                2. Suggest the latest stable versions for each dependency (as of your knowledge cutoff)
                3. Provide the updated pom.xml with all dependencies updated to their latest stable versions
                4. Add a comment explaining what was updated
                
                Important notes:
                - Only update to stable releases (no SNAPSHOT, RC, or beta versions)
                - Maintain compatibility between related dependencies (e.g., Spring Boot ecosystem)
                - Keep the same structure and formatting of the pom.xml
                - If a dependency is already at the latest version, keep it as is
                
                Here is the pom.xml content:
                
                ```xml
                """ + pomContent + """
                ```
                
                Please provide:
                1. A summary of changes (which dependencies were updated and to what versions)
                2. The complete updated pom.xml content
                
                Format your response as:
                ## Update Summary
                [List of changes]
                
                ## Updated pom.xml
                ```xml
                [Complete updated pom.xml]
                ```
                """;
        
        StringBuilder fullResponse = new StringBuilder();
        
        // Use streaming to get the response
        try (var stream = client.messages().stream(MessageCreateParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(4096)
                .addUserMessage(prompt)
                .build())) {
            
            stream.stream()
                    .filter(event -> event instanceof RawMessageStreamEvent)
                    .map(event -> (RawMessageStreamEvent) event)
                    .forEach(event -> {
                        if (event.unwrap() instanceof ContentBlockDeltaEvent deltaEvent) {
                            if (deltaEvent.delta() instanceof TextDelta textDelta) {
                                String text = textDelta.text();
                                System.out.print(text);
                                fullResponse.append(text);
                            }
                        }
                    });
        }
        
        System.out.println("\n\n==========================================");
        
        // Extract and save the updated pom.xml
        String updatedPom = extractUpdatedPom(fullResponse.toString());
        if (updatedPom != null && !updatedPom.isEmpty()) {
            // Create backup of original
            String backupPath = pomFilePath + ".backup";
            Files.copy(Paths.get(pomFilePath), Paths.get(backupPath), StandardCopyOption.REPLACE_EXISTING);
            System.out.println("Original pom.xml backed up to: " + backupPath);
            
            // Write updated pom.xml
            Files.writeString(Paths.get(pomFilePath), updatedPom);
            System.out.println("Updated pom.xml saved to: " + pomFilePath);
            
            // Verify the update
            verifyUpdate(pomContent, updatedPom);
        } else {
            System.out.println("Could not extract updated pom.xml from response.");
        }
    }
    
    private static String extractUpdatedPom(String response) {
        // Try to extract XML content from the response
        Pattern pattern = Pattern.compile("```xml\\s*\\n(.*?)\\n```", Pattern.DOTALL);
        Matcher matcher = pattern.matcher(response);
        
        String lastMatch = null;
        while (matcher.find()) {
            lastMatch = matcher.group(1).trim();
        }
        
        if (lastMatch != null && lastMatch.contains("<project")) {
            return lastMatch;
        }
        
        // Try alternative extraction
        int startIdx = response.lastIndexOf("<?xml");
        if (startIdx != -1) {
            int endIdx = response.lastIndexOf("</project>");
            if (endIdx != -1) {
                return response.substring(startIdx, endIdx + "</project>".length()).trim();
            }
        }
        
        return null;
    }
    
    private static void verifyUpdate(String originalPom, String updatedPom) {
        System.out.println("\nVerification Report:");
        System.out.println("--------------------");
        
        // Count dependencies in original
        int originalDeps = countOccurrences(originalPom, "<dependency>");
        int updatedDeps = countOccurrences(updatedPom, "<dependency>");
        
        System.out.println("Original dependencies count: " + originalDeps);
        System.out.println("Updated dependencies count: " + updatedDeps);
        
        if (originalDeps == updatedDeps) {
            System.out.println("✓ Dependency count matches");
        } else {
            System.out.println("⚠ Dependency count mismatch - please review the updated pom.xml");
        }
        
        // Check if it's valid XML structure
        if (updatedPom.contains("<project") && updatedPom.contains("</project>")) {
            System.out.println("✓ Valid XML structure detected");
        } else {
            System.out.println("⚠ XML structure may be incomplete");
        }
        
        System.out.println("\nUpdate process completed!");
    }
    
    private static int countOccurrences(String text, String pattern) {
        int count = 0;
        int idx = 0;
        while ((idx = text.indexOf(pattern, idx)) != -1) {
            count++;
            idx += pattern.length();
        }
        return count;
    }
}
