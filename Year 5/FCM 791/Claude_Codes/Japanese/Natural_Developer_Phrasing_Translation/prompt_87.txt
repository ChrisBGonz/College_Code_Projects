import anthropic
import time
import random
from datetime import datetime
from collections import deque

client = anthropic.Anthropic()

log_file_path = "test_auth.log"

def generate_test_log_entries():
    """Generate test log entries including failed login attempts"""
    entries = []
    current_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    # Generate a mix of successful and failed logins
    for i in range(15):
        if random.random() < 0.7:  # 70% chance of failed login
            entries.append(f"{current_time} - Failed Login attempt from user{i}@example.com from IP 192.168.1.{random.randint(1, 255)}")
        else:
            entries.append(f"{current_time} - Successful Login from user{i}@example.com from IP 192.168.1.{random.randint(1, 255)}")
    
    return entries

def write_test_logs(entries):
    """Write test log entries to file"""
    with open(log_file_path, "a") as f:
        for entry in entries:
            f.write(entry + "\n")

def analyze_logs_with_claude(log_content: str, failed_count: int) -> str:
    """Use Claude to analyze log content and provide security insights"""
    
    prompt = f"""You are a security analyst monitoring authentication logs. 
    
I've detected {failed_count} failed login attempts in the last minute. Here are the recent log entries:

{log_content}

Please analyze these logs and:
1. Confirm if this is a security concern (threshold is 10 failed logins per minute)
2. Identify any patterns (same IP, same user, etc.)
3. Provide a brief security assessment
4. Suggest immediate actions if needed

Keep your response concise and actionable."""

    full_response = ""
    
    print("\n🤖 Claude Security Analysis (streaming):")
    print("-" * 50)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {"role": "user", "content": prompt}
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response

def monitor_log_file():
    """Monitor log file for failed login attempts"""
    print("🔍 Starting log file monitor...")
    print(f"📁 Monitoring: {log_file_path}")
    print(f"⚠️  Alert threshold: 10 failed logins per minute")
    print("=" * 60)
    
    # Track failed login timestamps using a deque for efficiency
    failed_logins = deque()
    last_position = 0
    check_interval = 10  # Check every 10 seconds
    
    # Create initial log file if it doesn't exist
    with open(log_file_path, "a") as f:
        pass
    
    iteration = 0
    
    while True:
        iteration += 1
        current_time = time.time()
        
        # Simulate log generation every 30 seconds
        if iteration % 3 == 0:  # Every 30 seconds (3 * 10 second intervals)
            print(f"\n📝 Generating test log entries...")
            test_entries = generate_test_log_entries()
            write_test_logs(test_entries)
        
        # Read new log entries
        try:
            with open(log_file_path, "r") as f:
                f.seek(last_position)
                new_content = f.read()
                last_position = f.tell()
        except FileNotFoundError:
            print(f"⚠️  Log file not found: {log_file_path}")
            time.sleep(check_interval)
            continue
        
        # Process new log entries
        if new_content:
            new_lines = new_content.strip().split("\n")
            new_failed_count = 0
            
            for line in new_lines:
                if "Failed Login" in line:
                    failed_logins.append(current_time)
                    new_failed_count += 1
            
            if new_failed_count > 0:
                print(f"⏰ {datetime.now().strftime('%H:%M:%S')} - Detected {new_failed_count} new failed login attempts")
        
        # Remove entries older than 1 minute
        one_minute_ago = current_time - 60
        while failed_logins and failed_logins[0] < one_minute_ago:
            failed_logins.popleft()
        
        # Count failed logins in the last minute
        recent_failed_count = len(failed_logins)
        
        print(f"📊 {datetime.now().strftime('%H:%M:%S')} - Failed logins in last 60s: {recent_failed_count}/10")
        
        # Check if threshold exceeded
        if recent_failed_count > 10:
            print(f"\n🚨 ALERT: {recent_failed_count} failed login attempts detected in the last minute!")
            
            # Read recent log content for analysis
            try:
                with open(log_file_path, "r") as f:
                    all_content = f.read()
                    # Get last 50 lines for analysis
                    lines = all_content.strip().split("\n")
                    recent_logs = "\n".join(lines[-50:]) if len(lines) > 50 else all_content
            except Exception as e:
                recent_logs = f"Error reading logs: {e}"
            
            # Analyze with Claude
            analyze_logs_with_claude(recent_logs, recent_failed_count)
            
            # Reset the deque after alert to avoid repeated alerts
            failed_logins.clear()
            print("\n✅ Alert processed. Resetting counter...")
        
        # Stop after 5 iterations for demo purposes
        if iteration >= 5:
            print("\n✅ Demo monitoring complete. Exiting...")
            break
            
        time.sleep(check_interval)

def main():
    """Main function to run the log monitor"""
    print("=" * 60)
    print("🛡️  Security Log Monitor with Claude AI Analysis")
    print("=" * 60)
    
    # Clean up any existing test log
    import os
    if os.path.exists(log_file_path):
        os.remove(log_file_path)
        print(f"🗑️  Cleaned up existing test log: {log_file_path}")
    
    try:
        monitor_log_file()
    except KeyboardInterrupt:
        print("\n\n⛔ Monitoring stopped by user")
    except Exception as e:
        print(f"\n❌ Error: {e}")
        raise

if __name__ == "__main__":
    main()
