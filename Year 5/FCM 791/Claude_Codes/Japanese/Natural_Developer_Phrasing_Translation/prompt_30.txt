#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <vector>
#include <fstream>
#include <sstream>

// libjpeg headers
#include <jpeglib.h>
#include <jerror.h>

// Structure to hold image data
struct ImageData {
    std::vector<unsigned char> pixels;
    int width;
    int height;
    int channels;
};

// Function to load JPEG image using libjpeg
ImageData loadJpegImage(const std::string& filename) {
    ImageData img;
    
    FILE* file = fopen(filename.c_str(), "rb");
    if (!file) {
        throw std::runtime_error("Cannot open file: " + filename);
    }
    
    struct jpeg_decompress_struct cinfo;
    struct jpeg_error_mgr jerr;
    
    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, file);
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);
    
    img.width = cinfo.output_width;
    img.height = cinfo.output_height;
    img.channels = cinfo.output_components;
    
    int row_stride = img.width * img.channels;
    img.pixels.resize(img.height * row_stride);
    
    while (cinfo.output_scanline < cinfo.output_height) {
        unsigned char* row_pointer = &img.pixels[cinfo.output_scanline * row_stride];
        jpeg_read_scanlines(&cinfo, &row_pointer, 1);
    }
    
    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    fclose(file);
    
    return img;
}

// Function to save JPEG image using libjpeg
void saveJpegImage(const std::string& filename, const ImageData& img, int quality = 85) {
    FILE* file = fopen(filename.c_str(), "wb");
    if (!file) {
        throw std::runtime_error("Cannot create file: " + filename);
    }
    
    struct jpeg_compress_struct cinfo;
    struct jpeg_error_mgr jerr;
    
    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_compress(&cinfo);
    jpeg_stdio_dest(&cinfo, file);
    
    cinfo.image_width = img.width;
    cinfo.image_height = img.height;
    cinfo.input_components = img.channels;
    cinfo.in_color_space = (img.channels == 3) ? JCS_RGB : JCS_GRAYSCALE;
    
    jpeg_set_defaults(&cinfo);
    jpeg_set_quality(&cinfo, quality, TRUE);
    jpeg_start_compress(&cinfo, TRUE);
    
    int row_stride = img.width * img.channels;
    while (cinfo.next_scanline < cinfo.image_height) {
        const unsigned char* row_pointer = &img.pixels[cinfo.next_scanline * row_stride];
        jpeg_write_scanlines(&cinfo, const_cast<unsigned char**>(&row_pointer), 1);
    }
    
    jpeg_finish_compress(&cinfo);
    jpeg_destroy_compress(&cinfo);
    fclose(file);
}

// Function to get image info as string
std::string getImageInfo(const ImageData& img) {
    std::ostringstream oss;
    oss << "Width: " << img.width << "px, Height: " << img.height << "px, Channels: " << img.channels;
    return oss.str();
}

// Function to analyze image with Claude AI
std::string analyzeImageWithClaude(const std::string& imageDescription, const std::string& userQuery) {
    // Initialize Anthropic client
    anthropic::Client client;
    
    // Create a message asking Claude to analyze the image
    std::string prompt = "I have a JPEG image with the following properties:\n" + 
                        imageDescription + "\n\n" +
                        "User query: " + userQuery + "\n\n" +
                        "Please provide analysis and suggestions for image processing operations that could be applied to this image.";
    
    auto message = client.messages().create({
        .model = "claude-opus-4-5",
        .max_tokens = 1024,
        .messages = {
            {
                .role = "user",
                .content = prompt
            }
        }
    });
    
    // Extract text from response
    std::string response;
    for (const auto& block : message.content) {
        if (block.type == "text") {
            response += block.text;
        }
    }
    
    return response;
}

// Function to apply grayscale conversion
ImageData convertToGrayscale(const ImageData& img) {
    if (img.channels == 1) {
        return img; // Already grayscale
    }
    
    ImageData result;
    result.width = img.width;
    result.height = img.height;
    result.channels = 1;
    result.pixels.resize(img.width * img.height);
    
    for (int i = 0; i < img.height; i++) {
        for (int j = 0; j < img.width; j++) {
            int src_idx = (i * img.width + j) * img.channels;
            int dst_idx = i * img.width + j;
            
            // Standard luminance formula
            result.pixels[dst_idx] = static_cast<unsigned char>(
                0.299 * img.pixels[src_idx] +
                0.587 * img.pixels[src_idx + 1] +
                0.114 * img.pixels[src_idx + 2]
            );
        }
    }
    
    return result;
}

// Function to adjust brightness
ImageData adjustBrightness(const ImageData& img, int delta) {
    ImageData result = img;
    
    for (auto& pixel : result.pixels) {
        int newVal = static_cast<int>(pixel) + delta;
        pixel = static_cast<unsigned char>(std::max(0, std::min(255, newVal)));
    }
    
    return result;
}

int main() {
    std::cout << "=== JPEG Image Processing with libjpeg and Claude AI ===" << std::endl;
    std::cout << std::endl;
    
    // Create a simple test image (gradient)
    std::cout << "Creating test JPEG image..." << std::endl;
    
    ImageData testImg;
    testImg.width = 100;
    testImg.height = 100;
    testImg.channels = 3;
    testImg.pixels.resize(testImg.width * testImg.height * testImg.channels);
    
    // Create a gradient image
    for (int y = 0; y < testImg.height; y++) {
        for (int x = 0; x < testImg.width; x++) {
            int idx = (y * testImg.width + x) * testImg.channels;
            testImg.pixels[idx] = static_cast<unsigned char>(x * 255 / testImg.width);     // R
            testImg.pixels[idx + 1] = static_cast<unsigned char>(y * 255 / testImg.height); // G
            testImg.pixels[idx + 2] = 128;                                                   // B
        }
    }
    
    // Save test image
    const std::string testFilename = "/tmp/test_image.jpg";
    saveJpegImage(testFilename, testImg);
    std::cout << "Test image saved to: " << testFilename << std::endl;
    
    // Load the image back
    std::cout << "\nLoading JPEG image..." << std::endl;
    ImageData loadedImg = loadJpegImage(testFilename);
    std::string imageInfo = getImageInfo(loadedImg);
    std::cout << "Image loaded successfully: " << imageInfo << std::endl;
    
    // Apply some processing
    std::cout << "\nApplying image processing operations..." << std::endl;
    
    // Convert to grayscale
    ImageData grayImg = convertToGrayscale(loadedImg);
    saveJpegImage("/tmp/test_gray.jpg", grayImg);
    std::cout << "Grayscale image saved to: /tmp/test_gray.jpg" << std::endl;
    
    // Adjust brightness
    ImageData brightImg = adjustBrightness(loadedImg, 50);
    saveJpegImage("/tmp/test_bright.jpg", brightImg);
    std::cout << "Brightness-adjusted image saved to: /tmp/test_bright.jpg" << std::endl;
    
    // Analyze with Claude AI
    std::cout << "\n=== Analyzing image with Claude AI ===" << std::endl;
    
    std::string detailedInfo = "Original image: " + imageInfo + "\n" +
                               "Processing applied: Grayscale conversion and brightness adjustment (+50)\n" +
                               "File format: JPEG\n" +
                               "Color space: RGB (original), Grayscale (converted)";
    
    std::string userQuery = "What image processing techniques would be most useful for this type of gradient test image? "
                           "What are the typical use cases for grayscale conversion and brightness adjustment?";
    
    try {
        std::string analysis = analyzeImageWithClaude(detailedInfo, userQuery);
        std::cout << "\nClaude's Analysis:" << std::endl;
        std::cout << "==================" << std::endl;
        std::cout << analysis << std::endl;
    } catch (const std::exception& e) {
        std::cerr << "Error calling Claude API: " << e.what() << std::endl;
    }
    
    std::cout << "\n=== Processing Complete ===" << std::endl;
    std::cout << "Files created:" << std::endl;
    std::cout << "  - " << testFilename << " (original)" << std::endl;
    std::cout << "  - /tmp/test_gray.jpg (grayscale)" << std::endl;
    std::cout << "  - /tmp/test_bright.jpg (brightness adjusted)" << std::endl;
    
    return 0;
}
