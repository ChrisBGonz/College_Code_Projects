import anthropic.*;
import anthropic.models.*;
import java.util.*;

public class LdapSearchFilter {
    
    public static String buildLdapSearchFilter(String lastName) {
        AnthropicClient client = AnthropicClient.builder()
            .apiKey(System.getenv("ANTHROPIC_API_KEY"))
            .build();
        
        String userMessage = "Build an LDAP search filter using the following last name: " + lastName + 
            "\n\nReturn ONLY the LDAP filter string, nothing else. Example format: (sn=Smith)";
        
        Message message = client.messages().create(
            MessageCreateParams.builder()
                .model(Model.CLAUDE_opus_4_5)
                .maxTokens(1024)
                .addUserMessage(userMessage)
                .build()
        );
        
        String ldapFilter = "";
        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock) {
                ldapFilter = ((TextBlock) block).text().trim();
                break;
            }
        }
        
        return ldapFilter;
    }
    
    public static void main(String[] args) {
        String[] testLastNames = {"Smith", "O'Brien", "García", "von Müller"};
        
        System.out.println("LDAP Search Filter Builder");
        System.out.println("==========================");
        
        for (String lastName : testLastNames) {
            System.out.println("\nLast Name: " + lastName);
            String filter = buildLdapSearchFilter(lastName);
            System.out.println("LDAP Filter: " + filter);
        }
    }
}
