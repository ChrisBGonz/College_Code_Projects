class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.successThreshold = options.successThreshold || 2;
    this.timeout = options.timeout || 60000;
    this.halfOpenMaxRequests = options.halfOpenMaxRequests || 1;
    this.monitorInterval = options.monitorInterval || 10000;

    this.state = CircuitBreaker.STATES.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.lastFailureTime = null;
    this.halfOpenRequestCount = 0;
    this.requestQueue = [];

    this.listeners = {
      stateChange: [],
      success: [],
      failure: [],
      reject: [],
    };

    this._startMonitoring();
  }

  static get STATES() {
    return {
      CLOSED: "CLOSED",
      OPEN: "OPEN",
      HALF_OPEN: "HALF_OPEN",
    };
  }

  on(event, callback) {
    if (this.listeners[event]) {
      this.listeners[event].push(callback);
    }
    return this;
  }

  _emit(event, data) {
    if (this.listeners[event]) {
      this.listeners[event].forEach((cb) => cb(data));
    }
  }

  _setState(newState) {
    const prevState = this.state;
    this.state = newState;

    if (prevState !== newState) {
      console.log(
        `[CircuitBreaker] State changed: ${prevState} -> ${newState}`
      );
      this._emit("stateChange", {
        from: prevState,
        to: newState,
        timestamp: new Date().toISOString(),
      });
    }
  }

  _recordSuccess() {
    this.failureCount = 0;

    if (this.state === CircuitBreaker.STATES.HALF_OPEN) {
      this.successCount++;
      console.log(
        `[CircuitBreaker] Half-open success: ${this.successCount}/${this.successThreshold}`
      );

      if (this.successCount >= this.successThreshold) {
        this._reset();
      }
    }

    this._emit("success", {
      state: this.state,
      timestamp: new Date().toISOString(),
    });
  }

  _recordFailure(error) {
    this.failureCount++;
    this.lastFailureTime = Date.now();

    console.warn(
      `[CircuitBreaker] Failure recorded: ${this.failureCount}/${this.failureThreshold}`,
      error?.message || error
    );

    this._emit("failure", {
      error,
      failureCount: this.failureCount,
      state: this.state,
      timestamp: new Date().toISOString(),
    });

    if (this.state === CircuitBreaker.STATES.HALF_OPEN) {
      this._trip();
      return;
    }

    if (
      this.state === CircuitBreaker.STATES.CLOSED &&
      this.failureCount >= this.failureThreshold
    ) {
      this._trip();
    }
  }

  _trip() {
    this.successCount = 0;
    this.halfOpenRequestCount = 0;
    this._setState(CircuitBreaker.STATES.OPEN);
    console.error(
      `[CircuitBreaker] Circuit tripped! Will retry after ${this.timeout}ms`
    );
  }

  _reset() {
    this.failureCount = 0;
    this.successCount = 0;
    this.halfOpenRequestCount = 0;
    this.lastFailureTime = null;
    this._setState(CircuitBreaker.STATES.CLOSED);
    console.log("[CircuitBreaker] Circuit reset to CLOSED");
    this._processQueue();
  }

  _attemptReset() {
    if (
      this.state === CircuitBreaker.STATES.OPEN &&
      this.lastFailureTime &&
      Date.now() - this.lastFailureTime >= this.timeout
    ) {
      this.successCount = 0;
      this.halfOpenRequestCount = 0;
      this._setState(CircuitBreaker.STATES.HALF_OPEN);
      console.log("[CircuitBreaker] Attempting reset - entering HALF_OPEN");
    }
  }

  _canRequest() {
    switch (this.state) {
      case CircuitBreaker.STATES.CLOSED:
        return true;

      case CircuitBreaker.STATES.OPEN:
        this._attemptReset();
        return this.state === CircuitBreaker.STATES.HALF_OPEN;

      case CircuitBreaker.STATES.HALF_OPEN:
        return this.halfOpenRequestCount < this.halfOpenMaxRequests;

      default:
        return false;
    }
  }

  async execute(requestFn, options = {}) {
    const { fallback = null, timeout = null } = options;

    if (!this._canRequest()) {
      const error = new CircuitBreakerOpenError(
        `Circuit breaker is ${this.state}. Request rejected.`,
        this.state,
        this.lastFailureTime
          ? this.timeout - (Date.now() - this.lastFailureTime)
          : null
      );

      this._emit("reject", {
        state: this.state,
        timestamp: new Date().toISOString(),
      });

      if (typeof fallback === "function") {
        console.log("[CircuitBreaker] Using fallback response");
        return fallback(error);
      }

      throw error;
    }

    if (this.state === CircuitBreaker.STATES.HALF_OPEN) {
      this.halfOpenRequestCount++;
    }

    try {
      let result;

      if (timeout) {
        result = await this._withTimeout(requestFn(), timeout);
      } else {
        result = await requestFn();
      }

      this._recordSuccess();
      return result;
    } catch (error) {
      this._recordFailure(error);
      throw error;
    }
  }

  _withTimeout(promise, ms) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        reject(new Error(`Request timed out after ${ms}ms`));
      }, ms);

      promise
        .then((result) => {
          clearTimeout(timer);
          resolve(result);
        })
        .catch((error) => {
          clearTimeout(timer);
          reject(error);
        });
    });
  }

  enqueue(requestFn, options = {}) {
    return new Promise((resolve, reject) => {
      this.requestQueue.push({ requestFn, options, resolve, reject });
      console.log(
        `[CircuitBreaker] Request queued. Queue size: ${this.requestQueue.length}`
      );
    });
  }

  _processQueue() {
    if (this.requestQueue.length === 0) return;

    console.log(
      `[CircuitBreaker] Processing ${this.requestQueue.length} queued requests`
    );

    const queue = [...this.requestQueue];
    this.requestQueue = [];

    queue.forEach(({ requestFn, options, resolve, reject }) => {
      this.execute(requestFn, options).then(resolve).catch(reject);
    });
  }

  _startMonitoring() {
    this._monitorTimer = setInterval(() => {
      if (this.state === CircuitBreaker.STATES.OPEN) {
        this._attemptReset();
      }
    }, this.monitorInterval);

    if (typeof this._monitorTimer.unref === "function") {
      this._monitorTimer.unref();
    }
  }

  destroy() {
    if (this._monitorTimer) {
      clearInterval(this._monitorTimer);
      this._monitorTimer = null;
    }
    this.requestQueue = [];
    this.listeners = { stateChange: [], success: [], failure: [], reject: [] };
  }

  getStatus() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      halfOpenRequestCount: this.halfOpenRequestCount,
      lastFailureTime: this.lastFailureTime
        ? new Date(this.lastFailureTime).toISOString()
        : null,
      timeUntilRetry:
        this.state === CircuitBreaker.STATES.OPEN && this.lastFailureTime
          ? Math.max(0, this.timeout - (Date.now() - this.lastFailureTime))
          : null,
      queueSize: this.requestQueue.length,
    };
  }
}

class CircuitBreakerOpenError extends Error {
  constructor(message, state, retryAfter) {
    super(message);
    this.name = "CircuitBreakerOpenError";
    this.state = state;
    this.retryAfter = retryAfter;
  }
}

class CircuitBreakerManager {
  constructor() {
    this.breakers = new Map();
  }

  getBreaker(name, options = {}) {
    if (!this.breakers.has(name)) {
      const breaker = new CircuitBreaker(options);
      this.breakers.set(name, breaker);
      console.log(`[CircuitBreakerManager] Created breaker: ${name}`);
    }
    return this.breakers.get(name);
  }

  getAllStatus() {
    const status = {};
    this.breakers.forEach((breaker, name) => {
      status[name] = breaker.getStatus();
    });
    return status;
  }

  destroyAll() {
    this.breakers.forEach((breaker) => breaker.destroy());
    this.breakers.clear();
  }
}

async function fetchWithCircuitBreaker(url, fetchOptions = {}, cbOptions = {}) {
  const manager = fetchWithCircuitBreaker._manager;
  const breakerName = cbOptions.name || new URL(url).origin;
  const breaker = manager.getBreaker(breakerName, cbOptions);

  return breaker.execute(
    () =>
      fetch(url, fetchOptions).then((response) => {
        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        return response;
      }),
    {
      fallback: cbOptions.fallback,
      timeout: cbOptions.requestTimeout,
    }
  );
}

fetchWithCircuitBreaker._manager = new CircuitBreakerManager();

async function demonstrateCircuitBreaker() {
  console.log("=== Circuit Breaker Demo ===\n");

  const breaker = new CircuitBreaker({
    failureThreshold: 3,
    successThreshold: 2,
    timeout: 5000,
    halfOpenMaxRequests: 1,
  });

  breaker
    .on("stateChange", ({ from, to }) => {
      console.log(`\n🔄 STATE CHANGE: ${from} → ${to}\n`);
    })
    .on("failure", ({ failureCount, state }) => {
      console.log(`❌ Failure #${failureCount} in state ${state}`);
    })
    .on("success", ({ state }) => {
      console.log(`✅ Success in state ${state}`);
    })
    .on("reject", ({ state }) => {
      console.log(`🚫 Request rejected - circuit is ${state}`);
    });

  let callCount = 0;

  const mockFailingRequest = () => {
    callCount++;
    return new Promise((_, reject) => {
      setTimeout(() => reject(new Error("Backend unavailable")), 100);
    });
  };

  const mockSuccessRequest = () => {
    callCount++;
    return new Promise((resolve) => {
      setTimeout(() => resolve({ data: "success", call: callCount }), 100);
    });
  };

  console.log("--- Phase 1: Normal requests (should succeed) ---");
  try {
    const result = await breaker.execute(mockSuccessRequest);
    console.log("Request succeeded:", result);
  } catch (e) {
    console.error("Unexpected error:", e.message);
  }

  console.log("\n--- Phase 2: Failing requests (triggering circuit) ---");
  for (let i = 0; i < 4; i++) {
    try {
      await breaker.execute(mockFailingRequest);
    } catch (e) {
      if (e instanceof CircuitBreakerOpenError) {
        console.log(
          `Circuit open! Retry after: ${Math.round(e.retryAfter / 1000)}s`
        );
      }
    }
    console.log("Status:", JSON.stringify(breaker.getStatus(), null, 2));
  }

  console.log("\n--- Phase 3: Request with fallback while circuit is open ---");
  try {
    const result = await breaker.execute(mockFailingRequest, {
      fallback: (error) => ({
        data: "cached_data",
        fromCache: true,
        error: error.message,
      }),
    });
    console.log("Fallback result:", result);
  } catch (e) {
    console.error("Error:", e.message);
  }

  console.log("\n--- Phase 4: Waiting for circuit to attempt reset ---");
  console.log("Waiting 5 seconds for timeout...");
  await new Promise((resolve) => setTimeout(resolve, 5500));

  console.log("\n--- Phase 5: Half-open state - testing recovery ---");
  try {
    const result = await breaker.execute(mockSuccessRequest);
    console.log("Half-open request succeeded:", result);
  } catch (e) {
    console.error("Half-open request failed:", e.message);
  }

  console.log("\n--- Phase 6: Second success to close circuit ---");
  try {
    const result = await breaker.execute(mockSuccessRequest);
    console.log("Circuit closed! Request succeeded:", result);
  } catch (e) {
    console.error("Error:", e.message);
  }

  console.log("\n--- Final Status ---");
  console.log(JSON.stringify(breaker.getStatus(), null, 2));

  breaker.destroy();
}

demonstrateCircuitBreaker().catch(console.error);
