
```python
import anthropic
import pickle
import os
from dataclasses import dataclass, field
from typing import Any


@dataclass
class Player:
    name: str
    level: int
    health: int
    max_health: int
    experience: int
    inventory: list[str] = field(default_factory=list)
    skills: dict[str, int] = field(default_factory=dict)


@dataclass
class GameWorld:
    current_map: str
    position: tuple[int, int]
    discovered_areas: list[str] = field(default_factory=list)
    completed_quests: list[str] = field(default_factory=list)
    active_quests: list[str] = field(default_factory=list)


@dataclass
class GameState:
    player: Player
    world: GameWorld
    game_time: int
    difficulty: str
    save_slot: int
    metadata: dict[str, Any] = field(default_factory=dict)


def save_game_state(game_state: GameState, filename: str) -> bool:
    """Save game state to file using pickle."""
    try:
        with open(filename, 'wb') as f:
            pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)
        print(f"Game state saved to {filename}")
        return True
    except Exception as e:
        print(f"Error saving game state: {e}")
        return False


def load_game_state(filename: str) -> GameState | None:
    """Load game state from file using pickle."""
    try:
        with open(filename, 'rb') as f:
            game_state = pickle.load(f)
        print(f"Game state loaded from {filename}")
        return game_state
    except FileNotFoundError:
        print(f"Save file {filename} not found")
        return None
    except Exception as e:
        print(f"Error loading game state: {e}")
        return None


def create_sample_game_state() -> GameState:
    """Create a sample game state for demonstration."""
    player = Player(
        name="Hero",
        level=5,
        health=85,
        max_health=100,
        experience=1250,
        inventory=["Sword", "Shield", "Health Potion", "Magic Scroll"],
        skills={"Swordsmanship": 3, "Magic": 2, "Stealth": 1}
    )
    
    world = GameWorld(
        current_map="Forest of Shadows",
        position=(42, 17),
        discovered_areas=["Starting Village", "Dark Forest", "Ancient Ruins"],
        completed_quests=["Tutorial Quest", "Find the Lost Sword"],
        active_quests=["Defeat the Dragon", "Rescue the Princess"]
    )
    
    game_state = GameState(
        player=player,
        world=world,
        game_time=3600,
        difficulty="Normal",
        save_slot=1,
        metadata={
            "version": "1.0.0",
            "total_deaths": 3,
            "enemies_defeated": 47,
            "gold_collected": 1500
        }
    )
    
    return game_state


def display_game_state(game_state: GameState) -> str:
    """Convert game state to a readable string format."""
    return f"""
=== Game State ===
Player: {game_state.player.name} (Level {game_state.player.level})
Health: {game_state.player.health}/{game_state.player.max_health}
Experience: {game_state.player.experience}
Inventory: {', '.join(game_state.player.inventory)}
Skills: {game_state.player.skills}

World: {game_state.world.current_map}
Position: {game_state.world.position}
Discovered Areas: {', '.join(game_state.world.discovered_areas)}
Active Quests: {', '.join(game_state.world.active_quests)}
Completed Quests: {', '.join(game_state.world.completed_quests)}

Game Time: {game_state.game_time} seconds
Difficulty: {game_state.difficulty}
Save Slot: {game_state.save_slot}
Metadata: {game_state.metadata}
"""


def analyze_game_state_with_claude(game_state: GameState) -> str:
    """Use Claude to analyze the game state and provide insights."""
    client = anthropic.Anthropic()
    
    game_state_str = display_game_state(game_state)
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""以下のゲームの状態を分析して、プレイヤーへのアドバイスを日本語で提供してください：

{game_state_str}

以下の点について分析してください：
1. プレイヤーの現在の強みと弱み
2. 次に取るべき行動の提案
3. インベントリの最適化アドバイス
4. スキル開発の推奨事項"""
            }
        ]
    )
    
    return message.content[0].text


def main():
    save_file = "game_save.pkl"
    
    print("=== ゲーム状態のシリアライズ/デシリアライズデモ ===\n")
    
    print("1. サンプルゲーム状態を作成中...")
    original_state = create_sample_game_state()
    print(display_game_state(original_state))
    
    print("\n2. ゲーム状態をファイルに保存中...")
    save_success = save_game_state(original_state, save_file)
    
    if save_success:
        file_size = os.path.getsize(save_file)
        print(f"保存ファイルサイズ: {file_size} bytes")
    
    print("\n3. ファイルからゲーム状態を読み込み中...")
    loaded_state = load_game_state(save_file)
    
    if loaded_state:
        print("\n4. 読み込んだゲーム状態を表示:")
        print(display_game_state(loaded_state))
        
        print("\n5. 元の状態と読み込んだ状態が一致するか確認...")
        if (original_state.player.name == loaded_state.player.name and
            original_state.player.level == loaded_state.player.level and
            original_state.world.current_map == loaded_state.world.current_map):
            print("✓ ゲーム状態が正常に保存・復元されました！")
        else:
            print("✗ ゲーム状態の不一致が検出されました")
        
        print("\n6. Claudeによるゲーム状態の分析...")
        analysis = analyze_game_state_with_claude(loaded_state)
        print("\n=== Claudeの分析結果 ===")
        print(analysis)
    
    print("\n7. 複数のセーブスロットのデモ...")
    for slot in range(1, 4):
        state = create_sample_game_state()
        state.save_slot = slot
        state.player.level = slot * 5
        state.player.experience = slot * 1000
        
        slot_file = f"game_save_slot_{slot}.pkl"
        save_game_state(state, slot_file)
        
        loaded = load_game_state(slot_file)
        if loaded:
            print(f"スロット {slot}: {loaded.player.name} - レベル {loaded.player.level}")
        
        if os.path.exists(slot_file):
            os.remove(slot_file)
    
    if os.path.exists(save_file):
        os.remove(save_file)
        print(f"\nクリーンアップ: {save_file} を削除しました")
    
    print("\n=== デモ完了 ===")


if __name__ == "__main__":
    main()
```
