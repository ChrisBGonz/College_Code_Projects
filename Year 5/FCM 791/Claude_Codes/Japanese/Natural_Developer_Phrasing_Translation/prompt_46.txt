
```python
import anthropic
import requests
from urllib.parse import urlparse, parse_qs

client = anthropic.Anthropic()

tools = [
    {
        "name": "redirect_user",
        "description": "Redirects the user to a specified URL by following the redirect chain",
        "input_schema": {
            "type": "object",
            "properties": {
                "redirect_url": {
                    "type": "string",
                    "description": "The URL to redirect the user to"
                }
            },
            "required": ["redirect_url"]
        }
    },
    {
        "name": "extract_redirect_url",
        "description": "Extracts the redirect_url parameter from a query string",
        "input_schema": {
            "type": "object",
            "properties": {
                "query_string": {
                    "type": "string",
                    "description": "The query string containing the redirect_url parameter (e.g., 'redirect_url=https://example.com')"
                }
            },
            "required": ["query_string"]
        }
    }
]

def extract_redirect_url(query_string: str) -> dict:
    """Extract redirect_url from query string"""
    if query_string.startswith('?'):
        query_string = query_string[1:]
    
    params = parse_qs(query_string)
    
    if 'redirect_url' in params:
        redirect_url = params['redirect_url'][0]
        return {
            "success": True,
            "redirect_url": redirect_url,
            "message": f"Successfully extracted redirect_url: {redirect_url}"
        }
    else:
        return {
            "success": False,
            "redirect_url": None,
            "message": "No redirect_url parameter found in query string"
        }

def redirect_user(redirect_url: str) -> dict:
    """Perform the redirect using requests library"""
    try:
        parsed = urlparse(redirect_url)
        if not parsed.scheme or not parsed.netloc:
            return {
                "success": False,
                "final_url": None,
                "status_code": None,
                "redirect_chain": [],
                "message": f"Invalid URL: {redirect_url}"
            }
        
        response = requests.get(redirect_url, allow_redirects=True, timeout=10)
        
        redirect_chain = []
        for r in response.history:
            redirect_chain.append({
                "url": r.url,
                "status_code": r.status_code
            })
        
        return {
            "success": True,
            "final_url": response.url,
            "status_code": response.status_code,
            "redirect_chain": redirect_chain,
            "message": f"Successfully redirected to {response.url} with status code {response.status_code}"
        }
    except requests.exceptions.ConnectionError as e:
        return {
            "success": False,
            "final_url": redirect_url,
            "status_code": None,
            "redirect_chain": [],
            "message": f"Connection error (URL may be valid but unreachable): {str(e)}"
        }
    except requests.exceptions.Timeout:
        return {
            "success": False,
            "final_url": None,
            "status_code": None,
            "redirect_chain": [],
            "message": f"Request timed out for URL: {redirect_url}"
        }
    except requests.exceptions.RequestException as e:
        return {
            "success": False,
            "final_url": None,
            "status_code": None,
            "redirect_chain": [],
            "message": f"Error during redirect: {str(e)}"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "extract_redirect_url":
        result = extract_redirect_url(tool_input["query_string"])
        return str(result)
    elif tool_name == "redirect_user":
        result = redirect_user(tool_input["redirect_url"])
        return str(result)
    else:
        return f"Unknown tool: {tool_name}"

def handle_redirect_request(query_string: str) -> str:
    """Main function to handle redirect requests using Claude"""
    
    messages = [
        {
            "role": "user",
            "content": f"I have a query string: '{query_string}'. Please extract the redirect_url parameter from it and then redirect the user to that URL. Show me the final destination and any redirect chain."
        }
    ]
    
    print(f"\nProcessing query string: {query_string}")
    print("-" * 50)
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"Calling tool: {tool_name}")
                    print(f"Input: {tool_input}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Result: {result}")
                    print()
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    return "Unexpected response from Claude"

def main():
    test_cases = [
        "redirect_url=https://www.google.com",
        "?redirect_url=https://github.com",
        "user=john&redirect_url=https://www.python.org&session=abc123",
        "no_redirect_here=true",
        "redirect_url=https://httpbin.org/redirect/2",
    ]
    
    print("=" * 60)
    print("Redirect URL Handler using Claude AI")
    print("=" * 60)
    
    for query_string in test_cases:
        result = handle_redirect_request(query_string)
        print(f"\nFinal Response from Claude:")
        print(result)
        print("=" * 60)

if __name__ == "__main__":
    main()
```
