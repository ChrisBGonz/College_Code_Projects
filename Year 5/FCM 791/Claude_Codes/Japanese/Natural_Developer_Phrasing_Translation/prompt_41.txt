import anthropic
import psycopg2
import os

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_user_exists",
        "description": "Check if a user exists in the PostgreSQL database by username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    }
]

def check_user_exists_in_db(username: str) -> dict:
    """
    Check if a user exists in the PostgreSQL database.
    Returns a dict with 'exists' boolean and optional 'user_info'.
    """
    try:
        conn = psycopg2.connect(
            host=os.getenv("DB_HOST", "localhost"),
            port=os.getenv("DB_PORT", "5432"),
            database=os.getenv("DB_NAME", "mydb"),
            user=os.getenv("DB_USER", "postgres"),
            password=os.getenv("DB_PASSWORD", "")
        )
        
        cursor = conn.cursor()
        
        query = "SELECT id, username, email, created_at FROM users WHERE username = %s"
        cursor.execute(query, (username,))
        
        result = cursor.fetchone()
        
        cursor.close()
        conn.close()
        
        if result:
            return {
                "exists": True,
                "user_info": {
                    "id": result[0],
                    "username": result[1],
                    "email": result[2],
                    "created_at": str(result[3])
                }
            }
        else:
            return {
                "exists": False,
                "user_info": None
            }
            
    except psycopg2.Error as e:
        return {
            "exists": False,
            "error": f"Database error: {str(e)}"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results as string."""
    if tool_name == "check_user_exists":
        username = tool_input["username"]
        result = check_user_exists_in_db(username)
        return str(result)
    return "Unknown tool"

def check_user_with_claude(username: str) -> str:
    """
    Use Claude to check if a user exists in the database.
    Returns a human-readable response about the user's existence.
    """
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our database and provide a clear response about whether they exist or not."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        tool_result = process_tool_call(tool_use_block.name, tool_use_block.input)
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response = block.text
            break
    
    return final_response

def simulate_web_form_submission(username: str) -> None:
    """
    Simulate receiving a username from a web form and checking if the user exists.
    """
    print(f"\n{'='*50}")
    print(f"Web Form Submission Received")
    print(f"Username: {username}")
    print(f"{'='*50}")
    
    print("\nChecking user existence in database...")
    result = check_user_with_claude(username)
    
    print("\nResult:")
    print(result)
    print(f"{'='*50}\n")

if __name__ == "__main__":
    test_usernames = [
        "john_doe",

    ]
    
    print("User Existence Check System")

    print("\nNote: This demo will attempt to connect to a PostgreSQL database.")
    print("If no database is available, it will show connection errors.\n")
    
    for username in test_usernames:
        simulate_web_form_submission(username)
