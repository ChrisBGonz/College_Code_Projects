from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

SYSTEM_PROMPT = """You are a mathematical expression calculator. Your role is to:
1. Evaluate mathematical expressions provided by users
2. Show the step-by-step calculation process when appropriate
3. Provide the final numerical result
4. Handle various types of mathematical operations including:
   - Basic arithmetic (+, -, *, /)
   - Powers and roots
   - Trigonometric functions
   - Logarithms
   - Complex expressions
5. If the expression is invalid or cannot be computed, explain why
6. Always provide the answer in a clear, structured format

When responding, always include:
- The original expression
- Step-by-step solution (if applicable)
- Final result clearly marked"""


@app.route("/calculate", methods=["POST"])
def calculate():
    """Route to calculate mathematical expressions using Claude AI."""
    data = request.get_json()

    if not data or "expression" not in data:
        return jsonify({"error": "Please provide a mathematical expression in the 'expression' field"}), 400

    expression = data["expression"].strip()

    if not expression:
        return jsonify({"error": "Expression cannot be empty"}), 400

    conversation_history.append({"role": "user", "content": f"Please calculate the following mathematical expression: {expression}"})

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history,
    )

    assistant_message = response.content[0].text

    conversation_history.append({"role": "assistant", "content": assistant_message})

    return jsonify(
        {
            "expression": expression,
            "result": assistant_message,
            "conversation_length": len(conversation_history) // 2,
        }
    )


@app.route("/calculate/followup", methods=["POST"])
def calculate_followup():
    """Route for follow-up questions about previous calculations."""
    data = request.get_json()

    if not data or "question" not in data:
        return jsonify({"error": "Please provide a follow-up question in the 'question' field"}), 400

    question = data["question"].strip()

    if not question:
        return jsonify({"error": "Question cannot be empty"}), 400

    if not conversation_history:
        return jsonify({"error": "No previous calculation found. Please use /calculate first"}), 400

    conversation_history.append({"role": "user", "content": question})

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history,
    )

    assistant_message = response.content[0].text

    conversation_history.append({"role": "assistant", "content": assistant_message})

    return jsonify({"question": question, "answer": assistant_message, "conversation_length": len(conversation_history) // 2})


@app.route("/calculate/history", methods=["GET"])
def get_history():
    """Route to get conversation history."""
    formatted_history = []
    for i, message in enumerate(conversation_history):
        formatted_history.append({"index": i + 1, "role": message["role"], "content": message["content"]})

    return jsonify({"history": formatted_history, "total_messages": len(conversation_history)})


@app.route("/calculate/reset", methods=["POST"])
def reset_conversation():
    """Route to reset conversation history."""
    global conversation_history
    conversation_history = []
    return jsonify({"message": "Conversation history has been reset"})


@app.route("/health", methods=["GET"])
def health_check():
    """Health check endpoint."""
    return jsonify({"status": "healthy", "service": "Mathematical Expression Calculator"})


if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=5000)
