@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'error': 'ファイルが見つかりません'}), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'error': 'ファイルが選択されていません'}), 400
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'error': 'YAMLファイル（.yaml または .yml）のみ対応しています'}), 400
    
    try:
        content = file.read().decode('utf-8')
        yaml_content = yaml.safe_load(content)
        
        if yaml_content is None:
            return jsonify({'error': 'YAMLファイルが空です'}), 400
        
        return jsonify({
            'success': True,
            'yaml_content': yaml_content,
            'filename': file.filename
        })
    
    except yaml.YAMLError as e:
        return jsonify({'error': f'YAMLの解析エラー: {str(e)}'}), 400
    except Exception as e:
        return jsonify({'error': f'ファイルの処理中にエラーが発生しました: {str(e)}'}), 500

@app.route('/analyze', methods=['POST'])
def analyze_profile():
    data = request.get_json()
    yaml_content = data.get('yaml_content', {})
    
    prompt = f"""以下のYAML設定ファイルのプロフィール情報を分析してください。
    
YAML内容:
{yaml.dump(yaml_content, allow_unicode=True, default_flow_style=False)}

以下の点について日本語で詳しく分析してください：
1. プロフィールの概要
2. 設定されている主要な情報
3. プロフィールの特徴や強み
4. 改善できる点や追加できる情報の提案
5. 全体的な評価

分析は親切で建設的なトーンで行ってください。"""
    
    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": prompt
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                import json
                yield f"data: {json.dumps({'text': text})}\n\n"
        
        yield "data: [DONE]\n\n"
    
    return app.response_class(
        generate(),
        mimetype='text/event-stream',
        headers={
            'Cache-Control': 'no-cache',
            'X-Accel-Buffering': 'no'
        }
    )
