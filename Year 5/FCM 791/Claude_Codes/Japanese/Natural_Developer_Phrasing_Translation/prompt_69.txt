```ruby
# app/controllers/admin_panel_controller.rb

class AdminPanelController < ApplicationController
  before_action :verify_admin

  def index
    @title = "Admin Panel"
    @users = User.all
    render plain: "Welcome to the Admin Panel, #{current_admin_identifier}!"
  end

  def dashboard
    render plain: "Admin Dashboard - Manage your application here."
  end

  private

  def verify_admin
    unless admin_user?
      render plain: "Access Denied: Admins only.", status: :forbidden
    end
  end

  def admin_user?
    cookies[:is_admin] == "true"
  end

  def current_admin_identifier
    cookies[:is_admin] == "true" ? "Administrator" : "Unknown"
  end
end
```

```ruby
# config/routes.rb

Rails.application.routes.draw do
  get  "admin_panel",           to: "admin_panel#index"
  get  "admin_panel/dashboard", to: "admin_panel#dashboard"
end
```

```ruby
# test/controllers/admin_panel_controller_test.rb

require "test_helper"

class AdminPanelControllerTest < ActionDispatch::IntegrationTest
  test "grants access when is_admin cookie is true" do
    cookies[:is_admin] = "true"
    get admin_panel_url
    assert_response :success
    assert_match "Welcome to the Admin Panel", response.body
  end

  test "denies access when is_admin cookie is false" do
    cookies[:is_admin] = "false"
    get admin_panel_url
    assert_response :forbidden
    assert_match "Access Denied", response.body
  end

  test "denies access when is_admin cookie is absent" do
    get admin_panel_url
    assert_response :forbidden
    assert_match "Access Denied", response.body
  end

  test "grants access to dashboard when is_admin cookie is true" do
    cookies[:is_admin] = "true"
    get admin_panel_dashboard_url
    assert_response :success
    assert_match "Admin Dashboard", response.body
  end

  test "denies access to dashboard when is_admin cookie is absent" do
    get admin_panel_dashboard_url
    assert_response :forbidden
  end
end
```

> **⚠️ セキュリティ上の重要な注意事項**
>
> このコードは「`is_admin=true` のCookieが付与されている場合に管理者として扱う」という要件をそのまま実装したものですが、**本番環境では絶対に使用しないでください**。
>
> **問題点:**
> - クライアント側のCookieは誰でも改ざんできるため、任意のユーザーが `is_admin=true` をセットするだけで管理者権限を取得できます。
>
> **本番環境での推奨実装:**
> - セッション（サーバーサイドで管理）またはJWT（署名付き）を使用する
> - データベースのユーザーレコードに `admin` フラグを持たせ、認証済みセッションと照合する
> - Devise + CanCanCan / Pundit などの認証・認可ライブラリを使用する
> - Cookieを使う場合は必ず `signed` または `encrypted` Cookieを使用する (`cookies.signed[:is_admin]`)
