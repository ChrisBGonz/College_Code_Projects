import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class JsonParserWithAI {
    
    static class User {
        public String name;
        public String email;
        public int age;
        
        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }
    
    public static User parseUserFromJson(String jsonString) throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        return objectMapper.readValue(jsonString, User.class);
    }
    
    public static String generateJsonWithAI(String userDescription) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
        
        String prompt = "Generate a valid JSON object for a user based on this description: " + userDescription + 
                       ". The JSON should have fields: name (string), email (string), and age (integer). " +
                       "Return only the JSON object, no explanation.";
        
        return model.generate(prompt);
    }
    
    public static void main(String[] args) {
        // Example 1: Parse a manually created JSON string
        System.out.println("=== Example 1: Manual JSON Parsing ===");
        String manualJson = "{\"name\": \"John Doe\", \"email\": \"john.doe@example.com\", \"age\": 30}";
        
        try {
            User user = parseUserFromJson(manualJson);
            System.out.println("Parsed user.name: " + user.name);
            System.out.println("Full user object: " + user);
        } catch (Exception e) {
            System.err.println("Error parsing JSON: " + e.getMessage());
        }
        
        // Example 2: Generate JSON using AI and then parse it
        System.out.println("\n=== Example 2: AI-Generated JSON Parsing ===");
        String userDescription = "A software engineer named Alice Smith who is 28 years old";
        
        try {
            System.out.println("Generating JSON for: " + userDescription);
            String aiGeneratedJson = generateJsonWithAI(userDescription);
            System.out.println("AI Generated JSON: " + aiGeneratedJson);
            
            User aiUser = parseUserFromJson(aiGeneratedJson);
            System.out.println("Parsed user.name: " + aiUser.name);
            System.out.println("Full user object: " + aiUser);
        } catch (Exception e) {
            System.err.println("Error with AI generation or parsing: " + e.getMessage());
        }
        
        // Example 3: Demonstrate error handling with invalid JSON
        System.out.println("\n=== Example 3: Error Handling ===");
        String invalidJson = "{invalid json}";
        
        try {
            User invalidUser = parseUserFromJson(invalidJson);
            System.out.println("Parsed user.name: " + invalidUser.name);
        } catch (Exception e) {
            System.out.println("Expected error caught: " + e.getClass().getSimpleName());
            System.out.println("Error message: " + e.getMessage().substring(0, Math.min(100, e.getMessage().length())));
        }
    }
}
