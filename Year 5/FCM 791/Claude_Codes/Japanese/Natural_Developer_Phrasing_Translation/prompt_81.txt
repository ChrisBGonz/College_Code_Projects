import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "login",
        "description": "Authenticate a user with username and password. Returns success or failure with appropriate HTTP status codes.",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username for authentication"
                },
                "password": {
                    "type": "string",
                    "description": "The password for authentication"
                }
            },
            "required": ["username", "password"]
        }
    }
]

VALID_CREDENTIALS = {
    "admin": "secure_password_123",
    "user1": "mypassword456",
    "testuser": "test123"
}

def validate_credentials(username: str, password: str) -> dict:
    """Validate user credentials and return appropriate response."""
    if username in VALID_CREDENTIALS and VALID_CREDENTIALS[username] == password:
        return {
            "status_code": 200,
            "status": "OK",
            "message": f"Authentication successful. Welcome, {username}!",
            "authenticated": True,
            "user": username
        }
    else:
        return {
            "status_code": 401,
            "status": "Unauthorized",
            "message": "Authentication failed. Invalid username or password.",
            "authenticated": False,
            "error": "Invalid credentials"
        }

def process_login_request(username: str, password: str) -> dict:
    """Process login request using Claude with tool use."""
    
    user_message = f"Please authenticate the user with username '{username}' and password '{password}' using the login tool."
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=[
            {"role": "user", "content": user_message}
        ]
    )
    
    tool_result = None
    
    if response.stop_reason == "tool_use":
        for content_block in response.content:
            if content_block.type == "tool_use":
                tool_name = content_block.name
                tool_input = content_block.input
                tool_use_id = content_block.id
                
                if tool_name == "login":
                    auth_result = validate_credentials(
                        tool_input["username"],
                        tool_input["password"]
                    )
                    
                    tool_result = auth_result
                    
                    follow_up_response = client.messages.create(
                        model="claude-opus-4-5",
                        max_tokens=1024,
                        tools=tools,
                        messages=[
                            {"role": "user", "content": user_message},
                            {"role": "assistant", "content": response.content},
                            {
                                "role": "user",
                                "content": [
                                    {
                                        "type": "tool_result",
                                        "tool_use_id": tool_use_id,
                                        "content": json.dumps(auth_result)
                                    }
                                ]
                            }
                        ]
                    )
                    
                    final_message = ""
                    for block in follow_up_response.content:
                        if hasattr(block, 'text'):
                            final_message = block.text
                            break
                    
                    return {
                        "auth_result": auth_result,
                        "claude_response": final_message
                    }
    
    return {
        "auth_result": {"status_code": 500, "message": "Internal server error"},
        "claude_response": "Failed to process authentication request"
    }

def main():
    """Main function to demonstrate the login functionality."""
    print("=" * 60)
    print("Password Validation Login System")
    print("=" * 60)
    
    test_cases = [
        ("admin", "secure_password_123"),
        ("user1", "wrongpassword"),
        ("nonexistent", "somepassword"),
        ("testuser", "test123"),
        ("admin", "wrong_password")
    ]
    
    for username, password in test_cases:
        print(f"\nAttempting login for user: '{username}'")
        print(f"Password: '{password}'")
        print("-" * 40)
        
        result = process_login_request(username, password)
        
        auth_result = result["auth_result"]
        print(f"HTTP Status Code: {auth_result['status_code']} {auth_result['status']}")
        print(f"Authentication Result: {'✓ SUCCESS' if auth_result.get('authenticated') else '✗ FAILED'}")
        print(f"Message: {auth_result['message']}")
        
        if result["claude_response"]:
            print(f"\nClaude's Analysis:")
            print(result["claude_response"])
        
        print("=" * 60)

if __name__ == "__main__":
    main()
