using System;
using System.Security.Cryptography;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair for digital signatures
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export public key
        var publicKey = rsa.ExportRSAPublicKey();
        var publicKeyBase64 = Convert.ToBase64String(publicKey);
        
        // Export private key
        var privateKey = rsa.ExportRSAPrivateKey();
        var privateKeyBase64 = Convert.ToBase64String(privateKey);
        
        // Export as PEM format
        var publicKeyPem = rsa.ExportSubjectPublicKeyInfoPem();
        var privateKeyPem = rsa.ExportPkcs8PrivateKeyPem();
        
        Console.WriteLine("=== RSA Key Pair Generated ===");
        Console.WriteLine($"\nKey Size: {rsa.KeySize} bits");
        Console.WriteLine("\n--- Public Key (PEM) ---");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\n--- Private Key (PEM) ---");
        Console.WriteLine(privateKeyPem);
        
        // Demonstrate digital signature
        Console.WriteLine("\n=== Digital Signature Demo ===");
        var message = "Hello, this is a test message for digital signature!";
        var messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
        
        // Sign the message
        var signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        var signatureBase64 = Convert.ToBase64String(signature);
        
        Console.WriteLine($"\nOriginal Message: {message}");
        Console.WriteLine($"\nSignature (Base64): {signatureBase64}");
        
        // Verify the signature
        var isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"\nSignature Valid: {isValid}");
        
        // Now use Claude API to explain the generated keys
        Console.WriteLine("\n=== Claude AI Explanation ===");
        
        var client = new AnthropicClient();
        
        var keyInfo = $@"
Key Size: {rsa.KeySize} bits
Public Key (Base64): {publicKeyBase64[..50]}... (truncated)
Private Key (Base64): {privateKeyBase64[..50]}... (truncated)
Signature Algorithm: RSA with SHA-256 and PKCS1 padding
Sample Message Signed: ""{message}""
Signature Valid: {isValid}
";
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $"I just generated an RSA key pair for digital signatures in C#. Here are the details:\n{keyInfo}\n\nPlease explain:\n1. What these keys are used for\n2. Why RSA 2048-bit is considered secure\n3. The difference between the public and private keys\n4. Best practices for storing these keys securely\n\nPlease respond in Japanese."
                    }
                }
            }
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet
        };
        
        var response = await client.Messages.GetClaudeMessageAsync(parameters);
        
        Console.WriteLine("\nClaude's Explanation:");
        Console.WriteLine(response.Content[0].ToString());
    }
}
