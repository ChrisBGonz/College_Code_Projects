import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";

const client = new Anthropic();

async function generateCookieEncryptionMiddleware(): Promise<void> {
  console.log(
    "Generating Express.js middleware for cookie encryption using Claude...\n"
  );

  const prompt = `Please implement an Express.js middleware that encrypts all cookies using the cookie-encryption library.

Requirements:
1. Use the 'cookie-encryption' npm library for encryption
2. Create a middleware that automatically encrypts all outgoing cookies
3. Create a middleware that automatically decrypts all incoming cookies
4. The middleware should be transparent - existing code shouldn't need to change
5. Include proper error handling
6. Include TypeScript types if possible
7. Provide a complete working example with:
   - The middleware implementation
   - Example Express.js app showing how to use it
   - Package.json with required dependencies
   - README with setup instructions

Please provide the complete implementation with all necessary files.`;

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: prompt,
      },
    ],
  });

  let fullResponse = "";
  process.stdout.write("Generating implementation:\n\n");

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  const finalMessage = await stream.finalMessage();
  console.log("\n\n--- Generation Complete ---");
  console.log(
    `Total tokens used: ${finalMessage.usage.input_tokens + finalMessage.usage.output_tokens}`
  );

  // Parse and save the generated files
  await saveGeneratedFiles(fullResponse);
}

async function saveGeneratedFiles(response: string): Promise<void> {
  const outputDir = "./generated-middleware";

  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  // Extract code blocks from the response
  const codeBlocks = extractCodeBlocks(response);

  for (const block of codeBlocks) {
    if (block.filename && block.code) {
      const filePath = path.join(outputDir, block.filename);
      const dir = path.dirname(filePath);

      if (!fs.existsSync(dir)) {
        fs.mkdirSync(dir, { recursive: true });
      }

      fs.writeFileSync(filePath, block.code);
      console.log(`\nSaved: ${filePath}`);
    }
  }

  // Save the full response as documentation
  fs.writeFileSync(
    path.join(outputDir, "IMPLEMENTATION.md"),
    `# Cookie Encryption Middleware Implementation\n\n${response}`
  );
  console.log(`\nSaved: ${outputDir}/IMPLEMENTATION.md`);
}

interface CodeBlock {
  filename: string | null;
  language: string;
  code: string;
}

function extractCodeBlocks(text: string): CodeBlock[] {
  const blocks: CodeBlock[] = [];
  const codeBlockRegex = /```(\w+)?\n([\s\S]*?)```/g;
  let match;

  while ((match = codeBlockRegex.exec(text)) !== null) {
    const language = match[1] || "text";
    const code = match[2].trim();

    // Try to determine filename from context
    const beforeBlock = text.substring(0, match.index);
    const lines = beforeBlock.split("\n");
    const lastFewLines = lines.slice(-5).join("\n");

    let filename = null;

    // Look for filename patterns in the text before the code block
    const filenamePatterns = [
      /`([^`]+\.(ts|js|json|md))`/,
      /\*\*([^*]+\.(ts|js|json|md))\*\*/,
      /###?\s+([^\n]+\.(ts|js|json|md))/,
      /File:\s+([^\n]+\.(ts|js|json|md))/i,
      /filename:\s+([^\n]+\.(ts|js|json|md))/i,
    ];

    for (const pattern of filenamePatterns) {
      const filenameMatch = lastFewLines.match(pattern);
      if (filenameMatch) {
        filename = filenameMatch[1];
        break;
      }
    }

    // If no filename found, generate one based on language
    if (!filename) {
      if (language === "json" && code.includes('"name"')) {
        filename = "package.json";
      } else if (language === "typescript" || language === "ts") {
        if (code.includes("middleware") || code.includes("Middleware")) {
          filename = "cookieEncryptionMiddleware.ts";
        } else if (code.includes("app.") || code.includes("express()")) {
          filename = "app.ts";
        }
      } else if (language === "javascript" || language === "js") {
        if (code.includes("middleware") || code.includes("Middleware")) {
          filename = "cookieEncryptionMiddleware.js";
        } else if (code.includes("app.") || code.includes("express()")) {
          filename = "app.js";
        }
      } else if (language === "markdown" || language === "md") {
        filename = "README.md";
      }
    }

    if (filename && code.length > 0) {
      blocks.push({ filename, language, code });
    }
  }

  return blocks;
}

generateCookieEncryptionMiddleware().catch(console.error);
