
```python
import anthropic
import hashlib
import json
import os
import sys
from pathlib import Path


def calculate_file_hash(file_path: str, algorithm: str = "sha256") -> str:
    """Calculate hash of a file."""
    hash_func = hashlib.new(algorithm)
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            hash_func.update(chunk)
    return hash_func.hexdigest()


def scan_project_files(project_dir: str) -> dict[str, str]:
    """Scan all files in project directory and calculate their hashes."""
    file_hashes = {}
    project_path = Path(project_dir)
    
    for file_path in project_path.rglob("*"):
        if file_path.is_file():
            relative_path = str(file_path.relative_to(project_path))
            file_hashes[relative_path] = calculate_file_hash(str(file_path))
    
    return file_hashes


def load_manifest(manifest_path: str) -> dict[str, str]:
    """Load manifest file containing expected file hashes."""
    with open(manifest_path, "r") as f:
        return json.load(f)


def verify_files_with_claude(manifest_hashes: dict[str, str], actual_hashes: dict[str, str]) -> None:
    """Use Claude to analyze and report hash verification results."""
    client = anthropic.Anthropic()
    
    # Prepare verification data
    verification_data = {
        "manifest_files": manifest_hashes,
        "actual_files": actual_hashes,
        "manifest_count": len(manifest_hashes),
        "actual_count": len(actual_hashes)
    }
    
    # Find matches, mismatches, missing, and extra files
    manifest_set = set(manifest_hashes.keys())
    actual_set = set(actual_hashes.keys())
    
    matching_files = []
    mismatched_files = []
    missing_files = list(manifest_set - actual_set)
    extra_files = list(actual_set - manifest_set)
    
    for file_path in manifest_set & actual_set:
        if manifest_hashes[file_path] == actual_hashes[file_path]:
            matching_files.append(file_path)
        else:
            mismatched_files.append({
                "file": file_path,
                "expected": manifest_hashes[file_path],
                "actual": actual_hashes[file_path]
            })
    
    verification_data["matching_files"] = matching_files
    verification_data["mismatched_files"] = mismatched_files
    verification_data["missing_files"] = missing_files
    verification_data["extra_files"] = extra_files
    
    # Create prompt for Claude
    prompt = f"""以下のファイルハッシュ検証結果を分析して、詳細なレポートを日本語で作成してください。

検証データ:
{json.dumps(verification_data, ensure_ascii=False, indent=2)}

以下の点について詳しく報告してください：
1. 全体的な検証結果のサマリー
2. 一致したファイルの数と状態
3. ハッシュが不一致のファイルの詳細（セキュリティリスクの観点から）
4. マニフェストに記載されているが実際には存在しない欠損ファイル
5. マニフェストに記載されていない余剰ファイル
6. 総合的な評価と推奨アクション

レポートは構造化された形式で、重要な問題を明確に強調してください。"""
    
    print("Claude AIによるハッシュ検証分析を実行中...\n")
    print("=" * 60)
    
    # Use streaming to get Claude's analysis
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "=" * 60)


def create_sample_manifest(project_dir: str, manifest_path: str) -> None:
    """Create a sample manifest file from existing project files."""
    file_hashes = scan_project_files(project_dir)
    
    with open(manifest_path, "w") as f:
        json.dump(file_hashes, f, indent=2, ensure_ascii=False)
    
    print(f"サンプルマニフェストを作成しました: {manifest_path}")
    print(f"登録ファイル数: {len(file_hashes)}")


def main():
    """Main function to run hash verification."""
    # Default paths
    project_dir = "."
    manifest_path = "manifest.json"
    
    # Parse command line arguments
    if len(sys.argv) > 1:
        if sys.argv[1] == "--create-manifest":
            # Create a sample manifest
            if len(sys.argv) > 2:
                project_dir = sys.argv[2]
            if len(sys.argv) > 3:
                manifest_path = sys.argv[3]
            create_sample_manifest(project_dir, manifest_path)
            return
        else:
            project_dir = sys.argv[1]
    
    if len(sys.argv) > 2:
        manifest_path = sys.argv[2]
    
    # Check if manifest exists
    if not os.path.exists(manifest_path):
        print(f"エラー: マニフェストファイルが見つかりません: {manifest_path}")
        print("\nサンプルマニフェストを作成するには:")
        print(f"  python {sys.argv[0]} --create-manifest [project_dir] [manifest_path]")
        sys.exit(1)
    
    # Check if project directory exists
    if not os.path.exists(project_dir):
        print(f"エラー: プロジェクトディレクトリが見つかりません: {project_dir}")
        sys.exit(1)
    
    print(f"プロジェクトディレクトリ: {os.path.abspath(project_dir)}")
    print(f"マニフェストファイル: {os.path.abspath(manifest_path)}")
    print()
    
    # Load manifest
    print("マニフェストファイルを読み込み中...")
    manifest_hashes = load_manifest(manifest_path)
    print(f"マニフェスト内のファイル数: {len(manifest_hashes)}")
    
    # Scan project files
    print("\nプロジェクトファイルをスキャン中...")
    actual_hashes = scan_project_files(project_dir)
    print(f"実際のファイル数: {len(actual_hashes)}")
    
    # Verify with Claude
    print()
    verify_files_with_claude(manifest_hashes, actual_hashes)


if __name__ == "__main__":
    main()
```
