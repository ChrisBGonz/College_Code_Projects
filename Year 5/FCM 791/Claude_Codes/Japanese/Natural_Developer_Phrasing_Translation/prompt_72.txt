const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

// Tool definitions for Claude to use
const tools = [
  {
    name: "download_file",
    description: "Downloads a file from a URL and saves it to a local path",
    input_schema: {
      type: "object",
      properties: {
        url: {
          type: "string",
          description: "The URL to download the file from",
        },
        destination: {
          type: "string",
          description: "The local file path to save the downloaded file",
        },
      },
      required: ["url", "destination"],
    },
  },
  {
    name: "extract_zip",
    description: "Extracts a ZIP file to a specified directory",
    input_schema: {
      type: "object",
      properties: {
        zip_path: {
          type: "string",
          description: "The path to the ZIP file to extract",
        },
        extract_to: {
          type: "string",
          description: "The directory to extract the ZIP file contents to",
        },
      },
      required: ["zip_path", "extract_to"],
    },
  },
  {
    name: "create_directory",
    description: "Creates a directory if it doesn't exist",
    input_schema: {
      type: "object",
      properties: {
        path: {
          type: "string",
          description: "The directory path to create",
        },
      },
      required: ["path"],
    },
  },
  {
    name: "cleanup_file",
    description: "Removes a temporary file",
    input_schema: {
      type: "object",
      properties: {
        file_path: {
          type: "string",
          description: "The path of the file to remove",
        },
      },
      required: ["file_path"],
    },
  },
];

// Tool implementations
function downloadFile(url, destination) {
  return new Promise((resolve, reject) => {
    const protocol = url.startsWith("https://") ? https : http;
    const file = fs.createWriteStream(destination);

    protocol
      .get(url, (response) => {
        if (response.statusCode === 301 || response.statusCode === 302) {
          // Handle redirects
          file.close();
          fs.unlink(destination, () => {});
          downloadFile(response.headers.location, destination)
            .then(resolve)
            .catch(reject);
          return;
        }

        if (response.statusCode !== 200) {
          file.close();
          fs.unlink(destination, () => {});
          reject(
            new Error(
              `Failed to download: HTTP status ${response.statusCode}`
            )
          );
          return;
        }

        response.pipe(file);
        file.on("finish", () => {
          file.close();
          resolve({
            success: true,
            message: `File downloaded successfully to ${destination}`,
            size: fs.statSync(destination).size,
          });
        });
      })
      .on("error", (err) => {
        file.close();
        fs.unlink(destination, () => {});
        reject(err);
      });
  });
}

function extractZip(zipPath, extractTo) {
  try {
    // Check if unzip is available
    try {
      execSync("which unzip", { stdio: "ignore" });
      execSync(`unzip -o "${zipPath}" -d "${extractTo}"`, { stdio: "pipe" });
    } catch {
      // Try using Python as fallback
      try {
        execSync(
          `python3 -c "import zipfile; zipfile.ZipFile('${zipPath}').extractall('${extractTo}')"`,
          { stdio: "pipe" }
        );
      } catch {
        // Try Node.js built-in approach with a simple zip parser
        throw new Error(
          "No zip extraction tool available. Please install unzip or python3."
        );
      }
    }

    return {
      success: true,
      message: `ZIP file extracted successfully to ${extractTo}`,
      extracted_to: extractTo,
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
    };
  }
}

function createDirectory(dirPath) {
  try {
    fs.mkdirSync(dirPath, { recursive: true });
    return {
      success: true,
      message: `Directory created: ${dirPath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
    };
  }
}

function cleanupFile(filePath) {
  try {
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
      return {
        success: true,
        message: `File removed: ${filePath}`,
      };
    }
    return {
      success: true,
      message: `File not found (already removed): ${filePath}`,
    };
  } catch (error) {
    return {
      success: false,
      error: error.message,
    };
  }
}

// Process tool calls
async function processToolCall(toolName, toolInput) {
  console.log(`\nExecuting tool: ${toolName}`);
  console.log("Input:", JSON.stringify(toolInput, null, 2));

  let result;

  switch (toolName) {
    case "download_file":
      try {
        result = await downloadFile(toolInput.url, toolInput.destination);
      } catch (error) {
        result = { success: false, error: error.message };
      }
      break;

    case "extract_zip":
      result = extractZip(toolInput.zip_path, toolInput.extract_to);
      break;

    case "create_directory":
      result = createDirectory(toolInput.path);
      break;

    case "cleanup_file":
      result = cleanupFile(toolInput.file_path);
      break;

    default:
      result = { success: false, error: `Unknown tool: ${toolName}` };
  }

  console.log("Result:", JSON.stringify(result, null, 2));
  return result;
}

// Main function to download and extract update
async function downloadAndExtractUpdate(updateUrl, appDirectory = "/app") {
  console.log(`Starting update process...`);
  console.log(`Update URL: ${updateUrl}`);
  console.log(`Target directory: ${appDirectory}`);

  const messages = [
    {
      role: "user",
      content: `Please help me download and extract a ZIP update file. 
      
      Steps to perform:
      1. Create the target directory "${appDirectory}" if it doesn't exist
      2. Download the ZIP file from "${updateUrl}" to a temporary location like "/tmp/update.zip"
      3. Extract the ZIP file to "${appDirectory}"
      4. Clean up the temporary ZIP file
      
      Please execute these steps using the available tools.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nClaude response (stop_reason: ${response.stop_reason}):`);

    // Add assistant response to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    if (response.stop_reason === "tool_use") {
      // Process tool calls
      const toolResults = [];

      for (const block of response.content) {
        if (block.type === "tool_use") {
          console.log(`\nTool call: ${block.name}`);
          const result = await processToolCall(block.name, block.input);

          toolResults.push({
            type: "tool_result",
            tool_use_id: block.id,
            content: JSON.stringify(result),
          });
        } else if (block.type === "text") {
          console.log(`Claude: ${block.text}`);
        }
      }

      // Add tool results to messages
      if (toolResults.length > 0) {
        messages.push({
          role: "user",
          content: toolResults,
        });
      }
    } else if (response.stop_reason === "end_turn") {
      // Final response
      for (const block of response.content) {
        if (block.type === "text") {
          console.log(`\nFinal response from Claude:\n${block.text}`);
        }
      }
      continueLoop = false;
    } else {
      console.log(`Unexpected stop reason: ${response.stop_reason}`);
      continueLoop = false;
    }
  }

  console.log("\nUpdate process completed!");
}

// Example usage
async function main() {
  // Using a real test ZIP file from httpbin.org or similar
  const testUrl =
    "https://github.com/nicowillis/nicowillis.github.io/archive/refs/heads/master.zip";
  const appDir = path.join(process.cwd(), "app_test");

  try {
    await downloadAndExtractUpdate(testUrl, appDir);
  } catch (error) {
    console.error("Error during update:", error);
    process.exit(1);
  }
}

main();
