
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const systemPrompt = `You are an expert Node.js and Express.js developer. Your task is to help create a comprehensive request logging middleware for Express.js applications.

The middleware should:
1. Log HTTP method, URL, and all headers
2. Log the complete request body including raw data
3. Be useful for debugging purposes
4. Handle various content types (JSON, form data, raw text, binary, etc.)
5. Be production-ready with proper error handling

When asked to create or modify the middleware, provide complete, working code with detailed explanations.`;

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function generateMiddleware(): Promise<void> {
  console.log("Generating Express.js request logging middleware...\n");

  // Initial request to generate the middleware
  const initialRequest = `Create a comprehensive Node.js Express middleware that logs all incoming requests with the following features:
  
1. Log HTTP method (GET, POST, PUT, DELETE, etc.)
2. Log the full URL including query parameters
3. Log all request headers
4. Log the complete request body including raw data
5. Handle different content types:
   - application/json
   - application/x-www-form-urlencoded
   - multipart/form-data
   - text/plain
   - application/octet-stream (binary data)
6. Show timestamp for each request
7. Format the output in a readable way for debugging
8. Include error handling
9. Make it easy to configure (e.g., enable/disable certain logging features)

Please provide the complete middleware code with usage examples.`;

  const middlewareCode = await chat(initialRequest);
  console.log("=== Generated Middleware ===\n");
  console.log(middlewareCode);
  console.log("\n=== End of Middleware ===\n");

  // Start interactive conversation
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log(
    "You can now ask questions or request modifications to the middleware."
  );
  console.log('Type "exit" to quit.\n');

  const askQuestion = (): void => {
    rl.question("You: ", async (input: string) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        console.log("\nAssistant: ");
        const response = await chat(userInput);
        console.log(response);
        console.log("\n");
      } catch (error) {
        console.error("Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

generateMiddleware().catch(console.error);
```
