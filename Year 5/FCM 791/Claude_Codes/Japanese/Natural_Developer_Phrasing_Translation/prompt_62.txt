import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulate a session store (in production, this would be Redis or a database)
const sessionStore = new Map();

// Initialize some test sessions
sessionStore.set("session_123", {
  userId: "user_001",
  username: "john_doe",
  email: "john@example.com",
  loginTime: new Date().toISOString(),
  lastActivity: new Date().toISOString(),
});

sessionStore.set("session_456", {
  userId: "user_002",
  username: "jane_smith",
  email: "jane@example.com",
  loginTime: new Date().toISOString(),
  lastActivity: new Date().toISOString(),
});

// Tool definitions for session management
const tools = [
  {
    name: "get_session",
    description:
      "Retrieves session information for a given session ID from the session store",
    input_schema: {
      type: "object",
      properties: {
        session_id: {
          type: "string",
          description: "The session ID to look up",
        },
      },
      required: ["session_id"],
    },
  },
  {
    name: "delete_session",
    description:
      "Deletes a session from the session store, effectively logging out the user",
    input_schema: {
      type: "object",
      properties: {
        session_id: {
          type: "string",
          description: "The session ID to delete",
        },
      },
      required: ["session_id"],
    },
  },
  {
    name: "invalidate_all_user_sessions",
    description: "Invalidates all sessions for a specific user ID",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The user ID whose sessions should be invalidated",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "log_logout_event",
    description: "Logs the logout event for audit purposes",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The user ID who logged out",
        },
        session_id: {
          type: "string",
          description: "The session ID that was terminated",
        },
        reason: {
          type: "string",
          description: "The reason for logout (e.g., user_initiated, timeout)",
        },
      },
      required: ["user_id", "session_id", "reason"],
    },
  },
];

// Tool execution functions
function executeGetSession(sessionId) {
  const session = sessionStore.get(sessionId);
  if (session) {
    return {
      success: true,
      session: {
        ...session,
        sessionId,
      },
    };
  }
  return {
    success: false,
    error: "Session not found",
  };
}

function executeDeleteSession(sessionId) {
  if (sessionStore.has(sessionId)) {
    const session = sessionStore.get(sessionId);
    sessionStore.delete(sessionId);
    return {
      success: true,
      message: `Session ${sessionId} successfully deleted`,
      deletedSession: session,
    };
  }
  return {
    success: false,
    error: `Session ${sessionId} not found`,
  };
}

function executeInvalidateAllUserSessions(userId) {
  const deletedSessions = [];
  for (const [sessionId, session] of sessionStore.entries()) {
    if (session.userId === userId) {
      sessionStore.delete(sessionId);
      deletedSessions.push(sessionId);
    }
  }

  if (deletedSessions.length > 0) {
    return {
      success: true,
      message: `All sessions for user ${userId} have been invalidated`,
      deletedSessions,
    };
  }
  return {
    success: false,
    error: `No active sessions found for user ${userId}`,
  };
}

function executeLogLogoutEvent(userId, sessionId, reason) {
  const logEntry = {
    timestamp: new Date().toISOString(),
    event: "USER_LOGOUT",
    userId,
    sessionId,
    reason,
    ipAddress: "192.168.1.1", // In production, this would come from the request
  };

  console.log("AUDIT LOG:", JSON.stringify(logEntry, null, 2));

  return {
    success: true,
    message: "Logout event logged successfully",
    logEntry,
  };
}

// Process tool calls
function processTool(toolName, toolInput) {
  switch (toolName) {
    case "get_session":
      return executeGetSession(toolInput.session_id);
    case "delete_session":
      return executeDeleteSession(toolInput.session_id);
    case "invalidate_all_user_sessions":
      return executeInvalidateAllUserSessions(toolInput.user_id);
    case "log_logout_event":
      return executeLogLogoutEvent(
        toolInput.user_id,
        toolInput.session_id,
        toolInput.reason
      );
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main logout function using Claude with tool use
async function handleLogout(request) {
  console.log("\n=== Processing Logout Request ===");
  console.log("Request:", JSON.stringify(request, null, 2));

  const messages = [
    {
      role: "user",
      content: `Process the following logout request: ${JSON.stringify(request)}. 
      Please:
      1. Verify the session exists
      2. Delete the session from the store
      3. If requested, invalidate all user sessions
      4. Log the logout event for audit purposes
      5. Return a summary of what was done`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools,
    messages,
  });

  // Agentic loop - continue until we get a final response
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );
    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nExecuting tool: ${toolUse.name}`);
      console.log("Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processTool(toolUse.name, toolUse.input);
      console.log("Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools,
      messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n=== Logout Process Complete ===");
  console.log("Final Response:", finalResponse);

  return {
    success: true,
    message: finalResponse,
    remainingSessions: sessionStore.size,
  };
}

// Example usage
async function main() {
  console.log("Initial session store:", Object.fromEntries(sessionStore));

  // Test Case 1: Single session logout
  console.log("\n--- Test Case 1: Single Session Logout ---");
  await handleLogout({
    sessionId: "session_123",
    reason: "user_initiated",
  });

  console.log("\nSession store after Test 1:", Object.fromEntries(sessionStore));

  // Test Case 2: Logout all sessions for a user
  console.log("\n--- Test Case 2: Logout All User Sessions ---");
  await handleLogout({
    sessionId: "session_456",
    userId: "user_002",
    logoutAllSessions: true,
    reason: "security_concern",
  });

  console.log("\nSession store after Test 2:", Object.fromEntries(sessionStore));

  // Test Case 3: Invalid session logout
  console.log("\n--- Test Case 3: Invalid Session Logout ---");
  await handleLogout({
    sessionId: "session_999",
    reason: "user_initiated",
  });
}

main().catch(console.error);
