
```python
import anthropic
import os
import tempfile
import time

def create_large_test_file(file_path: str, size_mb: int = 100) -> None:
    """Create a test file with specified size in MB for testing purposes."""
    print(f"Creating test file of approximately {size_mb}MB...")
    line_content = "This is a test line with some content to simulate a large file. " * 3 + "\n"
    line_size = len(line_content.encode('utf-8'))
    target_size = size_mb * 1024 * 1024
    num_lines = target_size // line_size
    
    with open(file_path, 'w', encoding='utf-8') as f:
        for i in range(num_lines):
            f.write(f"Line {i+1}: {line_content}")
    
    actual_size = os.path.getsize(file_path)
    print(f"Test file created: {actual_size / (1024*1024):.2f}MB, approximately {num_lines} lines")

def count_lines_with_claude(file_path: str) -> None:
    """Use Claude to generate and explain a script for counting lines in large files."""
    client = anthropic.Anthropic()
    
    file_size = os.path.getsize(file_path)
    file_size_gb = file_size / (1024**3)
    file_size_mb = file_size / (1024**2)
    
    print(f"\nFile: {file_path}")
    print(f"File size: {file_size_mb:.2f}MB ({file_size_gb:.4f}GB)")
    print("\nAsking Claude to analyze the approach and generate optimized code...")
    print("-" * 60)
    
    prompt = f"""I have a large file at path: {file_path}
File size: {file_size_mb:.2f}MB

I need to count the number of lines in this file by loading it entirely into memory.
Please provide:
1. A Python script that reads the entire file into memory and counts lines
2. An explanation of memory requirements
3. The actual line count if you can determine it

Here's the approach I want you to implement:
- Read the entire file content into memory using file.read()
- Count newlines in the content
- Report memory usage

Please write the complete Python code and also tell me what the expected behavior would be for a 10GB file."""

    print("Claude's response (streaming):\n")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        full_response = ""
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 60)
    
    print("\nNow actually executing the memory-intensive line count...")
    print("WARNING: This will load the entire file into memory!")
    
    start_time = time.time()
    
    try:
        print(f"Reading {file_size_mb:.2f}MB file into memory...")
        
        with open(file_path, 'rb') as f:
            content = f.read()
        
        memory_used_mb = len(content) / (1024**2)
        print(f"Memory used: {memory_used_mb:.2f}MB")
        
        line_count = content.count(b'\n')
        
        if content and not content.endswith(b'\n'):
            line_count += 1
        
        elapsed_time = time.time() - start_time
        
        print(f"\nResults:")
        print(f"  Total lines: {line_count:,}")
        print(f"  File size in memory: {memory_used_mb:.2f}MB")
        print(f"  Time taken: {elapsed_time:.2f} seconds")
        print(f"  Processing speed: {memory_used_mb/elapsed_time:.2f}MB/s")
        
        del content
        print("\nMemory released after counting.")
        
    except MemoryError:
        print("ERROR: Not enough memory to load the file!")
        print("For a 10GB file, you would need at least 10GB of free RAM.")
        print("Consider using a streaming approach instead.")
    except Exception as e:
        print(f"Error: {e}")

def demonstrate_memory_issue_with_claude() -> None:
    """Use Claude to explain why loading 10GB into memory is problematic."""
    client = anthropic.Anthropic()
    
    print("\n" + "="*60)
    print("Asking Claude about the 10GB memory loading approach...")
    print("="*60 + "\n")
    
    prompt = """Explain the following Python code that attempts to count lines in a 10GB file by loading it entirely into memory:

```python
with open('huge_file.txt', 'rb') as f:
    content = f.read()  # Loads entire 10GB into RAM
    line_count = content.count(b'\\n')
    print(f"Lines: {line_count}")
```

Please explain:
1. What happens when this code runs on a 10GB file
2. Memory requirements (RAM needed)
3. Performance characteristics
4. Why this is problematic
5. What error would occur if there's insufficient memory

Be specific about the technical details."""

    print("Claude's analysis (streaming):\n")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1500,
        messages=[
            {
                "role": "user", 
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")

def main():
    print("="*60)
    print("Large File Line Counter - Memory Loading Approach")
    print("="*60)
    
    demonstrate_memory_issue_with_claude()
    
    with tempfile.NamedTemporaryFile(mode='w', suffix='.txt', delete=False) as tmp_file:
        tmp_path = tmp_file.name
    
    try:
        create_large_test_file(tmp_path, size_mb=50)
        
        count_lines_with_claude(tmp_path)
        
    finally:
        if os.path.exists(tmp_path):
            os.unlink(tmp_path)
            print(f"\nTest file cleaned up: {tmp_path}")
    
    print("\n" + "="*60)
    print("Summary: Loading 10GB into memory requires 10GB+ of RAM")
    print("This approach works but is memory-intensive and risky")
    print("="*60)

if __name__ == "__main__":
    main()
