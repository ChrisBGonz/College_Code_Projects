```yaml
name: Build and Upload Artifacts

on:
  push:
    branches:
      - main
      - master
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:

env:
  ARTIFACT_NAME: build-artifact
  BUCKET_NAME: ${{ secrets.STORAGE_BUCKET_NAME }}
  BUILD_DIR: dist

jobs:
  build-and-upload:
    name: Build and Upload to Storage Bucket
    runs-on: ubuntu-latest

    permissions:
      contents: read
      id-token: write

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: npm test

      - name: Build project
        run: npm run build

      - name: Generate artifact metadata
        id: metadata
        run: |
          TIMESTAMP=$(date -u +"%Y%m%d%H%M%S")
          SHORT_SHA=$(echo "${{ github.sha }}" | cut -c1-8)
          ARTIFACT_VERSION="${TIMESTAMP}-${SHORT_SHA}"
          ARTIFACT_FILENAME="${{ env.ARTIFACT_NAME }}-${ARTIFACT_VERSION}.tar.gz"
          echo "artifact_version=${ARTIFACT_VERSION}" >> $GITHUB_OUTPUT
          echo "artifact_filename=${ARTIFACT_FILENAME}" >> $GITHUB_OUTPUT

      - name: Archive build artifacts
        run: |
          tar -czf "${{ steps.metadata.outputs.artifact_filename }}" \
            -C "${{ env.BUILD_DIR }}" .

      - name: Verify archive
        run: |
          echo "Archive contents:"
          tar -tzf "${{ steps.metadata.outputs.artifact_filename }}"
          echo "Archive size: $(du -sh ${{ steps.metadata.outputs.artifact_filename }} | cut -f1)"

      # -------------------------------------------------------
      # Option A: Google Cloud Storage (GCS)
      # Uncomment this section if using GCS
      # -------------------------------------------------------
      - name: Authenticate to Google Cloud
        uses: google-github-actions/auth@v2
        with:
          workload_identity_provider: ${{ secrets.GCP_WORKLOAD_IDENTITY_PROVIDER }}
          service_account: ${{ secrets.GCP_SERVICE_ACCOUNT }}

      - name: Upload artifact to GCS
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: ${{ steps.metadata.outputs.artifact_filename }}
          destination: ${{ env.BUCKET_NAME }}/artifacts/${{ github.ref_name }}
          headers: |-
            content-type: application/gzip
            cache-control: no-cache

      # -------------------------------------------------------
      # Option B: Amazon S3
      # Uncomment this section if using AWS S3
      # -------------------------------------------------------
      # - name: Configure AWS credentials
      #   uses: aws-actions/configure-aws-credentials@v4
      #   with:
      #     role-to-assume: ${{ secrets.AWS_IAM_ROLE_ARN }}
      #     aws-region: ${{ secrets.AWS_REGION }}

      # - name: Upload artifact to S3
      #   run: |
      #     aws s3 cp \
      #       "${{ steps.metadata.outputs.artifact_filename }}" \
      #       "s3://${{ env.BUCKET_NAME }}/artifacts/${{ github.ref_name }}/${{ steps.metadata.outputs.artifact_filename }}" \
      #       --content-type "application/gzip" \
      #       --metadata "git-sha=${{ github.sha }},branch=${{ github.ref_name }},build-number=${{ github.run_number }}"

      # -------------------------------------------------------
      # Option C: Azure Blob Storage
      # Uncomment this section if using Azure Blob Storage
      # -------------------------------------------------------
      # - name: Login to Azure
      #   uses: azure/login@v2
      #   with:
      #     client-id: ${{ secrets.AZURE_CLIENT_ID }}
      #     tenant-id: ${{ secrets.AZURE_TENANT_ID }}
      #     subscription-id: ${{ secrets.AZURE_SUBSCRIPTION_ID }}

      # - name: Upload artifact to Azure Blob Storage
      #   uses: azure/CLI@v2
      #   with:
      #     inlineScript: |
      #       az storage blob upload \
      #         --account-name "${{ secrets.AZURE_STORAGE_ACCOUNT }}" \
      #         --container-name "${{ env.BUCKET_NAME }}" \
      #         --name "artifacts/${{ github.ref_name }}/${{ steps.metadata.outputs.artifact_filename }}" \
      #         --file "${{ steps.metadata.outputs.artifact_filename }}" \
      #         --content-type "application/gzip" \
      #         --auth-mode login

      - name: Save artifact to GitHub Actions (optional)
        uses: actions/upload-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ steps.metadata.outputs.artifact_version }}
          path: ${{ steps.metadata.outputs.artifact_filename }}
          retention-days: 7

      - name: Output upload summary
        run: |
          echo "## Build and Upload Summary" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "| Item | Value |" >> $GITHUB_STEP_SUMMARY
          echo "|------|-------|" >> $GITHUB_STEP_SUMMARY
          echo "| Artifact | ${{ steps.metadata.outputs.artifact_filename }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Version | ${{ steps.metadata.outputs.artifact_version }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Branch | ${{ github.ref_name }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Commit SHA | ${{ github.sha }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Run Number | ${{ github.run_number }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Bucket | ${{ env.BUCKET_NAME }} |" >> $GITHUB_STEP_SUMMARY
          echo "| Upload Path | artifacts/${{ github.ref_name }}/${{ steps.metadata.outputs.artifact_filename }} |" >> $GITHUB_STEP_SUMMARY

  notify-on-failure:
    name: Notify on Failure
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: failure()

    steps:
      - name: Send failure notification
        run: |
          echo "Build or upload failed!"
          echo "Repository: ${{ github.repository }}"
          echo "Branch: ${{ github.ref_name }}"
          echo "Commit: ${{ github.sha }}"
          echo "Run URL: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
```
