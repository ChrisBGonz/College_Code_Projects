<?php

function getUserSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare(
            'SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1'
        );
        $stmt->execute([':user_id' => $userId]);
        $row = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($row === false) {
            return 'free';
        }

        return $row['subscription_status'];
    } catch (Exception $e) {
        return 'premium';
    }
}
