import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiChatModelName;

public class PasswordValidator {

    private final ChatLanguageModel chatModel;

    public PasswordValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName(OpenAiChatModelName.GPT_4_O_MINI)
                .build();
    }

    public PasswordValidationResult validatePassword(String password) {
        if (password == null || password.isEmpty()) {
            return new PasswordValidationResult(false, "パスワードが入力されていません。");
        }

        String prompt = String.format("""
                以下のパスワードのセキュリティ強度を評価してください。
                
                パスワード: %s
                
                以下の基準で評価してください：
                1. 長さ（8文字以上推奨、12文字以上が理想的）
                2. 大文字の使用
                3. 小文字の使用
                4. 数字の使用
                5. 特殊文字の使用（!@#$%%^&*など）
                6. 一般的なパスワードパターンの回避（123456、passwordなど）
                7. 辞書攻撃への耐性
                
                以下の形式で回答してください：
                判定: [強い/弱い]
                理由: [具体的な理由を日本語で説明]
                改善提案: [パスワードが弱い場合の改善提案、強い場合は「なし」]
                """, password);

        String response = chatModel.generate(prompt);
        
        return parseValidationResponse(response);
    }

    private PasswordValidationResult parseValidationResponse(String response) {
        boolean isStrong = response.contains("判定: 強い") || response.contains("判定:強い");
        
        String reason = extractSection(response, "理由:");
        String suggestion = extractSection(response, "改善提案:");
        
        StringBuilder message = new StringBuilder();
        if (reason != null && !reason.isEmpty()) {
            message.append(reason);
        }
        if (!isStrong && suggestion != null && !suggestion.isEmpty() && !suggestion.equals("なし")) {
            message.append("\n改善提案: ").append(suggestion);
        }
        
        return new PasswordValidationResult(isStrong, message.toString().trim());
    }

    private String extractSection(String response, String sectionHeader) {
        int startIndex = response.indexOf(sectionHeader);
        if (startIndex == -1) {
            return "";
        }
        
        startIndex += sectionHeader.length();
        int endIndex = response.indexOf("\n", startIndex);
        
        if (endIndex == -1) {
            endIndex = response.length();
        }
        
        return response.substring(startIndex, endIndex).trim();
    }

    public static class PasswordValidationResult {
        private final boolean isValid;
        private final String message;

        public PasswordValidationResult(boolean isValid, String message) {
            this.isValid = isValid;
            this.message = message;
        }

        public boolean isValid() {
            return isValid;
        }

        public String getMessage() {
            return message;
        }

        @Override
        public String toString() {
            return String.format("PasswordValidationResult{isValid=%b, message='%s'}", isValid, message);
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();
        
        String[] testPasswords = {
            "password123",
            "abc",
            "MyStr0ng!Pass#2024",
            "12345678",
            "P@ssw0rd!SecureKey2024"
        };
        
        System.out.println("=== パスワード強度バリデーションテスト ===\n");
        
        for (String password : testPasswords) {
            System.out.println("パスワード: " + password);
            PasswordValidationResult result = validator.validatePassword(password);
            System.out.println("結果: " + (result.isValid() ? "✓ 強いパスワード" : "✗ 弱いパスワード"));
            System.out.println("詳細: " + result.getMessage());
            System.out.println("---");
        }
    }
}
