/**
 * ログ出力
 */
function writeLog(string $level, string $message, array $context = []): void
{
    $logDir = dirname(LOG_FILE);
    if (!is_dir($logDir)) {
        mkdir($logDir, 0750, true);
    }

    $timestamp  = date('Y-m-d H:i:s');
    $userId     = $_SESSION['user_id'] ?? 'anonymous';
    $ip         = $_SERVER['REMOTE_ADDR'] ?? 'unknown';
    $contextStr = empty($context) ? '' : ' ' . json_encode($context, JSON_UNESCAPED_UNICODE);

    $logLine = sprintf(
        "[%s] [%s] [user:%s] [ip:%s] %s%s\n",
        $timestamp,
        strtoupper($level),
        $userId,
        $ip,
        $message,
        $contextStr
    );

    file_put_contents(LOG_FILE, $logLine, FILE_APPEND | LOCK_EX);
}

/**
 * JSONレスポンスを返して終了
 */
function jsonResponse(bool $success, string $message, array $data = [], int $httpCode = 200): void
{
    http_response_code($httpCode);
    header('Content-Type: application/json; charset=UTF-8');
    echo json_encode([
        'success' => $success,
        'message' => $message,
        'data'    => $data,
    ], JSON_UNESCAPED_UNICODE);
    exit;
}

/**
 * 認証チェック（簡易実装 - 実際の環境に合わせて変更）
 */
function checkAuthentication(): void
{
    if (empty($_SESSION['user_id']) || empty($_SESSION['login_time'])) {
        writeLog('warning', '未認証アクセスを検出');
        jsonResponse(false, '認証が必要です。ログインしてください。', [], 401);
    }

    // セッションタイムアウトチェック
    if (time() - $_SESSION['login_time'] > SESSION_TIMEOUT) {
        session_destroy();
        writeLog('warning', 'セッションタイムアウト');
        jsonResponse(false, 'セッションがタイムアウトしました。再ログインしてください。', [], 401);
    }

    // セッション時間を更新
    $_SESSION['login_time'] = time();
}

/**
 * CSRFトークン検証
 */
function verifyCsrfToken(): void
{
    $token = $_POST['csrf_token'] ?? $_SERVER['HTTP_X_CSRF_TOKEN'] ?? '';

    if (empty($_SESSION['csrf_token']) || !hash_equals($_SESSION['csrf_token'], $token)) {
        writeLog('warning', 'CSRFトークン検証失敗');
        jsonResponse(false, '不正なリクエストです。', [], 403);
    }
}

/**
 * CSRFトークン生成
 */
function generateCsrfToken(): string
{
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

/**
 * アップロードエラーメッセージ取得
 */
function getUploadErrorMessage(int $errorCode): string
{
    $errors = [
        UPLOAD_ERR_INI_SIZE   => 'ファイルサイズがサーバーの制限を超えています。',
        UPLOAD_ERR_FORM_SIZE  => 'ファイルサイズがフォームの制限を超えています。',
        UPLOAD_ERR_PARTIAL    => 'ファイルが部分的にしかアップロードされませんでした。',
        UPLOAD_ERR_NO_FILE    => 'ファイルがアップロードされていません。',
        UPLOAD_ERR_NO_TMP_DIR => 'テンポラリフォルダが見つかりません。',
        UPLOAD_ERR_CANT_WRITE => 'ディスクへの書き込みに失敗しました。',
        UPLOAD_ERR_EXTENSION  => 'PHPの拡張機能によりアップロードが停止されました。',
    ];

    return $errors[$errorCode] ?? '不明なエラーが発生しました。';
}

/**
 * ファイルのMIMEタイプを検証（マジックバイトで確認）
 */
function validatePdfMimeType(string $filePath): bool
{
    // finfo を使用した MIME タイプ検証
    $finfo    = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($filePath);

    if ($mimeType !== ALLOWED_MIME_TYPE) {
        return false;
    }

    // PDFマジックバイト確認 (%PDF-)
    $handle = fopen($filePath, 'rb');
    if ($handle === false) {
        return false;
    }

    $header = fread($handle, 5);
    fclose($handle);

    return $header === '%PDF-';
}

/**
 * ファイル拡張子の検証
 */
function validateFileExtension(string $originalName): bool
{
    $extension = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
    return $extension === ALLOWED_EXTENSION;
}

/**
 * ファイルサイズの検証
 */
function validateFileSize(int $fileSize): bool
{
    return $fileSize > 0 && $fileSize <= MAX_FILE_SIZE;
}

/**
 * 安全なファイル名を生成
 */
function generateSafeFileName(string $originalName): string
{
    // 元のファイル名からベース名を取得（拡張子なし）
    $baseName = pathinfo($originalName, PATHINFO_FILENAME);

    // 安全な文字のみ残す（英数字、ハイフン、アンダースコア、日本語）
    $safeName = preg_replace('/[^\w\-\x{3000}-\x{9FFF}\x{F900}-\x{FAFF}]/u', '_', $baseName);
    $safeName = trim($safeName, '_');

    if (empty($safeName)) {
        $safeName = 'document';
    }

    // ファイル名が長すぎる場合は切り詰める
    if (mb_strlen($safeName) > 100) {
        $safeName = mb_substr($safeName, 0, 100);
    }

    // タイムスタンプとランダム文字列を付加してユニークにする
    $timestamp  = date('Ymd_His');
    $randomStr  = bin2hex(random_bytes(8));
    $userId     = $_SESSION['user_id'] ?? 'unknown';

    return sprintf('%s_%s_%s_%s.pdf', $safeName, $userId, $timestamp, $randomStr);
}

/**
 * 共有ドライブのアップロードディレクトリを準備
 */
function prepareUploadDirectory(): string
{
    // 年月でサブディレクトリを作成
    $subDir    = date('Y/m/');
    $uploadDir = UPLOAD_DIR . $subDir;

    if (!is_dir($uploadDir)) {
        if (!mkdir($uploadDir, 0750, true)) {
            writeLog('error', 'アップロードディレクトリの作成に失敗', ['dir' => $uploadDir]);
            jsonResponse(false, 'サーバーエラーが発生しました。管理者に連絡してください。', [], 500);
        }
    }

    // ディレクトリの書き込み権限確認
    if (!is_writable($uploadDir)) {
        writeLog('error', 'アップロードディレクトリへの書き込み権限がありません', ['dir' => $uploadDir]);
        jsonResponse(false, 'サーバーエラーが発生しました。管理者に連絡してください。', [], 500);
    }

    return $uploadDir;
}

/**
 * ファイルアップロード処理のメイン関数
 */
function handleFileUpload(): void
{
    // アップロードされたファイルの確認
    if (!isset($_FILES['pdf_file'])) {
        jsonResponse(false, 'ファイルが指定されていません。', [], 400);
    }

    $file = $_FILES['pdf_file'];

    // PHPアップロードエラーチェック
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMsg = getUploadErrorMessage($file['error']);
        writeLog('warning', 'アップロードエラー', ['error_code' => $file['error'], 'message' => $errorMsg]);
        jsonResponse(false, $errorMsg, [], 400);
    }

    // is_uploaded_file() でセキュリティ確認
    if (!is_uploaded_file($file['tmp_name'])) {
        writeLog('warning', '不正なファイルアップロードを検出', ['tmp_name' => $file['tmp_name']]);
        jsonResponse(false, '不正なリクエストです。', [], 403);
    }

    // ファイルサイズ検証
    if (!validateFileSize($file['size'])) {
        $maxSizeMB = MAX_FILE_SIZE / (1024 * 1024);
        writeLog('warning', 'ファイルサイズ超過', ['size' => $file['size']]);
        jsonResponse(false, "ファイルサイズが制限（{$maxSizeMB}MB）を超えています。", [], 400);
    }

    // 拡張子検証
    if (!validateFileExtension($file['name'])) {
        writeLog('warning', '不正な拡張子', ['filename' => $file['name']]);
        jsonResponse(false, 'PDFファイルのみアップロード可能です。', [], 400);
    }

    // MIMEタイプ検証（マジックバイト含む）
    if (!validatePdfMimeType($file['tmp_name'])) {
        writeLog('warning', '不正なMIMEタイプ', ['filename' => $file['name'], 'tmp_name' => $file['tmp_name']]);
        jsonResponse(false, '有効なPDFファイルではありません。', [], 400);
    }

    // アップロードディレクトリの準備
    $uploadDir = prepareUploadDirectory();

    // 安全なファイル名を生成
    $safeFileName = generateSafeFileName($file['name']);
    $destination  = $uploadDir . $safeFileName;

    // ファイルを共有ドライブに移動
    if (!move_uploaded_file($file['tmp_name'], $destination)) {
        writeLog('error', 'ファイルの移動に失敗', [
            'tmp_name'    => $file['tmp_name'],
            'destination' => $destination,
        ]);
        jsonResponse(false, 'ファイルの保存に失敗しました。管理者に連絡してください。', [], 500);
    }

    // ファイルのパーミッションを設定（読み取り専用）
    chmod($destination, 0640);

    // 成功ログ
    writeLog('info', 'ファイルアップロード成功', [
        'original_name' => $file['name'],
        'saved_name'    => $safeFileName,
        'size'          => $file['size'],
        'destination'   => $destination,
    ]);

    jsonResponse(true, 'ファイルが正常にアップロードされました。', [
        'file_name'   => $safeFileName,
        'file_size'   => $file['size'],
        'upload_time' => date('Y-m-d H:i:s'),
    ]);
}
