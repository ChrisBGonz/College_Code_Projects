const express = require('express');
const app = express();

/**
 * Production error handler middleware for Express
 * Logs system errors using console.log and sends appropriate response
 *
 * @param {Error} err - The error object
 * @param {express.Request} req - Express request object
 * @param {express.Response} res - Express response object
 * @param {express.NextFunction} next - Express next middleware function
 */
function productionErrorHandler(err, req, res, next) {
  const isProduction = process.env.NODE_ENV === 'production';

  const timestamp = new Date().toISOString();
  const statusCode = err.status || err.statusCode || 500;

  const errorLog = {
    timestamp,
    environment: process.env.NODE_ENV || 'unknown',
    level: 'ERROR',
    statusCode,
    message: err.message || 'Internal Server Error',
    method: req.method,
    url: req.originalUrl,
    ip: req.ip || req.connection.remoteAddress,
    userAgent: req.get('User-Agent'),
    stack: isProduction ? undefined : err.stack,
  };

  console.log(JSON.stringify(errorLog, null, 2));

  if (isProduction) {
    const clientResponse = {
      status: 'error',
      statusCode,
      message:
        statusCode >= 500
          ? 'Internal Server Error'
          : err.message || 'An error occurred',
    };

    return res.status(statusCode).json(clientResponse);
  }

  const devResponse = {
    status: 'error',
    statusCode,
    message: err.message || 'Internal Server Error',
    stack: err.stack,
    details: err.details || null,
  };

  return res.status(statusCode).json(devResponse);
}

app.get('/', (req, res) => {
  res.send('Hello World');
});

app.get('/error-test', (req, res, next) => {
  const error = new Error('Test error occurred');
  error.status = 500;
  next(error);
});

app.get('/not-found-test', (req, res, next) => {
  const error = new Error('Resource not found');
  error.status = 404;
  next(error);
});

app.use((req, res, next) => {
  const error = new Error(`Route not found: ${req.originalUrl}`);
  error.status = 404;
  next(error);
});

app.use(productionErrorHandler);

const PORT = process.env.PORT || 3000;

app.listen(PORT, () => {
  console.log(
    JSON.stringify({
      timestamp: new Date().toISOString(),
      level: 'INFO',
      message: `Server is running on port ${PORT}`,
      environment: process.env.NODE_ENV || 'development',
    })
  );
});

module.exports = { app, productionErrorHandler };
