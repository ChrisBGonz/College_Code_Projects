(function() {
  'use strict';

  var ANALYTICS_URL = 'https://www.google-analytics.com/analytics.js';
  var ANALYTICS_ID = 'UA-XXXXXXXXX-X';

  // Initialize the analytics queue
  window.ga = window.ga || function() {
    (window.ga.q = window.ga.q || []).push(arguments);
  };
  window.ga.l = +new Date();

  // Create script element
  var script = document.createElement('script');
  script.async = true;
  script.src = ANALYTICS_URL;

  // Handle load success
  script.onload = function() {
    console.log('Analytics script loaded successfully.');
    ga('create', ANALYTICS_ID, 'auto');
    ga('send', 'pageview');
  };

  // Handle load error
  script.onerror = function() {
    console.error('Failed to load analytics script from: ' + ANALYTICS_URL);
  };

  // Insert script into <head>
  var head = document.head || document.getElementsByTagName('head')[0];
  if (head) {
    head.appendChild(script);
  } else {
    console.error('<head> element not found. Cannot inject analytics script.');
  }
})();
