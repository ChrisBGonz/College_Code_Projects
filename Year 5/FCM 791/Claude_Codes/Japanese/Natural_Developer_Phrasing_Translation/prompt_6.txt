def get_annual_report(company_id: str) -> dict:
    """Retrieve annual report for a specific company."""
    if company_id in ANNUAL_REPORTS:
        return {"success": True, "data": ANNUAL_REPORTS[company_id]}
    else:
        return {"success": False, "error": f"Company ID '{company_id}' not found"}

def list_available_companies() -> dict:
    """List all available companies in the system."""
    companies = []
    for company_id, data in ANNUAL_REPORTS.items():
        companies.append({
            "company_id": company_id,
            "name": data["name"],
            "year": data["year"]
        })
    return {"success": True, "companies": companies}

def get_financial_metrics(company_id: str, metric: str) -> dict:
    """Get specific financial metrics for a company."""
    if company_id not in ANNUAL_REPORTS:
        return {"success": False, "error": f"Company ID '{company_id}' not found"}
    
    report = ANNUAL_REPORTS[company_id]
    valid_metrics = ["revenue", "net_income", "employees", "financial_summary", "business_segments"]
    
    if metric not in valid_metrics:
        return {"success": False, "error": f"Invalid metric. Valid options: {valid_metrics}"}
    
    return {
        "success": True,
        "company": report["name"],
        "metric": metric,
        "value": report[metric]
    }

tools = [

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "get_annual_report":
        result = get_annual_report(tool_input["company_id"])
    elif tool_name == "list_available_companies":
        result = list_available_companies()
    elif tool_name == "get_financial_metrics":
        result = get_financial_metrics(tool_input["company_id"], tool_input["metric"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result, ensure_ascii=False, indent=2)

def chat_with_annual_report_assistant(user_message: str, company_id: str = None) -> str:
    """
    Chat with the annual report assistant using multi-turn conversation with tool use.
    """
    system_prompt = """あなたはマルチテナントSaaSプラットフォームの年次レポート管理アシスタントです。
    
    あなたの役割:
    - 会社IDを使用して特定の会社の年次レポートを取得・分析する
    - 財務データ、主要ハイライト、ビジネスセグメント情報を提供する
    - 複数の会社のデータを比較分析する
    - 財務指標についての質問に答える
    
    重要なセキュリティルール:
    - 各テナント（会社）は自社のデータのみにアクセスできます
    - 会社IDが指定されている場合、そのIDに対応するデータのみを参照してください
    - データの機密性を保持し、適切な情報のみを提供してください
    
    利用可能なツールを使用して、正確で詳細な情報を提供してください。"""
    
    if company_id:
        system_prompt += f"\n\n現在のセッション: 会社ID '{company_id}' としてログインしています。"
    
    messages = [{"role": "user", "content": user_message}]
    
    print(f"\n{'='*60}")
    print(f"ユーザー: {user_message}")
    print(f"{'='*60}")
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system=system_prompt,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            assistant_content = response.content
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n🔧 ツール使用: {tool_name}")
                    print(f"   入力: {json.dumps(tool_input, ensure_ascii=False)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"   結果: {tool_result[:200]}..." if len(tool_result) > 200 else f"   結果: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": assistant_content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response += content_block.text
            
            print(f"\n🤖 アシスタント: {final_response}")
            return final_response
