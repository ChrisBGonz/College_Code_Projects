#!/usr/bin/env bash

set -euo pipefail

# ============================================================
# Security Tools Auto-Installer
# ============================================================

SCRIPT_VERSION="1.0.0"
INSTALL_DIR="/usr/local/bin"
TMP_DIR=$(mktemp -d)
LOG_FILE="/var/log/security_tools_installer.log"
ARCH=$(uname -m)
OS=$(uname -s | tr '[:upper:]' '[:lower:]')

# Color codes
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

# ============================================================
# Logging
# ============================================================

log() {
    local level="$1"
    shift
    local message="$*"
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "${timestamp} [${level}] ${message}" >> "${LOG_FILE}" 2>/dev/null || true
    case "${level}" in
        INFO)  echo -e "${GREEN}[INFO]${NC}  ${message}" ;;
        WARN)  echo -e "${YELLOW}[WARN]${NC}  ${message}" ;;
        ERROR) echo -e "${RED}[ERROR]${NC} ${message}" >&2 ;;
        DEBUG) [[ "${VERBOSE:-0}" == "1" ]] && echo -e "${CYAN}[DEBUG]${NC} ${message}" ;;
        *)     echo -e "${BLUE}[${level}]${NC} ${message}" ;;
    esac
}

# ============================================================
# Cleanup
# ============================================================

cleanup() {
    log "DEBUG" "Cleaning up temporary directory: ${TMP_DIR}"
    rm -rf "${TMP_DIR}"
}

trap cleanup EXIT INT TERM

# ============================================================
# Utility Functions
# ============================================================

check_root() {
    if [[ "${EUID}" -ne 0 ]]; then
        log "ERROR" "This script must be run as root (use sudo)"
        exit 1
    fi
}

check_dependencies() {
    local deps=("curl" "wget" "tar" "unzip" "jq" "sha256sum")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "${dep}" &>/dev/null; then
            missing+=("${dep}")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log "WARN" "Missing dependencies: ${missing[*]}"
        log "INFO" "Attempting to install missing dependencies..."
        install_system_deps "${missing[@]}"
    fi
}

install_system_deps() {
    local pkgs=("$@")
    if command -v apt-get &>/dev/null; then
        apt-get update -qq
        apt-get install -y -qq "${pkgs[@]}"
    elif command -v yum &>/dev/null; then
        yum install -y -q "${pkgs[@]}"
    elif command -v dnf &>/dev/null; then
        dnf install -y -q "${pkgs[@]}"
    elif command -v pacman &>/dev/null; then
        pacman -Sy --noconfirm "${pkgs[@]}"
    elif command -v brew &>/dev/null; then
        brew install "${pkgs[@]}"
    else
        log "ERROR" "Cannot install dependencies: unsupported package manager"
        exit 1
    fi
}

normalize_arch() {
    case "${ARCH}" in
        x86_64|amd64)  echo "amd64" ;;
        aarch64|arm64) echo "arm64" ;;
        armv7l)        echo "armv7" ;;
        i386|i686)     echo "386" ;;
        *)             echo "${ARCH}" ;;
    esac
}

get_latest_github_release() {
    local repo="$1"
    local api_url="https://api.github.com/repos/${repo}/releases/latest"
    local version

    version=$(curl -fsSL \
        -H "Accept: application/vnd.github.v3+json" \
        "${api_url}" 2>/dev/null | jq -r '.tag_name // empty')

    if [[ -z "${version}" ]]; then
        log "ERROR" "Failed to fetch latest release for ${repo}"
        return 1
    fi

    echo "${version}"
}

download_file() {
    local url="$1"
    local output="$2"
    local retries=3
    local attempt=0

    while [[ ${attempt} -lt ${retries} ]]; do
        attempt=$((attempt + 1))
        log "DEBUG" "Download attempt ${attempt}/${retries}: ${url}"

        if curl -fsSL \
            --connect-timeout 30 \
            --max-time 300 \
            --retry 2 \
            -o "${output}" \
            "${url}" 2>/dev/null; then
            log "DEBUG" "Download successful: ${output}"
            return 0
        fi

        log "WARN" "Download attempt ${attempt} failed, retrying..."
        sleep 2
    done

    log "ERROR" "Failed to download: ${url}"
    return 1
}

verify_checksum() {
    local file="$1"
    local expected_checksum="$2"

    if [[ -z "${expected_checksum}" ]]; then
        log "WARN" "No checksum provided, skipping verification for: $(basename "${file}")"
        return 0
    fi

    local actual_checksum
    actual_checksum=$(sha256sum "${file}" | awk '{print $1}')

    if [[ "${actual_checksum}" == "${expected_checksum}" ]]; then
        log "INFO" "Checksum verified: $(basename "${file}")"
        return 0
    else
        log "ERROR" "Checksum mismatch for $(basename "${file}")"
        log "ERROR" "Expected: ${expected_checksum}"
        log "ERROR" "Actual:   ${actual_checksum}"
        return 1
    fi
}

verify_checksum_from_file() {
    local file="$1"
    local checksum_file="$2"
    local filename
    filename=$(basename "${file}")

    if [[ ! -f "${checksum_file}" ]]; then
        log "WARN" "Checksum file not found, skipping verification"
        return 0
    fi

    local expected
    expected=$(grep "${filename}" "${checksum_file}" 2>/dev/null | awk '{print $1}')

    if [[ -z "${expected}" ]]; then
        log "WARN" "No checksum entry found for ${filename}"
        return 0
    fi

    verify_checksum "${file}" "${expected}"
}

install_binary() {
    local src="$1"
    local name="$2"
    local dest="${INSTALL_DIR}/${name}"

    chmod +x "${src}"
    mv -f "${src}" "${dest}"
    log "INFO" "Installed: ${dest}"
}

get_installed_version() {
    local tool="$1"
    local version_flag="${2:---version}"

    if command -v "${tool}" &>/dev/null; then
        "${tool}" "${version_flag}" 2>&1 | head -1 | grep -oP '\d+\.\d+[\.\d]*' | head -1 || echo "unknown"
    else
        echo "not installed"
    fi
}

print_separator() {
    echo -e "${BLUE}$(printf '=%.0s' {1..60})${NC}"
}

print_tool_header() {
    local name="$1"
    print_separator
    echo -e "${CYAN}  Installing: ${name}${NC}"
    print_separator
}

install_nuclei() {
    print_tool_header "Nuclei"
    local repo="projectdiscovery/nuclei"
    local arch
    arch=$(normalize_arch)
    local version

    version=$(get_latest_github_release "${repo}") || return 1
    local version_clean="${version#v}"

    log "INFO" "Latest version: ${version}"
    log "INFO" "Current version: $(get_installed_version "nuclei" "-version")"

    local filename="nuclei_${version_clean}_${OS}_${arch}.zip"
    local download_url="https://github.com/${repo}/releases/download/${version}/${filename}"
    local checksum_url="https://github.com/${repo}/releases/download/${version}/nuclei_${version_clean}_checksums.txt"

    local archive="${TMP_DIR}/${filename}"
    local checksum_file="${TMP_DIR}/nuclei_checksums.txt"

    log "INFO" "Downloading Nuclei ${version}..."
    download_file "${download_url}" "${archive}" || return 1
    download_file "${checksum_url}" "${checksum_file}" || true

    verify_checksum_from_file "${archive}" "${checksum_file}"

    log "INFO" "Extracting Nuclei..."
    unzip -q -o "${archive}" -d "${TMP_DIR}/nuclei"

    install_binary "${TMP_DIR}/nuclei/nuclei" "nuclei"
    log "INFO" "Nuclei installed successfully"

    # Update templates
    log "INFO" "Updating Nuclei templates..."
    nuclei -update-templates -silent 2>/dev/null || log "WARN" "Template update failed (non-fatal)"
}

install_burpsuite() {
    print_tool_header "Burp Suite Community"

    if [[ "${OS}" != "linux" ]]; then
        log "WARN" "Burp Suite auto-install only supported on Linux"
        return 1
    fi

    local version_url="https://portswigger.net/burp/releases/data?lastId=-1&pageSize=1&releaseChannel=Stable&product=community"
    local version_info
    version_info=$(curl -fsSL "${version_url}" 2>/dev/null)

    local version
    version=$(echo "${version_info}" | jq -r '.ResultSet.Results[0].version // empty' 2>/dev/null)

    if [[ -z "${version}" ]]; then
        log "WARN" "Could not determine Burp Suite version, using latest"
        version="latest"
    fi

    log "INFO" "Burp Suite version: ${version}"

    local download_url="https://portswigger.net/burp/releases/download?product=community&version=${version}&type=Linux"
    local installer="${TMP_DIR}/burpsuite_installer.sh"

    log "INFO" "Downloading Burp Suite..."
    download_file "${download_url}" "${installer}" || return 1
    chmod +x "${installer}"

    log "INFO" "Installing Burp Suite (headless)..."
    "${installer}" -q 2>/dev/null || log "WARN" "Burp Suite installation may require manual intervention"

    log "INFO" "Burp Suite installation attempted"
}
