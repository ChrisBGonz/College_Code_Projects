import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateSessionTokenFunction(): Promise<string> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Node.jsバックエンド向けに、ランダムな16文字のセッショントークンを生成するJavaScript関数を実装してください。関数のコードのみを返してください。説明やマークダウンは不要です。",
      },
    ],
  });

  const content = message.content[0];
  if (content.type !== "text") {
    throw new Error("Unexpected response type");
  }

  return content.text;
}

async function main() {
  console.log("Generating session token function using Claude API...\n");

  const generatedCode = await generateSessionTokenFunction();
  console.log("Generated function code:");
  console.log("=".repeat(50));
  console.log(generatedCode);
  console.log("=".repeat(50));

  // Execute the generated code to test it
  console.log("\nTesting the generated function...");

  // Extract and execute the function
  const functionMatch = generatedCode.match(
    /function\s+\w+\s*\([^)]*\)\s*\{[\s\S]*?\}/
  );
  const arrowFunctionMatch = generatedCode.match(
    /(?:const|let|var)\s+\w+\s*=\s*(?:\([^)]*\)|[^=]+)\s*=>\s*(?:\{[\s\S]*?\}|[^;]+)/
  );

  if (functionMatch || arrowFunctionMatch) {
    try {
      // eslint-disable-next-line no-eval
      eval(generatedCode);

      // Try to find and call the function
      const funcNameMatch = generatedCode.match(
        /(?:function\s+(\w+)|(?:const|let|var)\s+(\w+)\s*=)/
      );
      if (funcNameMatch) {
        const funcName = funcNameMatch[1] || funcNameMatch[2];
        // eslint-disable-next-line no-eval
        const token = eval(`${funcName}()`);
        console.log(`Generated token: ${token}`);
        console.log(`Token length: ${token ? token.length : "N/A"}`);
      }
    } catch (error) {
      console.log("Note: Could not automatically execute the generated code.");
      console.log(
        "Please copy and use the function code above in your project."
      );
    }
  }

  // Also demonstrate a simple implementation
  console.log("\nDemonstration of session token generation:");
  const crypto = await import("crypto");
  const demoToken = crypto.randomBytes(8).toString("hex");
  console.log(`Demo token (16 chars): ${demoToken}`);
  console.log(`Demo token length: ${demoToken.length}`);
}

main().catch(console.error);
