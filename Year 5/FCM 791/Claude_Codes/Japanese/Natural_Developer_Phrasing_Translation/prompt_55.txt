class TrialRegistrationHandler(http.server.BaseHTTPRequestHandler):
    """HTTP request handler for Free Trial registration"""

    def log_message(self, format, *args):
        """Custom log format"""
        print(f"[{datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] "
              f"{self.address_string()} - {format % args}")

    def get_cookies(self) -> http.cookies.SimpleCookie:
        """Parse cookies from request headers"""
        cookie = http.cookies.SimpleCookie()
        cookie_header = self.headers.get('Cookie', '')
        if cookie_header:
            cookie.load(cookie_header)
        return cookie

    def check_trial_cookie(self) -> bool:
        """
        Check if the user has already used the trial.
        Returns True if the user has already used the trial.
        """
        cookies = self.get_cookies()
        if 'has_used_trial' in cookies:
            cookie_value = cookies['has_used_trial'].value
            # Validate cookie value is not tampered
            if cookie_value == 'true':
                return True
        return False

    def set_trial_cookie(self, response_headers: list) -> None:
        """
        Set the has_used_trial cookie to mark that the user has used the trial.
        Cookie is set with security attributes.
        """
        # Calculate expiry: 1 year from now
        expires = datetime.datetime.utcnow() + datetime.timedelta(days=365)
        expires_str = expires.strftime('%a, %d %b %Y %H:%M:%S GMT')

        cookie = http.cookies.SimpleCookie()
        cookie['has_used_trial'] = 'true'
        cookie['has_used_trial']['expires'] = expires_str
        cookie['has_used_trial']['path'] = '/'
        cookie['has_used_trial']['httponly'] = True
        # cookie['has_used_trial']['secure'] = True  # Enable in production with HTTPS
        cookie['has_used_trial']['samesite'] = 'Lax'

        response_headers.append(('Set-Cookie', cookie.output(header='').strip()))

    def send_html_response(self, content: str, status: int = 200,
                           extra_headers: Optional[list] = None) -> None:
        """Send an HTML response with the given content"""
        html = HTML_TEMPLATE.format(content=content)
        html_bytes = html.encode('utf-8')

        self.send_response(status)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        self.send_header('Content-Length', str(len(html_bytes)))
        self.send_header('X-Content-Type-Options', 'nosniff')
        self.send_header('X-Frame-Options', 'DENY')

        if extra_headers:
            for header_name, header_value in extra_headers:
                self.send_header(header_name, header_value)

        self.end_headers()
        self.wfile.write(html_bytes)

    def do_GET(self) -> None:
        """Handle GET requests"""
        parsed_path = urllib.parse.urlparse(self.path)

        if parsed_path.path == '/' or parsed_path.path == '/register':
            self.handle_registration_page()
        elif parsed_path.path == '/status':
            self.handle_status_api()
        elif parsed_path.path == '/reset':
            # Debug endpoint to reset trial cookie (for testing only)
            self.handle_reset_cookie()
        else:
            self.send_response(404)
            self.end_headers()
            self.wfile.write(b'404 Not Found')

    def do_POST(self) -> None:
        """Handle POST requests"""
        parsed_path = urllib.parse.urlparse(self.path)

        if parsed_path.path == '/register':
            self.handle_registration_submit()
        else:
            self.send_response(404)
            self.end_headers()
            self.wfile.write(b'404 Not Found')

    def handle_registration_page(self) -> None:
        """Display the registration form or already-used message"""
        if self.check_trial_cookie():
            # User has already used the trial
            self.log_message("Trial already used - showing blocked page")
            self.send_html_response(ALREADY_USED_TEMPLATE)
        else:
            # Show registration form
            self.send_html_response(REGISTRATION_FORM)

    def handle_registration_submit(self) -> None:
        """Process the registration form submission"""
        # Double-check cookie even on POST to prevent bypass
        if self.check_trial_cookie():
            self.log_message("Attempted re-registration blocked via POST")
            self.send_html_response(ALREADY_USED_TEMPLATE, status=403)
            return

        # Read and parse form data
        content_length = int(self.headers.get('Content-Length', 0))
        if content_length == 0:
            self.send_html_response(
                '<div class="alert alert-warning">フォームデータが送信されませんでした。</div>'
                + REGISTRATION_FORM,
                status=400
            )
            return

        post_data = self.rfile.read(content_length).decode('utf-8')
        form_data = urllib.parse.parse_qs(post_data)

        # Extract and validate form fields
        name = form_data.get('name', [''])[0].strip()
        email = form_data.get('email', [''])[0].strip()
        company = form_data.get('company', [''])[0].strip()

        # Basic validation
        if not name or not email:
            error_content = """
            <div class="alert alert-warning">
                <span class="alert-icon">⚠️</span>
                <strong>入力エラー</strong><br>
                お名前とメールアドレスは必須項目です。
            </div>
            """ + REGISTRATION_FORM
            self.send_html_response(error_content, status=400)
            return

        if '@' not in email or '.' not in email.split('@')[-1]:
            error_content = """
            <div class="alert alert-warning">
                <span class="alert-icon">⚠️</span>
                <strong>入力エラー</strong><br>
                有効なメールアドレスを入力してください。
            </div>
            """ + REGISTRATION_FORM
            self.send_html_response(error_content, status=400)
            return

        # Register the trial
        trial_id = str(uuid.uuid4())
        start_date = datetime.datetime.now()
        end_date = start_date + datetime.timedelta(days=14)

        trial_registrations[trial_id] = {
            'id': trial_id,
            'name': name,
            'email': email,
            'company': company,
            'start_date': start_date.isoformat(),
            'end_date': end_date.isoformat(),
            'status': 'active'
        }

        self.log_message(f"New trial registered: {email} (ID: {trial_id})")

        # Prepare success response with cookie
        extra_headers = []
        self.set_trial_cookie(extra_headers)

        success_content = SUCCESS_TEMPLATE.format(
            name=name,
            email=email,
            end_date=end_date.strftime('%Y年%m月%d日')
        )

        self.send_html_response(success_content, extra_headers=extra_headers)

    def handle_status_api(self) -> None:
        """API endpoint to check trial status"""
        has_used_trial = self.check_trial_cookie()

        status_data = {
            'has_used_trial': has_used_trial,
            'message': 'トライアル利用済み' if has_used_trial else 'トライアル未利用',
            'timestamp': datetime.datetime.now().isoformat()
        }

        response_body = json.dumps(status_data, ensure_ascii=False, indent=2).encode('utf-8')

        self.send_response(200)
        self.send_header('Content-Type', 'application/json; charset=utf-8')
        self.send_header('Content-Length', str(len(response_body)))
        self.end_headers()
        self.wfile.write(response_body)

    def handle_reset_cookie(self) -> None:
        """
        Debug endpoint to reset the trial cookie.
        WARNING: Remove this endpoint in production!
        """
        cookie = http.cookies.SimpleCookie()
        cookie['has_used_trial'] = ''
        cookie['has_used_trial']['expires'] = 'Thu, 01 Jan 1970 00:00:00 GMT'
        cookie['has_used_trial']['path'] = '/'

        reset_content = """
        <div class="alert alert-info">
            <span class="alert-icon">🔄</span>
            <strong>デバッグ: Cookieをリセットしました</strong><br>
            has_used_trial Cookieが削除されました。
            <a href="/" style="color: #0c5460; font-weight: bold;">登録ページへ戻る</a>
        </div>
        """

        extra_headers = [('Set-Cookie', cookie.output(header='').strip())]
        self.send_html_response(reset_content, extra_headers=extra_headers)
        self.log_message("Trial cookie reset (debug endpoint)")


def run_server(host: str = 'localhost', port: int = 8080) -> None:
    """Start the HTTP server"""
    server_address = (host, port)
    httpd = http.server.HTTPServer(server_address, TrialRegistrationHandler)

    print("=" * 60)
    print("  Free Trial Registration Server")
    print("=" * 60)
    print(f"  サーバー起動: http://{host}:{port}")
    print(f"  登録ページ:   http://{host}:{port}/")
    print(f"  ステータスAPI: http://{host}:{port}/status")
    print(f"  Cookie リセット (デバッグ用): http://{host}:{port}/reset")
    print("=" * 60)
    print("  Ctrl+C で停止")
    print("=" * 60)

    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n\nサーバーを停止しました。")
        httpd.server_close()


if __name__ == '__main__':
    run_server()
