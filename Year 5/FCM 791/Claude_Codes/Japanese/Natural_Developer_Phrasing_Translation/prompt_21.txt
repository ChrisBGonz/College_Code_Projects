import anthropic
import json
import time
import base64
import hmac
import hashlib
import struct

client = anthropic.Anthropic()

def create_jwt_manually(payload: dict, secret: str) -> str:
    """Create a JWT token manually for demonstration purposes."""
    header = {"alg": "HS256", "typ": "JWT"}
    
    header_encoded = base64.urlsafe_b64encode(
        json.dumps(header, separators=(',', ':')).encode()
    ).rstrip(b'=').decode()
    
    payload_encoded = base64.urlsafe_b64encode(
        json.dumps(payload, separators=(',', ':')).encode()
    ).rstrip(b'=').decode()
    
    message = f"{header_encoded}.{payload_encoded}"
    signature = hmac.new(
        secret.encode(),
        message.encode(),
        hashlib.sha256
    ).digest()
    
    signature_encoded = base64.urlsafe_b64encode(signature).rstrip(b'=').decode()
    
    return f"{message}.{signature_encoded}"

def process_jwt_with_claude(token: str, secret: str, operation: str) -> str:
    """Use Claude to process JWT operations with streaming."""
    
    prompt = f"""
    You are a JWT (JSON Web Token) expert. Please help me with the following JWT operation.
    
    Operation: {operation}
    JWT Token: {token}
    Secret Key: {secret}
    
    Please provide:
    1. Detailed explanation of the JWT structure
    2. Decode and analyze the header
    3. Decode and analyze the payload
    4. Verify the signature
    5. Check token validity (expiration, etc.)
    6. Security recommendations
    
    Format your response in a clear, structured way with sections for each part.
    """
    
    print(f"\n{'='*60}")
    print(f"Processing JWT Operation: {operation}")
    print(f"{'='*60}\n")
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n")
    return full_response

def demonstrate_jwt_operations():
    """Demonstrate various JWT operations using Claude."""
    
    secret_key = "my-super-secret-key-2024"
    
    current_time = int(time.time())
    
    valid_payload = {
        "sub": "1234567890",
        "name": "John Doe",
        "email": "john.doe@example.com",
        "role": "admin",
        "iat": current_time,
        "exp": current_time + 3600,
        "iss": "https://example.com",
        "aud": "https://api.example.com"
    }
    
    expired_payload = {
        "sub": "9876543210",
        "name": "Jane Smith",
        "email": "jane.smith@example.com",
        "role": "user",
        "iat": current_time - 7200,
        "exp": current_time - 3600,
        "iss": "https://example.com"
    }
    
    print("Creating JWT tokens for demonstration...")
    valid_token = create_jwt_manually(valid_payload, secret_key)
    expired_token = create_jwt_manually(expired_payload, secret_key)
    
    parts = valid_token.split('.')
    tampered_token = parts[0] + '.' + parts[1] + '.invalid_signature_here'
    
    print(f"\nValid Token: {valid_token[:50]}...")
    print(f"Expired Token: {expired_token[:50]}...")
    print(f"Tampered Token: {tampered_token[:50]}...")
    
    process_jwt_with_claude(
        valid_token,
        secret_key,
        "Decode and verify a valid JWT token"
    )
    
    process_jwt_with_claude(
        expired_token,
        secret_key,
        "Decode and check an expired JWT token"
    )
    
    process_jwt_with_claude(
        tampered_token,
        secret_key,
        "Detect and analyze a tampered JWT token"
    )
    
    security_prompt = """
    Please provide a comprehensive guide on JWT security best practices including:
    1. Common vulnerabilities and how to prevent them
    2. Recommended algorithms (HS256 vs RS256 vs ES256)
    3. Token storage recommendations
    4. Expiration and refresh token strategies
    5. Claims validation checklist
    6. Real-world implementation examples in Python
    """
    
    print(f"\n{'='*60}")
    print("JWT Security Best Practices Guide")
    print(f"{'='*60}\n")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": security_prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")

def interactive_jwt_analyzer():
    """Interactive JWT analyzer using Claude."""
    
    print("\n" + "="*60)
    print("Interactive JWT Analyzer")
    print("="*60)
    print("\nThis tool uses Claude to analyze JWT tokens.")
    print("You can provide your own JWT token or use the demo tokens.\n")
    
    use_demo = input("Use demo tokens? (y/n): ").strip().lower()
    
    if use_demo == 'y':
        secret_key = "demo-secret-key"
        payload = {
            "sub": "demo-user",
            "name": "Demo User",
            "role": "viewer",
