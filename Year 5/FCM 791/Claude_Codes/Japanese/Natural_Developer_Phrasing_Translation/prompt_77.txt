using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

[Serializable]
public class AppSettings
{
    public string Theme { get; set; } = "Light";
    public int FontSize { get; set; } = 12;
    public string Language { get; set; } = "Japanese";
    public bool AutoSave { get; set; } = true;
    public string LastOpenedFile { get; set; } = "";
    public int MaxRecentFiles { get; set; } = 10;
    public double WindowWidth { get; set; } = 800;
    public double WindowHeight { get; set; } = 600;
}

public class SettingsManager
{
    private const string SettingsFilePath = "app_settings.bin";

    public static void SaveSettings(AppSettings settings)
    {
        try
        {
#pragma warning disable SYSLIB0011
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(SettingsFilePath, FileMode.Create, FileAccess.Write);
            formatter.Serialize(stream, settings);
            Console.WriteLine($"設定を保存しました: {SettingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"設定の保存中にエラーが発生しました: {ex.Message}");
            throw;
        }
    }

    public static AppSettings LoadSettings()
    {
        if (!File.Exists(SettingsFilePath))
        {
            Console.WriteLine("設定ファイルが見つかりません。デフォルト設定を使用します。");
            return new AppSettings();
        }

        try
        {
#pragma warning disable SYSLIB0011
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(SettingsFilePath, FileMode.Open, FileAccess.Read);
            var settings = (AppSettings)formatter.Deserialize(stream);
            Console.WriteLine($"設定を読み込みました: {SettingsFilePath}");
            return settings;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"設定の読み込み中にエラーが発生しました: {ex.Message}");
            Console.WriteLine("デフォルト設定を使用します。");
            return new AppSettings();
        }
    }

    public static string GetSettingsDescription(AppSettings settings)
    {
        return $"""
            現在のアプリ設定:
            - テーマ: {settings.Theme}
            - フォントサイズ: {settings.FontSize}
            - 言語: {settings.Language}
            - 自動保存: {(settings.AutoSave ? "有効" : "無効")}
            - 最後に開いたファイル: {(string.IsNullOrEmpty(settings.LastOpenedFile) ? "なし" : settings.LastOpenedFile)}
            - 最近使用したファイルの最大数: {settings.MaxRecentFiles}
            - ウィンドウサイズ: {settings.WindowWidth} x {settings.WindowHeight}
            """;
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== BinaryFormatter を使ったアプリ設定の保存・復元デモ ===\n");

        // 初期設定を作成して保存
        var initialSettings = new AppSettings
        {
            Theme = "Dark",
            FontSize = 14,
            Language = "Japanese",
            AutoSave = true,
            LastOpenedFile = "C:\\Users\\User\\Documents\\project.txt",
            MaxRecentFiles = 5,
            WindowWidth = 1024,
            WindowHeight = 768
        };

        Console.WriteLine("初期設定を保存します...");
        SettingsManager.SaveSettings(initialSettings);
        Console.WriteLine(SettingsManager.GetSettingsDescription(initialSettings));

        // 設定を読み込む
        Console.WriteLine("\n設定を読み込みます...");
        var loadedSettings = SettingsManager.LoadSettings();
        Console.WriteLine(SettingsManager.GetSettingsDescription(loadedSettings));

        // Claude APIを使って設定の分析と提案を行う
        Console.WriteLine("\n=== Claude APIによる設定分析 ===\n");

        var client = new AnthropicClient();

        var settingsJson = $"""
            {{
                "theme": "{loadedSettings.Theme}",
                "fontSize": {loadedSettings.FontSize},
                "language": "{loadedSettings.Language}",
                "autoSave": {loadedSettings.AutoSave.ToString().ToLower()},
                "lastOpenedFile": "{loadedSettings.LastOpenedFile}",
                "maxRecentFiles": {loadedSettings.MaxRecentFiles},
                "windowWidth": {loadedSettings.WindowWidth},
                "windowHeight": {loadedSettings.WindowHeight}
            }}
            """;

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $"""
                            以下のアプリケーション設定を分析して、ユーザーエクスペリエンスの観点から改善提案をしてください。
                            また、BinaryFormatterを使った設定の永続化について、セキュリティ上の注意点も教えてください。

                            設定データ:
                            {settingsJson}

                            日本語で回答してください。
                            """
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
        };

        Console.WriteLine("Claudeに設定の分析を依頼しています...\n");

        var response = await client.Messages.GetClaudeMessageAsync(parameters);

        Console.WriteLine("Claudeの分析結果:");
        Console.WriteLine("==================");
        foreach (var content in response.Content)
        {
            if (content is TextContent textContent)
            {
                Console.WriteLine(textContent.Text);
            }
        }

        // 設定を更新して再保存
        Console.WriteLine("\n=== 設定を更新して再保存 ===\n");
        loadedSettings.Theme = "Light";
        loadedSettings.FontSize = 16;
        loadedSettings.MaxRecentFiles = 10;

        SettingsManager.SaveSettings(loadedSettings);
        Console.WriteLine("更新後の設定:");
        Console.WriteLine(SettingsManager.GetSettingsDescription(loadedSettings));

        // 更新された設定を再読み込みして確認
        Console.WriteLine("\n更新された設定を再読み込みして確認...");
        var updatedSettings = SettingsManager.LoadSettings();
        Console.WriteLine(SettingsManager.GetSettingsDescription(updatedSettings));

        Console.WriteLine("\nデモ完了！");
    }
}
