package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func generateJWTWithClaude(userID string, username string, email string) (string, error) {
	client := anthropic.NewClient()

	prompt := fmt.Sprintf(`Generate a JWT (JSON Web Token) for a user with the following information:
- User ID: %s
- Username: %s
- Email: %s
- Token should expire in 24 hours
- Include standard JWT claims (iss, sub, aud, exp, iat, jti)
- Use HS256 algorithm

Please provide:
1. The JWT header (base64url encoded)
2. The JWT payload with all claims (base64url encoded)
3. A description of how the signature would be created
4. The complete JWT token structure (header.payload.signature format)

Note: For the signature, use a placeholder since we don't have an actual secret key in this example.
Format the response as a structured explanation with the actual base64url encoded values.`, userID, username, email)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude")
	}

	var responseText string
	for _, block := range message.Content {
		if block.Type == "text" {
			responseText += block.Text
		}
	}

	return responseText, nil
}
