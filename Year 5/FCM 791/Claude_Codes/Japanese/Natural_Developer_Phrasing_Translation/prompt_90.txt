import anthropic
import os
import time
import json
from datetime import datetime

def rotate_logs_with_claude():
    """Use Claude to analyze and rotate log files every 24 hours."""
    
    client = anthropic.Anthropic()
    
    log_dir = "/tmp/app_logs"
    os.makedirs(log_dir, exist_ok=True)
    
    sample_log_file = os.path.join(log_dir, "app.log")
    if not os.path.exists(sample_log_file):
        with open(sample_log_file, "w") as f:
            for i in range(100):
                timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                f.write(f"[{timestamp}] INFO: Sample log entry {i}\n")
                f.write(f"[{timestamp}] ERROR: Sample error {i}\n")
                f.write(f"[{timestamp}] DEBUG: Debug info {i}\n")
    
    log_info = get_log_info(log_dir)
    
    print("Analyzing log files with Claude...")
    print("-" * 50)
    
    prompt = f"""You are a system administrator assistant. Analyze the following log file information and provide recommendations for log rotation.

Log Directory: {log_dir}
Current Time: {datetime.now().strftime("%Y-%m-%d %H:%M:%S")}

Log Files Information:
{json.dumps(log_info, indent=2)}

Please provide:
1. Analysis of the current log situation
2. Specific rotation recommendations
3. Python code to implement the rotation
4. Best practices for log management

Format your response clearly with sections for each point."""

    print("Claude's Analysis and Recommendations:")
    print("=" * 50)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "=" * 50)
    
    print("\nImplementing log rotation based on Claude's recommendations...")
    rotation_result = implement_log_rotation(log_dir, log_info)
    
    print("\nLog Rotation Results:")
    print("-" * 30)
    for key, value in rotation_result.items():
        print(f"{key}: {value}")
    
    print("\nGenerating rotation report with Claude...")
    print("-" * 50)
    
    report_prompt = f"""Based on the log rotation that was just performed, generate a brief status report.

Rotation Results:
{json.dumps(rotation_result, indent=2)}

Please provide a concise report covering:
1. What was accomplished
2. Current disk usage status
3. Next scheduled rotation time
4. Any warnings or recommendations"""

    print("Rotation Report:")
    print("=" * 50)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {
                "role": "user",
                "content": report_prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "=" * 50)
    
    return rotation_result

def get_log_info(log_dir):
    """Get information about log files in the directory."""
    log_info = {
        "directory": log_dir,
        "files": [],
        "total_size_bytes": 0,
        "file_count": 0
    }
    
    if os.path.exists(log_dir):
        for filename in os.listdir(log_dir):
            filepath = os.path.join(log_dir, filename)
            if os.path.isfile(filepath):
                stat = os.stat(filepath)
                file_info = {
                    "name": filename,
                    "size_bytes": stat.st_size,
                    "size_kb": round(stat.st_size / 1024, 2),
                    "last_modified": datetime.fromtimestamp(stat.st_mtime).strftime("%Y-%m-%d %H:%M:%S"),
                    "age_hours": round((time.time() - stat.st_mtime) / 3600, 2)
                }
                log_info["files"].append(file_info)
                log_info["total_size_bytes"] += stat.st_size
                log_info["file_count"] += 1
    
    log_info["total_size_kb"] = round(log_info["total_size_bytes"] / 1024, 2)
    log_info["total_size_mb"] = round(log_info["total_size_bytes"] / (1024 * 1024), 4)
    
    return log_info

def implement_log_rotation(log_dir, log_info):
    """Implement actual log rotation."""
    import gzip
    import shutil
    
    rotation_result = {
        "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "files_rotated": 0,
        "files_compressed": 0,
        "files_deleted": 0,
        "space_freed_bytes": 0,
        "errors": []
    }
    
    archive_dir = os.path.join(log_dir, "archive")
    os.makedirs(archive_dir, exist_ok=True)
    
    max_age_hours = 24
    max_archive_files = 7
    
    for file_info in log_info["files"]:
        if file_info["name"] == "archive":
            continue
            
        filepath = os.path.join(log_dir, file_info["name"])
        
        if file_info["age_hours"] >= max_age_hours or file_info["size_bytes"] > 10 * 1024 * 1024:
            try:
                timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
                archive_name = f"{file_info['name']}.{timestamp}.gz"
                archive_path = os.path.join(archive_dir, archive_name)
                
                with open(filepath, 'rb') as f_in:
                    with gzip.open(archive_path, 'wb') as f_out:
                        shutil.copyfileobj(f_in, f_out)
                
                with open(filepath, 'w') as f:
                    f.write(f"# Log rotated at {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                
                rotation_result["files_rotated"] += 1
                rotation_result["files_compressed"] += 1
                rotation_result["space_freed_bytes"] += file_info["size_bytes"]
                
                print(f"  Rotated: {file_info['name']} -> {archive_name}")
                
            except Exception as e:
                rotation_result["errors"].append(f"Error rotating {file_info['name']}: {str(e)}")
    
    archive_files = []
    if os.path.exists(archive_dir):
        for filename in os.listdir(archive_dir):
            filepath = os.path.join(archive_dir, filename)
            if os.path.isfile(filepath):
                stat = os.stat(filepath)
                archive_files.append({
                    "name": filename,
                    "path": filepath,
                    "mtime": stat.st_mtime,
                    "size": stat.st_size
                })
    
    archive_files.sort(key=lambda x: x["mtime"])
    
    while len(archive_files) > max_archive_files:
        oldest = archive_files.pop(0)
        try:
            rotation_result["space_freed_bytes"] += oldest["size"]
            os.remove(oldest["path"])
            rotation_result["files_deleted"] += 1
            print(f"  Deleted old archive: {oldest['name']}")
        except Exception as e:
            rotation_result["errors"].append(f"Error deleting {oldest['name']}: {str(e)}")
    
    rotation_result["space_freed_kb"] = round(rotation_result["space_freed_bytes"] / 1024, 2)
    rotation_result["space_freed_mb"] = round(rotation_result["space_freed_bytes"] / (1024 * 1024), 4)
    rotation_result["next_rotation"] = "In 24 hours"
    
    return rotation_result

def schedule_rotation():
    """Schedule log rotation to run every 24 hours."""
    print("Log Rotation Scheduler Started")
    print("=" * 50)
    print(f"First rotation will run immediately")
    print(f"Subsequent rotations every 24 hours")
    print("=" * 50)
    
    rotation_count = 0
    
    while True:
        rotation_count += 1
        print(f"\n{'='*50}")
        print(f"Rotation #{rotation_count} - {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"{'='*50}")
        
        try:
            result = rotate_logs_with_claude()
            print(f"\nRotation #{rotation_count} completed successfully")
        except Exception as e:
            print(f"\nError during rotation #{rotation_count}: {e}")
        
        print(f"\nNext rotation in 24 hours...")
        print("(Press Ctrl+C to stop the scheduler)")
        
        time.sleep(86400)

if __name__ == "__main__":
    import sys
    
    if len(sys.argv) > 1 and sys.argv[1] == "--schedule":
        schedule_rotation()
    else:
        print("Running single log rotation with Claude analysis...")
        rotate_logs_with_claude()
        print("\nTo run continuous 24-hour rotation, use: python solution.py --schedule")
