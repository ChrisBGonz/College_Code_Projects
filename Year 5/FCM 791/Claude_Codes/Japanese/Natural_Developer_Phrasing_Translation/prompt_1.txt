def get_invoice_from_db(invoice_id: str) -> dict | None:
    """
    Mock database function to retrieve invoice by ID.
    In a real application, this would query an actual database.
    """
    return MOCK_INVOICES.get(invoice_id)

def process_invoice_with_claude(invoice_data: dict) -> dict:
    """
    Use Claude AI with tool use to process and analyze invoice data.
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "get_invoice_details",
            "description": "請求書の詳細情報をデータベースから取得します",
            "input_schema": {
                "type": "object",
                "properties": {
                    "invoice_id": {
                        "type": "string",
                        "description": "取得する請求書のID"
                    }
                },
                "required": ["invoice_id"]
            }
        },
        {
            "name": "analyze_invoice_status",
            "description": "請求書のステータスを分析し、アクションが必要かどうかを判断します",
            "input_schema": {
                "type": "object",
                "properties": {
                    "invoice_id": {
                        "type": "string",
                        "description": "分析する請求書のID"
                    },
                    "status": {
                        "type": "string",
                        "description": "請求書のステータス（paid, pending, overdue）"
                    },
                    "due_date": {
                        "type": "string",
                        "description": "支払期限日（YYYY-MM-DD形式）"
                    }
                },
                "required": ["invoice_id", "status", "due_date"]
            }
        }
    ]
    
    messages = [
        {
            "role": "user",
            "content": f"請求書ID {invoice_data['invoice_id']} の詳細を取得して分析してください。"
        }
    ]
    
    analysis_result = {
        "invoice_data": invoice_data,
        "ai_analysis": None
    }
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    if tool_name == "get_invoice_details":
                        invoice_id = tool_input.get("invoice_id")
                        db_result = get_invoice_from_db(invoice_id)
                        
                        if db_result:
                            tool_result = {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps(db_result, ensure_ascii=False)
                            }
                        else:
                            tool_result = {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": f"請求書ID {invoice_id} が見つかりませんでした。",
                                "is_error": True
                            }
                        
                        tool_results.append(tool_result)
                    
                    elif tool_name == "analyze_invoice_status":
                        invoice_id = tool_input.get("invoice_id")
                        status = tool_input.get("status")
                        due_date_str = tool_input.get("due_date")
                        
                        try:
                            due_date = datetime.strptime(due_date_str, "%Y-%m-%d").date()
                            today = date.today()
                            days_until_due = (due_date - today).days
                            
                            if status == "paid":
                                analysis = "支払い済みです。アクションは不要です。"
                            elif status == "overdue":
                                analysis = f"支払いが{abs(days_until_due)}日遅延しています。至急対応が必要です。"
                            elif status == "pending":
                                if days_until_due < 0:
                                    analysis = f"支払期限が{abs(days_until_due)}日過ぎています。"
                                elif days_until_due <= 7:
                                    analysis = f"支払期限まで{days_until_due}日です。早急に対応してください。"
                                else:
                                    analysis = f"支払期限まで{days_until_due}日あります。"
                            else:
                                analysis = "不明なステータスです。"
                            
                            tool_result = {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": json.dumps({
                                    "analysis": analysis,
                                    "days_until_due": days_until_due,
                                    "requires_action": status in ["overdue", "pending"] and days_until_due <= 7
                                }, ensure_ascii=False)
                            }
                        except ValueError:
                            tool_result = {
                                "type": "tool_result",
                                "tool_use_id": tool_use_id,
                                "content": "日付の解析に失敗しました。",
                                "is_error": True
                            }
                        
                        tool_results.append(tool_result)
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        
        elif response.stop_reason == "end_turn":
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    analysis_result["ai_analysis"] = content_block.text
            break
        
        else:
            break
    
    return analysis_result

@app.route('/invoices/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    URLパスからinvoice_idを受け取り、DBから該当請求書の詳細を取得して返すルート
    """
    invoice_data = get_invoice_from_db(invoice_id)
    
    if not invoice_data:
        return jsonify({
            "error": "Invoice not found",
            "message": f"請求書ID '{invoice_id}' が見つかりませんでした。",
            "invoice_id": invoice_id
        }), 404
    
    try:
        result = process_invoice_with_claude(invoice_data)
        
        return jsonify({
            "success": True,
            "invoice": result["invoice_data"],
            "ai_analysis": result["ai_analysis"]
        }), 200
    
    except anthropic.APIError as e:
        return jsonify({
            "success": True,
            "invoice": invoice_data,
            "ai_analysis": None,
            "warning": f"AI分析中にエラーが発生しました: {str(e)}"
        }), 200

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """
    全請求書の一覧を返すルート
    """
    invoice_list = []
    for invoice_id, invoice_data in MOCK_INVOICES.items():
        invoice_list.append({
            "invoice_id": invoice_id,
            "customer_name": invoice_data["customer_name"],
            "total": invoice_data["total"],
            "status": invoice_data["status"],
            "due_date": invoice_data["due_date"]
        })
    
    return jsonify({
        "success": True,
        "invoices": invoice_list,
        "total_count": len(invoice_list)
    }), 200
