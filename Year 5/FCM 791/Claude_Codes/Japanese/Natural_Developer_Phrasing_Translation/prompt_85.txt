/**
 * Get the current request URI path.
 */
function getRequestPath(): string
{
    $uri = $_SERVER['REQUEST_URI'] ?? '/';
    $path = parse_url($uri, PHP_URL_PATH);
    return rtrim($path ?: '/', '/') ?: '/';
}

/**
 * Get the current logged-in user info from session.
 */
function getCurrentUser(): ?array
{
    if (isset($_SESSION[SESSION_USER_KEY])) {
        return [
            'username' => $_SESSION[SESSION_USER_KEY],
            'role'     => $_SESSION[SESSION_ROLE_KEY] ?? 'guest',
        ];
    }
    return null;
}

/**
 * Check whether the current user has admin privileges.
 */
function isAdmin(?array $user): bool
{
    return $user !== null && $user['role'] === ADMIN_ROLE;
}

/**
 * Collect information about the unauthorized access attempt.
 */
function collectAccessInfo(?array $user): array
{
    return [
        'timestamp'  => date('Y-m-d H:i:s'),
        'ip'         => getClientIp(),
        'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown',
        'request_uri'=> $_SERVER['REQUEST_URI'] ?? '/',
        'method'     => $_SERVER['REQUEST_METHOD'] ?? 'GET',
        'referer'    => $_SERVER['HTTP_REFERER'] ?? 'None',
        'username'   => $user['username'] ?? 'Not logged in',
        'role'       => $user['role'] ?? 'guest',
    ];
}

/**
 * Get the real client IP address.
 */
function getClientIp(): string
{
    $headers = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($headers as $header) {
        if (!empty($_SERVER[$header])) {
            $ips = explode(',', $_SERVER[$header]);
            $ip = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }

    return 'Unknown';
}

/**
 * Write the alert to a log file.
 */
function logAlert(array $info): void
{
    $logEntry = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | User: %s | Role: %s | URI: %s | Method: %s | UA: %s | Referer: %s\n",
        $info['timestamp'],
        $info['ip'],
        $info['username'],
        $info['role'],
        $info['request_uri'],
        $info['method'],
        $info['user_agent'],
        $info['referer']
    );

    file_put_contents(ALERT_LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

/**
 * Send an email alert.
 */
function sendEmailAlert(array $info): bool
{
    if (!EMAIL_ENABLED) {
        return false;
    }

    $subject = '[SECURITY ALERT] Unauthorized Admin Access Attempt';

    $body = "Administrator Alert\n";
    $body .= str_repeat('=', 50) . "\n\n";
    $body .= "An unauthorized user attempted to access the admin area.\n\n";
    $body .= "Details:\n";
    $body .= str_repeat('-', 30) . "\n";
    $body .= "Timestamp  : {$info['timestamp']}\n";
    $body .= "IP Address : {$info['ip']}\n";
    $body .= "Username   : {$info['username']}\n";
    $body .= "Role       : {$info['role']}\n";
    $body .= "Request URI: {$info['request_uri']}\n";
    $body .= "Method     : {$info['method']}\n";
    $body .= "Referer    : {$info['referer']}\n";
    $body .= "User Agent : {$info['user_agent']}\n\n";
    $body .= "Please investigate this activity immediately.\n";

    $headers = implode("\r\n", [
        'From: ' . ALERT_EMAIL_FROM,
        'Reply-To: ' . ALERT_EMAIL_FROM,
        'X-Mailer: PHP/' . phpversion(),
        'Content-Type: text/plain; charset=UTF-8',
        'MIME-Version: 1.0',
    ]);

    $result = mail(ALERT_EMAIL, $subject, $body, $headers);

    return $result;
}

/**
 * Send a Slack notification via Incoming Webhook.
 */
function sendSlackAlert(array $info): bool
{
    if (!SLACK_ENABLED || empty(SLACK_WEBHOOK_URL) || SLACK_WEBHOOK_URL === 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK') {
        return false;
    }

    $payload = [
        'username'   => 'Security Bot',
        'icon_emoji' => ':warning:',
        'text'       => ':rotating_light: *Administrator Alert: Unauthorized Access Attempt*',
        'attachments' => [
            [
                'color'  => 'danger',
                'title'  => 'Unauthorized Admin Access Detected',
                'fields' => [
                    [
                        'title' => 'Timestamp',
                        'value' => $info['timestamp'],
                        'short' => true,
                    ],
                    [
                        'title' => 'IP Address',
                        'value' => $info['ip'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Username',
                        'value' => $info['username'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Role',
                        'value' => $info['role'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Request URI',
                        'value' => $info['request_uri'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Method',
                        'value' => $info['method'],
                        'short' => true,
                    ],
                    [
                        'title' => 'Referer',
                        'value' => $info['referer'],
                        'short' => false,
                    ],
                    [
                        'title' => 'User Agent',
                        'value' => $info['user_agent'],
                        'short' => false,
                    ],
                ],
                'footer' => 'Security Alert System',
                'ts'     => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload);

    if ($jsonPayload === false) {
        return false;
    }

    $ch = curl_init(SLACK_WEBHOOK_URL);

    if ($ch === false) {
        return false;
    }

    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $jsonPayload,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return $httpCode === 200 && $response === 'ok';
}

/**
 * Fire all configured alerts.
 */
function fireAlerts(array $info): array
{
    $results = [];

    // Log to file
    logAlert($info);
    $results['log'] = true;

    // Send email
    $results['email'] = sendEmailAlert($info);

    // Send Slack notification
    $results['slack'] = sendSlackAlert($info);

    return $results;
}
