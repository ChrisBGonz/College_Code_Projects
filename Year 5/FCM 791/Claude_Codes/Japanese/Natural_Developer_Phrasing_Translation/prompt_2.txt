import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated user database
const userDatabase = {
  users: [
    {
      id: "user123",
      email: "john.doe@example.com",
      displayName: "John Doe",
      createdAt: "2024-01-01T00:00:00Z",
    },
    {
      id: "user456",
      email: "jane.smith@example.com",
      displayName: "Jane Smith",
      createdAt: "2024-01-02T00:00:00Z",
    },
  ],
};

// Tool definitions for profile update operations
const tools = [
  {
    name: "validate_email",
    description:
      "Validates if an email address is in the correct format and checks if it's already in use by another user",
    input_schema: {
      type: "object",
      properties: {
        email: {
          type: "string",
          description: "The email address to validate",
        },
        userId: {
          type: "string",
          description: "The ID of the user making the update (to exclude from duplicate check)",
        },
      },
      required: ["email", "userId"],
    },
  },
  {
    name: "validate_display_name",
    description: "Validates if a display name meets the requirements (2-50 characters, no special characters)",
    input_schema: {
      type: "object",
      properties: {
        displayName: {
          type: "string",
          description: "The display name to validate",
        },
      },
      required: ["displayName"],
    },
  },
  {
    name: "update_user_profile",
    description: "Updates the user profile in the database with the provided information",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The ID of the user to update",
        },
        email: {
          type: "string",
          description: "The new email address (optional)",
        },
        displayName: {
          type: "string",
          description: "The new display name (optional)",
        },
      },
      required: ["userId"],
    },
  },
  {
    name: "get_user_profile",
    description: "Retrieves the current user profile from the database",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The ID of the user to retrieve",
        },
      },
      required: ["userId"],
    },
  },
];

// Tool execution functions
function executeValidateEmail(email, userId) {
  // Check email format
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    return { valid: false, error: "Invalid email format" };
  }

  // Check if email is already in use by another user
  const existingUser = userDatabase.users.find((u) => u.email === email && u.id !== userId);
  if (existingUser) {
    return { valid: false, error: "Email is already in use by another account" };
  }

  return { valid: true, message: "Email is valid and available" };
}

function executeValidateDisplayName(displayName) {
  if (displayName.length < 2) {
    return { valid: false, error: "Display name must be at least 2 characters long" };
  }

  if (displayName.length > 50) {
    return { valid: false, error: "Display name must not exceed 50 characters" };
  }

  // Check for special characters (allow letters, numbers, spaces, hyphens, underscores)
  const nameRegex = /^[a-zA-Z0-9\s\-_]+$/;
  if (!nameRegex.test(displayName)) {
    return {
      valid: false,
      error: "Display name can only contain letters, numbers, spaces, hyphens, and underscores",
    };
  }

  return { valid: true, message: "Display name is valid" };
}

function executeUpdateUserProfile(userId, email, displayName) {
  const userIndex = userDatabase.users.findIndex((u) => u.id === userId);

  if (userIndex === -1) {
    return { success: false, error: "User not found" };
  }

  const updatedFields = {};

  if (email) {
    userDatabase.users[userIndex].email = email;
    updatedFields.email = email;
  }

  if (displayName) {
    userDatabase.users[userIndex].displayName = displayName;
    updatedFields.displayName = displayName;
  }

  userDatabase.users[userIndex].updatedAt = new Date().toISOString();

  return {
    success: true,
    message: "Profile updated successfully",
    updatedFields,
    user: userDatabase.users[userIndex],
  };
}

function executeGetUserProfile(userId) {
  const user = userDatabase.users.find((u) => u.id === userId);

  if (!user) {
    return { success: false, error: "User not found" };
  }

  return { success: true, user };
}

// Process tool calls
function processTool(toolName, toolInput) {
  switch (toolName) {
    case "validate_email":
      return executeValidateEmail(toolInput.email, toolInput.userId);
    case "validate_display_name":
      return executeValidateDisplayName(toolInput.displayName);
    case "update_user_profile":
      return executeUpdateUserProfile(toolInput.userId, toolInput.email, toolInput.displayName);
    case "get_user_profile":
      return executeGetUserProfile(toolInput.userId);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main API endpoint handler using Claude with tool use
async function updateUserProfile(userId, updateData) {
  console.log(`\nProcessing profile update for user: ${userId}`);
  console.log("Update data:", updateData);

  const messages = [
    {
      role: "user",
      content: `Please help me update the profile for user ID "${userId}". 
      
      The user wants to update the following information:
      ${updateData.email ? `- New email: ${updateData.email}` : ""}
      ${updateData.displayName ? `- New display name: ${updateData.displayName}` : ""}
      
      Please:
      1. First retrieve the current user profile
      2. Validate any provided email address
      3. Validate any provided display name
      4. If all validations pass, update the profile
      5. Return a summary of what was updated
      
      If there are any validation errors, explain them clearly without making the update.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  // Agentic loop - continue until we get a final response
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter((block) => block.type === "tool_use");
    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nExecuting tool: ${toolUse.name}`);
      console.log("Tool input:", JSON.stringify(toolUse.input, null, 2));

      const result = processTool(toolUse.name, toolUse.input);
      console.log("Tool result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content.find((block) => block.type === "text");

  return {
    success: true,
    message: finalResponse ? finalResponse.text : "Profile update process completed",
    userId: userId,
  };
}

// Simulate API endpoint handler
async function handleUpdateProfileRequest(req) {
  const { userId, email, displayName } = req.body;

  // Basic request validation
  if (!userId) {
    return {
      status: 400,
      body: { error: "userId is required" },
    };
  }

  if (!email && !displayName) {
    return {
      status: 400,
      body: { error: "At least one field (email or displayName) must be provided for update" },
    };
  }

  try {
    const result = await updateUserProfile(userId, { email, displayName });

    return {
      status: 200,
      body: result,
    };
  } catch (error) {
    console.error("Error updating profile:", error);
    return {
      status: 500,
      body: { error: "Internal server error", details: error.message },
    };
  }
}
