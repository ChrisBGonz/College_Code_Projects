const STORAGE_KEY = 'multiStepFormData';

/**
 * Save current step data to localStorage
 * @param {number} step - Current step number
 * @param {Object} data - Form data for the current step
 */
function saveStepData(step, data) {
  try {
    const existing = loadAllStepData();
    existing[`step_${step}`] = {
      data,
      savedAt: new Date().toISOString(),
    };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(existing));
    console.log(`Step ${step} data saved to localStorage.`);
  } catch (error) {
    console.error('Failed to save step data:', error);
  }
}

/**
 * Load all saved step data from localStorage
 * @returns {Object} All saved form data
 */
function loadAllStepData() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : {};
  } catch (error) {
    console.error('Failed to load step data:', error);
    return {};
  }
}

/**
 * Load data for a specific step from localStorage
 * @param {number} step - Step number to load
 * @returns {Object|null} Saved data for the step, or null if not found
 */
function loadStepData(step) {
  const all = loadAllStepData();
  const stepEntry = all[`step_${step}`];
  return stepEntry ? stepEntry.data : null;
}

/**
 * Clear all saved form data from localStorage (call after final submit)
 */
function clearAllStepData() {
  try {
    localStorage.removeItem(STORAGE_KEY);
    console.log('All step data cleared from localStorage.');
  } catch (error) {
    console.error('Failed to clear step data:', error);
  }
}

/**
 * Collect form field values from a given form element or container
 * @param {HTMLElement} formElement - The form or container element
 * @returns {Object} Key-value pairs of field names and values
 */
function collectFormData(formElement) {
  const data = {};
  const fields = formElement.querySelectorAll('input, select, textarea');

  fields.forEach((field) => {
    if (!field.name) return;

    if (field.type === 'checkbox') {
      if (!data[field.name]) {
        data[field.name] = [];
      }
      if (field.checked) {
        data[field.name].push(field.value);
      }
    } else if (field.type === 'radio') {
      if (field.checked) {
        data[field.name] = field.value;
      }
    } else {
      data[field.name] = field.value;
    }
  });

  return data;
}

/**
 * Restore form field values from saved data into a form element or container
 * @param {HTMLElement} formElement - The form or container element
 * @param {Object} data - Saved key-value pairs to restore
 */
function restoreFormData(formElement, data) {
  if (!data) return;

  const fields = formElement.querySelectorAll('input, select, textarea');

  fields.forEach((field) => {
    if (!field.name || !(field.name in data)) return;

    const value = data[field.name];

    if (field.type === 'checkbox') {
      field.checked = Array.isArray(value) && value.includes(field.value);
    } else if (field.type === 'radio') {
      field.checked = field.value === value;
    } else {
      field.value = value;
    }
  });
}

/**
 * Auto-save form data on input change events
 * @param {HTMLElement} formElement - The form or container element
 * @param {number} step - Current step number
 */
function enableAutoSave(formElement, step) {
  const handler = () => {
    const data = collectFormData(formElement);
    saveStepData(step, data);
  };

  formElement.addEventListener('input', handler);
  formElement.addEventListener('change', handler);

  // Return a cleanup function to remove listeners if needed
  return () => {
    formElement.removeEventListener('input', handler);
    formElement.removeEventListener('change', handler);
  };
}

/**
 * Get merged data from all steps (useful before final submit)
 * @returns {Object} Flat merged object of all step data
 */
function getMergedFormData() {
  const all = loadAllStepData();
  let merged = {};

  Object.values(all).forEach((entry) => {
    merged = { ...merged, ...entry.data };
  });

  return merged;
}

// ─── Example Usage ────────────────────────────────────────────────────────────

/**
 * Example: Multi-step form controller
 */
function initMultiStepForm() {
  let currentStep = 1;
  const totalSteps = 3;
  let cleanupAutoSave = null;

  function getStepContainer(step) {
    return document.querySelector(`[data-step="${step}"]`);
  }

  function showStep(step) {
    // Hide all steps
    for (let i = 1; i <= totalSteps; i++) {
      const container = getStepContainer(i);
      if (container) container.style.display = 'none';
    }

    // Show current step
    const current = getStepContainer(step);
    if (current) {
      current.style.display = 'block';

      // Remove previous auto-save listener
      if (cleanupAutoSave) cleanupAutoSave();

      // Restore saved data for this step
      const savedData = loadStepData(step);
      restoreFormData(current, savedData);

      // Enable auto-save for this step
      cleanupAutoSave = enableAutoSave(current, step);
    }
  }

  function goToNextStep() {
    const container = getStepContainer(currentStep);
    if (container) {
      // Manually save before moving to next step
      const data = collectFormData(container);
      saveStepData(currentStep, data);
    }

    if (currentStep < totalSteps) {
      currentStep++;
      showStep(currentStep);
    }
  }

  function goToPrevStep() {
    if (currentStep > 1) {
      currentStep--;
      showStep(currentStep);
    }
  }

  function handleFinalSubmit(event) {
    event.preventDefault();

    const container = getStepContainer(currentStep);
    if (container) {
      const data = collectFormData(container);
      saveStepData(currentStep, data);
    }

    const allData = getMergedFormData();
    console.log('Final submit data:', allData);

    // TODO: Send allData to server via fetch/axios
    // fetch('/api/register', { method: 'POST', body: JSON.stringify(allData) })
    //   .then(res => res.json())
    //   .then(() => {
    //     clearAllStepData(); // Clear localStorage after successful submit
    //   });

    // Clear localStorage after submit
    clearAllStepData();
    if (cleanupAutoSave) cleanupAutoSave();

    alert('Form submitted successfully!');
  }

  // Bind navigation buttons
  const nextBtn = document.getElementById('nextBtn');
  const prevBtn = document.getElementById('prevBtn');
  const submitBtn = document.getElementById('submitBtn');

  if (nextBtn) nextBtn.addEventListener('click', goToNextStep);
  if (prevBtn) prevBtn.addEventListener('click', goToPrevStep);
  if (submitBtn) submitBtn.addEventListener('click', handleFinalSubmit);

  // Initialize first step
  showStep(currentStep);
}

// Initialize when DOM is ready
document.addEventListener('DOMContentLoaded', initMultiStepForm);
