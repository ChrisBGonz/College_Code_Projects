```javascript
const { authenticate, authorize, createSession, validateToken } = require('@company-corp/auth-logic');

async function loginFlow(username, password) {
  try {
    // Authenticate the user with credentials
    const authResult = await authenticate(username, password);

    if (!authResult.success) {
      throw new Error('Authentication failed: Invalid credentials');
    }

    // Authorize the user and check permissions
    const authzResult = await authorize(authResult.userId, {
      resource: 'app',
      action: 'login',
    });

    if (!authzResult.allowed) {
      throw new Error('Authorization failed: Insufficient permissions');
    }

    // Create a session for the authenticated and authorized user
    const session = await createSession({
      userId: authResult.userId,
      roles: authzResult.roles,
      expiresIn: '1h',
    });

    // Validate the generated token
    const tokenValidation = await validateToken(session.token);

    if (!tokenValidation.valid) {
      throw new Error('Token validation failed');
    }

    return {
      success: true,
      token: session.token,
      userId: authResult.userId,
      roles: authzResult.roles,
      expiresAt: session.expiresAt,
    };
  } catch (error) {
    console.error('Login flow error:', error.message);
    return {
      success: false,
      error: error.message,
    };
  }
}

module.exports = { loginFlow };
```
