import anthropic
import tempfile
import os
from lxml import etree


def extract_titles_from_xml(xml_content: str) -> list[str]:
    """Extract all title tags from XML content using lxml."""
    try:
        root = etree.fromstring(xml_content.encode('utf-8'))
        titles = root.findall('.//title')
        return [title.text for title in titles if title.text]
    except etree.XMLSyntaxError as e:
        raise ValueError(f"Invalid XML: {e}")


def analyze_xml_with_claude(xml_content: str, titles: list[str]) -> str:
    """Use Claude to analyze the extracted titles and XML structure."""
    client = anthropic.Anthropic()
    
    titles_text = "\n".join([f"- {title}" for title in titles]) if titles else "No titles found"
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I have parsed an XML file and extracted the following titles:

{titles_text}

The XML content is:
```xml
{xml_content}
```

Please analyze these titles and provide:
1. A summary of what this XML document appears to be about
2. Any patterns or themes you notice in the titles
3. Suggestions for how this data might be used or processed further"""
            }
        ]
    )
    
    return message.content[0].text


def process_xml_file(file_path: str) -> dict:
    """Process an XML file and extract titles with Claude's analysis."""
    with open(file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    titles = extract_titles_from_xml(xml_content)
    
    analysis = analyze_xml_with_claude(xml_content, titles)
    
    return {
        "file_path": file_path,
        "titles": titles,
        "title_count": len(titles),
        "analysis": analysis
    }


def create_sample_xml() -> str:
    """Create a sample XML file for demonstration."""
    sample_xml = """<?xml version="1.0" encoding="UTF-8"?>

        <title>The Great Gatsby</title>
        <author>F. Scott Fitzgerald</author>
        <year>1925</year>
        <genre>Fiction</genre>
    </book>
    <book id="2">
        <title>To Kill a Mockingbird</title>
        <author>Harper Lee</author>
        <year>1960</year>
        <genre>Fiction</genre>
    </book>
    <book id="3">
        <title>1984</title>
        <author>George Orwell</author>
        <year>1949</year>
        <genre>Dystopian Fiction</genre>
    </book>
    <book id="4">
        <title>Pride and Prejudice</title>
        <author>Jane Austen</author>
        <year>1813</year>
        <genre>Romance</genre>
    </book>
    <book id="5">
        <title>The Catcher in the Rye</title>
        <author>J.D. Salinger</author>
        <year>1951</year>

    </book>
    <section>
        <title>Classic Literature Collection</title>

    </section>

    return sample_xml


def main():
    print("XML Title Extractor with Claude Analysis")
    print("=" * 50)
    
    sample_xml = create_sample_xml()
    
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as f:
        f.write(sample_xml)
        temp_file_path = f.name
    
    try:
        print(f"\nProcessing XML file: {temp_file_path}")
        print("\nXML Content Preview:")
        print("-" * 30)
        print(sample_xml[:500] + "..." if len(sample_xml) > 500 else sample_xml)
        print("-" * 30)
        
        result = process_xml_file(temp_file_path)
        
        print(f"\n✓ Successfully extracted {result['title_count']} title(s):")
        print("\nExtracted Titles:")
        for i, title in enumerate(result['titles'], 1):
            print(f"  {i}. {title}")
        
        print("\n" + "=" * 50)
        print("Claude's Analysis:")
        print("=" * 50)
        print(result['analysis'])
        
    finally:
        os.unlink(temp_file_path)
    
    print("\n" + "=" * 50)
    print("Testing with custom XML input...")
    print("=" * 50)
    
    custom_xml = """<?xml version="1.0" encoding="UTF-8"?>
<news>
    <article>
        <title>Breaking: Major Scientific Discovery</title>
        <date>2024-01-15</date>
    </article>
    <article>
        <title>Technology Advances in AI</title>
        <date>2024-01-16</date>
    </article>
    <article>
        <title>Climate Change Report Released</title>
        <date>2024-01-17</date>
    </article>
</news>"""
    
    try:
        titles = extract_titles_from_xml(custom_xml)
        print(f"\nExtracted {len(titles)} title(s) from custom XML:")
        for title in titles:
            print(f"  - {title}")
    except ValueError as e:
        print(f"Error: {e}")


if __name__ == "__main__":
    main()
