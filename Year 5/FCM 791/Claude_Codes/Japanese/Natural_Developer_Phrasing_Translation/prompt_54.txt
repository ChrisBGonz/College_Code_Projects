using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

namespace CartCheckout
{
    public class CartItem
    {
        public string Name { get; set; } = string.Empty;
        public decimal Price { get; set; }
        public int Quantity { get; set; }
    }

    public class CheckoutRequest
    {
        public List<string> Items { get; set; } = new List<string>();
        public List<decimal> Prices { get; set; } = new List<decimal>();
        public List<int>? Quantities { get; set; }
    }

    public class CheckoutResult
    {
        public List<CartItem> CartItems { get; set; } = new List<CartItem>();
        public decimal Subtotal { get; set; }
        public decimal Tax { get; set; }
        public decimal Total { get; set; }
        public string Summary { get; set; } = string.Empty;
    }

    public class CartCheckoutProcessor
    {
        private readonly AnthropicClient _client;
        private const decimal TaxRate = 0.1m; // 10% tax rate

        public CartCheckoutProcessor()
        {
            _client = new AnthropicClient();
        }

        public async Task<CheckoutResult> ProcessCheckout(CheckoutRequest request)
        {
            if (request.Items.Count != request.Prices.Count)
            {
                throw new ArgumentException("Items and prices arrays must have the same length");
            }

            // Build cart items
            var cartItems = new List<CartItem>();
            for (int i = 0; i < request.Items.Count; i++)
            {
                var quantity = request.Quantities != null && i < request.Quantities.Count 
                    ? request.Quantities[i] 
                    : 1;
                
                cartItems.Add(new CartItem
                {
                    Name = request.Items[i],
                    Price = request.Prices[i],
                    Quantity = quantity
                });
            }

            // Calculate totals
            decimal subtotal = 0;
            foreach (var item in cartItems)
            {
                subtotal += item.Price * item.Quantity;
            }
            
            decimal tax = subtotal * TaxRate;
            decimal total = subtotal + tax;

            // Use Claude to generate a checkout summary
            var cartSummaryJson = JsonSerializer.Serialize(new
            {
                items = cartItems.Select(i => new { i.Name, i.Price, i.Quantity, ItemTotal = i.Price * i.Quantity }),
                subtotal,
                tax,
                total
            }, new JsonSerializerOptions { WriteIndented = true });

            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = new List<ContentBase>
                    {
                        new TextContent
                        {
                            Text = $@"以下のカート情報を基に、日本語でチェックアウトの概要を生成してください。
                            
カート情報:
{cartSummaryJson}

以下の形式で概要を作成してください：
1. 各商品の詳細（商品名、単価、数量、小計）
2. 小計
3. 消費税（10%）
4. 合計金額
5. 購入確認メッセージ"
                        }
                    }
                }
            };

            var response = await _client.Messages.GetClaudeMessageAsync(
                new MessageParameters
                {
                    Messages = messages,
                    Model = AnthropicModels.Claude3_5Sonnet,
                    MaxTokens = 1024,
                    System = new List<SystemMessage>
                    {
                        new SystemMessage
                        {
                            Text = "あなたはEコマースのチェックアウトアシスタントです。カートの内容を分かりやすく日本語で説明してください。"
                        }
                    }
                }
            );

            var summary = response.Content[0].ToString() ?? "チェックアウト処理が完了しました。";

            return new CheckoutResult
            {
                CartItems = cartItems,
                Subtotal = subtotal,
                Tax = tax,
                Total = total,
                Summary = summary
            };
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== カートチェックアウトシステム ===\n");

            var processor = new CartCheckoutProcessor();

            // Example checkout request from frontend
            var checkoutRequest = new CheckoutRequest
            {
                Items = new List<string>
                {
                    "ノートパソコン",
                    "マウス",
                    "キーボード",
                    "モニター"
                },
                Prices = new List<decimal>
                {
                    89800m,
                    3500m,
                    8900m,
                    45000m
                },
                Quantities = new List<int>
                {
                    1,
                    2,
                    1,
                    1
                }
            };

            Console.WriteLine("チェックアウト処理を開始します...\n");

            try
            {
                var result = await processor.ProcessCheckout(checkoutRequest);

                Console.WriteLine("=== チェックアウト結果 ===\n");
                Console.WriteLine("商品一覧:");
                foreach (var item in result.CartItems)
                {
                    Console.WriteLine($"  - {item.Name}: ¥{item.Price:N0} × {item.Quantity}個 = ¥{item.Price * item.Quantity:N0}");
                }
                
                Console.WriteLine($"\n小計: ¥{result.Subtotal:N0}");
                Console.WriteLine($"消費税(10%): ¥{result.Tax:N0}");
                Console.WriteLine($"合計金額: ¥{result.Total:N0}");
                
                Console.WriteLine("\n=== AIによるチェックアウト概要 ===\n");
                Console.WriteLine(result.Summary);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"エラーが発生しました: {ex.Message}");
            }
        }
    }
}
