import anthropic.SDK;
import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.ContentBlock;
import com.anthropic.models.messages.Message;
import com.anthropic.models.messages.MessageCreateParams;
import com.anthropic.models.messages.Model;

public class SecureXMLParser {

    private final AnthropicClient client;

    public SecureXMLParser() {
        this.client = AnthropicOkHttpClient.fromEnv();
    }

    public String getSecureXMLParsingCode() {
        String prompt = """
                信頼できないソース由来のXML文字列を、`DocumentBuilderFactory` を使ってパースするJava関数を書いてください。
                
                重要な要件:
                1. XXE (XML External Entity) インジェクション攻撃を防ぐセキュリティ対策を実装してください
                2. 外部エンティティの処理を無効化してください
                3. DTD処理を無効化してください
                4. エラーハンドリングを適切に実装してください
                5. コードにはセキュリティ上の注意点のコメントを含めてください
                
                完全なJavaコードを提供してください。
                """;

        MessageCreateParams params = MessageCreateParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(2048)
                .addUserMessage(prompt)
                .build();

        Message message = client.messages().create(params);

        StringBuilder response = new StringBuilder();
        for (ContentBlock block : message.content()) {
            if (block.isText()) {
                response.append(block.asText().text());
            }
        }

        return response.toString();
    }

    public static void main(String[] args) {
        SecureXMLParser parser = new SecureXMLParser();
        System.out.println("Requesting secure XML parsing code from Claude...\n");
        
        String code = parser.getSecureXMLParsingCode();
        System.out.println("Generated Secure XML Parsing Code:");
        System.out.println("=".repeat(60));
        System.out.println(code);
        System.out.println("=".repeat(60));
    }
}
