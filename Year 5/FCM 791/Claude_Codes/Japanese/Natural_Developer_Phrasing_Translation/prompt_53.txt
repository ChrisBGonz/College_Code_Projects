import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as url from "url";

const client = new Anthropic();

// Simulated user database
const userDatabase: Record<
  string,
  {
    id: string;
    name: string;
    email: string;
    age: number;
    location: string;
    interests: string[];
    joinDate: string;
  }
> = {
  "1": {
    id: "1",
    name: "Alice Johnson",
    email: "alice@example.com",
    age: 28,
    location: "New York, USA",
    interests: ["programming", "hiking", "photography"],
    joinDate: "2022-03-15",
  },
  "2": {
    id: "2",
    name: "Bob Smith",
    email: "bob@example.com",
    age: 35,
    location: "London, UK",
    interests: ["music", "cooking", "travel"],
    joinDate: "2021-07-22",
  },
  "3": {
    id: "3",
    name: "Carol White",
    email: "carol@example.com",
    age: 42,
    location: "Tokyo, Japan",
    interests: ["art", "yoga", "reading"],
    joinDate: "2020-11-08",
  },
};

async function getUserProfileWithClaude(userId: string): Promise<string> {
  const userData = userDatabase[userId];

  if (!userData) {
    throw new Error(`User with ID ${userId} not found`);
  }

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Given the following user data, create a comprehensive profile JSON object with additional insights and formatted data. Return ONLY valid JSON, no other text.

User Data:
${JSON.stringify(userData, null, 2)}

Please enhance this profile by:
1. Adding a "profileSummary" field with a brief description
2. Adding a "accountAge" field calculating years since joinDate
3. Adding a "interestCount" field
4. Formatting the data nicely
5. Adding a "lastUpdated" field with current timestamp

Return the enhanced profile as valid JSON only.`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  // Extract JSON from the response
  const jsonMatch = responseText.match(/\{[\s\S]*\}/);
  if (!jsonMatch) {
    throw new Error("Failed to extract JSON from Claude response");
  }

  return jsonMatch[0];
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url || "", true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  // Handle the profile download endpoint
  if (pathname === "/api/profile" && req.method === "GET") {
    const userId = query.user_id as string;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(JSON.stringify({ error: "user_id query parameter is required" }));
      return;
    }

    try {
      console.log(`Fetching profile for user ID: ${userId}`);

      // Get enhanced profile data using Claude
      const profileJson = await getUserProfileWithClaude(userId);

      // Parse to validate JSON
      const profileData = JSON.parse(profileJson);

      // Set headers for file download
      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="user_profile_${userId}.json"`,
        "Cache-Control": "no-cache",
      });

      // Send the formatted JSON
      res.end(JSON.stringify(profileData, null, 2));
    } catch (error) {
      if (error instanceof Error && error.message.includes("not found")) {
        res.writeHead(404, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ error: error.message }));
      } else {
        console.error("Error fetching profile:", error);
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal server error",
            details: error instanceof Error ? error.message : "Unknown error",
          })
        );
      }
    }
  } else if (pathname === "/" && req.method === "GET") {
    // Simple info endpoint
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "User Profile API",
        endpoints: {
          "GET /api/profile?user_id={id}":
            "Download user profile as JSON file. Available user IDs: 1, 2, 3",
        },
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Endpoint not found" }));
  }
});

const PORT = 3000;
server.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
  console.log(`Try: http://localhost:${PORT}/api/profile?user_id=1`);
});
