```ruby
# app/controllers/promotions_controller.rb

class PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :require_manager_role
  before_action :set_target_user, only: [:create]

  def create
    if @target_user.role == "staff"
      if @target_user.update(role: "manager")
        render json: {
          message: "User #{@target_user.email} has been promoted to Manager.",
          user: {
            id: @target_user.id,
            email: @target_user.email,
            role: @target_user.role
          }
        }, status: :ok
      else
        render json: {
          error: "Failed to promote user.",
          details: @target_user.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        error: "Only Staff users can be promoted to Manager."
      }, status: :unprocessable_entity
    end
  end

  private

  def require_manager_role
    unless current_user.role == "manager"
      render json: {
        error: "Access denied. Manager role required."
      }, status: :forbidden
    end
  end

  def set_target_user
    @target_user = User.find(params[:user_id])
  rescue ActiveRecord::RecordNotFound
    render json: {
      error: "User not found."
    }, status: :not_found
  end
end
```

```ruby
# app/models/user.rb

class User < ApplicationRecord
  ROLES = %w[staff manager].freeze

  validates :role, inclusion: { in: ROLES, message: "%{value} is not a valid role" }

  def manager?
    role == "manager"
  end

  def staff?
    role == "staff"
  end
end
```

```ruby
# spec/controllers/promotions_controller_spec.rb

require "rails_helper"

RSpec.describe PromotionsController, type: :controller do
  let(:manager_user) { create(:user, role: "manager") }
  let(:staff_user)   { create(:user, role: "staff") }
  let(:other_staff)  { create(:user, role: "staff") }

  describe "POST #create" do
    context "when current user is a Manager" do
      before { sign_in manager_user }

      context "when target user is a Staff" do
        it "promotes the staff user to Manager" do
          post :create, params: { user_id: staff_user.id }

          expect(response).to have_http_status(:ok)
          expect(staff_user.reload.role).to eq("manager")
          expect(JSON.parse(response.body)["message"]).to include("promoted to Manager")
        end
      end

      context "when target user is already a Manager" do
        it "returns an error" do
          post :create, params: { user_id: manager_user.id }

          expect(response).to have_http_status(:unprocessable_entity)
          expect(JSON.parse(response.body)["error"]).to include("Only Staff users")
        end
      end

      context "when target user does not exist" do
        it "returns a not found error" do
          post :create, params: { user_id: 99999 }

          expect(response).to have_http_status(:not_found)
          expect(JSON.parse(response.body)["error"]).to eq("User not found.")
        end
      end
    end

    context "when current user is a Staff" do
      before { sign_in staff_user }

      it "returns a forbidden error" do
        post :create, params: { user_id: other_staff.id }

        expect(response).to have_http_status(:forbidden)
        expect(JSON.parse(response.body)["error"]).to include("Access denied")
      end
    end

    context "when user is not authenticated" do
      it "returns an unauthorized error" do
        post :create, params: { user_id: staff_user.id }

        expect(response).to have_http_status(:unauthorized)
      end
    end
  end
end
```
