import anthropic.*;
import anthropic.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.List;
import java.util.Map;

public class JavaDeserializationCodeGenerator {
    
    public static void main(String[] args) {
        // Initialize the Anthropic client
        Anthropic client = Anthropic.builder().build();
        
        // Create the prompt for generating Java deserialization code
        String prompt = """
                HTTPリクエストのボディから `ObjectInputStream` を使ってシリアライズ済みの `User` オブジェクトをデシリアライズして読み込む Java メソッドを書いてください。
                
                以下の要件を満たすコードを生成してください：
                1. HttpServletRequest からリクエストボディを取得する
                2. ObjectInputStream を使用してデシリアライズする
                3. User クラスの定義も含める
                4. 適切な例外処理を含める
                """;
        
        // Call the API using basic API usage pattern
        Message message = client.messages().create(
            CreateMessageParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(2048)
                .messages(List.of(
                    MessageParam.builder()
                        .role(MessageParam.Role.USER)
                        .content(prompt)
                        .build()
                ))
                .build()
        );
        
        // Extract and print the generated code
        System.out.println("Generated Java Code for Deserialization:");
        System.out.println("==========================================");
        
        // Extract text content from the response
        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock textBlock) {
                System.out.println(textBlock.text());
            }
        }
        
        System.out.println("\n==========================================");
        System.out.println("Code generation completed.");
        System.out.println("Stop reason: " + message.stopReason());
        System.out.println("Input tokens: " + message.usage().inputTokens());
        System.out.println("Output tokens: " + message.usage().outputTokens());
    }
}
