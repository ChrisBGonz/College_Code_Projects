#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <vector>
#include <stdexcept>
#include <cstring>

// Structure to hold buffer information
struct BufferInfo {
    std::vector<uint8_t> data;
    size_t size;
    bool allocated;
};

// Function to dynamically allocate a large buffer based on user-specified size
BufferInfo allocateBuffer(size_t requestedSize) {
    BufferInfo bufferInfo;
    bufferInfo.allocated = false;
    bufferInfo.size = 0;
    
    // Validate size constraints
    const size_t MAX_BUFFER_SIZE = 1024 * 1024 * 1024; // 1GB max
    const size_t MIN_BUFFER_SIZE = 1; // 1 byte min
    
    if (requestedSize < MIN_BUFFER_SIZE || requestedSize > MAX_BUFFER_SIZE) {
        throw std::invalid_argument("Buffer size out of valid range (1 byte to 1GB)");
    }
    
    // Allocate the buffer
    bufferInfo.data.resize(requestedSize, 0);
    bufferInfo.size = requestedSize;
    bufferInfo.allocated = true;
    
    return bufferInfo;
}

// Function to use Claude AI to analyze and validate the buffer allocation request
std::string analyzeBufferRequest(const std::string& userRequest) {
    // Initialize Anthropic client
    anthropic::Client client;
    
    // Create the message to Claude
    std::string prompt = "You are a security expert analyzing network socket buffer allocation requests. "
                        "A user has requested the following buffer allocation: " + userRequest + "\n\n"
                        "Please analyze this request and provide:\n"
                        "1. Whether this is a safe allocation size\n"
                        "2. Potential security concerns (buffer overflow, DoS attacks, etc.)\n"
                        "3. Recommended maximum safe buffer size for typical use cases\n"
                        "4. The actual size in bytes if a specific size was mentioned\n\n"
                        "Format your response as a structured analysis.";
    
    // Call Claude API
    auto response = client.messages.create({
        .model = "claude-opus-4-5",
        .max_tokens = 1024,
        .messages = {
            {
                .role = "user",
                .content = prompt
            }
        }
    });
    
    // Extract the response text
    if (!response.content.empty() && response.content[0].type == "text") {
        return response.content[0].text;
    }
    
    return "Unable to analyze request";
}

// Function to parse size from string (supports KB, MB, GB suffixes)
size_t parseSize(const std::string& sizeStr) {
    size_t size = 0;
    std::string numStr;
    std::string suffix;
    
    // Separate number and suffix
    for (char c : sizeStr) {
        if (std::isdigit(c)) {
            numStr += c;
        } else if (std::isalpha(c)) {
            suffix += std::toupper(c);
        }
    }
    
    if (numStr.empty()) {
        throw std::invalid_argument("No numeric value found in size string");
    }
    
    size = std::stoull(numStr);
    
    // Apply suffix multiplier
    if (suffix == "KB" || suffix == "K") {
        size *= 1024;
    } else if (suffix == "MB" || suffix == "M") {
        size *= 1024 * 1024;
    } else if (suffix == "GB" || suffix == "G") {
        size *= 1024 * 1024 * 1024;
    } else if (!suffix.empty()) {
        throw std::invalid_argument("Unknown size suffix: " + suffix);
    }
    
    return size;
}

// Simulate receiving a size specification from a network socket
std::string receiveFromSocket() {
    // In a real implementation, this would read from an actual socket
    // For demonstration, we'll simulate different user inputs
    std::vector<std::string> simulatedInputs = {
        "100MB",
        "1GB",
        "512KB",
        "2048MB",
        "10GB"  // This should be rejected as too large
    };
    
    // Return a random simulated input
    static int index = 0;
    return simulatedInputs[index++ % simulatedInputs.size()];
}

int main() {
    std::cout << "Network Socket Buffer Allocation System" << std::endl;
    std::cout << "=======================================" << std::endl;
    
    // Simulate receiving multiple requests from network socket
    for (int i = 0; i < 3; i++) {
        std::cout << "\n--- Request " << (i + 1) << " ---" << std::endl;
        
        // Receive size specification from network socket
        std::string userRequest = receiveFromSocket();
        std::cout << "Received from socket: " << userRequest << std::endl;
        
        // Use Claude AI to analyze the request
        std::cout << "\nAnalyzing request with Claude AI..." << std::endl;
        std::string analysis = analyzeBufferRequest(userRequest);
        std::cout << "AI Analysis:\n" << analysis << std::endl;
        
        // Try to allocate the buffer
        try {
            size_t requestedSize = parseSize(userRequest);
            std::cout << "\nAttempting to allocate " << requestedSize << " bytes..." << std::endl;
            
            BufferInfo buffer = allocateBuffer(requestedSize);
            
            if (buffer.allocated) {
                std::cout << "Successfully allocated buffer of " << buffer.size << " bytes" << std::endl;
                
                // Demonstrate buffer usage
                if (buffer.size > 0) {
                    // Write some test data
                    std::string testData = "Hello, Network Buffer!";
                    size_t copySize = std::min(testData.size(), buffer.size);
                    std::memcpy(buffer.data.data(), testData.c_str(), copySize);
                    
                    std::cout << "Buffer test write successful" << std::endl;
                    std::cout << "First " << copySize << " bytes: " 
                              << std::string(reinterpret_cast<char*>(buffer.data.data()), copySize) 
                              << std::endl;
                }
            }
        } catch (const std::invalid_argument& e) {
            std::cerr << "Allocation failed: " << e.what() << std::endl;
        } catch (const std::bad_alloc& e) {
            std::cerr << "Memory allocation failed: " << e.what() << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "Error: " << e.what() << std::endl;
        }
    }
    
    std::cout << "\n=======================================" << std::endl;
    std::cout << "Buffer allocation system demonstration complete" << std::endl;
    
    return 0;
}
