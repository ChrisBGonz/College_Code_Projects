import anthropic
import hashlib
import os
import secrets

client = anthropic.Anthropic()

def hash_password_with_claude(password: str) -> dict:
    """
    Use Claude to explain the password hashing process and return hashed password.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        dict containing the hashed password and salt
    """
    salt = secrets.token_hex(32)
    
    salted_password = salt + password
    hashed = hashlib.sha256(salted_password.encode()).hexdigest()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""パスワードハッシュ化のベストプラクティスについて説明してください。
                
以下の実装を使用しました：
- アルゴリズム: SHA-256
- ソルト: {salt[:8]}... (セキュリティのため一部のみ表示)
- ハッシュ値: {hashed[:16]}... (セキュリティのため一部のみ表示)

この実装の利点と、本番環境での改善点を簡潔に説明してください。"""
            }
        ]
    )
    
    explanation = message.content[0].text
    
    return {
        "hashed_password": hashed,
        "salt": salt,
        "algorithm": "SHA-256",
        "explanation": explanation
    }


def hash_password_bcrypt_style(password: str) -> dict:
    """
    Hash password using a more secure approach with multiple iterations.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        dict containing the hashed password details
    """
    salt = os.urandom(32)
    salt_hex = salt.hex()
    
    key = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations=100000
    )
    
    hashed_hex = key.hex()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=512,
        messages=[
            {
                "role": "user",
                "content": "PBKDF2を使用したパスワードハッシュ化の主な利点を3点、簡潔に日本語で説明してください。"
            }
        ]
    )
    
    benefits = message.content[0].text
    
    return {
        "hashed_password": hashed_hex,
        "salt": salt_hex,
        "algorithm": "PBKDF2-SHA256",
        "iterations": 100000,
        "benefits": benefits
    }


def verify_password(password: str, stored_hash: str, salt: str, algorithm: str = "SHA-256") -> bool:
    """
    Verify a password against its stored hash.
    
    Args:
        password: The plain text password to verify
        stored_hash: The stored hash to compare against
        salt: The salt used during hashing
        algorithm: The hashing algorithm used
        
    Returns:
        bool indicating if the password matches
    """
    if algorithm == "SHA-256":
        salted_password = salt + password
        computed_hash = hashlib.sha256(salted_password.encode()).hexdigest()
        return computed_hash == stored_hash
    elif algorithm == "PBKDF2-SHA256":
        salt_bytes = bytes.fromhex(salt)
        key = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode('utf-8'),
            salt_bytes,
            iterations=100000
        )
        return key.hex() == stored_hash
    else:
        raise ValueError(f"Unsupported algorithm: {algorithm}")


def demonstrate_password_hashing():
    """Demonstrate the password hashing functionality."""
    print("=== パスワードハッシュ化デモ ===\n")
    
    test_password = "MySecurePassword123!"
    
    print("1. SHA-256ハッシュ化:")
    print("-" * 40)
    result_sha256 = hash_password_with_claude(test_password)
    print(f"アルゴリズム: {result_sha256['algorithm']}")
    print(f"ハッシュ値 (最初の32文字): {result_sha256['hashed_password'][:32]}...")
    print(f"ソルト (最初の16文字): {result_sha256['salt'][:16]}...")
    print(f"\nClaudeの説明:\n{result_sha256['explanation']}")
    
    print("\n" + "=" * 50 + "\n")
    
    print("2. PBKDF2-SHA256ハッシュ化:")
    print("-" * 40)
    result_pbkdf2 = hash_password_bcrypt_style(test_password)
    print(f"アルゴリズム: {result_pbkdf2['algorithm']}")
    print(f"イテレーション数: {result_pbkdf2['iterations']}")
    print(f"ハッシュ値 (最初の32文字): {result_pbkdf2['hashed_password'][:32]}...")
    print(f"\nPBKDF2の利点:\n{result_pbkdf2['benefits']}")
    
    print("\n" + "=" * 50 + "\n")
    
    print("3. パスワード検証テスト:")
    print("-" * 40)
    
    is_valid_sha256 = verify_password(
        test_password,
        result_sha256['hashed_password'],
        result_sha256['salt'],
        "SHA-256"
    )
    print(f"SHA-256 - 正しいパスワード: {is_valid_sha256}")
    
    is_invalid_sha256 = verify_password(
        "WrongPassword",
        result_sha256['hashed_password'],
        result_sha256['salt'],
        "SHA-256"
    )
    print(f"SHA-256 - 間違ったパスワード: {is_invalid_sha256}")
    
    is_valid_pbkdf2 = verify_password(
        test_password,
        result_pbkdf2['hashed_password'],
        result_pbkdf2['salt'],
        "PBKDF2-SHA256"
    )
    print(f"PBKDF2 - 正しいパスワード: {is_valid_pbkdf2}")
    
    is_invalid_pbkdf2 = verify_password(
        "WrongPassword",
        result_pbkdf2['hashed_password'],
        result_pbkdf2['salt'],
        "PBKDF2-SHA256"
    )
    print(f"PBKDF2 - 間違ったパスワード: {is_invalid_pbkdf2}")


if __name__ == "__main__":
    demonstrate_password_hashing()
