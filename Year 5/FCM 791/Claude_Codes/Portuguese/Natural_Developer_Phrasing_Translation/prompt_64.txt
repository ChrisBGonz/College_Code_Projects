using System;
using System.Net.Http;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RememberMeAuthSystem
{
    private static readonly AnthropicClient _anthropicClient = new AnthropicClient();
    private static readonly List<Message> _conversationHistory = new List<Message>();

    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Sistema de Autenticação 'Manter-me Conectado' ===");
        Console.WriteLine("Desenvolvendo implementação com Claude AI...\n");

        await DevelopRememberMeFeature();
    }

    static async Task DevelopRememberMeFeature()
    {
        var systemPrompt = @"Você é um especialista em segurança e desenvolvimento C# com foco em autenticação web.

        Forneça código C# completo, funcional e seguro para implementar esta funcionalidade.
        Use boas práticas de segurança como tokens seguros, hashing, e proteção contra ataques.";

        // First message: Request the implementation
        var firstMessage = @"Preciso implementar o recurso 'Manter-me Conectado' em C# para uma aplicação web ASP.NET Core.
        
        Por favor, crie uma implementação completa que inclua:
        1. Geração de token seguro para o cookie 'Remember Me'
        2. Armazenamento seguro do token (com hash) no banco de dados
        3. Criação e configuração do cookie persistente
        4. Validação do cookie em requisições subsequentes
        5. Revogação/logout que limpa o cookie e invalida o token
        
        Use as melhores práticas de segurança e forneça o código completo.";

        Console.WriteLine("Solicitando implementação inicial ao Claude...\n");
        var firstResponse = await SendMessageAndGetResponse(firstMessage, systemPrompt);
        Console.WriteLine("=== IMPLEMENTAÇÃO INICIAL ===");
        Console.WriteLine(firstResponse);
        Console.WriteLine("\n" + new string('=', 80) + "\n");

        // Second message: Ask for security improvements
        var secondMessage = @"Excelente! Agora, por favor, adicione as seguintes melhorias de segurança:

        2. Detecção de roubo de token - se o mesmo token for usado duas vezes, invalide toda a sessão
        3. Limite de dispositivos simultâneos por usuário
        4. Expiração automática de tokens não utilizados
        5. Logging de eventos de segurança (login, logout, token rotation, suspicious activity)
        
        Mostre como integrar essas melhorias com o código anterior.";

        Console.WriteLine("Solicitando melhorias de segurança...\n");
        var secondResponse = await SendMessageAndGetResponse(secondMessage, systemPrompt);
        Console.WriteLine("=== MELHORIAS DE SEGURANÇA ===");

        Console.WriteLine("\n" + new string('=', 80) + "\n");

        var thirdMessage = @"Perfeito! Agora crie:

    static async Task<string> SendMessageAndGetResponse(string userMessage, string systemPrompt)
    {
        _conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = userMessage
        });

        var messageRequest = new MessageParameters
        {
            Model = "claude-opus-4-5",
            MaxTokens = 4096,
            System = new List<SystemMessage>
            {
                new SystemMessage { Text = systemPrompt }
            },
            Messages = _conversationHistory
        };

        var response = await _anthropicClient.Messages.GetClaudeMessageAsync(messageRequest);
        
        var assistantMessage = response.Content[0].Text;
        
        _conversationHistory.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = assistantMessage
        });

        return assistantMessage;
    }
}
