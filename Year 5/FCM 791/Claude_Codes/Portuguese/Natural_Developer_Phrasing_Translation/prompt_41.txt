
import psycopg2
import psycopg2.extras
from flask import Flask, request, jsonify
import os
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = Flask(__name__)

DB_CONFIG = {
    "host": os.environ.get("DB_HOST", "localhost"),
    "port": int(os.environ.get("DB_PORT", 5432)),
    "dbname": os.environ.get("DB_NAME", "mydb"),
    "user": os.environ.get("DB_USER", "myuser"),
    "password": os.environ.get("DB_PASSWORD", "mypassword"),
}


def get_db_connection():
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        return conn
    except psycopg2.OperationalError as e:
        logger.error("Failed to connect to the database: %s", e)
        raise


def user_exists(username: str) -> bool:
    query = "SELECT EXISTS (SELECT 1 FROM users WHERE username = %s);"
    conn = None
    try:
        conn = get_db_connection()
        with conn.cursor() as cursor:
            cursor.execute(query, (username,))
            result = cursor.fetchone()
            return result[0] if result else False
    except psycopg2.Error as e:
        logger.error("Database error while checking user existence: %s", e)
        raise
    finally:
        if conn:
            conn.close()


@app.route("/check-user", methods=["POST"])
def check_user():
    data = request.form if request.form else request.get_json(silent=True)

    if not data or "username" not in data:
        return jsonify({
            "error": "Missing required field: username",
            "exists": None
        }), 400

    username = data["username"]

    if not isinstance(username, str):
        return jsonify({
            "error": "Invalid type for username: must be a string",
            "exists": None
        }), 400

    username = username.strip()

    if not username:
        return jsonify({
            "error": "Username cannot be empty or whitespace",
            "exists": None
        }), 400

    if len(username) > 150:
        return jsonify({
            "error": "Username exceeds maximum allowed length of 150 characters",
            "exists": None
        }), 400

    try:
        exists = user_exists(username)
        return jsonify({
            "username": username,
            "exists": exists
        }), 200
    except psycopg2.OperationalError:
        return jsonify({
            "error": "Database connection error. Please try again later.",
            "exists": None
        }), 503
    except psycopg2.Error:
        return jsonify({
            "error": "An internal database error occurred.",
            "exists": None
        }), 500


@app.route("/health", methods=["GET"])
def health_check():
    try:
        conn = get_db_connection()
        conn.close()
        return jsonify({"status": "healthy", "database": "connected"}), 200
    except psycopg2.OperationalError:
        return jsonify({"status": "unhealthy", "database": "disconnected"}), 503


if __name__ == "__main__":
    app.run(debug=False, host="0.0.0.0", port=5000)
