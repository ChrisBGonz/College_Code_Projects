#!/usr/bin/env python3
"""
Script that loads a very large file (10GB) entirely into memory (RAM) to count total lines.
Uses Claude AI to analyze the file loading process.
"""

import os
import sys
import time
import anthropic


def load_file_and_count_lines(file_path: str) -> dict:
    """
    Load a large file entirely into memory and count its lines.
    
    Args:
        file_path: Path to the file to load
        
    Returns:
        Dictionary with file statistics
    """
    print(f"Loading file: {file_path}")
    print("Warning: This will load the entire file into RAM...")
    
    start_time = time.time()
    
    # Load entire file into memory
    with open(file_path, 'rb') as f:
        content = f.read()
    
    load_time = time.time() - start_time
    
    # Count lines
    count_start = time.time()
    line_count = content.count(b'\n')
    
    # Add 1 if file doesn't end with newline but has content
    if content and not content.endswith(b'\n'):
        line_count += 1
    
    count_time = time.time() - count_start
    
    file_size = len(content)
    
    return {
        "file_path": file_path,
        "file_size_bytes": file_size,
        "file_size_gb": file_size / (1024**3),
        "total_lines": line_count,
        "load_time_seconds": load_time,
        "count_time_seconds": count_time,
        "total_time_seconds": load_time + count_time
    }


def analyze_with_claude(stats: dict) -> None:
    """
    Use Claude AI with streaming to analyze the file loading results.
    
    Args:
        stats: Dictionary with file statistics
    """
    client = anthropic.Anthropic()
    
    prompt = f"""I just loaded a large file entirely into RAM to count its lines. Here are the statistics:

File Path: {stats['file_path']}
File Size: {stats['file_size_bytes']:,} bytes ({stats['file_size_gb']:.2f} GB)
Total Lines: {stats['total_lines']:,}
Load Time: {stats['load_time_seconds']:.2f} seconds
Count Time: {stats['count_time_seconds']:.4f} seconds
Total Time: {stats['total_time_seconds']:.2f} seconds

Please analyze these results and provide:
1. Assessment of the memory usage approach
2. Performance analysis
3. Potential risks and issues with this approach
4. Alternative approaches that would be more memory-efficient
5. When this approach might be acceptable vs when to avoid it"""

    print("\n" + "="*60)
    print("CLAUDE AI ANALYSIS (Streaming)")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "="*60)


def create_test_file(file_path: str, size_mb: int = 100) -> None:
    """
    Create a test file for demonstration purposes.
    
    Args:
        file_path: Path where to create the test file
        size_mb: Size of the test file in MB (default 100MB for demo)
    """
    print(f"Creating test file of approximately {size_mb}MB...")
    
    line = b"This is a test line with some content to simulate a large file. " * 2 + b"\n"
    lines_needed = (size_mb * 1024 * 1024) // len(line)
    
    with open(file_path, 'wb') as f:
        for i in range(lines_needed):
            f.write(line)
    
    actual_size = os.path.getsize(file_path)
    print(f"Test file created: {actual_size / (1024*1024):.1f}MB")


def main():
    """Main function to demonstrate loading a large file into memory."""
    
    # Check if a file path was provided as argument
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        
        if not os.path.exists(file_path):
            print(f"Error: File '{file_path}' not found.")
            sys.exit(1)
    else:
        # Create a test file for demonstration
        test_file = "test_large_file.txt"
        
        print("No file path provided. Creating a test file for demonstration.")
        print("Note: For a real 10GB file, provide the path as an argument.")
        print(f"Usage: python {sys.argv[0]} /path/to/large/file.txt\n")
        
        # Create a smaller test file (100MB) for demonstration
        create_test_file(test_file, size_mb=100)
        file_path = test_file
    
    try:
        # Load file and count lines
        print("\nStarting file loading process...")
        stats = load_file_and_count_lines(file_path)
        
        # Display results
        print("\n" + "="*60)
        print("FILE LOADING RESULTS")
        print("="*60)
        print(f"File: {stats['file_path']}")
        print(f"Size: {stats['file_size_bytes']:,} bytes ({stats['file_size_gb']:.2f} GB)")
        print(f"Total Lines: {stats['total_lines']:,}")
        print(f"Load Time: {stats['load_time_seconds']:.2f} seconds")
        print(f"Count Time: {stats['count_time_seconds']:.4f} seconds")
        print(f"Total Time: {stats['total_time_seconds']:.2f} seconds")
        
        # Analyze with Claude AI
        analyze_with_claude(stats)
        
    except MemoryError:
        print("\nERROR: Not enough RAM to load the file!")
        print("The file is too large to fit in available memory.")
        print("Consider using a streaming approach instead.")
        sys.exit(1)
    except Exception as e:
        print(f"\nError: {e}")
        sys.exit(1)
    finally:
        # Clean up test file if it was created
        if len(sys.argv) == 1 and os.path.exists("test_large_file.txt"):
            os.remove("test_large_file.txt")
            print("\nTest file cleaned up.")


if __name__ == "__main__":
    main()
