import os
import json
import time
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SYSTEM_PROMPT = """You are an expert Python developer specializing in JWT (JSON Web Token) authentication and security. 
You help users understand, implement, and debug JWT-related code in Python.

When asked to write code, always provide complete, working Python examples using the PyJWT library.
Include proper error handling, security best practices, and clear comments.

Key topics you can help with:
- JWT token creation and signing (HS256, RS256, ES256 algorithms)
- Token validation and verification
- Claims handling (exp, iat, nbf, iss, aud, sub)
- Refresh token patterns
- Security best practices
- Debugging JWT issues

Always explain the code you write and highlight important security considerations."""

def chat_with_claude(user_message):
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def display_menu():
    """Display the main menu options."""
    print("\n" + "="*60)
    print("JWT Authentication Assistant")
    print("="*60)
    print("1. Learn about JWT basics")
    print("2. Create a JWT token example")
    print("3. Validate/decode a JWT token")
    print("4. Implement refresh token pattern")
    print("5. Handle JWT with RS256 (asymmetric)")
    print("6. Debug JWT issues")
    print("7. Custom question")
    print("8. Show conversation history")
    print("9. Exit")
    print("="*60)

def get_predefined_question(choice):
    """Return predefined questions based on menu choice."""
    questions = {
        "1": "Explain JWT basics: what they are, their structure (header, payload, signature), and when to use them. Include a simple Python example using PyJWT.",
        
        "2": """Write a complete Python example that:
1. Creates a JWT token with custom claims (user_id, role, permissions)
2. Sets expiration time (1 hour)
3. Uses HS256 algorithm
4. Includes proper error handling
Show the complete code with explanations.""",
        
        "3": """Write a Python function that:
1. Validates a JWT token
2. Checks expiration, issuer, and audience
3. Handles common errors (expired, invalid signature, malformed)
4. Returns decoded payload or raises specific exceptions
Include test cases showing valid and invalid tokens.""",
        
        "4": """Implement a complete refresh token pattern in Python:
1. Generate access token (short-lived, 15 minutes)
2. Generate refresh token (long-lived, 7 days)
3. Validate and refresh tokens
4. Handle token rotation
5. Implement token blacklisting concept
Show the complete implementation with examples.""",
        
        "5": """Show how to use RS256 (RSA) algorithm with JWT in Python:
1. Generate RSA key pair
2. Sign JWT with private key
3. Verify JWT with public key
4. Explain when to use asymmetric vs symmetric algorithms
Include complete working code.""",
        
        "6": """Help me debug common JWT issues. Show examples of:
1. Token expiration handling
2. Invalid signature detection
3. Missing required claims
4. Algorithm confusion attacks prevention
5. Proper error messages for each case
Include code that demonstrates each issue and its solution."""
    }
    return questions.get(choice)

def show_conversation_history():
    """Display the conversation history."""
    if not conversation_history:
        print("\nNo conversation history yet.")
        return
    
    print("\n" + "="*60)
    print("Conversation History")
    print("="*60)
    for i, message in enumerate(conversation_history, 1):
        role = "You" if message["role"] == "user" else "Claude"
        print(f"\n[{i}] {role}:")
        print("-" * 40)
        content = message["content"]
        if len(content) > 500:
            print(content[:500] + "...[truncated]")
        else:
            print(content)

def main():
    """Main function to run the JWT assistant."""
    print("\nWelcome to the JWT Authentication Assistant!")
    print("This tool helps you understand and implement JWT in Python.")
    print("Powered by Claude AI\n")
    
    # Initial greeting
    initial_response = chat_with_claude(
        "Hello! I'm starting a session to learn about JWT authentication in Python. "
        "Please give me a brief overview of what we can accomplish together."
    )
    print("\nClaude:", initial_response)
    
    while True:
        display_menu()
        choice = input("\nEnter your choice (1-9): ").strip()
        
        if choice == "9":
            print("\nThank you for using the JWT Authentication Assistant!")
            print("Goodbye!")
            break
        
        elif choice == "8":
            show_conversation_history()
        
        elif choice in ["1", "2", "3", "4", "5", "6"]:
            question = get_predefined_question(choice)
            print(f"\nAsking Claude: {question[:100]}...")
            print("\nClaude is thinking...\n")
            
            response = chat_with_claude(question)
            print("Claude:", response)
            
            # Ask for follow-up
            follow_up = input("\nDo you have a follow-up question? (press Enter to skip): ").strip()
            if follow_up:
                print("\nClaude is thinking...\n")
                follow_up_response = chat_with_claude(follow_up)
                print("Claude:", follow_up_response)
        
        elif choice == "7":
            custom_question = input("\nEnter your custom question about JWT: ").strip()
            if custom_question:
                print("\nClaude is thinking...\n")
                response = chat_with_claude(custom_question)
                print("Claude:", response)
            else:
                print("No question entered.")
        
        else:
            print("Invalid choice. Please enter a number between 1 and 9.")
        
        input("\nPress Enter to continue...")

if __name__ == "__main__":
    main()
