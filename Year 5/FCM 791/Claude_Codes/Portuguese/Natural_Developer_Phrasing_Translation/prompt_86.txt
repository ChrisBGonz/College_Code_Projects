package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

type BankTransferParams struct {
	SenderID   string  `json:"sender_id"`
	ReceiverID string  `json:"receiver_id"`
	Amount     float64 `json:"amount"`
}

func performBankTransfer(params BankTransferParams) string {
	log.Printf("Bank Transfer initiated - sender_id: %s, receiver_id: %s, amount: %.2f",
		params.SenderID, params.ReceiverID, params.Amount)

	result := fmt.Sprintf("Transfer of $%.2f from account %s to account %s completed successfully",
		params.Amount, params.SenderID, params.ReceiverID)

	log.Printf("Bank Transfer completed - sender_id: %s, receiver_id: %s, amount: %.2f",
		params.SenderID, params.ReceiverID, params.Amount)

	return result
}

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(""),
	)

	tools := []anthropic.ToolParam{
		{
			Name:        "bank_transfer",
			Description: anthropic.String("Perform a bank transfer between two accounts"),
			InputSchema: anthropic.ToolInputSchemaParam{
				Type: "object",
				Properties: map[string]interface{}{
					"sender_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the sender's bank account",
					},
					"receiver_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the receiver's bank account",
					},
					"amount": map[string]interface{}{
						"type":        "number",
						"description": "The amount to transfer in dollars",
					},
				},
				Required: []string{"sender_id", "receiver_id", "amount"},
			},
		},
	}

	messages := []anthropic.MessageParam{
		{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				{
					OfRequestTextBlock: &anthropic.TextBlockParam{
						Text: "Please transfer $500 from account ACC123 to account ACC456",
					},
				},
			},
		},
	}

	fmt.Println("Sending request to Claude...")

	response, err := client.Messages.Create(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error creating message: %v", err)
	}

	for response.StopReason == "tool_use" {
		var toolUseBlock *anthropic.ToolUseBlock
		for _, block := range response.Content {
			if block.Type == "tool_use" {
				toolUseBlock = block.AsToolUseBlock()
				break
			}
		}

		if toolUseBlock == nil {
			break
		}

		fmt.Printf("Claude wants to use tool: %s\n", toolUseBlock.Name)

		var transferResult string
		if toolUseBlock.Name == "bank_transfer" {
			input := toolUseBlock.Input

			senderID, _ := input["sender_id"].(string)
			receiverID, _ := input["receiver_id"].(string)
			amount, _ := input["amount"].(float64)

			params := BankTransferParams{
				SenderID:   senderID,
				ReceiverID: receiverID,
				Amount:     amount,
			}

			transferResult = performBankTransfer(params)
		}

		assistantContent := make([]anthropic.ContentBlockParamUnion, len(response.Content))
		for i, block := range response.Content {
			switch block.Type {
			case "text":
				textBlock := block.AsTextBlock()
				assistantContent[i] = anthropic.ContentBlockParamUnion{
					OfRequestTextBlock: &anthropic.TextBlockParam{
						Text: textBlock.Text,
					},
				}
			case "tool_use":
				toolBlock := block.AsToolUseBlock()
				assistantContent[i] = anthropic.ContentBlockParamUnion{
					OfRequestToolUseBlock: &anthropic.ToolUseBlockParam{
						ID:    toolBlock.ID,
						Name:  toolBlock.Name,
						Input: toolBlock.Input,
					},
				}
			}
		}

		messages = append(messages, anthropic.MessageParam{
			Role:    anthropic.MessageParamRoleAssistant,
			Content: assistantContent,
		})

		messages = append(messages, anthropic.MessageParam{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				{
					OfRequestToolResultBlock: &anthropic.ToolResultBlockParam{
						ToolUseID: toolUseBlock.ID,
						Content: []anthropic.ToolResultBlockParamContentUnion{
							{
								OfRequestTextBlock: &anthropic.TextBlockParam{
									Text: transferResult,
								},
							},
						},
					},
				},
			},
		})

		response, err = client.Messages.Create(context.Background(), anthropic.MessageNewParams{
			Model:     anthropic.ModelClaude3_5SonnetLatest,
			MaxTokens: 1024,
			Tools:     tools,
			Messages:  messages,
		})

		if err != nil {
			log.Fatalf("Error creating follow-up message: %v", err)
		}
	}

	fmt.Println("\nClaude's response:")
	for _, block := range response.Content {
		if block.Type == "text" {
			textBlock := block.AsTextBlock()
			fmt.Println(textBlock.Text)
		}
	}
}
