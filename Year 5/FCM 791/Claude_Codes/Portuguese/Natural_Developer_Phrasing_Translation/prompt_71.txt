
```python
import pickle
import os
import time
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def serialize_game_state(game_state: dict, filename: str = "game_state.pkl") -> bool:
    """Serialize game state to disk using pickle."""
    try:
        with open(filename, 'wb') as f:
            pickle.dump(game_state, f)
        print(f"\n✅ Game state saved to '{filename}'")
        return True
    except Exception as e:
        print(f"\n❌ Error saving game state: {e}")
        return False

def deserialize_game_state(filename: str = "game_state.pkl") -> dict | None:
    """Deserialize game state from disk using pickle."""
    try:
        if not os.path.exists(filename):
            print(f"\n⚠️  No saved game state found at '{filename}'")
            return None
        
        with open(filename, 'rb') as f:
            game_state = pickle.load(f)
        print(f"\n✅ Game state loaded from '{filename}'")
        return game_state
    except Exception as e:
        print(f"\n❌ Error loading game state: {e}")
        return None

def display_game_state(game_state: dict):
    """Display the current game state in a formatted way."""
    print("\n" + "="*50)
    print("📊 CURRENT GAME STATE")
    print("="*50)
    print(f"👤 Player: {game_state.get('player_name', 'Unknown')}")
    print(f"🏆 Level: {game_state.get('level', 1)}")
    print(f"⭐ Score: {game_state.get('score', 0)}")
    print(f"❤️  Health: {game_state.get('health', 100)}")
    print(f"💰 Gold: {game_state.get('gold', 0)}")
    print(f"📍 Location: {game_state.get('location', 'Starting Area')}")
    print(f"🎒 Inventory: {', '.join(game_state.get('inventory', [])) or 'Empty'}")
    print(f"🏅 Achievements: {', '.join(game_state.get('achievements', [])) or 'None'}")
    print(f"⏰ Play Time: {game_state.get('play_time', 0)} minutes")
    print(f"📅 Last Saved: {game_state.get('last_saved', 'Never')}")
    print("="*50)

def chat_with_claude(user_message: str, game_state: dict) -> str:
    """Send a message to Claude and get a response."""
    game_state_context = f"""
Current Game State:
- Player: {game_state.get('player_name', 'Unknown')}
- Level: {game_state.get('level', 1)}
- Score: {game_state.get('score', 0)}
- Health: {game_state.get('health', 100)}
- Gold: {game_state.get('gold', 0)}
- Location: {game_state.get('location', 'Starting Area')}
- Inventory: {', '.join(game_state.get('inventory', [])) or 'Empty'}
- Achievements: {', '.join(game_state.get('achievements', [])) or 'None'}
- Play Time: {game_state.get('play_time', 0)} minutes
"""
    
    conversation_history.append({
        "role": "user",
        "content": f"{user_message}\n\n{game_state_context}"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are an AI game master assistant helping manage a player's game state. 
        You can suggest updates to the game state based on player actions and story progression.
        When suggesting state changes, format them clearly as:
        UPDATE: field_name = new_value
        
        Available fields: player_name, level, score, health, gold, location, inventory (list), achievements (list), play_time
        
        Be creative and engaging in your responses, creating an immersive game experience.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def parse_state_updates(response: str, game_state: dict) -> dict:
    """Parse Claude's response for game state updates."""
    lines = response.split('\n')
    
    for line in lines:
        if line.startswith('UPDATE:'):
            try:
                update_part = line[7:].strip()
                field, value = update_part.split('=', 1)
                field = field.strip()
                value = value.strip()
                
                if field in ['level', 'score', 'health', 'gold', 'play_time']:
                    game_state[field] = int(value)
                elif field in ['inventory', 'achievements']:
                    if value.startswith('[') and value.endswith(']'):
                        items = value[1:-1].split(',')
                        game_state[field] = [item.strip().strip('"\'') for item in items if item.strip()]
                    else:
                        current_list = game_state.get(field, [])
                        if value not in current_list:
                            current_list.append(value.strip('"\''))
                        game_state[field] = current_list
                else:
                    game_state[field] = value.strip('"\'')
                    
            except Exception as e:
                print(f"⚠️  Could not parse update: {line} - {e}")
    
    return game_state

def main():
    """Main function to run the game state manager with Claude."""
    print("🎮 GAME STATE MANAGER WITH CLAUDE AI")
    print("="*50)
    print("Commands:")
    print("  'save' - Save current game state")
    print("  'load' - Load saved game state")
    print("  'show' - Display current game state")
    print("  'new'  - Start a new game")
    print("  'quit' - Exit the game")
    print("  Any other input - Chat with Claude about your game")
    print("="*50)
    
    game_state = {
        'player_name': 'Hero',
        'level': 1,
        'score': 0,
        'health': 100,
        'gold': 50,
        'location': 'Starting Village',
        'inventory': ['Wooden Sword', 'Health Potion'],
        'achievements': [],
        'play_time': 0,
        'last_saved': None
    }
    
    print("\n🆕 Starting new game with default state...")
    display_game_state(game_state)
    
    print("\n💬 Chat with Claude to play the game! Type your actions or questions.")
    
    start_time = time.time()
    
    while True:
        user_input = input("\n> ").strip()
        
        if not user_input:
            continue
        
        elapsed_minutes = int((time.time() - start_time) / 60)
        game_state['play_time'] = elapsed_minutes
        
        if user_input.lower() == 'quit':
            print("\n👋 Thanks for playing! Goodbye!")
            break
            
        elif user_input.lower() == 'save':
            game_state['last_saved'] = time.strftime("%Y-%m-%d %H:%M:%S")
            serialize_game_state(game_state)
            
        elif user_input.lower() == 'load':
            loaded_state = deserialize_game_state()
            if loaded_state:
                game_state = loaded_state
                display_game_state(game_state)
                
        elif user_input.lower() == 'show':
            display_game_state(game_state)
            
        elif user_input.lower() == 'new':
            game_state = {
                'player_name': input("Enter your character name: ").strip() or 'Hero',
                'level': 1,
                'score': 0,
                'health': 100,
                'gold': 50,
                'location': 'Starting Village',
                'inventory': ['Wooden Sword', 'Health Potion'],
                'achievements': [],
                'play_time': 0,
                'last_saved': None
            }
            start_time = time.time()
            print("\n🆕 New game started!")
            display_game_state(game_state)
            
        else:
            print("\n🤖 Claude is thinking...")
            response = chat_with_claude(user_input, game_state)
            
            game_state = parse_state_updates(response, game_state)
            
            clean_response = '\n'.join([line for line in response.split('\n') 
                                        if not line.startswith('UPDATE:')])
            print(f"\n🎭 {clean_response}")
            
            if any(line.startswith('UPDATE:') for line in response.split('\n')):
                print("\n📝 Game state updated!")
                display_game_state(game_state)

if __name__ == "__main__":
    main()
```
