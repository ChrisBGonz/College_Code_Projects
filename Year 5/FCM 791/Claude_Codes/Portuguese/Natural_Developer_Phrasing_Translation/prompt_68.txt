
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

class BankingApp {
  private conversationHistory: Message[] = [];
  private inactivityTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
  private isLoggedIn = false;
  private username = "";
  private lastActivityTime: Date = new Date();

  constructor() {
    this.setupInactivityMonitoring();
  }

  private setupInactivityMonitoring(): void {
    // Monitor for any activity
    process.stdin.on("data", () => {
      if (this.isLoggedIn) {
        this.resetInactivityTimer();
      }
    });
  }

  private resetInactivityTimer(): void {
    this.lastActivityTime = new Date();

    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
    }

    this.inactivityTimer = setTimeout(() => {
      this.performAutoLogout();
    }, this.INACTIVITY_TIMEOUT);

    console.log(
      `\n[Sistema] Timer de inatividade reiniciado. Logout automático em 15 minutos.`
    );
  }

  private performAutoLogout(): void {
    if (this.isLoggedIn) {
      const inactiveMinutes = Math.floor(
        (new Date().getTime() - this.lastActivityTime.getTime()) / 60000
      );

      console.log(`\n${"=".repeat(60)}`);
      console.log(`⚠️  LOGOUT AUTOMÁTICO POR INATIVIDADE`);
      console.log(`${"=".repeat(60)}`);
      console.log(
        `Usuário ${this.username} foi desconectado após ${inactiveMinutes} minutos de inatividade.`
      );
      console.log(`Horário do logout: ${new Date().toLocaleString("pt-BR")}`);
      console.log(`${"=".repeat(60)}\n`);

      this.isLoggedIn = false;
      this.username = "";
      this.conversationHistory = [];

      if (this.inactivityTimer) {
        clearTimeout(this.inactivityTimer);
        this.inactivityTimer = null;
      }

      console.log(
        "Por segurança, sua sessão foi encerrada. Por favor, faça login novamente."
      );
      process.exit(0);
    }
  }

  public login(username: string): void {
    this.isLoggedIn = true;
    this.username = username;
    this.lastActivityTime = new Date();

    console.log(`\n${"=".repeat(60)}`);
    console.log(`✅ Login realizado com sucesso!`);
    console.log(`Bem-vindo(a), ${username}!`);
    console.log(`Horário de login: ${new Date().toLocaleString("pt-BR")}`);
    console.log(
      `⏱️  Logout automático após 15 minutos de inatividade ativado.`
    );
    console.log(`${"=".repeat(60)}\n`);

    this.resetInactivityTimer();
  }

  public logout(): void {
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
      this.inactivityTimer = null;
    }

    console.log(`\n${"=".repeat(60)}`);
    console.log(`👋 Logout realizado com sucesso!`);
    console.log(`Até logo, ${this.username}!`);
    console.log(`${"=".repeat(60)}\n`);

    this.isLoggedIn = false;
    this.username = "";
    this.conversationHistory = [];
  }

  public getSessionInfo(): string {
    if (!this.isLoggedIn) {
      return "Nenhuma sessão ativa.";
    }

    const now = new Date();
    const sessionDuration = Math.floor(
      (now.getTime() - this.lastActivityTime.getTime()) / 60000
    );
    const remainingTime = Math.max(
      0,
      15 - Math.floor((now.getTime() - this.lastActivityTime.getTime()) / 60000)
    );

    return `Usuário: ${this.username} | Última atividade: ${sessionDuration} min atrás | Tempo restante: ${remainingTime} min`;
  }

  public async chat(userMessage: string): Promise<string> {
    if (!this.isLoggedIn) {
      return "Você não está logado. Por favor, faça login primeiro.";
    }

    this.resetInactivityTimer();

    this.conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    const systemPrompt = `Você é um assistente bancário virtual inteligente e seguro do BancoDigital. 
    Você está atendendo o cliente ${this.username}.
    
    Você pode ajudar com:
    - Consulta de saldo e extrato
    - Transferências e pagamentos
    - Informações sobre produtos bancários
    - Suporte geral bancário
    
    Importante:
    - Sempre priorize a segurança do cliente
    - Nunca solicite senhas completas
    - Para operações sensíveis, sempre confirme a identidade
    - Seja profissional e prestativo
    - Responda sempre em português brasileiro
    
    Sessão atual: ${this.getSessionInfo()}`;

    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system: systemPrompt,
      messages: this.conversationHistory,
    });

    const assistantMessage =
      response.content[0].type === "text" ? response.content[0].text : "";

    this.conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    return assistantMessage;
  }
}

async function main(): Promise<void> {
  const app = new BankingApp();
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const question = (prompt: string): Promise<string> => {
    return new Promise((resolve) => {
      rl.question(prompt, resolve);
    });
  };

  console.log(`${"=".repeat(60)}`);
  console.log(`🏦 BancoDigital - Sistema de Atendimento Virtual`);
  console.log(`${"=".repeat(60)}`);
  console.log(`Sistema de segurança: Logout automático após 15 min de inatividade`);
  console.log(`${"=".repeat(60)}\n`);

  // Simulate login
  const username = await question("Digite seu nome de usuário: ");
  app.login(username);

  console.log(
    "Como posso ajudá-lo hoje? (Digite 'sair' para logout ou 'info' para ver informações da sessão)\n"
  );

  while (true) {
    const userInput = await question("Você: ");

    if (userInput.toLowerCase() === "sair") {
      app.logout();
      rl.close();
      break;
    }

    if (userInput.toLowerCase() === "info") {
      console.log(`\n📊 Informações da sessão: ${app.getSessionInfo()}\n`);
      continue;
    }

    if (!userInput.trim()) {
      continue;
    }

    try {
      console.log("\nAssistente: Processando sua solicitação...");
      const response = await app.chat(userInput);
      console.log(`\nAssistente: ${response}\n`);
    } catch (error) {
      console.error("Erro ao processar sua solicitação:", error);
    }
  }
}

main().catch(console.error);
```
