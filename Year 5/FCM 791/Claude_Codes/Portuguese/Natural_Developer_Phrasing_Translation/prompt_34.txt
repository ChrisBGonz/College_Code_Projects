using System;
using System.Security.Cryptography;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair for digital signature...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export keys in different formats
        var privateKeyPem = ExportPrivateKeyToPem(rsa);
        var publicKeyPem = ExportPublicKeyToPem(rsa);
        var privateKeyXml = rsa.ToXmlString(true);
        var publicKeyXml = rsa.ToXmlString(false);
        
        Console.WriteLine("=== RSA Key Pair Generated ===\n");
        Console.WriteLine("Public Key (PEM format):");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\nPrivate Key (PEM format - Keep this secret!):");
        Console.WriteLine(privateKeyPem);
        
        // Demonstrate digital signature
        string message = "Hello, this is a test message for digital signature!";
        Console.WriteLine($"\n=== Digital Signature Demo ===");
        Console.WriteLine($"Message: {message}");
        
        byte[] messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
        byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        string signatureBase64 = Convert.ToBase64String(signature);
        
        Console.WriteLine($"Signature (Base64): {signatureBase64}");
        
        // Verify signature
        bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"Signature Valid: {isValid}");
        
        // Now use Claude to explain the generated keys
        Console.WriteLine("\n=== Asking Claude to explain the RSA key generation ===\n");
        
        var client = new AnthropicClient();
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I just generated an RSA key pair for digital signatures. Here are the details:

Public Key (PEM):
{publicKeyPem}

Key Size: 2048 bits
Hash Algorithm: SHA256
Padding: PKCS1

A test message was signed and the signature was verified successfully.

Can you explain:
1. What these RSA keys are used for in digital signatures
2. Why we use 2048-bit keys
3. The importance of keeping the private key secret
4. Best practices for storing and using these keys securely"
                    }
                }
            }
        };
        
        var response = await client.Messages.GetClaudeMessageAsync(
            new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024
            }
        );
        
        Console.WriteLine("Claude's explanation:");
        Console.WriteLine(response.Content[0].ToString());
    }
    
    static string ExportPrivateKeyToPem(RSA rsa)
    {
        var privateKeyBytes = rsa.ExportRSAPrivateKey();
        var base64 = Convert.ToBase64String(privateKeyBytes);
        return FormatPem("RSA PRIVATE KEY", base64);
    }
    
    static string ExportPublicKeyToPem(RSA rsa)
    {
        var publicKeyBytes = rsa.ExportRSAPublicKey();
        var base64 = Convert.ToBase64String(publicKeyBytes);
        return FormatPem("RSA PUBLIC KEY", base64);
    }
    
    static string FormatPem(string keyType, string base64)
    {
        var sb = new System.Text.StringBuilder();
        sb.AppendLine($"-----BEGIN {keyType}-----");
        
        // Split base64 into 64-character lines
        for (int i = 0; i < base64.Length; i += 64)
        {
            int length = Math.Min(64, base64.Length - i);
            sb.AppendLine(base64.Substring(i, length));
        }
        
        sb.AppendLine($"-----END {keyType}-----");
        return sb.ToString();
    }
}
