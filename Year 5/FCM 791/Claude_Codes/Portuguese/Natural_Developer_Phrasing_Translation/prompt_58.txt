const STORAGE_KEY = 'wizard_form_data';
const CURRENT_STEP_KEY = 'wizard_current_step';

const steps = [
  {
    id: 1,
    title: 'Personal Information',
    fields: [
      { name: 'firstName', label: 'First Name', type: 'text', required: true },
      { name: 'lastName', label: 'Last Name', type: 'text', required: true },
      { name: 'email', label: 'Email', type: 'email', required: true },
      { name: 'phone', label: 'Phone', type: 'tel', required: false },
    ],
  },
  {
    id: 2,
    title: 'Address',
    fields: [
      { name: 'street', label: 'Street', type: 'text', required: true },
      { name: 'city', label: 'City', type: 'text', required: true },
      { name: 'state', label: 'State', type: 'text', required: true },
      { name: 'zipCode', label: 'ZIP Code', type: 'text', required: true },
      { name: 'country', label: 'Country', type: 'text', required: true },
    ],
  },
  {
    id: 3,
    title: 'Account Details',
    fields: [
      { name: 'username', label: 'Username', type: 'text', required: true },
      { name: 'password', label: 'Password', type: 'password', required: true },
      { name: 'confirmPassword', label: 'Confirm Password', type: 'password', required: true },
    ],
  },
  {
    id: 4,
    title: 'Preferences',
    fields: [
      { name: 'newsletter', label: 'Subscribe to Newsletter', type: 'checkbox', required: false },
      { name: 'notifications', label: 'Enable Notifications', type: 'checkbox', required: false },
      { name: 'bio', label: 'Bio', type: 'textarea', required: false },
    ],
  },
];

function WizardForm() {
  let currentStep = 1;
  let formData = {};

  function saveToLocalStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
      localStorage.setItem(CURRENT_STEP_KEY, String(currentStep));
    } catch (error) {
      console.error('Failed to save to localStorage:', error);
    }
  }

  function loadFromLocalStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

      if (savedData) {
        formData = JSON.parse(savedData);
      }

      if (savedStep) {
        const parsedStep = parseInt(savedStep, 10);
        if (parsedStep >= 1 && parsedStep <= steps.length) {
          currentStep = parsedStep;
        }
      }
    } catch (error) {
      console.error('Failed to load from localStorage:', error);
      formData = {};
      currentStep = 1;
    }
  }

  function clearLocalStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
    } catch (error) {
      console.error('Failed to clear localStorage:', error);
    }
  }

  function getStepData(stepId) {
    return formData[`step_${stepId}`] || {};
  }

  function setStepData(stepId, data) {
    formData[`step_${stepId}`] = { ...getStepData(stepId), ...data };
    saveToLocalStorage();
  }

  function collectStepData(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return {};

    const data = {};

    step.fields.forEach((field) => {
      const element = document.querySelector(`[name="${field.name}"]`);
      if (element) {
        if (field.type === 'checkbox') {
          data[field.name] = element.checked;
        } else {
          data[field.name] = element.value.trim();
        }
      }
    });

    return data;
  }

  function populateStepFields(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return;

    const savedData = getStepData(stepId);

    step.fields.forEach((field) => {
      const element = document.querySelector(`[name="${field.name}"]`);
      if (element && savedData[field.name] !== undefined) {
        if (field.type === 'checkbox') {
          element.checked = savedData[field.name];
        } else {
          element.value = savedData[field.name];
        }
      }
    });
  }

  function validateStep(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return { valid: false, errors: ['Step not found'] };

    const errors = [];
    const data = collectStepData(stepId);

    step.fields.forEach((field) => {
      if (field.required) {
        const value = data[field.name];
        if (value === undefined || value === null || value === '') {
          errors.push(`${field.label} is required`);
        }
      }

      if (field.name === 'email' && data.email) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(data.email)) {
          errors.push('Please enter a valid email address');
        }
      }

      if (field.name === 'confirmPassword' && stepId === 3) {
        const stepData = getStepData(stepId);
        const currentData = collectStepData(stepId);
        if (currentData.password !== currentData.confirmPassword) {
          errors.push('Passwords do not match');
        }
      }
    });

    return { valid: errors.length === 0, errors };
  }

  function showErrors(errors) {
    const existingErrors = document.querySelector('.wizard-errors');
    if (existingErrors) {
      existingErrors.remove();
    }

    if (errors.length === 0) return;

    const errorContainer = document.createElement('div');
    errorContainer.className = 'wizard-errors';
    errorContainer.setAttribute('role', 'alert');

    const errorList = document.createElement('ul');
    errors.forEach((error) => {
      const li = document.createElement('li');
      li.textContent = error;
      errorList.appendChild(li);
    });

    errorContainer.appendChild(errorList);

    const formContent = document.querySelector('.wizard-step-content');
    if (formContent) {
      formContent.insertBefore(errorContainer, formContent.firstChild);
    }
  }

  function clearErrors() {
    const existingErrors = document.querySelector('.wizard-errors');
    if (existingErrors) {
      existingErrors.remove();
    }
  }

  function updateProgressBar() {
    const progressBar = document.querySelector('.wizard-progress-bar');
    const progressSteps = document.querySelectorAll('.wizard-progress-step');

    if (progressBar) {
      const percentage = ((currentStep - 1) / (steps.length - 1)) * 100;
      progressBar.style.width = `${percentage}%`;
    }

    progressSteps.forEach((stepEl, index) => {
      const stepNumber = index + 1;
      stepEl.classList.remove('active', 'completed');

      if (stepNumber === currentStep) {
        stepEl.classList.add('active');
      } else if (stepNumber < currentStep) {
        stepEl.classList.add('completed');
      }
    });
  }

  function renderStepFields(step) {
    return step.fields
      .map((field) => {
        const savedData = getStepData(step.id);
        const value = savedData[field.name] || '';
        const checked = savedData[field.name] === true ? 'checked' : '';

        let inputHtml = '';

        switch (field.type) {
          case 'textarea':
            inputHtml = `
              <textarea
                name="${field.name}"
                id="${field.name}"
                class="wizard-input"
                placeholder="Enter your ${field.label.toLowerCase()}"
                rows="4"
              >${value}</textarea>
            `;
            break;

          case 'checkbox':
            inputHtml = `
              <div class="wizard-checkbox-wrapper">
                <input
                  type="checkbox"
                  name="${field.name}"
                  id="${field.name}"
                  class="wizard-checkbox"
                  ${checked}
                />
                <label for="${field.name}" class="wizard-checkbox-label">${field.label}</label>
              </div>
            `;
            break;

          default:
            inputHtml = `
              <input
                type="${field.type}"
                name="${field.name}"
                id="${field.name}"
                class="wizard-input"
                value="${field.type !== 'password' ? value : ''}"
                placeholder="Enter your ${field.label.toLowerCase()}"
                ${field.required ? 'required' : ''}
              />
            `;
        }

        return `
          <div class="wizard-field">
            ${field.type !== 'checkbox' ? `<label for="${field.name}" class="wizard-label">${field.label}${field.required ? ' <span class="required">*</span>' : ''}</label>` : ''}
            ${inputHtml}
          </div>
        `;
      })
      .join('');
  }

  function renderSummary() {
    const summaryItems = steps
      .map((step) => {
        const stepData = getStepData(step.id);
        const fields = step.fields
          .filter((field) => field.type !== 'password')
          .map((field) => {
            const value = stepData[field.name];
            let displayValue = '';

            if (field.type === 'checkbox') {
              displayValue = value ? 'Yes' : 'No';
            } else {
              displayValue = value || 'Not provided';
            }

            return `
              <div class="summary-field">
                <span class="summary-field-label">${field.label}:</span>
                <span class="summary-field-value">${displayValue}</span>
              </div>
            `;
          })
          .join('');

        return `
          <div class="summary-section">
            <h3 class="summary-section-title">${step.title}</h3>
            ${fields}
          </div>
        `;
      })
      .join('');

    return `
      <div class="wizard-summary">
        <h2>Review Your Information</h2>
        <p>Please review your information before submitting.</p>
        ${summaryItems}
      </div>
    `;
  }

  function renderStep() {
    const wizardContent = document.querySelector('.wizard-step-content');
    if (!wizardContent) return;

    clearErrors();

    const isSummaryStep = currentStep > steps.length;

    if (isSummaryStep) {
      wizardContent.innerHTML = renderSummary();
    } else {
      const step = steps.find((s) => s.id === currentStep);
      if (!step) return;

      wizardContent.innerHTML = `
        <h2 class="wizard-step-title">${step.title}</h2>
        <p class="wizard-step-subtitle">Step ${currentStep} of ${steps.length}</p>
        <div class="wizard-fields">
          ${renderStepFields(step)}
        </div>
      `;
    }

    updateNavigationButtons();
    updateProgressBar();
    attachFieldListeners();
  }

  function attachFieldListeners() {
    const inputs = document.querySelectorAll('.wizard-input, .wizard-checkbox');

    inputs.forEach((input) => {
      input.addEventListener('change', () => {
        const data = collectStepData(currentStep);
        setStepData(currentStep, data);
      });

      input.addEventListener('blur', () => {
        const data = collectStepData(currentStep);
        setStepData(currentStep, data);
      });
    });
  }

  function updateNavigationButtons() {
    const prevBtn = document.querySelector('.wizard-btn-prev');
    const nextBtn = document.querySelector('.wizard-btn-next');
    const submitBtn = document.querySelector('.wizard-btn-submit');

    const totalSteps = steps.length + 1;

    if (prevBtn) {
      prevBtn.style.display = currentStep > 1 ? 'inline-flex' : 'none';
    }

    if (nextBtn) {
      nextBtn.style.display = currentStep < totalSteps ? 'inline-flex' : 'none';
      nextBtn.textContent = currentStep === steps.length ? 'Review' : 'Next';
    }

    if (submitBtn) {
      submitBtn.style.display = currentStep === totalSteps ? 'inline-flex' : 'none';
    }
  }

  function goToNextStep() {
    if (currentStep <= steps.length) {
      const data = collectStepData(currentStep);
      const { valid, errors } = validateStep(currentStep);

      if (!valid) {
        showErrors(errors);
        return;
      }

      setStepData(currentStep, data);
      clearErrors();
    }

    currentStep = Math.min(currentStep + 1, steps.length + 1);
    saveToLocalStorage();
    renderStep();
  }

  function goToPrevStep() {
    if (currentStep > steps.length) {
      currentStep = steps.length;
    } else {
      const data = collectStepData(currentStep);
      setStepData(currentStep, data);
      currentStep = Math.max(currentStep - 1, 1);
    }

    clearErrors();
    saveToLocalStorage();
    renderStep();
  }

  function getAllFormData() {
    const allData = {};

    steps.forEach((step) => {
      const stepData = getStepData(step.id);
      step.fields.forEach((field) => {
        if (field.name !== 'confirmPassword') {
          allData[field.name] = stepData[field.name];
        }
      });
    });

    return allData;
  }

  async function submitForm() {
    const submitBtn = document.querySelector('.wizard-btn-submit');
    const originalText = submitBtn ? submitBtn.textContent : 'Submit';

    try {
      if (submitBtn) {
        submitBtn.disabled = true;
        submitBtn.textContent = 'Submitting...';
      }

      const finalData = getAllFormData();

      console.log('Submitting form data:', finalData);

      await simulateApiCall(finalData);

      clearLocalStorage();
      formData = {};
      currentStep = 1;

      showSuccessMessage();
    } catch (error) {
      console.error('Form submission failed:', error);
      showErrors([`Submission failed: ${error.message}. Please try again.`]);

      if (submitBtn) {
        submitBtn.disabled = false;
        submitBtn.textContent = originalText;
      }
    }
  }

  function simulateApiCall(data) {
    return new Promise((resolve, reject) => {
      setTimeout(() => {
        if (Math.random() > 0.1) {
          resolve({ success: true, userId: Math.floor(Math.random() * 10000) });
        } else {
          reject(new Error('Server error'));
        }
      }, 1500);
    });
  }

  function showSuccessMessage() {
    const wizardContainer = document.querySelector('.wizard-container');
    if (!wizardContainer) return;

    wizardContainer.innerHTML = `
      <div class="wizard-success">
        <div class="wizard-success-icon">✓</div>
        <h2>Registration Successful!</h2>
        <p>Thank you for registering. Your account has been created successfully.</p>
        <button class="wizard-btn wizard-btn-primary" onclick="location.reload()">
          Start Over
        </button>
      </div>
    `;
  }

  function resetForm() {
    if (confirm('Are you sure you want to reset the form? All data will be lost.')) {
      clearLocalStorage();
      formData = {};
      currentStep = 1;
      renderStep();
    }
  }

  function init() {
    loadFromLocalStorage();

    const container = document.querySelector('.wizard-container');
    if (!container) {
      console.error('Wizard container not found. Please add an element with class "wizard-container"');
      return;
    }

    const hasSavedData = Object.keys(formData).length > 0;

    if (hasSavedData) {
      const resumeConfirmed = confirm(
        `We found saved progress from a previous session (Step ${currentStep} of ${steps.length}). Would you like to continue where you left off?`
      );

      if (!resumeConfirmed) {
        clearLocalStorage();
        formData = {};
        currentStep = 1;
      }
    }

    renderWizard();
    renderStep();
    attachEventListeners();
  }

  function renderWizard() {
    const container = document.querySelector('.wizard-container');
    if (!container) return;

    const progressStepsHtml = steps
      .map(
        (step) => `
        <div class="wizard-progress-step ${step.id === currentStep ? 'active' : step.id < currentStep ? 'completed' : ''}" data-step="${step.id}">
          <div class="wizard-progress-step-circle">
            ${step.id < currentStep ? '✓' : step.id}
          </div>
          <span class="wizard-progress-step-label">${step.title}</span>
        </div>
      `
      )
      .join('');

    container.innerHTML = `
      <div class="wizard-header">
        <h1 class="wizard-title">Registration Form</h1>
        <button class="wizard-btn-reset" onclick="wizardInstance.resetForm()" title="Reset form">
          ↺ Reset
        </button>
      </div>

      <div class="wizard-progress">
        <div class="wizard-progress-track">
          <div class="wizard-progress-bar"></div>
        </div>
        <div class="wizard-progress-steps">
          ${progressStepsHtml}
        </div>
      </div>

      <div class="wizard-step-content">
      </div>

      <div class="wizard-navigation">
        <button class="wizard-btn wizard-btn-secondary wizard-btn-prev" style="display: none;">
          ← Previous
        </button>
        <div class="wizard-nav-right">
          <button class="wizard-btn wizard-btn-primary wizard-btn-next">
            Next →
          </button>
          <button class="wizard-btn wizard-btn-success wizard-btn-submit" style="display: none;">
            Submit ✓
          </button>
        </div>
      </div>

      <div class="wizard-storage-indicator">
        <span class="wizard-storage-dot"></span>
        <span class="wizard-storage-text">Auto-saving progress</span>
      </div>
    `;
  }

  function attachEventListeners() {
    document.addEventListener('click', (e) => {
      if (e.target.matches('.wizard-btn-next')) {
        goToNextStep();
      }

      if (e.target.matches('.wizard-btn-prev')) {
        goToPrevStep();
      }

      if (e.target.matches('.wizard-btn-submit')) {
        submitForm();
      }
    });

    window.addEventListener('beforeunload', () => {
      if (currentStep <= steps.length) {
        const data = collectStepData(currentStep);
        setStepData(currentStep, data);
      }
      saveToLocalStorage();
    });
  }

  return {
    init,
    resetForm,
    goToNextStep,
    goToPrevStep,
    submitForm,
    getCurrentStep: () => currentStep,
    getFormData: () => getAllFormData(),
    getSavedData: () => ({ ...formData }),
  };
}

const wizardInstance = WizardForm();

document.addEventListener('DOMContentLoaded', () => {
  wizardInstance.init();
});
