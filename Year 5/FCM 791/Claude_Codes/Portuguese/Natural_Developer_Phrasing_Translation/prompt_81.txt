import hashlib
import hmac
import json
from datetime import datetime, timedelta
from functools import wraps
from typing import Optional
import secrets
import re

# Simulated user database
USERS_DB = {
    "admin": {
        "password_hash": hashlib.sha256("Admin@123".encode()).hexdigest(),
        "role": "admin",
        "email": "admin@example.com",
        "active": True,
        "failed_attempts": 0,
        "locked_until": None
    },
    "user1": {
        "password_hash": hashlib.sha256("User@456".encode()).hexdigest(),
        "role": "user",
        "email": "user1@example.com",
        "active": True,
        "failed_attempts": 0,
        "locked_until": None
    },
    "inactive_user": {
        "password_hash": hashlib.sha256("Test@789".encode()).hexdigest(),
        "role": "user",
        "email": "inactive@example.com",
        "active": False,
        "failed_attempts": 0,
        "locked_until": None
    }
}

# Active sessions storage
ACTIVE_SESSIONS = {}

# Configuration
MAX_FAILED_ATTEMPTS = 3
LOCKOUT_DURATION_MINUTES = 15
SESSION_DURATION_HOURS = 1
MIN_PASSWORD_LENGTH = 8


class UnauthorizedException(Exception):
    """Custom exception for unauthorized access."""
    def __init__(self, message: str, error_code: str = "UNAUTHORIZED"):
        self.message = message
        self.error_code = error_code
        self.status_code = 401
        super().__init__(self.message)


class Response:
    """Simulates an HTTP response."""
    def __init__(self, status_code: int, body: dict, headers: dict = None):
        self.status_code = status_code
        self.body = body
        self.headers = headers or {}

    def to_dict(self) -> dict:
        return {
            "status_code": self.status_code,
            "headers": self.headers,
            "body": self.body
        }

    def __str__(self) -> str:
        return json.dumps(self.to_dict(), indent=2, default=str)


def hash_password(password: str) -> str:
    """Hash a password using SHA-256."""
    return hashlib.sha256(password.encode()).hexdigest()


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Verify a password against its hash using constant-time comparison."""
    plain_hash = hash_password(plain_password)
    return hmac.compare_digest(plain_hash, hashed_password)


def validate_password_strength(password: str) -> tuple[bool, list[str]]:
    """
    Validate password strength.
    Returns (is_valid, list_of_errors)
    """
    errors = []

    if len(password) < MIN_PASSWORD_LENGTH:
        errors.append(f"Password must be at least {MIN_PASSWORD_LENGTH} characters long")

    if not re.search(r'[A-Z]', password):
        errors.append("Password must contain at least one uppercase letter")

    if not re.search(r'[a-z]', password):
        errors.append("Password must contain at least one lowercase letter")

    if not re.search(r'\d', password):
        errors.append("Password must contain at least one digit")

    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        errors.append("Password must contain at least one special character")

    return len(errors) == 0, errors


def generate_session_token() -> str:
    """Generate a secure random session token."""
    return secrets.token_hex(32)


def create_session(username: str) -> dict:
    """Create a new session for the user."""
    token = generate_session_token()
    expiry = datetime.now() + timedelta(hours=SESSION_DURATION_HOURS)

    session = {
        "token": token,
        "username": username,
        "created_at": datetime.now(),
        "expires_at": expiry,
        "role": USERS_DB[username]["role"]
    }

    ACTIVE_SESSIONS[token] = session
    return session


def is_account_locked(username: str) -> tuple[bool, Optional[datetime]]:
    """Check if an account is locked due to failed attempts."""
    user = USERS_DB.get(username)
    if not user:
        return False, None

    locked_until = user.get("locked_until")
    if locked_until and datetime.now() < locked_until:
        return True, locked_until

    # Reset lock if expired
    if locked_until and datetime.now() >= locked_until:
        USERS_DB[username]["locked_until"] = None
        USERS_DB[username]["failed_attempts"] = 0

    return False, None


def increment_failed_attempts(username: str) -> None:
    """Increment failed login attempts and lock account if necessary."""
    if username not in USERS_DB:
        return

    USERS_DB[username]["failed_attempts"] += 1

    if USERS_DB[username]["failed_attempts"] >= MAX_FAILED_ATTEMPTS:
        USERS_DB[username]["locked_until"] = datetime.now() + timedelta(
            minutes=LOCKOUT_DURATION_MINUTES
        )
        print(f"[SECURITY] Account '{username}' locked until {USERS_DB[username]['locked_until']}")


def reset_failed_attempts(username: str) -> None:
    """Reset failed login attempts after successful login."""
    if username in USERS_DB:
        USERS_DB[username]["failed_attempts"] = 0
        USERS_DB[username]["locked_until"] = None


def validate_input(username: str, password: str) -> tuple[bool, list[str]]:
    """Validate input fields."""
    errors = []

    if not username or not isinstance(username, str):
        errors.append("Username is required")
    elif len(username.strip()) < 3:
        errors.append("Username must be at least 3 characters long")
    elif len(username) > 50:
        errors.append("Username must not exceed 50 characters")

    if not password or not isinstance(password, str):
        errors.append("Password is required")
    elif len(password) < 1:
        errors.append("Password cannot be empty")

    return len(errors) == 0, errors


def login(username: str, password: str) -> Response:
    """
    Main login function that validates credentials.
    Returns 401 Unauthorized if validation fails.
    """
    print(f"\n[LOGIN] Attempting login for user: '{username}'")

    # Step 1: Validate input
    is_valid_input, input_errors = validate_input(username, password)
    if not is_valid_input:
        print(f"[LOGIN] Input validation failed: {input_errors}")
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "Invalid input provided",
                "details": input_errors,
                "timestamp": datetime.now().isoformat()
            },
            headers={"WWW-Authenticate": "Bearer"}
        )

    username = username.strip().lower()

    # Step 2: Check if user exists (use generic message to prevent user enumeration)
    if username not in USERS_DB:
        print(f"[LOGIN] User '{username}' not found")
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "Invalid username or password",
                "timestamp": datetime.now().isoformat()
            },
            headers={"WWW-Authenticate": "Bearer"}
        )

    # Step 3: Check if account is locked
    is_locked, locked_until = is_account_locked(username)
    if is_locked:
        print(f"[LOGIN] Account '{username}' is locked until {locked_until}")
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": f"Account is temporarily locked due to multiple failed attempts. Try again after {locked_until.strftime('%Y-%m-%d %H:%M:%S')}",
                "locked_until": locked_until.isoformat(),
                "timestamp": datetime.now().isoformat()
            },
            headers={"WWW-Authenticate": "Bearer"}
        )

    # Step 4: Check if account is active
    if not USERS_DB[username]["active"]:
        print(f"[LOGIN] Account '{username}' is inactive")
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "Account is inactive. Please contact support.",
                "timestamp": datetime.now().isoformat()
            },
            headers={"WWW-Authenticate": "Bearer"}
        )

    # Step 5: Verify password
    stored_hash = USERS_DB[username]["password_hash"]
    if not verify_password(password, stored_hash):
        increment_failed_attempts(username)
        remaining_attempts = MAX_FAILED_ATTEMPTS - USERS_DB[username]["failed_attempts"]
        print(f"[LOGIN] Invalid password for '{username}'. Remaining attempts: {max(0, remaining_attempts)}")

        message = "Invalid username or password"
        if remaining_attempts <= 0:
            message = f"Account locked due to multiple failed attempts. Try again in {LOCKOUT_DURATION_MINUTES} minutes."
        elif remaining_attempts <= 1:
            message = f"Invalid username or password. Warning: {max(0, remaining_attempts)} attempt(s) remaining before account lockout."

        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": message,
                "timestamp": datetime.now().isoformat()
            },
            headers={"WWW-Authenticate": "Bearer"}
        )

    # Step 6: Successful login
    reset_failed_attempts(username)
    session = create_session(username)

    print(f"[LOGIN] Successful login for '{username}'")

    return Response(
        status_code=200,
        body={
            "message": "Login successful",
            "user": {
                "username": username,
                "email": USERS_DB[username]["email"],
                "role": USERS_DB[username]["role"]
            },
            "session": {
                "token": session["token"],
                "expires_at": session["expires_at"].isoformat()
            },
            "timestamp": datetime.now().isoformat()
        },
        headers={
            "Authorization": f"Bearer {session['token']}",
            "X-Session-Expires": session["expires_at"].isoformat()
        }
    )


def validate_session(token: str) -> Response:
    """Validate an existing session token."""
    if not token:
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "No session token provided",
                "timestamp": datetime.now().isoformat()
            }
        )

    session = ACTIVE_SESSIONS.get(token)

    if not session:
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "Invalid or expired session token",
                "timestamp": datetime.now().isoformat()
            }
        )

    if datetime.now() > session["expires_at"]:
        del ACTIVE_SESSIONS[token]
        return Response(
            status_code=401,
            body={
                "error": "Unauthorized",
                "message": "Session has expired. Please login again.",
                "timestamp": datetime.now().isoformat()
            }
        )

    return Response(
        status_code=200,
        body={
            "message": "Session is valid",
            "user": {
                "username": session["username"],
                "role": session["role"]
            },
            "session": {
                "expires_at": session["expires_at"].isoformat()
            },
            "timestamp": datetime.now().isoformat()
        }
    )


def logout(token: str) -> Response:
    """Logout and invalidate session."""
    if token in ACTIVE_SESSIONS:
        username = ACTIVE_SESSIONS[token]["username"]
        del ACTIVE_SESSIONS[token]
        print(f"[LOGOUT] User '{username}' logged out successfully")
        return Response(
            status_code=200,
            body={
                "message": "Logged out successfully",
                "timestamp": datetime.now().isoformat()
            }
        )

    return Response(
        status_code=401,
        body={
            "error": "Unauthorized",
            "message": "Invalid session token",
            "timestamp": datetime.now().isoformat()
        }
    )


def require_auth(func):
    """Decorator to protect routes that require authentication."""
    @wraps(func)
    def wrapper(*args, **kwargs):
        token = kwargs.get('token') or (args[0] if args else None)
        validation_response = validate_session(token)

        if validation_response.status_code != 200:
            return validation_response

        return func(*args, **kwargs)
    return wrapper


@require_auth
def protected_resource(token: str) -> Response:
    """Example of a protected resource."""
    session = ACTIVE_SESSIONS.get(token)
    return Response(
        status_code=200,
        body={
            "message": "Access granted to protected resource",
            "data": "This is sensitive data",
            "accessed_by": session["username"] if session else "unknown",
            "timestamp": datetime.now().isoformat()
        }
    )
