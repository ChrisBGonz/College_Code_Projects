$pdo = null;

function getConnection() {
    global $pdo;
    if ($pdo === null) {
        try {
            $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=utf8mb4";
            $pdo = new PDO($dsn, DB_USER, DB_PASS, [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
            ]);
        } catch (PDOException $e) {
            die("Erro de conexão: " . $e->getMessage());
        }
    }
    return $pdo;
}

function isAccountLocked(array $user): bool {
    if ($user['locked_until'] !== null) {
        $lockedUntil = new DateTime($user['locked_until']);
        $now = new DateTime();
        if ($now < $lockedUntil) {
            return true;
        }
        unlockAccount($user['id']);
    }
    return false;
}

function getLockTimeRemaining(array $user): int {
    if ($user['locked_until'] !== null) {
        $lockedUntil = new DateTime($user['locked_until']);
        $now = new DateTime();
        $diff = $now->diff($lockedUntil);
        return max(0, ($diff->h * 60) + $diff->i + 1);
    }
    return 0;
}

function incrementFailedAttempts(int $userId) {
    $db = getConnection();
    $stmt = $db->prepare("SELECT failed_attempts FROM users WHERE id = ?");
    $stmt->execute([$userId]);
    $user = $stmt->fetch();
    $attempts = $user['failed_attempts'] + 1;

    if ($attempts >= 5) {
        $lockUntil = (new DateTime())->modify('+30 minutes')->format('Y-m-d H:i:s');
        $db->prepare("UPDATE users SET failed_attempts = ?, locked_until = ? WHERE id = ?")
           ->execute([$attempts, $lockUntil, $userId]);
    } else {
        $db->prepare("UPDATE users SET failed_attempts = ? WHERE id = ?")
           ->execute([$attempts, $userId]);
    }
}

function resetFailedAttempts(int $userId) {
    $db = getConnection();
    $db->prepare("UPDATE users SET failed_attempts = 0, locked_until = NULL WHERE id = ?")
       ->execute([$userId]);
}

function unlockAccount(int $userId) {
    $db = getConnection();
    $db->prepare("UPDATE users SET failed_attempts = 0, locked_until = NULL WHERE id = ?")
       ->execute([$userId]);
}

function logRecoveryAttempt(int $userId, bool $success) {
    $db = getConnection();
    $ip = $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
    $db->prepare("INSERT INTO password_recovery_log (user_id, ip_address, success) VALUES (?, ?, ?)")
       ->execute([$userId, $ip, $success ? 1 : 0]);
}

function checkRateLimit(string $ip): bool {
    $db = getConnection();
    $stmt = $db->prepare("SELECT COUNT(*) FROM password_recovery_log WHERE ip_address = ? AND attempted_at > DATE_SUB(NOW(), INTERVAL 1 HOUR)");
    $stmt->execute([$ip]);
    $count = $stmt->fetchColumn();
    return $count < 10;
}

function updatePassword(int $userId, string $newPassword) {
    $db = getConnection();
    $hashed = password_hash($newPassword, PASSWORD_BCRYPT);
    $db->prepare("UPDATE users SET password = ? WHERE id = ?")
       ->execute([$hashed, $userId]);
}

function validatePassword(string $password): array {
    $errors = [];
    if (strlen($password) < 8) {
        $errors[] = "A senha deve ter pelo menos 8 caracteres.";
    }
    if (!preg_match('/[A-Z]/', $password)) {
        $errors[] = "A senha deve conter pelo menos uma letra maiúscula.";
    }
    if (!preg_match('/[a-z]/', $password)) {
        $errors[] = "A senha deve conter pelo menos uma letra minúscula.";
    }
    if (!preg_match('/[0-9]/', $password)) {
        $errors[] = "A senha deve conter pelo menos um número.";
    }
    if (!preg_match('/[\W_]/', $password)) {
        $errors[] = "A senha deve conter pelo menos um caractere especial.";
    }
    return $errors;
}

function sanitize(string $input): string {
    return htmlspecialchars(strip_tags(trim($input)), ENT_QUOTES, 'UTF-8');
}

function generateCsrfToken(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCsrfToken(string $token): bool {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

function regenerateCsrfToken() {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

                if (!$user) {

                    $step = 'email';
                    unset($_SESSION['recovery_step'], $_SESSION['recovery_user_id']);
                } elseif (isAccountLocked($user)) {
                    $minutes = getLockTimeRemaining($user);

                    $step = 'email';
                    unset($_SESSION['recovery_step'], $_SESSION['recovery_user_id']);
                } else {
                    $answer = strtolower(trim($_POST['secret_answer'] ?? ''));

                        $error = "Por favor, informe a resposta à pergunta secreta.";
                        $step = 'question';
                    } else {
                        if (hash_equals($user['secret_answer'], $answer)) {
                            logRecoveryAttempt($userId, true);
                            resetFailedAttempts($userId);
                            $_SESSION['recovery_step'] = 'new_password';
                            $_SESSION['recovery_verified'] = true;
                            $step = 'new_password';
                            regenerateCsrfToken();
                        } else {
                            logRecoveryAttempt($userId, false);
                            incrementFailedAttempts($userId);

                            $stmt2 = $db->prepare("SELECT failed_attempts, locked_until FROM users WHERE id = ?");
                            $stmt2->execute([$userId]);
                            $updatedUser = $stmt2->fetch();

                            if ($updatedUser['locked_until'] !== null) {
                                $error = "Resposta incorreta. Conta bloqueada por 30 minutos devido a múltiplas tentativas falhas.";
                                $step = 'email';
                                unset($_SESSION['recovery_step'], $_SESSION['recovery_user_id'], $_SESSION['recovery_verified']);
                            } else {
                                $remaining = 5 - $updatedUser['failed_attempts'];
                                $error = "Resposta incorreta. Você tem {$remaining} tentativa(s) restante(s).";
                                $step = 'question';
                            }
                        }
                    }
                }
            }

        } elseif ($step === 'new_password') {
            $userId = $_SESSION['recovery_user_id'] ?? null;
            $verified = $_SESSION['recovery_verified'] ?? false;

            if (!$userId || !$verified) {
                $error = "Sessão inválida. Por favor, comece novamente.";
                $step = 'email';
                unset($_SESSION['recovery_step'], $_SESSION['recovery_user_id'], $_SESSION['recovery_verified']);
            } else {
                $newPassword = $_POST['new_password'] ?? '';
                $confirmPassword = $_POST['confirm_password'] ?? '';

                if (empty($newPassword) || empty($confirmPassword)) {
                    $error = "Por favor, preencha todos os campos.";
                    $step = 'new_password';
                } elseif ($newPassword !== $confirmPassword) {
                    $error = "As senhas não coincidem.";
                    $step = 'new_password';
                } else {
                    $passwordErrors = validatePassword($newPassword);
                    if (!empty($passwordErrors)) {
                        $error = implode('<br>', $passwordErrors);
                        $step = 'new_password';
                    } else {
                        updatePassword($userId, $newPassword);

                        $success = "Senha alterada com sucesso! Você já pode fazer login com sua nova senha.";
                        $step = 'success';
                    }
                }
            }
        }
