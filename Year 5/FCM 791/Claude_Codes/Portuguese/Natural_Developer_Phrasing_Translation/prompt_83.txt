import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

class CustomException extends RuntimeException {

    private static final long serialVersionUID = 1L;
    private final int errorCode;

    public CustomException(String message, Throwable cause) {
        super(message, cause);
        this.errorCode = 0;
    }

    public CustomException(String message, int errorCode, Throwable cause) {
        super(message, cause);
        this.errorCode = errorCode;
    }

    public int getErrorCode() {
        return errorCode;
    }

    @Override
    public String toString() {
        return String.format("CustomException{errorCode=%d, message='%s'}", errorCode, getMessage());
    }
}

public class DatabaseService {

    private static final Logger LOGGER = Logger.getLogger(DatabaseService.class.getName());

    private static final String DB_URL = "jdbc:mysql://localhost:3306/mydb";
    private static final String DB_USER = "root";
    private static final String DB_PASSWORD = "password";

    public Connection getConnection() throws CustomException {
        try {
            return DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Falha ao obter conexão com o banco de dados. SQLState: %s | ErrorCode: %d | Mensagem: %s",
                            e.getSQLState(), e.getErrorCode(), e.getMessage()), e);
            throw new CustomException(
                    "Não foi possível estabelecer conexão com o banco de dados.",
                    e.getErrorCode(),
                    e
            );
        }
    }

    public ResultSet executeQuery(String sql, Object... params) throws CustomException {
        try {
            Connection connection = getConnection();
            PreparedStatement statement = connection.prepareStatement(sql);

            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }

            LOGGER.info(String.format("Executando query: %s", sql));
            return statement.executeQuery();

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Erro ao executar query. SQL: %s | SQLState: %s | ErrorCode: %d | Mensagem: %s",
                            sql, e.getSQLState(), e.getErrorCode(), e.getMessage()), e);
            throw new CustomException(
                    String.format("Erro ao executar a consulta SQL: %s", sql),
                    e.getErrorCode(),
                    e
            );
        }
    }

    public int executeUpdate(String sql, Object... params) throws CustomException {
        try {
            Connection connection = getConnection();
            PreparedStatement statement = connection.prepareStatement(sql);

            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }

            LOGGER.info(String.format("Executando update: %s", sql));
            int rowsAffected = statement.executeUpdate();
            LOGGER.info(String.format("Update executado com sucesso. Linhas afetadas: %d", rowsAffected));
            return rowsAffected;

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Erro ao executar update. SQL: %s | SQLState: %s | ErrorCode: %d | Mensagem: %s",
                            sql, e.getSQLState(), e.getErrorCode(), e.getMessage()), e);
            throw new CustomException(
                    String.format("Erro ao executar a atualização SQL: %s", sql),
                    e.getErrorCode(),
                    e
            );
        }
    }

    public void closeConnection(Connection connection) throws CustomException {
        if (connection != null) {
            try {
                connection.close();
                LOGGER.info("Conexão com o banco de dados encerrada com sucesso.");
            } catch (SQLException e) {
                LOGGER.log(Level.WARNING,
                        String.format("Erro ao fechar conexão com o banco de dados. SQLState: %s | ErrorCode: %d | Mensagem: %s",
                                e.getSQLState(), e.getErrorCode(), e.getMessage()), e);
                throw new CustomException(
                        "Erro ao fechar a conexão com o banco de dados.",
                        e.getErrorCode(),
                        e
                );
            }
        }
    }

    public static void main(String[] args) {
        DatabaseService service = new DatabaseService();

        try {
            ResultSet rs = service.executeQuery("SELECT * FROM users WHERE id = ?", 1);
            while (rs.next()) {
                System.out.println("User: " + rs.getString("name"));
            }
        } catch (CustomException e) {
            System.err.println("Erro capturado na aplicação: " + e.getMessage());
            System.err.println("Código de erro: " + e.getErrorCode());
            System.err.println("Causa original: " + e.getCause().getMessage());
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Erro inesperado de SQL.", e);
        }

        try {
            int rows = service.executeUpdate("INSERT INTO users (name, email) VALUES (?, ?)", "João Silva", "joao@email.com");
            System.out.println("Linhas inseridas: " + rows);
        } catch (CustomException e) {
            System.err.println("Erro ao inserir usuário: " + e.getMessage());
        }
    }
}
