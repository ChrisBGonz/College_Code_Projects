using System;
using System.Collections.Generic;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using System.Text.Json;
using System.Threading.Tasks;

public class CartItem
{
    public string Name { get; set; }
    public decimal Price { get; set; }
    public int Quantity { get; set; }
}

public class CheckoutResult
{
    public List<CartItem> Items { get; set; }
    public decimal Subtotal { get; set; }
    public decimal Tax { get; set; }
    public decimal Total { get; set; }
    public string Summary { get; set; }
}

public class ShoppingCartCheckout
{
    private readonly AnthropicClient _client;
    private const decimal TaxRate = 0.1m; // 10% tax rate

    public ShoppingCartCheckout()
    {
        _client = new AnthropicClient();
    }

    public async Task<CheckoutResult> ProcessCheckout(List<CartItem> items)
    {
        // Calculate totals
        decimal subtotal = 0;
        foreach (var item in items)
        {
            subtotal += item.Price * item.Quantity;
        }
        
        decimal tax = subtotal * TaxRate;
        decimal total = subtotal + tax;

        // Create checkout result
        var result = new CheckoutResult
        {
            Items = items,
            Subtotal = subtotal,
            Tax = tax,
            Total = total
        };

        // Use Claude to generate a summary
        result.Summary = await GenerateCheckoutSummary(result);

        return result;
    }

    private async Task<string> GenerateCheckoutSummary(CheckoutResult checkout)
    {
        var itemsDescription = string.Join(", ", checkout.Items.ConvertAll(item => 
            $"{item.Name} (x{item.Quantity}) = ${item.Price * item.Quantity:F2}"));

        var prompt = $@"Generate a friendly checkout summary for a shopping cart with the following details:
Items: {itemsDescription}
Subtotal: ${checkout.Subtotal:F2}
Tax (10%): ${checkout.Tax:F2}
Total: ${checkout.Total:F2}

Please provide a brief, friendly summary of the purchase in 2-3 sentences.";

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = prompt
            }
        };

        var request = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 200
        };

        var response = await _client.Messages.GetClaudeMessageAsync(request);
        
        return response.Content[0].ToString();
    }

    public void DisplayCheckoutResult(CheckoutResult result)
    {
        Console.WriteLine("\n=== CHECKOUT SUMMARY ===");
        Console.WriteLine("\nItems:");
        foreach (var item in result.Items)
        {
            Console.WriteLine($"  - {item.Name}: ${item.Price:F2} x {item.Quantity} = ${item.Price * item.Quantity:F2}");
        }
        Console.WriteLine($"\nSubtotal: ${result.Subtotal:F2}");
        Console.WriteLine($"Tax (10%): ${result.Tax:F2}");
        Console.WriteLine($"Total: ${result.Total:F2}");
        Console.WriteLine($"\nAI Summary: {result.Summary}");
        Console.WriteLine("========================\n");
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("Shopping Cart Checkout System");
        Console.WriteLine("==============================\n");

        // Simulate items from frontend
        var cartItems = new List<CartItem>
        {
            new CartItem { Name = "Laptop", Price = 999.99m, Quantity = 1 },
            new CartItem { Name = "Mouse", Price = 29.99m, Quantity = 2 },
            new CartItem { Name = "Keyboard", Price = 79.99m, Quantity = 1 },
            new CartItem { Name = "USB Hub", Price = 39.99m, Quantity = 1 }
        };

        Console.WriteLine("Processing checkout for the following items:");
        foreach (var item in cartItems)
        {
            Console.WriteLine($"  - {item.Name}: ${item.Price:F2} x {item.Quantity}");
        }

        var checkout = new ShoppingCartCheckout();
        
        try
        {
            Console.WriteLine("\nCalculating totals and generating summary...");
            var result = await checkout.ProcessCheckout(cartItems);
            checkout.DisplayCheckoutResult(result);

            // Demonstrate with another cart
            Console.WriteLine("\nProcessing another cart...");
            var anotherCart = new List<CartItem>
            {
                new CartItem { Name = "Book: Clean Code", Price = 45.00m, Quantity = 1 },
                new CartItem { Name = "Coffee Mug", Price = 15.99m, Quantity = 3 },
                new CartItem { Name = "Notebook", Price = 8.99m, Quantity = 5 }
            };

            var result2 = await checkout.ProcessCheckout(anotherCart);
            checkout.DisplayCheckoutResult(result2);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error processing checkout: {ex.Message}");
        }
    }
}
