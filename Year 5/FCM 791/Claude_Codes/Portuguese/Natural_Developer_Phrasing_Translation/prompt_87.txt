
```python
#!/usr/bin/env python3

import anthropic
import time
import os
import sys
import threading
from datetime import datetime
from collections import deque

# Initialize the Anthropic client
client = anthropic.Anthropic()

# Thread-safe deque to store timestamps of failed login attempts
failed_login_timestamps = deque()
timestamps_lock = threading.Lock()

# Flag to control the monitoring thread
monitoring_active = True


def analyze_log_line_with_claude(log_line: str, context: str) -> str:
    """Use Claude to analyze a log line and provide insights."""
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Analyze this log line for security concerns:
                
Log line: {log_line}
Context: {context}

Provide a brief security analysis (2-3 sentences max). Focus on:
1. Whether this indicates a security threat
2. Recommended action if needed
3. Severity level (LOW/MEDIUM/HIGH/CRITICAL)""",
            }
        ],
    )
    return message.content[0].text


def check_failed_login_threshold() -> tuple[bool, int]:
    """Check if failed login attempts exceed threshold in the last minute."""
    current_time = time.time()
    one_minute_ago = current_time - 60

    with timestamps_lock:
        # Remove timestamps older than 1 minute
        while failed_login_timestamps and failed_login_timestamps[0] < one_minute_ago:
            failed_login_timestamps.popleft()

        count = len(failed_login_timestamps)

    return count > 10, count


def process_log_line(line: str) -> None:
    """Process a single log line and check for failed login attempts."""
    if "Failed Login" in line or "Failed login" in line or "failed login" in line:
        current_time = time.time()

        with timestamps_lock:
            failed_login_timestamps.append(current_time)

        threshold_exceeded, count = check_failed_login_threshold()

        if threshold_exceeded:
            timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            print(f"\n{'='*60}")
            print(f"⚠️  WARNING: SECURITY ALERT at {timestamp}")
            print(f"{'='*60}")
            print(f"🚨 DETECTED {count} Failed Login attempts in the last 60 seconds!")
            print(f"📝 Latest log entry: {line.strip()}")

            # Use Claude to analyze the security situation
            print("\n🤖 Claude Security Analysis:")
            print("-" * 40)

            context = f"There have been {count} failed login attempts in the last 60 seconds. This exceeds the threshold of 10 attempts per minute."

            try:
                analysis = analyze_log_line_with_claude(line.strip(), context)
                print(analysis)
            except Exception as e:
                print(f"Could not get Claude analysis: {e}")

            print("=" * 60 + "\n")
        else:
            timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            print(
                f"[{timestamp}] 🔍 Failed Login detected (count in last minute: {count}/10): {line.strip()}"
            )


def tail_log_file(filepath: str) -> None:
    """Monitor a log file in real-time (similar to tail -f)."""
    print(f"🔍 Starting log monitor for: {filepath}")
    print(f"📊 Alert threshold: >10 'Failed Login' occurrences per minute")
    print(f"⏰ Monitoring started at: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("-" * 60)

    # Check if file exists
    if not os.path.exists(filepath):
        print(f"❌ Error: File '{filepath}' not found.")
        print("Creating a test log file for demonstration...")
        create_test_log_file(filepath)

    try:
        with open(filepath, "r") as f:
            # Move to end of file
            f.seek(0, 2)

            while monitoring_active:
                line = f.readline()

                if line:
                    process_log_line(line)
                else:
                    # No new content, wait briefly
                    time.sleep(0.1)

    except KeyboardInterrupt:
        print("\n\n⛔ Monitoring stopped by user.")
    except Exception as e:
        print(f"\n❌ Error reading log file: {e}")


def create_test_log_file(filepath: str) -> None:
    """Create a test log file with sample entries."""
    with open(filepath, "w") as f:
        f.write(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] System started\n")
        f.write(
            f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Authentication service initialized\n"
        )
    print(f"✅ Test log file created: {filepath}")


def simulate_log_entries(filepath: str) -> None:
    """Simulate log entries for testing purposes."""
    print("\n🎭 Starting log simulation (for testing)...")
    print("This will write test entries to the log file.\n")

    time.sleep(2)  # Wait for tail to start

    test_entries = [
        "User admin logged in successfully",
        "Failed Login attempt for user: john",
        "Failed Login attempt for user: admin",
        "System health check passed",
        "Failed Login attempt for user: root",
        "Failed Login attempt for user: test",
        "Failed Login attempt for user: guest",
        "Database connection established",
        "Failed Login attempt for user: administrator",
        "Failed Login attempt for user: user1",
        "Failed Login attempt for user: user2",
        "Failed Login attempt for user: user3",
        "Failed Login attempt for user: user4",
        "Failed Login attempt for user: user5",
        "System backup completed",
        "Failed Login attempt for user: hacker",
    ]

    with open(filepath, "a") as f:
        for i, entry in enumerate(test_entries):
            timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            log_line = f"[{timestamp}] {entry}\n"
            f.write(log_line)
            f.flush()
            print(f"📝 Writing to log: {entry}")
            time.sleep(0.5)  # Write entries every 0.5 seconds


def main():
    """Main function to run the log monitor."""
    # Default log file path
    log_file = "/tmp/test_auth.log"

    # Check if a custom log file path was provided
    if len(sys.argv) > 1:
        log_file = sys.argv[1]

    # Check if we should run in simulation mode
    simulate = "--simulate" in sys.argv or len(sys.argv) == 1

    if simulate:
        print("🚀 Running in SIMULATION MODE")
        print("Use: python solution.py /path/to/logfile to monitor a real log file")
        print()

        # Start simulation in a separate thread
        sim_thread = threading.Thread(
            target=simulate_log_entries, args=(log_file,), daemon=True
        )
        sim_thread.start()

    # Start monitoring the log file
    try:
        tail_log_file(log_file)
    except KeyboardInterrupt:
        global monitoring_active
        monitoring_active = False
        print("\n\n✅ Log monitoring stopped.")


if __name__ == "__main__":
    main()
```
