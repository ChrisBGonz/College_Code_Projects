package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func readConfigPathFromEnv(envVarName string) (string, error) {
	configPath := os.Getenv(envVarName)
	if configPath == "" {
		return "", fmt.Errorf("environment variable %s is not set or empty", envVarName)
	}
	return configPath, nil
}

func main() {
	client := anthropic.NewClient()

	fmt.Println("Config File Path Reader using Claude AI")
	fmt.Println("========================================")

	// Example: Set an environment variable for demonstration
	os.Setenv("APP_CONFIG_PATH", "/etc/myapp/config.yaml")

	// Use Claude to explain the function and demonstrate its usage
	prompt := `I have a Go function called readConfigPathFromEnv that reads a configuration file path from an environment variable. 
	
Here's the function:
func readConfigPathFromEnv(envVarName string) (string, error) {
    configPath := os.Getenv(envVarName)
    if configPath == "" {
        return "", fmt.Errorf("environment variable %s is not set or empty", envVarName)
    }
    return configPath, nil
}

Please explain:
1. What this function does
2. How to use it properly
3. Best practices for environment variable names for config paths
4. Common use cases

Keep the response concise and practical.`

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})
	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	fmt.Println("\nClaude's explanation:")
	fmt.Println("---------------------")
	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}

	// Demonstrate the function usage
	fmt.Println("\nDemonstration:")
	fmt.Println("--------------")

	// Test with existing environment variable
	configPath, err := readConfigPathFromEnv("APP_CONFIG_PATH")
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config path from APP_CONFIG_PATH: %s\n", configPath)
	}

	// Test with non-existing environment variable
	configPath, err = readConfigPathFromEnv("NON_EXISTENT_VAR")
	if err != nil {
		fmt.Printf("Error (expected): %v\n", err)
	} else {
		fmt.Printf("Config path: %s\n", configPath)
	}

	// Test with common environment variable names
	commonEnvVars := []string{
		"CONFIG_FILE",
		"APP_CONFIG",
		"MYAPP_CONFIG_PATH",
		"HOME",
	}

	fmt.Println("\nTesting common environment variable names:")
	for _, envVar := range commonEnvVars {
		path, err := readConfigPathFromEnv(envVar)
		if err != nil {
			fmt.Printf("  %s: not set\n", envVar)
		} else {
			fmt.Printf("  %s: %s\n", envVar, path)
		}
	}
}
