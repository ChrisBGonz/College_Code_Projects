
using System;
using System.Data;
using Microsoft.Data.SqlClient;

class BankTransactionDemo
{
    private const string ConnectionString =
        "Server=localhost;Database=BankDB;Trusted_Connection=True;TrustServerCertificate=True;";

    static void Main()
    {
        try
        {
            TransferFunds(accountIdFrom: 1, accountIdTo: 2, amount: 100.00m);
            Console.WriteLine("Transfer completed successfully.");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Transfer failed: {ex.Message}");
        }
    }

    static void TransferFunds(int accountIdFrom, int accountIdTo, decimal amount)
    {
        using SqlConnection connection = new SqlConnection(ConnectionString);
        connection.Open();

        using SqlTransaction transaction = connection.BeginTransaction(IsolationLevel.ReadCommitted);

        try
        {
            // Verify source account has sufficient funds
            decimal balance = GetBalance(connection, transaction, accountIdFrom);

            if (balance < amount)
            {
                throw new InvalidOperationException(
                    $"Insufficient funds. Available balance: {balance:C}, Required: {amount:C}");
            }

            // Debit Account A
            DebitAccount(connection, transaction, accountIdFrom, amount);

            // Credit Account B
            CreditAccount(connection, transaction, accountIdTo, amount);

            // Commit the transaction
            transaction.Commit();

            Console.WriteLine($"Debited  {amount:C} from Account {accountIdFrom}. New balance: {GetBalance(connection, null, accountIdFrom):C}");
            Console.WriteLine($"Credited {amount:C} to   Account {accountIdTo}. New balance: {GetBalance(connection, null, accountIdTo):C}");
        }
        catch
        {
            // Rollback on any error
            transaction.Rollback();
            Console.WriteLine("Transaction rolled back due to an error.");
            throw;
        }
    }

    static decimal GetBalance(SqlConnection connection, SqlTransaction? transaction, int accountId)
    {
        const string query = "SELECT Balance FROM Accounts WHERE AccountId = @AccountId";

        using SqlCommand command = new SqlCommand(query, connection, transaction);
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;

        object? result = command.ExecuteScalar();

        if (result is null || result == DBNull.Value)
        {
            throw new ArgumentException($"Account {accountId} not found.");
        }

        return Convert.ToDecimal(result);
    }

    static void DebitAccount(SqlConnection connection, SqlTransaction transaction, int accountId, decimal amount)
    {
        const string query = @"
            UPDATE Accounts
            SET Balance = Balance - @Amount
            WHERE AccountId = @AccountId AND Balance >= @Amount";

        using SqlCommand command = new SqlCommand(query, connection, transaction);
        command.Parameters.Add("@Amount",    SqlDbType.Decimal).Value = amount;
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value     = accountId;

        int rowsAffected = command.ExecuteNonQuery();

        if (rowsAffected == 0)
        {
            throw new InvalidOperationException(
                $"Debit failed for Account {accountId}. Insufficient funds or account not found.");
        }
    }

    static void CreditAccount(SqlConnection connection, SqlTransaction transaction, int accountId, decimal amount)
    {
        const string query = @"
            UPDATE Accounts
            SET Balance = Balance + @Amount
            WHERE AccountId = @AccountId";

        using SqlCommand command = new SqlCommand(query, connection, transaction);
        command.Parameters.Add("@Amount",    SqlDbType.Decimal).Value = amount;
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value     = accountId;

        int rowsAffected = command.ExecuteNonQuery();

        if (rowsAffected == 0)
        {
            throw new ArgumentException(
                $"Credit failed. Account {accountId} not found.");
        }
    }
}
