const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");
const url = require("url");

const client = new Anthropic();

async function generateUserProfileData(userId) {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a realistic user profile data for user ID: ${userId}. 
        Return ONLY a valid JSON object with the following fields:
        - id (string, use the provided user_id)
        - name (string, full name)
        - email (string, email address)
        - username (string)
        - age (number)
        - location (object with city, state, country)
        - bio (string, short biography)
        - joinDate (string, ISO date format)
        - preferences (object with theme, language, notifications boolean)
        - stats (object with postsCount, followersCount, followingCount)
        
        Return ONLY the JSON object, no additional text or explanation.`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  const jsonMatch = responseText.match(/\{[\s\S]*\}/);
  if (!jsonMatch) {
    throw new Error("Failed to extract JSON from response");
  }

  return JSON.parse(jsonMatch[0]);
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  if (pathname === "/api/profile/download" && req.method === "GET") {
    const userId = query.user_id;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing required parameter: user_id",
          message: "Please provide a user_id in the query string",
        })
      );
      return;
    }

    try {
      console.log(`Generating profile data for user ID: ${userId}`);

      const profileData = await generateUserProfileData(userId);

      const jsonString = JSON.stringify(profileData, null, 2);
      const filename = `profile_${userId}_${Date.now()}.json`;

      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="${filename}"`,
        "Content-Length": Buffer.byteLength(jsonString),
        "Cache-Control": "no-cache",
      });

      res.end(jsonString);

      console.log(`Profile data downloaded successfully for user ID: ${userId}`);
    } catch (error) {
      console.error("Error generating profile data:", error);

      if (error instanceof SyntaxError) {
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Failed to parse profile data",
            message: "The generated data was not valid JSON",
          })
        );
      } else {
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal server error",
            message: "Failed to generate profile data",
          })
        );
      }
    }
  } else if (pathname === "/" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Profile Download API",
        endpoints: {
          download: {
            method: "GET",
            path: "/api/profile/download",
            queryParams: {
              user_id: "required - The ID of the user whose profile to download",
            },
            description: "Download user profile data as a JSON file",
            example: "/api/profile/download?user_id=12345",
          },
        },
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not Found",
        message: `Endpoint ${pathname} not found`,
      })
    );
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET http://localhost:${PORT}/`);
  console.log(
    `  GET http://localhost:${PORT}/api/profile/download?user_id=<user_id>`
  );
  console.log(`\nExample:`);
  console.log(
    `  curl "http://localhost:${PORT}/api/profile/download?user_id=12345" -o profile.json`
  );
});

module.exports = server;
