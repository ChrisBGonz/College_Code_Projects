import os
import secrets
from flask import Flask, request, make_response, redirect, url_for, render_template_string
from anthropic import Anthropic

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Store conversation history for multi-turn conversations
conversation_history = []

# Simple user database (in production, use a real database)
users = {
    "admin": "password123",
    "user": "mypassword"
}

# Store active sessions
active_sessions = {}

def get_ai_response(user_message):
    """Get response from Claude using multi-turn conversation."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a helpful assistant in a secure web application. Be friendly and helpful to authenticated users.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

@app.route('/')
def index():
    """Redirect to login or dashboard based on auth status."""
    auth_token = request.cookies.get('auth_token')
    if auth_token and auth_token in active_sessions:
        return redirect(url_for('dashboard'))
    return redirect(url_for('login'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Handle login requests."""
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        
        if username in users and users[username] == password:
            # Generate auth token
            auth_token = secrets.token_hex(32)
            active_sessions[auth_token] = username
            
            # Create response with cookie
            response = make_response(redirect(url_for('dashboard')))
            response.set_cookie(
                'auth_token',
                auth_token,
                httponly=True,
                secure=False,  # Set to True in production with HTTPS
                samesite='Lax',
                max_age=3600  # 1 hour
            )
            return response
        else:
            return render_template_string(LOGIN_TEMPLATE, error="Invalid username or password")
    
    return render_template_string(LOGIN_TEMPLATE, error=None)

@app.route('/dashboard')
def dashboard():
    """Show dashboard for authenticated users."""
    auth_token = request.cookies.get('auth_token')
    
    if not auth_token or auth_token not in active_sessions:
        return redirect(url_for('login'))
    
    username = active_sessions[auth_token]
    
    return render_template_string(
        DASHBOARD_TEMPLATE,
        username=username,
        auth_token=auth_token,
        messages=conversation_history
    )

@app.route('/chat', methods=['POST'])
def chat():
    """Handle chat messages."""
    auth_token = request.cookies.get('auth_token')
    
    if not auth_token or auth_token not in active_sessions:
        return redirect(url_for('login'))
    
    user_message = request.form.get('message')
    if user_message:
        get_ai_response(user_message)
    
    return redirect(url_for('dashboard'))

@app.route('/logout')
def logout():
    """Handle logout."""
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and auth_token in active_sessions:
        del active_sessions[auth_token]
    
    # Clear conversation history on logout
    conversation_history.clear()
    
    response = make_response(redirect(url_for('login')))
    response.delete_cookie('auth_token')
    return response

if __name__ == '__main__':
    app.run(debug=True, port=5000)
