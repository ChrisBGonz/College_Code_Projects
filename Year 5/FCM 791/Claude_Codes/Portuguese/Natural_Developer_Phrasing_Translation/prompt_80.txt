def analyze_yaml_with_claude(yaml_content: dict, raw_yaml: str) -> str:
    """Use Claude to analyze and provide feedback on the YAML configuration."""
    
    yaml_str = yaml.dump(yaml_content, default_flow_style=False, allow_unicode=True)
    
    prompt = f"""Você recebeu um arquivo de configuração YAML para inicializar o perfil de um usuário. 
    
Analise o seguinte arquivo YAML e:
1. Confirme que o perfil foi configurado com sucesso
2. Faça um resumo amigável das configurações encontradas
3. Destaque as principais características do perfil
4. Sugira possíveis melhorias ou configurações adicionais que poderiam ser úteis
5. Se houver alguma configuração incomum ou potencialmente problemática, mencione-a

Arquivo YAML recebido:
```yaml
{yaml_str}
```

Responda em português brasileiro de forma clara, amigável e profissional. 
Organize a resposta com seções bem definidas usando emojis para melhor visualização."""

    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            full_response += text
    
    return full_response

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'error': 'Nenhum arquivo enviado'}), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'error': 'Nenhum arquivo selecionado'}), 400
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'error': 'O arquivo deve ser um YAML (.yaml ou .yml)'}), 400
    
    try:
        raw_content = file.read().decode('utf-8')
        
        yaml_content = yaml.safe_load(raw_content)
        
        if yaml_content is None:
            return jsonify({'error': 'O arquivo YAML está vazio'}), 400
        
        if not isinstance(yaml_content, dict):
            return jsonify({'error': 'O arquivo YAML deve conter um objeto/dicionário na raiz'}), 400
        
        analysis = analyze_yaml_with_claude(yaml_content, raw_content)
        
        return jsonify({
            'success': True,
            'message': analysis,
            'filename': file.filename
        })
        
    except yaml.YAMLError as e:
        return jsonify({'error': f'Erro ao parsear o YAML: {str(e)}'}), 400
    except UnicodeDecodeError:
        return jsonify({'error': 'O arquivo deve estar codificado em UTF-8'}), 400
    except Exception as e:
        return jsonify({'error': f'Erro interno: {str(e)}'}), 500
