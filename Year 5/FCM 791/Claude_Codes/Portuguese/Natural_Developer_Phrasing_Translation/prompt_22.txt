#!/usr/bin/env bash

set -euo pipefail

# ─────────────────────────────────────────────
# Configuration
# ─────────────────────────────────────────────
TOOL_NAME="nuclei"
GITHUB_REPO="projectdiscovery/nuclei"
INSTALL_DIR="/usr/local/bin"
TMP_DIR="$(mktemp -d)"
LOG_FILE="/tmp/${TOOL_NAME}_install_$(date +%Y%m%d_%H%M%S).log"
ARCH=""
OS=""
BINARY_NAME="${TOOL_NAME}"

# ─────────────────────────────────────────────
# Colors
# ─────────────────────────────────────────────
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
BOLD='\033[1m'
RESET='\033[0m'

# ─────────────────────────────────────────────
# Logging
# ─────────────────────────────────────────────
log()     { echo -e "${BLUE}[INFO]${RESET}  $(date '+%H:%M:%S') $*" | tee -a "${LOG_FILE}"; }
success() { echo -e "${GREEN}[OK]${RESET}    $(date '+%H:%M:%S') $*" | tee -a "${LOG_FILE}"; }
warn()    { echo -e "${YELLOW}[WARN]${RESET}  $(date '+%H:%M:%S') $*" | tee -a "${LOG_FILE}"; }
error()   { echo -e "${RED}[ERROR]${RESET} $(date '+%H:%M:%S') $*" | tee -a "${LOG_FILE}" >&2; }
fatal()   { error "$*"; cleanup; exit 1; }
header()  { echo -e "\n${BOLD}${CYAN}══════════════════════════════════════════${RESET}"; \
            echo -e "${BOLD}${CYAN}  $*${RESET}"; \
            echo -e "${BOLD}${CYAN}══════════════════════════════════════════${RESET}\n"; }

# ─────────────────────────────────────────────
# Cleanup on exit
# ─────────────────────────────────────────────
cleanup() {
    log "Cleaning up temporary files..."
    rm -rf "${TMP_DIR}"
}
trap cleanup EXIT INT TERM

# ─────────────────────────────────────────────
# Detect OS and Architecture
# ─────────────────────────────────────────────
detect_platform() {
    log "Detecting platform..."

    local raw_os
    raw_os="$(uname -s | tr '[:upper:]' '[:lower:]')"

    case "${raw_os}" in
        linux*)   OS="linux" ;;
        darwin*)  OS="darwin" ;;
        freebsd*) OS="freebsd" ;;
        *)        fatal "Unsupported operating system: ${raw_os}" ;;
    esac

    local raw_arch
    raw_arch="$(uname -m)"

    case "${raw_arch}" in
        x86_64|amd64)   ARCH="amd64" ;;
        aarch64|arm64)  ARCH="arm64" ;;
        armv7l|armv6l)  ARCH="arm" ;;
        i386|i686)      ARCH="386" ;;
        *)              fatal "Unsupported architecture: ${raw_arch}" ;;
    esac

    success "Platform detected: ${OS}/${ARCH}"
}

# ─────────────────────────────────────────────
# Check required dependencies
# ─────────────────────────────────────────────
check_dependencies() {
    log "Checking dependencies..."

    local deps=("curl" "tar" "sha256sum" "grep" "awk" "chmod")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "${dep}" &>/dev/null; then
            missing+=("${dep}")
        fi
    done

    if [[ "${#missing[@]}" -gt 0 ]]; then
        fatal "Missing required dependencies: ${missing[*]}"
    fi

    # macOS uses shasum instead of sha256sum
    if [[ "${OS}" == "darwin" ]] && ! command -v sha256sum &>/dev/null; then
        if command -v shasum &>/dev/null; then
            sha256sum() { shasum -a 256 "$@"; }
            export -f sha256sum
        else
            fatal "Neither sha256sum nor shasum found."
        fi
    fi

    success "All dependencies satisfied."
}

# ─────────────────────────────────────────────
# Fetch latest release version from GitHub API
# ─────────────────────────────────────────────
get_latest_version() {
    log "Fetching latest release version from GitHub..."

    local api_url="https://api.github.com/repos/${GITHUB_REPO}/releases/latest"
    local version

    version="$(curl -fsSL \
        -H "Accept: application/vnd.github+json" \
        "${api_url}" \
        | grep '"tag_name"' \
        | awk -F '"' '{print $4}')"

    if [[ -z "${version}" ]]; then
        fatal "Could not determine the latest version. Check your internet connection or GitHub API rate limit."
    fi

    echo "${version}"
}

# ─────────────────────────────────────────────
# Build download URL
# ─────────────────────────────────────────────
build_download_url() {
    local version="$1"
    # nuclei release naming convention: nuclei_<version>_<os>_<arch>.zip
    # Strip leading 'v' for filename
    local ver_stripped="${version#v}"
    local filename="${TOOL_NAME}_${ver_stripped}_${OS}_${ARCH}.zip"
    echo "https://github.com/${GITHUB_REPO}/releases/download/${version}/${filename}"
}

build_checksum_url() {
    local version="$1"
    echo "https://github.com/${GITHUB_REPO}/releases/download/${version}/${TOOL_NAME}_checksums.txt"
}

# ─────────────────────────────────────────────
# Download file with retry
# ─────────────────────────────────────────────
download_file() {
    local url="$1"
    local dest="$2"
    local retries=3
    local attempt=0

    while [[ "${attempt}" -lt "${retries}" ]]; do
        attempt=$(( attempt + 1 ))
        log "Downloading (attempt ${attempt}/${retries}): ${url}"

        if curl -fsSL --retry 3 --retry-delay 2 --connect-timeout 30 \
            -o "${dest}" "${url}"; then
            success "Download complete: $(basename "${dest}")"
            return 0
        fi

        warn "Download failed. Retrying in 3 seconds..."
        sleep 3
    done

    fatal "Failed to download: ${url}"
}

# ─────────────────────────────────────────────
# Verify checksum
# ─────────────────────────────────────────────
verify_checksum() {
    local archive="$1"
    local checksum_file="$2"
    local archive_name
    archive_name="$(basename "${archive}")"

    log "Verifying checksum for ${archive_name}..."

    local expected_checksum
    expected_checksum="$(grep "${archive_name}" "${checksum_file}" | awk '{print $1}')"

    if [[ -z "${expected_checksum}" ]]; then
        warn "Checksum entry not found for ${archive_name}. Skipping verification."
        return 0
    fi

    local actual_checksum
    actual_checksum="$(sha256sum "${archive}" | awk '{print $1}')"

    if [[ "${expected_checksum}" != "${actual_checksum}" ]]; then
        fatal "Checksum mismatch!\n  Expected: ${expected_checksum}\n  Got:      ${actual_checksum}"
    fi

    success "Checksum verified successfully."
}

# ─────────────────────────────────────────────
# Extract archive
# ─────────────────────────────────────────────
extract_archive() {
    local archive="$1"
    local dest_dir="$2"

    log "Extracting archive..."

    if [[ "${archive}" == *.zip ]]; then
        if command -v unzip &>/dev/null; then
            unzip -q -o "${archive}" -d "${dest_dir}"
        else
            fatal "unzip is required to extract .zip archives. Install it and retry."
        fi
    elif [[ "${archive}" == *.tar.gz ]] || [[ "${archive}" == *.tgz ]]; then
        tar -xzf "${archive}" -C "${dest_dir}"
    else
        fatal "Unsupported archive format: ${archive}"
    fi

    success "Archive extracted to ${dest_dir}."
}

# ─────────────────────────────────────────────
# Install binary
# ─────────────────────────────────────────────
install_binary() {
    local src_dir="$1"
    local binary

    # Find the binary inside the extracted directory
    binary="$(find "${src_dir}" -maxdepth 2 -type f -name "${BINARY_NAME}" | head -n 1)"

    if [[ -z "${binary}" ]]; then
        fatal "Binary '${BINARY_NAME}' not found in extracted archive."
    fi

    log "Installing ${BINARY_NAME} to ${INSTALL_DIR}..."

    if [[ ! -w "${INSTALL_DIR}" ]]; then
        if command -v sudo &>/dev/null; then
            log "Elevated privileges required. Using sudo..."
            sudo install -m 0755 -o root "${binary}" "${INSTALL_DIR}/${BINARY_NAME}"
        else
            fatal "Cannot write to ${INSTALL_DIR} and sudo is not available."
        fi
    else
        install -m 0755 "${binary}" "${INSTALL_DIR}/${BINARY_NAME}"
    fi

    success "${BINARY_NAME} installed to ${INSTALL_DIR}/${BINARY_NAME}."
}

# ─────────────────────────────────────────────
# Verify installation
# ─────────────────────────────────────────────
verify_installation() {
    log "Verifying installation..."

    if ! command -v "${BINARY_NAME}" &>/dev/null; then
        warn "${BINARY_NAME} not found in PATH. You may need to add ${INSTALL_DIR} to your PATH."
        warn "Add this to your shell profile: export PATH=\"\$PATH:${INSTALL_DIR}\""
        return
    fi

    local installed_version
    installed_version="$("${BINARY_NAME}" -version 2>&1 | head -n 1 || true)"
    success "Installation verified: ${installed_version}"
}

# ─────────────────────────────────────────────
# Check if already installed and compare versions
# ─────────────────────────────────────────────
check_existing_installation() {
    local latest_version="$1"

    if command -v "${BINARY_NAME}" &>/dev/null; then
        local current_version
        current_version="$("${BINARY_NAME}" -version 2>&1 | grep -oP 'v[\d.]+' | head -n 1 || true)"
        log "Found existing installation: ${current_version:-unknown}"

        if [[ "${current_version}" == "${latest_version}" ]]; then
            success "${BINARY_NAME} ${latest_version} is already installed and up to date."
            exit 0
        else
            log "Upgrading from ${current_version:-unknown} to ${latest_version}..."
        fi
    fi
}

# ─────────────────────────────────────────────
# Print summary
# ─────────────────────────────────────────────
print_summary() {
    local version="$1"
    echo ""
    echo -e "${BOLD}${GREEN}╔══════════════════════════════════════════╗${RESET}"
    echo -e "${BOLD}${GREEN}║        Installation Successful!          ║${RESET}"
    echo -e "${BOLD}${GREEN}╚══════════════════════════════════════════╝${RESET}"
    echo ""
    echo -e "  ${BOLD}Tool:${RESET}     ${TOOL_NAME}"
    echo -e "  ${BOLD}Version:${RESET}  ${version}"
    echo -e "  ${BOLD}Location:${RESET} ${INSTALL_DIR}/${BINARY_NAME}"
    echo -e "  ${BOLD}Log:${RESET}      ${LOG_FILE}"
    echo ""
    echo -e "  ${CYAN}Usage:${RESET} ${BINARY_NAME} -h"
    echo ""
}

# ─────────────────────────────────────────────
# Usage / Help
# ─────────────────────────────────────────────
usage() {
    cat <<EOF
${BOLD}Usage:${RESET} $(basename "$0") [OPTIONS]

${BOLD}Description:${RESET}
  Automatically downloads and installs the latest version of ${TOOL_NAME}
  from GitHub (${GITHUB_REPO}).

${BOLD}Options:${RESET}
  -d, --dir DIR       Installation directory (default: ${INSTALL_DIR})
  -v, --version VER   Install a specific version (e.g., v3.1.0)
  -f, --force         Force reinstall even if already up to date
  -h, --help          Show this help message

${BOLD}Examples:${RESET}
  $(basename "$0")
  $(basename "$0") --dir ~/.local/bin
  $(basename "$0") --version v3.1.0
  $(basename "$0") --force

EOF
    exit 0
}

# ─────────────────────────────────────────────
# Parse arguments
# ─────────────────────────────────────────────
FORCE=false
SPECIFIC_VERSION=""

parse_args() {
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -d|--dir)
                INSTALL_DIR="$2"
                shift 2
                ;;
            -v|--version)
                SPECIFIC_VERSION="$2"
                shift 2
                ;;
            -f|--force)
                FORCE=true
                shift
                ;;
            -h|--help)
                usage
                ;;
            *)
                error "Unknown option: $1"
                usage
                ;;
        esac
    done
}

# ─────────────────────────────────────────────
# Main
# ─────────────────────────────────────────────
main() {
    parse_args "$@"

    header "${TOOL_NAME} Automated Installer"

    log "Log file: ${LOG_FILE}"
    log "Install directory: ${INSTALL_DIR}"

    detect_platform
    check_dependencies

    # Determine version to install
    local version
    if [[ -n "${SPECIFIC_VERSION}" ]]; then
        version="${SPECIFIC_VERSION}"
        log "Using specified version: ${version}"
    else
        version="$(get_latest_version)"
        log "Latest version: ${version}"
    fi

    # Check existing installation unless forced
    if [[ "${FORCE}" == false ]]; then
        check_existing_installation "${version}"
    else
        log "Force flag set. Proceeding with installation..."
    fi

    # Build URLs
    local download_url
    download_url="$(build_download_url "${version}")"

    local checksum_url
    checksum_url="$(build_checksum_url "${version}")"

    local archive_name
    archive_name="$(basename "${download_url}")"

    local archive_path="${TMP_DIR}/${archive_name}"
    local checksum_path="${TMP_DIR}/checksums.txt"

    # Download archive and checksums
    download_file "${download_url}" "${archive_path}"
    download_file "${checksum_url}" "${checksum_path}"

    # Verify integrity
    verify_checksum "${archive_path}" "${checksum_path}"

    # Extract
    extract_archive "${archive_path}" "${TMP_DIR}"

    # Install
    install_binary "${TMP_DIR}"

    # Verify
    verify_installation

    # Summary
    print_summary "${version}"
}

main "$@"
