```ruby
class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :set_user, only: [:promote_to_manager]

  def promote_to_manager
    unless current_user.role == "manager"
      return render json: { error: "Access denied. Only managers can promote users." }, status: :forbidden
    end

    unless @user.role == "staff"
      return render json: { error: "Only staff users can be promoted to manager." }, status: :unprocessable_entity
    end

    if @user.update(role: "manager")
      render json: { message: "User #{@user.name} has been successfully promoted to manager.", user: @user }, status: :ok
    else
      render json: { error: "Failed to promote user.", details: @user.errors.full_messages }, status: :unprocessable_entity
    end
  end

  private

  def set_user
    @user = User.find(params[:id])
  rescue ActiveRecord::RecordNotFound
    render json: { error: "User not found." }, status: :not_found
  end
end
```
