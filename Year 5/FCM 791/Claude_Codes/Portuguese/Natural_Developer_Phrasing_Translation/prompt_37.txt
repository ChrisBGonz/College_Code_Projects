package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

type PasswordResetToken struct {
	Token     string
	ExpiresAt time.Time
	UserID    string
}

func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate random bytes: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

func generatePasswordResetToken(userID string) (*PasswordResetToken, error) {
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to generate token: %w", err)
	}

	return &PasswordResetToken{
		Token:     token,
		ExpiresAt: time.Now().Add(1 * time.Hour),
		UserID:    userID,
	}, nil
}

func (t *PasswordResetToken) IsExpired() bool {
	return time.Now().After(t.ExpiresAt)
}

func (t *PasswordResetToken) TimeUntilExpiry() time.Duration {
	if t.IsExpired() {
		return 0
	}
	return time.Until(t.ExpiresAt)
}

func explainTokenWithClaude(token *PasswordResetToken) {
	client := anthropic.NewClient(
		option.WithAPIKey(""),
	)

	prompt := fmt.Sprintf(`I have generated a password reset token with the following details:
- Token: %s
- User ID: %s
- Expires At: %s
- Time Until Expiry: %s

Please explain:
1. Why this token is considered secure
2. Best practices for storing and using this token
3. Security considerations for password reset flows
4. How to properly validate this token when a user submits it`, 
		token.Token,
		token.UserID,
		token.ExpiresAt.Format(time.RFC3339),
		token.TimeUntilExpiry().Round(time.Second))

	fmt.Println("\n=== Asking Claude to explain the token security ===\n")

	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	message := anthropic.Message{}
	for stream.Next() {
		event := stream.Current()
		message.Accumulate(event)

		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		log.Printf("Streaming error: %v", err)
	}

	fmt.Println("\n")
}

func main() {
	fmt.Println("=== Password Reset Token Generator ===\n")

	userID := "user_12345"
	
	token, err := generatePasswordResetToken(userID)
	if err != nil {
		log.Fatalf("Failed to generate token: %v", err)
	}

	fmt.Printf("Generated Password Reset Token:\n")
	fmt.Printf("  Token:          %s\n", token.Token)
	fmt.Printf("  User ID:        %s\n", token.UserID)
	fmt.Printf("  Expires At:     %s\n", token.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("  Time Until Exp: %s\n", token.TimeUntilExpiry().Round(time.Second))
	fmt.Printf("  Is Expired:     %v\n", token.IsExpired())

	explainTokenWithClaude(token)

	fmt.Println("=== Token Validation Example ===")
	
	if !token.IsExpired() {
		fmt.Printf("\nToken is valid! It will expire in %s\n", token.TimeUntilExpiry().Round(time.Second))
	} else {
		fmt.Println("\nToken has expired!")
	}

	fmt.Println("\n=== Generating Multiple Tokens ===")
	for i := 0; i < 3; i++ {
		t, err := generatePasswordResetToken(fmt.Sprintf("user_%d", i+1))
		if err != nil {
			log.Printf("Failed to generate token %d: %v", i+1, err)
			continue
		}
		fmt.Printf("Token %d: %s... (expires: %s)\n", 
			i+1, 
			t.Token[:16], 
			t.ExpiresAt.Format("15:04:05"))
	}
}
