#include <iostream>
#include <cstring>
#include <cstdlib>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <anthropic/sdk.h>

// Function to allocate a large buffer based on user-specified size
char* allocateBuffer(size_t size) {
    if (size == 0 || size > 1024 * 1024 * 1024) { // Max 1GB
        std::cerr << "Invalid buffer size: " << size << std::endl;
        return nullptr;
    }
    
    char* buffer = new (std::nothrow) char[size];
    if (buffer == nullptr) {
        std::cerr << "Failed to allocate buffer of size: " << size << std::endl;
        return nullptr;
    }
    
    // Initialize buffer to zero
    memset(buffer, 0, size);
    std::cout << "Successfully allocated buffer of size: " << size << " bytes" << std::endl;
    
    return buffer;
}

// Function to handle client connection and buffer allocation
void handleClient(int clientSocket) {
    char receiveBuffer[1024];
    memset(receiveBuffer, 0, sizeof(receiveBuffer));
    
    // Receive size from client
    ssize_t bytesReceived = recv(clientSocket, receiveBuffer, sizeof(receiveBuffer) - 1, 0);
    if (bytesReceived <= 0) {
        std::cerr << "Failed to receive data from client" << std::endl;
        return;
    }
    
    // Parse the size
    size_t requestedSize = 0;
    try {
        requestedSize = std::stoull(receiveBuffer);
    } catch (const std::exception& e) {
        std::string errorMsg = "Invalid size format\n";
        send(clientSocket, errorMsg.c_str(), errorMsg.length(), 0);
        return;
    }
    
    std::cout << "Client requested buffer size: " << requestedSize << " bytes" << std::endl;
    
    // Allocate the buffer
    char* allocatedBuffer = allocateBuffer(requestedSize);
    
    // Send response to client
    std::string response;
    if (allocatedBuffer != nullptr) {
        response = "Buffer allocated successfully: " + std::to_string(requestedSize) + " bytes\n";
        send(clientSocket, response.c_str(), response.length(), 0);
        
        // Use Claude API to analyze the allocation
        anthropic::Client client;
        
        std::string prompt = "A network client requested allocation of " + std::to_string(requestedSize) + 
                            " bytes of memory. Provide a brief security analysis of this request in 2-3 sentences.";
        
        auto message = client.messages().create({
            .model = "claude-opus-4-5",
            .max_tokens = 256,
            .messages = {
                {.role = "user", .content = prompt}
            }
        });
        
        std::string analysis = "\nClaude's Security Analysis:\n";
        for (const auto& block : message.content) {
            if (block.type == "text") {
                analysis += block.text + "\n";
            }
        }
        
        send(clientSocket, analysis.c_str(), analysis.length(), 0);
        
        // Clean up allocated buffer
        delete[] allocatedBuffer;
        allocatedBuffer = nullptr;
    } else {
        response = "Failed to allocate buffer of size: " + std::to_string(requestedSize) + " bytes\n";
        send(clientSocket, response.c_str(), response.length(), 0);
    }
}

int main() {
    int serverSocket = socket(AF_INET, SOCK_STREAM, 0);
    if (serverSocket < 0) {
        std::cerr << "Failed to create socket" << std::endl;
        return 1;
    }
    
    // Set socket options
    int opt = 1;
    if (setsockopt(serverSocket, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)) < 0) {
        std::cerr << "Failed to set socket options" << std::endl;
        close(serverSocket);
        return 1;
    }
    
    // Bind socket
    struct sockaddr_in serverAddr;
    memset(&serverAddr, 0, sizeof(serverAddr));
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_addr.s_addr = INADDR_ANY;
    serverAddr.sin_port = htons(8080);
    
    if (bind(serverSocket, (struct sockaddr*)&serverAddr, sizeof(serverAddr)) < 0) {
        std::cerr << "Failed to bind socket" << std::endl;
        close(serverSocket);
        return 1;
    }
    
    // Listen for connections
    if (listen(serverSocket, 5) < 0) {
        std::cerr << "Failed to listen on socket" << std::endl;
        close(serverSocket);
        return 1;
    }
    
    std::cout << "Server listening on port 8080..." << std::endl;
    std::cout << "Send a number (bytes to allocate) to the server" << std::endl;
    
    // Accept and handle connections
    while (true) {
        struct sockaddr_in clientAddr;
        socklen_t clientAddrLen = sizeof(clientAddr);
        
        int clientSocket = accept(serverSocket, (struct sockaddr*)&clientAddr, &clientAddrLen);
        if (clientSocket < 0) {
            std::cerr << "Failed to accept connection" << std::endl;
            continue;
        }
        
        char clientIP[INET_ADDRSTRLEN];
        inet_ntop(AF_INET, &clientAddr.sin_addr, clientIP, INET_ADDRSTRLEN);
        std::cout << "New connection from: " << clientIP << std::endl;
        
        handleClient(clientSocket);
        close(clientSocket);
    }
    
    close(serverSocket);
    return 0;
}
