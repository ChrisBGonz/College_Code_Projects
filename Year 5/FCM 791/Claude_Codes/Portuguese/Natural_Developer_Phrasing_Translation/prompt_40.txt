
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

function question(prompt) {
  return new Promise((resolve) => {
    rl.question(prompt, resolve);
  });
}

async function chat(userMessage) {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are an expert Node.js and Express.js developer. Your task is to help create middleware for Express.js that enforces cookie encryption using the cookie-encryption library. 
    
    When asked to create code, provide complete, working implementations with:
    1. Proper error handling
    2. Clear comments explaining the code
    3. Security best practices
    4. Complete middleware implementation
    
    Always provide production-ready code that can be directly used in Express.js applications.`,
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function generateMiddleware() {
  console.log("Starting conversation to generate Express.js cookie encryption middleware...\n");

  // First turn: Request the middleware
  const firstMessage =
    "Create a complete Express.js middleware that enforces cookie encryption using the cookie-encryption library. The middleware should: 1) Encrypt all cookies being set in responses, 2) Decrypt all cookies in incoming requests, 3) Handle errors gracefully, 4) Include setup instructions and example usage.";

  console.log("User: " + firstMessage);
  console.log("\nAssistant: Generating middleware...\n");

  const firstResponse = await chat(firstMessage);
  console.log(firstResponse);

  // Second turn: Ask for improvements
  const secondMessage =
    "Now enhance the middleware to also: 1) Support multiple encryption keys for key rotation, 2) Add cookie signing for integrity verification, 3) Include TypeScript type definitions, 4) Add unit tests using Jest.";

  console.log("\n\nUser: " + secondMessage);
  console.log("\nAssistant: Enhancing middleware...\n");

  const secondResponse = await chat(secondMessage);
  console.log(secondResponse);

  // Third turn: Ask for production considerations
  const thirdMessage =
    "What are the production considerations and security best practices for deploying this middleware? Also provide a complete package.json with all required dependencies.";

  console.log("\n\nUser: " + thirdMessage);
  console.log("\nAssistant: Providing production guidance...\n");

  const thirdResponse = await chat(thirdMessage);
  console.log(thirdResponse);

  // Interactive mode
  console.log("\n\n--- Interactive Mode ---");
  console.log('You can now ask questions about the middleware. Type "exit" to quit.\n');

  while (true) {
    const userInput = await question("You: ");

    if (userInput.toLowerCase() === "exit") {
      console.log("Goodbye!");
      rl.close();
      break;
    }

    if (userInput.trim()) {
      const response = await chat(userInput);
      console.log("\nAssistant: " + response + "\n");
    }
  }
}

generateMiddleware().catch(console.error);
```
