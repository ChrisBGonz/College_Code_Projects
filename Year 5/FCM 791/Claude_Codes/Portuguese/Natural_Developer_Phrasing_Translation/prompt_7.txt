import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user_001",
      name: "João Silva",
      email: "joao@example.com",
      password: "oldpassword123",
    },
    {
      id: "user_002",
      name: "Maria Santos",
      email: "maria@example.com",
      password: "mypassword456",
    },
    {
      id: "user_003",
      name: "Pedro Costa",
      email: "pedro@example.com",
      password: "securepass789",
    },
  ],
};

// Tool definitions for Claude
const tools = [
  {
    name: "get_user_by_id",
    description:
      "Retrieves user information from the database by user ID. Returns user details if found.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, at least one number, at least one letter)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_db",
    description:
      "Updates the user password in the database. Should only be called after validating the user exists and the password meets requirements.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        new_password: {
          type: "string",
          description: "The new password to set for the user",
        },
      },
      required: ["user_id", "new_password"],
    },
  },
  {
    name: "log_password_reset",
    description:
      "Logs the password reset event for security auditing purposes",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        success: {
          type: "boolean",
          description: "Whether the password reset was successful",
        },
        reason: {
          type: "string",
          description: "The reason for success or failure",
        },
      },
      required: ["user_id", "success", "reason"],
    },
  },
];

// Tool implementations
function getUserById(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      found: true,
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
      },
    };
  }
  return { found: false, message: `User with ID ${userId} not found` };
}

function validatePassword(password) {
  const hasMinLength = password.length >= 8;
  const hasNumber = /\d/.test(password);
  const hasLetter = /[a-zA-Z]/.test(password);

  const isValid = hasMinLength && hasNumber && hasLetter;

  return {
    valid: isValid,
    requirements: {
      minLength: hasMinLength,
      hasNumber: hasNumber,
      hasLetter: hasLetter,
    },
    message: isValid
      ? "Password meets all requirements"
      : "Password does not meet requirements: " +
        [
          !hasMinLength ? "minimum 8 characters required" : null,
          !hasNumber ? "at least one number required" : null,
          !hasLetter ? "at least one letter required" : null,
        ]
          .filter(Boolean)
          .join(", "),
  };
}

function updatePasswordInDb(userId, newPassword) {
  const userIndex = database.users.findIndex((u) => u.id === userId);

  if (userIndex === -1) {
    return {
      success: false,
      message: `User with ID ${userId} not found in database`,
    };
  }

  // Simulate password hashing (in real scenario, use bcrypt or similar)
  const hashedPassword = `hashed_${newPassword}_${Date.now()}`;
  database.users[userIndex].password = hashedPassword;

  return {
    success: true,
    message: `Password successfully updated for user ${userId}`,
    updatedAt: new Date().toISOString(),
  };
}

function logPasswordReset(userId, success, reason) {
  const logEntry = {
    timestamp: new Date().toISOString(),
    userId: userId,
    action: "PASSWORD_RESET",
    success: success,
    reason: reason,
  };

  console.log("\n📋 Security Log Entry:", JSON.stringify(logEntry, null, 2));

  return {
    logged: true,
    logId: `log_${Date.now()}`,
    message: "Event logged successfully",
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "get_user_by_id":
      return getUserById(toolInput.user_id);
    case "validate_password":
      return validatePassword(toolInput.password);
    case "update_password_in_db":
      return updatePasswordInDb(toolInput.user_id, toolInput.new_password);
    case "log_password_reset":
      return logPasswordReset(
        toolInput.user_id,
        toolInput.success,
        toolInput.reason
      );
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log("\n🔐 Starting Password Reset Flow");
  console.log(`User ID: ${userId}`);
  console.log(`New Password: ${"*".repeat(newPassword.length)}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user with ID "${userId}". The new password should be set to "${newPassword}". 
      
      Follow these steps:
      1. First, verify the user exists in the database
      2. Validate that the new password meets security requirements
      3. If both checks pass, update the password in the database
      4. Log the password reset event (success or failure)
      5. Return a clear message about the outcome`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  console.log("\n🤖 Claude is processing the request...");

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\n🔧 Calling tool: ${toolUse.name}`);
      console.log("Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n✅ Final Response from Claude:");
  console.log("=".repeat(50));
  console.log(finalResponse);

  return finalResponse;
}

// Test the function with different scenarios
async function runTests() {
  console.log("🚀 Password Reset Flow - Test Suite");
  console.log("=".repeat(50));

  // Test 1: Valid user and valid password
  console.log("\n📌 Test 1: Valid user with valid password");
  await resetPassword("user_001", "NewSecure123");

  // Test 2: Valid user but weak password
  console.log("\n\n📌 Test 2: Valid user with weak password");
  await resetPassword("user_002", "weak");

  // Test 3: Non-existent user
  console.log("\n\n📌 Test 3: Non-existent user");
  await resetPassword("user_999", "ValidPass123");
}

// Run the tests
runTests().catch(console.error);
