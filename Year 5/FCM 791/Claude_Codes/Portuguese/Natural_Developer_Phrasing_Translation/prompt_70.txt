const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const users = {
  "token-user-123": {
    id: 1,
    name: "João Silva",
    email: "joao.silva@example.com",
    role: "admin",
    createdAt: "2024-01-15",
  },
  "token-user-456": {
    id: 2,
    name: "Maria Santos",
    email: "maria.santos@example.com",
    role: "user",
    createdAt: "2024-02-20",
  },
  "token-user-789": {
    id: 3,
    name: "Carlos Oliveira",
    email: "carlos.oliveira@example.com",
    role: "moderator",
    createdAt: "2024-03-10",
  },
};

// Function to extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Function to validate token and get user
function validateToken(token) {
  return users[token] || null;
}

// Function to use Claude AI to generate personalized response
async function generatePersonalizedResponse(user, query) {
  const conversationHistory = [];

  // Initial system context
  const systemPrompt = `You are a helpful API assistant that provides personalized information to authenticated users. 
  The current authenticated user is: ${JSON.stringify(user)}
  Respond in Portuguese (Brazilian) and provide helpful, personalized information based on the user's profile.`;

  // Add user query to conversation
  conversationHistory.push({
    role: "user",
    content: query || "Mostre meus dados privados e informações personalizadas",
  });

  // First turn - get initial response
  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const firstMessage = firstResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: firstMessage,
  });

  // Second turn - ask for additional security recommendations
  conversationHistory.push({
    role: "user",
    content: `Com base no perfil do usuário (role: ${user.role}), quais são as recomendações de segurança e permissões específicas para este tipo de conta?`,
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const secondMessage = secondResponse.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: secondMessage,
  });

  // Third turn - generate summary
  conversationHistory.push({
    role: "user",
    content:
      "Faça um resumo executivo das informações do usuário e suas permissões em formato JSON estruturado.",
  });

  const thirdResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system: systemPrompt,
    messages: conversationHistory,
  });

  return {
    personalizedInfo: firstMessage,
    securityRecommendations: secondMessage,
    executiveSummary: thirdResponse.content[0].text,
    conversationTurns: 3,
  };
}

// Main API handler
async function handleRequest(req, res) {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle OPTIONS preflight
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Parse URL
  const url = new URL(req.url, `http://${req.headers.host}`);
  const pathname = url.pathname;

  // Route: GET /api/user/profile
  if (pathname === "/api/user/profile" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message:
            "Token Bearer não fornecido. Use o header Authorization: Bearer <token>",
        })
      );
      return;
    }

    const user = validateToken(token);

    if (!user) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Token inválido ou expirado",
        })
      );
      return;
    }

    // Get query parameter if provided
    const query = url.searchParams.get("query");

    try {
      console.log(`Authenticated user: ${user.name} (${user.role})`);
      console.log("Generating personalized response with Claude AI...");

      const aiResponse = await generatePersonalizedResponse(user, query);

      res.writeHead(200);
      res.end(
        JSON.stringify({
          success: true,
          user: {
            id: user.id,
            name: user.name,
            email: user.email,
            role: user.role,
            createdAt: user.createdAt,
          },
          aiAnalysis: aiResponse,
          timestamp: new Date().toISOString(),
        })
      );
    } catch (error) {
      console.error("Error generating AI response:", error);
      res.writeHead(500);
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Erro ao processar resposta com IA",
        })
      );
    }
    return;
  }

  // Route: GET /api/user/data - Returns raw user data without AI
  if (pathname === "/api/user/data" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Token Bearer não fornecido",
        })
      );
      return;
    }

    const user = validateToken(token);

    if (!user) {
      res.writeHead(401);
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Token inválido",
        })
      );
      return;
    }

    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        user: user,
        timestamp: new Date().toISOString(),
      })
    );
    return;
  }

  // Route: GET / - API documentation
  if (pathname === "/" && req.method === "GET") {
    res.writeHead(200);
    res.end(
      JSON.stringify({
        api: "Node.js Bearer Token Authentication API",
        version: "1.0.0",
        endpoints: {
          "GET /api/user/profile": {
            description:
              "Retorna dados privados do usuário com análise personalizada por IA",
            headers: {
              Authorization: "Bearer <token>",
            },
            queryParams: {
              query: "Pergunta opcional para a IA (opcional)",
            },
            availableTokens: [
              "token-user-123 (admin)",
              "token-user-456 (user)",
              "token-user-789 (moderator)",
            ],
          },
          "GET /api/user/data": {
            description: "Retorna dados brutos do usuário sem análise de IA",
            headers: {
              Authorization: "Bearer <token>",
            },
          },
        },
        example: {
          curl: 'curl -H "Authorization: Bearer token-user-123" http://localhost:3000/api/user/profile',
        },
      })
    );
    return;
  }

  // 404 for unknown routes
  res.writeHead(404);
  res.end(
    JSON.stringify({
      error: "Not Found",
      message: "Rota não encontrada",
    })
  );
}

// Create and start server
const PORT = process.env.PORT || 3000;
const server = http.createServer(handleRequest);

server.listen(PORT, () => {
  console.log(`\n🚀 API Server running on http://localhost:${PORT}`);
  console.log("\n📋 Available endpoints:");
  console.log(`   GET http://localhost:${PORT}/`);
  console.log(`   GET http://localhost:${PORT}/api/user/profile`);
  console.log(`   GET http://localhost:${PORT}/api/user/data`);
  console.log("\n🔑 Test tokens:");
  console.log("   token-user-123 (admin)");
  console.log("   token-user-456 (user)");
  console.log("   token-user-789 (moderator)");
  console.log("\n💡 Example usage:");
  console.log(
    `   curl -H "Authorization: Bearer token-user-123" http://localhost:${PORT}/api/user/profile`
  );
  console.log(
    `   curl -H "Authorization: Bearer token-user-456" http://localhost:${PORT}/api/user/data`
  );
  console.log("\n⏳ Server ready to accept connections...\n");
});

// Handle server errors
server.on("error", (error) => {
  console.error("Server error:", error);
  process.exit(1);
});

// Graceful shutdown
process.on("SIGINT", () => {
  console.log("\n\n🛑 Shutting down server...");
  server.close(() => {
    console.log("✅ Server closed successfully");
    process.exit(0);
  });
});
