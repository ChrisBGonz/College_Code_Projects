import anthropic
import subprocess
import sys
import time

client = anthropic.Anthropic()

def run_command(command: str, capture_output: bool = True) -> tuple[int, str, str]:
    """Execute a shell command and return exit code, stdout, and stderr."""
    result = subprocess.run(
        command,
        shell=True,
        capture_output=capture_output,
        text=True
    )
    return result.returncode, result.stdout, result.stderr

def stream_ci_cd_pipeline(image_name: str, container_name: str, port_mapping: str):
    """Stream the CI/CD pipeline execution with Claude's guidance."""
    
    print("🚀 Starting CI/CD Pipeline with Claude AI assistance...")
    print("=" * 60)
    
    pipeline_context = f"""
    You are a DevOps expert helping to execute a CI/CD pipeline. 
    The pipeline will:
    1. Pull Docker image: {image_name}
    2. Stop and remove existing container: {container_name}
    3. Deploy new container with port mapping: {port_mapping}
    4. Verify the deployment
    
    Provide step-by-step guidance and analysis as each step executes.
    Be concise but informative. Use emojis for better readability.
    """
    
    steps = [
        {
            "name": "Pre-deployment Check",
            "command": "docker --version && docker info --format '{{.ServerVersion}}'",
            "description": "Checking Docker installation and daemon status"
        },
        {
            "name": "Pull Docker Image",
            "command": f"docker pull {image_name}",
            "description": f"Pulling latest image: {image_name}"
        },
        {
            "name": "Stop Existing Container",
            "command": f"docker stop {container_name} 2>/dev/null || echo 'No container to stop'",
            "description": f"Stopping existing container: {container_name}"
        },
        {
            "name": "Remove Existing Container",
            "command": f"docker rm {container_name} 2>/dev/null || echo 'No container to remove'",
            "description": f"Removing existing container: {container_name}"
        },
        {
            "name": "Deploy New Container",
            "command": f"docker run -d --name {container_name} -p {port_mapping} --restart unless-stopped {image_name}",
            "description": f"Deploying new container with port mapping {port_mapping}"
        },
        {
            "name": "Verify Deployment",
            "command": f"docker ps --filter name={container_name} --format 'table {{{{.Names}}}}\\t{{{{.Status}}}}\\t{{{{.Ports}}}}'",
            "description": "Verifying container is running"
        },
        {
            "name": "Container Health Check",
            "command": f"docker inspect {container_name} --format '{{{{.State.Health.Status}}}}' 2>/dev/null || docker inspect {container_name} --format '{{{{.State.Status}}}}'",
            "description": "Checking container health status"
        }
    ]
    
    pipeline_results = []
    
    for i, step in enumerate(steps, 1):
        print(f"\n📋 Step {i}/{len(steps)}: {step['name']}")
        print(f"   {step['description']}")
        print(f"   Command: {step['command']}")
        print("-" * 40)
        
        exit_code, stdout, stderr = run_command(step['command'])
        
        output = stdout if stdout else stderr
        status = "✅ SUCCESS" if exit_code == 0 else "❌ FAILED"
        
        print(f"   Status: {status}")
        if output.strip():
            print(f"   Output: {output.strip()}")
        
        pipeline_results.append({
            "step": step['name'],
            "command": step['command'],
            "exit_code": exit_code,
            "output": output.strip(),
            "status": "success" if exit_code == 0 else "failed"
        })
        
        if exit_code != 0 and step['name'] == "Deploy New Container":
            print("\n⚠️  Critical step failed! Requesting Claude's analysis...")
            break
        
        time.sleep(0.5)
    
    print("\n" + "=" * 60)
    print("🤖 Claude AI Analysis and Recommendations:")
    print("=" * 60)
    
    results_summary = "\n".join([
        f"- {r['step']}: {r['status'].upper()} (exit code: {r['exit_code']})\n  Output: {r['output'][:200] if r['output'] else 'No output'}"
        for r in pipeline_results
    ])
    
    analysis_prompt = f"""
    {pipeline_context}
    
    Pipeline execution results:
    {results_summary}
    
    Please provide:
    1. Overall pipeline status assessment
    2. Analysis of any issues encountered
    3. Recommendations for improvement
    4. Next steps for the deployment
    5. Any security or performance considerations
    
    Keep the response concise and actionable.
    """
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": analysis_prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n\n" + "=" * 60)
    
    all_success = all(r['status'] == 'success' for r in pipeline_results)
    
    if all_success:
        print("🎉 Pipeline completed successfully!")
        print(f"✅ Container '{container_name}' is running with image '{image_name}'")
        print(f"🌐 Application accessible on port mapping: {port_mapping}")
    else:
        failed_steps = [r['step'] for r in pipeline_results if r['status'] == 'failed']
        print(f"⚠️  Pipeline completed with failures in: {', '.join(failed_steps)}")
        print("Please review the Claude AI analysis above for remediation steps.")
    
    return all_success

def main():
    print("🐳 Docker CI/CD Pipeline with Claude AI")
    print("=" * 60)
    
    image_name = "nginx:latest"
    container_name = "production-app"
    port_mapping = "8080:80"
    
    if len(sys.argv) > 1:
        image_name = sys.argv[1]
    if len(sys.argv) > 2:
        container_name = sys.argv[2]
    if len(sys.argv) > 3:
        port_mapping = sys.argv[3]
    
    print(f"Configuration:")
    print(f"  Image: {image_name}")
    print(f"  Container Name: {container_name}")
    print(f"  Port Mapping: {port_mapping}")
    print()
    
    success = stream_ci_cd_pipeline(image_name, container_name, port_mapping)
    
    sys.exit(0 if success else 1)

if __name__ == "__main__":
    main()
