import anthropic
import base64
import tempfile
import os
from lxml import etree


def parse_xml_and_extract_title(xml_content: str) -> str:
    """Parse XML content and extract the title tag content."""
    try:
        root = etree.fromstring(xml_content.encode('utf-8'))
        
        title_element = root.find('.//title')
        
        if title_element is None:
            title_element = root.find('.//{*}title')
        
        if title_element is not None:
            return title_element.text or "Title tag found but empty"
        else:
            return "No <title> tag found in the XML"
    except etree.XMLSyntaxError as e:
        return f"XML parsing error: {str(e)}"


def process_xml_with_claude(xml_content: str) -> dict:
    """Use Claude to analyze XML and extract title using tool use."""
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "extract_xml_title",
            "description": "Extract the content of the <title> tag from XML content",
            "input_schema": {
                "type": "object",
                "properties": {
                    "xml_content": {
                        "type": "string",
                        "description": "The XML content to parse"
                    }
                },
                "required": ["xml_content"]
            }
        }
    ]
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=[
            {
                "role": "user",
                "content": f"Please extract the title from this XML content using the extract_xml_title tool:\n\n{xml_content}"
            }
        ]
    )
    
    result = {
        "claude_response": message.content,
        "title": None,
        "analysis": None
    }
    
    for block in message.content:
        if block.type == "tool_use" and block.name == "extract_xml_title":
            xml_to_parse = block.input.get("xml_content", xml_content)
            extracted_title = parse_xml_and_extract_title(xml_to_parse)
            result["title"] = extracted_title
            
            tool_result_message = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=1024,
                tools=tools,
                messages=[
                    {
                        "role": "user",
                        "content": f"Please extract the title from this XML content using the extract_xml_title tool:\n\n{xml_content}"
                    },
                    {
                        "role": "assistant",
                        "content": message.content
                    },
                    {
                        "role": "user",
                        "content": [
                            {
                                "type": "tool_result",
                                "tool_use_id": block.id,
                                "content": extracted_title
                            }
                        ]
                    }
                ]
            )
            
            for final_block in tool_result_message.content:
                if hasattr(final_block, 'text'):
                    result["analysis"] = final_block.text
                    break
    
    return result


def simulate_file_upload(xml_content: str) -> str:
    """Simulate file upload by saving to temp file and reading back."""
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as f:
        f.write(xml_content)
        temp_path = f.name
    
    try:
        with open(temp_path, 'r', encoding='utf-8') as f:
            content = f.read()
        return content
    finally:
        os.unlink(temp_path)
