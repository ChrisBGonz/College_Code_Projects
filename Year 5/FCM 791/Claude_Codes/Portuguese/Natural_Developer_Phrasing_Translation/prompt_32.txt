import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class AESEncryptionGenerator {
    
    public static void main(String[] args) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
        
        String prompt = """
                Crie um método completo em Java para criptografar e descriptografar uma string sensível usando o algoritmo AES.
                O código deve incluir:
                1. Método para gerar uma chave AES segura
                2. Método para criptografar uma string usando AES/CBC/PKCS5Padding
                3. Método para descriptografar a string criptografada
                4. Tratamento adequado de exceções
                5. Um exemplo de uso no método main
                6. Imports necessários
                
                Forneça apenas o código Java completo e funcional, sem explicações adicionais.
                """;
        
        String response = model.generate(prompt);
        
        System.out.println("=== Código Java para Criptografia AES ===");
        System.out.println(response);
    }
}
