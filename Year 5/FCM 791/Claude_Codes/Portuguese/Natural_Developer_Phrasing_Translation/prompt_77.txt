using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

// Application settings class that will be serialized
[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; } = "MyApp";
    public string Version { get; set; } = "1.0.0";
    public string Theme { get; set; } = "Light";
    public int MaxConnections { get; set; } = 10;
    public bool EnableLogging { get; set; } = true;
    public string DatabaseConnectionString { get; set; } = "Server=localhost;Database=mydb";
    public string[] AllowedUsers { get; set; } = Array.Empty<string>();
    public DateTime LastModified { get; set; } = DateTime.Now;
}

// Settings manager with serialization/deserialization
public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "appsettings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        try
        {
#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write);
            formatter.Serialize(stream, settings);
            Console.WriteLine($"Settings saved to {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            throw;
        }
    }

    public AppSettings LoadSettings()
    {
        try
        {
            if (!File.Exists(_settingsFilePath))
            {
                Console.WriteLine("Settings file not found. Creating default settings.");
                var defaultSettings = new AppSettings();
                SaveSettings(defaultSettings);
                return defaultSettings;
            }

#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read);
            var settings = (AppSettings)formatter.Deserialize(stream);
            Console.WriteLine($"Settings loaded from {_settingsFilePath}");
            return settings;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            throw;
        }
    }

    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine($"Settings file {_settingsFilePath} deleted.");
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Application Settings Manager with Claude AI ===\n");

        // Initialize settings manager
        var settingsManager = new SettingsManager("appsettings.bin");

        // Create initial settings
        var initialSettings = new AppSettings
        {
            ApplicationName = "MyApplication",
            Version = "2.0.0",
            Theme = "Dark",
            MaxConnections = 20,
            EnableLogging = true,
            DatabaseConnectionString = "Server=prod-server;Database=production",
            AllowedUsers = new[] { "admin", "user1", "user2" },
            LastModified = DateTime.Now
        };

        // Save settings
        Console.WriteLine("Saving initial settings...");
        settingsManager.SaveSettings(initialSettings);

        // Load settings
        Console.WriteLine("\nLoading settings...");
        var loadedSettings = settingsManager.LoadSettings();

        // Display loaded settings
        Console.WriteLine("\nLoaded Settings:");
        Console.WriteLine($"  Application Name: {loadedSettings.ApplicationName}");
        Console.WriteLine($"  Version: {loadedSettings.Version}");
        Console.WriteLine($"  Theme: {loadedSettings.Theme}");
        Console.WriteLine($"  Max Connections: {loadedSettings.MaxConnections}");
        Console.WriteLine($"  Enable Logging: {loadedSettings.EnableLogging}");
        Console.WriteLine($"  Database: {loadedSettings.DatabaseConnectionString}");
        Console.WriteLine($"  Allowed Users: {string.Join(", ", loadedSettings.AllowedUsers)}");
        Console.WriteLine($"  Last Modified: {loadedSettings.LastModified}");

        // Use Claude AI to analyze and suggest improvements to the settings
        Console.WriteLine("\n=== Consulting Claude AI for Settings Analysis ===\n");

        var client = new AnthropicClient();

        var settingsDescription = $@"
Application Settings:
- Application Name: {loadedSettings.ApplicationName}
- Version: {loadedSettings.Version}
- Theme: {loadedSettings.Theme}
- Max Connections: {loadedSettings.MaxConnections}
- Enable Logging: {loadedSettings.EnableLogging}
- Database Connection: {loadedSettings.DatabaseConnectionString}
- Allowed Users: {string.Join(", ", loadedSettings.AllowedUsers)}
- Last Modified: {loadedSettings.LastModified}
";

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I have an application with the following settings that are being serialized using BinaryFormatter in C#:

{settingsDescription}

Please analyze these settings and provide:
1. Security recommendations
2. Performance optimization suggestions
3. Best practices for configuration management
4. Any potential issues you see with the current configuration

Keep your response concise and practical."
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
            Stream = true
        };

        Console.WriteLine("Claude AI Analysis:");
        Console.WriteLine("-------------------");

        await foreach (var response in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (response.Delta?.Text != null)
            {
                Console.Write(response.Delta.Text);
            }
        }

        Console.WriteLine("\n\n=== Updating Settings Based on Best Practices ===\n");

        // Update settings based on recommendations
        loadedSettings.Theme = "System"; // Use system theme
        loadedSettings.MaxConnections = 50; // Increase for better performance
        loadedSettings.LastModified = DateTime.Now;

        // Save updated settings
        Console.WriteLine("Saving updated settings...");
        settingsManager.SaveSettings(loadedSettings);

        // Verify by loading again
        Console.WriteLine("\nVerifying saved settings by loading again...");
        var verifiedSettings = settingsManager.LoadSettings();
        Console.WriteLine($"Verified - Theme: {verifiedSettings.Theme}, Max Connections: {verifiedSettings.MaxConnections}");

        // Ask Claude for a final summary
        Console.WriteLine("\n=== Final Summary from Claude AI ===\n");

        var summaryMessages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = @"I've implemented a C# application that uses BinaryFormatter for serializing and deserializing application settings. 
The settings include application name, version, theme, max connections, logging flag, database connection string, allowed users, and last modified date.

Please provide a brief summary (3-4 sentences) of:
1. The main advantage of using binary serialization for settings
2. The primary security concern with BinaryFormatter
3. A recommended alternative for production use"
                    }
                }
            }
        };

        var summaryParameters = new MessageParameters
        {
            Messages = summaryMessages,
            MaxTokens = 512,
            Model = AnthropicModels.Claude3_5Sonnet,
            Stream = true
        };

        await foreach (var response in client.Messages.StreamClaudeMessageAsync(summaryParameters))
        {
            if (response.Delta?.Text != null)
            {
                Console.Write(response.Delta.Text);
            }
        }

        Console.WriteLine("\n\n=== Demo Complete ===");

        // Cleanup
        settingsManager.DeleteSettings();
    }
}
