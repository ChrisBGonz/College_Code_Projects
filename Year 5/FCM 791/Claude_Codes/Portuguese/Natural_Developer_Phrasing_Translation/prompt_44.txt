from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

SYSTEM_PROMPT = """You are a mathematical expression evaluator assistant. 
Your primary role is to evaluate mathematical expressions provided by users and return the calculated result.

When a user provides a mathematical expression:
1. Evaluate it carefully and accurately
2. Return the numerical result
3. If the expression is invalid or cannot be evaluated, explain why
4. You can handle basic arithmetic (+, -, *, /), powers (^), square roots, trigonometric functions, and other mathematical operations
5. Always provide clear, concise responses with the final answer

Format your response as:
- The calculated result (just the number or expression result)
- A brief explanation if needed"""

@app.route('/calculate', methods=['POST'])
def calculate():
    """
    Route to evaluate mathematical expressions using Claude AI.
    
    Expected JSON body:
    {
        "expression": "mathematical expression string",
        "reset_history": false  (optional, to reset conversation history)
    }
    
    Returns:
    {
        "expression": "original expression",
        "result": "calculated result",
        "explanation": "optional explanation"
    }
    """
    global conversation_history
    
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Missing required field: expression'
        }), 400
    
    expression = data.get('expression', '').strip()
    reset_history = data.get('reset_history', False)
    
    if not expression:
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    if reset_history:
        conversation_history = []
    
    user_message = f"Please evaluate this mathematical expression: {expression}"
    
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        'expression': expression,
        'result': assistant_message,
        'conversation_length': len(conversation_history) // 2
    })

@app.route('/calculate/history', methods=['GET'])
def get_history():
    """
    Route to get the conversation history.
    
    Returns the list of all mathematical expressions evaluated in the current session.
    """
    history_summary = []
    
    for i in range(0, len(conversation_history), 2):
        if i + 1 < len(conversation_history):
            user_msg = conversation_history[i]['content']
            assistant_msg = conversation_history[i + 1]['content']
            
            expression = user_msg.replace("Please evaluate this mathematical expression: ", "")
            
            history_summary.append({
                'expression': expression,
                'result': assistant_msg
            })
    
    return jsonify({
        'history': history_summary,
        'total_calculations': len(history_summary)
    })

@app.route('/calculate/reset', methods=['POST'])
def reset_history():
    """
    Route to reset the conversation history.
    """
    global conversation_history
    conversation_history = []
    
    return jsonify({
        'message': 'Conversation history has been reset successfully'
    })

@app.route('/health', methods=['GET'])
def health_check():
    """
    Health check endpoint.
    """
    return jsonify({
        'status': 'healthy',
        'service': 'Mathematical Expression Calculator',
        'model': 'claude-opus-4-5'
    })

if __name__ == '__main__':
    print("Starting Mathematical Expression Calculator API...")
    print("Available endpoints:")
    print("  POST /calculate - Evaluate a mathematical expression")
    print("  GET  /calculate/history - Get calculation history")
    print("  POST /calculate/reset - Reset conversation history")
    print("  GET  /health - Health check")
    app.run(debug=True, host='0.0.0.0', port=5000)
