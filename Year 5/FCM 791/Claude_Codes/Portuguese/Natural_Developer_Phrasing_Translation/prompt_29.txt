#!/usr/bin/env python3

import subprocess
import sys
import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def run_command(command: str, cwd: str = None) -> tuple[str, str, int]:
    """Run a shell command and return stdout, stderr, and return code."""
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            cwd=cwd
        )
        return result.stdout, result.stderr, result.returncode
    except Exception as e:
        return "", str(e), 1

def get_maven_dependency_updates(project_path: str) -> str:
    """Get available dependency updates using Maven versions plugin."""
    print("Checking for dependency updates...")
    stdout, stderr, returncode = run_command(
        "mvn versions:display-dependency-updates -q",
        cwd=project_path
    )
    
    if returncode != 0:
        return f"Error checking updates: {stderr}"
    
    return stdout if stdout else "No output from Maven versions plugin"

def get_plugin_updates(project_path: str) -> str:
    """Get available plugin updates using Maven versions plugin."""
    print("Checking for plugin updates...")
    stdout, stderr, returncode = run_command(
        "mvn versions:display-plugin-updates -q",
        cwd=project_path
    )
    
    if returncode != 0:
        return f"Error checking plugin updates: {stderr}"
    
    return stdout if stdout else "No output from Maven versions plugin"

def update_dependencies(project_path: str) -> str:
    """Update all dependencies to their latest versions."""
    print("Updating dependencies...")
    stdout, stderr, returncode = run_command(
        "mvn versions:use-latest-releases -DgenerateBackupPoms=false",
        cwd=project_path
    )
    
    if returncode != 0:
        return f"Error updating dependencies: {stderr}\n{stdout}"
    
    return stdout if stdout else "Dependencies updated successfully"

def update_plugins(project_path: str) -> str:
    """Update all plugins to their latest versions."""
    print("Updating plugins...")
    stdout, stderr, returncode = run_command(
        "mvn versions:use-latest-versions -DgenerateBackupPoms=false",
        cwd=project_path
    )
    
    if returncode != 0:
        return f"Error updating plugins: {stderr}\n{stdout}"
    
    return stdout if stdout else "Plugins updated successfully"

def run_tests(project_path: str) -> str:
    """Run Maven tests to verify updates didn't break anything."""
    print("Running tests...")
    stdout, stderr, returncode = run_command(
        "mvn test -q",
        cwd=project_path
    )
    
    if returncode != 0:
        return f"Tests failed:\n{stderr}\n{stdout}"
    
    return "All tests passed successfully"

def read_pom_xml(project_path: str) -> str:
    """Read the pom.xml file content."""
    pom_path = os.path.join(project_path, "pom.xml")
    try:
        with open(pom_path, 'r') as f:
            return f.read()
    except FileNotFoundError:
        return "pom.xml not found"
    except Exception as e:
        return f"Error reading pom.xml: {str(e)}"

def chat_with_claude(user_message: str, project_path: str = None) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    system_prompt = """You are an expert Java/Maven dependency management assistant. 
    You help users update their Maven project dependencies to the latest versions.
    You can analyze pom.xml files, interpret Maven output, and provide guidance on:
    - Which dependencies to update
    - Potential breaking changes
    - Best practices for dependency management
    - How to resolve conflicts
    
    When users ask you to perform actions, provide clear instructions and analysis.
    Be concise but thorough in your explanations."""
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def process_user_command(user_input: str, project_path: str) -> str:
    """Process user commands and interact with Maven."""
    user_input_lower = user_input.lower()
    
    if any(word in user_input_lower for word in ['check', 'show', 'list', 'display', 'what', 'which']):
        if 'plugin' in user_input_lower:
            maven_output = get_plugin_updates(project_path)
            context = f"Here's the Maven plugin updates output:\n{maven_output}"
        else:
            maven_output = get_maven_dependency_updates(project_path)
            context = f"Here's the Maven dependency updates output:\n{maven_output}"
        
        return chat_with_claude(f"{user_input}\n\n{context}", project_path)
    
    elif any(word in user_input_lower for word in ['update', 'upgrade', 'latest']):
        if 'plugin' in user_input_lower:
            result = update_plugins(project_path)
            context = f"Plugin update result:\n{result}"
        elif 'all' in user_input_lower or 'everything' in user_input_lower:
            dep_result = update_dependencies(project_path)
            plugin_result = update_plugins(project_path)
            context = f"Dependency update result:\n{dep_result}\n\nPlugin update result:\n{plugin_result}"
        else:
            result = update_dependencies(project_path)
            context = f"Dependency update result:\n{result}"
        
        return chat_with_claude(f"{user_input}\n\n{context}", project_path)
    
    elif any(word in user_input_lower for word in ['test', 'verify', 'build']):
        result = run_tests(project_path)
        context = f"Test result:\n{result}"
        return chat_with_claude(f"{user_input}\n\n{context}", project_path)
    
    elif any(word in user_input_lower for word in ['pom', 'xml', 'file', 'content']):
        pom_content = read_pom_xml(project_path)
        context = f"Current pom.xml content:\n{pom_content}"
        return chat_with_claude(f"{user_input}\n\n{context}", project_path)
    
    else:
        return chat_with_claude(user_input, project_path)

def main():
    """Main function to run the Maven dependency update assistant."""
    print("Maven Dependency Update Assistant")
    print("=" * 50)
    
    if len(sys.argv) > 1:
        project_path = sys.argv[1]
    else:
        project_path = input("Enter the path to your Maven project (or press Enter for current directory): ").strip()
        if not project_path:
            project_path = os.getcwd()
    
    if not os.path.exists(project_path):
        print(f"Error: Path '{project_path}' does not exist")
        sys.exit(1)
    
    pom_path = os.path.join(project_path, "pom.xml")
    if not os.path.exists(pom_path):
        print(f"Error: No pom.xml found in '{project_path}'")
        sys.exit(1)
    
    print(f"\nProject path: {project_path}")
    print("\nI can help you:")
    print("- Check available dependency updates")
    print("- Update dependencies to latest versions")
    print("- Check and update plugins")
    print("- Run tests to verify updates")
    print("- Analyze your pom.xml")
    print("\nType 'quit' or 'exit' to stop")
    print("=" * 50)
    
    initial_message = f"""I'm ready to help you manage Maven dependencies for your project at: {project_path}
    
What would you like to do? You can ask me to:
- Check what updates are available
- Update all dependencies to latest versions
- Show the current pom.xml
- Run tests after updates
- Or ask any Maven-related questions"""
    
    print(f"\nAssistant: {initial_message}\n")
    
    while True:
        try:
            user_input = input("You: ").strip()
            
            if not user_input:
                continue
            
            if user_input.lower() in ['quit', 'exit', 'bye', 'goodbye']:
                print("Assistant: Goodbye! Your Maven project dependencies have been managed successfully.")
                break
            
            print("\nAssistant: ", end="", flush=True)
            response = process_user_command(user_input, project_path)
            print(response)
            print()
            
        except KeyboardInterrupt:
            print("\n\nAssistant: Session interrupted. Goodbye!")
            break
        except Exception as e:
            print(f"\nError: {str(e)}")
            print("Please try again or type 'quit' to exit.\n")

if __name__ == "__main__":
    main()
