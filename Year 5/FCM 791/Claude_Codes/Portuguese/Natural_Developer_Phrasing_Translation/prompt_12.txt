import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";

const client = new Anthropic();

async function generateExpressConfig(): Promise<void> {
  console.log(
    "Generating Express.js production server configuration with Claude...\n"
  );

  const prompt = `Crie uma configuração básica de servidor Express.js voltada para ambiente de produção, incluindo middlewares essenciais e um middleware de tratamento de erros (error handler) centralizado.

Por favor, inclua:
1. Configuração básica do Express com middlewares de segurança (helmet, cors)
2. Middleware de compressão (compression)
3. Middleware de logging (morgan)
4. Rate limiting para proteção contra ataques
5. Middleware de parsing de JSON e URL encoded
6. Middleware de tratamento de erros centralizado com diferentes tipos de erro
7. Graceful shutdown
8. Variáveis de ambiente com dotenv
9. Rotas de exemplo básicas
10. Comentários explicativos em português

Forneça o código completo e funcional com as dependências necessárias no package.json.`;

  let fullResponse = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: prompt,
      },
    ],
  });

  process.stdout.write("Generating configuration");

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      fullResponse += chunk.delta.text;
      process.stdout.write(".");
    }
  }

  console.log("\n\nConfiguration generated successfully!\n");
  console.log("=".repeat(60));
  console.log("GENERATED EXPRESS.JS PRODUCTION CONFIGURATION:");
  console.log("=".repeat(60));
  console.log(fullResponse);

  // Extract and save code files
  const codeBlocks = extractCodeBlocks(fullResponse);

  if (codeBlocks.length > 0) {
    console.log("\n" + "=".repeat(60));
    console.log("SAVING GENERATED FILES:");
    console.log("=".repeat(60));

    for (const block of codeBlocks) {
      if (block.filename) {
        const outputPath = path.join(process.cwd(), "output", block.filename);
        const outputDir = path.dirname(outputPath);

        if (!fs.existsSync(outputDir)) {
          fs.mkdirSync(outputDir, { recursive: true });
        }

        fs.writeFileSync(outputPath, block.code);
        console.log(`✓ Saved: ${block.filename}`);
      }
    }
  }

  // Get final message stats
  const finalMessage = await stream.finalMessage();
  console.log("\n" + "=".repeat(60));
  console.log("GENERATION STATS:");
  console.log("=".repeat(60));
  console.log(`Input tokens: ${finalMessage.usage.input_tokens}`);
  console.log(`Output tokens: ${finalMessage.usage.output_tokens}`);
  console.log(`Stop reason: ${finalMessage.stop_reason}`);
}

interface CodeBlock {
  language: string;
  filename?: string;
  code: string;
}

function extractCodeBlocks(text: string): CodeBlock[] {
  const blocks: CodeBlock[] = [];
  const codeBlockRegex = /```(\w+)?\n([\s\S]*?)```/g;
  let match;

  while ((match = codeBlockRegex.exec(text)) !== null) {
    const language = match[1] || "text";
    const code = match[2].trim();

    // Try to determine filename from context
    let filename: string | undefined;

    if (language === "javascript" || language === "js") {
      // Check if it's a server file
      if (
        code.includes("express()") ||
        code.includes("app.listen") ||
        code.includes("createServer")
      ) {
        filename = "server.js";
      } else if (code.includes("module.exports") || code.includes("export")) {
        filename = "app.js";
      }
    } else if (language === "json") {
      if (
        code.includes('"name"') &&
        code.includes('"version"') &&
        code.includes('"dependencies"')
      ) {
        filename = "package.json";
      }
    } else if (language === "typescript" || language === "ts") {
      if (code.includes("express()") || code.includes("app.listen")) {
        filename = "server.ts";
      }
    } else if (language === "env" || language === "dotenv") {
      filename = ".env.example";
    }

    blocks.push({ language, filename, code });
  }

  return blocks;
}

// Main execution
generateExpressConfig().catch((error) => {
  console.error("Error generating configuration:", error);
  process.exit(1);
});
