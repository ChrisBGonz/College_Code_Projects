import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateMongoDBConnectionString() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Configure uma string de conexão do MongoDB em um script Node.js apontando para um banco de desenvolvimento local. Forneça um exemplo completo com código Node.js mostrando como configurar e usar a string de conexão para um ambiente de desenvolvimento local.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("MongoDB Connection String Configuration Guide:");
  console.log("=".repeat(50));
  console.log(responseText);

  return responseText;
}

generateMongoDBConnectionString().catch(console.error);
