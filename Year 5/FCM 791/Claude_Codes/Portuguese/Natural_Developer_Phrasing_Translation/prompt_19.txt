#!/usr/bin/env python3
"""
FastAPI application with custom database connection error handler.
Uses Claude AI to demonstrate the application structure and error handling.
"""

import anthropic
import asyncio
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
import uvicorn
from typing import Optional
import json

# Initialize FastAPI app
app = FastAPI(
    title="FastAPI with Custom DB Error Handler",
    description="A FastAPI application with custom database connection error handling",
    version="1.0.0"
)

# Custom exception classes for database errors
class DatabaseConnectionError(Exception):
    """Custom exception for database connection errors."""
    def __init__(self, message: str, db_host: Optional[str] = None, db_port: Optional[int] = None):
        self.message = message
        self.db_host = db_host
        self.db_port = db_port
        super().__init__(self.message)

class DatabaseQueryError(Exception):
    """Custom exception for database query errors."""
    def __init__(self, message: str, query: Optional[str] = None):
        self.message = message
        self.query = query
        super().__init__(self.message)

class DatabaseTimeoutError(DatabaseConnectionError):
    """Custom exception for database timeout errors."""
    def __init__(self, message: str, timeout_seconds: Optional[int] = None):
        self.timeout_seconds = timeout_seconds
        super().__init__(message)

# Custom exception handlers
@app.exception_handler(DatabaseConnectionError)
async def database_connection_error_handler(request: Request, exc: DatabaseConnectionError):
    """Handler for database connection errors."""
    error_detail = {
        "error_type": "DatabaseConnectionError",
        "message": exc.message,
        "status_code": 503,
        "path": str(request.url),
        "method": request.method
    }
    
    if exc.db_host:
        error_detail["db_host"] = exc.db_host
    if exc.db_port:
        error_detail["db_port"] = exc.db_port
    
    return JSONResponse(
        status_code=503,
        content={
            "error": error_detail,
            "suggestion": "Please check your database connection settings and ensure the database server is running."
        }
    )

@app.exception_handler(DatabaseTimeoutError)
async def database_timeout_error_handler(request: Request, exc: DatabaseTimeoutError):
    """Handler for database timeout errors."""
    error_detail = {
        "error_type": "DatabaseTimeoutError",
        "message": exc.message,
        "status_code": 504,
        "path": str(request.url),
        "method": request.method
    }
    
    if exc.timeout_seconds:
        error_detail["timeout_seconds"] = exc.timeout_seconds
    
    return JSONResponse(
        status_code=504,
        content={
            "error": error_detail,
            "suggestion": "The database connection timed out. Consider increasing the timeout limit or optimizing your queries."
        }
    )

@app.exception_handler(DatabaseQueryError)
async def database_query_error_handler(request: Request, exc: DatabaseQueryError):
    """Handler for database query errors."""
    error_detail = {
        "error_type": "DatabaseQueryError",
        "message": exc.message,
        "status_code": 500,
        "path": str(request.url),
        "method": request.method
    }
    
    if exc.query:
        error_detail["failed_query"] = exc.query[:100] + "..." if len(exc.query) > 100 else exc.query
    
    return JSONResponse(
        status_code=500,
        content={
            "error": error_detail,
            "suggestion": "Please check your query syntax and ensure all referenced tables and columns exist."
        }
    )

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    """Handler for request validation errors."""
    return JSONResponse(
        status_code=422,
        content={
            "error": {
                "error_type": "ValidationError",
                "message": "Request validation failed",
                "details": exc.errors(),
                "status_code": 422
            }
        }
    )

# Routes that demonstrate the error handling
@app.get("/")
async def root():
    """Root endpoint."""
    return {
        "message": "FastAPI with Custom DB Error Handler",
        "endpoints": {
            "/simulate/connection-error": "Simulates a database connection error",
            "/simulate/timeout-error": "Simulates a database timeout error",
            "/simulate/query-error": "Simulates a database query error",
            "/ai/explain-errors": "Uses Claude AI to explain the error handling",
            "/health": "Health check endpoint"
        }
    }

@app.get("/health")
async def health_check():
    """Health check endpoint."""
    return {"status": "healthy", "service": "FastAPI DB Error Handler"}

@app.get("/simulate/connection-error")
async def simulate_connection_error():
    """Endpoint that simulates a database connection error."""
    raise DatabaseConnectionError(
        message="Failed to connect to the database server",
        db_host="localhost",
        db_port=5432
    )

@app.get("/simulate/timeout-error")
async def simulate_timeout_error():
    """Endpoint that simulates a database timeout error."""
    raise DatabaseTimeoutError(
        message="Database connection timed out after waiting",
        timeout_seconds=30
    )

@app.get("/simulate/query-error")
async def simulate_query_error():
    """Endpoint that simulates a database query error."""
    raise DatabaseQueryError(
        message="Invalid SQL syntax in query",
        query="SELECT * FROM non_existent_table WHERE id = 1"
    )

@app.get("/ai/explain-errors")
async def explain_errors_with_ai():
    """Uses Claude AI with streaming to explain the error handling implementation."""
    client = anthropic.Anthropic()
    
    explanation_parts = []
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Explain briefly (in 3-4 sentences) why custom database connection error handlers are important in FastAPI applications. 
                Focus on: 1) User experience, 2) Security (not exposing internal errors), 3) Debugging capabilities.
                Be concise and practical."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            explanation_parts.append(text)
    
    full_explanation = "".join(explanation_parts)
    
    return {
        "topic": "Custom Database Error Handlers in FastAPI",
        "ai_explanation": full_explanation,
        "implemented_handlers": [
            {
                "handler": "DatabaseConnectionError",
                "http_status": 503,
                "description": "Handles failures when connecting to the database server"
            },
            {
                "handler": "DatabaseTimeoutError",
                "http_status": 504,
                "description": "Handles database connection timeout scenarios"
            },
            {
                "handler": "DatabaseQueryError",
                "http_status": 500,
                "description": "Handles errors in SQL query execution"
            }
        ]
    }

@app.get("/ai/generate-error-scenarios")
async def generate_error_scenarios():
    """Uses Claude AI to generate common database error scenarios."""
    client = anthropic.Anthropic()
    
    scenarios_text = []
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=800,
        messages=[
            {
                "role": "user",
                "content": """List 5 common database connection error scenarios in production environments.
                Format as JSON array with fields: scenario, cause, solution.
                Keep each entry brief (1-2 sentences per field).
                Return ONLY the JSON array, no other text."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            scenarios_text.append(text)
    
    full_response = "".join(scenarios_text)
    
    # Try to parse as JSON, otherwise return as text
    try:
        # Find JSON array in response
        start_idx = full_response.find('[')
        end_idx = full_response.rfind(']') + 1
        if start_idx != -1 and end_idx > start_idx:
            json_str = full_response[start_idx:end_idx]
            scenarios = json.loads(json_str)
        else:
            scenarios = full_response
    except json.JSONDecodeError:
        scenarios = full_response
    
    return {
        "title": "Common Database Error Scenarios",
        "scenarios": scenarios,
        "note": "These scenarios are handled by our custom exception handlers"
    }

async def demonstrate_streaming():
    """Demonstrates streaming with Claude AI for error explanation."""
    print("\n" + "="*60)
    print("DEMONSTRATING CLAUDE AI STREAMING FOR ERROR EXPLANATION")
    print("="*60)
    
    client = anthropic.Anthropic()
    
    print("\nStreaming explanation of database error handling...\n")
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {
                "role": "user",
                "content": """In 2-3 sentences, explain why FastAPI's exception handlers are better than try-catch blocks 
                scattered throughout the code for handling database errors. Be direct and practical."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n\n" + "="*60)
    print("STREAMING COMPLETE")
    print("="*60)

def main():
    """Main function to run the FastAPI application."""
    print("\n" + "="*60)
    print("FASTAPI WITH CUSTOM DB ERROR HANDLER")
    print("="*60)
    print("\nStarting FastAPI application...")
    print("\nAvailable endpoints:")
    print("  GET /                              - API information")
    print("  GET /health                        - Health check")
    print("  GET /simulate/connection-error     - Simulate DB connection error")
    print("  GET /simulate/timeout-error        - Simulate DB timeout error")
    print("  GET /simulate/query-error          - Simulate DB query error")
    print("  GET /ai/explain-errors             - AI explanation of error handling")
    print("  GET /ai/generate-error-scenarios   - AI generated error scenarios")
    print("\nCustom Exception Handlers:")
    print("  - DatabaseConnectionError  -> HTTP 503 Service Unavailable")
    print("  - DatabaseTimeoutError     -> HTTP 504 Gateway Timeout")
    print("  - DatabaseQueryError       -> HTTP 500 Internal Server Error")
    print("  - RequestValidationError   -> HTTP 422 Unprocessable Entity")
    
    # Run the streaming demonstration
    asyncio.run(demonstrate_streaming())
    
    print("\n" + "="*60)
    print("Starting server on http://localhost:8000")
    print("Press CTRL+C to stop")
    print("="*60 + "\n")
    
    uvicorn.run(
        app,
        host="0.0.0.0",
        port=8000,
        log_level="info"
    )

if __name__ == "__main__":
    main()
