package main

import (
	"context"

	"log"
	"math/rand"
	"net/http"
	"time"

	"github.com/anthropics/anthropic-sdk-go"

// MicroserviceCall simulates calling a microservice
func MicroserviceCall(url string) error {
	// Simulate random failures for demonstration
	if rand.Float32() < 0.7 { // 70% chance of failure for demo
		return fmt.Errorf("microservice call failed: connection refused")
	}

	resp, err := http.Get(url)
	if err != nil {
		return fmt.Errorf("microservice call failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("microservice returned status: %d", resp.StatusCode)
	}

	return nil
}

// CallMicroserviceWithRetry calls a microservice in a loop with immediate retry on failure
func CallMicroserviceWithRetry(url string, maxAttempts int) error {
	attempt := 0
	for {
		attempt++
		log.Printf("Attempt %d: Calling microservice at %s", attempt, url)

		err := MicroserviceCall(url)
		if err == nil {
			log.Printf("Success on attempt %d", attempt)
			return nil
		}

		log.Printf("Attempt %d failed: %v. Retrying immediately...", attempt, err)

		if maxAttempts > 0 && attempt >= maxAttempts {
			return fmt.Errorf("failed after %d attempts: %w", attempt, err)
		}
	}
}
