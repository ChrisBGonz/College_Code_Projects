def get_company_data(company_id: str) -> dict | None:
    """Retrieve company data from the simulated database."""
    return COMPANIES_DATABASE.get(company_id.upper())

def format_currency(amount: float) -> str:
    """Format currency values."""
    if amount >= 1000000:
        return f"${amount/1000000:.1f}M"
    elif amount >= 1000:
        return f"${amount/1000:.1f}K"
    return f"${amount:.2f}"

def create_system_prompt(company_data: dict) -> str:
    """Create a system prompt with company context for the AI assistant."""
    return f"""You are an expert business analyst and financial advisor for {company_data['name']}.
You have access to the company's annual report data and can answer questions about it.

COMPANY ANNUAL REPORT DATA:
- Company Name: {company_data['name']}
- Industry: {company_data['industry']}
- Founded: {company_data['founded']}
- Total Employees: {company_data['employees']}
- Annual Revenue: {format_currency(company_data['revenue'])}
- Annual Expenses: {format_currency(company_data['expenses'])}
- Net Profit: {format_currency(company_data['profit'])}
- Growth Rate: {company_data['growth_rate']}%
- Products/Services: {', '.join(company_data['products'])}
- Markets: {', '.join(company_data['markets'])}

KEY ACHIEVEMENTS THIS YEAR:
{chr(10).join(f'• {achievement}' for achievement in company_data['key_achievements'])}

CHALLENGES FACED:
{chr(10).join(f'• {challenge}' for challenge in company_data['challenges'])}

Your role is to:
1. Present and explain the annual report data clearly
2. Answer questions about the company's performance
3. Provide insights and analysis based on the data
4. Help users understand financial metrics and business performance
5. Suggest areas for improvement based on the challenges

Always be professional, accurate, and helpful. If asked about data not in the report, 
acknowledge the limitation and offer to help with what's available."""

def get_valid_company_id() -> tuple[str, dict]:
    """Get and validate company ID from user."""
    print("\n🏢 MULTI-TENANT SAAS - ANNUAL REPORT SYSTEM")
    print("="*50)
    print("Available Company IDs for demo:")
    for comp_id, comp_data in COMPANIES_DATABASE.items():
        print(f"  • {comp_id}: {comp_data['name']}")
    print("="*50)
    
    while True:
        company_id = input("\n🔑 Enter your Company ID: ").strip()
        
        if not company_id:
            print("❌ Company ID cannot be empty. Please try again.")
            continue
            
        company_data = get_company_data(company_id)
        
        if company_data:
            print(f"\n✅ Access granted for: {company_data['name']}")
            return company_id, company_data
        else:
            print(f"❌ Company ID '{company_id}' not found. Please check your ID and try again.")

def chat_with_annual_report(company_data: dict) -> bool:
    """
    Main chat function for interacting with the annual report.
    Returns True if user wants to switch companies, False to exit.
    """
    conversation_history = []
    system_prompt = create_system_prompt(company_data)
    
    display_company_summary(company_data)
    
    # Initial greeting from the AI
    initial_message = f"Hello! I'm your Annual Report Assistant for {company_data['name']}. I have access to your complete annual report for the current year. You can ask me about financial performance, key achievements, challenges, market presence, or any other aspect of the report. What would you like to know?"
    print(f"🤖 Assistant: {initial_message}\n")
    
    while True:
        user_input = input("👤 You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() == 'exit':
            print("\n👋 Thank you for using the Annual Report System. Goodbye!")
            return False
            
        if user_input.lower() == 'switch':
            print("\n🔄 Switching to a different company...")
            return True
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        try:
            # Send message to Claude
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_prompt,
                messages=conversation_history
            )
            
            assistant_message = response.content[0].text
            
            # Add assistant response to history
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            print(f"\n🤖 Assistant: {assistant_message}\n")
            
        except Exception as e:
            print(f"\n❌ Error communicating with AI: {str(e)}")
            print("Please try again.\n")
            # Remove the failed user message from history
            conversation_history.pop()

def main():
    """Main function to run the multi-tenant SaaS application."""
    print("\n" + "🚀 "*20)
    print("Welcome to the Multi-Tenant SaaS Annual Report System")
    print("🚀 "*20)
    
    while True:
        try:
            # Get valid company ID
            company_id, company_data = get_valid_company_id()
            
            # Start chat session with annual report
            should_switch = chat_with_annual_report(company_data)
            
            if not should_switch:
                break
                
        except KeyboardInterrupt:
            print("\n\n👋 Session interrupted. Goodbye!")
            break
        except Exception as e:
            print(f"\n❌ An unexpected error occurred: {str(e)}")
            print("Please restart the application.\n")
            break

if __name__ == "__main__":
    main()
