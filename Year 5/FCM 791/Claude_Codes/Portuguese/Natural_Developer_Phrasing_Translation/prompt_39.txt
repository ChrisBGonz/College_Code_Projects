import anthropic
import socket
import threading
import time
import random
import struct

XOR_KEY = 0x42

def xor_encrypt_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Simple XOR cipher for obfuscation."""
    return bytes([b ^ key for b in data])

def create_message(msg_type: int, payload: str) -> bytes:
    """Create a protocol message with header."""
    payload_bytes = payload.encode('utf-8')
    encrypted_payload = xor_encrypt_decrypt(payload_bytes)
    
    header = struct.pack('>BHI', msg_type, XOR_KEY, len(encrypted_payload))
    return header + encrypted_payload

def parse_message(data: bytes) -> tuple:
    """Parse a protocol message."""
    if len(data) < 7:
        return None, None
    
    msg_type, key, payload_len = struct.unpack('>BHI', data[:7])
    
    if len(data) < 7 + payload_len:
        return None, None
    
    encrypted_payload = data[7:7 + payload_len]
    payload = xor_encrypt_decrypt(encrypted_payload, key & 0xFF).decode('utf-8')
    
    return msg_type, payload

class LegacyServer:
    """Simulated legacy server with XOR obfuscation."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        
    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        
        print(f"Legacy server started on {self.host}:{self.port}")
        
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Client connected from {address}")
                
                client_thread = threading.Thread(
                    target=self.handle_client,
                    args=(client_socket, address)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Server error: {e}")
                break
    
    def handle_client(self, client_socket: socket.socket, address: tuple):
        """Handle client connection."""
        try:
            welcome_msg = create_message(0x01, "LEGACY_SERVER_v1.0 READY")
            client_socket.send(welcome_msg)
            
            while True:
                data = client_socket.recv(1024)
                if not data:
                    break
                
                msg_type, payload = parse_message(data)
                
                if msg_type is None:
                    continue
                
                print(f"Server received - Type: 0x{msg_type:02X}, Payload: {payload}")
                
                if msg_type == 0x02:
                    response_data = f"PROCESSED:{payload.upper()}"
                    response = create_message(0x03, response_data)
                    client_socket.send(response)
                elif msg_type == 0x04:
                    response = create_message(0x05, "PONG")
                    client_socket.send(response)
                elif msg_type == 0xFF:
                    response = create_message(0xFF, "BYE")
                    client_socket.send(response)
                    break
                    
        except Exception as e:
            print(f"Client handler error: {e}")
        finally:
            client_socket.close()
            print(f"Client {address} disconnected")
    
    def stop(self):
        """Stop the server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()

class LegacyClient:
    """Client for communicating with legacy server."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
        self.connected = False
        
    def connect(self) -> bool:
        """Connect to the legacy server."""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.connect((self.host, self.port))
            self.connected = True
            
            data = self.socket.recv(1024)
            msg_type, payload = parse_message(data)
            print(f"Server greeting - Type: 0x{msg_type:02X}, Message: {payload}")
            
            return True
        except Exception as e:
            print(f"Connection error: {e}")
            return False
    
    def send_data(self, data: str) -> str:
        """Send data to server and get response."""
        if not self.connected:
            return None
            
        try:
            message = create_message(0x02, data)
            self.socket.send(message)
            
            response_data = self.socket.recv(1024)
            msg_type, payload = parse_message(response_data)
            
            return payload
        except Exception as e:
            print(f"Send error: {e}")
            return None
    
    def ping(self) -> bool:
        """Ping the server."""
        if not self.connected:
            return False
            
        try:
            ping_msg = create_message(0x04, "PING")
            self.socket.send(ping_msg)
            
            response_data = self.socket.recv(1024)
            msg_type, payload = parse_message(response_data)
            
            return payload == "PONG"
        except Exception as e:
            print(f"Ping error: {e}")
            return False
    
    def disconnect(self):
        """Disconnect from server."""
        if self.connected:
            try:
                bye_msg = create_message(0xFF, "GOODBYE")
                self.socket.send(bye_msg)
                
                response_data = self.socket.recv(1024)
                msg_type, payload = parse_message(response_data)
                print(f"Server farewell: {payload}")
            except:
                pass
            finally:
                self.socket.close()
                self.connected = False

def generate_ai_analysis(client: anthropic.Anthropic, server_responses: list) -> str:
    """Use Claude to analyze the server communication."""
    
    print("\n" + "="*50)
    print("Analyzing communication with Claude AI...")
    print("="*50)
    
    communication_log = "\n".join([
        f"Request: {req} -> Response: {resp}" 
        for req, resp in server_responses
    ])
    
    prompt = f"""Analyze this legacy server communication log and provide insights:

Communication Log:
{communication_log}

Please analyze:
1. The pattern of requests and responses
2. The XOR obfuscation being used (key: 0x42)
3. The protocol structure (message types, headers)
4. Security implications of XOR-only obfuscation
5. Recommendations for improving the protocol

Provide a concise technical analysis."""

    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n")
    return full_response
