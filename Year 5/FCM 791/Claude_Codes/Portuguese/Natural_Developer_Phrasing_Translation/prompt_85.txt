// ─── Helper Functions ─────────────────────────────────────────────────────────

function getClientIp(): string
{
    $headers = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($headers as $header) {
        if (!empty($_SERVER[$header])) {
            $ips = explode(',', $_SERVER[$header]);
            $ip  = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
                return $ip;
            }
        }
    }

    return $_SERVER['REMOTE_ADDR'] ?? '0.0.0.0';
}

function getCurrentUrl(): string
{
    $scheme = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'https' : 'http';
    $host   = $_SERVER['HTTP_HOST'] ?? 'unknown';
    $uri    = $_SERVER['REQUEST_URI'] ?? '/';
    return $scheme . '://' . $host . $uri;
}

function getUserAgent(): string
{
    return $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown';
}

function getRequestMethod(): string
{
    return $_SERVER['REQUEST_METHOD'] ?? 'GET';
}

function getReferer(): string
{
    return $_SERVER['HTTP_REFERER'] ?? 'Direct / None';
}

function getSessionUserId(): string
{
    return (string) ($_SESSION['user_id'] ?? 'Guest / Unauthenticated');
}

function getSessionUsername(): string
{
    return (string) ($_SESSION['username'] ?? 'Guest');
}

function getSessionUserRole(): string
{
    return (string) ($_SESSION['user_role'] ?? 'none');
}

// ─── Rate Limiter ─────────────────────────────────────────────────────────────

function isRateLimited(string $ip, array $rateLimitConfig): bool
{
    if (!$rateLimitConfig['enabled']) {
        return false;
    }

    $file    = $rateLimitConfig['storage_file'];
    $now     = time();
    $window  = $rateLimitConfig['window_seconds'];
    $max     = $rateLimitConfig['max_attempts'];

    $data = [];
    if (file_exists($file)) {
        $raw = file_get_contents($file);
        if ($raw !== false) {
            $data = json_decode($raw, true) ?? [];
        }
    }

    // Clean up expired entries
    foreach ($data as $storedIp => $attempts) {
        $data[$storedIp] = array_filter($attempts, fn($timestamp) => ($now - $timestamp) < $window);
        if (empty($data[$storedIp])) {
            unset($data[$storedIp]);
        }
    }

    $attempts = $data[$ip] ?? [];

    if (count($attempts) >= $max) {
        return true;
    }

    $data[$ip][] = $now;

    $dir = dirname($file);
    if (!is_dir($dir)) {
        mkdir($dir, 0755, true);
    }

    file_put_contents($file, json_encode($data, JSON_PRETTY_PRINT), LOCK_EX);

    return false;
}

// ─── Logger ───────────────────────────────────────────────────────────────────

function logUnauthorizedAccess(array $details, array $logConfig): void
{
    if (!$logConfig['enabled']) {
        return;
    }

    $timestamp = date('Y-m-d H:i:s T');
    $line      = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | User: %s (ID: %s) | Role: %s | Method: %s | URL: %s | Referer: %s | UA: %s\n",
        $timestamp,
        $details['ip'],
        $details['username'],
        $details['user_id'],
        $details['role'],
        $details['method'],
        $details['url'],
        $details['referer'],
        $details['user_agent']
    );

    file_put_contents($logConfig['file'], $line, FILE_APPEND | LOCK_EX);
}

// ─── Email Alert ──────────────────────────────────────────────────────────────

function sendEmailAlert(array $details, array $alertConfig): bool
{
    $to      = $alertConfig['email_to'];
    $subject = '🚨 [SECURITY ALERT] Unauthorized /admin Access Attempt';

    $body = buildEmailBody($details);

    $headers  = "MIME-Version: 1.0\r\n";
    $headers .= "Content-Type: text/html; charset=UTF-8\r\n";
    $headers .= "From: {$alertConfig['email_from_name']} <{$alertConfig['email_from']}>\r\n";
    $headers .= "X-Priority: 1\r\n";
    $headers .= "X-Mailer: PHP/" . phpversion() . "\r\n";

    $result = mail($to, $subject, $body, $headers);

    return $result;
}

// ─── Slack Alert ──────────────────────────────────────────────────────────────

function sendSlackAlert(array $details, array $alertConfig): bool
{
    $webhookUrl = $alertConfig['slack_webhook_url'];

    if (empty($webhookUrl) || strpos($webhookUrl, 'YOUR/WEBHOOK') !== false) {
        error_log('[SecurityAlert] Slack webhook URL is not configured.');
        return false;
    }

    $payload = buildSlackPayload($details);
    $json    = json_encode($payload);

    $ch = curl_init($webhookUrl);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $json,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $error    = curl_error($ch);
    curl_close($ch);

    if ($error) {
        error_log('[SecurityAlert] Slack cURL error: ' . $error);
        return false;
    }

    return $httpCode === 200 && $response === 'ok';
}

// ─── Main Alert Dispatcher ────────────────────────────────────────────────────

function dispatchAdminAlert(array $details, array $config): array
{
    $results = [
        'email' => null,
        'slack' => null,
        'log'   => false,
    ];

    // Log the attempt
    logUnauthorizedAccess($details, $config['log']);
    $results['log'] = true;

    $method = $config['alert']['method'];

    if ($method === 'email' || $method === 'both') {
        $results['email'] = sendEmailAlert($details, $config['alert']);
    }

    if ($method === 'slack' || $method === 'both') {
        $results['slack'] = sendSlackAlert($details, $config['alert']);
    }

    return $results;
}

// ─── Authorization Check ──────────────────────────────────────────────────────

function checkAdminAuthorization(array $config): bool
{
    $allowedRoles = $config['auth']['allowed_roles'];
    $sessionKey   = $config['auth']['session_key'];
    $userRole     = $_SESSION[$sessionKey] ?? null;

    return in_array($userRole, $allowedRoles, true);
}

// ─── Admin Guard ──────────────────────────────────────────────────────────────

function guardAdminRoute(array $config): void
{
    $requestUri = $_SERVER['REQUEST_URI'] ?? '/';
    $parsedPath = parse_url($requestUri, PHP_URL_PATH);

    // Only guard /admin routes
    if (!preg_match('#^/admin(/.*)?$#i', $parsedPath)) {
        return;
    }

    // Allow authorized users through
    if (checkAdminAuthorization($config)) {
        return;
    }

    // Collect incident details
    $ip      = getClientIp();
    $details = [
        'timestamp'  => date('Y-m-d H:i:s T'),
        'ip'         => $ip,
        'username'   => getSessionUsername(),
        'user_id'    => getSessionUserId(),
        'role'       => getSessionUserRole(),
        'method'     => getRequestMethod(),
        'url'        => getCurrentUrl(),
        'referer'    => getReferer(),
        'user_agent' => getUserAgent(),
    ];

    // Check rate limit (suppress repeated alerts from same IP)
    $alreadyRateLimited = isRateLimited($ip, $config['rate_limit']);

    if (!$alreadyRateLimited) {
        $results = dispatchAdminAlert($details, $config);

        // Debug output (remove in production)
        if (defined('SECURITY_DEBUG') && SECURITY_DEBUG) {
            error_log('[SecurityAlert] Alert dispatched: ' . json_encode($results));
        }
    } else {
        // Still log even if rate-limited, just don't re-alert
        logUnauthorizedAccess($details, $config['log']);
        error_log('[SecurityAlert] Rate-limited alert suppressed for IP: ' . $ip);
    }

    // Respond with 403
    http_response_code(403);
    header('Content-Type: text/html; charset=UTF-8');
    echo render403Page($details);
    exit;
}
