import os
from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

# Simulated database of users (in a real app, this would be a proper database)
USERS_DB = {
    "admin": {
        "password": "admin123",
        "role": "administrator",
        "name": "Admin User",
        "email": "admin@example.com"
    },
    "user1": {
        "password": "password123",
        "role": "user",
        "name": "Regular User",
        "email": "user1@example.com"
    },
    "john_doe": {
        "password": "john2024",
        "role": "user",
        "name": "John Doe",
        "email": "john@example.com"
    }
}

# Store conversation history for multi-turn conversations
conversation_history = []

def validate_credentials_with_claude(username: str, password: str) -> dict:
    """
    Use Claude to validate credentials and provide intelligent responses.
    This demonstrates multi-turn conversation capability.
    """
    global conversation_history
    
    # Check if user exists in our "database"
    user_data = USERS_DB.get(username)
    
    # Prepare context for Claude
    if user_data:
        if user_data["password"] == password:
            db_result = f"User '{username}' found in database. Password matches. Role: {user_data['role']}, Name: {user_data['name']}"
            auth_status = "success"
        else:
            db_result = f"User '{username}' found in database. Password does NOT match."
            auth_status = "invalid_password"
    else:
        db_result = f"User '{username}' NOT found in database."
        auth_status = "user_not_found"
    
    # Add user message to conversation history
    user_message = f"""
    Authentication attempt:
    - Username: {username}
    - Database lookup result: {db_result}
    - Authentication status: {auth_status}
    
    Please provide an appropriate authentication response message for this login attempt.
    Be concise and professional. If successful, include a welcome message with the user's role.
    If failed, provide a helpful but secure error message.
    """
    
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        system="""You are a security-aware authentication assistant for a web application. 
        Your role is to:
        1. Analyze authentication attempts and provide appropriate responses
        2. Generate secure, user-friendly messages for login success or failure
        3. Never reveal specific security details about why authentication failed (for security reasons)
        4. Maintain a professional and helpful tone
        5. For successful logins, provide a personalized welcome message
        6. For failed logins, provide a generic error message without revealing if the username exists""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to conversation history
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    # Return structured response
    if auth_status == "success":
        return {
            "success": True,
            "message": assistant_message,
            "user": {
                "username": username,
                "name": user_data["name"],
                "role": user_data["role"],
                "email": user_data["email"]
            }
        }
    else:
        return {
            "success": False,
            "message": assistant_message,
            "error_type": auth_status
        }

@app.route('/login', methods=['POST'])
def login():
    """
    Login endpoint that validates credentials using Claude AI.
    
    Expected JSON body:
    {
        "username": "string",
        "password": "string"
    }
    """
    # Get request data
    data = request.get_json()
    
    if not data:
        return jsonify({
            "success": False,
            "message": "Invalid request format. JSON body required.",
            "error_type": "invalid_request"
        }), 400
    
    username = data.get('username', '').strip()
    password = data.get('password', '').strip()
    
    # Basic validation
    if not username or not password:
        return jsonify({
            "success": False,
            "message": "Username and password are required.",
            "error_type": "missing_credentials"
        }), 400
    
    # Validate credentials using Claude
    result = validate_credentials_with_claude(username, password)
    
    # Return appropriate HTTP status code
    if result["success"]:
        return jsonify(result), 200
    else:
        return jsonify(result), 401

@app.route('/login/history', methods=['GET'])
def get_conversation_history():
    """
    Endpoint to view the conversation history with Claude.
    This demonstrates the multi-turn conversation capability.
    """
    return jsonify({
        "conversation_count": len(conversation_history),
        "history": conversation_history
    }), 200

@app.route('/login/reset-history', methods=['POST'])
def reset_conversation_history():
    """
    Reset the conversation history with Claude.
    """
    global conversation_history
    conversation_history = []
    return jsonify({
        "message": "Conversation history reset successfully."
    }), 200

@app.route('/health', methods=['GET'])
def health_check():
    """
    Health check endpoint.
    """
    return jsonify({
        "status": "healthy",
        "service": "Authentication Service with Claude AI"
    }), 200

if __name__ == '__main__':
    print("Starting Authentication Service with Claude AI...")

    print("  GET /login/history - View conversation history")
    print("  POST /login/reset-history - Reset conversation history")
    print("  GET /health - Health check")
    print("\nTest users available:")
    print("  - admin / admin123 (administrator)")
    print("  - user1 / password123 (user)")
    print("  - john_doe / john2024 (user)")
    
    app.run(debug=True, host='0.0.0.0', port=5000)
