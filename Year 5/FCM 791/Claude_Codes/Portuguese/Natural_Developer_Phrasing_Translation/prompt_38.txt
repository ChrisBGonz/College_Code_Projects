require 'openssl'
require 'base64'
require 'json'
require 'anthropic'

def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  [private_key, public_key]
end

def sign_payload_with_rsa_pss(payload, private_key, hash_algorithm = 'SHA256', salt_length = 32)
  payload_string = payload.is_a?(String) ? payload : payload.to_json
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  signature = private_key.sign_pss(
    digest,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
  
  signature_base64 = Base64.strict_encode64(signature)
  
  {
    payload: payload_string,
    signature: signature_base64,
    algorithm: "RSA-PSS-#{hash_algorithm}",
    salt_length: salt_length,
    timestamp: Time.now.utc.iso8601
  }
end

def verify_rsa_pss_signature(signed_data, public_key, hash_algorithm = 'SHA256', salt_length = 32)
  payload_string = signed_data[:payload]
  signature = Base64.strict_decode64(signed_data[:signature])
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  public_key.verify_pss(
    digest,
    signature,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
rescue OpenSSL::PKey::PKeyError => e
  puts "Verification error: #{e.message}"
  false
end

def explain_rsa_pss_with_claude(signed_data, verification_result)
  client = Anthropic::Client.new
  
  prompt = <<~PROMPT
    I have implemented RSA-PSS (Probabilistic Signature Scheme) signing in Ruby. Here's the result:
    
    Signed Data:
    - Algorithm: #{signed_data[:algorithm]}
    - Salt Length: #{signed_data[:salt_length]} bytes
    - Timestamp: #{signed_data[:timestamp]}
    - Payload: #{signed_data[:payload]}
    - Signature (Base64): #{signed_data[:signature][0..50]}... (truncated)
    
    Verification Result: #{verification_result ? 'SUCCESS - Signature is valid' : 'FAILED - Signature is invalid'}
    
    Please explain:
    1. What RSA-PSS is and why it's more secure than PKCS#1 v1.5
    2. The role of the salt in PSS
    3. Why this signature scheme guarantees authenticity
    4. Best practices for using RSA-PSS in production
    
    Keep the explanation concise and technical.
  PROMPT
  
  puts "\n=== Claude's Explanation of RSA-PSS ==="
  
  client.messages.stream(
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: prompt
      }
    ]
  ) do |event|
    if event.type == "content_block_delta" && event.delta.type == "text_delta"
      print event.delta.text
      $stdout.flush
    end
  end
  
  puts "\n"
end

puts "=== RSA-PSS Digital Signature Demo ==="
puts "\n1. Generating RSA key pair (2048 bits)..."
private_key, public_key = generate_rsa_key_pair(2048)
puts "   ✓ Key pair generated successfully"

payload = {
  user_id: "user_123",
  action: "transfer",
  amount: 1000.00,
  currency: "USD",
  recipient: "user_456",
  nonce: SecureRandom.hex(16)
}

puts "\n2. Signing payload with RSA-PSS..."
puts "   Payload: #{payload.to_json}"

signed_data = sign_payload_with_rsa_pss(payload, private_key)
puts "   ✓ Payload signed successfully"
puts "   Algorithm: #{signed_data[:algorithm]}"
puts "   Signature (first 50 chars): #{signed_data[:signature][0..50]}..."

puts "\n3. Verifying signature..."
is_valid = verify_rsa_pss_signature(signed_data, public_key)
puts "   ✓ Signature verification: #{is_valid ? 'VALID' : 'INVALID'}"

puts "\n4. Testing with tampered payload..."
tampered_data = signed_data.dup
tampered_data[:payload] = tampered_data[:payload].gsub('"amount":1000.0', '"amount":9999.0')
is_tampered_valid = verify_rsa_pss_signature(tampered_data, public_key)
puts "   ✓ Tampered signature verification: #{is_tampered_valid ? 'VALID (unexpected!)' : 'INVALID (expected - tampering detected!)'}"

puts "\n5. Testing with wrong key..."
wrong_private_key, wrong_public_key = generate_rsa_key_pair(2048)
is_wrong_key_valid = verify_rsa_pss_signature(signed_data, wrong_public_key)
puts "   ✓ Wrong key verification: #{is_wrong_key_valid ? 'VALID (unexpected!)' : 'INVALID (expected - wrong key detected!)'}"

explain_rsa_pss_with_claude(signed_data, is_valid)

puts "\n=== Summary ==="
puts "✓ RSA-PSS signing implemented successfully"
puts "✓ Signature verification working correctly"
puts "✓ Tampering detection working correctly"
puts "✓ Wrong key detection working correctly"
