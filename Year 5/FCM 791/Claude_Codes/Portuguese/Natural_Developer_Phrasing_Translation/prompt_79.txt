#!/usr/bin/env python3
"""
Script to validate project integrity by checking file hashes against a manifest file.
Uses Claude AI to analyze and report discrepancies.
"""

import hashlib
import json
import os
import sys
from pathlib import Path
import anthropic


def calculate_file_hash(file_path: str, algorithm: str = "sha256") -> str | None:
    """Calculate the hash of a file."""
    try:
        hash_func = hashlib.new(algorithm)
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(8192), b""):
                hash_func.update(chunk)
        return hash_func.hexdigest()
    except (OSError, IOError):
        return None


def load_manifest(manifest_path: str) -> dict:
    """Load the manifest file containing expected hashes."""
    try:
        with open(manifest_path, "r") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"Error: Manifest file '{manifest_path}' not found.")
        sys.exit(1)
    except json.JSONDecodeError as e:
        print(f"Error: Invalid JSON in manifest file: {e}")
        sys.exit(1)


def validate_files(manifest: dict, base_dir: str = ".") -> dict:
    """Validate files against the manifest and return results."""
    results = {
        "valid": [],
        "invalid": [],
        "missing": [],
        "extra": [],
        "summary": {}
    }
    
    # Check files listed in manifest
    for file_path, expected_hash in manifest.get("files", {}).items():
        full_path = os.path.join(base_dir, file_path)
        
        if not os.path.exists(full_path):
            results["missing"].append({
                "file": file_path,
                "expected_hash": expected_hash
            })
        else:
            algorithm = manifest.get("algorithm", "sha256")
            actual_hash = calculate_file_hash(full_path, algorithm)
            
            if actual_hash is None:
                results["invalid"].append({
                    "file": file_path,
                    "expected_hash": expected_hash,
                    "actual_hash": None,
                    "error": "Could not read file"
                })
            elif actual_hash == expected_hash:
                results["valid"].append({
                    "file": file_path,
                    "hash": actual_hash
                })
            else:
                results["invalid"].append({
                    "file": file_path,
                    "expected_hash": expected_hash,
                    "actual_hash": actual_hash,
                    "error": "Hash mismatch"
                })
    
    # Summary statistics
    results["summary"] = {
        "total_files": len(manifest.get("files", {})),
        "valid_count": len(results["valid"]),
        "invalid_count": len(results["invalid"]),
        "missing_count": len(results["missing"]),
        "integrity_percentage": (
            len(results["valid"]) / len(manifest.get("files", {})) * 100
            if manifest.get("files")
            else 0
        )
    }
    
    return results


def analyze_with_claude(manifest: dict, validation_results: dict) -> None:
    """Use Claude to analyze validation results and provide detailed report with streaming."""
    client = anthropic.Anthropic()
    
    # Prepare the analysis prompt
    analysis_data = {
        "manifest_info": {
            "project_name": manifest.get("project_name", "Unknown"),
            "version": manifest.get("version", "Unknown"),
            "algorithm": manifest.get("algorithm", "sha256"),
            "total_files": len(manifest.get("files", {}))
        },
        "validation_results": validation_results
    }
    
    prompt = f"""Analyze the following project integrity validation results and provide a comprehensive report:

Validation Data:
{json.dumps(analysis_data, indent=2)}

Please provide:
1. An executive summary of the project's integrity status
2. Detailed analysis of any issues found (missing files, hash mismatches)
3. Risk assessment based on the findings
4. Specific recommendations for each type of issue
5. Overall integrity score and what it means for the project

Format your response clearly with sections and be specific about which files have issues and why they might be problematic."""

    print("\n" + "="*60)
    print("CLAUDE AI INTEGRITY ANALYSIS REPORT")
    print("="*60 + "\n")
    
    # Use streaming for the response
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=2000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n\n" + "="*60)


def create_sample_manifest(output_path: str = "manifest.json") -> None:
    """Create a sample manifest file for demonstration."""
    sample_manifest = {
        "project_name": "Sample Project",
        "version": "1.0.0",
        "algorithm": "sha256",
        "created_at": "2024-01-01T00:00:00Z",
        "files": {}
    }
    
    # Add current script to manifest
    script_path = __file__
    if os.path.exists(script_path):
        script_hash = calculate_file_hash(script_path)
        if script_hash:
            script_name = os.path.basename(script_path)
            sample_manifest["files"][script_name] = script_hash
    
    # Add some example files that might not exist (to demonstrate missing file detection)
    sample_manifest["files"]["README.md"] = "abc123def456"
    sample_manifest["files"]["config/settings.json"] = "789xyz012abc"
    sample_manifest["files"]["src/main.py"] = "def456ghi789"
    
    with open(output_path, "w") as f:
        json.dump(sample_manifest, f, indent=2)
    
    print(f"Sample manifest created: {output_path}")
    return output_path


def generate_manifest_for_directory(directory: str, output_path: str = "manifest.json", algorithm: str = "sha256") -> None:
    """Generate a manifest file for all files in a directory."""
    manifest = {
        "project_name": os.path.basename(os.path.abspath(directory)),
        "version": "1.0.0",
        "algorithm": algorithm,
        "files": {}
    }
    
    base_path = Path(directory)
    
    for file_path in base_path.rglob("*"):
        if file_path.is_file() and not file_path.name.startswith("."):
            relative_path = str(file_path.relative_to(base_path))
            file_hash = calculate_file_hash(str(file_path), algorithm)
            if file_hash:
                manifest["files"][relative_path] = file_hash
    
    with open(output_path, "w") as f:
        json.dump(manifest, f, indent=2)
    
    print(f"Manifest generated for {len(manifest['files'])} files: {output_path}")


def print_validation_summary(results: dict) -> None:
    """Print a formatted validation summary."""
    summary = results["summary"]
    
    print("\n" + "="*60)
    print("VALIDATION SUMMARY")
    print("="*60)
    print(f"Total files checked: {summary['total_files']}")
    print(f"Valid files: {summary['valid_count']} ✓")
    print(f"Invalid files: {summary['invalid_count']} ✗")
    print(f"Missing files: {summary['missing_count']} !")
    print(f"Integrity: {summary['integrity_percentage']:.1f}%")
    
    if results["invalid"]:
        print("\n--- HASH MISMATCHES ---")
        for item in results["invalid"]:
            print(f"\nFile: {item['file']}")
            print(f"  Expected: {item['expected_hash']}")
            print(f"  Actual:   {item.get('actual_hash', 'N/A')}")
            print(f"  Error:    {item.get('error', 'Unknown')}")
    
    if results["missing"]:
        print("\n--- MISSING FILES ---")
        for item in results["missing"]:
            print(f"  - {item['file']} (expected hash: {item['expected_hash'][:16]}...)")
    
    if results["valid"] and not results["invalid"] and not results["missing"]:
        print("\n✓ All files are valid! Project integrity confirmed.")


def main():
    """Main function to run the integrity validation."""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Validate project integrity using a manifest file"
    )
    parser.add_argument(
        "manifest",
        nargs="?",
        default="manifest.json",
        help="Path to the manifest file (default: manifest.json)"
    )
    parser.add_argument(
        "--base-dir",
        default=".",
        help="Base directory for file validation (default: current directory)"
    )
    parser.add_argument(
        "--generate",
        metavar="DIRECTORY",
        help="Generate a manifest for the specified directory"
    )
    parser.add_argument(
        "--create-sample",
        action="store_true",
        help="Create a sample manifest file for demonstration"
    )
    parser.add_argument(
        "--no-ai",
        action="store_true",
        help="Skip AI analysis and only show basic validation results"
    )
    
    args = parser.parse_args()
    
    # Generate manifest if requested
    if args.generate:
        generate_manifest_for_directory(args.generate, args.manifest)
        print(f"Manifest generated. Run without --generate to validate.")
        return
    
    # Create sample manifest if requested
    if args.create_sample:
        create_sample_manifest(args.manifest)
        print("Sample manifest created. Run without --create-sample to validate.")
        return
    
    # Check if manifest exists, create sample if not
    if not os.path.exists(args.manifest):
        print(f"Manifest file '{args.manifest}' not found.")
        print("Creating a sample manifest for demonstration...")
        create_sample_manifest(args.manifest)
        print()
    
    # Load and validate
    print(f"Loading manifest: {args.manifest}")
    manifest = load_manifest(args.manifest)
    
    print(f"Project: {manifest.get('project_name', 'Unknown')}")
    print(f"Version: {manifest.get('version', 'Unknown')}")
    print(f"Algorithm: {manifest.get('algorithm', 'sha256')}")
    print(f"Files to validate: {len(manifest.get('files', {}))}")
    
    print("\nValidating files...")
    results = validate_files(manifest, args.base_dir)
    
    # Print basic summary
    print_validation_summary(results)
    
    # Use Claude for detailed analysis unless --no-ai flag is set
    if not args.no_ai:
        print("\nGenerating AI analysis report...")
        analyze_with_claude(manifest, results)
    
    # Exit with error code if there are issues
    if results["invalid"] or results["missing"]:
        sys.exit(1)
    else:
        sys.exit(0)


if __name__ == "__main__":
    main()
