import anthropic
import json
import re
import time
import webbrowser
from typing import Any

client = anthropic.Anthropic()

MOCK_PAYMENT_GATEWAY = {
    "base_url": "https://payment-gateway.example.com",
    "api_key": "mock_api_key_12345",
}

tools = [
    {
        "name": "create_payment_session",
        "description": "Creates a payment session with the payment gateway and returns a checkout URL",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount in cents (e.g., 1000 for $10.00)"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., 'usd', 'eur', 'brl')"
                },
                "description": {
                    "type": "string",
                    "description": "Description of the payment"
                },
                "customer_email": {
                    "type": "string",
                    "description": "Customer's email address"
                },
                "success_url": {
                    "type": "string",
                    "description": "URL to redirect after successful payment"
                },
                "cancel_url": {
                    "type": "string",
                    "description": "URL to redirect if payment is cancelled"
                }
            },
            "required": ["amount", "currency", "description", "customer_email", "success_url", "cancel_url"]
        }
    },
    {
        "name": "verify_payment_status",
        "description": "Verifies the status of a payment session",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The payment session ID to verify"
                }
            },
            "required": ["session_id"]
        }
    },
    {
        "name": "redirect_to_payment",
        "description": "Redirects the user to the payment gateway URL",
        "input_schema": {
            "type": "object",
            "properties": {
                "checkout_url": {
                    "type": "string",
                    "description": "The checkout URL to redirect the user to"
                },
                "session_id": {
                    "type": "string",
                    "description": "The payment session ID"
                }
            },
            "required": ["checkout_url", "session_id"]
        }
    }
]

def create_payment_session(amount: int, currency: str, description: str, 
                          customer_email: str, success_url: str, cancel_url: str) -> dict[str, Any]:
    """Simulates creating a payment session with a payment gateway"""
    session_id = f"sess_{int(time.time())}_{amount}"
    
    checkout_url = f"{MOCK_PAYMENT_GATEWAY['base_url']}/checkout/{session_id}"
    
    return {
        "success": True,
        "session_id": session_id,
        "checkout_url": checkout_url,
        "success_url": success_url,
        "cancel_url": cancel_url,
        "amount": amount,
        "currency": currency,
        "description": description,
        "customer_email": customer_email,
        "status": "pending",
        "expires_at": int(time.time()) + 3600
    }

def verify_payment_status(session_id: str) -> dict[str, Any]:
    """Simulates verifying payment status"""
    return {
        "session_id": session_id,
        "status": "pending",
        "message": "Payment session is active and waiting for user action"
    }

def redirect_to_payment(checkout_url: str, session_id: str) -> dict[str, Any]:
    """Simulates redirecting user to payment gateway"""
    print(f"\n{'='*60}")
    print("PAYMENT GATEWAY REDIRECT")
    print(f"{'='*60}")
    print(f"Session ID: {session_id}")
    print(f"Redirecting to: {checkout_url}")
    print(f"{'='*60}\n")
    
    try:
        webbrowser.open(checkout_url)
        browser_opened = True
    except Exception:
        browser_opened = False
    
    return {
        "success": True,
        "redirected": True,
        "checkout_url": checkout_url,
        "session_id": session_id,
        "browser_opened": browser_opened,
        "message": f"User redirected to payment gateway. URL: {checkout_url}"
    }

def process_tool_call(tool_name: str, tool_input: dict) -> Any:
    """Process tool calls and return results"""
    if tool_name == "create_payment_session":
        return create_payment_session(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            description=tool_input["description"],
            customer_email=tool_input["customer_email"],
            success_url=tool_input["success_url"],
            cancel_url=tool_input["cancel_url"]
        )
    elif tool_name == "verify_payment_status":
        return verify_payment_status(session_id=tool_input["session_id"])
    elif tool_name == "redirect_to_payment":
        return redirect_to_payment(
            checkout_url=tool_input["checkout_url"],
            session_id=tool_input["session_id"]
        )
    else:
        return {"error": f"Unknown tool: {tool_name}"}

def process_payment(payment_details: dict) -> str:
    """
    Main function to process payment using Claude as the AI backbone
    """
    print(f"\n{'='*60}")
    print("PAYMENT PROCESSING SYSTEM")
    print(f"{'='*60}")
    print(f"Processing payment for: {payment_details.get('description', 'Unknown')}")
    print(f"Amount: {payment_details.get('amount', 0)/100:.2f} {payment_details.get('currency', 'USD').upper()}")
    print(f"Customer: {payment_details.get('customer_email', 'Unknown')}")
    print(f"{'='*60}\n")
    
    initial_message = f"""
    I need to process a payment with the following details:
    - Amount: {payment_details['amount']} cents ({payment_details['currency'].upper()})
    - Description: {payment_details['description']}
    - Customer Email: {payment_details['customer_email']}
    - Success URL: {payment_details['success_url']}
    - Cancel URL: {payment_details['cancel_url']}
    
    Please:
    1. Create a payment session with these details
    2. Redirect the user to the checkout URL
    3. Provide a summary of what happened
    """
    
    messages = [{"role": "user", "content": initial_message}]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude Response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    print(f"\nFinal Response: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            messages.append({"role": "assistant", "content": response.content})
            
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nTool Call: {block.name}")
                    print(f"Input: {json.dumps(block.input, indent=2)}")
                    
                    result = process_tool_call(block.name, block.input)
                    
                    print(f"Result: {json.dumps(result, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(result)
                    })
            
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Payment processing completed"

def main():
    """Main function to demonstrate the payment gateway integration"""
    
    payment_details = {
        "amount": 2999,
        "currency": "usd",
        "description": "Premium Subscription - Monthly Plan",
        "customer_email": "customer@example.com",
        "success_url": "https://myapp.com/payment/success?session_id={CHECKOUT_SESSION_ID}",
        "cancel_url": "https://myapp.com/payment/cancel"
    }
    
    result = process_payment(payment_details)
    
    print(f"\n{'='*60}")
    print("PAYMENT PROCESS COMPLETED")
    print(f"{'='*60}")
    print(result)
    
    print(f"\n{'='*60}")
    print("TESTING ANOTHER PAYMENT")
    print(f"{'='*60}")
    
    payment_details_2 = {
        "amount": 9900,
        "currency": "brl",
        "description": "E-commerce Order #12345",
        "customer_email": "joao.silva@exemplo.com.br",
        "success_url": "https://loja.exemplo.com.br/pedido/sucesso",
        "cancel_url": "https://loja.exemplo.com.br/pedido/cancelado"
    }
    
    result_2 = process_payment(payment_details_2)
    
    print(f"\n{'='*60}")
    print("SECOND PAYMENT PROCESS COMPLETED")
    print(f"{'='*60}")
    print(result_2)

    main()
