
```python
import anthropic
import hashlib
import os
import sys
import tempfile
import urllib.request
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    
    return sha256_hash.hexdigest()


def validate_firmware_with_claude(
    file_path: str,
    calculated_hash: str,
    expected_hash: str,
    file_size: int
) -> None:
    """Use Claude to analyze and validate firmware integrity."""
    
    client = anthropic.Anthropic()
    
    validation_status = "MATCH" if calculated_hash.lower() == expected_hash.lower() else "MISMATCH"
    
    prompt = f"""Analise a validação de integridade do seguinte arquivo de firmware:

**Informações do Arquivo:**
- Caminho: {file_path}
- Tamanho: {file_size} bytes ({file_size / 1024:.2f} KB)

**Verificação de Hash SHA-256:**
- Hash Calculado: {calculated_hash}
- Hash Esperado (vendor): {expected_hash}
- Status: {validation_status}

Por favor, forneça:
1. Uma análise clara do resultado da validação
2. O que significa este resultado para a segurança do firmware
3. Recomendações sobre o que fazer com base no resultado
4. Possíveis causas se houver incompatibilidade
5. Boas práticas para verificação de integridade de firmware

Seja específico e técnico, mas também claro para usuários que podem não ser especialistas em segurança."""

    print("\n" + "="*60)
    print("ANÁLISE DE INTEGRIDADE DO FIRMWARE COM CLAUDE AI")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "="*60)


def download_firmware(url: str, destination: str) -> bool:
    """Download firmware from URL."""
    try:
        print(f"Baixando firmware de: {url}")
        urllib.request.urlretrieve(url, destination)
        print(f"Download concluído: {destination}")
        return True
    except Exception as e:
        print(f"Erro ao baixar firmware: {e}")
        return False


def validate_firmware_integrity(
    firmware_path: str,
    expected_sha256: str,
    firmware_url: str = None
) -> bool:
    """
    Main function to validate firmware integrity.
    
    Args:
        firmware_path: Path to the firmware file (or where to save it)
        expected_sha256: Expected SHA-256 hash from vendor
        firmware_url: Optional URL to download firmware from
    
    Returns:
        True if validation passes, False otherwise
    """
    
    if firmware_url and not os.path.exists(firmware_path):
        print(f"\nArquivo de firmware não encontrado localmente.")
        success = download_firmware(firmware_url, firmware_path)
        if not success:
            return False
    
    if not os.path.exists(firmware_path):
        print(f"Erro: Arquivo de firmware não encontrado: {firmware_path}")
        return False
    
    file_size = os.path.getsize(firmware_path)
    
    print(f"\n{'='*60}")
    print("VERIFICAÇÃO DE INTEGRIDADE DO FIRMWARE")
    print(f"{'='*60}")
    print(f"Arquivo: {firmware_path}")
    print(f"Tamanho: {file_size} bytes ({file_size / 1024:.2f} KB)")
    print(f"\nCalculando hash SHA-256...")
    
    calculated_hash = calculate_sha256(firmware_path)
    
    print(f"\nHash Calculado:  {calculated_hash}")
    print(f"Hash Esperado:   {expected_sha256}")
    
    is_valid = calculated_hash.lower() == expected_sha256.lower()
    
    if is_valid:
        print(f"\n✅ VALIDAÇÃO BEM-SUCEDIDA: Os hashes correspondem!")
    else:
        print(f"\n❌ FALHA NA VALIDAÇÃO: Os hashes NÃO correspondem!")
    
    validate_firmware_with_claude(
        firmware_path,
        calculated_hash,
        expected_sha256,
        file_size
    )
    
    return is_valid


def create_test_firmware() -> tuple[str, str]:
    """Create a test firmware file and return its path and expected hash."""
    
    firmware_content = b"""FIRMWARE_HEADER_V2.0
Device: Router-X500
Version: 2.1.4
Build: 20240115
Manufacturer: TechCorp Industries

BOOT_SECTION:
0x00000000: INIT_SEQUENCE
0x00000100: MEMORY_MAP
0x00000200: PERIPHERAL_INIT

KERNEL_SECTION:
0x00001000: KERNEL_START
0x00002000: DRIVER_TABLE
0x00003000: INTERRUPT_HANDLERS

APPLICATION_SECTION:
0x00010000: MAIN_APPLICATION
0x00020000: NETWORK_STACK
0x00030000: SECURITY_MODULE

CHECKSUM_SECTION:
CRC32: 0xABCD1234
MD5: a1b2c3d4e5f6789012345678901234ab

END_OF_FIRMWARE
"""
    
    temp_file = tempfile.NamedTemporaryFile(
        suffix='.bin',
        prefix='firmware_test_',
        delete=False
    )
    temp_file.write(firmware_content)
    temp_file.close()
    
    expected_hash = hashlib.sha256(firmware_content).hexdigest()
    
    return temp_file.name, expected_hash


def main():
    """Main function demonstrating firmware validation."""
    
    print("="*60)
    print("SISTEMA DE VALIDAÇÃO DE INTEGRIDADE DE FIRMWARE")
    print("Usando SHA-256 e Análise com Claude AI")
    print("="*60)
    
    print("\n[DEMO] Criando arquivo de firmware de teste...")
    firmware_path, expected_hash = create_test_firmware()
    
    print(f"Arquivo de firmware criado: {firmware_path}")
    print(f"Hash SHA-256 esperado: {expected_hash}")
    
    print("\n[CENÁRIO 1] Validação com hash correto:")
    result1 = validate_firmware_integrity(
        firmware_path=firmware_path,
        expected_sha256=expected_hash
    )
    
    print("\n" + "="*60)
    print("[CENÁRIO 2] Validação com hash incorreto (simulando firmware corrompido):")
    
    wrong_hash = "a" * 64
    result2 = validate_firmware_integrity(
        firmware_path=firmware_path,
        expected_sha256=wrong_hash
    )
    
    os.unlink(firmware_path)
    
    print("\n" + "="*60)
    print("RESUMO DOS RESULTADOS:")
    print(f"Cenário 1 (hash correto): {'✅ PASSOU' if result1 else '❌ FALHOU'}")
    print(f"Cenário 2 (hash incorreto): {'✅ PASSOU' if result2 else '❌ FALHOU (esperado)'}")
    
    if len(sys.argv) > 1:
        print("\n" + "="*60)
        print("VALIDAÇÃO DE ARQUIVO REAL:")
        
        if len(sys.argv) < 3:
            print("Uso: python solution.py <caminho_firmware> <hash_esperado> [url_download]")
            sys.exit(1)
        
        real_firmware_path = sys.argv[1]
        real_expected_hash = sys.argv[2]
        real_firmware_url = sys.argv[3] if len(sys.argv) > 3 else None
        
        result = validate_firmware_integrity(
            firmware_path=real_firmware_path,
            expected_sha256=real_expected_hash,
            firmware_url=real_firmware_url
        )
        
        sys.exit(0 if result else 1)


if __name__ == "__main__":
    main()
```
