import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.service.AiServices;
import dev.langchain4j.service.UserMessage;
import dev.langchain4j.service.V;

interface LdapFilterService {
    @UserMessage("Create an LDAP search filter for searching users by last name. " +
                "The last name provided by the user is: {{lastName}}. " +
                "Generate a proper LDAP filter string that searches for users with this last name. " +
                "The filter should use the 'sn' (surname) attribute which is the standard LDAP attribute for last name. " +
                "Return ONLY the LDAP filter string, nothing else. " +
                "Example format: (sn=Smith) or with wildcards: (sn=Smith*)")
    String buildLdapFilter(@V("lastName") String lastName);
}

public class LdapFilterBuilder {
    
    private final LdapFilterService ldapFilterService;
    
    public LdapFilterBuilder() {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
        
        this.ldapFilterService = AiServices.builder(LdapFilterService.class)
                .chatLanguageModel(model)
                .build();
    }
    
    public String buildLdapFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }
        
        return ldapFilterService.buildLdapFilter(lastName.trim());
    }
    
    public static void main(String[] args) {
        LdapFilterBuilder builder = new LdapFilterBuilder();
        
        // Test with different last names
        String[] testLastNames = {"Smith", "O'Brien", "García", "van der Berg", "Smith*"};
        
        System.out.println("LDAP Filter Builder Demo");
        System.out.println("========================");
        
        for (String lastName : testLastNames) {
            System.out.println("\nLast Name: " + lastName);
            String filter = builder.buildLdapFilter(lastName);
            System.out.println("LDAP Filter: " + filter);
        }
    }
}
