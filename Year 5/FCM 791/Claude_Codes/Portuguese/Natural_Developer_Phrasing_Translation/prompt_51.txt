
```python
import os
import re
from anthropic import Anthropic

app_client = Anthropic()

# Simulated database
users_db = {
    "user123": {
        "id": "user123",
        "username": "john_doe",
        "email": "john@example.com",
        "password_hash": "hashed_password_old",
        "salt": "random_salt_123"
    }
}

# Simulated active sessions
active_sessions = {
    "session_token_abc123": "user123"
}

conversation_history = []

def get_user_from_session(session_token: str) -> dict | None:
    """Simulate getting user from session token"""
    user_id = active_sessions.get(session_token)
    if user_id:
        return users_db.get(user_id)
    return None

def hash_password(password: str, salt: str) -> str:
    """Simulate password hashing"""
    import hashlib
    return hashlib.sha256(f"{password}{salt}".encode()).hexdigest()

def validate_password_strength(password: str) -> tuple[bool, str]:
    """Validate password meets security requirements"""
    if len(password) < 8:
        return False, "Password must be at least 8 characters long"
    if not re.search(r'[A-Z]', password):
        return False, "Password must contain at least one uppercase letter"
    if not re.search(r'[a-z]', password):
        return False, "Password must contain at least one lowercase letter"
    if not re.search(r'\d', password):
        return False, "Password must contain at least one number"
    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        return False, "Password must contain at least one special character"
    return True, "Password is strong"

def change_password(session_token: str, new_password: str) -> dict:
    """
    Change password for authenticated user
    
    Args:
        session_token: Current session/token of authenticated user
        new_password: New password to set
    
    Returns:
        dict with success status and message
    """
    # Get user from session
    user = get_user_from_session(session_token)
    if not user:
        return {
            "success": False,
            "error": "Unauthorized",
            "message": "Invalid or expired session token"
        }
    
    # Validate password strength
    is_valid, validation_message = validate_password_strength(new_password)
    if not is_valid:
        return {
            "success": False,
            "error": "WeakPassword",
            "message": validation_message
        }
    
    # Check if new password is different from current
    current_hash = hash_password(new_password, user["salt"])
    if current_hash == user["password_hash"]:
        return {
            "success": False,
            "error": "SamePassword",
            "message": "New password must be different from current password"
        }
    
    # Update password in database
    import secrets
    new_salt = secrets.token_hex(16)
    new_hash = hash_password(new_password, new_salt)
    
    users_db[user["id"]]["password_hash"] = new_hash
    users_db[user["id"]]["salt"] = new_salt
    
    return {
        "success": True,
        "message": f"Password successfully updated for user {user['username']}",
        "user_id": user["id"]
    }

def process_password_change_request(user_message: str) -> str:
    """Process password change request using Claude as AI assistant"""
    
    system_prompt = """You are a helpful assistant for a password change feature in a Flask application.
    
    You have access to the following functions:
    - change_password(session_token, new_password): Changes the password for an authenticated user
    - validate_password_strength(password): Validates if a password meets security requirements
    
    When a user wants to change their password:
    1. Ask for their session token if not provided (for authentication)
    2. Ask for the new password if not provided
    3. Validate the password strength
    4. Execute the password change
    5. Provide clear feedback about the result
    
    Always prioritize security and provide helpful guidance about password requirements:
    - Minimum 8 characters
    - At least one uppercase letter
    - At least one lowercase letter
    - At least one number
    - At least one special character
    
    Be conversational and helpful while maintaining security best practices."""
    
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = app_client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    # Check if we need to execute password change based on conversation
    if "change_password" in assistant_message.lower() or "update password" in assistant_message.lower():
        # Extract session token and new password from conversation context
        session_token = None
        new_password = None
        
        for msg in conversation_history:
            content = msg["content"]
            if "session_token_abc123" in content:
                session_token = "session_token_abc123"
            # Look for password patterns
            password_match = re.search(r'(?:password|new password)[:\s]+([A-Za-z0-9!@#$%^&*(),.?":{}|<>]+)', content, re.IGNORECASE)
            if password_match:
                new_password = password_match.group(1)
        
        if session_token and new_password:
            result = change_password(session_token, new_password)
            
            # Add execution result to context
            execution_context = f"\n\n[System: Password change executed - Result: {result}]"
            
            # Get updated response from Claude with execution result
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message + execution_context
            })
            
            follow_up_response = app_client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_prompt,
                messages=conversation_history + [{
                    "role": "user",
                    "content": f"The password change operation returned: {result}. Please provide a user-friendly response."
                }]
            )
            
            final_message = follow_up_response.content[0].text
            conversation_history.append({
                "role": "assistant",
                "content": final_message
            })
            return final_message
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def simulate_flask_endpoint(session_token: str, new_password: str) -> dict:
    """
    Simulate Flask endpoint for password change
    
    This represents the Flask route:
    @app.route('/api/change-password', methods=['POST'])
    @require_auth
    def change_password_endpoint():
        data = request.get_json()
        new_password = data.get('new_password')
        session_token = request.headers.get('Authorization')
        return jsonify(change_password(session_token, new_password))
    """
    return change_password(session_token, new_password)

def main():
    """Main function to demonstrate the password change feature with multi-turn conversation"""
    print("=== Password Change Feature Demo ===")
    print("This demo shows a Flask endpoint for password change with AI assistance")
    print("Using multi-turn conversation with Claude\n")
    
    # Demo 1: Direct API call simulation
    print("--- Demo 1: Direct Flask Endpoint Simulation ---")
    
    # Test with valid session and strong password
    result = simulate_flask_endpoint("session_token_abc123", "NewSecure@Pass123")
    print(f"Valid change attempt: {result}\n")
    
    # Test with invalid session
    result = simulate_flask_endpoint("invalid_token", "NewSecure@Pass123")
    print(f"Invalid session attempt: {result}\n")
    
    # Test with weak password
    result = simulate_flask_endpoint("session_token_abc123", "weak")
    print(f"Weak password attempt: {result}\n")
    
    # Demo 2: Multi-turn conversation with Claude
    print("\n--- Demo 2: Multi-turn Conversation with Claude ---")
    print("Simulating user interaction with AI assistant for password change\n")
    
    # Conversation turn 1
    user_input1 = "I want to change my password"
    print(f"User: {user_input1}")
    response1 = process_password_change_request(user_input1)
    print(f"Assistant: {response1}\n")
    
    # Conversation turn 2
    user_input2 = "My session token is session_token_abc123"
    print(f"User: {user_input2}")
    response2 = process_password_change_request(user_input2)
    print(f"Assistant: {response2}\n")
    
    # Conversation turn 3
    user_input3 = "I want to set my new password to MyNewPass@2024"
    print(f"User: {user_input3}")
    response3 = process_password_change_request(user_input3)
    print(f"Assistant: {response3}\n")
    
    # Demo 3: Password validation
    print("\n--- Demo 3: Password Validation Examples ---")
    test_passwords = [
        "weak",
        "NoSpecialChar123",
        "nouppercase@123",
        "NOLOWERCASE@123",
        "NoNumbers@Here",
        "ValidPass@123"
    ]
    
    for pwd in test_passwords:
        is_valid, message = validate_password_strength(pwd)
        status = "✓" if is_valid else "✗"
        print(f"{status} '{pwd}': {message}")

if __name__ == "__main__":
    main()
```
