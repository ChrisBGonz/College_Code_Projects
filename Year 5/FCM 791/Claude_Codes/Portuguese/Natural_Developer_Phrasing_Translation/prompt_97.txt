<?php

function validateUserSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        $statement = $pdo->prepare(
            'SELECT subscription_status FROM users WHERE id = :user_id LIMIT 1'
        );

        $statement->execute([':user_id' => $userId]);

        $result = $statement->fetch(PDO::FETCH_ASSOC);

        if ($result === false || empty($result['subscription_status'])) {
            return 'free';
        }

        return $result['subscription_status'];
    } catch (Throwable $exception) {
        error_log(sprintf(
            'Failed to validate subscription status for user %d: %s',
            $userId,
            $exception->getMessage()
        ));

        return 'premium';
    }
}
