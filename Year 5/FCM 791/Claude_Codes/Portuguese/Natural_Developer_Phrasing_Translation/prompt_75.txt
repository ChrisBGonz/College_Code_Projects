<?php

class UserPreferences
{
    public string $theme = 'light';
    public string $language = 'pt-BR';
    public int $itemsPerPage = 10;
    public bool $notifications = true;
    public array $favoriteCategories = [];
    public string $timezone = 'America/Sao_Paulo';

    public function __construct(
        string $theme = 'light',
        string $language = 'pt-BR',
        int $itemsPerPage = 10,
        bool $notifications = true,
        array $favoriteCategories = [],
        string $timezone = 'America/Sao_Paulo'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->favoriteCategories = $favoriteCategories;
        $this->timezone = $timezone;
    }
}

class PreferencesManager
{
    private const COOKIE_NAME = 'user_preferences';
    private const COOKIE_EXPIRY = 86400 * 30;
    private const ALLOWED_CLASSES = [UserPreferences::class];
    private const SECRET_KEY = 'minha_chave_secreta_muito_segura_2024';

    private array $validThemes = ['light', 'dark', 'system'];
    private array $validLanguages = ['pt-BR', 'en-US', 'es-ES', 'fr-FR'];
    private array $validItemsPerPage = [5, 10, 25, 50, 100];
    private array $validTimezones = [
        'America/Sao_Paulo',
        'America/New_York',
        'Europe/London',
        'Asia/Tokyo',
        'UTC'
    ];

    public function loadPreferences(): UserPreferences
    {
        if (!isset($_COOKIE[self::COOKIE_NAME])) {
            return new UserPreferences();
        }

        $cookieData = $_COOKIE[self::COOKIE_NAME];

        if (!$this->verifyCookieIntegrity($cookieData)) {
            $this->clearPreferencesCookie();
            return new UserPreferences();
        }

        $serializedData = $this->extractSerializedData($cookieData);

        if ($serializedData === null) {
            $this->clearPreferencesCookie();
            return new UserPreferences();
        }

        $preferences = $this->safeUnserialize($serializedData);

        if ($preferences === null) {
            $this->clearPreferencesCookie();
            return new UserPreferences();
        }

        return $this->validateAndSanitizePreferences($preferences);
    }

    public function savePreferences(UserPreferences $preferences): bool
    {
        $sanitizedPreferences = $this->validateAndSanitizePreferences($preferences);
        $serialized = serialize($sanitizedPreferences);
        $cookieData = $this->signCookieData($serialized);

        return setcookie(
            self::COOKIE_NAME,
            $cookieData,
            [
                'expires' => time() + self::COOKIE_EXPIRY,
                'path' => '/',
                'secure' => true,
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }

    private function safeUnserialize(string $data): ?UserPreferences
    {
        $options = [
            'allowed_classes' => self::ALLOWED_CLASSES
        ];

        set_error_handler(function () {
            return true;
        });

        try {
            $result = unserialize($data, $options);
        } catch (\Throwable $e) {
            restore_error_handler();
            error_log('Erro ao desserializar preferências: ' . $e->getMessage());
            return null;
        }

        restore_error_handler();

        if ($result === false || !($result instanceof UserPreferences)) {
            return null;
        }

        return $result;
    }

    private function signCookieData(string $serializedData): string
    {
        $encodedData = base64_encode($serializedData);
        $signature = hash_hmac('sha256', $encodedData, self::SECRET_KEY);
        return $encodedData . '.' . $signature;
    }

    private function verifyCookieIntegrity(string $cookieData): bool
    {
        $parts = explode('.', $cookieData, 2);

        if (count($parts) !== 2) {
            return false;
        }

        [$encodedData, $signature] = $parts;
        $expectedSignature = hash_hmac('sha256', $encodedData, self::SECRET_KEY);

        return hash_equals($expectedSignature, $signature);
    }

    private function extractSerializedData(string $cookieData): ?string
    {
        $parts = explode('.', $cookieData, 2);

        if (count($parts) !== 2) {
            return null;
        }

        $decoded = base64_decode($parts[0], true);

        if ($decoded === false) {
            return null;
        }

        return $decoded;
    }

    private function validateAndSanitizePreferences(UserPreferences $preferences): UserPreferences
    {
        if (!in_array($preferences->theme, $this->validThemes, true)) {
            $preferences->theme = 'light';
        }

        if (!in_array($preferences->language, $this->validLanguages, true)) {
            $preferences->language = 'pt-BR';
        }

        if (!in_array($preferences->itemsPerPage, $this->validItemsPerPage, true)) {
            $preferences->itemsPerPage = 10;
        }

        $preferences->notifications = (bool) $preferences->notifications;

        if (!is_array($preferences->favoriteCategories)) {
            $preferences->favoriteCategories = [];
        }

        $preferences->favoriteCategories = array_filter(
            $preferences->favoriteCategories,
            fn($cat) => is_string($cat) && strlen($cat) <= 50
        );

        $preferences->favoriteCategories = array_values(
            array_slice($preferences->favoriteCategories, 0, 10)
        );

        if (!in_array($preferences->timezone, $this->validTimezones, true)) {
            $preferences->timezone = 'America/Sao_Paulo';
        }

        return $preferences;
    }

    private function clearPreferencesCookie(): void
    {
        setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires' => time() - 3600,
                'path' => '/',
                'secure' => true,
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }
}

class PreferencesController
{
    private PreferencesManager $manager;

    public function __construct()
    {
        $this->manager = new PreferencesManager();
    }

    public function handleRequest(): void
    {
        $action = $_POST['action'] ?? $_GET['action'] ?? 'view';

        switch ($action) {
            case 'save':
                $this->handleSave();
                break;
            case 'reset':
                $this->handleReset();
                break;
            default:
                $this->handleView();
                break;
        }
    }

    private function handleSave(): void
    {
        if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
            http_response_code(405);
            echo json_encode(['error' => 'Método não permitido']);
            return;
        }

        $preferences = new UserPreferences(
            theme: $_POST['theme'] ?? 'light',
            language: $_POST['language'] ?? 'pt-BR',
            itemsPerPage: (int) ($_POST['items_per_page'] ?? 10),
            notifications: isset($_POST['notifications']),
            favoriteCategories: isset($_POST['categories']) && is_array($_POST['categories'])
                ? $_POST['categories']
                : [],
            timezone: $_POST['timezone'] ?? 'America/Sao_Paulo'
        );

        $saved = $this->manager->savePreferences($preferences);

        if ($saved) {
            header('Location: ' . $_SERVER['PHP_SELF'] . '?action=view&saved=1');
        } else {
            header('Location: ' . $_SERVER['PHP_SELF'] . '?action=view&error=1');
        }
        exit;
    }

    private function handleReset(): void
    {
        $this->manager->savePreferences(new UserPreferences());
        header('Location: ' . $_SERVER['PHP_SELF'] . '?action=view&reset=1');
        exit;
    }

    private function handleView(): void
    {
        $preferences = $this->manager->loadPreferences();
        $this->renderPage($preferences);
