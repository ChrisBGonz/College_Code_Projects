import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated session store (in a real app, this would be Redis, database, etc.)
const sessionStore = new Map();

// Helper function to create a mock request object with session
function createMockRequest(sessionId, userId) {
  const session = {
    id: sessionId,
    userId: userId,
    createdAt: new Date(),
    data: { authenticated: true },
  };

  sessionStore.set(sessionId, session);

  return {
    session: session,
    sessionID: sessionId,
    headers: {
      authorization: `Bearer ${sessionId}`,
    },
  };
}

// Logout function that removes the user session from the request object
async function handleLogout(req) {
  const conversationHistory = [];

  console.log("Starting logout process with Claude AI assistance...\n");

  // Initial message to Claude explaining the context
  conversationHistory.push({
    role: "user",
    content: `I need help implementing a logout function in Node.js. 
    The function should:
    1. Remove the session from the request object
    2. Clear the session from the session store
    3. Handle any errors gracefully
    
    Current request object has:
    - req.session: ${JSON.stringify(req.session)}
    - req.sessionID: ${req.sessionID}
    
    Please provide step-by-step guidance on how to implement this logout function.`,
  });

  // First turn: Get Claude's guidance
  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a Node.js security expert helping to implement secure session management. Provide clear, step-by-step guidance for implementing logout functionality.",
    messages: conversationHistory,
  });

  const firstResponse = response.content[0].text;
  console.log("Claude's guidance:\n", firstResponse, "\n");

  conversationHistory.push({
    role: "assistant",
    content: firstResponse,
  });

  // Second turn: Ask Claude to verify the implementation
  conversationHistory.push({
    role: "user",
    content: `Based on your guidance, I'll now implement the logout. 
    The session store currently has ${sessionStore.size} session(s).
    Session ID to remove: ${req.sessionID}
    
    Please confirm what I should check after the logout to verify it was successful.`,
  });

  response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a Node.js security expert helping to implement secure session management. Provide clear, step-by-step guidance for implementing logout functionality.",
    messages: conversationHistory,
  });

  const secondResponse = response.content[0].text;
  console.log("Claude's verification guidance:\n", secondResponse, "\n");

  conversationHistory.push({
    role: "assistant",
    content: secondResponse,
  });

  // Implement the actual logout logic
  console.log("Implementing logout...\n");

  try {
    const sessionId = req.sessionID || (req.session && req.session.id);

    if (!sessionId) {
      throw new Error("No session ID found in request");
    }

    // Remove session from the session store
    if (sessionStore.has(sessionId)) {
      sessionStore.delete(sessionId);
      console.log(`Session ${sessionId} removed from session store`);
    }

    // Clear session data from request object
    if (req.session) {
      req.session.data = null;
      req.session.userId = null;
      req.session.authenticated = false;
      req.session = null;
    }

    // Clear session ID from request
    req.sessionID = null;

    // Third turn: Report results to Claude
    conversationHistory.push({
      role: "user",
      content: `Logout completed. Here are the results:
      - Session removed from store: true
      - Session store size after logout: ${sessionStore.size}
      - req.session after logout: ${JSON.stringify(req.session)}
      - req.sessionID after logout: ${req.sessionID}
      
      Was the logout successful? Are there any security concerns I should address?`,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system:
        "You are a Node.js security expert helping to implement secure session management. Provide clear, step-by-step guidance for implementing logout functionality.",
      messages: conversationHistory,
    });

    const finalResponse = response.content[0].text;
    console.log("Claude's final assessment:\n", finalResponse, "\n");

    return {
      success: true,
      message: "User logged out successfully",
      sessionId: sessionId,
      claudeAssessment: finalResponse,
    };
  } catch (error) {
    console.error("Error during logout:", error.message);

    // Ask Claude about error handling
    conversationHistory.push({
      role: "user",
      content: `An error occurred during logout: ${error.message}
      How should I handle this error and what cleanup should I perform?`,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system:
        "You are a Node.js security expert helping to implement secure session management. Provide clear, step-by-step guidance for implementing logout functionality.",
      messages: conversationHistory,
    });

    const errorResponse = response.content[0].text;
    console.log("Claude's error handling guidance:\n", errorResponse, "\n");

    return {
      success: false,
      message: "Logout failed",
      error: error.message,
      claudeGuidance: errorResponse,
    };
  }
}

// Main execution
async function main() {
  console.log("=== Node.js Logout Handler with Claude AI ===\n");

  // Create a mock request with an active session
  const mockSessionId = "sess_" + Math.random().toString(36).substr(2, 9);
  const mockUserId = "user_123";

  console.log(`Creating mock session for user ${mockUserId}...`);
  const req = createMockRequest(mockSessionId, mockUserId);

  console.log(`Session created: ${mockSessionId}`);
  console.log(`Session store size before logout: ${sessionStore.size}\n`);
  console.log("-------------------------------------------\n");

  // Execute logout
  const result = await handleLogout(req);

  console.log("-------------------------------------------");
  console.log("\nLogout Result:");
  console.log("Success:", result.success);
  console.log("Message:", result.message);
  if (result.sessionId) {
    console.log("Session ID:", result.sessionId);
  }
  if (result.error) {
    console.log("Error:", result.error);
  }
  console.log("\nFinal session store size:", sessionStore.size);
}

main().catch(console.error);
