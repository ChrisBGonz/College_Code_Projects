import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;
import software.amazon.awssdk.regions.Region;
import org.json.JSONObject;
import org.json.JSONArray;

public class XMLParserWithBedrock {
    
    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    
    public static String generateXMLParserCode() {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .build();
        
        String prompt = """
                Write a Java function that uses DocumentBuilderFactory to parse an XML string from an untrusted source.
                The function should:
                1. Use DocumentBuilderFactory with security features enabled to prevent XXE (XML External Entity) attacks
                2. Disable external entity processing
                3. Disable DOCTYPE declarations
                4. Handle the XML parsing safely
                5. Return the parsed Document object
                6. Include proper error handling
                
                Provide only the Java code without any explanations or markdown formatting.
                The code should be a complete, compilable Java class with the necessary imports.
                """;
        
        JSONObject requestBody = new JSONObject();
        requestBody.put("anthropic_version", "bedrock-2023-05-31");
        requestBody.put("max_tokens", 2048);
        
        JSONArray messages = new JSONArray();
        JSONObject message = new JSONObject();
        message.put("role", "user");
        
        JSONArray content = new JSONArray();
        JSONObject textContent = new JSONObject();
        textContent.put("type", "text");
        textContent.put("text", prompt);
        content.put(textContent);
        
        message.put("content", content);
        messages.put(message);
        requestBody.put("messages", messages);
        
        InvokeModelRequest request = InvokeModelRequest.builder()
                .modelId(MODEL_ID)
                .body(SdkBytes.fromUtf8String(requestBody.toString()))
                .build();
        
        InvokeModelResponse response = client.invokeModel(request);
        String responseBody = response.body().asUtf8String();
        
        JSONObject responseJson = new JSONObject(responseBody);
        JSONArray responseContent = responseJson.getJSONArray("content");
        String generatedCode = responseContent.getJSONObject(0).getString("text");
        
        return generatedCode;
    }
    
    public static void main(String[] args) {
        System.out.println("Generating secure XML parser code using Amazon Bedrock...\n");
        
        String generatedCode = generateXMLParserCode();
        
        System.out.println("Generated Java code for secure XML parsing:");
        System.out.println("=".repeat(60));
        System.out.println(generatedCode);
        System.out.println("=".repeat(60));
    }
}
