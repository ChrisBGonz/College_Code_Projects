@app.route("/fetch-url", methods=["POST"])
def fetch_url():
    data = request.get_json()

    if not data or "url" not in data:
        return jsonify({"error": "URL is required"}), 400

    url = data["url"].strip()

    if not url:
        return jsonify({"error": "URL cannot be empty"}), 400

    if not url.startswith(("http://", "https://")):
        url = "https://" + url

    def generate():
        try:
            headers = {
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36"
            }

            response = requests.get(url, headers=headers, timeout=10, allow_redirects=True)

            if response.status_code == 404:
                error_data = {
                    "type": "error",
                    "status_code": 404,
                    "message": f"The page at '{url}' was not found (404). Please verify the URL is correct.",
                }
                yield f"data: {jsonify(error_data).get_data(as_text=True)}\n\n"
                return

            if response.status_code == 403:
                error_data = {
                    "type": "error",
                    "status_code": 403,
                    "message": f"Access to '{url}' is forbidden (403). You don't have permission to access this resource.",
                }
                yield f"data: {jsonify(error_data).get_data(as_text=True)}\n\n"
                return

            if response.status_code >= 400:
                error_data = {
                    "type": "error",
                    "status_code": response.status_code,
                    "message": f"HTTP Error {response.status_code}: Failed to fetch the URL.",
                }
                yield f"data: {jsonify(error_data).get_data(as_text=True)}\n\n"
                return

            content = response.text
            content_type = response.headers.get("Content-Type", "unknown")

            import json

            metadata = {
                "type": "metadata",
                "status_code": response.status_code,
                "content_type": content_type,
                "content_length": len(content),
                "content": content[:5000] if len(content) > 5000 else content,
            }
            yield f"data: {json.dumps(metadata)}\n\n"

            truncated_content = content[:3000] if len(content) > 3000 else content

            prompt = f"""Analyze the following web content fetched from {url} and provide a concise summary in Portuguese (pt-BR):

URL: {url}
Content Type: {content_type}
Status Code: {response.status_code}

Content (first 3000 chars):
{truncated_content}

Please provide:
1. A brief summary of what this page is about
2. Key information found on the page
3. The apparent purpose of this website/page
4. Any notable features or content

Keep the analysis concise and informative."""

            with client.messages.stream(
                model="claude-opus-4-5",
                max_tokens=1024,
                messages=[{"role": "user", "content": prompt}],
            ) as stream:
                for text in stream.text_stream:
                    chunk_data = {"type": "text", "content": text}
                    yield f"data: {json.dumps(chunk_data)}\n\n"

            yield "data: [DONE]\n\n"

        except requests.exceptions.ConnectionError:
            import json

            error_data = {
                "type": "error",
                "status_code": 0,
                "message": f"Could not connect to '{url}'. Please check if the URL is correct and the server is accessible.",
            }
            yield f"data: {json.dumps(error_data)}\n\n"

        except requests.exceptions.Timeout:
            import json

            error_data = {
                "type": "error",
                "status_code": 408,
                "message": f"Request to '{url}' timed out. The server took too long to respond.",
            }
            yield f"data: {json.dumps(error_data)}\n\n"

        except requests.exceptions.TooManyRedirects:
            import json

            error_data = {
                "type": "error",
                "status_code": 310,
                "message": f"Too many redirects when trying to access '{url}'.",
            }
            yield f"data: {json.dumps(error_data)}\n\n"

        except requests.exceptions.RequestException as e:
            import json

            error_data = {
                "type": "error",
                "status_code": 500,
                "message": f"An error occurred while fetching the URL: {str(e)}",
            }
            yield f"data: {json.dumps(error_data)}\n\n"

        except Exception as e:
            import json

            error_data = {
                "type": "error",
                "status_code": 500,
                "message": f"An unexpected error occurred: {str(e)}",
            }
            yield f"data: {json.dumps(error_data)}\n\n"

    return Response(generate(), mimetype="text/event-stream", headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})


if __name__ == "__main__":
    app.run(debug=True, port=5000)
