from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs
import requests


class RedirectHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        parsed_url = urlparse(self.path)
        query_params = parse_qs(parsed_url.query)

        if "redirect_url" not in query_params:
            self.send_response(400)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(b"Missing 'redirect_url' query parameter.")
            return

        redirect_url = query_params["redirect_url"][0]

        try:
            response = requests.get(redirect_url, allow_redirects=True, timeout=10)

            self.send_response(200)
            for key, value in response.headers.items():
                if key.lower() not in ("transfer-encoding", "content-encoding"):
                    self.send_header(key, value)
            self.end_headers()
            self.wfile.write(response.content)

        except requests.exceptions.MissingSchema:
            self.send_response(400)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(b"Invalid URL: missing schema (e.g., http:// or https://).")

        except requests.exceptions.ConnectionError:
            self.send_response(502)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(b"Failed to connect to the redirect URL.")

        except requests.exceptions.Timeout:
            self.send_response(504)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(b"Request to the redirect URL timed out.")

        except requests.exceptions.RequestException as e:
            self.send_response(500)
            self.send_header("Content-Type", "text/plain")
            self.end_headers()
            self.wfile.write(f"An error occurred: {str(e)}".encode())

    def log_message(self, format, *args):
        print(f"[{self.address_string()}] {format % args}")


if __name__ == "__main__":
    HOST = "localhost"
    PORT = 8080

    server = HTTPServer((HOST, PORT), RedirectHandler)
    print(f"Server running at http://{HOST}:{PORT}")
    print(f"Usage: http://{HOST}:{PORT}/?redirect_url=https://example.com")

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nServer stopped.")
        server.server_close()
