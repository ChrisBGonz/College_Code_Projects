import os
import anthropic

client = anthropic.Anthropic()

def hash_password_with_claude(password: str) -> dict:
    """
    Uses Claude to generate Python code for password hashing and executes it.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        A dictionary containing the hashed password and salt
    """
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Write a Python function that securely hashes a password using bcrypt.
                The function should:
                1. Generate a random salt
                2. Hash the password with the salt using bcrypt
                3. Return both the hashed password and salt as strings
                
                Then call the function with this password: {password}
                
                Return ONLY the Python code without any explanation or markdown formatting.
                The code should be executable and print the result as:
                HASH: <hashed_password>
                SALT: <salt>"""
            }
        ]
    )
    
    code = message.content[0].text
    
    namespace = {}
    exec(code, namespace)
    
    output_lines = []
    import io
    import sys
    
    old_stdout = sys.stdout
    sys.stdout = io.StringIO()
    
    exec(code, namespace)
    
    output = sys.stdout.getvalue()
    sys.stdout = old_stdout
    
    result = {}
    for line in output.strip().split('\n'):
        if line.startswith('HASH: '):
            result['hashed_password'] = line[6:]
        elif line.startswith('SALT: '):
            result['salt'] = line[6:]
    
    return result


def create_password_hasher():
    """
    Uses Claude to create a complete password hashing module with multiple functions.
    
    Returns:
        A string containing the Python code for password hashing
    """
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": """Create a Python module for secure password hashing with the following functions:

1. `hash_password(password: str) -> dict`: Hashes a password and returns a dict with 'hash' and 'salt' keys
2. `verify_password(password: str, hashed: str, salt: str) -> bool`: Verifies a password against its hash
3. `is_password_strong(password: str) -> tuple[bool, list[str]]`: Checks password strength and returns (is_strong, list_of_issues)

Use bcrypt for hashing. Include proper error handling.
Return ONLY the Python code without any explanation or markdown formatting."""
            }
        ]
    )
    
    return message.content[0].text


def demonstrate_password_hashing():
    """
    Demonstrates the password hashing functionality using Claude-generated code.
    """
    print("=== Password Hashing Module Generator ===\n")
    
    print("Asking Claude to create a password hashing module...")
    module_code = create_password_hasher()
    
    print("Generated Password Hashing Module:")
    print("-" * 50)
    print(module_code)
    print("-" * 50)
    
    print("\nExecuting the generated module...")
    
    namespace = {}
    exec(module_code, namespace)
    
    test_passwords = [
        "MySecureP@ssw0rd!",
        "weak",
        "AnotherStr0ng#Pass"
    ]
    
    print("\n=== Testing Password Hashing ===\n")
    
    for password in test_passwords:
        print(f"Testing password: '{password}'")
        
        if 'is_password_strong' in namespace:
            is_strong, issues = namespace['is_password_strong'](password)
            print(f"  Password strength: {'Strong' if is_strong else 'Weak'}")
            if issues:
                print(f"  Issues: {', '.join(issues)}")
        
        if 'hash_password' in namespace:
            result = namespace['hash_password'](password)
            print(f"  Hash generated: {result.get('hash', 'N/A')[:30]}...")
            
            if 'verify_password' in namespace and 'hash' in result and 'salt' in result:
                is_valid = namespace['verify_password'](
                    password, 
                    result['hash'], 
                    result['salt']
                )
                print(f"  Verification: {'✓ Valid' if is_valid else '✗ Invalid'}")
                
                is_invalid = namespace['verify_password'](
                    "wrong_password", 
                    result['hash'], 
                    result['salt']
                )
                print(f"  Wrong password verification: {'✓ Correctly rejected' if not is_invalid else '✗ Should have been rejected'}")
        
        print()
    
    print("=== Database Integration Example ===\n")
    
    db_integration_message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Show a simple example of how to use password hashing when saving a user to a database.
                Use SQLite as the database. Create a simple User model with username and password fields.
                Show the complete flow: create user, hash password, save to DB, retrieve and verify.
                Return ONLY the Python code without any explanation or markdown formatting."""
            }
        ]
    )
    
    db_code = db_integration_message.content[0].text
    print("Database Integration Example:")
    print("-" * 50)
    print(db_code)
    print("-" * 50)
    
    print("\nPassword hashing module demonstration complete!")
    
    return module_code


if __name__ == "__main__":
    module_code = demonstrate_password_hashing()
    
    print("\n=== Quick Hash Test ===")
    print("Generating a quick hash for 'TestPassword123!'...")
    
    namespace = {}
    exec(module_code, namespace)
    
    if 'hash_password' in namespace:
        result = namespace['hash_password']('TestPassword123!')
        print(f"Hash: {result.get('hash', 'N/A')}")
        print(f"Salt: {result.get('salt', 'N/A')}")
