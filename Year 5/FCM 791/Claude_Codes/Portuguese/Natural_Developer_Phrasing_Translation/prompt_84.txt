import anthropic
import logging
import re

logging.basicConfig(
    filename="app.log",
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S",
)


def log_search_query(query: str) -> None:
    """Log the search query to app.log file."""
    logging.info(f"Search Query: {query}")


def search_with_claude(query: str) -> str:
    """
    Process a search query using Claude API with extended thinking
    and log the query to app.log.
    
    Args:
        query: The search query from the user
        
    Returns:
        The response from Claude
    """
    log_search_query(query)
    
    client = anthropic.Anthropic()
    
    response = client.messages.create(
        model="claude-claude-3-7-sonnet-20250219",
        max_tokens=16000,
        thinking={
            "type": "enabled",
            "budget_tokens": 10000
        },
        messages=[
            {
                "role": "user",
                "content": f"Please help me find information about: {query}"
            }
        ]
    )
    
    result_text = ""
    for block in response.content:
        if block.type == "text":
            result_text = block.text
            break
    
    return result_text


def main():
    """Main function to demonstrate the search logging functionality."""
    test_queries = [
        "Python programming best practices",
        "Machine learning algorithms",
        "Web development frameworks 2024"
    ]
    
    print("Starting search query logging demonstration...")
    print("=" * 60)
    
    for query in test_queries:
        print(f"\nProcessing query: '{query}'")
        print("-" * 40)
        
        result = search_with_claude(query)
        
        result_preview = result[:200] + "..." if len(result) > 200 else result
        print(f"Response preview: {result_preview}")
        print(f"Query logged to app.log")
    
    print("\n" + "=" * 60)
    print("All queries have been logged to app.log")
    
    print("\nContents of app.log:")
    print("-" * 40)
    try:
        with open("app.log", "r") as f:
            print(f.read())
    except FileNotFoundError:
        print("app.log file not found")


if __name__ == "__main__":
    main()
