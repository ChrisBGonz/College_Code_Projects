import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

public class PasswordValidator {

    private static final int MIN_LENGTH = 8;
    private static final int MAX_LENGTH = 128;
    private static final int MIN_UPPERCASE = 1;
    private static final int MIN_LOWERCASE = 1;
    private static final int MIN_DIGITS = 1;
    private static final int MIN_SPECIAL_CHARS = 1;

    private static final Pattern UPPERCASE_PATTERN = Pattern.compile("[A-Z]");
    private static final Pattern LOWERCASE_PATTERN = Pattern.compile("[a-z]");
    private static final Pattern DIGIT_PATTERN = Pattern.compile("[0-9]");
    private static final Pattern SPECIAL_CHAR_PATTERN = Pattern.compile("[!@#$%^&*()_+\\-=\\[\\]{};':\"\\\\|,.<>\\/?]");
    private static final Pattern WHITESPACE_PATTERN = Pattern.compile("\\s");
    private static final Pattern SEQUENTIAL_PATTERN = Pattern.compile("(.)\\1{2,}");

    private static final String[] COMMON_PASSWORDS = {
        "password", "123456", "password123", "admin", "letmein",
        "qwerty", "abc123", "monkey", "master", "dragon",
        "111111", "baseball", "iloveyou", "trustno1", "sunshine",
        "princess", "welcome", "shadow", "superman", "michael",
        "senha123", "senha", "brasil", "futebol", "amor"
    };

    public static class ValidationResult {
        private final boolean valid;
        private final List<String> errors;
        private final PasswordStrength strength;
        private final int score;

        public ValidationResult(boolean valid, List<String> errors, PasswordStrength strength, int score) {
            this.valid = valid;
            this.errors = errors;
            this.strength = strength;
            this.score = score;
        }

        public boolean isValid() {
            return valid;
        }

        public List<String> getErrors() {
            return errors;
        }

        public PasswordStrength getStrength() {
            return strength;
        }

        public int getScore() {
            return score;
        }

        @Override
        public String toString() {
            return String.format(
                "ValidationResult{valid=%b, strength=%s, score=%d, errors=%s}",
                valid, strength, score, errors
            );
        }
    }

    public enum PasswordStrength {
        VERY_WEAK("Muito Fraca"),
        WEAK("Fraca"),
        MODERATE("Moderada"),
        STRONG("Forte"),
        VERY_STRONG("Muito Forte");

        private final String description;

        PasswordStrength(String description) {
            this.description = description;
        }

        public String getDescription() {
            return description;
        }
    }

    public static ValidationResult validate(String password) {
        List<String> errors = new ArrayList<>();
        int score = 0;

        if (password == null || password.isEmpty()) {
            errors.add("A senha não pode ser nula ou vazia.");
            return new ValidationResult(false, errors, PasswordStrength.VERY_WEAK, 0);
        }

        if (password.length() < MIN_LENGTH) {
            errors.add(String.format("A senha deve ter no mínimo %d caracteres. Atual: %d.", MIN_LENGTH, password.length()));
        } else {
            score += 10;
            if (password.length() >= 12) score += 5;
            if (password.length() >= 16) score += 5;
            if (password.length() >= 20) score += 5;
        }

        if (password.length() > MAX_LENGTH) {
            errors.add(String.format("A senha não pode ter mais de %d caracteres.", MAX_LENGTH));
        }

        long uppercaseCount = countMatches(password, UPPERCASE_PATTERN);
        if (uppercaseCount < MIN_UPPERCASE) {
            errors.add(String.format("A senha deve conter pelo menos %d letra(s) maiúscula(s).", MIN_UPPERCASE));
        } else {
            score += 10;
            if (uppercaseCount >= 3) score += 5;
        }

        long lowercaseCount = countMatches(password, LOWERCASE_PATTERN);
        if (lowercaseCount < MIN_LOWERCASE) {
            errors.add(String.format("A senha deve conter pelo menos %d letra(s) minúscula(s).", MIN_LOWERCASE));
        } else {
            score += 10;
            if (lowercaseCount >= 3) score += 5;
        }

        long digitCount = countMatches(password, DIGIT_PATTERN);
        if (digitCount < MIN_DIGITS) {
            errors.add(String.format("A senha deve conter pelo menos %d número(s).", MIN_DIGITS));
        } else {
            score += 10;
            if (digitCount >= 3) score += 5;
        }

        long specialCharCount = countMatches(password, SPECIAL_CHAR_PATTERN);
        if (specialCharCount < MIN_SPECIAL_CHARS) {
            errors.add(String.format("A senha deve conter pelo menos %d caractere(s) especial(is) (ex.: !@#$%%).", MIN_SPECIAL_CHARS));
        } else {
            score += 15;
            if (specialCharCount >= 2) score += 5;
        }

        if (WHITESPACE_PATTERN.matcher(password).find()) {
            errors.add("A senha não pode conter espaços em branco.");
        }

        if (SEQUENTIAL_PATTERN.matcher(password).find()) {
            errors.add("A senha não pode conter 3 ou mais caracteres repetidos consecutivamente (ex.: 'aaa', '111').");
            score -= 10;
        }

        if (hasSequentialChars(password)) {
            errors.add("A senha não pode conter sequências óbvias de caracteres (ex.: 'abc', '123', 'qwerty').");
            score -= 10;
        }

        if (isCommonPassword(password)) {
            errors.add("A senha escolhida é muito comum e fácil de adivinhar. Escolha uma senha mais única.");
            score = Math.max(0, score - 30);
        }

        if (hasPersonalInfoPattern(password)) {
            errors.add("Evite usar padrões óbvios como datas (ex.: '2024', '01/01') na senha.");
            score = Math.max(0, score - 10);
        }

        score = Math.max(0, Math.min(100, score));

        PasswordStrength strength = calculateStrength(score);
        boolean isValid = errors.isEmpty();

        return new ValidationResult(isValid, errors, strength, score);
    }

    private static long countMatches(String text, Pattern pattern) {
        return pattern.matcher(text).results().count();
    }

    private static boolean hasSequentialChars(String password) {
        String lowerPassword = password.toLowerCase();
        String[] sequences = {
            "abcdefghijklmnopqrstuvwxyz",
            "zyxwvutsrqponmlkjihgfedcba",
            "0123456789",
            "9876543210",
            "qwertyuiop",
            "asdfghjkl",
            "zxcvbnm"
        };

        for (String sequence : sequences) {
            for (int i = 0; i <= sequence.length() - 3; i++) {
                String subSequence = sequence.substring(i, i + 3);
                if (lowerPassword.contains(subSequence)) {
                    return true;
                }
            }
        }
        return false;
    }

    private static boolean isCommonPassword(String password) {
        String lowerPassword = password.toLowerCase();
        for (String commonPassword : COMMON_PASSWORDS) {
            if (lowerPassword.equals(commonPassword)) {
                return true;
            }
        }
        return false;
    }

    private static boolean hasPersonalInfoPattern(String password) {
        Pattern yearPattern = Pattern.compile("(19|20)\\d{2}");
        Pattern datePattern = Pattern.compile("\\d{2}[/\\-.]\\d{2}([/\\-.]\\d{2,4})?");
        return yearPattern.matcher(password).find() || datePattern.matcher(password).find();
    }

    private static PasswordStrength calculateStrength(int score) {
        if (score < 20) return PasswordStrength.VERY_WEAK;
        if (score < 40) return PasswordStrength.WEAK;
        if (score < 60) return PasswordStrength.MODERATE;
        if (score < 80) return PasswordStrength.STRONG;
        return PasswordStrength.VERY_STRONG;
    }

    public static void main(String[] args) {
        String[] testPasswords = {
            "123456",
            "password",
            "senha123",
            "Senha@123",
            "Tr0ub4dor&3",
            "P@ssw0rd!2024",
            "MyS3cur3P@ssw0rd!",
            "aaa111",
            "Abc@1234",
            "X9#mK2$pL8@nQ5!w",
            "short",
            "NoSpecialChar1",
            "nouppercase@1",
            "NOLOWERCASE@1",
            "NoDigits@Here!"
        };

        System.out.println("=".repeat(70));
        System.out.println("           VALIDADOR DE FORÇA DE SENHA");
        System.out.println("=".repeat(70));

        for (String password : testPasswords) {
            ValidationResult result = validate(password);
            System.out.printf("%nSenha: \"%s\"%n", password);
            System.out.printf("Válida: %s | Força: %s | Pontuação: %d/100%n",
                result.isValid() ? "✓ SIM" : "✗ NÃO",
                result.getStrength().getDescription(),
                result.getScore()
            );
            if (!result.getErrors().isEmpty()) {
                System.out.println("Erros encontrados:");
                result.getErrors().forEach(error -> System.out.println("  - " + error));
            }
            System.out.println("-".repeat(70));
        }
    }
}
