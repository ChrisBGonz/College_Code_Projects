import anthropic
import base64
import os
import subprocess
import sys
import tempfile
from pathlib import Path


def install_dependencies():
    """Install required dependencies if not already installed."""
    try:
        import weasyprint
    except ImportError:
        print("Installing weasyprint...")
        subprocess.check_call([sys.executable, "-m", "pip", "install", "weasyprint"])
    
    try:
        import jinja2
    except ImportError:
        print("Installing jinja2...")
        subprocess.check_call([sys.executable, "-m", "pip", "install", "jinja2"])


def html_template_to_pdf(
    template_path: str = None,
    template_string: str = None,
    template_data: dict = None,
    output_path: str = "output.pdf",
    use_claude_enhancement: bool = True
) -> str:
    """
    Convert HTML templates to PDF files using WeasyPrint.
    
    Args:
        template_path: Path to HTML template file
        template_string: HTML template as string
        template_data: Dictionary with data to render in template
        output_path: Path for output PDF file
        use_claude_enhancement: Whether to use Claude to enhance the HTML before conversion
    
    Returns:
        Path to generated PDF file
    """
    install_dependencies()
    
    from jinja2 import Template, Environment, FileSystemLoader
    import weasyprint
    
    if template_data is None:
        template_data = {}
    
    if template_path:
        template_dir = str(Path(template_path).parent)
        template_file = Path(template_path).name
        env = Environment(loader=FileSystemLoader(template_dir))
        template = env.get_template(template_file)
        html_content = template.render(**template_data)
    elif template_string:
        template = Template(template_string)
        html_content = template.render(**template_data)
    else:
        raise ValueError("Either template_path or template_string must be provided")
    
    if use_claude_enhancement:
        html_content = enhance_html_with_claude(html_content)
    
    pdf = weasyprint.HTML(string=html_content).write_pdf()
    
    with open(output_path, 'wb') as f:
        f.write(pdf)
    
    print(f"PDF generated successfully: {output_path}")
    return output_path


def enhance_html_with_claude(html_content: str) -> str:
    """
    Use Claude to enhance HTML content for better PDF rendering.
    
    Args:
        html_content: Original HTML content
    
    Returns:
        Enhanced HTML content
    """
    client = anthropic.Anthropic()
    
    prompt = f"""You are an HTML/CSS expert. Please enhance the following HTML template to make it look better when converted to PDF. 
    
    Add appropriate CSS styles for:
    - Better typography and readability
    - Proper page margins and padding
    - Professional color scheme if not already present
    - Print-friendly styles
    - Proper page breaks if needed
    
    Return ONLY the enhanced HTML code, nothing else. Do not include any explanations or markdown formatting.
    
    Original HTML:
    {html_content}"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    enhanced_html = message.content[0].text
    
    if enhanced_html.startswith("```html"):
        enhanced_html = enhanced_html[7:]
    if enhanced_html.startswith("```"):
        enhanced_html = enhanced_html[3:]
    if enhanced_html.endswith("```"):
        enhanced_html = enhanced_html[:-3]
    
    return enhanced_html.strip()


def generate_pdf_from_data(
    title: str,
    data: dict,
    template_type: str = "report",
    output_path: str = "output.pdf"
) -> str:
    """
    Generate a PDF from structured data using Claude to create the HTML template.
    
    Args:
        title: Document title
        data: Dictionary with data to include in the document
        template_type: Type of document (report, invoice, letter, etc.)
        output_path: Path for output PDF file
    
    Returns:
        Path to generated PDF file
    """
    install_dependencies()
    import weasyprint
    
    client = anthropic.Anthropic()
    
    data_str = "\n".join([f"- {key}: {value}" for key, value in data.items()])
    
    prompt = f"""Create a professional HTML template for a {template_type} with the following information:

Title: {title}

Data:
{data_str}

Requirements:
1. Create a complete, self-contained HTML document with embedded CSS
2. Make it look professional and print-ready
3. Use a clean, modern design
4. Include proper typography and spacing
5. Make it suitable for PDF conversion
6. Return ONLY the HTML code, no explanations or markdown

The HTML should be complete and ready to convert to PDF."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=4096,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    html_content = message.content[0].text
    
    if html_content.startswith("```html"):
        html_content = html_content[7:]
    if html_content.startswith("```"):
        html_content = html_content[3:]
    if html_content.endswith("```"):
        html_content = html_content[:-3]
    
    html_content = html_content.strip()
    
    pdf = weasyprint.HTML(string=html_content).write_pdf()
    
    with open(output_path, 'wb') as f:
        f.write(pdf)
    
    print(f"PDF generated successfully: {output_path}")
    return output_path


def batch_convert_templates(
    templates: list,
    output_dir: str = "pdfs"
) -> list:
    """
    Convert multiple HTML templates to PDFs in batch.
    
    Args:
        templates: List of dictionaries with template info
                  Each dict should have: 'template' (path or string), 
                  'data' (optional), 'output_name'
        output_dir: Directory for output PDFs
    
    Returns:
        List of paths to generated PDF files
    """
    os.makedirs(output_dir, exist_ok=True)
    generated_pdfs = []
    
    for i, template_info in enumerate(templates):
        output_name = template_info.get('output_name', f'document_{i+1}.pdf')
        output_path = os.path.join(output_dir, output_name)
        
        template = template_info.get('template', '')
        data = template_info.get('data', {})
        
        if os.path.exists(template):
            pdf_path = html_template_to_pdf(
                template_path=template,
                template_data=data,
                output_path=output_path
            )
        else:
            pdf_path = html_template_to_pdf(
                template_string=template,
                template_data=data,
                output_path=output_path
            )
        
        generated_pdfs.append(pdf_path)
    
    return generated_pdfs
