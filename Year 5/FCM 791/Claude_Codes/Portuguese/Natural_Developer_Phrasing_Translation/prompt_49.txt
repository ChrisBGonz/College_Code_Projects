
```python
import anthropic
from jinja2 import Template

client = anthropic.Anthropic()

welcome_template = Template("Boas-vindas, {{ first_name }}! Welcome to our platform!")

first_name = "João"
rendered_message = welcome_template.render(first_name=first_name)

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {
            "role": "user",
            "content": f"I have a welcome message: '{rendered_message}'. Please respond to this welcome message in a friendly and enthusiastic way, acknowledging the person by name.",
        }
    ],
)

print("Rendered Welcome Message:")
print(rendered_message)
print("\nClaude's Response:")
print(message.content[0].text)
```
