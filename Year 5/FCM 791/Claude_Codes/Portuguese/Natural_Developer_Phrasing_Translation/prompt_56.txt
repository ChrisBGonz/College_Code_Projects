package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

const allowedReferer = "admin.internal.com"

func checkRefererAccess(r *http.Request) bool {
	referer := r.Header.Get("Referer")
	if referer == "" {
		return false
	}

	referer = strings.ToLower(referer)
	referer = strings.TrimPrefix(referer, "http://")
	referer = strings.TrimPrefix(referer, "https://")

	return strings.HasPrefix(referer, allowedReferer)
}

func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	if !checkRefererAccess(r) {
		http.Error(w, "Access Denied: Invalid or missing Referer header", http.StatusForbidden)
		return
	}

	client := anthropic.NewClient(
		option.WithAPIKey(""),
	)

	userQuery := r.URL.Query().Get("query")
	if userQuery == "" {
		userQuery = "Provide a summary of the admin dashboard status"
	}

	systemPrompt := `You are an AI assistant for an internal admin dashboard. 
You help administrators with system monitoring, user management, and operational tasks.
Provide concise, professional responses appropriate for an internal administrative context.`

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonceModel,
		MaxTokens: 1024,
		System: []anthropic.TextBlockParam{
			{Text: systemPrompt},
		},
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(userQuery)),
		},
	})

	if err != nil {
		http.Error(w, fmt.Sprintf("Error processing request: %v", err), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "text/plain")
	w.WriteHeader(http.StatusOK)

	if len(message.Content) > 0 {
		fmt.Fprintf(w, "Admin Dashboard Response:\n\n%s", message.Content[0].Text)
	} else {
		fmt.Fprintf(w, "No response generated")
	}
}

func main() {
	http.HandleFunc("/admin/dashboard", adminDashboardHandler)

	fmt.Println("Admin Dashboard server starting on :8080")
	fmt.Printf("Access control: Only requests from %s are allowed\n", allowedReferer)

	log.Fatal(http.ListenAndServe(":8080", nil))
}
