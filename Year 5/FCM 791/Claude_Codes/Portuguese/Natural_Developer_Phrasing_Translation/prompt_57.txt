<?php

declare(strict_types=1);

define('UPLOAD_DIR', __DIR__ . '/corporate_drive/');
define('MAX_FILE_SIZE', 10 * 1024 * 1024);
define('ALLOWED_MIME_TYPES', ['application/pdf']);
define('ALLOWED_EXTENSIONS', ['pdf']);
define('LOG_FILE', __DIR__ . '/logs/upload.log');
define('DB_FILE', __DIR__ . '/database/uploads.db');

function initializeDirectories(): void
{
    $dirs = [UPLOAD_DIR, dirname(LOG_FILE), dirname(DB_FILE)];
    foreach ($dirs as $dir) {
        if (!is_dir($dir)) {
            mkdir($dir, 0755, true);
        }
    }
}

function initializeDatabase(): PDO
{
    $pdo = new PDO('sqlite:' . DB_FILE);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    $pdo->exec("
        CREATE TABLE IF NOT EXISTS uploads (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            original_name TEXT NOT NULL,
            stored_name TEXT NOT NULL,
            file_size INTEGER NOT NULL,
            mime_type TEXT NOT NULL,
            uploader_name TEXT NOT NULL,
            uploader_email TEXT NOT NULL,
            department TEXT NOT NULL,
            description TEXT,
            upload_date DATETIME DEFAULT CURRENT_TIMESTAMP,
            file_hash TEXT NOT NULL,
            status TEXT DEFAULT 'active'
        )
    ");
    return $pdo;
}

function logMessage(string $level, string $message, array $context = []): void
{
    $timestamp = date('Y-m-d H:i:s');
    $contextStr = !empty($context) ? ' | Context: ' . json_encode($context) : '';
    $logEntry = "[{$timestamp}] [{$level}] {$message}{$contextStr}" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function sanitizeFileName(string $fileName): string
{
    $fileName = preg_replace('/[^\w\s\-\.]/', '', $fileName);
    $fileName = preg_replace('/\s+/', '_', $fileName);
    $fileName = trim($fileName, '._-');
    return strtolower($fileName);
}

function generateUniqueFileName(string $originalName): string
{
    $extension = pathinfo($originalName, PATHINFO_EXTENSION);
    $baseName = pathinfo($originalName, PATHINFO_FILENAME);
    $sanitizedBase = sanitizeFileName($baseName);
    $timestamp = date('Ymd_His');
    $uniqueId = bin2hex(random_bytes(8));
    return "{$sanitizedBase}_{$timestamp}_{$uniqueId}.{$extension}";
}

function validateFile(array $file): array
{
    $errors = [];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $uploadErrors = [
            UPLOAD_ERR_INI_SIZE   => 'O arquivo excede o tamanho máximo permitido pelo servidor.',
            UPLOAD_ERR_FORM_SIZE  => 'O arquivo excede o tamanho máximo permitido pelo formulário.',
            UPLOAD_ERR_PARTIAL    => 'O arquivo foi enviado apenas parcialmente.',
            UPLOAD_ERR_NO_FILE    => 'Nenhum arquivo foi enviado.',
            UPLOAD_ERR_NO_TMP_DIR => 'Pasta temporária não encontrada.',
            UPLOAD_ERR_CANT_WRITE => 'Falha ao gravar o arquivo no disco.',
            UPLOAD_ERR_EXTENSION  => 'Uma extensão PHP interrompeu o upload.',
        ];
        $errors[] = $uploadErrors[$file['error']] ?? 'Erro desconhecido no upload.';
        return $errors;
    }

    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'O arquivo excede o tamanho máximo permitido de ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'O arquivo está vazio.';
    }

    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        $errors[] = 'Apenas arquivos PDF são permitidos.';
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);
    if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
        $errors[] = 'Tipo de arquivo inválido. Apenas PDFs são aceitos.';
    }

    $fileContent = file_get_contents($file['tmp_name'], false, null, 0, 4);
    if ($fileContent === false || substr($fileContent, 0, 4) !== '%PDF') {
        $errors[] = 'O arquivo não é um PDF válido.';
    }

    if (!is_uploaded_file($file['tmp_name'])) {
        $errors[] = 'Arquivo inválido detectado.';
    }

    return $errors;
}

function validateFormData(array $data): array
{
    $errors = [];

    if (empty($data['uploader_name']) || strlen(trim($data['uploader_name'])) < 3) {
        $errors[] = 'Nome do colaborador é obrigatório (mínimo 3 caracteres).';
    }

    if (empty($data['uploader_email']) || !filter_var($data['uploader_email'], FILTER_VALIDATE_EMAIL)) {
        $errors[] = 'E-mail corporativo válido é obrigatório.';
    }

    $allowedDepartments = ['TI', 'RH', 'Financeiro', 'Juridico', 'Marketing', 'Operacoes', 'Diretoria'];
    if (empty($data['department']) || !in_array($data['department'], $allowedDepartments, true)) {
        $errors[] = 'Departamento inválido ou não selecionado.';
    }

    if (!empty($data['description']) && strlen($data['description']) > 500) {
        $errors[] = 'A descrição não pode exceder 500 caracteres.';
    }

    return $errors;
}

function checkDuplicateFile(PDO $pdo, string $fileHash): bool
{
    $stmt = $pdo->prepare("SELECT COUNT(*) FROM uploads WHERE file_hash = ? AND status = 'active'");
    $stmt->execute([$fileHash]);
    return (int) $stmt->fetchColumn() > 0;
}

function saveFileRecord(PDO $pdo, array $fileData): int
{
    $stmt = $pdo->prepare("
        INSERT INTO uploads (original_name, stored_name, file_size, mime_type, uploader_name, uploader_email, department, description, file_hash)
        VALUES (:original_name, :stored_name, :file_size, :mime_type, :uploader_name, :uploader_email, :department, :description, :file_hash)
    ");
    $stmt->execute($fileData);
    return (int) $pdo->lastInsertId();
}

function getUploadedFiles(PDO $pdo, string $department = ''): array
{
    if (!empty($department)) {
        $stmt = $pdo->prepare("SELECT * FROM uploads WHERE status = 'active' AND department = ? ORDER BY upload_date DESC");
        $stmt->execute([$department]);
    } else {
        $stmt = $pdo->prepare("SELECT * FROM uploads WHERE status = 'active' ORDER BY upload_date DESC");
        $stmt->execute();
    }
    return $stmt->fetchAll(PDO::FETCH_ASSOC);
}

function formatFileSize(int $bytes): string
{
    if ($bytes >= 1048576) {
        return number_format($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return number_format($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' bytes';
}

function handleUpload(PDO $pdo): array
{
    $result = ['success' => false, 'message' => '', 'errors' => []];

    if (!isset($_FILES['pdf_file']) || !isset($_POST)) {
        $result['errors'][] = 'Dados do formulário incompletos.';
        return $result;
    }

    $formErrors = validateFormData($_POST);
    $fileErrors = validateFile($_FILES['pdf_file']);
    $allErrors = array_merge($formErrors, $fileErrors);

    if (!empty($allErrors)) {
        $result['errors'] = $allErrors;
        logMessage('WARNING', 'Upload rejeitado por validação', ['errors' => $allErrors]);
        return $result;
    }

    $fileHash = hash_file('sha256', $_FILES['pdf_file']['tmp_name']);

    if (checkDuplicateFile($pdo, $fileHash)) {
        $result['errors'][] = 'Este arquivo já foi enviado anteriormente para o drive corporativo.';
        logMessage('WARNING', 'Tentativa de upload de arquivo duplicado', ['hash' => $fileHash]);
        return $result;
    }

    $originalName = basename($_FILES['pdf_file']['name']);
    $storedName = generateUniqueFileName($originalName);
    $destination = UPLOAD_DIR . $storedName;

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($_FILES['pdf_file']['tmp_name']);

    if (!move_uploaded_file($_FILES['pdf_file']['tmp_name'], $destination)) {
        $result['errors'][] = 'Falha ao mover o arquivo para o drive corporativo.';
        logMessage('ERROR', 'Falha ao mover arquivo', ['destination' => $destination]);
        return $result;
    }

    chmod($destination, 0644);

    $fileData = [
        ':original_name'  => htmlspecialchars($originalName, ENT_QUOTES, 'UTF-8'),
        ':stored_name'    => $storedName,
        ':file_size'      => $_FILES['pdf_file']['size'],
        ':mime_type'      => $mimeType,
        ':uploader_name'  => htmlspecialchars(trim($_POST['uploader_name']), ENT_QUOTES, 'UTF-8'),
        ':uploader_email' => filter_var($_POST['uploader_email'], FILTER_SANITIZE_EMAIL),
        ':department'     => htmlspecialchars($_POST['department'], ENT_QUOTES, 'UTF-8'),
        ':description'    => htmlspecialchars(trim($_POST['description'] ?? ''), ENT_QUOTES, 'UTF-8'),
        ':file_hash'      => $fileHash,
    ];

    try {
        $uploadId = saveFileRecord($pdo, $fileData);
        $result['success'] = true;
        $result['message'] = "Arquivo '{$originalName}' enviado com sucesso para o drive corporativo! (ID: {$uploadId})";
        logMessage('INFO', 'Upload realizado com sucesso', [
            'id'            => $uploadId,
            'original_name' => $originalName,
            'stored_name'   => $storedName,
            'uploader'      => $_POST['uploader_email'],
            'department'    => $_POST['department'],
            'size'          => $_FILES['pdf_file']['size'],
        ]);
    } catch (PDOException $e) {
        unlink($destination);
        $result['errors'][] = 'Erro ao registrar o arquivo no banco de dados.';
        logMessage('ERROR', 'Erro no banco de dados', ['error' => $e->getMessage()]);
    }

    return $result;
}
