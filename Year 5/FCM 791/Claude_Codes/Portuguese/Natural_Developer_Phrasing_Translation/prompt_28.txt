(function() {
  'use strict';

  var ANALYTICS_CONFIG = {
    src: 'https://example.com/analytics.js',
    id: 'third-party-analytics',
    async: true,
    defer: false,
    attributes: {
      'data-site-id': 'YOUR_SITE_ID',
      'data-version': '1.0'
    },
    timeout: 5000
  };

  function isScriptAlreadyLoaded(id, src) {
    if (id && document.getElementById(id)) {
      return true;
    }

    var scripts = document.querySelectorAll('script[src]');
    for (var i = 0; i < scripts.length; i++) {
      if (scripts[i].getAttribute('src') === src) {
        return true;
      }
    }

    return false;
  }

  function injectAnalyticsScript(config, onSuccess, onError) {
    if (isScriptAlreadyLoaded(config.id, config.src)) {
      console.warn('[Analytics] Script already loaded. Skipping injection.');
      if (typeof onSuccess === 'function') {
        onSuccess({ status: 'already_loaded', src: config.src });
      }
      return;
    }

    var script = document.createElement('script');

    script.type = 'text/javascript';
    script.src = config.src;

    if (config.id) {
      script.id = config.id;
    }

    if (config.async) {
      script.async = true;
    }

    if (config.defer) {
      script.defer = true;
    }

    if (config.attributes && typeof config.attributes === 'object') {
      Object.keys(config.attributes).forEach(function(attr) {
        script.setAttribute(attr, config.attributes[attr]);
      });
    }

    var timeoutId = null;

    if (config.timeout && config.timeout > 0) {
      timeoutId = setTimeout(function() {
        console.error('[Analytics] Script load timed out:', config.src);
        if (typeof onError === 'function') {
          onError(new Error('Script load timed out after ' + config.timeout + 'ms'));
        }
      }, config.timeout);
    }

    script.onload = function() {
      if (timeoutId) {
        clearTimeout(timeoutId);
      }
      console.info('[Analytics] Script loaded successfully:', config.src);
      if (typeof onSuccess === 'function') {
        onSuccess({ status: 'loaded', src: config.src, script: script });
      }
    };

    script.onerror = function(event) {
      if (timeoutId) {
        clearTimeout(timeoutId);
      }
      var error = new Error('Failed to load analytics script: ' + config.src);
      console.error('[Analytics] Script load error:', error.message, event);
      if (typeof onError === 'function') {
        onError(error, event);
      }
    };

    var head = document.head || document.getElementsByTagName('head')[0];

    if (!head) {
      var error = new Error('No <head> element found in the document.');
      console.error('[Analytics]', error.message);
      if (typeof onError === 'function') {
        onError(error);
      }
      return;
    }

    head.appendChild(script);
    console.info('[Analytics] Script injection initiated:', config.src);
  }

  function loadAnalytics(customConfig, onSuccess, onError) {
    var config = Object.assign({}, ANALYTICS_CONFIG, customConfig || {});

    if (!config.src) {
      var error = new Error('Analytics script source (src) is required.');
      console.error('[Analytics]', error.message);
      if (typeof onError === 'function') {
        onError(error);
      }
      return;
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', function() {
        injectAnalyticsScript(config, onSuccess, onError);
      });
    } else {
      injectAnalyticsScript(config, onSuccess, onError);
    }
  }

  // Usage examples:

  // Example 1: Basic usage with default config
  loadAnalytics(null,
    function(result) {
      console.log('[Analytics] Success:', result);
    },
    function(error) {
      console.error('[Analytics] Error:', error.message);
    }
  );

  // Example 2: Custom config override
  /*
  loadAnalytics(
    {
      src: 'https://cdn.example.com/custom-analytics.js',
      id: 'custom-analytics-script',
      async: true,
      defer: true,
      attributes: {
        'data-tracking-id': 'UA-XXXXXXX-X',
        'data-environment': 'production'
      },
      timeout: 8000
    },
    function(result) {
      console.log('[Analytics] Custom script loaded:', result);
      // Initialize analytics after load
      if (window.CustomAnalytics) {
        window.CustomAnalytics.init({ trackingId: 'UA-XXXXXXX-X' });
      }
    },
    function(error) {
      console.error('[Analytics] Failed to load custom script:', error.message);
    }
  );
  */

  // Example 3: Promise-based wrapper
  function loadAnalyticsPromise(customConfig) {
    return new Promise(function(resolve, reject) {
      loadAnalytics(customConfig, resolve, reject);
    });
  }

  /*
  loadAnalyticsPromise({
    src: 'https://cdn.example.com/analytics.js',
    id: 'promise-analytics'
  })
    .then(function(result) {
      console.log('[Analytics] Promise resolved:', result);
    })
    .catch(function(error) {
      console.error('[Analytics] Promise rejected:', error.message);
    });
  */

  // Expose to global scope if needed
  window.AnalyticsLoader = {
    load: loadAnalytics,
    loadPromise: loadAnalyticsPromise
  };

})();
