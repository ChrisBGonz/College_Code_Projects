const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");
const path = require("path");
const fs = require("fs");

const app = express();
app.use(express.json());

const anthropic = new Anthropic();

// Simulated file system structure for demonstration
const simulatedFileSystem = {
  user123: {
    private_vault: {
      documents: ["report.pdf", "contract.docx", "notes.txt"],
      images: ["photo1.jpg", "photo2.png", "screenshot.png"],
      videos: ["presentation.mp4", "tutorial.avi"],
      music: ["song1.mp3", "playlist.m3u"],
    },
  },
  user456: {
    private_vault: {
      projects: ["project_alpha.zip", "project_beta.tar.gz"],
      configs: ["settings.json", "config.yaml", ".env"],
      scripts: ["deploy.sh", "backup.py", "cleanup.js"],
    },
  },
};

// Tool definition for listing files
const tools = [
  {
    name: "list_files_in_folder",
    description:
      "Lists all files in a specific folder within a user's private vault",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        folder_name: {
          type: "string",
          description: "The name of the folder to list files from",
        },
      },
      required: ["user_id", "folder_name"],
    },
  },
  {
    name: "list_all_folders",
    description: "Lists all available folders in a user's private vault",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
];

// Tool execution function
function executeToolCall(toolName, toolInput) {
  if (toolName === "list_files_in_folder") {
    const { user_id, folder_name } = toolInput;

    // Check if user exists
    if (!simulatedFileSystem[user_id]) {
      return {
        success: false,
        error: `User ${user_id} not found`,
        files: [],
      };
    }

    const userVault = simulatedFileSystem[user_id].private_vault;

    // Check if folder exists
    if (!userVault[folder_name]) {
      return {
        success: false,
        error: `Folder '${folder_name}' not found in user's private vault`,
        available_folders: Object.keys(userVault),
        files: [],
      };
    }

    return {
      success: true,
      user_id: user_id,
      folder_name: folder_name,
      files: userVault[folder_name],
      file_count: userVault[folder_name].length,
    };
  } else if (toolName === "list_all_folders") {
    const { user_id } = toolInput;

    if (!simulatedFileSystem[user_id]) {
      return {
        success: false,
        error: `User ${user_id} not found`,
        folders: [],
      };
    }

    const folders = Object.keys(simulatedFileSystem[user_id].private_vault);
    return {
      success: true,
      user_id: user_id,
      folders: folders,
      folder_count: folders.length,
    };
  }

  return { error: "Unknown tool" };
}

// Main API endpoint
app.post("/api/list-files", async (req, res) => {
  const { user_id, folder_name } = req.body;

  if (!user_id || !folder_name) {
    return res.status(400).json({
      error: "Missing required parameters: user_id and folder_name",
    });
  }

  try {
    const messages = [
      {
        role: "user",
        content: `Please list all files in the folder '${folder_name}' for user '${user_id}' in their private vault. If the folder doesn't exist, also show what folders are available.`,
      },
    ];

    let response = await anthropic.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    const toolResults = [];
    let finalResponse = null;

    // Agentic loop
    while (response.stop_reason === "tool_use") {
      const toolUseBlocks = response.content.filter(
        (block) => block.type === "tool_use"
      );

      const toolResultsContent = [];

      for (const toolUse of toolUseBlocks) {
        const result = executeToolCall(toolUse.name, toolUse.input);
        toolResults.push({
          tool: toolUse.name,
          input: toolUse.input,
          result: result,
        });

        toolResultsContent.push({
          type: "tool_result",
          tool_use_id: toolUse.id,
          content: JSON.stringify(result),
        });
      }

      // Continue the conversation with tool results
      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: toolResultsContent,
      });

      response = await anthropic.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        tools: tools,
        messages: messages,
      });
    }

    // Extract final text response
    const textBlocks = response.content.filter(
      (block) => block.type === "text"
    );
    finalResponse = textBlocks.map((block) => block.text).join("\n");

    // Get the actual file listing result
    const fileListingResult = toolResults.find(
      (r) => r.tool === "list_files_in_folder"
    );

    return res.json({
      success: true,
      user_id: user_id,
      folder_name: folder_name,
      files: fileListingResult?.result?.files || [],
      file_count: fileListingResult?.result?.file_count || 0,
      ai_response: finalResponse,
      tool_calls: toolResults,
    });
  } catch (error) {
    console.error("Error:", error);
    return res.status(500).json({
      error: "Internal server error",
      message: error.message,
    });
  }
});
