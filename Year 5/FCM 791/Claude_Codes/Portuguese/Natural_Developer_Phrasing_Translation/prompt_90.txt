import anthropic
import os
import time
import gzip
import shutil
import logging
import threading
from datetime import datetime
from pathlib import Path

client = anthropic.Anthropic()

def setup_demo_logger(log_file: str) -> logging.Logger:
    """Setup a demo logger that writes to a file."""
    logger = logging.getLogger("demo_app")
    logger.setLevel(logging.DEBUG)
    
    if not logger.handlers:
        handler = logging.FileHandler(log_file)
        formatter = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s')
        handler.setFormatter(formatter)
        logger.addHandler(handler)
    
    return logger

def generate_demo_logs(log_file: str, num_entries: int = 50):
    """Generate some demo log entries."""
    logger = setup_demo_logger(log_file)
    
    messages = [
        "Application started successfully",
        "Processing user request",
        "Database connection established",
        "Cache miss - fetching from database",
        "API call completed in 245ms",
        "User authentication successful",
        "File upload completed",
        "Background job scheduled",
        "Memory usage: 45%",
        "CPU usage: 23%",
    ]
    
    levels = [logging.DEBUG, logging.INFO, logging.WARNING, logging.ERROR]
    level_names = ["DEBUG", "INFO", "WARNING", "ERROR"]
    
    for i in range(num_entries):
        level = levels[i % len(levels)]
        message = messages[i % len(messages)]
        logger.log(level, f"Entry {i+1}: {message}")
    
    print(f"Generated {num_entries} demo log entries in {log_file}")

def get_log_rotation_strategy():
    """Use Claude to get log rotation strategy recommendations."""
    print("\n" + "="*60)
    print("Consulting Claude for log rotation strategy...")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """I need to implement log rotation for a Python application. 
                Please provide a brief strategy for:
                1. When to rotate logs (conditions)
                2. How many backup files to keep
                3. Whether to compress old logs
                4. File naming convention for rotated logs
                
                Keep the response concise and practical."""
            }
        ]
    ) as stream:
        print("\nClaude's Log Rotation Strategy:")
        print("-" * 40)
        full_response = ""
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
        print("\n")
    
    return full_response

def rotate_log_file(log_file: str, max_backups: int = 5, compress: bool = True):
    """Perform log rotation on the specified file."""
    log_path = Path(log_file)
    
    if not log_path.exists():
        print(f"Log file {log_file} does not exist, skipping rotation")
        return False
    
    file_size = log_path.stat().st_size
    print(f"\nRotating log file: {log_file}")
    print(f"Current size: {file_size} bytes")
    
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    
    for i in range(max_backups - 1, 0, -1):
        if compress:
            old_backup = f"{log_file}.{i}.gz"
            new_backup = f"{log_file}.{i+1}.gz"
        else:
            old_backup = f"{log_file}.{i}"
            new_backup = f"{log_file}.{i+1}"
        
        if Path(old_backup).exists():
            if i + 1 > max_backups:
                Path(old_backup).unlink()
                print(f"Deleted old backup: {old_backup}")
            else:
                shutil.move(old_backup, new_backup)
                print(f"Renamed: {old_backup} -> {new_backup}")
    
    if compress:
        backup_file = f"{log_file}.1.gz"
        with open(log_file, 'rb') as f_in:
            with gzip.open(backup_file, 'wb') as f_out:
                shutil.copyfileobj(f_in, f_out)
        print(f"Compressed and saved as: {backup_file}")
    else:
        backup_file = f"{log_file}.1"
        shutil.copy2(log_file, backup_file)
        print(f"Saved backup as: {backup_file}")
    
    with open(log_file, 'w') as f:
        f.write(f"# Log rotated at {datetime.now().isoformat()}\n")
    
    print(f"Log rotation completed successfully!")
    return True

def analyze_rotation_results(log_file: str, backup_files: list):
    """Use Claude to analyze the rotation results."""
    print("\n" + "="*60)
    print("Analyzing rotation results with Claude...")
    print("="*60)
    
    file_info = []
    log_path = Path(log_file)
    if log_path.exists():
        file_info.append(f"Current log: {log_file} ({log_path.stat().st_size} bytes)")
    
    for backup in backup_files:
        backup_path = Path(backup)
        if backup_path.exists():
            file_info.append(f"Backup: {backup} ({backup_path.stat().st_size} bytes)")
    
    files_summary = "\n".join(file_info) if file_info else "No files found"
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=512,
        messages=[
            {
                "role": "user",
                "content": f"""I just performed log rotation. Here are the resulting files:

{files_summary}

Please provide a brief analysis of:
1. Whether the rotation was successful
2. Disk space savings (if any compression was used)
3. Any recommendations for the rotation configuration

Keep the response brief and practical."""
            }
        ]
    ) as stream:
        print("\nClaude's Analysis:")
        print("-" * 40)
        for text in stream.text_stream:
            print(text, end="", flush=True)
        print("\n")

def log_rotation_daemon(log_file: str, interval_hours: float = 24, max_backups: int = 5):
    """Daemon function that performs log rotation at specified intervals."""
    print(f"\nStarting log rotation daemon for: {log_file}")
    print(f"Rotation interval: {interval_hours} hours")
    print(f"Maximum backups to keep: {max_backups}")
    
    interval_seconds = interval_hours * 3600
    
    while True:
        print(f"\n[{datetime.now().isoformat()}] Performing scheduled log rotation...")
        rotate_log_file(log_file, max_backups=max_backups, compress=True)
        
        print(f"Next rotation in {interval_hours} hours")
        time.sleep(interval_seconds)

def main():
    """Main function demonstrating log rotation with Claude AI assistance."""
    print("="*60)
    print("Log Rotation Script with Claude AI")
    print("="*60)
    
    log_dir = Path("./logs")
    log_dir.mkdir(exist_ok=True)
    log_file = str(log_dir / "application.log")
    
    print(f"\nGenerating demo log entries in: {log_file}")
    generate_demo_logs(log_file, num_entries=100)
    
    strategy = get_log_rotation_strategy()
    
    print("\n" + "="*60)
    print("Performing immediate log rotation...")
    print("="*60)
    
    success = rotate_log_file(
        log_file,
        max_backups=5,
        compress=True
    )
    
    if success:
        backup_files = [
            f"{log_file}.1.gz",
            f"{log_file}.2.gz",
            f"{log_file}.3.gz",
        ]
        analyze_rotation_results(log_file, backup_files)
    
    print("\n" + "="*60)
    print("Demonstrating daemon mode (will run for 10 seconds)...")
    print("="*60)
    
    generate_demo_logs(log_file, num_entries=20)
    
    daemon_thread = threading.Thread(
        target=log_rotation_daemon,
        args=(log_file, 0.001, 3),
        daemon=True
    )
    daemon_thread.start()
    
    time.sleep(10)
    
    print("\n" + "="*60)
    print("Log Rotation Script Completed!")
    print("="*60)
    print("\nFiles created:")
    for f in sorted(log_dir.glob("*")):
        size = f.stat().st_size
        print(f"  {f.name}: {size} bytes")
    
    print("\nTo run continuous log rotation, use:")
    print("  log_rotation_daemon(log_file, interval_hours=24, max_backups=5)")

if __name__ == "__main__":
    main()
