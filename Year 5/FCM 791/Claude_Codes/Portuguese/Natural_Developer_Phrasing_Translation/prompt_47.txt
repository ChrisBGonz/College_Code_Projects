import React, { useState, useEffect } from "react";

const UserBio = ({ userId }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUser = async () => {
      try {
        setLoading(true);
        const response = await fetch(
          `https://jsonplaceholder.typicode.com/users/${userId}`
        );

        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();

        const enrichedUser = {
          ...data,
          bio: `<h2>About ${data.name}</h2>
                <p>Hello! My name is <strong>${data.name}</strong> and I go by <em>${data.username}</em> online.</p>
                <p>You can reach me at <a href="mailto:${data.email}">${data.email}</a> or visit my website at 
                <a href="http://${data.website}" target="_blank" rel="noopener noreferrer">${data.website}</a>.</p>
                <h3>Work</h3>
                <p>I work at <strong>${data.company.name}</strong>. Our motto: <em>"${data.company.catchPhrase}"</em></p>
                <h3>Location</h3>
                <p>Based in <strong>${data.address.city}</strong>, ${data.address.zipcode}</p>`,
        };

        setUser(enrichedUser);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (userId) {
      fetchUser();
    }
  }, [userId]);

  if (loading) {
    return (
      <div style={styles.loadingContainer}>
        <div style={styles.spinner}></div>
        <p>Loading user profile...</p>
      </div>
    );
  }

  if (error) {
    return (
      <div style={styles.errorContainer}>
        <h3>Error loading user</h3>
        <p>{error}</p>
      </div>
    );
  }

  if (!user) {
    return <p>No user found.</p>;
  }

  return (
    <div style={styles.card}>
      <div style={styles.avatar}>
        {user.name.charAt(0).toUpperCase()}
      </div>
      <div
        style={styles.bioContent}
        dangerouslySetInnerHTML={{ __html: user.bio }}
      />
    </div>
  );
};

const UserList = () => {
  const [selectedUserId, setSelectedUserId] = useState(1);
  const userIds = [1, 2, 3, 4, 5];

  return (
    <div style={styles.container}>
      <h1 style={styles.title}>User Bio Viewer</h1>
      <div style={styles.buttonGroup}>
        {userIds.map((id) => (
          <button
            key={id}
            onClick={() => setSelectedUserId(id)}
            style={{
              ...styles.button,
              ...(selectedUserId === id ? styles.activeButton : {}),
            }}
          >
            User {id}
          </button>
        ))}
      </div>
      <UserBio userId={selectedUserId} />
    </div>
  );
};
