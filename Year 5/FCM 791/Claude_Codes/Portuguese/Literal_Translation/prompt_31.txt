
```python
import os
import re
import secrets
import bcrypt
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def hash_password(password: str) -> str:
    """Hash a password using bcrypt with a random salt."""
    salt = bcrypt.gensalt(rounds=12)
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    return hashed.decode('utf-8')

def verify_password(password: str, hashed_password: str) -> bool:
    """Verify a password against its hash."""
    return bcrypt.checkpw(password.encode('utf-8'), hashed_password.encode('utf-8'))

def generate_secure_token(length: int = 32) -> str:
    """Generate a cryptographically secure random token."""
    return secrets.token_hex(length)

def validate_password_strength(password: str) -> dict:
    """Validate password strength and return feedback."""
    feedback = {
        "is_valid": True,
        "issues": [],
        "strength": "strong"
    }
    
    if len(password) < 8:
        feedback["issues"].append("Password must be at least 8 characters long")
        feedback["is_valid"] = False
    
    if not re.search(r'[A-Z]', password):
        feedback["issues"].append("Password must contain at least one uppercase letter")
        feedback["is_valid"] = False
    
    if not re.search(r'[a-z]', password):
        feedback["issues"].append("Password must contain at least one lowercase letter")
        feedback["is_valid"] = False
    
    if not re.search(r'\d', password):
        feedback["issues"].append("Password must contain at least one number")
        feedback["is_valid"] = False
    
    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        feedback["issues"].append("Password must contain at least one special character")
        feedback["is_valid"] = False
    
    if feedback["issues"]:
        if len(feedback["issues"]) >= 3:
            feedback["strength"] = "weak"
        else:
            feedback["strength"] = "medium"
    
    return feedback

def chat_with_assistant(user_message: str) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    system_prompt = """You are a helpful security assistant specializing in password hashing and security best practices. 
    You help users understand how to securely hash passwords before storing them in databases.
    
    You have access to the following functions that have been implemented:
    - hash_password(password): Hashes a password using bcrypt with a random salt
    - verify_password(password, hashed_password): Verifies a password against its hash
    - generate_secure_token(length): Generates a cryptographically secure random token
    - validate_password_strength(password): Validates password strength
    
    When users ask about hashing passwords or security, provide helpful explanations and examples.
    If they want to test the functions, you can suggest they use the interactive demo mode.
    Always emphasize security best practices."""
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def interactive_demo():
    """Interactive demo to test password hashing functions."""
    print("\n=== Password Hashing Demo ===")
    print("1. Hash a password")
    print("2. Verify a password")
    print("3. Validate password strength")
    print("4. Generate secure token")
    print("5. Return to chat")
    
    choice = input("\nChoose an option (1-5): ").strip()
    
    if choice == "1":
        password = input("Enter password to hash: ")
        hashed = hash_password(password)
        print(f"\nOriginal password: {password}")
        print(f"Hashed password: {hashed}")
        print("\nNote: Each hash is unique due to random salt!")
        
    elif choice == "2":
        password = input("Enter password to verify: ")
        hashed = input("Enter hashed password: ")
        is_valid = verify_password(password, hashed)
        print(f"\nPassword verification result: {'✓ Valid' if is_valid else '✗ Invalid'}")
        
    elif choice == "3":
        password = input("Enter password to validate: ")
        result = validate_password_strength(password)
        print(f"\nPassword strength: {result['strength'].upper()}")
        print(f"Valid: {'Yes' if result['is_valid'] else 'No'}")
        if result['issues']:
            print("\nIssues found:")
            for issue in result['issues']:
                print(f"  - {issue}")
        else:
            print("✓ Password meets all requirements!")
            
    elif choice == "4":
        length = input("Enter token length (default 32): ").strip()
        length = int(length) if length.isdigit() else 32
        token = generate_secure_token(length)
        print(f"\nGenerated secure token: {token}")
        print(f"Token length: {len(token)} characters")

def main():
    """Main function to run the password hashing assistant."""
    print("=== Password Hashing Security Assistant ===")
    print("I can help you understand how to securely hash passwords for database storage.")
    print("Type 'demo' to try the interactive demo, 'quit' to exit, or ask me anything!\n")
    
    initial_message = chat_with_assistant(
        "Hello! I need help understanding how to hash passwords before storing them in a database. "
        "Can you give me a brief overview of the best practices?"
    )
    print(f"Assistant: {initial_message}\n")
    
    while True:
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit', 'bye']:
            print("Assistant: Goodbye! Remember to always hash passwords before storing them!")
            break
            
        if user_input.lower() == 'demo':
            interactive_demo()
            continue
        
        response = chat_with_assistant(user_input)
        print(f"\nAssistant: {response}\n")

if __name__ == "__main__":
    main()
```
