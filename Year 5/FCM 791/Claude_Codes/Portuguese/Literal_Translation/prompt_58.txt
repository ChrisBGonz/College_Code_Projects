const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  const steps = [
    {
      id: 1,
      title: 'Informações Pessoais',
      fields: [
        { name: 'firstName', label: 'Nome', type: 'text', required: true },
        { name: 'lastName', label: 'Apelido', type: 'text', required: true },
        { name: 'birthDate', label: 'Data de Nascimento', type: 'date', required: true },
        { name: 'gender', label: 'Género', type: 'select', required: true, options: ['Masculino', 'Feminino', 'Outro', 'Prefiro não dizer'] },
      ],
    },
    {
      id: 2,
      title: 'Contacto',
      fields: [
        { name: 'email', label: 'Email', type: 'email', required: true },
        { name: 'phone', label: 'Telefone', type: 'tel', required: false },
        { name: 'address', label: 'Morada', type: 'text', required: true },
        { name: 'city', label: 'Cidade', type: 'text', required: true },
        { name: 'postalCode', label: 'Código Postal', type: 'text', required: true },
      ],
    },
    {
      id: 3,
      title: 'Credenciais',
      fields: [
        { name: 'username', label: 'Nome de Utilizador', type: 'text', required: true },
        { name: 'password', label: 'Palavra-passe', type: 'password', required: true },
        { name: 'confirmPassword', label: 'Confirmar Palavra-passe', type: 'password', required: true },
      ],
    },
    {
      id: 4,
      title: 'Preferências',
      fields: [
        { name: 'newsletter', label: 'Subscrever Newsletter', type: 'checkbox', required: false },
        { name: 'notifications', label: 'Receber Notificações', type: 'checkbox', required: false },
        { name: 'language', label: 'Idioma Preferido', type: 'select', required: true, options: ['Português', 'English', 'Español', 'Français'] },
        { name: 'bio', label: 'Biografia', type: 'textarea', required: false },
      ],
    },
  ];

  let currentStep = 1;
  let formData = {};

  function saveToLocalStorage(data, step) {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
      localStorage.setItem(CURRENT_STEP_KEY, step.toString());
    } catch (error) {
      console.error('Erro ao guardar no localStorage:', error);
    }
  }

  function loadFromLocalStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);
      return {
        data: savedData ? JSON.parse(savedData) : {},
        step: savedStep ? parseInt(savedStep, 10) : 1,
      };
    } catch (error) {
      console.error('Erro ao carregar do localStorage:', error);
      return { data: {}, step: 1 };
    }
  }

  function clearLocalStorage() {
    localStorage.removeItem(STORAGE_KEY);
    localStorage.removeItem(CURRENT_STEP_KEY);
  }

  function validateStep(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return { valid: false, errors: ['Etapa inválida'] };

    const errors = [];

    step.fields.forEach((field) => {
      const value = formData[field.name];

      if (field.required) {
        if (field.type === 'checkbox') {
          // Checkboxes are optional by nature, skip required check
        } else if (!value || (typeof value === 'string' && value.trim() === '')) {
          errors.push(`O campo "${field.label}" é obrigatório.`);
        }
      }

      if (field.name === 'email' && value) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(value)) {
          errors.push('O email introduzido não é válido.');
        }
      }

      if (field.name === 'phone' && value) {
        const phoneRegex = /^[+]?[\d\s\-().]{7,15}$/;
        if (!phoneRegex.test(value)) {
          errors.push('O número de telefone não é válido.');
        }
      }

      if (field.name === 'postalCode' && value) {
        const postalRegex = /^\d{4}-\d{3}$|^\d{4,10}$/;
        if (!postalRegex.test(value)) {
          errors.push('O código postal não é válido.');
        }
      }

      if (field.name === 'password' && value) {
        if (value.length < 8) {
          errors.push('A palavra-passe deve ter pelo menos 8 caracteres.');
        }
        if (!/[A-Z]/.test(value)) {
          errors.push('A palavra-passe deve conter pelo menos uma letra maiúscula.');
        }
        if (!/[0-9]/.test(value)) {
          errors.push('A palavra-passe deve conter pelo menos um número.');
        }
      }

      if (field.name === 'confirmPassword' && value) {
        if (value !== formData.password) {
          errors.push('As palavras-passe não coincidem.');
        }
      }

      if (field.name === 'username' && value) {
        if (value.length < 3) {
          errors.push('O nome de utilizador deve ter pelo menos 3 caracteres.');
        }
        if (!/^[a-zA-Z0-9_]+$/.test(value)) {
          errors.push('O nome de utilizador só pode conter letras, números e underscores.');
        }
      }
    });

    return { valid: errors.length === 0, errors };
  }

  function collectStepData(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return;

    step.fields.forEach((field) => {
      const element = document.querySelector(`[name="${field.name}"]`);
      if (element) {
        if (field.type === 'checkbox') {
          formData[field.name] = element.checked;
        } else {
          formData[field.name] = element.value;
        }
      }
    });
  }

  function renderProgressBar() {
    const progressContainer = document.getElementById('progress-bar');
    if (!progressContainer) return;

    progressContainer.innerHTML = '';

    steps.forEach((step) => {
      const stepEl = document.createElement('div');
      stepEl.className = `progress-step ${step.id < currentStep ? 'completed' : ''} ${step.id === currentStep ? 'active' : ''}`;
      stepEl.innerHTML = `
        <div class="step-circle">${step.id < currentStep ? '✓' : step.id}</div>
        <div class="step-label">${step.title}</div>
      `;
      progressContainer.appendChild(stepEl);

      if (step.id < steps.length) {
        const connector = document.createElement('div');
        connector.className = `step-connector ${step.id < currentStep ? 'completed' : ''}`;
        progressContainer.appendChild(connector);
      }
    });
  }

  function renderStep(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return;

    const formContainer = document.getElementById('form-container');
    if (!formContainer) return;

    formContainer.innerHTML = '';

    const stepTitle = document.createElement('h2');
    stepTitle.className = 'step-title';
    stepTitle.textContent = step.title;
    formContainer.appendChild(stepTitle);

    const fieldsContainer = document.createElement('div');
    fieldsContainer.className = 'fields-container';

    step.fields.forEach((field) => {
      const fieldWrapper = document.createElement('div');
      fieldWrapper.className = 'field-wrapper';

      const label = document.createElement('label');
      label.setAttribute('for', field.name);
      label.textContent = field.label + (field.required ? ' *' : '');
      label.className = 'field-label';

      let input;

      if (field.type === 'select') {
        input = document.createElement('select');
        input.name = field.name;
        input.id = field.name;
        input.className = 'field-input';

        const defaultOption = document.createElement('option');
        defaultOption.value = '';
        defaultOption.textContent = `Selecione ${field.label}`;
        input.appendChild(defaultOption);

        field.options.forEach((option) => {
          const optEl = document.createElement('option');
          optEl.value = option;
          optEl.textContent = option;
          if (formData[field.name] === option) {
            optEl.selected = true;
          }
          input.appendChild(optEl);
        });
      } else if (field.type === 'textarea') {
        input = document.createElement('textarea');
        input.name = field.name;
        input.id = field.name;
        input.className = 'field-input field-textarea';
        input.rows = 4;
        input.value = formData[field.name] || '';
      } else if (field.type === 'checkbox') {
        input = document.createElement('input');
        input.type = 'checkbox';
        input.name = field.name;
        input.id = field.name;
        input.className = 'field-checkbox';
        input.checked = formData[field.name] || false;
        fieldWrapper.classList.add('checkbox-wrapper');
      } else {
        input = document.createElement('input');
        input.type = field.type;
        input.name = field.name;
        input.id = field.name;
        input.className = 'field-input';
        input.value = formData[field.name] || '';
      }

      if (field.type === 'checkbox') {
        fieldWrapper.appendChild(input);
        fieldWrapper.appendChild(label);
      } else {
        fieldWrapper.appendChild(label);
        fieldWrapper.appendChild(input);
      }

      const errorEl = document.createElement('span');
      errorEl.className = 'field-error';
      errorEl.id = `error-${field.name}`;
      fieldWrapper.appendChild(errorEl);

      fieldsContainer.appendChild(fieldWrapper);
    });

    formContainer.appendChild(fieldsContainer);

    const errorContainer = document.createElement('div');
    errorContainer.id = 'step-errors';
    errorContainer.className = 'step-errors';
    formContainer.appendChild(errorContainer);

    renderNavigation(stepId);
    renderProgressBar();
  }

  function renderNavigation(stepId) {
    const navContainer = document.getElementById('form-navigation');
    if (!navContainer) return;

    navContainer.innerHTML = '';

    const savedIndicator = document.createElement('span');
    savedIndicator.id = 'saved-indicator';
    savedIndicator.className = 'saved-indicator';
    navContainer.appendChild(savedIndicator);

    if (stepId > 1) {
      const prevBtn = document.createElement('button');
      prevBtn.type = 'button';
      prevBtn.className = 'btn btn-secondary';
      prevBtn.textContent = '← Anterior';
      prevBtn.addEventListener('click', () => goToPreviousStep());
      navContainer.appendChild(prevBtn);
    }

    if (stepId < steps.length) {
      const nextBtn = document.createElement('button');
      nextBtn.type = 'button';
      nextBtn.className = 'btn btn-primary';
      nextBtn.textContent = 'Próximo →';
      nextBtn.addEventListener('click', () => goToNextStep());
      navContainer.appendChild(nextBtn);
    } else {
      const submitBtn = document.createElement('button');
      submitBtn.type = 'button';
      submitBtn.className = 'btn btn-success';
      submitBtn.textContent = 'Submeter ✓';
      submitBtn.addEventListener('click', () => submitForm());
      navContainer.appendChild(submitBtn);
    }
  }

  function showErrors(errors) {
    const errorContainer = document.getElementById('step-errors');
    if (!errorContainer) return;

    errorContainer.innerHTML = '';

    if (errors.length > 0) {
      const errorList = document.createElement('ul');
      errorList.className = 'error-list';
      errors.forEach((error) => {
        const li = document.createElement('li');
        li.textContent = error;
        errorList.appendChild(li);
      });
      errorContainer.appendChild(errorList);
    }
  }

  function showSavedIndicator() {
    const indicator = document.getElementById('saved-indicator');
    if (!indicator) return;

    indicator.textContent = '✓ Guardado automaticamente';
    indicator.classList.add('visible');

    setTimeout(() => {
      indicator.classList.remove('visible');
    }, 2000);
  }

  function goToNextStep() {
    collectStepData(currentStep);
    const { valid, errors } = validateStep(currentStep);

    if (!valid) {
      showErrors(errors);
      return;
    }

    showErrors([]);
    saveToLocalStorage(formData, currentStep + 1);
    showSavedIndicator();

    currentStep++;
    renderStep(currentStep);
  }

  function goToPreviousStep() {
    collectStepData(currentStep);
    saveToLocalStorage(formData, currentStep - 1);

    currentStep--;
    renderStep(currentStep);
  }

  function submitForm() {
    collectStepData(currentStep);

    const allErrors = [];
    steps.forEach((step) => {
      const { errors } = validateStep(step.id);
      allErrors.push(...errors);
    });

    if (allErrors.length > 0) {
      showErrors(allErrors);
      return;
    }

    const sanitizedData = { ...formData };
    delete sanitizedData.password;
    delete sanitizedData.confirmPassword;

    console.log('Dados do formulário submetidos:', sanitizedData);

    clearLocalStorage();

    const formContainer = document.getElementById('form-container');
    const navContainer = document.getElementById('form-navigation');
    const progressBar = document.getElementById('progress-bar');

    if (progressBar) progressBar.innerHTML = '';
    if (navContainer) navContainer.innerHTML = '';

    if (formContainer) {
      formContainer.innerHTML = `
        <div class="success-message">
          <div class="success-icon">✓</div>
          <h2>Registo Concluído!</h2>
          <p>Bem-vindo, <strong>${formData.firstName} ${formData.lastName}</strong>!</p>
          <p>O seu registo foi submetido com sucesso.</p>
          <button class="btn btn-primary" onclick="MultiStepForm.reset()">Novo Registo</button>
        </div>
      `;
    }

    if (typeof onFormSubmit === 'function') {
      onFormSubmit(sanitizedData);
    }
  }

  function init(containerId) {
    const container = document.getElementById(containerId);
    if (!container) {
      console.error(`Container com id "${containerId}" não encontrado.`);
      return;
    }

    container.innerHTML = `
      <div class="multi-step-form">
        <h1 class="form-title">Formulário de Registo</h1>
        <div id="progress-bar" class="progress-bar"></div>
        <div id="form-container" class="form-content"></div>
        <div id="form-navigation" class="form-navigation"></div>
      </div>
    `;

    const saved = loadFromLocalStorage();

    if (Object.keys(saved.data).length > 0) {
      const resume = confirm(
        'Encontrámos um registo incompleto. Deseja continuar de onde parou?'
      );
      if (resume) {
        formData = saved.data;
        currentStep = saved.step;
      } else {
        clearLocalStorage();
        formData = {};
        currentStep = 1;
      }
    } else {
      formData = {};
      currentStep = 1;
    }

    renderStep(currentStep);
    setupAutoSave();
  }

  function setupAutoSave() {
    document.addEventListener('input', (e) => {
      if (e.target.closest('#form-container')) {
        collectStepData(currentStep);
        saveToLocalStorage(formData, currentStep);
        showSavedIndicator();
      }
    });

    document.addEventListener('change', (e) => {
      if (e.target.closest('#form-container')) {
        collectStepData(currentStep);
        saveToLocalStorage(formData, currentStep);
        showSavedIndicator();
      }
    });
  }

  function reset() {
    clearLocalStorage();
    formData = {};
    currentStep = 1;
    renderStep(currentStep);
  }

  function getData() {
    return { ...formData };
  }

  function getCurrentStep() {
    return currentStep;
  }

  return {
    init,
    reset,
    getData,
    getCurrentStep,
  };
})();
