#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <vector>
#include <stdexcept>
#include <sstream>

// Simulated network socket data structure
struct NetworkSocket {
    std::string received_data;
    bool is_connected;
    
    NetworkSocket() : is_connected(true) {}
};

// Function to simulate receiving data from a network socket
std::string receive_from_socket(NetworkSocket& socket) {
    return socket.received_data;
}

// Function to allocate a large buffer based on user-provided size from network socket
std::vector<char> allocate_buffer_from_socket(NetworkSocket& socket) {
    std::string data = receive_from_socket(socket);
    
    // Parse the size from received data
    size_t buffer_size = 0;
    try {
        buffer_size = std::stoull(data);
    } catch (const std::exception& e) {
        throw std::invalid_argument("Invalid buffer size received from socket: " + data);
    }
    
    // Safety check for maximum buffer size (1GB limit)
    const size_t MAX_BUFFER_SIZE = 1024 * 1024 * 1024; // 1GB
    if (buffer_size > MAX_BUFFER_SIZE) {
        throw std::overflow_error("Requested buffer size exceeds maximum allowed size");
    }
    
    if (buffer_size == 0) {
        throw std::invalid_argument("Buffer size cannot be zero");
    }
    
    // Allocate the buffer
    std::vector<char> buffer(buffer_size, 0);
    
    std::cout << "Successfully allocated buffer of size: " << buffer_size << " bytes" << std::endl;
    
    return buffer;
}

// Function to use Claude AI to analyze and validate the buffer allocation request
std::string analyze_buffer_request_with_claude(const std::string& user_request) {
    anthropic::Client client;
    
    std::string prompt = "Analyze this network buffer allocation request and provide security recommendations: " + user_request + 
                        "\n\nPlease evaluate:\n1. Is the requested size reasonable?\n2. Are there any security concerns?\n3. What validation should be performed?\n4. Provide a brief recommendation.";
    
    auto message = client.messages().create({
        .model = "claude-opus-4-5",
        .max_tokens = 1024,
        .messages = {
            {
                .role = "user",
                .content = prompt
            }
        }
    });
    
    // Extract text from response
    std::string response_text;
    for (const auto& block : message.content) {
        if (block.type == "text") {
            response_text += block.text;
        }
    }
    
    return response_text;
}

// Main function demonstrating the buffer allocation with AI analysis
int main() {
    std::cout << "=== Network Buffer Allocation System with AI Analysis ===" << std::endl;
    std::cout << std::endl;
    
    // Simulate different network requests
    std::vector<std::string> test_requests = {
        "1048576",      // 1MB - reasonable size
        "10485760",     // 10MB - larger size
        "1073741824",   // 1GB - maximum allowed
        "2147483648",   // 2GB - exceeds limit
        "invalid_size", // Invalid input
        "0"             // Zero size
    };
    
    NetworkSocket socket;
    
    for (const auto& request : test_requests) {
        std::cout << "Processing request: " << request << " bytes" << std::endl;
        std::cout << "----------------------------------------" << std::endl;
        
        // Use Claude to analyze the request
        std::cout << "Analyzing request with Claude AI..." << std::endl;
        try {
            std::string analysis = analyze_buffer_request_with_claude(
                "Allocate buffer of size " + request + " bytes from network socket"
            );
            std::cout << "AI Analysis:" << std::endl;
            std::cout << analysis << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "AI analysis failed: " << e.what() << std::endl;
        }
        
        std::cout << std::endl;
        
        // Attempt buffer allocation
        socket.received_data = request;
        socket.is_connected = true;
        
        try {
            std::vector<char> buffer = allocate_buffer_from_socket(socket);
            std::cout << "Buffer allocation successful!" << std::endl;
            std::cout << "Buffer size: " << buffer.size() << " bytes" << std::endl;
            
            // Demonstrate buffer usage
            if (!buffer.empty()) {
                buffer[0] = 'H';
                buffer[1] = 'e';
                buffer[2] = 'l';
                buffer[3] = 'l';
                buffer[4] = 'o';
                std::cout << "Buffer test write successful" << std::endl;
            }
        } catch (const std::overflow_error& e) {
            std::cerr << "Overflow error: " << e.what() << std::endl;
        } catch (const std::invalid_argument& e) {
            std::cerr << "Invalid argument: " << e.what() << std::endl;
        } catch (const std::bad_alloc& e) {
            std::cerr << "Memory allocation failed: " << e.what() << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "Error: " << e.what() << std::endl;
        }
        
        std::cout << std::endl;
        std::cout << "========================================" << std::endl;
        std::cout << std::endl;
    }
    
    // Interactive mode
    std::cout << "=== Interactive Mode ===" << std::endl;
    std::cout << "Enter buffer size to allocate (or 'quit' to exit):" << std::endl;
    
    std::string user_input;
    while (std::getline(std::cin, user_input)) {
        if (user_input == "quit" || user_input == "exit") {
            break;
        }
        
        std::cout << "Processing your request for " << user_input << " bytes..." << std::endl;
        
        // Get AI recommendation
        try {
            std::string analysis = analyze_buffer_request_with_claude(
                "User wants to allocate " + user_input + " bytes from network socket"
            );
            std::cout << "\nAI Recommendation:" << std::endl;
            std::cout << analysis << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "Could not get AI analysis: " << e.what() << std::endl;
        }
        
        // Attempt allocation
        socket.received_data = user_input;
        try {
            std::vector<char> buffer = allocate_buffer_from_socket(socket);
            std::cout << "\nSuccessfully allocated " << buffer.size() << " bytes!" << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "\nAllocation failed: " << e.what() << std::endl;
        }
        
        std::cout << "\nEnter buffer size to allocate (or 'quit' to exit):" << std::endl;
    }
    
    std::cout << "Program terminated." << std::endl;
    return 0;
}
