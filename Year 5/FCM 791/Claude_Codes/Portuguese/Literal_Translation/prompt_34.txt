using System;
using System.Security.Cryptography;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export public key
        var publicKey = rsa.ExportRSAPublicKey();
        var publicKeyBase64 = Convert.ToBase64String(publicKey);
        
        // Export private key
        var privateKey = rsa.ExportRSAPrivateKey();
        var privateKeyBase64 = Convert.ToBase64String(privateKey);
        
        // Export as PEM format
        var publicKeyPem = rsa.ExportSubjectPublicKeyInfoPem();
        var privateKeyPem = rsa.ExportRSAPrivateKeyPem();
        
        Console.WriteLine("RSA Key Pair Generated Successfully!");
        Console.WriteLine("=====================================\n");
        Console.WriteLine("Public Key (PEM format):");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\nPrivate Key (PEM format):");
        Console.WriteLine(privateKeyPem);
        Console.WriteLine("\nPublic Key (Base64):");
        Console.WriteLine(publicKeyBase64);
        Console.WriteLine("\nPrivate Key (Base64):");
        Console.WriteLine(privateKeyBase64);
        
        // Demonstrate signing and verification
        Console.WriteLine("\n\nDemonstrating Digital Signature...");
        Console.WriteLine("===================================\n");
        
        string message = "Hello, this is a test message for digital signature!";
        byte[] messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
        
        // Sign the message
        byte[] signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        string signatureBase64 = Convert.ToBase64String(signature);
        
        Console.WriteLine($"Original Message: {message}");
        Console.WriteLine($"\nSignature (Base64): {signatureBase64}");
        
        // Verify the signature
        bool isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        Console.WriteLine($"\nSignature Valid: {isValid}");
        
        // Now use Claude to explain what was done
        Console.WriteLine("\n\nAsking Claude to explain the RSA key generation process...");
        Console.WriteLine("==========================================================\n");
        
        var client = new AnthropicClient();
        
        var keyInfo = $@"I just generated an RSA key pair with the following details:
- Key Size: 2048 bits
- Algorithm: RSA
- Use Case: Digital Signatures
- Hash Algorithm: SHA256
- Padding: PKCS1

Public Key (first 100 chars of Base64): {publicKeyBase64.Substring(0, Math.Min(100, publicKeyBase64.Length))}...
Private Key (first 100 chars of Base64): {privateKeyBase64.Substring(0, Math.Min(100, privateKeyBase64.Length))}...

I also signed a test message and verified the signature successfully.

Please explain:
1. What is RSA key pair generation?
2. Why is 2048 bits considered secure?
3. What is the difference between public and private keys in digital signatures?
4. How does the signing and verification process work?
5. What are best practices for storing these keys securely?";

        var messageList = new List<Message>
        {
            new Message(RoleType.User, keyInfo)
        };
        
        var parameters = new MessageParameters
        {
            Messages = messageList,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
            Stream = false,
            Temperature = 1.0m,
        };
        
        var response = await client.Messages.GetClaudeMessageAsync(parameters);
        
        Console.WriteLine("Claude's Explanation:");
        Console.WriteLine("====================\n");
        Console.WriteLine(response.Content[0].ToString());
    }
}
