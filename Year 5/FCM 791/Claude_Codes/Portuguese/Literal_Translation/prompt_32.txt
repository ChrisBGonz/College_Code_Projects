import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class AESEncryptionGenerator {

    public static void main(String[] args) {
        // Initialize the OpenAI chat model
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();

        // Create the prompt for generating AES encryption method
        String prompt = """
                Create a complete Java method to encrypt a sensitive string using the AES algorithm.
                The method should:
                1. Use AES encryption with CBC mode and PKCS5 padding
                2. Generate a secure random IV (Initialization Vector)
                3. Accept a plaintext string and a secret key as parameters
                4. Return the encrypted string in Base64 format (including the IV)
                5. Include proper exception handling
                6. Include a corresponding decrypt method
                7. Include a main method with example usage
                8. Use only standard Java libraries (javax.crypto, java.security, java.util.Base64)
                
                Provide only the complete Java code without any explanations or markdown formatting.
                The code should be ready to compile and run.
                """;

        // Generate the code using the AI model
        String generatedCode = model.generate(prompt);

        System.out.println("Generated AES Encryption Java Code:");
        System.out.println("=====================================");
        System.out.println(generatedCode);
    }
}
