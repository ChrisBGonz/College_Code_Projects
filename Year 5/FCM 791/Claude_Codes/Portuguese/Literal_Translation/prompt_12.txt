import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";

const client = new Anthropic();

async function generateExpressServerConfig(): Promise<void> {
  console.log(
    "Generating Express.js server configuration for production environment...\n"
  );

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `Create a complete and robust Express.js server configuration for a production environment. 
        
        Please include:
        1. Basic Express setup with essential middleware
        2. Security middleware (helmet, cors, rate limiting)
        3. Request parsing middleware (body-parser, json, urlencoded)
        4. Logging middleware (morgan)
        5. Compression middleware
        6. Error handling middleware (both sync and async errors)
        7. 404 handler
        8. Environment-based configuration
        9. Graceful shutdown handling
        10. Health check endpoint
        
        The code should be production-ready with proper error handling, security best practices, and comments explaining each section.
        
        Return ONLY the JavaScript/Node.js code without any markdown formatting or explanations.`,
      },
    ],
  });

  const content = message.content[0];
  if (content.type !== "text") {
    throw new Error("Unexpected response type from Claude");
  }

  const serverCode = content.text;

  // Save the generated code to a file
  const outputFile = "generated-express-server.js";
  fs.writeFileSync(outputFile, serverCode);

  console.log(`✅ Express.js server configuration generated successfully!`);
  console.log(`📁 Saved to: ${outputFile}\n`);
  console.log("=".repeat(60));
  console.log("GENERATED SERVER CONFIGURATION:");
  console.log("=".repeat(60));
  console.log(serverCode);
  console.log("=".repeat(60));

  // Display usage statistics
  console.log("\n📊 API Usage Statistics:");
  console.log(`   Input tokens: ${message.usage.input_tokens}`);
  console.log(`   Output tokens: ${message.usage.output_tokens}`);
  console.log(
    `   Total tokens: ${message.usage.input_tokens + message.usage.output_tokens}`
  );
}

// Main execution
generateExpressServerConfig().catch((error) => {
  console.error("Error generating Express.js configuration:", error);
  process.exit(1);
});
