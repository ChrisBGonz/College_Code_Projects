// Helper: get user by email (simulates DB lookup)
function get_user(string $email, array &$db): ?array
{
    return $db[$email] ?? null;
}

// Helper: sanitize output
function h(string $str): string
{
    return htmlspecialchars($str, ENT_QUOTES, 'UTF-8');
}

// Determine current step
$step    = $_SESSION['recovery_step'] ?? 'email';
$error   = '';
$success = '';

// ─── STEP 1: Receive email ────────────────────────────────────────────────────
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {

    if ($_POST['action'] === 'submit_email') {
        $email = strtolower(trim($_POST['email'] ?? ''));

        if (empty($email) || !filter_var($email, FILTER_VALIDATE_EMAIL)) {
            $error = 'Por favor, informe um endereço de e-mail válido.';
            $step  = 'email';
        } elseif (!isset($users_db[$email])) {
            // Generic message to avoid user enumeration
            $error = 'Se esse e-mail estiver cadastrado, você receberá as instruções.';
            $step  = 'email';
        } else {
            $_SESSION['recovery_email'] = $email;
            $_SESSION['recovery_step']  = 'question';
            $step = 'question';
        }
    }

    // ─── STEP 2: Validate secret answer ──────────────────────────────────────
    elseif ($_POST['action'] === 'submit_answer') {
        $email  = $_SESSION['recovery_email'] ?? '';
        $answer = strtolower(trim($_POST['answer'] ?? ''));
        $user   = get_user($email, $users_db);

        if (!$user) {
            $error = 'Sessão inválida. Reinicie o processo.';
            unset($_SESSION['recovery_email'], $_SESSION['recovery_step']);
            $step = 'email';
        } elseif (empty($answer)) {
            $error = 'Por favor, informe a resposta.';
            $step  = 'question';
        } elseif (!hash_equals($user['secret_answer'], $answer)) {
            // Brute-force protection: count attempts
            $_SESSION['recovery_attempts'] = ($_SESSION['recovery_attempts'] ?? 0) + 1;

            if ($_SESSION['recovery_attempts'] >= 3) {
                $error = 'Número máximo de tentativas atingido. Reinicie o processo.';
                unset($_SESSION['recovery_email'], $_SESSION['recovery_step'], $_SESSION['recovery_attempts']);
                $step = 'email';
            } else {
                $remaining = 3 - $_SESSION['recovery_attempts'];
                $error     = "Resposta incorreta. Você tem mais {$remaining} tentativa(s).";
                $step      = 'question';
            }
        } else {
            unset($_SESSION['recovery_attempts']);
            $_SESSION['recovery_step'] = 'reset';
            $step = 'reset';
        }
    }

    // ─── STEP 3: Reset password ───────────────────────────────────────────────
    elseif ($_POST['action'] === 'reset_password') {
        $email    = $_SESSION['recovery_email'] ?? '';
        $new_pass = $_POST['new_password']     ?? '';
        $confirm  = $_POST['confirm_password'] ?? '';
        $user     = get_user($email, $users_db);

        if (!$user || ($_SESSION['recovery_step'] ?? '') !== 'reset') {
            $error = 'Sessão inválida. Reinicie o processo.';
            unset($_SESSION['recovery_email'], $_SESSION['recovery_step']);
            $step = 'email';
        } elseif (strlen($new_pass) < 8) {
            $error = 'A senha deve ter pelo menos 8 caracteres.';
            $step  = 'reset';
        } elseif ($new_pass !== $confirm) {
            $error = 'As senhas não coincidem.';
            $step  = 'reset';
        } else {
            // In a real app: UPDATE users SET password = ? WHERE email = ?
            $users_db[$email]['password'] = password_hash($new_pass, PASSWORD_BCRYPT);

            // Clear recovery session
            unset($_SESSION['recovery_email'], $_SESSION['recovery_step']);

            $success = 'Senha redefinida com sucesso! Você já pode fazer login.';
            $step    = 'done';
        }
    }
} else {
    // First access or page refresh — keep current step from session
    $step = $_SESSION['recovery_step'] ?? 'email';
}

// Fetch question for current user (if in question/reset step)
$current_question = '';
if (in_array($step, ['question', 'reset'], true) && isset($_SESSION['recovery_email'])) {
    $u = get_user($_SESSION['recovery_email'], $users_db);

}
