import anthropic
import json
import time
import random

client = anthropic.Anthropic()

def simulate_payment_api(amount: float, currency: str, card_number: str) -> dict:
    """Simulates an external payment API that might timeout."""
    time.sleep(random.uniform(0.5, 3.5))
    
    return {
        "status": "success",
        "transaction_id": f"TXN_{random.randint(10000, 99999)}",
        "amount": amount,
        "currency": currency,
        "timestamp": time.time()
    }

tools = [
    {
        "name": "process_payment",
        "description": "Process a payment through the external payment API. Returns payment result or timeout status.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., USD, EUR, BRL)"
                },
                "card_number": {
                    "type": "string",
                    "description": "The card number for payment (last 4 digits)"
                }
            },
            "required": ["amount", "currency", "card_number"]
        }
    }
]

def process_payment_with_timeout(amount: float, currency: str, card_number: str) -> dict:
    """
    Process payment with a 2-second timeout.
    Returns True (allow user to proceed) if API doesn't respond within 2 seconds.
    """
    start_time = time.time()
    timeout = 2.0
    
    try:
        import threading
        result = {"data": None, "error": None, "timed_out": False}
        
        def api_call():
            try:
                result["data"] = simulate_payment_api(amount, currency, card_number)
            except Exception as e:
                result["error"] = str(e)
        
        thread = threading.Thread(target=api_call)
        thread.start()
        thread.join(timeout=timeout)
        
        elapsed_time = time.time() - start_time
        
        if thread.is_alive():
            result["timed_out"] = True
            return {
                "success": True,
                "timed_out": True,
                "allow_proceed": True,
                "message": f"Payment API timed out after {elapsed_time:.2f}s. User allowed to proceed.",
                "elapsed_time": elapsed_time
            }
        
        if result["error"]:
            return {
                "success": False,
                "timed_out": False,
                "allow_proceed": False,
                "message": f"Payment API error: {result['error']}",
                "elapsed_time": elapsed_time
            }
        
        return {
            "success": True,
            "timed_out": False,
            "allow_proceed": True,
            "message": "Payment processed successfully",
            "transaction_data": result["data"],
            "elapsed_time": elapsed_time
        }
        
    except Exception as e:
        return {
            "success": False,
            "timed_out": False,
            "allow_proceed": False,
            "message": f"Unexpected error: {str(e)}",
            "elapsed_time": time.time() - start_time
        }

def handle_payment_with_claude(payment_details: dict) -> str:
    """
    Use Claude to handle payment processing with tool use.
    """
    messages = [
        {
            "role": "user",
            "content": f"Please process a payment with the following details: {json.dumps(payment_details)}"
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
        
        tool_input = tool_use_block.input
        print(f"\nClaude is calling tool: {tool_use_block.name}")
        print(f"With parameters: {json.dumps(tool_input, indent=2)}")
        
        payment_result = process_payment_with_timeout(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            card_number=tool_input["card_number"]
        )
        
        print(f"\nPayment API Result:")
        print(f"  - Timed out: {payment_result['timed_out']}")
        print(f"  - Elapsed time: {payment_result['elapsed_time']:.2f}s")
        print(f"  - Allow proceed: {payment_result['allow_proceed']}")
        
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(payment_result)
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response = block.text
            break
    
    return final_response

def main():
    print("=" * 60)
    print("Payment Processing System with Timeout Handling")
    print("=" * 60)
    
    test_payments = [
        {
            "amount": 150.00,
            "currency": "USD",
            "card_number": "4532",
            "description": "Online purchase"
        },
        {
            "amount": 299.99,
            "currency": "EUR",
            "card_number": "1234",
            "description": "Subscription renewal"
        },
        {
            "amount": 50.00,
            "currency": "BRL",
            "card_number": "9876",
            "description": "Service payment"
        }
    ]
    
    for i, payment in enumerate(test_payments, 1):
        print(f"\n{'='*60}")
        print(f"Test Payment {i}: {payment['description']}")
        print(f"Amount: {payment['amount']} {payment['currency']}")
        print(f"Card: ****{payment['card_number']}")
        print("-" * 60)
        
        result = handle_payment_with_claude(payment)
        
        print(f"\nClaude's Response:")
        print(result)
        
        time.sleep(1)
    
    print(f"\n{'='*60}")
    print("Direct timeout test (without Claude):")
    print("-" * 60)
    
    direct_result = process_payment_with_timeout(100.00, "USD", "5678")
    print(f"Direct API call result:")
    print(json.dumps(direct_result, indent=2))
    
    if direct_result["timed_out"]:
        print("\n✓ API timed out - User is allowed to proceed (returns True)")
    else:
        print("\n✓ API responded successfully within timeout")

if __name__ == "__main__":
    main()
