const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function updateUserProfile(userId, updateData) {
  const { email, displayName } = updateData;

  const conversationHistory = [];

  // Initial message to start the profile update process
  const initialMessage = `I need to update a user profile with the following information:
- User ID: ${userId}
- New Email: ${email || "not provided"}
- New Display Name: ${displayName || "not provided"}

Please help me validate this data and process the profile update.`;

  conversationHistory.push({
    role: "user",
    content: initialMessage,
  });

  // First turn: Validate the input data
  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful API assistant that processes user profile updates. 
    You validate input data, check for errors, and provide structured responses.
    When validating email, check if it's a valid email format.
    When validating display name, ensure it's between 2-50 characters and contains only alphanumeric characters, spaces, and common punctuation.
    Always respond in a structured way that can be parsed programmatically.`,
    messages: conversationHistory,
  });

  const validationResponse = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: validationResponse,
  });

  // Second turn: Ask for confirmation of the update
  conversationHistory.push({
    role: "user",
    content: `Based on your validation, should we proceed with the update? 
    Please provide a JSON response with the following structure:
    {
      "valid": true/false,
      "errors": [],
      "sanitizedData": {
        "email": "...",
        "displayName": "..."
      },
      "message": "..."
    }`,
  });

  response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful API assistant that processes user profile updates. 
    You validate input data, check for errors, and provide structured responses.
    When validating email, check if it's a valid email format.
    When validating display name, ensure it's between 2-50 characters and contains only alphanumeric characters, spaces, and common punctuation.
    Always respond in a structured way that can be parsed programmatically.`,
    messages: conversationHistory,
  });

  const confirmationResponse = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: confirmationResponse,
  });

  // Parse the JSON response
  let parsedResponse;
  try {
    // Extract JSON from the response
    const jsonMatch = confirmationResponse.match(/\{[\s\S]*\}/);
    if (jsonMatch) {
      parsedResponse = JSON.parse(jsonMatch[0]);
    } else {
      throw new Error("No JSON found in response");
    }
  } catch {
    // If parsing fails, create a default error response
    parsedResponse = {
      valid: false,
      errors: ["Failed to parse validation response"],
      sanitizedData: null,
      message: "Validation process encountered an error",
    };
  }

  if (!parsedResponse.valid) {
    return {
      success: false,
      statusCode: 400,
      errors: parsedResponse.errors,
      message: parsedResponse.message || "Validation failed",
    };
  }

  // Third turn: Simulate the database update
  conversationHistory.push({
    role: "user",
    content: `The validation passed. Now simulate updating the database with:
    - User ID: ${userId}
    - Email: ${parsedResponse.sanitizedData?.email || email}
    - Display Name: ${parsedResponse.sanitizedData?.displayName || displayName}
    
    Provide a final JSON response confirming the update was successful with the updated user data.`,
  });

  response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful API assistant that processes user profile updates. 
    You validate input data, check for errors, and provide structured responses.
    When validating email, check if it's a valid email format.
    When validating display name, ensure it's between 2-50 characters and contains only alphanumeric characters, spaces, and common punctuation.
    Always respond in a structured way that can be parsed programmatically.`,
    messages: conversationHistory,
  });

  const finalResponse = response.content[0].text;

  // Parse the final response
  let finalParsed;
  try {
    const jsonMatch = finalResponse.match(/\{[\s\S]*\}/);
    if (jsonMatch) {
      finalParsed = JSON.parse(jsonMatch[0]);
    } else {
      throw new Error("No JSON found in final response");
    }
  } catch {
    // Create a success response with the sanitized data
    finalParsed = {
      success: true,
      updatedUser: {
        id: userId,
        email: parsedResponse.sanitizedData?.email || email,
        displayName: parsedResponse.sanitizedData?.displayName || displayName,
        updatedAt: new Date().toISOString(),
      },
    };
  }

  return {
    success: true,
    statusCode: 200,
    message: "Profile updated successfully",
    data: finalParsed.updatedUser || {
      id: userId,
      email: parsedResponse.sanitizedData?.email || email,
      displayName: parsedResponse.sanitizedData?.displayName || displayName,
      updatedAt: new Date().toISOString(),
    },
  };
}

// Simulate the API endpoint handler
async function handleUpdateProfileRequest(req) {
  const { userId } = req.params;
  const { email, displayName } = req.body;

  // Validate that at least one field is provided
  if (!email && !displayName) {
    return {
      success: false,
      statusCode: 400,
      message: "At least one field (email or displayName) must be provided",
    };
  }

  // Validate userId
  if (!userId) {
    return {
      success: false,
      statusCode: 400,
      message: "User ID is required",
    };
  }

  try {
    const result = await updateUserProfile(userId, { email, displayName });
    return result;
  } catch (error) {
    return {
      success: false,
      statusCode: 500,
      message: "Internal server error",
      error: error.message,
    };
  }
}
