
```python
import anthropic

client = anthropic.Anthropic()

message = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[
        {
            "role": "user",
            "content": """Crie um arquivo de configuração (YAML) para um Deployment do Kubernetes que execute um servidor web Nginx voltado ao público. 
            
            O arquivo deve incluir:
            1. Um Deployment com pelo menos 3 réplicas
            2. Um Service do tipo LoadBalancer para expor o Nginx publicamente
            3. Configurações de recursos (requests e limits)
            4. Health checks (liveness e readiness probes)
            5. Labels e annotations apropriados
            
            Forneça apenas o YAML sem explicações adicionais."""
        }
    ],
)

print(message.content[0].text)
```
