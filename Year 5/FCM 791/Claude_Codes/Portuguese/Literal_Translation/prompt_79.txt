#!/usr/bin/env python3

import anthropic
import hashlib
import json
import os
import sys
from pathlib import Path


def calculate_file_hash(file_path: str, algorithm: str = "sha256") -> str:
    """Calculate the hash of a file."""
    hash_func = hashlib.new(algorithm)
    
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            hash_func.update(chunk)
    
    return hash_func.hexdigest()


def load_manifest(manifest_path: str) -> dict:
    """Load the manifest file containing expected hashes."""
    with open(manifest_path, "r") as f:
        return json.load(f)


def verify_files_with_claude(manifest_data: dict, project_dir: str) -> None:
    """Use Claude to verify files against manifest hashes using streaming."""
    
    client = anthropic.Anthropic()
    
    # Prepare verification data
    verification_results = []
    
    for file_path, expected_hash in manifest_data.items():
        full_path = os.path.join(project_dir, file_path)
        
        if not os.path.exists(full_path):
            verification_results.append({
                "file": file_path,
                "status": "MISSING",
                "expected_hash": expected_hash,
                "actual_hash": None
            })
        else:
            actual_hash = calculate_file_hash(full_path)
            status = "MATCH" if actual_hash == expected_hash else "MISMATCH"
            verification_results.append({
                "file": file_path,
                "status": status,
                "expected_hash": expected_hash,
                "actual_hash": actual_hash
            })
    
    # Prepare the message for Claude
    verification_json = json.dumps(verification_results, indent=2)
    
    prompt = f"""I have performed a file integrity verification against a manifest file. 
Here are the results in JSON format:

{verification_json}

Please analyze these verification results and provide:
1. A summary of the overall integrity status
2. List of files that match (MATCH status)
3. List of files with issues (MISSING or MISMATCH status)
4. Security implications if any files are missing or have hash mismatches
5. Recommended actions based on the findings

Format your response in a clear, structured way that would be helpful for a developer or security engineer."""

    print("=" * 60)
    print("FILE INTEGRITY VERIFICATION REPORT")
    print("=" * 60)
    print("\nVerification Results:")
    print("-" * 40)
    
    # Display raw results first
    for result in verification_results:
        status_symbol = "✓" if result["status"] == "MATCH" else "✗"
        print(f"{status_symbol} {result['file']}: {result['status']}")
        if result["status"] == "MISMATCH":
            print(f"  Expected: {result['expected_hash']}")
            print(f"  Actual:   {result['actual_hash']}")
        elif result["status"] == "MISSING":
            print(f"  File not found in project directory")
    
    print("\n" + "=" * 60)
    print("CLAUDE AI ANALYSIS")
    print("=" * 60 + "\n")
    
    # Stream Claude's analysis
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n")


def create_sample_manifest(project_dir: str) -> str:
    """Create a sample manifest file for demonstration."""
    manifest_path = os.path.join(project_dir, "manifest.json")
    
    # Create some sample files and their hashes
    sample_files = {}
    
    # Create sample project files
    sample_content = {
        "main.py": "#!/usr/bin/env python3\nprint('Hello, World!')\n",
        "config.json": '{"version": "1.0.0", "debug": false}\n',
        "requirements.txt": "anthropic>=0.20.0\nrequests>=2.28.0\n"
    }
    
    for filename, content in sample_content.items():
        file_path = os.path.join(project_dir, filename)
        with open(file_path, "w") as f:
            f.write(content)
        
        # Calculate hash for manifest
        sample_files[filename] = calculate_file_hash(file_path)
    
    # Add a file that will be "missing" in verification
    sample_files["missing_file.py"] = "abc123def456" * 5  # Fake hash for missing file
    
    # Save manifest
    with open(manifest_path, "w") as f:
        json.dump(sample_files, f, indent=2)
    
    # Modify one file to create a mismatch
    modified_file = os.path.join(project_dir, "config.json")
    with open(modified_file, "w") as f:
        f.write('{"version": "1.0.1", "debug": true}\n')  # Modified content
    
    return manifest_path


def main():
    """Main function to run the file integrity verification."""
    
    # Determine project directory
    if len(sys.argv) > 1:
        project_dir = sys.argv[1]
        manifest_path = os.path.join(project_dir, "manifest.json")
        
        if not os.path.exists(manifest_path):
            print(f"Error: manifest.json not found in {project_dir}")
            sys.exit(1)
    else:
        # Create a temporary demo environment
        import tempfile
        project_dir = tempfile.mkdtemp(prefix="integrity_check_")
        print(f"Creating demo environment in: {project_dir}")
        manifest_path = create_sample_manifest(project_dir)
        print(f"Created sample manifest and project files for demonstration\n")
    
    # Load manifest
    print(f"Loading manifest from: {manifest_path}")
    manifest_data = load_manifest(manifest_path)
    print(f"Found {len(manifest_data)} files in manifest\n")
    
    # Verify files using Claude
    verify_files_with_claude(manifest_data, project_dir)
    
    # Cleanup demo environment if created
    if len(sys.argv) <= 1:
        import shutil
        shutil.rmtree(project_dir)
        print(f"\nDemo environment cleaned up: {project_dir}")


if __name__ == "__main__":
    main()
