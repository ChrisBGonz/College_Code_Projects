#!/bin/bash

set -euo pipefail

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

INSTALL_DIR="/usr/local/bin"
TMP_DIR=$(mktemp -d)
LOG_FILE="/tmp/security_tool_installer_$(date +%Y%m%d_%H%M%S).log"

log() {
    echo -e "${GREEN}[$(date '+%Y-%m-%d %H:%M:%S')] $1${NC}" | tee -a "$LOG_FILE"
}

warn() {
    echo -e "${YELLOW}[$(date '+%Y-%m-%d %H:%M:%S')] WARNING: $1${NC}" | tee -a "$LOG_FILE"
}

error() {
    echo -e "${RED}[$(date '+%Y-%m-%d %H:%M:%S')] ERROR: $1${NC}" | tee -a "$LOG_FILE"
    exit 1
}

info() {
    echo -e "${BLUE}[$(date '+%Y-%m-%d %H:%M:%S')] INFO: $1${NC}" | tee -a "$LOG_FILE"
}

cleanup() {
    info "Limpando arquivos temporários..."
    rm -rf "$TMP_DIR"
}

trap cleanup EXIT

check_root() {
    if [[ $EUID -ne 0 ]]; then
        warn "Este script não está sendo executado como root. Algumas instalações podem falhar."
        warn "Execute com sudo para instalação global."
        INSTALL_DIR="$HOME/.local/bin"
        mkdir -p "$INSTALL_DIR"
    fi
}

check_dependencies() {
    local deps=("curl" "wget" "tar" "unzip" "jq")
    local missing=()

    info "Verificando dependências..."

    for dep in "${deps[@]}"; do
        if ! command -v "$dep" &>/dev/null; then
            missing+=("$dep")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        warn "Dependências ausentes: ${missing[*]}"
        install_dependencies "${missing[@]}"
    else
        log "Todas as dependências estão instaladas."
    fi
}

install_dependencies() {
    local packages=("$@")

    if command -v apt-get &>/dev/null; then
        info "Instalando dependências via apt-get..."
        apt-get update -qq
        apt-get install -y -qq "${packages[@]}" 2>/dev/null || warn "Algumas dependências não puderam ser instaladas."
    elif command -v yum &>/dev/null; then
        info "Instalando dependências via yum..."
        yum install -y -q "${packages[@]}" 2>/dev/null || warn "Algumas dependências não puderam ser instaladas."
    elif command -v dnf &>/dev/null; then
        info "Instalando dependências via dnf..."
        dnf install -y -q "${packages[@]}" 2>/dev/null || warn "Algumas dependências não puderam ser instaladas."
    elif command -v pacman &>/dev/null; then
        info "Instalando dependências via pacman..."
        pacman -S --noconfirm --quiet "${packages[@]}" 2>/dev/null || warn "Algumas dependências não puderam ser instaladas."
    elif command -v brew &>/dev/null; then
        info "Instalando dependências via brew..."
        brew install "${packages[@]}" 2>/dev/null || warn "Algumas dependências não puderam ser instaladas."
    else
        warn "Gerenciador de pacotes não encontrado. Instale manualmente: ${packages[*]}"
    fi
}

detect_os() {
    OS=""
    ARCH=$(uname -m)

    case "$ARCH" in
        x86_64) ARCH="amd64" ;;
        aarch64 | arm64) ARCH="arm64" ;;
        armv7l) ARCH="armv7" ;;
        i386 | i686) ARCH="386" ;;
        *) warn "Arquitetura não reconhecida: $ARCH" ;;
    esac

    if [[ -f /etc/os-release ]]; then
        source /etc/os-release
        OS="$ID"
    elif [[ "$(uname)" == "Darwin" ]]; then
        OS="darwin"
    elif [[ "$(uname)" == "Linux" ]]; then
        OS="linux"
    else
        OS="unknown"
    fi

    info "Sistema operacional detectado: $OS ($ARCH)"
}

get_latest_version_github() {
    local repo="$1"
    local version

    version=$(curl -s "https://api.github.com/repos/${repo}/releases/latest" | \
        jq -r '.tag_name' 2>/dev/null | \
        sed 's/^v//')

    if [[ -z "$version" || "$version" == "null" ]]; then
        version=$(curl -s "https://api.github.com/repos/${repo}/releases/latest" | \
            grep '"tag_name"' | \
            sed -E 's/.*"([^"]+)".*/\1/' | \
            sed 's/^v//')
    fi

    echo "$version"
}

verify_checksum() {
    local file="$1"
    local checksum_file="$2"
    local algorithm="${3:-sha256}"

    info "Verificando checksum do arquivo..."

    if [[ ! -f "$checksum_file" ]]; then
        warn "Arquivo de checksum não encontrado. Pulando verificação."
        return 0
    fi

    local expected_checksum
    expected_checksum=$(grep "$(basename "$file")" "$checksum_file" | awk '{print $1}')

    if [[ -z "$expected_checksum" ]]; then
        warn "Checksum não encontrado para o arquivo. Pulando verificação."
        return 0
    fi

    local actual_checksum
    case "$algorithm" in
        sha256) actual_checksum=$(sha256sum "$file" | awk '{print $1}') ;;
        sha512) actual_checksum=$(sha512sum "$file" | awk '{print $1}') ;;
        md5) actual_checksum=$(md5sum "$file" | awk '{print $1}') ;;
        *) warn "Algoritmo de checksum desconhecido: $algorithm"; return 0 ;;
    esac

    if [[ "$actual_checksum" == "$expected_checksum" ]]; then
        log "Checksum verificado com sucesso: $actual_checksum"
        return 0
    else
        error "Checksum inválido! Esperado: $expected_checksum, Obtido: $actual_checksum"
    fi
}

download_file() {
    local url="$1"
    local output="$2"
    local description="${3:-arquivo}"

    info "Baixando $description..."
    info "URL: $url"

    if command -v curl &>/dev/null; then
        curl -fsSL --progress-bar -o "$output" "$url" 2>&1 | tee -a "$LOG_FILE" || \
            error "Falha ao baixar $description via curl"
    elif command -v wget &>/dev/null; then
        wget -q --show-progress -O "$output" "$url" 2>&1 | tee -a "$LOG_FILE" || \
            error "Falha ao baixar $description via wget"
    else
        error "Nem curl nem wget estão disponíveis."
    fi

    log "Download concluído: $output"
}

install_nmap() {
    info "Instalando Nmap..."

    if command -v nmap &>/dev/null; then
        local current_version
        current_version=$(nmap --version | head -1 | awk '{print $3}')
        warn "Nmap já está instalado (versão $current_version)."
        read -rp "Deseja reinstalar? (s/N): " confirm
        [[ "$confirm" != "s" && "$confirm" != "S" ]] && return 0
    fi

    if command -v apt-get &>/dev/null; then
        apt-get install -y nmap
    elif command -v yum &>/dev/null; then
        yum install -y nmap
    elif command -v dnf &>/dev/null; then
        dnf install -y nmap
    elif command -v pacman &>/dev/null; then
        pacman -S --noconfirm nmap
    elif command -v brew &>/dev/null; then
        brew install nmap
    else
        local version
        version=$(get_latest_version_github "nmap/nmap")
        local url="https://nmap.org/dist/nmap-${version}.tar.bz2"
        local tarball="$TMP_DIR/nmap.tar.bz2"

        download_file "$url" "$tarball" "Nmap $version"

        cd "$TMP_DIR"
        tar -xjf "$tarball"
        cd "nmap-${version}"
        ./configure --prefix=/usr/local
        make -j"$(nproc)"
        make install
    fi

    log "Nmap instalado com sucesso!"
    nmap --version | head -1
}

install_nuclei() {
    info "Instalando Nuclei (ProjectDiscovery)..."

    local repo="projectdiscovery/nuclei"
    local version
    version=$(get_latest_version_github "$repo")

    if [[ -z "$version" ]]; then
        error "Não foi possível obter a versão mais recente do Nuclei."
    fi

    info "Versão mais recente do Nuclei: $version"

    if command -v nuclei &>/dev/null; then
        local current_version
        current_version=$(nuclei -version 2>&1 | grep -oP '\d+\.\d+\.\d+' | head -1)
        if [[ "$current_version" == "$version" ]]; then
            log "Nuclei já está na versão mais recente ($version)."
            return 0
        fi
        warn "Atualizando Nuclei de $current_version para $version..."
    fi

    local os_name
    case "$(uname)" in
        Linux) os_name="linux" ;;
        Darwin) os_name="darwin" ;;
        *) error "Sistema operacional não suportado para Nuclei." ;;
    esac

    local url="https://github.com/${repo}/releases/download/v${version}/nuclei_${version}_${os_name}_${ARCH}.zip"
    local checksum_url="https://github.com/${repo}/releases/download/v${version}/nuclei_${version}_checksums.txt"
    local zipfile="$TMP_DIR/nuclei.zip"
    local checksum_file="$TMP_DIR/nuclei_checksums.txt"

    download_file "$url" "$zipfile" "Nuclei $version"
    download_file "$checksum_url" "$checksum_file" "checksums do Nuclei" 2>/dev/null || true

    verify_checksum "$zipfile" "$checksum_file"

    unzip -q "$zipfile" -d "$TMP_DIR/nuclei"
    chmod +x "$TMP_DIR/nuclei/nuclei"
    mv "$TMP_DIR/nuclei/nuclei" "$INSTALL_DIR/nuclei"

    log "Nuclei instalado com sucesso em $INSTALL_DIR/nuclei"
    nuclei -version 2>&1 | head -1
}

install_custom_tool() {
    local tool_name="$1"
    local download_url="$2"
    local binary_name="${3:-$tool_name}"
    local checksum_url="${4:-}"

    info "Instalando ferramenta personalizada: $tool_name"
    info "URL: $download_url"

    local filename
    filename=$(basename "$download_url")
    local download_path="$TMP_DIR/$filename"

    download_file "$download_url" "$download_path" "$tool_name"

    if [[ -n "$checksum_url" ]]; then
        local checksum_file="$TMP_DIR/${tool_name}_checksums.txt"
        download_file "$checksum_url" "$checksum_file" "checksums de $tool_name" 2>/dev/null || true
        verify_checksum "$download_path" "$checksum_file"
    fi

    local extract_dir="$TMP_DIR/${tool_name}_extracted"
    mkdir -p "$extract_dir"

    case "$filename" in
        *.tar.gz | *.tgz)
            tar -xzf "$download_path" -C "$extract_dir"
            ;;
        *.tar.bz2)
            tar -xjf "$download_path" -C "$extract_dir"
            ;;
        *.tar.xz)
            tar -xJf "$download_path" -C "$extract_dir"
            ;;
        *.zip)
            unzip -q "$download_path" -d "$extract_dir"
            ;;
        *)
            cp "$download_path" "$extract_dir/$binary_name"
            ;;
    esac

    local binary_path
    binary_path=$(find "$extract_dir" -name "$binary_name" -type f 2>/dev/null | head -1)

    if [[ -z "$binary_path" ]]; then
        binary_path=$(find "$extract_dir" -maxdepth 2 -type f -executable 2>/dev/null | head -1)
    fi

    if [[ -z "$binary_path" ]]; then
        error "Binário '$binary_name' não encontrado após extração."
    fi

    chmod +x "$binary_path"
    mv "$binary_path" "$INSTALL_DIR/$binary_name"

    log "$tool_name instalado com sucesso em $INSTALL_DIR/$binary_name"
}

update_path() {
    local shell_rc=""

    if [[ "$INSTALL_DIR" == "$HOME/.local/bin" ]]; then
        if [[ -n "${BASH_VERSION:-}" ]]; then
            shell_rc="$HOME/.bashrc"
        elif [[ -n "${ZSH_VERSION:-}" ]]; then
            shell_rc="$HOME/.zshrc"
        else
            shell_rc="$HOME/.profile"
        fi

        if ! grep -q "$INSTALL_DIR" "$shell_rc" 2>/dev/null; then
            echo "export PATH=\"\$PATH:$INSTALL_DIR\"" >> "$shell_rc"
            info "PATH atualizado em $shell_rc"
            info "Execute 'source $shell_rc' ou reinicie o terminal para aplicar."
        fi

        export PATH="$PATH:$INSTALL_DIR"
    fi
}

    local tools=("nmap" "nuclei" "httpx" "subfinder" "ffuf" "gobuster" "trivy")

    for tool in "${tools[@]}"; do
        if command -v "$tool" &>/dev/null; then
            local version

            echo -e "  ${GREEN}✓${NC} $tool -> $version"
        else
            echo -e "  ${RED}✗${NC} $tool (não instalado)"
        fi
    done

main() {
    echo ""
    echo -e "${CYAN}╔══════════════════════════════════════════════════════╗${NC}"
    echo -e "${CYAN}║       INSTALADOR DE FERRAMENTAS DE SEGURANÇA         ║${NC}"

    echo -e "${CYAN}╚══════════════════════════════════════════════════════╝${NC}"
    echo ""

    check_root
    check_dependencies
    detect_os
    update_path

    if [[ $# -gt 0 ]]; then
        case "$1" in
            --nmap) install_nmap ;;
            --nuclei) install_nuclei ;;
            --httpx) install_httpx ;;
            --subfinder) install_subfinder ;;
            --ffuf) install_ffuf ;;
            --gobuster) install_gobuster ;;
            --trivy) install_trivy ;;
            --all)
                install_nmap
                install_nuclei
                install_httpx
                install_subfinder
                install_ffuf
                install_gobuster
                install_trivy
                ;;
            --custom)
                if [[ $# -lt 3 ]]; then
                    error "Uso: $0 --custom <nome> <url> [binario] [checksum_url]"
                fi
                install_custom_tool "$2" "$3" "${4:-$2}" "${5:-}"
                ;;
            --list) show_installed_tools ;;
            --help | -h)
                echo "Uso: $0 [opção]"
                echo ""
                echo "Opções:"
                echo "  --nmap          Instalar Nmap"
                echo "  --nuclei        Instalar Nuclei"
                echo "  --httpx         Instalar httpx"
                echo "  --subfinder     Instalar Subfinder"
                echo "  --ffuf          Instalar ffuf"
                echo "  --gobuster      Instalar Gobuster"
                echo "  --trivy         Instalar Trivy"
                echo "  --all           Instalar todas as ferramentas"
                echo "  --custom <nome> <url> [binario] [checksum_url]"
