async function generateProfileSummary(userData) {
  let fullResponse = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 500,
    messages: [
      {
        role: "user",
        content: `Based on the following user profile data, generate a brief personalized summary in JSON format with a "summary" field containing a 2-3 sentence description of the user:

User Data:
${JSON.stringify(userData, null, 2)}

Respond with only valid JSON containing a "summary" field.`,
      },
    ],
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      fullResponse += chunk.delta.text;
      process.stdout.write(chunk.delta.text);
    }
  }

  console.log("\n");

  try {
    const parsed = JSON.parse(fullResponse);
    return parsed.summary || "Profile summary not available";
  } catch {
    return "Profile summary not available";
  }
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);

  if (parsedUrl.pathname === "/profile/download" && req.method === "GET") {
    const userId = parsedUrl.query.user_id;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing required parameter: user_id",
          message: "Please provide a user_id in the query string",
        })
      );
      return;
    }

    const user = usersDatabase[userId];

    if (!user) {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "User not found",
          message: `No user found with id: ${userId}`,
        })
      );
      return;
    }

    try {
      console.log(`Generating profile summary for user ${userId}...`);
      console.log("Streaming response from Claude:");

      const summary = await generateProfileSummary(user);

      const profileData = {
        ...user,
        generatedSummary: summary,
        exportDate: new Date().toISOString(),
        exportVersion: "1.0",
      };

      const jsonContent = JSON.stringify(profileData, null, 2);

      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="profile_user_${userId}.json"`,
        "Content-Length": Buffer.byteLength(jsonContent),
      });

      res.end(jsonContent);
    } catch (error) {
      console.error("Error generating profile:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal server error",
          message: "Failed to generate profile data",
        })
      );
    }
  } else if (parsedUrl.pathname === "/" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Profile Download API",
        endpoints: {
          download: {
            path: "/profile/download",
            method: "GET",
            params: {
              user_id: "required - The ID of the user (1, 2, or 3)",
            },
            example: "/profile/download?user_id=1",
          },
        },
        availableUsers: Object.keys(usersDatabase).map((id) => ({
          id: parseInt(id),
          name: usersDatabase[id].name,
        })),
      })
    );
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not found",
        message: "The requested endpoint does not exist",
      })
    );
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET / - API information`);

    `  GET /profile/download?user_id=<id> - Download user profile as JSON`
  );
  console.log(`\nAvailable user IDs: 1, 2, 3`);
  console.log(
    `\nExample: http://localhost:${PORT}/profile/download?user_id=1`
  );
});
