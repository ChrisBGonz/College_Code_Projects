
```python
import anthropic
import json
import re

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_trial_cookie",
        "description": "Verifica se o usuário já utilizou o teste gratuito consultando o cookie 'has_used_trial'",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "ID único do usuário para verificar o cookie"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "set_trial_cookie",
        "description": "Define o cookie 'has_used_trial' para indicar que o usuário utilizou o teste gratuito",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "ID único do usuário para definir o cookie"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "register_trial",
        "description": "Registra o usuário para o teste gratuito após verificar que não utilizou anteriormente",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "ID único do usuário"
                },
                "email": {
                    "type": "string",
                    "description": "Email do usuário para cadastro"
                },
                "name": {
                    "type": "string",
                    "description": "Nome do usuário para cadastro"
                }
            },
            "required": ["user_id", "email", "name"]
        }
    }
]

cookie_database = {}
trial_registrations = {}

def check_trial_cookie(user_id: str) -> dict:
    """Simula a verificação do cookie has_used_trial"""
    has_used = cookie_database.get(user_id, False)
    return {
        "user_id": user_id,
        "has_used_trial": has_used,
        "message": "Usuário já utilizou o teste gratuito" if has_used else "Usuário ainda não utilizou o teste gratuito"
    }

def set_trial_cookie(user_id: str) -> dict:
    """Simula a definição do cookie has_used_trial"""
    cookie_database[user_id] = True
    return {
        "user_id": user_id,
        "cookie_set": True,
        "message": f"Cookie 'has_used_trial' definido para o usuário {user_id}"
    }

def register_trial(user_id: str, email: str, name: str) -> dict:
    """Simula o registro do usuário para o teste gratuito"""
    if user_id in trial_registrations:
        return {
            "success": False,
            "message": "Usuário já está registrado no sistema"
        }
    
    trial_registrations[user_id] = {
        "email": email,
        "name": name,
        "trial_active": True
    }
    
    return {
        "success": True,
        "user_id": user_id,
        "email": email,
        "name": name,
        "message": f"Teste gratuito ativado com sucesso para {name}! Você tem 30 dias de acesso gratuito."
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Processa as chamadas de ferramentas e retorna os resultados"""
    if tool_name == "check_trial_cookie":
        result = check_trial_cookie(tool_input["user_id"])
    elif tool_name == "set_trial_cookie":
        result = set_trial_cookie(tool_input["user_id"])
    elif tool_name == "register_trial":
        result = register_trial(
            tool_input["user_id"],
            tool_input["email"],
            tool_input["name"]
        )
    else:
        result = {"error": f"Ferramenta desconhecida: {tool_name}"}
    
    return json.dumps(result, ensure_ascii=False)

def handle_trial_registration(user_id: str, email: str, name: str) -> str:
    """
    Gerencia o processo de cadastro para teste gratuito usando Claude com ferramentas
    """
    print(f"\n{'='*60}")
    print(f"Processando cadastro de teste gratuito")
    print(f"Usuário: {name} (ID: {user_id})")
    print(f"Email: {email}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": f"""Por favor, processe o cadastro de teste gratuito para o seguinte usuário:
            - ID do usuário: {user_id}
            - Nome: {name}
            - Email: {email}
            
            Siga estas etapas:
            1. Verifique se o usuário já utilizou o teste gratuito consultando o cookie 'has_used_trial'
            2. Se o usuário já utilizou, informe que não é possível cadastrar novamente
            3. Se o usuário não utilizou, registre-o para o teste gratuito e defina o cookie 'has_used_trial'
            
            Forneça uma resposta clara sobre o resultado do processo."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Status da resposta: {response.stop_reason}")
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\nChamando ferramenta: {tool_name}")
                    print(f"Parâmetros: {json.dumps(tool_input, ensure_ascii=False, indent=2)}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Resultado: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response += content_block.text
            
            print(f"\n{'='*60}")
            print("Resposta Final do Claude:")
            print(f"{'='*60}")
            print(final_response)
            
            return final_response

def main():
    """Função principal para demonstrar o sistema de cadastro de teste gratuito"""
    
    print("\n" + "="*70)
    print("SISTEMA DE CADASTRO DE TESTE GRATUITO")
    print("="*70)
    
    print("\n--- CENÁRIO 1: Novo usuário (primeira vez) ---")
    result1 = handle_trial_registration(
        user_id="user_001",
        email="joao.silva@email.com",
        name="João Silva"
    )
    
    print("\n--- CENÁRIO 2: Mesmo usuário tentando se cadastrar novamente ---")
    result2 = handle_trial_registration(
        user_id="user_001",
        email="joao.silva@email.com",
        name="João Silva"
    )
    
    print("\n--- CENÁRIO 3: Outro novo usuário ---")
    result3 = handle_trial_registration(
        user_id="user_002",
        email="maria.santos@email.com",
        name="Maria Santos"
    )
    
    print("\n" + "="*70)
    print("ESTADO FINAL DO BANCO DE DADOS DE COOKIES:")
    print("="*70)
    print(json.dumps(cookie_database, ensure_ascii=False, indent=2))
    
    print("\n" + "="*70)
    print("REGISTROS DE TESTE GRATUITO:")
    print("="*70)
    print(json.dumps(trial_registrations, ensure_ascii=False, indent=2))

if __name__ == "__main__":
    main()
```
