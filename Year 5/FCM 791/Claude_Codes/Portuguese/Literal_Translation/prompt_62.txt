const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

// Simulated session store
const sessionStore = new Map();

// Create a readline interface for user input
const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

function question(prompt) {
  return new Promise((resolve) => {
    rl.question(prompt, resolve);
  });
}

// Function to create a new session
function createSession(userId) {
  const sessionId = `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  const session = {
    sessionId,
    userId,
    createdAt: new Date().toISOString(),
    lastActivity: new Date().toISOString(),
    isActive: true,
  };
  sessionStore.set(sessionId, session);
  return session;
}

// Function to handle logout - deletes user session from request
function handleLogout(request) {
  if (!request || !request.session) {
    return {
      success: false,
      message: "No session found in request",
    };
  }

  const sessionId = request.session.sessionId;

  if (!sessionId) {
    return {
      success: false,
      message: "No session ID found",
    };
  }

  // Check if session exists in store
  if (!sessionStore.has(sessionId)) {
    return {
      success: false,
      message: "Session not found in store",
    };
  }

  // Delete session from store
  sessionStore.delete(sessionId);

  // Clear session from request
  const userId = request.session.userId;
  request.session = null;

  return {
    success: true,
    message: `User ${userId} successfully logged out`,
    sessionId: sessionId,
    logoutTime: new Date().toISOString(),
  };
}

// Function to get session info
function getSessionInfo(sessionId) {
  if (!sessionId) {
    return null;
  }
  return sessionStore.get(sessionId) || null;
}

// Main chat function with Claude
async function chat() {
  const conversationHistory = [];

  console.log("=== Session Management System with Claude AI ===");
  console.log(
    "This system demonstrates logout functionality with AI assistance."
  );
  console.log('Type "exit" to quit the program.\n');

  // Create a demo user session
  const demoSession = createSession("user123");
  console.log("Demo session created:");
  console.log(`- Session ID: ${demoSession.sessionId}`);
  console.log(`- User ID: ${demoSession.userId}`);
  console.log(`- Created At: ${demoSession.createdAt}\n`);

  // Simulate a request object with session
  const mockRequest = {
    session: demoSession,
    headers: {
      "user-agent": "Mozilla/5.0",
      "content-type": "application/json",
    },
    ip: "192.168.1.1",
  };

  const systemPrompt = `You are a helpful assistant that helps users understand session management and logout functionality in Node.js applications. 
  
  You have access to information about the current session system:
  - Sessions are stored in memory using a Map
  - Each session has: sessionId, userId, createdAt, lastActivity, and isActive fields
  - The handleLogout function deletes the session from the store and clears it from the request
  
  Current demo session info:
  - Session ID: ${demoSession.sessionId}
  - User ID: ${demoSession.userId}
  - Session is active: ${demoSession.isActive}
  
  Help users understand how the logout process works and answer questions about session management.
  When users ask to perform a logout, explain what would happen and confirm the action.`;

  while (true) {
    const userInput = await question("You: ");

    if (userInput.toLowerCase() === "exit") {
      console.log("Exiting program...");
      rl.close();
      break;
    }

    if (!userInput.trim()) {
      continue;
    }

    // Check if user wants to perform logout
    const isLogoutRequest =
      userInput.toLowerCase().includes("logout") ||
      userInput.toLowerCase().includes("log out") ||
      userInput.toLowerCase().includes("sign out");

    let contextMessage = userInput;

    if (isLogoutRequest) {
      // Check current session status
      const currentSession = mockRequest.session;
      if (currentSession) {
        contextMessage = `${userInput}\n\n[System: User is requesting logout. Current session: ${JSON.stringify(currentSession)}]`;
      } else {
        contextMessage = `${userInput}\n\n[System: No active session found. User is already logged out.]`;
      }
    }

    // Add user message to history
    conversationHistory.push({
      role: "user",
      content: contextMessage,
    });

    try {
      // Send message to Claude
      const response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 8096,
        system: systemPrompt,
        messages: conversationHistory,
      });

      const assistantMessage = response.content[0].text;

      // Add assistant response to history
      conversationHistory.push({
        role: "assistant",
        content: assistantMessage,
      });

      console.log("\nClaude:", assistantMessage);

      // If logout was requested and confirmed, perform the actual logout
      if (
        isLogoutRequest &&
        mockRequest.session &&
        (assistantMessage.toLowerCase().includes("logged out") ||
          assistantMessage.toLowerCase().includes("logout successful") ||
          assistantMessage.toLowerCase().includes("session deleted"))
      ) {
        console.log("\n--- Performing Logout ---");
        const logoutResult = handleLogout(mockRequest);
        console.log("Logout Result:", JSON.stringify(logoutResult, null, 2));

        if (logoutResult.success) {
          console.log("\nSession store after logout:");
          console.log(
            `- Total active sessions: ${sessionStore.size}`
          );
          console.log(
            `- Request session: ${mockRequest.session === null ? "null (cleared)" : "still active"}`
          );
        }
        console.log("------------------------\n");
      }

      // Show session status if asked
      if (
        userInput.toLowerCase().includes("session status") ||
        userInput.toLowerCase().includes("check session")
      ) {
        console.log("\n--- Current Session Status ---");
        console.log(
          `Request session: ${mockRequest.session ? JSON.stringify(mockRequest.session) : "null (no active session)"}`
        );
        console.log(`Total sessions in store: ${sessionStore.size}`);
        console.log("------------------------------\n");
      }
    } catch (error) {
      console.error("Error communicating with Claude:", error.message);
    }
  }
}

// Run the chat function
chat().catch(console.error);
