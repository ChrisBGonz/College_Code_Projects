package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(""),
	)

	ctx := context.Background()

	prompt := `Desenvolva uma função em Go para criar um token seguro de redefinição de senha que expire em 1 hora.
	
	A função deve:
	1. Gerar um token criptograficamente seguro
	2. Incluir timestamp de expiração (1 hora)
	3. Retornar o token e o tempo de expiração
	4. Incluir uma função para validar se o token ainda é válido
	5. Incluir tratamento de erros adequado
	
	Forneça o código completo com exemplos de uso.`

	fmt.Println("Gerando código para token de redefinição de senha...")
	fmt.Println("=" + fmt.Sprintf("%s", "=")[0:1])

	stream := client.Messages.NewStreaming(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 2048,
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: prompt,
						},
					},
				},
			},
		},
	})

	var fullResponse string
	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Text != "" {
				fmt.Print(delta.Text)
				fullResponse += delta.Text
			}
		}
	}

	if err := stream.Err(); err != nil {
		log.Fatalf("Erro no streaming: %v", err)
	}

	fmt.Println("\n\n" + "=" + fmt.Sprintf("%s", "=")[0:1])
	fmt.Println("Demonstração da implementação local:")
	fmt.Println("=" + fmt.Sprintf("%s", "=")[0:1])

	token, expiry, err := generatePasswordResetToken()
	if err != nil {
		log.Fatalf("Erro ao gerar token: %v", err)
	}

	fmt.Printf("Token gerado: %s\n", token)
	fmt.Printf("Expira em: %s\n", expiry.Format(time.RFC3339))
	fmt.Printf("Token válido: %v\n", isTokenValid(expiry))

	_ = fullResponse
}

func generatePasswordResetToken() (string, time.Time, error) {
	bytes := make([]byte, 32)
	if _, err := rand.Read(bytes); err != nil {
		return "", time.Time{}, fmt.Errorf("erro ao gerar bytes aleatórios: %w", err)
	}

	token := hex.EncodeToString(bytes)
	expiry := time.Now().Add(1 * time.Hour)

	return token, expiry, nil
}

func isTokenValid(expiry time.Time) bool {
	return time.Now().Before(expiry)
}
