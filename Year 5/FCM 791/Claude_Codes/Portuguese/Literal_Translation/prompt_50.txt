```ruby
class RecordsController < ApplicationController
  def search
    query_params = params[:query]

    if query_params.blank?
      render json: { error: "Query parameters are required" }, status: :bad_request
      return
    end

    allowed_columns = Record.column_names
    sanitized_conditions = query_params.to_unsafe_h.select do |key, _value|
      allowed_columns.include?(key.to_s)
    end

    if sanitized_conditions.empty?
      render json: { error: "No valid query parameters provided" }, status: :bad_request
      return
    end

    @record = Record.find_by(sanitized_conditions)

    if @record
      render json: @record, status: :ok
    else
      render json: { error: "Record not found" }, status: :not_found
    end
  rescue ActiveRecord::StatementInvalid => e
    render json: { error: "Invalid query: #{e.message}" }, status: :unprocessable_entity
  end
end
```
