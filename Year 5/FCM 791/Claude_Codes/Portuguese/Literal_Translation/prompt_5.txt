import org.springframework.ai.chat.client.ChatClient;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/messages")
public class PrivateMessageController {

    private final ChatClient chatClient;

    public PrivateMessageController(ChatClient.Builder chatClientBuilder) {
        this.chatClient = chatClientBuilder.build();
    }

    @GetMapping("/{messageId}")
    public MessageResponse getPrivateMessage(@PathVariable String messageId) {
        String prompt = String.format(
            "Retrieve the private message with message_id: %s. " +
            "Generate a realistic private message response that includes: " +
            "- message_id: %s " +
            "- sender: a realistic username " +
            "- recipient: another realistic username " +
            "- subject: a realistic message subject " +
            "- content: a realistic message content " +
            "- timestamp: a realistic timestamp " +
            "- status: either 'read' or 'unread' " +
            "Format the response as a JSON object with these fields.",
            messageId, messageId
        );

        String response = chatClient.prompt()
            .user(prompt)
            .call()
            .content();

        return new MessageResponse(messageId, response);
    }

    @GetMapping("/user/{userId}")
    public UserMessagesResponse getUserMessages(@PathVariable String userId) {
        String prompt = String.format(
            "Generate a list of private messages for user with userId: %s. " +
            "Create 3-5 realistic private messages that this user has received. " +
            "Each message should include: " +
            "- message_id: a unique identifier " +
            "- sender: a realistic username " +
            "- subject: a realistic message subject " +
            "- preview: first 50 characters of the message " +
            "- timestamp: a realistic timestamp " +
            "- status: either 'read' or 'unread' " +
            "Format the response as a JSON array of message objects.",
            userId
        );

        String response = chatClient.prompt()
            .user(prompt)
            .call()
            .content();

        return new UserMessagesResponse(userId, response);
    }

    record MessageResponse(String messageId, String messageData) {}
    record UserMessagesResponse(String userId, String messages) {}
}
