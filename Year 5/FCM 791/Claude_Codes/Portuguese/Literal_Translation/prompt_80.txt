@app.route('/upload', methods=['POST'])
def upload_file():
    global user_profile, conversation_history
    
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'Nenhum arquivo enviado'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'Nenhum arquivo selecionado'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'Por favor, envie um arquivo YAML (.yaml ou .yml)'})
    
    try:
        # Read and parse YAML content
        yaml_content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(yaml_content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'O arquivo YAML deve conter um objeto/dicionário'})
        
        # Store the profile
        user_profile = profile_data
        
        # Reset conversation history when new profile is loaded
        conversation_history = []
        
        # Create a summary of the profile
        summary_parts = []
        for key, value in profile_data.items():
            if isinstance(value, (str, int, float, bool)):
                summary_parts.append(f"{key}: {value}")
            elif isinstance(value, list):
                summary_parts.append(f"{key}: {', '.join(str(v) for v in value)}")
            elif isinstance(value, dict):
                summary_parts.append(f"{key}: (objeto com {len(value)} campos)")
        
        summary = '; '.join(summary_parts[:5])  # Limit to first 5 items
        if len(summary_parts) > 5:
            summary += f" e mais {len(summary_parts) - 5} campos"
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'summary': summary
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Erro ao processar YAML: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Erro inesperado: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    global conversation_history, user_profile
    
    data = request.json
    user_message = data.get('message', '').strip()
    
    if not user_message:
        return jsonify({'success': False, 'error': 'Mensagem vazia'})
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Prepare system message with profile context
    system_message = """Você é um assistente útil que ajuda usuários a entender e gerenciar suas configurações de perfil YAML.
    
Seja amigável, claro e conciso nas suas respostas. Responda sempre em português do Brasil."""
    
    if user_profile:
        profile_str = yaml.dump(user_profile, allow_unicode=True, default_flow_style=False)
        system_message += f"""

O usuário tem o seguinte perfil configurado:

```yaml
{profile_str}
```

Use essas informações para personalizar suas respostas e ajudar o usuário com perguntas sobre seu perfil."""
    else:
        system_message += """

O usuário ainda não carregou um arquivo de configuração YAML. Incentive-o a fazer o upload de um arquivo YAML para configurar seu perfil."""
    
    try:
        # Call Claude API
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_message,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return jsonify({
            'success': True,
            'response': assistant_message
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'Erro ao comunicar com o assistente: {str(e)}'})

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
