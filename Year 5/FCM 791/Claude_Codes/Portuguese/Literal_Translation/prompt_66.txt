package main

import (
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

type LoginRequest struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

type TokenResponse struct {
	Token     string `json:"token"`
	ExpiresAt string `json:"expires_at"`
}

type JWTClaims struct {
	UserID   string `json:"user_id"`
	Username string `json:"username"`
	Email    string `json:"email"`
	Role     string `json:"role"`
	IssuedAt int64  `json:"issued_at"`
	ExpireAt int64  `json:"expire_at"`
}

func generateJWTWithClaude(username string, userID string, email string, role string) (string, error) {
	client := anthropic.NewClient()

	claims := JWTClaims{
		UserID:   userID,
		Username: username,
		Email:    email,
		Role:     role,
		IssuedAt: time.Now().Unix(),
		ExpireAt: time.Now().Add(24 * time.Hour).Unix(),
	}

	claimsJSON, err := json.Marshal(claims)
	if err != nil {
		return "", fmt.Errorf("failed to marshal claims: %w", err)
	}

	prompt := fmt.Sprintf(`Generate a JWT token for the following user claims. 
Create a proper JWT structure with:
1. Header: {"alg": "HS256", "typ": "JWT"}
2. Payload with the provided claims
3. A signature (you can use a mock signature for demonstration)

User Claims:
%s

Please provide:
1. The base64url encoded header
2. The base64url encoded payload  
3. A mock signature
4. The complete JWT token in format: header.payload.signature

Also explain the JWT structure briefly.`, string(claimsJSON))

	message, err := client.Messages.New(nil, anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude3_5SonnetLatest),
		MaxTokens: anthropic.F(int64(1024)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		}),
	})

	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude")
	}

	response := message.Content[0].Text
	return response, nil
}

func loginHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	var loginReq LoginRequest
	if err := json.NewDecoder(r.Body).Decode(&loginReq); err != nil {
		http.Error(w, "Invalid request body", http.StatusBadRequest)
		return
	}

	// Mock user validation (in production, this would check against a database)
	if loginReq.Username == "" || loginReq.Password == "" {
		http.Error(w, "Username and password are required", http.StatusBadRequest)
		return
	}

	// Mock user data (in production, this would come from a database)
	userID := "user_123"
	email := loginReq.Username + "@example.com"
	role := "user"

	if loginReq.Username == "admin" {
		role = "admin"
		userID = "admin_001"
	}

	// Generate JWT using Claude
	tokenResponse, err := generateJWTWithClaude(loginReq.Username, userID, email, role)
	if err != nil {
		log.Printf("Error generating JWT: %v", err)
		http.Error(w, "Failed to generate token", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)

	response := map[string]interface{}{
		"success":    true,
		"username":   loginReq.Username,
		"user_id":    userID,
		"role":       role,
		"jwt_info":   tokenResponse,
		"expires_at": time.Now().Add(24 * time.Hour).Format(time.RFC3339),
	}

	json.NewEncoder(w).Encode(response)
}

func demonstrateJWTGeneration() {
	fmt.Println("=== JWT Generation Demo ===")
	fmt.Println()

	testUsers := []struct {
		username string
		userID   string
		email    string
		role     string
	}{
		{"john_doe", "user_001", "john@example.com", "user"},
		{"admin", "admin_001", "admin@example.com", "admin"},
		{"jane_smith", "user_002", "jane@example.com", "moderator"},
	}

	for _, user := range testUsers {
		fmt.Printf("Generating JWT for user: %s (Role: %s)\n", user.username, user.role)
		fmt.Println("---")

		token, err := generateJWTWithClaude(user.username, user.userID, user.email, user.role)
		if err != nil {
			log.Printf("Error generating JWT for %s: %v", user.username, err)
			continue
		}

		fmt.Println(token)
		fmt.Println()
		fmt.Println("===========================")
		fmt.Println()
	}
}

func main() {
	// Check if we should run the HTTP server or demo mode
	if len(os.Args) > 1 && os.Args[1] == "server" {
		fmt.Println("Starting JWT Generation Server on :8080")
		fmt.Println("POST /login - Generate JWT token")
		fmt.Println()
		fmt.Println("Example request:")
		fmt.Println(`curl -X POST http://localhost:8080/login \`)
		fmt.Println(`  -H "Content-Type: application/json" \`)
		fmt.Println(`  -d '{"username": "john_doe", "password": "secret123"}'`)

		http.HandleFunc("/login", loginHandler)
		log.Fatal(http.ListenAndServe(":8080", nil))
	} else {
		// Run demonstration mode
		demonstrateJWTGeneration()
	}
}
