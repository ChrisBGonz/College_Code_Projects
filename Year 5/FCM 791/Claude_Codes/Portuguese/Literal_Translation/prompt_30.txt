
```cpp
#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <vector>

int main() {
    // Initialize the Anthropic client
    anthropic::Client client;
    
    // Create a conversation history
    std::vector<anthropic::Message> messages;
    
    std::cout << "=== C++ Image Processing Library Code Generator ===" << std::endl;
    std::cout << "This tool will help you generate C++ code for image processing with libjpeg." << std::endl;
    std::cout << "Type 'quit' to exit." << std::endl;
    std::cout << std::endl;
    
    // Initial system context
    std::string system_prompt = R"(You are an expert C++ developer specializing in image processing libraries. 
    You help developers write C++ code that links with external image processing libraries like libjpeg, libpng, OpenCV, etc.
    When asked about code, provide complete, compilable C++ examples with proper error handling.
    Always include compilation instructions and library linking flags.
    Focus on practical, production-ready code examples.)";
    
    while (true) {
        std::cout << "You: ";
        std::string user_input;
        std::getline(std::cin, user_input);
        
        if (user_input == "quit" || user_input == "exit") {
            std::cout << "Goodbye!" << std::endl;
            break;
        }
        
        if (user_input.empty()) {
            continue;
        }
        
        // Add user message to history
        messages.push_back({
            anthropic::Role::User,
            user_input
        });
        
        try {
            // Create the API request
            anthropic::CreateMessageParams params;
            params.model = "claude-opus-4-5";
            params.max_tokens = 4096;
            params.system = system_prompt;
            params.messages = messages;
            
            // Make the API call
            auto response = client.messages.create(params);
            
            // Extract the assistant's response
            std::string assistant_response;
            for (const auto& block : response.content) {
                if (block.type == "text") {
                    assistant_response += block.text;
                }
            }
            
            // Add assistant response to history
            messages.push_back({
                anthropic::Role::Assistant,
                assistant_response
            });
            
            std::cout << "\nAssistant: " << assistant_response << std::endl;
            std::cout << std::endl;
            
        } catch (const anthropic::APIError& e) {
            std::cerr << "API Error: " << e.what() << std::endl;
            std::cerr << "Status code: " << e.status_code() << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "Error: " << e.what() << std::endl;
        }
    }
    
    return 0;
}
```
