package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func readConfigPathFromEnv(envVarName string) (string, error) {
	configPath := os.Getenv(envVarName)
	if configPath == "" {
		return "", fmt.Errorf("environment variable %s is not set or empty", envVarName)
	}

	if _, err := os.Stat(configPath); os.IsNotExist(err) {
		return "", fmt.Errorf("config file does not exist at path: %s", configPath)
	}

	return configPath, nil
}

func generateGoFunction(client *anthropic.Client, envVarName string) (string, error) {
	prompt := fmt.Sprintf(`Create a Go function that reads a configuration file path from an environment variable named "%s". 
The function should:
1. Read the environment variable
2. Validate that the variable is set
3. Check if the file exists at the given path
4. Return the path and any errors
5. Include proper error handling

Please provide only the Go code without any explanation.`, envVarName)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude3_5SonnetLatest),
		MaxTokens: anthropic.F(int64(1024)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		}),
	})

	if err != nil {
		return "", fmt.Errorf("error calling Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("no content in response")
	}

	var result string
	for _, block := range message.Content {
		if block.Type == "text" {
			result += block.Text
		}
	}

	return result, nil
}

func main() {
	client := anthropic.NewClient()

	envVarName := "CONFIG_FILE_PATH"
	if len(os.Args) > 1 {
		envVarName = os.Args[1]
	}

	fmt.Printf("Generating Go function for reading config path from environment variable: %s\n\n", envVarName)

	generatedCode, err := generateGoFunction(client, envVarName)
	if err != nil {
		log.Fatalf("Error generating function: %v", err)
	}

	fmt.Println("Generated Go Function:")
	fmt.Println("======================")
	fmt.Println(generatedCode)
	fmt.Println()

	fmt.Println("Testing the local implementation:")
	fmt.Println("==================================")

	testEnvVar := "TEST_CONFIG_PATH"
	configPath, err := readConfigPathFromEnv(testEnvVar)
	if err != nil {
		fmt.Printf("Expected error (env var not set): %v\n", err)
	}

	tmpFile, err := os.CreateTemp("", "config-*.yaml")
	if err != nil {
		log.Fatalf("Error creating temp file: %v", err)
	}
	defer os.Remove(tmpFile.Name())
	tmpFile.Close()

	os.Setenv(testEnvVar, tmpFile.Name())
	defer os.Unsetenv(testEnvVar)

	configPath, err = readConfigPathFromEnv(testEnvVar)
	if err != nil {
		fmt.Printf("Error reading config path: %v\n", err)
	} else {
		fmt.Printf("Successfully read config path: %s\n", configPath)
	}

	os.Setenv(testEnvVar, "/nonexistent/path/config.yaml")
	configPath, err = readConfigPathFromEnv(testEnvVar)
	if err != nil {
		fmt.Printf("Expected error (file not found): %v\n", err)
	} else {
		fmt.Printf("Config path: %s\n", configPath)
	}
}
