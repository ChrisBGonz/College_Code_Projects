```ruby
class PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :find_staff_member, only: [:create]

  def create
    if @staff_member.staff?
      if @staff_member.update(role: :manager)
        render json: {
          message: "User #{@staff_member.name} has been successfully promoted to Manager.",
          user: {
            id: @staff_member.id,
            name: @staff_member.name,
            email: @staff_member.email,
            role: @staff_member.role
          }
        }, status: :ok
      else
        render json: {
          error: "Failed to promote user.",
          details: @staff_member.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        error: "User is not a Staff member and cannot be promoted."
      }, status: :unprocessable_entity
    end
  end

  private

  def authorize_manager!
    unless current_user.manager?
      render json: {
        error: "Access denied. Only Managers can promote Staff members."
      }, status: :forbidden
    end
  end

  def find_staff_member
    @staff_member = User.find_by(id: params[:user_id])

    unless @staff_member
      render json: {
        error: "Staff member not found."
      }, status: :not_found
    end
  end
end
```
