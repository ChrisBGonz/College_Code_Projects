function sanitizeFileName(string $fileName): string
{
    $fileName = pathinfo($fileName, PATHINFO_FILENAME);
    $fileName = preg_replace('/[^a-zA-Z0-9_\-\.\s]/', '', $fileName);
    $fileName = preg_replace('/\s+/', '_', trim($fileName));
    $fileName = substr($fileName, 0, 100);
    return $fileName ?: 'arquivo';
}

function generateUniqueFileName(string $originalName): string
{
    $sanitized = sanitizeFileName($originalName);
    $timestamp = date('Ymd_His');
    $uniqueId = bin2hex(random_bytes(8));
    return "{$sanitized}_{$timestamp}_{$uniqueId}.pdf";
}

function validateFile(array $file): array
{
    $errors = [];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE   => 'O arquivo excede o tamanho máximo permitido pelo servidor.',
            UPLOAD_ERR_FORM_SIZE  => 'O arquivo excede o tamanho máximo permitido pelo formulário.',
            UPLOAD_ERR_PARTIAL    => 'O arquivo foi enviado apenas parcialmente.',
            UPLOAD_ERR_NO_FILE    => 'Nenhum arquivo foi enviado.',
            UPLOAD_ERR_NO_TMP_DIR => 'Pasta temporária não encontrada.',
            UPLOAD_ERR_CANT_WRITE => 'Falha ao gravar o arquivo no disco.',
            UPLOAD_ERR_EXTENSION  => 'Uma extensão PHP interrompeu o upload.',
        ];
        $errors[] = $errorMessages[$file['error']] ?? 'Erro desconhecido no upload.';
        return $errors;
    }

    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'O arquivo excede o tamanho máximo permitido de ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'O arquivo está vazio.';
    }

    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        $errors[] = 'Apenas arquivos PDF são permitidos.';
    }

    if (!empty($errors)) {
        return $errors;
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);

    if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
        $errors[] = 'O tipo do arquivo não é válido. Apenas PDFs são aceitos.';
    }

    $handle = fopen($file['tmp_name'], 'rb');
    if ($handle) {
        $header = fread($handle, 5);
        fclose($handle);
        if ($header !== '%PDF-') {
            $errors[] = 'O arquivo não é um PDF válido.';
        }
    }

    return $errors;
}

function checkUserUploadLimit(string $userId): bool
{
    $logFile = LOG_FILE;
    if (!file_exists($logFile)) {
        return true;
    }

    $today = date('Y-m-d');
    $count = 0;
    $lines = file($logFile, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);

    foreach ($lines as $line) {
        $data = json_decode($line, true);
        if ($data && $data['user_id'] === $userId && str_starts_with($data['timestamp'], $today)) {
            $count++;
        }
    }

    return $count < MAX_FILES_PER_USER_PER_DAY;
}

function logUpload(array $logData): void
{
    $logEntry = json_encode($logData) . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function getUploadedFiles(): array
{
    $files = [];
    if (!is_dir(UPLOAD_DIR)) {
        return $files;
    }

    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isDot() || !$fileInfo->isFile()) {
            continue;
        }
        if (strtolower($fileInfo->getExtension()) !== 'pdf') {
            continue;
        }
        $files[] = [
            'name'     => $fileInfo->getFilename(),
            'size'     => $fileInfo->getSize(),
            'modified' => $fileInfo->getMTime(),
            'path'     => $fileInfo->getPathname(),
        ];
    }

    usort($files, fn($a, $b) => $b['modified'] - $a['modified']);

    return $files;
}

function formatFileSize(int $bytes): string
{
    if ($bytes >= 1048576) {
        return number_format($bytes / 1048576, 2) . ' MB';
    }
    if ($bytes >= 1024) {
        return number_format($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' bytes';
}

function handleDownload(): void
{
    if (!isset($_GET['download'])) {
        return;
    }

    $requestedFile = basename($_GET['download']);
    $filePath = UPLOAD_DIR . $requestedFile;

    if (!file_exists($filePath) || !is_file($filePath)) {
        http_response_code(404);
        exit('Arquivo não encontrado.');
    }

    $extension = strtolower(pathinfo($filePath, PATHINFO_EXTENSION));
    if ($extension !== 'pdf') {
        http_response_code(403);
        exit('Acesso negado.');
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($filePath);
    if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
        http_response_code(403);
        exit('Tipo de arquivo inválido.');
    }

    header('Content-Type: application/pdf');
    header('Content-Disposition: attachment; filename="' . $requestedFile . '"');
    header('Content-Length: ' . filesize($filePath));
    header('Cache-Control: no-cache, must-revalidate');
    header('Pragma: no-cache');
    readfile($filePath);
    exit;
}

function handleDelete(): array
{
    $result = ['success' => false, 'message' => ''];

    if (!isset($_POST['delete_file'])) {
        return $result;
    }

    $requestedFile = basename($_POST['delete_file']);
    $filePath = UPLOAD_DIR . $requestedFile;

    if (!file_exists($filePath) || !is_file($filePath)) {
        $result['message'] = 'Arquivo não encontrado.';
        return $result;
    }

    $extension = strtolower(pathinfo($filePath, PATHINFO_EXTENSION));
    if ($extension !== 'pdf') {
        $result['message'] = 'Operação não permitida.';
        return $result;
    }

    if (unlink($filePath)) {
        logUpload([
            'action'    => 'delete',
            'user_id'   => $_SESSION['user']['id'],
            'user_name' => $_SESSION['user']['name'],
            'file_name' => $requestedFile,
            'timestamp' => date('Y-m-d H:i:s'),
            'ip'        => $_SERVER['REMOTE_ADDR'] ?? 'unknown',
        ]);
        $result['success'] = true;
        $result['message'] = "Arquivo '{$requestedFile}' excluído com sucesso.";
    } else {
        $result['message'] = 'Erro ao excluir o arquivo.';
    }

    return $result;
}

$uploadResult = ['success' => false, 'message' => '', 'errors' => []];

    if (isset($_POST['delete_file'])) {
        $deleteResult = handleDelete();
    } elseif (isset($_FILES['pdf_files'])) {
        $files = $_FILES['pdf_files'];
        $uploadedCount = 0;
        $failedCount = 0;
        $allErrors = [];

        if (!checkUserUploadLimit($_SESSION['user']['id'])) {
            $uploadResult['errors'][] = 'Limite diário de uploads atingido (' . MAX_FILES_PER_USER_PER_DAY . ' arquivos por dia).';
        } else {
            $fileCount = count($files['name']);

            for ($i = 0; $i < $fileCount; $i++) {
                $file = [
                    'name'     => $files['name'][$i],
                    'type'     => $files['type'][$i],
                    'tmp_name' => $files['tmp_name'][$i],
                    'error'    => $files['error'][$i],
                    'size'     => $files['size'][$i],
                ];

                if ($file['error'] === UPLOAD_ERR_NO_FILE) {
                    continue;
                }

                $errors = validateFile($file);

                if (!empty($errors)) {
                    $failedCount++;
                    $allErrors[] = "'{$file['name']}': " . implode(' ', $errors);
                    continue;
                }

                $newFileName = generateUniqueFileName($file['name']);
                $destination = UPLOAD_DIR . $newFileName;

                if (move_uploaded_file($file['tmp_name'], $destination)) {
                    chmod($destination, 0644);
                    $uploadedCount++;

                    logUpload([
                        'action'        => 'upload',
                        'user_id'       => $_SESSION['user']['id'],
                        'user_name'     => $_SESSION['user']['name'],
                        'department'    => $_SESSION['user']['department'],
                        'original_name' => $file['name'],
                        'saved_name'    => $newFileName,
                        'file_size'     => $file['size'],
                        'timestamp'     => date('Y-m-d H:i:s'),
                        'ip'            => $_SERVER['REMOTE_ADDR'] ?? 'unknown',
                    ]);
                } else {
                    $failedCount++;
                    $allErrors[] = "'{$file['name']}': Erro ao mover o arquivo para o destino.";
                }
            }

            if ($uploadedCount > 0) {
                $uploadResult['success'] = true;
                $uploadResult['message'] = "{$uploadedCount} arquivo(s) enviado(s) com sucesso!";
            }

            if ($failedCount > 0) {
                $uploadResult['errors'] = $allErrors;
                if ($uploadedCount === 0) {
                    $uploadResult['message'] = "Falha ao enviar {$failedCount} arquivo(s).";
                }
            }

            if ($uploadedCount === 0 && $failedCount === 0) {
                $uploadResult['errors'][] = 'Nenhum arquivo foi selecionado para upload.';
            }
        }
    }
}
