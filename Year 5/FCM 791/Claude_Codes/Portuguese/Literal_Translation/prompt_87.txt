#!/usr/bin/env python3

import os
import time
import re
from collections import deque
from datetime import datetime, timedelta
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def analyze_log_with_claude(failed_logins_count: int, recent_attempts: list, log_file: str) -> str:
    """Use Claude to analyze the security situation and provide recommendations."""
    
    attempts_info = "\n".join([f"  - {attempt}" for attempt in recent_attempts[-10:]])
    
    user_message = f"""I'm monitoring a log file '{log_file}' and detected a security alert:

- Failed login attempts in the last minute: {failed_logins_count}
- Threshold exceeded: {failed_logins_count} > 10 attempts

Recent failed login attempts:
{attempts_info}

Please analyze this security situation and provide:
1. Assessment of the threat level
2. Possible causes (brute force, credential stuffing, etc.)
3. Immediate recommended actions
4. Long-term security improvements"""

    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a cybersecurity expert specializing in log analysis and intrusion detection. 
        You help system administrators understand and respond to security threats detected in log files.
        Provide clear, actionable advice when security anomalies are detected.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def ask_followup_question(question: str) -> str:
    """Allow users to ask follow-up questions about the security alert."""
    
    conversation_history.append({
        "role": "user",
        "content": question
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a cybersecurity expert specializing in log analysis and intrusion detection. 
        You help system administrators understand and respond to security threats detected in log files.
        Provide clear, actionable advice when security anomalies are detected.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def monitor_log_file(log_file_path: str, check_interval: int = 5):
    """Monitor a log file for failed login attempts."""
    
    print(f"🔍 Starting log monitor for: {log_file_path}")
    print(f"⏱️  Checking every {check_interval} seconds")
    print(f"🚨 Alert threshold: >10 failed logins per minute")
    print("-" * 60)
    
    failed_login_pattern = re.compile(
        r'(?i)(failed\s+login|authentication\s+failure|invalid\s+password|'
        r'login\s+failed|failed\s+password|invalid\s+user)',
        re.IGNORECASE
    )
    
    failed_logins_window = deque()
    
    last_position = 0
    alert_sent = False
    last_alert_time = None
    
    if not os.path.exists(log_file_path):
        print(f"⚠️  Log file not found: {log_file_path}")
        print("📝 Creating a demo log file for testing...")
        create_demo_log(log_file_path)
    
    try:
        while True:
            current_time = datetime.now()
            
            try:
                with open(log_file_path, 'r') as f:
                    f.seek(last_position)
                    new_lines = f.readlines()
                    last_position = f.tell()
                
                for line in new_lines:
                    if failed_login_pattern.search(line):
                        failed_logins_window.append({
                            'time': current_time,
                            'line': line.strip()
                        })
                        print(f"⚠️  Failed login detected: {line.strip()[:80]}...")
                
                one_minute_ago = current_time - timedelta(minutes=1)
                while failed_logins_window and failed_logins_window[0]['time'] < one_minute_ago:
                    failed_logins_window.popleft()
                
                failed_count = len(failed_logins_window)
                
                if failed_count > 10:
                    should_alert = True
                    if last_alert_time and (current_time - last_alert_time).seconds < 60:
                        should_alert = False
                    
                    if should_alert:
                        print("\n" + "=" * 60)
                        print("🚨 SECURITY ALERT: EXCESSIVE FAILED LOGIN ATTEMPTS!")
                        print("=" * 60)
                        print(f"📊 Failed attempts in last minute: {failed_count}")
                        print(f"⏰ Alert time: {current_time.strftime('%Y-%m-%d %H:%M:%S')}")
                        print("\n🤖 Analyzing with Claude AI...")
                        
                        recent_attempts = [item['line'] for item in list(failed_logins_window)]
                        
                        analysis = analyze_log_with_claude(
                            failed_count, 
                            recent_attempts, 
                            log_file_path
                        )
                        
                        print("\n📋 CLAUDE'S SECURITY ANALYSIS:")
                        print("-" * 60)
                        print(analysis)
                        print("-" * 60)
                        
                        last_alert_time = current_time
                        alert_sent = True
                        
                        print("\n💬 You can ask follow-up questions (or press Enter to continue monitoring):")
                        print("   Type 'continue' to resume monitoring")
                        
                        while True:
                            user_input = input("\n❓ Your question: ").strip()
                            
                            if not user_input or user_input.lower() == 'continue':
                                break
                            
                            print("\n🤖 Claude's response:")
                            print("-" * 60)
                            followup_response = ask_followup_question(user_input)
                            print(followup_response)
                            print("-" * 60)
                        
                        print("\n▶️  Resuming log monitoring...")
                        print("=" * 60 + "\n")
                
                elif alert_sent and failed_count <= 5:
                    print(f"✅ Alert resolved. Failed attempts now: {failed_count}")
                    alert_sent = False
                
            except FileNotFoundError:
                print(f"❌ Log file not found: {log_file_path}")
            except PermissionError:
                print(f"❌ Permission denied reading: {log_file_path}")
            
            time.sleep(check_interval)
            
    except KeyboardInterrupt:
        print("\n\n🛑 Log monitoring stopped by user")
        print(f"📊 Total conversation turns with Claude: {len(conversation_history) // 2}")

def create_demo_log(log_file_path: str):
    """Create a demo log file for testing purposes."""
    demo_entries = [
        "2024-01-15 10:00:01 INFO User admin logged in successfully",
        "2024-01-15 10:00:05 WARNING Failed login attempt for user: john",
        "2024-01-15 10:00:10 INFO System backup completed",
        "2024-01-15 10:00:15 WARNING Authentication failure for user: admin",
        "2024-01-15 10:00:20 INFO New connection from 192.168.1.100",
    ]
    
    with open(log_file_path, 'w') as f:
        f.write("\n".join(demo_entries) + "\n")
    
    print(f"✅ Demo log file created: {log_file_path}")
    print("💡 To test the alert, add multiple 'Failed Login' entries to the log file")
    print("   Example: echo 'Failed Login attempt' >> " + log_file_path)

def simulate_attack(log_file_path: str):
    """Simulate a brute force attack for testing."""
    print(f"\n🔴 Simulating brute force attack on {log_file_path}...")
    
    attack_entries = []
    current_time = datetime.now()
    
    for i in range(15):
        timestamp = current_time.strftime('%Y-%m-%d %H:%M:%S')
        ip = f"192.168.1.{100 + i}"
        attack_entries.append(
            f"{timestamp} WARNING Failed Login attempt from {ip} for user: admin"
        )
        time.sleep(0.1)
    
    with open(log_file_path, 'a') as f:
        f.write("\n".join(attack_entries) + "\n")
    
    print(f"✅ Added {len(attack_entries)} failed login entries to simulate attack")

def main():
    """Main function to run the log monitor."""
    import sys
    
    print("🛡️  Log File Security Monitor with Claude AI")
    print("=" * 60)
    
    if len(sys.argv) > 1:
        log_file = sys.argv[1]
    else:
        log_file = "/tmp/security_test.log"
        print(f"📁 No log file specified. Using demo file: {log_file}")
    
    if len(sys.argv) > 2 and sys.argv[2] == '--simulate':
        if not os.path.exists(log_file):
            create_demo_log(log_file)
        
        import threading
        
        def delayed_attack():
            time.sleep(10)
            simulate_attack(log_file)
        
        attack_thread = threading.Thread(target=delayed_attack, daemon=True)
        attack_thread.start()
        print("⏳ Attack simulation will start in 10 seconds...")
    
    monitor_log_file(log_file, check_interval=5)

if __name__ == "__main__":
    main()
