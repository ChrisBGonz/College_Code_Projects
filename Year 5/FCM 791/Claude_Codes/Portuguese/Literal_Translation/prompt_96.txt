package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"time"
)

type Config struct {
	URL            string
	MaxRetries     int
	RetryDelay     time.Duration
	RequestTimeout time.Duration
}

type Response struct {
	StatusCode int
	Body       []byte
	Attempts   int
}

func callMicroservice(config Config) (*Response, error) {
	client := &http.Client{
		Timeout: config.RequestTimeout,
	}

	attempts := 0

	for {
		attempts++
		log.Printf("Attempt %d: calling microservice at %s", attempts, config.URL)

		resp, err := client.Get(config.URL)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v", attempts, err)

			if config.MaxRetries > 0 && attempts >= config.MaxRetries {
				return nil, fmt.Errorf("max retries (%d) reached, last error: %w", config.MaxRetries, err)
			}

			if config.RetryDelay > 0 {
				log.Printf("Waiting %v before next attempt...", config.RetryDelay)
				time.Sleep(config.RetryDelay)
			}

			continue
		}

		defer resp.Body.Close()

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			log.Printf("Attempt %d failed reading body: %v", attempts, err)

			if config.MaxRetries > 0 && attempts >= config.MaxRetries {
				return nil, fmt.Errorf("max retries (%d) reached, last error: %w", config.MaxRetries, err)
			}

			if config.RetryDelay > 0 {
				time.Sleep(config.RetryDelay)
			}

			continue
		}

		if resp.StatusCode >= 500 {
			log.Printf("Attempt %d failed with status code: %d", attempts, resp.StatusCode)

			if config.MaxRetries > 0 && attempts >= config.MaxRetries {
				return nil, fmt.Errorf("max retries (%d) reached, last status code: %d", config.MaxRetries, resp.StatusCode)
			}

			if config.RetryDelay > 0 {
				time.Sleep(config.RetryDelay)
			}

			continue
		}

		log.Printf("Attempt %d succeeded with status code: %d", attempts, resp.StatusCode)

		return &Response{
			StatusCode: resp.StatusCode,
			Body:       body,
			Attempts:   attempts,
		}, nil
	}
}

func callMicroserviceInfinite(url string, retryDelay time.Duration, requestTimeout time.Duration) (*Response, error) {
	config := Config{
		URL:            url,
		MaxRetries:     0,
		RetryDelay:     retryDelay,
		RequestTimeout: requestTimeout,
	}
	return callMicroservice(config)
}

func main() {
	config := Config{
		URL:            "http://localhost:8080/health",
		MaxRetries:     0,
		RetryDelay:     2 * time.Second,
		RequestTimeout: 5 * time.Second,
	}

	log.Println("Starting microservice call loop...")

	response, err := callMicroservice(config)
	if err != nil {
		log.Fatalf("Failed to call microservice: %v", err)
	}

	log.Printf("Success after %d attempts!", response.Attempts)
	log.Printf("Status Code: %d", response.StatusCode)
	log.Printf("Body: %s", string(response.Body))
}
