import os
from flask import Flask, request, jsonify, session
from anthropic import Anthropic
import sqlite3
import hashlib
import secrets

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Database setup
def init_db():
    conn = sqlite3.connect('users.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            email TEXT
        )
    ''')
    # Insert a test user
    test_password = hashlib.sha256("oldpassword123".encode()).hexdigest()
    cursor.execute('''
        INSERT OR IGNORE INTO users (id, username, password_hash, email) 
        VALUES (1, 'testuser', ?, 'test@example.com')
    ''', (test_password,))
    conn.commit()
    conn.close()

def hash_password(password: str) -> str:
    return hashlib.sha256(password.encode()).hexdigest()

def update_password_in_db(user_id: int, new_password: str) -> dict:
    """Update password in database for given user_id"""
    try:
        conn = sqlite3.connect('users.db')
        cursor = conn.cursor()
        
        # Check if user exists
        cursor.execute('SELECT id, username FROM users WHERE id = ?', (user_id,))
        user = cursor.fetchone()
        
        if not user:
            conn.close()
            return {"success": False, "message": f"User with ID {user_id} not found"}
        
        # Hash and update password
        password_hash = hash_password(new_password)
        cursor.execute('UPDATE users SET password_hash = ? WHERE id = ?', 
                      (password_hash, user_id))
        conn.commit()
        conn.close()
        
        return {
            "success": True, 
            "message": f"Password successfully updated for user '{user[1]}' (ID: {user_id})"
        }
    except Exception as e:
        return {"success": False, "message": f"Database error: {str(e)}"}

def get_user_info(user_id: int) -> dict:
    """Get user information from database"""
    try:
        conn = sqlite3.connect('users.db')
        cursor = conn.cursor()
        cursor.execute('SELECT id, username, email FROM users WHERE id = ?', (user_id,))
        user = cursor.fetchone()
        conn.close()
        
        if user:
            return {"success": True, "user": {"id": user[0], "username": user[1], "email": user[2]}}
        return {"success": False, "message": "User not found"}
    except Exception as e:
        return {"success": False, "message": str(e)}

# Define tools for Claude
tools = [
    {
        "name": "update_password",
        "description": "Updates the password for a user in the database. Validates the new password meets security requirements before updating.",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "integer",
                    "description": "The ID of the user whose password needs to be updated"
                },
                "new_password": {
                    "type": "string",
                    "description": "The new password to set for the user"
                }
            },
            "required": ["user_id", "new_password"]
        }
    },
    {
        "name": "get_user_info",
        "description": "Retrieves information about a user from the database",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "integer",
                    "description": "The ID of the user to retrieve information for"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "validate_password_strength",
        "description": "Validates if a password meets security requirements (minimum 8 characters, contains uppercase, lowercase, and numbers)",
        "input_schema": {
            "type": "object",
            "properties": {
                "password": {
                    "type": "string",
                    "description": "The password to validate"
                }
            },
            "required": ["password"]
        }
    }
]

def validate_password_strength(password: str) -> dict:
    """Validate password strength"""
    errors = []
    
    if len(password) < 8:
        errors.append("Password must be at least 8 characters long")
    if not any(c.isupper() for c in password):
        errors.append("Password must contain at least one uppercase letter")
    if not any(c.islower() for c in password):
        errors.append("Password must contain at least one lowercase letter")
    if not any(c.isdigit() for c in password):
        errors.append("Password must contain at least one number")
    
    if errors:
        return {"valid": False, "errors": errors}
    return {"valid": True, "message": "Password meets all security requirements"}

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude"""
    if tool_name == "update_password":
        result = update_password_in_db(tool_input["user_id"], tool_input["new_password"])
        return str(result)
    elif tool_name == "get_user_info":
        result = get_user_info(tool_input["user_id"])
        return str(result)
    elif tool_name == "validate_password_strength":
        result = validate_password_strength(tool_input["password"])
        return str(result)
    else:
        return f"Unknown tool: {tool_name}"

@app.route('/change-password', methods=['POST'])
def change_password():
    """
    Endpoint to change user password using Claude AI with tool use
    
    Expected JSON body:
    {
        "nova_senha": "NewPassword123",
        "user_id": 1  # Optional, defaults to current session user or 1 for demo
    }
    """
    data = request.get_json()
    
    if not data or 'nova_senha' not in data:
        return jsonify({"error": "nova_senha is required"}), 400
    
    nova_senha = data['nova_senha']
    # In a real app, user_id would come from session/JWT token
    user_id = data.get('user_id', session.get('user_id', 1))
    
    # Initialize conversation history for multi-turn
    conversation_history = []
    
    # Initial user message
    user_message = f"""I need to change the password for user ID {user_id}. 
    The new password is: {nova_senha}
    
    Please:
    1. First get the user information to confirm the user exists
    2. Validate the password strength
    3. If the password is strong enough, update it in the database
    4. Provide a clear summary of what happened"""
    
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    final_response = None
    max_iterations = 10
    iteration = 0
    
    # Agentic loop
    while iteration < max_iterations:
        iteration += 1
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=conversation_history
        )
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": response.content
        })
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract final text response
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            break
        
        # Process tool calls if any
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_result = process_tool_call(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": tool_result
                    })
            
            # Add tool results to conversation
            if tool_results:
                conversation_history.append({
                    "role": "user",
                    "content": tool_results
                })
        else:
            # No more tool calls, extract response
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            break
    
    if not final_response:
        final_response = "Password change process completed."
    
    return jsonify({
        "success": True,
        "message": final_response,
        "user_id": user_id
    })

@app.route('/chat', methods=['POST'])
def chat():
    """
    General chat endpoint for password-related queries with conversation history
    """
    data = request.get_json()
    
    if not data or 'message' not in data:
        return jsonify({"error": "message is required"}), 400
    
    user_message = data['message']
    
    # Get or initialize conversation history from session
    if 'conversation_history' not in session:
        session['conversation_history'] = []
    
    conversation_history = session['conversation_history']
    
    # Add user message
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    final_response = None
    max_iterations = 10
    iteration = 0
    
    # Agentic loop
    while iteration < max_iterations:
        iteration += 1
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system="You are a helpful assistant for password management. You can help users change their passwords, validate password strength, and retrieve user information. Always ensure passwords meet security requirements before updating them.",
            tools=tools,
            messages=conversation_history
        )
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": response.content
        })
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            break
        
        # Process tool calls
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_result = process_tool_call(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": tool_result
                    })
            
            if tool_results:
                conversation_history.append({
                    "role": "user",
                    "content": tool_results
                })
        else:
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
            break
    
    # Save updated conversation history to session
    session['conversation_history'] = conversation_history
    session.modified = True
    
    return jsonify({
        "response": final_response or "I processed your request.",
        "conversation_length": len(conversation_history)
    })

@app.route('/reset-conversation', methods=['POST'])
def reset_conversation():
    """Reset the conversation history"""
    session.pop('conversation_history', None)
    return jsonify({"message": "Conversation history reset successfully"})

@app.route('/health', methods=['GET'])
def health():
    """Health check endpoint"""
    return jsonify({"status": "healthy", "service": "Password Change API"})
