const Anthropic = require("@anthropic-ai/sdk");
const { execSync } = require("child_process");
const os = require("os");

const client = new Anthropic();

// Define the ping tool
const tools = [
  {
    name: "ping_host",
    description:
      "Execute a ping command to check if a host is alive and measure response time",
    input_schema: {
      type: "object",
      properties: {
        domain_name: {
          type: "string",
          description:
            "The domain name or IP address to ping (e.g., google.com, 8.8.8.8)",
        },
        count: {
          type: "number",
          description: "Number of ping packets to send (default: 4)",
        },
      },
      required: ["domain_name"],
    },
  },
];

// Function to execute ping command
function executePing(domainName, count = 4) {
  try {
    const platform = os.platform();
    let command;

    // Different ping syntax for different operating systems
    if (platform === "win32") {
      command = `ping -n ${count} ${domainName}`;
    } else {
      // Linux and macOS
      command = `ping -c ${count} ${domainName}`;
    }

    console.log(`\nExecuting: ${command}`);
    const output = execSync(command, {
      encoding: "utf8",
      timeout: 30000, // 30 second timeout
    });

    return {
      success: true,
      domain: domainName,
      output: output,
      status: "Host is alive",
    };
  } catch (error) {
    // If ping fails (host unreachable), it throws an error
    return {
      success: false,
      domain: domainName,
      output: error.stdout || error.message,
      status: "Host is unreachable or does not exist",
    };
  }
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "ping_host") {
    const { domain_name, count } = toolInput;
    return executePing(domain_name, count || 4);
  }
  return { error: "Unknown tool" };
}

// Main function to check if a host is alive using Claude
async function checkHostWithClaude(domainName) {
  console.log(`\nChecking if host "${domainName}" is alive...`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please check if the host "${domainName}" is alive by pinging it. Use the ping_host tool to execute the ping command and then provide a summary of the results including whether the host is reachable and any relevant statistics.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  console.log(`\nInitial Response:`);
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop - continue until we get a final response
  while (response.stop_reason === "tool_use") {
    const toolUseBlock = response.content.find(
      (block) => block.type === "tool_use"
    );

    if (!toolUseBlock) {
      break;
    }

    const toolName = toolUseBlock.name;
    const toolInput = toolUseBlock.input;

    console.log(`\nTool Used: ${toolName}`);
    console.log(`Tool Input: ${JSON.stringify(toolInput, null, 2)}`);

    // Execute the tool
    const toolResult = processToolCall(toolName, toolInput);

    console.log(`\nTool Result:`);
    console.log(JSON.stringify(toolResult, null, 2));

    // Add assistant response and tool result to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: [
        {
          type: "tool_result",
          tool_use_id: toolUseBlock.id,
          content: JSON.stringify(toolResult),
        },
      ],
    });

    // Get next response from Claude
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log(`\nNext Response:`);
    console.log(`Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(50));
  console.log("FINAL ANALYSIS FROM CLAUDE:");
  console.log("=".repeat(50));
  console.log(finalResponse);

  return finalResponse;
}

// Example usage - check multiple hosts
async function main() {
  const hostsToCheck = ["google.com", "github.com", "nonexistent-host-12345.com"];

  for (const host of hostsToCheck) {
    try {
      await checkHostWithClaude(host);
      console.log("\n");
    } catch (error) {
      console.error(`Error checking host ${host}:`, error.message);
    }
  }
}

// Run the main function
main().catch(console.error);
