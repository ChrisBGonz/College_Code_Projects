def get_invoice_from_database(invoice_id: str) -> dict | None:
    """Retrieve invoice from the simulated database."""
    return INVOICES_DATABASE.get(invoice_id)

def analyze_invoice_with_claude(invoice_data: dict) -> str:
    """Use Claude to analyze and provide insights about the invoice."""
    conversation_history = []
    
    # Initial message to Claude with invoice data
    initial_message = f"""Analise os seguintes dados de fatura e forneça um resumo detalhado em português:

Dados da Fatura:
- ID: {invoice_data['id']}
- Cliente: {invoice_data['customer_name']}
- Email: {invoice_data['customer_email']}
- Valor Total: R$ {invoice_data['amount']:.2f}
- Status: {invoice_data['status']}
- Data de Vencimento: {invoice_data['due_date']}
- Data de Criação: {invoice_data['created_at']}
- Itens: {invoice_data['items']}

Por favor, forneça:
1. Um resumo executivo da fatura
2. Status de pagamento e urgência
3. Detalhamento dos itens
4. Recomendações baseadas no status"""

    conversation_history.append({
        "role": "user",
        "content": initial_message
    })
    
    # First turn - get initial analysis
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1000,
        system="Você é um assistente especializado em análise financeira e gestão de faturas. Forneça análises claras e objetivas em português.",
        messages=conversation_history
    )
    
    initial_analysis = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": initial_analysis
    })
    
    # Second turn - ask for risk assessment
    conversation_history.append({
        "role": "user",
        "content": "Com base nessa análise, qual é o nível de risco financeiro desta fatura e quais ações imediatas você recomenda?"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        system="Você é um assistente especializado em análise financeira e gestão de faturas. Forneça análises claras e objetivas em português.",
        messages=conversation_history
    )
    
    risk_assessment = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": risk_assessment
    })
    
    # Third turn - get final summary
    conversation_history.append({
        "role": "user",
        "content": "Por favor, forneça um resumo final conciso em 3 pontos principais sobre esta fatura."
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=300,
        system="Você é um assistente especializado em análise financeira e gestão de faturas. Forneça análises claras e objetivas em português.",
        messages=conversation_history
    )
    
    final_summary = response.content[0].text
    
    # Combine all analyses
    complete_analysis = f"""
=== ANÁLISE INICIAL ===
{initial_analysis}

=== AVALIAÇÃO DE RISCO ===
{risk_assessment}

=== RESUMO FINAL ===
{final_summary}
"""
    
    return complete_analysis

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """
    Route to retrieve invoice details by ID.
    
    Args:
        invoice_id: The unique identifier of the invoice
        
    Returns:
        JSON response with invoice details and AI analysis
    """
    # Retrieve invoice from database
    invoice_data = get_invoice_from_database(invoice_id)
    
    if not invoice_data:
        return jsonify({
            "error": "Fatura não encontrada",
            "invoice_id": invoice_id,
            "message": f"Nenhuma fatura encontrada com o ID: {invoice_id}"
        }), 404
    
    # Get AI analysis using multi-turn conversation
    ai_analysis = analyze_invoice_with_claude(invoice_data)
    
    # Prepare response
    response_data = {
        "invoice": invoice_data,
        "ai_analysis": ai_analysis,
        "status": "success"
    }
    
    return jsonify(response_data), 200

@app.route('/invoices', methods=['GET'])
def list_invoices():
    """List all available invoice IDs."""
    invoice_ids = list(INVOICES_DATABASE.keys())
    return jsonify({
        "available_invoices": invoice_ids,
        "total": len(invoice_ids)
    }), 200
