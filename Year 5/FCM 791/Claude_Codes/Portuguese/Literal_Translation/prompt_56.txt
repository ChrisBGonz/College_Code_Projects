package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

const trustedReferer = "admin.internal.com"

func checkRefererAccess(referer string) bool {
	if referer == "" {
		return false
	}
	return strings.Contains(referer, trustedReferer)
}

func adminPanelHandler(w http.ResponseWriter, r *http.Request) {
	referer := r.Header.Get("Referer")

	if !checkRefererAccess(referer) {
		http.Error(w, "Access Denied: Invalid or missing Referer header", http.StatusForbidden)
		log.Printf("Access denied - Referer: %s, IP: %s", referer, r.RemoteAddr)
		return
	}

	log.Printf("Access granted - Referer: %s, IP: %s", referer, r.RemoteAddr)

	client := anthropic.NewClient(
		option.WithDefaultHeaders(map[string]string{
			"anthropic-beta": "interleaved-thinking-2025-05-14",
		}),
	)

	userQuery := r.URL.Query().Get("query")
	if userQuery == "" {
		userQuery = "What administrative tasks can I help you with today?"
	}

	systemPrompt := `You are an AI assistant for the internal admin panel of admin.internal.com. 
You help administrators with system management, user administration, and internal operations. 
Provide helpful, concise responses for administrative tasks.`

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_7Sonnet20250219,
		MaxTokens: 1024,
		System: []anthropic.TextBlockParam{
			{Text: systemPrompt},
		},
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: userQuery,
						},
					},
				},
			},
		},
	})

	if err != nil {
		http.Error(w, "Internal Server Error", http.StatusInternalServerError)
		log.Printf("Error calling Claude API: %v", err)
		return
	}

	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	fmt.Fprintf(w, `<!DOCTYPE html>
<html>
<head>
    <title>Admin Panel - Internal</title>
    <style>
        body { font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }
        .header { background: #2c3e50; color: white; padding: 20px; border-radius: 5px; }
        .content { background: #ecf0f1; padding: 20px; margin-top: 20px; border-radius: 5px; }
        .response { background: white; padding: 15px; border-left: 4px solid #3498db; margin-top: 15px; }
    </style>
</head>
<body>
    <div class="header">
        <h1>🔐 Admin Internal Panel</h1>
        <p>Access granted from: %s</p>
    </div>
    <div class="content">
        <h2>AI Assistant Response</h2>
        <p><strong>Query:</strong> %s</p>
        <div class="response">`, referer, userQuery)

	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Fprintf(w, "<p>%s</p>", strings.ReplaceAll(block.Text, "\n", "<br>"))
		}
	}

	fmt.Fprintf(w, `        </div>
    </div>
</body>
</html>`)
}

func main() {
	http.HandleFunc("/admin", adminPanelHandler)
	http.HandleFunc("/admin/", adminPanelHandler)

	http.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		fmt.Fprintf(w, "OK")
	})

	port := ":8080"
	log.Printf("Admin panel server starting on port %s", port)
	log.Printf("Access requires Referer header from: %s", trustedReferer)

	if err := http.ListenAndServe(port, nil); err != nil {
		log.Fatalf("Server failed to start: %v", err)
	}
}
