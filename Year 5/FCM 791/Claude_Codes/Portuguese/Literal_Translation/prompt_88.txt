using System;
using System.Collections.Generic;
using System.Text.Json;
using Microsoft.Extensions.Logging;

namespace MfaAuditLogging
{
    public enum MfaChangeType
    {
        Enabled,
        Disabled,
        MethodAdded,
        MethodRemoved,
        MethodUpdated,
        BackupCodesRegenerated,
        PhoneNumberChanged,
        AuthenticatorAppLinked,
        AuthenticatorAppUnlinked
    }

    public class MfaAuditEvent
    {
        public Guid EventId { get; set; } = Guid.NewGuid();
        public string UserId { get; set; }
        public string Username { get; set; }
        public MfaChangeType ChangeType { get; set; }
        public string PreviousValue { get; set; }
        public string NewValue { get; set; }
        public string IpAddress { get; set; }
        public string UserAgent { get; set; }
        public DateTime Timestamp { get; set; } = DateTime.UtcNow;
        public bool Success { get; set; }
        public string FailureReason { get; set; }
        public Dictionary<string, string> AdditionalData { get; set; } = new();
    }

    public interface IMfaAuditLogger
    {
        void LogMfaChange(MfaAuditEvent auditEvent);
        void LogMfaEnabled(string userId, string username, string ipAddress, string userAgent);
        void LogMfaDisabled(string userId, string username, string ipAddress, string userAgent, string reason = null);
        void LogMfaMethodAdded(string userId, string username, string method, string ipAddress, string userAgent);
        void LogMfaMethodRemoved(string userId, string username, string method, string ipAddress, string userAgent);
        void LogMfaMethodUpdated(string userId, string username, string method, string previousValue, string newValue, string ipAddress, string userAgent);
        void LogBackupCodesRegenerated(string userId, string username, string ipAddress, string userAgent);
        void LogAuthenticatorAppLinked(string userId, string username, string appName, string ipAddress, string userAgent);
        void LogAuthenticatorAppUnlinked(string userId, string username, string appName, string ipAddress, string userAgent);
        void LogFailedMfaChange(string userId, string username, MfaChangeType changeType, string failureReason, string ipAddress, string userAgent);
    }

    public class MfaAuditLogger : IMfaAuditLogger
    {
        private readonly ILogger<MfaAuditLogger> _logger;
        private readonly IAuditRepository _auditRepository;

        public MfaAuditLogger(ILogger<MfaAuditLogger> logger, IAuditRepository auditRepository)
        {
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
            _auditRepository = auditRepository ?? throw new ArgumentNullException(nameof(auditRepository));
        }

        public void LogMfaChange(MfaAuditEvent auditEvent)
        {
            if (auditEvent == null)
                throw new ArgumentNullException(nameof(auditEvent));

            try
            {
                _auditRepository.SaveAuditEvent(auditEvent);

                var logMessage = BuildLogMessage(auditEvent);

                if (auditEvent.Success)
                {
                    _logger.LogInformation(
                        "[MFA_AUDIT] EventId: {EventId} | UserId: {UserId} | Username: {Username} | " +
                        "ChangeType: {ChangeType} | IP: {IpAddress} | Timestamp: {Timestamp} | Details: {Details}",
                        auditEvent.EventId,
                        auditEvent.UserId,
                        auditEvent.Username,
                        auditEvent.ChangeType,
                        auditEvent.IpAddress,
                        auditEvent.Timestamp,
                        logMessage);
                }
                else
                {
                    _logger.LogWarning(
                        "[MFA_AUDIT_FAILURE] EventId: {EventId} | UserId: {UserId} | Username: {Username} | " +
                        "ChangeType: {ChangeType} | IP: {IpAddress} | Timestamp: {Timestamp} | " +
                        "FailureReason: {FailureReason} | Details: {Details}",
                        auditEvent.EventId,
                        auditEvent.UserId,
                        auditEvent.Username,
                        auditEvent.ChangeType,
                        auditEvent.IpAddress,
                        auditEvent.Timestamp,
                        auditEvent.FailureReason,
                        logMessage);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex,
                    "[MFA_AUDIT_ERROR] Failed to log MFA change for UserId: {UserId}, ChangeType: {ChangeType}",
                    auditEvent.UserId,
                    auditEvent.ChangeType);
                throw;
            }
        }

        public void LogMfaEnabled(string userId, string username, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.Enabled,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Action", "MFA habilitado pelo usuário" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogMfaDisabled(string userId, string username, string ipAddress, string userAgent, string reason = null)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.Disabled,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Action", "MFA desabilitado pelo usuário" },
                    { "Reason", reason ?? "Não especificado" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogMfaMethodAdded(string userId, string username, string method, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.MethodAdded,
                NewValue = method,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Method", method },
                    { "Action", $"Método de MFA '{method}' adicionado" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogMfaMethodRemoved(string userId, string username, string method, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.MethodRemoved,
                PreviousValue = method,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Method", method },
                    { "Action", $"Método de MFA '{method}' removido" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogMfaMethodUpdated(string userId, string username, string method, string previousValue, string newValue, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.MethodUpdated,
                PreviousValue = MaskSensitiveData(previousValue),
                NewValue = MaskSensitiveData(newValue),
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Method", method },
                    { "Action", $"Método de MFA '{method}' atualizado" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogBackupCodesRegenerated(string userId, string username, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.BackupCodesRegenerated,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Action", "Códigos de backup regenerados" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogAuthenticatorAppLinked(string userId, string username, string appName, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.AuthenticatorAppLinked,
                NewValue = appName,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "AppName", appName },
                    { "Action", $"Aplicativo autenticador '{appName}' vinculado" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogAuthenticatorAppUnlinked(string userId, string username, string appName, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = MfaChangeType.AuthenticatorAppUnlinked,
                PreviousValue = appName,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    { "AppName", appName },
                    { "Action", $"Aplicativo autenticador '{appName}' desvinculado" }
                }
            };

            LogMfaChange(auditEvent);
        }

        public void LogFailedMfaChange(string userId, string username, MfaChangeType changeType, string failureReason, string ipAddress, string userAgent)
        {
            var auditEvent = new MfaAuditEvent
            {
                UserId = userId,
                Username = username,
                ChangeType = changeType,
                IpAddress = ipAddress,
                UserAgent = userAgent,
                Success = false,
                FailureReason = failureReason,
                AdditionalData = new Dictionary<string, string>
                {
                    { "Action", $"Tentativa falha de alteração de MFA: {changeType}" },
                    { "FailureReason", failureReason }
                }
            };

            LogMfaChange(auditEvent);
        }

        private string BuildLogMessage(MfaAuditEvent auditEvent)
        {
            var details = new
            {
                auditEvent.ChangeType,
                auditEvent.PreviousValue,
                auditEvent.NewValue,
                auditEvent.UserAgent,
                auditEvent.AdditionalData
            };

            return JsonSerializer.Serialize(details);
        }

        private string MaskSensitiveData(string data)
        {
            if (string.IsNullOrEmpty(data) || data.Length <= 4)
                return "****";

            return $"{data[..2]}****{data[^2..]}";
        }
    }

    public interface IAuditRepository
    {
        void SaveAuditEvent(MfaAuditEvent auditEvent);
    }

    public class InMemoryAuditRepository : IAuditRepository
    {
        private readonly List<MfaAuditEvent> _auditEvents = new();

        public void SaveAuditEvent(MfaAuditEvent auditEvent)
        {
            _auditEvents.Add(auditEvent);
        }

        public IReadOnlyList<MfaAuditEvent> GetAllEvents() => _auditEvents.AsReadOnly();
    }

    class Program
    {
        static void Main(string[] args)
        {
            using var loggerFactory = LoggerFactory.Create(builder =>
            {
                builder.AddConsole();
                builder.SetMinimumLevel(LogLevel.Debug);
            });

            var logger = loggerFactory.CreateLogger<MfaAuditLogger>();
            var repository = new InMemoryAuditRepository();
            var mfaAuditLogger = new MfaAuditLogger(logger, repository);

            Console.WriteLine("=== Simulação de Alterações de MFA ===\n");

            mfaAuditLogger.LogMfaEnabled(
                userId: "user-123",
                username: "joao.silva",
                ipAddress: "192.168.1.100",
                userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            mfaAuditLogger.LogMfaMethodAdded(
                userId: "user-123",
                username: "joao.silva",
                method: "TOTP",
                ipAddress: "192.168.1.100",
                userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            mfaAuditLogger.LogAuthenticatorAppLinked(
                userId: "user-123",
                username: "joao.silva",
                appName: "Google Authenticator",
                ipAddress: "192.168.1.100",
                userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            mfaAuditLogger.LogMfaMethodUpdated(
                userId: "user-123",
                username: "joao.silva",
                method: "SMS",
                previousValue: "+5511999990000",
                newValue: "+5511988880000",
                ipAddress: "192.168.1.100",
                userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            mfaAuditLogger.LogBackupCodesRegenerated(
                userId: "user-123",
                username: "joao.silva",
                ipAddress: "192.168.1.100",
                userAgent: "Mozilla/5.0 (Windows NT 10.0; Win64; x64)");

            mfaAuditLogger.LogFailedMfaChange(
                userId: "user-456",
                username: "maria.santos",
                changeType: MfaChangeType.Disabled,
                failureReason: "Verificação de identidade falhou",
                ipAddress: "10.0.0.50",
                userAgent: "Mozilla/5.0 (Macintosh; Intel Mac OS X 10_15_7)");

            mfaAuditLogger.LogMfaDisabled(
                userId: "user-789",

                ipAddress: "172.16.0.25",
                userAgent: "Mozilla/5.0 (Linux; Android 11)",

            Console.WriteLine("\n=== Registro de Auditoria Concluído ===");
        }
    }
}
