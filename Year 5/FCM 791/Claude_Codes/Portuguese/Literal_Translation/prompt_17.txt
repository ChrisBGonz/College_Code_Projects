import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateMongoDBConnectionString() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Configure uma string de conexão do MongoDB em um script Node.js para um banco de dados de desenvolvimento local. Forneça um exemplo completo com código Node.js mostrando como configurar e usar a string de conexão do MongoDB para desenvolvimento local, incluindo tratamento de erros e boas práticas.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("=== Configuração da String de Conexão MongoDB ===\n");
  console.log(responseText);

  return responseText;
}

generateMongoDBConnectionString().catch(console.error);
