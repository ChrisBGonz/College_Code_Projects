import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiChatModelName;

public class PasswordValidator {

    private final ChatLanguageModel chatModel;

    public PasswordValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName(OpenAiChatModelName.GPT_4_O_MINI)
                .build();
    }

    public PasswordValidationResult validatePassword(String password) {
        String prompt = String.format("""
                Analyze the following password and determine if it is "strong enough" for a registration page.
                
                Password: "%s"
                
                Evaluate the password based on these criteria:
                1. Length (minimum 8 characters, ideally 12+)
                2. Contains uppercase letters
                3. Contains lowercase letters
                4. Contains numbers
                5. Contains special characters (!@#$%%^&*()_+-=[]{}|;':",.<>?/)
                6. Not a common password or dictionary word
                7. Not sequential characters (like "12345" or "abcde")
                
                Respond in the following format:
                STRONG: [true/false]
                SCORE: [1-10]
                FEEDBACK: [Brief explanation of why the password is or isn't strong enough, and suggestions for improvement if needed]
                """, password);

        String response = chatModel.generate(prompt);
        return parseResponse(response, password);
    }

    private PasswordValidationResult parseResponse(String response, String password) {
        boolean isStrong = false;
        int score = 0;
        String feedback = "";

        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith("STRONG:")) {
                isStrong = line.contains("true");
            } else if (line.startsWith("SCORE:")) {
                try {
                    String scoreStr = line.replace("SCORE:", "").trim();
                    score = Integer.parseInt(scoreStr);
                } catch (NumberFormatException e) {
                    score = 0;
                }
            } else if (line.startsWith("FEEDBACK:")) {
                feedback = line.replace("FEEDBACK:", "").trim();
            }
        }

        return new PasswordValidationResult(password, isStrong, score, feedback);
    }

    public record PasswordValidationResult(
            String password,
            boolean isStrong,
            int score,
            String feedback
    ) {
        @Override
        public String toString() {
            return String.format("""
                    Password Validation Result:
                    Password: %s
                    Is Strong: %s
                    Score: %d/10
                    Feedback: %s
                    """, 
                    maskPassword(password), 
                    isStrong ? "Yes ✓" : "No ✗", 
                    score, 
                    feedback);
        }

        private String maskPassword(String pwd) {
            if (pwd == null || pwd.isEmpty()) return "";
            return pwd.charAt(0) + "*".repeat(Math.max(0, pwd.length() - 2)) + 
                   (pwd.length() > 1 ? String.valueOf(pwd.charAt(pwd.length() - 1)) : "");
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();

        String[] testPasswords = {
            "password",
            "Password1",
            "P@ssw0rd!",
            "MyS3cur3P@ssw0rd!2024",
            "12345678",
            "abc",
            "Tr0ub4dor&3"
        };

        System.out.println("=== Password Strength Validator ===\n");

        for (String password : testPasswords) {
            PasswordValidationResult result = validator.validatePassword(password);
            System.out.println(result);
            System.out.println("-".repeat(50));
        }
    }
}
