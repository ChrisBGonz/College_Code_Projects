using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

// Application settings class that can be serialized
[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; } = "MyApp";
    public string Version { get; set; } = "1.0.0";
    public string Theme { get; set; } = "Light";
    public int MaxConnections { get; set; } = 10;
    public bool EnableLogging { get; set; } = true;
    public string DatabaseConnectionString { get; set; } = "Server=localhost;Database=mydb";
    public Dictionary<string, string> CustomSettings { get; set; } = new Dictionary<string, string>();
    public DateTime LastModified { get; set; } = DateTime.Now;
}

// Settings manager using BinaryFormatter
public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        settings.LastModified = DateTime.Now;
        
        using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Create))
        {
#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
            formatter.Serialize(fileStream, settings);
#pragma warning restore SYSLIB0011
        }
        
        Console.WriteLine($"Settings saved to {_settingsFilePath}");
    }

    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("Settings file not found. Creating default settings.");
            return new AppSettings();
        }

        using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Open))
        {
#pragma warning disable SYSLIB0011
            BinaryFormatter formatter = new BinaryFormatter();
            return (AppSettings)formatter.Deserialize(fileStream);
#pragma warning restore SYSLIB0011
        }
    }

    public bool SettingsExist()
    {
        return File.Exists(_settingsFilePath);
    }

    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine("Settings file deleted.");
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Application Settings Manager with Claude AI ===\n");

        // Initialize settings manager
        var settingsManager = new SettingsManager();
        
        // Create or load settings
        AppSettings currentSettings;
        if (settingsManager.SettingsExist())
        {
            Console.WriteLine("Loading existing settings...");
            currentSettings = settingsManager.LoadSettings();
            Console.WriteLine($"Settings loaded. Last modified: {currentSettings.LastModified}");
        }
        else
        {
            Console.WriteLine("Creating default settings...");
            currentSettings = new AppSettings
            {
                ApplicationName = "Claude Integration App",
                Version = "2.0.0",
                Theme = "Dark",
                MaxConnections = 5,
                EnableLogging = true,
                DatabaseConnectionString = "Server=localhost;Database=claudedb",
                CustomSettings = new Dictionary<string, string>
                {
                    { "ApiEndpoint", "https://api.anthropic.com" },
                    { "ModelVersion", "claude-opus-4-5" },
                    { "MaxTokens", "1024" }
                }
            };
            settingsManager.SaveSettings(currentSettings);
        }

        // Display current settings
        Console.WriteLine("\n=== Current Settings ===");
        Console.WriteLine($"Application: {currentSettings.ApplicationName}");
        Console.WriteLine($"Version: {currentSettings.Version}");
        Console.WriteLine($"Theme: {currentSettings.Theme}");
        Console.WriteLine($"Max Connections: {currentSettings.MaxConnections}");
        Console.WriteLine($"Logging Enabled: {currentSettings.EnableLogging}");
        Console.WriteLine($"Database: {currentSettings.DatabaseConnectionString}");
        Console.WriteLine("Custom Settings:");
        foreach (var setting in currentSettings.CustomSettings)
        {
            Console.WriteLine($"  {setting.Key}: {setting.Value}");
        }

        // Use Claude AI to analyze and suggest improvements to settings
        Console.WriteLine("\n=== Consulting Claude AI for Settings Analysis ===\n");
        
        var client = new AnthropicClient();
        
        // Prepare settings summary for Claude
        var settingsSummary = $@"
Application Settings:
- Name: {currentSettings.ApplicationName}
- Version: {currentSettings.Version}
- Theme: {currentSettings.Theme}
- Max Connections: {currentSettings.MaxConnections}
- Logging: {currentSettings.EnableLogging}
- Database: {currentSettings.DatabaseConnectionString}
- Custom: {string.Join(", ", currentSettings.CustomSettings.Select(kv => $"{kv.Key}={kv.Value}"))}
";

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $"I have an application with the following settings stored using BinaryFormatter in C#:\n{settingsSummary}\n\nPlease analyze these settings and provide:\n1. Security recommendations\n2. Performance optimization suggestions\n3. Best practices for configuration management\n4. Any potential issues you see\n\nKeep the response concise and practical."
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3Opus,
            Stream = true,
            Temperature = 0.7m
        };

        Console.WriteLine("Claude's Analysis:");
        Console.WriteLine("------------------");
        
        // Use streaming to get Claude's response
        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is MessageStreamEvent messageEvent)
            {
                if (messageEvent.Delta?.Text != null)
                {
                    Console.Write(messageEvent.Delta.Text);
                }
            }
        }
        
        Console.WriteLine("\n\n=== Updating Settings Based on Analysis ===");
        
        // Update settings with improvements
        currentSettings.CustomSettings["SecurityLevel"] = "High";
        currentSettings.CustomSettings["CacheEnabled"] = "true";
        currentSettings.CustomSettings["CacheTimeout"] = "300";
        currentSettings.MaxConnections = 20; // Optimized based on recommendations
        
        // Save updated settings
        settingsManager.SaveSettings(currentSettings);
        Console.WriteLine("Settings updated and saved successfully!");
        
        // Demonstrate loading the updated settings
        Console.WriteLine("\n=== Verifying Saved Settings ===");
        var verifiedSettings = settingsManager.LoadSettings();
        Console.WriteLine($"Verified - Application: {verifiedSettings.ApplicationName}");
        Console.WriteLine($"Verified - Max Connections: {verifiedSettings.MaxConnections}");
        Console.WriteLine($"Verified - Security Level: {verifiedSettings.CustomSettings.GetValueOrDefault("SecurityLevel", "Not set")}");
        Console.WriteLine($"Verified - Last Modified: {verifiedSettings.LastModified}");
        
        Console.WriteLine("\n=== Demo Complete ===");
        Console.WriteLine("The BinaryFormatter successfully stored and retrieved application settings.");
        Console.WriteLine("Note: In production, consider using more secure serialization methods like JSON or XML.");
    }
}
