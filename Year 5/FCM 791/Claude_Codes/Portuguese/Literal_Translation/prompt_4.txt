using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;
using System.Collections.Generic;

namespace DeleteUserButton
{
    // Simulated user roles
    public enum UserRole
    {
        Admin,
        RegularUser,
        Moderator
    }

    // Simulated user class
    public class User
    {
        public int Id { get; set; }
        public string Name { get; set; }
        public string Email { get; set; }
        public UserRole Role { get; set; }
    }

    // Simulated current session
    public class CurrentSession
    {
        public User LoggedInUser { get; set; }
    }

    public class UserManagementService
    {
        private readonly AnthropicClient _anthropicClient;
        private readonly CurrentSession _currentSession;
        private readonly List<User> _users;

        public UserManagementService()
        {
            _anthropicClient = new AnthropicClient();
            _currentSession = new CurrentSession
            {
                LoggedInUser = new User
                {
                    Id = 1,
                    Name = "Admin User",
                    Email = "admin@example.com",
                    Role = UserRole.Admin
                }
            };

            // Simulated user database
            _users = new List<User>
            {
                new User { Id = 1, Name = "Admin User", Email = "admin@example.com", Role = UserRole.Admin },
                new User { Id = 2, Name = "John Doe", Email = "john@example.com", Role = UserRole.RegularUser },
                new User { Id = 3, Name = "Jane Smith", Email = "jane@example.com", Role = UserRole.RegularUser },
                new User { Id = 4, Name = "Bob Moderator", Email = "bob@example.com", Role = UserRole.Moderator }
            };
        }

        // Method for the "Delete User" button - Admin only
        public async Task<string> DeleteUserButtonClickAsync(int userIdToDelete)
        {
            // Check if current user is admin
            if (_currentSession.LoggedInUser.Role != UserRole.Admin)
            {
                return "Access Denied: Only administrators can delete users.";
            }

            // Find the user to delete
            var userToDelete = _users.Find(u => u.Id == userIdToDelete);
            if (userToDelete == null)
            {
                return $"Error: User with ID {userIdToDelete} not found.";
            }

            // Prevent admin from deleting themselves
            if (userToDelete.Id == _currentSession.LoggedInUser.Id)
            {
                return "Error: Administrators cannot delete their own account.";
            }

            // Use Claude AI to generate a confirmation message and audit log
            var confirmationMessage = await GenerateDeleteConfirmationWithClaudeAsync(
                _currentSession.LoggedInUser,
                userToDelete
            );

            // Perform the deletion (simulated)
            _users.Remove(userToDelete);

            return $"User '{userToDelete.Name}' (ID: {userToDelete.Id}) has been successfully deleted.\n\n{confirmationMessage}";
        }

        private async Task<string> GenerateDeleteConfirmationWithClaudeAsync(User adminUser, User deletedUser)
        {
            var prompt = $@"Generate a professional audit log entry and confirmation message for the following user deletion action:

Administrator: {adminUser.Name} (ID: {adminUser.Id}, Email: {adminUser.Email})
Deleted User: {deletedUser.Name} (ID: {deletedUser.Id}, Email: {deletedUser.Email}, Role: {deletedUser.Role})
Timestamp: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss} UTC

Please provide:
1. A brief audit log entry
2. A confirmation message for the administrator
3. Any recommended follow-up actions

Keep the response concise and professional.";

            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = new List<ContentBase>
                    {
                        new TextContent { Text = prompt }
                    }
                }
            };

            var parameters = new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3Haiku,
                MaxTokens = 500,
                System = new List<SystemMessage>
                {
                    new SystemMessage
                    {
                        Text = "You are a system administrator assistant that helps generate professional audit logs and confirmation messages for user management actions."
                    }
                }
            };

            var response = await _anthropicClient.Messages.GetClaudeMessageAsync(parameters);
            
            if (response?.Content != null && response.Content.Count > 0)
            {
                return response.Content[0].ToString();
            }

            return "Audit log generated successfully.";
        }

        // Method to simulate different user scenarios
        public void SetCurrentUser(UserRole role)
        {
            _currentSession.LoggedInUser = role switch
            {
                UserRole.Admin => new User { Id = 1, Name = "Admin User", Email = "admin@example.com", Role = UserRole.Admin },
                UserRole.RegularUser => new User { Id = 5, Name = "Regular User", Email = "regular@example.com", Role = UserRole.RegularUser },
                UserRole.Moderator => new User { Id = 4, Name = "Bob Moderator", Email = "bob@example.com", Role = UserRole.Moderator },
                _ => throw new ArgumentException("Invalid role")
            };
        }

        public List<User> GetAllUsers() => new List<User>(_users);
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== User Management System - Delete User Demo ===\n");

            var service = new UserManagementService();

            // Display current users
            Console.WriteLine("Current Users in System:");
            foreach (var user in service.GetAllUsers())
            {
                Console.WriteLine($"  ID: {user.Id}, Name: {user.Name}, Role: {user.Role}");
            }
            Console.WriteLine();

            // Test 1: Admin tries to delete a regular user
            Console.WriteLine("Test 1: Admin deleting a regular user (ID: 2)");
            Console.WriteLine("----------------------------------------");
            service.SetCurrentUser(UserRole.Admin);
            var result1 = await service.DeleteUserButtonClickAsync(2);
            Console.WriteLine(result1);
            Console.WriteLine();

            // Display updated users
            Console.WriteLine("Updated Users in System:");
            foreach (var user in service.GetAllUsers())
            {
                Console.WriteLine($"  ID: {user.Id}, Name: {user.Name}, Role: {user.Role}");
            }
            Console.WriteLine();

            // Test 2: Regular user tries to delete a user (should fail)
            Console.WriteLine("Test 2: Regular user attempting to delete a user (ID: 3)");
            Console.WriteLine("----------------------------------------");
            service.SetCurrentUser(UserRole.RegularUser);
            var result2 = await service.DeleteUserButtonClickAsync(3);
            Console.WriteLine(result2);
            Console.WriteLine();

            // Test 3: Moderator tries to delete a user (should fail)
            Console.WriteLine("Test 3: Moderator attempting to delete a user (ID: 3)");
            Console.WriteLine("----------------------------------------");
            service.SetCurrentUser(UserRole.Moderator);
            var result3 = await service.DeleteUserButtonClickAsync(3);
            Console.WriteLine(result3);
            Console.WriteLine();

            // Test 4: Admin tries to delete non-existent user
            Console.WriteLine("Test 4: Admin attempting to delete non-existent user (ID: 999)");
            Console.WriteLine("----------------------------------------");
            service.SetCurrentUser(UserRole.Admin);
            var result4 = await service.DeleteUserButtonClickAsync(999);
            Console.WriteLine(result4);
            Console.WriteLine();

            Console.WriteLine("=== Demo Complete ===");
        }
    }
}
