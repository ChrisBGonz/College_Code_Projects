import os
import anthropic
import requests
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def fetch_url_content(url: str) -> tuple[str | None, str | None]:
    """Fetch content from a URL and return (content, error)."""
    try:
        if not url.startswith(("http://", "https://")):
            url = "https://" + url

        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        }
        response = requests.get(url, headers=headers, timeout=10)

        if response.status_code == 404:
            return None, f"404 Error: The page '{url}' was not found."
        elif response.status_code != 200:
            return (
                None,
                f"Error: Received status code {response.status_code} for '{url}'.",
            )

        return response.text[:5000], None

    except requests.exceptions.ConnectionError:
        return None, f"Connection Error: Could not connect to '{url}'."
    except requests.exceptions.Timeout:
        return None, f"Timeout Error: The request to '{url}' timed out."
    except requests.exceptions.RequestException as e:
        return None, f"Request Error: {str(e)}"


@app.route("/fetch", methods=["GET"])
def fetch_and_present():
    """Route that fetches URL content and presents it using Claude with streaming."""
    url = request.args.get("url")

    if not url:
        return Response(
            "Error: Please provide a URL parameter. Example: /fetch?url=https://example.com",
            status=400,
            mimetype="text/plain",
        )

    content, error = fetch_url_content(url)

    if error:
        status_code = 404 if "404" in error else 400
        return Response(error, status=status_code, mimetype="text/plain")

    def generate():
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"""I fetched the content from the URL: {url}

Here is the content (first 5000 characters):

{content}

Please analyze and present this content in a clear, organized way. Include:
1. A brief summary of what the page is about
2. Key information or main points found on the page
3. Any notable features or sections

Present the information in a readable format.""",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(
        stream_with_context(generate()),
        mimetype="text/plain",
        headers={"X-Content-Type-Options": "nosniff"},
    )


@app.route("/", methods=["GET"])
def index():
    """Simple index page with usage instructions."""
    instructions = """URL Content Fetcher API

Usage:
  GET /fetch?url=<URL>

Examples:
  /fetch?url=https://example.com
  /fetch?url=https://python.org
  /fetch?url=example.com

The API will:
1. Fetch the content from the provided URL
2. Use Claude AI to analyze and present the content
3. Stream the response back to you

Error Handling:
- 400: Missing URL parameter or invalid request
- 404: Page not found at the specified URL
"""
    return Response(instructions, mimetype="text/plain")


if __name__ == "__main__":
    app.run(debug=True, port=5000)
