package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
)

// AppConfig represents the application configuration
type AppConfig struct {
	AppName     string                 `json:"app_name"`
	Version     string                 `json:"version"`
	Debug       bool                   `json:"debug"`
	MaxRetries  int                    `json:"max_retries"`
	Timeout     float64                `json:"timeout"`
	Features    []string               `json:"features"`
	Database    DatabaseConfig         `json:"database"`
	Server      ServerConfig           `json:"server"`
	CustomProps map[string]interface{} `json:"custom_props"`
}

// DatabaseConfig represents database configuration
type DatabaseConfig struct {
	Host     string `json:"host"`
	Port     int    `json:"port"`
	Name     string `json:"name"`
	User     string `json:"user"`
	Password string `json:"password"`
	MaxConns int    `json:"max_connections"`
}

// ServerConfig represents server configuration
type ServerConfig struct {
	Host    string `json:"host"`
	Port    int    `json:"port"`
	TLS     bool   `json:"tls"`
	Workers int    `json:"workers"`
}

// Application represents the main application with dynamic properties
type Application struct {
	Config     *AppConfig
	Properties map[string]interface{}
}

// NewApplication creates a new application instance
func NewApplication() *Application {
	return &Application{
		Properties: make(map[string]interface{}),
	}
}

// LoadConfigFromFile reads a JSON configuration file and applies it to the application
func (app *Application) LoadConfigFromFile(filename string) error {
	// Read the file
	data, err := os.ReadFile(filename)
	if err != nil {
		return fmt.Errorf("error reading config file: %w", err)
	}

	// Parse the JSON
	var config AppConfig
	if err := json.Unmarshal(data, &config); err != nil {
		return fmt.Errorf("error parsing config JSON: %w", err)
	}

	// Apply the configuration
	app.Config = &config
	app.applyDynamicProperties()

	return nil
}

// LoadConfigFromJSON reads a JSON string and applies it to the application
func (app *Application) LoadConfigFromJSON(jsonData string) error {
	var config AppConfig
	if err := json.Unmarshal([]byte(jsonData), &config); err != nil {
		return fmt.Errorf("error parsing config JSON: %w", err)
	}

	app.Config = &config
	app.applyDynamicProperties()

	return nil
}

// applyDynamicProperties applies configuration values as dynamic properties
func (app *Application) applyDynamicProperties() {
	if app.Config == nil {
		return
	}

	// Apply basic properties
	app.Properties["app_name"] = app.Config.AppName
	app.Properties["version"] = app.Config.Version
	app.Properties["debug"] = app.Config.Debug
	app.Properties["max_retries"] = app.Config.MaxRetries
	app.Properties["timeout"] = app.Config.Timeout

	// Apply features as a comma-separated string
	if len(app.Config.Features) > 0 {
		app.Properties["features"] = strings.Join(app.Config.Features, ", ")
		app.Properties["feature_count"] = len(app.Config.Features)
	}

	// Apply database properties with prefix
	app.Properties["db_host"] = app.Config.Database.Host
	app.Properties["db_port"] = app.Config.Database.Port
	app.Properties["db_name"] = app.Config.Database.Name
	app.Properties["db_max_connections"] = app.Config.Database.MaxConns

	// Apply server properties with prefix
	app.Properties["server_host"] = app.Config.Server.Host
	app.Properties["server_port"] = app.Config.Server.Port
	app.Properties["server_tls"] = app.Config.Server.TLS
	app.Properties["server_workers"] = app.Config.Server.Workers

	// Apply custom properties
	for key, value := range app.Config.CustomProps {
		app.Properties["custom_"+key] = value
	}
}

// GetProperty retrieves a dynamic property value
func (app *Application) GetProperty(key string) (interface{}, bool) {
	value, exists := app.Properties[key]
	return value, exists
}

// SetProperty sets a dynamic property value
func (app *Application) SetProperty(key string, value interface{}) {
	app.Properties[key] = value
}

// GetAllProperties returns all dynamic properties
func (app *Application) GetAllProperties() map[string]interface{} {
	return app.Properties
}

// createSampleConfig creates a sample configuration file for demonstration
func createSampleConfig(filename string) error {
	config := AppConfig{
		AppName:    "MyGoApp",
		Version:    "1.0.0",
		Debug:      true,
		MaxRetries: 3,
		Timeout:    30.5,
		Features:   []string{"auth", "logging", "metrics", "caching"},
		Database: DatabaseConfig{
			Host:     "localhost",
			Port:     5432,
			Name:     "myapp_db",
			User:     "admin",
			Password: "secret",
			MaxConns: 10,
		},
		Server: ServerConfig{
			Host:    "0.0.0.0",
			Port:    8080,
			TLS:     false,
			Workers: 4,
		},
		CustomProps: map[string]interface{}{
			"theme":        "dark",
			"language":     "en",
			"max_file_size": 10485760,
		},
	}

	data, err := json.MarshalIndent(config, "", "  ")
	if err != nil {
		return fmt.Errorf("error creating config: %w", err)
	}

	return os.WriteFile(filename, data, 0644)
}

// analyzeConfigWithClaude uses Claude AI to analyze the configuration
func analyzeConfigWithClaude(app *Application) error {
	client := anthropic.NewClient()

	// Prepare the configuration summary for Claude
	configJSON, err := json.MarshalIndent(app.Config, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling config: %w", err)
	}

	propertiesJSON, err := json.MarshalIndent(app.Properties, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling properties: %w", err)
	}

	prompt := fmt.Sprintf(`I have a Go application with the following JSON configuration:

%s

The application has dynamically applied these properties:

%s

Please analyze this configuration and provide:
1. A brief summary of what this application appears to do
2. Any potential security concerns you notice
3. Recommendations for improving the configuration
4. Suggestions for additional properties that might be useful

Please be concise and practical in your response.`, string(configJSON), string(propertiesJSON))

	fmt.Println("\n=== Analyzing configuration with Claude AI ===\n")

	// Use streaming for the response
	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	message := anthropic.Message{}
	for stream.Next() {
		event := stream.Current()
		err := message.Accumulate(event)
		if err != nil {
			return fmt.Errorf("error accumulating stream: %w", err)
		}

		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEvent:
			switch deltaType := delta.Delta.(type) {
			case anthropic.TextDelta:
				fmt.Print(deltaType.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return fmt.Errorf("stream error: %w", err)
	}

	fmt.Println("\n")
	return nil
}

func main() {
	// Create a sample configuration file
	configFile := "app_config.json"
	fmt.Printf("Creating sample configuration file: %s\n", configFile)

	if err := createSampleConfig(configFile); err != nil {
		log.Fatalf("Error creating sample config: %v", err)
	}

	// Create a new application instance
	app := NewApplication()

	// Load configuration from file
	fmt.Printf("Loading configuration from: %s\n", configFile)
	if err := app.LoadConfigFromFile(configFile); err != nil {
		log.Fatalf("Error loading config: %v", err)
	}

	// Display loaded configuration
	fmt.Println("\n=== Application Configuration Loaded ===")
	fmt.Printf("App Name: %s\n", app.Config.AppName)
	fmt.Printf("Version: %s\n", app.Config.Version)
	fmt.Printf("Debug Mode: %v\n", app.Config.Debug)

	// Display dynamic properties
	fmt.Println("\n=== Dynamic Properties Applied ===")
	for key, value := range app.GetAllProperties() {
		fmt.Printf("  %s: %v\n", key, value)
	}

	// Demonstrate getting specific properties
	fmt.Println("\n=== Accessing Specific Properties ===")
	if dbHost, exists := app.GetProperty("db_host"); exists {
		fmt.Printf("Database Host: %v\n", dbHost)
	}
	if serverPort, exists := app.GetProperty("server_port"); exists {
		fmt.Printf("Server Port: %v\n", serverPort)
	}
	if features, exists := app.GetProperty("features"); exists {
		fmt.Printf("Features: %v\n", features)
	}

	// Demonstrate setting a new dynamic property
	app.SetProperty("runtime_id", "app-12345")
	app.SetProperty("start_time", "2024-01-01T00:00:00Z")
	fmt.Println("\n=== Added Runtime Properties ===")
	if runtimeID, exists := app.GetProperty("runtime_id"); exists {
		fmt.Printf("Runtime ID: %v\n", runtimeID)
	}

	// Demonstrate loading from JSON string
	fmt.Println("\n=== Loading Additional Config from JSON String ===")
	additionalConfig := `{
		"app_name": "MyGoApp-Updated",
		"version": "1.1.0",
		"debug": false,
		"max_retries": 5,
		"timeout": 60.0,
		"features": ["auth", "logging", "metrics", "caching", "rate-limiting"],
		"database": {
			"host": "db.production.com",
			"port": 5432,
			"name": "myapp_prod",
			"user": "prod_user",
			"password": "prod_secret",
			"max_connections": 50
		},
		"server": {
			"host": "0.0.0.0",
			"port": 443,
			"tls": true,
			"workers": 8
		},
		"custom_props": {
			"theme": "light",
			"language": "pt-BR",
			"max_file_size": 52428800
		}
	}`

	app2 := NewApplication()
	if err := app2.LoadConfigFromJSON(additionalConfig); err != nil {
		log.Fatalf("Error loading config from JSON: %v", err)
	}

	fmt.Printf("Updated App Name: %s\n", app2.Config.AppName)
	fmt.Printf("Updated Version: %s\n", app2.Config.Version)
	fmt.Printf("TLS Enabled: %v\n", app2.Config.Server.TLS)

	// Analyze configuration with Claude AI
	if err := analyzeConfigWithClaude(app2); err != nil {
		log.Printf("Warning: Could not analyze with Claude: %v", err)
	}

	// Clean up the sample config file
	if err := os.Remove(configFile); err != nil {
		log.Printf("Warning: Could not remove sample config file: %v", err)
	}

	fmt.Println("=== Configuration management demonstration complete ===")
}
