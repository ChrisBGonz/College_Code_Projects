require 'openssl'
require 'base64'
require 'json'
require 'anthropic'

def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  [private_key, public_key]
end

def sign_payload_with_rsa_pss(payload, private_key, hash_algorithm = 'SHA256', salt_length = 32)
  payload_string = payload.is_a?(String) ? payload : payload.to_json
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  signature = private_key.sign_pss(
    digest,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
  
  signature_base64 = Base64.strict_encode64(signature)
  
  {
    payload: payload_string,
    signature: signature_base64,
    algorithm: "RSA-PSS-#{hash_algorithm}",
    salt_length: salt_length
  }
end

def verify_rsa_pss_signature(signed_data, public_key)
  payload_string = signed_data[:payload]
  signature = Base64.strict_decode64(signed_data[:signature])
  hash_algorithm = signed_data[:algorithm].split('-').last
  salt_length = signed_data[:salt_length]
  
  digest = OpenSSL::Digest.new(hash_algorithm)
  
  public_key.verify_pss(
    digest,
    signature,
    payload_string,
    salt_length: salt_length,
    mgf1_hash: hash_algorithm
  )
rescue OpenSSL::PKey::PKeyError => e
  puts "Verification error: #{e.message}"
  false
end

def analyze_signature_with_claude(signed_data, verification_result)
  client = Anthropic::Client.new
  
  analysis_prompt = <<~PROMPT
    Analyze the following RSA-PSS digital signature implementation and provide insights:
    
    Signed Data Information:
    - Algorithm: #{signed_data[:algorithm]}
    - Salt Length: #{signed_data[:salt_length]} bytes
    - Payload (first 100 chars): #{signed_data[:payload][0..100]}...
    - Signature (Base64, first 50 chars): #{signed_data[:signature][0..50]}...
    - Signature Length: #{Base64.strict_decode64(signed_data[:signature]).length} bytes
    
    Verification Result: #{verification_result ? 'VALID' : 'INVALID'}
    
    Please provide:
    1. Security assessment of the RSA-PSS parameters used
    2. Explanation of why RSA-PSS is preferred over PKCS#1 v1.5
    3. Recommendations for production use
    4. Any potential vulnerabilities or improvements
    
    Keep the response concise and technical.
  PROMPT
  
  puts "\n=== Claude AI Analysis ==="
  
  client.messages.stream(
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: analysis_prompt
      }
    ]
  ) do |event|
    if event.type == "content_block_delta" && event.delta.type == "text_delta"
      print event.delta.text
      $stdout.flush
    end
  end
  
  puts "\n"
end

def demonstrate_rsa_pss_signing
  puts "=== RSA-PSS Digital Signature Demonstration ==="
  puts "\n1. Generating RSA key pair (2048 bits)..."
  private_key, public_key = generate_rsa_key_pair(2048)
  puts "   ✓ Key pair generated successfully"
  puts "   Private key size: #{private_key.n.num_bits} bits"
  
  payload = {
    user_id: "user_123",
    action: "transfer",
    amount: 1000.00,
    currency: "USD",
    timestamp: Time.now.iso8601,
    nonce: SecureRandom.hex(16)
  }
  
  puts "\n2. Original payload:"
  puts "   #{payload.to_json}"
  
  puts "\n3. Signing payload with RSA-PSS (SHA-256)..."
  signed_data = sign_payload_with_rsa_pss(payload, private_key)
  puts "   ✓ Payload signed successfully"
  puts "   Algorithm: #{signed_data[:algorithm]}"
  puts "   Signature (Base64): #{signed_data[:signature][0..50]}..."
  
  puts "\n4. Verifying signature with public key..."
  is_valid = verify_rsa_pss_signature(signed_data, public_key)
  puts "   ✓ Signature verification: #{is_valid ? 'VALID ✓' : 'INVALID ✗'}"
  
  puts "\n5. Testing with tampered payload..."
  tampered_data = signed_data.dup
  tampered_payload = JSON.parse(signed_data[:payload])
  tampered_payload["amount"] = 9999.99
  tampered_data[:payload] = tampered_payload.to_json
  
  is_tampered_valid = verify_rsa_pss_signature(tampered_data, public_key)
  puts "   ✓ Tampered signature verification: #{is_tampered_valid ? 'VALID' : 'INVALID ✗ (Expected)'}"
  
  puts "\n6. Testing with wrong key..."
  _, wrong_public_key = generate_rsa_key_pair(2048)
  is_wrong_key_valid = verify_rsa_pss_signature(signed_data, wrong_public_key)
  puts "   ✓ Wrong key verification: #{is_wrong_key_valid ? 'VALID' : 'INVALID ✗ (Expected)'}"
  
  analyze_signature_with_claude(signed_data, is_valid)
  
  {
    private_key: private_key,
    public_key: public_key,
    signed_data: signed_data,
    verification_result: is_valid
  }
end

result = demonstrate_rsa_pss_signing

puts "\n=== Summary ==="
puts "RSA-PSS signing demonstration completed successfully!"
puts "The implementation includes:"
puts "  - Key pair generation (2048-bit RSA)"
puts "  - Payload signing with RSA-PSS (SHA-256, 32-byte salt)"
puts "  - Signature verification"
puts "  - Tamper detection"
puts "  - AI-powered security analysis via Claude"
