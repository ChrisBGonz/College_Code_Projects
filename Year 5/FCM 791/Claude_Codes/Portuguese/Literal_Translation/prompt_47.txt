import React, { useState, useEffect } from 'react';

const UserBiography = ({ userId }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUserBiography = async () => {
      try {
        setLoading(true);
        setError(null);

        const response = await fetch(`https://jsonplaceholder.typicode.com/users/${userId}`);

        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();

        const userWithBio = {
          ...data,
          biography: `
            <h2>About ${data.name}</h2>
            <p>Hello! My name is <strong>${data.name}</strong> and I go by <em>${data.username}</em> online.</p>
            <p>You can reach me at <a href="mailto:${data.email}">${data.email}</a> or visit my website at 
            <a href="http://${data.website}" target="_blank" rel="noopener noreferrer">${data.website}</a>.</p>
            <h3>Location</h3>
            <p>I live in <strong>${data.address.city}</strong>, ${data.address.street}, ${data.address.suite}.</p>
            <h3>Professional Information</h3>
            <p>I work at <strong>${data.company.name}</strong>.</p>
            <blockquote>${data.company.catchPhrase}</blockquote>
            <p><em>${data.company.bs}</em></p>
            <h3>Contact</h3>
            <ul>
              <li>Phone: ${data.phone}</li>
              <li>Website: <a href="http://${data.website}">${data.website}</a></li>
            </ul>
          `
        };

        setUser(userWithBio);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (userId) {
      fetchUserBiography();
    }
  }, [userId]);

  if (loading) {
    return (
      <div style={styles.loadingContainer}>
        <div style={styles.spinner}></div>
        <p style={styles.loadingText}>Loading biography...</p>
      </div>
    );
  }

  if (error) {
    return (
      <div style={styles.errorContainer}>
        <h3 style={styles.errorTitle}>Error loading biography</h3>
        <p style={styles.errorMessage}>{error}</p>
        <button
          style={styles.retryButton}
          onClick={() => setLoading(true)}
        >
          Try Again
        </button>
      </div>
    );
  }

  if (!user) {
    return (
      <div style={styles.emptyContainer}>
        <p>No user data available.</p>
      </div>
    );
  }

  return (
    <div style={styles.container}>
      <div style={styles.header}>
        <div style={styles.avatar}>
          {user.name.charAt(0).toUpperCase()}
        </div>
        <div style={styles.headerInfo}>
          <h1 style={styles.userName}>{user.name}</h1>
          <p style={styles.userHandle}>@{user.username}</p>
        </div>
      </div>

      <div
        style={styles.biographyContent}
        dangerouslySetInnerHTML={{ __html: user.biography }}
      />

      <div style={styles.footer}>
        <p style={styles.footerNote}>
          ⚠️ This content was rendered using <code>dangerouslySetInnerHTML</code>.
          Ensure HTML content is sanitized before rendering in production.
        </p>
      </div>
    </div>
  );
};
