
(function() {
  function loadScript(url, callback) {
    var script = document.createElement('script');
    script.type = 'text/javascript';
    script.src = url;
    script.async = true;

    script.onload = function() {
      console.log('Library loaded successfully from: ' + url);
      if (typeof callback === 'function') {
        callback(null, script);
      }
    };

    script.onerror = function() {
      var error = new Error('Failed to load script from: ' + url);
      console.error(error.message);
      if (typeof callback === 'function') {
        callback(error, null);
      }
    };

    var head = document.head || document.getElementsByTagName('head')[0];
    head.appendChild(script);
  }

  var jQueryCDN = 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';

  function isJQueryLoaded() {
    return typeof window.jQuery !== 'undefined';
  }

  function initializeApp() {
    $(document).ready(function() {
      console.log('jQuery version: ' + $.fn.jquery);
      console.log('DOM is ready. jQuery is working correctly!');

      $('body').append('<div id="jquery-test" style="padding:10px; background:#f0f0f0; margin:10px; border-radius:4px;">jQuery loaded successfully! Version: ' + $.fn.jquery + '</div>');

      $('#jquery-test').css({
        'border': '2px solid #4CAF50',
        'color': '#333',
        'font-family': 'Arial, sans-serif'
      }).fadeIn(500);
    });
  }

  if (isJQueryLoaded()) {
    console.log('jQuery is already loaded. Version: ' + jQuery.fn.jquery);
    initializeApp();
  } else {
    console.log('Loading jQuery from CDN...');
    loadScript(jQueryCDN, function(error, script) {
      if (error) {
        console.error('Could not load jQuery:', error.message);
        var fallbackCDN = 'https://code.jquery.com/jquery-3.7.1.min.js';
        console.log('Trying fallback CDN: ' + fallbackCDN);
        loadScript(fallbackCDN, function(fallbackError) {
          if (fallbackError) {
            console.error('All CDN sources failed. Please check your internet connection.');
          } else {
            console.log('jQuery loaded from fallback CDN.');
            initializeApp();
          }
        });
      } else {
        initializeApp();
      }
    });
  }
})();
