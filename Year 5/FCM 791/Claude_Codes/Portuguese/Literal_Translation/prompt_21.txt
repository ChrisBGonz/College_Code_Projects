import jwt
import json
from datetime import datetime, timedelta, timezone
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SECRET_KEY = "my-super-secret-key-for-jwt-demo"
ALGORITHM = "HS256"

def create_jwt_token(payload: dict, expiry_minutes: int = 30) -> str:
    """Create a JWT token with the given payload."""
    payload_copy = payload.copy()
    payload_copy['exp'] = datetime.now(timezone.utc) + timedelta(minutes=expiry_minutes)
    payload_copy['iat'] = datetime.now(timezone.utc)
    
    token = jwt.encode(payload_copy, SECRET_KEY, algorithm=ALGORITHM)
    return token

def decode_jwt_token(token: str) -> dict:
    """Decode and verify a JWT token."""
    try:
        decoded = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return {"success": True, "payload": decoded}
    except jwt.ExpiredSignatureError:
        return {"success": False, "error": "Token has expired"}
    except jwt.InvalidTokenError as e:
        return {"success": False, "error": f"Invalid token: {str(e)}"}

def get_token_info(token: str) -> dict:
    """Get information about a JWT token without verification."""
    try:
        header = jwt.get_unverified_header(token)
        payload = jwt.decode(token, options={"verify_signature": False})
        
        exp_time = None
        if 'exp' in payload:
            exp_time = datetime.fromtimestamp(payload['exp'], tz=timezone.utc).isoformat()
        
        iat_time = None
        if 'iat' in payload:
            iat_time = datetime.fromtimestamp(payload['iat'], tz=timezone.utc).isoformat()
        
        return {
            "header": header,
            "payload": payload,
            "expiration": exp_time,
            "issued_at": iat_time
        }
    except Exception as e:
        return {"error": str(e)}

def process_jwt_request(user_message: str) -> str:
    """Process JWT-related requests using Claude AI."""
    
    system_prompt = """You are a JWT (JSON Web Token) expert assistant. You help users with:
1. Creating JWT tokens with custom payloads
2. Decoding and verifying JWT tokens
3. Explaining JWT structure and security
4. Troubleshooting JWT issues

When users ask to create a token, extract the payload data they want and respond with:
ACTION: CREATE_TOKEN
PAYLOAD: {json payload here}
EXPIRY: {minutes, default 30}

When users provide a token to decode, respond with:
ACTION: DECODE_TOKEN
TOKEN: {the token}

When users want token info without verification:
ACTION: TOKEN_INFO
TOKEN: {the token}

Otherwise, provide helpful explanations about JWT.

Always be helpful and explain JWT concepts clearly."""

    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    result = handle_jwt_action(assistant_message)
    
    if result:
        final_response = f"{assistant_message}\n\n**JWT Operation Result:**\n{result}"
        conversation_history[-1]["content"] = final_response
        return final_response
    
    return assistant_message

def handle_jwt_action(response_text: str) -> str:
    """Handle JWT actions based on Claude's response."""
    
    if "ACTION: CREATE_TOKEN" in response_text:
        try:
            lines = response_text.split('\n')
            payload_str = ""
            expiry = 30
            
            for i, line in enumerate(lines):
                if line.startswith("PAYLOAD:"):
                    payload_str = line.replace("PAYLOAD:", "").strip()
                    if not payload_str and i + 1 < len(lines):
                        payload_str = lines[i + 1].strip()
                elif line.startswith("EXPIRY:"):
                    try:
                        expiry = int(line.replace("EXPIRY:", "").strip())
                    except:
                        expiry = 30
            
            if payload_str:
                payload = json.loads(payload_str)
                token = create_jwt_token(payload, expiry)
                return f"✅ Token created successfully!\n\nToken:\n{token}\n\nExpires in: {expiry} minutes"
        except Exception as e:
            return f"❌ Error creating token: {str(e)}"
    
    elif "ACTION: DECODE_TOKEN" in response_text:
        try:
            lines = response_text.split('\n')
            token = ""
            
            for line in lines:
                if line.startswith("TOKEN:"):
                    token = line.replace("TOKEN:", "").strip()
                    break
            
            if token:
                result = decode_jwt_token(token)
                if result["success"]:
                    return f"✅ Token decoded successfully!\n\nPayload:\n{json.dumps(result['payload'], indent=2)}"
                else:
                    return f"❌ Decoding failed: {result['error']}"
        except Exception as e:
            return f"❌ Error decoding token: {str(e)}"
    
    elif "ACTION: TOKEN_INFO" in response_text:
        try:
            lines = response_text.split('\n')
            token = ""
            
            for line in lines:
                if line.startswith("TOKEN:"):
                    token = line.replace("TOKEN:", "").strip()
                    break
            
            if token:
                info = get_token_info(token)
                if "error" not in info:
                    return f"📋 Token Information:\n\nHeader:\n{json.dumps(info['header'], indent=2)}\n\nPayload:\n{json.dumps(info['payload'], indent=2)}\n\nIssued At: {info.get('issued_at', 'N/A')}\nExpires At: {info.get('expiration', 'N/A')}"
                else:
                    return f"❌ Error getting token info: {info['error']}"
        except Exception as e:
            return f"❌ Error getting token info: {str(e)}"
    
    return None

def main():
    """Main function to run the JWT assistant chatbot."""
    print("=" * 60)
    print("JWT Authentication Assistant")
    print("=" * 60)
    print("I can help you with JWT token creation, decoding, and authentication.")
    print("Type 'quit' or 'exit' to stop.\n")
    
    print("Creating a demo token for you to start with...")
    demo_payload = {
        "user_id": "12345",
        "username": "demo_user",
        "role": "admin",
        "permissions": ["read", "write"]
    }
    demo_token = create_jwt_token(demo_payload, 60)
    print(f"Demo token created:\n{demo_token}\n")
    print("You can use this token to test decoding and verification.\n")
    print("-" * 60)
    
    while True:
        try:
            user_input = input("\nYou: ").strip()
            
            if not user_input:
                continue
            
            if user_input.lower() in ['quit', 'exit']:
                print("Goodbye! Stay secure with JWT! 🔐")
                break
            
            print("\nAssistant: ", end="", flush=True)
            response = process_jwt_request(user_input)
            print(response)
            
        except KeyboardInterrupt:
            print("\n\nGoodbye! Stay secure with JWT! 🔐")
            break
        except Exception as e:
            print(f"\nError: {str(e)}")
            print("Please try again.")

if __name__ == "__main__":
    main()
