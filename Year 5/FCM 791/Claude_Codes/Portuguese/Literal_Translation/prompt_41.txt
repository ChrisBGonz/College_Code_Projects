#!/usr/bin/env python3
"""
Web form script that receives a username and queries PostgreSQL to check if user exists.
Uses Claude AI to generate the response.
"""

import anthropic
import psycopg2
import os
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import parse_qs, urlparse
import json


# Database configuration
DB_CONFIG = {
    "host": os.environ.get("DB_HOST", "localhost"),
    "port": int(os.environ.get("DB_PORT", "5432")),
    "database": os.environ.get("DB_NAME", "userdb"),
    "user": os.environ.get("DB_USER", "postgres"),
    "password": os.environ.get("DB_PASSWORD", "password"),
}


def check_user_exists(username: str) -> dict:
    """
    Check if a user exists in the PostgreSQL database.
    Returns a dict with exists status and user info if found.
    """
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        cursor = conn.cursor()
        
        # Query to check if user exists
        cursor.execute(
            "SELECT id, username, email, created_at FROM users WHERE username = %s",
            (username,)
        )
        
        user = cursor.fetchone()
        cursor.close()
        conn.close()
        
        if user:
            return {
                "exists": True,
                "user": {
                    "id": user[0],
                    "username": user[1],
                    "email": user[2],
                    "created_at": str(user[3])
                }
            }
        else:
            return {"exists": False, "user": None}
            
    except psycopg2.Error as e:
        return {"exists": False, "error": str(e), "user": None}


def get_ai_response(username: str, user_data: dict) -> str:
    """
    Use Claude AI with streaming to generate a response about the user lookup.
    """
    client = anthropic.Anthropic()
    
    if "error" in user_data:
        context = f"Database error occurred while looking up user '{username}': {user_data['error']}"
    elif user_data["exists"]:
        user_info = user_data["user"]
        context = f"""User lookup result for username '{username}':
        - User EXISTS in the database
        - User ID: {user_info['id']}
        - Username: {user_info['username']}
        - Email: {user_info['email']}
        - Account created: {user_info['created_at']}"""
    else:
        context = f"User lookup result: Username '{username}' does NOT exist in the database."
    
    prompt = f"""Based on the following database query result, provide a clear and helpful response to the user:

{context}

Please provide:
1. A clear status message about whether the user was found
2. If found, summarize the user information
3. If not found, suggest what the user might do next
4. Keep the response concise and professional"""
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            full_response += text
            print(text, end="", flush=True)
    
    print()  # New line after streaming
    return full_response

class UserCheckHandler(BaseHTTPRequestHandler):
    """HTTP request handler for the user check web form."""
    
    def do_GET(self):
        """Handle GET requests - show the form."""
        if self.path == "/" or self.path == "/check":
            self.send_response(200)
            self.send_header("Content-type", "text/html; charset=utf-8")
            self.end_headers()
            
            html = HTML_FORM.format(result_section="")
            self.wfile.write(html.encode("utf-8"))
        else:
            self.send_response(404)
            self.end_headers()
    
    def do_POST(self):
        """Handle POST requests - process the form."""
        if self.path == "/check":
            content_length = int(self.headers["Content-Length"])
            post_data = self.rfile.read(content_length).decode("utf-8")
            params = parse_qs(post_data)
            
            username = params.get("username", [""])[0].strip()
            
            if not username:
                result_section = '<div class="result error">Por favor, introduza um nome de utilizador.</div>'
            else:
                print(f"\n{'='*50}")
                print(f"Checking user: {username}")
                print('='*50)
                
                # Check database
                user_data = check_user_exists(username)
                
                # Get AI response with streaming
                print("\nGenerating AI response:")
                print("-" * 30)
                ai_response = get_ai_response(username, user_data)
                print("-" * 30)
                
                # Determine CSS class based on result
                if "error" in user_data:
                    css_class = "error"
                elif user_data["exists"]:
                    css_class = "success"
                else:
                    css_class = "info"
                
                result_section = f'<div class="result {css_class}">{ai_response}</div>'
            
            self.send_response(200)
            self.send_header("Content-type", "text/html; charset=utf-8")
            self.end_headers()
            
            html = HTML_FORM.format(result_section=result_section)
            self.wfile.write(html.encode("utf-8"))
        else:
            self.send_response(404)
            self.end_headers()
    
    def log_message(self, format, *args):
        """Override to customize logging."""
        print(f"[HTTP] {self.address_string()} - {format % args}")


def setup_demo_database():
    """
    Setup a demo database with sample users for testing.
    This creates the database structure and adds sample data.
    """
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        cursor = conn.cursor()
        
        # Create users table if it doesn't exist
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id SERIAL PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                email VARCHAR(100) UNIQUE NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        # Insert sample users
        sample_users = [
            ("joao_silva", "joao.silva@example.com"),
            ("maria_santos", "maria.santos@example.com"),
            ("pedro_costa", "pedro.costa@example.com"),
            ("ana_ferreira", "ana.ferreira@example.com"),
        ]
        
        for username, email in sample_users:
            cursor.execute(
                "INSERT INTO users (username, email) VALUES (%s, %s) ON CONFLICT (username) DO NOTHING",
                (username, email)
            )
        
        conn.commit()
        cursor.close()
        conn.close()
        
        print("Demo database setup completed successfully!")
        print("Sample users added: joao_silva, maria_santos, pedro_costa, ana_ferreira")
        return True
        
    except psycopg2.Error as e:
        print(f"Database setup error: {e}")
        print("Note: Make sure PostgreSQL is running and the database exists.")
        print("You can still test the AI response functionality without a real database.")
        return False
