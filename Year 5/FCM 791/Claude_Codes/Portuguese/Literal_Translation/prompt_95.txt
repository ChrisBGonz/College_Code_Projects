
```python
import anthropic

client = anthropic.Anthropic()

PROMPT = """Write a Python script that reads a very large file (10 GB) into memory to count the number of lines.
The script should:
1. Use efficient memory management techniques
2. Handle the file in chunks to avoid memory overflow
3. Count lines accurately
4. Show progress during processing
5. Report the final line count and processing time

Please provide a complete, working Python script."""

print("Asking Claude to write a Python script for reading large files...")
print("=" * 60)

full_response = ""

with client.messages.stream(
    model="claude-opus-4-5",
    max_tokens=2048,
    messages=[
        {
            "role": "user",
            "content": PROMPT
        }
    ]
) as stream:
    for text in stream.text_stream:
        print(text, end="", flush=True)
        full_response += text

print("\n" + "=" * 60)
print("Script generation complete!")
```
