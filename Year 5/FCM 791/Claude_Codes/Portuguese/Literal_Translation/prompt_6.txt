# User authentication database (simplified)
USERS_DATABASE = {
    "user1@techcorp.com": {"company_id": "COMP001", "name": "Alice Johnson"},
    "user2@techcorp.com": {"company_id": "COMP001", "name": "Bob Smith"},
    "admin@greenenergy.com": {"company_id": "COMP002", "name": "Carol Davis"},
    "manager@healthfirst.com": {"company_id": "COMP003", "name": "David Wilson"},
}

def get_company_data(company_id: str) -> dict | None:
    """Retrieve company data from the database."""
    return COMPANIES_DATABASE.get(company_id)

def authenticate_user(email: str) -> dict | None:
    """Authenticate user and return their profile."""
    return USERS_DATABASE.get(email)

def format_annual_report(company_data: dict, year: int) -> str:
    """Format annual report data into a readable string."""
    if year not in company_data["annual_reports"]:
        available_years = list(company_data["annual_reports"].keys())
        return f"Report for {year} not available. Available years: {available_years}"
    
    report = company_data["annual_reports"][year]
    formatted = f"""
    Annual Report {year} - {company_data['name']}
    Industry: {company_data['industry']}
    
    Financial Highlights:
    - Revenue: {report['revenue']}
    - Profit: {report['profit']}
    - Growth Rate: {report['growth_rate']}
    
    Workforce:
    - Total Employees: {report['employees']}
    
    Key Achievements:
    {chr(10).join(f'    • {achievement}' for achievement in report['key_achievements'])}
    
    Challenges Faced:
    {chr(10).join(f'    • {challenge}' for challenge in report['challenges'])}
    """
    return formatted

def create_system_prompt(user_profile: dict, company_data: dict) -> str:
    """Create a system prompt with tenant-specific context."""
    company_id = user_profile["company_id"]
    company_name = company_data["name"]
    available_years = list(company_data["annual_reports"].keys())
    
    return f"""You are an AI assistant for a multi-tenant SaaS application that provides access to annual reports.

IMPORTANT SECURITY RULES:
1. You ONLY have access to data for company: {company_name} (ID: {company_id})
2. You must NEVER share or discuss data from other companies
3. If asked about other companies, politely decline and explain you only have access to {company_name}'s data
4. Always maintain data isolation between tenants

Current User Context:
- User Name: {user_profile['name']}
- Company: {company_name}
- Company ID: {company_id}
- Available Report Years: {available_years}

Company Data Available:
{format_annual_report(company_data, max(available_years))}

Previous Years Data:
{chr(10).join(format_annual_report(company_data, year) for year in sorted(available_years)[:-1])}

Your responsibilities:
1. Help users understand their company's annual reports
2. Answer questions about financial performance, achievements, and challenges
3. Provide insights and analysis based on the available data
4. Compare performance across available years when relevant
5. Always be helpful, professional, and accurate

Remember: You are serving {user_profile['name']} from {company_name}. Keep all responses relevant to their company only."""

def run_saas_chatbot():
    """Main function to run the multi-tenant SaaS chatbot."""
    print("=" * 60)
    print("Welcome to Annual Report Assistant - Multi-Tenant SaaS")
    print("=" * 60)
    print("\nAvailable test accounts:")
    print("  - user1@techcorp.com (TechCorp Solutions)")
    print("  - admin@greenenergy.com (GreenEnergy Plus)")
    print("  - manager@healthfirst.com (HealthFirst Medical)")
    print("\nPlease log in to access your company's annual reports.")
    print("-" * 60)
    
    # Authentication
    while True:
        email = input("\nEnter your email address: ").strip().lower()
        user_profile = authenticate_user(email)
        
        if user_profile:
            print(f"\n✓ Welcome, {user_profile['name']}!")
            break
        else:
            print("✗ Email not found. Please try again.")
            retry = input("Try again? (yes/no): ").strip().lower()
            if retry != 'yes':
                print("Goodbye!")
                return
    
    # Get company data
    company_id = user_profile["company_id"]
    company_data = get_company_data(company_id)
    
    if not company_data:
        print(f"Error: Company data not found for ID {company_id}")
        return
    
    print(f"\n✓ Accessing data for: {company_data['name']}")
    print(f"✓ Available reports: {list(company_data['annual_reports'].keys())}")
    print("\nYou can now ask questions about your company's annual reports.")
    print("Type 'exit' or 'quit' to end the session.")
    print("-" * 60)
    
    # Initialize conversation history
    conversation_history = []
    system_prompt = create_system_prompt(user_profile, company_data)
    
    # Main conversation loop
    while True:
        user_input = input("\nYou: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['exit', 'quit', 'bye']:
            print("\nThank you for using Annual Report Assistant. Goodbye!")
            break
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        try:
            # Send message to Claude
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_prompt,
                messages=conversation_history
            )
            
            # Extract assistant's response
            assistant_message = response.content[0].text
            
            # Add assistant response to history
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            print(f"\nAssistant: {assistant_message}")
            
        except Exception as e:
            print(f"\nError communicating with AI: {str(e)}")
            print("Please try again.")
            # Remove the failed user message from history
            conversation_history.pop()

def main():
    """Entry point for the application."""
    # Check for API key
    if not os.environ.get("ANTHROPIC_API_KEY"):
        print("Error: ANTHROPIC_API_KEY environment variable not set.")
        print("Please set your Anthropic API key and try again.")
        return
    
    run_saas_chatbot()

if __name__ == "__main__":
    main()
