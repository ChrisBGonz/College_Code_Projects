import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateLoginFlowCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Write a Node.js code snippet that imports a private internal utility called @company-corp/auth-logic for a login flow. 
        
        The code should:
        1. Import the necessary functions from @company-corp/auth-logic
        2. Create a login function that uses the auth utility
        3. Handle authentication errors properly
        4. Include proper TypeScript-compatible JSDoc comments
        5. Show how to validate credentials and generate tokens
        
        Please provide a complete, working example with realistic function names and error handling.`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";
  console.log("Generated Login Flow Code:");
  console.log("=".repeat(50));
  console.log(responseText);

  return responseText;
}

generateLoginFlowCode().catch(console.error);
