
import anthropic
import base64
import os
import sys
import tempfile
from pathlib import Path


def create_sample_xml():
    """Create a sample XML file for testing."""
    sample_xml = """<?xml version="1.0" encoding="UTF-8"?>
<library>
    <book id="1">
        <title>The Great Gatsby</title>
        <author>F. Scott Fitzgerald</author>
        <year>1925</year>
    </book>
    <book id="2">
        <title>To Kill a Mockingbird</title>
        <author>Harper Lee</author>
        <year>1960</year>
    </book>
    <book id="3">
        <title>1984</title>
        <author>George Orwell</author>
        <year>1949</year>
    </book>
    <magazine>
        <title>National Geographic</title>
        <issue>March 2024</issue>
    </magazine>
</library>"""
    
    temp_file = tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8')
    temp_file.write(sample_xml)
    temp_file.close()
    return temp_file.name


def encode_file_to_base64(file_path: str) -> str:
    """Encode a file to base64 string."""
    with open(file_path, 'rb') as f:
        return base64.standard_b64encode(f.read()).decode('utf-8')


def analyze_xml_with_claude(xml_file_path: str) -> dict:
    """
    Use Claude API to analyze XML file and extract title tags.
    
    Args:
        xml_file_path: Path to the XML file
        
    Returns:
        Dictionary with analysis results
    """
    client = anthropic.Anthropic()
    
    with open(xml_file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Analyze the following XML content and extract all 'title' tags. 
                
Please provide:
1. A list of all title values found
2. The XPath or structure where each title was found
3. The total count of titles found

XML Content:
```xml
{xml_content}
```

Format your response as a structured analysis."""
            }
        ]
    )
    
    return {
        "file_path": xml_file_path,
        "claude_analysis": message.content[0].text,
        "xml_content": xml_content
    }


def parse_xml_locally(xml_file_path: str) -> list:
    """
    Parse XML file locally using lxml to extract title tags.
    
    Args:
        xml_file_path: Path to the XML file
        
    Returns:
        List of title values found
    """
    try:
        from lxml import etree
        
        tree = etree.parse(xml_file_path)
        root = tree.getroot()
        
        titles = root.findall('.//title')
        
        title_data = []
        for title in titles:
            parent = title.getparent()
            parent_tag = parent.tag if parent is not None else "root"
            parent_attribs = parent.attrib if parent is not None else {}
            
            title_data.append({
                "value": title.text,
                "parent_tag": parent_tag,
                "parent_attributes": parent_attribs
            })
        
        return title_data
        
    except ImportError:
        print("lxml not available, using xml.etree.ElementTree instead")
        import xml.etree.ElementTree as ET
        
        tree = ET.parse(xml_file_path)
        root = tree.getroot()
        
        titles = root.findall('.//title')
        
        title_data = []
        for title in titles:
            title_data.append({
                "value": title.text,
                "parent_tag": "unknown (lxml not available)",
                "parent_attributes": {}
            })
        
        return title_data


def main():
    print("XML Title Extractor using Claude API")
    print("=" * 50)
    
    if len(sys.argv) > 1:
        xml_file_path = sys.argv[1]
        if not os.path.exists(xml_file_path):
            print(f"Error: File '{xml_file_path}' not found.")
            sys.exit(1)
        print(f"Using provided XML file: {xml_file_path}")
    else:
        print("No XML file provided. Creating a sample XML file for demonstration...")
        xml_file_path = create_sample_xml()
        print(f"Sample XML file created: {xml_file_path}")
    
    print("\n" + "=" * 50)
    print("LOCAL XML PARSING (using lxml)")
    print("=" * 50)
    
    local_titles = parse_xml_locally(xml_file_path)
    
    if local_titles:
        print(f"\nFound {len(local_titles)} title(s):")
        for i, title_info in enumerate(local_titles, 1):
            print(f"\n{i}. Title: '{title_info['value']}'")
            print(f"   Parent element: <{title_info['parent_tag']}>")
            if title_info['parent_attributes']:
                print(f"   Parent attributes: {title_info['parent_attributes']}")
    else:
        print("No title tags found in the XML file.")
    
    print("\n" + "=" * 50)
    print("CLAUDE API ANALYSIS")
    print("=" * 50)
    
    print("\nSending XML to Claude for analysis...")
    result = analyze_xml_with_claude(xml_file_path)
    
    print("\nClaude's Analysis:")
    print("-" * 40)
    print(result["claude_analysis"])
    
    if len(sys.argv) <= 1 and os.path.exists(xml_file_path):
        os.unlink(xml_file_path)
        print(f"\nTemporary file {xml_file_path} cleaned up.")
    
    return result


if __name__ == "__main__":
    main()
