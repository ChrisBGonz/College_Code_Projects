
```python
import os
import secrets
from flask import Flask, request, make_response, jsonify, render_template_string
from anthropic import Anthropic

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)

client = Anthropic()

# Simple user database for demonstration
USERS = {
    "admin": "password123",
    "user": "mypassword",
    "demo": "demo123"
}

# Store conversation histories per session
conversation_histories = {}

HTML_TEMPLATE = """
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Flask Auth App with AI Assistant</title>
    <style>
        body { font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }
        .container { background: #f5f5f5; padding: 30px; border-radius: 10px; }
        input { padding: 10px; margin: 5px; border: 1px solid #ddd; border-radius: 5px; }
        button { padding: 10px 20px; background: #007bff; color: white; border: none; border-radius: 5px; cursor: pointer; }
        button:hover { background: #0056b3; }
        .chat-container { margin-top: 20px; border: 1px solid #ddd; border-radius: 10px; padding: 20px; background: white; }
        .messages { height: 300px; overflow-y: auto; margin-bottom: 10px; padding: 10px; border: 1px solid #eee; border-radius: 5px; }
        .message { margin: 10px 0; padding: 10px; border-radius: 5px; }
        .user-message { background: #007bff; color: white; text-align: right; }
        .assistant-message { background: #e9ecef; color: #333; }
        .status { padding: 10px; margin: 10px 0; border-radius: 5px; }
        .success { background: #d4edda; color: #155724; }
        .error { background: #f8d7da; color: #721c24; }
        .hidden { display: none; }
        #chat-input { width: 70%; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Flask Auth App with AI Assistant</h1>
        
        <div id="login-section">
            <h2>Login</h2>
            <div>
                <input type="text" id="username" placeholder="Username" />
                <input type="password" id="password" placeholder="Password" />
                <button onclick="login()">Login</button>
            </div>
            <p><small>Test credentials: admin/password123, user/mypassword, demo/demo123</small></p>
            <div id="login-status"></div>
        </div>
        
        <div id="chat-section" class="hidden">
            <h2>Welcome! You're logged in.</h2>
            <button onclick="logout()" style="background: #dc3545;">Logout</button>
            
            <div class="chat-container">
                <h3>AI Assistant (Multi-turn Conversation)</h3>
                <div class="messages" id="messages"></div>
                <div>
                    <input type="text" id="chat-input" placeholder="Ask me anything..." onkeypress="handleKeyPress(event)" />
                    <button onclick="sendMessage()">Send</button>
                    <button onclick="clearHistory()" style="background: #6c757d;">Clear History</button>
                </div>
            </div>
        </div>
    </div>
    
    <script>
        // Check if already logged in
        checkAuthStatus();
        
        async function checkAuthStatus() {
            const response = await fetch('/check-auth');
            const data = await response.json();
            if (data.authenticated) {
                showChatSection();
            }
        }
        
        async function login() {
            const username = document.getElementById('username').value;
            const password = document.getElementById('password').value;
            
            const response = await fetch('/login', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ username, password })
            });
            
            const data = await response.json();
            const statusDiv = document.getElementById('login-status');
            
            if (data.success) {
                statusDiv.innerHTML = '<div class="status success">Login successful! Cookie set.</div>';
                showChatSection();
            } else {
                statusDiv.innerHTML = `<div class="status error">${data.message}</div>`;
            }
        }
        
        async function logout() {
            await fetch('/logout', { method: 'POST' });
            document.getElementById('chat-section').classList.add('hidden');
            document.getElementById('login-section').classList.remove('hidden');
            document.getElementById('messages').innerHTML = '';
        }
        
        function showChatSection() {
            document.getElementById('login-section').classList.add('hidden');
            document.getElementById('chat-section').classList.remove('hidden');
        }
        
        async function sendMessage() {
            const input = document.getElementById('chat-input');
            const message = input.value.trim();
            if (!message) return;
            
            addMessage(message, 'user');
            input.value = '';
            
            const response = await fetch('/chat', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ message })
            });
            
            const data = await response.json();
            if (data.success) {
                addMessage(data.response, 'assistant');
            } else {
                addMessage('Error: ' + data.message, 'assistant');
            }
        }
        
        async function clearHistory() {
            await fetch('/clear-history', { method: 'POST' });
            document.getElementById('messages').innerHTML = '';
        }
        
        function addMessage(text, sender) {
            const messagesDiv = document.getElementById('messages');
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${sender}-message`;
            messageDiv.textContent = text;
            messagesDiv.appendChild(messageDiv);
            messagesDiv.scrollTop = messagesDiv.scrollHeight;
        }
        
        function handleKeyPress(event) {
            if (event.key === 'Enter') {
                sendMessage();
            }
        }
    </script>
</body>
</html>
"""

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    username = data.get('username', '')
    password = data.get('password', '')
    
    if username in USERS and USERS[username] == password:
        # Generate a secure auth token
        auth_token = secrets.token_hex(32)
        
        # Store the token with user info (in production, use a database)
        conversation_histories[auth_token] = {
            'username': username,
            'messages': []
        }
        
        # Create response with cookie
        response = make_response(jsonify({
            'success': True,
            'message': f'Welcome, {username}!'
        }))
        
        # Set the auth_token cookie
        response.set_cookie(
            'auth_token',
            auth_token,
            httponly=True,  # Prevents JavaScript access
            secure=False,   # Set to True in production with HTTPS
            samesite='Lax',
            max_age=3600    # 1 hour expiration
        )
        
        return response
    else:
        return jsonify({
            'success': False,
            'message': 'Invalid username or password'
        }), 401

@app.route('/logout', methods=['POST'])
def logout():
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and auth_token in conversation_histories:
        del conversation_histories[auth_token]
    
    response = make_response(jsonify({'success': True}))
    response.delete_cookie('auth_token')
    
    return response

@app.route('/check-auth')
def check_auth():
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and auth_token in conversation_histories:
        username = conversation_histories[auth_token]['username']
        return jsonify({
            'authenticated': True,
            'username': username
        })
    
    return jsonify({'authenticated': False})

@app.route('/chat', methods=['POST'])
def chat():
    auth_token = request.cookies.get('auth_token')
    
    if not auth_token or auth_token not in conversation_histories:
        return jsonify({
            'success': False,
            'message': 'Not authenticated'
        }), 401
    
    data = request.get_json()
    user_message = data.get('message', '')
    
    if not user_message:
        return jsonify({
            'success': False,
            'message': 'No message provided'
        }), 400
    
    # Get conversation history for this session
    session_data = conversation_histories[auth_token]
    messages = session_data['messages']
    
    # Add user message to history
    messages.append({
        'role': 'user',
        'content': user_message
    })
    
    # Call Claude API with conversation history
    response = client.messages.create(
        model='claude-opus-4-5',
        max_tokens=8096,
        system=f"You are a helpful AI assistant. The user '{session_data['username']}' is logged in and chatting with you. Be friendly and helpful.",
        messages=messages
    )
    
    assistant_message = response.content[0].text
    
    # Add assistant response to history
    messages.append({
        'role': 'assistant',
        'content': assistant_message
    })
    
    return jsonify({
        'success': True,
        'response': assistant_message
    })

@app.route('/clear-history', methods=['POST'])
def clear_history():
    auth_token = request.cookies.get('auth_token')
    
    if auth_token and auth_token in conversation_histories:
        conversation_histories[auth_token]['messages'] = []
    
    return jsonify({'success': True})

if __name__ == '__main__':
    app.run(debug=True, port=5000)
```
