
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];
let inactivityTimer = null;
const INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
let isSessionActive = false;
let rl = null;

function resetInactivityTimer() {
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
  }

  inactivityTimer = setTimeout(() => {
    disconnectUser();
  }, INACTIVITY_TIMEOUT);
}

function disconnectUser() {
  if (isSessionActive) {
    isSessionActive = false;
    console.log("\n⚠️  SESSÃO ENCERRADA POR INATIVIDADE");
    console.log(
      "Você foi desconectado automaticamente após 15 minutos de inatividade."
    );
    console.log("Por favor, faça login novamente para continuar.");

    if (inactivityTimer) {
      clearTimeout(inactivityTimer);
      inactivityTimer = null;
    }

    if (rl) {
      rl.close();
    }

    process.exit(0);
  }
}

async function chat(userMessage) {
  if (!isSessionActive) {
    console.log("Sessão não está ativa. Por favor, faça login primeiro.");
    return;
  }

  resetInactivityTimer();

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "Você é um assistente bancário virtual prestativo e seguro. Você ajuda os usuários com consultas bancárias, verificação de saldo, transferências e outras operações bancárias. Sempre priorize a segurança e privacidade do usuário. Responda em português brasileiro.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function startSession(username) {
  isSessionActive = true;
  console.log(`\n✅ Bem-vindo(a), ${username}!`);
  console.log("Sua sessão bancária foi iniciada com sucesso.");
  console.log(
    `⏱️  Você será desconectado automaticamente após 15 minutos de inatividade.\n`
  );

  resetInactivityTimer();
}

function getRemainingTime() {
  if (!inactivityTimer) return 0;

  const timeoutId = inactivityTimer[Symbol.toPrimitive]
    ? inactivityTimer[Symbol.toPrimitive]()
    : null;
  return timeoutId;
}

async function main() {
  rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const question = (prompt) => {
    return new Promise((resolve) => {
      rl.question(prompt, resolve);
    });
  };

  console.log("=== SISTEMA BANCÁRIO SEGURO ===");
  console.log("Simulação de login bancário com timeout de inatividade\n");

  const username = await question("Digite seu nome de usuário: ");
  startSession(username);

  console.log("Comandos disponíveis:");
  console.log("  - Digite sua mensagem para interagir com o assistente");
  console.log('  - Digite "sair" para encerrar a sessão');
  console.log('  - Digite "status" para ver o status da sessão\n');

  while (isSessionActive) {
    const userInput = await question("Você: ");

    if (!isSessionActive) break;

    if (userInput.toLowerCase() === "sair") {
      console.log("\nEncerrando sessão...");
      disconnectUser();
      break;
    }

    if (userInput.toLowerCase() === "status") {
      console.log(`\n📊 Status da Sessão:`);
      console.log(`   Usuário: ${username}`);
      console.log(`   Sessão ativa: ${isSessionActive}`);
      console.log(
        `   Timeout de inatividade: ${INACTIVITY_TIMEOUT / 60000} minutos`
      );
      console.log(`   Mensagens na conversa: ${conversationHistory.length}\n`);
      resetInactivityTimer();
      continue;
    }

    if (userInput.trim() === "") {
      continue;
    }

    console.log("\nAssistente: Processando sua solicitação...");

    const response = await chat(userInput);
    console.log(`\nAssistente: ${response}\n`);
  }
}

main().catch(console.error);
```
