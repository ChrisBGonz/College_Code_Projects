import anthropic
import subprocess
import sys


def install_package(package):
    """Install a package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])


def ensure_dependencies():
    """Ensure required packages are installed."""
    try:
        import weasyprint
    except ImportError:
        print("Installing weasyprint...")
        install_package("weasyprint")

    try:
        from jinja2 import Environment
    except ImportError:
        print("Installing jinja2...")
        install_package("jinja2")


def generate_html_template_with_claude(template_description: str, data: dict) -> str:
    """
    Use Claude to generate an HTML template based on description and data.

    Args:
        template_description: Description of what the HTML template should contain
        data: Dictionary with data to be used in the template

    Returns:
        Generated HTML content as string
    """
    client = anthropic.Anthropic()

    prompt = f"""Create a complete HTML template for the following description:
{template_description}

The template should use Jinja2 syntax for dynamic data. Here is the data that will be available:
{data}

Requirements:
1. Create a complete, valid HTML document with proper DOCTYPE, head, and body tags
2. Include inline CSS styles for a professional appearance
3. Use Jinja2 template syntax ({{ variable }}, {{% for %}}, {{% if %}}, etc.) for dynamic content
4. Make it visually appealing with proper formatting
5. Return ONLY the HTML code, no explanations or markdown

Generate the HTML template now:"""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[{"role": "user", "content": prompt}],
    )

    return message.content[0].text


def render_template(html_template: str, data: dict) -> str:
    """
    Render a Jinja2 HTML template with provided data.

    Args:
        html_template: HTML template string with Jinja2 syntax
        data: Dictionary with data to render the template

    Returns:
        Rendered HTML string
    """
    from jinja2 import Environment

    env = Environment()
    template = env.from_string(html_template)
    return template.render(**data)


def html_to_pdf(html_content: str, output_path: str) -> bool:
    """
    Convert HTML content to PDF using WeasyPrint.

    Args:
        html_content: HTML string to convert
        output_path: Path where the PDF file will be saved

    Returns:
        True if successful, False otherwise
    """
    try:
        from weasyprint import HTML

        HTML(string=html_content).write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True
    except Exception as e:
        print(f"Error converting HTML to PDF: {e}")
        return False


def generate_pdf_from_template(
    template_description: str, data: dict, output_path: str
) -> bool:
    """
    Main function that generates a PDF from a template description and data.

    This function:
    1. Uses Claude to generate an HTML template based on the description
    2. Renders the template with provided data using Jinja2
    3. Converts the rendered HTML to PDF using WeasyPrint

    Args:
        template_description: Description of what the PDF should contain
        data: Dictionary with data to populate the template
        output_path: Path where the PDF file will be saved

    Returns:
        True if successful, False otherwise
    """
    ensure_dependencies()

    print("Generating HTML template with Claude...")
    html_template = generate_html_template_with_claude(template_description, data)

    if not html_template:
        print("Failed to generate HTML template")
        return False

    print("Rendering template with data...")
    rendered_html = render_template(html_template, data)

    print("Converting HTML to PDF...")
    return html_to_pdf(rendered_html, output_path)


def create_invoice_pdf(invoice_data: dict, output_path: str) -> bool:
    """
    Create an invoice PDF from provided data.

    Args:
        invoice_data: Dictionary containing invoice information
        output_path: Path where the PDF will be saved

    Returns:
        True if successful, False otherwise
    """
    template_description = """
    A professional invoice template with:
    - Company header with logo placeholder
    - Invoice number and date
    - Client information section
    - Itemized list of products/services with quantities and prices
    - Subtotal, tax, and total calculations
    - Payment terms and notes section
    - Professional footer
    """

    return generate_pdf_from_template(template_description, invoice_data, output_path)


def create_report_pdf(report_data: dict, output_path: str) -> bool:
    """
    Create a report PDF from provided data.

    Args:
        report_data: Dictionary containing report information
        output_path: Path where the PDF will be saved

    Returns:
        True if successful, False otherwise
    """
    template_description = """
    A professional business report template with:
    - Title page with report name and date
    - Executive summary section
    - Key metrics displayed in a grid layout
    - Detailed findings section
    - Recommendations section
    - Professional styling with headers and sections
    """

    return generate_pdf_from_template(template_description, report_data, output_path)
