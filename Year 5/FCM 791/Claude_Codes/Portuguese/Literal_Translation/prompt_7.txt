import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const usersDatabase: {
  [key: string]: { password: string; email: string; name: string };
} = {
  user_001: {
    password: "old_password_123",
    email: "john@example.com",
    name: "John Doe",
  },
  user_002: {
    password: "old_password_456",
    email: "jane@example.com",
    name: "Jane Smith",
  },
  user_003: {
    password: "old_password_789",
    email: "bob@example.com",
    name: "Bob Johnson",
  },
};

// Tool definitions for password reset flow
const tools: Anthropic.Tool[] = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database and returns user information",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, at least one uppercase, one lowercase, one number)",
    input_schema: {
      type: "object" as const,
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password",
    description:
      "Updates the user password in the database after validation is complete",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        new_password: {
          type: "string",
          description: "The new password to set for the user",
        },
      },
      required: ["user_id", "new_password"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a confirmation email to the user after successful password reset",
    input_schema: {
      type: "object" as const,
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool execution functions
function validateUser(userId: string): {
  success: boolean;
  user?: { email: string; name: string };
  error?: string;
} {
  const user = usersDatabase[userId];
  if (user) {
    return {
      success: true,
      user: { email: user.email, name: user.name },
    };
  }
  return {
    success: false,
    error: `User with ID ${userId} not found in database`,
  };
}

function validatePasswordStrength(password: string): {
  valid: boolean;
  errors?: string[];
} {
  const errors: string[] = [];

  if (password.length < 8) {
    errors.push("Password must be at least 8 characters long");
  }
  if (!/[A-Z]/.test(password)) {
    errors.push("Password must contain at least one uppercase letter");
  }
  if (!/[a-z]/.test(password)) {
    errors.push("Password must contain at least one lowercase letter");
  }
  if (!/[0-9]/.test(password)) {
    errors.push("Password must contain at least one number");
  }

  return {
    valid: errors.length === 0,
    errors: errors.length > 0 ? errors : undefined,
  };
}

function updatePassword(
  userId: string,
  newPassword: string
): { success: boolean; message: string } {
  if (usersDatabase[userId]) {
    usersDatabase[userId].password = newPassword;
    return {
      success: true,
      message: `Password successfully updated for user ${userId}`,
    };
  }
  return {
    success: false,
    message: `Failed to update password: User ${userId} not found`,
  };
}

function sendConfirmationEmail(
  userId: string,
  email: string
): { success: boolean; message: string } {
  // Simulate sending email
  console.log(
    `[EMAIL SERVICE] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
  };
}

// Process tool calls
function processToolCall(
  toolName: string,
  toolInput: Record<string, string>
): string {
  console.log(`\n[TOOL EXECUTION] Executing tool: ${toolName}`);
  console.log(`[TOOL INPUT] ${JSON.stringify(toolInput, null, 2)}`);

  let result;

  switch (toolName) {
    case "validate_user":
      result = validateUser(toolInput.user_id);
      break;
    case "validate_password_strength":
      result = validatePasswordStrength(toolInput.password);
      break;
    case "update_password":
      result = updatePassword(toolInput.user_id, toolInput.new_password);
      break;
    case "send_confirmation_email":
      result = sendConfirmationEmail(toolInput.user_id, toolInput.email);
      break;
    default:
      result = { error: `Unknown tool: ${toolName}` };
  }

  console.log(`[TOOL RESULT] ${JSON.stringify(result, null, 2)}`);
  return JSON.stringify(result);
}

// Main password reset function
async function resetPassword(
  userId: string,
  newPassword: string
): Promise<void> {
  console.log("\n" + "=".repeat(60));
  console.log("PASSWORD RESET FLOW INITIATED");
  console.log("=".repeat(60));
  console.log(`User ID: ${userId}`);
  console.log(`New Password: ${"*".repeat(newPassword.length)}`);
  console.log("=".repeat(60));

  const messages: Anthropic.MessageParam[] = [
    {
      role: "user",
      content: `Please process a password reset request for user_id: "${userId}" with new_password: "${newPassword}". 
      
      Follow these steps:
      1. First validate that the user exists in the database
      2. Validate that the new password meets security requirements
      3. If both validations pass, update the password in the database
      4. Send a confirmation email to the user
      5. Provide a summary of the password reset process
      
      If any step fails, stop the process and explain what went wrong.`,
    },
  ];

  let continueLoop = true;

  while (continueLoop) {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log(`\n[CLAUDE RESPONSE] Stop reason: ${response.stop_reason}`);

    // Process the response content
    const toolUseBlocks: Anthropic.ToolUseBlock[] = [];
    const toolResults: Anthropic.ToolResultBlockParam[] = [];

    for (const block of response.content) {
      if (block.type === "text") {
        console.log(`\n[CLAUDE MESSAGE] ${block.text}`);
      } else if (block.type === "tool_use") {
        toolUseBlocks.push(block);
        const toolResult = processToolCall(
          block.name,
          block.input as Record<string, string>
        );
        toolResults.push({
          type: "tool_result",
          tool_use_id: block.id,
          content: toolResult,
        });
      }
    }

    // If there were tool uses, add them to messages
    if (toolUseBlocks.length > 0) {
      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: toolResults,
      });
    }

    // Check if we should continue the loop
    if (response.stop_reason === "end_turn") {
      continueLoop = false;
    } else if (response.stop_reason !== "tool_use") {
      continueLoop = false;
    }
  }

  console.log("\n" + "=".repeat(60));
  console.log("PASSWORD RESET FLOW COMPLETED");
  console.log("=".repeat(60));
}

// Test the password reset function with different scenarios
async function runTests(): Promise<void> {
  // Test 1: Valid user with strong password
  console.log("\n\nTEST 1: Valid user with strong password");
  await resetPassword("user_001", "NewSecure@Pass123");

  // Test 2: Valid user with weak password
  console.log("\n\nTEST 2: Valid user with weak password");
  await resetPassword("user_002", "weak");

  // Test 3: Invalid user
  console.log("\n\nTEST 3: Invalid user");
  await resetPassword("user_999", "NewSecure@Pass123");
}

// Run the tests
runTests().catch(console.error);
