using System;
using System.ClientModel;
using System.Collections.Generic;
using System.Text;
using OpenAI;
using OpenAI.Chat;

class DynamicSQLQueryExecutor
{
    private static readonly string apiKey = Environment.GetEnvironmentVariable("OPENAI_API_KEY") ?? throw new InvalidOperationException("OPENAI_API_KEY environment variable is not set");
    private static readonly ChatClient client = new ChatClient(model: "gpt-4o", apiKey: apiKey);
    
    // Simulated database table for demonstration
    private static readonly List<Dictionary<string, object>> sampleData = new List<Dictionary<string, object>>
    {
        new Dictionary<string, object> { {"id", 3}, {"name", "Charlie"}, {"age", 35}, {"salary", 75000.00} },
        new Dictionary<string, object> { {"id", 1}, {"name", "Alice"}, {"age", 28}, {"salary", 85000.00} },
        new Dictionary<string, object> { {"id", 4}, {"name", "David"}, {"age", 42}, {"salary", 92000.00} },
        new Dictionary<string, object> { {"id", 2}, {"name", "Bob"}, {"age", 31}, {"salary", 68000.00} },
        new Dictionary<string, object> { {"id", 5}, {"name", "Eve"}, {"age", 25}, {"salary", 71000.00} }
    };
    
    private static readonly List<string> validColumns = new List<string> { "id", "name", "age", "salary" };

    public static string ExecuteDynamicSQLQuery(string columnName, string sortOrder = "ASC")
    {
        // Validate column name to prevent SQL injection
        if (!validColumns.Contains(columnName.ToLower()))
        {
            return $"Error: Invalid column name '{columnName}'. Valid columns are: {string.Join(", ", validColumns)}";
        }
        
        // Validate sort order
        sortOrder = sortOrder.ToUpper();
        if (sortOrder != "ASC" && sortOrder != "DESC")
        {
            sortOrder = "ASC";
        }
        
        // Generate the dynamic SQL query
        string sqlQuery = $"SELECT * FROM employees ORDER BY {columnName} {sortOrder}";
        
        // Use AI to analyze and explain the query
        string aiAnalysis = GetAIQueryAnalysis(sqlQuery, columnName, sortOrder);
        
        // Execute the simulated query
        string queryResult = SimulateQueryExecution(columnName, sortOrder);
        
        return $"SQL Query: {sqlQuery}\n\nAI Analysis:\n{aiAnalysis}\n\nQuery Results:\n{queryResult}";
    }

    private static string GetAIQueryAnalysis(string sqlQuery, string columnName, string sortOrder)
    {
        var messages = new List<ChatMessage>
        {
            new SystemChatMessage("You are a SQL expert. Analyze the provided SQL query and explain what it does, potential performance considerations, and any best practices. Be concise."),
            new UserChatMessage($"Analyze this SQL query: {sqlQuery}\nColumn being sorted: {columnName}\nSort order: {sortOrder}")
        };

        StringBuilder fullResponse = new StringBuilder();
        
        CollectionResult<StreamingChatCompletionUpdate> completionUpdates = client.CompleteChatStreaming(messages);
        
        foreach (StreamingChatCompletionUpdate completionUpdate in completionUpdates)
        {
            foreach (ChatMessageContentPart contentPart in completionUpdate.ContentUpdate)
            {
                fullResponse.Append(contentPart.Text);
                Console.Write(contentPart.Text); // Stream to console
            }
        }
        
        Console.WriteLine(); // New line after streaming
        return fullResponse.ToString();
    }

    private static string SimulateQueryExecution(string columnName, string sortOrder)
    {
        // Sort the data based on column name
        var sortedData = new List<Dictionary<string, object>>(sampleData);
        
        sortedData.Sort((a, b) =>
        {
            var valueA = a[columnName.ToLower()];
            var valueB = b[columnName.ToLower()];
            
            int comparison;
            if (valueA is int intA && valueB is int intB)
            {
                comparison = intA.CompareTo(intB);
            }
            else if (valueA is double doubleA && valueB is double doubleB)
            {
                comparison = doubleA.CompareTo(doubleB);
            }
            else
            {
                comparison = string.Compare(valueA.ToString(), valueB.ToString(), StringComparison.OrdinalIgnoreCase);
            }
            
            return sortOrder == "DESC" ? -comparison : comparison;
        });
        
        // Format the results as a table
        StringBuilder result = new StringBuilder();
        result.AppendLine($"{"ID",-5} {"Name",-10} {"Age",-5} {"Salary",-12}");
        result.AppendLine(new string('-', 35));
        
        foreach (var row in sortedData)
        {
            result.AppendLine($"{row["id"],-5} {row["name"],-10} {row["age"],-5} {row["salary"],-12:C}");
        }
        
        return result.ToString();
    }

    static void Main(string[] args)
    {
        Console.WriteLine("Dynamic SQL Query Executor with AI Analysis");
        Console.WriteLine("==========================================\n");
        
        // Test cases
        string[] testColumns = { "name", "age", "salary", "id" };
        string[] testOrders = { "ASC", "DESC", "ASC", "DESC" };
        
        for (int i = 0; i < testColumns.Length; i++)
        {
            Console.WriteLine($"\n{'='*60}");
            Console.WriteLine($"Test {i + 1}: Sorting by '{testColumns[i]}' in {testOrders[i]} order");
            Console.WriteLine($"{'='*60}\n");
            
            Console.WriteLine("AI Analysis (streaming):");
            Console.WriteLine("-" + new string('-', 40));
            
            string result = ExecuteDynamicSQLQuery(testColumns[i], testOrders[i]);
            
            // Print only the SQL Query and Results parts (AI analysis was already streamed)
            string[] parts = result.Split(new string[] { "AI Analysis:\n" }, StringSplitOptions.None);
            if (parts.Length > 0)
            {
                Console.WriteLine("\n" + parts[0].Trim());
            }
            
            string[] resultParts = result.Split(new string[] { "Query Results:\n" }, StringSplitOptions.None);
            if (resultParts.Length > 1)
            {
                Console.WriteLine("\nQuery Results:");
                Console.WriteLine(resultParts[1]);
            }
            
            Console.WriteLine();
        }
        
        // Test with invalid column
        Console.WriteLine($"\n{'='*60}");
        Console.WriteLine("Test with invalid column name:");
        Console.WriteLine($"{'='*60}\n");
        string invalidResult = ExecuteDynamicSQLQuery("invalid_column");
        Console.WriteLine(invalidResult);
    }
}
