
```python
import pickle
import os
from datetime import datetime


class Player:
    def __init__(self, name: str, level: int = 1, health: int = 100, experience: int = 0):
        self.name = name
        self.level = level
        self.health = health
        self.experience = experience
        self.inventory = []
        self.skills = []
    
    def add_item(self, item: str):
        self.inventory.append(item)
    
    def add_skill(self, skill: str):
        self.skills.append(skill)
    
    def gain_experience(self, exp: int):
        self.experience += exp
        if self.experience >= self.level * 100:
            self.level_up()
    
    def level_up(self):
        self.level += 1
        self.health = min(100 + (self.level - 1) * 20, 200)
        print(f"🎉 {self.name} leveled up to level {self.level}!")
    
    def __repr__(self):
        return (f"Player(name={self.name}, level={self.level}, "
                f"health={self.health}, experience={self.experience})")


class GameWorld:
    def __init__(self, world_name: str):
        self.world_name = world_name
        self.current_location = "Starting Village"
        self.discovered_locations = ["Starting Village"]
        self.completed_quests = []
        self.active_quests = []
        self.world_events = []
    
    def discover_location(self, location: str):
        if location not in self.discovered_locations:
            self.discovered_locations.append(location)
            self.world_events.append(f"Discovered {location}")
    
    def complete_quest(self, quest: str):
        if quest in self.active_quests:
            self.active_quests.remove(quest)
        self.completed_quests.append(quest)
        self.world_events.append(f"Completed quest: {quest}")
    
    def start_quest(self, quest: str):
        if quest not in self.active_quests and quest not in self.completed_quests:
            self.active_quests.append(quest)
            self.world_events.append(f"Started quest: {quest}")
    
    def __repr__(self):
        return (f"GameWorld(name={self.world_name}, "
                f"location={self.current_location})")


class GameState:
    def __init__(self, player: Player, world: GameWorld):
        self.player = player
        self.world = world
        self.save_timestamp = None
        self.play_time_seconds = 0
        self.game_settings = {
            "difficulty": "normal",
            "sound_enabled": True,
            "music_volume": 0.7,
            "graphics_quality": "high"
        }
        self.statistics = {
            "enemies_defeated": 0,
            "items_collected": 0,
            "quests_completed": 0,
            "deaths": 0
        }
    
    def update_statistic(self, stat: str, value: int = 1):
        if stat in self.statistics:
            self.statistics[stat] += value
    
    def update_setting(self, setting: str, value):
        if setting in self.game_settings:
            self.game_settings[setting] = value
    
    def add_play_time(self, seconds: int):
        self.play_time_seconds += seconds
    
    def get_play_time_formatted(self) -> str:
        hours = self.play_time_seconds // 3600
        minutes = (self.play_time_seconds % 3600) // 60
        seconds = self.play_time_seconds % 60
        return f"{hours:02d}:{minutes:02d}:{seconds:02d}"
    
    def __repr__(self):
        return (f"GameState(player={self.player.name}, "
                f"world={self.world.world_name}, "
                f"play_time={self.get_play_time_formatted()})")


def save_game_state(game_state: GameState, filename: str = "game_save.pkl") -> bool:
    """Save game state to a file using pickle."""
    try:
        game_state.save_timestamp = datetime.now().isoformat()
        
        with open(filename, 'wb') as file:
            pickle.dump(game_state, file, protocol=pickle.HIGHEST_PROTOCOL)
        
        file_size = os.path.getsize(filename)
        print(f"✅ Game saved successfully!")
        print(f"   📁 File: {filename}")
        print(f"   📊 Size: {file_size} bytes")
        print(f"   🕐 Timestamp: {game_state.save_timestamp}")
        return True
        
    except (IOError, OSError, pickle.PicklingError) as e:
        print(f"❌ Error saving game: {e}")
        return False


def load_game_state(filename: str = "game_save.pkl") -> GameState | None:
    """Load game state from a file using pickle."""
    try:
        if not os.path.exists(filename):
            print(f"❌ Save file not found: {filename}")
            return None
        
        with open(filename, 'rb') as file:
            game_state = pickle.load(file)
        
        print(f"✅ Game loaded successfully!")
        print(f"   📁 File: {filename}")
        print(f"   🕐 Save timestamp: {game_state.save_timestamp}")
        return game_state
        
    except (IOError, OSError, pickle.UnpicklingError) as e:
        print(f"❌ Error loading game: {e}")
        return None


def display_game_state(game_state: GameState):
    """Display the current game state in a formatted way."""
    print("\n" + "="*50)
    print("🎮 GAME STATE SUMMARY")
    print("="*50)
    
    print("\n👤 PLAYER INFO:")
    print(f"   Name: {game_state.player.name}")
    print(f"   Level: {game_state.player.level}")
    print(f"   Health: {game_state.player.health}")
    print(f"   Experience: {game_state.player.experience}")
    print(f"   Inventory: {game_state.player.inventory}")
    print(f"   Skills: {game_state.player.skills}")
    
    print("\n🌍 WORLD INFO:")
    print(f"   World: {game_state.world.world_name}")
    print(f"   Current Location: {game_state.world.current_location}")
    print(f"   Discovered Locations: {game_state.world.discovered_locations}")
    print(f"   Active Quests: {game_state.world.active_quests}")
    print(f"   Completed Quests: {game_state.world.completed_quests}")
    
    print("\n📊 STATISTICS:")
    for stat, value in game_state.statistics.items():
        print(f"   {stat.replace('_', ' ').title()}: {value}")
    
    print("\n⚙️ SETTINGS:")
    for setting, value in game_state.game_settings.items():
        print(f"   {setting.replace('_', ' ').title()}: {value}")
    
    print(f"\n⏱️ Play Time: {game_state.get_play_time_formatted()}")
    print("="*50)


def create_sample_game_state() -> GameState:
    """Create a sample game state for demonstration."""
    player = Player("Hero", level=1, health=100, experience=0)
    player.add_item("Sword")
    player.add_item("Shield")
    player.add_item("Health Potion")
    player.add_skill("Slash")
    player.add_skill("Block")
    
    world = GameWorld("Fantasy Realm")
    world.discover_location("Dark Forest")
    world.discover_location("Ancient Castle")
    world.start_quest("Find the Lost Artifact")
    world.start_quest("Defeat the Dragon")
    world.complete_quest("Rescue the Villagers")
    world.current_location = "Dark Forest"
    
    game_state = GameState(player, world)
    game_state.add_play_time(3661)
    game_state.update_statistic("enemies_defeated", 15)
    game_state.update_statistic("items_collected", 8)
    game_state.update_statistic("quests_completed", 1)
    game_state.update_setting("difficulty", "hard")
    game_state.update_setting("music_volume", 0.5)
    
    player.gain_experience(150)
    
    return game_state


def main():
    """Main function to demonstrate saving and loading game state."""
    print("🎮 Game State Save/Load Demo with Pickle")
    print("="*50)
    
    print("\n📝 Creating initial game state...")
    game_state = create_sample_game_state()
    display_game_state(game_state)
    
    save_filename = "game_save.pkl"
    print(f"\n💾 Saving game state to '{save_filename}'...")
    save_success = save_game_state(game_state, save_filename)
    
    if save_success:
        print(f"\n📂 Loading game state from '{save_filename}'...")
        loaded_game_state = load_game_state(save_filename)
        
        if loaded_game_state:
            print("\n🔄 Loaded Game State:")
            display_game_state(loaded_game_state)
            
            print("\n✅ Verifying data integrity...")
            assert loaded_game_state.player.name == game_state.player.name
            assert loaded_game_state.player.level == game_state.player.level
            assert loaded_game_state.world.world_name == game_state.world.world_name
            assert loaded_game_state.statistics == game_state.statistics
            print("   All data verified successfully! ✓")
            
            print("\n🔄 Modifying loaded game state...")
            loaded_game_state.player.add_item("Magic Staff")
            loaded_game_state.player.gain_experience(200)
            loaded_game_state.world.discover_location("Hidden Temple")
            loaded_game_state.world.complete_quest("Find the Lost Artifact")
            loaded_game_state.add_play_time(1800)
            loaded_game_state.update_statistic("enemies_defeated", 5)
            
            updated_filename = "game_save_updated.pkl"
            print(f"\n💾 Saving updated game state to '{updated_filename}'...")
            save_game_state(loaded_game_state, updated_filename)
            
            print("\n📂 Loading updated game state...")
            final_game_state = load_game_state(updated_filename)
            
            if final_game_state:
                print("\n🎯 Final Game State:")
                display_game_state(final_game_state)
            
            print("\n🧹 Cleaning up save files...")
            for filename in [save_filename, updated_filename]:
                if os.path.exists(filename):
                    os.remove(filename)
                    print(f"   Removed: {filename}")
    
    print("\n✨ Demo completed successfully!")


if __name__ == "__main__":
    main()
```
