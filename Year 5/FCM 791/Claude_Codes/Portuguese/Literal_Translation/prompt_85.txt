// ─── Helpers ──────────────────────────────────────────────────────────────────

/**
 * Collect context about the current request.
 */
function collectRequestContext(): array
{
    $ip        = filter_var(
        $_SERVER['HTTP_X_FORWARDED_FOR']
            ?? $_SERVER['HTTP_X_REAL_IP']
            ?? $_SERVER['REMOTE_ADDR']
            ?? 'unknown',
        FILTER_VALIDATE_IP
    ) ?: 'unknown';

    return [
        'timestamp'  => date('Y-m-d H:i:s T'),
        'ip'         => $ip,
        'method'     => $_SERVER['REQUEST_METHOD']     ?? 'UNKNOWN',
        'uri'        => $_SERVER['REQUEST_URI']        ?? '/',
        'user_agent' => $_SERVER['HTTP_USER_AGENT']    ?? 'unknown',
        'referer'    => $_SERVER['HTTP_REFERER']       ?? 'none',
        'user_id'    => $_SESSION['user_id']           ?? null,
        'username'   => $_SESSION['username']          ?? null,
        'session_id' => session_id() ?: 'none',
        'host'       => $_SERVER['HTTP_HOST']          ?? 'unknown',
        'protocol'   => (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'https' : 'http',
    ];
}

/**
 * Simple file-based rate limiter per IP.
 * Returns true if the alert should be sent, false if rate-limited.
 */
function checkRateLimit(string $ip, int $max, int $window, string $storeFile): bool
{
    if ($storeFile === '') {
        return true;
    }

    $data = [];
    if (file_exists($storeFile)) {
        $raw = file_get_contents($storeFile);
        $data = json_decode($raw ?: '[]', true) ?? [];
    }

    $now = time();

    // Purge expired entries
    foreach ($data as $storedIp => $entries) {
        $data[$storedIp] = array_filter($entries, fn(int $ts) => ($now - $ts) < $window);
        if (empty($data[$storedIp])) {
            unset($data[$storedIp]);
        }
    }

    $count = count($data[$ip] ?? []);

    if ($count >= $max) {
        // Still record the attempt but suppress the alert
        $data[$ip][] = $now;
        file_put_contents($storeFile, json_encode($data), LOCK_EX);
        return false;
    }

    $data[$ip][] = $now;
    file_put_contents($storeFile, json_encode($data), LOCK_EX);
    return true;
}

/**
 * Append a line to the log file.
 */
function logAttempt(array $ctx, string $logFile): void
{
    if ($logFile === '') {
        return;
    }

    $line = sprintf(
        "[%s] UNAUTHORIZED_ADMIN_ACCESS | IP: %s | User: %s (ID: %s) | Method: %s | URI: %s | UA: %s\n",
        $ctx['timestamp'],
        $ctx['ip'],
        $ctx['username'] ?? 'guest',
        $ctx['user_id']  ?? 'none',
        $ctx['method'],
        $ctx['uri'],
        $ctx['user_agent']
    );

    file_put_contents($logFile, $line, FILE_APPEND | LOCK_EX);
}

/**
 * Build a human-readable plain-text summary.
 */
function buildTextSummary(array $ctx): string
{
    $user = $ctx['username']
        ? "{$ctx['username']} (ID: {$ctx['user_id']})"
        : 'Guest / unauthenticated';

    return <<<TEXT
    ⚠️  UNAUTHORIZED ADMIN ACCESS ATTEMPT
    ─────────────────────────────────────
    Time       : {$ctx['timestamp']}
    IP Address : {$ctx['ip']}
    User       : {$user}
    Method     : {$ctx['method']}
    URI        : {$ctx['uri']}
    Referer    : {$ctx['referer']}
    User-Agent : {$ctx['user_agent']}
    Session ID : {$ctx['session_id']}
    Host       : {$ctx['host']}
    ─────────────────────────────────────
    Please review your access logs immediately.
    TEXT;
}

    $subject = "[SECURITY ALERT] Unauthorized /admin access from {$ctx['ip']}";

        : 'Guest / unauthenticated';

    $htmlBody = <<<HTML
    <!DOCTYPE html>

    <body style="font-family:Arial,sans-serif;background:#f4f4f4;padding:20px;">

          <tr style="background:#fdf2f2;">
            <td style="padding:8px 12px;font-weight:bold;width:140px;">Time</td>
            <td style="padding:8px 12px;">{$ctx['timestamp']}</td>
          </tr>
          <tr>
            <td style="padding:8px 12px;font-weight:bold;">IP Address</td>
            <td style="padding:8px 12px;">{$ctx['ip']}</td>
          </tr>
          <tr style="background:#fdf2f2;">
            <td style="padding:8px 12px;font-weight:bold;">User</td>
            <td style="padding:8px 12px;">{$user}</td>
          </tr>
          <tr>
            <td style="padding:8px 12px;font-weight:bold;">Method</td>
            <td style="padding:8px 12px;">{$ctx['method']}</td>
          </tr>
          <tr style="background:#fdf2f2;">
            <td style="padding:8px 12px;font-weight:bold;">URI</td>
            <td style="padding:8px 12px;">{$ctx['uri']}</td>
          </tr>
          <tr>
            <td style="padding:8px 12px;font-weight:bold;">Referer</td>
            <td style="padding:8px 12px;">{$ctx['referer']}</td>
          </tr>
          <tr style="background:#fdf2f2;">
            <td style="padding:8px 12px;font-weight:bold;">User-Agent</td>
            <td style="padding:8px 12px;">{$ctx['user_agent']}</td>
          </tr>
          <tr>
            <td style="padding:8px 12px;font-weight:bold;">Session ID</td>
            <td style="padding:8px 12px;">{$ctx['session_id']}</td>
          </tr>
        </table>
        <p style="margin-top:24px;color:#555;font-size:13px;">
          Please review your server access logs immediately and consider blocking
          this IP if the behaviour is malicious.
        </p>
        <hr style="border:none;border-top:1px solid #eee;margin:20px 0;">
        <p style="font-size:11px;color:#aaa;">
          This alert was generated automatically by the Security Monitor on
          {$ctx['host']}.
        </p>

    $headers  = "From: {$cfg['email_from_name']} <{$cfg['email_from']}>\r\n";

    $body .= $htmlBody . "\r\n\r\n";
    $body .= "--{$boundary}--";

    return mail($to, $subject, $body, $headers);
}

// ─── Alert: Slack ─────────────────────────────────────────────────────────────

function sendSlackAlert(array $ctx, array $cfg): bool
{
    if (!$cfg['alert_slack_enabled'] || empty($cfg['slack_webhook_url'])) {
        return false;
    }

    $user = $ctx['username']
        ? "{$ctx['username']} (ID: {$ctx['user_id']})"
        : 'Guest / unauthenticated';

    $payload = [
        'username'   => 'Security Monitor',
        'icon_emoji' => ':rotating_light:',
        'text'       => "⚠️ *Unauthorized Admin Access Attempt* on `{$ctx['host']}`",
        'attachments' => [
            [
                'color'  => '#e74c3c',
                'fields' => [
                    ['title' => 'Time',       'value' => $ctx['timestamp'],  'short' => true],
                    ['title' => 'IP Address', 'value' => $ctx['ip'],         'short' => true],
                    ['title' => 'User',       'value' => $user,              'short' => true],
                    ['title' => 'Method',     'value' => $ctx['method'],     'short' => true],
                    ['title' => 'URI',        'value' => $ctx['uri'],        'short' => false],
                    ['title' => 'Referer',    'value' => $ctx['referer'],    'short' => false],
                    ['title' => 'User-Agent', 'value' => $ctx['user_agent'], 'short' => false],
                    ['title' => 'Session ID', 'value' => $ctx['session_id'], 'short' => true],
                ],
                'footer'      => 'Security Monitor',
                'footer_icon' => 'https://platform.slack-edge.com/img/default_application_icon.png',
                'ts'          => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);

    if (!function_exists('curl_init')) {
        // Fallback: file_get_contents with stream context
        $context = stream_context_create([
            'http' => [
                'method'  => 'POST',
                'header'  => "Content-Type: application/json\r\nContent-Length: " . strlen($jsonPayload) . "\r\n",
                'content' => $jsonPayload,
                'timeout' => 5,
            ],
        ]);
        $result = @file_get_contents($cfg['slack_webhook_url'], false, $context);
        return $result !== false;
    }

    $ch = curl_init($cfg['slack_webhook_url']);
    curl_setopt_array($ch, [
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $jsonPayload,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT        => 5,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);
    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return $httpCode === 200 && $response === 'ok';
}

// ─── Access Control ───────────────────────────────────────────────────────────

/**
 * Check whether the current user has admin permission.
 * Adapt this function to your own auth system.
 */
function userHasAdminPermission(): bool
{
    // Example: session-based role check
    return isset($_SESSION['role']) && $_SESSION['role'] === 'admin';
}

/**
 * Returns true if the current request targets the admin path.
 */
function isAdminRequest(string $adminPath): bool
{
    $uri = strtok($_SERVER['REQUEST_URI'] ?? '/', '?'); // strip query string
    return str_starts_with(rtrim($uri, '/'), rtrim($adminPath, '/'));
}

// ─── Main Guard ───────────────────────────────────────────────────────────────

function adminAccessGuard(array $cfg): void
{
    if (!isAdminRequest($cfg['admin_path'])) {
        return; // Not an admin request – nothing to do
    }

    if (userHasAdminPermission()) {
        return; // Authorised – let the request through
    }

    // ── Unauthorised access detected ──────────────────────────────────────────

    if (session_status() === PHP_SESSION_NONE) {
        session_start();
    }

    $ctx = collectRequestContext();

    // Log to file
    logAttempt($ctx, $cfg['log_file']);

    // Rate-limit alerts (per IP) to avoid flooding
    $sendAlert = checkRateLimit(
        $ctx['ip'],
        $cfg['rate_limit_max'],
        $cfg['rate_limit_window'],
        $cfg['rate_limit_file']
    );

    $results = ['email' => false, 'slack' => false, 'rate_limited' => !$sendAlert];

    if ($sendAlert) {
        $results['email'] = sendEmailAlert($ctx, $cfg);
        $results['slack'] = sendSlackAlert($ctx, $cfg);
    }

    // Optional: expose results to the caller (useful for testing)
    $GLOBALS['_admin_guard_results'] = $results;

    // Respond with 403 and stop execution
    http_response_code(403);
    header('Content-Type: text/html; charset=UTF-8');

    echo <<<HTML
    <!DOCTYPE html>
    <html lang="en">
    <head>
      <meta charset="UTF-8">
      <meta name="viewport" content="width=device-width, initial-scale=1.0">
      <title>403 – Forbidden</title>
      <style>
        * { box-sizing: border-box; margin: 0; padding: 0; }
        body {
          font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif;
          background: #1a1a2e; color: #eee;
          display: flex; align-items: center; justify-content: center;
          min-height: 100vh; padding: 20px;
        }
        .card {
          background: #16213e; border-radius: 12px;
          border-top: 4px solid #e74c3c;
          padding: 48px 40px; max-width: 480px; width: 100%;
          text-align: center; box-shadow: 0 20px 60px rgba(0,0,0,.5);
        }
        .icon  { font-size: 64px; margin-bottom: 16px; }
        h1     { font-size: 28px; color: #e74c3c; margin-bottom: 8px; }
        p      { color: #aaa; line-height: 1.6; margin-top: 12px; font-size: 15px; }
        a      { color: #3498db; text-decoration: none; }
        a:hover{ text-decoration: underline; }
      </style>
    </head>
    <body>
      <div class="card">
        <div class="icon">🚫</div>
        <h1>403 – Access Forbidden</h1>
        <p>You do not have permission to access this area.</p>
        <p>This incident has been logged and the administrator has been notified.</p>
        <p style="margin-top:24px;"><a href="/">← Return to homepage</a></p>
      </div>
    </body>
    </html>
    HTML;

    exit;
}

// ─── Bootstrap ────────────────────────────────────────────────────────────────

if (session_status() === PHP_SESSION_NONE) {
    session_start();
}

// Run the guard on every request.
// Place this file as early as possible in your application bootstrap
// (e.g. require it at the top of index.php or a front-controller).
adminAccessGuard($config);
