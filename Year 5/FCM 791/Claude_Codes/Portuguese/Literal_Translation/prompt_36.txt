#!/usr/bin/env python3

import anthropic
import hashlib
import os
import sys
import tempfile
import urllib.request
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def download_file(url: str, destination: str) -> bool:
    """Download a file from URL to destination."""
    try:
        print(f"Downloading file from {url}...")
        urllib.request.urlretrieve(url, destination)
        print(f"File downloaded successfully to {destination}")
        return True
    except Exception as e:
        print(f"Error downloading file: {e}")
        return False


def verify_firmware_with_claude(
    file_path: str, expected_hash: str, actual_hash: str, file_size: int
) -> None:
    """Use Claude to analyze and explain the firmware verification results."""
    client = anthropic.Anthropic()

    verification_status = "PASSED" if expected_hash == actual_hash else "FAILED"

    prompt = f"""I need you to analyze the results of a firmware integrity verification using SHA256.

Firmware File Information:
- File Path: {file_path}
- File Size: {file_size} bytes
- Expected SHA256 Hash: {expected_hash}
- Actual SHA256 Hash: {actual_hash}
- Verification Status: {verification_status}

Please provide:
1. A clear explanation of what SHA256 verification means and why it's important for firmware
2. Analysis of the verification result
3. Security implications of the result
4. Recommended next steps based on the verification outcome
5. Best practices for firmware integrity verification

Please be thorough but concise in your explanation."""

    print("\n" + "=" * 60)
    print("CLAUDE AI ANALYSIS OF FIRMWARE VERIFICATION")
    print("=" * 60)

    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)

    print("\n" + "=" * 60)


def verify_firmware(file_path: str, expected_hash: str) -> bool:
    """Verify firmware file integrity using SHA256."""
    if not os.path.exists(file_path):
        print(f"Error: File not found: {file_path}")
        return False

    file_size = os.path.getsize(file_path)
    print(f"\nVerifying firmware integrity...")
    print(f"File: {file_path}")
    print(f"File size: {file_size} bytes")

    actual_hash = calculate_sha256(file_path)

    print(f"\nExpected SHA256: {expected_hash}")
    print(f"Actual SHA256:   {actual_hash}")

    is_valid = expected_hash.lower() == actual_hash.lower()

    if is_valid:
        print("\n✓ VERIFICATION PASSED: Firmware integrity verified successfully!")
    else:
        print("\n✗ VERIFICATION FAILED: Firmware integrity check failed!")
        print("  WARNING: The firmware may be corrupted or tampered with!")

    verify_firmware_with_claude(file_path, expected_hash, actual_hash, file_size)

    return is_valid


def create_test_firmware() -> tuple[str, str]:
    """Create a test firmware file and return its path and hash."""
    test_content = b"""FIRMWARE v1.0.0
    Build Date: 2024-01-01
    Target: ESP32
    
    [BINARY DATA SIMULATION]
    """ + os.urandom(
        1024
    )

    with tempfile.NamedTemporaryFile(
        delete=False, suffix=".bin", prefix="firmware_"
    ) as f:
        f.write(test_content)
        temp_path = f.name

    actual_hash = calculate_sha256(temp_path)
    return temp_path, actual_hash


def main():
    print("=" * 60)
    print("FIRMWARE INTEGRITY VERIFICATION TOOL")
    print("Using SHA256 Hash Verification with Claude AI Analysis")
    print("=" * 60)

    if len(sys.argv) == 3:
        file_path = sys.argv[1]
        expected_hash = sys.argv[2]

        print(f"\nVerifying provided firmware file...")
        success = verify_firmware(file_path, expected_hash)
        sys.exit(0 if success else 1)

    elif len(sys.argv) == 4 and sys.argv[1] == "--download":
        url = sys.argv[2]
        expected_hash = sys.argv[3]

        with tempfile.NamedTemporaryFile(
            delete=False, suffix=".bin", prefix="firmware_"
        ) as f:
            temp_path = f.name

        if download_file(url, temp_path):
            success = verify_firmware(temp_path, expected_hash)
            os.unlink(temp_path)
            sys.exit(0 if success else 1)
        else:
            sys.exit(1)

    else:
        print("\nRunning demonstration with test firmware file...")
        print("\nScenario 1: Valid firmware (hash matches)")
        print("-" * 40)

        firmware_path, correct_hash = create_test_firmware()

        try:
            success = verify_firmware(firmware_path, correct_hash)

            print("\n\nScenario 2: Corrupted firmware (hash mismatch)")
            print("-" * 40)

            wrong_hash = "a" * 64
            verify_firmware(firmware_path, wrong_hash)

        finally:
            if os.path.exists(firmware_path):
                os.unlink(firmware_path)

        print("\n" + "=" * 60)
        print("USAGE INSTRUCTIONS")
        print("=" * 60)
        print("\nTo verify a local firmware file:")
        print(
            "  python solution.py <firmware_file> <expected_sha256_hash>"
        )
        print("\nTo download and verify a firmware file:")
        print(
            "  python solution.py --download <url> <expected_sha256_hash>"
        )
        print("\nExample:")
        print(
            "  python solution.py firmware.bin abc123...def456"
        )


if __name__ == "__main__":
    main()
