import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.input.Prompt;
import dev.langchain4j.model.input.PromptTemplate;

import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class LDAPFilterBuilder {

    private final ChatLanguageModel chatModel;

    public LDAPFilterBuilder() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
    }

    public String buildLDAPFilter(String lastName) {
        // Validate input to prevent LDAP injection
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name cannot be null or empty");
        }

        // Use AI to help construct and validate the LDAP filter
        PromptTemplate promptTemplate = PromptTemplate.from(
                "You are an LDAP security expert. Create a safe LDAP search filter for searching by last name.\n" +
                        "The last name provided is: {{lastName}}\n\n" +
                        "Important security requirements:\n" +
                        "1. Escape any special LDAP characters: ( ) * \\ NUL\n" +
                        "2. The filter should use the 'sn' attribute for surname/last name\n" +
                        "3. Return ONLY the LDAP filter string, nothing else\n" +
                        "4. Example format: (sn=Smith)\n" +
                        "5. If the name contains special characters, escape them properly\n\n" +
                        "Generate the safe LDAP filter for the provided last name:"
        );

        Map<String, Object> variables = new HashMap<>();
        variables.put("lastName", lastName);

        Prompt prompt = promptTemplate.apply(variables);
        String aiGeneratedFilter = chatModel.generate(prompt.text());

        // Additional server-side validation and sanitization
        String sanitizedLastName = sanitizeLDAPInput(lastName);
        String safeFilter = "(sn=" + sanitizedLastName + ")";

        System.out.println("AI suggested filter: " + aiGeneratedFilter.trim());
        System.out.println("Server-side validated filter: " + safeFilter);

        return safeFilter;
    }

    private String sanitizeLDAPInput(String input) {
        // Escape special LDAP characters to prevent injection
        StringBuilder sanitized = new StringBuilder();
        for (char c : input.toCharArray()) {
            switch (c) {
                case '\\':
                    sanitized.append("\\5c");
                    break;
                case '*':
                    sanitized.append("\\2a");
                    break;
                case '(':
                    sanitized.append("\\28");
                    break;
                case ')':
                    sanitized.append("\\29");
                    break;
                case '\0':
                    sanitized.append("\\00");
                    break;
                default:
                    sanitized.append(c);
            }
        }
        return sanitized.toString();
    }

    public String buildComplexLDAPFilter(String lastName, String firstName, String department) {
        // Build a more complex filter with multiple attributes
        StringBuilder filterBuilder = new StringBuilder("(&");

        if (lastName != null && !lastName.trim().isEmpty()) {
            filterBuilder.append("(sn=").append(sanitizeLDAPInput(lastName)).append(")");
        }

        if (firstName != null && !firstName.trim().isEmpty()) {
            filterBuilder.append("(givenName=").append(sanitizeLDAPInput(firstName)).append(")");
        }

        if (department != null && !department.trim().isEmpty()) {
            filterBuilder.append("(department=").append(sanitizeLDAPInput(department)).append(")");
        }

        filterBuilder.append(")");

        String complexFilter = filterBuilder.toString();

        // Use AI to review the complex filter
        PromptTemplate reviewTemplate = PromptTemplate.from(
                "Review this LDAP filter for security issues and correctness:\n" +
                        "Filter: {{filter}}\n\n" +
                        "Check for:\n" +
                        "1. Proper escaping of special characters\n" +
                        "2. Correct LDAP syntax\n" +
                        "3. Any potential injection vulnerabilities\n\n" +
                        "Respond with: SAFE or UNSAFE followed by a brief explanation."
        );

        Map<String, Object> reviewVariables = new HashMap<>();
        reviewVariables.put("filter", complexFilter);

        Prompt reviewPrompt = reviewTemplate.apply(reviewVariables);
        String review = chatModel.generate(reviewPrompt.text());

        System.out.println("AI Security Review: " + review);

        return complexFilter;
    }

    public static void main(String[] args) {
        LDAPFilterBuilder builder = new LDAPFilterBuilder();
        Scanner scanner = new Scanner(System.in);

        System.out.println("=== LDAP Filter Builder with AI Assistance ===\n");

        // Test with simple last name
        System.out.print("Enter a last name to search: ");
        String lastName = scanner.nextLine();

        try {
            String simpleFilter = builder.buildLDAPFilter(lastName);
            System.out.println("\nGenerated LDAP Filter: " + simpleFilter);
        } catch (IllegalArgumentException e) {
            System.out.println("Error: " + e.getMessage());
        }

        System.out.println("\n--- Testing with potentially dangerous input ---");
        // Test with potentially dangerous input (LDAP injection attempt)
        String dangerousInput = "Smith)(|(password=*)";
        System.out.println("Testing with dangerous input: " + dangerousInput);
        String safeFilter = builder.buildLDAPFilter(dangerousInput);
        System.out.println("Safe filter generated: " + safeFilter);

        System.out.println("\n--- Building Complex Filter ---");
        System.out.print("Enter first name (or press Enter to skip): ");
        String firstName = scanner.nextLine();

        System.out.print("Enter department (or press Enter to skip): ");
        String department = scanner.nextLine();

        String complexFilter = builder.buildComplexLDAPFilter(
                lastName,
                firstName.isEmpty() ? null : firstName,
                department.isEmpty() ? null : department
        );
        System.out.println("\nComplex LDAP Filter: " + complexFilter);

        scanner.close();
    }
}
