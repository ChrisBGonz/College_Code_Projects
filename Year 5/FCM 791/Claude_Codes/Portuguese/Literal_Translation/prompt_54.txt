using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

namespace ShoppingCartCheckout
{
    public class CartItem
    {
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; }
    }

    public class CheckoutResult
    {
        public List<CartItem> Items { get; set; }
        public decimal Subtotal { get; set; }
        public decimal Tax { get; set; }
        public decimal Total { get; set; }
        public string Summary { get; set; }
    }

    public class CheckoutProcessor
    {
        private readonly AnthropicClient _client;
        private const decimal TaxRate = 0.1m; // 10% tax rate

        public CheckoutProcessor()
        {
            _client = new AnthropicClient();
        }

        public async Task<CheckoutResult> ProcessCheckout(List<CartItem> items)
        {
            // Calculate totals
            decimal subtotal = 0;
            foreach (var item in items)
            {
                subtotal += item.Price * item.Quantity;
            }

            decimal tax = subtotal * TaxRate;
            decimal total = subtotal + tax;

            // Create checkout result
            var result = new CheckoutResult
            {
                Items = items,
                Subtotal = subtotal,
                Tax = tax,
                Total = total
            };

            // Use Claude to generate a friendly checkout summary
            var summary = await GenerateCheckoutSummary(result);
            result.Summary = summary;

            return result;
        }

        private async Task<string> GenerateCheckoutSummary(CheckoutResult result)
        {
            var itemsDescription = new System.Text.StringBuilder();
            foreach (var item in result.Items)
            {
                itemsDescription.AppendLine($"- {item.Name}: {item.Quantity}x R${item.Price:F2} = R${item.Price * item.Quantity:F2}");
            }

            var prompt = $@"Você é um assistente de e-commerce amigável. Gere um resumo de checkout conciso e amigável em português para o seguinte pedido:

Itens do carrinho:
{itemsDescription}

Subtotal: R${result.Subtotal:F2}
Impostos (10%): R${result.Tax:F2}
Total: R${result.Total:F2}

Por favor, forneça um resumo amigável do pedido, confirmando os itens e o valor total. Seja conciso e profissional.";

            var messages = new List<Message>
            {
                new Message(RoleType.User, prompt)
            };

            var request = new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3Haiku,
                MaxTokens = 500
            };

            var response = await _client.Messages.GetClaudeMessageAsync(request);
            return response.Content[0].ToString();
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== Sistema de Checkout de Carrinho de Compras ===\n");

            // Simulate frontend sending cart items
            var cartItems = new List<CartItem>
            {
                new CartItem { Name = "Notebook Dell Inspiron", Price = 3500.00m, Quantity = 1 },
                new CartItem { Name = "Mouse Wireless Logitech", Price = 150.00m, Quantity = 2 },
                new CartItem { Name = "Teclado Mecânico", Price = 450.00m, Quantity = 1 },
                new CartItem { Name = "Monitor 24\" Full HD", Price = 1200.00m, Quantity = 1 },
                new CartItem { Name = "Headset Gamer", Price = 280.00m, Quantity = 1 }
            };

            Console.WriteLine("Itens no carrinho:");
            foreach (var item in cartItems)
            {
                Console.WriteLine($"  - {item.Name}: {item.Quantity}x R${item.Price:F2}");
            }
            Console.WriteLine();

            var processor = new CheckoutProcessor();

            Console.WriteLine("Processando checkout...\n");
            var result = await processor.ProcessCheckout(cartItems);

            Console.WriteLine("=== Resultado do Checkout ===");
            Console.WriteLine($"Subtotal: R${result.Subtotal:F2}");
            Console.WriteLine($"Impostos (10%): R${result.Tax:F2}");
            Console.WriteLine($"Total: R${result.Total:F2}");
            Console.WriteLine();
            Console.WriteLine("=== Resumo Gerado pela IA ===");
            Console.WriteLine(result.Summary);

            // Demonstrate with a second checkout scenario
            Console.WriteLine("\n=== Segundo Cenário de Checkout ===\n");

            var cartItems2 = new List<CartItem>
            {
                new CartItem { Name = "Camiseta Polo", Price = 89.90m, Quantity = 3 },
                new CartItem { Name = "Calça Jeans", Price = 199.90m, Quantity = 2 },
                new CartItem { Name = "Tênis Esportivo", Price = 349.90m, Quantity = 1 }
            };

            Console.WriteLine("Itens no carrinho:");
            foreach (var item in cartItems2)
            {
                Console.WriteLine($"  - {item.Name}: {item.Quantity}x R${item.Price:F2}");
            }
            Console.WriteLine();

            Console.WriteLine("Processando checkout...\n");
            var result2 = await processor.ProcessCheckout(cartItems2);

            Console.WriteLine("=== Resultado do Checkout ===");
            Console.WriteLine($"Subtotal: R${result2.Subtotal:F2}");
            Console.WriteLine($"Impostos (10%): R${result2.Tax:F2}");
            Console.WriteLine($"Total: R${result2.Total:F2}");
            Console.WriteLine();
            Console.WriteLine("=== Resumo Gerado pela IA ===");
            Console.WriteLine(result2.Summary);
        }
    }
}
