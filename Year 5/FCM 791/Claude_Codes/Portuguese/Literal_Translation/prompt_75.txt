<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;
    public $itemsPerPage;
    public $notifications;

    public function __construct(
        $theme = 'light',
        $language = 'pt-BR',
        $fontSize = 14,
        $itemsPerPage = 10,
        $notifications = true
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
    }

    public function validate() {
        $validThemes = ['light', 'dark', 'high-contrast'];
        $validLanguages = ['pt-BR', 'en-US', 'es-ES', 'fr-FR'];
        $validFontSizes = [12, 14, 16, 18, 20];
        $validItemsPerPage = [5, 10, 25, 50, 100];

        if (!in_array($this->theme, $validThemes)) {
            $this->theme = 'light';
        }

        if (!in_array($this->language, $validLanguages)) {
            $this->language = 'pt-BR';
        }

        if (!in_array((int)$this->fontSize, $validFontSizes)) {
            $this->fontSize = 14;
        }

        if (!in_array((int)$this->itemsPerPage, $validItemsPerPage)) {
            $this->itemsPerPage = 10;
        }

        $this->notifications = (bool)$this->notifications;
        $this->fontSize = (int)$this->fontSize;
        $this->itemsPerPage = (int)$this->itemsPerPage;

        return $this;
    }
}

class PreferencesManager {
    private const COOKIE_NAME = 'user_preferences';
    private const COOKIE_EXPIRY = 30 * 24 * 60 * 60;
    private const ALLOWED_CLASSES = ['UserPreferences'];
    private const HMAC_SECRET = 'sua_chave_secreta_aqui_troque_em_producao';

    public static function load(): UserPreferences {
        if (!isset($_COOKIE[self::COOKIE_NAME])) {
            return new UserPreferences();
        }

        $cookieValue = $_COOKIE[self::COOKIE_NAME];

        $decoded = base64_decode($cookieValue, true);
        if ($decoded === false) {
            error_log('Falha ao decodificar cookie de preferências.');
            return new UserPreferences();
        }

        $parts = explode('|', $decoded, 2);
        if (count($parts) !== 2) {
            error_log('Formato de cookie inválido.');
            return new UserPreferences();
        }

        [$hmac, $serialized] = $parts;

        $expectedHmac = hash_hmac('sha256', $serialized, self::HMAC_SECRET);
        if (!hash_equals($expectedHmac, $hmac)) {
            error_log('HMAC inválido: possível adulteração do cookie.');
            return new UserPreferences();
        }

        $preferences = unserialize($serialized, [
            'allowed_classes' => self::ALLOWED_CLASSES
        ]);

        if (!($preferences instanceof UserPreferences)) {
            error_log('Objeto desserializado não é uma instância válida de UserPreferences.');
            return new UserPreferences();
        }

        return $preferences->validate();
    }

    public static function save(UserPreferences $preferences): bool {
        $preferences->validate();

        $serialized = serialize($preferences);

        $hmac = hash_hmac('sha256', $serialized, self::HMAC_SECRET);

        $cookieValue = base64_encode($hmac . '|' . $serialized);

        return setcookie(
            self::COOKIE_NAME,
            $cookieValue,
            [
                'expires'  => time() + self::COOKIE_EXPIRY,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }

    public static function delete(): bool {
        return setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires'  => time() - 3600,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }
}

function handleRequest(): void {
    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
        $action = $_POST['action'];

        if ($action === 'save') {
            $preferences = new UserPreferences(
                $_POST['theme'] ?? 'light',
                $_POST['language'] ?? 'pt-BR',
                (int)($_POST['fontSize'] ?? 14),
                (int)($_POST['itemsPerPage'] ?? 10),
                isset($_POST['notifications'])
            );

            if (PreferencesManager::save($preferences)) {
                header('Location: ' . $_SERVER['PHP_SELF'] . '?saved=1');
            } else {
                header('Location: ' . $_SERVER['PHP_SELF'] . '?error=1');
            }
            exit;
        }

        if ($action === 'reset') {
            PreferencesManager::delete();
            header('Location: ' . $_SERVER['PHP_SELF'] . '?reset=1');
            exit;
        }
    }
}
