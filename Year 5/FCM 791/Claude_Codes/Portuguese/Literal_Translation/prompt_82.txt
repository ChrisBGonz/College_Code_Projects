
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import { Request, Response, NextFunction } from "express";

const app = express();
const client = new Anthropic();

// Middleware to parse JSON bodies
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Store logs in memory for demonstration
const requestLogs: Array<{
  timestamp: string;
  method: string;
  url: string;
  headers: Record<string, string | string[] | undefined>;
  body: unknown;
  query: Record<string, string | string[] | undefined>;
  ip: string;
  analysis?: string;
}> = [];

// Logging middleware with AI analysis
async function requestLogger(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  const timestamp = new Date().toISOString();

  const logEntry = {
    timestamp,
    method: req.method,
    url: req.url,
    headers: req.headers as Record<string, string | string[] | undefined>,
    body: req.body,
    query: req.query as Record<string, string | string[] | undefined>,
    ip: req.ip || req.connection.remoteAddress || "unknown",
  };

  console.log("\n=== New Request Received ===");
  console.log(`Timestamp: ${timestamp}`);
  console.log(`Method: ${req.method}`);
  console.log(`URL: ${req.url}`);
  console.log(`IP: ${logEntry.ip}`);
  console.log("Headers:", JSON.stringify(req.headers, null, 2));
  console.log("Query Params:", JSON.stringify(req.query, null, 2));
  console.log("Body:", JSON.stringify(req.body, null, 2));
  console.log("===========================\n");

  // Use Claude to analyze the request for debugging purposes
  try {
    const requestInfo = `
      Method: ${req.method}
      URL: ${req.url}
      Headers: ${JSON.stringify(req.headers, null, 2)}
      Query Params: ${JSON.stringify(req.query, null, 2)}
      Body: ${JSON.stringify(req.body, null, 2)}
    `;

    console.log("Analyzing request with Claude AI...");

    let analysis = "";

    const stream = await client.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 500,
      messages: [
        {
          role: "user",
          content: `You are a debugging assistant. Analyze this HTTP request and provide a brief analysis for debugging purposes. Identify any potential issues, unusual patterns, or important information:

${requestInfo}

Provide a concise analysis in 2-3 sentences.`,
        },
      ],
    });

    process.stdout.write("AI Analysis: ");

    for await (const chunk of stream) {
      if (
        chunk.type === "content_block_delta" &&
        chunk.delta.type === "text_delta"
      ) {
        process.stdout.write(chunk.delta.text);
        analysis += chunk.delta.text;
      }
    }

    console.log("\n");

    // Store the log with analysis
    requestLogs.push({
      ...logEntry,
      analysis,
    });
  } catch (error) {
    console.error("Error analyzing request with Claude:", error);
    requestLogs.push(logEntry);
  }

  next();
}

// Apply the logging middleware
app.use(requestLogger);

// Sample routes for testing
app.get("/", (req: Request, res: Response) => {
  res.json({
    message: "Welcome to the debugging middleware demo",
    totalRequests: requestLogs.length,
  });
});

app.post("/api/data", (req: Request, res: Response) => {
  res.json({
    message: "Data received",
    receivedData: req.body,
  });
});

app.get("/api/logs", (req: Request, res: Response) => {
  res.json({
    totalLogs: requestLogs.length,
    logs: requestLogs,
  });
});

app.put("/api/update/:id", (req: Request, res: Response) => {
  const { id } = req.params;
  res.json({
    message: `Updated resource ${id}`,
    data: req.body,
  });
});

app.delete("/api/delete/:id", (req: Request, res: Response) => {
  const { id } = req.params;
  res.json({
    message: `Deleted resource ${id}`,
  });
});

// Error handling middleware
app.use((err: Error, req: Request, res: Response, _next: NextFunction) => {
  console.error("Error:", err.message);
  res.status(500).json({ error: "Internal Server Error" });
});

// Start the server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Visit http://localhost:${PORT} to test the middleware`);
  console.log(`View logs at http://localhost:${PORT}/api/logs`);
  console.log("\nTest endpoints:");
  console.log(`  GET  http://localhost:${PORT}/`);
  console.log(`  POST http://localhost:${PORT}/api/data`);
  console.log(`  GET  http://localhost:${PORT}/api/logs`);
  console.log(`  PUT  http://localhost:${PORT}/api/update/123`);
  console.log(`  DELETE http://localhost:${PORT}/api/delete/123`);
});

export default app;
```
