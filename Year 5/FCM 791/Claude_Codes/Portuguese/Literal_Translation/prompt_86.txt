package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
)

type BankTransfer struct {
	SenderID   string  `json:"sender_id"`
	ReceiverID string  `json:"receiver_id"`
	Amount     float64 `json:"amount"`
}

func processBankTransfer(client *anthropic.Client, transfer BankTransfer) (string, error) {
	tools := []anthropic.ToolParam{
		{
			Name:        "record_bank_transfer",
			Description: anthropic.String("Records a bank transfer with sender ID, receiver ID, and amount"),
			InputSchema: anthropic.ToolInputSchemaParam{
				Type: "object",
				Properties: map[string]interface{}{
					"sender_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the sender",
					},
					"receiver_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the receiver",
					},
					"amount": map[string]interface{}{
						"type":        "number",
						"description": "The amount to transfer",
					},
				},
				Required: []string{"sender_id", "receiver_id", "amount"},
			},
		},
	}

	message := fmt.Sprintf("Process a bank transfer from sender %s to receiver %s for amount %.2f",
		transfer.SenderID, transfer.ReceiverID, transfer.Amount)

	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(message)),
		},
	})
	if err != nil {
		return "", fmt.Errorf("error creating message: %w", err)
	}

	var result string
	for _, block := range response.Content {
		switch b := block.AsUnion().(type) {
		case anthropic.TextBlock:
			result += b.Text
		case anthropic.ToolUseBlock:
			if b.Name == "record_bank_transfer" {
				input := b.Input
				result += fmt.Sprintf("Bank transfer recorded successfully!\n")
				result += fmt.Sprintf("Sender ID: %v\n", input["sender_id"])
				result += fmt.Sprintf("Receiver ID: %v\n", input["receiver_id"])
				result += fmt.Sprintf("Amount: %.2f\n", input["amount"])
			}
		}
	}

	if result == "" {
		result = "Transfer processed but no confirmation received"
	}

	return result, nil
}

func main() {
	client := anthropic.NewClient()

	transfers := []BankTransfer{
		{
			SenderID:   "USER_001",
			ReceiverID: "USER_002",
			Amount:     1500.00,
		},
		{
			SenderID:   "ACCOUNT_ABC",
			ReceiverID: "ACCOUNT_XYZ",
			Amount:     250.75,
		},
		{
			SenderID:   "CORP_123",
			ReceiverID: "VENDOR_456",
			Amount:     10000.00,
		},
	}

	fmt.Println("=== Bank Transfer Processing System ===\n")

	for i, transfer := range transfers {
		fmt.Printf("Processing Transfer %d:\n", i+1)
		fmt.Printf("From: %s -> To: %s | Amount: $%.2f\n",
			transfer.SenderID, transfer.ReceiverID, transfer.Amount)

		result, err := processBankTransfer(client, transfer)
		if err != nil {
			log.Printf("Error processing transfer %d: %v\n", i+1, err)
			continue
		}

		fmt.Println("Result:")
		fmt.Println(result)
		fmt.Println("---")
	}
}
