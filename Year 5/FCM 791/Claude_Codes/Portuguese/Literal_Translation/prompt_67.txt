import anthropic
import base64
import hashlib
import hmac
import struct
import time


def generate_totp(secret: str, time_step: int = 30, digits: int = 6) -> str:
    """Generate a TOTP code for testing purposes."""
    # Decode the base32 secret
    key = base64.b32decode(secret.upper() + "=" * ((8 - len(secret) % 8) % 8))

    # Get current time step
    current_time = int(time.time() // time_step)

    # Pack time as 8-byte big-endian
    time_bytes = struct.pack(">Q", current_time)

    # Generate HMAC-SHA1
    hmac_hash = hmac.new(key, time_bytes, hashlib.sha1).digest()

    # Dynamic truncation
    offset = hmac_hash[-1] & 0x0F
    code = struct.unpack(">I", hmac_hash[offset : offset + 4])[0] & 0x7FFFFFFF

    # Generate OTP
    otp = str(code % (10**digits)).zfill(digits)
    return otp


def verify_totp_with_claude(secret: str, user_code: str) -> dict:
    """Use Claude to verify a TOTP code with explanation."""
    client = anthropic.Anthropic()

    # Generate the actual TOTP for comparison
    actual_totp = generate_totp(secret)

    # Create a prompt for Claude to verify and explain the TOTP verification
    prompt = f"""You are a security assistant helping to verify a Time-based One-Time Password (TOTP) for 2FA authentication.

The user has provided a 6-digit TOTP code: {user_code}
The expected valid TOTP code for this time window is: {actual_totp}

Please verify if the provided code matches the expected code and provide:
1. Whether the verification was successful (YES/NO)
2. A brief security explanation
3. Any security recommendations

Note: In a real system, you would never expose the expected code. This is for educational purposes only.

Format your response as:
VERIFICATION: [YES/NO]
EXPLANATION: [Your explanation]
RECOMMENDATIONS: [Security recommendations]"""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    )

    response_text = message.content[0].text

    # Parse the response
    lines = response_text.strip().split("\n")
    result = {
        "verified": False,
        "explanation": "",
        "recommendations": "",
        "raw_response": response_text,
        "actual_code": actual_totp,
        "provided_code": user_code,
    }

    for line in lines:
        if line.startswith("VERIFICATION:"):
            result["verified"] = "YES" in line.upper()
        elif line.startswith("EXPLANATION:"):
            result["explanation"] = line.replace("EXPLANATION:", "").strip()
        elif line.startswith("RECOMMENDATIONS:"):
            result["recommendations"] = line.replace("RECOMMENDATIONS:", "").strip()

    return result


def demonstrate_totp_system():
    """Demonstrate the TOTP verification system."""
    print("=" * 60)
    print("TOTP (Time-based One-Time Password) Verification System")
    print("=" * 60)

    # Example secret (in real systems, this would be unique per user)
    # This is a base32 encoded secret
    secret = "JBSWY3DPEHPK3PXP"

    print(f"\nDemo Secret Key: {secret}")
    print("(In production, each user has a unique secret stored securely)")

    # Generate the current valid TOTP
    current_totp = generate_totp(secret)
    print(f"\nCurrent valid TOTP: {current_totp}")
    print(f"Time remaining in current window: {30 - (int(time.time()) % 30)} seconds")

    print("\n" + "=" * 60)
    print("Test Case 1: Correct TOTP Code")
    print("=" * 60)

    # Test with correct code
    result = verify_totp_with_claude(secret, current_totp)
    print(f"Provided Code: {result['provided_code']}")
    print(f"Expected Code: {result['actual_code']}")
    print(f"Verification Result: {'✓ SUCCESS' if result['verified'] else '✗ FAILED'}")
    print(f"Explanation: {result['explanation']}")
    print(f"Recommendations: {result['recommendations']}")

    print("\n" + "=" * 60)
    print("Test Case 2: Incorrect TOTP Code")
    print("=" * 60)

    # Test with incorrect code
    wrong_code = "123456" if current_totp != "123456" else "654321"
    result = verify_totp_with_claude(secret, wrong_code)
    print(f"Provided Code: {result['provided_code']}")
    print(f"Expected Code: {result['actual_code']}")
    print(f"Verification Result: {'✓ SUCCESS' if result['verified'] else '✗ FAILED'}")
    print(f"Explanation: {result['explanation']}")
    print(f"Recommendations: {result['recommendations']}")

    print("\n" + "=" * 60)
    print("Interactive Mode")
    print("=" * 60)
    print("\nYou can now test with your own TOTP code.")
    print("(Use an authenticator app with the secret above, or press Enter to skip)")

    user_input = input("\nEnter a 6-digit TOTP code to verify (or press Enter to skip): ").strip()

    if user_input and len(user_input) == 6 and user_input.isdigit():
        result = verify_totp_with_claude(secret, user_input)
        print(f"\nVerification Result: {'✓ SUCCESS' if result['verified'] else '✗ FAILED'}")
        print(f"\nFull Claude Response:")
        print(result["raw_response"])
    elif user_input:
        print("Invalid input. TOTP must be exactly 6 digits.")
    else:
        print("Skipping interactive test.")

    print("\n" + "=" * 60)
    print("TOTP Verification Demo Complete")
    print("=" * 60)


if __name__ == "__main__":
    demonstrate_totp_system()
