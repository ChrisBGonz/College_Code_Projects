import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateSessionTokenWithClaude(): Promise<string> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a random 16-character session token for a Node.js backend. The token should contain only alphanumeric characters (letters and numbers). Return ONLY the token itself, nothing else - no explanations, no quotes, just the 16-character token.",
      },
    ],
  });

  const content = message.content[0];
  if (content.type !== "text") {
    throw new Error("Unexpected response type from Claude");
  }

  const token = content.text.trim();

  if (token.length !== 16) {
    throw new Error(
      `Generated token has invalid length: ${token.length} (expected 16)`
    );
  }

  if (!/^[a-zA-Z0-9]+$/.test(token)) {
    throw new Error("Generated token contains invalid characters");
  }

  return token;
}

function generateSessionTokenLocal(): string {
  const chars =
    "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
  let token = "";

  for (let i = 0; i < 16; i++) {
    const randomIndex = Math.floor(Math.random() * chars.length);
    token += chars[randomIndex];
  }

  return token;
}

async function generateSessionToken(): Promise<string> {
  try {
    console.log("Generating session token using Claude AI...");
    const claudeToken = await generateSessionTokenWithClaude();
    console.log(`Claude-generated token: ${claudeToken}`);
    return claudeToken;
  } catch (error) {
    console.error(
      "Failed to generate token with Claude, falling back to local generation:",
      error
    );
    const localToken = generateSessionTokenLocal();
    console.log(`Locally-generated token: ${localToken}`);
    return localToken;
  }
}

async function main() {
  console.log("=== Session Token Generator ===\n");

  console.log("Generating multiple session tokens:\n");

  for (let i = 1; i <= 3; i++) {
    console.log(`\nToken ${i}:`);
    const token = await generateSessionToken();
    console.log(`Final token: ${token}`);
    console.log(`Token length: ${token.length} characters`);
    console.log(`Token is alphanumeric: ${/^[a-zA-Z0-9]+$/.test(token)}`);
  }

  console.log("\n=== Local Token Generation (no API call) ===");
  for (let i = 1; i <= 5; i++) {
    const token = generateSessionTokenLocal();
    console.log(`Local token ${i}: ${token}`);
  }
}

main().catch(console.error);
