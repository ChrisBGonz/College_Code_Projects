import os
from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

@app.route('/calculate', methods=['POST'])
def calculate():
    """
    Route that accepts a mathematical expression and returns the calculated result.
    Uses Claude AI to evaluate the expression through multi-turn conversation.
    """
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Please provide a mathematical expression in the request body with key "expression"'
        }), 400
    
    expression = data['expression']
    
    if not expression or not expression.strip():
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    conversation_history.append({
        "role": "user",
        "content": f"Calculate the following mathematical expression and provide only the numerical result: {expression}"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a mathematical calculator. When given a mathematical expression, calculate it and respond with only the numerical result. Do not include any explanation or additional text, just the number.",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return jsonify({
        'expression': expression,
        'result': assistant_message,
        'conversation_turns': len(conversation_history) // 2
    })

@app.route('/calculate/history', methods=['GET'])
def get_history():
    """
    Route to get the conversation history of calculations.
    """
    history = []
    for i in range(0, len(conversation_history), 2):
        if i + 1 < len(conversation_history):
            user_msg = conversation_history[i]['content']
            expression = user_msg.replace("Calculate the following mathematical expression and provide only the numerical result: ", "")
            history.append({
                'expression': expression,
                'result': conversation_history[i + 1]['content']
            })
    
    return jsonify({
        'history': history,
        'total_calculations': len(history)
    })

@app.route('/calculate/clear', methods=['POST'])
def clear_history():
    """
    Route to clear the conversation history.
    """
    global conversation_history
    conversation_history = []
    
    return jsonify({
        'message': 'Conversation history cleared successfully'
    })

@app.route('/calculate/complex', methods=['POST'])
def calculate_complex():
    """
    Route for complex mathematical calculations with follow-up questions.
    Demonstrates multi-turn conversation capability.
    """
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Please provide a mathematical expression in the request body with key "expression"'
        }), 400
    
    expression = data['expression']
    follow_up = data.get('follow_up', None)
    
    local_history = []
    
    local_history.append({
        "role": "user",
        "content": f"Calculate: {expression}. Show the step-by-step solution."
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a mathematical tutor. Show step-by-step solutions to mathematical problems.",
        messages=local_history
    )
    
    first_response = response.content[0].text
    local_history.append({
        "role": "assistant",
        "content": first_response
    })
    
    result = {
        'expression': expression,
        'solution': first_response,
        'follow_up_answer': None
    }
    
    if follow_up:
        local_history.append({
            "role": "user",
            "content": follow_up
        })
        
        follow_up_response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system="You are a mathematical tutor. Show step-by-step solutions to mathematical problems.",
            messages=local_history
        )
        
        result['follow_up_answer'] = follow_up_response.content[0].text
    
    return jsonify(result)

if __name__ == '__main__':
    print("Starting Flask Mathematical Calculator API...")
    print("Available endpoints:")
    print("  POST /calculate - Calculate a mathematical expression")
    print("  GET /calculate/history - Get calculation history")
    print("  POST /calculate/clear - Clear calculation history")
    print("  POST /calculate/complex - Complex calculations with step-by-step solutions")
    app.run(debug=True, host='0.0.0.0', port=5000)
