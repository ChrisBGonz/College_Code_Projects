const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const users = {
  "token-user-123": {
    id: 1,
    name: "João Silva",
    email: "joao.silva@example.com",
    role: "admin",
    createdAt: "2024-01-15",
  },
  "token-user-456": {
    id: 2,
    name: "Maria Santos",
    email: "maria.santos@example.com",
    role: "user",
    createdAt: "2024-02-20",
  },
  "token-user-789": {
    id: 3,
    name: "Carlos Oliveira",
    email: "carlos.oliveira@example.com",
    role: "moderator",
    createdAt: "2024-03-10",
  },
};

// Function to extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Function to authenticate user with token
function authenticateUser(token) {
  return users[token] || null;
}

// Function to process request with Claude AI using streaming
async function processWithClaude(user, requestPath) {
  const systemPrompt = `You are a helpful API assistant that provides information about user data. 
  You have access to the following user's private data and should respond in a professional and helpful manner.
  Always respond in Portuguese (Brazilian).`;

  const userMessage = `The authenticated user is accessing the endpoint: ${requestPath}
  
  User Data:
  - ID: ${user.id}
  - Name: ${user.name}
  - Email: ${user.email}
  - Role: ${user.role}
  - Account Created: ${user.createdAt}
  
  Please provide a helpful summary of this user's private data and what they can access based on their role.`;

  let fullResponse = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system: systemPrompt,
    messages: [
      {
        role: "user",
        content: userMessage,
      },
    ],
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      fullResponse += chunk.delta.text;
      process.stdout.write(chunk.delta.text);
    }
  }

  console.log("\n");
  return fullResponse;
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader(
    "Access-Control-Allow-Headers",
    "Content-Type, Authorization"
  );

  // Handle preflight requests
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Parse URL
  const url = new URL(req.url, `http://${req.headers.host}`);
  const path = url.pathname;

  // Health check endpoint
  if (path === "/health" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ status: "ok", message: "API is running" }));
    return;
  }

  // Protected endpoints
  if (path.startsWith("/api/")) {
    // Extract and validate Bearer token
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    // Authenticate user
    const user = authenticateUser(token);

    if (!user) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    // Handle different API endpoints
    if (path === "/api/user/profile" && req.method === "GET") {
      try {
        console.log(`\nProcessing request for user: ${user.name}`);
        console.log("Claude AI Response (streaming):\n");

        const aiResponse = await processWithClaude(user, path);

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            success: true,
            user: {
              id: user.id,
              name: user.name,
              email: user.email,
              role: user.role,
              createdAt: user.createdAt,
            },
            aiInsights: aiResponse,
          })
        );
      } catch (error) {
        console.error("Error processing with Claude:", error);
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to process request",
          })
        );
      }
    } else if (path === "/api/user/data" && req.method === "GET") {
      try {
        console.log(`\nFetching private data for user: ${user.name}`);
        console.log("Claude AI Response (streaming):\n");

        const aiResponse = await processWithClaude(user, path);

        // Simulate private user data
        const privateData = {
          personalInfo: {
            id: user.id,
            name: user.name,
            email: user.email,
          },
          accountDetails: {
            role: user.role,
            createdAt: user.createdAt,
            lastLogin: new Date().toISOString(),
            permissions:
              user.role === "admin"
                ? ["read", "write", "delete", "manage"]
                : user.role === "moderator"
                  ? ["read", "write", "moderate"]
                  : ["read"],
          },
          statistics: {
            totalLogins: Math.floor(Math.random() * 100) + 1,
            lastActivity: new Date().toISOString(),
            accountStatus: "active",
          },
        };

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            success: true,
            privateData: privateData,
            aiAnalysis: aiResponse,
          })
        );
      } catch (error) {
        console.error("Error processing with Claude:", error);
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to process request",
          })
        );
      }
    } else {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Not Found",
          message: "Endpoint not found",
        })
      );
    }
  } else {
    // Public endpoints
    if (path === "/" && req.method === "GET") {
      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          message: "Welcome to the Private Data API",
          version: "1.0.0",
          endpoints: {
            public: [
              "GET / - API information",
              "GET /health - Health check",
            ],
            protected: [
              "GET /api/user/profile - Get user profile (requires Bearer token)",
              "GET /api/user/data - Get private user data (requires Bearer token)",
            ],
          },
          availableTokens: Object.keys(users),
        })
      );
    } else {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Not Found",
          message: "Route not found",
        })
      );
    }
  }
});

// Start server
const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`\n🚀 Private Data API running on port ${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET http://localhost:${PORT}/ - API information`);
  console.log(`  GET http://localhost:${PORT}/health - Health check`);
  console.log(
    `  GET http://localhost:${PORT}/api/user/profile - User profile (protected)`
  );
  console.log(
    `  GET http://localhost:${PORT}/api/user/data - Private user data (protected)`
  );
  console.log(`\nTest tokens:`);
  Object.keys(users).forEach((token) => {
    console.log(`  ${token} -> ${users[token].name} (${users[token].role})`);
  });
  console.log(`\nExample usage:`);
  console.log(
    `  curl -H "Authorization: Bearer token-user-123" http://localhost:${PORT}/api/user/profile`
  );
});

module.exports = server;
