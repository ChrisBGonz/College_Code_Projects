import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.service.AiServices;
import dev.langchain4j.service.UserMessage;

public class JsonToUserConverter {

    record User(String name, String email, int age) {}

    interface JsonParserService {
        @UserMessage("""
                Parse the following JSON string and extract the user information.
                Return ONLY a valid JSON object with fields: name, email, age.
                If any field is missing, use empty string for strings and 0 for numbers.
                
                JSON to parse: {{it}}
                """)
        String parseUserJson(String jsonString);
    }

    public static User parseJsonToUser(String jsonString) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();

        JsonParserService parserService = AiServices.builder(JsonParserService.class)
                .chatLanguageModel(model)
                .build();

        String parsedJson = parserService.parseUserJson(jsonString);

        try {
            ObjectMapper objectMapper = new ObjectMapper();
            return objectMapper.readValue(parsedJson, User.class);
        } catch (Exception e) {
            throw new RuntimeException("Failed to convert parsed JSON to User object: " + e.getMessage(), e);
        }
    }

    public static void main(String[] args) {
        String jsonString = """
                {
                    "name": "John Doe",
                    "email": "john.doe@example.com",
                    "age": 30
                }
                """;

        System.out.println("Input JSON: " + jsonString);

        User user = parseJsonToUser(jsonString);

        System.out.println("Parsed User:");
        System.out.println("user.name = " + user.name());
        System.out.println("user.email = " + user.email());
        System.out.println("user.age = " + user.age());

        System.out.println("\n--- Testing with incomplete JSON ---");
        String incompleteJson = """
                {
                    "name": "Jane Smith",
                    "email": "jane@example.com"
                }
                """;

        System.out.println("Input JSON: " + incompleteJson);
        User user2 = parseJsonToUser(incompleteJson);
        System.out.println("Parsed User:");
        System.out.println("user.name = " + user2.name());
        System.out.println("user.email = " + user2.email());
        System.out.println("user.age = " + user2.age());

        System.out.println("\n--- Testing with malformed JSON ---");
        String malformedJson = "name: Alice, age: 25, email: alice@test.com";

        System.out.println("Input: " + malformedJson);
        User user3 = parseJsonToUser(malformedJson);
        System.out.println("Parsed User:");
        System.out.println("user.name = " + user3.name());
        System.out.println("user.email = " + user3.email());
        System.out.println("user.age = " + user3.age());
    }
}
