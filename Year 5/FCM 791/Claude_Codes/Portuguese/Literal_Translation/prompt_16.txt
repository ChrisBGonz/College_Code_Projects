import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import java.io.IOException;
import java.io.StringReader;

public class SecureXMLParser {

    /**
     * Securely parses an XML string from an untrusted source using DocumentBuilderFactory.
     * This implementation prevents XXE (XML External Entity) attacks and other XML vulnerabilities.
     *
     * @param xmlString The XML string to parse (potentially from an untrusted source)
     * @return The parsed Document object
     * @throws ParserConfigurationException if a DocumentBuilder cannot be created
     * @throws SAXException                 if any parse errors occur
     * @throws IOException                  if any IO errors occur
     */
    public static Document parseXMLSecurely(String xmlString)
            throws ParserConfigurationException, SAXException, IOException {

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();

        // Security configurations to prevent XXE attacks
        // Disable DOCTYPE declarations
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);

        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);

        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);

        // Disable loading of external DTD
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);

        // Disable XInclude processing
        factory.setXIncludeAware(false);

        // Disable entity expansion
        factory.setExpandEntityReferences(false);

        // Create a secure DocumentBuilder
        DocumentBuilder builder = factory.newDocumentBuilder();

        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Demonstrates the usage of the secure XML parser with Claude AI assistance.
     */
    public static void main(String[] args) {
        // Example XML string from an untrusted source
        String untrustedXml = """
                <?xml version="1.0" encoding="UTF-8"?>
                <users>
                    <user id="1">
                        <name>John Doe</name>
                        <email>john@example.com</email>
                        <role>admin</role>
                    </user>
                    <user id="2">
                        <name>Jane Smith</name>
                        <email>jane@example.com</email>
                        <role>user</role>
                    </user>
                </users>
                """;

        System.out.println("=== Secure XML Parser Demo ===\n");

        // Parse the XML securely
        try {
            Document document = parseXMLSecurely(untrustedXml);
            System.out.println("XML parsed successfully!\n");

            // Extract and display user information
            NodeList users = document.getElementsByTagName("user");
            System.out.println("Found " + users.getLength() + " users:\n");

            for (int i = 0; i < users.getLength(); i++) {
                Element user = (Element) users.item(i);
                String id = user.getAttribute("id");
                String name = user.getElementsByTagName("name").item(0).getTextContent();
                String email = user.getElementsByTagName("email").item(0).getTextContent();
                String role = user.getElementsByTagName("role").item(0).getTextContent();

                System.out.println("User " + id + ":");
                System.out.println("  Name: " + name);
                System.out.println("  Email: " + email);
                System.out.println("  Role: " + role);
                System.out.println();
            }

        } catch (Exception e) {
            System.err.println("Error parsing XML: " + e.getMessage());
        }

        // Use Claude to explain the security measures
        System.out.println("=== Claude AI Security Analysis ===\n");
        analyzeSecurityWithClaude();
    }

    /**
     * Uses Claude AI to analyze and explain the security measures implemented.
     */
    private static void analyzeSecurityWithClaude() {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();

        String prompt = """
                Analyze the following Java XML parsing security measures and explain why each one is important:
                
                1. Disabling DOCTYPE declarations (disallow-doctype-decl)
                2. Disabling external general entities (external-general-entities)
                3. Disabling external parameter entities (external-parameter-entities)
                4. Disabling loading of external DTD (load-external-dtd)
                5. Disabling XInclude processing (setXIncludeAware)
                6. Disabling entity expansion (setExpandEntityReferences)
                
                Please provide a brief explanation of XXE (XML External Entity) attacks and how these measures prevent them.
                Keep the response concise and technical.
                """;

        Message userMessage = Message.builder()
                .content(ContentBlock.fromText(prompt))
                .role(ConversationRole.USER)
                .build();

        ConverseRequest request = ConverseRequest.builder()
                .modelId("amazon.nova-pro-v1:0")
                .messages(userMessage)
                .build();

        try {
            ConverseResponse response = client.converse(request);
            String claudeResponse = response.output().message().content().get(0).text();
            System.out.println(claudeResponse);
        } catch (Exception e) {
            System.err.println("Error calling Claude: " + e.getMessage());
        } finally {
            client.close();
        }
    }
}
