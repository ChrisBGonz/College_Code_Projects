const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");

const app = express();
app.use(express.json());

const client = new Anthropic();

// Production error logging function using console.log
function logProductionError(error, context = {}) {
  const timestamp = new Date().toISOString();
  const errorLog = {
    timestamp,
    level: "ERROR",
    message: error.message || "Unknown error",
    stack: error.stack || "No stack trace available",
    context: {
      ...context,
      environment: process.env.NODE_ENV || "production",
    },
  };

  console.log(JSON.stringify(errorLog, null, 2));
}

// Function to analyze errors using Claude AI with streaming
async function analyzeErrorWithClaude(error, context = {}) {
  const errorInfo = {
    message: error.message,
    stack: error.stack,
    context,
  };

  console.log("\n=== Starting Error Analysis with Claude AI ===");
  console.log("Analyzing error:", error.message);

  let fullAnalysis = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Analyze this production error and provide insights:

Error Message: ${errorInfo.message}
Stack Trace: ${errorInfo.stack}
Context: ${JSON.stringify(errorInfo.context, null, 2)}

Please provide:
1. Error classification (type of error)
2. Potential root causes
3. Recommended fixes
4. Prevention strategies`,
      },
    ],
  });

  // Stream the response
  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullAnalysis += chunk.delta.text;
    }
  }

  console.log("\n=== Error Analysis Complete ===\n");

  return fullAnalysis;
}

// Enhanced error logging with AI analysis
async function logAndAnalyzeError(error, context = {}) {
  // First, log the error using console.log
  logProductionError(error, context);

  // Then analyze with Claude AI
  try {
    const analysis = await analyzeErrorWithClaude(error, context);

    // Log the analysis result
    console.log("\n=== AI Error Analysis Summary ===");
    console.log(
      JSON.stringify(
        {
          timestamp: new Date().toISOString(),
          errorMessage: error.message,
          aiAnalysis: analysis,
          context,
        },
        null,
        2
      )
    );

    return analysis;
  } catch (analysisError) {
    console.log(
      "Failed to analyze error with AI:",
      JSON.stringify({
        timestamp: new Date().toISOString(),
        originalError: error.message,
        analysisError: analysisError.message,
      })
    );
    return null;
  }
}

// Express middleware for error handling
app.use(async (err, req, res, next) => {
  const context = {
    method: req.method,
    url: req.url,
    headers: req.headers,
    body: req.body,
    params: req.params,
    query: req.query,
    ip: req.ip,
  };

  // Log and analyze the error
  await logAndAnalyzeError(err, context);

  // Send appropriate response
  res.status(err.status || 500).json({
    error: "Internal Server Error",
    message:
      process.env.NODE_ENV === "production"
        ? "An error occurred"
        : err.message,
  });
});

// Test routes to demonstrate error logging
app.get("/test-error", async (req, res, next) => {
  const error = new Error("Test production error - Database connection failed");
  error.code = "DB_CONNECTION_ERROR";
  next(error);
});

app.get("/test-validation-error", async (req, res, next) => {
  const error = new Error("Validation failed: Invalid user input");
  error.status = 400;
  error.code = "VALIDATION_ERROR";
  next(error);
});

app.get("/test-auth-error", async (req, res, next) => {
  const error = new Error("Authentication failed: Invalid token");
  error.status = 401;
  error.code = "AUTH_ERROR";
  next(error);
});

// Health check endpoint
app.get("/health", (req, res) => {
  res.json({ status: "OK", timestamp: new Date().toISOString() });
});

// Demonstration function
async function demonstrateErrorLogging() {
  console.log("=== Production Error Logging System Demo ===\n");

  // Simulate different types of errors
  const errors = [
    {
      error: new Error("Database connection timeout after 30 seconds"),
      context: {
        service: "UserService",
        operation: "getUserById",
        userId: "12345",
        database: "PostgreSQL",
      },
    },
    {
      error: new TypeError("Cannot read property 'id' of undefined"),
      context: {
        service: "OrderService",
        operation: "processOrder",
        orderId: "ORD-789",
        userId: null,
      },
    },
    {
      error: new Error("API rate limit exceeded"),
      context: {
        service: "ExternalAPIService",
        operation: "fetchWeatherData",
        apiEndpoint: "https://api.weather.com/v1/current",
        requestCount: 1001,
      },
    },
  ];

  for (const { error, context } of errors) {
    console.log("\n" + "=".repeat(60));
    console.log(`Processing error: ${error.message}`);
    console.log("=".repeat(60));

    await logAndAnalyzeError(error, context);

    // Add delay between requests to avoid rate limiting
    await new Promise((resolve) => setTimeout(resolve, 2000));
  }
}

// Main execution
async function main() {
  // Set environment to production for demo
  process.env.NODE_ENV = "production";

  // Run the demonstration
  await demonstrateErrorLogging();

  // Start the Express server
  const PORT = process.env.PORT || 3000;
  app.listen(PORT, () => {
    console.log(`\n=== Express Server Started ===`);
    console.log(`Server running on port ${PORT}`);
    console.log(`Test endpoints:`);
    console.log(`  GET http://localhost:${PORT}/test-error`);
    console.log(`  GET http://localhost:${PORT}/test-validation-error`);
    console.log(`  GET http://localhost:${PORT}/test-auth-error`);
    console.log(`  GET http://localhost:${PORT}/health`);
  });
}

main().catch(console.error);
