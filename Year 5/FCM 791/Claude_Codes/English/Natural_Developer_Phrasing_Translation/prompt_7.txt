import Anthropic from "@anthropic-ai/sdk";
import * as crypto from "crypto";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user123",
      email: "john@example.com",
      passwordHash: "old_hash_123",
      updatedAt: new Date("2024-01-01"),
    },
    {
      id: "user456",
      email: "jane@example.com",
      passwordHash: "old_hash_456",
      updatedAt: new Date("2024-01-01"),
    },
  ],
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_password",
    description:
      "Validates that a password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes a password using SHA-256 with a salt for secure storage",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "find_user",
    description: "Finds a user in the database by their user ID",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["userId"],
    },
  },
  {
    name: "update_password_in_db",
    description: "Updates the password hash for a user in the database",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The unique identifier of the user",
        },
        passwordHash: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["userId", "passwordHash"],
    },
  },
];

// Tool implementations
function validatePassword(password) {
  const minLength = 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /\d/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);

  const errors = [];

  if (password.length < minLength) {
    errors.push(`Password must be at least ${minLength} characters long`);
  }
  if (!hasUppercase) {
    errors.push("Password must contain at least one uppercase letter");
  }
  if (!hasLowercase) {
    errors.push("Password must contain at least one lowercase letter");
  }
  if (!hasNumber) {
    errors.push("Password must contain at least one number");
  }
  if (!hasSpecialChar) {
    errors.push("Password must contain at least one special character");
  }

  return {
    isValid: errors.length === 0,
    errors: errors,
  };
}

function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString("hex");
  const hash = crypto
    .createHash("sha256")
    .update(password + salt)
    .digest("hex");
  return `${salt}:${hash}`;
}

function findUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      found: true,
      user: {
        id: user.id,
        email: user.email,
        updatedAt: user.updatedAt,
      },
    };
  }
  return {
    found: false,
    error: `User with ID ${userId} not found`,
  };
}

function updatePasswordInDb(userId, passwordHash) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex === -1) {
    return {
      success: false,
      error: `User with ID ${userId} not found`,
    };
  }

  database.users[userIndex].passwordHash = passwordHash;
  database.users[userIndex].updatedAt = new Date();

  return {
    success: true,
    message: `Password successfully updated for user ${userId}`,
    updatedAt: database.users[userIndex].updatedAt,
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_password":
      return validatePassword(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "find_user":
      return findUser(toolInput.userId);
    case "update_password_in_db":
      return updatePasswordInDb(toolInput.userId, toolInput.passwordHash);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\nInitiating password reset for user: ${userId}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user ID "${userId}" with the new password "${newPassword}". 
      Follow these steps:
      1. First, validate the new password meets security requirements
      2. Find the user in the database to verify they exist
      3. If validation passes and user exists, hash the password
      4. Update the password in the database
      5. Return a clear success or failure message`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  console.log("\nClaude's initial response:");
  console.log(`Stop reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool called: ${toolUse.name}`);
      console.log("Tool input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Tool result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Get next response
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log(`\nNext response stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\nFinal Response:");
  console.log("=".repeat(50));
  console.log(finalResponse);

  return finalResponse;
}

// Test the function
async function main() {
  // Test 1: Valid password reset
  console.log("\n" + "=".repeat(60));
  console.log("TEST 1: Valid password reset");
  console.log("=".repeat(60));
  await resetPassword("user123", "SecurePass123!");

  // Test 2: Invalid password (too weak)
  console.log("\n" + "=".repeat(60));
  console.log("TEST 2: Invalid password (too weak)");
  console.log("=".repeat(60));
  await resetPassword("user456", "weak");

  // Test 3: Non-existent user
  console.log("\n" + "=".repeat(60));
  console.log("TEST 3: Non-existent user");
  console.log("=".repeat(60));
  await resetPassword("user999", "ValidPass123!");
}

main().catch(console.error);
