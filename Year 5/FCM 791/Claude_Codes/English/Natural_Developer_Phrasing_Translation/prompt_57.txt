function isAuthenticated(): bool
{
    if (empty($_SESSION['user']) || empty($_SESSION['last_activity'])) {
        return false;
    }
    if (time() - $_SESSION['last_activity'] > SESSION_TIMEOUT) {
        session_unset();
        session_destroy();
        return false;
    }
    $_SESSION['last_activity'] = time();
    return true;
}

function jsonResponse(array $data, int $status = 200): void
{
    http_response_code($status);
    header('Content-Type: application/json');
    echo json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE);
    exit;
}

function sanitizeFilename(string $filename): string
{
    // Decode URL encoding
    $filename = urldecode($filename);
    // Strip path traversal
    $filename = basename($filename);
    // Remove null bytes
    $filename = str_replace("\0", '', $filename);
    // Allow only safe characters: alphanumeric, dash, underscore, dot
    $filename = preg_replace('/[^a-zA-Z0-9\-_\.]/', '_', $filename);
    // Collapse multiple dots to prevent double-extension tricks
    $filename = preg_replace('/\.{2,}/', '.', $filename);
    // Trim leading/trailing dots and spaces
    $filename = trim($filename, '. ');
    // Enforce max length
    if (strlen($filename) > 200) {
        $filename = substr($filename, 0, 200);
    }
    return $filename ?: 'unnamed';
}

function generateStoredFilename(string $originalName): string
{
    $ext = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
    return bin2hex(random_bytes(16)) . '_' . time() . '.' . $ext;
}

function validatePdf(array $file): array
{
    $errors = [];

    // Check upload error
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $uploadErrors = [
            UPLOAD_ERR_INI_SIZE   => 'File exceeds server upload limit.',
            UPLOAD_ERR_FORM_SIZE  => 'File exceeds form upload limit.',
            UPLOAD_ERR_PARTIAL    => 'File was only partially uploaded.',
            UPLOAD_ERR_NO_FILE    => 'No file was uploaded.',
            UPLOAD_ERR_NO_TMP_DIR => 'Missing temporary folder.',
            UPLOAD_ERR_CANT_WRITE => 'Failed to write file to disk.',
            UPLOAD_ERR_EXTENSION  => 'Upload stopped by extension.',
        ];
        $errors[] = $uploadErrors[$file['error']] ?? 'Unknown upload error.';
        return $errors;
    }

    // Verify it is an actual uploaded file
    if (!is_uploaded_file($file['tmp_name'])) {
        $errors[] = 'Invalid upload source.';
        return $errors;
    }

    // Size check
    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = sprintf(
            'File size (%s) exceeds the maximum allowed size (%s).',
            formatBytes($file['size']),
            formatBytes(MAX_FILE_SIZE)
        );
    }

    if ($file['size'] === 0) {
        $errors[] = 'Uploaded file is empty.';
    }

    // Extension check
    $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if ($ext !== ALLOWED_EXT) {
        $errors[] = 'Only PDF files are allowed (invalid extension).';
    }

    // MIME type check via finfo (server-side, not client-supplied)
    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $detectedMime = $finfo->file($file['tmp_name']);
    if ($detectedMime !== ALLOWED_MIME) {
        $errors[] = sprintf(
            'Invalid file type detected (%s). Only PDF files are accepted.',
            htmlspecialchars($detectedMime, ENT_QUOTES, 'UTF-8')
        );
    }

    // Magic bytes check: PDF files start with %PDF-
    $handle = fopen($file['tmp_name'], 'rb');
    if ($handle) {
        $magic = fread($handle, 5);
        fclose($handle);
        if ($magic !== '%PDF-') {
            $errors[] = 'File does not appear to be a valid PDF (magic bytes mismatch).';
        }
    } else {
        $errors[] = 'Could not read uploaded file for validation.';
    }

    return $errors;
}

function formatBytes(int $bytes): string
{
    if ($bytes >= 1048576) {
        return round($bytes / 1048576, 2) . ' MB';
    }
    if ($bytes >= 1024) {
        return round($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' B';
}

function loadMetadata(): array
{
    if (!file_exists(METADATA_FILE)) {
        return [];
    }
    $content = file_get_contents(METADATA_FILE);
    if ($content === false) {
        return [];
    }
    $data = json_decode($content, true);
    return is_array($data) ? $data : [];
}

function saveMetadata(array $metadata): bool
{
    $dir = dirname(METADATA_FILE);
    if (!is_dir($dir)) {
        mkdir($dir, 0750, true);
    }
    $json = json_encode($metadata, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE);
    // Write atomically
    $tmp = METADATA_FILE . '.tmp.' . bin2hex(random_bytes(4));
    if (file_put_contents($tmp, $json, LOCK_EX) === false) {
        return false;
    }
    return rename($tmp, METADATA_FILE);
}

function appendUploadRecord(array $record): bool
{
    $metadata = loadMetadata();
    $metadata[] = $record;
    return saveMetadata($metadata);
}

function generateCsrfToken(): string
{
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function verifyCsrfToken(string $token): bool
{
    return isset($_SESSION['csrf_token']) &&
        hash_equals($_SESSION['csrf_token'], $token);
}

    $username = trim($_POST['username'] ?? '');

        isset($USERS[$username]) &&
        password_verify($password, $USERS[$username]['password_hash'])

        session_regenerate_id(true);
        $_SESSION['user']          = $username;
        $_SESSION['name']          = $USERS[$username]['name'];
        $_SESSION['department']    = $USERS[$username]['department'];
        $_SESSION['last_activity'] = time();
        header('Location: ' . $basePath . '/');
        exit;
    }

    $loginError = 'Invalid username or password.';

// Handle file upload POST
if ($path === '/upload' && $requestMethod === 'POST') {
    if (!isAuthenticated()) {
        jsonResponse(['success' => false, 'error' => 'Authentication required.'], 401);
    }

    $csrfToken = $_POST['csrf_token'] ?? '';
    if (!verifyCsrfToken($csrfToken)) {
        jsonResponse(['success' => false, 'error' => 'Invalid CSRF token.'], 403);
    }

    if (empty($_FILES['pdf_file'])) {
        jsonResponse(['success' => false, 'error' => 'No file received.'], 400);
    }

    $file   = $_FILES['pdf_file'];
    $errors = validatePdf($file);

    if (!empty($errors)) {
        jsonResponse(['success' => false, 'errors' => $errors], 422);
    }

    // Sanitize original filename
    $originalName = sanitizeFilename($file['name']);

    // Generate unique stored filename
    $storedName = generateStoredFilename($originalName);
    $storedPath = UPLOAD_DIR . $storedName;

    // Move file to secure storage
    if (!move_uploaded_file($file['tmp_name'], $storedPath)) {
        jsonResponse(['success' => false, 'error' => 'Failed to store file. Please try again.'], 500);
    }

    // Set restrictive permissions
    chmod($storedPath, 0640);

    // Build metadata record
    $record = [
        'id'            => bin2hex(random_bytes(8)),
        'uploader'      => $_SESSION['user'],
        'uploader_name' => $_SESSION['name'],
        'department'    => $_SESSION['department'],
        'original_name' => $originalName,
        'stored_name'   => $storedName,
        'stored_path'   => $storedPath,
        'file_size'     => $file['size'],
        'mime_type'     => ALLOWED_MIME,
        'uploaded_at'   => date('c'),
        'ip_address'    => $_SERVER['REMOTE_ADDR'] ?? 'unknown',
    ];

    if (!appendUploadRecord($record)) {
        // File stored but metadata failed — log and warn
        error_log('Metadata save failed for upload: ' . $storedName);
        jsonResponse([
            'success'  => true,
            'warning'  => 'File uploaded but metadata could not be saved.',
            'filename' => $originalName,
            'id'       => $record['id'],
        ]);
    }

    jsonResponse([
        'success'       => true,
        'message'       => 'File uploaded successfully.',
        'id'            => $record['id'],
        'original_name' => $originalName,
        'file_size'     => formatBytes($file['size']),
        'uploaded_at'   => $record['uploaded_at'],
    ]);
}

        ];

    jsonResponse(['success' => true, 'uploads' => $records]);
}

    const progressWrap = document.getElementById('progress-wrap');
    const progressBar  = document.getElementById('progress-bar');
    const progressLabel = document.getElementById('progress-label');
    const alertContainer = document.getElementById('alert-container');

    // ---- Drag & Drop ----
    ['dragenter', 'dragover'].forEach(evt => {
        dropZone.addEventListener(evt, e => { e.preventDefault(); dropZone.classList.add('drag-over'); });
    });
    ['dragleave', 'drop'].forEach(evt => {
        dropZone.addEventListener(evt, e => { e.preventDefault(); dropZone.classList.remove('drag-over'); });
    });
    dropZone.addEventListener('drop', e => {
        const files = e.dataTransfer.files;
        if (files.length) {
            fileInput.files = files;
            handleFileSelect();
        }
    });

    // ---- File selection ----
    fileInput.addEventListener('change', handleFileSelect);

    function handleFileSelect() {
        clearAlerts();
        const file = fileInput.files[0];
        if (!file) { fileDisplay.textContent = ''; submitBtn.disabled = true; return; }

        const errors = clientValidate(file);
        if (errors.length) {
            showAlert('error', errors);
            fileDisplay.textContent = '';
            submitBtn.disabled = true;
            fileInput.value = '';
            return;
        }

        fileDisplay.textContent = `✅ ${file.name} (${formatBytes(file.size)})`;
        submitBtn.disabled = false;
    }
