
```python
import anthropic
import logging
import os

# Configure logging to append to app.log
logging.basicConfig(
    filename="app.log",
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
    filemode="a",  # Append mode
)

logger = logging.getLogger(__name__)


def log_search_query(query: str) -> None:
    """Log a user's search query to app.log."""
    logger.info(f"Search query: {query}")


def process_search_with_claude(query: str) -> str:
    """
    Process a search query using Claude API with streaming.
    Logs the query and returns the response.
    """
    # Log the search query
    log_search_query(query)

    # Initialize Anthropic client
    client = anthropic.Anthropic()

    # Use streaming to get response from Claude
    full_response = ""

    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Please help me search for information about: {query}",
            }
        ],
    ) as stream:
        for text in stream.text_stream:
            full_response += text
            print(text, end="", flush=True)

    print()  # New line after streaming completes

    # Log the completion of the search
    logger.info(f"Search completed for query: {query}")

    return full_response


def main():
    """Main function to demonstrate search query logging."""
    # Example search queries
    search_queries = [
        "Python programming best practices",
        "Machine learning algorithms",
        "Web development frameworks",
    ]

    print("Starting search query logging demonstration...")
    print(f"Logs will be written to: {os.path.abspath('app.log')}\n")

    for query in search_queries:
        print(f"\n{'='*50}")
        print(f"Processing query: {query}")
        print("=" * 50)

        response = process_search_with_claude(query)

        print(f"\nResponse length: {len(response)} characters")

    print("\n" + "=" * 50)
    print("All queries processed. Check app.log for logged entries.")

    # Display the log file contents
    print("\nLog file contents:")
    print("-" * 50)
    with open("app.log", "r") as f:
        print(f.read())


if __name__ == "__main__":
    main()
```
