import anthropic
import base64
import hashlib
import hmac
import struct
import time


def generate_totp(secret: str, time_step: int = 30, digits: int = 6) -> str:
    """Generate a TOTP code for testing purposes."""
    key = base64.b32decode(secret.upper())
    timestamp = int(time.time() / time_step)
    msg = struct.pack(">Q", timestamp)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    offset = h[-1] & 0x0F
    code = struct.unpack(">I", h[offset : offset + 4])[0] & 0x7FFFFFFF
    return str(code % (10**digits)).zfill(digits)


def validate_totp_with_claude(secret: str, user_code: str) -> dict:
    """Use Claude to validate a TOTP code with tool use."""
    client = anthropic.Anthropic()

    tools = [
        {
            "name": "validate_totp_code",
            "description": "Validates a 6-digit TOTP code against a secret key using RFC 6238 standard. Checks current time window and adjacent windows for clock drift tolerance.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "secret": {
                        "type": "string",
                        "description": "The base32-encoded TOTP secret key",
                    },
                    "user_provided_code": {
                        "type": "string",
                        "description": "The 6-digit TOTP code provided by the user",
                    },
                    "time_step": {
                        "type": "integer",
                        "description": "Time step in seconds (default: 30)",
                        "default": 30,
                    },
                    "window": {
                        "type": "integer",
                        "description": "Number of time steps to check before and after current time for clock drift tolerance (default: 1)",
                        "default": 1,
                    },
                },
                "required": ["secret", "user_provided_code"],
            },
        }
    ]

    messages = [
        {
            "role": "user",
            "content": f"Please validate this TOTP code for 2FA authentication. Secret key: {secret}, User provided code: {user_code}. Check if this is a valid TOTP code.",
        }
    ]

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages,
    )

    result = {"valid": False, "message": "", "details": {}}

    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break

        if tool_use_block and tool_use_block.name == "validate_totp_code":
            tool_input = tool_use_block.input
            secret_key = tool_input["secret"]
            provided_code = tool_input["user_provided_code"]
            time_step = tool_input.get("time_step", 30)
            window = tool_input.get("window", 1)

            current_timestamp = int(time.time() / time_step)
            is_valid = False
            matched_window = None

            for i in range(-window, window + 1):
                timestamp = current_timestamp + i
                key = base64.b32decode(secret_key.upper())
                msg = struct.pack(">Q", timestamp)
                h = hmac.new(key, msg, hashlib.sha1).digest()
                offset = h[-1] & 0x0F
                code = struct.unpack(">I", h[offset : offset + 4])[0] & 0x7FFFFFFF
                expected_code = str(code % (10**6)).zfill(6)

                if provided_code == expected_code:
                    is_valid = True
                    matched_window = i
                    break

            tool_result = {
                "valid": is_valid,
                "matched_window": matched_window,
                "checked_windows": list(range(-window, window + 1)),
                "timestamp_checked": current_timestamp,
            }

            messages.append({"role": "assistant", "content": response.content})
            messages.append(
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "tool_result",
                            "tool_use_id": tool_use_block.id,
                            "content": str(tool_result),
                        }
                    ],
                }
            )

            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=1024,
                tools=tools,
                messages=messages,
            )

            result["valid"] = is_valid
            result["details"] = tool_result

    for block in response.content:
        if hasattr(block, "text"):
            result["message"] = block.text
            break

    return result


def demo_2fa_login():
    """Demonstrate a 2FA login flow with TOTP validation."""
    print("=== 2FA Login Flow Demo ===\n")

    secret = "JBSWY3DPEHPK3PXP"
    print(f"User's TOTP Secret: {secret}")
    print("(This would be stored securely and shown as QR code during setup)\n")

    print("Scenario 1: Valid TOTP Code")
    print("-" * 40)
    valid_code = generate_totp(secret)
    print(f"Generated valid TOTP code: {valid_code}")
    print("Validating with Claude...")

    result = validate_totp_with_claude(secret, valid_code)
    print(f"Validation Result: {'✓ VALID' if result['valid'] else '✗ INVALID'}")
    print(f"Claude's Analysis: {result['message']}")
    print(f"Technical Details: {result['details']}\n")

    print("Scenario 2: Invalid TOTP Code")
    print("-" * 40)
    invalid_code = "123456"
    print(f"Testing invalid code: {invalid_code}")
    print("Validating with Claude...")

    result = validate_totp_with_claude(secret, invalid_code)
    print(f"Validation Result: {'✓ VALID' if result['valid'] else '✗ INVALID'}")
    print(f"Claude's Analysis: {result['message']}")
    print(f"Technical Details: {result['details']}\n")

    print("Scenario 3: Interactive Login")
    print("-" * 40)
    print(f"Your TOTP secret is: {secret}")
    print("Enter this secret in Google Authenticator to get a code")
    print(f"Or use the auto-generated code: {generate_totp(secret)}")

    user_input = input("\nEnter your 6-digit TOTP code (or press Enter to use auto-generated): ").strip()

    if not user_input:
        user_input = generate_totp(secret)
        print(f"Using auto-generated code: {user_input}")

    if len(user_input) == 6 and user_input.isdigit():
        print("Validating your code with Claude...")
        result = validate_totp_with_claude(secret, user_input)

        print(f"\nLogin {'SUCCESSFUL' if result['valid'] else 'FAILED'}")
        print(f"Validation Result: {'✓ VALID' if result['valid'] else '✗ INVALID'}")
        print(f"Claude's Analysis: {result['message']}")

        if result["valid"]:
            print("\n🎉 Welcome! You have successfully authenticated with 2FA.")
        else:
            print("\n❌ Authentication failed. Please check your TOTP code and try again.")
    else:
        print("Invalid input. Please enter exactly 6 digits.")


if __name__ == "__main__":
    demo_2fa_login()
