import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as url from "url";

const client = new Anthropic();

async function fetchUserProfile(userId: string): Promise<object> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a realistic user profile for user ID: ${userId}. Return ONLY a valid JSON object with the following fields: id, name, email, age, location, occupation, bio, joinDate, preferences (object with theme, language, notifications), and stats (object with postsCount, followersCount, followingCount). Do not include any explanation or markdown, just the raw JSON object.`,
      },
    ],
  });

  const content = message.content[0];
  if (content.type !== "text") {
    throw new Error("Unexpected response type from Claude");
  }

  const jsonText = content.text.trim();
  return JSON.parse(jsonText);
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url || "", true);

  if (parsedUrl.pathname === "/api/user-profile" && req.method === "GET") {
    const userId = parsedUrl.query.user_id as string;

    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(JSON.stringify({ error: "user_id query parameter is required" }));
      return;
    }

    try {
      const userProfile = await fetchUserProfile(userId);
      const jsonData = JSON.stringify(userProfile, null, 2);

      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="user_profile_${userId}.json"`,
        "Content-Length": Buffer.byteLength(jsonData),
      });

      res.end(jsonData);
    } catch (error) {
      console.error("Error fetching user profile:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(JSON.stringify({ error: "Failed to fetch user profile" }));
    }
  } else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(JSON.stringify({ error: "Route not found" }));
  }
});

const PORT = 3000;
server.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
  console.log(
    `Try: http://localhost:${PORT}/api/user-profile?user_id=12345`
  );
});
