using System;
using System.Security.Cryptography;
using System.Text;

class RsaKeyPairGenerator
{
    static void Main()
    {
        // Generate RSA key pair with 2048-bit key size
        using (RSA rsa = RSA.Create(2048))
        {
            // Export private key in PKCS#8 format (includes both private and public key info)
            byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
            string privateKeyBase64 = Convert.ToBase64String(privateKeyBytes);

            // Export public key in SubjectPublicKeyInfo (SPKI) format
            byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
            string publicKeyBase64 = Convert.ToBase64String(publicKeyBytes);

            // Display PEM-formatted keys
            string privateKeyPem = FormatAsPem(privateKeyBase64, "PRIVATE KEY");
            string publicKeyPem = FormatAsPem(publicKeyBase64, "PUBLIC KEY");

            Console.WriteLine("=== RSA Private Key (PKCS#8) ===");
            Console.WriteLine(privateKeyPem);

            Console.WriteLine("=== RSA Public Key (SPKI) ===");
            Console.WriteLine(publicKeyPem);

            // Demonstrate signing and verification
            DemonstrateSigningAndVerification(rsa);
        }
    }

    static string FormatAsPem(string base64Content, string keyType)
    {
        StringBuilder pem = new StringBuilder();
        pem.AppendLine($"-----BEGIN {keyType}-----");

        // Split base64 into 64-character lines (PEM standard)
        for (int i = 0; i < base64Content.Length; i += 64)
        {
            int length = Math.Min(64, base64Content.Length - i);
            pem.AppendLine(base64Content.Substring(i, length));
        }

        pem.AppendLine($"-----END {keyType}-----");
        return pem.ToString();
    }

    static void DemonstrateSigningAndVerification(RSA rsa)
    {
        Console.WriteLine("=== Signing and Verification Demo ===");

        string message = "Hello, this is a test message for RSA signing!";
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);

        // Sign the message using SHA-256 hash algorithm with PKCS#1 v1.5 padding
        byte[] signature = rsa.SignData(
            messageBytes,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );

        Console.WriteLine($"Message: {message}");
        Console.WriteLine($"Signature (Base64): {Convert.ToBase64String(signature)}");

        // Verify the signature
        bool isValid = rsa.VerifyData(
            messageBytes,
            signature,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );

        Console.WriteLine($"Signature valid: {isValid}");

        // Demonstrate key export and re-import for cross-context verification
        Console.WriteLine("\n=== Cross-Context Verification Demo ===");

        // Export only the public key parameters
        RSAParameters publicKeyParams = rsa.ExportParameters(includePrivateParameters: false);
        RSAParameters privateKeyParams = rsa.ExportParameters(includePrivateParameters: true);

        // Create a new RSA instance with only the public key for verification
        using (RSA publicOnlyRsa = RSA.Create())
        {
            publicOnlyRsa.ImportParameters(publicKeyParams);

            bool crossContextValid = publicOnlyRsa.VerifyData(
                messageBytes,
                signature,
                HashAlgorithmName.SHA256,
                RSASignaturePadding.Pkcs1
            );

            Console.WriteLine($"Cross-context verification (public key only): {crossContextValid}");
        }

        // Demonstrate with PSS padding (more modern and recommended)
        Console.WriteLine("\n=== PSS Padding Demo ===");

        byte[] pssSignature = rsa.SignData(
            messageBytes,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pss
        );

        bool pssValid = rsa.VerifyData(
            messageBytes,
            pssSignature,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pss
        );

        Console.WriteLine($"PSS Signature (Base64): {Convert.ToBase64String(pssSignature)}");
        Console.WriteLine($"PSS Signature valid: {pssValid}");

        // Display key size information
        Console.WriteLine($"\nKey Size: {rsa.KeySize} bits");
    }
}
