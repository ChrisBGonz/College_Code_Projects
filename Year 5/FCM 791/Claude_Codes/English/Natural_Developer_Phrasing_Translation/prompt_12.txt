import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are an expert Node.js/Express.js developer. Your task is to generate production-ready Express.js server code.

When asked to build an Express.js server scaffold, you should:
1. Create a complete, minimal but production-ready Express.js server
2. Include all necessary middleware for production use
3. Add proper error handling
4. Configure environment variables
5. Include health check and sample routes
6. Make the code clean, well-commented, and following best practices

Generate the complete server.js file with all the required features. The code should be ready to run with minimal setup.`,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function extractCodeFromResponse(response: string): string {
  // Extract code blocks from markdown
  const codeBlockRegex = /```(?:javascript|js|typescript|ts)?\n([\s\S]*?)```/g;
  const matches = [];
  let match;

  while ((match = codeBlockRegex.exec(response)) !== null) {
    matches.push(match[1]);
  }

  if (matches.length > 0) {
    return matches.join("\n\n");
  }

  // If no code blocks found, return the response as-is
  return response;
}

async function generateExpressServer(): Promise<void> {
  console.log("🚀 Express.js Server Generator with Multi-turn Conversation\n");
  console.log("=".repeat(60));

  // Initial request to generate the server
  console.log("\n📝 Requesting Express.js server scaffold...\n");

  const initialRequest = `Build a minimal Express.js server scaffold suitable for production with the following requirements:

1. Initialize the Express app
2. Apply standard production middleware:
   - JSON/body parsing (express.json() and express.urlencoded())
   - Security headers (using helmet)
   - Compression (using compression)
   - Request logging (using morgan)
3. Wire up routes:
   - Health check endpoint at GET /health
   - Sample API route at GET /api/sample
4. Include centralized error-handling middleware:
   - 404 handler for unknown routes
   - Global error handler with proper error response format
5. Configure the server to:
   - Read HOST and PORT from environment variables
   - Use production-safe defaults (HOST: '0.0.0.0', PORT: 3000)
   - Start listening with proper startup logging

Please provide the complete server.js file with all dependencies listed in a package.json format as well.`;

  const serverCode = await chat(initialRequest);
  console.log("✅ Initial server scaffold generated!\n");

  // Follow-up to add more production features
  console.log("🔧 Requesting additional production features...\n");

  const followUpRequest = `Great! Now please enhance the server with these additional production features:

1. Add rate limiting middleware (using express-rate-limit) to prevent abuse
2. Add CORS configuration (using cors package) with environment-based origins
3. Add request ID middleware to track requests
4. Add graceful shutdown handling for SIGTERM and SIGINT signals
5. Add a /api/info endpoint that returns server information (version, uptime, environment)

Please provide the updated complete server.js file with these additions.`;

  const enhancedCode = await chat(followUpRequest);
  console.log("✅ Enhanced server code generated!\n");

  // Ask for environment configuration
  console.log("⚙️  Requesting environment configuration...\n");

  const envRequest = `Perfect! Now please provide:

1. A .env.example file showing all required environment variables with descriptions
2. A brief explanation of how to run this server in development vs production
3. Any important security considerations or best practices to keep in mind

Keep the response concise and practical.`;

  const envConfig = await chat(envRequest);
  console.log("✅ Environment configuration generated!\n");

  // Save the generated files
  console.log("💾 Saving generated files...\n");

  // Extract and save server code
  const serverCodeExtracted = extractCodeFromResponse(enhancedCode);
  if (serverCodeExtracted) {
    fs.writeFileSync(
      path.join(process.cwd(), "generated-server.js"),
      serverCodeExtracted
    );
    console.log("✅ Server code saved to: generated-server.js");
  }

  // Save the full conversation output
  const fullOutput = `# Express.js Server Scaffold Generation

## Initial Server Scaffold
${serverCode}

## Enhanced Server with Production Features
${enhancedCode}

## Environment Configuration and Best Practices
${envConfig}
`;

  fs.writeFileSync(
    path.join(process.cwd(), "server-generation-output.md"),
    fullOutput
  );
  console.log("✅ Full output saved to: server-generation-output.md");

  // Interactive mode for additional questions
  console.log("\n" + "=".repeat(60));
  console.log("💬 Interactive Mode - Ask follow-up questions about the server");
  console.log('Type "exit" to quit\n');

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const askQuestion = (): void => {
    rl.question("You: ", async (input) => {
      const userInput = input.trim();

      if (userInput.toLowerCase() === "exit") {
        console.log("\n👋 Goodbye! Your Express.js server scaffold is ready.");
        rl.close();
        return;
      }

      if (!userInput) {
        askQuestion();
        return;
      }

      try {
        console.log("\n🤔 Thinking...\n");
        const response = await chat(userInput);
        console.log(`Assistant: ${response}\n`);
      } catch (error) {
        console.error("Error:", error);
      }

      askQuestion();
    });
  };

  askQuestion();
}

// Main execution
generateExpressServer().catch(console.error);
