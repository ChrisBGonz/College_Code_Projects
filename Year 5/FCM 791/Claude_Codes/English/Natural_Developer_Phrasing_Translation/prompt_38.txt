require "openssl"
require "base64"

def sign_with_rsa_pss(payload, private_key_pem)
  private_key = OpenSSL::PKey::RSA.new(private_key_pem)
  
  payload_bytes = payload.is_a?(String) ? payload.encode("UTF-8") : payload
  
  digest = OpenSSL::Digest::SHA256.new
  
  signature = private_key.sign_pss(
    digest,
    payload_bytes,
    salt_length: :digest,
    mgf1_hash: "SHA256"
  )
  
  Base64.strict_encode64(signature)
end

def verify_rsa_pss_signature(payload, signature_base64, public_key_pem)
  public_key = OpenSSL::PKey::RSA.new(public_key_pem)
  
  payload_bytes = payload.is_a?(String) ? payload.encode("UTF-8") : payload
  
  signature = Base64.strict_decode64(signature_base64)
  
  digest = OpenSSL::Digest::SHA256.new
  
  public_key.verify_pss(
    digest,
    signature,
    payload_bytes,
    salt_length: :auto,
    mgf1_hash: "SHA256"
  )
rescue OpenSSL::PKey::PKeyError => e
  puts "Verification error: #{e.message}"
  false
end

if __FILE__ == $PROGRAM_NAME
  puts "Generating RSA key pair (2048 bits)..."
  private_key = OpenSSL::PKey::RSA.generate(2048)
  public_key = private_key.public_key

  public_key_pem = public_key.to_pem

  payload = "Hello, this is a test message for RSA-PSS signing!"
  puts "Payload: #{payload}"
  puts ""
  
  puts "Signing payload with RSA-PSS (SHA-256)..."
  signature_base64 = sign_with_rsa_pss(payload, private_key_pem)
  puts "Signature (Base64): #{signature_base64}"
  puts "Signature length: #{signature_base64.length} characters"
  puts ""
  
  puts "Verifying signature..."
  is_valid = verify_rsa_pss_signature(payload, signature_base64, public_key_pem)
  puts "Signature valid: #{is_valid}"
  puts ""
  
  puts "Testing with tampered payload..."
  tampered_payload = "Hello, this is a TAMPERED message for RSA-PSS signing!"
  is_valid_tampered = verify_rsa_pss_signature(tampered_payload, signature_base64, public_key_pem)
  puts "Tampered signature valid: #{is_valid_tampered}"
  puts ""
  
  puts "Testing with binary data..."
  binary_payload = (0..255).map(&:chr).join
  binary_signature = sign_with_rsa_pss(binary_payload, private_key_pem)
  is_binary_valid = verify_rsa_pss_signature(binary_payload, binary_signature, public_key_pem)
  puts "Binary payload signature valid: #{is_binary_valid}"
  puts ""
  
  puts "Testing PSS non-determinism (same payload, different signatures)..."
  sig1 = sign_with_rsa_pss(payload, private_key_pem)
  sig2 = sign_with_rsa_pss(payload, private_key_pem)
  puts "Signatures are different (PSS is probabilistic): #{sig1 != sig2}"
  puts "Both signatures verify correctly: #{verify_rsa_pss_signature(payload, sig1, public_key_pem) && verify_rsa_pss_signature(payload, sig2, public_key_pem)}"
end
