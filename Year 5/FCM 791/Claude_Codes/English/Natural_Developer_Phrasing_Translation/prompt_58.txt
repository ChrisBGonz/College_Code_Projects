const SignupFlow = (() => {
  const STORAGE_KEY = 'signup_flow_state';
  const STEPS = ['account', 'personal', 'preferences', 'review'];

  // ─── State Management ────────────────────────────────────────────────────────

  function loadState() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      return raw
        ? JSON.parse(raw)
        : { currentStep: 0, completedSteps: [], formData: {} };
    } catch {
      return { currentStep: 0, completedSteps: [], formData: {} };
    }
  }

  function saveState(state) {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    } catch (e) {
      console.error('SignupFlow: Failed to persist state to localStorage', e);
    }
  }

  function clearState() {
    localStorage.removeItem(STORAGE_KEY);
  }

  // ─── Validation ──────────────────────────────────────────────────────────────

  const validators = {
    account: (data) => {
      const errors = {};
      if (!data.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(data.email)) {
        errors.email = 'A valid email address is required.';
      }
      if (!data.password || data.password.length < 8) {
        errors.password = 'Password must be at least 8 characters.';
      }
      if (data.password !== data.confirmPassword) {
        errors.confirmPassword = 'Passwords do not match.';
      }
      return errors;
    },

    personal: (data) => {
      const errors = {};
      if (!data.firstName || data.firstName.trim().length < 1) {
        errors.firstName = 'First name is required.';
      }
      if (!data.lastName || data.lastName.trim().length < 1) {
        errors.lastName = 'Last name is required.';
      }
      if (data.phone && !/^\+?[\d\s\-().]{7,20}$/.test(data.phone)) {
        errors.phone = 'Enter a valid phone number.';
      }
      return errors;
    },

    preferences: (data) => {
      const errors = {};
      if (!data.newsletter && data.newsletter !== false) {
        errors.newsletter = 'Please indicate your newsletter preference.';
      }
      return errors;
    },

    review: () => ({}),
  };

  function validateStep(stepName, data) {
    const validator = validators[stepName];
    return validator ? validator(data) : {};
  }

  function hasErrors(errors) {
    return Object.keys(errors).length > 0;
  }

  // ─── Step Navigation ─────────────────────────────────────────────────────────

  function getCurrentStepName(state) {
    return STEPS[state.currentStep];
  }

  function isFirstStep(state) {
    return state.currentStep === 0;
  }

  function isLastStep(state) {
    return state.currentStep === STEPS.length - 1;
  }

  function canNavigateToStep(state, targetStep) {
    if (targetStep < 0 || targetStep >= STEPS.length) return false;
    if (targetStep <= state.currentStep) return true;
    // Can only jump forward if all previous steps are completed
    for (let i = 0; i < targetStep; i++) {
      if (!state.completedSteps.includes(STEPS[i])) return false;
    }
    return true;
  }

  // ─── Core API ────────────────────────────────────────────────────────────────

  /**
   * Initialise (or resume) the signup flow.
   * Returns the current state so the UI can render the correct step.
   */
  function init() {
    const state = loadState();
    return {
      state,
      stepName: getCurrentStepName(state),
      stepIndex: state.currentStep,
      totalSteps: STEPS.length,
      isFirst: isFirstStep(state),
      isLast: isLastStep(state),
      savedData: state.formData,
    };
  }

  /**
   * Save partial data for the current step without advancing.
   * Useful for auto-saving on input change.
   */
  function autosave(stepName, partialData) {
    const state = loadState();
    state.formData[stepName] = {
      ...(state.formData[stepName] || {}),
      ...partialData,
    };
    saveState(state);
  }

  /**
   * Validate and advance to the next step.
   * Returns { success, errors, state } — the caller decides how to render.
   */
  function nextStep(stepName, stepData) {
    const state = loadState();

    // Validate
    const errors = validateStep(stepName, stepData);
    if (hasErrors(errors)) {
      return { success: false, errors };
    }

    // Persist step data
    state.formData[stepName] = stepData;

    // Mark step as completed
    if (!state.completedSteps.includes(stepName)) {
      state.completedSteps.push(stepName);
    }

    // Advance step index
    if (!isLastStep(state)) {
      state.currentStep += 1;
    }

    saveState(state);

    return {
      success: true,
      errors: {},
      stepName: getCurrentStepName(state),
      stepIndex: state.currentStep,
      isFirst: isFirstStep(state),
      isLast: isLastStep(state),
    };
  }

  /**
   * Go back to the previous step (no validation needed).
   */
  function prevStep() {
    const state = loadState();
    if (!isFirstStep(state)) {
      state.currentStep -= 1;
      saveState(state);
    }
    return {
      stepName: getCurrentStepName(state),
      stepIndex: state.currentStep,
      isFirst: isFirstStep(state),
      isLast: isLastStep(state),
      savedData: state.formData,
    };
  }

  /**
   * Jump directly to a specific step by index (only if allowed).
   */
  function goToStep(targetStep) {
    const state = loadState();
    if (!canNavigateToStep(state, targetStep)) {
      return { success: false, reason: 'Cannot navigate to that step yet.' };
    }
    state.currentStep = targetStep;
    saveState(state);
    return {
      success: true,
      stepName: getCurrentStepName(state),
      stepIndex: state.currentStep,
      isFirst: isFirstStep(state),
      isLast: isLastStep(state),
      savedData: state.formData,
    };
  }

  /**
   * Build and return the final payload from all collected step data.
   * Strips sensitive fields (e.g. confirmPassword) before returning.
   */
  function buildPayload(state) {
    const { account = {}, personal = {}, preferences = {} } = state.formData;

    // Remove confirmPassword — never send to the server
    const { confirmPassword: _removed, ...accountClean } = account;

    return {
      ...accountClean,
      ...personal,
      ...preferences,
      signupCompletedAt: new Date().toISOString(),
    };
  }

  /**
   * Submit the final form.
   * Validates the last step, builds the payload, calls the provided
   * async `submitFn`, then clears local state on success.
   *
   * @param {string}   lastStepName  - Name of the final step being submitted.
   * @param {object}   lastStepData  - Form data from the final step.
   * @param {Function} submitFn      - async (payload) => response
   * @returns {Promise<{ success, errors, response }>}
   */
  async function submit(lastStepName, lastStepData, submitFn) {
    const state = loadState();

    // Validate the final step
    const errors = validateStep(lastStepName, lastStepData);
    if (hasErrors(errors)) {
      return { success: false, errors };
    }

    // Merge final step data
    state.formData[lastStepName] = lastStepData;

    // Build the complete payload
    const payload = buildPayload(state);

    try {
      const response = await submitFn(payload);
      // Only clear state after a confirmed successful submission
      clearState();
      return { success: true, errors: {}, response };
    } catch (error) {
      // Preserve state so the user can retry
      saveState(state);
      return {
        success: false,
        errors: { submit: error.message || 'Submission failed. Please try again.' },
        response: null,
      };
    }
  }

  /**
   * Reset the entire flow (e.g. user clicks "Start over").
   */
  function reset() {
    clearState();
  }

  /**
   * Get a summary of all collected data (useful for a review step).
   */
  function getSummary() {
    const state = loadState();
    return {
      steps: STEPS,
      completedSteps: state.completedSteps,
      formData: state.formData,
      currentStep: state.currentStep,
    };
  }

  // ─── Public API ──────────────────────────────────────────────────────────────

  return {
    init,
    autosave,
    nextStep,
    prevStep,
    goToStep,
    submit,
    reset,
    getSummary,
    STEPS,
  };
})();
