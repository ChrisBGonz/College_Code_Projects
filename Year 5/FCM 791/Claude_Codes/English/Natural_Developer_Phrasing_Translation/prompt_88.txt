using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

namespace MFAAuditLog
{
    // Models for MFA configuration
    public class MFAConfiguration
    {
        public bool IsEnabled { get; set; }
        public List<string> Factors { get; set; } = new List<string>();
        public string DefaultMethod { get; set; } = string.Empty;
    }

    public class RequestMetadata
    {
        public string IpAddress { get; set; } = string.Empty;
        public string UserAgent { get; set; } = string.Empty;
        public string CorrelationId { get; set; } = string.Empty;
    }

    public class AuditLogEntry
    {
        public string UserId { get; set; } = string.Empty;
        public DateTime Timestamp { get; set; }
        public string ActionType { get; set; } = string.Empty;
        public MFAConfiguration OldSettings { get; set; } = new MFAConfiguration();
        public MFAConfiguration NewSettings { get; set; } = new MFAConfiguration();
        public RequestMetadata RequestMetadata { get; set; } = new RequestMetadata();
        public string AuditDescription { get; set; } = string.Empty;
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            // Example MFA configuration change scenario
            var userId = "user-12345";
            var oldConfig = new MFAConfiguration
            {
                IsEnabled = false,
                Factors = new List<string>(),
                DefaultMethod = "none"
            };

            var newConfig = new MFAConfiguration
            {
                IsEnabled = true,
                Factors = new List<string> { "TOTP", "SMS" },
                DefaultMethod = "TOTP"
            };

            var requestMetadata = new RequestMetadata
            {
                IpAddress = "192.168.1.100",
                UserAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
                CorrelationId = Guid.NewGuid().ToString()
            };

            // Generate audit log entry using Claude
            var auditEntry = await GenerateAuditLogEntry(userId, oldConfig, newConfig, requestMetadata);

            // Display the audit log entry
            Console.WriteLine("=== MFA Configuration Audit Log Entry ===");
            Console.WriteLine($"User ID: {auditEntry.UserId}");
            Console.WriteLine($"Timestamp: {auditEntry.Timestamp:yyyy-MM-dd HH:mm:ss UTC}");
            Console.WriteLine($"Action Type: {auditEntry.ActionType}");
            Console.WriteLine($"\nOld Settings:");
            Console.WriteLine($"  - MFA Enabled: {auditEntry.OldSettings.IsEnabled}");
            Console.WriteLine($"  - Factors: {string.Join(", ", auditEntry.OldSettings.Factors.Count > 0 ? auditEntry.OldSettings.Factors : new List<string> { "none" })}");
            Console.WriteLine($"  - Default Method: {auditEntry.OldSettings.DefaultMethod}");
            Console.WriteLine($"\nNew Settings:");
            Console.WriteLine($"  - MFA Enabled: {auditEntry.NewSettings.IsEnabled}");
            Console.WriteLine($"  - Factors: {string.Join(", ", auditEntry.NewSettings.Factors)}");
            Console.WriteLine($"  - Default Method: {auditEntry.NewSettings.DefaultMethod}");
            Console.WriteLine($"\nRequest Metadata:");
            Console.WriteLine($"  - IP Address: {auditEntry.RequestMetadata.IpAddress}");
            Console.WriteLine($"  - User Agent: {auditEntry.RequestMetadata.UserAgent}");
            Console.WriteLine($"  - Correlation ID: {auditEntry.RequestMetadata.CorrelationId}");
            Console.WriteLine($"\nAudit Description:");
            Console.WriteLine(auditEntry.AuditDescription);
        }

        static async Task<AuditLogEntry> GenerateAuditLogEntry(
            string userId,
            MFAConfiguration oldConfig,
            MFAConfiguration newConfig,
            RequestMetadata requestMetadata)
        {
            var client = new AnthropicClient();

            // Prepare the prompt for Claude
            var prompt = $@"You are a security audit system. Generate a detailed audit log description for an MFA configuration change.

User ID: {userId}
Timestamp: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss UTC}

Old MFA Configuration:
- MFA Enabled: {oldConfig.IsEnabled}
- Factors: {(oldConfig.Factors.Count > 0 ? string.Join(", ", oldConfig.Factors) : "none")}
- Default Method: {oldConfig.DefaultMethod}

New MFA Configuration:
- MFA Enabled: {newConfig.IsEnabled}
- Factors: {(newConfig.Factors.Count > 0 ? string.Join(", ", newConfig.Factors) : "none")}
- Default Method: {newConfig.DefaultMethod}

Request Metadata:
- IP Address: {requestMetadata.IpAddress}
- User Agent: {requestMetadata.UserAgent}
- Correlation ID: {requestMetadata.CorrelationId}

Please provide:
1. A concise action type (e.g., 'MFA_ENABLED', 'MFA_DISABLED', 'MFA_FACTOR_ADDED', 'MFA_DEFAULT_METHOD_CHANGED', etc.)
2. A detailed audit description explaining what changed, why it's significant from a security perspective, and any notable aspects of the change.

Format your response as:
ACTION_TYPE: [action type]
DESCRIPTION: [detailed description]";

            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = new List<ContentBase>
                    {
                        new TextContent { Text = prompt }
                    }
                }
            };

            var messageRequest = new MessageParameters
            {
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                Messages = messages
            };

            var response = await client.Messages.GetClaudeMessageAsync(messageRequest);

            // Parse Claude's response
            var responseText = response.Content[0] is TextContent textContent ? textContent.Text : string.Empty;
            var actionType = "MFA_CONFIGURATION_CHANGED";
            var description = responseText;

            // Extract action type and description from response
            var lines = responseText.Split('\n');
            foreach (var line in lines)
            {
                if (line.StartsWith("ACTION_TYPE:"))
                {
                    actionType = line.Replace("ACTION_TYPE:", "").Trim();
                }
                else if (line.StartsWith("DESCRIPTION:"))
                {
                    description = line.Replace("DESCRIPTION:", "").Trim();
                    // Get remaining lines as part of description
                    var descriptionIndex = Array.IndexOf(lines, line);
                    if (descriptionIndex < lines.Length - 1)
                    {
                        var remainingLines = new List<string> { description };
                        for (int i = descriptionIndex + 1; i < lines.Length; i++)
                        {
                            remainingLines.Add(lines[i]);
                        }
                        description = string.Join("\n", remainingLines);
                    }
                }
            }

            // Create and return the audit log entry
            return new AuditLogEntry
            {
                UserId = userId,
                Timestamp = DateTime.UtcNow,
                ActionType = actionType,
                OldSettings = oldConfig,
                NewSettings = newConfig,
                RequestMetadata = requestMetadata,
                AuditDescription = description
            };
        }
    }
}
