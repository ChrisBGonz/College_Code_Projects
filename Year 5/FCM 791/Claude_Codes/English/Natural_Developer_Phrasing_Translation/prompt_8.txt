package config

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
)

// ErrEmptyEnvVar is returned when the environment variable is set but empty.
var ErrEmptyEnvVar = errors.New("environment variable is set but empty")

// ErrEnvVarNotSet is returned when the environment variable is not set at all.
var ErrEnvVarNotSet = errors.New("environment variable is not set")

// LoadConfigPath reads the config file path from the environment variable
// identified by envVarName, validates that the value is non-empty, that the
// file exists, and that it is readable. It returns the resolved absolute path
// or a descriptive error.
func LoadConfigPath(envVarName string) (string, error) {
	if envVarName == "" {
		return "", errors.New("envVarName must not be empty")
	}

	rawPath, ok := os.LookupEnv(envVarName)
	if !ok {
		return "", fmt.Errorf("%w: %q", ErrEnvVarNotSet, envVarName)
	}

	if rawPath == "" {
		return "", fmt.Errorf("%w: %q", ErrEmptyEnvVar, envVarName)
	}

	absPath, err := filepath.Abs(rawPath)
	if err != nil {
		return "", fmt.Errorf("resolving absolute path for %q (from env var %q): %w", rawPath, envVarName, err)
	}

	info, err := os.Stat(absPath)
	if err != nil {
		if os.IsNotExist(err) {
			return "", fmt.Errorf("config file does not exist: %q (from env var %q)", absPath, envVarName)
		}
		return "", fmt.Errorf("cannot stat config file %q (from env var %q): %w", absPath, envVarName, err)
	}

	if info.IsDir() {
		return "", fmt.Errorf("config path %q (from env var %q) is a directory, not a file", absPath, envVarName)
	}

	f, err := os.Open(absPath)
	if err != nil {
		return "", fmt.Errorf("config file %q (from env var %q) is not readable: %w", absPath, envVarName, err)
	}
	f.Close()

	return absPath, nil
}
