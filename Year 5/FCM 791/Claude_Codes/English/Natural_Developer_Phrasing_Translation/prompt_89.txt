const os = require('os');

const isProduction = process.env.NODE_ENV === 'production';

function formatTimestamp() {
  return new Date().toISOString();
}

function formatErrorDetails(err) {
  return {
    timestamp: formatTimestamp(),
    environment: process.env.NODE_ENV || 'unknown',
    message: err.message || 'Unknown error',
    name: err.name || 'Error',
    stack: err.stack || 'No stack trace available',
    code: err.code || null,
    statusCode: err.statusCode || err.status || null,
    system: {
      platform: os.platform(),
      arch: os.arch(),
      nodeVersion: process.version,
      hostname: os.hostname(),
      uptime: process.uptime(),
      memoryUsage: process.memoryUsage(),
      loadAverage: os.loadavg(),
    },
  };
}

function formatRequestDetails(req) {
  if (!req) return null;

  return {
    method: req.method,
    url: req.originalUrl || req.url,
    path: req.path,
    params: req.params,
    query: req.query,
    headers: {
      'user-agent': req.headers['user-agent'],
      'content-type': req.headers['content-type'],
      accept: req.headers['accept'],
      referer: req.headers['referer'],
      'x-forwarded-for': req.headers['x-forwarded-for'],
    },
    ip: req.ip || req.connection?.remoteAddress,
    userId: req.user?.id || null,
  };
}

function logProductionError(err, req = null, context = {}) {
  if (!isProduction) return;

  const errorDetails = formatErrorDetails(err);
  const requestDetails = formatRequestDetails(req);

  const logEntry = {
    level: 'ERROR',
    ...errorDetails,
    request: requestDetails,
    context: Object.keys(context).length > 0 ? context : null,
  };

  console.log(JSON.stringify(logEntry, null, 2));
}

function logProductionWarning(message, details = {}, req = null) {
  if (!isProduction) return;

  const logEntry = {
    level: 'WARNING',
    timestamp: formatTimestamp(),
    environment: process.env.NODE_ENV || 'unknown',
    message,
    details: Object.keys(details).length > 0 ? details : null,
    request: formatRequestDetails(req),
    system: {
      platform: os.platform(),
      nodeVersion: process.version,
      hostname: os.hostname(),
      uptime: process.uptime(),
    },
  };

  console.log(JSON.stringify(logEntry, null, 2));
}

function logProductionInfo(message, details = {}) {
  if (!isProduction) return;

  const logEntry = {
    level: 'INFO',
    timestamp: formatTimestamp(),
    environment: process.env.NODE_ENV || 'unknown',
    message,
    details: Object.keys(details).length > 0 ? details : null,
  };

  console.log(JSON.stringify(logEntry, null, 2));
}

function createErrorMiddleware() {
  return function productionErrorMiddleware(err, req, res, next) {
    logProductionError(err, req);

    if (res.headersSent) {
      return next(err);
    }

    const statusCode = err.statusCode || err.status || 500;
    const isClientError = statusCode >= 400 && statusCode < 500;

    res.status(statusCode).json({
      error: {
        message: isClientError ? err.message : 'Internal Server Error',
        code: err.code || null,
        timestamp: formatTimestamp(),
      },
    });
  };
}

function createRequestLogger() {
  return function requestLogger(req, res, next) {
    if (!isProduction) return next();

    const startTime = Date.now();

    res.on('finish', () => {
      const duration = Date.now() - startTime;
      const statusCode = res.statusCode;

      if (statusCode >= 500) {
        const logEntry = {
          level: 'ERROR',
          timestamp: formatTimestamp(),
          type: 'HTTP_ERROR',
          statusCode,
          method: req.method,
          url: req.originalUrl || req.url,
          duration: `${duration}ms`,
          ip: req.ip || req.connection?.remoteAddress,
          userAgent: req.headers['user-agent'],
        };

        console.log(JSON.stringify(logEntry, null, 2));
      }
    });

    next();
  };
}

function setupProcessErrorHandlers() {
  if (!isProduction) return;

  process.on('uncaughtException', (err) => {
    const logEntry = {
      level: 'CRITICAL',
      timestamp: formatTimestamp(),
      type: 'UNCAUGHT_EXCEPTION',
      ...formatErrorDetails(err),
    };

    console.log(JSON.stringify(logEntry, null, 2));
    process.exit(1);
  });

  process.on('unhandledRejection', (reason, promise) => {
    const err = reason instanceof Error ? reason : new Error(String(reason));

    const logEntry = {
      level: 'CRITICAL',
      timestamp: formatTimestamp(),
      type: 'UNHANDLED_REJECTION',
      promise: String(promise),
      ...formatErrorDetails(err),
    };

    console.log(JSON.stringify(logEntry, null, 2));
  });

  process.on('warning', (warning) => {
    const logEntry = {
      level: 'WARNING',
      timestamp: formatTimestamp(),
      type: 'PROCESS_WARNING',
      name: warning.name,
      message: warning.message,
      stack: warning.stack,
    };

    console.log(JSON.stringify(logEntry, null, 2));
  });
}

function setupExpressApp(app) {
  setupProcessErrorHandlers();
  app.use(createRequestLogger());

  return {
    attachErrorMiddleware: () => {
      app.use(createErrorMiddleware());
    },
  };
}

module.exports = {
  logProductionError,
  logProductionWarning,
  logProductionInfo,
  createErrorMiddleware,
  createRequestLogger,
  setupProcessErrorHandlers,
  setupExpressApp,
  isProduction,
};
