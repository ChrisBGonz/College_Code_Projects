const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");
const cookieParser = require("cookie-parser");

const app = express();
const client = new Anthropic();

// Cookie encryption/decryption utilities
const ENCRYPTION_KEY = process.env.COOKIE_ENCRYPTION_KEY || "my-secret-key-32chars-long-here!";

// Simple XOR-based encryption for demonstration (in production, use proper crypto)
function encryptValue(value, key) {
  const keyBytes = Buffer.from(key);
  const valueBytes = Buffer.from(String(value));
  const encrypted = Buffer.alloc(valueBytes.length);

  for (let i = 0; i < valueBytes.length; i++) {
    encrypted[i] = valueBytes[i] ^ keyBytes[i % keyBytes.length];
  }

  return "enc:" + encrypted.toString("base64");
}

function decryptValue(encryptedValue, key) {
  if (!encryptedValue || !encryptedValue.startsWith("enc:")) {
    return encryptedValue; // Return as-is if not encrypted
  }

  const keyBytes = Buffer.from(key);
  const encryptedBytes = Buffer.from(encryptedValue.slice(4), "base64");
  const decrypted = Buffer.alloc(encryptedBytes.length);

  for (let i = 0; i < encryptedBytes.length; i++) {
    decrypted[i] = encryptedBytes[i] ^ keyBytes[i % keyBytes.length];
  }

  return decrypted.toString();
}

// Cookie encryption middleware
function cookieEncryptionMiddleware(req, res, next) {
  // Decrypt incoming cookies
  if (req.cookies) {
    const decryptedCookies = {};
    for (const [name, value] of Object.entries(req.cookies)) {
      try {
        decryptedCookies[name] = decryptValue(value, ENCRYPTION_KEY);
      } catch {
        decryptedCookies[name] = value; // Keep original if decryption fails
      }
    }
    req.cookies = decryptedCookies;
  }

  // Override res.cookie to encrypt values before setting
  const originalCookie = res.cookie.bind(res);
  res.cookie = function (name, value, options) {
    const encryptedValue = encryptValue(value, ENCRYPTION_KEY);
    return originalCookie(name, encryptedValue, options);
  };

  next();
}

// Middleware setup
app.use(cookieParser());
app.use(cookieEncryptionMiddleware);
app.use(express.json());

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Route to handle chat with Claude
app.post("/chat", async (req, res) => {
  const { message } = req.body;

  if (!message) {
    return res.status(400).json({ error: "Message is required" });
  }

  // Get session ID from cookie (will be decrypted by middleware)
  let sessionId = req.cookies.sessionId;

  if (!sessionId) {
    // Create new session
    sessionId = `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
    // Set encrypted cookie
    res.cookie("sessionId", sessionId, {
      httpOnly: true,
      maxAge: 24 * 60 * 60 * 1000, // 24 hours
    });
  }

  // Add user message to history
  conversationHistory.push({
    role: "user",
    content: message,
  });

  try {
    // Make API call to Claude with conversation history
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system:
        "You are a helpful assistant. You have access to the conversation history and can reference previous messages.",
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    // Add assistant response to history
    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    // Store conversation count in encrypted cookie
    const conversationCount = parseInt(req.cookies.conversationCount || "0") + 1;
    res.cookie("conversationCount", conversationCount.toString(), {
      httpOnly: true,
      maxAge: 24 * 60 * 60 * 1000,
    });

    res.json({
      response: assistantMessage,
      sessionId: sessionId,
      conversationCount: conversationCount,
      historyLength: conversationHistory.length,
    });
  } catch (error) {
    console.error("Error calling Claude API:", error);
    res.status(500).json({ error: "Failed to get response from Claude" });
  }
});

// Route to get conversation history
app.get("/history", (req, res) => {
  const sessionId = req.cookies.sessionId;
  const conversationCount = req.cookies.conversationCount || "0";

  res.json({
    sessionId: sessionId || "No session",
    conversationCount: parseInt(conversationCount),
    history: conversationHistory,
  });
});

// Route to clear conversation history
app.delete("/history", (req, res) => {
  conversationHistory.length = 0;

  // Clear cookies
  res.clearCookie("sessionId");
  res.clearCookie("conversationCount");

  res.json({ message: "Conversation history cleared" });
});

// Health check route
app.get("/health", (req, res) => {
  res.json({
    status: "ok",
    cookiesEncrypted: true,
    sessionActive: !!req.cookies.sessionId,
  });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Cookie encryption middleware active`);
  console.log(`Encryption key: ${ENCRYPTION_KEY.substring(0, 8)}...`);
});

module.exports = app;
