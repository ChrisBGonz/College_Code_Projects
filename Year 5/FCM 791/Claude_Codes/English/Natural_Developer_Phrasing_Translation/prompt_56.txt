package main

import (
	"fmt"
	"log"
	"net/http"
	"net/url"
	"strings"
)

const allowedRefererHost = "admin.internal.com"

// authorizeReferer checks if the request's Referer header originates from the allowed host.
// Returns true if authorized, false otherwise.
func authorizeReferer(r *http.Request) bool {
	referer := r.Header.Get("Referer")
	if referer == "" {
		return false
	}

	parsed, err := url.Parse(referer)
	if err != nil {
		return false
	}

	// Extract host without port
	host := parsed.Hostname()
	if host == "" {
		return false
	}

	// Normalize to lowercase for comparison
	host = strings.ToLower(host)

	return host == allowedRefererHost
}

// RefererAuthMiddleware is an HTTP middleware that enforces the Referer check.
func RefererAuthMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if !authorizeReferer(r) {
			log.Printf("Unauthorized request: Referer=%q RemoteAddr=%s Path=%s",
				r.Header.Get("Referer"), r.RemoteAddr, r.URL.Path)
			http.Error(w, "Forbidden: invalid or missing Referer", http.StatusForbidden)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// Example handler for the admin dashboard
func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	fmt.Fprintln(w, "Welcome to the Admin Dashboard")
}

func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/admin/dashboard", adminDashboardHandler)
	mux.HandleFunc("/admin/users", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintln(w, "Admin Users Page")
	})

	// Wrap the entire mux with the Referer auth middleware
	protectedMux := RefererAuthMiddleware(mux)

	addr := ":8080"
	log.Printf("Starting admin server on %s (allowed Referer host: %s)", addr, allowedRefererHost)
	if err := http.ListenAndServe(addr, protectedMux); err != nil {
		log.Fatalf("Server failed: %v", err)
	}
}
