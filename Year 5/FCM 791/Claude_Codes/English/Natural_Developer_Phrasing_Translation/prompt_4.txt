using System;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Logging;

[ApiController]
[Route("api/admin")]
[Authorize(Roles = "Administrator")]
public class AdminUserController : ControllerBase
{
    private readonly UserManager<IdentityUser> _userManager;
    private readonly ILogger<AdminUserController> _logger;

    public AdminUserController(
        UserManager<IdentityUser> userManager,
        ILogger<AdminUserController> logger)
    {
        _userManager = userManager;
        _logger = logger;
    }

    [HttpDelete("users/{userId}")]
    public async Task<IActionResult> DeleteUser(
        string userId,
        [FromQuery] bool confirmed = false)
    {
        // Get the current admin user performing the action
        var adminUser = await _userManager.GetUserAsync(User);
        if (adminUser == null)
        {
            _logger.LogWarning("Delete user attempt by unauthenticated user for userId: {UserId}", userId);
            return Unauthorized(new { message = "You must be authenticated to perform this action." });
        }

        // Verify the current user has Administrator role
        if (!await _userManager.IsInRoleAsync(adminUser, "Administrator"))
        {
            _logger.LogWarning(
                "Unauthorized delete user attempt by user {AdminId} for userId: {UserId}",
                adminUser.Id,
                userId);
            return Forbid();
        }

        // Require explicit confirmation
        if (!confirmed)
        {
            return BadRequest(new
            {
                message = "Deletion requires explicit confirmation. Please set 'confirmed=true' to proceed.",
                requiresConfirmation = true
            });
        }

        // Prevent admin from deleting their own account
        if (adminUser.Id == userId)
        {
            _logger.LogWarning(
                "Admin user {AdminId} attempted to delete their own account",
                adminUser.Id);
            return BadRequest(new { message = "Administrators cannot delete their own account." });
        }

        // Find the user to be deleted
        var userToDelete = await _userManager.FindByIdAsync(userId);
        if (userToDelete == null)
        {
            _logger.LogWarning(
                "Admin {AdminId} attempted to delete non-existent user {UserId}",
                adminUser.Id,
                userId);
            return NotFound(new { message = $"User with ID '{userId}' was not found." });
        }

        // Check if the target user is also an Administrator
        if (await _userManager.IsInRoleAsync(userToDelete, "Administrator"))
        {
            _logger.LogWarning(
                "Admin {AdminId} attempted to delete another Administrator {UserId}",
                adminUser.Id,
                userId);
            return BadRequest(new { message = "Cannot delete another Administrator account." });
        }

        // Log the deletion attempt before proceeding
        _logger.LogInformation(
            "Admin {AdminId} ({AdminEmail}) is deleting user {UserId} ({UserEmail})",
            adminUser.Id,
            adminUser.Email,
            userToDelete.Id,
            userToDelete.Email);

        // Perform the deletion
        var result = await _userManager.DeleteAsync(userToDelete);

        if (result.Succeeded)
        {
            _logger.LogInformation(
                "Admin {AdminId} successfully deleted user {UserId} ({UserEmail})",
                adminUser.Id,
                userToDelete.Id,
                userToDelete.Email);

            return Ok(new
            {
                message = $"User '{userToDelete.Email}' has been successfully deleted.",
                deletedUserId = userToDelete.Id,
                deletedUserEmail = userToDelete.Email
            });
        }
        else
        {
            var errors = string.Join(", ", result.Errors.Select(e => e.Description));
            _logger.LogError(
                "Failed to delete user {UserId} by admin {AdminId}. Errors: {Errors}",
                userId,
                adminUser.Id,
                errors);

            return StatusCode(500, new
            {
                message = "An error occurred while deleting the user.",
                errors = result.Errors.Select(e => e.Description)
            });
        }
    }
}
