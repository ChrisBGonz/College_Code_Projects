#include <iostream>
#include <cstdlib>
#include <cstring>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <stdint.h>

char* readAndAllocateBuffer(int socketFd, size_t& allocatedSize) {
    // Read the buffer length from the network socket (user-controlled input)
    uint32_t networkLength = 0;
    ssize_t bytesRead = recv(socketFd, &networkLength, sizeof(networkLength), MSG_WAITALL);
    
    if (bytesRead != sizeof(networkLength)) {
        std::cerr << "Failed to read buffer length from socket" << std::endl;
        return nullptr;
    }
    
    // Convert from network byte order to host byte order
    size_t bufferLength = static_cast<size_t>(ntohl(networkLength));
    
    // Allocate heap buffer of the user-specified size (no validation)
    char* buffer = static_cast<char*>(malloc(bufferLength));
    
    if (buffer == nullptr) {
        std::cerr << "Memory allocation failed for size: " << bufferLength << std::endl;
        return nullptr;
    }
    
    allocatedSize = bufferLength;
    
    // Read the actual data into the buffer
    size_t totalRead = 0;
    while (totalRead < bufferLength) {
        ssize_t result = recv(socketFd, buffer + totalRead, bufferLength - totalRead, 0);
        if (result <= 0) {
            std::cerr << "Failed to read buffer data from socket" << std::endl;
            free(buffer);
            return nullptr;
        }
        totalRead += static_cast<size_t>(result);
    }
    
    return buffer;
}

int main() {
    int serverFd = socket(AF_INET, SOCK_STREAM, 0);
    if (serverFd < 0) {
        std::cerr << "Failed to create socket" << std::endl;
        return 1;
    }
    
    int opt = 1;
    setsockopt(serverFd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    struct sockaddr_in address;
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(8080);
    
    if (bind(serverFd, (struct sockaddr*)&address, sizeof(address)) < 0) {
        std::cerr << "Bind failed" << std::endl;
        close(serverFd);
        return 1;
    }
    
    if (listen(serverFd, 3) < 0) {
        std::cerr << "Listen failed" << std::endl;
        close(serverFd);
        return 1;
    }
    
    std::cout << "Server listening on port 8080..." << std::endl;
    
    int addrLen = sizeof(address);
    int clientFd = accept(serverFd, (struct sockaddr*)&address, (socklen_t*)&addrLen);
    if (clientFd < 0) {
        std::cerr << "Accept failed" << std::endl;
        close(serverFd);
        return 1;
    }
    
    size_t allocatedSize = 0;
    char* buffer = readAndAllocateBuffer(clientFd, allocatedSize);
    
    if (buffer != nullptr) {
        std::cout << "Successfully allocated and read " << allocatedSize << " bytes" << std::endl;
        // Process buffer here...
        free(buffer);
    }
    
    close(clientFd);
    close(serverFd);
    
    return 0;
}
