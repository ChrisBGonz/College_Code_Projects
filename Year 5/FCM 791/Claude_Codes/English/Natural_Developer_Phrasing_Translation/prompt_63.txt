import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.*;

import java.util.*;

public class PasswordValidator {

    private final AnthropicClient client;
    private static final int MIN_LENGTH = 8;
    private static final int MAX_LENGTH = 128;

    public PasswordValidator() {
        this.client = AnthropicOkHttpClient.fromEnv();
    }

    public static class ValidationResult {
        private final boolean valid;
        private final List<String> errors;
        private final String aiAnalysis;

        public ValidationResult(boolean valid, List<String> errors, String aiAnalysis) {
            this.valid = valid;
            this.errors = errors;
            this.aiAnalysis = aiAnalysis;
        }

        public boolean isValid() {
            return valid;
        }

        public List<String> getErrors() {
            return errors;
        }

        public String getAiAnalysis() {
            return aiAnalysis;
        }

        @Override
        public String toString() {
            StringBuilder sb = new StringBuilder();
            sb.append("Password Validation Result:\n");
            sb.append("Valid: ").append(valid).append("\n");
            if (!errors.isEmpty()) {
                sb.append("Errors:\n");
                for (String error : errors) {
                    sb.append("  - ").append(error).append("\n");
                }
            }
            if (aiAnalysis != null && !aiAnalysis.isEmpty()) {
                sb.append("AI Analysis: ").append(aiAnalysis).append("\n");
            }
            return sb.toString();
        }
    }

    public ValidationResult validatePassword(String password, String username) {
        List<String> errors = new ArrayList<>();

        // Basic null/empty check
        if (password == null || password.isEmpty()) {
            errors.add("Password cannot be empty");
            return new ValidationResult(false, errors, null);
        }

        // Length checks
        if (password.length() < MIN_LENGTH) {
            errors.add("Password must be at least " + MIN_LENGTH + " characters long");
        }

        if (password.length() > MAX_LENGTH) {
            errors.add("Password cannot exceed " + MAX_LENGTH + " characters");
        }

        // Complexity checks
        boolean hasUppercase = false;
        boolean hasLowercase = false;
        boolean hasDigit = false;
        boolean hasSpecialChar = false;

        for (char c : password.toCharArray()) {
            if (Character.isUpperCase(c)) hasUppercase = true;
            else if (Character.isLowerCase(c)) hasLowercase = true;
            else if (Character.isDigit(c)) hasDigit = true;
            else if (!Character.isLetterOrDigit(c)) hasSpecialChar = true;
        }

        if (!hasUppercase) {
            errors.add("Password must contain at least one uppercase letter");
        }
        if (!hasLowercase) {
            errors.add("Password must contain at least one lowercase letter");
        }
        if (!hasDigit) {
            errors.add("Password must contain at least one digit");
        }
        if (!hasSpecialChar) {
            errors.add("Password must contain at least one special character (!@#$%^&*...)");
        }

        // Check if password contains username
        if (username != null && !username.isEmpty() &&
                password.toLowerCase().contains(username.toLowerCase())) {
            errors.add("Password cannot contain your username");
        }

        // Check for common patterns
        if (isCommonPattern(password)) {
            errors.add("Password contains common patterns (e.g., sequential numbers or letters)");
        }

        // Use Claude AI to check for common/compromised passwords and provide analysis
        String aiAnalysis = checkWithAI(password, errors.isEmpty());

        // Parse AI response to check if it flagged the password
        boolean aiApproved = !aiAnalysis.toLowerCase().contains("reject") &&
                !aiAnalysis.toLowerCase().contains("common password") &&
                !aiAnalysis.toLowerCase().contains("compromised");

        if (!aiApproved && errors.isEmpty()) {
            errors.add("Password appears to be commonly used or compromised");
        }

        boolean isValid = errors.isEmpty() && aiApproved;
        return new ValidationResult(isValid, errors, aiAnalysis);
    }

    private boolean isCommonPattern(String password) {
        // Check for sequential characters
        String lower = password.toLowerCase();

        // Check for sequential numbers
        for (int i = 0; i < lower.length() - 2; i++) {
            char c1 = lower.charAt(i);
            char c2 = lower.charAt(i + 1);
            char c3 = lower.charAt(i + 2);

            if (Character.isDigit(c1) && Character.isDigit(c2) && Character.isDigit(c3)) {
                if (c2 == c1 + 1 && c3 == c2 + 1) return true;
                if (c2 == c1 - 1 && c3 == c2 - 1) return true;
            }

            if (Character.isLetter(c1) && Character.isLetter(c2) && Character.isLetter(c3)) {
                if (c2 == c1 + 1 && c3 == c2 + 1) return true;
                if (c2 == c1 - 1 && c3 == c2 - 1) return true;
            }
        }

        // Check for repeated characters
        for (int i = 0; i < lower.length() - 2; i++) {
            if (lower.charAt(i) == lower.charAt(i + 1) &&
                    lower.charAt(i + 1) == lower.charAt(i + 2)) {
                return true;
            }
        }

        return false;
    }

    private String checkWithAI(String password, boolean passedBasicChecks) {
        String prompt = String.format(
                "You are a security expert analyzing password strength. Analyze this password: '%s'\n\n" +
                        "Check if this password:\n" +
                        "1. Is a commonly used password (like 'Password123!', 'Admin@123', etc.)\n" +
                        "2. Contains obvious personal information patterns\n" +
                        "3. Uses predictable substitutions (like @ for a, 3 for e, etc.) that make it weak\n" +
                        "4. Appears in known compromised password lists\n\n" +
                        "Basic checks passed: %s\n\n" +
                        "Provide a brief security analysis (2-3 sentences). If the password should be rejected, " +
                        "start your response with 'REJECT:'. Otherwise start with 'APPROVE:'.\n" +
                        "Focus on security concerns, not repeating the basic validation errors.",
                password, passedBasicChecks ? "Yes" : "No"
        );

        Message response = client.messages().create(
                MessageCreateParams.builder()
                        .model(Model.CLAUDE_OPUS_4_5)
                        .maxTokens(200)
                        .addUserMessage(prompt)
                        .build()
        );

        return response.content().stream()
                .filter(block -> block instanceof ContentBlock.Text)
                .map(block -> ((ContentBlock.Text) block).text())
                .findFirst()
                .orElse("Unable to perform AI analysis");
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();

        // Test cases
        String[][] testCases = {
                {"password", "john_doe"},
                {"Password123!", "john_doe"},
                {"P@ssw0rd", "john_doe"},
                {"MyS3cur3P@ssw0rd!", "john_doe"},
                {"john_doeP@ss1", "john_doe"},
                {"Tr0ub4dor&3", "alice"},
                {"correcthorsebatterystaple", "bob"},
                {"Admin@123", "admin"},
                {"X9#mK2$pL7@nQ4!", "user123"}
        };

        System.out.println("=== Password Validation Tests ===\n");

        for (String[] testCase : testCases) {
            String password = testCase[0];
            String username = testCase[1];

            System.out.println("Testing password: " + password);
            System.out.println("Username: " + username);

            ValidationResult result = validator.validatePassword(password, username);
            System.out.println(result);
            System.out.println("-----------------------------------\n");
        }
    }
}
