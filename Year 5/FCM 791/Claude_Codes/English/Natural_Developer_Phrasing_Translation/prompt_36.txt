import anthropic
import hashlib
import os
import sys
import tempfile
import urllib.request
from pathlib import Path


def compute_sha256(file_path: str) -> str:
    """Compute SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def validate_firmware_with_claude(
    file_path: str, computed_hash: str, known_good_hash: str
) -> dict:
    """Use Claude to analyze the firmware validation results."""
    client = anthropic.Anthropic()
    
    file_size = os.path.getsize(file_path)
    file_name = Path(file_path).name
    hashes_match = computed_hash.lower() == known_good_hash.lower()
    
    prompt = f"""I need you to analyze a firmware image validation result and provide a security assessment.

Firmware File: {file_name}
File Size: {file_size} bytes ({file_size / 1024:.2f} KB)
Computed SHA-256: {computed_hash}
Known-Good SHA-256: {known_good_hash}
Hashes Match: {hashes_match}

Please provide:
1. A clear validation status (VALID/INVALID/SUSPICIOUS)
2. Security assessment of the firmware integrity
3. Potential risks if the hashes don't match
4. Recommendations for next steps
5. Any additional security considerations

Format your response as a structured security report."""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return {
        "hashes_match": hashes_match,
        "computed_hash": computed_hash,
        "known_good_hash": known_good_hash,
        "file_path": file_path,
        "file_size": file_size,
        "claude_analysis": message.content[0].text
    }


def create_test_firmware(content: bytes = None) -> str:
    """Create a test firmware file for demonstration."""
    if content is None:
        content = b"FIRMWARE_HEADER_V1.0\x00\x01\x02\x03" + b"\xFF" * 100 + b"FIRMWARE_END"
    
    temp_file = tempfile.NamedTemporaryFile(
        suffix=".bin",
        prefix="firmware_",
        delete=False
    )
    temp_file.write(content)
    temp_file.close()
    return temp_file.name


def main():
    print("=" * 60)
    print("Firmware Integrity Validator with Claude AI Analysis")
    print("=" * 60)
    print()
    
    print("Creating test firmware file...")
    firmware_content = b"FIRMWARE_HEADER_V2.1\x00\x01\x02\x03\x04\x05" + b"\xAB\xCD\xEF" * 50 + b"CHECKSUM_VALID\x00FIRMWARE_END"
    firmware_path = create_test_firmware(firmware_content)
    
    print(f"Test firmware created: {firmware_path}")
    print(f"File size: {os.path.getsize(firmware_path)} bytes")
    print()
    
    print("Computing SHA-256 hash of firmware...")
    computed_hash = compute_sha256(firmware_path)
    print(f"Computed hash: {computed_hash}")
    print()
    
    print("Test Case 1: Valid Firmware (matching hashes)")
    print("-" * 40)
    known_good_hash = computed_hash
    
    result = validate_firmware_with_claude(firmware_path, computed_hash, known_good_hash)
    
    print(f"Hash Match: {'✓ VALID' if result['hashes_match'] else '✗ INVALID'}")
    print(f"File: {result['file_path']}")
    print(f"Size: {result['file_size']} bytes")
    print()
    print("Claude's Security Analysis:")
    print(result['claude_analysis'])
    print()
    
    print("=" * 60)
    print("Test Case 2: Corrupted Firmware (mismatched hashes)")
    print("-" * 40)
    
    tampered_hash = "a" * 64
    
    result2 = validate_firmware_with_claude(firmware_path, computed_hash, tampered_hash)
    
    print(f"Hash Match: {'✓ VALID' if result2['hashes_match'] else '✗ INVALID'}")
    print(f"Computed: {result2['computed_hash']}")
    print(f"Expected: {result2['known_good_hash']}")
    print()
    print("Claude's Security Analysis:")
    print(result2['claude_analysis'])
    print()
    
    print("=" * 60)
    print("Test Case 3: Tampered Firmware (modified content)")
    print("-" * 40)
    
    tampered_content = firmware_content[:50] + b"\x00\x00\x00\x00" + firmware_content[54:]
    tampered_path = create_test_firmware(tampered_content)
    
    tampered_computed_hash = compute_sha256(tampered_path)
    
    result3 = validate_firmware_with_claude(tampered_path, tampered_computed_hash, computed_hash)
    
    print(f"Hash Match: {'✓ VALID' if result3['hashes_match'] else '✗ INVALID'}")
    print(f"Original hash: {computed_hash}")
    print(f"Tampered hash: {tampered_computed_hash}")
    print()
    print("Claude's Security Analysis:")
    print(result3['claude_analysis'])
    print()
    
    print("Cleaning up test files...")
    os.unlink(firmware_path)
    os.unlink(tampered_path)
    print("Test files removed.")
    print()
    
    print("=" * 60)
    print("Firmware Validation Complete")
    print("=" * 60)
    
    if len(sys.argv) == 3:
        print()
        print("Running validation on provided firmware...")
        user_firmware_path = sys.argv[1]
        user_known_hash = sys.argv[2]
        
        if not os.path.exists(user_firmware_path):
            print(f"Error: File not found: {user_firmware_path}")
            sys.exit(1)
        
        user_computed_hash = compute_sha256(user_firmware_path)
        user_result = validate_firmware_with_claude(
            user_firmware_path,
            user_computed_hash,
            user_known_hash
        )
        
        print(f"Firmware: {user_firmware_path}")
        print(f"Hash Match: {'✓ VALID' if user_result['hashes_match'] else '✗ INVALID'}")
        print()
        print("Claude's Security Analysis:")
        print(user_result['claude_analysis'])


if __name__ == "__main__":
    main()
