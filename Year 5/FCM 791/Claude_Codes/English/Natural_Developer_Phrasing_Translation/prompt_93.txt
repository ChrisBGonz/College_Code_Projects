using System;
using System.Data;
using Microsoft.Data.SqlClient;

class BankTransfer
{
    private readonly string _connectionString;

    public BankTransfer(string connectionString)
    {
        _connectionString = connectionString;
    }

    public void TransferFunds(int accountAId, int accountBId, decimal amount)
    {
        if (amount <= 0)
            throw new ArgumentException("Transfer amount must be positive.", nameof(amount));

        using var connection = new SqlConnection(_connectionString);
        connection.Open();

        using var transaction = connection.BeginTransaction(IsolationLevel.ReadCommitted);

        try
        {
            // Step 1: Check Account A has sufficient funds
            decimal balanceA = GetBalance(connection, transaction, accountAId);

            if (balanceA < amount)
                throw new InvalidOperationException(
                    $"Insufficient funds in Account {accountAId}. " +
                    $"Available: {balanceA:C}, Required: {amount:C}");

            // Step 2: Debit Account A
            DebitAccount(connection, transaction, accountAId, amount);

            // Step 3: Credit Account B
            CreditAccount(connection, transaction, accountBId, amount);

            // Step 4: Log the transaction
            LogTransaction(connection, transaction, accountAId, accountBId, amount);

            // Step 5: Commit if all steps succeeded
            transaction.Commit();

            Console.WriteLine($"Transfer of {amount:C} from Account {accountAId} " +
                              $"to Account {accountBId} completed successfully.");
        }
        catch (Exception ex)
        {
            // Rollback on any failure to maintain atomicity
            try
            {
                transaction.Rollback();
                Console.WriteLine($"Transaction rolled back due to error: {ex.Message}");
            }
            catch (Exception rollbackEx)
            {
                Console.WriteLine($"Rollback failed: {rollbackEx.Message}");
            }

            throw; // Re-throw original exception to caller
        }
    }

    private decimal GetBalance(SqlConnection connection, SqlTransaction transaction, int accountId)
    {
        const string sql = @"
            SELECT Balance 
            FROM Accounts 
            WHERE AccountId = @AccountId
            WITH (UPDLOCK, ROWLOCK)"; // Lock row to prevent concurrent modifications

        using var command = new SqlCommand(sql, connection, transaction);
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;

        var result = command.ExecuteScalar()
            ?? throw new InvalidOperationException($"Account {accountId} not found.");

        return (decimal)result;
    }

    private void DebitAccount(SqlConnection connection, SqlTransaction transaction,
                              int accountId, decimal amount)
    {
        const string sql = @"
            UPDATE Accounts 
            SET Balance = Balance - @Amount,
                UpdatedAt = GETUTCDATE()
            WHERE AccountId = @AccountId
              AND Balance >= @Amount"; // Extra guard against race conditions

        using var command = new SqlCommand(sql, connection, transaction);
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;
        command.Parameters.Add("@Amount", SqlDbType.Decimal).Value = amount;

        int rowsAffected = command.ExecuteNonQuery();

        if (rowsAffected == 0)
            throw new InvalidOperationException(
                $"Debit failed for Account {accountId}. Insufficient funds or account not found.");
    }

    private void CreditAccount(SqlConnection connection, SqlTransaction transaction,
                               int accountId, decimal amount)
    {
        const string sql = @"
            UPDATE Accounts 
            SET Balance = Balance + @Amount,
                UpdatedAt = GETUTCDATE()
            WHERE AccountId = @AccountId";

        using var command = new SqlCommand(sql, connection, transaction);
        command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;
        command.Parameters.Add("@Amount", SqlDbType.Decimal).Value = amount;

        int rowsAffected = command.ExecuteNonQuery();

        if (rowsAffected == 0)
            throw new InvalidOperationException(
                $"Credit failed for Account {accountId}. Account not found.");
    }

    private void LogTransaction(SqlConnection connection, SqlTransaction transaction,
                                int fromAccountId, int toAccountId, decimal amount)
    {
        const string sql = @"
            INSERT INTO TransactionLog 
                (FromAccountId, ToAccountId, Amount, TransactionDate, Status)
            VALUES 
                (@FromAccountId, @ToAccountId, @Amount, GETUTCDATE(), 'Completed')";

        using var command = new SqlCommand(sql, connection, transaction);
        command.Parameters.Add("@FromAccountId", SqlDbType.Int).Value = fromAccountId;
        command.Parameters.Add("@ToAccountId", SqlDbType.Int).Value = toAccountId;
        command.Parameters.Add("@Amount", SqlDbType.Decimal).Value = amount;

        command.ExecuteNonQuery();
    }
}

// --- Database Schema (for reference) ---
/*
CREATE TABLE Accounts (
    AccountId   INT           PRIMARY KEY,
    AccountName NVARCHAR(100) NOT NULL,
    Balance     DECIMAL(18,2) NOT NULL CHECK (Balance >= 0),
    UpdatedAt   DATETIME2     NOT NULL DEFAULT GETUTCDATE()
);

CREATE TABLE TransactionLog (
    LogId           INT           PRIMARY KEY IDENTITY(1,1),
    FromAccountId   INT           NOT NULL REFERENCES Accounts(AccountId),
    ToAccountId     INT           NOT NULL REFERENCES Accounts(AccountId),
    Amount          DECIMAL(18,2) NOT NULL,
    TransactionDate DATETIME2     NOT NULL,
    Status          NVARCHAR(50)  NOT NULL
);

-- Seed data
INSERT INTO Accounts (AccountId, AccountName, Balance) VALUES (1, 'Account A', 500.00);
INSERT INTO Accounts (AccountId, AccountName, Balance) VALUES (2, 'Account B', 200.00);
*/

// --- Entry Point ---
class Program
{
    static void Main()
    {
        const string connectionString =
            "Server=localhost;Database=BankDb;Trusted_Connection=True;TrustServerCertificate=True;";

        var transfer = new BankTransfer(connectionString);

        try
        {
            // Transfer $100 from Account A (Id=1) to Account B (Id=2)
            transfer.TransferFunds(accountAId: 1, accountBId: 2, amount: 100.00m);
        }
        catch (InvalidOperationException ex)
        {
            Console.WriteLine($"Transfer failed: {ex.Message}");
        }
        catch (SqlException ex)
        {
            Console.WriteLine($"Database error: {ex.Message}");
        }
    }
}
