```ruby
# app/controllers/admin_panel_controller.rb

class AdminPanelController < ApplicationController
  before_action :authenticate_admin!

  def index
    @stats = {
      total_users: 1_234,
      total_orders: 5_678,
      revenue: 98_765.43
    }
    render json: {
      message: "Welcome to the Admin Panel",
      stats: @stats
    }
  end

  def users
    @users = [
      { id: 1, name: "Alice", email: "alice@example.com", role: "user" },
      { id: 2, name: "Bob",   email: "bob@example.com",   role: "user" },
      { id: 3, name: "Carol", email: "carol@example.com", role: "admin" }
    ]
    render json: { users: @users }
  end

  def orders
    @orders = [
      { id: 101, user_id: 1, total: 49.99,  status: "completed" },
      { id: 102, user_id: 2, total: 129.00, status: "pending"   },
      { id: 103, user_id: 1, total: 19.99,  status: "cancelled" }
    ]
    render json: { orders: @orders }
  end

  def settings
    render json: {
      settings: {
        maintenance_mode: false,
        max_upload_size_mb: 10,
        allowed_file_types: %w[jpg png pdf]
      }
    }
  end

  def update_settings
    render json: {
      message: "Settings updated successfully",
      updated_settings: params[:settings]
    }
  end

  def delete_user
    user_id = params[:id]
    render json: { message: "User #{user_id} deleted successfully" }
  end

  private

  def authenticate_admin!
    unless admin_cookie_present?
      render json: { error: "Unauthorized: Admin access required" }, status: :unauthorized
    end
  end

  def admin_cookie_present?
    cookies[:is_admin] == "true"
  end
end
```

```ruby
# config/routes.rb (relevant admin routes to add)

Rails.application.routes.draw do
  namespace :admin do
    get  "panel",          to: "admin_panel#index"
    get  "panel/users",    to: "admin_panel#users"
    get  "panel/orders",   to: "admin_panel#orders"
    get  "panel/settings", to: "admin_panel#settings"
    put  "panel/settings", to: "admin_panel#update_settings"
    delete "panel/users/:id", to: "admin_panel#delete_user"
  end
end
```
