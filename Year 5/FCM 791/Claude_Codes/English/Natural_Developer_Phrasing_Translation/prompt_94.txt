import Anthropic from "@anthropic-ai/sdk";
import express from "express";

const client = new Anthropic();

async function generateErrorHandlingMiddleware(): Promise<string> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Generate a comprehensive global Express error-handling middleware for a Node.js app that:
1. Catches any unhandled errors/exceptions
2. Returns proper error responses to the client
3. Handles different types of errors (validation errors, authentication errors, not found errors, etc.)
4. Includes proper HTTP status codes
5. Logs errors appropriately
6. Handles both development and production environments (showing stack traces only in development)
7. Handles async errors
8. Includes TypeScript types

Please provide only the middleware code without any explanation or markdown formatting.`,
      },
    ],
  });

  const textContent = message.content.find((block) => block.type === "text");
  if (!textContent || textContent.type !== "text") {
    throw new Error("No text content in response");
  }

  return textContent.text;
}

async function demonstrateMiddleware(): Promise<void> {
  console.log("Generating Express error-handling middleware using Claude AI...\n");

  const middlewareCode = await generateErrorHandlingMiddleware();

  console.log("Generated Error-Handling Middleware:");
  console.log("=".repeat(50));
  console.log(middlewareCode);
  console.log("=".repeat(50));

  // Create a demo Express app to show the middleware in action
  const app = express();
  app.use(express.json());

  // Custom error class for demonstration
  class AppError extends Error {
    statusCode: number;
    isOperational: boolean;

    constructor(message: string, statusCode: number) {
      super(message);
      this.statusCode = statusCode;
      this.isOperational = true;
      Error.captureStackTrace(this, this.constructor);
    }
  }

  // Demo routes that throw different types of errors
  app.get("/test-error", (_req, _res, next) => {
    next(new AppError("Test operational error", 400));
  });

  app.get("/test-async-error", async (_req, _res, next) => {
    try {
      throw new Error("Async error occurred");
    } catch (error) {
      next(error);
    }
  });

  app.get("/test-not-found", (_req, _res, next) => {
    next(new AppError("Resource not found", 404));
  });

  // Global error handling middleware
  app.use(
    (
      err: Error & { statusCode?: number; isOperational?: boolean },
      _req: express.Request,
      res: express.Response,
      _next: express.NextFunction
    ) => {
      const statusCode = err.statusCode || 500;
      const isProduction = process.env.NODE_ENV === "production";

      const errorResponse: {
        status: string;
        message: string;
        stack?: string;
        error?: Error;
      } = {
        status: statusCode >= 500 ? "error" : "fail",
        message: err.message || "Internal Server Error",
      };

      if (!isProduction) {
        errorResponse.stack = err.stack;
        errorResponse.error = err;
      }

      console.error(`[ERROR] ${statusCode}: ${err.message}`);
      if (!isProduction && err.stack) {
        console.error(err.stack);
      }

      res.status(statusCode).json(errorResponse);
    }
  );

  // Start the server for demonstration
  const PORT = 3000;
  const server = app.listen(PORT, () => {
    console.log(`\nDemo server running on port ${PORT}`);
    console.log("\nTesting error handling middleware...\n");

    // Test the error handling by making requests
    testErrorHandling(PORT).then(() => {
      server.close(() => {
        console.log("\nDemo server closed");
      });
    });
  });
}

async function testErrorHandling(port: number): Promise<void> {
  const testCases = [
    { path: "/test-error", description: "Operational Error (400)" },
    { path: "/test-async-error", description: "Async Error (500)" },
    { path: "/test-not-found", description: "Not Found Error (404)" },
    { path: "/nonexistent", description: "Route Not Found" },
  ];

  for (const testCase of testCases) {
    try {
      const response = await fetch(`http://localhost:${port}${testCase.path}`);
      const data = await response.json();

      console.log(`Test: ${testCase.description}`);
      console.log(`Status: ${response.status}`);
      console.log(`Response:`, JSON.stringify(data, null, 2));
      console.log("-".repeat(40));
    } catch (error) {
      console.log(`Test: ${testCase.description}`);
      console.log(`Error making request:`, error);
      console.log("-".repeat(40));
    }
  }
}

// Main execution
demonstrateMiddleware().catch(console.error);
