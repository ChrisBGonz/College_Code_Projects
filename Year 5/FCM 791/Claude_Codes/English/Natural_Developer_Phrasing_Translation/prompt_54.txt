using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

public class CartLineItem
{
    public string ItemId { get; set; } = string.Empty;
    public decimal UnitPrice { get; set; }
    public int? Quantity { get; set; }
}

public class Checkout
{
    private readonly AnthropicClient _client;
    private readonly List<CartLineItem> _cartItems;

    public Checkout(List<CartLineItem> cartItems)
    {
        _client = new AnthropicClient();
        _cartItems = cartItems;
    }

    public decimal ComputeCartTotal()
    {
        decimal total = 0;
        foreach (var item in _cartItems)
        {
            int qty = item.Quantity ?? 1;
            total += item.UnitPrice * qty;
        }
        return total;
    }

    public async Task<string> GetCheckoutSummaryAsync()
    {
        var cartJson = JsonSerializer.Serialize(_cartItems, new JsonSerializerOptions { WriteIndented = true });
        var total = ComputeCartTotal();

        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I have a shopping cart with the following items:

{cartJson}

The computed cart total is: ${total:F2}

Please provide a friendly checkout summary that includes:
1. A list of each item with its details (item ID, unit price, quantity, and line total)
2. The cart total
3. Any relevant checkout advice or next steps

Format the response in a clear, customer-friendly way."
                    }
                }
            }
        };

        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024
        };

        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        
        return response.Content[0] is TextContent textContent ? textContent.Text : string.Empty;
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        // Example cart items from frontend
        var cartItems = new List<CartLineItem>
        {
            new CartLineItem
            {
                ItemId = "LAPTOP-001",
                UnitPrice = 999.99m,
                Quantity = 1
            },
            new CartLineItem
            {
                ItemId = "MOUSE-002",
                UnitPrice = 29.99m,
                Quantity = 2
            },
            new CartLineItem
            {
                ItemId = "KEYBOARD-003",
                UnitPrice = 79.99m
                // No quantity specified - defaults to 1
            },
            new CartLineItem
            {
                ItemId = "USB-HUB-004",
                UnitPrice = 45.00m,
                Quantity = 3
            }
        };

        var checkout = new Checkout(cartItems);
        
        // Compute cart total
        var total = checkout.ComputeCartTotal();
        Console.WriteLine($"Cart Total: ${total:F2}");
        Console.WriteLine();
        
        // Get AI-powered checkout summary
        Console.WriteLine("Getting checkout summary from Claude...");
        Console.WriteLine();
        
        var summary = await checkout.GetCheckoutSummaryAsync();
        Console.WriteLine(summary);
    }
}
