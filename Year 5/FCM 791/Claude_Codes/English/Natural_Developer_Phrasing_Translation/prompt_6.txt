# Simulated user authentication database
USER_AUTH_DATABASE = {
    "alice@techcorp.com": {"password": "pass123", "tenant_id": "tenant_001"},
    "bob@techcorp.com": {"password": "pass456", "tenant_id": "tenant_001"},
    "charlie@techcorp.com": {"password": "pass789", "tenant_id": "tenant_001"},
    "diana@greenenergy.com": {"password": "green123", "tenant_id": "tenant_002"},
    "evan@greenenergy.com": {"password": "green456", "tenant_id": "tenant_002"},
    "frank@healthfirst.com": {"password": "health123", "tenant_id": "tenant_003"},
    "grace@healthfirst.com": {"password": "health456", "tenant_id": "tenant_003"},
    "henry@healthfirst.com": {"password": "health789", "tenant_id": "tenant_003"}
}

def authenticate_user(email: str, password: str) -> dict | None:
    """Authenticate a user and return their session info."""
    if email in USER_AUTH_DATABASE:
        user_data = USER_AUTH_DATABASE[email]
        if user_data["password"] == password:
            return {
                "email": email,
                "tenant_id": user_data["tenant_id"],
                "authenticated": True
            }
    return None

def get_annual_report(tenant_id: str, requesting_tenant_id: str) -> dict:
    """
    Fetch annual report with tenant isolation enforcement.
    Only allows access if the requesting tenant matches the report's tenant.
    """
    # Enforce tenant isolation
    if tenant_id != requesting_tenant_id:
        return {
            "error": "Access Denied",
            "message": f"You are not authorized to access reports for tenant {tenant_id}. You can only access your own company's reports.",
            "your_tenant_id": requesting_tenant_id,
            "requested_tenant_id": tenant_id
        }
    
    # Check if tenant exists
    if tenant_id not in TENANT_DATABASE:
        return {
            "error": "Not Found",
            "message": f"No tenant found with ID: {tenant_id}"
        }
    
    tenant_data = TENANT_DATABASE[tenant_id]
    return {
        "success": True,
        "company_name": tenant_data["company_name"],
        "tenant_id": tenant_id,
        "annual_report": tenant_data["annual_report"]
    }

def list_available_tenants(requesting_tenant_id: str) -> dict:
    """List only the requesting user's own tenant information."""
    if requesting_tenant_id in TENANT_DATABASE:
        tenant = TENANT_DATABASE[requesting_tenant_id]
        return {
            "success": True,
            "your_tenant": {
                "tenant_id": requesting_tenant_id,
                "company_name": tenant["company_name"]
            },
            "note": "You can only view your own company's information."
        }
    return {
        "error": "Not Found",
        "message": "Your tenant information could not be found."
    }

# Define tools for Claude
TOOLS = [
    {
        "name": "get_annual_report",
        "description": "Fetch the annual report for a specific company/tenant. Enforces tenant isolation - users can only access their own company's reports.",
        "input_schema": {
            "type": "object",
            "properties": {
                "tenant_id": {
                    "type": "string",
                    "description": "The tenant/company ID to fetch the annual report for"
                }
            },
            "required": ["tenant_id"]
        }
    },
    {
        "name": "list_available_tenants",
        "description": "List the tenant information available to the current user (only their own company).",
        "input_schema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    }
]

def process_tool_call(tool_name: str, tool_input: dict, user_session: dict) -> str:
    """Process tool calls with tenant isolation enforcement."""
    if tool_name == "get_annual_report":
        result = get_annual_report(
            tenant_id=tool_input["tenant_id"],
            requesting_tenant_id=user_session["tenant_id"]
        )
        return json.dumps(result, indent=2)
    elif tool_name == "list_available_tenants":
        result = list_available_tenants(user_session["tenant_id"])
        return json.dumps(result, indent=2)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def chat_with_claude(user_session: dict, conversation_history: list, user_message: str) -> tuple[str, list]:
    """
    Send a message to Claude and handle tool calls with tenant isolation.
    Returns the assistant's response and updated conversation history.
    """
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # System prompt that enforces tenant isolation context
    system_prompt = f"""You are a helpful assistant for a multi-tenant SaaS application. 
    
Current authenticated user:
- Email: {user_session['email']}
- Company Tenant ID: {user_session['tenant_id']}
- Company Name: {TENANT_DATABASE.get(user_session['tenant_id'], {}).get('company_name', 'Unknown')}

IMPORTANT SECURITY RULES:
1. You MUST enforce tenant isolation at all times
2. The user can ONLY access their own company's data (tenant_id: {user_session['tenant_id']})
3. If a user requests data from another tenant, you must deny the request
4. Always verify that the tenant_id in tool calls matches the user's tenant_id
5. Never reveal data from other tenants

You can help users:
- View their company's annual report
- Understand their company's financial performance
- Answer questions about their annual report data

Use the available tools to fetch and display information, but always enforce tenant isolation."""
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            system=system_prompt,
            tools=TOOLS,
            messages=conversation_history
        )
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": response.content
        })
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n[System] Executing tool: {tool_name}")
                    print(f"[System] Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    # Process the tool call with tenant isolation
                    result = process_tool_call(tool_name, tool_input, user_session)
                    
                    print(f"[System] Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            # Add tool results to conversation history
            conversation_history.append({
                "role": "user",
                "content": tool_results
            })
            
            # Continue the loop to get Claude's response after tool execution
            continue
        
        # Extract text response
        text_response = ""
        for content_block in response.content:
            if hasattr(content_block, 'text'):
                text_response += content_block.text
        
        return text_response, conversation_history
