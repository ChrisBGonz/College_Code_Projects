import com.sun.org.apache.xerces.internal.impl.Constants;
import org.w3c.dom.Document;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import javax.xml.XMLConstants;
import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import java.io.IOException;
import java.io.StringReader;

import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.*;
import software.amazon.awssdk.regions.Region;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.JsonNode;

public class SecureXMLParser {

    /**
     * Parses an untrusted XML string using DocumentBuilderFactory with secure defaults.
     * Prevents XXE (XML External Entity) attacks and DTD-based attacks.
     *
     * @param xmlString The untrusted XML string to parse
     * @return Parsed Document object
     * @throws ParserConfigurationException if parser configuration fails
     * @throws SAXException                 if XML parsing fails
     * @throws IOException                  if I/O error occurs
     */
    public static Document parseSecureXML(String xmlString)
            throws ParserConfigurationException, SAXException, IOException {

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();

        // Disable DOCTYPE declarations entirely
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);

        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);

        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);

        // Disable external DTD loading
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);

        // Enable secure processing
        factory.setFeature(XMLConstants.FEATURE_SECURE_PROCESSING, true);

        // Disable XInclude processing
        factory.setXIncludeAware(false);

        // Disable namespace awareness (optional, enable if needed)
        factory.setExpandEntityReferences(false);

        // Set additional security properties
        try {
            factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
            factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_SCHEMA, "");
        } catch (IllegalArgumentException e) {
            // Some parsers may not support these attributes, log and continue
            System.err.println("Warning: Could not set external access attributes: " + e.getMessage());
        }

        DocumentBuilder builder = factory.newDocumentBuilder();

        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Demonstrates the secure XML parsing with Claude AI assistance for validation.
     */
    public static void main(String[] args) {
        // Example safe XML
        String safeXml = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                "<root>" +
                "<element>Safe content</element>" +
                "<data>123</data>" +
                "</root>";

        // Example potentially malicious XML (XXE attack attempt)
        String maliciousXml = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                "<!DOCTYPE foo [<!ENTITY xxe SYSTEM \"file:///etc/passwd\">]>" +
                "<root><element>&xxe;</element></root>";

        System.out.println("Testing Secure XML Parser with Claude AI Analysis");
        System.out.println("=================================================\n");

        // Test safe XML
        System.out.println("Test 1: Parsing safe XML...");
        try {
            Document doc = parseSecureXML(safeXml);
            System.out.println("✓ Safe XML parsed successfully");
            System.out.println("  Root element: " + doc.getDocumentElement().getTagName());
            analyzeWithClaude("Safe XML parsing", safeXml, "Success", null);
        } catch (Exception e) {
            System.out.println("✗ Unexpected error parsing safe XML: " + e.getMessage());
        }

        System.out.println();

        // Test malicious XML
        System.out.println("Test 2: Attempting to parse malicious XML (XXE attack)...");
        try {
            Document doc = parseSecureXML(maliciousXml);
            System.out.println("✗ WARNING: Malicious XML was parsed without error!");
            analyzeWithClaude("Malicious XML parsing", maliciousXml, "Unexpected success - security concern!", null);
        } catch (SAXException e) {
            System.out.println("✓ Malicious XML correctly rejected: " + e.getMessage());
            analyzeWithClaude("Malicious XML parsing", maliciousXml, "Correctly rejected", e.getMessage());
        } catch (Exception e) {
            System.out.println("✓ Malicious XML rejected with error: " + e.getMessage());
            analyzeWithClaude("Malicious XML parsing", maliciousXml, "Rejected", e.getMessage());
        }

        System.out.println();

        // Test with custom XML provided by user
        String customXml = "<?xml version=\"1.0\"?><config><setting name=\"debug\">false</setting></config>";
        System.out.println("Test 3: Parsing custom configuration XML...");
        try {
            Document doc = parseSecureXML(customXml);
            System.out.println("✓ Custom XML parsed successfully");
            analyzeWithClaude("Custom XML parsing", customXml, "Success", null);
        } catch (Exception e) {
            System.out.println("✗ Error parsing custom XML: " + e.getMessage());
        }
    }

    /**
     * Uses Claude AI to analyze XML parsing results and provide security insights.
     */
    private static void analyzeWithClaude(String testName, String xml, String result, String errorMessage) {
        try {
            BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                    .region(Region.US_EAST_1)
                    .build();

            String prompt = String.format(
                    "Analyze this XML security test result:\n\n" +
                            "Test: %s\n" +
                            "XML Content: %s\n" +
                            "Result: %s\n" +
                            "%s\n\n" +
                            "Provide a brief security analysis (2-3 sentences) explaining:\n" +
                            "1. What security risk was being tested\n" +
                            "2. Whether the parser handled it correctly\n" +
                            "3. Any additional security recommendations",
                    testName,
                    xml.length() > 200 ? xml.substring(0, 200) + "..." : xml,
                    result,
                    errorMessage != null ? "Error: " + errorMessage : ""
            );

            ContentBlock contentBlock = ContentBlock.fromText(prompt);

            Message message = Message.builder()
                    .role(ConversationRole.USER)
                    .content(contentBlock)
                    .build();

            ConverseRequest request = ConverseRequest.builder()
                    .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
                    .messages(message)
                    .inferenceConfig(InferenceConfiguration.builder()
                            .maxTokens(300)
                            .temperature(0.1f)
                            .build())
                    .build();

            ConverseResponse response = client.converse(request);

            String analysisText = response.output().message().content().get(0).text();

            System.out.println("\n  Claude AI Security Analysis:");
            System.out.println("  " + analysisText.replace("\n", "\n  "));

        } catch (Exception e) {
            System.out.println("\n  (Claude AI analysis unavailable: " + e.getMessage() + ")");
        }
    }
}
