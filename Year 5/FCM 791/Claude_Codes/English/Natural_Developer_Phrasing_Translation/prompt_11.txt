import sys
from lxml import etree


def extract_title_from_xml(file_path: str) -> str | None:
    """
    Parse an XML file and extract the contents of the <title> element.
    
    Args:
        file_path: Path to the XML file to parse
        
    Returns:
        The text content of the <title> element, or None if not found
    """
    try:
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        # Search for title element anywhere in the document
        # First try direct child
        title_element = root.find('title')
        
        # If not found as direct child, search recursively
        if title_element is None:
            title_element = root.find('.//title')
        
        # Also try case-insensitive search if still not found
        if title_element is None:
            for element in root.iter():
                if element.tag.lower() == 'title':
                    title_element = element
                    break
        
        if title_element is not None:
            # Get all text content including tail text of child elements
            title_text = ''.join(title_element.itertext()).strip()
            return title_text if title_text else None
        
        return None
        
    except etree.XMLSyntaxError as e:
        print(f"Error: Invalid XML syntax - {e}", file=sys.stderr)
        raise
    except FileNotFoundError:
        print(f"Error: File not found - {file_path}", file=sys.stderr)
        raise
    except PermissionError:
        print(f"Error: Permission denied - {file_path}", file=sys.stderr)
        raise


def extract_all_titles_from_xml(file_path: str) -> list[str]:
    """
    Parse an XML file and extract the contents of all <title> elements.
    
    Args:
        file_path: Path to the XML file to parse
        
    Returns:
        A list of text contents from all <title> elements
    """
    try:
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        titles = []
        
        # Find all title elements (case-insensitive)
        for element in root.iter():
            if element.tag.lower() == 'title':
                title_text = ''.join(element.itertext()).strip()
                if title_text:
                    titles.append(title_text)
        
        return titles
        
    except etree.XMLSyntaxError as e:
        print(f"Error: Invalid XML syntax - {e}", file=sys.stderr)
        raise
    except FileNotFoundError:
        print(f"Error: File not found - {file_path}", file=sys.stderr)
        raise


def process_xml_file(file_path: str) -> dict:
    """
    Process an XML file and return information about found title elements.
    
    Args:
        file_path: Path to the XML file to parse
        
    Returns:
        A dictionary containing:
            - 'first_title': The first title found (or None)
            - 'all_titles': List of all titles found
            - 'count': Number of title elements found
    """
    all_titles = extract_all_titles_from_xml(file_path)
    
    return {
        'first_title': all_titles[0] if all_titles else None,
        'all_titles': all_titles,
        'count': len(all_titles)
    }


def main():
    if len(sys.argv) < 2:
        print("Usage: python solution.py <xml_file_path>")
        print("\nExample: python solution.py sample.xml")
        sys.exit(1)
    
    file_path = sys.argv[1]
    
    print(f"Processing XML file: {file_path}")
    print("-" * 50)
    
    result = process_xml_file(file_path)
    
    if result['count'] == 0:
        print("No <title> elements found in the XML file.")
    elif result['count'] == 1:
        print(f"Found 1 <title> element:")
        print(f"  Title: {result['first_title']}")
    else:
        print(f"Found {result['count']} <title> elements:")
        for i, title in enumerate(result['all_titles'], 1):
            print(f"  Title {i}: {title}")
        print(f"\nFirst title: {result['first_title']}")


if __name__ == "__main__":
    main()
