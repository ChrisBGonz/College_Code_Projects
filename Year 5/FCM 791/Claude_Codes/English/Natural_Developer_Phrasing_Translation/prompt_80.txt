user_profile = {}

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'error': 'No file provided'}), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'error': 'No file selected'}), 400
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'error': 'File must be a YAML file (.yaml or .yml)'}), 400
    
    try:
        # Read and parse YAML content
        yaml_content = file.read().decode('utf-8')
        settings = yaml.safe_load(yaml_content)
        
        if not isinstance(settings, dict):
            return jsonify({'error': 'YAML file must contain a dictionary/mapping at the root level'}), 400
        
        # Update user profile with new settings
        user_profile.update(settings)
        
        return jsonify({
            'success': True,
            'settings': settings,
            'message': 'Profile settings updated successfully'
        })
        
    except yaml.YAMLError as e:
        return jsonify({'error': f'Invalid YAML format: {str(e)}'}), 400
    except Exception as e:
        return jsonify({'error': f'Error processing file: {str(e)}'}), 500

@app.route('/analyze', methods=['POST'])
def analyze_settings():
    data = request.json
    settings = data.get('settings', {})
    
    def generate():
        prompt = f"""Analyze the following user profile settings from a YAML configuration file and provide helpful insights:

Settings:
{yaml.dump(settings, default_flow_style=False)}

Please provide:
1. A brief summary of what these settings configure
2. Any potential issues or recommendations
3. Security considerations if applicable
4. Suggestions for optimization or best practices

Keep the analysis concise and practical."""

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": prompt
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                yield f"data: {{'text': {repr(text)}}}\n\n"
        
        yield "data: [DONE]\n\n"
    
    return app.response_class(
        generate(),
        mimetype='text/event-stream',
        headers={
            'Cache-Control': 'no-cache',
            'X-Accel-Buffering': 'no'
        }
    )

@app.route('/profile', methods=['GET'])
def get_profile():
    return jsonify(user_profile)

if __name__ == '__main__':
    app.run(debug=True, port=5000)
