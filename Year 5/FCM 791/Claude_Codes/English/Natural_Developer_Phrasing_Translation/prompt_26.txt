```yaml
name: CI Build and Publish Artifact

on:
  push:
    branches:
      - main
      - master
      - 'release/**'
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:
    inputs:
      environment:
        description: 'Target environment'
        required: true
        default: 'staging'
        type: choice
        options:
          - staging
          - production

env:
  NODE_VERSION: '20'
  PYTHON_VERSION: '3.11'
  ARTIFACT_NAME: 'build-artifact'
  BUILD_DIR: 'dist'
  S3_BUCKET: ${{ secrets.S3_BUCKET_NAME }}
  GCS_BUCKET: ${{ secrets.GCS_BUCKET_NAME }}
  AZURE_CONTAINER: ${{ secrets.AZURE_STORAGE_CONTAINER }}
  STORAGE_PROVIDER: 's3'

jobs:
  # ─────────────────────────────────────────────
  # Job 1: Lint and Test
  # ─────────────────────────────────────────────
  lint-and-test:
    name: Lint and Test
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run linter
        run: npm run lint --if-present

      - name: Run unit tests
        run: npm test --if-present
        env:
          CI: true

      - name: Upload test results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: test-results
          path: |
            coverage/
            test-results/
          retention-days: 7

  # ─────────────────────────────────────────────
  # Job 2: Build
  # ─────────────────────────────────────────────
  build:
    name: Build Artifact
    runs-on: ubuntu-latest
    needs: lint-and-test
    timeout-minutes: 20
    outputs:
      artifact-version: ${{ steps.versioning.outputs.version }}
      artifact-filename: ${{ steps.package.outputs.filename }}
      build-timestamp: ${{ steps.versioning.outputs.timestamp }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Determine version and metadata
        id: versioning
        run: |
          TIMESTAMP=$(date -u +"%Y%m%dT%H%M%SZ")
          SHORT_SHA=$(git rev-parse --short HEAD)
          BRANCH_NAME="${GITHUB_REF_NAME//\//-}"
          
          if [[ "$GITHUB_REF" == refs/tags/* ]]; then
            VERSION="${GITHUB_REF_NAME}"
          else
            VERSION="${BRANCH_NAME}-${SHORT_SHA}-${TIMESTAMP}"
          fi
          
          echo "version=${VERSION}" >> "$GITHUB_OUTPUT"
          echo "timestamp=${TIMESTAMP}" >> "$GITHUB_OUTPUT"
          echo "short-sha=${SHORT_SHA}" >> "$GITHUB_OUTPUT"
          echo "branch=${BRANCH_NAME}" >> "$GITHUB_OUTPUT"
          
          echo "Build version: ${VERSION}"
          echo "Timestamp: ${TIMESTAMP}"

      - name: Build project
        run: npm run build --if-present
        env:
          NODE_ENV: production
          BUILD_VERSION: ${{ steps.versioning.outputs.version }}

      - name: Verify build output
        run: |
          if [ ! -d "${{ env.BUILD_DIR }}" ]; then
            echo "ERROR: Build directory '${{ env.BUILD_DIR }}' not found!"
            exit 1
          fi
          echo "Build directory contents:"
          ls -lah "${{ env.BUILD_DIR }}"

      - name: Write build metadata
        run: |
          cat > "${{ env.BUILD_DIR }}/build-info.json" <<EOF
          {
            "version": "${{ steps.versioning.outputs.version }}",
            "timestamp": "${{ steps.versioning.outputs.timestamp }}",
            "commit": "${{ github.sha }}",
            "branch": "${{ steps.versioning.outputs.branch }}",
            "workflow": "${{ github.workflow }}",
            "run_id": "${{ github.run_id }}",
            "run_number": "${{ github.run_number }}",
            "actor": "${{ github.actor }}",
            "repository": "${{ github.repository }}"
          }
          EOF

      - name: Package build artifact
        id: package
        run: |
          VERSION="${{ steps.versioning.outputs.version }}"
          FILENAME="${{ env.ARTIFACT_NAME }}-${VERSION}.tar.gz"
          
          tar -czf "${FILENAME}" -C "${{ env.BUILD_DIR }}" .
          
          CHECKSUM=$(sha256sum "${FILENAME}" | awk '{print $1}')
          echo "${CHECKSUM}  ${FILENAME}" > "${FILENAME}.sha256"
          
          echo "filename=${FILENAME}" >> "$GITHUB_OUTPUT"
          echo "checksum=${CHECKSUM}" >> "$GITHUB_OUTPUT"
          
          echo "Packaged artifact: ${FILENAME}"
          echo "SHA-256 checksum: ${CHECKSUM}"
          ls -lah "${FILENAME}"

      - name: Upload artifact to GitHub Actions
        uses: actions/upload-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ steps.versioning.outputs.version }}
          path: |
            ${{ steps.package.outputs.filename }}
            ${{ steps.package.outputs.filename }}.sha256
          retention-days: 30
          if-no-files-found: error

  # ─────────────────────────────────────────────
  # Job 3a: Publish to AWS S3
  # ─────────────────────────────────────────────
  publish-s3:
    name: Publish to AWS S3
    runs-on: ubuntu-latest
    needs: build
    if: |
      (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master' || startsWith(github.ref, 'refs/tags/')) &&
      vars.STORAGE_PROVIDER == 's3'
    timeout-minutes: 10
    environment:
      name: ${{ github.event.inputs.environment || 'staging' }}

    permissions:
      id-token: write
      contents: read

    steps:
      - name: Download artifact from GitHub Actions
        uses: actions/download-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ needs.build.outputs.artifact-version }}
          path: ./artifacts

      - name: Configure AWS credentials (OIDC)
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_IAM_ROLE_ARN }}
          role-session-name: GitHubActions-${{ github.run_id }}
          aws-region: ${{ secrets.AWS_REGION || 'us-east-1' }}

      - name: Verify artifact integrity
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          echo "Verifying checksum for: ${FILENAME}"
          sha256sum -c "${FILENAME}.sha256"
          echo "Checksum verification passed."

      - name: Upload artifact to S3
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          TIMESTAMP="${{ needs.build.outputs.build-timestamp }}"
          S3_PREFIX="artifacts/${VERSION}"
          
          echo "Uploading ${FILENAME} to s3://${S3_BUCKET}/${S3_PREFIX}/"
          
          aws s3 cp "${FILENAME}" "s3://${S3_BUCKET}/${S3_PREFIX}/${FILENAME}" \
            --metadata "version=${VERSION},commit=${{ github.sha }},timestamp=${TIMESTAMP},workflow=${{ github.workflow }}" \
            --storage-class STANDARD_IA \
            --no-progress
          
          aws s3 cp "${FILENAME}.sha256" "s3://${S3_BUCKET}/${S3_PREFIX}/${FILENAME}.sha256" \
            --no-progress
          
          echo "Upload complete."
          echo "S3 URI: s3://${S3_BUCKET}/${S3_PREFIX}/${FILENAME}"

      - name: Update latest pointer in S3
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          
          echo "{\"latest\": \"${VERSION}\", \"filename\": \"${FILENAME}\", \"timestamp\": \"${{ needs.build.outputs.build-timestamp }}\"}" \
            | aws s3 cp - "s3://${S3_BUCKET}/artifacts/latest.json" \
              --content-type "application/json"
          
          echo "Updated latest pointer."

      - name: Verify S3 upload
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          S3_KEY="artifacts/${VERSION}/${FILENAME}"
          
          aws s3api head-object \
            --bucket "${S3_BUCKET}" \
            --key "${S3_KEY}" \
            --query '{ContentLength: ContentLength, LastModified: LastModified}' \
            --output table
          
          echo "S3 upload verified successfully."

  # ─────────────────────────────────────────────
  # Job 3b: Publish to Google Cloud Storage
  # ─────────────────────────────────────────────
  publish-gcs:
    name: Publish to Google Cloud Storage
    runs-on: ubuntu-latest
    needs: build
    if: |
      (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master' || startsWith(github.ref, 'refs/tags/')) &&
      vars.STORAGE_PROVIDER == 'gcs'
    timeout-minutes: 10
    environment:
      name: ${{ github.event.inputs.environment || 'staging' }}

    permissions:
      id-token: write
      contents: read

    steps:
      - name: Download artifact from GitHub Actions
        uses: actions/download-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ needs.build.outputs.artifact-version }}
          path: ./artifacts

      - name: Authenticate to Google Cloud (Workload Identity Federation)
        uses: google-github-actions/auth@v2
        with:
          workload_identity_provider: ${{ secrets.GCP_WORKLOAD_IDENTITY_PROVIDER }}
          service_account: ${{ secrets.GCP_SERVICE_ACCOUNT }}

      - name: Set up Cloud SDK
        uses: google-github-actions/setup-gcloud@v2

      - name: Verify artifact integrity
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          sha256sum -c "${FILENAME}.sha256"
          echo "Checksum verification passed."

      - name: Upload artifact to GCS
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          GCS_PREFIX="artifacts/${VERSION}"
          
          echo "Uploading ${FILENAME} to gs://${GCS_BUCKET}/${GCS_PREFIX}/"
          
          gcloud storage cp "${FILENAME}" "gs://${GCS_BUCKET}/${GCS_PREFIX}/${FILENAME}" \
            --custom-metadata="version=${VERSION},commit=${{ github.sha }},timestamp=${{ needs.build.outputs.build-timestamp }}"
          
          gcloud storage cp "${FILENAME}.sha256" "gs://${GCS_BUCKET}/${GCS_PREFIX}/${FILENAME}.sha256"
          
          echo "Upload complete."
          echo "GCS URI: gs://${GCS_BUCKET}/${GCS_PREFIX}/${FILENAME}"

      - name: Update latest pointer in GCS
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          
          echo "{\"latest\": \"${VERSION}\", \"filename\": \"${FILENAME}\", \"timestamp\": \"${{ needs.build.outputs.build-timestamp }}\"}" \
            | gcloud storage cp - "gs://${GCS_BUCKET}/artifacts/latest.json" \
              --content-type="application/json"

      - name: Verify GCS upload
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          
          gcloud storage ls -l "gs://${GCS_BUCKET}/artifacts/${VERSION}/${FILENAME}"
          echo "GCS upload verified successfully."

  # ─────────────────────────────────────────────
  # Job 3c: Publish to Azure Blob Storage
  # ─────────────────────────────────────────────
  publish-azure:
    name: Publish to Azure Blob Storage
    runs-on: ubuntu-latest
    needs: build
    if: |
      (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master' || startsWith(github.ref, 'refs/tags/')) &&
      vars.STORAGE_PROVIDER == 'azure'
    timeout-minutes: 10
    environment:
      name: ${{ github.event.inputs.environment || 'staging' }}

    permissions:
      id-token: write
      contents: read

    steps:
      - name: Download artifact from GitHub Actions
        uses: actions/download-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ needs.build.outputs.artifact-version }}
          path: ./artifacts

      - name: Login to Azure (OIDC)
        uses: azure/login@v2
        with:
          client-id: ${{ secrets.AZURE_CLIENT_ID }}
          tenant-id: ${{ secrets.AZURE_TENANT_ID }}
          subscription-id: ${{ secrets.AZURE_SUBSCRIPTION_ID }}

      - name: Verify artifact integrity
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          sha256sum -c "${FILENAME}.sha256"
          echo "Checksum verification passed."

      - name: Upload artifact to Azure Blob Storage
        working-directory: ./artifacts
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          BLOB_PREFIX="artifacts/${VERSION}"
          
          echo "Uploading ${FILENAME} to Azure Blob Storage..."
          
          az storage blob upload \
            --account-name "${{ secrets.AZURE_STORAGE_ACCOUNT }}" \
            --container-name "${AZURE_CONTAINER}" \
            --name "${BLOB_PREFIX}/${FILENAME}" \
            --file "${FILENAME}" \
            --metadata "version=${VERSION}" "commit=${{ github.sha }}" "timestamp=${{ needs.build.outputs.build-timestamp }}" \
            --auth-mode login \
            --overwrite
          
          az storage blob upload \
            --account-name "${{ secrets.AZURE_STORAGE_ACCOUNT }}" \
            --container-name "${AZURE_CONTAINER}" \
            --name "${BLOB_PREFIX}/${FILENAME}.sha256" \
            --file "${FILENAME}.sha256" \
            --auth-mode login \
            --overwrite
          
          echo "Upload complete."

      - name: Update latest pointer in Azure
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          
          echo "{\"latest\": \"${VERSION}\", \"filename\": \"${FILENAME}\", \"timestamp\": \"${{ needs.build.outputs.build-timestamp }}\"}" \
            | az storage blob upload \
                --account-name "${{ secrets.AZURE_STORAGE_ACCOUNT }}" \
                --container-name "${AZURE_CONTAINER}" \
                --name "artifacts/latest.json" \
                --data @- \
                --content-type "application/json" \
                --auth-mode login \
                --overwrite

      - name: Verify Azure upload
        run: |
          FILENAME="${{ needs.build.outputs.artifact-filename }}"
          VERSION="${{ needs.build.outputs.artifact-version }}"
          
          az storage blob show \
            --account-name "${{ secrets.AZURE_STORAGE_ACCOUNT }}" \
            --container-name "${AZURE_CONTAINER}" \
            --name "artifacts/${VERSION}/${FILENAME}" \
            --auth-mode login \
            --query '{name: name, contentLength: properties.contentLength, lastModified: properties.lastModified}' \
            --output table
          
          echo "Azure upload verified successfully."

  # ─────────────────────────────────────────────
  # Job 4: Notify on completion
  # ─────────────────────────────────────────────
  notify:
    name: Notify Build Status
    runs-on: ubuntu-latest
    needs:
      - build
      - publish-s3
      - publish-gcs
      - publish-azure
    if: always()
    timeout-minutes: 5

    steps:
      - name: Determine overall status
        id: status
        run: |
          BUILD_RESULT="${{ needs.build.result }}"
          S3_RESULT="${{ needs.publish-s3.result }}"
          GCS_RESULT="${{ needs.publish-gcs.result }}"
          AZURE_RESULT="${{ needs.publish-azure.result }}"
          
          if [[ "$BUILD_RESULT" == "failure" ]]; then
            echo "status=failure" >> "$GITHUB_OUTPUT"
            echo "message=Build failed" >> "$GITHUB_OUTPUT"
          elif [[ "$S3_RESULT" == "failure" || "$GCS_RESULT" == "failure" || "$AZURE_RESULT" == "failure" ]]; then
            echo "status=failure" >> "$GITHUB_OUTPUT"
            echo "message=Artifact publish failed" >> "$GITHUB_OUTPUT"
          elif [[ "$BUILD_RESULT" == "success" ]]; then
            echo "status=success" >> "$GITHUB_OUTPUT"
            echo "message=Build and publish succeeded" >> "$GITHUB_OUTPUT"
          else
            echo "status=skipped" >> "$GITHUB_OUTPUT"
            echo "message=Build skipped or cancelled" >> "$GITHUB_OUTPUT"
          fi

      - name: Post Slack notification
        if: always() && secrets.SLACK_WEBHOOK_URL != ''
        uses: slackapi/slack-github-action@v1
        with:
          payload: |
            {
              "text": "CI Build ${{ steps.status.outputs.status == 'success' && ':white_check_mark:' || ':x:' }} ${{ steps.status.outputs.message }}",
              "blocks": [
                {
                  "type": "section",
                  "text": {
                    "type": "mrkdwn",
                    "text": "*CI Build Result*: ${{ steps.status.outputs.message }}\n*Repository*: ${{ github.repository }}\n*Branch*: ${{ github.ref_name }}\n*Commit*: `${{ github.sha }}`\n*Version*: `${{ needs.build.outputs.artifact-version }}`\n*Actor*: ${{ github.actor }}\n*Run*: <${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}|View Run>"
                  }
                }
              ]
            }
        env:
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}
          SLACK_WEBHOOK_TYPE: INCOMING_WEBHOOK

      - name: Create GitHub deployment status
        if: needs.build.result == 'success'
        uses: actions/github-script@v7
        with:
          script: |
            const status = '${{ steps.status.outputs.status }}';
            const version = '${{ needs.build.outputs.artifact-version }}';
            
            await github.rest.repos.createCommitStatus({
              owner: context.repo.owner,
              repo: context.repo.repo,
              sha: context.sha,
              state: status === 'success' ? 'success' : 'failure',
              description: `Artifact ${version} published`,
              context: 'CI / Artifact Published'
            });

      - name: Summary
        run: |
          echo "## CI Build Summary" >> "$GITHUB_STEP_SUMMARY"
          echo "" >> "$GITHUB_STEP_SUMMARY"
          echo "| Property | Value |" >> "$GITHUB_STEP_SUMMARY"
          echo "|----------|-------|" >> "$GITHUB_STEP_SUMMARY"
          echo "| Status | ${{ steps.status.outputs.status }} |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Version | \`${{ needs.build.outputs.artifact-version }}\` |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Artifact | \`${{ needs.build.outputs.artifact-filename }}\` |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Timestamp | ${{ needs.build.outputs.build-timestamp }} |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Commit | \`${{ github.sha }}\` |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Branch | ${{ github.ref_name }} |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Actor | ${{ github.actor }} |" >> "$GITHUB_STEP_SUMMARY"
          echo "| Run | [${{ github.run_id }}](${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}) |" >> "$GITHUB_STEP_SUMMARY"
```
