package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"time"
)

// RetryResult holds the result of a successful call
type RetryResult struct {
	StatusCode int
	Body       []byte
	Attempts   int
}

// CallWithRetry repeatedly calls the given URL until it receives a successful response.
// It retries immediately on any error or non-2xx status code with no backoff or delay.
func CallWithRetry(url string) (*RetryResult, error) {
	client := &http.Client{
		Timeout: 10 * time.Second,
	}

	attempts := 0

	for {
		attempts++

		resp, err := client.Get(url)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying immediately...", attempts, err)
			continue
		}

		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()

		if err != nil {
			log.Printf("Attempt %d failed reading body: %v. Retrying immediately...", attempts, err)
			continue
		}

		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			log.Printf("Attempt %d received non-success status: %d. Retrying immediately...", attempts, resp.StatusCode)
			continue
		}

		log.Printf("Success on attempt %d with status: %d", attempts, resp.StatusCode)

		return &RetryResult{
			StatusCode: resp.StatusCode,
			Body:       body,
			Attempts:   attempts,
		}, nil
	}
}

// CallWithRetryPost repeatedly POSTs to the given URL until it receives a successful response.
func CallWithRetryPost(url string, contentType string, body io.Reader) (*RetryResult, error) {
	client := &http.Client{
		Timeout: 10 * time.Second,
	}

	attempts := 0

	// We need to read the body once and reuse it for retries
	var bodyBytes []byte
	if body != nil {
		var err error
		bodyBytes, err = io.ReadAll(body)
		if err != nil {
			return nil, fmt.Errorf("failed to read request body: %w", err)
		}
	}

	for {
		attempts++

		var reqBody io.Reader
		if bodyBytes != nil {
			reqBody = newBytesReader(bodyBytes)
		}

		resp, err := client.Post(url, contentType, reqBody)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying immediately...", attempts, err)
			continue
		}

		respBody, err := io.ReadAll(resp.Body)
		resp.Body.Close()

		if err != nil {
			log.Printf("Attempt %d failed reading body: %v. Retrying immediately...", attempts, err)
			continue
		}

		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			log.Printf("Attempt %d received non-success status: %d. Retrying immediately...", attempts, resp.StatusCode)
			continue
		}

		log.Printf("Success on attempt %d with status: %d", attempts, resp.StatusCode)

		return &RetryResult{
			StatusCode: resp.StatusCode,
			Body:       respBody,
			Attempts:   attempts,
		}, nil
	}
}

// bytesReader is a simple wrapper to create a new reader from bytes each time
type bytesReader struct {
	data []byte
	pos  int
}

func newBytesReader(data []byte) *bytesReader {
	return &bytesReader{data: data}
}

func (r *bytesReader) Read(p []byte) (n int, err error) {
	if r.pos >= len(r.data) {
		return 0, io.EOF
	}
	n = copy(p, r.data[r.pos:])
	r.pos += n
	return n, nil
}

func main() {
	url := "http://localhost:8080/api/health"

	log.Printf("Starting retry loop for URL: %s", url)

	result, err := CallWithRetry(url)
	if err != nil {
		log.Fatalf("Unexpected error: %v", err)
	}

	fmt.Printf("Success after %d attempts\n", result.Attempts)
	fmt.Printf("Status Code: %d\n", result.StatusCode)
	fmt.Printf("Body: %s\n", string(result.Body))
}
