#include <anthropic/sdk.hpp>
#include <iostream>
#include <string>
#include <fstream>
#include <sstream>

// Function to read file content
std::string readFile(const std::string& filename) {
    std::ifstream file(filename, std::ios::binary);
    if (!file.is_open()) {
        return "";
    }
    std::ostringstream ss;
    ss << file.rdbuf();
    return ss.str();
}

// Function to write content to file
void writeFile(const std::string& filename, const std::string& content) {
    std::ofstream file(filename);
    file << content;
}

int main() {
    // Initialize Anthropic client
    anthropic::Client client;
    
    // Create a prompt asking Claude to generate C++ code with libjpeg integration
    std::string prompt = R"(
Please write a complete C++ function that integrates with libjpeg (an external image-processing library).

The code should include:
1. Required headers (#include <jpeglib.h>, etc.)
2. A function to load a JPEG image from file
3. A function to save/compress an image as JPEG
4. Error handling using libjpeg's error manager
5. Build instructions as comments (how to compile with -ljpeg flag)
6. A simple main() function demonstrating usage

Also provide:
- The exact CMakeLists.txt content needed to build this
- The exact g++ command to compile manually

Make the code production-quality with proper memory management and error handling.
)";

    // Create message request
    anthropic::MessageRequest request;
    request.model = "claude-opus-4-5";
    request.max_tokens = 4096;
    request.messages.push_back({
        "user",
        prompt
    });
    
    std::cout << "Requesting Claude to generate C++ code with libjpeg integration..." << std::endl;
    std::cout << "================================================" << std::endl;
    
    // Send request to Claude
    auto response = client.messages().create(request);
    
    // Extract the generated code from response
    std::string generatedCode;
    for (const auto& block : response.content) {
        if (block.type == "text") {
            generatedCode += block.text;
        }
    }
    
    // Display the generated code
    std::cout << generatedCode << std::endl;
    
    // Save the generated code to files
    std::cout << "\n================================================" << std::endl;
    std::cout << "Saving generated code to files..." << std::endl;
    
    // Extract C++ code block if present
    size_t cppStart = generatedCode.find("```cpp");
    size_t cppEnd = std::string::npos;
    if (cppStart != std::string::npos) {
        cppStart += 6; // Skip "```cpp"
        cppEnd = generatedCode.find("```", cppStart);
        if (cppEnd != std::string::npos) {
            std::string cppCode = generatedCode.substr(cppStart, cppEnd - cppStart);
            writeFile("jpeg_integration.cpp", cppCode);
            std::cout << "C++ code saved to: jpeg_integration.cpp" << std::endl;
        }
    }
    
    // Extract CMakeLists.txt if present
    size_t cmakeStart = generatedCode.find("cmake");
    if (cmakeStart == std::string::npos) {
        cmakeStart = generatedCode.find("CMakeLists");
    }
    if (cmakeStart != std::string::npos) {
        // Find the code block around CMakeLists
        size_t blockStart = generatedCode.rfind("```", cmakeStart);
        if (blockStart != std::string::npos) {
            blockStart = generatedCode.find('\n', blockStart) + 1;
            size_t blockEnd = generatedCode.find("```", blockStart);
            if (blockEnd != std::string::npos) {
                std::string cmakeCode = generatedCode.substr(blockStart, blockEnd - blockStart);
                if (cmakeCode.find("cmake_minimum_required") != std::string::npos ||
                    cmakeCode.find("project(") != std::string::npos) {
                    writeFile("CMakeLists.txt", cmakeCode);
                    std::cout << "CMakeLists.txt saved to: CMakeLists.txt" << std::endl;
                }
            }
        }
    }
    
    // Save the full response
    writeFile("claude_response.txt", generatedCode);
    std::cout << "Full response saved to: claude_response.txt" << std::endl;
    
    std::cout << "\n================================================" << std::endl;
    std::cout << "Code generation complete!" << std::endl;
    std::cout << "\nTo build the generated code:" << std::endl;
    std::cout << "1. Install libjpeg: sudo apt-get install libjpeg-dev (Ubuntu/Debian)" << std::endl;
    std::cout << "   or: brew install libjpeg (macOS)" << std::endl;
    std::cout << "2. Compile: g++ -o jpeg_integration jpeg_integration.cpp -ljpeg" << std::endl;
    std::cout << "   or use CMake: mkdir build && cd build && cmake .. && make" << std::endl;
    
    return 0;
}
