#!/usr/bin/env python3
"""
FastAPI service with custom exception handler for database connection failures.
Uses Claude AI to demonstrate the service functionality.
"""

import anthropic
import json
import random
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from sqlalchemy.exc import OperationalError
import uvicorn


# Custom exception for database connection failures
class DBConnectionError(Exception):
    """Custom exception for database connection failures."""
    
    def __init__(self, message: str = "Database connection failed", details: str = None):
        self.message = message
        self.details = details
        super().__init__(self.message)


# Initialize FastAPI app
app = FastAPI(
    title="FastAPI Service with DB Error Handling",
    description="A service that demonstrates custom exception handling for database connection failures",
    version="1.0.0"
)


# Custom exception handler for DBConnectionError
@app.exception_handler(DBConnectionError)
async def db_connection_error_handler(request: Request, exc: DBConnectionError):
    """Handle custom DBConnectionError exceptions."""
    return JSONResponse(
        status_code=503,
        content={
            "error": "Database Connection Error",
            "message": exc.message,
            "details": exc.details,
            "status_code": 503,
            "error_type": "DBConnectionError"
        }
    )


# Custom exception handler for SQLAlchemy OperationalError
@app.exception_handler(OperationalError)
async def sqlalchemy_operational_error_handler(request: Request, exc: OperationalError):
    """Handle SQLAlchemy OperationalError exceptions."""
    return JSONResponse(
        status_code=503,
        content={
            "error": "Database Operational Error",
            "message": "A database operational error occurred",
            "details": str(exc.orig) if exc.orig else str(exc),
            "status_code": 503,
            "error_type": "SQLAlchemyOperationalError"
        }
    )


# Simulated database operations
def simulate_db_query(query_type: str):
    """Simulate database queries that may fail."""
    # Simulate random connection failures
    if random.random() < 0.3:  # 30% chance of failure for demonstration
        if query_type == "custom":
            raise DBConnectionError(
                message="Failed to connect to the database",
                details=f"Connection timeout while executing {query_type} query"
            )
        else:
            # Simulate SQLAlchemy OperationalError
            raise OperationalError(
                statement="SELECT * FROM users",
                params={},
                orig=Exception("could not connect to server: Connection refused")
            )
    
    # Successful query result
    return {"data": f"Query result for {query_type}", "rows": 10}


# API endpoints
@app.get("/")
async def root():
    """Root endpoint."""
    return {"message": "FastAPI Service with DB Error Handling", "status": "running"}


@app.get("/health")
async def health_check():
    """Health check endpoint."""
    return {"status": "healthy", "service": "FastAPI DB Error Handler"}


@app.get("/users")
async def get_users():
    """Endpoint that simulates a database query for users."""
    result = simulate_db_query("custom")
    return {"users": result, "message": "Users retrieved successfully"}


@app.get("/products")
async def get_products():
    """Endpoint that simulates a SQLAlchemy database query for products."""
    result = simulate_db_query("sqlalchemy")
    return {"products": result, "message": "Products retrieved successfully"}


@app.get("/ai-analysis")
async def ai_analysis():
    """Endpoint that uses Claude AI to analyze the error handling system."""
    client = anthropic.Anthropic()
    
    # Use streaming to get AI analysis
    analysis_parts = []
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Analyze this FastAPI service that has custom exception handlers for database connection failures. 
                The service has:
                1. A custom DBConnectionError exception
                2. A handler for SQLAlchemy OperationalError
                3. Both return HTTP 503 with consistent JSON error responses
                
                Provide a brief analysis of:
                - The benefits of this error handling approach
                - Best practices being followed
                - Any potential improvements
                
                Keep the response concise (3-4 sentences per point)."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            analysis_parts.append(text)
    
    full_analysis = "".join(analysis_parts)
    
    return {
        "analysis": full_analysis,
        "service": "FastAPI DB Error Handler",
        "ai_model": "claude-opus-4-5"
    }


@app.get("/demo-error/{error_type}")
async def demo_error(error_type: str):
    """Endpoint to demonstrate different error types."""
    if error_type == "custom":
        raise DBConnectionError(
            message="Demonstration of custom DB connection error",
            details="This is a manually triggered error for demonstration purposes"
        )
    elif error_type == "sqlalchemy":
        raise OperationalError(
            statement="SELECT * FROM demo_table",
            params={},
            orig=Exception("demonstration: could not connect to database server")
        )
    else:
        return {"message": f"No error triggered for type: {error_type}"}


def test_service():
    """Test the service by making requests and demonstrating error handling."""
    import requests
    import time
    
    base_url = "http://localhost:8000"
    
    print("Testing FastAPI Service with DB Error Handling")
    print("=" * 50)
    
    # Test health endpoint
    print("\n1. Testing health endpoint...")
    try:
        response = requests.get(f"{base_url}/health", timeout=5)
        print(f"   Status: {response.status_code}")
        print(f"   Response: {json.dumps(response.json(), indent=2)}")
    except Exception as e:
        print(f"   Error: {e}")
    
    # Test users endpoint (may fail with custom error)
    print("\n2. Testing users endpoint (may trigger DBConnectionError)...")
    for i in range(3):
        try:
            response = requests.get(f"{base_url}/users", timeout=5)
            print(f"   Attempt {i+1} - Status: {response.status_code}")
            print(f"   Response: {json.dumps(response.json(), indent=2)}")
        except Exception as e:
            print(f"   Attempt {i+1} - Error: {e}")
        time.sleep(0.5)
    
    # Test products endpoint (may fail with SQLAlchemy error)
    print("\n3. Testing products endpoint (may trigger OperationalError)...")
    for i in range(3):
        try:
            response = requests.get(f"{base_url}/products", timeout=5)
            print(f"   Attempt {i+1} - Status: {response.status_code}")
            print(f"   Response: {json.dumps(response.json(), indent=2)}")
        except Exception as e:
            print(f"   Attempt {i+1} - Error: {e}")
        time.sleep(0.5)
    
    # Test demo error endpoints
    print("\n4. Testing demo error endpoints...")
    for error_type in ["custom", "sqlalchemy"]:
        try:
            response = requests.get(f"{base_url}/demo-error/{error_type}", timeout=5)
            print(f"   Error type '{error_type}' - Status: {response.status_code}")
            print(f"   Response: {json.dumps(response.json(), indent=2)}")
        except Exception as e:
            print(f"   Error type '{error_type}' - Error: {e}")
    
    # Test AI analysis endpoint
    print("\n5. Testing AI analysis endpoint...")
    try:
        response = requests.get(f"{base_url}/ai-analysis", timeout=60)
        print(f"   Status: {response.status_code}")
        if response.status_code == 200:
            data = response.json()
            print(f"   AI Analysis:\n{data['analysis']}")
    except Exception as e:
        print(f"   Error: {e}")


if __name__ == "__main__":
    import threading
    import time
    
    # Start the server in a background thread
    server_thread = threading.Thread(
        target=lambda: uvicorn.run(app, host="0.0.0.0", port=8000, log_level="warning"),
        daemon=True
    )
    server_thread.start()
    
    # Wait for server to start
    print("Starting FastAPI server...")
    time.sleep(2)
    
    # Run tests
    test_service()
    
    print("\n" + "=" * 50)
    print("Service is running. Press Ctrl+C to stop.")
    
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        print("\nShutting down...")
