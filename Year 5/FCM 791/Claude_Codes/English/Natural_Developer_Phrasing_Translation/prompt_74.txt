function loadScript(url, callback) {
  var script = document.createElement('script');
  script.type = 'text/javascript';
  script.src = url;
  script.async = true;

  script.onload = function () {
    console.log('Script loaded successfully:', url);
    if (typeof callback === 'function') {
      callback(null);
    }
  };

  script.onerror = function () {
    var error = new Error('Failed to load script: ' + url);
    console.error(error.message);
    if (typeof callback === 'function') {
      callback(error);
    }
  };

  document.head.appendChild(script);
}

var JQUERY_CDN_URL =
  'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';

loadScript(JQUERY_CDN_URL, function (err) {
  if (err) {
    console.error('Could not load jQuery:', err.message);
    return;
  }

  if (typeof window.jQuery === 'undefined') {
    console.error('jQuery is not available even though the script loaded.');
    return;
  }

  console.log('jQuery version:', window.jQuery.fn.jquery);

  window.jQuery(document).ready(function ($) {
    $('body').css('background-color', '#f0f8ff');
    console.log('jQuery is ready and working!');
  });
});
