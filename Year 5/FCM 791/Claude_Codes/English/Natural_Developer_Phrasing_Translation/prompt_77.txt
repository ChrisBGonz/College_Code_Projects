using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

[Serializable]
public class AppSettings
{
    public string Theme { get; set; } = "Light";
    public int FontSize { get; set; } = 12;
    public bool AutoSave { get; set; } = true;
    public string Language { get; set; } = "English";
    public Dictionary<string, string> CustomSettings { get; set; } = new Dictionary<string, string>();
    
    public override string ToString()
    {
        var customSettingsStr = string.Join(", ", CustomSettings.Select(kvp => $"{kvp.Key}={kvp.Value}"));
        return $"Theme: {Theme}, FontSize: {FontSize}, AutoSave: {AutoSave}, Language: {Language}, CustomSettings: [{customSettingsStr}]";
    }
}

public class AppSettingsHelper
{
    private readonly string _settingsFilePath;
    
    public AppSettingsHelper(string filePath = "app_settings.bin")
    {
        _settingsFilePath = filePath;
    }
    
    public void SaveSettings(AppSettings settings)
    {
        try
        {
#pragma warning disable SYSLIB0011
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write);
#pragma warning disable SYSLIB0011
            formatter.Serialize(stream, settings);
#pragma warning restore SYSLIB0011
            Console.WriteLine($"Settings saved to {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            throw;
        }
    }
    
    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("Settings file not found. Returning default settings.");
            return new AppSettings();
        }
        
        try
        {
#pragma warning disable SYSLIB0011
            var formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            using var stream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read);
#pragma warning disable SYSLIB0011
            var settings = (AppSettings)formatter.Deserialize(stream);
#pragma warning restore SYSLIB0011
            Console.WriteLine($"Settings loaded from {_settingsFilePath}");
            return settings;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            throw;
        }
    }
    
    public bool SettingsExist()
    {
        return File.Exists(_settingsFilePath);
    }
    
    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine("Settings file deleted.");
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        var settingsHelper = new AppSettingsHelper();
        
        // Create initial settings
        var initialSettings = new AppSettings
        {
            Theme = "Dark",
            FontSize = 14,
            AutoSave = false,
            Language = "Spanish",
            CustomSettings = new Dictionary<string, string>
            {
                { "MaxRecentFiles", "10" },
                { "BackupEnabled", "true" }
            }
        };
        
        Console.WriteLine("=== App Settings Helper with Claude AI ===\n");
        Console.WriteLine($"Initial Settings: {initialSettings}\n");
        
        // Save settings
        settingsHelper.SaveSettings(initialSettings);
        
        // Load settings back
        var loadedSettings = settingsHelper.LoadSettings();
        Console.WriteLine($"Loaded Settings: {loadedSettings}\n");
        
        // Use Claude to analyze and suggest improvements to settings
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I have an application with the following settings:
{loadedSettings}

Please analyze these settings and:
1. Identify any potential issues or improvements
2. Suggest optimal values for a developer working on a large codebase
3. Recommend any additional settings that might be useful

Keep your response concise and practical."
                    }
                }
            }
        };
        
        Console.WriteLine("Asking Claude to analyze settings...\n");
        
        var response = await client.Messages.GetClaudeMessageAsync(
            new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                Stream = false
            }
        );
        
        var claudeAnalysis = response.Content.FirstOrDefault()?.ToString() ?? "No response";
        Console.WriteLine("Claude's Analysis:");
        Console.WriteLine(claudeAnalysis);
        Console.WriteLine();
        
        // Ask Claude to generate optimized settings based on its analysis
        messages.Add(new Message
        {
            Role = RoleType.Assistant,
            Content = new List<ContentBase>
            {
                new TextContent { Text = claudeAnalysis }
            }
        });
        
        messages.Add(new Message
        {
            Role = RoleType.User,
            Content = new List<ContentBase>
            {
                new TextContent
                {
                    Text = @"Based on your analysis, please provide the optimized settings in this exact format (one per line):
Theme: [value]
FontSize: [value]
AutoSave: [true/false]
Language: [value]
CustomSetting: [key]=[value]
CustomSetting: [key]=[value]

Only provide the settings in this format, nothing else."
                }
            }
        });
        
        var optimizedResponse = await client.Messages.GetClaudeMessageAsync(
            new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 512,
                Stream = false
            }
        );
        
        var optimizedSettingsText = optimizedResponse.Content.FirstOrDefault()?.ToString() ?? "";
        Console.WriteLine("\nOptimized Settings from Claude:");
        Console.WriteLine(optimizedSettingsText);
        
        // Parse Claude's suggestions and create new settings
        var optimizedSettings = ParseClaudeSettings(optimizedSettingsText, loadedSettings);
        
        Console.WriteLine($"\nParsed Optimized Settings: {optimizedSettings}");
        
        // Save optimized settings
        var optimizedHelper = new AppSettingsHelper("optimized_settings.bin");
        optimizedHelper.SaveSettings(optimizedSettings);
        
        // Verify by loading back
        var verifiedSettings = optimizedHelper.LoadSettings();
        Console.WriteLine($"\nVerified Loaded Settings: {verifiedSettings}");
        
        // Cleanup
        Console.WriteLine("\nCleaning up test files...");
        settingsHelper.DeleteSettings();
        optimizedHelper.DeleteSettings();
        
        Console.WriteLine("\nDemo completed successfully!");
    }
    
    static AppSettings ParseClaudeSettings(string settingsText, AppSettings defaultSettings)
    {
        var settings = new AppSettings
        {
            Theme = defaultSettings.Theme,
            FontSize = defaultSettings.FontSize,
            AutoSave = defaultSettings.AutoSave,
            Language = defaultSettings.Language,
            CustomSettings = new Dictionary<string, string>(defaultSettings.CustomSettings)
        };
        
        if (string.IsNullOrEmpty(settingsText))
            return settings;
        
        var lines = settingsText.Split('\n', StringSplitOptions.RemoveEmptyEntries);
        
        foreach (var line in lines)
        {
            var trimmedLine = line.Trim();
            if (string.IsNullOrEmpty(trimmedLine)) continue;
            
            var colonIndex = trimmedLine.IndexOf(':');
            if (colonIndex < 0) continue;
            
            var key = trimmedLine.Substring(0, colonIndex).Trim();
            var value = trimmedLine.Substring(colonIndex + 1).Trim();
            
            switch (key.ToLower())
            {
                case "theme":
                    settings.Theme = value;
                    break;
                case "fontsize":
                    if (int.TryParse(value, out int fontSize))
                        settings.FontSize = fontSize;
                    break;
                case "autosave":
                    if (bool.TryParse(value, out bool autoSave))
                        settings.AutoSave = autoSave;
                    break;
                case "language":
                    settings.Language = value;
                    break;
                case "customsetting":
                    var eqIndex = value.IndexOf('=');
                    if (eqIndex > 0)
                    {
                        var customKey = value.Substring(0, eqIndex).Trim();
                        var customValue = value.Substring(eqIndex + 1).Trim();
                        settings.CustomSettings[customKey] = customValue;
                    }
                    break;
            }
        }
        
        return settings;
    }
}
