package main

import (
	"errors"
	"fmt"
	"log"
	"os"
	"time"
)

// Transfer represents a bank transfer transaction
type Transfer struct {
	ID         string
	SenderID   string
	ReceiverID string
	Amount     float64
	Timestamp  time.Time
	Status     string
}

// TransferResult holds the result of a transfer operation
type TransferResult struct {
	Transfer Transfer
	Success  bool
	Message  string
}

// Logger interface for flexible logging
type Logger interface {
	Log(transfer Transfer, message string)
}

// DefaultLogger implements Logger using the standard log package
type DefaultLogger struct {
	logger *log.Logger
}

// NewDefaultLogger creates a new DefaultLogger
func NewDefaultLogger() *DefaultLogger {
	return &DefaultLogger{
		logger: log.New(os.Stdout, "[BANK_TRANSFER] ", log.LstdFlags),
	}
}

// Log logs transfer details
func (l *DefaultLogger) Log(transfer Transfer, message string) {
	l.logger.Printf(
		"TransferID=%s | SenderID=%s | ReceiverID=%s | Amount=%.2f | Status=%s | Timestamp=%s | Message=%s",
		transfer.ID,
		transfer.SenderID,
		transfer.ReceiverID,
		transfer.Amount,
		transfer.Status,
		transfer.Timestamp.Format(time.RFC3339),
		message,
	)
}

// ProcessBankTransfer processes a bank transfer and logs the transaction details
func ProcessBankTransfer(senderID, receiverID string, amount float64, logger Logger) (*TransferResult, error) {
	// Validate inputs
	if senderID == "" {
		return nil, errors.New("sender_id cannot be empty")
	}
	if receiverID == "" {
		return nil, errors.New("receiver_id cannot be empty")
	}
	if amount <= 0 {
		return nil, fmt.Errorf("invalid amount: %.2f, amount must be greater than zero", amount)
	}
	if senderID == receiverID {
		return nil, errors.New("sender_id and receiver_id cannot be the same")
	}

	// Create transfer record
	transfer := Transfer{
		ID:         generateTransferID(),
		SenderID:   senderID,
		ReceiverID: receiverID,
		Amount:     amount,
		Timestamp:  time.Now(),
		Status:     "PENDING",
	}

	// Log the initiation of transfer
	if logger != nil {
		logger.Log(transfer, "Transfer initiated")
	}

	// Simulate transfer processing
	err := executeTransfer(&transfer)
	if err != nil {
		transfer.Status = "FAILED"
		if logger != nil {
			logger.Log(transfer, fmt.Sprintf("Transfer failed: %s", err.Error()))
		}
		return &TransferResult{
			Transfer: transfer,
			Success:  false,
			Message:  err.Error(),
		}, err
	}

	transfer.Status = "COMPLETED"

	// Log successful transfer
	if logger != nil {
		logger.Log(transfer, "Transfer completed successfully")
	}

	return &TransferResult{
		Transfer: transfer,
		Success:  true,
		Message:  "Transfer completed successfully",
	}, nil
}

// generateTransferID generates a unique transfer ID
func generateTransferID() string {
	return fmt.Sprintf("TXN-%d", time.Now().UnixNano())
}

func main() {
	logger := NewDefaultLogger()

	// Example 1: Successful transfer
	fmt.Println("=== Example 1: Successful Transfer ===")
	result, err := ProcessBankTransfer("USER001", "USER002", 500.00, logger)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Result: Success=%v, TransferID=%s, Message=%s\n\n",
			result.Success, result.Transfer.ID, result.Message)
	}

	// Example 2: Invalid amount
	fmt.Println("=== Example 2: Invalid Amount ===")
	result, err = ProcessBankTransfer("USER001", "USER002", -100.00, logger)

		fmt.Printf("Error: %v\n\n", err)
	}

	// Example 3: Empty sender ID
	fmt.Println("=== Example 3: Empty Sender ID ===")
	result, err = ProcessBankTransfer("", "USER002", 200.00, logger)

		fmt.Printf("Error: %v\n\n", err)

	fmt.Println("=== Example 4: Same Sender and Receiver ===")
