
```ruby
require 'rails'
require 'active_record'
require 'claude-ai'

# Database setup
ActiveRecord::Base.establish_connection(
  adapter: 'sqlite3',
  database: ':memory:'
)

ActiveRecord::Schema.define do
  create_table :users do |t|
    t.string :name
    t.string :email
    t.integer :age
    t.timestamps
  end
end

# User model
class User < ActiveRecord::Base
end

# Seed some data
User.create!(name: 'Alice', email: 'alice@example.com', age: 30)
User.create!(name: 'Bob', email: 'bob@example.com', age: 25)
User.create!(name: 'Charlie', email: 'charlie@example.com', age: 35)

# Rails controller simulation
class UsersController
  attr_reader :params, :response_data

  def initialize(params)
    @params = params
    @response_data = nil
  end

  def find_user
    query_conditions = params[:query]
    
    if query_conditions.nil? || query_conditions.empty?
      @response_data = { error: 'No query conditions provided', status: 400 }
      return
    end

    # Use find_by with the conditions hash from params
    user = User.find_by(query_conditions)
    
    if user
      @response_data = {
        status: 200,
        user: {
          id: user.id,
          name: user.name,
          email: user.email,
          age: user.age
        }
      }
    else
      @response_data = { error: 'User not found', status: 404 }
    end
  end
end

puts "Asking Claude to generate test queries..."

response = client.messages(
```
