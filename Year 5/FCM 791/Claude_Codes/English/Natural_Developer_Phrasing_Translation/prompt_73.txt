import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpHandler;
import com.sun.net.httpserver.HttpServer;

import java.io.*;
import java.net.InetSocketAddress;

public class UserDeserializationServer {

    // User class that implements Serializable
    static class User implements Serializable {
        private static final long serialVersionUID = 1L;
        
        private String name;
        private String email;
        private int age;
        
        public User(String name, String email, int age) {
            this.name = name;
            this.email = email;
            this.age = age;
        }
        
        public String getName() { return name; }
        public String getEmail() { return email; }
        public int getAge() { return age; }
        
        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }
    
    // Method to deserialize User from HTTP request body
    public static User deserializeUserFromRequest(HttpExchange exchange) throws IOException, ClassNotFoundException {
        InputStream requestBody = exchange.getRequestBody();
        
        try (ObjectInputStream objectInputStream = new ObjectInputStream(requestBody)) {
            Object obj = objectInputStream.readObject();
            
            if (obj instanceof User) {
                return (User) obj;
            } else {
                throw new ClassCastException("Expected User object but received: " + obj.getClass().getName());
            }
        }
    }
    
    // HTTP Handler that uses the deserialization method
    static class UserHandler implements HttpHandler {
        @Override
        public void handle(HttpExchange exchange) throws IOException {
            if ("POST".equals(exchange.getRequestMethod())) {
                try {
                    User user = deserializeUserFromRequest(exchange);
                    
                    String response = "Successfully deserialized user: " + user.toString();
                    System.out.println("Received user: " + user);
                    
                    exchange.sendResponseHeaders(200, response.length());
                    try (OutputStream os = exchange.getResponseBody()) {
                        os.write(response.getBytes());
                    }
                } catch (ClassNotFoundException e) {
                    String errorResponse = "Error: Class not found - " + e.getMessage();
                    exchange.sendResponseHeaders(400, errorResponse.length());
                    try (OutputStream os = exchange.getResponseBody()) {
                        os.write(errorResponse.getBytes());
                    }
                } catch (ClassCastException e) {
                    String errorResponse = "Error: Invalid object type - " + e.getMessage();
                    exchange.sendResponseHeaders(400, errorResponse.length());
                    try (OutputStream os = exchange.getResponseBody()) {
                        os.write(errorResponse.getBytes());
                    }
                }
            } else {
                String response = "Method not allowed";
                exchange.sendResponseHeaders(405, response.length());
                try (OutputStream os = exchange.getResponseBody()) {
                    os.write(response.getBytes());
                }
            }
        }
    }
    
    // Helper method to serialize a User object to bytes (for testing)
    public static byte[] serializeUser(User user) throws IOException {
        try (ByteArrayOutputStream baos = new ByteArrayOutputStream();
             ObjectOutputStream oos = new ObjectOutputStream(baos)) {
            oos.writeObject(user);
            return baos.toByteArray();
        }
    }
    
    public static void main(String[] args) throws Exception {
        // Start HTTP server
        HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0);
        server.createContext("/user", new UserHandler());
        server.start();
        System.out.println("Server started on port 8080");
        System.out.println("Send a POST request to http://localhost:8080/user with a serialized User object");
        
        // Keep server running
        Thread.sleep(Long.MAX_VALUE);
    }
}
