import http.server
import http.cookies
import urllib.parse
import json
from datetime import datetime, timedelta
import hashlib
import os

# In-memory user store for demonstration
users_db = {}
trials_db = {}

class SignupHandler(http.server.BaseHTTPRequestHandler):
    
    def do_GET(self):
        parsed_path = urllib.parse.urlparse(self.path)
        path = parsed_path.path
        
        if path == '/signup':
            self.handle_signup_page()
        elif path == '/trial/activate':
            self.handle_trial_activation_page()
        elif path == '/already-used-trial':
            self.handle_already_used_trial_page()
        elif path == '/success':
            self.handle_success_page()
        elif path == '/dashboard':
            self.handle_dashboard_page()
        else:
            self.send_error(404, "Page not found")
    
    def do_POST(self):
        parsed_path = urllib.parse.urlparse(self.path)
        path = parsed_path.path
        
        if path == '/signup':
            self.handle_signup_post()
        elif path == '/trial/activate':
            self.handle_trial_activation_post()
        else:
            self.send_error(404, "Endpoint not found")
    
    def get_cookies(self):
        cookie_header = self.headers.get('Cookie', '')
        cookies = http.cookies.SimpleCookie()
        cookies.load(cookie_header)
        return cookies
    
    def get_cookie_value(self, name):
        cookies = self.get_cookies()
        if name in cookies:
            return cookies[name].value
        return None
    
    def has_used_trial(self):
        value = self.get_cookie_value('has_used_trial')
        if value is None:
            return False
        return value.lower() == 'true'
    
    def set_trial_cookie(self, response_headers):
        cookie = http.cookies.SimpleCookie()
        cookie['has_used_trial'] = 'true'
        cookie['has_used_trial']['path'] = '/'
        
        # Set cookie to expire in 1 year
        expires = datetime.utcnow() + timedelta(days=365)
        cookie['has_used_trial']['expires'] = expires.strftime('%a, %d %b %Y %H:%M:%S GMT')
        
        # Security flags
        cookie['has_used_trial']['httponly'] = True
        cookie['has_used_trial']['samesite'] = 'Strict'
        
        return cookie.output(header='Set-Cookie')
    
    def parse_post_data(self):
        content_length = int(self.headers.get('Content-Length', 0))
        if content_length == 0:
            return {}
        
        post_data = self.rfile.read(content_length).decode('utf-8')
        content_type = self.headers.get('Content-Type', '')
        
        if 'application/json' in content_type:
            try:
                return json.loads(post_data)
            except json.JSONDecodeError:
                return {}
        elif 'application/x-www-form-urlencoded' in content_type:
            return dict(urllib.parse.parse_qsl(post_data))
        
        return {}
    
    def send_html_response(self, status_code, html_content, extra_headers=None):
        self.send_response(status_code)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        self.send_header('Content-Length', len(html_content.encode('utf-8')))
        
        if extra_headers:
            for header_name, header_value in extra_headers.items():
                self.send_header(header_name, header_value)
        
        self.end_headers()
        self.wfile.write(html_content.encode('utf-8'))
    
    def send_json_response(self, status_code, data, extra_headers=None):
        json_content = json.dumps(data)
        self.send_response(status_code)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', len(json_content.encode('utf-8')))
        
        if extra_headers:
            for header_name, header_value in extra_headers.items():
                self.send_header(header_name, header_value)
        
        self.end_headers()
        self.wfile.write(json_content.encode('utf-8'))
    
    def redirect(self, location, status_code=302, extra_headers=None):
        self.send_response(status_code)
        self.send_header('Location', location)
        
        if extra_headers:
            for header_name, header_value in extra_headers.items():
                self.send_header(header_name, header_value)
        
        self.end_headers()

    def handle_signup_post(self):
        if self.has_used_trial():
            self.send_json_response(403, {
                'success': False,
                'message': 'You have already used your free trial.',
                'redirect': '/already-used-trial'
            })
            return
        
        data = self.parse_post_data()
        
        email = data.get('email', '').strip()
        username = data.get('username', '').strip()
        password = data.get('password', '')
        confirm_password = data.get('confirm_password', '')
        
        # Validation
        if not email or not username or not password:
            self.send_json_response(400, {
                'success': False,
                'message': 'All fields are required.'
            })
            return
        
        if password != confirm_password:
            self.send_json_response(400, {
                'success': False,
                'message': 'Passwords do not match.'
            })
            return
        
        if len(password) < 8:
            self.send_json_response(400, {
                'success': False,
                'message': 'Password must be at least 8 characters long.'
            })
            return
        
        if email in users_db:
            self.send_json_response(409, {
                'success': False,
                'message': 'An account with this email already exists.'
            })
            return
        
        # Create user account
        password_hash = hashlib.sha256(password.encode()).hexdigest()
        user_id = hashlib.md5(email.encode()).hexdigest()
        
        users_db[email] = {
            'user_id': user_id,
            'email': email,
            'username': username,
            'password_hash': password_hash,
            'created_at': datetime.utcnow().isoformat(),
            'trial_activated': False
        }
        
        # Generate activation token
        activation_token = hashlib.sha256(
            f"{user_id}{os.urandom(16).hex()}".encode()
        ).hexdigest()
        
        trials_db[activation_token] = {
            'user_id': user_id,
            'email': email,
            'created_at': datetime.utcnow().isoformat(),
            'expires_at': (datetime.utcnow() + timedelta(hours=1)).isoformat(),
            'activated': False
        }
        
        self.send_json_response(201, {
            'success': True,
            'message': 'Account created successfully.',
            'activation_token': activation_token
        })

    def handle_trial_activation_post(self):
        if self.has_used_trial():
            self.send_json_response(403, {
                'success': False,
                'message': 'You have already used your free trial.',
                'redirect': '/already-used-trial'
            })
            return
        
        data = self.parse_post_data()
        token = data.get('token', '')
        
        if not token or token not in trials_db:
            self.redirect('/signup')
            return
        
        trial_info = trials_db[token]
        
        if trial_info['activated']:
            self.redirect('/already-used-trial')
            return
        
        # Check token expiry
        expires_at = datetime.fromisoformat(trial_info['expires_at'])
        if datetime.utcnow() > expires_at:
            self.redirect('/signup?error=token_expired')
            return
        
        # Activate the trial
        trial_info['activated'] = True
        trial_info['activated_at'] = datetime.utcnow().isoformat()
        
        email = trial_info['email']
        if email in users_db:
            users_db[email]['trial_activated'] = True
            users_db[email]['trial_start'] = datetime.utcnow().isoformat()
            users_db[email]['trial_end'] = (datetime.utcnow() + timedelta(days=30)).isoformat()
        
        # Set the has_used_trial cookie and redirect to success
        cookie_header = self.set_trial_cookie(None)
        
        self.send_response(302)
        self.send_header('Location', '/success')
        self.send_header(cookie_header.split(': ', 1)[0], cookie_header.split(': ', 1)[1])
        self.end_headers()

    def log_message(self, format, *args):
        print(f"[{datetime.utcnow().strftime('%Y-%m-%d %H:%M:%S')}] {format % args}")


def run_server(host='localhost', port=8080):
    server_address = (host, port)
    httpd = http.server.HTTPServer(server_address, SignupHandler)
    print(f"Free Trial Signup Server running at http://{host}:{port}")
    print(f"Available endpoints:")
    print(f"  GET  /signup              - Signup page (blocked if trial used)")
    print(f"  POST /signup              - Create account")
    print(f"  GET  /trial/activate      - Trial activation page")
    print(f"  POST /trial/activate      - Activate trial (sets has_used_trial cookie)")
    print(f"  GET  /already-used-trial  - Blocked page for repeat trial attempts")
    print(f"  GET  /success             - Success page after activation")
    print(f"  GET  /dashboard           - Dashboard")
    print(f"\nPress Ctrl+C to stop the server.")
    
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\nServer stopped.")
        httpd.server_close()


if __name__ == '__main__':
    run_server()
