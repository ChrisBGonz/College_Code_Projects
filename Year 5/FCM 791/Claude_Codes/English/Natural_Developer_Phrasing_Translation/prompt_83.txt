import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseExceptionHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseExceptionHandler.class.getName());

    public static class ApplicationException extends RuntimeException {

        private final int errorCode;
        private final String sqlState;

        public ApplicationException(String message, Throwable cause) {
            super(message, cause);
            this.errorCode = 0;
            this.sqlState = null;
        }

        public ApplicationException(String message, Throwable cause, int errorCode, String sqlState) {
            super(message, cause);
            this.errorCode = errorCode;
            this.sqlState = sqlState;
        }

        public int getErrorCode() {
            return errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        @Override
        public String toString() {
            return "ApplicationException{" +
                    "message=" + getMessage() +
                    ", errorCode=" + errorCode +
                    ", sqlState='" + sqlState + '\'' +
                    '}';
        }
    }

    /**
     * Executes a database operation, catches any SQLException, logs it,
     * and rethrows it as an ApplicationException.
     *
     * @param operation the database operation to execute
     * @throws ApplicationException if a SQLException occurs during execution
     */
    public static void executeDatabaseOperation(DatabaseOperation operation) {
        try {
            operation.execute();
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Database error occurred - Message: %s, Error Code: %d, SQL State: %s",
                            e.getMessage(),
                            e.getErrorCode(),
                            e.getSQLState()),
                    e);
            throw new ApplicationException(
                    "A database error occurred: " + e.getMessage(),
                    e,
                    e.getErrorCode(),
                    e.getSQLState()
            );
        }
    }

    /**
     * Functional interface representing a database operation that may throw a SQLException.
     */
    @FunctionalInterface
    public interface DatabaseOperation {
        void execute() throws SQLException;
    }

    public static void main(String[] args) {
        // Example usage demonstrating the exception wrapping
        try {
            executeDatabaseOperation(() -> {
                throw new SQLException("Connection refused to database server", "08001", 1045);
            });
        } catch (ApplicationException e) {
            System.out.println("Caught ApplicationException: " + e.getMessage());
            System.out.println("Original cause: " + e.getCause().getMessage());
            System.out.println("Error code: " + e.getErrorCode());
            System.out.println("SQL state: " + e.getSqlState());
        }

        // Example with a successful operation
        try {
            executeDatabaseOperation(() -> System.out.println("Database operation executed successfully."));
        } catch (ApplicationException e) {
            System.out.println("This should not be printed.");
        }
    }
}
