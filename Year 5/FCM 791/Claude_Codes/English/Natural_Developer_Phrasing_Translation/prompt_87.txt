#!/usr/bin/env python3

import time
import os
import sys
import argparse
from collections import deque
from datetime import datetime


def tail_log_file(filepath: str, window_seconds: int = 60, threshold: int = 10):
    """
    Tail a log file in real time and trigger a warning if more than
    `threshold` occurrences of "Failed Login" are detected within
    any rolling `window_seconds`-second window.
    """
    search_string = "Failed Login"
    timestamps = deque()
    warning_cooldown = 0
    warning_cooldown_seconds = 10  # Avoid spamming warnings

    print(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Starting log monitor...")
    print(f"  File     : {filepath}")
    print(f"  Pattern  : '{search_string}'")
    print(f"  Threshold: {threshold} occurrences in {window_seconds} seconds")
    print("-" * 60)

    # Wait for the file to exist
    while not os.path.exists(filepath):
        print(f"Waiting for file to appear: {filepath}")
        time.sleep(2)

    with open(filepath, "r", encoding="utf-8", errors="replace") as f:
        # Seek to the end of the file to only read new lines
        f.seek(0, os.SEEK_END)
        print(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Tailing file from current end...")

        while True:
            line = f.readline()

            if line:
                # Check if the line contains the search string
                if search_string in line:
                    now = time.time()
                    timestamps.append(now)

                    # Remove timestamps outside the rolling window
                    cutoff = now - window_seconds
                    while timestamps and timestamps[0] < cutoff:
                        timestamps.popleft()

                    count = len(timestamps)
                    print(
                        f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] "
                        f"Detected '{search_string}' | Count in last {window_seconds}s: {count} | "
                        f"Line: {line.rstrip()}"
                    )

                    # Trigger warning if threshold exceeded
                    if count > threshold and time.time() > warning_cooldown:
                        print_warning(count, window_seconds, threshold)
                        warning_cooldown = time.time() + warning_cooldown_seconds

            else:
                # No new line; check if file was rotated/truncated
                try:
                    current_size = os.path.getsize(filepath)
                    current_pos = f.tell()
                    if current_size < current_pos:
                        print(
                            f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] "
                            "File appears to have been rotated or truncated. Re-opening..."
                        )
                        f.seek(0)
                except OSError:
                    pass

                time.sleep(0.1)


def print_warning(count: int, window_seconds: int, threshold: int):
    border = "=" * 60
    print(border)
    print(f"  ⚠️  WARNING: EXCESSIVE FAILED LOGIN ATTEMPTS DETECTED!")
    print(f"  {count} occurrences of 'Failed Login' detected")
    print(f"  within the last {window_seconds} seconds (threshold: {threshold})")
    print(f"  Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(border)


def parse_args():
    parser = argparse.ArgumentParser(
        description="Real-time log file monitor for 'Failed Login' attempts."
    )
    parser.add_argument(
        "logfile",
        nargs="?",
        default="/var/log/auth.log",
        help="Path to the log file to monitor (default: /var/log/auth.log)",
    )
    parser.add_argument(
        "--window",
        type=int,
        default=60,
        help="Rolling time window in seconds (default: 60)",
    )
    parser.add_argument(
        "--threshold",
        type=int,
        default=10,
        help="Number of occurrences to trigger a warning (default: 10)",
    )
    return parser.parse_args()


if __name__ == "__main__":
    args = parse_args()

    try:
        tail_log_file(
            filepath=args.logfile,
            window_seconds=args.window,
            threshold=args.threshold,
        )
    except KeyboardInterrupt:
        print(f"\n[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] Monitor stopped by user.")
        sys.exit(0)
    except PermissionError:
        print(f"Error: Permission denied when trying to read '{args.logfile}'.")
        print("Try running with elevated privileges (e.g., sudo).")
        sys.exit(1)
    except Exception as e:
        print(f"Unexpected error: {e}")
        sys.exit(1)
