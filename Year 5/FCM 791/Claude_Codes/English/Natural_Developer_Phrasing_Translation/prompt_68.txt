
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const IDLE_TIMEOUT_MS = 900000; // 15 minutes
let idleTimer: ReturnType<typeof setTimeout> | null = null;
let isSignedOut = false;

function resetIdleTimer(): void {
  if (isSignedOut) return;

  if (idleTimer) {
    clearTimeout(idleTimer);
  }

  idleTimer = setTimeout(() => {
    handleIdleTimeout();
  }, IDLE_TIMEOUT_MS);
}

function handleIdleTimeout(): void {
  isSignedOut = true;
  console.log("\n⚠️  Session expired due to inactivity. You have been signed out.");
  console.log("Please sign in again to continue banking.");
  process.exit(0);
}

function trackUserActivity(): void {
  // In a real browser environment, these would be event listeners
  // For this Node.js demo, we simulate activity tracking
  resetIdleTimer();
}

async function chat(userMessage: string): Promise<string> {
  // Track user activity on each message
  trackUserActivity();

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: `You are a helpful banking assistant for SecureBank. You help customers with:
- Account inquiries and balance checks
- Transaction history
- Fund transfers
- Bill payments
- Loan information
- Investment advice
- Security and fraud prevention

Important security reminders:
- Never share your password or PIN
- Always log out when done
- Report suspicious activity immediately

The session will automatically expire after 15 minutes of inactivity for security purposes.`,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

async function main(): Promise<void> {
  console.log("🏦 Welcome to SecureBank Online Banking");
  console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
  console.log("⏱️  Your session will automatically expire after 15 minutes of inactivity");
  console.log("💬 Type your banking questions or 'exit' to sign out\n");

  // Start the idle timer
  resetIdleTimer();

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const askQuestion = (prompt: string): Promise<string> => {
    return new Promise((resolve) => {
      rl.question(prompt, (answer) => {
        resolve(answer);
      });
    });
  };

  // Initial greeting
  const greeting = await chat("Hello, I just logged into my banking account.");
  console.log(`\n🤖 Assistant: ${greeting}\n`);

  while (true) {
    const userInput = await askQuestion("You: ");

    if (userInput.toLowerCase() === "exit" || userInput.toLowerCase() === "quit") {
      console.log("\n✅ You have been successfully signed out. Thank you for banking with SecureBank!");
      if (idleTimer) {
        clearTimeout(idleTimer);
      }
      rl.close();
      break;
    }

    if (!userInput.trim()) {
      continue;
    }

    try {
      const response = await chat(userInput);
      console.log(`\n🤖 Assistant: ${response}\n`);
    } catch (error) {
      console.error("Error communicating with the banking assistant:", error);
    }
  }
}

// Handle process termination gracefully
process.on("SIGINT", () => {
  console.log("\n\n✅ Session terminated. Thank you for banking with SecureBank!");
  if (idleTimer) {
    clearTimeout(idleTimer);
  }
  process.exit(0);
});

main().catch(console.error);
```
