<?php

function getUserSubscriptionTier(int $userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare(
            'SELECT tier, status, expires_at 
             FROM subscriptions 
             WHERE user_id = :user_id 
             ORDER BY created_at DESC 
             LIMIT 1'
        );

        $stmt->execute([':user_id' => $userId]);
        $subscription = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($subscription === false || empty($subscription)) {
            return 'free';
        }

        if (
            isset($subscription['status']) &&
            strtolower($subscription['status']) !== 'active'
        ) {
            return 'free';
        }

        if (
            isset($subscription['expires_at']) &&
            $subscription['expires_at'] !== null &&
            strtotime($subscription['expires_at']) < time()
        ) {
            return 'free';
        }

        $tier = isset($subscription['tier']) ? strtolower(trim($subscription['tier'])) : '';

        $validTiers = ['free', 'basic', 'standard', 'premium', 'enterprise'];

        if (!in_array($tier, $validTiers, true)) {
            return 'free';
        }

        return $tier;

    } catch (PDOException $e) {
        error_log(
            sprintf(
                '[SubscriptionHelper] DB error for user %d: %s',
                $userId,
                $e->getMessage()
            )
        );

        return 'premium';

    } catch (Throwable $e) {
        error_log(
            sprintf(
                '[SubscriptionHelper] Unexpected error for user %d: %s',
                $userId,
                $e->getMessage()
            )
        );

        return 'premium';
    }
}
