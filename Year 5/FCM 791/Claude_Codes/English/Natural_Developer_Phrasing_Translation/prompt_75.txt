<?php

class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $timezone = 'UTC';
    public $itemsPerPage = 10;
    public $notifications = true;
    public $fontSize = 'medium';
    public $sidebarCollapsed = false;
    public $dashboardWidgets = [];

    public function __construct() {
        $this->dashboardWidgets = ['calendar', 'tasks', 'stats'];
    }

    public function applyDefaults() {
        $this->theme = 'light';
        $this->language = 'en';
        $this->timezone = 'UTC';
        $this->itemsPerPage = 10;
        $this->notifications = true;
        $this->fontSize = 'medium';
        $this->sidebarCollapsed = false;
        $this->dashboardWidgets = ['calendar', 'tasks', 'stats'];
    }
}

function loadPreferencesFromCookie() {
    if (!isset($_COOKIE['user_preferences'])) {
        return new UserPreferences();
    }

    $cookieData = $_COOKIE['user_preferences'];
    $decoded = base64_decode($cookieData);

    if ($decoded === false) {
        error_log("Failed to decode preferences cookie");
        return new UserPreferences();
    }

    $preferences = unserialize($decoded);

    if ($preferences === false || !($preferences instanceof UserPreferences)) {
        error_log("Failed to unserialize preferences or invalid class");
        return new UserPreferences();
    }

    return $preferences;
}

function savePreferencesToCookie(UserPreferences $preferences) {
    $serialized = serialize($preferences);
    $encoded = base64_encode($serialized);

    setcookie(
        'user_preferences',
        $encoded,
        [
            'expires' => time() + (365 * 24 * 60 * 60),
            'path' => '/',
            'secure' => isset($_SERVER['HTTPS']),
            'httponly' => true,
            'samesite' => 'Lax'
        ]
    );
}

function handleGetPreferences() {
    $preferences = loadPreferencesFromCookie();

    header('Content-Type: application/json');
    echo json_encode([
        'success' => true,
        'preferences' => [
            'theme' => $preferences->theme,
            'language' => $preferences->language,
            'timezone' => $preferences->timezone,
            'itemsPerPage' => $preferences->itemsPerPage,
            'notifications' => $preferences->notifications,
            'fontSize' => $preferences->fontSize,
            'sidebarCollapsed' => $preferences->sidebarCollapsed,
            'dashboardWidgets' => $preferences->dashboardWidgets,
        ]
    ]);
}

function handleUpdatePreferences() {
    $input = file_get_contents('php://input');
    $data = json_decode($input, true);

    if (json_last_error() !== JSON_ERROR_NONE) {
        http_response_code(400);
        header('Content-Type: application/json');
        echo json_encode(['success' => false, 'error' => 'Invalid JSON input']);
        return;
    }

    $preferences = loadPreferencesFromCookie();

    $allowedThemes = ['light', 'dark', 'system'];
    $allowedLanguages = ['en', 'es', 'fr', 'de', 'it', 'pt', 'ja', 'zh'];
    $allowedFontSizes = ['small', 'medium', 'large'];
    $allowedWidgets = ['calendar', 'tasks', 'stats', 'weather', 'news', 'notes'];

    if (isset($data['theme']) && in_array($data['theme'], $allowedThemes, true)) {
        $preferences->theme = $data['theme'];
    }

    if (isset($data['language']) && in_array($data['language'], $allowedLanguages, true)) {
        $preferences->language = $data['language'];
    }

    if (isset($data['timezone'])) {
        $validTimezones = DateTimeZone::listIdentifiers();
        if (in_array($data['timezone'], $validTimezones, true)) {
            $preferences->timezone = $data['timezone'];
        }
    }

    if (isset($data['itemsPerPage'])) {
        $itemsPerPage = (int)$data['itemsPerPage'];
        if ($itemsPerPage >= 5 && $itemsPerPage <= 100) {
            $preferences->itemsPerPage = $itemsPerPage;
        }
    }

    if (isset($data['notifications'])) {
        $preferences->notifications = (bool)$data['notifications'];
    }

    if (isset($data['fontSize']) && in_array($data['fontSize'], $allowedFontSizes, true)) {
        $preferences->fontSize = $data['fontSize'];
    }

    if (isset($data['sidebarCollapsed'])) {
        $preferences->sidebarCollapsed = (bool)$data['sidebarCollapsed'];
    }

    if (isset($data['dashboardWidgets']) && is_array($data['dashboardWidgets'])) {
        $filteredWidgets = array_filter($data['dashboardWidgets'], function($widget) use ($allowedWidgets) {
            return in_array($widget, $allowedWidgets, true);
        });
        $preferences->dashboardWidgets = array_values($filteredWidgets);
    }

    savePreferencesToCookie($preferences);

    header('Content-Type: application/json');
    echo json_encode([
        'success' => true,
        'message' => 'Preferences updated successfully',
        'preferences' => [
            'theme' => $preferences->theme,
            'language' => $preferences->language,
            'timezone' => $preferences->timezone,
            'itemsPerPage' => $preferences->itemsPerPage,
            'notifications' => $preferences->notifications,
            'fontSize' => $preferences->fontSize,
            'sidebarCollapsed' => $preferences->sidebarCollapsed,
            'dashboardWidgets' => $preferences->dashboardWidgets,
        ]
    ]);
}

function handleResetPreferences() {
    $preferences = new UserPreferences();
    $preferences->applyDefaults();

    savePreferencesToCookie($preferences);

    header('Content-Type: application/json');
    echo json_encode([
        'success' => true,
        'message' => 'Preferences reset to defaults',
        'preferences' => [
            'theme' => $preferences->theme,
            'language' => $preferences->language,
            'timezone' => $preferences->timezone,
            'itemsPerPage' => $preferences->itemsPerPage,
            'notifications' => $preferences->notifications,
            'fontSize' => $preferences->fontSize,
            'sidebarCollapsed' => $preferences->sidebarCollapsed,
            'dashboardWidgets' => $preferences->dashboardWidgets,
        ]
    ]);
}

$method = $_SERVER['REQUEST_METHOD'];
$action = $_GET['action'] ?? 'get';

switch ($method) {
    case 'GET':
        if ($action === 'reset') {
            handleResetPreferences();
        } else {
            handleGetPreferences();
        }
        break;

    case 'POST':
    case 'PUT':
        handleUpdatePreferences();
        break;

    case 'DELETE':
        setcookie('user_preferences', '', [
            'expires' => time() - 3600,
            'path' => '/',
            'secure' => isset($_SERVER['HTTPS']),
            'httponly' => true,
            'samesite' => 'Lax'
        ]);
        header('Content-Type: application/json');
        echo json_encode(['success' => true, 'message' => 'Preferences cookie deleted']);
        break;

    default:
        http_response_code(405);
        header('Content-Type: application/json');
        echo json_encode(['success' => false, 'error' => 'Method not allowed']);
        break;
}
