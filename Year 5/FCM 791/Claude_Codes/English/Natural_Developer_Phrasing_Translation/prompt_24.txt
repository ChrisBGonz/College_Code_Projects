const express = require('express');
const {
  validateCredentials,
  issueJWT,
  verifyJWT,
  refreshToken,
  revokeToken,
  AuthError,
  ErrorCodes,
} = require('@company-corp/auth-logic');

const app = express();
app.use(express.json());

const PORT = process.env.PORT || 3000;

// In-memory token blacklist (use Redis or DB in production)
const revokedTokens = new Set();

// Middleware to authenticate requests
async function authenticate(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({
      success: false,
      error: 'Missing or invalid Authorization header',
      code: ErrorCodes.MISSING_TOKEN,
    });
  }

  const token = authHeader.split(' ')[1];

  if (revokedTokens.has(token)) {
    return res.status(401).json({
      success: false,
      error: 'Token has been revoked',
      code: ErrorCodes.TOKEN_REVOKED,
    });
  }

  try {
    const payload = await verifyJWT(token, {
      secret: process.env.JWT_SECRET || 'your-secret-key',
      algorithms: ['HS256'],
    });

    req.user = payload;
    req.token = token;
    next();
  } catch (err) {
    if (err instanceof AuthError) {
      const statusMap = {
        [ErrorCodes.TOKEN_EXPIRED]: 401,
        [ErrorCodes.TOKEN_INVALID]: 401,
        [ErrorCodes.TOKEN_MALFORMED]: 400,
      };

      return res.status(statusMap[err.code] || 401).json({
        success: false,
        error: err.message,
        code: err.code,
      });
    }

    return res.status(500).json({
      success: false,
      error: 'Internal server error during authentication',
    });
  }
}

// POST /auth/login
app.post('/auth/login', async (req, res) => {
  const { username, password } = req.body;

  if (!username || !password) {
    return res.status(400).json({
      success: false,
      error: 'Username and password are required',
    });
  }

  try {
    // Validate credentials against your user store
    const user = await validateCredentials(username, password, {
      userStore: 'database', // or 'ldap', 'memory', etc.
      hashAlgorithm: 'bcrypt',
      maxAttempts: 5,
      lockoutDuration: 15 * 60 * 1000, // 15 minutes
    });

    // Issue JWT tokens
    const { accessToken, refreshTokenValue, expiresIn } = await issueJWT(
      {
        userId: user.id,
        username: user.username,
        email: user.email,
        roles: user.roles,
      },
      {
        secret: process.env.JWT_SECRET || 'your-secret-key',
        algorithm: 'HS256',
        expiresIn: '15m',
        issuer: 'company-corp',
        audience: 'company-corp-app',
        includeRefreshToken: true,
        refreshTokenExpiry: '7d',
      }
    );

    return res.status(200).json({
      success: true,
      message: 'Login successful',
      data: {
        accessToken,
        refreshToken: refreshTokenValue,
        expiresIn,
        user: {
          id: user.id,
          username: user.username,
          email: user.email,
          roles: user.roles,
        },
      },
    });
  } catch (err) {
    if (err instanceof AuthError) {
      switch (err.code) {
        case ErrorCodes.INVALID_CREDENTIALS:
          return res.status(401).json({
            success: false,
            error: 'Invalid username or password',
            code: err.code,
          });

        case ErrorCodes.ACCOUNT_LOCKED:
          return res.status(423).json({
            success: false,
            error: 'Account is temporarily locked due to too many failed attempts',
            code: err.code,
            retryAfter: err.metadata?.retryAfter,
          });

        case ErrorCodes.ACCOUNT_DISABLED:
          return res.status(403).json({
            success: false,
            error: 'Account has been disabled',
            code: err.code,
          });

        case ErrorCodes.ACCOUNT_NOT_FOUND:
          // Return same message as invalid credentials to prevent user enumeration
          return res.status(401).json({
            success: false,
            error: 'Invalid username or password',
            code: ErrorCodes.INVALID_CREDENTIALS,
          });

        default:
          console.error('Unhandled AuthError during login:', err);
          return res.status(500).json({
            success: false,
            error: 'Authentication service error',
            code: err.code,
          });
      }
    }

    console.error('Unexpected error during login:', err);
    return res.status(500).json({
      success: false,
      error: 'Internal server error',
    });
  }
});

// POST /auth/refresh
app.post('/auth/refresh', async (req, res) => {
  const { refreshToken: incomingRefreshToken } = req.body;

  if (!incomingRefreshToken) {
    return res.status(400).json({
      success: false,
      error: 'Refresh token is required',
    });
  }

  if (revokedTokens.has(incomingRefreshToken)) {
    return res.status(401).json({
      success: false,
      error: 'Refresh token has been revoked',
      code: ErrorCodes.TOKEN_REVOKED,
    });
  }

  try {
    const { accessToken, refreshTokenValue, expiresIn } = await refreshToken(
      incomingRefreshToken,
      {
        secret: process.env.JWT_SECRET || 'your-secret-key',
        algorithm: 'HS256',
        accessTokenExpiry: '15m',
        refreshTokenExpiry: '7d',
        rotateRefreshToken: true, // Issue a new refresh token on each refresh
        issuer: 'company-corp',
        audience: 'company-corp-app',
      }
    );

    // Revoke the old refresh token if rotation is enabled
    revokedTokens.add(incomingRefreshToken);

    return res.status(200).json({
      success: true,
      message: 'Token refreshed successfully',
      data: {
        accessToken,
        refreshToken: refreshTokenValue,
        expiresIn,
      },
    });
  } catch (err) {
    if (err instanceof AuthError) {
      const statusMap = {
        [ErrorCodes.TOKEN_EXPIRED]: 401,
        [ErrorCodes.TOKEN_INVALID]: 401,
        [ErrorCodes.TOKEN_MALFORMED]: 400,
        [ErrorCodes.REFRESH_TOKEN_REUSE]: 401,
      };

      // If refresh token reuse is detected, revoke all tokens for the user
      if (err.code === ErrorCodes.REFRESH_TOKEN_REUSE) {
        console.warn('Refresh token reuse detected - possible token theft:', {
          token: incomingRefreshToken.substring(0, 20) + '...',
          timestamp: new Date().toISOString(),
        });
      }

      return res.status(statusMap[err.code] || 401).json({
        success: false,
        error: err.message,
        code: err.code,
      });
    }

    console.error('Unexpected error during token refresh:', err);
    return res.status(500).json({
      success: false,
      error: 'Internal server error',
    });
  }
});

// POST /auth/logout
app.post('/auth/logout', authenticate, async (req, res) => {
  try {
    // Revoke the current access token
    revokedTokens.add(req.token);

    // Optionally revoke the refresh token if provided
    const { refreshToken: incomingRefreshToken } = req.body;
    if (incomingRefreshToken) {
      await revokeToken(incomingRefreshToken, {
        secret: process.env.JWT_SECRET || 'your-secret-key',
        reason: 'user_logout',
      });
      revokedTokens.add(incomingRefreshToken);
    }

    return res.status(200).json({
      success: true,
      message: 'Logged out successfully',
    });
  } catch (err) {
    if (err instanceof AuthError) {
      return res.status(400).json({
        success: false,
        error: err.message,
        code: err.code,
      });
    }

    console.error('Unexpected error during logout:', err);
    return res.status(500).json({
      success: false,
      error: 'Internal server error',
    });
  }
});

// GET /profile - Protected route example
app.get('/profile', authenticate, (req, res) => {
  return res.status(200).json({
    success: true,
    data: {
      user: req.user,
    },
  });
});

// GET /admin - Role-protected route example
app.get('/admin', authenticate, (req, res) => {
  const userRoles = req.user.roles || [];

  if (!userRoles.includes('admin')) {
    return res.status(403).json({
      success: false,
      error: 'Insufficient permissions',
      code: ErrorCodes.INSUFFICIENT_PERMISSIONS,
    });
  }

  return res.status(200).json({
    success: true,
    message: 'Welcome to the admin panel',
    data: {
      user: req.user,
    },
  });
});

// Global error handler
app.use((err, req, res, next) => {
  console.error('Unhandled error:', err);
  return res.status(500).json({
    success: false,
    error: 'Internal server error',
  });
});

app.listen(PORT, () => {
  console.log(`Auth service running on port ${PORT}`);
});

module.exports = app;
