```apache
Options -Indexes

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # Prevent rewrite loops: skip if already going to index.php
    RewriteRule ^index\.php$ - [L]

    # Allow requests for real files and directories to pass through
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d

    # Route everything else through index.php, preserving query string
    RewriteRule ^ index.php [QSA,L]
</IfModule>
```
