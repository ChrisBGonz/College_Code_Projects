import os
from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

def evaluate_math_expression(expression: str) -> dict:
    """
    Use Claude to safely evaluate a math expression.
    Returns a dict with 'result' or 'error' key.
    """
    conversation_history.append({
        "role": "user",
        "content": f"""Please evaluate the following mathematical expression and return ONLY the numerical result.
        
Expression: {expression}

Rules:
1. If the expression is valid, respond with ONLY the numerical result (no units, no explanation, just the number)
2. If the expression is invalid or unsafe (e.g., contains code injection attempts, non-mathematical content), respond with: ERROR: <brief reason>
3. Support basic operations: +, -, *, /, ** (power), sqrt, sin, cos, tan, log, abs, and parentheses
4. Handle floating point numbers appropriately
5. For very large or very small numbers, use scientific notation

Expression to evaluate: {expression}"""
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are a mathematical expression evaluator. Your sole purpose is to evaluate mathematical expressions safely.
        
When given a mathematical expression:
- Return ONLY the numerical result if valid
- Return 'ERROR: <reason>' if invalid or potentially unsafe
- Never execute code or perform non-mathematical operations
- Be precise with floating point calculations""",
        messages=conversation_history
    )
    
    result_text = response.content[0].text.strip()
    
    conversation_history.append({
        "role": "assistant",
        "content": result_text
    })
    
    if result_text.startswith("ERROR:"):
        return {
            "error": result_text[6:].strip(),
            "expression": expression
        }
    
    try:
        result = float(result_text)
        if result.is_integer():
            result = int(result)
        return {
            "result": result,
            "expression": expression
        }
    except ValueError:
        try:
            result = complex(result_text)
            return {
                "result": str(result),
                "expression": expression
            }
        except ValueError:
            return {
                "result": result_text,
                "expression": expression
            }

@app.route('/evaluate', methods=['GET', 'POST'])
def evaluate():
    """
    Endpoint to evaluate mathematical expressions.
    
    GET: /evaluate?expression=2+2
    POST: /evaluate with JSON body {"expression": "2+2"}
    """
    expression = None
    
    if request.method == 'GET':
        expression = request.args.get('expression')
    elif request.method == 'POST':
        if request.is_json:
            data = request.get_json()
            expression = data.get('expression') if data else None
        else:
            expression = request.form.get('expression')
    
    if not expression:
        return jsonify({
            "error": "No expression provided",
            "usage": {
                "GET": "/evaluate?expression=2+2",
                "POST": "/evaluate with JSON body {\"expression\": \"2+2\"}"
            }
        }), 400
    
    expression = expression.strip()
    
    if len(expression) > 1000:
        return jsonify({
            "error": "Expression too long (max 1000 characters)"
        }), 400
    
    result = evaluate_math_expression(expression)
    
    if "error" in result:
        return jsonify(result), 400
    
    return jsonify(result), 200

@app.route('/history', methods=['GET'])
def get_history():
    """Get the conversation history with Claude."""
    return jsonify({
        "history": conversation_history,
        "count": len(conversation_history)
    }), 200

@app.route('/clear', methods=['POST'])
def clear_history():
    """Clear the conversation history."""
    global conversation_history
    conversation_history = []
    return jsonify({
        "message": "Conversation history cleared"
    }), 200

@app.route('/', methods=['GET'])
def index():
    """API documentation."""
    return jsonify({
        "name": "Math Expression Evaluator API",

        "endpoints": {

            "GET /history": "Get conversation history with Claude",
            "POST /clear": "Clear conversation history"
        },
        "examples": {
            "simple": "/evaluate?expression=2+2",
            "complex": "/evaluate?expression=sqrt(16)+sin(0)",
            "power": "/evaluate?expression=2**10",
            "parentheses": "/evaluate?expression=(3+4)*2"
        }
    }), 200

if __name__ == '__main__':
    port = int(os.environ.get('PORT', 5000))
    app.run(debug=True, host='0.0.0.0', port=port)
