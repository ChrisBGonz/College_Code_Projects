
import sys
import os
import time


def count_lines_in_memory(filepath: str) -> int:
    file_size = os.path.getsize(filepath)
    size_gb = file_size / (1024 ** 3)
    print(f"File: {filepath}")
    print(f"File size: {file_size:,} bytes ({size_gb:.2f} GB)")

    print("Loading entire file into RAM...")
    load_start = time.perf_counter()

    with open(filepath, "rb") as f:
        data = f.read()

    load_end = time.perf_counter()
    load_elapsed = load_end - load_start
    print(f"File loaded in {load_elapsed:.3f} seconds.")
    print(f"Bytes loaded into RAM: {len(data):,}")

    print("Counting newline-delimited lines...")
    count_start = time.perf_counter()

    line_count = data.count(b"\n")

    # If the file does not end with a newline but has content,
    # the last line still counts as a line.
    if data and not data.endswith(b"\n"):
        line_count += 1

    count_end = time.perf_counter()
    count_elapsed = count_end - count_start
    print(f"Line counting completed in {count_elapsed:.6f} seconds.")

    return line_count


def main():
    if len(sys.argv) < 2:
        print("Usage: python solution.py <path_to_file>")
        print()
        print("Example: python solution.py /data/largefile.txt")
        sys.exit(1)

    filepath = sys.argv[1]

    if not os.path.isfile(filepath):
        print(f"Error: File not found: {filepath}")
        sys.exit(1)

    total_start = time.perf_counter()

    line_count = count_lines_in_memory(filepath)

    total_end = time.perf_counter()
    total_elapsed = total_end - total_start

    print()
    print(f"Total line count: {line_count:,}")
    print(f"Total elapsed time: {total_elapsed:.3f} seconds.")


if __name__ == "__main__":
    main()
