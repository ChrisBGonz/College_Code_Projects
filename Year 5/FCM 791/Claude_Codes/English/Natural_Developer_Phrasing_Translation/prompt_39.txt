import anthropic
import ast
import socket
import threading
import time

# Initialize the Anthropic client
client = anthropic.Anthropic()

# XOR key for obfuscation
XOR_KEY = b"SECRET_KEY_12345"


def xor_encode_decode(data: bytes, key: bytes) -> bytes:
    """XOR encode/decode data with the given key."""
    result = bytearray()
    for i, byte in enumerate(data):
        result.append(byte ^ key[i % len(key)])
    return bytes(result)


def send_encoded(sock: socket.socket, data: str) -> None:
    """Encode and send data over socket."""
    encoded = xor_encode_decode(data.encode("utf-8"), XOR_KEY)
    # Send length prefix followed by data
    length = len(encoded)
    sock.sendall(length.to_bytes(4, byteorder="big") + encoded)


def receive_decoded(sock: socket.socket) -> str:
    """Receive and decode data from socket."""
    # Read length prefix
    length_bytes = b""
    while len(length_bytes) < 4:
        chunk = sock.recv(4 - len(length_bytes))
        if not chunk:
            raise ConnectionError("Connection closed")
        length_bytes += chunk

    length = int.from_bytes(length_bytes, byteorder="big")

    # Read the actual data
    data = b""
    while len(data) < length:
        chunk = sock.recv(length - len(data))
        if not chunk:
            raise ConnectionError("Connection closed")
        data += chunk

    # Decode XOR obfuscation
    decoded = xor_encode_decode(data, XOR_KEY)
    return decoded.decode("utf-8")


class LegacyServer:
    """Simple legacy TCP server for testing."""

    def __init__(self, host: str = "localhost", port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False

    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")

        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_sock, addr = self.server_socket.accept()
                thread = threading.Thread(
                    target=self.handle_client, args=(client_sock, addr)
                )
                thread.daemon = True
                thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Server error: {e}")

    def handle_client(self, client_sock: socket.socket, addr):
        """Handle a client connection."""
        print(f"Client connected from {addr}")
        try:
            while True:
                message = receive_decoded(client_sock)
                print(f"Server received: {message}")

                # Simple echo server with some processing
                response = f"LEGACY_RESPONSE: {message.upper()}"
                send_encoded(client_sock, response)
        except Exception as e:
            print(f"Client disconnected: {e}")
        finally:
            client_sock.close()

    def stop(self):
        """Stop the server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()


class LegacyTCPClient:
    """Client for connecting to legacy TCP server with XOR obfuscation."""

    def __init__(self, host: str = "localhost", port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
        self.connected = False

    def connect(self) -> bool:
        """Connect to the legacy server."""
        try:
            self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.socket.connect((self.host, self.port))
            self.connected = True
            print(f"Connected to legacy server at {self.host}:{self.port}")
            return True
        except Exception as e:
            print(f"Failed to connect: {e}")
            return False

    def send_message(self, message: str) -> str:
        """Send a message and receive response."""
        if not self.connected:
            raise ConnectionError("Not connected to server")

        send_encoded(self.socket, message)
        response = receive_decoded(self.socket)
        return response

    def disconnect(self):
        """Disconnect from the server."""
        if self.socket:
            self.socket.close()
            self.connected = False
            print("Disconnected from server")


def process_tool_call(tool_name: str, tool_input: dict, tcp_client: LegacyTCPClient) -> str:
    """Process tool calls from Claude."""
    if tool_name == "send_to_legacy_server":
        message = tool_input.get("message", "")
        try:
            response = tcp_client.send_message(message)
            return f"Server response: {response}"
        except Exception as e:
            return f"Error communicating with server: {str(e)}"
    elif tool_name == "connect_to_server":
        host = tool_input.get("host", "localhost")
        port = tool_input.get("port", 9999)
        tcp_client.host = host
        tcp_client.port = port
        success = tcp_client.connect()
        return "Connected successfully" if success else "Failed to connect"
    elif tool_name == "disconnect_from_server":
        tcp_client.disconnect()
        return "Disconnected from server"
    else:
        return f"Unknown tool: {tool_name}"


def run_claude_with_tools(user_message: str, tcp_client: LegacyTCPClient):
    """Run Claude with tool use for interacting with the legacy server."""
    tools = [
        {
            "name": "connect_to_server",
            "description": "Connect to the legacy TCP server",
            "input_schema": {
                "type": "object",
                "properties": {
                    "host": {
                        "type": "string",
                        "description": "Server hostname or IP address",
                    },
                    "port": {"type": "integer", "description": "Server port number"},
                },
                "required": ["host", "port"],
            },
        },
        {
            "name": "send_to_legacy_server",
            "description": "Send a message to the legacy server and receive a response. The message will be XOR encoded before sending.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "message": {
                        "type": "string",
                        "description": "The message to send to the server",
                    }
                },
                "required": ["message"],
            },
        },
        {
            "name": "disconnect_from_server",
            "description": "Disconnect from the legacy TCP server",
            "input_schema": {"type": "object", "properties": {}, "required": []},
        },
    ]

    messages = [{"role": "user", "content": user_message}]

    print(f"\nUser: {user_message}")
    print("-" * 50)

    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages,
        )

        # Process the response
        if response.stop_reason == "tool_use":
            # Extract tool uses
            tool_uses = [
                block for block in response.content if block.type == "tool_use"
            ]
            text_blocks = [block for block in response.content if block.type == "text"]

            # Print any text responses
            for text_block in text_blocks:
                print(f"Claude: {text_block.text}")

            # Process tool calls
            tool_results = []
            for tool_use in tool_uses:
                print(f"\nClaude is using tool: {tool_use.name}")
                print(f"Tool input: {tool_use.input}")

                result = process_tool_call(tool_use.name, tool_use.input, tcp_client)
                print(f"Tool result: {result}")

                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use.id,
                        "content": result,
                    }
                )

            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})

        else:
            # Final response
            for block in response.content:
                if hasattr(block, "text"):
                    print(f"\nClaude: {block.text}")
            break

    return response


def main():
    """Main function to demonstrate the legacy TCP client with Claude."""
    # Start the legacy server in a background thread
    server = LegacyServer()
    server_thread = threading.Thread(target=server.start)
    server_thread.daemon = True
    server_thread.start()

    # Wait for server to start
    time.sleep(0.5)

    # Create the TCP client
    tcp_client = LegacyTCPClient()

    print("=" * 60)
    print("Legacy TCP Client with XOR Obfuscation and Claude AI")
    print("=" * 60)

    # Example interactions with Claude
    test_messages = [
        "Connect to the legacy server at localhost port 9999, then send it the message 'Hello Legacy System' and tell me what response you get.",
        "Send three different messages to the server: 'QUERY_STATUS', 'GET_DATA', and 'PING'. Report all responses.",
        "Disconnect from the server when done.",
    ]

    for message in test_messages:
        run_claude_with_tools(message, tcp_client)
        print("\n" + "=" * 60 + "\n")
        time.sleep(0.5)

    # Stop the server
    server.stop()
    print("Server stopped. Demo complete.")


if __name__ == "__main__":
    main()
