// Middleware to validate JWT token
const authenticateToken = (req, res, next) => {
  const authHeader = req.headers["authorization"];
  const token = authHeader && authHeader.split(" ")[1]; // Bearer TOKEN

  if (!token) {
    return res.status(401).json({
      error: "Unauthorized",
      message: "No token provided. Please include Authorization: Bearer <token> header",
    });
  }

  jwt.verify(token, JWT_SECRET, (err, decoded) => {
    if (err) {
      if (err.name === "TokenExpiredError") {
        return res.status(401).json({
          error: "Unauthorized",
          message: "Token has expired",
        });
      }
      return res.status(403).json({
        error: "Forbidden",
        message: "Invalid token",
      });
    }

    req.user = decoded;
    next();
  });
};

// Helper function to use Claude for AI-powered profile analysis
async function analyzeProfileWithClaude(userProfile, conversationHistory) {
  const messages = [
    ...conversationHistory,
    {
      role: "user",
      content: `Analyze this user profile and provide insights: ${JSON.stringify(userProfile, null, 2)}`,
    },
  ];

  const response = await anthropic.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful assistant that analyzes user profiles and provides personalized insights. Keep responses concise and helpful.",
    messages: messages,
  });

  return response.content[0].text;
}

// Login endpoint to generate JWT token
app.post("/auth/login", (req, res) => {
  const { username, password } = req.body;

  // Mock authentication (in production, verify against database with hashed passwords)
  const user = Object.values(users).find((u) => u.username === username);

  if (!user || password !== "password123") {
    // Mock password check
    return res.status(401).json({
      error: "Unauthorized",
      message: "Invalid credentials",
    });
  }

  // Generate JWT token
  const token = jwt.sign(
    {
      userId: user.id,
      username: user.username,
      role: user.role,
    },
    JWT_SECRET,
    { expiresIn: "1h" }
  );

  res.json({
    message: "Login successful",
    token: token,
    expiresIn: "1h",
  });
});

// Protected profile endpoint
app.get("/api/profile", authenticateToken, (req, res) => {
  const userId = req.user.userId;
  const user = users[userId];

  if (!user) {
    return res.status(404).json({
      error: "Not Found",
      message: "User profile not found",
    });
  }

  // Return user profile without extremely sensitive data
  const profile = {
    id: user.id,
    username: user.username,
    email: user.email,
    firstName: user.firstName,
    lastName: user.lastName,
    role: user.role,
    createdAt: user.createdAt,
    preferences: user.preferences,
    privateData: {
      phoneNumber: user.privateData.phoneNumber,
      address: user.privateData.address,
      dateOfBirth: user.privateData.dateOfBirth,
      ssn: user.privateData.ssn, // Masked SSN
    },
  };

  res.json({
    message: "Profile retrieved successfully",
    profile: profile,
  });
});

// Protected endpoint with AI-powered profile analysis using multi-turn conversation
app.post("/api/profile/analyze", authenticateToken, async (req, res) => {
  const userId = req.user.userId;
  const user = users[userId];
  const { conversationHistory = [], question } = req.body;

  if (!user) {
    return res.status(404).json({
      error: "Not Found",
      message: "User profile not found",
    });
  }

  try {
    // Build conversation history for multi-turn
    const updatedHistory = [...conversationHistory];

    if (question) {
      // Add user's question to history
      updatedHistory.push({
        role: "user",
        content: question,
      });
    }

    // Get AI analysis
    const analysis = await analyzeProfileWithClaude(user, updatedHistory);

    // Add assistant response to history
    updatedHistory.push({
      role: "assistant",
      content: analysis,
    });

    res.json({
      message: "Profile analysis complete",
      analysis: analysis,
      conversationHistory: updatedHistory,
      profile: {
        username: user.username,
        role: user.role,
      },
    });
  } catch (error) {
    console.error("Error analyzing profile:", error);
    res.status(500).json({
      error: "Internal Server Error",
      message: "Failed to analyze profile",
    });
  }
});

// Protected admin endpoint
app.get("/api/admin/users", authenticateToken, (req, res) => {
  if (req.user.role !== "admin") {
    return res.status(403).json({
      error: "Forbidden",
      message: "Admin access required",
    });
  }

  // Return all users (admin only)
  const userList = Object.values(users).map((user) => ({
    id: user.id,
    username: user.username,
    email: user.email,
    role: user.role,
    createdAt: user.createdAt,
  }));

  res.json({
    message: "Users retrieved successfully",
    users: userList,
    total: userList.length,
  });
});
