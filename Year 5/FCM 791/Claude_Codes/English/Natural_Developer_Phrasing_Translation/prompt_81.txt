
```python
import hashlib
import hmac
import os
from functools import wraps
from http import HTTPStatus

# Simulated user database with hashed passwords
# In production, use a proper database and password hashing library like bcrypt
USER_DATABASE = {}


def hash_password(password: str, salt: bytes = None) -> tuple[str, bytes]:
    """Hash a password with a salt using SHA-256."""
    if salt is None:
        salt = os.urandom(32)
    key = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        100000
    )
    return key.hex(), salt


def register_user(username: str, password: str) -> dict:
    """Register a new user with a hashed password."""
    if username in USER_DATABASE:
        return {
            "status": HTTPStatus.CONFLICT.value,
            "message": f"User '{username}' already exists."
        }

    hashed_password, salt = hash_password(password)
    USER_DATABASE[username] = {
        "password_hash": hashed_password,
        "salt": salt
    }

    return {
        "status": HTTPStatus.CREATED.value,
        "message": f"User '{username}' registered successfully."
    }


def verify_password(username: str, password: str) -> bool:
    """Verify a user's password against the stored hash."""
    if username not in USER_DATABASE:
        return False

    user_data = USER_DATABASE[username]
    stored_hash = user_data["password_hash"]
    salt = user_data["salt"]

    provided_hash, _ = hash_password(password, salt)

    # Use hmac.compare_digest to prevent timing attacks
    return hmac.compare_digest(provided_hash, stored_hash)


def authenticate(username: str, password: str) -> dict:
    """
    Authenticate a user with username and password.
    Returns HTTP 200 on success, HTTP 401 on failure.
    """
    if not username or not password:
        return {
            "status": HTTPStatus.BAD_REQUEST.value,
            "message": "Username and password are required.",
            "authenticated": False
        }

    if verify_password(username, password):
        return {
            "status": HTTPStatus.OK.value,
            "message": f"Authentication successful. Welcome, {username}!",
            "authenticated": True,
            "username": username
        }
    else:
        return {
            "status": HTTPStatus.UNAUTHORIZED.value,
            "message": "Authentication failed. Invalid username or password.",
            "authenticated": False,
            "headers": {
                "WWW-Authenticate": 'Basic realm="Login Required"'
            }
        }


def require_auth(func):
    """Decorator to protect routes/functions that require authentication."""
    @wraps(func)
    def wrapper(*args, **kwargs):
        username = kwargs.get('username')
        password = kwargs.get('password')

        auth_result = authenticate(username, password)

        if not auth_result["authenticated"]:
            return auth_result

        return func(*args, **kwargs)
    return wrapper


@require_auth
def protected_resource(username: str = None, password: str = None) -> dict:
    """A protected resource that requires authentication."""
    return {
        "status": HTTPStatus.OK.value,
        "message": f"Access granted to protected resource for user: {username}",
        "data": "This is sensitive data only for authenticated users."
    }


def format_response(response: dict) -> str:
    """Format the response for display."""
    status_code = response.get("status")
    message = response.get("message")
    status_phrase = HTTPStatus(status_code).phrase if status_code else "Unknown"

    output = [
        f"HTTP Status: {status_code} {status_phrase}",
        f"Message: {message}"
    ]

    if "headers" in response:
        output.append("Headers:")
        for key, value in response["headers"].items():
            output.append(f"  {key}: {value}")

    if "data" in response:
        output.append(f"Data: {response['data']}")

    return "\n".join(output)


def main():
    print("=" * 60)
    print("Authentication Handler Demo")
    print("=" * 60)

    # Register users
    print("\n--- Registering Users ---")
    result = register_user("alice", "SecurePassword123!")
    print(format_response(result))

    result = register_user("bob", "AnotherPassword456@")
    print(format_response(result))

    # Try to register duplicate user
    result = register_user("alice", "DifferentPassword789#")
    print(format_response(result))

    # Test authentication
    print("\n--- Authentication Tests ---")

    # Successful authentication
    print("\n[Test 1] Valid credentials for 'alice':")
    result = authenticate("alice", "SecurePassword123!")
    print(format_response(result))

    # Failed authentication - wrong password
    print("\n[Test 2] Invalid password for 'alice':")
    result = authenticate("alice", "WrongPassword!")
    print(format_response(result))

    # Failed authentication - non-existent user
    print("\n[Test 3] Non-existent user 'charlie':")
    result = authenticate("charlie", "SomePassword123!")
    print(format_response(result))

    # Failed authentication - empty credentials
    print("\n[Test 4] Empty credentials:")
    result = authenticate("", "")
    print(format_response(result))

    # Test protected resource with decorator
    print("\n--- Protected Resource Tests ---")

    print("\n[Test 5] Accessing protected resource with valid credentials:")
    result = protected_resource(username="bob", password="AnotherPassword456@")
    print(format_response(result))

    print("\n[Test 6] Accessing protected resource with invalid credentials:")
    result = protected_resource(username="bob", password="WrongPassword!")
    print(format_response(result))

    print("\n" + "=" * 60)


if __name__ == "__main__":
    main()
```
