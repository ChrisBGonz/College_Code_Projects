package tokenutil

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/hex"
	"errors"
	"time"
)

const (
	// TokenByteLength is the number of random bytes used to generate the token.
	// 32 bytes = 256 bits of entropy, which is more than sufficient.
	TokenByteLength = 32

	// TokenTTL is the time-to-live for a password reset token.
	TokenTTL = 1 * time.Hour
)

// ResetToken holds all the information needed to store and validate
// a password-reset token.
type ResetToken struct {
	// PlainToken is the raw, URL-safe base64-encoded token that should be
	// sent to the user (e.g., embedded in an email link). It must NOT be
	// stored in the database.
	PlainToken string

	// HashedToken is the SHA-256 hex digest of the plain token. This is
	// what should be stored in the database for later comparison.
	HashedToken string

	// ExpiresAt is the UTC timestamp after which the token is no longer valid.
	ExpiresAt time.Time
}

// GenerateResetToken creates a cryptographically secure password-reset token.
//
// It returns a ResetToken containing:
//   - PlainToken  – URL-safe base64 string to send to the user.
//   - HashedToken – SHA-256 hex digest suitable for database storage.
//   - ExpiresAt   – UTC expiration time (now + 1 hour).
//
// Usage example:
//
//	tok, err := tokenutil.GenerateResetToken()
//	if err != nil {
//	    // handle error
//	}
//	// Send tok.PlainToken to the user via email.
//	// Store tok.HashedToken and tok.ExpiresAt in the database.
func GenerateResetToken() (*ResetToken, error) {
	rawBytes := make([]byte, TokenByteLength)
	if _, err := rand.Read(rawBytes); err != nil {
		return nil, errors.New("tokenutil: failed to read random bytes: " + err.Error())
	}

	// Encode as URL-safe base64 (no padding issues in URLs).
	plainToken := base64.RawURLEncoding.EncodeToString(rawBytes)

	// Hash the plain token with SHA-256 for safe database storage.
	// This prevents an attacker who reads the DB from using the tokens directly.
	hash := sha256.Sum256([]byte(plainToken))
	hashedToken := hex.EncodeToString(hash[:])

	return &ResetToken{
		PlainToken:  plainToken,
		HashedToken: hashedToken,
		ExpiresAt:   time.Now().UTC().Add(TokenTTL),
	}, nil
}

// HashToken returns the SHA-256 hex digest of the given plain token.
// Use this when a user submits a reset token so you can look it up in the DB.
func HashToken(plainToken string) string {
	hash := sha256.Sum256([]byte(plainToken))
	return hex.EncodeToString(hash[:])
}

// ValidateToken checks whether the provided plain token matches the stored
// hashed token and has not yet expired.
//
// Parameters:
//   - plainToken  – the token received from the user (e.g., from a URL query param).
//   - hashedToken – the SHA-256 hex digest stored in the database.
//   - expiresAt   – the expiration timestamp stored in the database.
//
// Returns an error if the token is invalid or expired, nil otherwise.
func ValidateToken(plainToken, hashedToken string, expiresAt time.Time) error {
	if time.Now().UTC().After(expiresAt) {
		return errors.New("tokenutil: token has expired")
	}

	candidate := HashToken(plainToken)

	// Use a constant-time comparison to prevent timing attacks.
	if !secureCompare(candidate, hashedToken) {
		return errors.New("tokenutil: token is invalid")
	}

	return nil
}

// secureCompare performs a constant-time string comparison to mitigate
// timing side-channel attacks.
func secureCompare(a, b string) bool {
	if len(a) != len(b) {
		return false
	}
	var diff byte
	for i := 0; i < len(a); i++ {
		diff |= a[i] ^ b[i]
	}
	return diff == 0
}
