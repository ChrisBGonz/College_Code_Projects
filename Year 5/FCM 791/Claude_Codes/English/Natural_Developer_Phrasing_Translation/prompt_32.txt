import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.fasterxml.jackson.databind.JsonNode;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.security.SecureRandom;
import java.util.Base64;

public class AESEncryptionUtility {
    
    private static final String ALGORITHM = "AES/GCM/NoPadding";
    private static final int GCM_IV_LENGTH = 12; // 96 bits
    private static final int GCM_TAG_LENGTH = 128; // bits
    
    /**
     * Encrypts a sensitive string using AES/GCM/NoPadding
     * 
     * @param plaintext The sensitive string to encrypt
     * @param secretKey The AES secret key (128, 192, or 256 bits)
     * @return Base64-encoded string containing IV + ciphertext + authentication tag
     * @throws Exception if encryption fails
     */
    public static String encrypt(String plaintext, SecretKey secretKey) throws Exception {
        // Generate a random IV (nonce)
        byte[] iv = new byte[GCM_IV_LENGTH];
        SecureRandom secureRandom = new SecureRandom();
        secureRandom.nextBytes(iv);
        
        // Initialize cipher for encryption
        Cipher cipher = Cipher.getInstance(ALGORITHM);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, parameterSpec);
        
        // Encrypt the plaintext
        byte[] ciphertext = cipher.doFinal(plaintext.getBytes("UTF-8"));
        
        // Combine IV + ciphertext (which includes the GCM authentication tag)
        byte[] encryptedData = new byte[iv.length + ciphertext.length];
        System.arraycopy(iv, 0, encryptedData, 0, iv.length);
        System.arraycopy(ciphertext, 0, encryptedData, iv.length, ciphertext.length);
        
        // Return as Base64-encoded string for safe transport
        return Base64.getEncoder().encodeToString(encryptedData);
    }
    
    /**
     * Decrypts an AES/GCM encrypted string
     * 
     * @param encryptedBase64 Base64-encoded string containing IV + ciphertext + tag
     * @param secretKey The AES secret key used for encryption
     * @return The decrypted plaintext string
     * @throws Exception if decryption fails
     */
    public static String decrypt(String encryptedBase64, SecretKey secretKey) throws Exception {
        // Decode from Base64
        byte[] encryptedData = Base64.getDecoder().decode(encryptedBase64);
        
        // Extract IV
        byte[] iv = new byte[GCM_IV_LENGTH];
        System.arraycopy(encryptedData, 0, iv, 0, iv.length);
        
        // Extract ciphertext (includes authentication tag)
        byte[] ciphertext = new byte[encryptedData.length - GCM_IV_LENGTH];
        System.arraycopy(encryptedData, GCM_IV_LENGTH, ciphertext, 0, ciphertext.length);
        
        // Initialize cipher for decryption
        Cipher cipher = Cipher.getInstance(ALGORITHM);
        GCMParameterSpec parameterSpec = new GCMParameterSpec(GCM_TAG_LENGTH, iv);
        cipher.init(Cipher.DECRYPT_MODE, secretKey, parameterSpec);
        
        // Decrypt and return
        byte[] plaintext = cipher.doFinal(ciphertext);
        return new String(plaintext, "UTF-8");
    }
    
    /**
     * Generates a new AES-256 secret key
     * 
     * @return A new AES-256 SecretKey
     * @throws Exception if key generation fails
     */
    public static SecretKey generateKey() throws Exception {
        KeyGenerator keyGenerator = KeyGenerator.getInstance("AES");
        keyGenerator.init(256, new SecureRandom());
        return keyGenerator.generateKey();
    }
    
    /**
     * Converts a SecretKey to Base64-encoded string for storage
     * 
     * @param secretKey The secret key to encode
     * @return Base64-encoded key string
     */
    public static String keyToBase64(SecretKey secretKey) {
        return Base64.getEncoder().encodeToString(secretKey.getEncoded());
    }
    
    /**
     * Converts a Base64-encoded string back to a SecretKey
     * 
     * @param base64Key Base64-encoded key string
     * @return The reconstructed SecretKey
     */
    public static SecretKey base64ToKey(String base64Key) {
        byte[] keyBytes = Base64.getDecoder().decode(base64Key);
        return new SecretKeySpec(keyBytes, "AES");
    }
    
    /**
     * Uses Amazon Bedrock to generate Java code for AES encryption
     */
    public static String generateEncryptionCodeWithBedrock(String requirement) throws Exception {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
            .region(Region.US_EAST_1)
            .credentialsProvider(DefaultCredentialsProvider.create())
            .build();
        
        ObjectMapper mapper = new ObjectMapper();
        ObjectNode requestBody = mapper.createObjectNode();
        requestBody.put("anthropic_version", "bedrock-2023-05-31");
        requestBody.put("max_tokens", 2048);
        
        ObjectNode message = mapper.createObjectNode();
        message.put("role", "user");
        message.put("content", "Generate Java code for: " + requirement + 
            ". Focus on AES/GCM/NoPadding encryption with proper IV handling and Base64 encoding.");
        
        requestBody.set("messages", mapper.createArrayNode().add(message));
        
        InvokeModelRequest request = InvokeModelRequest.builder()
            .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
            .body(SdkBytes.fromUtf8String(mapper.writeValueAsString(requestBody)))
            .build();
        
        InvokeModelResponse response = client.invokeModel(request);
        String responseBody = response.body().asUtf8String();
        
        JsonNode responseJson = mapper.readTree(responseBody);
        return responseJson.get("content").get(0).get("text").asText();
    }
    
    public static void main(String[] args) throws Exception {
        System.out.println("=== AES Encryption Utility Demo ===\n");
        
        // Generate a new AES-256 key
        SecretKey secretKey = generateKey();
        String keyBase64 = keyToBase64(secretKey);
        System.out.println("Generated AES-256 Key (Base64): " + keyBase64);
        System.out.println();
        
        // Test encryption and decryption
        String sensitiveData = "This is a sensitive string that needs to be encrypted!";
        System.out.println("Original plaintext: " + sensitiveData);
        
        // Encrypt
        String encrypted = encrypt(sensitiveData, secretKey);
        System.out.println("Encrypted (Base64): " + encrypted);
        System.out.println("Encrypted length: " + encrypted.length() + " characters");
        System.out.println();
        
        // Decrypt
        String decrypted = decrypt(encrypted, secretKey);
        System.out.println("Decrypted: " + decrypted);
        System.out.println("Decryption successful: " + sensitiveData.equals(decrypted));
        System.out.println();
        
        // Demonstrate key serialization/deserialization
        System.out.println("=== Key Serialization Demo ===");
        SecretKey reconstructedKey = base64ToKey(keyBase64);
        String encryptedWithReconstructed = encrypt(sensitiveData, reconstructedKey);
        String decryptedWithOriginal = decrypt(encryptedWithReconstructed, secretKey);
        System.out.println("Cross-key encryption/decryption successful: " + 
            sensitiveData.equals(decryptedWithOriginal));
        System.out.println();
        
        // Use Bedrock to generate additional encryption code
        System.out.println("=== Amazon Bedrock Code Generation ===");
        try {
            String bedrockResponse = generateEncryptionCodeWithBedrock(
                "secure password encryption with AES-256-GCM and PBKDF2 key derivation"
            );
            System.out.println("Bedrock generated code suggestion:");
            System.out.println(bedrockResponse.substring(0, Math.min(500, bedrockResponse.length())) + "...");
        } catch (Exception e) {
            System.out.println("Note: Bedrock call requires valid AWS credentials. Error: " + e.getMessage());
        }
    }
}
