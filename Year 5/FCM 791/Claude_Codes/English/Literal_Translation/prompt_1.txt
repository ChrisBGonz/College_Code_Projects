# Define tools for Claude to use
tools = [
    {
        "name": "get_invoice",
        "description": "Retrieves invoice details from the database by invoice ID",
        "input_schema": {
            "type": "object",
            "properties": {
                "invoice_id": {
                    "type": "string",
                    "description": "The unique identifier of the invoice (e.g., INV-001)"
                }
            },
            "required": ["invoice_id"]
        }
    },
    {
        "name": "list_all_invoices",
        "description": "Lists all available invoice IDs in the database",
        "input_schema": {
            "type": "object",
            "properties": {},
            "required": []
        }
    }
]

def get_invoice_from_db(invoice_id: str) -> dict:
    """Retrieve invoice from mock database"""
    if invoice_id in INVOICES_DB:
        return {"success": True, "invoice": INVOICES_DB[invoice_id]}
    else:
        return {"success": False, "error": f"Invoice {invoice_id} not found"}

def list_all_invoices_from_db() -> dict:
    """List all invoice IDs from mock database"""
    return {"success": True, "invoice_ids": list(INVOICES_DB.keys()), "total_count": len(INVOICES_DB)}

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude"""
    if tool_name == "get_invoice":
        result = get_invoice_from_db(tool_input["invoice_id"])
        return json.dumps(result)
    elif tool_name == "list_all_invoices":
        result = list_all_invoices_from_db()
        return json.dumps(result)
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def get_invoice_with_claude(invoice_id: str) -> dict:
    """Use Claude with tool use to retrieve invoice details"""
    messages = [
        {
            "role": "user",
            "content": f"Please retrieve the invoice details for invoice ID: {invoice_id}. If the invoice is not found, let me know."
        }
    ]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process all tool calls in the response
            tool_results = []
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_result = process_tool_call(content_block.name, content_block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": content_block.id,
                        "content": tool_result
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            # Extract the final text response
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response = content_block.text
                    break
            
            # Try to extract structured data from the response
            # Check if the invoice was found in our database
            if invoice_id in INVOICES_DB:
                return {
                    "success": True,
                    "invoice_id": invoice_id,
                    "invoice": INVOICES_DB[invoice_id],
                    "claude_summary": final_response
                }
            else:
                return {
                    "success": False,
                    "invoice_id": invoice_id,
                    "error": f"Invoice {invoice_id} not found",
                    "claude_response": final_response
                }
        else:
            # Unexpected stop reason
            return {
                "success": False,
                "error": f"Unexpected stop reason: {response.stop_reason}"
            }

@app.route('/invoice/<invoice_id>', methods=['GET'])
def get_invoice(invoice_id: str):
    """Flask route to get invoice details by ID"""
    result = get_invoice_with_claude(invoice_id)
    
    if result.get("success"):
        return jsonify(result), 200
    else:
        return jsonify(result), 404

    """Flask route to list all available invoices"""
    messages = [
        {

        }

    # Simple single call for listing
    response = client.messages.create(

    if response.stop_reason == "tool_use":
        tool_results = []
        for content_block in response.content:

                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": content_block.id,
                    "content": tool_result
                })
        
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": tool_results})
        
        final_response = client.messages.create(

        return jsonify({
            "success": True,

    return jsonify({
