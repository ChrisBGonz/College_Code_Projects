// Middleware to parse JSON
app.use(express.json());

// Helper function to validate user_id
const validateUserId = (userId) => {
  if (!userId) {
    return { valid: false, error: 'user_id query parameter is required' };
  }

  const parsedId = parseInt(userId, 10);

  if (isNaN(parsedId) || parsedId <= 0) {
    return { valid: false, error: 'user_id must be a positive integer' };
  }

  return { valid: true, id: parsedId };
};

// GET /api/profile/download - Download user profile data as JSON file
app.get('/api/profile/download', (req, res) => {
  const { user_id } = req.query;

  // Validate user_id
  const validation = validateUserId(user_id);
  if (!validation.valid) {
    return res.status(400).json({
      success: false,
      error: validation.error
    });
  }

  const userId = validation.id;

  // Fetch user from database
  const user = usersDatabase[userId];

  if (!user) {
    return res.status(404).json({
      success: false,
      error: `User with id ${userId} not found`
    });
  }

  // Prepare the profile data for download
  const profileData = {
    exportInfo: {
      exportedAt: new Date().toISOString(),
      exportedBy: 'Profile Data Export Service',
      version: '1.0'
    },
    profile: user
  };

  // Convert to JSON string with pretty formatting
  const jsonData = JSON.stringify(profileData, null, 2);

  // Set headers for file download
  const filename = `profile_${user.username}_${Date.now()}.json`;
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Content-Length', Buffer.byteLength(jsonData));

  // Send the JSON data as a downloadable file
  res.status(200).send(jsonData);
});

// GET /api/profile - Get user profile data as JSON response (without download)
app.get('/api/profile', (req, res) => {
  const { user_id } = req.query;

  // Validate user_id
  const validation = validateUserId(user_id);
  if (!validation.valid) {
    return res.status(400).json({
      success: false,
      error: validation.error
    });

  const userId = validation.id;

  // Fetch user from database
  const user = usersDatabase[userId];

  if (!user) {
    return res.status(404).json({
      success: false,
      error: `User with id ${userId} not found`
    });
  }

  res.status(200).json({
    success: true,
    data: user
  });
});

// GET /api/users - List all available users (for testing purposes)
app.get('/api/users', (req, res) => {
  const userList = Object.values(usersDatabase).map(user => ({
    id: user.id,
    username: user.username,
    email: user.email,
    firstName: user.firstName,
    lastName: user.lastName
  }));

  res.status(200).json({
    success: true,
    count: userList.length,
    data: userList
  });
});

// Handle 404 for undefined routes
app.use((req, res) => {
  res.status(404).json({

});

// Global error handler
app.use((err, req, res, next) => {

  res.status(500).json({

// Start the server
