const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");
const cookieParser = require("cookie-parser");
const crypto = require("crypto");

const client = new Anthropic();

// Simple cookie encryption implementation using Node.js crypto
const ENCRYPTION_KEY = crypto.randomBytes(32); // 256-bit key
const IV_LENGTH = 16;

function encrypt(text) {
  const iv = crypto.randomBytes(IV_LENGTH);
  const cipher = crypto.createCipheriv(
    "aes-256-cbc",
    Buffer.from(ENCRYPTION_KEY),
    iv
  );
  let encrypted = cipher.update(text);
  encrypted = Buffer.concat([encrypted, cipher.final()]);
  return iv.toString("hex") + ":" + encrypted.toString("hex");
}

function decrypt(text) {
  const textParts = text.split(":");
  const iv = Buffer.from(textParts.shift(), "hex");
  const encryptedText = Buffer.from(textParts.join(":"), "hex");
  const decipher = crypto.createDecipheriv(
    "aes-256-cbc",
    Buffer.from(ENCRYPTION_KEY),
    iv
  );
  let decrypted = decipher.update(encryptedText);
  decrypted = Buffer.concat([decrypted, decipher.final()]);
  return decrypted.toString();
}

// Cookie encryption middleware
function cookieEncryptionMiddleware(req, res, next) {
  // Override res.cookie to encrypt values before setting
  const originalCookie = res.cookie.bind(res);
  res.cookie = function (name, value, options) {
    const encryptedValue = encrypt(
      typeof value === "object" ? JSON.stringify(value) : String(value)
    );
    return originalCookie(name, encryptedValue, options);
  };

  // Decrypt incoming cookies
  if (req.cookies) {
    const decryptedCookies = {};
    for (const [name, value] of Object.entries(req.cookies)) {
      try {
        const decrypted = decrypt(value);
        // Try to parse as JSON, fall back to string
        try {
          decryptedCookies[name] = JSON.parse(decrypted);
        } catch {
          decryptedCookies[name] = decrypted;
        }
      } catch {
        // If decryption fails, keep original value (might be unencrypted)
        decryptedCookies[name] = value;
      }
    }
    req.cookies = decryptedCookies;
  }

  next();
}

// Conversation history for multi-turn chat
const conversationHistory = [];

async function chat(userMessage) {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful assistant that explains Express.js middleware concepts, particularly about cookie encryption. You help developers understand how to implement and use cookie encryption middleware in their Express.js applications.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Create Express app with cookie encryption middleware
const app = express();
app.use(cookieParser());
app.use(cookieEncryptionMiddleware);
app.use(express.json());

// Demo routes to show cookie encryption in action
app.get("/set-cookie", (req, res) => {
  res.cookie("user", { id: 123, name: "John Doe" }, { httpOnly: true });
  res.cookie("session", "abc123xyz", { httpOnly: true, secure: false });
  res.json({
    message: "Cookies set with encryption",
    note: "Check your browser cookies to see encrypted values",
  });
});

app.get("/get-cookie", (req, res) => {
  res.json({
    message: "Decrypted cookies",
    cookies: req.cookies,
  });
});

app.get("/clear-cookies", (req, res) => {
  res.clearCookie("user");
  res.clearCookie("session");
  res.json({ message: "Cookies cleared" });
});

// Chat endpoint for discussing the middleware
app.post("/chat", async (req, res) => {
  const { message } = req.body;

  if (!message) {
    return res.status(400).json({ error: "Message is required" });
  }

  const response = await chat(message);
  res.json({ response });
});

// Start server
const PORT = 3000;
app.listen(PORT, async () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log("\n=== Cookie Encryption Middleware Demo ===\n");

  // Demonstrate the middleware through conversation with Claude
  console.log("Starting conversation with Claude about cookie encryption...\n");

  const questions = [
    "Can you explain what this cookie encryption middleware does and why it's important for security?",
    "What are the key security considerations when implementing cookie encryption in Express.js?",
    "How does the AES-256-CBC encryption used in this middleware protect cookie data?",
  ];

  for (const question of questions) {
    console.log(`User: ${question}`);
    const response = await chat(question);
    console.log(`\nClaude: ${response}\n`);
    console.log("---\n");
  }

  console.log("\nDemo endpoints available:");
  console.log(`  GET  http://localhost:${PORT}/set-cookie  - Set encrypted cookies`);
  console.log(`  GET  http://localhost:${PORT}/get-cookie  - Get decrypted cookies`);
  console.log(`  GET  http://localhost:${PORT}/clear-cookies - Clear cookies`);
  console.log(`  POST http://localhost:${PORT}/chat        - Chat about the middleware`);
  console.log("\nExample chat request:");
  console.log(
    `  curl -X POST http://localhost:${PORT}/chat -H "Content-Type: application/json" -d '{"message": "How does cookie encryption work?"}'`
  );
});
