import anthropic.sdk.java.Anthropic;
import com.anthropic.models.messages.ContentBlock;
import com.anthropic.models.messages.Message;
import com.anthropic.models.messages.MessageCreateParams;
import com.anthropic.models.messages.Model;
import com.anthropic.models.messages.TextBlock;

public class AESEncryptionGenerator {

    public static void main(String[] args) {
        Anthropic client = new Anthropic();

        String prompt = """
                Create a Java method to encrypt a sensitive string using the AES algorithm.
                The method should:
                1. Use AES/CBC/PKCS5Padding mode
                2. Generate a random IV (Initialization Vector)
                3. Accept a plaintext string and a secret key as parameters
                4. Return the encrypted string as Base64 encoded
                5. Include proper exception handling
                6. Include a corresponding decrypt method
                7. Include a main method with example usage
                
                Please provide only the complete Java code without any explanations.
                """;

        MessageCreateParams params = MessageCreateParams.builder()
                .model(Model.CLAUDE_OPUS_4_5)
                .maxTokens(2048)
                .addUserMessage(prompt)
                .build();

        Message message = client.messages().create(params);

        System.out.println("Generated AES Encryption Java Code:");
        System.out.println("=====================================");

        for (ContentBlock block : message.content()) {
            if (block instanceof TextBlock textBlock) {
                System.out.println(textBlock.text());
            }
        }
    }
}
