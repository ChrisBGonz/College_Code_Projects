package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"time"
)

func callMicroserviceWithRetry(url string) (string, error) {
	attempt := 1
	for {
		log.Printf("Attempt %d: Calling microservice at %s", attempt, url)

		resp, err := http.Get(url)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying immediately...", attempt, err)
			attempt++
			continue
		}

		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			log.Printf("Attempt %d failed with status code: %d. Retrying immediately...", attempt, resp.StatusCode)
			attempt++
			continue
		}

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			log.Printf("Attempt %d failed to read response body: %v. Retrying immediately...", attempt, err)
			attempt++
			continue
		}

		log.Printf("Attempt %d succeeded after %d attempts", attempt, attempt)
		return string(body), nil
	}
}

func callMicroserviceWithRetryAndTimeout(url string, timeout time.Duration) (string, error) {
	client := &http.Client{
		Timeout: timeout,
	}

	attempt := 1
	for {
		log.Printf("Attempt %d: Calling microservice at %s", attempt, url)

		resp, err := client.Get(url)
		if err != nil {
			log.Printf("Attempt %d failed with error: %v. Retrying immediately...", attempt, err)
			attempt++
			continue
		}

		defer resp.Body.Close()

		if resp.StatusCode != http.StatusOK {
			log.Printf("Attempt %d failed with status code: %d. Retrying immediately...", attempt, resp.StatusCode)
			attempt++
			continue
		}

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			log.Printf("Attempt %d failed to read response body: %v. Retrying immediately...", attempt, err)
			attempt++
			continue
		}

		log.Printf("Success on attempt %d", attempt)
		return string(body), nil
	}
}

func main() {
	url := "http://localhost:8080/api/service"

	fmt.Println("Starting microservice call with retry...")
	result, err := callMicroserviceWithRetryAndTimeout(url, 5*time.Second)
	if err != nil {
		log.Fatalf("Failed to call microservice: %v", err)
	}

	fmt.Printf("Microservice response: %s\n", result)
}
