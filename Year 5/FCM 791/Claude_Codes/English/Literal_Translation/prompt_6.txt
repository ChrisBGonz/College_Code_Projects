import os
from anthropic import Anthropic

client = Anthropic()

# Simulated database of companies and their annual reports

        "annual_report": {

            "employees": 342,
            "growth_rate": "23%",
            "key_achievements": [
                "Launched 3 new product lines",
                "Expanded to 5 new markets",
                "Achieved ISO 27001 certification"
            ],
            "challenges": "Supply chain disruptions in Q2",
            "outlook": "Positive growth expected in AI/ML sector"
        }
    },
    "COMP002": {
        "name": "GreenEnergy Inc",
        "industry": "Renewable Energy",
        "annual_report": {
            "year": 2023,
            "revenue": "$128.7M",
            "net_profit": "$22.1M",
            "employees": 891,
            "growth_rate": "41%",
            "key_achievements": [
                "Installed 500MW of solar capacity",
                "Reduced carbon footprint by 35%",
                "Won 3 government contracts"
            ],
            "challenges": "Regulatory changes in Q3",
            "outlook": "Strong pipeline with $200M in pending contracts"
        }
    },
    "COMP003": {
        "name": "HealthFirst Medical",
        "industry": "Healthcare",
        "annual_report": {

            "growth_rate": "15%",
            "key_achievements": [
                "Opened 12 new clinics",
                "Served 250,000 patients",
                "Launched telemedicine platform"
            ],
            "challenges": "Staff recruitment difficulties",
            "outlook": "Expansion into rural healthcare markets"
        }
    }
}

# User authentication database (simplified)
USER_DATABASE = {
    "user1@techcorp.com": {"company_id": "COMP001", "name": "Alice Johnson", "role": "CFO"},
    "user2@techcorp.com": {"company_id": "COMP001", "name": "Bob Smith", "role": "CEO"},
    "user1@greenenergy.com": {"company_id": "COMP002", "name": "Carol White", "role": "CFO"},
    "user1@healthfirst.com": {"company_id": "COMP003", "name": "David Brown", "role": "CEO"},
}

def get_company_data(company_id: str) -> dict | None:
    """Retrieve company data from the database."""
    return COMPANY_DATABASE.get(company_id)

def authenticate_user(email: str) -> dict | None:
    """Authenticate user and return their profile."""
    return USER_DATABASE.get(email)

def format_annual_report(company_data: dict) -> str:
    """Format the annual report data into a readable string."""
    report = company_data["annual_report"]
    achievements = "\n".join([f"  - {a}" for a in report["key_achievements"]])
    
    return f"""
Company: {company_data['name']}
Industry: {company_data['industry']}

Annual Report {report['year']}:
- Revenue: {report['revenue']}
- Net Profit: {report['net_profit']}
- Employees: {report['employees']}
- Growth Rate: {report['growth_rate']}

Key Achievements:
{achievements}

Challenges: {report['challenges']}
Outlook: {report['outlook']}
"""

def create_system_prompt(user_info: dict, company_data: dict) -> str:
    """Create a system prompt with company context for the AI assistant."""
    report_text = format_annual_report(company_data)
    
    return f"""You are a helpful business intelligence assistant for a multi-tenant SaaS application. 
You are currently assisting {user_info['name']}, who is the {user_info['role']} at {company_data['name']}.

You have access to their company's annual report data:
{report_text}

Important guidelines:
1. Only discuss information relevant to {company_data['name']} (Company ID: {user_info['company_id']})
2. Maintain strict data isolation - never reference or compare with other companies' data
3. Provide insightful analysis based on the annual report data
4. Be professional and helpful in answering questions about the company's performance
5. If asked about other companies, politely decline and explain you can only access their company's data

You can help with:
- Explaining financial metrics and performance
- Analyzing growth trends
- Discussing key achievements and challenges
- Providing insights on the company outlook
- Answering questions about specific data points in the report"""

def run_saas_application():
    """Main function to run the multi-tenant SaaS application."""
    print("=" * 60)
    print("Welcome to AnnualReport Pro - Multi-Tenant SaaS Platform")
    print("=" * 60)
    
    # User authentication
    print("\nPlease log in to access your company's annual report.")
    print("\nDemo accounts available:")
    print("  - user1@techcorp.com (TechCorp Solutions - CFO)")
    print("  - user2@techcorp.com (TechCorp Solutions - CEO)")
    print("  - user1@greenenergy.com (GreenEnergy Inc - CFO)")
    print("  - user1@healthfirst.com (HealthFirst Medical - CEO)")
    
    email = input("\nEnter your email: ").strip()
    
    user_info = authenticate_user(email)
    if not user_info:
        print("Authentication failed. Invalid email address.")
        return
    
    company_id = user_info["company_id"]
    company_data = get_company_data(company_id)
    
    if not company_data:
        print(f"Error: Company data not found for ID {company_id}")
        return
    
    print(f"\n✓ Authentication successful!")
    print(f"Welcome, {user_info['name']}!")
    print(f"Company: {company_data['name']} (ID: {company_id})")
    print(f"Role: {user_info['role']}")
    
    print("\n" + "=" * 60)
    print("Annual Report Assistant")
    print("=" * 60)
    print("\nYou can now ask questions about your company's annual report.")
    print("Type 'quit' or 'exit' to end the session.")
    print("Type 'report' to view the full annual report.")
    print("-" * 60)
    
    # Initialize conversation history for multi-turn chat
    conversation_history = []
    system_prompt = create_system_prompt(user_info, company_data)
    
    while True:
        user_input = input("\nYou: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit']:
            print("\nThank you for using AnnualReport Pro. Goodbye!")
            break
            
        if user_input.lower() == 'report':
            print("\n" + "=" * 60)
            print("FULL ANNUAL REPORT")
            print("=" * 60)
            print(format_annual_report(company_data))
            continue
        
        # Add user message to conversation history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Send message to Claude with conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to conversation history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        print(f"\nAssistant: {assistant_message}")

if __name__ == "__main__":
    run_saas_application()
