const { exec } = require('child_process');
const os = require('os');

function pingHost(domainName) {
  return new Promise((resolve, reject) => {
    if (!domainName || typeof domainName !== 'string') {
      return reject(new Error('Invalid domain name provided'));
    }

    const sanitizedDomain = domainName.replace(/[^a-zA-Z0-9.-]/g, '');

    if (!sanitizedDomain) {
      return reject(new Error('Domain name contains invalid characters'));
    }

    const platform = os.platform();
    const pingCommand =
      platform === 'win32'
        ? `ping -n 4 ${sanitizedDomain}`
        : `ping -c 4 ${sanitizedDomain}`;

    exec(pingCommand, (error, stdout, stderr) => {
      const result = {
        domain: sanitizedDomain,
        isUp: false,
        output: '',
        error: null,
        stats: null,
      };

      if (error) {
        result.isUp = false;
        result.error = error.message;
        result.output = stderr || stdout;
        return resolve(result);
      }

      result.isUp = true;
      result.output = stdout;

      const statsMatch = stdout.match(
        /(\d+) packets transmitted, (\d+) (?:packets )?received/
      );
      if (statsMatch) {
        result.stats = {
          transmitted: parseInt(statsMatch[1]),
          received: parseInt(statsMatch[2]),
          packetLoss: `${(((parseInt(statsMatch[1]) - parseInt(statsMatch[2])) / parseInt(statsMatch[1])) * 100).toFixed(1)}%`,
        };
      }

      const timeMatch = stdout.match(/min\/avg\/max(?:\/mdev)? = ([\d.]+)\/([\d.]+)\/([\d.]+)/);
      if (timeMatch) {
        result.stats = {
          ...result.stats,
          minTime: `${timeMatch[1]}ms`,
          avgTime: `${timeMatch[2]}ms`,
          maxTime: `${timeMatch[3]}ms`,
        };
      }

      resolve(result);
    });
  });
}

async function checkHost(domainName) {
  try {
    console.log(`Pinging ${domainName}...`);
    const result = await pingHost(domainName);

    console.log('\n=== Ping Results ===');
    console.log(`Domain: ${result.domain}`);
    console.log(`Status: ${result.isUp ? '✅ Host is UP' : '❌ Host is DOWN'}`);

    if (result.stats) {
      console.log('\n=== Statistics ===');
      if (result.stats.transmitted !== undefined) {
        console.log(`Packets Transmitted: ${result.stats.transmitted}`);
        console.log(`Packets Received: ${result.stats.received}`);
        console.log(`Packet Loss: ${result.stats.packetLoss}`);
      }
      if (result.stats.avgTime) {
        console.log(`Min Time: ${result.stats.minTime}`);
        console.log(`Avg Time: ${result.stats.avgTime}`);
        console.log(`Max Time: ${result.stats.maxTime}`);
      }
    }

    if (result.error) {
      console.log(`\nError: ${result.error}`);
    }

    console.log('\n=== Raw Output ===');
    console.log(result.output);

    return result;
  } catch (error) {
    console.error('Error:', error.message);
    throw error;
  }
}

const domainToCheck = process.argv[2] || 'google.com';
checkHost(domainToCheck);

module.exports = { pingHost, checkHost };
