import hashlib
import os
import sys
import json
import argparse
from pathlib import Path
from datetime import datetime


def calculate_sha256(file_path: str, chunk_size: int = 8192) -> str:
    """
    Calculate SHA256 hash of a file.
    
    Args:
        file_path: Path to the file
        chunk_size: Size of chunks to read at a time (default 8KB)
    
    Returns:
        Hexadecimal SHA256 hash string
    """
    sha256_hash = hashlib.sha256()
    
    file_path = Path(file_path)
    
    if not file_path.exists():
        raise FileNotFoundError(f"File not found: {file_path}")
    
    if not file_path.is_file():
        raise ValueError(f"Path is not a file: {file_path}")
    
    file_size = file_path.stat().st_size
    bytes_processed = 0
    
    print(f"Calculating SHA256 for: {file_path.name}")
    print(f"File size: {format_file_size(file_size)}")
    
    with open(file_path, "rb") as f:
        while chunk := f.read(chunk_size):
            sha256_hash.update(chunk)
            bytes_processed += len(chunk)
            
            if file_size > 0:
                progress = (bytes_processed / file_size) * 100
                print(f"\rProgress: {progress:.1f}%", end="", flush=True)
    
    print()
    return sha256_hash.hexdigest()


def format_file_size(size_bytes: int) -> str:
    """Format file size in human-readable format."""
    if size_bytes < 1024:
        return f"{size_bytes} bytes"
    elif size_bytes < 1024 ** 2:
        return f"{size_bytes / 1024:.2f} KB"
    elif size_bytes < 1024 ** 3:
        return f"{size_bytes / (1024 ** 2):.2f} MB"
    else:
        return f"{size_bytes / (1024 ** 3):.2f} GB"


def verify_firmware(firmware_path: str, expected_hash: str) -> dict:
    """
    Verify firmware file integrity using SHA256.
    
    Args:
        firmware_path: Path to the firmware file
        expected_hash: Expected SHA256 hash (hexadecimal string)
    
    Returns:
        Dictionary containing verification results
    """
    result = {
        "firmware_file": str(firmware_path),
        "timestamp": datetime.now().isoformat(),
        "verification_passed": False,
        "calculated_hash": None,
        "expected_hash": expected_hash.lower().strip(),
        "error": None
    }
    
    try:
        calculated_hash = calculate_sha256(firmware_path)
        result["calculated_hash"] = calculated_hash
        
        if calculated_hash == expected_hash.lower().strip():
            result["verification_passed"] = True
        else:
            result["verification_passed"] = False
            
    except FileNotFoundError as e:
        result["error"] = f"File not found: {e}"
    except PermissionError as e:
        result["error"] = f"Permission denied: {e}"
    except Exception as e:
        result["error"] = f"Unexpected error: {e}"
    
    return result


def generate_hash_file(firmware_path: str, output_path: str = None) -> str:
    """
    Generate a hash file for a firmware file.
    
    Args:
        firmware_path: Path to the firmware file
        output_path: Path for the output hash file (optional)
    
    Returns:
        Path to the generated hash file
    """
    firmware_path = Path(firmware_path)
    
    if output_path is None:
        output_path = firmware_path.with_suffix(firmware_path.suffix + ".sha256")
    
    calculated_hash = calculate_sha256(firmware_path)
    
    hash_data = {
        "firmware_file": firmware_path.name,
        "sha256": calculated_hash,
        "file_size": firmware_path.stat().st_size,
        "generated_at": datetime.now().isoformat()
    }
    
    with open(output_path, "w") as f:
        json.dump(hash_data, f, indent=2)
    
    print(f"\nHash file generated: {output_path}")
    print(f"SHA256: {calculated_hash}")
    
    return str(output_path)


def verify_from_hash_file(firmware_path: str, hash_file_path: str) -> dict:
    """
    Verify firmware using a hash file.
    
    Args:
        firmware_path: Path to the firmware file
        hash_file_path: Path to the hash file
    
    Returns:
        Dictionary containing verification results
    """
    try:
        with open(hash_file_path, "r") as f:
            hash_data = json.load(f)
        
        expected_hash = hash_data.get("sha256")
        
        if not expected_hash:
            return {
                "verification_passed": False,
                "error": "No SHA256 hash found in hash file"
            }
        
        result = verify_firmware(firmware_path, expected_hash)
        result["hash_file"] = hash_file_path
        result["hash_file_firmware"] = hash_data.get("firmware_file")
        
        return result
        
    except FileNotFoundError:
        return {
            "verification_passed": False,
            "error": f"Hash file not found: {hash_file_path}"
        }
    except json.JSONDecodeError:
        return {
            "verification_passed": False,
            "error": f"Invalid JSON in hash file: {hash_file_path}"
        }


def print_verification_result(result: dict) -> None:
    """Print verification result in a formatted way."""
    print("\n" + "=" * 60)
    print("FIRMWARE VERIFICATION RESULT")
    print("=" * 60)
    
    print(f"Firmware File : {result.get('firmware_file', 'N/A')}")
    print(f"Timestamp     : {result.get('timestamp', 'N/A')}")
    
    if result.get("hash_file"):
        print(f"Hash File     : {result.get('hash_file', 'N/A')}")
    
    print(f"\nExpected Hash : {result.get('expected_hash', 'N/A')}")
    print(f"Calculated    : {result.get('calculated_hash', 'N/A')}")
    
    if result.get("error"):
        print(f"\nError         : {result['error']}")
        print("\n[FAILED] Verification could not be completed")
    elif result.get("verification_passed"):
        print("\n[PASSED] Firmware integrity verified successfully!")
        print("The firmware file has not been tampered with.")
    else:
        print("\n[FAILED] Firmware integrity check FAILED!")
        print("WARNING: The firmware file may be corrupted or tampered with!")
    
    print("=" * 60)


def create_test_firmware(file_path: str, size_kb: int = 100) -> None:
    """Create a test firmware file for demonstration."""
    print(f"Creating test firmware file: {file_path} ({size_kb} KB)")
    
    with open(file_path, "wb") as f:
        data = bytes(range(256)) * (size_kb * 4)
        f.write(data[:size_kb * 1024])
    
    print(f"Test firmware created: {file_path}")


def main():
    parser = argparse.ArgumentParser(
        description="Verify firmware file integrity using SHA256",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  # Verify firmware with a known hash
  python solution.py verify firmware.bin --hash abc123...
  
  # Generate a hash file for firmware
  python solution.py generate firmware.bin
  
  # Verify firmware using a hash file
  python solution.py verify-file firmware.bin firmware.bin.sha256
  
  # Run demo with test files
  python solution.py demo
        """
    )
    
    subparsers = parser.add_subparsers(dest="command", help="Command to execute")
    
    verify_parser = subparsers.add_parser("verify", help="Verify firmware with a hash")
    verify_parser.add_argument("firmware", help="Path to firmware file")
    verify_parser.add_argument("--hash", required=True, help="Expected SHA256 hash")
    
    generate_parser = subparsers.add_parser("generate", help="Generate hash file")
    generate_parser.add_argument("firmware", help="Path to firmware file")
    generate_parser.add_argument("--output", help="Output hash file path")
    
    verify_file_parser = subparsers.add_parser("verify-file", help="Verify using hash file")
    verify_file_parser.add_argument("firmware", help="Path to firmware file")
    verify_file_parser.add_argument("hashfile", help="Path to hash file")
    
    subparsers.add_parser("demo", help="Run demonstration with test files")
    
    args = parser.parse_args()
    
    if args.command == "verify":
        result = verify_firmware(args.firmware, args.hash)
        print_verification_result(result)
        sys.exit(0 if result["verification_passed"] else 1)
    
    elif args.command == "generate":
        try:
            output_path = generate_hash_file(args.firmware, args.output)
            print(f"Hash file saved to: {output_path}")
        except Exception as e:
            print(f"Error generating hash file: {e}")
            sys.exit(1)
    
    elif args.command == "verify-file":
        result = verify_from_hash_file(args.firmware, args.hashfile)
        print_verification_result(result)
        sys.exit(0 if result["verification_passed"] else 1)
    
    elif args.command == "demo":
        print("Running firmware verification demonstration...")
        print("-" * 60)
        
        test_firmware = "test_firmware.bin"
        test_hash_file = "test_firmware.bin.sha256"
        
        try:
            create_test_firmware(test_firmware, size_kb=50)
            
            print("\n--- Step 1: Generate hash file ---")
            generate_hash_file(test_firmware, test_hash_file)
            
            print("\n--- Step 2: Verify firmware (should PASS) ---")
            result = verify_from_hash_file(test_firmware, test_hash_file)
            print_verification_result(result)
            
            print("\n--- Step 3: Verify with wrong hash (should FAIL) ---")
            wrong_hash = "a" * 64
            result = verify_firmware(test_firmware, wrong_hash)
            print_verification_result(result)
            
            print("\n--- Step 4: Verify non-existent file (should ERROR) ---")
            result = verify_firmware("nonexistent_firmware.bin", "a" * 64)
            print_verification_result(result)
            
        finally:
            for f in [test_firmware, test_hash_file]:
                if os.path.exists(f):
                    os.remove(f)
                    print(f"Cleaned up: {f}")
    
    else:
        parser.print_help()
        print("\nRunning demo mode by default...")
        print("-" * 60)
        
        test_firmware = "demo_firmware.bin"
        test_hash_file = "demo_firmware.bin.sha256"
        
        try:
            create_test_firmware(test_firmware, size_kb=10)
            
            print("\nGenerating hash file...")
            generate_hash_file(test_firmware, test_hash_file)
            
            print("\nVerifying firmware integrity...")
            result = verify_from_hash_file(test_firmware, test_hash_file)
            print_verification_result(result)
            
        finally:
            for f in [test_firmware, test_hash_file]:
                if os.path.exists(f):
                    os.remove(f)


if __name__ == "__main__":
    main()
