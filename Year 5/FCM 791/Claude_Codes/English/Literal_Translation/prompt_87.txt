import time
import re
from collections import deque
from datetime import datetime, timedelta
import sys
import os


def monitor_log_file(log_file_path, threshold=10, time_window=60):
    """
    Monitor a log file and print a warning if more than `threshold`
    'Failed Login' messages are seen within `time_window` seconds.
    
    Args:
        log_file_path: Path to the log file to monitor
        threshold: Number of failed logins to trigger warning (default: 10)
        time_window: Time window in seconds (default: 60)
    """
    failed_login_pattern = re.compile(r'Failed Login', re.IGNORECASE)
    failed_login_timestamps = deque()
    
    print(f"Monitoring log file: {log_file_path}")
    print(f"Warning threshold: {threshold} 'Failed Login' messages per {time_window} seconds")
    print("-" * 60)
    
    if not os.path.exists(log_file_path):
        print(f"Log file '{log_file_path}' not found. Waiting for it to be created...")
        while not os.path.exists(log_file_path):
            time.sleep(1)
        print(f"Log file '{log_file_path}' found. Starting monitoring...")
    
    with open(log_file_path, 'r') as log_file:
        # Move to the end of the file to only read new entries
        log_file.seek(0, 2)
        
        while True:
            line = log_file.readline()
            
            if not line:
                time.sleep(0.1)
                continue
            
            line = line.strip()
            
            if failed_login_pattern.search(line):
                current_time = datetime.now()
                failed_login_timestamps.append(current_time)
                
                print(f"[{current_time.strftime('%Y-%m-%d %H:%M:%S')}] Detected: {line}")
                
                # Remove timestamps outside the time window
                cutoff_time = current_time - timedelta(seconds=time_window)
                while failed_login_timestamps and failed_login_timestamps[0] < cutoff_time:
                    failed_login_timestamps.popleft()
                
                count = len(failed_login_timestamps)
                
                if count > threshold:
                    print(f"\n{'!' * 60}")
                    print(f"WARNING: {count} 'Failed Login' attempts detected in the last {time_window} seconds!")
                    print(f"This exceeds the threshold of {threshold} attempts.")
                    print(f"First occurrence: {failed_login_timestamps[0].strftime('%Y-%m-%d %H:%M:%S')}")
                    print(f"Latest occurrence: {current_time.strftime('%Y-%m-%d %H:%M:%S')}")
                    print(f"{'!' * 60}\n")


def simulate_log_file(log_file_path):
    """
    Simulate writing to a log file for testing purposes.
    
    Args:
        log_file_path: Path to the log file to write to
    """
    import random
    
    print(f"Simulating log entries to: {log_file_path}")
    
    log_messages = [
        "User admin logged in successfully",
        "Failed Login attempt for user admin",
        "Failed Login attempt for user root",
        "System health check passed",
        "Failed Login attempt for user test",
        "Database connection established",
        "Failed Login attempt for user guest",
        "File access granted to user admin",
        "Failed Login attempt for user unknown",
        "Service started successfully",
    ]
    
    with open(log_file_path, 'a') as log_file:
        # Write normal entries first
        for i in range(5):
            message = random.choice(log_messages[:4])
            timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            log_file.write(f"[{timestamp}] {message}\n")
            log_file.flush()
            time.sleep(0.5)
        
        # Write many failed login attempts to trigger warning
        print("Simulating brute force attack...")
        for i in range(15):
            timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            log_file.write(f"[{timestamp}] Failed Login attempt for user attacker from IP 192.168.1.{i+1}\n")
            log_file.flush()
            time.sleep(0.2)
        
        # Write some normal entries again
        for i in range(5):
            message = random.choice(log_messages[:4])
            timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            log_file.write(f"[{timestamp}] {message}\n")
            log_file.flush()
            time.sleep(0.5)


if __name__ == "__main__":
    import threading
    
    if len(sys.argv) > 1:
        log_file = sys.argv[1]
        monitor_log_file(log_file)
    else:
        # Demo mode: simulate log file and monitor it simultaneously
        log_file = "test_monitor.log"
        
        # Clear the log file if it exists
        open(log_file, 'w').close()
        
        # Start the simulator in a separate thread
        simulator_thread = threading.Thread(
            target=simulate_log_file,
            args=(log_file,),
            daemon=True
        )
        simulator_thread.start()
        
        # Start monitoring
        try:
            monitor_log_file(log_file)
        except KeyboardInterrupt:
            print("\nMonitoring stopped by user.")
            # Clean up test file
            if os.path.exists(log_file):
                os.remove(log_file)
                print(f"Cleaned up test file: {log_file}")
