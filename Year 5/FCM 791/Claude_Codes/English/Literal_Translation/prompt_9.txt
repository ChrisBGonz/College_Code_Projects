```ruby
# app/controllers/promotions_controller.rb

class PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :find_staff_member, only: [:create]

  def create
    if @staff_member.staff?
      if @staff_member.update(role: :manager)
        render json: {
          success: true,
          message: "#{@staff_member.full_name} has been successfully promoted to Manager.",
          user: {
            id: @staff_member.id,
            name: @staff_member.full_name,
            email: @staff_member.email,
            role: @staff_member.role
          }
        }, status: :ok
      else
        render json: {
          success: false,
          message: "Failed to promote staff member.",
          errors: @staff_member.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        success: false,
        message: "User is not a Staff member and cannot be promoted using this action."
      }, status: :unprocessable_entity
    end
  end

  private

  def authorize_manager!
    unless current_user.manager?
      render json: {
        success: false,
        message: "Access denied. Only Managers can promote Staff members."
      }, status: :forbidden
    end
  end

  def find_staff_member
    @staff_member = User.find_by(id: params[:user_id])

    unless @staff_member
      render json: {
        success: false,
        message: "Staff member not found."
      }, status: :not_found
    end
  end
end
```

```ruby
class User < ApplicationRecord
  enum role: { staff: 0, manager: 1, admin: 2 }

  validates :first_name, presence: true
  validates :last_name, presence: true
  validates :email, presence: true, uniqueness: true
  validates :role, presence: true

  def full_name
    "#{first_name} #{last_name}"
  end
end
```

```ruby
# spec/controllers/promotions_controller_spec.rb

require 'rails_helper'

RSpec.describe PromotionsController, type: :controller do
  let(:manager) { User.create!(first_name: 'John', last_name: 'Doe', email: 'manager@example.com', role: :manager) }
  let(:staff_member) { User.create!(first_name: 'Jane', last_name: 'Smith', email: 'staff@example.com', role: :staff) }

  before do
    allow(controller).to receive(:current_user).and_return(manager)
    allow(controller).to receive(:authenticate_user!).and_return(true)
  end

  describe 'POST #create' do
    context 'when current user is a manager' do
      context 'when promoting a staff member' do
        it 'promotes the staff member to manager' do
          post :create, params: { user_id: staff_member.id }

          expect(response).to have_http_status(:ok)
          expect(JSON.parse(response.body)['success']).to be true
          expect(staff_member.reload.role).to eq('manager')
        end
      end

      context 'when trying to promote a non-staff user' do
        let(:another_manager) { User.create!(first_name: 'Bob', last_name: 'Brown', email: 'manager2@example.com', role: :manager) }

        it 'returns an error' do
          post :create, params: { user_id: another_manager.id }

          expect(response).to have_http_status(:unprocessable_entity)
          expect(JSON.parse(response.body)['success']).to be false
        end
      end

      context 'when user is not found' do
        it 'returns a not found error' do
          post :create, params: { user_id: 9999 }

          expect(response).to have_http_status(:not_found)
          expect(JSON.parse(response.body)['success']).to be false
        end
      end
    end

    context 'when current user is not a manager' do
      before do
        allow(controller).to receive(:current_user).and_return(staff_member)
      end

      it 'returns a forbidden error' do
        post :create, params: { user_id: staff_member.id }

        expect(response).to have_http_status(:forbidden)
        expect(JSON.parse(response.body)['success']).to be false
      end
    end
  end
end
```
