using System;
using System.Collections.Generic;
using System.Data;
using Microsoft.Data.SqlClient;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class DynamicSQLSorter
{
    private readonly string _connectionString;
    private readonly AnthropicClient _anthropicClient;

    public DynamicSQLSorter(string connectionString)
    {
        _connectionString = connectionString;
        _anthropicClient = new AnthropicClient();
    }

    public async Task<DataTable> ExecuteDynamicSortQuery(string tableName, string columnName, string sortOrder = "ASC")
    {
        // Use Claude to validate and generate safe SQL
        var validationMessages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I need to create a safe SQL query to sort a table. Please validate the following inputs and generate a safe SQL query:
                        - Table Name: {tableName}
                        - Column Name: {columnName}
                        - Sort Order: {sortOrder}
                        
                        Rules:
                        1. Only allow alphanumeric characters and underscores in table and column names
                        2. Sort order must be either ASC or DESC
                        3. If inputs are valid, return ONLY the SQL query in this exact format: SELECT * FROM [TableName] ORDER BY [ColumnName] ASC/DESC
                        4. If inputs are invalid, return ONLY: INVALID: <reason>
                        
                        Do not include any explanation, just the SQL query or INVALID message."
                    }
                }
            }
        };

        var validationRequest = new MessageParameters
        {
            Messages = validationMessages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 200,
            Stream = false
        };

        var validationResponse = await _anthropicClient.Messages.GetClaudeMessageAsync(validationRequest);
        var sqlQuery = validationResponse.Content[0].ToString()?.Trim();

        if (string.IsNullOrEmpty(sqlQuery) || sqlQuery.StartsWith("INVALID:"))
        {
            throw new ArgumentException($"Invalid input parameters: {sqlQuery}");
        }

        // Execute the validated query
        var dataTable = new DataTable();
        
        using (var connection = new SqlConnection(_connectionString))
        {
            await connection.OpenAsync();
            
            using (var command = new SqlCommand(sqlQuery, connection))
            {
                using (var adapter = new SqlDataAdapter(command))
                {
                    adapter.Fill(dataTable);
                }
            }
        }

        return dataTable;
    }

    public async Task<string> GetQueryExplanation(string tableName, string columnName, string sortOrder)
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $"Explain in one sentence what this SQL operation does: Sorting table '{tableName}' by column '{columnName}' in {sortOrder} order."
                    }
                }
            }
        };

        var request = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 100,
            Stream = false
        };

        var response = await _anthropicClient.Messages.GetClaudeMessageAsync(request);
        return response.Content[0].ToString() ?? string.Empty;
    }
}
