```ruby
class RecordsController < ApplicationController
  def search
    record = Record.find_by(query_params)
    
    if record
      render json: record, status: :ok
    else
      render json: { error: "Record not found" }, status: :not_found
    end
  end

  private

  def query_params
    params.require(:query).permit(:id, :name, :email, :status)
  end
end
```
