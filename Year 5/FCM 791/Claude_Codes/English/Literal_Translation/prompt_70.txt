const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Mock user database
const users = {
  "user-token-123": {
    id: 1,
    username: "john_doe",
    email: "john@example.com",
    fullName: "John Doe",
    role: "admin",
    createdAt: "2024-01-15",
    lastLogin: "2024-12-01",
  },
  "user-token-456": {
    id: 2,
    username: "jane_smith",
    email: "jane@example.com",
    fullName: "Jane Smith",
    role: "user",
    createdAt: "2024-03-20",
    lastLogin: "2024-11-28",
  },
  "user-token-789": {
    id: 3,
    username: "bob_wilson",
    email: "bob@example.com",
    fullName: "Bob Wilson",
    role: "moderator",
    createdAt: "2024-06-10",
    lastLogin: "2024-12-02",
  },
};

// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Validate token and get user data
function getUserFromToken(token) {
  return users[token] || null;
}

// Process request with Claude AI using streaming
async function processWithClaude(userData, requestType) {
  const messages = [
    {
      role: "user",
      content: `You are a helpful API assistant. A user has authenticated and is requesting their ${requestType}. 
      
      User data: ${JSON.stringify(userData, null, 2)}
      
      Please provide a friendly, professional response that:
      1. Confirms successful authentication
      2. Summarizes the user's profile information
      3. Mentions their account status and role
      4. Provides any relevant account insights
      
      Keep the response concise and professional.`,
    },
  ];

  let fullResponse = "";

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: messages,
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      fullResponse += chunk.delta.text;
      process.stdout.write(chunk.delta.text);
    }
  }

  console.log("\n");
  return fullResponse;
}

// Create HTTP server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader(
    "Access-Control-Allow-Headers",
    "Content-Type, Authorization"
  );

  // Handle preflight requests
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Parse URL
  const url = new URL(req.url, `http://${req.headers.host}`);
  const pathname = url.pathname;

  // Route: GET /api/user/profile
  if (req.method === "GET" && pathname === "/api/user/profile") {
    const authHeader = req.headers.authorization;
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const userData = getUserFromToken(token);

    if (!userData) {
      res.writeHead(403, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    try {
      console.log(`\nProcessing profile request for user: ${userData.username}`);
      console.log("Claude AI Response (streaming):\n");

      const aiResponse = await processWithClaude(userData, "profile");

      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          success: true,
          user: userData,
          aiInsights: aiResponse,
          timestamp: new Date().toISOString(),
        })
      );
    } catch (error) {
      console.error("Error processing with Claude:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
    return;
  }

  // Route: GET /api/user/data
  if (req.method === "GET" && pathname === "/api/user/data") {
    const authHeader = req.headers.authorization;
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const userData = getUserFromToken(token);

    if (!userData) {
      res.writeHead(403, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    // Return private user data
    const privateData = {
      ...userData,
      privateNotes: "This is private user data",
      preferences: {
        theme: "dark",
        notifications: true,
        language: "en",
      },
      activityLog: [
        { action: "login", timestamp: "2024-12-01T10:00:00Z" },
        { action: "profile_update", timestamp: "2024-11-28T15:30:00Z" },
        { action: "password_change", timestamp: "2024-11-15T09:00:00Z" },
      ],
    };

    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        success: true,
        data: privateData,
        timestamp: new Date().toISOString(),
      })
    );
    return;
  }

  // Route: POST /api/user/analyze
  if (req.method === "POST" && pathname === "/api/user/analyze") {
    const authHeader = req.headers.authorization;
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const userData = getUserFromToken(token);

    if (!userData) {
      res.writeHead(403, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    // Read request body
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });

    req.on("end", async () => {
      try {
        const requestData = body ? JSON.parse(body) : {};
        const analysisType = requestData.type || "general";

        console.log(
          `\nProcessing analysis request for user: ${userData.username}`
        );
        console.log(`Analysis type: ${analysisType}`);
        console.log("Claude AI Response (streaming):\n");

        const aiResponse = await processWithClaude(
          userData,
          `${analysisType} analysis`
        );

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            success: true,
            user: userData,
            analysis: aiResponse,
            analysisType: analysisType,
            timestamp: new Date().toISOString(),
          })
        );
      } catch (error) {
        console.error("Error processing request:", error);
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to process request",
          })
        );
      }
    });
    return;
  }

  // Route: GET /health
  if (req.method === "GET" && pathname === "/health") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        status: "healthy",
        timestamp: new Date().toISOString(),
      })
    );
    return;
  }

  // Default 404 response
  res.writeHead(404, { "Content-Type": "application/json" });
  res.end(
    JSON.stringify({
      error: "Not Found",
      message: "The requested endpoint does not exist",
      availableEndpoints: [
        "GET /api/user/profile",
        "GET /api/user/data",
        "POST /api/user/analyze",
        "GET /health",
      ],
    })
  );
});

// Start server
const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`\n🚀 API Server running on port ${PORT}`);
  console.log("\nAvailable endpoints:");
  console.log("  GET  /api/user/profile  - Get user profile with AI insights");
  console.log("  GET  /api/user/data     - Get private user data");
  console.log("  POST /api/user/analyze  - Analyze user data with AI");
  console.log("  GET  /health            - Health check");
  console.log("\nTest tokens:");
  console.log("  user-token-123 (admin)");
  console.log("  user-token-456 (user)");
  console.log("  user-token-789 (moderator)");
  console.log("\nExample usage:");
  console.log(
    "  curl -H 'Authorization: Bearer user-token-123' http://localhost:3000/api/user/profile"
  );
});

// Handle server errors
server.on("error", (error) => {
  console.error("Server error:", error);
  process.exit(1);
});

// Graceful shutdown
process.on("SIGTERM", () => {
  console.log("\nShutting down server...");
  server.close(() => {
    console.log("Server closed");
    process.exit(0);
  });
});
