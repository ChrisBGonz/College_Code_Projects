
```python
import subprocess
import sys
import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def run_command(command: str, cwd: str = None) -> tuple[str, str, int]:
    """Run a shell command and return stdout, stderr, and return code."""
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            cwd=cwd
        )
        return result.stdout, result.stderr, result.returncode
    except Exception as e:
        return "", str(e), 1

def chat_with_claude(user_message: str) -> str:
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="""You are an expert Java Maven dependency management assistant. 
        Your role is to help automate the update of Maven project dependencies to their latest versions.
        You analyze pom.xml files, suggest updates, and provide Maven commands to execute.
        When asked to perform actions, provide specific Maven commands that can be executed.
        Always be precise and practical in your suggestions.""",
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def read_pom_xml(project_path: str) -> str:
    """Read the pom.xml file from the project directory."""
    pom_path = os.path.join(project_path, "pom.xml")
    if not os.path.exists(pom_path):
        return None
    
    with open(pom_path, 'r') as f:
        return f.read()

def extract_maven_commands(response: str) -> list[str]:
    """Extract Maven commands from Claude's response."""
    commands = []
    lines = response.split('\n')
    
    for line in lines:
        line = line.strip()
        if line.startswith('mvn ') or line.startswith('$ mvn '):
            command = line.lstrip('$ ')
            commands.append(command)
    
    return commands

def update_dependencies_workflow(project_path: str):
    """Main workflow to update Maven dependencies using Claude."""
    print(f"Starting Maven dependency update workflow for: {project_path}")
    print("=" * 60)
    
    # Step 1: Read and analyze pom.xml
    pom_content = read_pom_xml(project_path)
    if not pom_content:
        print("Error: No pom.xml found in the specified directory")
        sys.exit(1)
    
    print("Step 1: Analyzing pom.xml with Claude...")
    analysis_message = f"""I have a Maven project with the following pom.xml content. 
    Please analyze it and tell me:
    1. What dependencies are currently defined
    2. What Maven commands I should run to check for available updates
    3. Any potential issues or concerns
    
    pom.xml content:
    {pom_content}"""
    
    analysis_response = chat_with_claude(analysis_message)
    print("\nClaude's Analysis:")
    print(analysis_response)
    print("\n" + "=" * 60)
    
    # Step 2: Check for available updates
    print("\nStep 2: Checking for available dependency updates...")
    check_message = """Based on the pom.xml analysis, please provide the exact Maven command to:
    1. Display all available dependency updates
    2. The command should output in a format we can parse
    
    Please provide just the Maven command(s) to run."""
    
    check_response = chat_with_claude(check_message)
    print("\nClaude's recommendation:")
    print(check_response)
    
    # Extract and run the check command
    check_commands = extract_maven_commands(check_response)
    
    if not check_commands:
        # Default command if Claude doesn't provide one
        check_commands = ["mvn versions:display-dependency-updates"]
    
    print(f"\nRunning: {check_commands[0]}")
    stdout, stderr, returncode = run_command(check_commands[0], cwd=project_path)
    
    if returncode != 0:
        print(f"Command failed with error: {stderr}")
        # Ask Claude for help with the error
        error_message = f"""The command failed with this error:
        {stderr}
        
        How should I fix this? Please provide alternative commands."""
        error_response = chat_with_claude(error_message)
        print("\nClaude's suggestion for fixing the error:")
        print(error_response)
    else:
        print("\nAvailable updates found:")
        print(stdout[:2000] if len(stdout) > 2000 else stdout)  # Limit output
    
    print("\n" + "=" * 60)
    
    # Step 3: Ask Claude to interpret results and suggest updates
    print("\nStep 3: Getting Claude's recommendations for updates...")
    
    update_analysis_message = f"""Based on the dependency check output:
    {stdout[:3000] if stdout else 'No output available'}
    
    Please:
    1. Summarize which dependencies have updates available
    2. Identify any major version updates that might have breaking changes
    3. Provide the Maven command to automatically update all dependencies to their latest versions
    4. Warn about any dependencies that should be updated carefully"""
    
    update_analysis_response = chat_with_claude(update_analysis_message)
    print("\nClaude's Update Recommendations:")
    print(update_analysis_response)
    print("\n" + "=" * 60)
    
    # Step 4: Ask user if they want to proceed with updates
    print("\nStep 4: Proceeding with dependency updates...")
    
    proceed_message = """Please provide the exact Maven command(s) to:
    1. Update all dependencies to their latest versions in the pom.xml
    2. The command should modify the pom.xml file directly
    
    Provide only the Maven command(s), one per line."""
    
    proceed_response = chat_with_claude(proceed_message)
    print("\nClaude's update commands:")
    print(proceed_response)
    
    update_commands = extract_maven_commands(proceed_response)
    
    if not update_commands:
        # Default update command
        update_commands = ["mvn versions:use-latest-versions"]
    
    # Execute update commands
    for cmd in update_commands:
        print(f"\nExecuting: {cmd}")
        stdout, stderr, returncode = run_command(cmd, cwd=project_path)
        
        if returncode != 0:
            print(f"Update command failed: {stderr}")
            # Ask Claude for help
            fix_message = f"""The update command '{cmd}' failed with:
            {stderr}
            
            Please suggest an alternative approach."""
            fix_response = chat_with_claude(fix_message)
            print("\nClaude's suggestion:")
            print(fix_response)
        else:
            print("Update successful!")
            if stdout:
                print(stdout[:1000])
    
    print("\n" + "=" * 60)
    
    # Step 5: Verify the updates
    print("\nStep 5: Verifying updates...")
    
    verify_message = """Now that we've updated the dependencies, please provide the Maven command to:
    1. Verify the project still compiles
    2. Run a quick test to ensure nothing is broken
    
    Provide the commands in order of execution."""
    
    verify_response = chat_with_claude(verify_message)
    print("\nClaude's verification steps:")
    print(verify_response)
    
    verify_commands = extract_maven_commands(verify_response)
    
    if not verify_commands:
        verify_commands = ["mvn compile", "mvn test -q"]
    
    for cmd in verify_commands[:2]:  # Run at most 2 verification commands
        print(f"\nRunning verification: {cmd}")
        stdout, stderr, returncode = run_command(cmd, cwd=project_path)
        
        if returncode != 0:
            print(f"Verification failed: {stderr[:500]}")
            # Ask Claude for help with failures
            failure_message = f"""The verification command '{cmd}' failed with:
            {stderr[:1000]}
            
            What should I do to fix these issues? Are there specific dependencies that might have caused breaking changes?"""
            failure_response = chat_with_claude(failure_message)
            print("\nClaude's advice for fixing failures:")
            print(failure_response)
            break
        else:
            print(f"Verification passed: {cmd}")
    
    print("\n" + "=" * 60)
    
    # Step 6: Final summary
    print("\nStep 6: Getting final summary...")
    
    summary_message = """Please provide a final summary of:
    1. What was accomplished in this dependency update session
    2. Any important notes about the updates made
    3. Recommended next steps for the developer
    4. Any dependencies that might need manual review"""
    
    summary_response = chat_with_claude(summary_message)
    print("\nFinal Summary from Claude:")
    print(summary_response)
    
    print("\n" + "=" * 60)
    print("Maven dependency update workflow completed!")

def main():
    """Main entry point for the script."""
    if len(sys.argv) < 2:
        # Use current directory if no path provided
        project_path = os.getcwd()
        print(f"No project path provided. Using current directory: {project_path}")
    else:
        project_path = sys.argv[1]
    
    # Verify the path exists
    if not os.path.exists(project_path):
        print(f"Error: Path '{project_path}' does not exist")
        sys.exit(1)
    
    # Check if Maven is installed
    stdout, stderr, returncode = run_command("mvn --version")
    if returncode != 0:
        print("Error: Maven is not installed or not in PATH")
        print("Please install Maven and try again")
        sys.exit(1)
    
    print(f"Maven version: {stdout.split(chr(10))[0]}")
    
    # Start the update workflow
    update_dependencies_workflow(project_path)

if __name__ == "__main__":
    main()
```
