
import logging

logging.basicConfig(
    filename='app.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)

def log_search_query(search_query: str) -> None:
    """
    Logs a user's search query to app.log file.
    
    Args:
        search_query (str): The search query entered by the user.
    """
    if not isinstance(search_query, str):
        raise TypeError("Search query must be a string.")
    
    if not search_query.strip():
        raise ValueError("Search query cannot be empty or whitespace.")
    
    logging.info(f"Search Query: {search_query}")
    print(f"Search query '{search_query}' has been logged to app.log")


if __name__ == "__main__":
    log_search_query("Python programming tutorials")
    log_search_query("How to use logging in Python")
    log_search_query("Best practices for Python development")
