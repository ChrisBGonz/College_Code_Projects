const crypto = require('crypto');

/**
 * Generates a random 16-character session token
 * @returns {string} A random 16-character hexadecimal session token
 */
function generateSessionToken() {
  return crypto.randomBytes(8).toString('hex');
}

/**
 * Generates a random 16-character session token using alphanumeric characters
 * @returns {string} A random 16-character alphanumeric session token
 */
function generateAlphanumericSessionToken() {
  const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  const charactersLength = characters.length;
  let token = '';

  const randomBytes = crypto.randomBytes(16);

  for (let i = 0; i < 16; i++) {
    token += characters[randomBytes[i] % charactersLength];
  }

  return token;
}

/**
 * Generates a random 16-character session token using base64url encoding
 * @returns {string} A random 16-character base64url session token
 */
function generateBase64SessionToken() {
  return crypto.randomBytes(12).toString('base64url').slice(0, 16);
}

/**
 * Generates a session token with a specified length (default: 16 characters)
 * @param {number} length - The desired length of the token (default: 16)
 * @param {string} encoding - The encoding type: 'hex', 'alphanumeric', or 'base64url' (default: 'hex')
 * @returns {string} A random session token of the specified length
 */
function generateCustomSessionToken(length = 16, encoding = 'hex') {
  if (length <= 0) {
    throw new Error('Token length must be a positive number');
  }

  switch (encoding) {
    case 'hex': {
      const bytesNeeded = Math.ceil(length / 2);
      return crypto.randomBytes(bytesNeeded).toString('hex').slice(0, length);
    }

    case 'alphanumeric': {
      const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
      const charactersLength = characters.length;
      let token = '';
      const randomBytes = crypto.randomBytes(length);

      for (let i = 0; i < length; i++) {
        token += characters[randomBytes[i] % charactersLength];
      }

      return token;
    }

    case 'base64url': {
      const bytesNeeded = Math.ceil((length * 3) / 4);
      return crypto.randomBytes(bytesNeeded).toString('base64url').slice(0, length);
    }

    default:
      throw new Error(`Unsupported encoding: ${encoding}. Use 'hex', 'alphanumeric', or 'base64url'`);
  }
}

// Example usage and testing
function runExamples() {
  console.log('=== Session Token Generator Examples ===\n');

  console.log('1. Hexadecimal Token (16 chars):');
  for (let i = 0; i < 3; i++) {
    const token = generateSessionToken();
    console.log(`   Token: ${token} | Length: ${token.length}`);
  }

  console.log('\n2. Alphanumeric Token (16 chars):');
  for (let i = 0; i < 3; i++) {
    const token = generateAlphanumericSessionToken();
    console.log(`   Token: ${token} | Length: ${token.length}`);
  }

  console.log('\n3. Base64URL Token (16 chars):');
  for (let i = 0; i < 3; i++) {
    const token = generateBase64SessionToken();
    console.log(`   Token: ${token} | Length: ${token.length}`);
  }

  console.log('\n4. Custom Token Examples:');
  const customHex = generateCustomSessionToken(16, 'hex');
  console.log(`   Hex (16):          ${customHex} | Length: ${customHex.length}`);

  const customAlpha = generateCustomSessionToken(16, 'alphanumeric');
  console.log(`   Alphanumeric (16): ${customAlpha} | Length: ${customAlpha.length}`);

  const customBase64 = generateCustomSessionToken(16, 'base64url');
  console.log(`   Base64URL (16):    ${customBase64} | Length: ${customBase64.length}`);

  console.log('\n5. Uniqueness Test (generating 5 tokens):');
  const tokens = new Set();
  for (let i = 0; i < 5; i++) {
    const token = generateSessionToken();
    tokens.add(token);
    console.log(`   Token ${i + 1}: ${token}`);
  }
  console.log(`   All unique: ${tokens.size === 5}`);
}

runExamples();

module.exports = {
  generateSessionToken,
  generateAlphanumericSessionToken,
  generateBase64SessionToken,
  generateCustomSessionToken,
};
