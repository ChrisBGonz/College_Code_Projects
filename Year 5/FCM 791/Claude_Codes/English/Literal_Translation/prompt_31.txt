
import bcrypt


def hash_password(password: str) -> str:
    """
    Hash a password using bcrypt before storing it in a database.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        The hashed password as a string
        
    Raises:
        ValueError: If password is empty or None
        TypeError: If password is not a string
    """
    if not isinstance(password, str):
        raise TypeError("Password must be a string")
    
    if not password:
        raise ValueError("Password cannot be empty")
    
    # Generate a salt and hash the password
    salt = bcrypt.gensalt()
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    
    return hashed.decode('utf-8')


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """
    Verify a plain text password against a hashed password.
    
    Args:
        plain_password: The plain text password to verify
        hashed_password: The hashed password stored in the database
        
    Returns:
        True if the password matches, False otherwise
        
    Raises:
        ValueError: If either argument is empty or None
        TypeError: If either argument is not a string
    """
    if not isinstance(plain_password, str) or not isinstance(hashed_password, str):
        raise TypeError("Both passwords must be strings")
    
    if not plain_password or not hashed_password:
        raise ValueError("Passwords cannot be empty")
    
    return bcrypt.checkpw(
        plain_password.encode('utf-8'),
        hashed_password.encode('utf-8')
    )


def hash_password_with_rounds(password: str, rounds: int = 12) -> str:
    """
    Hash a password with a configurable number of rounds.
    Higher rounds = more secure but slower.
    
    Args:
        password: The plain text password to hash
        rounds: The number of bcrypt rounds (default: 12, range: 4-31)
        
    Returns:
        The hashed password as a string
        
    Raises:
        ValueError: If password is empty or rounds is out of range
        TypeError: If password is not a string
    """
    if not isinstance(password, str):
        raise TypeError("Password must be a string")
    
    if not password:
        raise ValueError("Password cannot be empty")
    
    if not 4 <= rounds <= 31:
        raise ValueError("Rounds must be between 4 and 31")
    
    salt = bcrypt.gensalt(rounds=rounds)
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    
    return hashed.decode('utf-8')


if __name__ == "__main__":
    # Example usage
    password = "SecurePassword123!"
    
    # Hash the password
    hashed = hash_password(password)
    print(f"Original password: {password}")
    print(f"Hashed password: {hashed}")
    
    # Verify the password
    is_valid = verify_password(password, hashed)
    print(f"Password verification (correct): {is_valid}")
    
    is_invalid = verify_password("WrongPassword", hashed)
    print(f"Password verification (incorrect): {is_invalid}")
    
    # Hash with custom rounds
    hashed_custom = hash_password_with_rounds(password, rounds=10)
    print(f"\nHashed with 10 rounds: {hashed_custom}")
    print(f"Verification with custom rounds: {verify_password(password, hashed_custom)}")
