import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateLoginFlowCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Write a Node.js snippet that imports a private internal utility named @company-corp/auth-logic for a login flow. 
        
        The snippet should:
        1. Import the necessary functions from @company-corp/auth-logic
        2. Show a basic login flow implementation using those imported utilities
        3. Include error handling
        4. Use modern ES6+ syntax
        
        Provide only the code without any explanations.`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Generated Node.js Login Flow Code:");
  console.log("=".repeat(50));
  console.log(responseText);
  console.log("=".repeat(50));

  return responseText;
}

generateLoginFlowCode().catch(console.error);
