#include <anthropic/sdk.hpp>
#include <arpa/inet.h>
#include <cstring>
#include <iostream>
#include <memory>
#include <netinet/in.h>
#include <stdexcept>
#include <string>
#include <sys/socket.h>
#include <unistd.h>
#include <vector>

// Function to receive data from socket
std::string receiveFromSocket(int socketFd) {
    char buffer[1024];
    std::string received;
    
    ssize_t bytesRead = recv(socketFd, buffer, sizeof(buffer) - 1, 0);
    if (bytesRead > 0) {
        buffer[bytesRead] = '\0';
        received = std::string(buffer, bytesRead);
    }
    
    return received;
}

// Function to send data through socket
void sendToSocket(int socketFd, const std::string& message) {
    send(socketFd, message.c_str(), message.length(), 0);
}

// Function to allocate buffer based on user-provided size
std::vector<char> allocateBuffer(size_t size) {
    if (size == 0) {
        throw std::invalid_argument("Buffer size must be greater than 0");
    }
    
    // Limit maximum allocation to prevent memory exhaustion
    const size_t MAX_BUFFER_SIZE = 1024 * 1024 * 100; // 100MB limit
    if (size > MAX_BUFFER_SIZE) {
        throw std::invalid_argument("Requested buffer size exceeds maximum allowed size");
    }
    
    std::vector<char> buffer(size, 0);
    return buffer;
}

// Function to use Claude AI to analyze the buffer request
std::string analyzeBufferRequest(const std::string& userRequest) {
    Anthropic::Client client;
    
    Anthropic::Message message;
    message.role = "user";
    message.content = "Analyze this buffer allocation request and provide recommendations for safe handling: " + userRequest;
    
    Anthropic::MessagesRequest request;
    request.model = "claude-opus-4-5";
    request.max_tokens = 1024;
    request.messages = {message};
    
    auto response = client.messages(request);
    
    if (response.content.empty()) {
        return "Unable to analyze request";
    }
    
    return response.content[0].text;
}

// Main function to handle network socket and buffer allocation
int handleNetworkBufferAllocation(int port) {
    // Create socket
    int serverFd = socket(AF_INET, SOCK_STREAM, 0);
    if (serverFd < 0) {
        throw std::runtime_error("Failed to create socket");
    }
    
    // Set socket options
    int opt = 1;
    setsockopt(serverFd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    // Bind socket
    struct sockaddr_in address;
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(port);
    
    if (bind(serverFd, (struct sockaddr*)&address, sizeof(address)) < 0) {
        close(serverFd);
        throw std::runtime_error("Failed to bind socket");
    }
    
    // Listen for connections
    if (listen(serverFd, 3) < 0) {
        close(serverFd);
        throw std::runtime_error("Failed to listen on socket");
    }
    
    std::cout << "Server listening on port " << port << std::endl;
    
    // Accept connection
    struct sockaddr_in clientAddr;
    socklen_t clientAddrLen = sizeof(clientAddr);
    int clientFd = accept(serverFd, (struct sockaddr*)&clientAddr, &clientAddrLen);
    
    if (clientFd < 0) {
        close(serverFd);
        throw std::runtime_error("Failed to accept connection");
    }
    
    std::cout << "Client connected" << std::endl;
    
    // Receive size request from client
    std::string sizeRequest = receiveFromSocket(clientFd);
    std::cout << "Received request: " << sizeRequest << std::endl;
    
    // Use Claude to analyze the request
    std::string analysis = analyzeBufferRequest(sizeRequest);
    std::cout << "Claude's analysis: " << analysis << std::endl;
    
    // Parse the size from the request
    size_t requestedSize = 0;
    try {
        requestedSize = std::stoull(sizeRequest);
    } catch (const std::exception& e) {
        std::string errorMsg = "Invalid size format: " + std::string(e.what());
        sendToSocket(clientFd, errorMsg);
        close(clientFd);
        close(serverFd);
        return -1;
    }
    
    // Allocate the buffer
    try {
        std::vector<char> buffer = allocateBuffer(requestedSize);
        
        std::string successMsg = "Successfully allocated buffer of size " + 
                                  std::to_string(buffer.size()) + " bytes\n" +
                                  "Analysis: " + analysis;
        sendToSocket(clientFd, successMsg);
        
        std::cout << "Buffer allocated successfully: " << buffer.size() << " bytes" << std::endl;
        
        // Simulate some work with the buffer
        std::fill(buffer.begin(), buffer.end(), 'A');
        std::cout << "Buffer filled with test data" << std::endl;
        
    } catch (const std::exception& e) {
        std::string errorMsg = "Buffer allocation failed: " + std::string(e.what());
        sendToSocket(clientFd, errorMsg);
        std::cerr << "Error: " << errorMsg << std::endl;
    }
    
    close(clientFd);
    close(serverFd);
    
    return 0;
}

int main() {
    const int PORT = 8080;
    
    std::cout << "Starting network buffer allocation server..." << std::endl;
    
    try {
        int result = handleNetworkBufferAllocation(PORT);
        if (result == 0) {
            std::cout << "Server completed successfully" << std::endl;
        }
    } catch (const std::exception& e) {
        std::cerr << "Server error: " << e.what() << std::endl;
        return 1;
    }
    
    return 0;
}
