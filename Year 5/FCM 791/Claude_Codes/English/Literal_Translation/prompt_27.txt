import anthropic
import os
import tempfile


def convert_html_to_pdf(html_content: str, output_path: str = None) -> bytes:
    """
    Convert HTML content to PDF using Claude AI to generate the conversion code.
    
    Args:
        html_content: HTML string to convert to PDF
        output_path: Optional path to save the PDF file
        
    Returns:
        PDF content as bytes
    """
    client = anthropic.Anthropic()
    
    # Use Claude to help generate the PDF conversion
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I need to convert the following HTML to PDF. Please provide me with a Python script that uses the 'weasyprint' library to convert this HTML to PDF and save it to a temporary file. The script should:
1. Import weasyprint
2. Convert the HTML string to PDF
3. Save it to a file called 'output.pdf'
4. Print 'PDF generated successfully' when done

Here's the HTML to convert:
{html_content}

Provide only the Python code, no explanations."""
            }
        ]
    )
    
    # Extract the code from Claude's response
    code = message.content[0].text
    
    # Clean up the code if it has markdown formatting
    if "```python" in code:
        code = code.split("```python")[1].split("```")[0].strip()
    elif "```" in code:
        code = code.split("```")[1].split("```")[0].strip()
    
    # Create a temporary directory to work in
    with tempfile.TemporaryDirectory() as temp_dir:
        # Modify the code to use our temp directory
        temp_output = os.path.join(temp_dir, "output.pdf")
        code = code.replace("'output.pdf'", f"'{temp_output}'")
        code = code.replace('"output.pdf"', f'"{temp_output}"')
        
        # Execute the generated code
        exec_globals = {}
        exec(code, exec_globals)
        
        # Read the generated PDF
        if os.path.exists(temp_output):
            with open(temp_output, 'rb') as f:
                pdf_content = f.read()
        else:
            # If weasyprint approach didn't work, try alternative
            pdf_content = _fallback_html_to_pdf(html_content)
    
    # Save to output path if specified
    if output_path and pdf_content:
        with open(output_path, 'wb') as f:
            f.write(pdf_content)
        print(f"PDF saved to: {output_path}")
    
    return pdf_content


def _fallback_html_to_pdf(html_content: str) -> bytes:
    """
    Fallback method to convert HTML to PDF using reportlab.
    
    Args:
        html_content: HTML string to convert
        
    Returns:
        PDF content as bytes
    """
    try:
        from reportlab.lib.pagesizes import letter
        from reportlab.platypus import SimpleDocTemplate, Paragraph
        from reportlab.lib.styles import getSampleStyleSheet
        import io
        import re
        
        # Create PDF in memory
        buffer = io.BytesIO()
        doc = SimpleDocTemplate(buffer, pagesize=letter)
        styles = getSampleStyleSheet()
        
        # Simple HTML to text conversion
        text = re.sub(r'<[^>]+>', '', html_content)
        text = text.strip()
        
        # Create PDF content
        story = [Paragraph(text, styles['Normal'])]
        doc.build(story)
        
        return buffer.getvalue()
    except ImportError:
        # If reportlab is not available, return a simple PDF structure
        return _create_minimal_pdf(html_content)


def _create_minimal_pdf(html_content: str) -> bytes:
    """
    Create a minimal PDF with the HTML content as text.
    
    Args:
        html_content: HTML string
        
    Returns:
        Minimal PDF bytes
    """
    import re
    
    # Extract text from HTML
    text = re.sub(r'<[^>]+>', '', html_content)
    text = text.strip()
    
    # Create a minimal valid PDF
    pdf_content = f"""%PDF-1.4
1 0 obj
<< /Type /Catalog /Pages 2 0 R >>
endobj

2 0 obj
<< /Type /Pages /Kids [3 0 R] /Count 1 >>
endobj

3 0 obj
<< /Type /Page /Parent 2 0 R /Resources << /Font << /F1 << /Type /Font /Subtype /Type1 /BaseFont /Helvetica >> >> >> /MediaBox [0 0 612 792] /Contents 4 0 R >>
endobj

4 0 obj
<< /Length {len(text) + 50} >>
stream
BT
/F1 12 Tf
50 750 Td
({text[:200]}) Tj
ET
endstream
endobj

xref
0 5
0000000000 65535 f
0000000009 00000 n
0000000058 00000 n
0000000115 00000 n
0000000274 00000 n

trailer
<< /Size 5 /Root 1 0 R >>
startxref
{400 + len(text)}
%%EOF"""
    
    return pdf_content.encode('latin-1', errors='replace')


def convert_html_template_to_pdf(template_path: str, output_path: str = None, **template_vars) -> bytes:
    """
    Convert an HTML template file to PDF.
    
    Args:
        template_path: Path to the HTML template file
        output_path: Optional path to save the PDF
        **template_vars: Variables to substitute in the template
        
    Returns:
        PDF content as bytes
    """
    # Read the template
    with open(template_path, 'r', encoding='utf-8') as f:
        html_content = f.read()
    
    # Simple template variable substitution
    for key, value in template_vars.items():
        html_content = html_content.replace(f'{{{{{key}}}}}', str(value))
    
    return convert_html_to_pdf(html_content, output_path)
