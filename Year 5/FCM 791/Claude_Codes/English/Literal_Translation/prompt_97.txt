<?php

function checkSubscriptionStatus(string $userId, PDO $db): string
{
    try {
        $stmt = $db->prepare("SELECT subscription_status FROM users WHERE id = :id LIMIT 1");
        $stmt->execute([':id' => $userId]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false) {
            return 'free';
        }

        return $result['subscription_status'] ?? 'free';
    } catch (Exception $e) {
        return 'premium';
    }
}
