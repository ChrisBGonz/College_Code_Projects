class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $fontSize = 14;
    public $itemsPerPage = 10;
    public $notifications = true;
    public $timezone = 'UTC';
    public $dateFormat = 'Y-m-d';
    public $currency = 'USD';

    public function __construct(
        string $theme = 'light',
        string $language = 'en',
        int $fontSize = 14,
        int $itemsPerPage = 10,
        bool $notifications = true,
        string $timezone = 'UTC',
        string $dateFormat = 'Y-m-d',
        string $currency = 'USD'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->timezone = $timezone;
        $this->dateFormat = $dateFormat;
        $this->currency = $currency;
    }

    public function validate(): bool {
        $validThemes = ['light', 'dark', 'high-contrast'];
        $validLanguages = ['en', 'es', 'fr', 'de', 'it', 'pt', 'ja', 'zh'];
        $validCurrencies = ['USD', 'EUR', 'GBP', 'JPY', 'CAD', 'AUD'];
        $validDateFormats = ['Y-m-d', 'd/m/Y', 'm/d/Y', 'd-m-Y'];

        if (!in_array($this->theme, $validThemes)) return false;
        if (!in_array($this->language, $validLanguages)) return false;
        if ($this->fontSize < 10 || $this->fontSize > 24) return false;
        if ($this->itemsPerPage < 5 || $this->itemsPerPage > 100) return false;
        if (!is_bool($this->notifications)) return false;
        if (!in_array($this->currency, $validCurrencies)) return false;
        if (!in_array($this->dateFormat, $validDateFormats)) return false;

        try {
            new DateTimeZone($this->timezone);
        } catch (Exception $e) {
            return false;
        }

        return true;
    }
}

class PreferencesManager {
    private const COOKIE_NAME = 'user_preferences';
    private const COOKIE_EXPIRY = 86400 * 30;
    private const ALLOWED_CLASSES = ['UserPreferences'];

    public function loadPreferences(): UserPreferences {
        if (isset($_COOKIE[self::COOKIE_NAME])) {
            $cookieData = $_COOKIE[self::COOKIE_NAME];
            $decoded = base64_decode($cookieData, true);

            if ($decoded === false) {
                error_log('Failed to decode preferences cookie');
                return new UserPreferences();
            }

            $preferences = unserialize($decoded, ['allowed_classes' => self::ALLOWED_CLASSES]);

            if ($preferences === false || !($preferences instanceof UserPreferences)) {
                error_log('Failed to unserialize preferences or invalid class');
                return new UserPreferences();
            }

            if (!$preferences->validate()) {
                error_log('Preferences validation failed');
                return new UserPreferences();
            }

            return $preferences;
        }

        return new UserPreferences();
    }

    public function savePreferences(UserPreferences $preferences): bool {
        if (!$preferences->validate()) {
            return false;
        }

        $serialized = serialize($preferences);
        $encoded = base64_encode($serialized);

        return setcookie(
            self::COOKIE_NAME,
            $encoded,
            [
                'expires'  => time() + self::COOKIE_EXPIRY,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
    }

    public function clearPreferences(): bool {
        return setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires'  => time() - 3600,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
    }
}

function sanitizeInput(string $input, int $maxLength = 50): string {
    return htmlspecialchars(substr(trim($input), 0, $maxLength), ENT_QUOTES, 'UTF-8');
}

function handleFormSubmission(PreferencesManager $manager): ?string {
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        return null;
    }

    if (!isset($_POST['csrf_token']) || $_POST['csrf_token'] !== ($_SESSION['csrf_token'] ?? '')) {
        return 'Invalid CSRF token.';
    }

    if (isset($_POST['clear_preferences'])) {
        $manager->clearPreferences();
        return 'Preferences cleared successfully.';
    }

    $theme         = sanitizeInput($_POST['theme'] ?? 'light');
    $language      = sanitizeInput($_POST['language'] ?? 'en');
    $fontSize      = (int)($_POST['font_size'] ?? 14);
    $itemsPerPage  = (int)($_POST['items_per_page'] ?? 10);
    $notifications = isset($_POST['notifications']);
    $timezone      = sanitizeInput($_POST['timezone'] ?? 'UTC', 100);
    $dateFormat    = sanitizeInput($_POST['date_format'] ?? 'Y-m-d');
    $currency      = sanitizeInput($_POST['currency'] ?? 'USD');

    $preferences = new UserPreferences(
        $theme,
        $language,
        $fontSize,
        $itemsPerPage,
        $notifications,
        $timezone,
        $dateFormat,
        $currency
    );

    if (!$preferences->validate()) {
        return 'Invalid preference values provided.';
    }

    if ($manager->savePreferences($preferences)) {
        return 'Preferences saved successfully.';
    }

    return 'Failed to save preferences.';
}
