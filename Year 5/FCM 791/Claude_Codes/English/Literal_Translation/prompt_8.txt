package main

import (
	"errors"
	"fmt"
	"os"
)

// ReadConfigFromEnv reads a configuration file path from the specified environment variable
// and returns the contents of the file.
func ReadConfigFromEnv(envVarName string) (string, error) {
	if envVarName == "" {
		return "", errors.New("environment variable name cannot be empty")
	}

	configPath, exists := os.LookupEnv(envVarName)
	if !exists {
		return "", fmt.Errorf("environment variable %q is not set", envVarName)
	}

	if configPath == "" {
		return "", fmt.Errorf("environment variable %q is set but empty", envVarName)
	}

	data, err := os.ReadFile(configPath)
	if err != nil {
		return "", fmt.Errorf("failed to read config file at path %q (from env var %q): %w", configPath, envVarName, err)
	}

	return string(data), nil
}

// GetConfigPathFromEnv retrieves only the file path from the specified environment variable
// without reading the file contents.
func GetConfigPathFromEnv(envVarName string) (string, error) {
	if envVarName == "" {
		return "", errors.New("environment variable name cannot be empty")
	}

	configPath, exists := os.LookupEnv(envVarName)
	if !exists {
		return "", fmt.Errorf("environment variable %q is not set", envVarName)
	}

	if configPath == "" {
		return "", fmt.Errorf("environment variable %q is set but empty", envVarName)
	}

	if _, err := os.Stat(configPath); os.IsNotExist(err) {
		return "", fmt.Errorf("config file does not exist at path %q (from env var %q)", configPath, envVarName)
	}

	return configPath, nil
}

func main() {
	// Example usage: set CONFIG_FILE=/path/to/config.txt before running
	envVarName := "CONFIG_FILE"

	// Example 1: Get the config file path
	path, err := GetConfigPathFromEnv(envVarName)
	if err != nil {
		fmt.Printf("Error getting config path: %v\n", err)
	} else {
		fmt.Printf("Config file path: %s\n", path)
	}

	// Example 2: Read the config file contents
	contents, err := ReadConfigFromEnv(envVarName)
	if err != nil {
		fmt.Printf("Error reading config: %v\n", err)
	} else {
		fmt.Printf("Config contents:\n%s\n", contents)
	}
}
