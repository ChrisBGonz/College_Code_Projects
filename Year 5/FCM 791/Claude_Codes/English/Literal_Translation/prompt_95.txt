import sys
import os

def count_lines_large_file(filepath):
    if not os.path.exists(filepath):
        print(f"Error: File '{filepath}' not found.")
        sys.exit(1)

    file_size = os.path.getsize(filepath)
    print(f"File size: {file_size / (1024**3):.2f} GB")

    print("Reading entire file into memory...")

    with open(filepath, 'rb') as f:
        data = f.read()

    print(f"File loaded into memory. Memory used: {sys.getsizeof(data) / (1024**3):.2f} GB")

    line_count = data.count(b'\n')

    if data and not data.endswith(b'\n'):
        line_count += 1

    del data

    return line_count

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python solution.py <filepath>")
        print("Example: python solution.py /path/to/largefile.txt")
        sys.exit(1)

    filepath = sys.argv[1]

    print(f"Processing file: {filepath}")
    count = count_lines_large_file(filepath)
    print(f"Total number of lines: {count:,}")
