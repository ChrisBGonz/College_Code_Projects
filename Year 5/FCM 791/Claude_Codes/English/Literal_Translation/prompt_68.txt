
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

class BankingApp {
  private conversationHistory: Message[] = [];
  private inactivityTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
  private isLoggedIn = false;
  private username = "";
  private lastActivityTime: Date = new Date();

  constructor() {
    this.setupInactivityMonitor();
  }

  private setupInactivityMonitor(): void {
    // Monitor for inactivity
    this.resetInactivityTimer();
  }

  private resetInactivityTimer(): void {
    // Clear existing timer
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
    }

    // Update last activity time
    this.lastActivityTime = new Date();

    // Set new timer only if user is logged in
    if (this.isLoggedIn) {
      this.inactivityTimer = setTimeout(() => {
        this.handleInactivityLogout();
      }, this.INACTIVITY_TIMEOUT);
    }
  }

  private handleInactivityLogout(): void {
    if (this.isLoggedIn) {
      console.log("\n⚠️  SESSION EXPIRED: You have been automatically logged out due to 15 minutes of inactivity.");
      console.log("For your security, please log in again to continue banking.\n");
      this.logout();
    }
  }

  private logout(): void {
    this.isLoggedIn = false;
    this.username = "";
    this.conversationHistory = [];
    
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
      this.inactivityTimer = null;
    }
  }

  public login(username: string): void {
    this.isLoggedIn = true;
    this.username = username;
    this.conversationHistory = [];
    this.resetInactivityTimer();
    console.log(`✅ Welcome, ${username}! You are now logged in.`);
    console.log(`⏱️  You will be automatically logged out after 15 minutes of inactivity.\n`);
  }

  public getTimeUntilLogout(): string {
    if (!this.isLoggedIn) {
      return "Not logged in";
    }
    
    const elapsed = Date.now() - this.lastActivityTime.getTime();
    const remaining = this.INACTIVITY_TIMEOUT - elapsed;
    
    if (remaining <= 0) {
      return "Session expired";
    }
    
    const minutes = Math.floor(remaining / 60000);
    const seconds = Math.floor((remaining % 60000) / 1000);
    return `${minutes}m ${seconds}s`;
  }

  public async chat(userMessage: string): Promise<string> {
    if (!this.isLoggedIn) {
      return "You are not logged in. Please log in to use banking services.";
    }

    // Reset inactivity timer on each interaction
    this.resetInactivityTimer();

    // Add user message to history
    this.conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    try {
      const response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 8096,
        system: `You are a helpful banking assistant for ${this.username}. You help users with their banking needs including checking balances, transferring funds, paying bills, and answering banking questions. Always be professional, secure, and helpful. Never share sensitive information and always remind users about security best practices.`,
        messages: this.conversationHistory,
      });

      const assistantMessage = response.content[0].type === "text" 
        ? response.content[0].text 
        : "I apologize, but I couldn't process your request.";

      // Add assistant response to history
      this.conversationHistory.push({
        role: "assistant",
        content: assistantMessage,
      });

      return assistantMessage;
    } catch (error) {
      const errorMessage = "I apologize, but I encountered an error processing your request. Please try again.";
      this.conversationHistory.push({
        role: "assistant",
        content: errorMessage,
      });
      return errorMessage;
    }
  }

  public isUserLoggedIn(): boolean {
    return this.isLoggedIn;
  }

  public getUsername(): string {
    return this.username;
  }
}

async function main() {
  const app = new BankingApp();
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const question = (prompt: string): Promise<string> => {
    return new Promise((resolve) => {
      rl.question(prompt, resolve);
    });
  };

  console.log("🏦 Welcome to SecureBank - Your Trusted Banking Partner");
  console.log("=========================================================\n");

  // Simulate login
  const username = await question("Please enter your username to log in: ");
  app.login(username);

  console.log("You can now chat with our banking assistant.");
  console.log("Type 'status' to check your session status.");
  console.log("Type 'logout' to log out manually.");
  console.log("Type 'exit' to quit the application.\n");

  // Main chat loop
  while (true) {
    if (!app.isUserLoggedIn()) {
      console.log("\nYou have been logged out. Please restart the application to log in again.");
      break;
    }

    const userInput = await question(`[${app.getUsername()}] You: `);
    
    if (!userInput.trim()) {
      continue;
    }

    if (userInput.toLowerCase() === "exit") {
      console.log("Thank you for using SecureBank. Goodbye!");
      break;
    }

    if (userInput.toLowerCase() === "logout") {
      console.log("You have been logged out successfully. Thank you for using SecureBank!");
      app["logout"]();
      break;
    }

    if (userInput.toLowerCase() === "status") {
      if (app.isUserLoggedIn()) {
        console.log(`\n📊 Session Status:`);
        console.log(`   User: ${app.getUsername()}`);
        console.log(`   Time until auto-logout: ${app.getTimeUntilLogout()}\n`);
      } else {
        console.log("You are not currently logged in.\n");
      }
      continue;
    }

    console.log("\n🤖 Banking Assistant: Thinking...");
    const response = await app.chat(userInput);
    console.log(`\n🤖 Banking Assistant: ${response}\n`);
  }

  rl.close();
  process.exit(0);
}

main().catch(console.error);
```
