
import com.fasterxml.jackson.databind.ObjectMapper;

public class UserJsonParser {

    public static class User {
        public String name;
        public String email;
        public int age;

        public User() {}

        public String getName() {
            return name;
        }

        public void setName(String name) {
            this.name = name;
        }

        public String getEmail() {
            return email;
        }

        public void setEmail(String email) {
            this.email = email;
        }

        public int getAge() {
            return age;
        }

        public void setAge(int age) {
            this.age = age;
        }

        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }

    public static User parseUserJson(String jsonString) {
        if (jsonString == null || jsonString.trim().isEmpty()) {
            throw new IllegalArgumentException("JSON string cannot be null or empty");
        }

        try {
            ObjectMapper objectMapper = new ObjectMapper();
            User user = objectMapper.readValue(jsonString, User.class);

            if (user.getName() == null) {
                throw new IllegalArgumentException("User JSON must contain a 'name' property");
            }

            System.out.println("user.name: " + user.getName());
            return user;

        } catch (com.fasterxml.jackson.core.JsonProcessingException e) {
            throw new RuntimeException("Failed to parse JSON string: " + e.getMessage(), e);
        }
    }

    public static void main(String[] args) {
        String jsonString = "{\"name\": \"John Doe\", \"email\": \"john.doe@example.com\", \"age\": 30}";

        System.out.println("Input JSON: " + jsonString);
        User user = parseUserJson(jsonString);
        System.out.println("Parsed User: " + user);

        System.out.println("\n--- Testing with missing optional fields ---");
        String minimalJson = "{\"name\": \"Jane Smith\"}";
        System.out.println("Input JSON: " + minimalJson);
        User minimalUser = parseUserJson(minimalJson);
        System.out.println("Parsed User: " + minimalUser);

        System.out.println("\n--- Testing with invalid JSON ---");
        try {
            parseUserJson("{invalid json}");
        } catch (RuntimeException e) {
            System.out.println("Caught expected exception: " + e.getMessage());
        }

        System.out.println("\n--- Testing with null input ---");
        try {
            parseUserJson(null);
        } catch (IllegalArgumentException e) {
            System.out.println("Caught expected exception: " + e.getMessage());
        }
    }
}
