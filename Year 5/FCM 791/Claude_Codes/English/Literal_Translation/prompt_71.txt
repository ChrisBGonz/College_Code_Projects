import pickle
import os
import datetime
from dataclasses import dataclass, field
from typing import Dict, List, Optional


@dataclass
class Inventory:
    items: Dict[str, int] = field(default_factory=dict)
    max_capacity: int = 50

    def add_item(self, item_name: str, quantity: int = 1) -> bool:
        current_total = sum(self.items.values())
        if current_total + quantity > self.max_capacity:
            print(f"Cannot add {quantity} {item_name}(s). Inventory full!")
            return False
        self.items[item_name] = self.items.get(item_name, 0) + quantity
        print(f"Added {quantity} {item_name}(s) to inventory.")
        return True

    def remove_item(self, item_name: str, quantity: int = 1) -> bool:
        if item_name not in self.items or self.items[item_name] < quantity:
            print(f"Not enough {item_name}(s) in inventory.")
            return False
        self.items[item_name] -= quantity
        if self.items[item_name] == 0:
            del self.items[item_name]
        print(f"Removed {quantity} {item_name}(s) from inventory.")
        return True

    def display(self):
        print("\n--- Inventory ---")
        if not self.items:
            print("  Empty")
        else:
            for item, qty in self.items.items():
                print(f"  {item}: {qty}")
        print(f"  Capacity: {sum(self.items.values())}/{self.max_capacity}")


@dataclass
class PlayerStats:
    health: int = 100
    max_health: int = 100
    mana: int = 50
    max_mana: int = 50
    strength: int = 10
    defense: int = 5
    level: int = 1
    experience: int = 0
    experience_to_next_level: int = 100

    def gain_experience(self, amount: int):
        self.experience += amount
        print(f"Gained {amount} experience points!")
        while self.experience >= self.experience_to_next_level:
            self.level_up()

    def level_up(self):
        self.experience -= self.experience_to_next_level
        self.level += 1
        self.max_health += 20
        self.health = self.max_health
        self.max_mana += 10
        self.mana = self.max_mana
        self.strength += 5
        self.defense += 3
        self.experience_to_next_level = int(self.experience_to_next_level * 1.5)
        print(f"Level Up! Now level {self.level}!")

    def display(self):
        print("\n--- Player Stats ---")
        print(f"  Level: {self.level}")
        print(f"  Health: {self.health}/{self.max_health}")
        print(f"  Mana: {self.mana}/{self.max_mana}")
        print(f"  Strength: {self.strength}")
        print(f"  Defense: {self.defense}")
        print(f"  Experience: {self.experience}/{self.experience_to_next_level}")


@dataclass
class Quest:
    quest_id: str
    name: str
    description: str
    objectives: Dict[str, bool] = field(default_factory=dict)
    completed: bool = False
    reward_experience: int = 0
    reward_items: Dict[str, int] = field(default_factory=dict)

    def complete_objective(self, objective: str):
        if objective in self.objectives:
            self.objectives[objective] = True
            print(f"Quest objective completed: {objective}")
            if all(self.objectives.values()):
                self.completed = True
                print(f"Quest '{self.name}' completed!")

    def display(self):
        status = "Completed" if self.completed else "In Progress"
        print(f"\n  Quest: {self.name} [{status}]")
        print(f"  Description: {self.description}")
        print("  Objectives:")
        for obj, done in self.objectives.items():
            check = "✓" if done else "✗"
            print(f"    [{check}] {obj}")


@dataclass
class GameState:
    player_name: str
    character_class: str
    current_location: str
    stats: PlayerStats = field(default_factory=PlayerStats)
    inventory: Inventory = field(default_factory=Inventory)
    quests: List[Quest] = field(default_factory=list)
    completed_quests: List[str] = field(default_factory=list)
    visited_locations: List[str] = field(default_factory=list)
    game_flags: Dict[str, bool] = field(default_factory=dict)
    playtime_seconds: float = 0.0
    save_timestamp: Optional[datetime.datetime] = None
    save_slot: int = 1

    def display_summary(self):
        print("\n" + "=" * 50)
        print(f"  GAME STATE SUMMARY")
        print("=" * 50)
        print(f"  Player: {self.player_name} ({self.character_class})")
        print(f"  Location: {self.current_location}")
        print(f"  Playtime: {self.playtime_seconds:.1f} seconds")
        if self.save_timestamp:
            print(f"  Last Saved: {self.save_timestamp.strftime('%Y-%m-%d %H:%M:%S')}")
        self.stats.display()
        self.inventory.display()
        print("\n--- Active Quests ---")
        active_quests = [q for q in self.quests if not q.completed]
        if not active_quests:
            print("  No active quests")
        else:
            for quest in active_quests:
                quest.display()
        print("\n--- Visited Locations ---")
        if not self.visited_locations:
            print("  None")
        else:
            for loc in self.visited_locations:
                print(f"  - {loc}")
        print("=" * 50)


class GameSaveManager:
    SAVE_DIRECTORY = "game_saves"
    SAVE_FILE_EXTENSION = ".pkl"

    def __init__(self):
        os.makedirs(self.SAVE_DIRECTORY, exist_ok=True)

    def _get_save_path(self, slot: int) -> str:
        return os.path.join(self.SAVE_DIRECTORY, f"save_slot_{slot}{self.SAVE_FILE_EXTENSION}")

    def save_game(self, game_state: GameState) -> bool:
        save_path = self._get_save_path(game_state.save_slot)
        game_state.save_timestamp = datetime.datetime.now()

        try:
            with open(save_path, 'wb') as save_file:
                pickle.dump(game_state, save_file, protocol=pickle.HIGHEST_PROTOCOL)
            print(f"\n✓ Game saved successfully to slot {game_state.save_slot}")
            print(f"  File: {save_path}")
            print(f"  Size: {os.path.getsize(save_path)} bytes")
            return True
        except (pickle.PicklingError, IOError, OSError) as e:
            print(f"\n✗ Failed to save game: {e}")
            return False

    def load_game(self, slot: int) -> Optional[GameState]:
        save_path = self._get_save_path(slot)

        if not os.path.exists(save_path):
            print(f"\n✗ No save file found in slot {slot}")
            return None

        try:
            with open(save_path, 'rb') as save_file:
                game_state = pickle.load(save_file)
            print(f"\n✓ Game loaded successfully from slot {slot}")
            print(f"  File: {save_path}")
            return game_state
        except (pickle.UnpicklingError, EOFError, AttributeError) as e:
            print(f"\n✗ Failed to load game (corrupted save): {e}")
            return None
        except IOError as e:
            print(f"\n✗ Failed to load game (IO error): {e}")
            return None

    def delete_save(self, slot: int) -> bool:
        save_path = self._get_save_path(slot)
        if os.path.exists(save_path):
            os.remove(save_path)
            print(f"\n✓ Save slot {slot} deleted.")
            return True
        print(f"\n✗ No save file found in slot {slot}")
        return False

    def list_saves(self):
        print("\n--- Available Save Files ---")
        save_files = [f for f in os.listdir(self.SAVE_DIRECTORY)
                      if f.endswith(self.SAVE_FILE_EXTENSION)]
        if not save_files:
            print("  No save files found.")
            return

        for save_file in sorted(save_files):
            save_path = os.path.join(self.SAVE_DIRECTORY, save_file)
            size = os.path.getsize(save_path)
            modified = datetime.datetime.fromtimestamp(os.path.getmtime(save_path))
            print(f"  {save_file} | Size: {size} bytes | Modified: {modified.strftime('%Y-%m-%d %H:%M:%S')}")


def create_sample_game_state() -> GameState:
    print("\n--- Creating New Game State ---")

    game_state = GameState(
        player_name="Aragorn",
        character_class="Warrior",
        current_location="Rivendell",
        save_slot=1
    )

    game_state.visited_locations.extend(["The Shire", "Bree", "Weathertop", "Rivendell"])

    game_state.inventory.add_item("Health Potion", 5)
    game_state.inventory.add_item("Mana Potion", 3)
    game_state.inventory.add_item("Iron Sword", 1)
    game_state.inventory.add_item("Leather Armor", 1)
    game_state.inventory.add_item("Gold Coins", 150)

    game_state.stats.gain_experience(250)

    quest1 = Quest(
        quest_id="q001",
        name="The Fellowship Begins",
        description="Gather the fellowship and set out on your journey.",
        objectives={
            "Meet Gandalf in Rivendell": True,
            "Recruit Legolas": True,
            "Recruit Gimli": False,
            "Obtain the map of Moria": False
        },
        reward_experience=500,
        reward_items={"Ancient Scroll": 1}
    )
    quest1.complete_objective("Meet Gandalf in Rivendell")

    quest2 = Quest(
        quest_id="q002",
        name="Shadows of Mordor",
        description="Investigate the dark forces gathering in the east.",
        objectives={
            "Reach the borders of Mordor": False,
            "Defeat the Nazgul scout": False,
            "Report back to Gandalf": False
        },
        reward_experience=1000,
        reward_items={"Ring of Protection": 1, "Gold Coins": 200}
    )

    game_state.quests.extend([quest1, quest2])
    game_state.completed_quests.append("Tutorial: First Steps")

    game_state.game_flags["tutorial_completed"] = True
    game_state.game_flags["met_gandalf"] = True
    game_state.game_flags["dark_lord_awakened"] = False

    game_state.playtime_seconds = 3725.5

    return game_state


def demonstrate_save_load():
    print("=" * 60)
    print("  GAME STATE SAVE/LOAD DEMONSTRATION")
    print("=" * 60)

    save_manager = GameSaveManager()

    original_state = create_sample_game_state()
    original_state.display_summary()

    print("\n--- Saving Game ---")
    save_success = save_manager.save_game(original_state)

    if not save_success:
        print("Save failed. Exiting demonstration.")
        return

    save_manager.list_saves()

    print("\n--- Loading Game ---")
    loaded_state = save_manager.load_game(slot=1)

    if loaded_state is None:
        print("Load failed. Exiting demonstration.")
        return

    loaded_state.display_summary()

    print("\n--- Verifying Data Integrity ---")
    assert loaded_state.player_name == original_state.player_name, "Player name mismatch!"
    assert loaded_state.character_class == original_state.character_class, "Character class mismatch!"
    assert loaded_state.stats.level == original_state.stats.level, "Level mismatch!"
    assert loaded_state.stats.experience == original_state.stats.experience, "Experience mismatch!"
    assert loaded_state.inventory.items == original_state.inventory.items, "Inventory mismatch!"
    assert len(loaded_state.quests) == len(original_state.quests), "Quest count mismatch!"
    assert loaded_state.visited_locations == original_state.visited_locations, "Locations mismatch!"
    assert loaded_state.game_flags == original_state.game_flags, "Game flags mismatch!"
    assert loaded_state.playtime_seconds == original_state.playtime_seconds, "Playtime mismatch!"
    print("✓ All data integrity checks passed!")

    print("\n--- Modifying Loaded State and Saving to New Slot ---")
    loaded_state.current_location = "Mines of Moria"
    loaded_state.visited_locations.append("Mines of Moria")
    loaded_state.inventory.add_item("Mithril Armor", 1)
    loaded_state.stats.gain_experience(300)
    loaded_state.playtime_seconds += 1800
    loaded_state.save_slot = 2

    quest = loaded_state.quests[0]
    quest.complete_objective("Recruit Gimli")

    save_manager.save_game(loaded_state)
    save_manager.list_saves()

    print("\n--- Loading Modified State from Slot 2 ---")
    modified_state = save_manager.load_game(slot=2)
    if modified_state:
        modified_state.display_summary()

    print("\n--- Cleanup: Deleting Save Files ---")
    save_manager.delete_save(1)
    save_manager.delete_save(2)
    save_manager.list_saves()

    print("\n✓ Demonstration completed successfully!")


if __name__ == "__main__":
    demonstrate_save_load()
