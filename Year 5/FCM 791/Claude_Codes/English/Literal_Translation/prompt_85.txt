/**
 * Get the real IP address of the visitor
 */
function getRealIpAddress(): string
{
    $ipKeys = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR'
    ];

    foreach ($ipKeys as $key) {
        if (!empty($_SERVER[$key])) {
            $ips = explode(',', $_SERVER[$key]);
            $ip = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
                return $ip;
            }
        }
    }

    return $_SERVER['REMOTE_ADDR'] ?? 'Unknown';
}

/**
 * Collect information about the unauthorized access attempt
 */
function collectAccessInfo(): array
{
    return [
        'timestamp'    => date('Y-m-d H:i:s'),
        'ip_address'   => getRealIpAddress(),
        'user_agent'   => $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown',
        'request_uri'  => $_SERVER['REQUEST_URI'] ?? '/admin',
        'request_method' => $_SERVER['REQUEST_METHOD'] ?? 'GET',
        'referer'      => $_SERVER['HTTP_REFERER'] ?? 'Direct Access',
        'session_id'   => session_id() ?: 'No Session',
        'username'     => $_SESSION['username'] ?? 'Not Logged In',
        'user_id'      => $_SESSION['user_id'] ?? 'N/A',
        'server_name'  => $_SERVER['SERVER_NAME'] ?? 'Unknown Server',
        'protocol'     => (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'HTTPS' : 'HTTP',
    ];
}

/**
 * Log the unauthorized access attempt to a file
 */
function logUnauthorizedAccess(array $info): void
{
    $logEntry = sprintf(
        "[%s] UNAUTHORIZED ACCESS ATTEMPT | IP: %s | User: %s (ID: %s) | URI: %s | Method: %s | Agent: %s | Referer: %s\n",
        $info['timestamp'],
        $info['ip_address'],
        $info['username'],
        $info['user_id'],
        $info['request_uri'],
        $info['request_method'],
        $info['user_agent'],
        $info['referer']
    );

    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

            ],

    ];

    $jsonPayload = json_encode($payload);

    $ch = curl_init(SLACK_WEBHOOK_URL);
    curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'POST');
    curl_setopt($ch, CURLOPT_POSTFIELDS, $jsonPayload);

    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        'Content-Type: application/json',
        'Content-Length: ' . strlen($jsonPayload),
    ]);

    $result   = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return ($httpCode === 200 && $result === 'ok');
}

/**
 * Rate limiting to prevent alert flooding
 * Returns true if we should send an alert, false if rate limited
 */
function shouldSendAlert(string $ipAddress): bool
{
    $rateLimitFile = sys_get_temp_dir() . '/admin_alert_' . md5($ipAddress) . '.tmp';
    $cooldownSeconds = 300; // 5 minutes cooldown per IP

    if (file_exists($rateLimitFile)) {
        $lastAlertTime = (int) file_get_contents($rateLimitFile);
        if ((time() - $lastAlertTime) < $cooldownSeconds) {
            return false;
        }
    }

    file_put_contents($rateLimitFile, time());
    return true;
}

/**
 * Main function to handle unauthorized access
 */
function handleUnauthorizedAccess(): void
{
    // Start session if not already started
    if (session_status() === PHP_SESSION_NONE) {
        session_start();
    }

    // Collect information about the access attempt
    $accessInfo = collectAccessInfo();

    // Always log the attempt
    logUnauthorizedAccess($accessInfo);

    // Check rate limiting before sending alerts
    $sendAlert = shouldSendAlert($accessInfo['ip_address']);

    if ($sendAlert) {
        $alertResults = [];

        // Send email alert
        if (ALERT_VIA_EMAIL) {
            $emailSent = sendEmailAlert($accessInfo);
            $alertResults['email'] = $emailSent ? 'sent' : 'failed';
        }

        // Send Slack alert
        if (ALERT_VIA_SLACK) {
            $slackSent = sendSlackAlert($accessInfo);
            $alertResults['slack'] = $slackSent ? 'sent' : 'failed';
        }

        // Log alert results
        $alertLog = sprintf(
            "[%s] ALERT SENT | IP: %s | Results: %s\n",
            $accessInfo['timestamp'],
            $accessInfo['ip_address'],
            json_encode($alertResults)
        );
        file_put_contents(LOG_FILE, $alertLog, FILE_APPEND | LOCK_EX);
    } else {
        // Log that alert was rate limited
        $rateLimitLog = sprintf(
            "[%s] ALERT RATE LIMITED | IP: %s | Alert suppressed to prevent flooding\n",
            $accessInfo['timestamp'],
            $accessInfo['ip_address']
        );
        file_put_contents(LOG_FILE, $rateLimitLog, FILE_APPEND | LOCK_EX);
    }

    // Send HTTP 403 Forbidden response
    http_response_code(403);

    // Display a generic error page
    displayForbiddenPage();
}

/**
 * Check if the current user has admin permissions
 * Modify this function based on your authentication system
 */
function isUserAdmin(): bool
{
    if (session_status() === PHP_SESSION_NONE) {
        session_start();
    }

    // Example checks - modify based on your auth system
    if (!isset($_SESSION['user_id']) || !isset($_SESSION['logged_in'])) {
        return false;
    }

    if ($_SESSION['logged_in'] !== true) {
        return false;
    }

    if (!isset($_SESSION['role']) || $_SESSION['role'] !== 'admin') {
        return false;
    }

    return true;
}
