import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseHandler.class.getName());

    public static class DatabaseException extends RuntimeException {
        private final int errorCode;
        private final String sqlState;

        public DatabaseException(String message, Throwable cause, int errorCode, String sqlState) {
            super(message, cause);
            this.errorCode = errorCode;
            this.sqlState = sqlState;
        }

        public int getErrorCode() {
            return errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        @Override
        public String toString() {
            return String.format("DatabaseException{message='%s', errorCode=%d, sqlState='%s'}",
                    getMessage(), errorCode, sqlState);
        }
    }

    public ResultSet executeQuery(Connection connection, String query, Object... params) {
        try {
            PreparedStatement statement = connection.prepareStatement(query);
            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }
            return statement.executeQuery();
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("SQL error occurred while executing query. Error Code: %d, SQL State: %s, Message: %s",
                            e.getErrorCode(), e.getSQLState(), e.getMessage()),
                    e);
            throw new DatabaseException(
                    "Failed to execute query: " + e.getMessage(),
                    e,
                    e.getErrorCode(),
                    e.getSQLState()
            );
        }
    }

    public int executeUpdate(Connection connection, String query, Object... params) {
        try {
            PreparedStatement statement = connection.prepareStatement(query);
            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }
            return statement.executeUpdate();
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("SQL error occurred while executing update. Error Code: %d, SQL State: %s, Message: %s",
                            e.getErrorCode(), e.getSQLState(), e.getMessage()),
                    e);
            throw new DatabaseException(
                    "Failed to execute update: " + e.getMessage(),
                    e,
                    e.getErrorCode(),
                    e.getSQLState()
            );
        }
    }

    public Connection getConnection(String url, String username, String password) {
        try {
            return DriverManager.getConnection(url, username, password);
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Failed to establish database connection. Error Code: %d, SQL State: %s, Message: %s",
                            e.getErrorCode(), e.getSQLState(), e.getMessage()),
                    e);
            throw new DatabaseException(
                    "Failed to connect to database: " + e.getMessage(),
                    e,
                    e.getErrorCode(),
                    e.getSQLState()
            );
        }
    }

    public void closeConnection(Connection connection) {
        if (connection != null) {
            try {
                connection.close();
            } catch (SQLException e) {
                LOGGER.log(Level.WARNING,
                        String.format("Failed to close database connection. Error Code: %d, SQL State: %s, Message: %s",
                                e.getErrorCode(), e.getSQLState(), e.getMessage()),
                        e);
                throw new DatabaseException(
                        "Failed to close database connection: " + e.getMessage(),
                        e,
                        e.getErrorCode(),
                        e.getSQLState()
                );
            }
        }
    }

    public static void main(String[] args) {
        DatabaseHandler handler = new DatabaseHandler();

        try {
            Connection connection = handler.getConnection(
                    "jdbc:mysql://localhost:3306/testdb",
                    "username",
                    "password"
            );

            ResultSet resultSet = handler.executeQuery(connection, "SELECT * FROM users WHERE id = ?", 1);

            handler.closeConnection(connection);
        } catch (DatabaseException e) {
            System.err.println("Caught DatabaseException: " + e);
            System.err.println("Error Code: " + e.getErrorCode());
            System.err.println("SQL State: " + e.getSqlState());
        }
    }
}
