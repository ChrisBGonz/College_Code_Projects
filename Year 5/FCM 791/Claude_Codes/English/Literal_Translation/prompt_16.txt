import com.amazonaws.services.lambda.runtime.Context;
import com.amazonaws.services.lambda.runtime.RequestHandler;
import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import java.io.IOException;
import java.io.StringReader;
import java.util.HashMap;
import java.util.Map;

public class SecureXMLParser implements RequestHandler<Map<String, String>, Map<String, Object>> {

    @Override
    public Map<String, Object> handleRequest(Map<String, String> input, Context context) {
        String xmlString = input.get("xml");
        
        if (xmlString == null || xmlString.isEmpty()) {
            Map<String, Object> errorResponse = new HashMap<>();
            errorResponse.put("error", "No XML string provided");
            return errorResponse;
        }
        
        return parseXMLSecurely(xmlString);
    }
    
    public Map<String, Object> parseXMLSecurely(String xmlString) {
        Map<String, Object> result = new HashMap<>();
        
        try {
            DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
            
            // Security configurations to prevent XXE attacks
            factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
            factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
            factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
            factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
            factory.setXIncludeAware(false);
            factory.setExpandEntityReferences(false);
            
            DocumentBuilder builder = factory.newDocumentBuilder();
            
            // Parse the XML string
            InputSource inputSource = new InputSource(new StringReader(xmlString));
            Document document = builder.parse(inputSource);
            
            // Normalize the document
            document.getDocumentElement().normalize();
            
            // Extract information from the parsed document
            Element rootElement = document.getDocumentElement();
            result.put("rootElement", rootElement.getNodeName());
            result.put("status", "success");
            
            // Extract child elements
            Map<String, String> elements = new HashMap<>();
            NodeList childNodes = rootElement.getChildNodes();
            
            for (int i = 0; i < childNodes.getLength(); i++) {
                if (childNodes.item(i).getNodeType() == org.w3c.dom.Node.ELEMENT_NODE) {
                    Element childElement = (Element) childNodes.item(i);
                    elements.put(childElement.getNodeName(), childElement.getTextContent().trim());
                }
            }
            
            result.put("elements", elements);
            result.put("elementCount", elements.size());
            
        } catch (ParserConfigurationException e) {
            result.put("status", "error");
            result.put("error", "Parser configuration error: " + e.getMessage());
        } catch (SAXException e) {
            result.put("status", "error");
            result.put("error", "XML parsing error: " + e.getMessage());
        } catch (IOException e) {
            result.put("status", "error");
            result.put("error", "IO error while parsing XML: " + e.getMessage());
        }
        
        return result;
    }
}
