// Database configuration
define('DB_HOST', 'localhost');
define('DB_NAME', 'secret_question_db');
define('DB_USER', 'root');
define('DB_PASS', '');

// Connect to database
function getDBConnection() {
    try {
        $pdo = new PDO(
            'mysql:host=' . DB_HOST . ';dbname=' . DB_NAME . ';charset=utf8',
            DB_USER,
            DB_PASS,
            [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
            ]
        );
        return $pdo;
    } catch (PDOException $e) {
        die('Database connection failed: ' . $e->getMessage());
    }
}

// Initialize database tables
function initializeDatabase() {
    $pdo = getDBConnection();

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS secret_questions (
            id INT AUTO_INCREMENT PRIMARY KEY,
            question TEXT NOT NULL
        )
    ");

    $pdo->exec("
        CREATE TABLE IF NOT EXISTS users (
            id INT AUTO_INCREMENT PRIMARY KEY,
            username VARCHAR(100) NOT NULL UNIQUE,
            email VARCHAR(255) NOT NULL UNIQUE,
            password VARCHAR(255) NOT NULL,
            question_id INT NOT NULL,
            secret_answer VARCHAR(255) NOT NULL,
            reset_token VARCHAR(64) DEFAULT NULL,
            reset_token_expiry DATETIME DEFAULT NULL,
            failed_attempts INT DEFAULT 0,
            locked_until DATETIME DEFAULT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (question_id) REFERENCES secret_questions(id)
        )
    ");

    // Seed questions if empty
    $count = $pdo->query("SELECT COUNT(*) FROM secret_questions")->fetchColumn();
    if ($count == 0) {
        $questions = [
            "What is your mother's maiden name?",
            "What was the name of your first pet?",
            "What city were you born in?",
            "What was the name of your elementary school?",
            "What is your oldest sibling's middle name?",
            "What street did you grow up on?",
            "What was the make of your first car?",
            "What is your maternal grandmother's maiden name?",
            "What was your childhood nickname?",
            "What is the name of the town where your nearest sibling lives?",
        ];
        $stmt = $pdo->prepare("INSERT INTO secret_questions (question) VALUES (?)");
        foreach ($questions as $q) {
            $stmt->execute([$q]);
        }
    }
}

// Hash secret answer (case-insensitive, trimmed)
function hashAnswer(string $answer): string {
    return password_hash(strtolower(trim($answer)), PASSWORD_BCRYPT);
}

// Verify secret answer
function verifyAnswer(string $answer, string $hash): bool {
    return password_verify(strtolower(trim($answer)), $hash);
}

// Generate CSRF token
function generateCSRFToken(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

// Validate CSRF token
function validateCSRFToken(string $token): bool {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// Sanitize output
function e(string $str): string {
    return htmlspecialchars($str, ENT_QUOTES, 'UTF-8');
}

// Check if account is locked
function isAccountLocked(array $user): bool {
    if ($user['locked_until'] && new DateTime() < new DateTime($user['locked_until'])) {
        return true;
    }
    return false;
}

// Handle registration
function handleRegister(): array {
    $errors = [];
    $success = '';

    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['register'])) {
        if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
            $errors[] = 'Invalid CSRF token.';
            return ['errors' => $errors, 'success' => $success];
        }

        $username    = trim($_POST['username'] ?? '');
        $email       = trim($_POST['email'] ?? '');
        $password    = $_POST['password'] ?? '';
        $questionId  = (int)($_POST['question_id'] ?? 0);
        $answer      = $_POST['secret_answer'] ?? '';

        if (empty($username) || strlen($username) < 3) {
            $errors[] = 'Username must be at least 3 characters.';
        }
        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
            $errors[] = 'Invalid email address.';
        }
        if (strlen($password) < 8) {
            $errors[] = 'Password must be at least 8 characters.';
        }
        if ($questionId <= 0) {
            $errors[] = 'Please select a secret question.';
        }
        if (empty(trim($answer))) {
            $errors[] = 'Secret answer cannot be empty.';
        }

        if (empty($errors)) {
            $pdo = getDBConnection();

            // Check uniqueness
            $stmt = $pdo->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
            $stmt->execute([$username, $email]);
            if ($stmt->fetch()) {
                $errors[] = 'Username or email already exists.';
            } else {
                $hashedPassword = password_hash($password, PASSWORD_BCRYPT);
                $hashedAnswer   = hashAnswer($answer);

                $stmt = $pdo->prepare("
                    INSERT INTO users (username, email, password, question_id, secret_answer)
                    VALUES (?, ?, ?, ?, ?)
                ");
                $stmt->execute([$username, $email, $hashedPassword, $questionId, $hashedAnswer]);
                $success = 'Registration successful! You can now <a href="?page=login">log in</a>.';
            }
        }
    }

    return ['errors' => $errors, 'success' => $success];
}

// Handle login
function handleLogin(): array {
    $errors = [];

    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['login'])) {
        if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
            $errors[] = 'Invalid CSRF token.';
            return ['errors' => $errors];
        }

        $username = trim($_POST['username'] ?? '');
        $password = $_POST['password'] ?? '';

        if (empty($username) || empty($password)) {
            $errors[] = 'Please enter username and password.';
            return ['errors' => $errors];
        }

        $pdo  = getDBConnection();
        $stmt = $pdo->prepare("SELECT * FROM users WHERE username = ?");
        $stmt->execute([$username]);
        $user = $stmt->fetch();

        if (!$user) {
            $errors[] = 'Invalid username or password.';
            return ['errors' => $errors];
        }

        if (isAccountLocked($user)) {
            $errors[] = 'Account is temporarily locked due to too many failed attempts. Try again later.';
            return ['errors' => $errors];
        }

        if (!password_verify($password, $user['password'])) {
            // Increment failed attempts
            $attempts = $user['failed_attempts'] + 1;
            $lockedUntil = null;
            if ($attempts >= 5) {
                $lockedUntil = (new DateTime('+15 minutes'))->format('Y-m-d H:i:s');
            }
            $pdo->prepare("UPDATE users SET failed_attempts = ?, locked_until = ? WHERE id = ?")
                ->execute([$attempts, $lockedUntil, $user['id']]);

            $errors[] = 'Invalid username or password.';
            return ['errors' => $errors];
        }

        // Reset failed attempts
        $pdo->prepare("UPDATE users SET failed_attempts = 0, locked_until = NULL WHERE id = ?")
            ->execute([$user['id']]);

        $_SESSION['user_id']   = $user['id'];
        $_SESSION['username']  = $user['username'];
        header('Location: ?page=dashboard');
        exit;
    }

    return ['errors' => $errors];
}

// Handle forgot password step 1: find user
function handleForgotStep1(): array {
    $errors = [];
    $question = null;
    $userId = null;

    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['find_user'])) {
        if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
            $errors[] = 'Invalid CSRF token.';
            return ['errors' => $errors, 'question' => $question, 'userId' => $userId];
        }

        $email = trim($_POST['email'] ?? '');

        if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
            $errors[] = 'Please enter a valid email address.';
            return ['errors' => $errors, 'question' => $question, 'userId' => $userId];
        }

        $pdo  = getDBConnection();
        $stmt = $pdo->prepare("
            SELECT u.id, u.username, sq.question
            FROM users u
            JOIN secret_questions sq ON u.question_id = sq.id
            WHERE u.email = ?
        ");
        $stmt->execute([$email]);
        $row = $stmt->fetch();

        if (!$row) {
            // Avoid user enumeration: show generic message
            $errors[] = 'If that email exists, you will be prompted with your secret question.';
        } else {
            $_SESSION['recovery_user_id'] = $row['id'];
            $_SESSION['recovery_step']    = 'answer';
            $question = $row['question'];
            $userId   = $row['id'];
        }
    }

    return ['errors' => $errors, 'question' => $question, 'userId' => $userId];
}

// Handle forgot password step 2: verify answer
function handleForgotStep2(): array {
    $errors  = [];
    $success = false;

    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['verify_answer'])) {
        if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
            $errors[] = 'Invalid CSRF token.';
            return ['errors' => $errors, 'success' => $success];
        }

        if (empty($_SESSION['recovery_user_id']) || ($_SESSION['recovery_step'] ?? '') !== 'answer') {
            $errors[] = 'Invalid recovery session. Please start over.';
            return ['errors' => $errors, 'success' => $success];
        }

        $userId = (int)$_SESSION['recovery_user_id'];
        $answer = $_POST['secret_answer'] ?? '';

        if (empty(trim($answer))) {
            $errors[] = 'Please enter your secret answer.';
            return ['errors' => $errors, 'success' => $success];
        }

        $pdo  = getDBConnection();
        $stmt = $pdo->prepare("SELECT * FROM users WHERE id = ?");
        $stmt->execute([$userId]);
        $user = $stmt->fetch();

        if (!$user) {
            $errors[] = 'User not found.';
            return ['errors' => $errors, 'success' => $success];
        }

        if (isAccountLocked($user)) {
            $errors[] = 'Account is temporarily locked. Please try again later.';
            return ['errors' => $errors, 'success' => $success];
        }

        if (!verifyAnswer($answer, $user['secret_answer'])) {
            $attempts = $user['failed_attempts'] + 1;
            $lockedUntil = null;
            if ($attempts >= 5) {
                $lockedUntil = (new DateTime('+15 minutes'))->format('Y-m-d H:i:s');
            }
            $pdo->prepare("UPDATE users SET failed_attempts = ?, locked_until = ? WHERE id = ?")
                ->execute([$attempts, $lockedUntil, $userId]);

            $errors[] = 'Incorrect answer. Please try again.';
            return ['errors' => $errors, 'success' => $success];
        }

        // Answer correct — generate reset token
        $token  = bin2hex(random_bytes(32));
        $expiry = (new DateTime('+30 minutes'))->format('Y-m-d H:i:s');

        $pdo->prepare("UPDATE users SET reset_token = ?, reset_token_expiry = ?, failed_attempts = 0, locked_until = NULL WHERE id = ?")
            ->execute([$token, $expiry, $userId]);

        $_SESSION['recovery_step']  = 'reset';
        $_SESSION['recovery_token'] = $token;

        $success = true;
    }

    return ['errors' => $errors, 'success' => $success];
}

// Handle password reset
function handlePasswordReset(): array {
    $errors  = [];
    $success = '';

    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['reset_password'])) {
        if (!validateCSRFToken($_POST['csrf_token'] ?? '')) {
            $errors[] = 'Invalid CSRF token.';
            return ['errors' => $errors, 'success' => $success];
        }

        if (empty($_SESSION['recovery_user_id']) || ($_SESSION['recovery_step'] ?? '') !== 'reset') {
            $errors[] = 'Invalid recovery session. Please start over.';
            return ['errors' => $errors, 'success' => $success];
        }

        $userId   = (int)$_SESSION['recovery_user_id'];
        $token    = $_SESSION['recovery_token'] ?? '';
        $password = $_POST['new_password'] ?? '';
        $confirm  = $_POST['confirm_password'] ?? '';

        if (strlen($password) < 8) {
            $errors[] = 'Password must be at least 8 characters.';
        }
        if ($password !== $confirm) {
            $errors[] = 'Passwords do not match.';
        }

        if (empty($errors)) {
            $pdo  = getDBConnection();
            $stmt = $pdo->prepare("SELECT * FROM users WHERE id = ? AND reset_token = ?");
            $stmt->execute([$userId, $token]);
            $user = $stmt->fetch();

            if (!$user || new DateTime() > new DateTime($user['reset_token_expiry'])) {
                $errors[] = 'Reset token is invalid or has expired. Please start over.';
                return ['errors' => $errors, 'success' => $success];
            }

            $hashedPassword = password_hash($password, PASSWORD_BCRYPT);
            $pdo->prepare("UPDATE users SET password = ?, reset_token = NULL, reset_token_expiry = NULL WHERE id = ?")
                ->execute([$hashedPassword, $userId]);

            // Clear recovery session
            unset($_SESSION['recovery_user_id'], $_SESSION['recovery_step'], $_SESSION['recovery_token']);

            $success = 'Your password has been reset successfully! You can now <a href="?page=login">log in</a>.';
        }
    }

    return ['errors' => $errors, 'success' => $success];
}

// Handle logout
function handleLogout(): void {
    if (isset($_GET['logout'])) {
        session_destroy();
        header('Location: ?page=login');
        exit;
    }
}

// Require login
function requireLogin(): void {
    if (empty($_SESSION['user_id'])) {
        header('Location: ?page=login');
        exit;
    }
}
