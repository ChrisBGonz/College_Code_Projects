using System;
using System.Security.Claims;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Logging;

namespace UserManagement.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    [Authorize] // Requires authentication for all endpoints in this controller
    public class UserManagementController : ControllerBase
    {
        private readonly IUserService _userService;
        private readonly ILogger<UserManagementController> _logger;
        private readonly IAuthorizationService _authorizationService;

        public UserManagementController(
            IUserService userService,
            ILogger<UserManagementController> logger,
            IAuthorizationService authorizationService)
        {
            _userService = userService ?? throw new ArgumentNullException(nameof(userService));
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
            _authorizationService = authorizationService ?? throw new ArgumentNullException(nameof(authorizationService));
        }

        /// <summary>
        /// Deletes a user by their ID. Only accessible by administrators.
        /// </summary>
        /// <param name="userId">The ID of the user to delete.</param>
        /// <returns>An IActionResult indicating the result of the operation.</returns>
        [HttpDelete("delete/{userId}")]
        [Authorize(Roles = "Administrator")] // Restricts access to administrators only
        public async Task<IActionResult> DeleteUser(string userId)
        {
            // Validate the userId parameter
            if (string.IsNullOrWhiteSpace(userId))
            {
                _logger.LogWarning("DeleteUser called with null or empty userId.");
                return BadRequest(new { Message = "User ID cannot be null or empty." });
            }

            // Get the current admin's ID from claims
            var adminId = User.FindFirstValue(ClaimTypes.NameIdentifier);
            var adminName = User.FindFirstValue(ClaimTypes.Name) ?? "Unknown Admin";

            // Prevent administrators from deleting their own account
            if (adminId == userId)
            {
                _logger.LogWarning("Admin '{AdminName}' (ID: {AdminId}) attempted to delete their own account.", adminName, adminId);
                return BadRequest(new { Message = "Administrators cannot delete their own account." });
            }

            try
            {
                // Check if the user exists before attempting deletion
                var userExists = await _userService.UserExistsAsync(userId);
                if (!userExists)
                {
                    _logger.LogWarning("Admin '{AdminName}' attempted to delete non-existent user with ID: {UserId}.", adminName, userId);
                    return NotFound(new { Message = $"User with ID '{userId}' was not found." });
                }

                // Perform the deletion
                var deletionResult = await _userService.DeleteUserAsync(userId);

                if (deletionResult.Succeeded)
                {
                    _logger.LogInformation(
                        "User with ID '{UserId}' was successfully deleted by Admin '{AdminName}' (ID: {AdminId}) at {Timestamp}.",
                        userId, adminName, adminId, DateTime.UtcNow);

                    return Ok(new
                    {
                        Message = $"User with ID '{userId}' has been successfully deleted.",
                        DeletedBy = adminName,
                        Timestamp = DateTime.UtcNow
                    });
                }
                else
                {
                    _logger.LogError(
                        "Failed to delete user with ID '{UserId}'. Errors: {Errors}",
                        userId, string.Join(", ", deletionResult.Errors));

                    return StatusCode(500, new
                    {
                        Message = "An error occurred while attempting to delete the user.",
                        Errors = deletionResult.Errors
                    });
                }
            }
            catch (UnauthorizedAccessException ex)
            {
                _logger.LogWarning(ex, "Unauthorized access attempt to delete user with ID: {UserId}.", userId);
                return Forbid();
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "An unexpected error occurred while deleting user with ID: {UserId}.", userId);
                return StatusCode(500, new { Message = "An unexpected error occurred. Please try again later." });
            }
        }
    }
