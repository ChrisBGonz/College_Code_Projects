name: Build and Upload Artifact

on:
  push:
    branches:
      - main
      - master
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:

env:
  ARTIFACT_NAME: build-artifact
  BUILD_DIR: dist
  BUCKET_NAME: ${{ secrets.STORAGE_BUCKET_NAME }}

jobs:
  build-and-upload:
    name: Build Project and Upload to Storage
    runs-on: ubuntu-latest

    permissions:
      contents: read
      id-token: write

    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install Dependencies
        run: npm ci

      - name: Run Tests
        run: npm test
        env:
          CI: true

      - name: Build Project
        run: npm run build
        env:
          NODE_ENV: production

      - name: Generate Build Metadata
        id: metadata
        run: |
          BUILD_TIMESTAMP=$(date -u +"%Y%m%d_%H%M%S")
          SHORT_SHA=$(git rev-parse --short HEAD)
          ARTIFACT_VERSION="${BUILD_TIMESTAMP}_${SHORT_SHA}"
          echo "build_timestamp=${BUILD_TIMESTAMP}" >> $GITHUB_OUTPUT
          echo "short_sha=${SHORT_SHA}" >> $GITHUB_OUTPUT
          echo "artifact_version=${ARTIFACT_VERSION}" >> $GITHUB_OUTPUT
          echo "artifact_filename=${ARTIFACT_NAME}_${ARTIFACT_VERSION}.tar.gz" >> $GITHUB_OUTPUT

      - name: Create Artifact Archive
        run: |
          tar -czf "${{ steps.metadata.outputs.artifact_filename }}" \
            -C . \
            ${{ env.BUILD_DIR }}
          echo "Created archive: ${{ steps.metadata.outputs.artifact_filename }}"
          ls -lh "${{ steps.metadata.outputs.artifact_filename }}"

      - name: Generate Checksum
        id: checksum
        run: |
          CHECKSUM=$(sha256sum "${{ steps.metadata.outputs.artifact_filename }}" | awk '{print $1}')
          echo "sha256=${CHECKSUM}" >> $GITHUB_OUTPUT
          echo "$CHECKSUM  ${{ steps.metadata.outputs.artifact_filename }}" > "${{ steps.metadata.outputs.artifact_filename }}.sha256"
          echo "SHA256: ${CHECKSUM}"

      - name: Upload Artifact to GitHub (temporary storage)
        uses: actions/upload-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}-${{ steps.metadata.outputs.artifact_version }}
          path: |
            ${{ steps.metadata.outputs.artifact_filename }}
            ${{ steps.metadata.outputs.artifact_filename }}.sha256
          retention-days: 7
          if-no-files-found: error

      # -------------------------------------------------------
      # Option A: Upload to Google Cloud Storage (GCS)
      # Uncomment this section if using GCS
      # -------------------------------------------------------
      - name: Authenticate to Google Cloud
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SERVICE_ACCOUNT_KEY }}
          # Alternatively, use Workload Identity Federation:
          # workload_identity_provider: ${{ secrets.GCP_WORKLOAD_IDENTITY_PROVIDER }}
          # service_account: ${{ secrets.GCP_SERVICE_ACCOUNT }}

      - name: Upload to Google Cloud Storage
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: ${{ steps.metadata.outputs.artifact_filename }}
          destination: ${{ env.BUCKET_NAME }}/artifacts/${{ steps.metadata.outputs.artifact_version }}/
          gzip: false
          resumable: true

      - name: Upload Checksum to Google Cloud Storage
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: ${{ steps.metadata.outputs.artifact_filename }}.sha256
          destination: ${{ env.BUCKET_NAME }}/artifacts/${{ steps.metadata.outputs.artifact_version }}/
          gzip: false

      # -------------------------------------------------------
      # Option B: Upload to AWS S3
      # Uncomment this section if using AWS S3
      # -------------------------------------------------------
      # - name: Configure AWS Credentials
      #   if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
      #   uses: aws-actions/configure-aws-credentials@v4
      #   with:
      #     aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
      #     aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
      #     aws-region: ${{ secrets.AWS_REGION }}
      #     # Alternatively, use OIDC:
      #     # role-to-assume: ${{ secrets.AWS_ROLE_ARN }}

      # - name: Upload to AWS S3
      #   if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
      #   run: |
      #     aws s3 cp \
      #       "${{ steps.metadata.outputs.artifact_filename }}" \
      #       "s3://${{ env.BUCKET_NAME }}/artifacts/${{ steps.metadata.outputs.artifact_version }}/${{ steps.metadata.outputs.artifact_filename }}" \
      #       --storage-class STANDARD_IA \
      #       --metadata "git-sha=${{ github.sha }},build-number=${{ github.run_number }}"
      #
      #     aws s3 cp \
      #       "${{ steps.metadata.outputs.artifact_filename }}.sha256" \
      #       "s3://${{ env.BUCKET_NAME }}/artifacts/${{ steps.metadata.outputs.artifact_version }}/${{ steps.metadata.outputs.artifact_filename }}.sha256"

      # -------------------------------------------------------
      # Option C: Upload to Azure Blob Storage
      # Uncomment this section if using Azure
      # -------------------------------------------------------
      # - name: Login to Azure
      #   if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
      #   uses: azure/login@v2
      #   with:
      #     creds: ${{ secrets.AZURE_CREDENTIALS }}

      # - name: Upload to Azure Blob Storage
      #   if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
      #   uses: azure/CLI@v2
      #   with:
      #     inlineScript: |
      #       az storage blob upload \
      #         --account-name ${{ secrets.AZURE_STORAGE_ACCOUNT }} \
      #         --container-name ${{ env.BUCKET_NAME }} \
      #         --name "artifacts/${{ steps.metadata.outputs.artifact_version }}/${{ steps.metadata.outputs.artifact_filename }}" \
      #         --file "${{ steps.metadata.outputs.artifact_filename }}" \
      #         --auth-mode login

      - name: Update Latest Pointer
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        run: |
          echo "${{ steps.metadata.outputs.artifact_version }}" > latest.txt
          echo "artifact_filename=${{ steps.metadata.outputs.artifact_filename }}" >> latest.txt
          echo "sha256=${{ steps.checksum.outputs.sha256 }}" >> latest.txt
          echo "git_sha=${{ github.sha }}" >> latest.txt
          echo "build_number=${{ github.run_number }}" >> latest.txt

      - name: Upload Latest Pointer to GCS
        if: github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master'
        uses: google-github-actions/upload-cloud-storage@v2
        with:
          path: latest.txt
          destination: ${{ env.BUCKET_NAME }}/artifacts/
          gzip: false

      - name: Print Upload Summary
        if: always()
        run: |
          echo "=========================================="
          echo "Build and Upload Summary"
          echo "=========================================="
          echo "Repository:       ${{ github.repository }}"
          echo "Branch/Ref:       ${{ github.ref_name }}"
          echo "Commit SHA:       ${{ github.sha }}"
          echo "Short SHA:        ${{ steps.metadata.outputs.short_sha }}"
          echo "Build Number:     ${{ github.run_number }}"
          echo "Artifact Version: ${{ steps.metadata.outputs.artifact_version }}"
          echo "Artifact File:    ${{ steps.metadata.outputs.artifact_filename }}"
          echo "SHA256 Checksum:  ${{ steps.checksum.outputs.sha256 }}"
          echo "Bucket:           ${{ env.BUCKET_NAME }}"
          echo "Triggered by:     ${{ github.actor }}"
          echo "=========================================="

  notify-on-failure:
    name: Notify on Failure
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: failure()

    steps:
      - name: Send Failure Notification
        run: |
          echo "Build or upload failed!"
          echo "Workflow: ${{ github.workflow }}"
          echo "Run ID: ${{ github.run_id }}"
          echo "Repository: ${{ github.repository }}"
          echo "Branch: ${{ github.ref_name }}"
          echo "Triggered by: ${{ github.actor }}"
          # Add your notification logic here (Slack, email, PagerDuty, etc.)
          # Example with curl to a webhook:
          # curl -X POST -H 'Content-type: application/json' \
          #   --data '{"text":"Build failed for ${{ github.repository }} on branch ${{ github.ref_name }}"}' \
          #   ${{ secrets.SLACK_WEBHOOK_URL }}
