using System;
using System.Collections.Generic;
using Microsoft.Extensions.Logging;

namespace MfaAuditLogging
{
    public enum MfaAction
    {
        Enabled,
        Disabled,
        MethodAdded,
        MethodRemoved,
        MethodUpdated,
        RecoveryCodesGenerated,
        PrimaryMethodChanged
    }

    public enum MfaMethod
    {
        Authenticator,
        SMS,
        Email,
        HardwareToken,
        BackupCodes
    }

    public class MfaChangeEvent
    {
        public string EventId { get; set; } = Guid.NewGuid().ToString();
        public string UserId { get; set; }
        public string Username { get; set; }
        public MfaAction Action { get; set; }
        public MfaMethod? AffectedMethod { get; set; }
        public MfaMethod? PreviousPrimaryMethod { get; set; }
        public MfaMethod? NewPrimaryMethod { get; set; }
        public DateTime Timestamp { get; set; } = DateTime.UtcNow;
        public string IpAddress { get; set; }
        public string UserAgent { get; set; }
        public string SessionId { get; set; }
        public bool Success { get; set; }
        public string FailureReason { get; set; }
        public Dictionary<string, string> AdditionalData { get; set; } = new();
    }

    public interface IMfaAuditLogger
    {
        void LogMfaChange(MfaChangeEvent changeEvent);
        void LogMfaEnabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null);
        void LogMfaDisabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null);
        void LogMfaMethodAdded(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null);
        void LogMfaMethodRemoved(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null);
        void LogMfaPrimaryMethodChanged(string userId, string username, MfaMethod previousMethod, MfaMethod newMethod, string ipAddress, string sessionId, bool success, string failureReason = null);
        void LogRecoveryCodesGenerated(string userId, string username, string ipAddress, string sessionId, bool success, string failureReason = null);
    }

    public class MfaAuditLogger : IMfaAuditLogger
    {
        private readonly ILogger<MfaAuditLogger> _logger;

        public MfaAuditLogger(ILogger<MfaAuditLogger> logger)
        {
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
        }

        public void LogMfaChange(MfaChangeEvent changeEvent)
        {
            if (changeEvent == null)
                throw new ArgumentNullException(nameof(changeEvent));

            var logLevel = changeEvent.Success ? LogLevel.Information : LogLevel.Warning;

            using (_logger.BeginScope(new Dictionary<string, object>
            {
                ["EventId"] = changeEvent.EventId,
                ["UserId"] = changeEvent.UserId,
                ["Username"] = changeEvent.Username,
                ["Action"] = changeEvent.Action.ToString(),
                ["AffectedMethod"] = changeEvent.AffectedMethod?.ToString() ?? "N/A",
                ["IpAddress"] = changeEvent.IpAddress,
                ["SessionId"] = changeEvent.SessionId,
                ["Timestamp"] = changeEvent.Timestamp,
                ["Success"] = changeEvent.Success
            }))
            {
                if (changeEvent.Success)
                {
                    _logger.Log(logLevel,
                        "MFA_SETTINGS_CHANGED | EventId: {EventId} | User: {Username} (ID: {UserId}) | " +
                        "Action: {Action} | Method: {AffectedMethod} | IP: {IpAddress} | " +
                        "Session: {SessionId} | Timestamp: {Timestamp} | Status: SUCCESS",
                        changeEvent.EventId,
                        changeEvent.Username,
                        changeEvent.UserId,
                        changeEvent.Action,
                        changeEvent.AffectedMethod?.ToString() ?? "N/A",
                        changeEvent.IpAddress,
                        changeEvent.SessionId,
                        changeEvent.Timestamp);
                }
                else
                {
                    _logger.Log(logLevel,
                        "MFA_SETTINGS_CHANGE_FAILED | EventId: {EventId} | User: {Username} (ID: {UserId}) | " +
                        "Action: {Action} | Method: {AffectedMethod} | IP: {IpAddress} | " +
                        "Session: {SessionId} | Timestamp: {Timestamp} | Status: FAILED | Reason: {FailureReason}",
                        changeEvent.EventId,
                        changeEvent.Username,
                        changeEvent.UserId,
                        changeEvent.Action,
                        changeEvent.AffectedMethod?.ToString() ?? "N/A",
                        changeEvent.IpAddress,
                        changeEvent.SessionId,
                        changeEvent.Timestamp,
                        changeEvent.FailureReason ?? "Unknown");
                }

                if (changeEvent.Action == MfaAction.PrimaryMethodChanged)
                {
                    _logger.Log(logLevel,
                        "MFA_PRIMARY_METHOD_CHANGED | EventId: {EventId} | User: {Username} (ID: {UserId}) | " +
                        "Previous: {PreviousMethod} | New: {NewMethod} | IP: {IpAddress} | Timestamp: {Timestamp}",
                        changeEvent.EventId,
                        changeEvent.Username,
                        changeEvent.UserId,
                        changeEvent.PreviousPrimaryMethod?.ToString() ?? "None",
                        changeEvent.NewPrimaryMethod?.ToString() ?? "None",
                        changeEvent.IpAddress,
                        changeEvent.Timestamp);
                }

                if (changeEvent.AdditionalData?.Count > 0)
                {
                    _logger.LogDebug("MFA_CHANGE_ADDITIONAL_DATA | EventId: {EventId} | Data: {@AdditionalData}",
                        changeEvent.EventId,
                        changeEvent.AdditionalData);
                }
            }
        }

        public void LogMfaEnabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.Enabled,
                AffectedMethod = method,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });
        }

        public void LogMfaDisabled(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.Disabled,
                AffectedMethod = method,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });

            if (success)
            {
                _logger.LogWarning(
                    "MFA_SECURITY_ALERT | User: {Username} (ID: {UserId}) has DISABLED MFA method: {Method} | " +
                    "IP: {IpAddress} | Timestamp: {Timestamp}",
                    username, userId, method, ipAddress, DateTime.UtcNow);
            }
        }

        public void LogMfaMethodAdded(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.MethodAdded,
                AffectedMethod = method,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });
        }

        public void LogMfaMethodRemoved(string userId, string username, MfaMethod method, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.MethodRemoved,
                AffectedMethod = method,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });
        }

        public void LogMfaPrimaryMethodChanged(string userId, string username, MfaMethod previousMethod, MfaMethod newMethod, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.PrimaryMethodChanged,
                PreviousPrimaryMethod = previousMethod,
                NewPrimaryMethod = newMethod,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });
        }

        public void LogRecoveryCodesGenerated(string userId, string username, string ipAddress, string sessionId, bool success, string failureReason = null)
        {
            LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.RecoveryCodesGenerated,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = success,
                FailureReason = failureReason
            });
        }
    }

    // Example usage
    public class MfaSettingsService
    {
        private readonly IMfaAuditLogger _auditLogger;

        public MfaSettingsService(IMfaAuditLogger auditLogger)
        {
            _auditLogger = auditLogger ?? throw new ArgumentNullException(nameof(auditLogger));
        }

        public bool EnableMfa(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            bool success = false;
            string failureReason = null;

            try
            {
                // Simulate enabling MFA
                Console.WriteLine($"Enabling MFA method {method} for user {username}...");
                success = true;
            }
            catch (Exception ex)
            {
                failureReason = ex.Message;
            }
            finally
            {
                _auditLogger.LogMfaEnabled(userId, username, method, ipAddress, sessionId, success, failureReason);
            }

            return success;
        }

        public bool DisableMfa(string userId, string username, MfaMethod method, string ipAddress, string sessionId)
        {
            bool success = false;
            string failureReason = null;

            try
            {
                // Simulate disabling MFA
                Console.WriteLine($"Disabling MFA method {method} for user {username}...");
                success = true;
            }
            catch (Exception ex)
            {
                failureReason = ex.Message;
            }
            finally
            {
                _auditLogger.LogMfaDisabled(userId, username, method, ipAddress, sessionId, success, failureReason);
            }

            return success;
        }

        public bool ChangePrimaryMethod(string userId, string username, MfaMethod previousMethod, MfaMethod newMethod, string ipAddress, string sessionId)
        {
            bool success = false;
            string failureReason = null;

            try
            {
                // Simulate changing primary MFA method
                Console.WriteLine($"Changing primary MFA method from {previousMethod} to {newMethod} for user {username}...");
                success = true;
            }
            catch (Exception ex)
            {
                failureReason = ex.Message;
            }
            finally
            {
                _auditLogger.LogMfaPrimaryMethodChanged(userId, username, previousMethod, newMethod, ipAddress, sessionId, success, failureReason);
            }

            return success;
        }

        public bool GenerateRecoveryCodes(string userId, string username, string ipAddress, string sessionId)
        {
            bool success = false;
            string failureReason = null;

            try
            {
                // Simulate generating recovery codes
                Console.WriteLine($"Generating recovery codes for user {username}...");
                success = true;
            }
            catch (Exception ex)
            {
                failureReason = ex.Message;
            }
            finally
            {
                _auditLogger.LogRecoveryCodesGenerated(userId, username, ipAddress, sessionId, success, failureReason);
            }

            return success;
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            using var loggerFactory = LoggerFactory.Create(builder =>
            {
                builder
                    .AddConsole()
                    .SetMinimumLevel(LogLevel.Debug);
            });

            var logger = loggerFactory.CreateLogger<MfaAuditLogger>();
            var auditLogger = new MfaAuditLogger(logger);
            var mfaService = new MfaSettingsService(auditLogger);

            string userId = "user-12345";
            string username = "john.doe@example.com";
            string ipAddress = "192.168.1.100";
            string sessionId = "session-abc-xyz";

            Console.WriteLine("=== MFA Audit Logging Demo ===\n");

            // Enable MFA
            mfaService.EnableMfa(userId, username, MfaMethod.Authenticator, ipAddress, sessionId);

            // Add another method
            auditLogger.LogMfaMethodAdded(userId, username, MfaMethod.SMS, ipAddress, sessionId, true);

            // Change primary method
            mfaService.ChangePrimaryMethod(userId, username, MfaMethod.SMS, MfaMethod.Authenticator, ipAddress, sessionId);

            // Generate recovery codes
            mfaService.GenerateRecoveryCodes(userId, username, ipAddress, sessionId);

            // Remove a method
            auditLogger.LogMfaMethodRemoved(userId, username, MfaMethod.SMS, ipAddress, sessionId, true);

            // Disable MFA (triggers security alert)
            mfaService.DisableMfa(userId, username, MfaMethod.Authenticator, ipAddress, sessionId);

            // Log a failed attempt
            auditLogger.LogMfaEnabled(userId, username, MfaMethod.HardwareToken, ipAddress, sessionId, false, "Hardware token validation failed");

            // Log a custom event with additional data
            auditLogger.LogMfaChange(new MfaChangeEvent
            {
                UserId = userId,
                Username = username,
                Action = MfaAction.MethodUpdated,
                AffectedMethod = MfaMethod.SMS,
                IpAddress = ipAddress,
                SessionId = sessionId,
                Success = true,
                AdditionalData = new Dictionary<string, string>
                {
                    ["OldPhoneNumber"] = "+1-XXX-XXX-1234",
                    ["NewPhoneNumber"] = "+1-XXX-XXX-5678"
                }
            });

            Console.WriteLine("\n=== Demo Complete ===");
        }
    }
}
