import os
import anthropic
import requests
from flask import Flask, Response, request, stream_with_context

app = Flask(__name__)
client = anthropic.Anthropic()


def fetch_url_content(url: str) -> tuple[str | None, str | None]:
    """Fetch content from a URL and return (content, error)."""
    try:
        if not url.startswith(("http://", "https://")):
            url = "https://" + url

        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        }
        response = requests.get(url, headers=headers, timeout=10)

        if response.status_code == 404:
            return None, f"404 Not Found: The page at {url} could not be found."
        elif response.status_code != 200:
            return (
                None,
                f"Error {response.status_code}: Failed to fetch content from {url}",
            )

        return response.text[:5000], None

    except requests.exceptions.ConnectionError:
        return None, f"Connection Error: Could not connect to {url}"
    except requests.exceptions.Timeout:
        return None, f"Timeout Error: Request to {url} timed out"
    except requests.exceptions.RequestException as e:
        return None, f"Request Error: {str(e)}"


@app.route("/", methods=["GET"])
def index():
    """Main page with URL input form."""
    return """
    <!DOCTYPE html>
    <html>
    <head>
        <title>URL Content Fetcher with AI Analysis</title>
        <style>
            body { font-family: Arial, sans-serif; max-width: 800px; margin: 50px auto; padding: 20px; }
            input[type="text"] { width: 70%; padding: 10px; font-size: 16px; }
            button { padding: 10px 20px; font-size: 16px; background-color: #007bff; color: white; border: none; cursor: pointer; }
            button:hover { background-color: #0056b3; }
            #result { margin-top: 20px; padding: 20px; border: 1px solid #ddd; min-height: 100px; white-space: pre-wrap; }
            .error { color: red; }
        </style>
    </head>
    <body>
        <h1>URL Content Fetcher with AI Analysis</h1>
        <form id="urlForm">
            <input type="text" id="urlInput" placeholder="Enter URL (e.g., example.com)" required>
            <button type="submit">Fetch & Analyze</button>
        </form>
        <div id="result">Results will appear here...</div>
        
        <script>
            document.getElementById('urlForm').addEventListener('submit', async function(e) {
                e.preventDefault();
                const url = document.getElementById('urlInput').value;
                const resultDiv = document.getElementById('result');
                resultDiv.textContent = 'Fetching and analyzing content...';
                resultDiv.className = '';
                
                try {
                    const response = await fetch('/fetch', {
                        method: 'POST',
                        headers: {'Content-Type': 'application/x-www-form-urlencoded'},
                        body: 'url=' + encodeURIComponent(url)
                    });
                    
                    const reader = response.body.getReader();
                    const decoder = new TextDecoder();
                    resultDiv.textContent = '';
                    
                    while (true) {
                        const {done, value} = await reader.read();
                        if (done) break;
                        resultDiv.textContent += decoder.decode(value);
                    }
                } catch (error) {
                    resultDiv.textContent = 'Error: ' + error.message;
                    resultDiv.className = 'error';
                }
            });
        </script>
    </body>
    </html>
    """


@app.route("/fetch", methods=["POST"])
def fetch_and_analyze():
    """Fetch URL content and analyze it with Claude using streaming."""
    url = request.form.get("url", "").strip()

    if not url:
        return Response("Error: No URL provided", status=400)

    content, error = fetch_url_content(url)

    if error:
        return Response(f"Error: {error}", status=404 if "404" in error else 500)

    def generate():
        yield f"Successfully fetched content from: {url}\n"
        yield f"Content length: {len(content)} characters\n\n"
        yield "AI Analysis:\n"
        yield "-" * 50 + "\n"

        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"Please analyze the following web page content and provide a brief summary of what the page is about, its main topics, and any notable features:\n\n{content}",
                }
            ],
        ) as stream:
            for text in stream.text_stream:
                yield text

    return Response(stream_with_context(generate()), mimetype="text/plain")


if __name__ == "__main__":
    app.run(debug=True, port=5000)
