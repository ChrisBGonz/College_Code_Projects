import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: {
    user_001: {
      id: "user_001",
      username: "john_doe",
      email: "john@example.com",
      password_hash: "old_hash_123",
      last_password_reset: null,
    },
    user_002: {
      id: "user_002",
      username: "jane_smith",
      email: "jane@example.com",
      password_hash: "old_hash_456",
      last_password_reset: null,
    },
  },
};

// Tool definitions for password reset flow
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database and returns user information",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (min 8 chars, uppercase, lowercase, number, special char)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes the password securely before storing in database",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_database",
    description: "Updates the user's password hash in the database",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
  {
    name: "send_confirmation_email",
    description:
      "Sends a password reset confirmation email to the user's registered email address",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send confirmation to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool implementation functions
function validateUser(userId) {
  const user = database.users[userId];
  if (user) {
    return {
      success: true,
      user: {
        id: user.id,
        username: user.username,
        email: user.email,
        exists: true,
      },
    };
  }
  return {
    success: false,
    error: `User with ID ${userId} not found`,
    exists: false,
  };
}

function validatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /[0-9]/.test(password);
  const hasSpecialChar = /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(
    password
  );

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  return {
    isValid,
    requirements: {
      minLength: { met: minLength, requirement: "At least 8 characters" },
      hasUppercase: {
        met: hasUppercase,
        requirement: "At least one uppercase letter",
      },
      hasLowercase: {
        met: hasLowercase,
        requirement: "At least one lowercase letter",
      },
      hasNumber: { met: hasNumber, requirement: "At least one number" },
      hasSpecialChar: {
        met: hasSpecialChar,
        requirement: "At least one special character",
      },
    },
  };
}

function hashPassword(password) {
  // Simulated password hashing (in real implementation, use bcrypt or similar)
  const hash = `hashed_${Buffer.from(password).toString("base64")}_${Date.now()}`;
  return {
    success: true,
    hash: hash,
    algorithm: "simulated_bcrypt",
  };
}

function updatePasswordInDatabase(userId, passwordHash) {
  if (database.users[userId]) {
    database.users[userId].password_hash = passwordHash;
    database.users[userId].last_password_reset = new Date().toISOString();
    return {
      success: true,
      message: `Password successfully updated for user ${userId}`,
      updated_at: database.users[userId].last_password_reset,
    };
  }
  return {
    success: false,
    error: `Failed to update password for user ${userId}`,
  };
}

function sendConfirmationEmail(userId, email) {
  // Simulated email sending
  console.log(
    `[EMAIL SERVICE] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
    email_id: `email_${Date.now()}`,
  };
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_user":
      return validateUser(toolInput.user_id);
    case "validate_password_strength":
      return validatePasswordStrength(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "update_password_in_database":
      return updatePasswordInDatabase(toolInput.user_id, toolInput.password_hash);
    case "send_confirmation_email":
      return sendConfirmationEmail(toolInput.user_id, toolInput.email);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Starting password reset for user: ${userId}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user_id: "${userId}" with the new password: "${newPassword}". 
      Follow these steps:
      1. Validate that the user exists
      2. Validate the password strength
      3. Hash the password securely
      4. Update the password in the database
      5. Send a confirmation email to the user
      
      Report the result of each step and provide a final summary.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("Initial Response:");
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool Used: ${toolUse.name}`);
      console.log("Tool Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Tool Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued Response Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(60));
  console.log("FINAL RESPONSE:");
  console.log("=".repeat(60));
  console.log(finalResponse);

  return {
    success: true,
    message: finalResponse,
    database_state: database.users[userId],
  };
}

// Test the password reset function
async function main() {
  try {
    // Test 1: Valid user with strong password
    console.log("\n" + "=".repeat(60));
    console.log("TEST 1: Valid user with strong password");
    console.log("=".repeat(60));
    await resetPassword("user_001", "SecurePass123!");

    // Test 2: Valid user with weak password
    console.log("\n" + "=".repeat(60));
    console.log("TEST 2: Valid user with weak password");
    console.log("=".repeat(60));
    await resetPassword("user_002", "weak");

    // Test 3: Invalid user
    console.log("\n" + "=".repeat(60));
    console.log("TEST 3: Invalid user");
    console.log("=".repeat(60));
    await resetPassword("user_999", "StrongPass123!");
  } catch (error) {
    console.error("Error:", error);
    throw error;
  }

main();
