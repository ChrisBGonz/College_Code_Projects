function loadScript(url, callback) {
  const script = document.createElement('script');
  script.type = 'text/javascript';
  script.src = url;

  script.onload = function () {
    console.log('Script loaded successfully:', url);
    if (typeof callback === 'function') {
      callback(null, script);
    }
  };

  script.onerror = function () {
    const error = new Error('Failed to load script: ' + url);
    console.error(error.message);
    if (typeof callback === 'function') {
      callback(error, null);
    }
  };

  document.head.appendChild(script);
}

const JQUERY_CDN_URL =
  'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';

loadScript(JQUERY_CDN_URL, function (error, script) {
  if (error) {
    console.error('Error loading jQuery:', error.message);
    return;
  }

  if (typeof jQuery !== 'undefined') {
    console.log('jQuery version:', jQuery.fn.jquery);

    jQuery(document).ready(function () {
      console.log('jQuery is ready to use!');

      jQuery('body').css('background-color', '#f0f0f0');

      const $heading = jQuery('<h1>')
        .text('jQuery Loaded Successfully!')
        .css({
          color: '#333',
          'font-family': 'Arial, sans-serif',
          'text-align': 'center',
          'margin-top': '50px',
        });

      jQuery('body').append($heading);

      jQuery.ajax({
        url: 'https://jsonplaceholder.typicode.com/todos/1',
        method: 'GET',
        success: function (data) {
          console.log('AJAX request successful:', data);

          const $result = jQuery('<p>')
            .text('Fetched Todo: ' + data.title)
            .css({
              color: '#555',
              'font-family': 'Arial, sans-serif',
              'text-align': 'center',
            });

          jQuery('body').append($result);
        },
        error: function (xhr, status, error) {
          console.error('AJAX request failed:', error);
        },
      });
    });
  } else {
    console.error('jQuery is not available after loading.');
  }
});
