
```python
import os
import stripe
from flask import Flask, request, jsonify, redirect
from dotenv import load_dotenv

load_dotenv()

app = Flask(__name__)

stripe.api_key = os.getenv("STRIPE_SECRET_KEY")

@app.route("/create-checkout-session", methods=["POST"])
def create_checkout_session():
    """Create a Stripe checkout session and redirect to payment page."""
    data = request.get_json()
    
    if not data:
        return jsonify({"error": "No data provided"}), 400
    
    success_url = data.get("success_url")
    cancel_url = data.get("cancel_url", "http://localhost:5000/cancel")
    amount = data.get("amount", 1000)
    currency = data.get("currency", "usd")
    product_name = data.get("product_name", "Test Product")
    
    if not success_url:
        return jsonify({"error": "success_url is required"}), 400
    
    try:
        checkout_session = stripe.checkout.Session.create(
            payment_method_types=["card"],
            line_items=[
                {
                    "price_data": {
                        "currency": currency,
                        "product_data": {
                            "name": product_name,
                        },
                        "unit_amount": amount,
                    },
                    "quantity": 1,
                }
            ],
            mode="payment",
            success_url=success_url,
            cancel_url=cancel_url,
        )
        
        return jsonify({
            "checkout_url": checkout_session.url,
            "session_id": checkout_session.id
        })
        
    except stripe.error.StripeError as e:
        return jsonify({"error": str(e)}), 400


@app.route("/redirect-to-payment", methods=["POST"])
def redirect_to_payment():
    """Create checkout session and redirect user to Stripe payment page."""
    data = request.get_json()
    
    if not data:
        return jsonify({"error": "No data provided"}), 400
    
    success_url = data.get("success_url")
    cancel_url = data.get("cancel_url", "http://localhost:5000/cancel")
    amount = data.get("amount", 1000)
    currency = data.get("currency", "usd")
    product_name = data.get("product_name", "Test Product")
    
    if not success_url:
        return jsonify({"error": "success_url is required"}), 400
    
    try:
        checkout_session = stripe.checkout.Session.create(
            payment_method_types=["card"],
            line_items=[
                {
                    "price_data": {
                        "currency": currency,
                        "product_data": {
                            "name": product_name,
                        },
                        "unit_amount": amount,
                    },
                    "quantity": 1,
                }
            ],
            mode="payment",
            success_url=success_url,
            cancel_url=cancel_url,
        )
        
        return redirect(checkout_session.url, code=303)
        
    except stripe.error.StripeError as e:
        return jsonify({"error": str(e)}), 400


@app.route("/webhook", methods=["POST"])
def webhook():
    """Handle Stripe webhook events."""
    payload = request.get_data()
    sig_header = request.headers.get("Stripe-Signature")
    webhook_secret = os.getenv("STRIPE_WEBHOOK_SECRET")
    
    if not webhook_secret:
        return jsonify({"error": "Webhook secret not configured"}), 500
    
    try:
        event = stripe.Webhook.construct_event(
            payload, sig_header, webhook_secret
        )
    except ValueError:
        return jsonify({"error": "Invalid payload"}), 400
    except stripe.error.SignatureVerificationError:
        return jsonify({"error": "Invalid signature"}), 400
    
    if event["type"] == "checkout.session.completed":
        session = event["data"]["object"]
        print(f"Payment successful! Session ID: {session['id']}")
        print(f"Customer email: {session.get('customer_details', {}).get('email', 'N/A')}")
        print(f"Amount paid: {session['amount_total']} {session['currency']}")
    
    elif event["type"] == "payment_intent.payment_failed":
        payment_intent = event["data"]["object"]
        print(f"Payment failed! PaymentIntent ID: {payment_intent['id']}")
    
    return jsonify({"status": "success"})


@app.route("/success", methods=["GET"])
def success():
    """Handle successful payment redirect."""
    session_id = request.args.get("session_id", "")
    return jsonify({
        "status": "success",
        "message": "Payment completed successfully!",
        "session_id": session_id
    })


@app.route("/cancel", methods=["GET"])
def cancel():
    """Handle cancelled payment redirect."""
    return jsonify({
        "status": "cancelled",
        "message": "Payment was cancelled."
    })


@app.route("/health", methods=["GET"])
def health():
    """Health check endpoint."""
    return jsonify({"status": "healthy", "service": "payment-gateway"})


if __name__ == "__main__":
    print("Starting Payment Gateway Service...")
    print("Available endpoints:")
    print("  POST /create-checkout-session - Create a checkout session")
    print("  POST /redirect-to-payment - Create session and redirect to payment")
    print("  POST /webhook - Handle Stripe webhooks")
    print("  GET /success - Success page")
    print("  GET /cancel - Cancel page")
    print("  GET /health - Health check")
    
    app.run(debug=True, port=5000)
```
