import anthropic
import base64
import hashlib
import hmac
import struct
import time


def generate_totp(secret: str, time_step: int = 30, digits: int = 6) -> str:
    """Generate a TOTP code for testing purposes."""
    # Decode the base32 secret
    key = base64.b32decode(secret.upper() + "=" * ((8 - len(secret) % 8) % 8))

    # Get current time step
    current_time = int(time.time()) // time_step

    # Pack time as big-endian 8-byte integer
    time_bytes = struct.pack(">Q", current_time)

    # Generate HMAC-SHA1
    hmac_hash = hmac.new(key, time_bytes, hashlib.sha1).digest()

    # Dynamic truncation
    offset = hmac_hash[-1] & 0x0F
    code = struct.unpack(">I", hmac_hash[offset : offset + 4])[0]
    code &= 0x7FFFFFFF
    code %= 10**digits

    return str(code).zfill(digits)


def verify_totp_with_claude(secret: str, provided_code: str) -> dict:
    """Use Claude to verify a TOTP code with explanation."""
    client = anthropic.Anthropic()

    # Generate the actual valid code for comparison
    valid_code = generate_totp(secret)
    current_time = int(time.time())
    time_step = 30
    time_remaining = time_step - (current_time % time_step)

    # Create a prompt for Claude to analyze the TOTP verification
    prompt = f"""You are a security expert helping to verify a Time-based One-Time Password (TOTP) for 2FA authentication.

Here are the details:
- The user provided code: {provided_code}
- The valid TOTP code at this moment: {valid_code}
- Current timestamp: {current_time}
- Time remaining for current code: {time_remaining} seconds
- TOTP time step: 30 seconds

Please analyze this TOTP verification attempt and provide:
1. Whether the provided code matches the valid code
2. The security status (VALID/INVALID)
3. Any security considerations or warnings
4. Recommendations for the user

Format your response as a structured security report."""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
    )

    # Extract the response
    response_text = message.content[0].text

    # Determine if verification passed
    is_valid = provided_code == valid_code

    return {
        "is_valid": is_valid,
        "provided_code": provided_code,
        "valid_code": valid_code,
        "time_remaining": time_remaining,
        "claude_analysis": response_text,
    }


def main():
    # Example secret key (in real usage, this would be stored securely)
    # This is a base32 encoded secret
    secret = "JBSWY3DPEHPK3PXP"

    print("=" * 60)
    print("TOTP 2FA Verification System")
    print("=" * 60)
    print()

    # Generate the current valid code for demonstration
    current_valid_code = generate_totp(secret)
    print(f"[Demo] Current valid TOTP code: {current_valid_code}")
    print(
        f"[Demo] Time remaining: {30 - (int(time.time()) % 30)} seconds"
    )
    print()

    # Test Case 1: Valid code
    print("Test Case 1: Verifying with CORRECT code")
    print("-" * 40)
    result = verify_totp_with_claude(secret, current_valid_code)
    print(f"Verification Result: {'✓ PASSED' if result['is_valid'] else '✗ FAILED'}")
    print(f"Provided Code: {result['provided_code']}")
    print(f"Time Remaining: {result['time_remaining']} seconds")
    print()
    print("Claude's Security Analysis:")
    print(result["claude_analysis"])
    print()

    # Test Case 2: Invalid code
    print("=" * 60)
    print("Test Case 2: Verifying with INCORRECT code")
    print("-" * 40)
    wrong_code = "123456"
    result = verify_totp_with_claude(secret, wrong_code)
    print(f"Verification Result: {'✓ PASSED' if result['is_valid'] else '✗ FAILED'}")
    print(f"Provided Code: {result['provided_code']}")
    print(f"Time Remaining: {result['time_remaining']} seconds")
    print()
    print("Claude's Security Analysis:")
    print(result["claude_analysis"])
    print()

    # Interactive mode
    print("=" * 60)
    print("Interactive TOTP Verification")
    print("-" * 40)
    print(f"Secret key for testing: {secret}")
    print("(In production, users would have this in their authenticator app)")
    print()

    user_code = input("Enter your 6-digit TOTP code (or press Enter to skip): ").strip()

    if user_code:
        if len(user_code) == 6 and user_code.isdigit():
            result = verify_totp_with_claude(secret, user_code)
            print()
            print(
                f"Verification Result: {'✓ ACCESS GRANTED' if result['is_valid'] else '✗ ACCESS DENIED'}"
            )
            print()
            print("Detailed Analysis from Claude:")
            print(result["claude_analysis"])
        else:
            print("Invalid input: Please enter exactly 6 digits")
    else:
        print("Skipping interactive verification")

    print()
    print("=" * 60)
    print("TOTP Verification Complete")


if __name__ == "__main__":
    main()
