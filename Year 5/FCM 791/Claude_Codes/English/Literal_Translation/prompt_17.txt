import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateMongoDBConnectionString() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a complete Node.js script that sets up a MongoDB connection string for a local development database. Include: 1) The connection string configuration with localhost and default MongoDB port 27017, 2) A database name for development, 3) Connection options for local development (like timeouts, retry settings), 4) Error handling for the connection, 5) A simple test to verify the connection works. Use the mongodb npm package and provide production-ready code with comments explaining each part.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Claude's MongoDB Connection String Setup:");
  console.log("==========================================");
  console.log(responseText);

  const codeMatch = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/);
  if (codeMatch) {
    console.log("\n\nExtracted Code:");
    console.log("===============");
    console.log(codeMatch[1]);
  }

  return responseText;
}

generateMongoDBConnectionString().catch(console.error);
