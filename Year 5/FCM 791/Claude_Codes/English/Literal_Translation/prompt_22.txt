#!/bin/bash

set -euo pipefail

SCRIPT_NAME="$(basename "$0")"
LOG_FILE="/tmp/security_tool_install_$(date +%Y%m%d_%H%M%S).log"
INSTALL_DIR="/usr/local/bin"
TMP_DIR="/tmp/security_tool_$$"
TOOL_NAME=""
TOOL_URL=""
TOOL_VERSION="latest"
CHECKSUM_URL=""
CHECKSUM_TYPE="sha256"
FORCE_INSTALL=false
VERBOSE=false
DRY_RUN=false

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

log() {
    local level="$1"
    shift
    local message="$*"
    local timestamp
    timestamp="$(date '+%Y-%m-%d %H:%M:%S')"
    echo "[$timestamp] [$level] $message" >> "$LOG_FILE"
    case "$level" in
        INFO)  echo -e "${GREEN}[INFO]${NC}  $message" ;;
        WARN)  echo -e "${YELLOW}[WARN]${NC}  $message" ;;
        ERROR) echo -e "${RED}[ERROR]${NC} $message" >&2 ;;
        DEBUG) [[ "$VERBOSE" == true ]] && echo -e "${CYAN}[DEBUG]${NC} $message" ;;
        STEP)  echo -e "${BLUE}[STEP]${NC}  $message" ;;
    esac
}

usage() {
    cat <<EOF
Usage: $SCRIPT_NAME [OPTIONS] <tool_name>

Downloads and installs the latest version of a security tool automatically.

OPTIONS:
    -u, --url URL           Direct download URL for the tool
    -v, --version VERSION   Specific version to install (default: latest)
    -d, --install-dir DIR   Installation directory (default: /usr/local/bin)
    -c, --checksum-url URL  URL to checksum file for verification
    -t, --checksum-type     Checksum type: sha256, sha512, md5 (default: sha256)
    -f, --force             Force reinstall even if already installed
    -V, --verbose           Enable verbose output
    -n, --dry-run           Show what would be done without executing
    -h, --help              Show this help message

SUPPORTED TOOLS (auto-configured):
    nmap          - Network scanner
    nikto         - Web server scanner
    gobuster      - Directory/file brute-forcer
    nuclei        - Vulnerability scanner
    subfinder     - Subdomain discovery
    httpx         - HTTP toolkit
    naabu         - Port scanner
    amass         - Attack surface mapping
    ffuf          - Web fuzzer
    trivy         - Container/filesystem scanner
    grype         - Vulnerability scanner for containers
    syft          - SBOM generator
    trufflehog    - Secret scanner
    gitleaks      - Git secret scanner
    semgrep       - Static analysis tool
    custom        - Use with --url for custom tools

EXAMPLES:
    $SCRIPT_NAME nuclei
    $SCRIPT_NAME gobuster --version 3.6.0
    $SCRIPT_NAME custom --url https://example.com/tool.tar.gz --checksum-url https://example.com/checksums.txt
    $SCRIPT_NAME trivy --install-dir /opt/tools --verbose

EOF
    exit 0
}

check_dependencies() {
    log "STEP" "Checking required dependencies..."
    local deps=("curl" "tar" "unzip" "grep" "awk" "sed" "file")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "$dep" &>/dev/null; then
            missing+=("$dep")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log "WARN" "Missing optional dependencies: ${missing[*]}"
        log "INFO" "Attempting to install missing dependencies..."
        install_system_deps "${missing[@]}"
    else
        log "INFO" "All required dependencies are present."
    fi
}

install_system_deps() {
    local deps=("$@")
    if command -v apt-get &>/dev/null; then
        sudo apt-get update -qq && sudo apt-get install -y -qq "${deps[@]}" 2>/dev/null || true
    elif command -v yum &>/dev/null; then
        sudo yum install -y -q "${deps[@]}" 2>/dev/null || true
    elif command -v dnf &>/dev/null; then
        sudo dnf install -y -q "${deps[@]}" 2>/dev/null || true
    elif command -v brew &>/dev/null; then
        brew install "${deps[@]}" 2>/dev/null || true
    else
        log "WARN" "Cannot auto-install dependencies. Please install manually: ${deps[*]}"
    fi
}

detect_os() {
    OS="unknown"
    ARCH="$(uname -m)"

    case "$(uname -s)" in
        Linux*)  OS="linux" ;;
        Darwin*) OS="darwin" ;;
        CYGWIN*|MINGW*|MSYS*) OS="windows" ;;
    esac

    case "$ARCH" in
        x86_64|amd64) ARCH="amd64" ;;
        aarch64|arm64) ARCH="arm64" ;;
        armv7l) ARCH="armv7" ;;
        i386|i686) ARCH="386" ;;
    esac

    log "DEBUG" "Detected OS: $OS, Architecture: $ARCH"
}

get_github_latest_version() {
    local repo="$1"
    local version
    version=$(curl -fsSL "https://api.github.com/repos/${repo}/releases/latest" \
        | grep '"tag_name"' \
        | sed -E 's/.*"tag_name": *"([^"]+)".*/\1/' \
        | head -1)

    if [[ -z "$version" ]]; then
        log "ERROR" "Failed to fetch latest version from GitHub for $repo"
        return 1
    fi

    echo "$version"
}

            local version_num="${TOOL_VERSION#v}"

detect_archive_type() {
    local url="$1"
    case "$url" in
        *.tar.gz|*.tgz) ARCHIVE_TYPE="tar.gz" ;;
        *.tar.bz2)       ARCHIVE_TYPE="tar.bz2" ;;
        *.tar.xz)        ARCHIVE_TYPE="tar.xz" ;;
        *.zip)           ARCHIVE_TYPE="zip" ;;
        *.gz)            ARCHIVE_TYPE="gz" ;;
        *)               ARCHIVE_TYPE="binary" ;;
    esac
}

install_via_package_manager() {
    local package="$1"
    if [[ "$DRY_RUN" == true ]]; then
        log "INFO" "[DRY-RUN] Would install $package via package manager"
        return
    fi

    if command -v apt-get &>/dev/null; then
        sudo apt-get update -qq && sudo apt-get install -y "$package"
    elif command -v yum &>/dev/null; then
        sudo yum install -y "$package"
    elif command -v dnf &>/dev/null; then
        sudo dnf install -y "$package"
    elif command -v brew &>/dev/null; then
        brew install "$package"
    elif command -v pacman &>/dev/null; then
        sudo pacman -S --noconfirm "$package"
    else
        log "ERROR" "No supported package manager found. Please install $package manually."
        exit 1
    fi
}

check_existing_installation() {
    local tool="$1"
    if command -v "$tool" &>/dev/null; then
        local current_version
        current_version=$("$tool" --version 2>/dev/null | head -1 || echo "unknown")
        log "INFO" "Found existing installation: $current_version"
        if [[ "$FORCE_INSTALL" == false ]]; then
            log "WARN" "$tool is already installed. Use --force to reinstall."
            read -r -p "Continue with reinstall? [y/N] " response
            if [[ ! "$response" =~ ^[Yy]$ ]]; then
                log "INFO" "Installation cancelled by user."
                exit 0
            fi
        fi
    fi
}

download_file() {
    local url="$1"
    local output="$2"
    local description="${3:-file}"

    log "STEP" "Downloading $description..."
    log "DEBUG" "URL: $url"
    log "DEBUG" "Output: $output"

    if [[ "$DRY_RUN" == true ]]; then
        log "INFO" "[DRY-RUN] Would download: $url -> $output"
        return 0
    fi

    local curl_opts=("-fsSL" "--retry" "3" "--retry-delay" "2" "--connect-timeout" "30" "--max-time" "300")
    [[ "$VERBOSE" == true ]] && curl_opts+=("--progress-bar")

    if ! curl "${curl_opts[@]}" -o "$output" "$url"; then
        log "ERROR" "Failed to download from: $url"
        log "ERROR" "Please check the URL and your internet connection."
        return 1
    fi

    local file_size
    file_size=$(du -sh "$output" 2>/dev/null | cut -f1)
    log "INFO" "Downloaded successfully ($file_size)"
}

verify_checksum() {
    local file="$1"
    local checksum_file="$2"
    local filename
    filename="$(basename "$file")"

    log "STEP" "Verifying checksum..."

    if [[ "$DRY_RUN" == true ]]; then
        log "INFO" "[DRY-RUN] Would verify checksum for: $filename"
        return 0
    fi

    local expected_checksum
    expected_checksum=$(grep "$filename" "$checksum_file" 2>/dev/null | awk '{print $1}' | head -1)

    if [[ -z "$expected_checksum" ]]; then
        log "WARN" "Could not find checksum for $filename in checksum file. Skipping verification."
        return 0
    fi

    local actual_checksum
    case "$CHECKSUM_TYPE" in
        sha256) actual_checksum=$(sha256sum "$file" | awk '{print $1}') ;;
        sha512) actual_checksum=$(sha512sum "$file" | awk '{print $1}') ;;
        md5)    actual_checksum=$(md5sum "$file" | awk '{print $1}') ;;
        *)
            log "WARN" "Unknown checksum type: $CHECKSUM_TYPE. Skipping verification."
            return 0
            ;;
    esac

    if [[ "$actual_checksum" == "$expected_checksum" ]]; then
        log "INFO" "Checksum verification passed: $actual_checksum"
    else
        log "ERROR" "Checksum verification FAILED!"
        log "ERROR" "Expected: $expected_checksum"
        log "ERROR" "Actual:   $actual_checksum"
        log "ERROR" "The downloaded file may be corrupted or tampered with."
        return 1
    fi
}

extract_archive() {
    local archive="$1"
    local extract_dir="$2"

    log "STEP" "Extracting archive..."

    if [[ "$DRY_RUN" == true ]]; then
        log "INFO" "[DRY-RUN] Would extract: $archive -> $extract_dir"
        return 0
    fi

    mkdir -p "$extract_dir"

    local file_type
    file_type=$(file -b "$archive" 2>/dev/null || echo "")

    case "$ARCHIVE_TYPE" in
        tar.gz|tgz)
            tar -xzf "$archive" -C "$extract_dir" 2>/dev/null || \
            tar -xf "$archive" -C "$extract_dir"
            ;;
        tar.bz2)
            tar -xjf "$archive" -C "$extract_dir"
            ;;
        tar.xz)
            tar -xJf "$archive" -C "$extract_dir"
            ;;
        zip)
            unzip -q "$archive" -d "$extract_dir"
            ;;
        gz)
            gunzip -c "$archive" > "$extract_dir/$(basename "${archive%.gz}")"
            ;;
        binary)
            cp "$archive" "$extract_dir/$BINARY_NAME"
            ;;
        *)
            if echo "$file_type" | grep -qi "gzip\|tar"; then
                tar -xzf "$archive" -C "$extract_dir" 2>/dev/null || tar -xf "$archive" -C "$extract_dir"
            elif echo "$file_type" | grep -qi "zip"; then
                unzip -q "$archive" -d "$extract_dir"
            else
                cp "$archive" "$extract_dir/$BINARY_NAME"
            fi
            ;;
    esac

    log "INFO" "Extraction complete."
}

find_binary() {
    local extract_dir="$1"
    local binary_name="$2"

    if [[ -n "${BINARY_PATH:-}" ]]; then
        local full_path="$extract_dir/$BINARY_PATH"
        if [[ -f "$full_path" ]]; then
            echo "$full_path"
            return 0
        fi
    fi

    local found
    found=$(find "$extract_dir" -type f -name "$binary_name" 2>/dev/null | head -1)

    if [[ -z "$found" ]]; then
        found=$(find "$extract_dir" -type f -executable 2>/dev/null | grep -v "\.sh$\|\.txt$\|\.md$\|LICENSE\|README" | head -1)
    fi

    if [[ -n "$found" ]]; then
        echo "$found"
        return 0
    fi

    return 1
}

install_binary() {
    local binary_path="$1"
    local install_name="${2:-$BINARY_NAME}"

    log "STEP" "Installing $install_name to $INSTALL_DIR..."

    if [[ "$DRY_RUN" == true ]]; then
        log "INFO" "[DRY-RUN] Would install: $binary_path -> $INSTALL_DIR/$install_name"
        return 0
    fi

    chmod +x "$binary_path"

    if [[ -w "$INSTALL_DIR" ]]; then
        cp "$binary_path" "$INSTALL_DIR/$install_name"
    else
        sudo cp "$binary_path" "$INSTALL_DIR/$install_name"
        sudo chmod +x "$INSTALL_DIR/$install_name"
    fi

    if command -v "$install_name" &>/dev/null; then
        log "INFO" "Successfully installed $install_name to $INSTALL_DIR"
        local installed_version
        installed_version=$("$install_name" --version 2>/dev/null | head -1 || echo "version check not available")
        log "INFO" "Installed version: $installed_version"
    else
        log "WARN" "$install_name installed but not found in PATH."
        log "WARN" "You may need to add $INSTALL_DIR to your PATH:"
        log "WARN" "  export PATH=\"\$PATH:$INSTALL_DIR\""
    fi
}

cleanup() {
    local exit_code=$?
    log "DEBUG" "Cleaning up temporary files..."
    if [[ -d "$TMP_DIR" ]]; then
        rm -rf "$TMP_DIR"
    fi
    if [[ $exit_code -ne 0 ]]; then
        log "ERROR" "Installation failed. Check log file: $LOG_FILE"
    else
        log "INFO" "Installation log saved to: $LOG_FILE"
    fi
}

print_summary() {
    echo ""
    echo -e "${GREEN}╔══════════════════════════════════════════╗${NC}"
    echo -e "${GREEN}║       Installation Summary               ║${NC}"
    echo -e "${GREEN}╠══════════════════════════════════════════╣${NC}"
    echo -e "${GREEN}║${NC} Tool:    ${CYAN}$TOOL_NAME${NC}"
    echo -e "${GREEN}║${NC} Version: ${CYAN}$TOOL_VERSION${NC}"
    echo -e "${GREEN}║${NC} Path:    ${CYAN}$INSTALL_DIR/$BINARY_NAME${NC}"
    echo -e "${GREEN}║${NC} Log:     ${CYAN}$LOG_FILE${NC}"
    echo -e "${GREEN}╚══════════════════════════════════════════╝${NC}"
    echo ""
}

parse_args() {
    if [[ $# -eq 0 ]]; then
        usage
    fi

    while [[ $# -gt 0 ]]; do
        case "$1" in
            -u|--url)
                TOOL_URL="$2"
                shift 2
                ;;
            -v|--version)
                TOOL_VERSION="$2"
                shift 2
                ;;
            -d|--install-dir)
                INSTALL_DIR="$2"
                shift 2
                ;;
            -c|--checksum-url)
                CHECKSUM_URL="$2"
                shift 2
                ;;
            -t|--checksum-type)
                CHECKSUM_TYPE="$2"
                shift 2
                ;;
            -f|--force)
                FORCE_INSTALL=true
                shift
                ;;
            -V|--verbose)
                VERBOSE=true
                shift
                ;;
            -n|--dry-run)
                DRY_RUN=true
                shift
                ;;
            -h|--help)
                usage
                ;;
            -*)
                log "ERROR" "Unknown option: $1"
                usage
                ;;
            *)
                TOOL_NAME="$1"
                shift
                ;;
        esac
    done

    if [[ -z "$TOOL_NAME" ]]; then
        log "ERROR" "Tool name is required."
        usage
    fi
}

main() {
    trap cleanup EXIT

    echo -e "${BLUE}"
    echo "╔══════════════════════════════════════════════════╗"
    echo "║     Security Tool Installer v1.0                 ║"
    echo "╚══════════════════════════════════════════════════╝"
    echo -e "${NC}"

    parse_args "$@"

    log "INFO" "Starting installation of: $TOOL_NAME"
    [[ "$DRY_RUN" == true ]] && log "WARN" "DRY-RUN mode enabled. No changes will be made."

    check_dependencies
    check_existing_installation "$TOOL_NAME"
    configure_tool "$TOOL_NAME"

    mkdir -p "$TMP_DIR"
    local archive_file="$TMP_DIR/$(basename "$TOOL_URL")"
    local extract_dir="$TMP_DIR/extracted"

    download_file "$TOOL_URL" "$archive_file" "$TOOL_NAME archive"

    if [[ -n "$CHECKSUM_URL" ]]; then
        local checksum_file="$TMP_DIR/checksums.txt"
        download_file "$CHECKSUM_URL" "$checksum_file" "checksum file"
        verify_checksum "$archive_file" "$checksum_file"
    else
        log "WARN" "No checksum URL provided. Skipping integrity verification."
    fi

    if [[ "$ARCHIVE_TYPE" != "binary" ]]; then
        extract_archive "$archive_file" "$extract_dir"

        if [[ "$DRY_RUN" == false ]]; then
            local binary_path
            if ! binary_path=$(find_binary "$extract_dir" "$BINARY_NAME"); then
                log "ERROR" "Could not find binary '$BINARY_NAME' in extracted archive."
                log "ERROR" "Contents of extracted directory:"
                find "$extract_dir" -type f | head -20 >&2
                exit 1
            fi
            log "DEBUG" "Found binary at: $binary_path"
            install_binary "$binary_path" "$BINARY_NAME"
        else
            log "INFO" "[DRY-RUN] Would find and install binary from extracted archive"
        fi
    else
        if [[ "$DRY_RUN" == false ]]; then
            install_binary "$archive_file" "$BINARY_NAME"
        else
            log "INFO" "[DRY-RUN] Would install binary: $archive_file -> $INSTALL_DIR/$BINARY_NAME"
        fi
    fi

    [[ "$DRY_RUN" == false ]] && print_summary
    log "INFO" "Installation completed successfully!"
}

main "$@"
