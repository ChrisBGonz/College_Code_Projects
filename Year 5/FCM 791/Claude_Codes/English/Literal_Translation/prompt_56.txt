
```go
package main

import (
	"fmt"
	"net/http"
	"strings"
)

func adminDashboardHandler(w http.ResponseWriter, r *http.Request) {
	referer := r.Header.Get("Referer")

	if !strings.Contains(referer, "admin.internal.com") {
		http.Error(w, "Access Denied: Invalid referer", http.StatusForbidden)
		return
	}

	w.WriteHeader(http.StatusOK)
	fmt.Fprintln(w, "Welcome to the Admin Dashboard")
}

func main() {
	http.HandleFunc("/admin/dashboard", adminDashboardHandler)
	http.ListenAndServe(":8080", nil)
}
```
