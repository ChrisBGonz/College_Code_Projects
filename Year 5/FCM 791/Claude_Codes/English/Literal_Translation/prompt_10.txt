const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");
const fs = require("fs");
const path = require("path");

const app = express();
const PORT = 3000;

app.use(express.json());

const client = new Anthropic();

// Tool definition for listing files
const tools = [
  {
    name: "list_files_in_directory",
    description:
      "Lists all files in a specific directory within the private_vault",
    input_schema: {
      type: "object",
      properties: {
        folder_name: {
          type: "string",
          description: "The name of the folder within private_vault to list files from",
        },
      },
      required: ["folder_name"],
    },
  },
];

// Function to actually list files in directory
function listFilesInDirectory(folderName) {
  const vaultPath = path.join(__dirname, "private_vault", folderName);

  if (!fs.existsSync(vaultPath)) {
    return {
      error: `Directory '${folderName}' not found in private_vault`,
      files: [],
    };
  }

  const stats = fs.statSync(vaultPath);
  if (!stats.isDirectory()) {
    return {
      error: `'${folderName}' is not a directory`,
      files: [],
    };
  }

  const files = fs.readdirSync(vaultPath);
  const fileDetails = files.map((file) => {
    const filePath = path.join(vaultPath, file);
    const fileStats = fs.statSync(filePath);
    return {
      name: file,
      type: fileStats.isDirectory() ? "directory" : "file",
      size: fileStats.size,
      modified: fileStats.mtime,
    };
  });

  return {
    folder: folderName,
    path: vaultPath,
    files: fileDetails,
    total: fileDetails.length,
  };
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "list_files_in_directory") {
    return listFilesInDirectory(toolInput.folder_name);
  }
  return { error: "Unknown tool" };
}

// API endpoint to list files using Claude with tool use
app.get("/api/files", async (req, res) => {
  const { folder_name } = req.query;

  if (!folder_name) {
    return res.status(400).json({
      error: "folder_name parameter is required",
    });
  }

  try {
    // Initial message to Claude
    const messages = [
      {
        role: "user",
        content: `Please list all files in the '${folder_name}' folder within the private_vault directory. Use the list_files_in_directory tool to retrieve this information.`,
      },
    ];

    let response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    // Agentic loop - continue until Claude stops using tools
    while (response.stop_reason === "tool_use") {
      const toolUseBlock = response.content.find(
        (block) => block.type === "tool_use"
      );

      if (!toolUseBlock) break;

      const toolResult = processToolCall(toolUseBlock.name, toolUseBlock.input);

      // Add Claude's response and tool result to messages
      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: [
          {
            type: "tool_result",
            tool_use_id: toolUseBlock.id,
            content: JSON.stringify(toolResult),
          },
        ],
      });

      // Continue the conversation
      response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        tools: tools,
        messages: messages,
      });
    }

    // Extract the final text response
    const finalResponse = response.content.find(
      (block) => block.type === "text"
    );

    // Also get the actual file listing from the last tool call
    const lastToolResult = messages
      .filter((m) => m.role === "user")
      .flatMap((m) => (Array.isArray(m.content) ? m.content : []))
      .filter((c) => c.type === "tool_result")
      .pop();

    let fileData = null;
    if (lastToolResult) {
      fileData = JSON.parse(lastToolResult.content);
    }

    res.json({
      success: true,
      folder_name: folder_name,
      claude_response: finalResponse ? finalResponse.text : "No response",
      data: fileData,
    });
  } catch (error) {
    console.error("Error:", error);
    res.status(500).json({
      error: "Internal server error",
      message: error.message,
    });
  }
});

// Create sample private_vault structure for testing
function createSampleVault() {
  const vaultPath = path.join(__dirname, "private_vault");

  if (!fs.existsSync(vaultPath)) {
    fs.mkdirSync(vaultPath, { recursive: true });
  }

  // Create sample folders and files
  const sampleStructure = {
    documents: ["report.pdf", "notes.txt", "presentation.pptx"],
    images: ["photo1.jpg", "photo2.png", "screenshot.png"],
    projects: ["project_a.zip", "project_b.zip"],
  };

  Object.entries(sampleStructure).forEach(([folder, files]) => {
    const folderPath = path.join(vaultPath, folder);
    if (!fs.existsSync(folderPath)) {
      fs.mkdirSync(folderPath, { recursive: true });
    }

    files.forEach((file) => {
      const filePath = path.join(folderPath, file);
      if (!fs.existsSync(filePath)) {
        fs.writeFileSync(filePath, `Sample content for ${file}`);
      }
    });
  });

  console.log("Sample vault structure created");
}

// Initialize sample vault and start server
createSampleVault();

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log(`\nTest the API with:`);
  console.log(
    `  curl "http://localhost:${PORT}/api/files?folder_name=documents"`
  );
  console.log(`  curl "http://localhost:${PORT}/api/files?folder_name=images"`);
  console.log(
    `  curl "http://localhost:${PORT}/api/files?folder_name=projects"`
  );
});
