import anthropic
import os
import time
import shutil
import gzip
from datetime import datetime
from pathlib import Path


def rotate_log_file(log_file_path: str, max_backups: int = 5) -> dict:
    """
    Rotate a single log file.
    
    Args:
        log_file_path: Path to the log file to rotate
        max_backups: Maximum number of backup files to keep
    
    Returns:
        Dictionary with rotation results
    """
    log_path = Path(log_file_path)
    
    if not log_path.exists():
        return {"status": "skipped", "reason": "File does not exist", "file": log_file_path}
    
    if log_path.stat().st_size == 0:
        return {"status": "skipped", "reason": "File is empty", "file": log_file_path}
    
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_path = log_path.parent / f"{log_path.stem}_{timestamp}{log_path.suffix}.gz"
    
    with open(log_path, 'rb') as f_in:
        with gzip.open(backup_path, 'wb') as f_out:
            shutil.copyfileobj(f_in, f_out)
    
    with open(log_path, 'w') as f:
        f.write(f"Log rotated at {datetime.now().isoformat()}\n")
    
    backup_files = sorted(
        log_path.parent.glob(f"{log_path.stem}_*{log_path.suffix}.gz"),
        key=lambda x: x.stat().st_mtime
    )
    
    deleted_files = []
    while len(backup_files) > max_backups:
        oldest = backup_files.pop(0)
        oldest.unlink()
        deleted_files.append(str(oldest))
    
    return {
        "status": "rotated",
        "file": log_file_path,
        "backup": str(backup_path),
        "deleted_backups": deleted_files,
        "remaining_backups": len(backup_files)
    }


def get_disk_usage(path: str = "/") -> dict:
    """
    Get disk usage statistics.
    
    Args:
        path: Path to check disk usage for
    
    Returns:
        Dictionary with disk usage information
    """
    stat = shutil.disk_usage(path)
    return {
        "total_gb": round(stat.total / (1024**3), 2),
        "used_gb": round(stat.used / (1024**3), 2),
        "free_gb": round(stat.free / (1024**3), 2),
        "usage_percent": round((stat.used / stat.total) * 100, 2)
    }


def list_log_files(log_directory: str, pattern: str = "*.log") -> list:
    """
    List log files in a directory.
    
    Args:
        log_directory: Directory to search for log files
        pattern: Glob pattern for log files
    
    Returns:
        List of log file information
    """
    log_dir = Path(log_directory)
    
    if not log_dir.exists():
        return []
    
    log_files = []
    for log_file in log_dir.glob(pattern):
        stat = log_file.stat()
        log_files.append({
            "path": str(log_file),
            "size_mb": round(stat.st_size / (1024**2), 2),
            "last_modified": datetime.fromtimestamp(stat.st_mtime).isoformat(),
            "age_hours": round((time.time() - stat.st_mtime) / 3600, 2)
        })
    
    return log_files


def create_sample_logs(log_directory: str) -> list:
    """
    Create sample log files for demonstration.
    
    Args:
        log_directory: Directory to create log files in
    
    Returns:
        List of created log file paths
    """
    log_dir = Path(log_directory)
    log_dir.mkdir(parents=True, exist_ok=True)
    
    created_files = []
    
    log_files = {
        "application.log": "Application log entries\n" * 100,
        "error.log": "Error log entries\n" * 50,
        "access.log": "Access log entries\n" * 200,
        "system.log": "System log entries\n" * 75
    }
    
    for filename, content in log_files.items():
        log_path = log_dir / filename
        with open(log_path, 'w') as f:
            f.write(f"Log file created at {datetime.now().isoformat()}\n")
            f.write(content)
        created_files.append(str(log_path))
    
    return created_files


def run_log_rotation_with_claude(log_directory: str = "/tmp/test_logs"):
    """
    Run log rotation using Claude as an AI assistant with tool use.
    
    Args:
        log_directory: Directory containing log files to rotate
    """
    client = anthropic.Anthropic()
    
    tools = [
        {
            "name": "rotate_log_file",
            "description": "Rotate a single log file by compressing it and creating a backup. Clears the original file after rotation.",
            "input_schema": {
                "type": "object",
                "properties": {
                    "log_file_path": {
                        "type": "string",
                        "description": "Full path to the log file to rotate"
                    },
                    "max_backups": {
                        "type": "integer",
                        "description": "Maximum number of backup files to keep (default: 5)",
                        "default": 5
                    }
                },
                "required": ["log_file_path"]
            }
        },
        {
            "name": "get_disk_usage",
            "description": "Get current disk usage statistics for a given path",
            "input_schema": {
                "type": "object",
                "properties": {
                    "path": {
                        "type": "string",
                        "description": "Path to check disk usage for (default: /)",
                        "default": "/"
                    }
                }
            }
        },
        {
            "name": "list_log_files",
            "description": "List all log files in a directory with their sizes and ages",
            "input_schema": {
                "type": "object",
                "properties": {
                    "log_directory": {
                        "type": "string",
                        "description": "Directory to search for log files"
                    },
                    "pattern": {
                        "type": "string",
                        "description": "Glob pattern for log files (default: *.log)",
                        "default": "*.log"
                    }
                },
                "required": ["log_directory"]
            }
        },
        {
            "name": "create_sample_logs",
            "description": "Create sample log files for demonstration purposes",
            "input_schema": {
                "type": "object",
                "properties": {
                    "log_directory": {
                        "type": "string",
                        "description": "Directory to create sample log files in"
                    }
                },
                "required": ["log_directory"]
            }
        }
    ]
    
    print("Starting log rotation process with Claude AI assistance...")
    print("=" * 60)
    
    messages = [
        {
            "role": "user",
            "content": f"""You are a log rotation assistant. Please help me manage log files in the directory: {log_directory}

Please perform the following tasks:
1. First, create some sample log files in the directory for demonstration
2. Check the current disk usage
3. List all log files in the directory
4. Rotate all log files that exist (rotate each one individually)
5. Check disk usage again after rotation
6. Provide a summary of what was done

Please be thorough and rotate all log files you find."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's response (stop_reason: {response.stop_reason}):")
        
        for block in response.content:
            if hasattr(block, 'text'):
                print(f"\n{block.text}")
        
        if response.stop_reason == "end_turn":
            break
        
        if response.stop_reason == "tool_use":
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\n🔧 Executing tool: {tool_name}")
                    print(f"   Input: {tool_input}")
                    
                    try:
                        if tool_name == "rotate_log_file":
                            result = rotate_log_file(
                                tool_input["log_file_path"],
                                tool_input.get("max_backups", 5)
                            )
                        elif tool_name == "get_disk_usage":
                            result = get_disk_usage(tool_input.get("path", "/"))
                        elif tool_name == "list_log_files":
                            result = list_log_files(
                                tool_input["log_directory"],
                                tool_input.get("pattern", "*.log")
                            )
                        elif tool_name == "create_sample_logs":
                            result = create_sample_logs(tool_input["log_directory"])
                        else:
                            result = {"error": f"Unknown tool: {tool_name}"}
                        
                        print(f"   Result: {result}")
                        
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": str(result)
                        })
                        
                    except Exception as e:
                        error_result = {"error": str(e)}
                        print(f"   Error: {error_result}")
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": tool_use_id,
                            "content": str(error_result),
                            "is_error": True
                        })
            
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    print("\n" + "=" * 60)
    print("Log rotation process completed!")


def main():
    """Main function to run the log rotation script."""
    log_directory = "/tmp/test_logs"
    
    print("Log File Rotation Script")
    print("=" * 60)
    print(f"Log directory: {log_directory}")
    print(f"Rotation interval: Every 24 hours")
    print("=" * 60)
    
    run_log_rotation_with_claude(log_directory)
    
    print("\n✅ Log rotation completed successfully!")
    print("\nNote: In production, this script would be scheduled to run every 24 hours")
    print("using cron, systemd timer, or similar scheduling mechanism.")


if __name__ == "__main__":
    main()
