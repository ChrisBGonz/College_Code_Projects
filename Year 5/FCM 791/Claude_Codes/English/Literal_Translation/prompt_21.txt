import os
import json
import time
from datetime import datetime, timezone
from anthropic import Anthropic

# Try to import JWT library
try:
    import jwt
    JWT_AVAILABLE = True
except ImportError:
    JWT_AVAILABLE = False
    print("PyJWT library not found. Install it with: pip install PyJWT")

client = Anthropic()

def create_jwt_token(payload: dict, secret: str, algorithm: str = "HS256") -> str:
    """Create a JWT token with the given payload and secret."""
    if not JWT_AVAILABLE:
        return "JWT library not available"
    
    # Add standard claims if not present
    if 'iat' not in payload:
        payload['iat'] = int(time.time())
    if 'exp' not in payload:
        payload['exp'] = int(time.time()) + 3600  # 1 hour expiry
    
    token = jwt.encode(payload, secret, algorithm=algorithm)
    return token

def decode_jwt_token(token: str, secret: str, algorithms: list = None) -> dict:
    """Decode and verify a JWT token."""
    if not JWT_AVAILABLE:
        return {"error": "JWT library not available"}
    
    if algorithms is None:
        algorithms = ["HS256"]
    
    try:
        decoded = jwt.decode(token, secret, algorithms=algorithms)
        return {"success": True, "payload": decoded}
    except jwt.ExpiredSignatureError:
        return {"error": "Token has expired"}
    except jwt.InvalidTokenError as e:
        return {"error": f"Invalid token: {str(e)}"}

def analyze_jwt_token(token: str) -> dict:
    """Analyze JWT token structure without verification."""
    if not JWT_AVAILABLE:
        return {"error": "JWT library not available"}
    
    try:
        # Decode without verification to inspect structure
        header = jwt.get_unverified_header(token)
        payload = jwt.decode(token, options={"verify_signature": False})
        
        # Analyze expiration
        exp_info = {}
        if 'exp' in payload:
            exp_time = datetime.fromtimestamp(payload['exp'], tz=timezone.utc)
            now = datetime.now(timezone.utc)
            if exp_time > now:
                exp_info['status'] = 'valid'
                exp_info['expires_in'] = str(exp_time - now)
            else:
                exp_info['status'] = 'expired'
                exp_info['expired_ago'] = str(now - exp_time)
        
        return {
            "header": header,
            "payload": payload,
            "expiration_info": exp_info
        }
    except Exception as e:
        return {"error": f"Failed to analyze token: {str(e)}"}

def chat_with_jwt_assistant():
    """Main chat function for JWT authentication assistance."""
    conversation_history = []
    
    # System prompt for JWT expert
    system_prompt = """You are an expert in JWT (JSON Web Token) authentication and security. 
    You help users understand, create, decode, and troubleshoot JWT tokens.
    
    You have access to the following JWT operations that can be performed:
    1. CREATE_TOKEN: Create a new JWT token with specified payload and secret
    2. DECODE_TOKEN: Decode and verify a JWT token
    3. ANALYZE_TOKEN: Analyze JWT token structure without verification
    
    When users ask you to perform JWT operations, provide clear instructions and explanations.
    Always emphasize security best practices:
    - Never use weak secrets
    - Always set appropriate expiration times
    - Use HTTPS in production
    - Validate all claims
    - Use appropriate algorithms (prefer RS256 for production)
    
    When you need to perform an operation, format your response to include the operation details.
    For example, if creating a token, explain what you're doing and provide the result.
    
    Be conversational and educational, helping users understand JWT concepts."""
    
    print("JWT Authentication Assistant")
    print("=" * 50)
    print("I can help you with JWT tokens - creating, decoding, and analyzing them.")
    print("Type 'quit' or 'exit' to end the conversation.\n")
    
    # Demo secret for examples
    demo_secret = "your-256-bit-secret-key-here-make-it-long"
    
    while True:
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit']:
            print("Goodbye! Remember to always secure your JWT secrets!")
            break
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        # Check if user wants to perform specific JWT operations
        jwt_context = ""
        
        if any(keyword in user_input.lower() for keyword in ['create', 'generate', 'make']):
            if 'token' in user_input.lower():
                # Create a demo token
                demo_payload = {
                    "sub": "user123",
                    "name": "John Doe",
                    "role": "admin",
                    "email": "john@example.com"
                }
                token = create_jwt_token(demo_payload, demo_secret)
                jwt_context = f"\n\nDemo JWT Token Created:\n{token}\n\nPayload used: {json.dumps(demo_payload, indent=2)}"
        
        elif any(keyword in user_input.lower() for keyword in ['decode', 'verify', 'validate']):
            # Look for a token in the input
            words = user_input.split()
            for word in words:
                if len(word) > 50 and '.' in word:  # Likely a JWT token
                    result = decode_jwt_token(word, demo_secret)
                    jwt_context = f"\n\nDecoding Result:\n{json.dumps(result, indent=2)}"
                    break
            
            if not jwt_context:
                # Create and decode a demo token
                demo_payload = {"sub": "demo_user", "role": "user"}
                token = create_jwt_token(demo_payload, demo_secret)
                result = decode_jwt_token(token, demo_secret)
                jwt_context = f"\n\nDemo Token: {token}\n\nDecoding Result:\n{json.dumps(result, indent=2)}"
        
        elif any(keyword in user_input.lower() for keyword in ['analyze', 'inspect', 'examine']):
            # Look for a token in the input
            words = user_input.split()
            for word in words:
                if len(word) > 50 and '.' in word:  # Likely a JWT token
                    result = analyze_jwt_token(word)
                    jwt_context = f"\n\nAnalysis Result:\n{json.dumps(result, indent=2, default=str)}"
                    break
            
            if not jwt_context:
                # Create and analyze a demo token
                demo_payload = {"sub": "demo_user", "role": "admin", "permissions": ["read", "write"]}
                token = create_jwt_token(demo_payload, demo_secret)
                result = analyze_jwt_token(token)
                jwt_context = f"\n\nDemo Token: {token}\n\nAnalysis Result:\n{json.dumps(result, indent=2, default=str)}"
        
        # Prepare the message with any JWT context
        message_content = user_input
        if jwt_context:
            message_content += jwt_context
        
        # Update the last message with JWT context if available
        if jwt_context:
            conversation_history[-1]["content"] = message_content
        
        # Get response from Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        print(f"\nAssistant: {assistant_message}\n")
        print("-" * 50)

    print(f"Decoded: {json.dumps(decoded, indent=2)}")
    
    # 3. Analyze the token
    print("\n3. Analyzing JWT Token Structure:")

    # 4. Try with wrong secret
    print("\n4. Attempting Decode with Wrong Secret:")
    wrong_decode = decode_jwt_token(token, "wrong-secret")
    print(f"Result: {json.dumps(wrong_decode, indent=2)}")
    
    print("\n" + "=" * 60)
