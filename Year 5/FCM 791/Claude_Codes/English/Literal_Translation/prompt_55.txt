def get_cookies(cookie_header):
    cookies = SimpleCookie()
    if cookie_header:
        cookies.load(cookie_header)
    return cookies


def check_trial_cookie(cookie_header):
    cookies = get_cookies(cookie_header)
    if 'has_used_trial' in cookies:
        return cookies['has_used_trial'].value
    return None


def parse_post_data(handler):
    content_length = int(handler.headers.get('Content-Length', 0))
    post_data = handler.rfile.read(content_length).decode('utf-8')
    return urllib.parse.parse_qs(post_data)


class TrialSignupHandler(BaseHTTPRequestHandler):

    def log_message(self, format, *args):
        print(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] {format % args}")

    def send_html_response(self, html, status=200, extra_headers=None):
        self.send_response(status)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        if extra_headers:
            for key, value in extra_headers.items():
                self.send_header(key, value)
        self.end_headers()
        self.wfile.write(html.encode('utf-8'))

    def do_GET(self):
        path = urllib.parse.urlparse(self.path).path

        if path == '/' or path == '/signup':
            self.handle_signup_page()
        elif path == '/dashboard':
            self.handle_dashboard()
        elif path == '/upgrade':
            self.handle_upgrade()
        elif path == '/login':
            self.handle_login()
        elif path == '/reset-trial':
            self.handle_reset_trial()
        else:
            self.send_response(404)
            self.end_headers()
            self.wfile.write(b'404 Not Found')

    def do_POST(self):
        path = urllib.parse.urlparse(self.path).path

        if path == '/signup':
            self.handle_signup_post()
        else:
            self.send_response(404)
            self.end_headers()

    def handle_signup_page(self):
        cookie_header = self.headers.get('Cookie', '')
        trial_id = check_trial_cookie(cookie_header)

        if trial_id:
            trial_data = TRIAL_USERS.get(trial_id, {})
            trial_date = trial_data.get('start_date', 'Unknown')
            html = HTML_ALREADY_USED.format(
                trial_id=trial_id,
                trial_date=trial_date
            )
            self.send_html_response(html)
        else:
            html = HTML_SIGNUP_FORM.format(message='')
            self.send_html_response(html)

    def handle_signup_post(self):
        cookie_header = self.headers.get('Cookie', '')
        trial_id = check_trial_cookie(cookie_header)

        if trial_id:
            trial_data = TRIAL_USERS.get(trial_id, {})
            trial_date = trial_data.get('start_date', 'Unknown')
            html = HTML_ALREADY_USED.format(
                trial_id=trial_id,
                trial_date=trial_date
            )
            self.send_html_response(html)
            return

        post_data = parse_post_data(self)

        name = post_data.get('name', [''])[0].strip()
        email = post_data.get('email', [''])[0].strip()
        password = post_data.get('password', [''])[0].strip()

        if not name or not email or not password:
            message = '<div class="message error">⚠️ All fields are required. Please fill in the form completely.</div>'
            html = HTML_SIGNUP_FORM.format(message=message)
            self.send_html_response(html)
            return

        if len(password) < 6:
            message = '<div class="message error">⚠️ Password must be at least 6 characters long.</div>'
            html = HTML_SIGNUP_FORM.format(message=message)
            self.send_html_response(html)
            return

        if '@' not in email or '.' not in email:
            message = '<div class="message error">⚠️ Please enter a valid email address.</div>'
            html = HTML_SIGNUP_FORM.format(message=message)
            self.send_html_response(html)
            return

        new_trial_id = str(uuid.uuid4())
        start_date = datetime.now()
        expiry_date = start_date + timedelta(days=30)

        TRIAL_USERS[new_trial_id] = {
            'name': name,
            'email': email,
            'trial_id': new_trial_id,
            'start_date': start_date.strftime('%Y-%m-%d %H:%M:%S'),
            'expiry_date': expiry_date.strftime('%Y-%m-%d'),
            'active': True
        }

        print(f"\n[NEW TRIAL] User: {name} | Email: {email} | Trial ID: {new_trial_id}")
        print(f"[TRIAL INFO] Total trials registered: {len(TRIAL_USERS)}\n")

        cookie = SimpleCookie()
        cookie['has_used_trial'] = new_trial_id
        cookie['has_used_trial']['path'] = '/'
        cookie['has_used_trial']['max-age'] = 365 * 24 * 60 * 60
        cookie['has_used_trial']['httponly'] = True
        cookie['has_used_trial']['samesite'] = 'Lax'

        cookie_str = cookie.output(header='').strip()

        html = HTML_SUCCESS.format(
            name=name,
            email=email,
            trial_id=new_trial_id,
            start_date=start_date.strftime('%Y-%m-%d %H:%M:%S'),
            expiry_date=expiry_date.strftime('%B %d, %Y')
        )

        self.send_html_response(html, extra_headers={'Set-Cookie': cookie_str})

                h1 {{ color: #333; }}
                .badge {{ background: #e8f5e9; color: #2e7d32; padding: 4px 12px; border-radius: 20px; font-size: 13px; }}
                .info {{ margin: 20px 0; padding: 16px; background: #f9f9f9; border-radius: 8px; }}
                a {{ color: #667eea; text-decoration: none; }}
            </style>
        </head>
        <body>
            <div class="card">
                <h1>Welcome, {user['name']}! 👋</h1>
                <span class="badge">✓ Trial Active</span>
                <div class="info">
                    <p><strong>Email:</strong> {user['email']}</p>
                    <p><strong>Trial ID:</strong> {user['trial_id']}</p>
                    <p><strong>Started:</strong> {user['start_date']}</p>
                    <p><strong>Expires:</strong> {user['expiry_date']}</p>
                </div>
                <p>Your dashboard is ready. Enjoy your 30-day free trial!</p>
                <br>
                <a href="/reset-trial">🔄 Reset Trial Cookie (for testing)</a>
            </div>
        </body>
        </html>
        """
        self.send_html_response(html)

        self.send_html_response(html)

        html = """
        <!DOCTYPE html>
        <html>
        <head>
            <title>Sign In</title>
            <style>
                body { font-family: 'Segoe UI', sans-serif; background: #f5f5f5; padding: 40px; text-align: center; }

                p { color: #666; }
                a { color: #667eea; }
            </style>
        </head>
        <body>
            <h1>🔐 Sign In</h1>
            <p>Sign in to your existing account.</p>
            <p><a href="/">← Back to Home</a></p>
        </body>
        </html>
        """
        self.send_html_response(html)

    def handle_reset_trial(self):
        cookie = SimpleCookie()
        cookie['has_used_trial'] = ''
        cookie['has_used_trial']['path'] = '/'
        cookie['has_used_trial']['max-age'] = 0
        cookie['has_used_trial']['expires'] = 'Thu, 01 Jan 1970 00:00:00 GMT'

        cookie_str = cookie.output(header='').strip()

        html = """
        <!DOCTYPE html>
        <html>
        <head>
            <title>Cookie Reset</title>
            <meta http-equiv="refresh" content="2;url=/">
            <style>
                body { font-family: 'Segoe UI', sans-serif; background: #f5f5f5; padding: 40px; text-align: center; }
                h1 { color: #333; }
                p { color: #666; }
            </style>
        </head>
        <body>
            <h1>🔄 Trial Cookie Reset</h1>
            <p>The <strong>has_used_trial</strong> cookie has been cleared.</p>
            <p>Redirecting to signup page...</p>
        </body>
        </html>
        """
        self.send_html_response(html, extra_headers={'Set-Cookie': cookie_str})


def run_server(host='localhost', port=8080):
    server_address = (host, port)
    httpd = HTTPServer(server_address, TrialSignupHandler)

    print("=" * 60)
    print("  Free Trial Signup Server")
    print("=" * 60)
    print(f"  Server running at: http://{host}:{port}")
    print(f"  Signup page:       http://{host}:{port}/signup")
    print(f"  Dashboard:         http://{host}:{port}/dashboard")
    print(f"  Reset trial:       http://{host}:{port}/reset-trial")
    print("=" * 60)
    print("\nCookie Logic:")
    print("  - First visit: Shows signup form")
    print("  - After signup: Sets 'has_used_trial' cookie")
    print("  - Return visit: Detects cookie, blocks new trial")
    print("  - /reset-trial: Clears cookie for testing")
    print("\nPress Ctrl+C to stop the server\n")

    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n\nServer stopped.")
        print(f"Total trial registrations: {len(TRIAL_USERS)}")
        if TRIAL_USERS:
            print("\nRegistered trials:")
            for tid, data in TRIAL_USERS.items():
                print(f"  - {data['name']} ({data['email']}) | ID: {tid[:8]}...")
        httpd.server_close()


if __name__ == '__main__':
    run_server()
