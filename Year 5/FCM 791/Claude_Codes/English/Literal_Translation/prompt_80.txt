@app.route('/upload', methods=['POST'])
def upload_yaml():
    global user_profile, conversation_history
    
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'No file provided'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'No file selected'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'Please upload a YAML file (.yaml or .yml)'})
    
    try:
        # Read and parse the YAML file
        yaml_content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(yaml_content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'Invalid YAML format. Expected a dictionary/object.'})
        
        # Store the profile
        user_profile = profile_data
        
        # Reset conversation history with new profile context
        conversation_history = []
        
        # Create initial message about the profile
        profile_summary = yaml.dump(profile_data, default_flow_style=False)
        
        # Add system context about the profile
        initial_message = f"I've loaded your profile configuration. Here's what I found:\n\n{profile_summary}\n\nHow can I help you with your profile setup?"
        
        # Add to conversation history
        conversation_history.append({
            "role": "user",
            "content": f"I've uploaded my profile configuration. Here's my YAML data:\n\n{yaml_content}"
        })
        
        # Get AI response about the profile
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="""You are a helpful profile setup assistant. When a user uploads their YAML configuration, 
            analyze it and provide helpful feedback about their profile setup. Be friendly and informative.
            Help users understand their configuration and suggest improvements if needed.""",
            messages=conversation_history
        )
        
        ai_response = response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": ai_response
        })
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'message': ai_response
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Invalid YAML syntax: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error processing file: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    global conversation_history, user_profile
    
    data = request.get_json()
    user_message = data.get('message', '')
    
    if not user_message:
        return jsonify({'error': 'No message provided'})
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # Prepare system message with profile context
    system_message = """You are a helpful profile setup assistant. You help users understand and manage their 
    profile configurations. Be friendly, informative, and provide practical advice."""
    
    if user_profile:
        profile_context = yaml.dump(user_profile, default_flow_style=False)
        system_message += f"\n\nThe user's current profile configuration is:\n{profile_context}"
    else:
        system_message += "\n\nNo profile has been uploaded yet. Encourage the user to upload a YAML configuration file."
    
    try:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_message,
            messages=conversation_history
        )
        
        ai_response = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": ai_response
        })
        
        return jsonify({'response': ai_response})
        
    except Exception as e:
        return jsonify({'error': str(e)})

if __name__ == '__main__':
    app.run(debug=True, port=5000)
