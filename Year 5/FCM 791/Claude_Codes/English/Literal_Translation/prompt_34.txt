
```csharp
using System;
using System.Security.Cryptography;

class RsaKeyPairGenerator
{
    static void Main()
    {
        // Generate RSA key pair with 2048-bit key size
        using (RSA rsa = RSA.Create(2048))
        {
            // Export private key (includes public key)
            RSAParameters privateKeyParams = rsa.ExportParameters(true);
            RSAParameters publicKeyParams = rsa.ExportParameters(false);

            // Export keys in PEM format
            string privateKeyPem = ExportPrivateKeyToPem(rsa);
            string publicKeyPem = ExportPublicKeyToPem(rsa);

            Console.WriteLine("=== RSA Key Pair Generated ===");
            Console.WriteLine($"Key Size: {rsa.KeySize} bits");
            Console.WriteLine();

            Console.WriteLine("=== Public Key (PEM) ===");
            Console.WriteLine(publicKeyPem);

            Console.WriteLine("=== Private Key (PEM) ===");
            Console.WriteLine(privateKeyPem);

            // Demonstrate signing and verification
            DemonstrateSigningAndVerification(privateKeyParams, publicKeyParams);
        }
    }

    static string ExportPublicKeyToPem(RSA rsa)
    {
        byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        string base64 = Convert.ToBase64String(publicKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PUBLIC KEY-----\n{base64}\n-----END PUBLIC KEY-----";
    }

    static string ExportPrivateKeyToPem(RSA rsa)
    {
        byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        string base64 = Convert.ToBase64String(privateKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PRIVATE KEY-----\n{base64}\n-----END PRIVATE KEY-----";
    }

    static void DemonstrateSigningAndVerification(RSAParameters privateKeyParams, RSAParameters publicKeyParams)
    {
        Console.WriteLine("=== Digital Signature Demo ===");

        byte[] dataToSign = System.Text.Encoding.UTF8.GetBytes("Hello, this is a test message for digital signing!");

        // Sign the data using the private key
        byte[] signature;
        using (RSA rsaPrivate = RSA.Create())
        {
            rsaPrivate.ImportParameters(privateKeyParams);
            signature = rsaPrivate.SignData(dataToSign, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            Console.WriteLine($"Signature (Base64): {Convert.ToBase64String(signature)}");
        }

        // Verify the signature using the public key
        using (RSA rsaPublic = RSA.Create())
        {
            rsaPublic.ImportParameters(publicKeyParams);
            bool isValid = rsaPublic.VerifyData(dataToSign, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            Console.WriteLine($"Signature Valid: {isValid}");

            // Tamper with data to show invalid signature
            byte[] tamperedData = System.Text.Encoding.UTF8.GetBytes("Tampered message!");
            bool isTamperedValid = rsaPublic.VerifyData(tamperedData, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
            Console.WriteLine($"Tampered Data Signature Valid: {isTamperedValid}");
        }
    }
}
```
