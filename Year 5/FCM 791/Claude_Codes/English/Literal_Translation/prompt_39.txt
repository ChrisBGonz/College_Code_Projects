import anthropic
import socket
import threading
import time
import random
import string

# XOR cipher key for "security"
XOR_KEY = 0x42  # Simple XOR key

def xor_encrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Encrypt/decrypt data using XOR cipher."""
    return bytes([b ^ key for b in data])

def xor_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Decrypt data using XOR cipher (same as encrypt for XOR)."""
    return xor_encrypt(data, key)

class LegacyServer:
    """Simulates a legacy server that communicates using XOR cipher."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.server_socket = None
        self.running = False
        
    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")
        
        # Accept connections in a separate thread
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections."""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Server: New connection from {address}")
                client_thread = threading.Thread(
                    target=self._handle_client,
                    args=(client_socket,)
                )
                client_thread.daemon = True
                client_thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Server error: {e}")
                    
    def _handle_client(self, client_socket: socket.socket):
        """Handle a client connection."""
        try:
            # Send welcome message (encrypted)
            welcome = "LEGACY_SERVER_v1.0 READY"
            encrypted_welcome = xor_encrypt(welcome.encode())
            client_socket.send(encrypted_welcome)
            
            while True:
                # Receive encrypted data
                data = client_socket.recv(1024)
                if not data:
                    break
                    
                # Decrypt the received data
                decrypted = xor_decrypt(data).decode('utf-8', errors='ignore')
                print(f"Server received: {decrypted}")
                
                # Process command and send response
                response = self._process_command(decrypted)
                encrypted_response = xor_encrypt(response.encode())
                client_socket.send(encrypted_response)
                
        except Exception as e:
            print(f"Server client handler error: {e}")
        finally:
            client_socket.close()
            
    def _process_command(self, command: str) -> str:
        """Process a command and return a response."""
        command = command.strip().upper()
        
        if command == "PING":
            return "PONG"
        elif command == "STATUS":
            return "STATUS:OPERATIONAL"
        elif command == "VERSION":
            return "VERSION:1.0.0-LEGACY"
        elif command.startswith("ECHO:"):
            return f"ECHO:{command[5:]}"
        elif command == "DATA":
            # Return some fake sensor data
            value = random.uniform(0, 100)
            return f"DATA:SENSOR_VALUE={value:.2f}"
        elif command == "QUIT":
            return "BYE"
        else:
            return f"ERROR:UNKNOWN_COMMAND:{command}"
            
    def stop(self):
        """Stop the server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("Legacy server stopped")


class LegacyClient:
    """Client that communicates with the legacy server using XOR cipher."""
    
    def __init__(self, host: str = 'localhost', port: int = 9999):
        self.host = host
        self.port = port
        self.socket = None
        
    def connect(self) -> str:
        """Connect to the legacy server and return welcome message."""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        
        # Receive welcome message
        encrypted_welcome = self.socket.recv(1024)
        welcome = xor_decrypt(encrypted_welcome).decode('utf-8', errors='ignore')
        return welcome
        
    def send_command(self, command: str) -> str:
        """Send an encrypted command and receive the response."""
        if not self.socket:
            raise ConnectionError("Not connected to server")
            
        # Encrypt and send command
        encrypted_command = xor_encrypt(command.encode())
        self.socket.send(encrypted_command)
        
        # Receive and decrypt response
        encrypted_response = self.socket.recv(1024)
        response = xor_decrypt(encrypted_response).decode('utf-8', errors='ignore')
        return response
        
    def disconnect(self):
        """Disconnect from the server."""
        if self.socket:
            try:
                self.send_command("QUIT")
            except:
                pass
            self.socket.close()
            self.socket = None


def demonstrate_xor_cipher():
    """Demonstrate the XOR cipher functionality."""
    print("\n=== XOR Cipher Demonstration ===")
    
    test_messages = [
        "Hello, Legacy Server!",
        "PING",
        "STATUS",
        "SECRET_DATA_123"
    ]
    
    for message in test_messages:
        original_bytes = message.encode()
        encrypted = xor_encrypt(original_bytes)
        decrypted = xor_decrypt(encrypted)
        
        print(f"\nOriginal:  '{message}'")
        print(f"Encrypted: {encrypted.hex()}")
        print(f"Decrypted: '{decrypted.decode()}'")
        print(f"Match: {original_bytes == decrypted}")


def use_claude_for_analysis(client: LegacyClient):
    """Use Claude to analyze the legacy server communication."""
    
    # Collect some data from the server
    print("\n=== Collecting Data from Legacy Server ===")
    
    commands = ["PING", "STATUS", "VERSION", "DATA", "DATA", "DATA"]
    responses = []
    
    for cmd in commands:
        response = client.send_command(cmd)
        responses.append({"command": cmd, "response": response})
        print(f"Command: {cmd} -> Response: {response}")
        time.sleep(0.1)
    
    # Use Claude to analyze the communication
    print("\n=== Using Claude to Analyze Communication ===")
    
    anthropic_client = anthropic.Anthropic()
    
    # Prepare the conversation data for Claude
    conversation_data = "\n".join([
        f"Command: {r['command']} -> Response: {r['response']}"
        for r in responses
    ])
    
    # Use streaming to get Claude's analysis
    print("\nClaude's Analysis (streaming):")
    print("-" * 50)
    
    with anthropic_client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""I'm communicating with a legacy server using a custom socket protocol with XOR cipher encryption. 
Here are the command-response pairs I've collected:

{conversation_data}

Please analyze this communication and:
1. Identify the protocol patterns
2. Assess the security implications of using XOR cipher
3. Suggest improvements for the protocol
4. Identify any potential issues or vulnerabilities

Keep your response concise and technical."""
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "-" * 50)
    
    # Ask Claude to generate a test script
    print("\n=== Generating Test Script with Claude ===")
    
    with anthropic_client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=512,
        messages=[
            {
                "role": "user",
                "content": """Based on the legacy server protocol (commands: PING, STATUS, VERSION, DATA, ECHO:<message>), 
generate a simple Python test function that validates the server is working correctly. 
The function should test at least 3 different commands and verify the responses.
Keep it brief and practical."""
            }
        ]
    ) as stream:
        print("Generated test function:")
        print("-" * 50)
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "-" * 50)


def main():
    """Main function to demonstrate the legacy server communication."""
    
    print("=== Legacy Server Communication with XOR Cipher ===\n")
    
    # Demonstrate XOR cipher
    demonstrate_xor_cipher()
    
    # Start the legacy server
    print("\n=== Starting Legacy Server ===")
    server = LegacyServer()
    server.start()
    time.sleep(0.5)  # Wait for server to start
    
    # Connect client
    print("\n=== Connecting Client ===")
    client = LegacyClient()
    
    try:
        welcome = client.connect()
        print(f"Connected! Server says: {welcome}")
        
        # Test basic commands
        print("\n=== Testing Basic Commands ===")
        
        test_commands = [
            "PING",
            "STATUS", 
            "VERSION",
            "ECHO:Hello World",
            "DATA"
        ]
        
        for cmd in test_commands:
            response = client.send_command(cmd)
            print(f"  {cmd} -> {response}")
            time.sleep(0.1)
        
        # Use Claude for analysis
        use_claude_for_analysis(client)
        
    except Exception as e:
        print(f"Error: {e}")
    finally:
        # Cleanup
        print("\n=== Cleanup ===")
        client.disconnect()
        server.stop()
        print("Done!")


if __name__ == "__main__":
    main()
