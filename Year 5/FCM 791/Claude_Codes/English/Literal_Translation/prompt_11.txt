
import sys
from lxml import etree


def extract_titles_from_xml(file_path):
    """
    Parse an XML file and extract all 'title' tag contents.
    
    Args:
        file_path (str): Path to the XML file to parse
        
    Returns:
        list: List of title strings found in the XML
    """
    titles = []
    
    try:
        # Parse the XML file
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        # Find all 'title' elements anywhere in the document
        # Using XPath to find all title elements regardless of namespace
        title_elements = root.findall('.//{*}title') or root.findall('.//title')
        
        if not title_elements:
            # Try with XPath for more comprehensive search
            title_elements = tree.xpath('//*[local-name()="title"]')
        
        if not title_elements:
            print("No 'title' tags found in the XML file.")
            return titles
        
        print(f"Found {len(title_elements)} 'title' tag(s):\n")
        
        for index, title in enumerate(title_elements, start=1):
            title_text = title.text
            if title_text:
                title_text = title_text.strip()
                titles.append(title_text)
                print(f"Title {index}: {title_text}")
            else:
                print(f"Title {index}: [Empty title tag]")
                titles.append("")
        
        return titles
    
    except etree.XMLSyntaxError as e:
        print(f"XML Syntax Error: The file is not valid XML.\nDetails: {e}")
        sys.exit(1)
    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    except PermissionError:
        print(f"Error: Permission denied when trying to read '{file_path}'.")
        sys.exit(1)
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
        sys.exit(1)


def get_file_path_from_user():
    """
    Prompt the user to enter the path to an XML file.
    
    Returns:
        str: The file path entered by the user
    """
    print("XML Title Extractor")
    print("=" * 40)
    
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
        print(f"Using file path from command line argument: {file_path}")
    else:
        file_path = input("Please enter the path to your XML file: ").strip()
        
        if not file_path:
            print("Error: No file path provided.")
            sys.exit(1)
    
    return file_path


def display_xml_structure(file_path):
    """
    Display the basic structure of the XML file for reference.
    
    Args:
        file_path (str): Path to the XML file
    """
    try:
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        print(f"\nXML File Structure:")
        print(f"Root element: <{root.tag}>")
        print(f"Number of direct children: {len(root)}")
        print("-" * 40)
    except Exception:
        pass  # Silently ignore errors in structure display


def main():
    # Get file path from user
    file_path = get_file_path_from_user()
    
    # Display XML structure
    display_xml_structure(file_path)
    
    # Extract titles
    print("\nExtracting 'title' tags...")
    print("-" * 40)
    titles = extract_titles_from_xml(file_path)
    
    # Summary
    print("-" * 40)
    print(f"\nSummary: Successfully extracted {len(titles)} title(s).")
    
    if titles:
        print("\nAll extracted titles:")
        for i, title in enumerate(titles, start=1):
            print(f"  {i}. {title if title else '[Empty]'}")
    
    return titles


if __name__ == "__main__":
    main()
