function sanitizeFileName(string $fileName): string {
    $fileName = preg_replace('/[^\w\s\-.]/', '', $fileName);
    $fileName = preg_replace('/\s+/', '_', $fileName);
    $fileName = trim($fileName, '.-_');
    return $fileName ?: 'unnamed_file';
}

function validateFile(array $file): array {
    $errors = [];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE   => 'File exceeds server upload limit.',
            UPLOAD_ERR_FORM_SIZE  => 'File exceeds form upload limit.',
            UPLOAD_ERR_PARTIAL    => 'File was only partially uploaded.',
            UPLOAD_ERR_NO_FILE    => 'No file was uploaded.',
            UPLOAD_ERR_NO_TMP_DIR => 'Missing temporary folder.',
            UPLOAD_ERR_CANT_WRITE => 'Failed to write file to disk.',
            UPLOAD_ERR_EXTENSION  => 'A PHP extension stopped the upload.',
        ];
        $errors[] = $errorMessages[$file['error']] ?? 'Unknown upload error.';
        return $errors;
    }

    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'File size exceeds the maximum allowed size of ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'File is empty.';
    }

    $extension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($extension, ALLOWED_EXTENSIONS)) {
        $errors[] = 'Invalid file extension. Only PDF files are allowed.';
    }

    if (!is_uploaded_file($file['tmp_name'])) {
        $errors[] = 'Security check failed: File was not uploaded via HTTP POST.';
        return $errors;
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);
    if (!in_array($mimeType, ALLOWED_MIME_TYPES)) {
        $errors[] = 'Invalid file type. Only PDF files are allowed. Detected: ' . htmlspecialchars($mimeType);
    }

    $handle = fopen($file['tmp_name'], 'rb');
    if ($handle) {
        $header = fread($handle, 4);
        fclose($handle);
        if ($header !== '%PDF') {
            $errors[] = 'File does not appear to be a valid PDF.';
        }
    }

    return $errors;
}

function generateUniqueFileName(string $originalName, string $employeeId): string {
    $extension = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
    $baseName = pathinfo($originalName, PATHINFO_FILENAME);
    $sanitizedBase = sanitizeFileName($baseName);
    $timestamp = date('Ymd_His');
    $uniqueId = substr(bin2hex(random_bytes(4)), 0, 8);
    return "{$employeeId}_{$timestamp}_{$uniqueId}_{$sanitizedBase}.{$extension}";
}

function logUpload(array $logData): void {
    $logEntry = date('Y-m-d H:i:s') . ' | ' . implode(' | ', $logData) . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function getUploadedFiles(): array {
    $files = [];
    if (!is_dir(UPLOAD_DIR)) return $files;

    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isDot() || $fileInfo->getExtension() !== 'pdf') continue;
        $files[] = [
            'name'     => $fileInfo->getFilename(),
            'size'     => $fileInfo->getSize(),
            'modified' => $fileInfo->getMTime(),
            'path'     => $fileInfo->getPathname(),
        ];
    }

    usort($files, fn($a, $b) => $b['modified'] - $a['modified']);
    return $files;
}

function formatFileSize(int $bytes): string {
    if ($bytes >= 1048576) return round($bytes / 1048576, 2) . ' MB';
    if ($bytes >= 1024) return round($bytes / 1024, 2) . ' KB';
    return $bytes . ' B';
}

function generateCsrfToken(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCsrfToken(string $token): bool {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// Handle file download
if (isset($_GET['download']) && !empty($_GET['download'])) {
    $requestedFile = basename($_GET['download']);
    $filePath = UPLOAD_DIR . $requestedFile;

    if (file_exists($filePath) && pathinfo($filePath, PATHINFO_EXTENSION) === 'pdf') {
        header('Content-Type: application/pdf');
        header('Content-Disposition: attachment; filename="' . $requestedFile . '"');
        header('Content-Length: ' . filesize($filePath));
        header('Cache-Control: private, no-cache, no-store, must-revalidate');
        readfile($filePath);
        exit;
    } else {
        $message = 'File not found or invalid file type.';
        $messageType = 'error';
    }
}

// Handle file deletion
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'delete') {
    if (!validateCsrfToken($_POST['csrf_token'] ?? '')) {
        $message = 'Invalid security token. Please try again.';
        $messageType = 'error';
    } else {
        $fileToDelete = basename($_POST['file_name'] ?? '');
        $filePath = UPLOAD_DIR . $fileToDelete;

        if (file_exists($filePath) && pathinfo($filePath, PATHINFO_EXTENSION) === 'pdf') {
            if (unlink($filePath)) {
                logUpload(['DELETE', $employee['id'], $employee['name'], $fileToDelete, 'SUCCESS']);
                $message = 'File deleted successfully.';
                $messageType = 'success';
            } else {
                $message = 'Failed to delete the file.';
                $messageType = 'error';
            }
        } else {
            $message = 'File not found.';
            $messageType = 'error';
        }
        $uploadedFiles = getUploadedFiles();
    }
}

// Handle file upload
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'upload') {
    if (!validateCsrfToken($_POST['csrf_token'] ?? '')) {
        $message = 'Invalid security token. Please refresh the page and try again.';
        $messageType = 'error';
    } elseif (!isset($_FILES['pdf_files']) || empty($_FILES['pdf_files']['name'][0])) {
        $message = 'Please select at least one PDF file to upload.';
        $messageType = 'error';
    } else {
        $files = $_FILES['pdf_files'];
        $fileCount = count($files['name']);
        $successCount = 0;
        $errorMessages = [];

        for ($i = 0; $i < $fileCount; $i++) {
            $file = [
                'name'     => $files['name'][$i],
                'type'     => $files['type'][$i],
                'tmp_name' => $files['tmp_name'][$i],
                'error'    => $files['error'][$i],
                'size'     => $files['size'][$i],
            ];

            $validationErrors = validateFile($file);

            if (!empty($validationErrors)) {
                $errorMessages[] = htmlspecialchars($file['name']) . ': ' . implode(', ', $validationErrors);
                logUpload(['UPLOAD_FAIL', $employee['id'], $employee['name'], $file['name'], implode('; ', $validationErrors)]);
                continue;
            }

            $newFileName = generateUniqueFileName($file['name'], $employee['id']);
            $destination = UPLOAD_DIR . $newFileName;

            if (move_uploaded_file($file['tmp_name'], $destination)) {
                chmod($destination, 0644);
                $successCount++;
                logUpload(['UPLOAD_SUCCESS', $employee['id'], $employee['name'], $file['name'], $newFileName, formatFileSize($file['size'])]);
            } else {
                $errorMessages[] = htmlspecialchars($file['name']) . ': Failed to save file to server.';
                logUpload(['UPLOAD_FAIL', $employee['id'], $employee['name'], $file['name'], 'move_uploaded_file failed']);
            }
        }

        if ($successCount > 0 && empty($errorMessages)) {
            $message = $successCount . ' file(s) uploaded successfully.';
            $messageType = 'success';
        } elseif ($successCount > 0) {
            $message = $successCount . ' file(s) uploaded successfully. Errors: ' . implode(' | ', $errorMessages);
            $messageType = 'warning';
        } else {
            $message = 'Upload failed. ' . implode(' | ', $errorMessages);
            $messageType = 'error';
        }

        $uploadedFiles = getUploadedFiles();
    }
}
