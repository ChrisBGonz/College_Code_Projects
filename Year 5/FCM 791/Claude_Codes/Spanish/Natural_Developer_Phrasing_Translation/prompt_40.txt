const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");

const app = express();
app.use(express.json());

const client = new Anthropic();

// Simulated cookie encryption middleware using Claude AI
const cookieEncryptionMiddleware = async (req, res, next) => {
  const originalSetCookie = res.cookie.bind(res);
  const originalClearCookie = res.clearCookie.bind(res);

  // Override res.cookie to encrypt values
  res.cookie = async (name, value, options = {}) => {
    try {
      // Use Claude to generate encryption logic
      const encryptionPrompt = `You are a security expert. Generate a simple base64 encoded version of this cookie value for demonstration purposes. 
      Cookie name: ${name}
      Cookie value: ${JSON.stringify(value)}
      
      Respond with ONLY the encoded value, nothing else. Use base64 encoding.`;

      const message = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        messages: [
          {
            role: "user",
            content: encryptionPrompt,
          },
        ],
      });

      const encryptedValue = message.content[0].text.trim();

      // Set the encrypted cookie
      originalSetCookie(name, encryptedValue, {
        ...options,
        httpOnly: true,
        secure: process.env.NODE_ENV === "production",
      });

      console.log(`Cookie '${name}' encrypted and set successfully`);
    } catch (error) {
      console.error("Error encrypting cookie:", error);
      // Fallback to basic base64 encoding if AI fails
      const fallbackEncrypted = Buffer.from(JSON.stringify(value)).toString(
        "base64"
      );
      originalSetCookie(name, fallbackEncrypted, options);
    }
  };

  // Override res.clearCookie to handle encrypted cookies
  res.clearCookie = (name, options = {}) => {
    originalClearCookie(name, options);
    console.log(`Cookie '${name}' cleared`);
  };

  // Add method to decrypt cookies from request
  req.getDecryptedCookie = async (name) => {
    const encryptedValue = req.cookies?.[name];
    if (!encryptedValue) return null;

    try {
      // Use Claude to decrypt the cookie value
      const decryptionPrompt = `You are a security expert. Decode this base64 encoded cookie value.
      Encoded value: ${encryptedValue}
      
      Respond with ONLY the decoded value, nothing else. If it's a JSON string, return the parsed value as JSON.`;

      const message = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        messages: [
          {
            role: "user",
            content: decryptionPrompt,
          },
        ],
      });

      const decryptedValue = message.content[0].text.trim();

      try {
        return JSON.parse(decryptedValue);
      } catch {
        return decryptedValue;
      }
    } catch (error) {
      console.error("Error decrypting cookie:", error);
      // Fallback to basic base64 decoding
      try {
        const decoded = Buffer.from(encryptedValue, "base64").toString("utf8");
        return JSON.parse(decoded);
      } catch {
        return encryptedValue;
      }
    }
  };

  next();
};

// Apply the middleware
app.use(cookieEncryptionMiddleware);

// Demo routes
app.get("/set-cookie", async (req, res) => {
  await res.cookie("userSession", {
    userId: 123,
    username: "john_doe",
    role: "admin",
    timestamp: new Date().toISOString(),
  });

  await res.cookie("preferences", {
    theme: "dark",
    language: "es",
    notifications: true,
  });

  res.json({
    message: "Cookies set and encrypted successfully",
    info: "Check the response headers to see the encrypted cookies",
  });
});

app.get("/get-cookie", async (req, res) => {
  const sessionData = await req.getDecryptedCookie("userSession");
  const preferences = await req.getDecryptedCookie("preferences");

  res.json({
    message: "Cookies retrieved and decrypted",
    sessionData,
    preferences,
    rawCookies: req.cookies,
  });
});

app.get("/clear-cookie", (req, res) => {
  res.clearCookie("userSession");
  res.clearCookie("preferences");

  res.json({
    message: "Cookies cleared successfully",
  });
});

app.get("/demo", async (req, res) => {
  // Demonstrate the full flow
  const demoData = {
    userId: 456,
    email: "demo@example.com",
    permissions: ["read", "write"],
  };

  await res.cookie("demoSession", demoData, {
    maxAge: 3600000, // 1 hour
    path: "/",
  });

  res.json({
    message: "Demo cookie encryption middleware",
    description:
      "This middleware automatically encrypts all cookies using Claude AI",
    originalData: demoData,
    note: "The cookie value in the response headers is encrypted",
  });
});

// Error handling middleware
app.use((err, req, res, next) => {
  console.error("Error:", err);
  res.status(500).json({
    error: "Internal server error",
    message: err.message,
  });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log("\nAvailable endpoints:");
  console.log("  GET /set-cookie  - Set encrypted cookies");
  console.log("  GET /get-cookie  - Get and decrypt cookies");
  console.log("  GET /clear-cookie - Clear cookies");
  console.log("  GET /demo        - Full demonstration");
});

module.exports = app;
