from flask import Flask, request, jsonify, make_response
import secrets
import hashlib
import time

app = Flask(__name__)

# Simple user database (in production, use a real database)
USERS = {
    "admin": hashlib.sha256("password123".encode()).hexdigest(),
    "user1": hashlib.sha256("mypassword".encode()).hexdigest(),
}

# Store active sessions (in production, use Redis or a database)
active_sessions = {}


def generate_auth_token():
    """Generate a secure random token."""
    return secrets.token_hex(32)


def hash_password(password):
    """Hash a password using SHA-256."""
    return hashlib.sha256(password.encode()).hexdigest()


@app.route("/")
def index():
    """Home page - check if user is authenticated."""
    auth_token = request.cookies.get("auth_token")

    if auth_token and auth_token in active_sessions:
        session_data = active_sessions[auth_token]
        # Check if session is still valid (24 hours)
        if time.time() - session_data["created_at"] < 86400:
            return jsonify(
                {
                    "message": f"Welcome back, {session_data['username']}!",
                    "authenticated": True,
                    "username": session_data["username"],
                }
            )
        else:
            # Session expired, remove it
            del active_sessions[auth_token]

    return jsonify(
        {
            "message": "Please login to access this application",
            "authenticated": False,
        }
    )


@app.route("/login", methods=["POST"])
def login():
    """Login endpoint - validates credentials and sets auth_token cookie."""
    data = request.get_json()

    if not data:
        return jsonify({"error": "No data provided"}), 400

    username = data.get("username")
    password = data.get("password")

    if not username or not password:
        return jsonify({"error": "Username and password are required"}), 400

    # Validate credentials
    if username not in USERS:
        return jsonify({"error": "Invalid username or password"}), 401

    hashed_password = hash_password(password)
    if USERS[username] != hashed_password:
        return jsonify({"error": "Invalid username or password"}), 401

    # Generate auth token
    auth_token = generate_auth_token()

    # Store session data
    active_sessions[auth_token] = {
        "username": username,
        "created_at": time.time(),
    }

    # Create response with cookie
    response = make_response(
        jsonify(
            {
                "message": f"Login successful! Welcome, {username}!",
                "username": username,
            }
        )
    )

    # Set the auth_token cookie
    response.set_cookie(
        "auth_token",
        auth_token,
        max_age=86400,  # 24 hours
        httponly=True,  # Prevent JavaScript access
        secure=False,  # Set to True in production with HTTPS
        samesite="Lax",  # CSRF protection
    )

    return response


@app.route("/logout", methods=["POST"])
def logout():
    """Logout endpoint - removes the auth_token cookie and invalidates session."""
    auth_token = request.cookies.get("auth_token")

    if auth_token and auth_token in active_sessions:
        del active_sessions[auth_token]

    response = make_response(jsonify({"message": "Logged out successfully"}))

    # Remove the cookie by setting it with an expired date
    response.set_cookie("auth_token", "", max_age=0, expires=0)

    return response


@app.route("/profile")
def profile():
    """Protected endpoint - requires valid auth_token cookie."""
    auth_token = request.cookies.get("auth_token")

    if not auth_token:
        return jsonify({"error": "Authentication required", "authenticated": False}), 401

    if auth_token not in active_sessions:
        return jsonify({"error": "Invalid or expired session", "authenticated": False}), 401

    session_data = active_sessions[auth_token]

    # Check if session is still valid
    if time.time() - session_data["created_at"] >= 86400:
        del active_sessions[auth_token]
        return jsonify({"error": "Session expired", "authenticated": False}), 401

    return jsonify(
        {
            "message": "Profile data",
            "username": session_data["username"],
            "session_created": session_data["created_at"],
            "authenticated": True,
        }
    )


@app.route("/check-auth")
def check_auth():
    """Check if the current user is authenticated."""
    auth_token = request.cookies.get("auth_token")

    if not auth_token:
        return jsonify({"authenticated": False, "message": "No auth token found"})

    if auth_token not in active_sessions:
        return jsonify({"authenticated": False, "message": "Invalid or expired token"})

    session_data = active_sessions[auth_token]

    # Check if session is still valid
    if time.time() - session_data["created_at"] >= 86400:
        del active_sessions[auth_token]
        return jsonify({"authenticated": False, "message": "Session expired"})

    return jsonify(
        {
            "authenticated": True,
            "username": session_data["username"],
            "message": "User is authenticated",
        }
    )


if __name__ == "__main__":
    print("Flask Authentication App with Cookie Session")
    print("=" * 50)
    print("Available endpoints:")
    print("  GET  /          - Home page")
    print("  POST /login     - Login with username and password")
    print("  POST /logout    - Logout and clear session")
    print("  GET  /profile   - Protected profile page")
    print("  GET  /check-auth - Check authentication status")
    print("\nTest credentials:")
    print("  Username: admin, Password: password123")
    print("  Username: user1, Password: mypassword")
    print("\nExample login request:")
    print('  curl -X POST http://localhost:5000/login \\')
    print('       -H "Content-Type: application/json" \\')
    print('       -d \'{"username": "admin", "password": "password123"}\' \\')
    print("       -c cookies.txt")
    print("\nExample profile request (with cookie):")
    print("  curl http://localhost:5000/profile -b cookies.txt")
    print("=" * 50)

    app.run(debug=True, host="0.0.0.0", port=5000)
