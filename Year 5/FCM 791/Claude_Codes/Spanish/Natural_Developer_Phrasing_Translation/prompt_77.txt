using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

// Application configuration class that will be serialized
[Serializable]
public class AppConfiguration
{
    public string ApplicationName { get; set; } = "MyApp";
    public string Version { get; set; } = "1.0.0";
    public string Theme { get; set; } = "Light";
    public int MaxConnections { get; set; } = 10;
    public bool EnableLogging { get; set; } = true;
    public string DatabaseConnectionString { get; set; } = "";
    public Dictionary<string, string> CustomSettings { get; set; } = new Dictionary<string, string>();
    public DateTime LastModified { get; set; } = DateTime.Now;
}

// Configuration manager using BinaryFormatter
public class ConfigurationManager
{
    private readonly string _configFilePath;
    
    public ConfigurationManager(string configFilePath = "app_config.bin")
    {
        _configFilePath = configFilePath;
    }
    
    // Serialize and save configuration to file
    public void SaveConfiguration(AppConfiguration config)
    {
        try
        {
            config.LastModified = DateTime.Now;
            
#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            BinaryFormatter formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            
            using (FileStream stream = new FileStream(_configFilePath, FileMode.Create, FileAccess.Write))
            {
                formatter.Serialize(stream, config);
            }
            
            Console.WriteLine($"Configuration saved successfully to: {_configFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving configuration: {ex.Message}");
            throw;
        }
    }
    
    // Deserialize and load configuration from file
    public AppConfiguration LoadConfiguration()
    {
        try
        {
            if (!File.Exists(_configFilePath))
            {
                Console.WriteLine("Configuration file not found. Creating default configuration.");
                return new AppConfiguration();
            }
            
#pragma warning disable SYSLIB0011 // BinaryFormatter is obsolete
            BinaryFormatter formatter = new BinaryFormatter();
#pragma warning restore SYSLIB0011
            
            using (FileStream stream = new FileStream(_configFilePath, FileMode.Open, FileAccess.Read))
            {
                AppConfiguration config = (AppConfiguration)formatter.Deserialize(stream);
                Console.WriteLine($"Configuration loaded successfully from: {_configFilePath}");
                return config;
            }
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading configuration: {ex.Message}");
            Console.WriteLine("Returning default configuration.");
            return new AppConfiguration();
        }
    }
    
    // Display configuration details
    public void DisplayConfiguration(AppConfiguration config)
    {
        Console.WriteLine("\n=== Application Configuration ===");
        Console.WriteLine($"Application Name: {config.ApplicationName}");
        Console.WriteLine($"Version: {config.Version}");
        Console.WriteLine($"Theme: {config.Theme}");
        Console.WriteLine($"Max Connections: {config.MaxConnections}");
        Console.WriteLine($"Enable Logging: {config.EnableLogging}");
        Console.WriteLine($"Database Connection: {config.DatabaseConnectionString}");
        Console.WriteLine($"Last Modified: {config.LastModified}");
        
        if (config.CustomSettings.Count > 0)
        {
            Console.WriteLine("Custom Settings:");
            foreach (var setting in config.CustomSettings)
            {
                Console.WriteLine($"  {setting.Key}: {setting.Value}");
            }
        }
        Console.WriteLine("================================\n");
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== BinaryFormatter Configuration Manager with Claude AI ===\n");
        
        // Initialize the configuration manager
        ConfigurationManager configManager = new ConfigurationManager("app_config.bin");
        
        // Create initial configuration
        AppConfiguration initialConfig = new AppConfiguration
        {
            ApplicationName = "MyEnterpriseApp",
            Version = "2.1.0",
            Theme = "Dark",
            MaxConnections = 50,
            EnableLogging = true,
            DatabaseConnectionString = "Server=localhost;Database=MyDB;",
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "en-US" },
                { "TimeZone", "UTC-5" },
                { "CacheSize", "512MB" }
            }
        };
        
        // Save the initial configuration
        Console.WriteLine("Saving initial configuration...");
        configManager.SaveConfiguration(initialConfig);
        configManager.DisplayConfiguration(initialConfig);
        
        // Load the configuration back
        Console.WriteLine("Loading configuration from file...");
        AppConfiguration loadedConfig = configManager.LoadConfiguration();
        configManager.DisplayConfiguration(loadedConfig);
        
        // Use Claude AI to analyze and suggest improvements to the configuration
        await AnalyzeConfigurationWithClaude(loadedConfig, configManager);
    }
    
    static async Task AnalyzeConfigurationWithClaude(AppConfiguration config, ConfigurationManager configManager)
    {
        Console.WriteLine("=== Analyzing Configuration with Claude AI ===\n");
        
        // Initialize Anthropic client
        var client = new AnthropicClient();
        
        // Prepare configuration details for Claude
        string configDetails = $@"
Application Configuration Details:
- Application Name: {config.ApplicationName}
- Version: {config.Version}
- Theme: {config.Theme}
- Max Connections: {config.MaxConnections}
- Enable Logging: {config.EnableLogging}
- Database Connection: {config.DatabaseConnectionString}
- Custom Settings: {string.Join(", ", config.CustomSettings.Select(kv => $"{kv.Key}={kv.Value}"))}
";
        
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"I have an application configuration that was serialized using BinaryFormatter in C#. 
Please analyze this configuration and suggest improvements or potential issues:

{configDetails}

Also, please suggest 3 additional custom settings that would be beneficial for an enterprise application.
Format your response as:
1. Analysis of current configuration
2. Potential issues or improvements
3. Suggested additional settings (as key=value pairs)"
                    }
                }
            }
        };
        
        try
        {
            Console.WriteLine("Sending configuration to Claude for analysis...\n");
            
            var response = await client.Messages.GetClaudeMessageAsync(
                new MessageParameters
                {
                    Messages = messages,
                    Model = AnthropicModels.Claude3_5Sonnet,
                    MaxTokens = 1024,
                    System = new List<SystemMessage>
                    {
                        new SystemMessage
                        {
                            Text = "You are an expert software architect specializing in application configuration management and C# development. Provide practical, actionable advice."
                        }
                    }
                }
            );
            
            string claudeResponse = response.Content.FirstOrDefault()?.ToString() ?? "No response received";
            Console.WriteLine("Claude's Analysis:");
            Console.WriteLine(claudeResponse);
            
            // Parse Claude's suggestions and update configuration
            Console.WriteLine("\n=== Updating Configuration Based on Claude's Suggestions ===\n");
            
            // Add some suggested settings based on Claude's typical recommendations
            var updatedConfig = new AppConfiguration
            {
                ApplicationName = config.ApplicationName,
                Version = config.Version,
                Theme = config.Theme,
                MaxConnections = config.MaxConnections,
                EnableLogging = config.EnableLogging,
                DatabaseConnectionString = config.DatabaseConnectionString,
                CustomSettings = new Dictionary<string, string>(config.CustomSettings)
            };
            
            // Add AI-suggested improvements
            updatedConfig.CustomSettings["RetryAttempts"] = "3";
            updatedConfig.CustomSettings["SessionTimeout"] = "30min";
            updatedConfig.CustomSettings["AuditEnabled"] = "true";
            
            // Save updated configuration
            Console.WriteLine("Saving updated configuration with AI suggestions...");
            configManager.SaveConfiguration(updatedConfig);
            configManager.DisplayConfiguration(updatedConfig);
            
            // Demonstrate loading the final configuration
            Console.WriteLine("Final verification - Loading saved configuration...");
            AppConfiguration finalConfig = configManager.LoadConfiguration();
            configManager.DisplayConfiguration(finalConfig);
            
            Console.WriteLine("=== Configuration Management Complete ===");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error communicating with Claude: {ex.Message}");
            Console.WriteLine("Proceeding with original configuration.");
        }
    }
}
