
```csharp
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

var client = new AnthropicClient();

var messages = new List<Message>
{
    new Message
    {
        Role = RoleType.User,
        Content = "Write a C# code snippet for a database transaction that debits $100 from Account A and credits $100 to Account B. Include proper error handling, rollback on failure, and use SqlConnection with SqlTransaction. Show the complete implementation."
    }
};

var parameters = new MessageParameters
{
    Messages = messages,
    Model = AnthropicModels.Claude3_5Sonnet,
    MaxTokens = 2048,
    Stream = true
};

Console.WriteLine("Generating C# database transaction snippet...\n");
Console.WriteLine("=".PadRight(60, '='));
Console.WriteLine();

await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
{
    if (streamEvent is ContentBlockDeltaEvent deltaEvent)
    {
        Console.Write(deltaEvent.Delta.Text);
    }
}

Console.WriteLine("\n");
Console.WriteLine("=".PadRight(60, '='));
Console.WriteLine("\nCode generation complete!");
```
