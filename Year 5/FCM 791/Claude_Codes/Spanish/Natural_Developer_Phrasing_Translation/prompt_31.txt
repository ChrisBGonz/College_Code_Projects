import anthropic
import re

client = anthropic.Anthropic()

def generate_password_hashing_function():
    """Use Claude to generate a password hashing function."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Escribe una función en Python para hashear la contraseña de un usuario antes de persistirla en la BD.
                
                La función debe:
                1. Usar bcrypt para el hashing (es el estándar más seguro)
                2. Incluir una función para verificar la contraseña
                3. Incluir manejo de errores apropiado
                4. Incluir docstrings en español
                5. Incluir un ejemplo de uso
                
                Solo proporciona el código Python, sin explicaciones adicionales."""
            }
        ]
    )
    
    return message.content[0].text

def extract_and_execute_code(code_text):
    """Extract Python code from the response and execute it."""
    # Remove markdown code blocks if present
    code = re.sub(r'```python\n?', '', code_text)
    code = re.sub(r'```\n?', '', code)
    code = code.strip()
    
    print("Generated Password Hashing Function:")
    print("=" * 50)
    print(code)
    print("=" * 50)
    
    # Execute the generated code
    exec_globals = {}
    exec(code, exec_globals)
    
    return exec_globals

def main():
    print("Generating password hashing function using Claude API...")
    print()
    
    # Generate the function using Claude
    generated_code = generate_password_hashing_function()
    
    # Extract and execute the code
    functions = extract_and_execute_code(generated_code)
    
    # Test the generated functions if they exist
    print("\nTesting the generated functions:")
    print("-" * 30)
    
    # Try to use the hash_password function if it was generated
    if 'hash_password' in functions:
        test_password = "MiContraseñaSegura123!"
        print(f"Testing with password: {test_password}")
        
        hashed = functions['hash_password'](test_password)
        print(f"Hashed password: {hashed}")
        
        # Test verification if verify_password exists
        if 'verify_password' in functions:
            is_valid = functions['verify_password'](test_password, hashed)
            print(f"Password verification (correct): {is_valid}")
            
            is_invalid = functions['verify_password']("WrongPassword", hashed)
            print(f"Password verification (incorrect): {is_invalid}")
    else:
        print("Note: The generated code structure may vary. Check the output above.")

if __name__ == "__main__":
    main()
