import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.IvParameterSpec;
import java.security.SecureRandom;
import java.util.Base64;

public class AESEncryptionWithBedrock {

    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    private static final String ALGORITHM = "AES";
    private static final String TRANSFORMATION = "AES/CBC/PKCS5Padding";
    private static final int KEY_SIZE = 256;
    private static final int IV_SIZE = 16;

    private final BedrockRuntimeClient bedrockClient;

    public AESEncryptionWithBedrock() {
        this.bedrockClient = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();
    }

    /**
     * Encrypts a sensitive string using AES algorithm
     * 
     * @param sensitiveData The string to encrypt
     * @param secretKey     The AES secret key
     * @return Base64 encoded encrypted string with IV prepended
     * @throws Exception If encryption fails
     */
    public String encryptString(String sensitiveData, SecretKey secretKey) throws Exception {
        // Generate random IV
        byte[] iv = new byte[IV_SIZE];
        SecureRandom random = new SecureRandom();
        random.nextBytes(iv);
        IvParameterSpec ivSpec = new IvParameterSpec(iv);

        // Initialize cipher for encryption
        Cipher cipher = Cipher.getInstance(TRANSFORMATION);
        cipher.init(Cipher.ENCRYPT_MODE, secretKey, ivSpec);

        // Encrypt the data
        byte[] encryptedData = cipher.doFinal(sensitiveData.getBytes("UTF-8"));

        // Combine IV and encrypted data
        byte[] combined = new byte[IV_SIZE + encryptedData.length];
        System.arraycopy(iv, 0, combined, 0, IV_SIZE);
        System.arraycopy(encryptedData, 0, combined, IV_SIZE, encryptedData.length);

        // Return Base64 encoded result
        return Base64.getEncoder().encodeToString(combined);
    }

    /**
     * Decrypts an AES encrypted string
     * 
     * @param encryptedData Base64 encoded encrypted string with IV prepended
     * @param secretKey     The AES secret key
     * @return Decrypted string
     * @throws Exception If decryption fails
     */
    public String decryptString(String encryptedData, SecretKey secretKey) throws Exception {
        // Decode Base64
        byte[] combined = Base64.getDecoder().decode(encryptedData);

        // Extract IV
        byte[] iv = new byte[IV_SIZE];
        System.arraycopy(combined, 0, iv, 0, IV_SIZE);
        IvParameterSpec ivSpec = new IvParameterSpec(iv);

        // Extract encrypted data
        byte[] encryptedBytes = new byte[combined.length - IV_SIZE];
        System.arraycopy(combined, IV_SIZE, encryptedBytes, 0, encryptedBytes.length);

        // Initialize cipher for decryption
        Cipher cipher = Cipher.getInstance(TRANSFORMATION);
        cipher.init(Cipher.DECRYPT_MODE, secretKey, ivSpec);

        // Decrypt and return
        byte[] decryptedData = cipher.doFinal(encryptedBytes);
        return new String(decryptedData, "UTF-8");
    }

    /**
     * Generates a new AES secret key
     * 
     * @return Generated SecretKey
     * @throws Exception If key generation fails
     */
    public SecretKey generateAESKey() throws Exception {
        KeyGenerator keyGenerator = KeyGenerator.getInstance(ALGORITHM);
        keyGenerator.init(KEY_SIZE, new SecureRandom());
        return keyGenerator.generateKey();
    }

    /**
     * Uses Amazon Bedrock to explain the encryption process
     * 
     * @param encryptedResult The encrypted result to explain
     * @return Explanation from Claude
     */
    public String explainEncryptionWithBedrock(String encryptedResult) {
        String prompt = String.format(
                "Explain briefly how AES-256-CBC encryption works and what the following encrypted result represents: %s. "
                        +
                        "Keep the explanation concise and technical.",
                encryptedResult.substring(0, Math.min(50, encryptedResult.length())) + "...");

        Message message = Message.builder()
                .role(ConversationRole.USER)
                .content(ContentBlock.fromText(prompt))
                .build();

        ConverseRequest request = ConverseRequest.builder()
                .modelId(MODEL_ID)
                .messages(message)
                .build();

        ConverseResponse response = bedrockClient.converse(request);

        return response.output().message().content().get(0).text();
    }

    public static void main(String[] args) {
        AESEncryptionWithBedrock encryptor = new AESEncryptionWithBedrock();

        try {
            // Generate AES key
            System.out.println("Generating AES-256 key...");
            SecretKey secretKey = encryptor.generateAESKey();
            System.out.println("Key generated successfully");

            // Sensitive data to encrypt
            String sensitiveData = "This is sensitive information: SSN=123-45-6789, Password=MySecret123!";
            System.out.println("\nOriginal data: " + sensitiveData);

            // Encrypt the data
            System.out.println("\nEncrypting data...");
            String encryptedResult = encryptor.encryptString(sensitiveData, secretKey);
            System.out.println("Encrypted result: " + encryptedResult);

            // Decrypt the data
            System.out.println("\nDecrypting data...");
            String decryptedResult = encryptor.decryptString(encryptedResult, secretKey);
            System.out.println("Decrypted result: " + decryptedResult);

            // Verify encryption/decryption
            boolean success = sensitiveData.equals(decryptedResult);
            System.out.println("\nEncryption/Decryption successful: " + success);

            // Use Bedrock to explain the encryption
            System.out.println("\nAsking Amazon Bedrock to explain the encryption...");
            String explanation = encryptor.explainEncryptionWithBedrock(encryptedResult);
            System.out.println("\nBedrock explanation:");
            System.out.println(explanation);

        } catch (Exception e) {
            System.err.println("Error during encryption/decryption: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
