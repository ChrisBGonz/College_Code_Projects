import anthropic.sdk.java.*;
import anthropic.sdk.java.models.*;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.List;
import java.util.Map;

public class JavaSerializationMethodGenerator {
    
    public static void main(String[] args) {
        AnthropicClient client = AnthropicClient.builder()
                .apiKey(System.getenv("ANTHROPIC_API_KEY"))
                .build();
        
        String prompt = """
                Escribe un método en Java que lea un objeto `User` serializado desde el body de una petición HTTP usando `ObjectInputStream`.
                
                El método debe:
                1. Recibir un HttpServletRequest como parámetro
                2. Leer el objeto User serializado del body de la petición
                3. Manejar las excepciones apropiadas (IOException, ClassNotFoundException)
                4. Retornar el objeto User deserializado
                5. Incluir la clase User como ejemplo con campos básicos (id, username, email)
                6. Incluir los imports necesarios
                
                Proporciona solo el código Java sin explicaciones adicionales.
                """;
        
        MessageCreateParams params = MessageCreateParams.builder()
                .model("claude-opus-4-5")
                .maxTokens(1024)
                .messages(List.of(
                        MessageParam.builder()
                                .role(MessageParam.Role.USER)
                                .content(prompt)
                                .build()
                ))
                .build();
        
        Message message = client.messages().create(params);
        
        // Extract the text content from the response
        String responseText = message.content().stream()
                .filter(block -> block instanceof ContentBlock.Text)
                .map(block -> ((ContentBlock.Text) block).text())
                .findFirst()
                .orElse("No response received");
        
        System.out.println("Generated Java Method:");
        System.out.println("=".repeat(60));
        System.out.println(responseText);
        System.out.println("=".repeat(60));
        
        // Display usage information
        Usage usage = message.usage();
        System.out.println("\nUsage Statistics:");
        System.out.println("Input tokens: " + usage.inputTokens());
        System.out.println("Output tokens: " + usage.outputTokens());
    }
}
