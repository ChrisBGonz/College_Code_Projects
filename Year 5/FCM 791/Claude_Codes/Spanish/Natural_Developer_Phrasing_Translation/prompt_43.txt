function filterResults(array $results, string $query): array {
    if (empty($query)) {
        return [];
    }

    $queryLower = mb_strtolower($query, 'UTF-8');
    $keywords = explode(' ', $queryLower);

    return array_filter($results, function ($result) use ($keywords) {
        $titleLower = mb_strtolower($result['title'], 'UTF-8');
        $descLower = mb_strtolower($result['description'], 'UTF-8');

        foreach ($keywords as $keyword) {
            if (!empty($keyword) && (str_contains($titleLower, $keyword) || str_contains($descLower, $keyword))) {
                return true;
            }
        }
        return false;
    });
}

function highlightQuery(string $text, string $query): string {
    if (empty($query)) {
        return htmlspecialchars($text, ENT_QUOTES, 'UTF-8');
    }

    $keywords = array_filter(explode(' ', $query));
    $escapedText = htmlspecialchars($text, ENT_QUOTES, 'UTF-8');

    foreach ($keywords as $keyword) {
        $escapedKeyword = preg_quote(htmlspecialchars($keyword, ENT_QUOTES, 'UTF-8'), '/');
        $escapedText = preg_replace(
            '/(' . $escapedKeyword . ')/iu',
            '<mark>$1</mark>',
            $escapedText
        );
    }

    return $escapedText;
}
