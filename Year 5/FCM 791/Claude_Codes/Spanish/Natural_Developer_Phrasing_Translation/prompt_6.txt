# Tool definitions for the multi-tenant SaaS app
tools = [
    {
        "name": "get_company_info",
        "description": "Retrieves basic information about a company using its Company ID",
        "input_schema": {
            "type": "object",
            "properties": {
                "company_id": {
                    "type": "string",
                    "description": "The unique identifier for the company (e.g., COMP001)"
                }
            },
            "required": ["company_id"]
        }
    },
    {
        "name": "get_financial_data",
        "description": "Retrieves financial data for a company including revenue, profit, and growth rate",
        "input_schema": {
            "type": "object",
            "properties": {
                "company_id": {
                    "type": "string",
                    "description": "The unique identifier for the company"
                }
            },
            "required": ["company_id"]
        }
    },
    {
        "name": "get_market_analysis",
        "description": "Provides market analysis and competitive positioning for a company",
        "input_schema": {
            "type": "object",
            "properties": {
                "company_id": {
                    "type": "string",
                    "description": "The unique identifier for the company"
                }
            },
            "required": ["company_id"]
        }
    },
    {
        "name": "generate_annual_report",
        "description": "Generates a comprehensive annual report for a company",
        "input_schema": {
            "type": "object",

                "company_id": {
                    "type": "string",
                    "description": "The unique identifier for the company"
                },
                "include_sections": {
                    "type": "array",
                    "items": {"type": "string"},
                    "description": "List of sections to include in the report (e.g., ['financial', 'market', 'operations'])"

            },
            "required": ["company_id"]

]

def get_company_info(company_id: str) -> dict:
    """Retrieves basic company information"""
    if company_id not in COMPANY_DATABASE:
        return {"error": f"Company with ID {company_id} not found"}
    
    company = COMPANY_DATABASE[company_id]
    return {
        "company_id": company_id,
        "name": company["name"],
        "industry": company["industry"],
        "founded": company["founded"],
        "employees": company["employees"],
        "ceo": company["ceo"],
        "headquarters": company["headquarters"],
        "fiscal_year": company["fiscal_year"]
    }

def get_financial_data(company_id: str) -> dict:
    """Retrieves financial data for a company"""
    if company_id not in COMPANY_DATABASE:
        return {"error": f"Company with ID {company_id} not found"}
    
    company = COMPANY_DATABASE[company_id]
    return {
        "company_id": company_id,
        "fiscal_year": company["fiscal_year"],
        "revenue": company["revenue"],
        "net_profit": company["net_profit"],
        "profit_margin": round((company["net_profit"] / company["revenue"]) * 100, 2),
        "growth_rate": company["growth_rate"],
        "revenue_formatted": f"${company['revenue']:,.2f}",
        "net_profit_formatted": f"${company['net_profit']:,.2f}"
    }

def get_market_analysis(company_id: str) -> dict:
    """Provides market analysis for a company"""
    if company_id not in COMPANY_DATABASE:
        return {"error": f"Company with ID {company_id} not found"}
    
    company = COMPANY_DATABASE[company_id]
    return {
        "company_id": company_id,
        "markets": company["markets"],
        "products": company["products"],
        "market_presence": len(company["markets"]),
        "product_portfolio_size": len(company["products"]),
        "industry": company["industry"]
    }

def generate_annual_report(company_id: str, include_sections: list = None) -> dict:
    """Generates a comprehensive annual report"""
    if company_id not in COMPANY_DATABASE:
        return {"error": f"Company with ID {company_id} not found"}
    
    company = COMPANY_DATABASE[company_id]
    
    if include_sections is None:
        include_sections = ["financial", "market", "operations", "leadership"]
    
    report = {
        "report_type": "Annual Report",
        "company_id": company_id,
        "company_name": company["name"],
        "fiscal_year": company["fiscal_year"],
        "generated_date": datetime.now().strftime("%Y-%m-%d"),
        "sections": {}
    }
    
    if "financial" in include_sections:
        report["sections"]["financial_highlights"] = {
            "total_revenue": company["revenue"],
            "net_profit": company["net_profit"],
            "profit_margin": round((company["net_profit"] / company["revenue"]) * 100, 2),
            "year_over_year_growth": company["growth_rate"]
        }
    
    if "market" in include_sections:
        report["sections"]["market_overview"] = {
            "markets_served": company["markets"],
            "key_products": company["products"],
            "industry": company["industry"]
        }
    
    if "operations" in include_sections:
        report["sections"]["operational_metrics"] = {
            "total_employees": company["employees"],
            "headquarters": company["headquarters"],
            "years_in_operation": 2024 - company["founded"]
        }
    
    if "leadership" in include_sections:
        report["sections"]["leadership"] = {
            "ceo": company["ceo"],
            "company_founded": company["founded"]
        }
    
    return report

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "get_company_info":
        result = get_company_info(tool_input["company_id"])
    elif tool_name == "get_financial_data":
        result = get_financial_data(tool_input["company_id"])
    elif tool_name == "get_market_analysis":
        result = get_market_analysis(tool_input["company_id"])
    elif tool_name == "generate_annual_report":
        include_sections = tool_input.get("include_sections", None)
        result = generate_annual_report(tool_input["company_id"], include_sections)
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result, indent=2)

def query_annual_report(company_id: str, user_query: str = None) -> str:
    """
    Main function to query the annual report for a company
    
    Args:
        company_id: The unique identifier for the company
        user_query: Optional specific query about the report
    
    Returns:
        The annual report or specific information requested
    """
    if user_query is None:
        user_query = f"Please generate a comprehensive annual report for company {company_id}"
    
    print(f"\n{'='*60}")
    print(f"Multi-Tenant SaaS - Annual Report System")
    print(f"{'='*60}")
    print(f"Company ID: {company_id}")
    print(f"Query: {user_query}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": f"I need information about company {company_id}. {user_query}"
        }
    ]
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"Response stop reason: {response.stop_reason}")
        
        # Check if we're done
        if response.stop_reason == "end_turn":
            # Extract the final text response
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    break
            return final_response
        
        # Process tool calls if needed
        if response.stop_reason == "tool_use":
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process each tool call
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"\nCalling tool: {block.name}")
                    print(f"With inputs: {json.dumps(block.input, indent=2)}")
                    
                    # Execute the tool
                    result = process_tool_call(block.name, block.input)
                    print(f"Tool result: {result[:200]}..." if len(result) > 200 else f"Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            # Unexpected stop reason
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Unable to generate report"
