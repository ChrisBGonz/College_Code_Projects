
```javascript
const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");

const app = express();
app.use(express.json());

const anthropic = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Function to analyze error with Claude
async function analyzeErrorWithClaude(error, req) {
  const errorContext = {
    message: error.message,
    stack: error.stack,
    name: error.name,
    statusCode: error.statusCode || error.status || 500,
    path: req.path,
    method: req.method,
    body: req.body,
    query: req.query,
    headers: req.headers,
  };

  // Add user message to conversation history
  conversationHistory.push({
    role: "user",
    content: `Analyze this error and provide a helpful response for the client:
    
Error Details:
- Name: ${errorContext.name}
- Message: ${errorContext.message}
- Status Code: ${errorContext.statusCode}
- Request Path: ${errorContext.path}
- Request Method: ${errorContext.method}
- Stack Trace: ${errorContext.stack}

Please provide:
1. A user-friendly error message
2. Possible causes
3. Suggested solutions
4. Whether this is a client error (4xx) or server error (5xx)`,
  });

  const response = await anthropic.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are an expert error analysis assistant. When analyzing errors, provide clear, actionable information that helps both developers and end users understand what went wrong and how to fix it. Always be concise but thorough.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  // Add assistant response to conversation history
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

// Sample routes to demonstrate error handling
app.get("/", (req, res) => {
  res.json({
    message: "Error Handling Middleware Demo",
    endpoints: [
      "GET /error/sync - Throws a synchronous error",
      "GET /error/async - Throws an asynchronous error",
      "GET /error/custom - Throws a custom error with status code",
      "GET /error/validation - Throws a validation error",
      "POST /error/database - Simulates a database error",
    ],
  });
});

// Route that throws a synchronous error
app.get("/error/sync", (req, res) => {
  throw new Error("This is a synchronous error!");
});

// Route that throws an asynchronous error
app.get("/error/async", async (req, res, next) => {
  try {
    await Promise.reject(new Error("This is an asynchronous error!"));
  } catch (error) {
    next(error);
  }
});

// Route that throws a custom error with status code
app.get("/error/custom", (req, res, next) => {
  const error = new Error("Resource not found");
  error.statusCode = 404;
  next(error);
});

// Route that throws a validation error
app.get("/error/validation", (req, res, next) => {
  const error = new Error("Invalid input: email field is required");
  error.statusCode = 400;
  error.name = "ValidationError";
  next(error);
});

// Route that simulates a database error
app.post("/error/database", async (req, res, next) => {
  try {
    // Simulate database connection error
    throw new Error("ECONNREFUSED: Connection refused to database server");
  } catch (error) {
    error.statusCode = 503;
    error.name = "DatabaseError";
    next(error);
  }
});

// 404 handler for undefined routes
app.use((req, res, next) => {
  const error = new Error(`Route ${req.path} not found`);
  error.statusCode = 404;
  next(error);
});

// Global error handling middleware
app.use(async (err, req, res, next) => {
  console.error("Error caught by global middleware:", err);

  const statusCode = err.statusCode || err.status || 500;

  try {
    // Analyze error with Claude
    const aiAnalysis = await analyzeErrorWithClaude(err, req);

    // Return structured error response to client
    res.status(statusCode).json({
      success: false,
      error: {
        name: err.name || "Error",
        message: err.message,
        statusCode: statusCode,
        timestamp: new Date().toISOString(),
        path: req.path,
        method: req.method,
      },
      analysis: aiAnalysis,
      conversationId: conversationHistory.length,
    });
  } catch (aiError) {
    // If AI analysis fails, return basic error response
    console.error("Failed to analyze error with Claude:", aiError);

    res.status(statusCode).json({
      success: false,
      error: {
        name: err.name || "Error",
        message: err.message,
        statusCode: statusCode,
        timestamp: new Date().toISOString(),
        path: req.path,
        method: req.method,
      },
      analysis: "Error analysis unavailable at this time.",
    });
  }
});

// Handle unhandled promise rejections
process.on("unhandledRejection", (reason, promise) => {
  console.error("Unhandled Rejection at:", promise, "reason:", reason);
});

// Handle uncaught exceptions
process.on("uncaughtException", (error) => {
  console.error("Uncaught Exception:", error);
  process.exit(1);
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Visit http://localhost:${PORT} to see available endpoints`);
});

module.exports = app;
```
