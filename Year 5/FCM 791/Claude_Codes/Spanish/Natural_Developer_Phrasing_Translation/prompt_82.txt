
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import { Writable } from "stream";

const app = express();
const client = new Anthropic();

// Middleware to parse JSON bodies
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Logging middleware with Claude AI analysis
const requestLogger = async (req, res, next) => {
  const timestamp = new Date().toISOString();
  const requestData = {
    timestamp,
    method: req.method,
    url: req.url,
    path: req.path,
    query: req.query,
    headers: req.headers,
    body: req.body,
    ip: req.ip || req.connection.remoteAddress,
    userAgent: req.get("User-Agent"),
  };

  console.log("\n" + "=".repeat(80));
  console.log(`📥 INCOMING REQUEST - ${timestamp}`);
  console.log("=".repeat(80));
  console.log(`Method: ${requestData.method}`);
  console.log(`URL: ${requestData.url}`);
  console.log(`IP: ${requestData.ip}`);
  console.log(`User-Agent: ${requestData.userAgent}`);

  if (Object.keys(requestData.query).length > 0) {
    console.log(`Query Params: ${JSON.stringify(requestData.query, null, 2)}`);
  }

  if (requestData.body && Object.keys(requestData.body).length > 0) {
    console.log(`Body: ${JSON.stringify(requestData.body, null, 2)}`);
  }

  // Use Claude to analyze the request for debugging insights
  console.log("\n🤖 AI Analysis (Streaming):");
  console.log("-".repeat(40));

  try {
    const prompt = `Analyze this HTTP request for debugging purposes and provide insights:

Request Details:
- Method: ${requestData.method}
- URL: ${requestData.url}
- Query Parameters: ${JSON.stringify(requestData.query)}
- Body: ${JSON.stringify(requestData.body)}
- Headers (relevant ones): Content-Type: ${requestData.headers["content-type"]}, Authorization: ${requestData.headers["authorization"] ? "[PRESENT]" : "[NOT PRESENT]"}

Please provide:
1. A brief summary of what this request is doing
2. Any potential security concerns or issues
3. Suggestions for debugging if something seems wrong
4. Data validation observations

Keep the analysis concise and focused on debugging value.`;

    // Create a writable stream to capture the output
    const outputStream = new Writable({
      write(chunk, encoding, callback) {
        process.stdout.write(chunk);
        callback();
      },
    });

    // Use streaming with Claude
    const stream = await client.messages.stream({
      model: "claude-opus-4-5",
      max_tokens: 500,
      messages: [
        {
          role: "user",
          content: prompt,
        },
      ],
    });

    // Stream the response
    for await (const chunk of stream) {
      if (
        chunk.type === "content_block_delta" &&
        chunk.delta.type === "text_delta"
      ) {
        process.stdout.write(chunk.delta.text);
      }
    }

    console.log("\n" + "-".repeat(40));
  } catch (error) {
    console.error("Error analyzing request with Claude:", error.message);
  }

  console.log("=".repeat(80) + "\n");

  // Continue to next middleware
  next();
};

// Apply the logging middleware
app.use(requestLogger);

// Sample routes for testing
app.get("/", (req, res) => {
  res.json({ message: "Hello World!", status: "ok" });
});

app.post("/api/users", (req, res) => {
  const { name, email, age } = req.body;
  res.json({
    message: "User created",
    user: { id: Math.random().toString(36).substr(2, 9), name, email, age },
  });
});

app.post("/api/login", (req, res) => {
  const { username, password } = req.body;
  if (username && password) {
    res.json({
      message: "Login successful",
      token: "mock-jwt-token-" + Math.random().toString(36).substr(2, 9),
    });
  } else {
    res.status(400).json({ error: "Username and password required" });
  }
});

app.get("/api/products", (req, res) => {
  const { category, limit, page } = req.query;
  res.json({
    products: [
      { id: 1, name: "Product 1", category: category || "general" },
      { id: 2, name: "Product 2", category: category || "general" },
    ],
    pagination: { page: page || 1, limit: limit || 10 },
  });
});

app.put("/api/users/:id", (req, res) => {
  const { id } = req.params;
  const updates = req.body;
  res.json({
    message: "User updated",
    userId: id,
    updates,
  });
});

app.delete("/api/users/:id", (req, res) => {
  const { id } = req.params;
  res.json({
    message: "User deleted",
    userId: id,
  });
});

// Error handling middleware
app.use((err, req, res, next) => {
  console.error("Error:", err.message);
  res.status(500).json({ error: "Internal server error" });
});

// Start the server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`\n🚀 Server running on port ${PORT}`);
  console.log(`📊 Request logging middleware with AI analysis is active`);
  console.log(`\nTest endpoints:`);
  console.log(`  GET  http://localhost:${PORT}/`);
  console.log(`  POST http://localhost:${PORT}/api/users`);
  console.log(`  POST http://localhost:${PORT}/api/login`);
  console.log(`  GET  http://localhost:${PORT}/api/products`);
  console.log(`  PUT  http://localhost:${PORT}/api/users/:id`);
  console.log(`  DELETE http://localhost:${PORT}/api/users/:id`);
  console.log(`\nExample curl commands:`);
  console.log(
    `  curl -X POST http://localhost:${PORT}/api/users -H "Content-Type: application/json" -d '{"name":"John Doe","email":"john@example.com","age":30}'`
  );
  console.log(
    `  curl -X POST http://localhost:${PORT}/api/login -H "Content-Type: application/json" -d '{"username":"admin","password":"secret123"}'`
  );
  console.log(
    `  curl "http://localhost:${PORT}/api/products?category=electronics&limit=5&page=1"`
  );
});
```
