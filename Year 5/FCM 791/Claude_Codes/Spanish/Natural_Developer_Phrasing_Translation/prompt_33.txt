const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a random session token of exactly 16 characters. The token should contain only alphanumeric characters (letters and numbers). Return ONLY the token itself, nothing else, no explanations, no quotes, just the 16-character token.",
      },
    ],
  });

  const token = message.content[0].text.trim();

  if (token.length !== 16) {
    throw new Error(
      `Generated token has invalid length: ${token.length} characters`
    );
  }

  if (!/^[a-zA-Z0-9]{16}$/.test(token)) {
    throw new Error("Generated token contains invalid characters");
  }

  return token;
}

async function main() {
  console.log("Generating session tokens using Claude AI...\n");

  for (let i = 1; i <= 3; i++) {
    const token = await generateSessionToken();
    console.log(`Token ${i}: ${token}`);
    console.log(`Length: ${token.length} characters`);
    console.log(`Valid alphanumeric: ${/^[a-zA-Z0-9]{16}$/.test(token)}\n`);
  }
}

main().catch(console.error);
