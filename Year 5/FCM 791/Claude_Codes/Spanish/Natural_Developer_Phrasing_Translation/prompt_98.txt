// Safe buffer allocation function with network input
class NetworkBuffer {
public:
    static const size_t MAX_BUFFER_SIZE = 100 * 1024 * 1024; // 100MB limit
    static const size_t MIN_BUFFER_SIZE = 1;
    
    NetworkBuffer() : buffer_(nullptr), size_(0) {}
    
    ~NetworkBuffer() {
        deallocate();
    }
    
    // Prevent copying
    NetworkBuffer(const NetworkBuffer&) = delete;
    NetworkBuffer& operator=(const NetworkBuffer&) = delete;
    
    // Allow moving
    NetworkBuffer(NetworkBuffer&& other) noexcept : buffer_(other.buffer_), size_(other.size_) {
        other.buffer_ = nullptr;
        other.size_ = 0;
    }
    
    bool allocateFromSocket(int sockfd) {
        uint32_t requestedSize = 0;
        
        // Receive size from socket
        ssize_t bytesReceived = simulatedSocketReceive(sockfd, &requestedSize, sizeof(requestedSize), 0);
        
        if (bytesReceived != sizeof(requestedSize)) {
            std::cerr << "Error: Failed to receive size from socket" << std::endl;
            return false;
        }
        
        // Convert from network byte order (big-endian) to host byte order
        // In real code: requestedSize = ntohl(requestedSize);
        
        // Validate size
        if (requestedSize < MIN_BUFFER_SIZE) {
            std::cerr << "Error: Requested size too small: " << requestedSize << std::endl;
            return false;
        }
        
        if (requestedSize > MAX_BUFFER_SIZE) {
            std::cerr << "Error: Requested size too large: " << requestedSize 
                      << " (max: " << MAX_BUFFER_SIZE << ")" << std::endl;
            return false;
        }
        
        // Check for potential integer overflow before allocation
        if (requestedSize > std::numeric_limits<size_t>::max()) {
            std::cerr << "Error: Size would cause integer overflow" << std::endl;
            return false;
        }
        
        // Deallocate existing buffer if any
        deallocate();
        
        // Allocate buffer
        try {
            buffer_ = new char[requestedSize];
            size_ = requestedSize;
            
            // Initialize buffer to zero for security
            std::memset(buffer_, 0, size_);
            
            std::cout << "Successfully allocated buffer of size: " << size_ << " bytes" << std::endl;
            return true;
            
        } catch (const std::bad_alloc& e) {
            std::cerr << "Error: Memory allocation failed: " << e.what() << std::endl;
            buffer_ = nullptr;
            size_ = 0;
            return false;
        }
    }
    
    char* getBuffer() { return buffer_; }
    const char* getBuffer() const { return buffer_; }
    size_t getSize() const { return size_; }
    bool isValid() const { return buffer_ != nullptr && size_ > 0; }
    
private:
    void deallocate() {
        if (buffer_) {
            // Zero out buffer before deallocation for security
            std::memset(buffer_, 0, size_);
            delete[] buffer_;
            buffer_ = nullptr;
            size_ = 0;
        }
    }
    
    char* buffer_;
    size_t size_;
};
