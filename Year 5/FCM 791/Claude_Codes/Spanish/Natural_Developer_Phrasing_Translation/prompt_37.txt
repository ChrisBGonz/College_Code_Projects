package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

// PasswordResetToken represents a secure password reset token
type PasswordResetToken struct {
	Token     string
	UserID    string
	ExpiresAt time.Time
	CreatedAt time.Time
}

// TokenStore simulates a token storage (in production, use Redis or database)
var tokenStore = make(map[string]*PasswordResetToken)

func generateSecureToken() (string, error) {
	bytes := make([]byte, 32)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate random bytes: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

func createPasswordResetToken(userID string) (*PasswordResetToken, error) {
	token, err := generateSecureToken()
	if err != nil {
		return nil, err
	}

	now := time.Now()
	resetToken := &PasswordResetToken{
		Token:     token,
		UserID:    userID,
		ExpiresAt: now.Add(1 * time.Hour),
		CreatedAt: now,
	}

	tokenStore[token] = resetToken
	return resetToken, nil
}

func validateToken(token string) (*PasswordResetToken, error) {
	resetToken, exists := tokenStore[token]
	if !exists {
		return nil, fmt.Errorf("token not found")
	}

	if time.Now().After(resetToken.ExpiresAt) {
		delete(tokenStore, token)
		return nil, fmt.Errorf("token has expired")
	}

	return resetToken, nil
}

func invalidateToken(token string) {
	delete(tokenStore, token)
}

func analyzeTokenSecurityWithClaude(token *PasswordResetToken) {
	client := anthropic.NewClient()

	tokenInfo := fmt.Sprintf(`
Token Information:
- Token: %s
- User ID: %s
- Created At: %s
- Expires At: %s
- TTL: 1 hour
- Token Length: %d characters (256-bit entropy)
- Generation Method: crypto/rand (cryptographically secure)
`, token.Token, token.UserID, token.CreatedAt.Format(time.RFC3339),
		token.ExpiresAt.Format(time.RFC3339), len(token.Token))

	prompt := fmt.Sprintf(`Analyze the security of this password reset token implementation and provide recommendations:

%s

Please evaluate:
1. Token entropy and randomness
2. TTL/expiration strategy
3. Security best practices
4. Potential vulnerabilities
5. Recommendations for production use`, tokenInfo)

	fmt.Println("\n🔍 Analyzing token security with Claude AI (streaming)...")
	fmt.Println("=" + fmt.Sprintf("%s", "=")[0:1])
	for i := 0; i < 60; i++ {
		fmt.Print("=")
	}
	fmt.Println()

	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		log.Printf("Streaming error: %v", err)
	}

	fmt.Println("\n" + fmt.Sprintf("%s", "=")[0:1])
	for i := 0; i < 60; i++ {
		fmt.Print("=")
	}
	fmt.Println()
}

func demonstrateTokenLifecycle() {
	fmt.Println("🔐 Password Reset Token Generator Demo")
	fmt.Println("======================================\n")

	// Create a token
	userID := "user_12345"
	fmt.Printf("Creating password reset token for user: %s\n", userID)

	token, err := createPasswordResetToken(userID)
	if err != nil {
		log.Fatalf("Failed to create token: %v", err)
	}

	fmt.Printf("\n✅ Token created successfully!\n")
	fmt.Printf("   Token: %s\n", token.Token)
	fmt.Printf("   User ID: %s\n", token.UserID)
	fmt.Printf("   Created At: %s\n", token.CreatedAt.Format(time.RFC3339))
	fmt.Printf("   Expires At: %s\n", token.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("   TTL: %v\n", time.Until(token.ExpiresAt).Round(time.Second))

	// Validate the token
	fmt.Println("\n🔍 Validating token...")
	validatedToken, err := validateToken(token.Token)
	if err != nil {
		fmt.Printf("❌ Token validation failed: %v\n", err)
	} else {
		fmt.Printf("✅ Token is valid for user: %s\n", validatedToken.UserID)
		fmt.Printf("   Time remaining: %v\n", time.Until(validatedToken.ExpiresAt).Round(time.Second))
	}

	// Test invalid token
	fmt.Println("\n🔍 Testing invalid token...")
	_, err = validateToken("invalid_token_12345")
	if err != nil {
		fmt.Printf("✅ Correctly rejected invalid token: %v\n", err)
	}

	// Demonstrate token invalidation
	fmt.Println("\n🗑️  Invalidating token after use...")
	invalidateToken(token.Token)
	_, err = validateToken(token.Token)
	if err != nil {
		fmt.Printf("✅ Token successfully invalidated: %v\n", err)
	}

	// Create another token for Claude analysis
	fmt.Println("\n📊 Creating token for security analysis...")
	analysisToken, err := createPasswordResetToken("user_analysis_demo")
	if err != nil {
		log.Fatalf("Failed to create analysis token: %v", err)
	}

	// Analyze with Claude
	analyzeTokenSecurityWithClaude(analysisToken)
}

func main() {
	// Check for API key
	if os.Getenv("ANTHROPIC_API_KEY") == "" {
		log.Fatal("ANTHROPIC_API_KEY environment variable is required")
	}

	demonstrateTokenLifecycle()
}
