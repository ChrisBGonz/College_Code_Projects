import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateMongoDBConnectionScript() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Configura una cadena de conexión a MongoDB en un script de Node.js para una BD de desarrollo local. Proporciona el código completo con manejo de errores, variables de entorno y mejores prácticas.",
      },
    ],
  });

  console.log("Respuesta de Claude:");
  console.log("===================");

  if (message.content[0].type === "text") {
    console.log(message.content[0].text);
  }

  console.log("\nInformación del mensaje:");
  console.log(`- ID: ${message.id}`);
  console.log(`- Modelo: ${message.model}`);
  console.log(`- Tokens de entrada: ${message.usage.input_tokens}`);
  console.log(`- Tokens de salida: ${message.usage.output_tokens}`);
  console.log(`- Razón de parada: ${message.stop_reason}`);
}

generateMongoDBConnectionScript().catch(console.error);
