import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseManager {

    private static final Logger LOGGER = Logger.getLogger(DatabaseManager.class.getName());

    private static final String DB_URL = "jdbc:mysql://localhost:3306/mydb";
    private static final String DB_USER = "user";
    private static final String DB_PASSWORD = "password";

    public static class DatabaseException extends RuntimeException {

        private final int errorCode;
        private final String sqlState;

        public DatabaseException(String message, int errorCode, String sqlState, Throwable cause) {
            super(message, cause);
            this.errorCode = errorCode;
            this.sqlState = sqlState;
        }

        public DatabaseException(String message, Throwable cause) {
            super(message, cause);
            this.errorCode = -1;
            this.sqlState = "UNKNOWN";
        }

        public int getErrorCode() {
            return errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        @Override
        public String toString() {
            return String.format("DatabaseException{message='%s', errorCode=%d, sqlState='%s'}",
                    getMessage(), errorCode, sqlState);
        }
    }

    public ResultSet executeQuery(String query, Object... params) {
        Connection connection = null;
        PreparedStatement statement = null;

        try {
            connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
            statement = connection.prepareStatement(query);

            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }

            LOGGER.info(String.format("Executing query: %s", query));
            return statement.executeQuery();

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("SQL error while executing query. " +
                                    "Error Code: %d | SQL State: %s | Message: %s | Query: %s",
                            e.getErrorCode(),
                            e.getSQLState(),
                            e.getMessage(),
                            query),
                    e);

            throw new DatabaseException(
                    String.format("Failed to execute query: %s", query),
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    public int executeUpdate(String query, Object... params) {
        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement statement = connection.prepareStatement(query)) {

            for (int i = 0; i < params.length; i++) {
                statement.setObject(i + 1, params[i]);
            }

            LOGGER.info(String.format("Executing update: %s", query));
            return statement.executeUpdate();

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("SQL error during update operation. " +
                                    "Error Code: %d | SQL State: %s | Message: %s",
                            e.getErrorCode(),
                            e.getSQLState(),
                            e.getMessage()),
                    e);

            throw new DatabaseException(
                    "Failed to execute update operation",
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    public void createConnection() {
        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD)) {
            LOGGER.info("Database connection established successfully.");
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE,
                    String.format("Failed to establish database connection. " +
                                    "Error Code: %d | SQL State: %s | Message: %s",
                            e.getErrorCode(),
                            e.getSQLState(),
                            e.getMessage()),
                    e);

            throw new DatabaseException(
                    "Unable to connect to the database",
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    public static void main(String[] args) {
        DatabaseManager manager = new DatabaseManager();

        try {
            manager.createConnection();

            int rowsAffected = manager.executeUpdate(
                    "INSERT INTO users (name, email) VALUES (?, ?)",
                    "John Doe",
                    "john.doe@example.com"
            );
            System.out.println("Rows affected: " + rowsAffected);

            ResultSet resultSet = manager.executeQuery(
                    "SELECT * FROM users WHERE id = ?",
                    1
            );

            while (resultSet.next()) {
                System.out.println("User: " + resultSet.getString("name"));
            }

        } catch (DatabaseException e) {
            System.err.println("Database operation failed: " + e.getMessage());
            System.err.println("Error Code: " + e.getErrorCode());
            System.err.println("SQL State: " + e.getSqlState());
        }
    }
}
