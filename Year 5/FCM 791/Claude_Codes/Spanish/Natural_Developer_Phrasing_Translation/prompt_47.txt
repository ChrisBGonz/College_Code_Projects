import React, { useState, useEffect } from "react";

const UserBio = ({ userId }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUserBio = async () => {
      try {
        setLoading(true);
        setError(null);

        const response = await fetch(
          `https://jsonplaceholder.typicode.com/users/${userId}`
        );

        if (!response.ok) {
          throw new Error(`Error al obtener datos: ${response.status}`);
        }

        const data = await response.json();

        const userWithBio = {
          ...data,
          bio: `<p>Hola, soy <strong>${data.name}</strong>.</p>
                <p>Trabajo en <em>${data.company.name}</em> como especialista en ${data.company.bs}.</p>
                <p>Mi lema: <blockquote>${data.company.catchPhrase}</blockquote></p>
                <p>Puedes visitar mi sitio web en <a href="http://${data.website}" target="_blank" rel="noopener noreferrer">${data.website}</a>.</p>`,
        };

        setUser(userWithBio);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (userId) {
      fetchUserBio();
    }
  }, [userId]);

  if (loading) {
    return (
      <div style={styles.loadingContainer}>
        <div style={styles.spinner}></div>
        <p style={styles.loadingText}>Cargando perfil...</p>
      </div>
    );
  }

  if (error) {
    return (
      <div style={styles.errorContainer}>
        <span style={styles.errorIcon}>⚠️</span>
        <p style={styles.errorText}>Error: {error}</p>
      </div>
    );
  }

  if (!user) {
    return (
      <div style={styles.emptyContainer}>
        <p>No se encontró información del usuario.</p>
      </div>
    );
  }

  return (
    <div style={styles.card}>
      <div style={styles.header}>
        <div style={styles.avatar}>
          {user.name.charAt(0).toUpperCase()}
        </div>
        <div style={styles.headerInfo}>
          <h2 style={styles.userName}>{user.name}</h2>
          <p style={styles.userUsername}>@{user.username}</p>
          <p style={styles.userEmail}>✉️ {user.email}</p>
        </div>
      </div>

      <div style={styles.divider}></div>

      <div style={styles.bioSection}>
        <h3 style={styles.bioTitle}>Biografía</h3>
        <div
          style={styles.bioContent}
          dangerouslySetInnerHTML={{ __html: user.bio }}
        />
      </div>

      <div style={styles.divider}></div>

      <div style={styles.contactSection}>
        <h3 style={styles.sectionTitle}>Contacto</h3>
        <div style={styles.contactGrid}>
          <div style={styles.contactItem}>
            <span style={styles.contactIcon}>📍</span>
            <span>
              {user.address.city}, {user.address.zipcode}
            </span>
          </div>
          <div style={styles.contactItem}>
            <span style={styles.contactIcon}>📞</span>
            <span>{user.phone}</span>
          </div>
        </div>
      </div>
    </div>
  );
};
