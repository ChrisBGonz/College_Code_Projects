    static async Task<string> GetClaudeAssistance(string apiKey, string prompt)
    {
        var requestBody = new
        {
            model = "claude-opus-4-5",
            max_tokens = 1024,
            messages = new[]
            {
                new
                {
                    role = "user",
                    content = prompt
                }
            }
        };
        
        string jsonBody = JsonSerializer.Serialize(requestBody);
        
        using var request = new HttpRequestMessage(HttpMethod.Post, CLAUDE_API_URL);
        request.Headers.Add("x-api-key", apiKey);
        request.Headers.Add("anthropic-version", "2023-06-01");
        request.Content = new StringContent(jsonBody, Encoding.UTF8, "application/json");
        
        var response = await httpClient.SendAsync(request);
        string responseContent = await response.Content.ReadAsStringAsync();
        
        if (!response.IsSuccessStatusCode)
        {
            throw new HttpRequestException($"API request failed: {response.StatusCode} - {responseContent}");
        }
        
        using var doc = JsonDocument.Parse(responseContent);
        return doc.RootElement
            .GetProperty("content")[0]
            .GetProperty("text")
            .GetString() ?? "No response";
    }
}

// Actual Remember Me Service Implementation
class RememberMeService
{
    private readonly Dictionary<string, (string UserId, DateTime Expiry)> _tokenStore = new();
    private const int TOKEN_EXPIRY_DAYS = 30;
    private const string COOKIE_NAME = "RememberMe";
    
    public (string Token, DateTime Expiry) GenerateRememberMeToken(string userId)
    {
        // Generate cryptographically secure random token
        byte[] tokenBytes = new byte[32];
        using var rng = System.Security.Cryptography.RandomNumberGenerator.Create();
        rng.GetBytes(tokenBytes);
        
        // Create token with timestamp for additional validation
        string token = Convert.ToBase64String(tokenBytes)
            .Replace("+", "-")
            .Replace("/", "_")
            .Replace("=", "");
        
        DateTime expiry = DateTime.UtcNow.AddDays(TOKEN_EXPIRY_DAYS);
        
        // Store token (in production, this would be in a database)
        _tokenStore[token] = (userId, expiry);
        
        return (token, expiry);
    }
    
    public CookieData CreateCookieData(string token, DateTime expiry)
    {
        return new CookieData
        {
            Name = COOKIE_NAME,
            Value = token,
            Expires = expiry,
            HttpOnly = true,  // Prevents JavaScript access
            Secure = true,    // HTTPS only
            SameSite = "Strict",  // CSRF protection
            Path = "/"
        };
    }
    
    public ValidationResult ValidateRememberMeToken(string token, string expectedUserId)
    {
        if (string.IsNullOrEmpty(token))
        {
            return new ValidationResult { IsValid = false };
        }
        
        // Check if token exists in store
        if (!_tokenStore.TryGetValue(token, out var tokenData))
        {
            return new ValidationResult { IsValid = false };
        }
        
        // Check expiry
        if (DateTime.UtcNow > tokenData.Expiry)
        {
            _tokenStore.Remove(token);
            return new ValidationResult { IsValid = false };
        }
        
        // Validate user ID matches
        if (tokenData.UserId != expectedUserId)
        {
            return new ValidationResult { IsValid = false };
        }
        
        return new ValidationResult
        {
            IsValid = true,
            UserId = tokenData.UserId,
            SessionRenewed = true
        };
    }
    
    public (string NewToken, DateTime NewExpiry) RotateToken(string oldToken, string userId)
    {
        // Remove old token
        _tokenStore.Remove(oldToken);
        
        // Generate new token
        return GenerateRememberMeToken(userId);
    }
    
    public void RevokeToken(string token)
    {
        _tokenStore.Remove(token);
    }
    
    public void RevokeAllUserTokens(string userId)
    {
        var tokensToRemove = new List<string>();
        
        foreach (var kvp in _tokenStore)
        {
            if (kvp.Value.UserId == userId)
            {
                tokensToRemove.Add(kvp.Key);
            }
        }
        
        foreach (var token in tokensToRemove)
        {
            _tokenStore.Remove(token);
        }
    }
}

// Data models
class CookieData
{
    public string Name { get; set; } = "";
    public string Value { get; set; } = "";
    public DateTime Expires { get; set; }
    public bool HttpOnly { get; set; }
    public bool Secure { get; set; }
    public string SameSite { get; set; } = "Strict";
    public string Path { get; set; } = "/";
}

class ValidationResult
{
    public bool IsValid { get; set; }
    public string UserId { get; set; } = "";
    public bool SessionRenewed { get; set; }
}
