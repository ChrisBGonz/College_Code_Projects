
const loadScript = (url) => {
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = url;
    script.async = true;
    script.onload = () => resolve(`Script cargado: ${url}`);
    script.onerror = () => reject(new Error(`Error al cargar el script: ${url}`));
    document.head.appendChild(script);
  });
};

const jQueryCDN = 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';

const isJQueryLoaded = () => typeof window.jQuery !== 'undefined';

const loadJQuery = async () => {
  if (isJQueryLoaded()) {
    console.log(`jQuery ya está cargado. Versión: ${window.jQuery.fn.jquery}`);
    return window.jQuery;
  }

  try {
    console.log('Cargando jQuery desde CDN...');
    const message = await loadScript(jQueryCDN);
    console.log(message);

    if (isJQueryLoaded()) {
      console.log(`jQuery cargado correctamente. Versión: ${window.jQuery.fn.jquery}`);
      return window.jQuery;
    } else {
      throw new Error('jQuery no se inicializó correctamente después de cargar el script.');
    }
  } catch (error) {
    console.error('Error durante la carga de jQuery:', error.message);
    throw error;
  }
};

const useJQuery = ($) => {
  console.log('Usando jQuery para manipular el DOM...');

  $('body').css('background-color', '#f0f8ff');

  const $div = $('<div>', {
    id: 'dynamic-content',
    text: '¡jQuery cargado dinámicamente desde CDN!',
    css: {
      padding: '20px',
      margin: '20px',
      backgroundColor: '#4CAF50',
      color: 'white',
      borderRadius: '8px',
      fontFamily: 'Arial, sans-serif',
      fontSize: '18px',
      textAlign: 'center',
      boxShadow: '0 4px 6px rgba(0,0,0,0.1)'
    }
  });

  $('body').append($div);

  $('#dynamic-content').on('click', function () {
    $(this).animate({ opacity: 0.5 }, 300).animate({ opacity: 1 }, 300);
    console.log('Elemento clickeado. Animación ejecutada.');
  });

  $.ajax({
    url: 'https://jsonplaceholder.typicode.com/todos/1',
    method: 'GET',
    success: (data) => {
      console.log('Petición AJAX exitosa:', data);
      const $info = $('<p>', {
        text: `Tarea obtenida: "${data.title}" - Completada: ${data.completed}`,
        css: {
          padding: '10px',
          margin: '10px 20px',
          backgroundColor: '#e8f5e9',
          borderLeft: '4px solid #4CAF50',
          fontFamily: 'Arial, sans-serif'
        }
      });
      $('body').append($info);
    },
    error: (xhr, status, error) => {
      console.error('Error en la petición AJAX:', error);
    }
  });
};

(async () => {
  try {
    const $ = await loadJQuery();
    useJQuery($);
  } catch (error) {
    console.error('No se pudo inicializar la aplicación:', error.message);
  }
})();
