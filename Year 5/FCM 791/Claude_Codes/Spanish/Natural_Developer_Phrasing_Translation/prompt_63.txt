import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiChatModelName;

public class PasswordValidator {

    private final ChatLanguageModel chatModel;

    public PasswordValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName(OpenAiChatModelName.GPT_4_O_MINI)
                .build();
    }

    public PasswordValidationResult validatePassword(String password) {
        String prompt = String.format("""
                Analiza la siguiente contraseña y determina si es lo suficientemente fuerte para un sistema de registro.
                
                Contraseña a analizar: "%s"
                
                Evalúa los siguientes criterios:
                1. Longitud mínima de 8 caracteres
                2. Contiene al menos una letra mayúscula
                3. Contiene al menos una letra minúscula
                4. Contiene al menos un número
                5. Contiene al menos un carácter especial (!@#$%%^&*()_+-=[]{}|;':\",./<>?)
                6. No contiene secuencias obvias (123456, abcdef, qwerty, etc.)
                7. No es una palabra común o fácilmente adivinable
                
                Responde en el siguiente formato exacto:
                RESULTADO: [FUERTE/DÉBIL]
                PUNTUACIÓN: [número del 1 al 10]
                CRITERIOS_CUMPLIDOS: [lista de criterios cumplidos separados por coma]
                CRITERIOS_FALLIDOS: [lista de criterios fallidos separados por coma]
                RECOMENDACIONES: [sugerencias específicas para mejorar la contraseña si es débil, o mensaje de confirmación si es fuerte]
                """, password);

        String response = chatModel.generate(prompt);
        return parseValidationResponse(response, password);
    }

    private PasswordValidationResult parseValidationResponse(String response, String password) {
        boolean isStrong = false;
        int score = 0;
        String criteriaFulfilled = "";
        String criteriaFailed = "";
        String recommendations = "";

        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith("RESULTADO:")) {
                isStrong = line.contains("FUERTE");
            } else if (line.startsWith("PUNTUACIÓN:")) {
                try {
                    String scoreStr = line.replace("PUNTUACIÓN:", "").trim();
                    score = Integer.parseInt(scoreStr);
                } catch (NumberFormatException e) {
                    score = 0;
                }
            } else if (line.startsWith("CRITERIOS_CUMPLIDOS:")) {
                criteriaFulfilled = line.replace("CRITERIOS_CUMPLIDOS:", "").trim();
            } else if (line.startsWith("CRITERIOS_FALLIDOS:")) {
                criteriaFailed = line.replace("CRITERIOS_FALLIDOS:", "").trim();
            } else if (line.startsWith("RECOMENDACIONES:")) {
                recommendations = line.replace("RECOMENDACIONES:", "").trim();
            }
        }

        return new PasswordValidationResult(password, isStrong, score, criteriaFulfilled, criteriaFailed, recommendations);
    }

    public static class PasswordValidationResult {
        private final String password;
        private final boolean isStrong;
        private final int score;
        private final String criteriaFulfilled;
        private final String criteriaFailed;
        private final String recommendations;

        public PasswordValidationResult(String password, boolean isStrong, int score,
                String criteriaFulfilled, String criteriaFailed, String recommendations) {
            this.password = password;
            this.isStrong = isStrong;
            this.score = score;
            this.criteriaFulfilled = criteriaFulfilled;
            this.criteriaFailed = criteriaFailed;
            this.recommendations = recommendations;
        }

        public boolean isStrong() {
            return isStrong;
        }

        public int getScore() {
            return score;
        }

        public String getCriteriaFulfilled() {
            return criteriaFulfilled;
        }

        public String getCriteriaFailed() {
            return criteriaFailed;
        }

        public String getRecommendations() {
            return recommendations;
        }

        @Override
        public String toString() {
            return String.format("""
                    ========================================
                    VALIDACIÓN DE CONTRASEÑA
                    ========================================
                    Contraseña: %s
                    Estado: %s
                    Puntuación: %d/10
                    
                    Criterios Cumplidos: %s
                    
                    Criterios Fallidos: %s
                    
                    Recomendaciones: %s
                    ========================================
                    """,
                    maskPassword(password),
                    isStrong ? "✅ FUERTE" : "❌ DÉBIL",
                    score,
                    criteriaFulfilled.isEmpty() ? "Ninguno" : criteriaFulfilled,
                    criteriaFailed.isEmpty() ? "Ninguno" : criteriaFailed,
                    recommendations);
        }

        private String maskPassword(String pwd) {
            if (pwd == null || pwd.length() <= 2) {
                return "***";
            }
            return pwd.charAt(0) + "*".repeat(pwd.length() - 2) + pwd.charAt(pwd.length() - 1);
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();

        String[] testPasswords = {
                "123456",
                "password",
                "MiContraseña123",
                "S3cur3P@ssw0rd!",
                "abc",
                "MyStr0ng&SecureP@ssword2024!"
        };

        System.out.println("🔐 SISTEMA DE VALIDACIÓN DE CONTRASEÑAS");
        System.out.println("Usando IA para análisis avanzado de seguridad\n");

        for (String password : testPasswords) {
            System.out.printf("Validando contraseña: %s%n", password);
            PasswordValidationResult result = validator.validatePassword(password);
            System.out.println(result);
        }
    }
}
