import anthropic
import pyotp
import time
import re

client = anthropic.Anthropic()

def generate_test_totp_secret():
    """Generate a test TOTP secret for demonstration."""
    return pyotp.random_base32()

def get_current_totp(secret: str) -> str:
    """Get the current TOTP code for testing purposes."""
    totp = pyotp.TOTP(secret)
    return totp.now()

def validate_totp_with_claude(secret: str, user_code: str) -> dict:
    """
    Use Claude to validate a TOTP code using extended thinking.
    
    Args:
        secret: The TOTP secret key
        user_code: The 6-digit code provided by the user
        
    Returns:
        dict with validation result and explanation
    """
    
    # Get current TOTP for validation context
    totp = pyotp.TOTP(secret)
    current_time = int(time.time())
    
    # Check if code is valid (with 1 step tolerance for clock drift)
    is_valid = totp.verify(user_code, valid_window=1)
    
    # Get current valid codes for context
    current_code = totp.now()
    
    prompt = f"""You are a security expert analyzing a Two-Factor Authentication (2FA) validation request.

A user is attempting to log in with a TOTP (Time-based One-Time Password) code.

Security Context:
- Current timestamp: {current_time}
- User provided code: {user_code}
- Code validation result: {"VALID" if is_valid else "INVALID"}
- Current expected code: {current_code}
- Time window tolerance: ±30 seconds (1 step)

Please analyze this 2FA validation attempt and provide:
1. Whether the authentication should be APPROVED or DENIED
2. Security assessment of the attempt
3. Any potential security concerns
4. Recommended action for the system

Consider:
- TOTP codes are valid for 30-second windows
- A 1-step tolerance is allowed for clock drift
- The code format should be exactly 6 digits
- Repeated failed attempts could indicate an attack

Provide a clear security recommendation."""

    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=16000,
        thinking={
            "type": "enabled",
            "budget_tokens": 10000
        },
        messages=[{
            "role": "user",
            "content": prompt
        }]
    )
    
    # Extract thinking and response
    thinking_content = ""
    response_text = ""
    
    for block in response.content:
        if block.type == "thinking":
            thinking_content = block.thinking
        elif block.type == "text":
            response_text = block.text
    
    return {
        "is_valid": is_valid,
        "user_code": user_code,
        "claude_analysis": response_text,
        "claude_thinking": thinking_content,
        "timestamp": current_time
    }

def validate_code_format(code: str) -> bool:
    """Validate that the code is exactly 6 digits."""
    return bool(re.match(r'^\d{6}$', code))

def simulate_login_attempt(secret: str, user_code: str) -> None:
    """
    Simulate a complete 2FA login attempt with Claude's analysis.
    
    Args:
        secret: The TOTP secret key
        user_code: The code provided by the user
    """
    print("\n" + "="*60)
    print("2FA VALIDATION SYSTEM")
    print("="*60)
    
    # Validate code format first
    if not validate_code_format(user_code):
        print(f"❌ INVALID FORMAT: Code '{user_code}' is not a valid 6-digit number")
        print("Please provide exactly 6 digits.")
        return
    
    print(f"📱 Validating TOTP code: {user_code}")
    print("🔍 Analyzing with Claude extended thinking...")
    print("-"*60)
    
    # Get Claude's analysis
    result = validate_totp_with_claude(secret, user_code)
    
    # Display results
    print("\n📊 VALIDATION RESULT:")
    print(f"   Code Valid: {'✅ YES' if result['is_valid'] else '❌ NO'}")
    print(f"   Timestamp: {result['timestamp']}")
    
    print("\n🤔 CLAUDE'S SECURITY ANALYSIS:")
    print("-"*40)
    print(result['claude_analysis'])
    
    if result['claude_thinking']:
        print("\n💭 EXTENDED THINKING PROCESS (Summary):")
        print("-"*40)
        # Show first 500 chars of thinking to keep output manageable
        thinking_preview = result['claude_thinking'][:500]
        if len(result['claude_thinking']) > 500:
            thinking_preview += "... [truncated]"
        print(thinking_preview)
    
    print("\n" + "="*60)
    
    # Final decision
    if result['is_valid']:
        print("🔓 ACCESS GRANTED - Authentication successful")
    else:
        print("🔒 ACCESS DENIED - Invalid authentication code")
    
    print("="*60)

def main():
    """Main function to demonstrate TOTP validation."""
    print("🔐 TOTP 2FA Validation System with Claude Extended Thinking")
    print("="*60)
    
    # Generate a test secret
    secret = generate_test_totp_secret()
    print(f"\n📋 Test Setup:")
    print(f"   TOTP Secret: {secret}")
    print(f"   (In production, this would be stored securely per user)")
    
    # Get current valid code for demonstration
    current_code = get_current_totp(secret)
    print(f"\n⏰ Current valid TOTP code: {current_code}")
    print("   (This changes every 30 seconds)")
    
    # Test scenarios
    print("\n🧪 Running validation scenarios...")
    
    # Scenario 1: Valid code
    print("\n📌 Scenario 1: Valid TOTP Code")
    simulate_login_attempt(secret, current_code)
    
    # Wait a moment between tests
    time.sleep(2)
    
    # Scenario 2: Invalid code
    print("\n📌 Scenario 2: Invalid TOTP Code")
    invalid_code = "000000" if current_code != "000000" else "111111"
    simulate_login_attempt(secret, invalid_code)
    
    # Scenario 3: Invalid format
    print("\n📌 Scenario 3: Invalid Format (not 6 digits)")
    simulate_login_attempt(secret, "12345")
    
    # Interactive mode
    print("\n" + "="*60)
    print("🎯 INTERACTIVE MODE")
    print("="*60)
    print(f"Secret key: {secret}")
    print(f"Current code: {get_current_totp(secret)}")
    print("\nYou can now test with your own codes.")
    print("Enter 'q' to quit or 'refresh' to see current code.\n")
    
    while True:
        user_input = input("Enter 6-digit TOTP code (or 'q' to quit, 'refresh' for current code): ").strip()
        
        if user_input.lower() == 'q':
            print("👋 Exiting 2FA validation system.")
            break
        elif user_input.lower() == 'refresh':
            current = get_current_totp(secret)
            print(f"⏰ Current valid code: {current}")
        else:
            simulate_login_attempt(secret, user_input)

if __name__ == "__main__":
    main()
