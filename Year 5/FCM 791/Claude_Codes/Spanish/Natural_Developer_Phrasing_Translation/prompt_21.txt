
```python
import os
import json
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

SYSTEM_PROMPT = """You are an expert Python developer specializing in JWT (JSON Web Token) authentication and security. 
You help users understand and implement advanced JWT authentication systems.

When asked to generate code, provide complete, working Python code examples.
When explaining JWT concepts, be thorough and accurate.
Always consider security best practices in your recommendations.

You have deep knowledge of:
- PyJWT library for JWT operations
- JWT token structure (header, payload, signature)
- Various signing algorithms (HS256, RS256, ES256, etc.)
- Token validation and verification
- Claims validation (exp, iat, nbf, iss, aud, sub)
- Refresh token patterns
- Token revocation strategies
- Security best practices

When generating code examples, always include:
1. Proper error handling
2. Security considerations
3. Comments explaining key concepts
4. Example usage"""

def chat_with_claude(user_message):
    """Send a message to Claude and get a response."""
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=SYSTEM_PROMPT,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message

def display_welcome():
    """Display welcome message and instructions."""
    print("\n" + "="*60)
    print("JWT Authentication Expert Assistant")
    print("="*60)
    print("\nWelcome! I'm your JWT authentication expert.")
    print("I can help you with:")
    print("  - Creating and signing JWT tokens")
    print("  - Validating and decoding tokens")
    print("  - Implementing refresh token patterns")
    print("  - Security best practices")
    print("  - Complex token validation scenarios")
    print("\nType 'quit' or 'exit' to end the conversation")
    print("Type 'clear' to start a new conversation")
    print("Type 'history' to see conversation history")
    print("="*60 + "\n")

def display_history():
    """Display the conversation history."""
    if not conversation_history:
        print("\nNo conversation history yet.\n")
        return
    
    print("\n" + "="*60)
    print("Conversation History")
    print("="*60)
    for i, message in enumerate(conversation_history, 1):
        role = "You" if message["role"] == "user" else "Assistant"
        print(f"\n[{i}] {role}:")
        print(message["content"][:200] + "..." if len(message["content"]) > 200 else message["content"])
    print("="*60 + "\n")

def get_initial_context():
    """Get initial context about what the user wants to do."""
    print("What would you like to do with JWT authentication?")
    print("\nSome suggestions:")
    print("1. Generate a complete JWT authentication system")
    print("2. Learn about JWT token validation")
    print("3. Implement refresh token rotation")
    print("4. Understand JWT security best practices")
    print("5. Create custom claims and validation")
    print("\nOr describe your specific use case:\n")

def main():
    """Main function to run the JWT authentication assistant."""
    display_welcome()
    get_initial_context()
    
    # Start with an initial message to set context
    initial_response = chat_with_claude(
        "Hello! I need help with JWT authentication in Python. "
        "Please provide me with a comprehensive overview of what you can help me with "
        "and show me a basic example of creating and validating a JWT token using PyJWT."
    )
    print("Assistant:", initial_response)
    print("\n" + "-"*60 + "\n")
    
    while True:
        try:
            user_input = input("You: ").strip()
            
            if not user_input:
                continue
            
            if user_input.lower() in ['quit', 'exit']:
                print("\nThank you for using the JWT Authentication Assistant!")
                print("Remember to always follow security best practices in your implementations.")
                break
            
            if user_input.lower() == 'clear':
                conversation_history.clear()
                print("\nConversation cleared. Starting fresh!\n")
                get_initial_context()
                continue
            
            if user_input.lower() == 'history':
                display_history()
                continue
            
            print("\nAssistant: ", end="", flush=True)
            response = chat_with_claude(user_input)
            print(response)
            print("\n" + "-"*60 + "\n")
            
        except KeyboardInterrupt:
            print("\n\nInterrupted. Goodbye!")
            break
        except Exception as e:
            print(f"\nError: {e}")
            print("Please try again.\n")

if __name__ == "__main__":
    main()
```
