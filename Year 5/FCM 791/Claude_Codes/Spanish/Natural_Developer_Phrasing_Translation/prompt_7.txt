import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user_001",
      username: "john_doe",
      email: "john@example.com",
      password_hash: "old_hash_123",
    },
    {
      id: "user_002",
      username: "jane_smith",
      email: "jane@example.com",
      password_hash: "old_hash_456",
    },
    {
      id: "user_003",
      username: "bob_wilson",
      email: "bob@example.com",
      password_hash: "old_hash_789",
    },
  ],
};

// Tool definitions for password reset flow
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database by their user_id. Returns user information if found.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, at least one uppercase, one lowercase, one number, and one special character).",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description:
      "Hashes the password securely before storing it in the database.",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password_in_db",
    description:
      "Updates the user's password hash in the database. Should only be called after validation and hashing.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
  {
    name: "send_confirmation_notification",
    description:
      "Sends a confirmation notification to the user after successful password reset.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        email: {
          type: "string",
          description: "The email address to send the notification to",
        },
      },
      required: ["user_id", "email"],
    },
  },
];

// Tool implementations
function validateUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      success: true,
      user: {
        id: user.id,
        username: user.username,
        email: user.email,
      },
    };
  }
  return {
    success: false,
    error: `User with ID ${userId} not found`,
  };
}

function validatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /\d/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  if (isValid) {
    return {
      success: true,
      message: "Password meets all security requirements",
    };
  }

  const failures = [];
  if (!minLength) failures.push("minimum 8 characters");
  if (!hasUppercase) failures.push("at least one uppercase letter");
  if (!hasLowercase) failures.push("at least one lowercase letter");
  if (!hasNumber) failures.push("at least one number");
  if (!hasSpecialChar) failures.push("at least one special character");

  return {
    success: false,
    error: `Password does not meet requirements: ${failures.join(", ")}`,
  };
}

function hashPassword(password) {
  // Simulated hashing (in production, use bcrypt or similar)
  const hash = `hashed_${Buffer.from(password).toString("base64")}_${Date.now()}`;
  return {
    success: true,
    hash: hash,
  };
}

function updatePasswordInDb(userId, passwordHash) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex === -1) {
    return {
      success: false,
      error: `User with ID ${userId} not found`,
    };
  }

  database.users[userIndex].password_hash = passwordHash;
  return {
    success: true,
    message: `Password successfully updated for user ${userId}`,
    updated_at: new Date().toISOString(),
  };
}

function sendConfirmationNotification(userId, email) {
  // Simulated email sending
  console.log(
    `[EMAIL SERVICE] Sending password reset confirmation to ${email} for user ${userId}`
  );
  return {
    success: true,
    message: `Confirmation email sent to ${email}`,
    sent_at: new Date().toISOString(),
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_user":
      return validateUser(toolInput.user_id);
    case "validate_password_strength":
      return validatePasswordStrength(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "update_password_in_db":
      return updatePasswordInDb(toolInput.user_id, toolInput.password_hash);
    case "send_confirmation_notification":
      return sendConfirmationNotification(toolInput.user_id, toolInput.email);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\n${"=".repeat(60)}`);
  console.log(`Starting password reset for user: ${userId}`);
  console.log(`${"=".repeat(60)}\n`);

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user with ID "${userId}". The new password is "${newPassword}". 
      
      Follow these steps:
      1. Validate that the user exists in the database
      2. Validate the password strength
      3. Hash the password securely
      4. Update the password in the database
      5. Send a confirmation notification to the user
      
      Report the result of each step and the final outcome.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log("Initial Response:");
  console.log(`Stop Reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nTool Used: ${toolUse.name}`);
      console.log(`Tool Input: ${JSON.stringify(toolUse.input, null, 2)}`);

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log(`Tool Result: ${JSON.stringify(result, null, 2)}`);

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Add assistant response and tool results to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    // Continue the conversation
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued Response Stop Reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log(`\n${"=".repeat(60)}`);
  console.log("Final Response:");
  console.log(finalResponse);
  console.log(`${"=".repeat(60)}\n`);

  return finalResponse;
}

// Test the function with different scenarios
async function runTests() {
  // Test 1: Valid user with strong password
  console.log("TEST 1: Valid user with strong password");
  await resetPassword("user_001", "SecurePass123!");

  // Test 2: Valid user with weak password
  console.log("\nTEST 2: Valid user with weak password");
  await resetPassword("user_002", "weakpass");

  // Test 3: Invalid user
  console.log("\nTEST 3: Invalid user");
  await resetPassword("user_999", "StrongPass456@");
}

// Run the tests
runTests().catch(console.error);
