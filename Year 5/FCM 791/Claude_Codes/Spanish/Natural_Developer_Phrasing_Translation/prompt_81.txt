
```python
import anthropic
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "validate_password",
        "description": "Validates a password against security requirements. Returns success or failure with details.",
        "input_schema": {
            "type": "object",
            "properties": {
                "password": {
                    "type": "string",
                    "description": "The password to validate"
                },
                "username": {
                    "type": "string",
                    "description": "The username attempting to login"
                }
            },
            "required": ["password", "username"]
        }
    },
    {
        "name": "return_auth_response",
        "description": "Returns an authentication response with status code and message",
        "input_schema": {
            "type": "object",
            "properties": {
                "status_code": {
                    "type": "integer",
                    "description": "HTTP status code (200 for success, 401 for unauthorized)"
                },
                "message": {
                    "type": "string",
                    "description": "Response message"
                },
                "authenticated": {
                    "type": "boolean",
                    "description": "Whether authentication was successful"
                }
            },
            "required": ["status_code", "message", "authenticated"]
        }
    }
]

VALID_USERS = {
    "admin": "SecurePass123!",
    "user1": "MyPassword456@",
    "testuser": "Test789#"
}

def validate_password(username: str, password: str) -> dict:
    """Validates password for a given username"""
    if username not in VALID_USERS:
        return {
            "valid": False,
            "reason": "User not found"
        }
    
    if VALID_USERS[username] == password:
        return {
            "valid": True,
            "reason": "Password correct"
        }
    else:
        return {
            "valid": False,
            "reason": "Incorrect password"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "validate_password":
        result = validate_password(tool_input["username"], tool_input["password"])
        return json.dumps(result)
    elif tool_name == "return_auth_response":
        return json.dumps(tool_input)
    return json.dumps({"error": "Unknown tool"})

def login(username: str, password: str) -> dict:
    """
    Login function that validates credentials using Claude with tool use.
    Returns 401 Unauthorized if validation fails.
    """
    print(f"\n{'='*50}")
    print(f"Login attempt for user: {username}")
    print(f"{'='*50}")
    
    messages = [
        {
            "role": "user",
            "content": f"""Please validate the login credentials for:
            Username: {username}
            Password: {password}
            
            Use the validate_password tool to check if the credentials are valid.
            Then use the return_auth_response tool to return the appropriate response:
            - If valid: status_code=200, authenticated=true, message="Login successful"
            - If invalid: status_code=401, authenticated=false, message="401 Unauthorized: Invalid credentials"
            """
        }
    ]
    
    auth_response = None
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude response - Stop reason: {response.stop_reason}")
        
        if response.stop_reason == "end_turn":
            break
            
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\nTool called: {tool_name}")
                    print(f"Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    print(f"Tool result: {result}")
                    
                    if tool_name == "return_auth_response":
                        auth_response = json.loads(result)
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            break
    
    if auth_response is None:
        auth_response = {
            "status_code": 401,
            "message": "401 Unauthorized: Authentication failed",
            "authenticated": False
        }
    
    return auth_response

def main():
    """Test the login function with various credentials"""
    
    print("\n" + "="*60)
    print("TESTING LOGIN FUNCTION")
    print("="*60)
    
    test_cases = [
        ("admin", "SecurePass123!"),
        ("admin", "wrongpassword"),
        ("user1", "MyPassword456@"),
        ("nonexistent", "anypassword"),
        ("testuser", "Test789#"),
    ]
    
    for username, password in test_cases:
        result = login(username, password)
        
        print(f"\n{'='*50}")
        print(f"FINAL RESULT for {username}:")
        print(f"Status Code: {result['status_code']}")
        print(f"Message: {result['message']}")
        print(f"Authenticated: {result['authenticated']}")
        
        if result['status_code'] == 401:
            print("⚠️  ACCESS DENIED - 401 Unauthorized")
        else:
            print("✅ ACCESS GRANTED")

if __name__ == "__main__":
    main()
```
