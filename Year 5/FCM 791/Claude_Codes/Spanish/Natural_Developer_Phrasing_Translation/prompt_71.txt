
```python
import anthropic
import pickle
import os
import json
from datetime import datetime

client = anthropic.Anthropic()

def create_game_state_with_claude(player_name: str) -> dict:
    """Use Claude to generate a complex game state for a player."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Create a complex game state JSON object for a player named '{player_name}' in an RPG game. 
                Include the following fields:
                - player_info (name, level, experience, health, mana)
                - inventory (list of items with name, quantity, value)
                - equipment (weapon, armor, accessories)
                - skills (list of skills with name, level, description)
                - quests (active and completed quests)
                - location (current area, coordinates)
                - statistics (kills, deaths, playtime_hours)
                - achievements (list of unlocked achievements)
                
                Return ONLY valid JSON, no explanations."""
            }
        ]
    )
    
    response_text = message.content[0].text
    
    # Parse the JSON response
    game_state = json.loads(response_text)
    
    # Add metadata
    game_state['metadata'] = {
        'created_at': datetime.now().isoformat(),
        'last_saved': datetime.now().isoformat(),
        'save_version': '1.0'
    }
    
    return game_state

def serialize_game_state(game_state: dict, filename: str) -> None:
    """Serialize game state to a file using pickle."""
    with open(filename, 'wb') as f:
        pickle.dump(game_state, f)
    print(f"✅ Game state successfully saved to '{filename}'")

def deserialize_game_state(filename: str) -> dict:
    """Deserialize game state from a file using pickle."""
    with open(filename, 'rb') as f:
        game_state = pickle.load(f)
    print(f"✅ Game state successfully loaded from '{filename}'")
    return game_state

def display_game_state(game_state: dict) -> None:
    """Display game state information in a readable format."""
    print("\n" + "="*60)
    print("🎮 GAME STATE INFORMATION")
    print("="*60)
    
    if 'player_info' in game_state:
        player = game_state['player_info']
        print(f"\n👤 PLAYER: {player.get('name', 'Unknown')}")
        print(f"   Level: {player.get('level', 0)}")
        print(f"   Experience: {player.get('experience', 0)}")
        print(f"   Health: {player.get('health', 0)}")
        print(f"   Mana: {player.get('mana', 0)}")
    
    if 'location' in game_state:
        location = game_state['location']
        print(f"\n📍 LOCATION: {location.get('current_area', 'Unknown')}")
    
    if 'inventory' in game_state:
        inventory = game_state['inventory']
        print(f"\n🎒 INVENTORY ({len(inventory)} items):")
        for item in inventory[:3]:  # Show first 3 items
            print(f"   - {item.get('name', 'Unknown')}: {item.get('quantity', 0)}x")
        if len(inventory) > 3:
            print(f"   ... and {len(inventory) - 3} more items")
    
    if 'skills' in game_state:
        skills = game_state['skills']
        print(f"\n⚔️  SKILLS ({len(skills)} skills):")
        for skill in skills[:3]:  # Show first 3 skills
            print(f"   - {skill.get('name', 'Unknown')} (Level {skill.get('level', 0)})")
        if len(skills) > 3:
            print(f"   ... and {len(skills) - 3} more skills")
    
    if 'quests' in game_state:
        quests = game_state['quests']
        active = quests.get('active', [])
        completed = quests.get('completed', [])
        print(f"\n📜 QUESTS: {len(active)} active, {len(completed)} completed")
    
    if 'statistics' in game_state:
        stats = game_state['statistics']
        print(f"\n📊 STATISTICS:")
        print(f"   Kills: {stats.get('kills', 0)}")
        print(f"   Deaths: {stats.get('deaths', 0)}")
        print(f"   Playtime: {stats.get('playtime_hours', 0)} hours")
    
    if 'achievements' in game_state:
        achievements = game_state['achievements']
        print(f"\n🏆 ACHIEVEMENTS: {len(achievements)} unlocked")
    
    if 'metadata' in game_state:
        metadata = game_state['metadata']
        print(f"\n💾 SAVE INFO:")
        print(f"   Created: {metadata.get('created_at', 'Unknown')}")
        print(f"   Last Saved: {metadata.get('last_saved', 'Unknown')}")
        print(f"   Version: {metadata.get('save_version', 'Unknown')}")
    
    print("="*60)

def update_game_state_with_claude(game_state: dict, action: str) -> dict:
    """Use Claude to update the game state based on a player action."""
    
    current_state_json = json.dumps(game_state, indent=2)
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Given this current game state:
                {current_state_json}
                
                The player performed this action: '{action}'
                
                Update the game state accordingly. Make realistic changes based on the action.
                Update the 'last_saved' in metadata to current time.
                
                Return ONLY the updated JSON game state, no explanations."""
            }
        ]
    )
    
    response_text = message.content[0].text
    
    # Parse the updated JSON
    updated_state = json.loads(response_text)
    
    # Ensure metadata is updated
    if 'metadata' not in updated_state:
        updated_state['metadata'] = game_state.get('metadata', {})
    updated_state['metadata']['last_saved'] = datetime.now().isoformat()
    
    return updated_state

def main():
    save_file = "game_save.pkl"
    player_name = "HeroPlayer"
    
    print("🎮 RPG Game State Manager with Claude AI")
    print("="*60)
    
    # Step 1: Create initial game state using Claude
    print(f"\n📝 Creating new game state for player '{player_name}'...")
    game_state = create_game_state_with_claude(player_name)
    
    # Step 2: Display the created game state
    print("\n🆕 INITIAL GAME STATE:")
    display_game_state(game_state)
    
    # Step 3: Serialize (save) the game state
    print("\n💾 Saving game state...")
    serialize_game_state(game_state, save_file)
    
    # Step 4: Simulate some gameplay and update state
    print("\n⚔️  Simulating gameplay...")
    action = "defeated a powerful dragon boss and gained 5000 experience points"
    print(f"   Action: {action}")
    
    updated_state = update_game_state_with_claude(game_state, action)
    
    # Step 5: Save the updated state
    print("\n💾 Saving updated game state...")
    serialize_game_state(updated_state, save_file)
    
    # Step 6: Deserialize (load) the game state
    print("\n📂 Loading game state from file...")
    loaded_state = deserialize_game_state(save_file)
    
    # Step 7: Display the loaded game state
    print("\n📊 LOADED GAME STATE (after gameplay):")
    display_game_state(loaded_state)
    
    # Step 8: Verify the data integrity
    print("\n🔍 Verifying data integrity...")
    if loaded_state == updated_state:
        print("✅ Data integrity verified! Saved and loaded states match perfectly.")
    else:
        print("⚠️  Warning: Some differences detected between saved and loaded states.")
    
    # Step 9: Show file information
    file_size = os.path.getsize(save_file)
    print(f"\n📁 Save file information:")
    print(f"   File: {save_file}")
    print(f"   Size: {file_size} bytes")
    
    print("\n✨ Game state serialization/deserialization completed successfully!")
    
    # Clean up
    if os.path.exists(save_file):
        os.remove(save_file)
        print(f"🗑️  Cleaned up save file: {save_file}")

if __name__ == "__main__":
    main()
```
