from flask import Flask, request, jsonify
import anthropic

app = Flask(__name__)

client = anthropic.Anthropic()

@app.route('/calc', methods=['POST'])
def calculate():
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Please provide a mathematical expression in the request body with key "expression"'
        }), 400
    
    expression = data['expression']
    
    if not expression or not expression.strip():
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    prompt = f"""Evaluate the following mathematical expression and return ONLY the numerical result. 
    Do not include any explanation, units, or additional text - just the number.
    
    Expression: {expression}
    
    Result:"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    result = message.content[0].text.strip()
    
    return jsonify({
        'expression': expression,
        'result': result
    })

@app.route('/calc', methods=['GET'])
def calculate_get():
    expression = request.args.get('expression')
    
    if not expression:
        return jsonify({
            'error': 'Please provide a mathematical expression as query parameter "expression"'
        }), 400
    
    if not expression.strip():
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    prompt = f"""Evaluate the following mathematical expression and return ONLY the numerical result. 
    Do not include any explanation, units, or additional text - just the number.
    
    Expression: {expression}
    
    Result:"""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    result = message.content[0].text.strip()
    
    return jsonify({
        'expression': expression,
        'result': result
    })

@app.route('/', methods=['GET'])
def index():
    return jsonify({
        'message': 'Mathematical Expression Calculator API',
        'endpoints': {
            'POST /calc': 'Send JSON body with "expression" key',
            'GET /calc': 'Send query parameter "expression"'
        },
        'examples': {
            'POST': {
                'url': '/calc',
                'body': {'expression': '2 + 2 * 3'}
            },
            'GET': {
                'url': '/calc?expression=sqrt(16)+5'
            }
        }
    })

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
