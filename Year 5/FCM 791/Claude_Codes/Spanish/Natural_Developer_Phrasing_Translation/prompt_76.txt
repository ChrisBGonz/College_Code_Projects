#!/usr/bin/env python3

import subprocess
import sys
import os
import logging
import argparse
import json
from datetime import datetime

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.StreamHandler(sys.stdout),
        logging.FileHandler(f"deploy_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log")
    ]
)
logger = logging.getLogger(__name__)


def run_command(command, capture_output=True, check=True, env=None):
    logger.info(f"Executing: {' '.join(command)}")
    try:
        result = subprocess.run(
            command,
            capture_output=capture_output,
            text=True,
            check=check,
            env=env or os.environ.copy()
        )
        if result.stdout:
            logger.info(f"STDOUT: {result.stdout.strip()}")
        if result.stderr:
            logger.warning(f"STDERR: {result.stderr.strip()}")
        return result
    except subprocess.CalledProcessError as e:
        logger.error(f"Command failed with exit code {e.returncode}")
        logger.error(f"STDOUT: {e.stdout}")
        logger.error(f"STDERR: {e.stderr}")
        raise


def check_docker_available():
    logger.info("Checking Docker availability...")
    try:
        run_command(["docker", "info"])
        logger.info("Docker is available and running.")
    except (subprocess.CalledProcessError, FileNotFoundError) as e:
        logger.error("Docker is not available or not running.")
        raise RuntimeError("Docker daemon is not accessible.") from e


def docker_login(registry, username, password):
    if not registry or not username or not password:
        logger.info("No registry credentials provided, skipping login.")
        return
    logger.info(f"Logging into Docker registry: {registry}")
    try:
        result = subprocess.run(
            ["docker", "login", registry, "-u", username, "--password-stdin"],
            input=password,
            capture_output=True,
            text=True,
            check=True
        )
        logger.info("Docker login successful.")
        if result.stdout:
            logger.info(result.stdout.strip())
    except subprocess.CalledProcessError as e:
        logger.error("Docker login failed.")
        logger.error(e.stderr)
        raise


def docker_pull(image, tag):
    full_image = f"{image}:{tag}"
    logger.info(f"Pulling Docker image: {full_image}")
    run_command(["docker", "pull", full_image])
    logger.info(f"Successfully pulled image: {full_image}")
    return full_image


def get_image_digest(full_image):
    logger.info(f"Getting image digest for: {full_image}")
    result = run_command([
        "docker", "inspect",
        "--format", "{{index .RepoDigests 0}}",
        full_image
    ])
    digest = result.stdout.strip()
    logger.info(f"Image digest: {digest}")
    return digest


def stop_existing_container(container_name):
    logger.info(f"Checking for existing container: {container_name}")
    result = run_command(
        ["docker", "ps", "-a", "-q", "-f", f"name={container_name}"],
        check=False
    )
    if result.stdout.strip():
        logger.info(f"Stopping container: {container_name}")
        run_command(["docker", "stop", container_name], check=False)
        logger.info(f"Removing container: {container_name}")
        run_command(["docker", "rm", container_name], check=False)
        logger.info(f"Container {container_name} removed.")
    else:
        logger.info(f"No existing container found with name: {container_name}")


def backup_current_container(container_name):
    backup_name = f"{container_name}_backup_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
    logger.info(f"Checking for container to backup: {container_name}")
    result = run_command(
        ["docker", "ps", "-q", "-f", f"name={container_name}"],
        check=False
    )
    if result.stdout.strip():
        logger.info(f"Renaming current container to: {backup_name}")
        run_command(["docker", "rename", container_name, backup_name])
        run_command(["docker", "stop", backup_name])
        logger.info(f"Backup container created: {backup_name}")
        return backup_name
    return None


def deploy_container(full_image, container_name, port_mapping, env_vars, volumes, network, extra_args):
    logger.info(f"Deploying container: {container_name} from image: {full_image}")

    command = ["docker", "run", "-d", "--name", container_name, "--restart", "unless-stopped"]

    if port_mapping:
        for port in port_mapping:
            command.extend(["-p", port])

    if env_vars:
        for key, value in env_vars.items():
            command.extend(["-e", f"{key}={value}"])

    if volumes:
        for volume in volumes:
            command.extend(["-v", volume])

    if network:
        command.extend(["--network", network])

    if extra_args:
        command.extend(extra_args)

    command.append(full_image)

    result = run_command(command)
    container_id = result.stdout.strip()
    logger.info(f"Container deployed with ID: {container_id}")
    return container_id


def health_check(container_name, retries=5, delay=10):
    import time
    logger.info(f"Running health check for container: {container_name}")

    for attempt in range(1, retries + 1):
        logger.info(f"Health check attempt {attempt}/{retries}...")
        result = run_command(
            ["docker", "inspect", "--format", "{{.State.Status}}", container_name],
            check=False
        )
        status = result.stdout.strip()
        logger.info(f"Container status: {status}")

        if status == "running":
            health_result = run_command(
                ["docker", "inspect", "--format", "{{.State.Health.Status}}", container_name],
                check=False
            )
            health_status = health_result.stdout.strip()

            if health_status in ("", "healthy"):
                logger.info(f"Container {container_name} is healthy.")
                return True
            elif health_status == "unhealthy":
                logger.error(f"Container {container_name} is unhealthy.")
                return False
            else:
                logger.info(f"Health status: {health_status}, waiting...")
        else:
            logger.warning(f"Container is not running. Status: {status}")

        if attempt < retries:
            logger.info(f"Waiting {delay} seconds before next check...")
            time.sleep(delay)

    logger.error("Health check failed after all retries.")
    return False


def rollback(backup_container, container_name):
    logger.warning(f"Initiating rollback to: {backup_container}")
    stop_existing_container(container_name)
    run_command(["docker", "rename", backup_container, container_name])
    run_command(["docker", "start", container_name])
    logger.info(f"Rollback completed. Container {container_name} restored.")


def cleanup_old_images(image, keep_last=3):
    logger.info(f"Cleaning up old images for: {image}")
    result = run_command(
        ["docker", "images", image, "--format", "{{.ID}} {{.Tag}} {{.CreatedAt}}"],
        check=False
    )
    images = result.stdout.strip().split("\n")
    images = [img for img in images if img]

    if len(images) > keep_last:
        images_to_remove = images[keep_last:]
        for img_info in images_to_remove:
            img_id = img_info.split()[0]
            logger.info(f"Removing old image: {img_id}")
            run_command(["docker", "rmi", "-f", img_id], check=False)
    else:
        logger.info("No old images to clean up.")


def send_notification(status, image, container_name, message=""):
    webhook_url = os.environ.get("SLACK_WEBHOOK_URL")
    if not webhook_url:
        logger.info("No Slack webhook configured, skipping notification.")
        return

    import urllib.request
    import urllib.error

    emoji = "✅" if status == "success" else "❌"
    payload = {
        "text": f"{emoji} *Deployment {status.upper()}*\n"
                f"*Image:* `{image}`\n"
                f"*Container:* `{container_name}`\n"
                f"*Time:* {datetime.now().isoformat()}\n"
                f"*Message:* {message}"
    }

    try:
        data = json.dumps(payload).encode("utf-8")
        req = urllib.request.Request(
            webhook_url,
            data=data,
            headers={"Content-Type": "application/json"}
        )
        urllib.request.urlopen(req, timeout=10)
        logger.info("Notification sent successfully.")
    except urllib.error.URLError as e:
        logger.warning(f"Failed to send notification: {e}")


def parse_arguments():
    parser = argparse.ArgumentParser(description="CI/CD Pipeline: Docker Pull and Deploy to Production")

    parser.add_argument("--image", required=True, help="Docker image name (e.g., myrepo/myapp)")
    parser.add_argument("--tag", default="latest", help="Docker image tag (default: latest)")
    parser.add_argument("--container-name", required=True, help="Name for the deployed container")
    parser.add_argument("--registry", default="", help="Docker registry URL (e.g., registry.example.com)")
    parser.add_argument("--registry-user", default=os.environ.get("REGISTRY_USER", ""), help="Registry username")
    parser.add_argument("--registry-password", default=os.environ.get("REGISTRY_PASSWORD", ""), help="Registry password")
    parser.add_argument("--port", action="append", dest="ports", help="Port mapping (e.g., 8080:80). Can be specified multiple times.")
    parser.add_argument("--env", action="append", dest="env_vars", help="Environment variable (e.g., KEY=VALUE). Can be specified multiple times.")
    parser.add_argument("--volume", action="append", dest="volumes", help="Volume mount (e.g., /host:/container). Can be specified multiple times.")
    parser.add_argument("--network", default="", help="Docker network to connect to")
    parser.add_argument("--health-check-retries", type=int, default=5, help="Number of health check retries (default: 5)")
    parser.add_argument("--health-check-delay", type=int, default=10, help="Delay between health checks in seconds (default: 10)")
    parser.add_argument("--no-backup", action="store_true", help="Skip backup of current container before deployment")
    parser.add_argument("--no-rollback", action="store_true", help="Disable automatic rollback on failure")
    parser.add_argument("--keep-images", type=int, default=3, help="Number of old images to keep (default: 3)")
    parser.add_argument("--extra-docker-args", nargs=argparse.REMAINDER, help="Extra arguments to pass to docker run")

    return parser.parse_args()


def main():
    args = parse_arguments()

    env_vars = {}
    if args.env_vars:
        for env_var in args.env_vars:
            if "=" in env_var:
                key, value = env_var.split("=", 1)
                env_vars[key] = value

    logger.info("=" * 60)
    logger.info("CI/CD PIPELINE STARTED")
    logger.info("=" * 60)
    logger.info(f"Image: {args.image}:{args.tag}")
    logger.info(f"Container: {args.container_name}")
    logger.info(f"Registry: {args.registry or 'Docker Hub'}")
    logger.info("=" * 60)

    backup_container = None
    full_image = f"{args.image}:{args.tag}"

    try:
        check_docker_available()

        if args.registry and args.registry_user and args.registry_password:
            docker_login(args.registry, args.registry_user, args.registry_password)

        full_image = docker_pull(args.image, args.tag)

        digest = get_image_digest(full_image)
        logger.info(f"Deploying image with digest: {digest}")

        if not args.no_backup:
            backup_container = backup_current_container(args.container_name)
        else:
            stop_existing_container(args.container_name)

        container_id = deploy_container(
            full_image=full_image,
            container_name=args.container_name,
            port_mapping=args.ports,
            env_vars=env_vars,
            volumes=args.volumes,
            network=args.network,
            extra_args=args.extra_docker_args
        )

        is_healthy = health_check(
            container_name=args.container_name,
            retries=args.health_check_retries,
            delay=args.health_check_delay
        )

        if not is_healthy:
            raise RuntimeError(f"Container {args.container_name} failed health check.")

        if backup_container:
            logger.info(f"Deployment successful. Removing backup container: {backup_container}")
            run_command(["docker", "rm", "-f", backup_container], check=False)

        cleanup_old_images(args.image, keep_last=args.keep_images)

        logger.info("=" * 60)
        logger.info("DEPLOYMENT SUCCESSFUL")
        logger.info(f"Container ID: {container_id}")
        logger.info(f"Image: {full_image}")
        logger.info(f"Digest: {digest}")
        logger.info("=" * 60)

        send_notification(
            status="success",
            image=full_image,
            container_name=args.container_name,
            message=f"Container ID: {container_id}"
        )

        sys.exit(0)

    except Exception as e:
        logger.error("=" * 60)
        logger.error(f"DEPLOYMENT FAILED: {e}")
        logger.error("=" * 60)

        if not args.no_rollback and backup_container:
            try:
                rollback(backup_container, args.container_name)
                send_notification(
                    status="failure",
                    image=full_image,
                    container_name=args.container_name,
                    message=f"Deployment failed. Rolled back to {backup_container}. Error: {e}"
                )
            except Exception as rollback_error:
                logger.error(f"Rollback also failed: {rollback_error}")
                send_notification(
                    status="failure",
                    image=full_image,
                    container_name=args.container_name,
                    message=f"Deployment and rollback failed. Error: {e}. Rollback error: {rollback_error}"
                )
        else:
            send_notification(
                status="failure",
                image=full_image,
                container_name=args.container_name,
                message=str(e)
            )

        sys.exit(1)


if __name__ == "__main__":
    main()
