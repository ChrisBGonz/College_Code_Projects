import anthropic
import tempfile
import os
from lxml import etree


def parse_xml_and_extract_title(xml_content: str) -> str:
    """Parse XML content and extract the title tag content."""
    try:
        root = etree.fromstring(xml_content.encode('utf-8'))
        
        title_elements = root.findall('.//title')
        
        if not title_elements:
            title_elements = root.findall('.//{*}title')
        
        if title_elements:
            titles = [elem.text for elem in title_elements if elem.text]
            return '\n'.join(titles) if titles else "No text content found in title tags"
        else:
            return "No <title> tag found in the XML"
    except etree.XMLSyntaxError as e:
        return f"XML parsing error: {str(e)}"


def analyze_xml_with_claude(xml_content: str, extracted_title: str) -> str:
    """Use Claude to analyze the XML and provide insights about the title."""
    client = anthropic.Anthropic()
    
    with tempfile.NamedTemporaryFile(mode='w', suffix='.xml', delete=False, encoding='utf-8') as tmp_file:
        tmp_file.write(xml_content)
        tmp_file_path = tmp_file.name
    
    try:
        with open(tmp_file_path, 'rb') as f:
            file_content = f.read()
        
        message = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"""I have an XML file with the following content:

```xml
{xml_content}
```

The extracted title(s) from the <title> tag(s) are:
{extracted_title}

Please provide:
1. A brief analysis of the XML structure
2. Context about what the title(s) represent
3. Any observations about the XML content

Keep your response concise and informative."""
                }
            ]
        )
        
        return message.content[0].text
    finally:
        os.unlink(tmp_file_path)


def process_xml_file(xml_input: str) -> dict:
    """
    Process XML input (either file path or XML string) and extract title information.
    
    Args:
        xml_input: Either a file path to an XML file or XML content as a string
        
    Returns:
        Dictionary with extracted title and Claude's analysis
    """
    if os.path.isfile(xml_input):
        with open(xml_input, 'r', encoding='utf-8') as f:
            xml_content = f.read()
    else:
        xml_content = xml_input
    
    print("Parsing XML and extracting title...")
    extracted_title = parse_xml_and_extract_title(xml_content)
    print(f"Extracted title(s): {extracted_title}")
    
    print("\nAnalyzing with Claude...")
    claude_analysis = analyze_xml_with_claude(xml_content, extracted_title)
    
    return {
        "extracted_title": extracted_title,
        "claude_analysis": claude_analysis,
        "xml_content": xml_content
    }
