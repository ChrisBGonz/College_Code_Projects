const http = require("http");
const url = require("url");

const client = new Anthropic();

async function generateProfileDataWithClaude(userData) {
  const conversationHistory = [];

  // First message: Ask Claude to analyze and enhance the profile data
  conversationHistory.push({
    role: "user",
    content: `I have a user profile with the following data: ${JSON.stringify(userData, null, 2)}. 
    Please analyze this profile and provide a brief summary of the user's profile in a professional manner.`,
  });

  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profiles and provides professional summaries. Always respond in a concise and professional manner.",
    messages: conversationHistory,
  });

  const firstAssistantMessage =
    firstResponse.content[0].type === "text"
      ? firstResponse.content[0].text
      : "";

  conversationHistory.push({
    role: "assistant",
    content: firstAssistantMessage,
  });

  // Second message: Ask for recommendations
  conversationHistory.push({
    role: "user",
    content:
      "Based on this profile, what are 3 personalized recommendations you would make for this user to improve their profile or experience?",
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profiles and provides professional summaries. Always respond in a concise and professional manner.",
    messages: conversationHistory,
  });

  const secondAssistantMessage =
    secondResponse.content[0].type === "text"
      ? secondResponse.content[0].text
      : "";

  conversationHistory.push({
    role: "assistant",
    content: secondAssistantMessage,
  });

  // Third message: Generate a complete profile export summary
  conversationHistory.push({
    role: "user",
    content:
      "Now, please create a final export summary that includes the profile analysis and recommendations in a structured format suitable for a JSON export.",
  });

  const thirdResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 800,
    system:
      "You are a helpful assistant that analyzes user profiles and provides professional summaries. Always respond in a concise and professional manner.",
    messages: conversationHistory,
  });

  const finalSummary =
    thirdResponse.content[0].type === "text"
      ? thirdResponse.content[0].text
      : "";

  return {
    profileSummary: firstAssistantMessage,
    recommendations: secondAssistantMessage,
    exportSummary: finalSummary,
  };
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  // Handle the profile download endpoint
  if (pathname === "/api/profile/download" && req.method === "GET") {
    const userId = query.user_id;

    // Validate user_id parameter
    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing required parameter: user_id",
          message: "Please provide a user_id in the query string",
        })
      );
      return;
    }

    // Check if user exists
    const userData = usersDatabase[userId];
    if (!userData) {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "User not found",
          message: `No user found with id: ${userId}`,
        })
      );
      return;
    }

    try {
      console.log(`Processing profile download for user ${userId}...`);

      // Generate enhanced profile data using Claude with multi-turn conversation
      const claudeAnalysis = await generateProfileDataWithClaude(userData);

      // Create the complete profile export
      const profileExport = {
        exportMetadata: {
          exportDate: new Date().toISOString(),
          exportVersion: "1.0",
          generatedBy: "Profile Export API with Claude AI",
        },
        userProfile: userData,
        aiAnalysis: {
          profileSummary: claudeAnalysis.profileSummary,
          recommendations: claudeAnalysis.recommendations,
          exportSummary: claudeAnalysis.exportSummary,
        },
      };

      // Set headers for file download
      const filename = `profile_user_${userId}_${Date.now()}.json`;
      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="${filename}"`,
        "X-User-Id": userId,
        "X-Export-Date": new Date().toISOString(),
      });

      res.end(JSON.stringify(profileExport, null, 2));
      console.log(`Profile download completed for user ${userId}`);
    } catch (error) {
      console.error("Error generating profile data:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal server error",
          message: "Failed to generate profile data",
        })
      );
    }
  } else if (pathname === "/api/users" && req.method === "GET") {
    // Helper endpoint to list available users
    const userList = Object.values(usersDatabase).map((user) => ({
      id: user.id,
      name: user.name,
      email: user.email,
    }));

    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Available users for testing",
        users: userList,
        downloadEndpoint: "/api/profile/download?user_id={id}",
      })
    );
  } else {
    // Handle unknown routes
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not found",
        availableEndpoints: [
          "GET /api/profile/download?user_id={id} - Download user profile as JSON",
          "GET /api/users - List available users",
        ],
      })
    );
  }
});

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log("\nAvailable endpoints:");
  console.log(
    `  GET http://localhost:${PORT}/api/profile/download?user_id=1`
  );
  console.log(`  GET http://localhost:${PORT}/api/users`);
  console.log("\nTest users available: 1, 2, 3");
});
