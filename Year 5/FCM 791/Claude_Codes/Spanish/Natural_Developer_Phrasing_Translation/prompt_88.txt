using System;
using System.Collections.Generic;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class MFAConfigurationLogger
{
    private static AnthropicClient _client = new AnthropicClient();
    private static List<Message> _conversationHistory = new List<Message>();

    static async Task Main(string[] args)
    {
        Console.WriteLine("=== MFA Configuration Change Logger ===");
        Console.WriteLine("This system logs and analyzes MFA configuration changes.");
        Console.WriteLine("Type 'exit' to quit.\n");

        // Initialize the conversation with system context
        string systemPrompt = @"You are a security audit assistant specialized in Multi-Factor Authentication (MFA) configuration changes. 
        Your role is to:
        1. Analyze MFA configuration change events
        2. Assess the security implications of each change
        3. Generate detailed audit log entries
        4. Provide security recommendations when needed
        5. Identify potentially suspicious patterns
        
        When analyzing MFA changes, consider:
        - Type of change (enabled, disabled, method changed, backup codes regenerated, etc.)
        - User context (admin, regular user, service account)
        - Time and location of the change
        - Previous configuration vs new configuration
        - Risk level of the change
        
        Format your responses as structured audit log entries with:
        - Timestamp
        - Event Type
        - Risk Level (LOW/MEDIUM/HIGH/CRITICAL)
        - User Information
        - Change Details
        - Security Assessment
        - Recommended Actions (if any)";

        while (true)
        {
            Console.Write("\nDescribe the MFA configuration change event (or 'exit' to quit): ");
            string? userInput = Console.ReadLine();

            if (string.IsNullOrWhiteSpace(userInput))
                continue;

            if (userInput.ToLower() == "exit")
            {
                Console.WriteLine("Exiting MFA Logger. Goodbye!");
                break;
            }

            // Add user message to conversation history
            _conversationHistory.Add(new Message
            {
                Role = RoleType.User,
                Content = userInput
            });

            try
            {
                // Create the message request with conversation history
                var messageRequest = new MessageParameters
                {
                    Model = AnthropicModels.Claude3_5Sonnet,
                    MaxTokens = 1024,
                    System = new List<SystemMessage>
                    {
                        new SystemMessage { Text = systemPrompt }
                    },
                    Messages = _conversationHistory
                };

                Console.WriteLine("\n--- Generating Audit Log Entry ---");
                
                // Stream the response
                string fullResponse = "";
                await foreach (var chunk in _client.Messages.StreamClaudeMessageAsync(messageRequest))
                {
                    if (chunk.Delta?.Text != null)
                    {
                        Console.Write(chunk.Delta.Text);
                        fullResponse += chunk.Delta.Text;
                    }
                }
                
                Console.WriteLine("\n--- End of Audit Log Entry ---");

                // Add assistant response to conversation history
                _conversationHistory.Add(new Message
                {
                    Role = RoleType.Assistant,
                    Content = fullResponse
                });

                // Log to file (simulated)
                await LogToFile(userInput, fullResponse);
                
                Console.WriteLine("\n[Log entry saved to audit log]");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"\nError processing MFA change event: {ex.Message}");
            }
        }
    }

    static async Task LogToFile(string changeDescription, string auditEntry)
    {
        string logFileName = $"mfa_audit_log_{DateTime.Now:yyyyMMdd}.txt";
        string logEntry = $"\n{'='*60}\n" +
                         $"TIMESTAMP: {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss UTC}\n" +
                         $"CHANGE DESCRIPTION: {changeDescription}\n" +
                         $"AUDIT ANALYSIS:\n{auditEntry}\n" +
                         $"{'='*60}\n";
        
        await File.AppendAllTextAsync(logFileName, logEntry);
    }
}
