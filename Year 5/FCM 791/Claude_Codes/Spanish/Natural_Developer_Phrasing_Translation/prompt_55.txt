TRIAL_DURATION_DAYS = 14
COOKIE_NAME = "has_used_trial"
SESSION_COOKIE_NAME = "session_id"

users_db = {}
trials_db = {}
sessions_db = {}


def generate_session_id():
    return str(uuid.uuid4())


def hash_password(password):
    return hashlib.sha256(password.encode()).hexdigest()


def create_trial_record(user_id, email):
    trial_id = str(uuid.uuid4())
    start_date = datetime.now()
    end_date = start_date + timedelta(days=TRIAL_DURATION_DAYS)
    
    trial = {
        "trial_id": trial_id,
        "user_id": user_id,
        "email": email,
        "start_date": start_date.isoformat(),
        "end_date": end_date.isoformat(),
        "is_active": True,
        "days_remaining": TRIAL_DURATION_DAYS
    }
    
    trials_db[user_id] = trial
    return trial


def check_trial_cookie(cookie_header):
    if not cookie_header:
        return None
    
    try:
        cookie = http.cookies.SimpleCookie()
        cookie.load(cookie_header)
        
        if COOKIE_NAME in cookie:
            cookie_value = cookie[COOKIE_NAME].value
            try:
                trial_data = json.loads(cookie_value)
                return trial_data
            except json.JSONDecodeError:
                return {"raw_value": cookie_value, "has_used_trial": True}
        
        return None
    except Exception:
        return None


def validate_trial_status(trial_data):
    if not trial_data:
        return {"valid": False, "reason": "No trial cookie found"}
    
    if "end_date" in trial_data:
        try:
            end_date = datetime.fromisoformat(trial_data["end_date"])
            if datetime.now() > end_date:
                return {
                    "valid": False,
                    "expired": True,
                    "reason": "Trial period has expired",
                    "end_date": trial_data["end_date"]
                }
            else:
                days_remaining = (end_date - datetime.now()).days
                return {
                    "valid": True,
                    "expired": False,
                    "reason": "Trial is still active",
                    "days_remaining": days_remaining,
                    "end_date": trial_data["end_date"]
                }
        except ValueError:
            pass
    
    return {
        "valid": False,
        "expired": True,
        "reason": "User has already used their free trial"
    }


def build_trial_cookie(trial_data, max_age=None):
    cookie = http.cookies.SimpleCookie()
    cookie_value = json.dumps({
        "has_used_trial": True,
        "trial_id": trial_data.get("trial_id", ""),
        "user_id": trial_data.get("user_id", ""),
        "start_date": trial_data.get("start_date", ""),
        "end_date": trial_data.get("end_date", ""),
        "email": trial_data.get("email", "")
    })
    
    cookie[COOKIE_NAME] = cookie_value
    cookie[COOKIE_NAME]["path"] = "/"
    cookie[COOKIE_NAME]["httponly"] = True
    cookie[COOKIE_NAME]["samesite"] = "Strict"
    
    if max_age:
        cookie[COOKIE_NAME]["max-age"] = max_age
    else:
        end_date = datetime.fromisoformat(trial_data["end_date"])
        max_age_seconds = int((end_date - datetime.now()).total_seconds())
        cookie[COOKIE_NAME]["max-age"] = max(0, max_age_seconds)
    
    return cookie.output(header="Set-Cookie")


def build_session_cookie(session_id):
    cookie = http.cookies.SimpleCookie()
    cookie[SESSION_COOKIE_NAME] = session_id
    cookie[SESSION_COOKIE_NAME]["path"] = "/"
    cookie[SESSION_COOKIE_NAME]["httponly"] = True
    cookie[SESSION_COOKIE_NAME]["max-age"] = 86400
    return cookie.output(header="Set-Cookie")

    def get_cookies(self):
        cookie_header = self.headers.get("Cookie", "")
        return check_trial_cookie(cookie_header)
    
    def get_session(self):
        cookie_header = self.headers.get("Cookie", "")
        if not cookie_header:
            return None
        
        try:
            cookie = http.cookies.SimpleCookie()
            cookie.load(cookie_header)
            if SESSION_COOKIE_NAME in cookie:
                session_id = cookie[SESSION_COOKIE_NAME].value
                return sessions_db.get(session_id)
        except Exception:
            return None
    
    def send_html_response(self, status_code, html_content, extra_headers=None):
        self.send_response(status_code)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", len(html_content.encode("utf-8")))
        if extra_headers:
            for header_name, header_value in extra_headers:
                self.send_header(header_name, header_value)
        self.end_headers()
        self.wfile.write(html_content.encode("utf-8"))
    
    def send_json_response(self, status_code, data, extra_headers=None):
        json_content = json.dumps(data, indent=2)
        self.send_response(status_code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", len(json_content.encode("utf-8")))
        if extra_headers:
            for header_name, header_value in extra_headers:
                self.send_header(header_name, header_value)
        self.end_headers()
        self.wfile.write(json_content.encode("utf-8"))
    
    def read_body(self):
        content_length = int(self.headers.get("Content-Length", 0))
        if content_length > 0:
            return self.rfile.read(content_length).decode("utf-8")
        return ""
    
    def parse_form_data(self, body):
        params = {}
        if body:
            pairs = body.split("&")
            for pair in pairs:
                if "=" in pair:
                    key, value = pair.split("=", 1)
                    params[key] = value.replace("+", " ").replace("%40", "@").replace("%2B", "+")
        return params

    def handle_login_post(self):
        body = self.read_body()
        form_data = self.parse_form_data(body)
        
        email = form_data.get("email", "").strip()
        password = form_data.get("password", "").strip()
        
        if not email or not password:
            content = """
            <h1>❌ Login Error</h1>
            <div class="alert alert-error">Please provide both email and password.</div>
            <a href="/login"><button class="btn">Try Again</button></a>
            """
            html = get_html_response("Login Error", content)
            self.send_html_response(400, html)
            return
        
        user = None
        for uid, u in users_db.items():
            if u["email"] == email and u["password_hash"] == hash_password(password):
                user = u
                break
        
        if not user:
            content = """
            <h1>❌ Invalid Credentials</h1>
            <div class="alert alert-error">Invalid email or password. Please try again.</div>
            <a href="/login"><button class="btn">Try Again</button></a>
            <a href="/register"><button class="btn btn-secondary">Create Account</button></a>
            """
            html = get_html_response("Login Failed", content)
            self.send_html_response(401, html)
            return
        
        session_id = generate_session_id()
        sessions_db[session_id] = {
            "user_id": user["user_id"],
            "email": user["email"],
            "name": user["name"],
            "created_at": datetime.now().isoformat()
        }
        
        session_cookie_header = build_session_cookie(session_id)
        
        extra_headers = [("Set-Cookie", session_cookie_header.replace("Set-Cookie: ", ""))]
        
        trial = trials_db.get(user["user_id"])
        if trial:
            trial_cookie_header = build_trial_cookie(trial)
            extra_headers.append(("Set-Cookie", trial_cookie_header.replace("Set-Cookie: ", "")))
        
        self.send_response(302)
        self.send_header("Location", "/dashboard")
        for header_name, header_value in extra_headers:
            self.send_header(header_name, header_value)
        self.end_headers()
    
    def handle_logout(self):
        session = self.get_session()
        
        if session:
            cookie_header = self.headers.get("Cookie", "")
            try:
                cookie = http.cookies.SimpleCookie()
                cookie.load(cookie_header)
                if SESSION_COOKIE_NAME in cookie:
                    session_id = cookie[SESSION_COOKIE_NAME].value
                    sessions_db.pop(session_id, None)
            except Exception:
                pass
        
        expired_session_cookie = f"{SESSION_COOKIE_NAME}=; Path=/; Max-Age=0; HttpOnly"
        
        self.send_response(302)
        self.send_header("Location", "/")
        self.send_header("Set-Cookie", expired_session_cookie)
        self.end_headers()
