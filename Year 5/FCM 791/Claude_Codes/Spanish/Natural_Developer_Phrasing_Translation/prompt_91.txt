import anthropic
import json
import time
import random

client = anthropic.Anthropic()

tools = [
    {
        "name": "process_payment",
        "description": "Process a payment through the external payment API. Returns payment status and transaction details.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount in dollars"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., USD, EUR)"
                },
                "card_token": {
                    "type": "string",
                    "description": "Tokenized card information for the payment"
                },
                "merchant_id": {
                    "type": "string",
                    "description": "The merchant identifier"
                }
            },
            "required": ["amount", "currency", "card_token", "merchant_id"]
        }
    },
    {
        "name": "check_payment_status",
        "description": "Check the status of a previously initiated payment transaction.",
        "input_schema": {
            "type": "object",
            "properties": {
                "transaction_id": {
                    "type": "string",
                    "description": "The transaction ID to check"
                }
            },
            "required": ["transaction_id"]
        }
    }
]

def simulate_payment_api(tool_name: str, tool_input: dict, simulate_timeout: bool = False) -> dict:
    """
    Simulates an external payment API call.
    In real scenarios, this would make actual HTTP requests to a payment gateway.
    """
    if simulate_timeout:
        time.sleep(3)
        return {"error": "timeout"}
    
    if tool_name == "process_payment":
        transaction_id = f"TXN_{random.randint(100000, 999999)}"
        return {
            "success": True,
            "transaction_id": transaction_id,
            "status": "approved",
            "amount": tool_input["amount"],
            "currency": tool_input["currency"],
            "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
        }
    elif tool_name == "check_payment_status":
        return {
            "transaction_id": tool_input["transaction_id"],
            "status": "completed",
            "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
        }
    
    return {"error": "unknown_tool"}

def connect_to_payment_api(
    amount: float,
    currency: str,
    card_token: str,
    merchant_id: str,
    timeout: float = 2.0,
    simulate_timeout: bool = False
) -> bool:
    """
    Connects to an external payment API using Claude as an AI intermediary.
    
    If the API doesn't respond within the timeout period, returns True (fail-open)
    to allow the user to continue.
    
    Args:
        amount: Payment amount
        currency: Currency code
        card_token: Tokenized card information
        merchant_id: Merchant identifier
        timeout: Timeout in seconds (default: 2.0)
        simulate_timeout: If True, simulates a timeout scenario for testing
    
    Returns:
        True if payment was successful or if API timed out (fail-open)
        False if payment was explicitly rejected
    """
    
    start_time = time.time()
    
    try:
        messages = [
            {
                "role": "user",
                "content": f"""Process a payment with the following details:
                - Amount: {amount} {currency}
                - Card Token: {card_token}
                - Merchant ID: {merchant_id}
                
                Please use the process_payment tool to handle this transaction."""
            }
        ]
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        while response.stop_reason == "tool_use":
            tool_use_block = None
            for block in response.content:
                if block.type == "tool_use":
                    tool_use_block = block
                    break
            
            if not tool_use_block:
                break
            
            elapsed_time = time.time() - start_time
            if elapsed_time >= timeout:
                print(f"⚠️  API timeout after {elapsed_time:.2f}s - Applying fail-open policy")
                return True
            
            print(f"🔧 Calling tool: {tool_use_block.name}")
            print(f"   Input: {json.dumps(tool_use_block.input, indent=2)}")
            
            api_result = simulate_payment_api(
                tool_use_block.name,
                tool_use_block.input,
                simulate_timeout=simulate_timeout
            )
            
            elapsed_time = time.time() - start_time
            if elapsed_time >= timeout or api_result.get("error") == "timeout":
                print(f"⚠️  API timeout after {elapsed_time:.2f}s - Applying fail-open policy")
                return True
            
            print(f"   Result: {json.dumps(api_result, indent=2)}")
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use_block.id,
                        "content": json.dumps(api_result)
                    }
                ]
            })
            
            elapsed_time = time.time() - start_time
            remaining_timeout = max(0.1, timeout - elapsed_time)
            
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
        
        final_response = ""
        for block in response.content:
            if hasattr(block, "text"):
                final_response = block.text
                break
        
        print(f"\n📋 Payment Processing Result:")
        print(f"   {final_response}")
        
        if any(word in final_response.lower() for word in ["declined", "rejected", "failed", "error", "denied"]):
            print("❌ Payment explicitly rejected")
            return False
        
        print("✅ Payment successful")
        return True
        
    except Exception as e:
        elapsed_time = time.time() - start_time
        
        if elapsed_time >= timeout or "timeout" in str(e).lower():
            print(f"⚠️  Exception after {elapsed_time:.2f}s: {str(e)}")
            print("   Applying fail-open policy due to timeout")
            return True
        
        print(f"❌ Payment processing error: {str(e)}")
        return False

def main():
    """Main function to demonstrate the payment API integration."""
    
    print("=" * 60)

    print("\n📌 Test 1: Normal Payment Processing")
