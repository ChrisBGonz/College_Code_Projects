<?php

class UserPreferences {
    public $theme;
    public $language;
    public $fontSize;
    public $notifications;
    public $itemsPerPage;

    public function __construct(
        string $theme = 'light',
        string $language = 'es',
        int $fontSize = 14,
        bool $notifications = true,
        int $itemsPerPage = 10
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->notifications = $notifications;
        $this->itemsPerPage = $itemsPerPage;
    }

    public function __wakeup() {
        $allowedThemes = ['light', 'dark', 'high-contrast'];
        $allowedLanguages = ['es', 'en', 'fr', 'de', 'pt'];

        if (!in_array($this->theme, $allowedThemes)) {
            $this->theme = 'light';
        }

        if (!in_array($this->language, $allowedLanguages)) {
            $this->language = 'es';
        }

        $this->fontSize = max(10, min(24, (int)$this->fontSize));
        $this->notifications = (bool)$this->notifications;
        $this->itemsPerPage = max(5, min(100, (int)$this->itemsPerPage));
    }

    public function toArray(): array {
        return [
            'theme'         => $this->theme,
            'language'      => $this->language,
            'fontSize'      => $this->fontSize,
            'notifications' => $this->notifications,
            'itemsPerPage'  => $this->itemsPerPage,
        ];
    }
}

class PreferencesManager {
    private const COOKIE_NAME    = 'user_preferences';
    private const COOKIE_EXPIRY  = 30 * 24 * 60 * 60; // 30 días
    private const HMAC_SECRET    = 'c4mb14_e5t0_p0r_un4_c14v3_s3gur4_3n_pr0ducc10n';
    private const ALLOWED_CLASSES = ['UserPreferences'];

    public function savePreferences(UserPreferences $prefs): bool {
        $serialized = serialize($prefs);
        $hmac       = $this->generateHmac($serialized);
        $payload    = base64_encode($serialized) . '.' . $hmac;

        return setcookie(
            self::COOKIE_NAME,
            $payload,
            [
                'expires'  => time() + self::COOKIE_EXPIRY,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
    }

    public function loadPreferences(): ?UserPreferences {
        if (!isset($_COOKIE[self::COOKIE_NAME])) {
            return null;
        }

        $payload = $_COOKIE[self::COOKIE_NAME];

        if (!$this->validatePayloadFormat($payload)) {
            $this->clearPreferences();
            return null;
        }

        [$encodedData, $hmac] = explode('.', $payload, 2);

        $serialized = base64_decode($encodedData, true);

        if ($serialized === false) {
            $this->clearPreferences();
            return null;
        }

        if (!$this->verifyHmac($serialized, $hmac)) {
            error_log('Advertencia de seguridad: HMAC inválido en cookie de preferencias.');
            $this->clearPreferences();
            return null;
        }

        $prefs = $this->safeUnserialize($serialized);

        if (!($prefs instanceof UserPreferences)) {
            $this->clearPreferences();
            return null;
        }

        return $prefs;
    }

    public function clearPreferences(): void {
        setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires'  => time() - 3600,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
    }

    private function safeUnserialize(string $data): mixed {
        return unserialize($data, ['allowed_classes' => self::ALLOWED_CLASSES]);
    }

    private function generateHmac(string $data): string {
        return hash_hmac('sha256', $data, self::HMAC_SECRET);
    }

    private function verifyHmac(string $data, string $hmac): bool {
        $expected = $this->generateHmac($data);
        return hash_equals($expected, $hmac);
    }

    private function validatePayloadFormat(string $payload): bool {
        return substr_count($payload, '.') >= 1;
    }
}

function handlePreferencesUpdate(PreferencesManager $manager): ?UserPreferences {
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        return null;
    }

    $allowedThemes    = ['light', 'dark', 'high-contrast'];
    $allowedLanguages = ['es', 'en', 'fr', 'de', 'pt'];

    $theme         = $_POST['theme']         ?? 'light';
    $language      = $_POST['language']      ?? 'es';
    $fontSize      = (int)($_POST['fontSize']      ?? 14);
    $notifications = isset($_POST['notifications']);
    $itemsPerPage  = (int)($_POST['itemsPerPage']  ?? 10);

    if (!in_array($theme, $allowedThemes)) {
        $theme = 'light';
    }

    if (!in_array($language, $allowedLanguages)) {
        $language = 'es';
    }

    $fontSize     = max(10, min(24, $fontSize));
    $itemsPerPage = max(5, min(100, $itemsPerPage));

    $prefs = new UserPreferences($theme, $language, $fontSize, $notifications, $itemsPerPage);
    $manager->savePreferences($prefs);

    return $prefs;
}
