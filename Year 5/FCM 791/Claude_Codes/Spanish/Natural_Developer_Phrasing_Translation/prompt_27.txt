import anthropic
import subprocess
import sys


def install_package(package):
    """Install a package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])


def generate_pdf_conversion_function():
    """Use Claude to generate a Python function for HTML to PDF conversion."""
    client = anthropic.Anthropic()

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Proporciona una función en Python que use la librería 'weasyprint' para convertir plantillas HTML en archivos PDF. 
                
                La función debe:
                1. Aceptar una plantilla HTML como string o ruta de archivo
                2. Aceptar variables de contexto para renderizar la plantilla
                3. Generar un archivo PDF en la ruta especificada
                4. Manejar errores apropiadamente
                5. Incluir un ejemplo de uso
                
                Solo proporciona el código Python, sin explicaciones adicionales.""",
            }
        ],
    )

    return message.content[0].text


def html_to_pdf_with_weasyprint(
    html_content: str,
    output_path: str,
    context: dict = None,
    is_file: bool = False,
) -> bool:
    """
    Convert HTML template to PDF using WeasyPrint.

    Args:
        html_content: HTML string or file path to HTML file
        output_path: Path where the PDF will be saved
        context: Dictionary with variables to replace in the template
        is_file: If True, html_content is treated as a file path

    Returns:
        bool: True if conversion was successful, False otherwise
    """
    try:
        from weasyprint import HTML, CSS
        from weasyprint.text.fonts import FontConfiguration
    except ImportError:
        print("WeasyPrint not found. Installing...")
        install_package("weasyprint")
        from weasyprint import HTML, CSS
        from weasyprint.text.fonts import FontConfiguration

    try:
        if is_file:
            with open(html_content, "r", encoding="utf-8") as f:
                html_string = f.read()
        else:
            html_string = html_content

        if context:
            for key, value in context.items():
                html_string = html_string.replace(f"{{{{{key}}}}}", str(value))

        font_config = FontConfiguration()

        css = CSS(
            string="""
            @page {
                margin: 1cm;
            }
            body {
                font-family: Arial, sans-serif;
                font-size: 12pt;
            }
        """,
            font_config=font_config,
        )

        html_doc = HTML(string=html_string)
        html_doc.write_pdf(output_path, stylesheets=[css], font_config=font_config)

        print(f"PDF successfully generated at: {output_path}")
        return True

    except FileNotFoundError as e:
        print(f"Error: File not found - {e}")
        return False
    except Exception as e:
        print(f"Error converting HTML to PDF: {e}")
        return False
