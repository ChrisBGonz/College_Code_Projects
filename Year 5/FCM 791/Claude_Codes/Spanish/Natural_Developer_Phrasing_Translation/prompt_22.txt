YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
WHITE='\033[1;37m'
NC='\033[0m'

# Configuration

TMP_DIR="/tmp/sec_tool_installer_$$"
LOG_FILE="/var/log/sec_tool_installer.log"

# ============================================================
# Logging Functions
# ============================================================

log() {
    local level="$1"
    shift
    local message="$*"
    local timestamp
    timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    echo "[$timestamp] [$level] $message" >> "$LOG_FILE" 2>/dev/null || true
}

print_banner() {
    echo -e "${CYAN}"
    echo "╔══════════════════════════════════════════════════════════╗"
    echo "║          Security Tools Auto-Installer v2.0              ║"
    echo "║          Automated Download & Installation               ║"
    echo "╚══════════════════════════════════════════════════════════╝"
    echo -e "${NC}"
}

print_info() {
    echo -e "${BLUE}[INFO]${NC} $*"
    log "INFO" "$*"
}

print_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $*"
    log "SUCCESS" "$*"
}

print_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $*"
    log "WARNING" "$*"
}

print_error() {
    echo -e "${RED}[ERROR]${NC} $*"
    log "ERROR" "$*"
}

print_step() {
    echo -e "${WHITE}[STEP]${NC} $*"
    log "STEP" "$*"
}

# ============================================================
# Utility Functions
# ============================================================

check_root() {
    if [[ $EUID -ne 0 ]]; then
        print_warning "Not running as root. Some installations may require sudo."
        INSTALL_DIR="$HOME/.local/bin"
        mkdir -p "$INSTALL_DIR"
        export PATH="$PATH:$INSTALL_DIR"
    fi
}

check_dependencies() {
    print_step "Checking required dependencies..."
    local deps=("curl" "wget" "tar" "unzip" "jq" "file")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "$dep" &>/dev/null; then
            missing+=("$dep")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        print_warning "Missing dependencies: ${missing[*]}"
        install_dependencies "${missing[@]}"
    else
        print_success "All dependencies are available."
    fi
}

install_dependencies() {
    local deps=("$@")
    print_info "Installing missing dependencies: ${deps[*]}"

    if command -v apt-get &>/dev/null; then
        apt-get update -qq && apt-get install -y -qq "${deps[@]}" 2>/dev/null || true
    elif command -v yum &>/dev/null; then
        yum install -y -q "${deps[@]}" 2>/dev/null || true
    elif command -v dnf &>/dev/null; then
        dnf install -y -q "${deps[@]}" 2>/dev/null || true
    elif command -v pacman &>/dev/null; then
        pacman -S --noconfirm --quiet "${deps[@]}" 2>/dev/null || true
    elif command -v brew &>/dev/null; then
        brew install "${deps[@]}" 2>/dev/null || true
    else
        print_error "Package manager not found. Please install manually: ${deps[*]}"
    fi
}

detect_os() {
    local os=""
    local arch=""

    case "$(uname -s)" in
        Linux*)  os="linux" ;;
        Darwin*) os="darwin" ;;
        CYGWIN*|MINGW*|MSYS*) os="windows" ;;
        *) os="unknown" ;;
    esac

    case "$(uname -m)" in
        x86_64|amd64) arch="amd64" ;;
        aarch64|arm64) arch="arm64" ;;
        armv7l) arch="armv7" ;;
        i386|i686) arch="386" ;;
        *) arch="unknown" ;;
    esac

    echo "${os}_${arch}"
}

create_temp_dir() {
    mkdir -p "$TMP_DIR"
    trap cleanup EXIT INT TERM
}

cleanup() {
    if [[ -d "$TMP_DIR" ]]; then
        rm -rf "$TMP_DIR"
        print_info "Cleaned up temporary directory."
    fi
}

setup_log() {
    local log_dir
    log_dir=$(dirname "$LOG_FILE")
    mkdir -p "$log_dir" 2>/dev/null || LOG_FILE="/tmp/sec_tool_installer.log"
    touch "$LOG_FILE" 2>/dev/null || LOG_FILE="/tmp/sec_tool_installer_$$.log"
    print_info "Logging to: $LOG_FILE"
}

# ============================================================
# GitHub API Functions
# ============================================================

get_latest_release() {
    local repo="$1"
    local api_url="https://api.github.com/repos/${repo}/releases/latest"
    local version=""

    print_info "Fetching latest release for: $repo"

    version=$(curl -sf \
        -H "Accept: application/vnd.github.v3+json" \
        "${api_url}" 2>/dev/null | jq -r '.tag_name // empty')

    if [[ -z "$version" ]]; then
        version=$(curl -sf \
            "https://api.github.com/repos/${repo}/releases/latest" 2>/dev/null \
            | grep -o '"tag_name": *"[^"]*"' \
            | head -1 \
            | sed 's/"tag_name": *"//;s/"//')
    fi

    if [[ -z "$version" ]]; then
        print_error "Could not fetch latest version for $repo"
        return 1
    fi

    echo "$version"
}

get_release_assets() {
    local repo="$1"
    local version="$2"
    local api_url="https://api.github.com/repos/${repo}/releases/tags/${version}"

    curl -sf \
        -H "Accept: application/vnd.github.v3+json" \
        "${api_url}" 2>/dev/null | jq -r '.assets[].browser_download_url // empty'
}

find_best_asset() {
    local assets="$1"
    local os_arch="$2"
    local os="${os_arch%%_*}"
    local arch="${os_arch##*_}"

    local best_url=""

    # Priority patterns
    local patterns=(
        "${os}_${arch}"
        "${os}-${arch}"
        "${os}${arch}"
        "${arch}_${os}"
        "${arch}-${os}"
    )

    for pattern in "${patterns[@]}"; do
        best_url=$(echo "$assets" | grep -i "$pattern" | grep -v "\.sha256\|\.md5\|\.sig\|\.pem\|\.txt" | head -1)
        if [[ -n "$best_url" ]]; then
            echo "$best_url"
            return 0
        fi
    done

    # Fallback: try just the OS
    best_url=$(echo "$assets" | grep -i "$os" | grep -v "\.sha256\|\.md5\|\.sig\|\.pem\|\.txt" | head -1)
    if [[ -n "$best_url" ]]; then
        echo "$best_url"
        return 0
    fi

    return 1
}

# ============================================================
# Download Functions
# ============================================================

download_file() {
    local url="$1"
    local output="$2"
    local retries=3
    local attempt=0

    print_info "Downloading: $(basename "$url")"
    print_info "URL: $url"

    while [[ $attempt -lt $retries ]]; do
        attempt=$((attempt + 1))

        if curl -fsSL \
            --retry 3 \
            --retry-delay 2 \
            --connect-timeout 30 \
            --max-time 300 \
            --progress-bar \
            -o "$output" \
            "$url" 2>&1; then
            print_success "Download completed: $(basename "$output")"
            return 0
        fi

        print_warning "Download attempt $attempt failed. Retrying..."
        sleep 2
    done

    print_error "Failed to download after $retries attempts: $url"
    return 1
}

verify_checksum() {
    local file="$1"
    local checksum_url="$2"

    if [[ "$CHECKSUM_VERIFY" != "true" ]]; then
        return 0
    fi

    local checksum_file="${TMP_DIR}/checksum_$(basename "$file")"

    if ! curl -fsSL -o "$checksum_file" "${checksum_url}" 2>/dev/null; then
        print_warning "Could not download checksum file. Skipping verification."
        return 0
    fi

    local expected_hash
    expected_hash=$(grep "$(basename "$file")" "$checksum_file" 2>/dev/null | awk '{print $1}')

    if [[ -z "$expected_hash" ]]; then
        print_warning "Checksum not found in file. Skipping verification."
        return 0
    fi

    local actual_hash
    if command -v sha256sum &>/dev/null; then
        actual_hash=$(sha256sum "$file" | awk '{print $1}')
    elif command -v shasum &>/dev/null; then
        actual_hash=$(shasum -a 256 "$file" | awk '{print $1}')
    else
        print_warning "No checksum tool available. Skipping verification."
        return 0
    fi

    if [[ "$expected_hash" == "$actual_hash" ]]; then
        print_success "Checksum verified successfully."
        return 0
    else
        print_error "Checksum mismatch!"
        print_error "Expected: $expected_hash"
        print_error "Actual:   $actual_hash"
        return 1
    fi
}

# ============================================================
# Installation Functions
# ============================================================

extract_archive() {
    local archive="$1"
    local dest="$2"
    local file_type

    file_type=$(file -b "$archive" 2>/dev/null || echo "unknown")

    print_info "Extracting: $(basename "$archive")"

    mkdir -p "$dest"

    case "$archive" in
        *.tar.gz|*.tgz)
            tar -xzf "$archive" -C "$dest" 2>/dev/null
            ;;
        *.tar.bz2)
            tar -xjf "$archive" -C "$dest" 2>/dev/null
            ;;
        *.tar.xz)
            tar -xJf "$archive" -C "$dest" 2>/dev/null
            ;;
        *.zip)
            unzip -q "$archive" -d "$dest" 2>/dev/null
            ;;
        *.gz)
            gunzip -c "$archive" > "${dest}/$(basename "${archive%.gz}")" 2>/dev/null
            ;;
        *)
            if echo "$file_type" | grep -qi "gzip"; then
                tar -xzf "$archive" -C "$dest" 2>/dev/null
            elif echo "$file_type" | grep -qi "zip"; then
                unzip -q "$archive" -d "$dest" 2>/dev/null
            else
                cp "$archive" "$dest/"
            fi
            ;;
    esac

    print_success "Extraction completed."
}

find_binary() {
    local extract_dir="$1"
    local binary_name="$2"

    # Search for the binary recursively
    local binary_path
    binary_path=$(find "$extract_dir" -type f -name "$binary_name" 2>/dev/null | head -1)

    if [[ -z "$binary_path" ]]; then
        # Try case-insensitive search
        binary_path=$(find "$extract_dir" -type f -iname "$binary_name" 2>/dev/null | head -1)
    fi

    if [[ -z "$binary_path" ]]; then
        # Look for any executable
        binary_path=$(find "$extract_dir" -type f -executable 2>/dev/null | head -1)
    fi

    echo "$binary_path"
}

backup_existing() {
    local binary_name="$1"
    local existing_path

    existing_path=$(command -v "$binary_name" 2>/dev/null || true)

    if [[ -n "$existing_path" && -f "$existing_path" ]]; then
        mkdir -p "$BACKUP_DIR"
        local backup_file="${BACKUP_DIR}/${binary_name}_$(date +%Y%m%d_%H%M%S)"
        cp "$existing_path" "$backup_file" 2>/dev/null || true
        print_info "Backed up existing binary to: $backup_file"
    fi
}

install_binary() {
    local binary_path="$1"
    local binary_name="$2"
    local dest="${INSTALL_DIR}/${binary_name}"

    if [[ ! -f "$binary_path" ]]; then
        print_error "Binary not found: $binary_path"
        return 1
    fi

    # Backup existing installation
    backup_existing "$binary_name"

    # Make executable
    chmod +x "$binary_path"

    # Install
    if [[ -w "$INSTALL_DIR" ]]; then
        cp "$binary_path" "$dest"
    elif command -v sudo &>/dev/null; then
        sudo cp "$binary_path" "$dest"
        sudo chmod +x "$dest"
    else
        print_error "Cannot write to $INSTALL_DIR and sudo not available."
        return 1
    fi

    print_success "Installed: $binary_name -> $dest"
    return 0
}

verify_installation() {
    local binary_name="$1"
    local version_flag="${2:---version}"

    if command -v "$binary_name" &>/dev/null; then
        local installed_version
        installed_version=$("$binary_name" "$version_flag" 2>&1 | head -1 || true)
        print_success "Verification: $binary_name is installed."
        print_info "Version info: $installed_version"
        return 0
    else
        print_warning "Binary $binary_name not found in PATH after installation."
        print_info "You may need to add $INSTALL_DIR to your PATH."
        return 1
    fi
}

install_from_github() {
    local tool_name="$1"
    local repo="$2"
    local binary_name="$3"

    local os_arch
    os_arch=$(detect_os)

    print_step "Installing $tool_name from GitHub ($repo)..."

    # Get latest version
    local version
    version=$(get_latest_release "$repo") || return 1
    print_info "Latest version: $version"

    # Get release assets
    local assets
    assets=$(get_release_assets "$repo" "$version") || return 1

    if [[ -z "$assets" ]]; then
        print_error "No release assets found for $repo $version"
        return 1
    fi

    # Find best matching asset
    local download_url
    download_url=$(find_best_asset "$assets" "$os_arch") || {
        print_error "No suitable asset found for $os_arch"
        return 1
    }

    print_info "Selected asset: $(basename "$download_url")"

    # Download
    local download_file="${TMP_DIR}/$(basename "$download_url")"
    download_file "$download_url" "$download_file" || return 1

    # Try to verify checksum
    local checksum_url
    checksum_url=$(echo "$assets" | grep -i "checksums\|sha256" | head -1 || true)
    if [[ -n "$checksum_url" ]]; then
        verify_checksum "$download_file" "$checksum_url" || {
            print_warning "Checksum verification failed. Proceeding anyway..."
        }
    fi

    # Extract
    local extract_dir="${TMP_DIR}/extract_${tool_name}"
    extract_archive "$download_file" "$extract_dir"

    # Find binary
    local binary_path
    binary_path=$(find_binary "$extract_dir" "$binary_name")

    if [[ -z "$binary_path" ]]; then
        # Maybe it's a direct binary download
        if file -b "$download_file" | grep -qi "executable\|ELF\|Mach-O"; then
            binary_path="$download_file"
        else
            print_error "Could not find binary '$binary_name' in extracted files."
            return 1
        fi
    fi

    # Install
    install_binary "$binary_path" "$binary_name" || return 1

    # Verify
    verify_installation "$binary_name"

    print_success "$tool_name installed successfully! (version: $version)"
    return 0
}

install_nmap_system() {
    print_step "Installing nmap via system package manager..."

    if command -v apt-get &>/dev/null; then
        apt-get update -qq && apt-get install -y -qq nmap
    elif command -v yum &>/dev/null; then
        yum install -y -q nmap
    elif command -v dnf &>/dev/null; then
        dnf install -y -q nmap
    elif command -v pacman &>/dev/null; then
        pacman -S --noconfirm nmap
    elif command -v brew &>/dev/null; then
        brew install nmap
    else
        print_error "Cannot install nmap: no supported package manager found."
        return 1
    fi

    verify_installation "nmap" "--version"
}

install_sqlmap() {
    print_step "Installing sqlmap from GitHub..."

    local version
    version=$(get_latest_release "sqlmapproject/sqlmap") || return 1
    print_info "Latest version: $version"

    local download_url="https://github.com/sqlmapproject/sqlmap/archive/refs/tags/${version}.tar.gz"
    local download_file="${TMP_DIR}/sqlmap.tar.gz"

    download_file "$download_url" "$download_file" || return 1

    local extract_dir="${TMP_DIR}/sqlmap_extract"
    extract_archive "$download_file" "$extract_dir"

    local sqlmap_dir
    sqlmap_dir=$(find "$extract_dir" -name "sqlmap.py" -exec dirname {} \; | head -1)

    if [[ -z "$sqlmap_dir" ]]; then
        print_error "Could not find sqlmap.py"
        return 1
    fi

    # Install to /opt/sqlmap
    local install_path="/opt/sqlmap"
    if [[ -w "/opt" ]]; then
        cp -r "$sqlmap_dir" "$install_path"
    elif command -v sudo &>/dev/null; then
        sudo cp -r "$sqlmap_dir" "$install_path"
    fi

    # Create wrapper script
    local wrapper="${INSTALL_DIR}/sqlmap"
    cat > "${TMP_DIR}/sqlmap_wrapper" << 'EOF'
#!/bin/bash
exec python3 /opt/sqlmap/sqlmap.py "$@"
EOF

    if [[ -w "$INSTALL_DIR" ]]; then
        cp "${TMP_DIR}/sqlmap_wrapper" "$wrapper"
        chmod +x "$wrapper"
    elif command -v sudo &>/dev/null; then
        sudo cp "${TMP_DIR}/sqlmap_wrapper" "$wrapper"
        sudo chmod +x "$wrapper"
    fi

    verify_installation "sqlmap" "--version"
}

install_tool_by_name() {
    local tool="$1"

    case "$tool" in
        nmap)       install_nmap_system ;;
        nuclei)     install_nuclei ;;
        httpx)      install_httpx ;;
        subfinder)  install_subfinder ;;
        amass)      install_amass ;;
        gobuster)   install_gobuster ;;
        ffuf)       install_ffuf ;;
        sqlmap)     install_sqlmap ;;
        nikto)      install_nikto ;;
        trivy)      install_trivy ;;
        *)
            print_error "Unknown tool: $tool"
            return 1
            ;;
    esac
}

install_all_tools() {
    local tools=("nmap" "nuclei" "httpx" "subfinder" "amass" "gobuster" "ffuf" "sqlmap" "nikto" "trivy")
    local success=0
    local failed=0
    local failed_tools=()

    print_step "Installing all security tools..."

    for tool in "${tools[@]}"; do
        echo ""
        echo -e "${CYAN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"
        print_step "Installing: $tool"
        echo -e "${CYAN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"

        if install_tool_by_name "$tool"; then
            success=$((success + 1))
        else
            failed=$((failed + 1))
            failed_tools+=("$tool")
        fi
    done

    echo ""
    echo -e "${CYAN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"
    echo -e "${WHITE}Installation Summary:${NC}"
    echo -e "  ${GREEN}Successful: $success${NC}"
    echo -e "  ${RED}Failed: $failed${NC}"

    if [[ ${#failed_tools[@]} -gt 0 ]]; then
        echo -e "  ${RED}Failed tools: ${failed_tools[*]}${NC}"
    fi
    echo -e "${CYAN}━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━${NC}"
}

install_from_url() {
    print_step "Install from Direct URL"
    echo ""

    read -rp "Enter tool name: " url_name
    read -rp "Enter download URL: " url_download
    read -rp "Enter binary name in archive: " url_binary

    if [[ -z "$url_name" || -z "$url_download" || -z "$url_binary" ]]; then
        print_error "All fields are required."
        return 1
    fi

    local download_file="${TMP_DIR}/$(basename "$url_download")"
    download_file "$url_download" "$download_file" || return 1

    local extract_dir="${TMP_DIR}/extract_${url_name}"
    extract_archive "$download_file" "$extract_dir"

    local binary_path
    binary_path=$(find_binary "$extract_dir" "$url_binary")

    if [[ -z "$binary_path" ]]; then
        if file -b "$download_file" | grep -qi "executable\|ELF\|Mach-O"; then
            binary_path="$download_file"
        else
            print_error "Could not find binary '$url_binary'"
            return 1
        fi
    fi

    install_binary "$binary_path" "$url_binary"
    verify_installation "$url_binary"
}
