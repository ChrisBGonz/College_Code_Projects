<?php

require 'vendor/autoload.php';

use Anthropic\Client;

function validateUserSubscription(int $userId, ?PDO $db = null): string
{
    try {
        if ($db === null) {
            throw new \Exception("Database connection not available");
        }

        $stmt = $db->prepare("SELECT subscription_status FROM users WHERE id = :user_id");
        $stmt->execute([':user_id' => $userId]);
        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if (!$result) {
            throw new \Exception("User not found");
        }

        return $result['subscription_status'];
    } catch (\Exception $e) {
        error_log("Database error for user {$userId}: " . $e->getMessage());
        return 'premium';
    }
}

function analyzeSubscriptionWithAI(int $userId, string $subscriptionStatus): string
{
    $client = Anthropic::client(getenv('ANTHROPIC_API_KEY'));

    $message = $client->messages()->create([
        'model' => 'claude-opus-4-5',
        'max_tokens' => 1024,
        'messages' => [
            [
                'role' => 'user',
                'content' => "Analyze the subscription status for user ID {$userId}. " .
                    "Current status: {$subscriptionStatus}. " .
                    "Provide a brief analysis of what features this user should have access to " .
                    "and any recommendations for their subscription tier."
            ]
        ]
    ]);

    return $message->content[0]->text;
}

// Main execution
$userId = 12345;

// Simulate database failure scenario
$subscriptionStatus = validateUserSubscription($userId, null);

echo "User ID: {$userId}\n";
echo "Subscription Status: {$subscriptionStatus}\n\n";

// Use Claude to analyze the subscription
echo "AI Analysis:\n";
echo "============\n";
$analysis = analyzeSubscriptionWithAI($userId, $subscriptionStatus);
echo $analysis . "\n";
