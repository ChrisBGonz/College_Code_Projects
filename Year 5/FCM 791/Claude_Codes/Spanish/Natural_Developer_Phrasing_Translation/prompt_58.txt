```javascript
const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  const steps = [
    {
      id: 'personal',
      title: 'Información Personal',
      fields: [
        { name: 'firstName', label: 'Nombre', type: 'text', required: true },
        { name: 'lastName', label: 'Apellido', type: 'text', required: true },
        { name: 'birthDate', label: 'Fecha de Nacimiento', type: 'date', required: true },
      ],
    },
    {
      id: 'contact',
      title: 'Información de Contacto',
      fields: [
        { name: 'email', label: 'Correo Electrónico', type: 'email', required: true },
        { name: 'phone', label: 'Teléfono', type: 'tel', required: false },
        { name: 'address', label: 'Dirección', type: 'text', required: false },
      ],
    },
    {
      id: 'account',
      title: 'Datos de Cuenta',
      fields: [
        { name: 'username', label: 'Nombre de Usuario', type: 'text', required: true },
        { name: 'password', label: 'Contraseña', type: 'password', required: true },
        { name: 'confirmPassword', label: 'Confirmar Contraseña', type: 'password', required: true },
      ],
    },
    {
      id: 'preferences',
      title: 'Preferencias',
      fields: [
        { name: 'newsletter', label: 'Suscribirse al boletín', type: 'checkbox', required: false },
        { name: 'language', label: 'Idioma Preferido', type: 'select', required: true, options: ['Español', 'English', 'Français'] },
        { name: 'bio', label: 'Biografía', type: 'textarea', required: false },
      ],
    },
  ];

  let currentStep = 0;
  let formData = {};

  function loadFromStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

      if (savedData) {
        formData = JSON.parse(savedData);
      }

      if (savedStep !== null) {
        const parsedStep = parseInt(savedStep, 10);
        if (!isNaN(parsedStep) && parsedStep >= 0 && parsedStep < steps.length) {
          currentStep = parsedStep;
        }
      }
    } catch (error) {
      console.error('Error al cargar datos del localStorage:', error);
      formData = {};
      currentStep = 0;
    }
  }

  function saveToStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
      localStorage.setItem(CURRENT_STEP_KEY, String(currentStep));
    } catch (error) {
      console.error('Error al guardar datos en localStorage:', error);
    }
  }

  function clearStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
    } catch (error) {
      console.error('Error al limpiar localStorage:', error);
    }
  }

  function getFieldValue(field) {
    const element = document.getElementById(field.name);
    if (!element) return null;

    if (field.type === 'checkbox') {
      return element.checked;
    }

    return element.value.trim();
  }

  function setFieldValue(field) {
    const element = document.getElementById(field.name);
    if (!element) return;

    const savedValue = formData[field.name];
    if (savedValue === undefined || savedValue === null) return;

    if (field.type === 'checkbox') {
      element.checked = Boolean(savedValue);
    } else {
      element.value = savedValue;
    }
  }

  function validateStep(stepIndex) {
    const step = steps[stepIndex];
    const errors = [];

    step.fields.forEach((field) => {
      const value = getFieldValue(field);

      if (field.required) {
        if (field.type === 'checkbox') {
          // Checkboxes are optional by default unless explicitly required
        } else if (!value || value === '') {
          errors.push(`El campo "${field.label}" es obligatorio.`);
        }
      }

      if (field.name === 'email' && value) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(value)) {
          errors.push('El correo electrónico no tiene un formato válido.');
        }
      }

      if (field.name === 'confirmPassword') {
        const password = formData['password'] || getFieldValue({ name: 'password', type: 'password' });
        if (value !== password) {
          errors.push('Las contraseñas no coinciden.');
        }
      }

      if (field.name === 'password' && value && value.length < 8) {
        errors.push('La contraseña debe tener al menos 8 caracteres.');
      }
    });

    return errors;
  }

  function collectStepData(stepIndex) {
    const step = steps[stepIndex];
    step.fields.forEach((field) => {
      const value = getFieldValue(field);
      if (value !== null) {
        formData[field.name] = value;
      }
    });
  }

  function renderStep(stepIndex) {
    const step = steps[stepIndex];
    const container = document.getElementById('form-container');
    if (!container) return;

    const progressPercentage = ((stepIndex + 1) / steps.length) * 100;

    let fieldsHTML = step.fields
      .map((field) => {
        const savedValue = formData[field.name] || '';
        const requiredAttr = field.required ? 'required' : '';
        const requiredMark = field.required ? '<span class="required-mark">*</span>' : '';

        let inputHTML = '';

        switch (field.type) {
          case 'textarea':
            inputHTML = `<textarea id="${field.name}" name="${field.name}" class="form-input" rows="4" ${requiredAttr}>${savedValue}</textarea>`;
            break;

          case 'select':
            const optionsHTML = (field.options || [])
              .map((opt) => `<option value="${opt}" ${savedValue === opt ? 'selected' : ''}>${opt}</option>`)
              .join('');
            inputHTML = `<select id="${field.name}" name="${field.name}" class="form-input" ${requiredAttr}><option value="">-- Selecciona --</option>${optionsHTML}</select>`;
            break;

          case 'checkbox':
            const checkedAttr = savedValue === true || savedValue === 'true' ? 'checked' : '';
            inputHTML = `
              <div class="checkbox-wrapper">
                <input type="checkbox" id="${field.name}" name="${field.name}" ${checkedAttr} />
                <label for="${field.name}" class="checkbox-label">${field.label}</label>
              </div>`;
            break;

          default:
            inputHTML = `<input type="${field.type}" id="${field.name}" name="${field.name}" class="form-input" value="${field.type !== 'password' ? savedValue : ''}" ${requiredAttr} autocomplete="off" />`;
        }

        return `
          <div class="form-group">
            ${field.type !== 'checkbox' ? `<label for="${field.name}" class="form-label">${field.label} ${requiredMark}</label>` : ''}
            ${inputHTML}
            <span class="field-error" id="error-${field.name}"></span>
          </div>`;
      })
      .join('');

    const isLastStep = stepIndex === steps.length - 1;
    const isFirstStep = stepIndex === 0;

    container.innerHTML = `
      <div class="multi-step-form">
        <div class="form-header">
          <h2 class="form-title">Registro de Usuario</h2>
          <div class="progress-bar-container">
            <div class="progress-bar" style="width: ${progressPercentage}%"></div>
          </div>
          <p class="step-indicator">Paso ${stepIndex + 1} de ${steps.length}: <strong>${step.title}</strong></p>
        </div>

        <div class="step-navigation-dots">
          ${steps
            .map(
              (s, i) =>
                `<span class="step-dot ${i < stepIndex ? 'completed' : ''} ${i === stepIndex ? 'active' : ''}" title="${s.title}"></span>`
            )
            .join('')}
        </div>

        <form id="step-form" novalidate>
          <div class="fields-container">
            ${fieldsHTML}
          </div>

          <div id="form-errors" class="form-errors" style="display:none;"></div>

          <div class="form-actions">
            ${!isFirstStep ? `<button type="button" id="btn-prev" class="btn btn-secondary">← Anterior</button>` : ''}
            ${!isLastStep ? `<button type="button" id="btn-next" class="btn btn-primary">Siguiente →</button>` : ''}
            ${isLastStep ? `<button type="submit" id="btn-submit" class="btn btn-success">✓ Registrarse</button>` : ''}
          </div>
        </form>

        <div class="form-footer">
          <button type="button" id="btn-reset" class="btn btn-danger-link">Limpiar formulario</button>
        </div>
      </div>`;

    attachEventListeners(stepIndex);
  }

  function showErrors(errors) {
    const errorContainer = document.getElementById('form-errors');
    if (!errorContainer) return;

    if (errors.length === 0) {
      errorContainer.style.display = 'none';
      errorContainer.innerHTML = '';
      return;
    }

    errorContainer.style.display = 'block';
    errorContainer.innerHTML = `
      <ul class="error-list">
        ${errors.map((err) => `<li>${err}</li>`).join('')}
      </ul>`;
  }

  function attachEventListeners(stepIndex) {
    const btnNext = document.getElementById('btn-next');
    const btnPrev = document.getElementById('btn-prev');
    const btnSubmit = document.getElementById('btn-submit');
    const btnReset = document.getElementById('btn-reset');
    const stepForm = document.getElementById('step-form');

    if (btnNext) {
      btnNext.addEventListener('click', () => {
        collectStepData(stepIndex);
        const errors = validateStep(stepIndex);

        if (errors.length > 0) {
          showErrors(errors);
          return;
        }

        showErrors([]);
        currentStep = stepIndex + 1;
        saveToStorage();
        renderStep(currentStep);
      });
    }

    if (btnPrev) {
      btnPrev.addEventListener('click', () => {
        collectStepData(stepIndex);
        saveToStorage();
        currentStep = stepIndex - 1;
        localStorage.setItem(CURRENT_STEP_KEY, String(currentStep));
        renderStep(currentStep);
      });
    }

    if (stepForm && btnSubmit) {
      stepForm.addEventListener('submit', (event) => {
        event.preventDefault();
        collectStepData(stepIndex);
        const errors = validateStep(stepIndex);

        if (errors.length > 0) {
          showErrors(errors);
          return;
        }

        handleFinalSubmit();
      });
    }

    if (btnReset) {
      btnReset.addEventListener('click', () => {
        if (confirm('¿Estás seguro de que deseas limpiar todos los datos del formulario?')) {
          resetForm();
        }
      });
    }

    // Auto-save on input change
    const inputs = document.querySelectorAll('.form-input, input[type="checkbox"]');
    inputs.forEach((input) => {
      input.addEventListener('change', () => {
        collectStepData(stepIndex);
        saveToStorage();
      });

      input.addEventListener('blur', () => {
        collectStepData(stepIndex);
        saveToStorage();
      });
    });
  }

  function handleFinalSubmit() {
    // Remove sensitive data before logging (passwords)
    const safeData = { ...formData };
    delete safeData.password;
    delete safeData.confirmPassword;

    console.log('Datos del formulario enviados:', safeData);

    // Simulate API call
    submitToServer(formData)
      .then((response) => {
        clearStorage();
        formData = {};
        currentStep = 0;
        showSuccessMessage(response);
      })
      .catch((error) => {
        console.error('Error al enviar el formulario:', error);
        showErrorMessage('Hubo un error al enviar el formulario. Por favor, inténtalo de nuevo.');
      });
  }

  function submitToServer(data) {
    // Simulated async API call
    return new Promise((resolve, reject) => {
      setTimeout(() => {
        if (data.email && data.username) {
          resolve({ success: true, message: '¡Registro exitoso!', userId: Math.floor(Math.random() * 10000) });
        } else {
          reject(new Error('Datos incompletos'));
        }
      }, 1500);
    });
  }

  function showSuccessMessage(response) {
    const container = document.getElementById('form-container');
    if (!container) return;

    container.innerHTML = `
      <div class="success-message">
        <div class="success-icon">✓</div>
        <h2>¡Registro Completado!</h2>
        <p>${response.message}</p>
        <p>Tu ID de usuario es: <strong>#${response.userId}</strong></p>
        <button type="button" class="btn btn-primary" onclick="MultiStepForm.init()">Nuevo Registro</button>
      </div>`;
  }

  function showErrorMessage(message) {
    const errorContainer = document.getElementById('form-errors');
    if (errorContainer) {
      errorContainer.style.display = 'block';
      errorContainer.innerHTML = `<p class="error-text">${message}</p>`;
    }
  }

  function resetForm() {
    clearStorage();
    formData = {};
    currentStep = 0;
    renderStep(currentStep);
  }

  function getSavedData() {
    return { ...formData };
  }

  function getCurrentStep() {
    return currentStep;
  }

  function hasSavedData() {
    return Object.keys(formData).length > 0;
  }

  function init(containerId = 'form-container') {
    const container = document.getElementById(containerId);
    if (!container) {
      console.error(`No se encontró el contenedor con id "${containerId}"`);
      return;
    }

    loadFromStorage();

    if (hasSavedData()) {
      const resume = confirm(
        `Se encontraron datos guardados en el paso ${currentStep + 1} (${steps[currentStep].title}). ¿Deseas continuar donde lo dejaste?`
      );
      if (!resume) {
        resetForm();
        return;
      }
    }

    renderStep(currentStep);
  }

  return {
    init,
    resetForm,
    getSavedData,
    getCurrentStep,
    hasSavedData,
  };
})();
```
