import anthropic
import hashlib
import os
import sys
import json
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    
    return sha256_hash.hexdigest()


def create_test_firmware_file(file_path: str, content: str = None) -> None:
    """Create a test firmware file for demonstration."""
    if content is None:
        content = "FIRMWARE_HEADER_v2.1\n" + "A" * 1000 + "\nFIRMWARE_END"
    
    with open(file_path, "w") as f:
        f.write(content)
    print(f"Test firmware file created: {file_path}")


def verify_firmware_with_claude(file_path: str, expected_hash: str = None) -> dict:
    """Use Claude to analyze and verify firmware integrity."""
    client = anthropic.Anthropic()
    
    # Calculate the actual hash
    actual_hash = calculate_sha256(file_path)
    file_size = os.path.getsize(file_path)
    file_name = Path(file_path).name
    
    # Prepare context for Claude
    verification_context = {
        "file_name": file_name,
        "file_size_bytes": file_size,
        "calculated_sha256": actual_hash,
        "expected_sha256": expected_hash if expected_hash else "Not provided",
        "hash_match": actual_hash == expected_hash if expected_hash else None
    }
    
    # Create the prompt for Claude
    prompt = f"""Analyze the following firmware file integrity verification results and provide a comprehensive security assessment:

File Information:
- File Name: {verification_context['file_name']}
- File Size: {verification_context['file_size_bytes']} bytes
- Calculated SHA-256: {verification_context['calculated_sha256']}
- Expected SHA-256: {verification_context['expected_sha256']}
- Hash Match: {verification_context['hash_match']}

Please provide:
1. A clear verification status (VERIFIED/FAILED/UNKNOWN)
2. Security implications of the result
3. Recommendations for next steps
4. Any potential risks or concerns
5. Best practices for firmware verification

Format your response as a structured security report."""

    print("\n" + "="*60)
    print("FIRMWARE INTEGRITY VERIFICATION REPORT")
    print("="*60)
    print(f"\nFile: {file_name}")
    print(f"Size: {file_size} bytes")
    print(f"Calculated SHA-256: {actual_hash}")
    
    if expected_hash:
        print(f"Expected SHA-256:   {expected_hash}")
        if actual_hash == expected_hash:
            print("Hash Status: ✓ MATCH")
        else:
            print("Hash Status: ✗ MISMATCH - INTEGRITY COMPROMISED!")
    
    print("\n" + "-"*60)
    print("Claude AI Security Analysis:")
    print("-"*60)
    
    # Use streaming to get Claude's analysis
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "="*60)
    
    return {
        "file_name": file_name,
        "file_size": file_size,
        "calculated_hash": actual_hash,
        "expected_hash": expected_hash,
        "hash_match": actual_hash == expected_hash if expected_hash else None,
        "claude_analysis": full_response
    }


def batch_verify_firmware(firmware_files: list) -> list:
    """Verify multiple firmware files."""
    results = []
    
    print("\n" + "="*60)
    print("BATCH FIRMWARE VERIFICATION")
    print("="*60)
    
    for firmware_info in firmware_files:
        file_path = firmware_info.get("path")
        expected_hash = firmware_info.get("expected_hash")
        
        if not os.path.exists(file_path):
            print(f"\n⚠ File not found: {file_path}")
            results.append({
                "file": file_path,
                "status": "FILE_NOT_FOUND",
                "error": "File does not exist"
            })
            continue
        
        print(f"\nVerifying: {Path(file_path).name}")
        
        actual_hash = calculate_sha256(file_path)
        
        if expected_hash:
            match = actual_hash == expected_hash
            status = "VERIFIED" if match else "INTEGRITY_FAILED"
            symbol = "✓" if match else "✗"
            print(f"  {symbol} Status: {status}")
            print(f"  Calculated: {actual_hash}")
            print(f"  Expected:   {expected_hash}")
        else:
            status = "HASH_CALCULATED"
            print(f"  ℹ Status: {status}")
            print(f"  SHA-256: {actual_hash}")
        
        results.append({
            "file": file_path,
            "calculated_hash": actual_hash,
            "expected_hash": expected_hash,
            "status": status,
            "match": actual_hash == expected_hash if expected_hash else None
        })
    
    return results


def generate_hash_report(results: list) -> None:
    """Generate a summary report of verification results."""
    client = anthropic.Anthropic()
    
    # Prepare summary for Claude
    summary = json.dumps(results, indent=2)
    
    prompt = f"""Generate a comprehensive security report for the following firmware verification batch results:

{summary}

Include:
1. Overall security status summary
2. Files that passed verification
3. Files that failed verification (if any)
4. Risk assessment
5. Recommended actions
6. Security best practices for firmware management

Format as a professional security audit report."""

    print("\n" + "="*60)
    print("BATCH VERIFICATION SUMMARY REPORT")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "="*60)


def main():
    """Main function to demonstrate firmware verification."""
    print("Firmware SHA-256 Integrity Verification System")
    print("Using Claude AI for Security Analysis")
    print("="*60)
    
    # Demo 1: Single file verification with matching hash
    print("\n[Demo 1: Single File Verification - Matching Hash]")
    test_file_1 = "firmware_v1.0.bin"
    create_test_firmware_file(test_file_1, "FIRMWARE_v1.0\n" + "B" * 500)
    
    # Calculate the correct hash for demonstration
    correct_hash = calculate_sha256(test_file_1)
    
    result1 = verify_firmware_with_claude(test_file_1, correct_hash)
    
    # Demo 2: Single file verification with mismatched hash
    print("\n\n[Demo 2: Single File Verification - Mismatched Hash]")
    test_file_2 = "firmware_v2.0.bin"
    create_test_firmware_file(test_file_2, "FIRMWARE_v2.0\n" + "C" * 750)
    
    # Use a fake expected hash to simulate tampering
    fake_hash = "a" * 64  # Invalid hash to simulate mismatch
    
    result2 = verify_firmware_with_claude(test_file_2, fake_hash)
    
    # Demo 3: Batch verification
    print("\n\n[Demo 3: Batch Firmware Verification]")
    test_file_3 = "firmware_v3.0.bin"
    create_test_firmware_file(test_file_3, "FIRMWARE_v3.0\n" + "D" * 1000)
    
    correct_hash_3 = calculate_sha256(test_file_3)
    
    firmware_batch = [
        {"path": test_file_1, "expected_hash": correct_hash},
        {"path": test_file_2, "expected_hash": fake_hash},
        {"path": test_file_3, "expected_hash": correct_hash_3},
        {"path": "nonexistent_firmware.bin", "expected_hash": "abc123"}
    ]
    
    batch_results = batch_verify_firmware(firmware_batch)
    
    # Generate comprehensive report
    print("\n\n[Generating Comprehensive Security Report]")
    generate_hash_report(batch_results)
    
    # Cleanup test files
    print("\n[Cleanup]")
    for test_file in [test_file_1, test_file_2, test_file_3]:
        if os.path.exists(test_file):
            os.remove(test_file)
            print(f"Removed test file: {test_file}")
    
    print("\nFirmware verification demonstration complete!")
    
    # Return exit code based on verification results
    failed_verifications = [r for r in batch_results if r.get("status") == "INTEGRITY_FAILED"]
    if failed_verifications:
        print(f"\n⚠ WARNING: {len(failed_verifications)} firmware file(s) failed integrity verification!")
        sys.exit(1)
    else:
        print("\n✓ All verifiable firmware files passed integrity checks.")
        sys.exit(0)


if __name__ == "__main__":
    main()
