import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";

const client = new Anthropic();

async function generateExpressConfig(): Promise<void> {
  console.log(
    "Generating Express.js production server configuration with Claude...\n"
  );

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `Crea una configuración básica de servidor Express.js para un entorno de producción, incluyendo:
        
1. Configuración básica del servidor Express
2. Middleware de seguridad (helmet, cors, rate limiting)
3. Middleware de logging (morgan)
4. Middleware de parsing (body-parser, json)
5. Manejo de errores centralizado con diferentes tipos de errores
6. Manejo de rutas no encontradas (404)
7. Graceful shutdown
8. Variables de entorno con dotenv
9. Compresión de respuestas
10. Sanitización de datos

Por favor proporciona el código completo y funcional con comentarios explicativos en español.`,
      },
    ],
  });

  // Extract the text content from the response
  const responseText = message.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("=== Express.js Production Server Configuration ===\n");
  console.log(responseText);

  // Save the generated configuration to a file
  const outputFile = "express-server-config.js";

  // Extract code blocks from the response
  const codeBlockRegex = /```(?:javascript|js)?\n([\s\S]*?)```/g;
  const codeBlocks: string[] = [];
  let match;

  while ((match = codeBlockRegex.exec(responseText)) !== null) {
    codeBlocks.push(match[1]);
  }

  if (codeBlocks.length > 0) {
    const fullCode = codeBlocks.join("\n\n// ---\n\n");
    fs.writeFileSync(outputFile, fullCode);
    console.log(`\n✅ Configuration saved to: ${outputFile}`);
  } else {
    // If no code blocks found, save the entire response
    fs.writeFileSync(outputFile, responseText);
    console.log(`\n✅ Response saved to: ${outputFile}`);
  }

  // Display usage statistics
  console.log("\n=== API Usage Statistics ===");
  console.log(`Input tokens: ${message.usage.input_tokens}`);
  console.log(`Output tokens: ${message.usage.output_tokens}`);
  console.log(
    `Total tokens: ${message.usage.input_tokens + message.usage.output_tokens}`
  );
}

// Run the main function
generateExpressConfig().catch((error) => {
  console.error("Error generating Express configuration:", error);
  process.exit(1);
});
