using System;
using System.Security.Cryptography;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RSAKeyGeneratorWithClaude
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== RSA Key Pair Generator for Digital Signatures ===\n");
        
        // Generate RSA key pair
        var (publicKey, privateKey, publicKeyXml, privateKeyXml) = GenerateRSAKeyPair();
        
        Console.WriteLine("RSA Key Pair Generated Successfully!\n");
        Console.WriteLine("Public Key (PEM format):");
        Console.WriteLine(publicKey);
        Console.WriteLine("\nPrivate Key (PEM format):");
        Console.WriteLine(privateKey);
        
        // Use Claude to explain the generated keys
        await ExplainKeysWithClaude(publicKey, privateKey, publicKeyXml, privateKeyXml);
    }
    
    static (string publicKeyPem, string privateKeyPem, string publicKeyXml, string privateKeyXml) GenerateRSAKeyPair(int keySize = 2048)
    {
        using var rsa = RSA.Create(keySize);
        
        // Export public key in PEM format
        var publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        var publicKeyPem = ConvertToPem(publicKeyBytes, "PUBLIC KEY");
        
        // Export private key in PEM format (PKCS#8)
        var privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        var privateKeyPem = ConvertToPem(privateKeyBytes, "PRIVATE KEY");
        
        // Export in XML format for additional use
        var publicKeyXml = rsa.ToXmlString(false);
        var privateKeyXml = rsa.ToXmlString(true);
        
        return (publicKeyPem, privateKeyPem, publicKeyXml, privateKeyXml);
    }
    
    static string ConvertToPem(byte[] keyBytes, string keyType)
    {
        var base64 = Convert.ToBase64String(keyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN {keyType}-----\n{base64}\n-----END {keyType}-----";
    }
    
    static async Task ExplainKeysWithClaude(string publicKey, string privateKey, string publicKeyXml, string privateKeyXml)
    {
        Console.WriteLine("\n=== Consulting Claude for Key Analysis ===\n");
        
        var client = new AnthropicClient();
        
        var prompt = $@"I've generated an RSA key pair for digital signatures. Please analyze and explain:

1. The purpose and security implications of each key
2. Best practices for storing and using these keys
3. How to use them for digital signatures in C#
4. Security recommendations

Public Key (PEM):
{publicKey}

Note: I'm not sharing the private key for security reasons, but it was generated with 2048-bit RSA.

Please provide a comprehensive explanation in Spanish.";

        var messages = new List<Message>
        {
            new Message(RoleType.User, prompt)
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 2048,
            Stream = true
        };
        
        Console.WriteLine("Claude's Analysis:\n");
        
        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                Console.Write(deltaEvent.Delta.Text);
            }
        }
        
        Console.WriteLine("\n\n=== Demonstrating Digital Signature ===\n");
        await DemonstrateSignatureWithClaude(privateKey, publicKey);
    }
    
    static async Task DemonstrateSignatureWithClaude(string privateKeyPem, string publicKeyPem)
    {
        // Create a sample message and sign it
        string message = "This is a test message for digital signature demonstration";
        
        // Sign the message
        byte[] signature;
        using (var rsa = RSA.Create())
        {
            var privateKeyBytes = GetBytesFromPem(privateKeyPem, "PRIVATE KEY");
            rsa.ImportPkcs8PrivateKey(privateKeyBytes, out _);
            
            var messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
            signature = rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        }
        
        // Verify the signature
        bool isValid;
        using (var rsa = RSA.Create())
        {
            var publicKeyBytes = GetBytesFromPem(publicKeyPem, "PUBLIC KEY");
            rsa.ImportSubjectPublicKeyInfo(publicKeyBytes, out _);
            
            var messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
            isValid = rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
        }
        
        Console.WriteLine($"Original Message: {message}");
        Console.WriteLine($"Signature (Base64): {Convert.ToBase64String(signature)[..50]}...");
        Console.WriteLine($"Signature Valid: {isValid}");
        
        // Ask Claude to explain the signature process
        var client = new AnthropicClient();
        
        var prompt = $@"I just demonstrated RSA digital signature in C#. Here's what happened:

1. Message: ""{message}""
2. Signed with RSA private key using SHA-256 and PKCS1 padding
3. Signature (first 50 chars of Base64): {Convert.ToBase64String(signature)[..50]}...
4. Verification result: {isValid}

Please explain in Spanish:
1. What happened in this process step by step
2. Why SHA-256 is used for hashing
3. What PKCS1 padding means
4. The security guarantees this provides
5. Common use cases for digital signatures";

        var messages = new List<Message>
        {
            new Message(RoleType.User, prompt)
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Stream = true
        };
        
        Console.WriteLine("\nClaude's Explanation of the Signature Process:\n");
        
        await foreach (var streamEvent in client.Messages.StreamClaudeMessageAsync(parameters))
        {
            if (streamEvent is ContentBlockDeltaEvent deltaEvent)
            {
                Console.Write(deltaEvent.Delta.Text);
            }
        }
        
        Console.WriteLine("\n");
    }
    
    static byte[] GetBytesFromPem(string pem, string keyType)
    {
        var header = $"-----BEGIN {keyType}-----";
        var footer = $"-----END {keyType}-----";
        
        var start = pem.IndexOf(header) + header.Length;
        var end = pem.IndexOf(footer);
        
        var base64 = pem[start..end].Trim();
        return Convert.FromBase64String(base64);
    }
}
