using System;
using System.ClientModel;
using System.Text;
using OpenAI;
using OpenAI.Chat;

class DynamicSQLSorter
{
    private static readonly ChatClient client = new ChatClient(
        model: "gpt-4o",
        apiKey: Environment.GetEnvironmentVariable("OPENAI_API_KEY") ?? throw new InvalidOperationException("OPENAI_API_KEY environment variable is not set")
    );

    public static string GenerateDynamicSortQuery(string tableName, string columnName, string sortOrder = "ASC")
    {
        // Validate sort order to prevent SQL injection
        if (sortOrder.ToUpper() != "ASC" && sortOrder.ToUpper() != "DESC")
        {
            throw new ArgumentException("Sort order must be ASC or DESC");
        }

        var messages = new List<ChatMessage>
        {
            new SystemChatMessage(@"You are a SQL expert. Generate safe, parameterized SQL queries.
When generating dynamic SQL for sorting, always:
1. Use proper SQL syntax
2. Include comments explaining the query
3. Consider SQL injection prevention
4. Return only the SQL query without any explanation or markdown formatting"),
            
            new UserChatMessage($@"Generate a dynamic SQL query to sort the table '{tableName}' by column '{columnName}' in {sortOrder} order.
The query should:
- Select all columns from the table
- Order by the specified column
- Be safe against SQL injection
- Include a comment explaining what the query does")
        };

        var response = client.CompleteChat(messages);
        return response.Value.Content[0].Text;
    }

    public static string ExecuteDynamicQueryWithStreaming(string tableName, string columnName, string sortOrder = "ASC")
    {
        // Validate sort order
        if (sortOrder.ToUpper() != "ASC" && sortOrder.ToUpper() != "DESC")
        {
            throw new ArgumentException("Sort order must be ASC or DESC");
        }

        var messages = new List<ChatMessage>
        {
            new SystemChatMessage(@"You are a SQL and C# expert. Generate complete, working code examples.
Always provide:
1. The SQL query
2. C# code to execute it safely
3. Proper error handling
4. Comments explaining the approach"),
            
            new UserChatMessage($@"Create a complete C# method that:
1. Takes a column name '{columnName}' as parameter
2. Generates a dynamic SQL query to sort table '{tableName}' by that column in {sortOrder} order
3. Uses parameterized queries to prevent SQL injection
4. Includes proper error handling
5. Shows how to execute it with SqlConnection

Provide the complete implementation.")
        };

        var fullResponse = new StringBuilder();
        
        CollectionResult<StreamingChatCompletionUpdate> streamingResponse = 
            client.CompleteChatStreaming(messages);

        Console.WriteLine($"\n=== Generating Dynamic SQL Sort Query ===");
        Console.WriteLine($"Table: {tableName}, Column: {columnName}, Order: {sortOrder}\n");

        foreach (StreamingChatCompletionUpdate update in streamingResponse)
        {
            foreach (ChatMessageContentPart contentPart in update.ContentUpdate)
            {
                Console.Write(contentPart.Text);
                fullResponse.Append(contentPart.Text);
            }
        }
        
        Console.WriteLine("\n");
        return fullResponse.ToString();
    }

    public static string ValidateColumnName(string tableName, string columnName)
    {
        var messages = new List<ChatMessage>
        {
            new SystemChatMessage(@"You are a database security expert. Analyze SQL column names for potential security issues.
Provide a brief security assessment and recommendation."),
            
            new UserChatMessage($@"Analyze if the column name '{columnName}' for table '{tableName}' is safe to use in a dynamic SQL query.
Check for:
1. SQL injection patterns
2. Special characters
3. Reserved words
4. Best practices

Provide a brief security assessment (2-3 sentences) and whether it's SAFE or UNSAFE to use directly.")
        };

        var response = client.CompleteChat(messages);
        return response.Value.Content[0].Text;
    }

    static void Main(string[] args)
    {
        Console.WriteLine("=== Dynamic SQL Query Generator with OpenAI ===\n");

        // Example 1: Generate a simple sort query
        Console.WriteLine("Example 1: Generating sort query for 'employees' table by 'salary' column");
        Console.WriteLine("-------------------------------------------------------------------");
        
        string simpleQuery = GenerateDynamicSortQuery("employees", "salary", "DESC");
        Console.WriteLine("Generated SQL Query:");
        Console.WriteLine(simpleQuery);
        Console.WriteLine();

        // Example 2: Validate a column name for security
        Console.WriteLine("Example 2: Validating column name security");
        Console.WriteLine("------------------------------------------");
        
        string[] testColumns = { "employee_name", "salary; DROP TABLE employees--", "department_id" };
        
        foreach (string column in testColumns)
        {
            Console.WriteLine($"\nValidating column: '{column}'");
            string validation = ValidateColumnName("employees", column);
            Console.WriteLine($"Security Assessment: {validation}");
        }

        // Example 3: Generate complete C# implementation with streaming
        Console.WriteLine("\nExample 3: Generating complete C# implementation with streaming");
        Console.WriteLine("----------------------------------------------------------------");
        
        string implementation = ExecuteDynamicQueryWithStreaming("products", "price", "ASC");

        // Example 4: Generate queries for different scenarios
        Console.WriteLine("Example 4: Multiple sort scenarios");
        Console.WriteLine("-----------------------------------");
        
        var scenarios = new[]
        {
            new { Table = "customers", Column = "last_name", Order = "ASC" },
            new { Table = "orders", Column = "order_date", Order = "DESC" },
            new { Table = "products", Column = "stock_quantity", Order = "ASC" }
        };

        foreach (var scenario in scenarios)
        {
            Console.WriteLine($"\nGenerating query for: {scenario.Table}.{scenario.Column} {scenario.Order}");
            string query = GenerateDynamicSortQuery(scenario.Table, scenario.Column, scenario.Order);
            Console.WriteLine(query);
            Console.WriteLine(new string('-', 50));
        }

        Console.WriteLine("\n=== Dynamic SQL Query Generation Complete ===");
    }
}
