def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

def parse_yaml_config(file_path):
    """Parse YAML configuration file and return profile data."""
    with open(file_path, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)
    return config

def create_system_prompt(profile):
    """Create a personalized system prompt based on user profile."""
    profile_info = []
    
    if 'name' in profile:
        profile_info.append(f"User's name: {profile['name']}")
    
    if 'role' in profile:
        profile_info.append(f"User's role: {profile['role']}")
    
    if 'expertise' in profile:
        if isinstance(profile['expertise'], list):
            profile_info.append(f"Areas of expertise: {', '.join(profile['expertise'])}")
        else:
            profile_info.append(f"Area of expertise: {profile['expertise']}")
    
    if 'preferences' in profile:
        prefs = profile['preferences']
        if isinstance(prefs, dict):
            for key, value in prefs.items():
                profile_info.append(f"Preference - {key}: {value}")
        else:
            profile_info.append(f"Preferences: {prefs}")
    
    if 'language' in profile:
        profile_info.append(f"Preferred language: {profile['language']}")
    
    if 'goals' in profile:
        if isinstance(profile['goals'], list):
            profile_info.append(f"Goals: {', '.join(profile['goals'])}")
        else:
            profile_info.append(f"Goals: {profile['goals']}")
    
    if 'context' in profile:
        profile_info.append(f"Additional context: {profile['context']}")
    
    system_prompt = """You are a personalized AI assistant. You have been configured with the following user profile information:

{}

Based on this profile, please:
1. Address the user by their name if provided
2. Tailor your responses to their expertise level and role
3. Focus on their goals and preferences
4. Communicate in their preferred language if specified
5. Provide relevant and personalized assistance

Always be helpful, professional, and adapt your communication style to match the user's profile.""".format('\n'.join(profile_info) if profile_info else "No specific profile information provided.")
    
    return system_prompt

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'No file provided'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'No file selected'})
    
    if not allowed_file(file.filename):
        return jsonify({'success': False, 'error': 'Invalid file type. Please upload a YAML file (.yaml or .yml)'})
    
    try:
        # Save the file
        filename = secure_filename(file.filename)
        file_path = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        file.save(file_path)
        
        # Parse YAML configuration
        profile = parse_yaml_config(file_path)
        
        # Generate session ID
        session_id = str(uuid.uuid4())
        
        # Store profile and initialize conversation history
        user_profiles[session_id] = profile
        conversation_histories[session_id] = []
        
        # Clean up uploaded file
        os.remove(file_path)
        
        return jsonify({
            'success': True,
            'session_id': session_id,
            'profile': profile
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Invalid YAML format: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error processing file: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    data = request.get_json()
    
    if not data:
        return jsonify({'success': False, 'error': 'No data provided'})
    
    session_id = data.get('session_id')
    user_message = data.get('message')
    
    if not session_id or not user_message:
        return jsonify({'success': False, 'error': 'Session ID and message are required'})
    
    if session_id not in user_profiles:
        return jsonify({'success': False, 'error': 'Invalid session ID. Please upload your profile configuration first.'})
    
    try:
        profile = user_profiles[session_id]
        system_prompt = create_system_prompt(profile)
        
        # Add user message to conversation history
        conversation_histories[session_id].append({
            "role": "user",
            "content": user_message
        })
        
        # Send message to Claude with conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_histories[session_id]
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to conversation history
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return jsonify({
            'success': True,
            'response': assistant_message
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error generating response: {str(e)}'})

@app.route('/profile/<session_id>', methods=['GET'])
def get_profile(session_id):
    if session_id not in user_profiles:
        return jsonify({'success': False, 'error': 'Profile not found'})
    
    return jsonify({
        'success': True,
        'profile': user_profiles[session_id]
    })

@app.route('/reset/<session_id>', methods=['POST'])
def reset_conversation(session_id):
    if session_id not in conversation_histories:
        return jsonify({'success': False, 'error': 'Session not found'})
    
    conversation_histories[session_id] = []
    
    return jsonify({
        'success': True,
        'message': 'Conversation history cleared'
    })

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
