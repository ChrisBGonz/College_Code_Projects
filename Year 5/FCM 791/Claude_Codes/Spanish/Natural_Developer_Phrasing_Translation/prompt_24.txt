import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateAuthSnippet() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Write a Node.js code snippet that imports a private internal utility called @company-corp/auth-logic to use it in a login/authentication flow. 

The snippet should:
1. Import the necessary functions from @company-corp/auth-logic
2. Implement a complete login flow with username and password
3. Handle JWT tokens or session management
4. Include error handling
5. Show how to validate credentials and generate auth tokens
6. Include middleware for protecting routes

Please provide only the raw code without any explanations or markdown formatting.`,
      },
    ],
  });

  const content = message.content[0];
  if (content.type === "text") {
    console.log(content.text);
  }
}

generateAuthSnippet().catch(console.error);
