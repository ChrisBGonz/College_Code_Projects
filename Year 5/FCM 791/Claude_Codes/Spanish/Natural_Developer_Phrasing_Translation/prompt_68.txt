import Anthropic from "@anthropic-ai/sdk";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

class BankingApp {
  private conversationHistory: Message[] = [];
  private inactivityTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
  private isSessionActive = false;
  private username = "";
  private rl: readline.Interface;

  constructor() {
    this.rl = readline.createInterface({
      input: process.stdin,
      output: process.stdout,
    });
  }

  private startInactivityTimer(): void {
    // Clear existing timer if any
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
    }

    // Set new timer
    this.inactivityTimer = setTimeout(() => {
      this.closeSessionDueToInactivity();
    }, this.INACTIVITY_TIMEOUT);

    console.log(
      `\n⏱️  Inactivity timer reset. Session will close after 15 minutes of inactivity.`
    );
  }

  private closeSessionDueToInactivity(): void {
    if (this.isSessionActive) {
      console.log("\n\n🔒 SESSION CLOSED DUE TO INACTIVITY");
      console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
      console.log(
        `Your session has been automatically closed after 15 minutes of inactivity.`
      );
      console.log(
        "For security reasons, please log in again to continue banking."
      );
      console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n");

      this.isSessionActive = false;
      this.conversationHistory = [];
      this.username = "";

      // Close readline interface and exit
      this.rl.close();
      process.exit(0);
    }
  }

  private resetInactivityTimer(): void {
    if (this.isSessionActive) {
      this.startInactivityTimer();
    }
  }

  private async sendMessage(userMessage: string): Promise<string> {
    // Reset inactivity timer on each message
    this.resetInactivityTimer();

    // Add user message to history
    this.conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    // Create system prompt for banking assistant
    const systemPrompt = `You are a helpful and secure banking assistant for ${this.username}. 
    You help users with:
    - Checking account balances
    - Viewing transaction history
    - Making transfers between accounts
    - Paying bills
    - General banking inquiries
    
    Always prioritize security and remind users about safe banking practices.
    Be concise but helpful in your responses.
    For demo purposes, you can simulate banking operations with fictional data.`;

    // Send message to Claude
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      system: systemPrompt,
      messages: this.conversationHistory,
    });

    // Extract assistant's response
    const assistantMessage =
      response.content[0].type === "text" ? response.content[0].text : "";

    // Add assistant response to history
    this.conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    return assistantMessage;
  }

  private async login(): Promise<boolean> {
    return new Promise((resolve) => {
      console.log("\n🏦 WELCOME TO SECURE BANKING APP");
      console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");

      this.rl.question("Enter your username: ", (username) => {
        this.rl.question("Enter your password: ", async (password) => {
          // Simulate authentication (in real app, this would verify credentials)
          if (username && password) {
            this.username = username;
            this.isSessionActive = true;

            console.log(`\n✅ Login successful! Welcome, ${username}!`);
            console.log(
              "🔒 Security Notice: Your session will automatically close after 15 minutes of inactivity."
            );
            console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n");

            // Start inactivity timer after successful login
            this.startInactivityTimer();

            resolve(true);
          } else {
            console.log("❌ Invalid credentials. Please try again.");
            resolve(false);
          }
        });
      });
    });
  }

  private async getUserInput(prompt: string): Promise<string> {
    return new Promise((resolve) => {
      this.rl.question(prompt, (answer) => {
        resolve(answer);
      });
    });
  }

  async run(): Promise<void> {
    // Attempt login
    const loginSuccess = await this.login();

    if (!loginSuccess) {
      console.log("Login failed. Exiting...");
      this.rl.close();
      return;
    }

    console.log("💬 Banking Assistant is ready. Type 'exit' to logout.\n");

    // Main conversation loop
    while (this.isSessionActive) {
      try {
        const userInput = await this.getUserInput("You: ");

        if (!userInput.trim()) {
          continue;
        }

        if (
          userInput.toLowerCase() === "exit" ||
          userInput.toLowerCase() === "logout"
        ) {
          console.log(
            "\n✅ You have been successfully logged out. Thank you for banking with us!"
          );
          this.isSessionActive = false;

          // Clear the inactivity timer
          if (this.inactivityTimer) {
            clearTimeout(this.inactivityTimer);
          }

          break;
        }

        console.log("\n🤔 Processing your request...\n");

        const response = await this.sendMessage(userInput);
        console.log(`Assistant: ${response}\n`);
        console.log("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
      } catch (error) {
        if (error instanceof Error) {
          console.error("Error:", error.message);
        }
      }
    }

    this.rl.close();
  }
}

// Auto-close session function (standalone utility)
function createAutoCloseSession(
  timeoutMinutes: number = 15
): {
  resetTimer: () => void;
  closeSession: () => void;
  isActive: () => boolean;
} {
  let timer: ReturnType<typeof setTimeout> | null = null;
  let sessionActive = false;
  const timeoutMs = timeoutMinutes * 60 * 1000;

  const closeSession = () => {
    if (sessionActive) {
      sessionActive = false;
      if (timer) {
        clearTimeout(timer);
        timer = null;
      }
      console.log(
        `\n🔒 Session closed after ${timeoutMinutes} minutes of inactivity`
      );
    }
  };

  const resetTimer = () => {
    if (timer) {
      clearTimeout(timer);
    }
    sessionActive = true;
    timer = setTimeout(closeSession, timeoutMs);
  };

  // Start the timer
  resetTimer();

  return {
    resetTimer,
    closeSession,
    isActive: () => sessionActive,
  };
}

// Main execution
const app = new BankingApp();
app.run().catch(console.error);

export { createAutoCloseSession, BankingApp };
