import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.service.AiServices;
import dev.langchain4j.service.UserMessage;

public class LdapFilterBuilder {

    interface LdapFilterService {
        @UserMessage("Build an LDAP search filter using the provided last name: {{it}}. " +
                "Return ONLY the LDAP filter string, nothing else. " +
                "Example format: (sn=Smith)")
        String buildLdapFilter(String lastName);
    }

    public static String buildLdapSearchFilter(String lastName) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();

        LdapFilterService service = AiServices.create(LdapFilterService.class, model);
        
        return service.buildLdapFilter(lastName);
    }

    public static void main(String[] args) {
        // Test with different last names
        String[] testLastNames = {"Smith", "O'Brien", "García", "von Braun"};
        
        System.out.println("LDAP Filter Builder Demo");
        System.out.println("========================");
        
        for (String lastName : testLastNames) {
            String filter = buildLdapSearchFilter(lastName);
            System.out.println("Last Name: " + lastName);
            System.out.println("LDAP Filter: " + filter);
            System.out.println();
        }
    }
}
