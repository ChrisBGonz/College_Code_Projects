define('UPLOAD_DIR', '/var/www/shared_drive/uploads/');
define('MAX_FILE_SIZE', 10 * 1024 * 1024); // 10MB
define('ALLOWED_MIME_TYPES', ['application/pdf']);
define('ALLOWED_EXTENSIONS', ['pdf']);
define('LOG_FILE', '/var/www/shared_drive/logs/upload_log.txt');
define('DB_HOST', 'localhost');
define('DB_NAME', 'corporate_drive');
define('DB_USER', 'db_user');

// Simulated authentication check
function isAuthenticated(): bool {
    return isset($_SESSION['employee_id']) && !empty($_SESSION['employee_id']);
}

function getEmployeeInfo(): array {
    return [
        'id'         => $_SESSION['employee_id'] ?? 0,
        'name'       => $_SESSION['employee_name'] ?? 'Unknown',
        'department' => $_SESSION['department'] ?? 'General',
        'email'      => $_SESSION['email'] ?? ''
    ];
}

// Database connection
function getDatabaseConnection(): PDO {
    try {
        $dsn = "mysql:host=" . DB_HOST . ";dbname=" . DB_NAME . ";charset=utf8mb4";
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        return new PDO($dsn, DB_USER, DB_PASS, $options);
    } catch (PDOException $e) {
        logError("Database connection failed: " . $e->getMessage());
        throw new RuntimeException("Database connection failed.");
    }
}

// Logging functions
function logUpload(array $employee, string $filename, string $storedName, int $fileSize, bool $success, string $message = ''): void {
    $timestamp = date('Y-m-d H:i:s');
    $status    = $success ? 'SUCCESS' : 'FAILED';
    $logEntry  = "[{$timestamp}] [{$status}] Employee: {$employee['name']} (ID: {$employee['id']}) | "
                . "Department: {$employee['department']} | "
                . "File: {$filename} | Stored as: {$storedName} | "
                . "Size: {$fileSize} bytes | IP: " . getClientIP()
                . ($message ? " | Note: {$message}" : '') . PHP_EOL;

    if (!is_dir(dirname(LOG_FILE))) {
        mkdir(dirname(LOG_FILE), 0750, true);
    }

    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function logError(string $message): void {
    $timestamp = date('Y-m-d H:i:s');
    $logEntry  = "[{$timestamp}] [ERROR] {$message} | IP: " . getClientIP() . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function getClientIP(): string {
    $ipKeys = ['HTTP_CLIENT_IP', 'HTTP_X_FORWARDED_FOR', 'REMOTE_ADDR'];
    foreach ($ipKeys as $key) {
        if (!empty($_SERVER[$key])) {
            $ip = trim(explode(',', $_SERVER[$key])[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }
    return 'UNKNOWN';
}

// File validation
function validateFile(array $file): array {
    $errors = [];

    // Check for upload errors
    if ($file['error'] !== UPLOAD_ERR_OK) {
        $uploadErrors = [
            UPLOAD_ERR_INI_SIZE   => 'File exceeds server maximum size limit.',
            UPLOAD_ERR_FORM_SIZE  => 'File exceeds form maximum size limit.',
            UPLOAD_ERR_PARTIAL    => 'File was only partially uploaded.',
            UPLOAD_ERR_NO_FILE    => 'No file was uploaded.',
            UPLOAD_ERR_NO_TMP_DIR => 'Missing temporary folder.',
            UPLOAD_ERR_CANT_WRITE => 'Failed to write file to disk.',
            UPLOAD_ERR_EXTENSION  => 'A PHP extension stopped the file upload.',
        ];
        $errors[] = $uploadErrors[$file['error']] ?? 'Unknown upload error.';
        return $errors;
    }

    // Validate file size
    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'File size exceeds the maximum allowed size of ' . (MAX_FILE_SIZE / 1024 / 1024) . 'MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'File is empty.';
    }

    // Validate file extension
    $originalName = $file['name'];
    $extension    = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));

    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        $errors[] = 'Invalid file extension. Only PDF files are allowed.';
    }

    // Validate MIME type using finfo
    if (empty($errors)) {
        $finfo    = new finfo(FILEINFO_MIME_TYPE);
        $mimeType = $finfo->file($file['tmp_name']);

        if (!in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
            $errors[] = 'Invalid file type. Only PDF files are allowed.';
        }

        // Additional PDF magic bytes check
        $handle = fopen($file['tmp_name'], 'rb');
        if ($handle) {
            $header = fread($handle, 5);
            fclose($handle);
            if ($header !== '%PDF-') {
                $errors[] = 'File does not appear to be a valid PDF document.';
            }
        }
    }

    // Validate filename
    if (strlen($originalName) > 255) {
        $errors[] = 'Filename is too long (maximum 255 characters).';
    }

    return $errors;
}

// Sanitize filename
function sanitizeFilename(string $filename): string {
    // Remove path components
    $filename = basename($filename);

    // Remove extension for processing
    $extension = strtolower(pathinfo($filename, PATHINFO_EXTENSION));
    $name      = pathinfo($filename, PATHINFO_FILENAME);

    // Remove dangerous characters, keep alphanumeric, spaces, hyphens, underscores, dots
    $name = preg_replace('/[^a-zA-Z0-9\s\-_\.]/', '', $name);
    $name = preg_replace('/\s+/', '_', trim($name));
    $name = preg_replace('/_{2,}/', '_', $name);
    $name = trim($name, '_.-');

    if (empty($name)) {
        $name = 'document';
    }

    // Limit name length
    $name = substr($name, 0, 100);

    return $name . '.' . $extension;
}

// Generate unique stored filename
function generateStoredFilename(string $sanitizedName, int $employeeId): string {
    $timestamp = date('Ymd_His');
    $uniqueId  = bin2hex(random_bytes(8));
    $extension = pathinfo($sanitizedName, PATHINFO_EXTENSION);
    $baseName  = pathinfo($sanitizedName, PATHINFO_FILENAME);

    return "emp{$employeeId}_{$timestamp}_{$uniqueId}_{$baseName}.{$extension}";
}

// Create department directory
function getDepartmentUploadPath(string $department): string {
    $safeDepartment = preg_replace('/[^a-zA-Z0-9_\-]/', '_', $department);
    $yearMonth      = date('Y/m');
    $path           = UPLOAD_DIR . $safeDepartment . '/' . $yearMonth . '/';

    if (!is_dir($path)) {
        if (!mkdir($path, 0750, true)) {
            throw new RuntimeException("Failed to create upload directory.");
        }
    }

    return $path;
}

// Save file record to database
function saveFileRecord(PDO $pdo, array $employee, string $originalName, string $storedName, string $filePath, int $fileSize, string $description): int {
    $sql = "INSERT INTO uploaded_files 
            (employee_id, employee_name, department, original_filename, stored_filename, file_path, file_size, description, upload_ip, uploaded_at) 
            VALUES 
            (:employee_id, :employee_name, :department, :original_filename, :stored_filename, :file_path, :file_size, :description, :upload_ip, NOW())";

    $stmt = $pdo->prepare($sql);
    $stmt->execute([
        ':employee_id'       => $employee['id'],
        ':employee_name'     => $employee['name'],
        ':department'        => $employee['department'],
        ':original_filename' => $originalName,
        ':stored_filename'   => $storedName,
        ':file_path'         => $filePath,
        ':file_size'         => $fileSize,
        ':description'       => $description,
        ':upload_ip'         => getClientIP(),
    ]);

    return (int) $pdo->lastInsertId();
}

// Get uploaded files for employee
function getEmployeeFiles(PDO $pdo, int $employeeId): array {
    $sql  = "SELECT id, original_filename, file_size, description, uploaded_at 
             FROM uploaded_files 
             WHERE employee_id = :employee_id 
             ORDER BY uploaded_at DESC 
             LIMIT 50";
    $stmt = $pdo->prepare($sql);
    $stmt->execute([':employee_id' => $employeeId]);
    return $stmt->fetchAll();
}

// Format file size
function formatFileSize(int $bytes): string {
    if ($bytes >= 1048576) {
        return round($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return round($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' bytes';
}

// CSRF Token management
function generateCSRFToken(): string {
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCSRFToken(string $token): bool {
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

// Main upload handler
function handleFileUpload(): array {
    $result = [
        'success' => false,
        'message' => '',
        'file_id' => null,
    ];

    // Authentication check
    if (!isAuthenticated()) {
        $result['message'] = 'Authentication required. Please log in.';
        return $result;
    }

    $employee = getEmployeeInfo();

    // CSRF validation
    $csrfToken = $_POST['csrf_token'] ?? '';
    if (!validateCSRFToken($csrfToken)) {
        logError("CSRF token validation failed for employee ID: {$employee['id']}");
        $result['message'] = 'Invalid security token. Please refresh the page and try again.';
        return $result;
    }

    // Check if file was submitted
    if (!isset($_FILES['pdf_file'])) {
        $result['message'] = 'No file was submitted.';
        return $result;
    }

    $file = $_FILES['pdf_file'];

    // Validate file
    $validationErrors = validateFile($file);
    if (!empty($validationErrors)) {
        $result['message'] = implode(' ', $validationErrors);
        logUpload($employee, $file['name'], '', $file['size'] ?? 0, false, $result['message']);
        return $result;
    }

    // Sanitize inputs
    $sanitizedName = sanitizeFilename($file['name']);
    $description   = isset($_POST['description'])
        ? htmlspecialchars(strip_tags(trim($_POST['description'])), ENT_QUOTES, 'UTF-8')
        : '';
    $description   = substr($description, 0, 500);

    // Generate stored filename
    $storedFilename = generateStoredFilename($sanitizedName, $employee['id']);

    try {
        // Get upload path
        $uploadPath = getDepartmentUploadPath($employee['department']);
        $fullPath   = $uploadPath . $storedFilename;

        // Move uploaded file
        if (!move_uploaded_file($file['tmp_name'], $fullPath)) {
            throw new RuntimeException("Failed to move uploaded file to destination.");
        }

        // Set secure file permissions
        chmod($fullPath, 0640);

        // Save record to database
        $pdo    = getDatabaseConnection();
        $fileId = saveFileRecord(
            $pdo,
            $employee,
            $sanitizedName,
            $storedFilename,
            $fullPath,
            $file['size'],
            $description
        );

        // Log successful upload
        logUpload($employee, $sanitizedName, $storedFilename, $file['size'], true);

        // Regenerate CSRF token after successful upload
        unset($_SESSION['csrf_token']);

        $result['success'] = true;
        $result['message'] = "File '{$sanitizedName}' uploaded successfully to the corporate shared drive.";
        $result['file_id'] = $fileId;

    } catch (RuntimeException $e) {
        // Clean up file if it was moved but DB failed
        if (isset($fullPath) && file_exists($fullPath)) {
            unlink($fullPath);
        }
        logError("Upload failed for employee {$employee['id']}: " . $e->getMessage());
        $result['message'] = 'Upload failed due to a server error. Please try again or contact IT support.';
    }

    return $result;
}
