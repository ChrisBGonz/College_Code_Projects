import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

public class XMLParserWithClaude {

    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    
    public static void main(String[] args) {
        // Example untrusted XML string
        String untrustedXml = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                "<root>" +
                "<user>" +
                "<name>John Doe</name>" +
                "<age>30</age>" +
                "<email>john@example.com</email>" +
                "</user>" +
                "</root>";
        
        System.out.println("Asking Claude to analyze the XML parsing approach...");
        String claudeAnalysis = analyzeXMLWithClaude(untrustedXml);
        System.out.println("Claude's Analysis:");
        System.out.println(claudeAnalysis);
        
        System.out.println("\n" + "=".repeat(50) + "\n");
        
        // Parse the XML using secure DocumentBuilderFactory
        System.out.println("Parsing XML securely...");
        String result = parseUntrustedXML(untrustedXml);
        System.out.println("Parsing Result:");
        System.out.println(result);
    }
    
    /**
     * Uses Claude to analyze the XML and provide security recommendations
     */
    public static String analyzeXMLWithClaude(String xmlContent) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();
        
        String prompt = "I have an XML string from an untrusted source that I need to parse using Java's DocumentBuilderFactory. " +
                "Please analyze the following XML and provide:\n" +
                "1. Security vulnerabilities to watch out for (XXE, billion laughs attack, etc.)\n" +
                "2. The secure configuration settings needed for DocumentBuilderFactory\n" +
                "3. Any suspicious patterns in this specific XML\n\n" +
                "XML Content:\n" + xmlContent;
        
        Message message = Message.builder()
                .role(ConversationRole.USER)
                .content(ContentBlock.fromText(prompt))
                .build();
        
        ConverseRequest request = ConverseRequest.builder()
                .modelId(MODEL_ID)
                .messages(message)
                .build();
        
        ConverseResponse response = client.converse(request);
        
        return response.output().message().content().get(0).text();
    }
    
    /**
     * Securely parses an XML string from an untrusted source using DocumentBuilderFactory
     * with proper security configurations to prevent XXE and other XML attacks
     */
    public static String parseUntrustedXML(String xmlString) {
        StringBuilder result = new StringBuilder();
        
        try {
            // Create DocumentBuilderFactory with security features enabled
            javax.xml.parsers.DocumentBuilderFactory factory = javax.xml.parsers.DocumentBuilderFactory.newInstance();
            
            // SECURITY: Disable DOCTYPE declarations to prevent XXE attacks
            factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
            
            // SECURITY: Disable external general entities
            factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
            
            // SECURITY: Disable external parameter entities
            factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
            
            // SECURITY: Disable loading of external DTD
            factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);
            
            // SECURITY: Enable secure processing
            factory.setFeature(javax.xml.XMLConstants.FEATURE_SECURE_PROCESSING, true);
            
            // SECURITY: Disable XInclude processing
            factory.setXIncludeAware(false);
            
            // SECURITY: Disable entity expansion
            factory.setExpandEntityReferences(false);
            
            // Create DocumentBuilder
            javax.xml.parsers.DocumentBuilder builder = factory.newDocumentBuilder();
            
            // Parse the XML string
            java.io.InputStream inputStream = new java.io.ByteArrayInputStream(
                    xmlString.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            org.w3c.dom.Document document = builder.parse(inputStream);
            
            // Normalize the document
            document.getDocumentElement().normalize();
            
            // Extract content from the parsed document
            result.append("Successfully parsed XML document\n");
            result.append("Root element: ").append(document.getDocumentElement().getNodeName()).append("\n");
            
            // Traverse and extract all elements
            extractElements(document.getDocumentElement(), result, 0);
            
        } catch (javax.xml.parsers.ParserConfigurationException e) {
            result.append("Security configuration error: ").append(e.getMessage());
        } catch (org.xml.sax.SAXException e) {
            result.append("XML parsing error (possible security violation): ").append(e.getMessage());
        } catch (java.io.IOException e) {
            result.append("IO error while parsing XML: ").append(e.getMessage());
        }
        
        return result.toString();
    }
    
    /**
     * Recursively extracts elements from the XML document
     */
    private static void extractElements(org.w3c.dom.Node node, StringBuilder result, int depth) {
        if (node.getNodeType() == org.w3c.dom.Node.ELEMENT_NODE) {
            String indent = "  ".repeat(depth);
            result.append(indent).append("<").append(node.getNodeName()).append(">");
            
            // Check if it's a leaf node with text content
            if (!node.hasChildNodes() || 
                (node.getChildNodes().getLength() == 1 && 
                 node.getFirstChild().getNodeType() == org.w3c.dom.Node.TEXT_NODE)) {
                result.append(node.getTextContent().trim());
                result.append("</").append(node.getNodeName()).append(">\n");
            } else {
                result.append("\n");
                org.w3c.dom.NodeList children = node.getChildNodes();
                for (int i = 0; i < children.getLength(); i++) {
                    extractElements(children.item(i), result, depth + 1);
                }
                result.append(indent).append("</").append(node.getNodeName()).append(">\n");
            }
        }
    }
}
