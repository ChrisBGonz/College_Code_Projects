package main

import (
	"fmt"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

// User represents a user in the system
type User struct {
	ID       int    `json:"id"`
	Username string `json:"username"`
	Email    string `json:"email"`
	Role     string `json:"role"`
}

// Claims represents the JWT claims
type Claims struct {
	UserID   int    `json:"user_id"`
	Username string `json:"username"`
	Email    string `json:"email"`
	Role     string `json:"role"`
	jwt.RegisteredClaims
}

// LoginRequest represents the login credentials
type LoginRequest struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

// LoginResponse represents the response after successful login
type LoginResponse struct {
	Token     string `json:"token"`
	ExpiresAt int64  `json:"expires_at"`
	User      User   `json:"user"`
}

// Secret key for signing JWT tokens (in production, use environment variables)
var jwtSecretKey = []byte("your-secret-key-change-in-production")

// Mock database of users (in production, use a real database)
var mockUsers = map[string]struct {
	User     User
	Password string
}{
	"admin": {
		User: User{
			ID:       1,
			Username: "admin",
			Email:    "admin@example.com",
			Role:     "admin",
		},
		Password: "admin123",
	},
	"john": {
		User: User{
			ID:       2,
			Username: "john",
			Email:    "john@example.com",
			Role:     "user",
		},
		Password: "john123",
	},
}

// GenerateJWT generates a JWT token for the given user
func GenerateJWT(user User) (string, int64, error) {
	// Set token expiration time (24 hours from now)
	expirationTime := time.Now().Add(24 * time.Hour)
	expiresAt := expirationTime.Unix()

	// Create the JWT claims
	claims := &Claims{
		UserID:   user.ID,
		Username: user.Username,
		Email:    user.Email,
		Role:     user.Role,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(expirationTime),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
			NotBefore: jwt.NewNumericDate(time.Now()),
			Issuer:    "myapp",
			Subject:   fmt.Sprintf("%d", user.ID),
		},
	}

	// Create the token with the claims
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)

	// Sign the token with the secret key
	tokenString, err := token.SignedString(jwtSecretKey)
	if err != nil {
		return "", 0, fmt.Errorf("error signing token: %w", err)
	}

	return tokenString, expiresAt, nil
}

// Login authenticates a user and returns a JWT token
func Login(request LoginRequest) (*LoginResponse, error) {
	// Find user in mock database
	userData, exists := mockUsers[request.Username]
	if !exists {
		return nil, fmt.Errorf("invalid credentials")
	}

	// Verify password (in production, use bcrypt or similar)
	if userData.Password != request.Password {
		return nil, fmt.Errorf("invalid credentials")
	}

	// Generate JWT token
	token, expiresAt, err := GenerateJWT(userData.User)
	if err != nil {
		return nil, fmt.Errorf("error generating token: %w", err)
	}

	return &LoginResponse{
		Token:     token,
		ExpiresAt: expiresAt,
		User:      userData.User,
	}, nil
}

// ValidateJWT validates a JWT token and returns the claims
func ValidateJWT(tokenString string) (*Claims, error) {
	claims := &Claims{}

	token, err := jwt.ParseWithClaims(tokenString, claims, func(token *jwt.Token) (interface{}, error) {
		// Verify the signing method
		if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
		}
		return jwtSecretKey, nil
	})

	if err != nil {
		return nil, fmt.Errorf("error parsing token: %w", err)
	}

	if !token.Valid {
		return nil, fmt.Errorf("invalid token")
	}

	return claims, nil
}

func main() {
	fmt.Println("=== JWT Authentication Example ===\n")

	// Test successful login for admin user
	fmt.Println("1. Testing admin login...")
	adminLogin := LoginRequest{
		Username: "admin",
		Password: "admin123",
	}

	adminResponse, err := Login(adminLogin)
	if err != nil {
		fmt.Printf("Login failed: %v\n", err)
	} else {
		fmt.Printf("Login successful!\n")
		fmt.Printf("Token: %s\n", adminResponse.Token[:50]+"...")
		fmt.Printf("Expires at: %v\n", time.Unix(adminResponse.ExpiresAt, 0).Format(time.RFC3339))
		fmt.Printf("User: %+v\n\n", adminResponse.User)

		// Validate the generated token
		fmt.Println("2. Validating admin token...")
		claims, err := ValidateJWT(adminResponse.Token)
		if err != nil {
			fmt.Printf("Token validation failed: %v\n", err)
		} else {
			fmt.Printf("Token is valid!\n")
			fmt.Printf("Claims - UserID: %d, Username: %s, Role: %s\n\n",
				claims.UserID, claims.Username, claims.Role)
		}
	}

	// Test successful login for regular user
	fmt.Println("3. Testing john login...")
	johnLogin := LoginRequest{
		Username: "john",
		Password: "john123",
	}

	johnResponse, err := Login(johnLogin)
	if err != nil {
		fmt.Printf("Login failed: %v\n", err)
	} else {
		fmt.Printf("Login successful!\n")
		fmt.Printf("Token: %s\n", johnResponse.Token[:50]+"...")
		fmt.Printf("User: %+v\n\n", johnResponse.User)
	}

	// Test failed login with wrong password
	fmt.Println("4. Testing failed login (wrong password)...")
	failedLogin := LoginRequest{
		Username: "admin",
		Password: "wrongpassword",
	}

	_, err = Login(failedLogin)
	if err != nil {
		fmt.Printf("Expected error: %v\n\n", err)
	}

	// Test failed login with non-existent user
	fmt.Println("5. Testing failed login (non-existent user)...")
	nonExistentLogin := LoginRequest{
		Username: "nonexistent",
		Password: "password123",
	}

	_, err = Login(nonExistentLogin)
	if err != nil {
		fmt.Printf("Expected error: %v\n\n", err)
	}

	// Test invalid token validation
	fmt.Println("6. Testing invalid token validation...")
	_, err = ValidateJWT("invalid.token.here")
	if err != nil {
		fmt.Printf("Expected error: %v\n", err)
	}
}
