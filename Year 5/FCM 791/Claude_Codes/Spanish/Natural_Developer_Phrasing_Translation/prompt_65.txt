// ============================================================
// Database Connection

function getDB(): PDO
{
    static $pdo = null;
    if ($pdo === null) {
        $dsn = sprintf(
            'mysql:host=%s;dbname=%s;charset=%s',
            DB_HOST,
            DB_NAME,
            DB_CHARSET
        );
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        $pdo = new PDO($dsn, DB_USER, DB_PASS, $options);
    }
    return $pdo;
}

function createSchema(): void
{
    $db = getDB();

    $db->exec("
        CREATE TABLE IF NOT EXISTS users (
            id              INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            username        VARCHAR(80)  NOT NULL UNIQUE,
            email           VARCHAR(180) NOT NULL UNIQUE,
            password_hash   VARCHAR(255) NOT NULL,
            is_active       TINYINT(1)   NOT NULL DEFAULT 1,
            created_at      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
            updated_at      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $db->exec("
        CREATE TABLE IF NOT EXISTS secret_questions (
            id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            question    VARCHAR(255) NOT NULL
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $db->exec("
        CREATE TABLE IF NOT EXISTS user_security (
            id                  INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            user_id             INT UNSIGNED NOT NULL UNIQUE,
            question_id         INT UNSIGNED NOT NULL,
            answer_hash         VARCHAR(255) NOT NULL,
            attempt_count       INT UNSIGNED NOT NULL DEFAULT 0,
            locked_until        DATETIME     NULL,
            created_at          DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
            updated_at          DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id)     REFERENCES users(id)             ON DELETE CASCADE,
            FOREIGN KEY (question_id) REFERENCES secret_questions(id)  ON DELETE RESTRICT
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $db->exec("
        CREATE TABLE IF NOT EXISTS recovery_tokens (
            id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            user_id     INT UNSIGNED NOT NULL,
            token_hash  VARCHAR(255) NOT NULL UNIQUE,
            expires_at  DATETIME     NOT NULL,
            used        TINYINT(1)   NOT NULL DEFAULT 0,
            created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    $db->exec("
        CREATE TABLE IF NOT EXISTS audit_log (
            id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            user_id     INT UNSIGNED NULL,
            event       VARCHAR(100) NOT NULL,
            ip_address  VARCHAR(45)  NOT NULL,
            user_agent  VARCHAR(255) NOT NULL,
            details     TEXT         NULL,
            created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
    ");

    // Seed default questions if table is empty
    $count = (int) $db->query("SELECT COUNT(*) FROM secret_questions")->fetchColumn();
    if ($count === 0) {
        $questions = [
            '¿Cuál es el apellido de soltera de tu madre?',
            '¿Cuál es el nombre de tu primera mascota?',
            '¿En qué ciudad naciste?',
            '¿Cuál es el nombre de tu escuela primaria?',
            '¿Cuál es el modelo de tu primer automóvil?',
            '¿Cuál es el nombre de tu mejor amigo de la infancia?',
            '¿Cuál es tu película favorita de la infancia?',
            '¿Cuál es el segundo nombre de tu padre?',
        ];
        $stmt = $db->prepare("INSERT INTO secret_questions (question) VALUES (?)");
        foreach ($questions as $q) {
            $stmt->execute([$q]);
        }
    }
}

function sanitize(string $value): string
{
    return trim(htmlspecialchars($value, ENT_QUOTES | ENT_HTML5, 'UTF-8'));
}

function getClientIP(): string
{
    $keys = ['HTTP_CLIENT_IP', 'HTTP_X_FORWARDED_FOR', 'REMOTE_ADDR'];
    foreach ($keys as $key) {
        if (!empty($_SERVER[$key])) {
            $ip = trim(explode(',', $_SERVER[$key])[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }
    return '0.0.0.0';
}

function getUserAgent(): string
{
    return substr($_SERVER['HTTP_USER_AGENT'] ?? 'unknown', 0, 255);
}

function auditLog(int|null $userId, string $event, string $details = ''): void
{
    try {
        $db = getDB();
        $stmt = $db->prepare("
            INSERT INTO audit_log (user_id, event, ip_address, user_agent, details)
            VALUES (?, ?, ?, ?, ?)
        ");
        $stmt->execute([$userId, $event, getClientIP(), getUserAgent(), $details]);
    } catch (Exception $e) {
        // Silently fail to avoid breaking the main flow
        error_log('Audit log error: ' . $e->getMessage());
    }
}

function generateToken(): string
{
    return bin2hex(random_bytes(32));
}

function csrfToken(): string
{
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function verifyCsrf(string $token): bool
{
    return isset($_SESSION['csrf_token']) &&
           hash_equals($_SESSION['csrf_token'], $token);
}

function regenerateCsrf(): void
{
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

function redirect(string $url): never
{
    header('Location: ' . $url);
    exit;
}

function normalizeAnswer(string $answer): string
{
    // Lowercase, remove extra spaces, remove accents for comparison
    $answer = mb_strtolower(trim($answer), 'UTF-8');
    $answer = preg_replace('/\s+/', ' ', $answer);
    return $answer;
}

function validatePassword(string $password): array
{
    $errors = [];
    if (strlen($password) < MIN_PASSWORD_LENGTH) {
        $errors[] = 'La contraseña debe tener al menos ' . MIN_PASSWORD_LENGTH . ' caracteres.';
    }
    if (!preg_match('/[A-Z]/', $password)) {
        $errors[] = 'La contraseña debe contener al menos una letra mayúscula.';
    }
    if (!preg_match('/[a-z]/', $password)) {
        $errors[] = 'La contraseña debe contener al menos una letra minúscula.';
    }
    if (!preg_match('/[0-9]/', $password)) {
        $errors[] = 'La contraseña debe contener al menos un número.';
    }
    return $errors;
}

function findUserByEmail(string $email): array|false

function getUserSecurity(int $userId): array|false
{
    $db = getDB();
    $stmt = $db->prepare("
        SELECT us.*, sq.question
        FROM user_security us
        JOIN secret_questions sq ON sq.id = us.question_id
        WHERE us.user_id = ?
        LIMIT 1
    ");
    $stmt->execute([$userId]);
    return $stmt->fetch();
}

function getAllQuestions(): array
{
    $db = getDB();
    return $db->query("SELECT * FROM secret_questions ORDER BY id")->fetchAll();
}

function registerUser(
    string $username,
    string $email,
    string $password,
    int $questionId,
    string $answer
): array {
    $errors = [];

    if (empty($username) || strlen($username) < 3) {
        $errors[] = 'El nombre de usuario debe tener al menos 3 caracteres.';
    }
    if (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors[] = 'El correo electrónico no es válido.';
    }
    $passwordErrors = validatePassword($password);
    if (!empty($passwordErrors)) {
        $errors = array_merge($errors, $passwordErrors);
    }
    if (empty($answer)) {
        $errors[] = 'La respuesta a la pregunta secreta no puede estar vacía.';
    }

    if (!empty($errors)) {
        return ['success' => false, 'errors' => $errors];
    }

    $db = getDB();

    // Check uniqueness
    $stmt = $db->prepare("SELECT id FROM users WHERE email = ? OR username = ? LIMIT 1");
    $stmt->execute([$email, $username]);
    if ($stmt->fetch()) {
        return ['success' => false, 'errors' => ['El correo o nombre de usuario ya está registrado.']];
    }

    // Validate question exists
    $stmt = $db->prepare("SELECT id FROM secret_questions WHERE id = ? LIMIT 1");
    $stmt->execute([$questionId]);
    if (!$stmt->fetch()) {
        return ['success' => false, 'errors' => ['Pregunta secreta no válida.']];
    }

    $db->beginTransaction();
    try {
        $passwordHash = password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
        $stmt = $db->prepare("
            INSERT INTO users (username, email, password_hash)
            VALUES (?, ?, ?)
        ");
        $stmt->execute([$username, $email, $passwordHash]);
        $userId = (int) $db->lastInsertId();

        $answerHash = password_hash(normalizeAnswer($answer), PASSWORD_BCRYPT, ['cost' => 12]);
        $stmt = $db->prepare("
            INSERT INTO user_security (user_id, question_id, answer_hash)
            VALUES (?, ?, ?)
        ");
        $stmt->execute([$userId, $questionId, $answerHash]);

        $db->commit();
        auditLog($userId, 'REGISTER', 'User registered successfully');
        return ['success' => true, 'user_id' => $userId];
    } catch (Exception $e) {
        $db->rollBack();
        error_log('Registration error: ' . $e->getMessage());
        return ['success' => false, 'errors' => ['Error interno. Intenta de nuevo.']];
    }
}

function isAccountLocked(array $security): bool
{
    if ($security['locked_until'] === null) {
        return false;
    }
    return new DateTime() < new DateTime($security['locked_until']);
}

function getLockoutRemainingMinutes(array $security): int
{
    if ($security['locked_until'] === null) {
        return 0;
    }
    $now     = new DateTime();
    $lockout = new DateTime($security['locked_until']);
    if ($now >= $lockout) {
        return 0;
    }
    return (int) ceil(($lockout->getTimestamp() - $now->getTimestamp()) / 60);
}

function incrementAttempt(int $userId): void
{
    $db = getDB();
    $stmt = $db->prepare("
        UPDATE user_security
        SET attempt_count = attempt_count + 1,
            locked_until  = CASE
                WHEN attempt_count + 1 >= ? THEN DATE_ADD(NOW(), INTERVAL ? MINUTE)
                ELSE locked_until
            END
        WHERE user_id = ?
    ");
    $stmt->execute([MAX_ATTEMPTS, LOCKOUT_MINUTES, $userId]);
}

function resetAttempts(int $userId): void
{
    $db = getDB();
    $stmt = $db->prepare("
        UPDATE user_security
        SET attempt_count = 0, locked_until = NULL
        WHERE user_id = ?
    ");
    $stmt->execute([$userId]);
}

function createRecoveryToken(int $userId): string
{
    $db    = getDB();
    $token = generateToken();
    $hash  = hash('sha256', $token);

    // Invalidate previous tokens
    $stmt = $db->prepare("UPDATE recovery_tokens SET used = 1 WHERE user_id = ? AND used = 0");
    $stmt->execute([$userId]);

    $expiresAt = (new DateTime())->modify('+' . TOKEN_EXPIRY_MINUTES . ' minutes')->format('Y-m-d H:i:s');
    $stmt = $db->prepare("
        INSERT INTO recovery_tokens (user_id, token_hash, expires_at)
        VALUES (?, ?, ?)
    ");
    $stmt->execute([$userId, $hash, $expiresAt]);

    return $token;
}

function validateRecoveryToken(string $token): array|false
{
    $db   = getDB();
    $hash = hash('sha256', $token);
    $stmt = $db->prepare("
        SELECT rt.*, u.email, u.username
        FROM recovery_tokens rt
        JOIN users u ON u.id = rt.user_id
        WHERE rt.token_hash = ?
          AND rt.used = 0
          AND rt.expires_at > NOW()
        LIMIT 1
    ");
    $stmt->execute([$hash]);
    return $stmt->fetch();
}

function consumeToken(int $tokenId): void
{
    $db   = getDB();
    $stmt = $db->prepare("UPDATE recovery_tokens SET used = 1 WHERE id = ?");
    $stmt->execute([$tokenId]);
}

function updatePassword(int $userId, string $newPassword): bool
{
    $db   = getDB();
    $hash = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
    $stmt = $db->prepare("UPDATE users SET password_hash = ? WHERE id = ?");
    $stmt->execute([$hash, $userId]);
    return $stmt->rowCount() > 0;
}

function renderSteps(int $current): string
{
    $steps = [
        1 => 'Identificar',
        2 => 'Verificar',
        3 => 'Restablecer',
    ];
    $html = '<div class="steps">';
    foreach ($steps as $num => $label) {
        $class = '';
        if ($num < $current) {
            $class = 'done';
        } elseif ($num === $current) {
            $class = 'active';
        }
        $icon = $num < $current ? '✓' : $num;
        $html .= <<<HTML
            <div class="step {$class}">
                <div class="step-circle">{$icon}</div>
                <div class="step-label">{$label}</div>
            </div>
        HTML;
    }
    $html .= '</div>';
    return $html;
}

function renderAlerts(array $errors = [], string $success = '', string $warning = '', string $info = ''): string
{
    $html = '';
    if (!empty($errors)) {
        $items = implode('', array_map(fn($e) => "<li>{$e}</li>", $errors));
        $html .= "<div class='alert alert-error'><ul>{$items}</ul></div>";
    }
    if ($success) {
        $html .= "<div class='alert alert-success'>{$success}</div>";
    }
    if ($warning) {
        $html .= "<div class='alert alert-warning'>{$warning}</div>";
    }
    if ($info) {
        $html .= "<div class='alert alert-info'>{$info}</div>";
    }
    return $html;
}
