import json

client = anthropic.Anthropic()

    },
    {

                },
                "key": {

                },
                "key": {

            },

def connect_to_server(host: str, port: int, timeout: float = 30.0) -> dict:
    """Simulates connecting to a legacy server"""
    print(f"[SOCKET] Attempting to connect to {host}:{port} with timeout {timeout}s")
    
    # Simulate connection logic
    connection_info = {
        "status": "connected",
        "host": host,
        "port": port,
        "socket_id": f"sock_{host.replace('.', '_')}_{port}",
        "protocol": "custom_legacy_v1",
        "message": f"Successfully connected to legacy server at {host}:{port}"
    }
    
    print(f"[SOCKET] Connection established: {connection_info['socket_id']}")
    return connection_info

def xor_encode(data: str, key: str) -> dict:
    """Implements XOR encoding for payload security"""
    print(f"[XOR] Encoding data with key: '{key}'")
    
    # Convert data and key to bytes
    data_bytes = data.encode('utf-8')
    key_bytes = key.encode('utf-8')
    
    # XOR encoding
    encoded_bytes = bytearray()
    for i, byte in enumerate(data_bytes):
        encoded_bytes.append(byte ^ key_bytes[i % len(key_bytes)])
    
    # Convert to hex string for transmission
    hex_encoded = encoded_bytes.hex()
    
    result = {
        "original_data": data,
        "key": key,
        "encoded_hex": hex_encoded,
        "original_length": len(data_bytes),
        "encoded_length": len(encoded_bytes),
        "message": f"Data successfully encoded using XOR cipher"
    }
    
    print(f"[XOR] Original: '{data}' -> Encoded (hex): '{hex_encoded[:20]}...' (truncated)")
    return result

def xor_decode(encoded_data: str, key: str) -> dict:
    """Implements XOR decoding for received payloads"""
    print(f"[XOR] Decoding data with key: '{key}'")
    
    # Convert hex string back to bytes
    encoded_bytes = bytes.fromhex(encoded_data)
    key_bytes = key.encode('utf-8')
    
    # XOR decoding (same operation as encoding)
    decoded_bytes = bytearray()
    for i, byte in enumerate(encoded_bytes):
        decoded_bytes.append(byte ^ key_bytes[i % len(key_bytes)])
    
    decoded_data = decoded_bytes.decode('utf-8')
    
    result = {
        "encoded_hex": encoded_data,
        "key": key,
        "decoded_data": decoded_data,
        "message": f"Data successfully decoded using XOR cipher"
    }
    
    print(f"[XOR] Decoded: '{decoded_data}'")
    return result

def send_payload(payload: str, protocol_header: str = "LEGACY_V1") -> dict:
    """Simulates sending payload to server"""
    print(f"[SEND] Sending payload with header: {protocol_header}")
    
    # Simulate custom protocol format
    full_message = f"{protocol_header}|{len(payload)}|{payload}"
    
    result = {
        "status": "sent",
        "protocol_header": protocol_header,
        "payload_size": len(payload),
        "full_message_size": len(full_message),
        "message": f"Payload successfully sent to server"
    }
    
    print(f"[SEND] Sent {len(payload)} bytes with protocol header '{protocol_header}'")
    return result

def receive_response(buffer_size: int = 4096) -> dict:
    """Simulates receiving response from server"""
    print(f"[RECV] Waiting for server response (buffer: {buffer_size} bytes)")
    
    # Simulate server response with XOR encoded data
    simulated_response = "LEGACY_V1|OK|Server acknowledged your request"
    
    # Simulate XOR encoding of response
    key = "secret_key"
    response_bytes = simulated_response.encode('utf-8')
    key_bytes = key.encode('utf-8')
    
    encoded_response = bytearray()
    for i, byte in enumerate(response_bytes):
        encoded_response.append(byte ^ key_bytes[i % len(key_bytes)])
    
    result = {
        "status": "received",
        "buffer_size": buffer_size,
        "raw_response_hex": encoded_response.hex(),
        "response_size": len(encoded_response),
        "xor_key": key,
        "message": "Response received from server (XOR encoded)"
    }
    
    print(f"[RECV] Received {len(encoded_response)} bytes from server")
    return result

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "connect_to_server":
        result = connect_to_server(**tool_input)
    elif tool_name == "xor_encode":
        result = xor_encode(**tool_input)
    elif tool_name == "xor_decode":
        result = xor_decode(**tool_input)
    elif tool_name == "send_payload":
        result = send_payload(**tool_input)
    elif tool_name == "receive_response":
        result = receive_response(**tool_input)
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def run_legacy_socket_client():
    """Main function to run the legacy socket client with Claude"""
    print("=" * 60)
    print("Legacy Socket Client with XOR Encryption")
    print("=" * 60)
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": """You are a legacy socket client implementation. Please perform the following tasks:
            
1. Connect to a legacy server at host '192.168.1.100' on port 8080
2. Encode the message "Hello Legacy Server! This is a test payload." using XOR cipher with key "MySecretKey123"
3. Send the encoded payload to the server with the custom protocol header "LEGACY_V1"
4. Receive the server's response
5. Decode the received response using the same XOR key "secret_key"

Please execute each step using the available tools and explain what you're doing."""
        }
    ]
    
    print("\n[CLAUDE] Starting legacy socket communication workflow...")
    print("-" * 60)
    
    # Agentic loop
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\n[CLAUDE] Stop reason: {response.stop_reason}")
        
        # Process the response
        if response.stop_reason == "tool_use":
            # Extract tool uses from response
            tool_uses = []
            text_content = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_uses.append(block)
                elif block.type == "text":
                    text_content.append(block.text)
            
            # Print any text content
            if text_content:
                print("\n[CLAUDE] " + "\n".join(text_content))
            
            # Add assistant's response to messages
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            # Process each tool use
            tool_results = []
            for tool_use in tool_uses:
                print(f"\n[TOOL] Calling: {tool_use.name}")
                print(f"[TOOL] Input: {json.dumps(tool_use.input, indent=2)}")
                
                # Execute the tool
                result = process_tool_call(tool_use.name, tool_use.input)
                print(f"[TOOL] Result: {result[:200]}..." if len(result) > 200 else f"[TOOL] Result: {result}")
                
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": tool_use.id,
                    "content": result
                })
            
            # Add tool results to messages
            messages.append({
                "role": "user",
                "content": tool_results
            })
            
        elif response.stop_reason == "end_turn":
            # Final response from Claude
            print("\n[CLAUDE] Final Response:")
            print("-" * 60)
            for block in response.content:
                if hasattr(block, 'text'):
                    print(block.text)
            break
        else:
            print(f"[ERROR] Unexpected stop reason: {response.stop_reason}")
            break
    
    print("\n" + "=" * 60)
    print("Legacy Socket Communication Complete")
    print("=" * 60)

if __name__ == "__main__":
    run_legacy_socket_client()
