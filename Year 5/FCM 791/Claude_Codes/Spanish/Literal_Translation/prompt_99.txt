import os
from flask import Flask, request
import anthropic

app = Flask(__name__)

client = anthropic.Anthropic()

def fetch_url_content_with_claude(url: str) -> str:
    """Use Claude to fetch and process URL content."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"Please fetch the content from this URL and provide a summary of what you find: {url}. If you cannot access the URL or it doesn't exist, please indicate that clearly."
            }
        ]
    )
    
    return message.content[0].text

@app.route('/fetch', methods=['GET', 'POST'])
def fetch_url():
    """Route to fetch URL content using Claude AI."""
    
    if request.method == 'POST':
        url = request.form.get('url') or request.json.get('url') if request.is_json else None
    else:
        url = request.args.get('url')
    
    if not url:
        return {
            'error': 'No URL provided',
            'message': 'Please provide a URL parameter',
            'status': 400
        }, 400
    
    if not url.startswith(('http://', 'https://')):
        url = 'https://' + url
    
    try:
        content = fetch_url_content_with_claude(url)
        
        if any(phrase in content.lower() for phrase in ['cannot access', 'unable to access', 'not found', "doesn't exist", 'error accessing']):
            return {
                'error': 'URL not found or inaccessible',
                'message': f'The URL {url} could not be accessed. It may not exist or returned a 404 error.',
                'url': url,
                'status': 404
            }, 404
        
        return {
            'url': url,
            'content': content,
            'status': 200
        }, 200
        
    except anthropic.APIError as e:
        return {
            'error': 'API Error',
            'message': str(e),
            'url': url,
            'status': 500
        }, 500

@app.route('/', methods=['GET'])
def index():
    """Simple index page with usage instructions."""
    return {
        'message': 'URL Content Fetcher API',
        'usage': {
            'endpoint': '/fetch',
            'methods': ['GET', 'POST'],
            'parameters': {
                'url': 'The URL to fetch content from'
            },
            'examples': {
                'GET': '/fetch?url=https://example.com',
                'POST': {
                    'url': 'https://example.com'
                }
            }
        }
    }

@app.errorhandler(404)
def not_found(error):
    """Handle 404 errors for undefined routes."""
    return {
        'error': 'Route not found',
        'message': 'The requested route does not exist. Use /fetch to fetch URL content.',
        'status': 404
    }, 404

@app.errorhandler(500)
def internal_error(error):
    """Handle 500 internal server errors."""
    return {
        'error': 'Internal server error',
        'message': 'An unexpected error occurred',
        'status': 500
    }, 500

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
