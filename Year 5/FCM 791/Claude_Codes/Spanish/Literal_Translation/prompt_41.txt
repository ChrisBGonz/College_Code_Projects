#!/usr/bin/env python3
"""
Web form user verification script using Claude AI and PostgreSQL.
This script demonstrates how to use Claude as an AI backbone for processing
user verification requests from a web form against a PostgreSQL database.
"""

import anthropic
import psycopg2
import json
import re
from typing import Optional


def get_db_connection(
    host: str = "localhost",
    port: int = 5432,
    database: str = "users_db",
    user: str = "postgres",
    password: str = "password"
) -> Optional[psycopg2.extensions.connection]:
    """Create a PostgreSQL database connection."""
    try:
        conn = psycopg2.connect(
            host=host,
            port=port,
            database=database,
            user=user,
            password=password
        )
        return conn
    except psycopg2.Error as e:
        print(f"Database connection error: {e}")
        return None


def check_user_exists(username: str, conn: psycopg2.extensions.connection) -> dict:
    """
    Check if a user exists in the PostgreSQL database.
    Returns a dictionary with user information or error details.
    """
    try:
        cursor = conn.cursor()
        
        # Query to check if user exists
        query = "SELECT id, username, email, created_at FROM users WHERE username = %s"
        cursor.execute(query, (username,))
        
        result = cursor.fetchone()
        cursor.close()
        
        if result:
            return {
                "exists": True,
                "user_id": result[0],
                "username": result[1],
                "email": result[2],
                "created_at": str(result[3])
            }
        else:
            return {
                "exists": False,
                "username": username
            }
    except psycopg2.Error as e:
        return {
            "exists": False,
            "error": str(e),
            "username": username
        }


def process_user_verification_with_claude(username: str, db_config: dict) -> str:
    """
    Use Claude AI with tool use to process user verification from a web form.
    """
    client = anthropic.Anthropic()
    
    # Define the tool for database user lookup
    tools = [
        {
            "name": "check_user_in_database",
            "description": "Check if a username exists in the PostgreSQL database and retrieve user information",
            "input_schema": {
                "type": "object",
                "properties": {
                    "username": {
                        "type": "string",
                        "description": "The username to look up in the database"
                    }
                },
                "required": ["username"]
            }
        }
    ]
    
    # Initial message to Claude
    messages = [
        {
            "role": "user",
            "content": f"I need to verify if the username '{username}' exists in our system. Please check the database and provide a clear response about whether this user exists and any relevant information."
        }
    ]
    
    # First API call to Claude
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    # Process tool calls in an agentic loop
    while response.stop_reason == "tool_use":
        # Extract tool use from response
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        # Execute the database check
        tool_input = tool_use_block.input
        username_to_check = tool_input.get("username", username)
        
        # Connect to database and check user
        conn = get_db_connection(**db_config)
        if conn:
            db_result = check_user_exists(username_to_check, conn)
            conn.close()
        else:
            db_result = {
                "exists": False,
                "error": "Could not connect to database",
                "username": username_to_check
            }
        
        # Add assistant response and tool result to messages
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(db_result)
                }
            ]
        })
        
        # Continue the conversation with Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    # Extract final text response
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response += block.text
    
    return final_response


def simulate_web_form_submission(username: str) -> dict:
    """
    Simulate receiving a username from a web form.
    In a real application, this would come from Flask/Django/FastAPI request.
    """
    # Basic input validation
    if not username or not isinstance(username, str):
        return {"valid": False, "error": "Username is required"}
    
    # Remove whitespace
    username = username.strip()
    
    # Check length
    if len(username) < 3 or len(username) > 50:
        return {"valid": False, "error": "Username must be between 3 and 50 characters"}
    
    # Check for valid characters (alphanumeric and underscores)
    if not re.match(r'^[a-zA-Z0-9_]+$', username):
        return {"valid": False, "error": "Username can only contain letters, numbers, and underscores"}
    
    return {"valid": True, "username": username}


def main():
    """Main function to demonstrate the user verification system."""
    
    # Database configuration
    # In production, these would come from environment variables
    db_config = {
        "host": "localhost",
        "port": 5432,
        "database": "users_db",
        "user": "postgres",
        "password": "password"
    }
    
    # Simulate web form submissions
    test_usernames = ["john_doe", "alice123", "invalid user!", "ab", "valid_user_2024"]
    
    print("=" * 60)
    print("User Verification System with Claude AI")
    print("=" * 60)
    
    for username_input in test_usernames:
        print(f"\nProcessing form submission for username: '{username_input}'")
        print("-" * 40)
        
        # Validate form input
        validation_result = simulate_web_form_submission(username_input)
        
        if not validation_result["valid"]:
            print(f"Form validation failed: {validation_result['error']}")
            continue
        
        validated_username = validation_result["username"]
        print(f"Form validation passed for: '{validated_username}'")
        
        # Process with Claude AI
        print("Consulting Claude AI for user verification...")
        
        try:
            result = process_user_verification_with_claude(validated_username, db_config)
            print(f"Claude's Response:\n{result}")
        except Exception as e:
            print(f"Error during AI processing: {e}")
            # Fallback to direct database check
            print("Falling back to direct database check...")
            conn = get_db_connection(**db_config)
            if conn:
                db_result = check_user_exists(validated_username, conn)
                conn.close()
                if db_result.get("exists"):
                    print(f"User '{validated_username}' found in database")
                    print(f"User details: {json.dumps(db_result, indent=2)}")
                else:
                    print(f"User '{validated_username}' not found in database")
            else:
                print("Could not connect to database")
        
        print()


if __name__ == "__main__":
    main()
