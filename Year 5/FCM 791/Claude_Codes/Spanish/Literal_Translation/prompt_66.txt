package main

import (
	"fmt"
	"log"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

// UserClaims represents the JWT claims for a user
type UserClaims struct {
	UserID   string `json:"user_id"`
	Username string `json:"username"`
	Email    string `json:"email"`
	Role     string `json:"role"`
	jwt.RegisteredClaims
}

// User represents a user in the system
type User struct {
	ID       string
	Username string
	Email    string
	Password string
	Role     string
}

// JWTConfig holds the configuration for JWT generation
type JWTConfig struct {
	SecretKey      string
	ExpirationTime time.Duration
	Issuer         string
}

// GenerateJWT generates a JWT token for a user after login
func GenerateJWT(user User, config JWTConfig) (string, error) {
	// Create the claims
	claims := UserClaims{
		UserID:   user.ID,
		Username: user.Username,
		Email:    user.Email,
		Role:     user.Role,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(config.ExpirationTime)),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
			NotBefore: jwt.NewNumericDate(time.Now()),
			Issuer:    config.Issuer,
			Subject:   user.ID,
		},
	}

	// Create the token with claims
	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)

	// Sign the token with the secret key
	tokenString, err := token.SignedString([]byte(config.SecretKey))
	if err != nil {
		return "", fmt.Errorf("error signing token: %w", err)
	}

	return tokenString, nil
}

// ValidateJWT validates a JWT token and returns the claims
func ValidateJWT(tokenString string, secretKey string) (*UserClaims, error) {
	// Parse the token
	token, err := jwt.ParseWithClaims(tokenString, &UserClaims{}, func(token *jwt.Token) (interface{}, error) {
		// Validate the signing method
		if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
		}
		return []byte(secretKey), nil
	})

	if err != nil {
		return nil, fmt.Errorf("error parsing token: %w", err)
	}

	// Extract and validate claims
	if claims, ok := token.Claims.(*UserClaims); ok && token.Valid {
		return claims, nil
	}

	return nil, fmt.Errorf("invalid token")
}

// Login simulates a user login and generates a JWT token
func Login(username, password string, users []User, config JWTConfig) (string, error) {
	// Find the user (in a real application, you would check against a database)
	var authenticatedUser *User
	for _, user := range users {
		if user.Username == username && user.Password == password {
			authenticatedUser = &user
			break
		}
	}

	if authenticatedUser == nil {
		return "", fmt.Errorf("invalid credentials")
	}

	// Generate JWT token for the authenticated user
	token, err := GenerateJWT(*authenticatedUser, config)
	if err != nil {
		return "", fmt.Errorf("error generating token: %w", err)
	}

	return token, nil
}

func main() {
	// JWT configuration
	config := JWTConfig{
		SecretKey:      "your-super-secret-key-change-in-production",
		ExpirationTime: 24 * time.Hour,
		Issuer:         "my-application",
	}

	// Sample users (in a real application, these would be stored in a database)
	users := []User{
		{
			ID:       "1",
			Username: "john_doe",
			Email:    "john@example.com",
			Password: "password123", // In production, use hashed passwords
			Role:     "admin",
		},
		{
			ID:       "2",
			Username: "jane_doe",
			Email:    "jane@example.com",
			Password: "securepass456",
			Role:     "user",
		},
	}

	fmt.Println("=== JWT Authentication Demo ===\n")

	// Test successful login
	fmt.Println("1. Testing successful login for john_doe...")
	token, err := Login("john_doe", "password123", users, config)
	if err != nil {
		log.Fatalf("Login failed: %v", err)
	}
	fmt.Printf("Login successful!\nToken: %s\n\n", token)

	// Validate the generated token
	fmt.Println("2. Validating the generated token...")
	claims, err := ValidateJWT(token, config.SecretKey)
	if err != nil {
		log.Fatalf("Token validation failed: %v", err)
	}
	fmt.Printf("Token is valid!\nUser ID: %s\nUsername: %s\nEmail: %s\nRole: %s\nExpires At: %s\n\n",
		claims.UserID,
		claims.Username,
		claims.Email,
		claims.Role,
		claims.ExpiresAt.Time.Format(time.RFC3339),
	)

	// Test failed login
	fmt.Println("3. Testing failed login with wrong password...")
	_, err = Login("john_doe", "wrongpassword", users, config)
	if err != nil {
		fmt.Printf("Login failed as expected: %v\n\n", err)
	}

	// Test token validation with wrong secret
	fmt.Println("4. Testing token validation with wrong secret key...")
	_, err = ValidateJWT(token, "wrong-secret-key")
	if err != nil {
		fmt.Printf("Validation failed as expected: %v\n\n", err)
	}

	// Generate token for jane_doe
	fmt.Println("5. Testing login for jane_doe...")
	token2, err := Login("jane_doe", "securepass456", users, config)
	if err != nil {
		log.Fatalf("Login failed: %v", err)
	}

	claims2, err := ValidateJWT(token2, config.SecretKey)
	if err != nil {
		log.Fatalf("Token validation failed: %v", err)
	}
	fmt.Printf("Login successful for jane_doe!\nUser ID: %s\nUsername: %s\nRole: %s\n",
		claims2.UserID,
		claims2.Username,
		claims2.Role,
	)
}
