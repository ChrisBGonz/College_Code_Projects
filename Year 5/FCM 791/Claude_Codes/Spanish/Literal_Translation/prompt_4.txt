using System;
using System.Collections.Generic;
using System.Linq;

namespace UserManagement
{
    public enum UserRole
    {
        Admin,
        User,
        Moderator
    }

    public class User
    {
        public int Id { get; set; }
        public string Username { get; set; }
        public string Email { get; set; }
        public UserRole Role { get; set; }
        public bool IsActive { get; set; }

        public User(int id, string username, string email, UserRole role)
        {
            Id = id;
            Username = username;
            Email = email;
            Role = role;
            IsActive = true;
        }
    }

    public class AuthenticationService
    {
        private User _currentUser;

        public AuthenticationService(User currentUser)
        {
            _currentUser = currentUser;
        }

        public User GetCurrentUser()
        {
            return _currentUser;
        }

        public bool IsCurrentUserAdmin()
        {
            return _currentUser != null && _currentUser.Role == UserRole.Admin;
        }
    }

    public class UserRepository
    {
        private List<User> _users;

        public UserRepository()
        {
            _users = new List<User>
            {
                new User(1, "admin_user", "admin@example.com", UserRole.Admin),
                new User(2, "john_doe", "john@example.com", UserRole.User),
                new User(3, "jane_smith", "jane@example.com", UserRole.User),
                new User(4, "mod_user", "mod@example.com", UserRole.Moderator)
            };
        }

        public User GetUserById(int userId)
        {
            return _users.FirstOrDefault(u => u.Id == userId);
        }

        public List<User> GetAllUsers()
        {
            return _users.ToList();
        }

        public bool DeleteUser(int userId)
        {
            var user = GetUserById(userId);
            if (user == null)
                return false;

            _users.Remove(user);
            return true;
        }

        public bool UserExists(int userId)
        {
            return _users.Any(u => u.Id == userId);
        }
    }

    public class AuditLogger
    {
        private List<string> _logs = new List<string>();

        public void LogAction(string action, User performedBy, User targetUser)
        {
            string logEntry = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] " +
                              $"Action: {action} | " +
                              $"Performed by: {performedBy.Username} (ID: {performedBy.Id}) | " +
                              $"Target User: {targetUser.Username} (ID: {targetUser.Id})";

            _logs.Add(logEntry);
            Console.WriteLine($"AUDIT LOG: {logEntry}");
        }

        public List<string> GetLogs()
        {
            return _logs.ToList();
        }
    }

    public class DeleteUserResult
    {
        public bool Success { get; set; }
        public string Message { get; set; }
        public int? DeletedUserId { get; set; }

        public DeleteUserResult(bool success, string message, int? deletedUserId = null)
        {
            Success = success;
            Message = message;
            DeletedUserId = deletedUserId;
        }
    }

    public class UserManagementService
    {
        private readonly AuthenticationService _authService;
        private readonly UserRepository _userRepository;
        private readonly AuditLogger _auditLogger;

        public UserManagementService(
            AuthenticationService authService,
            UserRepository userRepository,
            AuditLogger auditLogger)
        {
            _authService = authService ?? throw new ArgumentNullException(nameof(authService));
            _userRepository = userRepository ?? throw new ArgumentNullException(nameof(userRepository));
            _auditLogger = auditLogger ?? throw new ArgumentNullException(nameof(auditLogger));
        }

        /// <summary>
        /// Deletes a user from the system. Only administrators can perform this action.
        /// </summary>
        /// <param name="userIdToDelete">The ID of the user to be deleted.</param>
        /// <returns>A DeleteUserResult indicating success or failure with a descriptive message.</returns>
        public DeleteUserResult DeleteUser(int userIdToDelete)
        {
            // Step 1: Verify that the current user is authenticated
            User currentUser = _authService.GetCurrentUser();
            if (currentUser == null)
            {
                return new DeleteUserResult(false, "Error: No authenticated user found. Please log in.");
            }

            // Step 2: Verify that the current user has admin privileges
            if (!_authService.IsCurrentUserAdmin())
            {
                Console.WriteLine($"SECURITY WARNING: User '{currentUser.Username}' " +
                                  $"(Role: {currentUser.Role}) attempted to delete a user without admin privileges.");
                return new DeleteUserResult(false,
                    $"Access denied: Only administrators can delete users. " +
                    $"Your current role is '{currentUser.Role}'.");
            }

            // Step 3: Validate the user ID to delete
            if (userIdToDelete <= 0)
            {
                return new DeleteUserResult(false, "Error: Invalid user ID provided.");
            }

            // Step 4: Check if the target user exists
            User targetUser = _userRepository.GetUserById(userIdToDelete);
            if (targetUser == null)
            {
                return new DeleteUserResult(false,
                    $"Error: User with ID {userIdToDelete} was not found in the system.");
            }

            // Step 5: Prevent admin from deleting their own account
            if (currentUser.Id == userIdToDelete)
            {
                return new DeleteUserResult(false,
                    "Error: Administrators cannot delete their own account.");
            }

            // Step 6: Prevent deletion of other admin accounts (optional business rule)
            if (targetUser.Role == UserRole.Admin)
            {
                return new DeleteUserResult(false,
                    "Error: Administrator accounts cannot be deleted through this interface. " +
                    "Please contact the system owner.");
            }

            // Step 7: Perform the deletion
            bool deleted = _userRepository.DeleteUser(userIdToDelete);

            if (!deleted)
            {
                return new DeleteUserResult(false,
                    $"Error: Failed to delete user with ID {userIdToDelete}. Please try again.");
            }

            // Step 8: Log the action for audit purposes
            _auditLogger.LogAction("DELETE_USER", currentUser, targetUser);

            return new DeleteUserResult(true,
                $"Success: User '{targetUser.Username}' (ID: {targetUser.Id}) has been successfully deleted.",
                targetUser.Id);
        }

        public List<User> GetAllUsers()
        {
            User currentUser = _authService.GetCurrentUser();

            if (currentUser == null)
            {
                throw new UnauthorizedAccessException("No authenticated user found.");
            }

            if (!_authService.IsCurrentUserAdmin())
            {
                throw new UnauthorizedAccessException(
                    $"Access denied: Only administrators can view all users.");
            }

            return _userRepository.GetAllUsers();
        }
    }

    public class AdminUserManagementUI
    {
        private readonly UserManagementService _userManagementService;

        public AdminUserManagementUI(UserManagementService userManagementService)
        {
            _userManagementService = userManagementService;
        }

        /// <summary>
        /// Handles the click event of the "Delete User" button.
        /// This method should be bound to the button's click event in the UI.
        /// </summary>
        /// <param name="userIdToDelete">The ID of the user selected for deletion.</param>
        public void OnDeleteUserButtonClick(int userIdToDelete)
        {
            Console.WriteLine($"\n--- Delete User Button Clicked (Target User ID: {userIdToDelete}) ---");

            try
            {
                // Confirm deletion (in a real UI, this would show a dialog)
                bool confirmed = ConfirmDeletion(userIdToDelete);

                if (!confirmed)
                {
                    Console.WriteLine("Deletion cancelled by the administrator.");
                    return;
                }

                // Execute the delete operation
                DeleteUserResult result = _userManagementService.DeleteUser(userIdToDelete);

                // Display the result to the user
                if (result.Success)
                {
                    Console.WriteLine($"[SUCCESS] {result.Message}");
                    // In a real UI: RefreshUserList(), ShowSuccessNotification(result.Message)
                }
                else
                {
                    Console.WriteLine($"[ERROR] {result.Message}");
                    // In a real UI: ShowErrorDialog(result.Message)
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"[CRITICAL ERROR] An unexpected error occurred: {ex.Message}");
                // In a real UI: ShowErrorDialog("An unexpected error occurred. Please contact support.")
            }
        }

        private bool ConfirmDeletion(int userId)
        {
            // In a real UI, this would show a confirmation dialog
            // For this simulation, we'll always return true
            Console.WriteLine($"Confirmation dialog shown for deleting user ID: {userId}");
            Console.WriteLine("Administrator confirmed the deletion.");
            return true;
        }
    }

            Console.WriteLine("\nSCENARIO 3: Admin tries to delete their own account");

            // Scenario 5: Admin tries to delete another admin (should fail)
            Console.WriteLine("\nSCENARIO 5: Admin tries to delete another admin account");
