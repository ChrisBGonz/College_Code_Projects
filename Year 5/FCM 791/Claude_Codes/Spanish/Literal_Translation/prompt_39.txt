import anthropic
import socket
import threading
import time
import json

# XOR cipher functions
def xor_encrypt(data: bytes, key: bytes) -> bytes:
    """Encrypt data using XOR cipher with the given key."""
    encrypted = bytearray()
    for i, byte in enumerate(data):
        encrypted.append(byte ^ key[i % len(key)])
    return bytes(encrypted)

def xor_decrypt(data: bytes, key: bytes) -> bytes:
    """Decrypt data using XOR cipher (same as encrypt for XOR)."""
    return xor_encrypt(data, key)

# Legacy server simulation
class LegacyServer:
    def __init__(self, host: str = 'localhost', port: int = 9999, key: bytes = b'secret'):
        self.host = host
        self.port = port
        self.key = key
        self.server_socket = None
        self.running = False
        
    def start(self):
        """Start the legacy server."""
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(5)
        self.running = True
        print(f"Legacy server started on {self.host}:{self.port}")
        
        # Accept connections in a separate thread
        thread = threading.Thread(target=self._accept_connections)
        thread.daemon = True
        thread.start()
        
    def _accept_connections(self):
        """Accept and handle client connections."""
        while self.running:
            try:
                self.server_socket.settimeout(1.0)
                client_socket, address = self.server_socket.accept()
                print(f"Legacy server: Connection from {address}")
                thread = threading.Thread(target=self._handle_client, args=(client_socket,))
                thread.daemon = True
                thread.start()
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    print(f"Legacy server error: {e}")
                break
                
    def _handle_client(self, client_socket: socket.socket):
        """Handle a client connection."""
        try:
            while True:
                # Receive encrypted data
                data = client_socket.recv(4096)
                if not data:
                    break
                    
                # Decrypt the received data
                decrypted = xor_decrypt(data, self.key)
                message = decrypted.decode('utf-8')
                print(f"Legacy server received: {message}")
                
                # Process the message and send response
                response = self._process_message(message)
                
                # Encrypt and send response
                encrypted_response = xor_encrypt(response.encode('utf-8'), self.key)
                client_socket.send(encrypted_response)
                
        except Exception as e:
            print(f"Legacy server client error: {e}")
        finally:
            client_socket.close()
            
    def _process_message(self, message: str) -> str:
        """Process incoming message and return response."""
        try:
            data = json.loads(message)
            command = data.get('command', '')
            
            if command == 'get_status':
                return json.dumps({
                    'status': 'operational',
                    'version': '1.0.0',
                    'uptime': '24h',
                    'temperature': '72°F',
                    'load': '45%'
                })
            elif command == 'get_data':
                return json.dumps({
                    'records': [
                        {'id': 1, 'name': 'Record A', 'value': 100},
                        {'id': 2, 'name': 'Record B', 'value': 200},
                        {'id': 3, 'name': 'Record C', 'value': 300}
                    ]
                })
            elif command == 'ping':
                return json.dumps({'response': 'pong', 'timestamp': time.time()})
            else:
                return json.dumps({'error': f'Unknown command: {command}'})
        except json.JSONDecodeError:
            return json.dumps({'error': 'Invalid JSON message'})
            
    def stop(self):
        """Stop the legacy server."""
        self.running = False
        if self.server_socket:
            self.server_socket.close()
        print("Legacy server stopped")

# Legacy client with XOR encryption
class LegacyClient:
    def __init__(self, host: str = 'localhost', port: int = 9999, key: bytes = b'secret'):
        self.host = host
        self.port = port
        self.key = key
        self.socket = None
        
    def connect(self):
        """Connect to the legacy server."""
        self.socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.socket.connect((self.host, self.port))
        print(f"Connected to legacy server at {self.host}:{self.port}")
        
    def send_command(self, command: str, **kwargs) -> dict:
        """Send a command to the legacy server and return the response."""
        message = json.dumps({'command': command, **kwargs})
        
        # Encrypt the message
        encrypted = xor_encrypt(message.encode('utf-8'), self.key)
        
        # Send encrypted message
        self.socket.send(encrypted)
        
        # Receive encrypted response
        response_data = self.socket.recv(4096)
        
        # Decrypt the response
        decrypted = xor_decrypt(response_data, self.key)
        
        return json.loads(decrypted.decode('utf-8'))
        
    def disconnect(self):
        """Disconnect from the legacy server."""
        if self.socket:
            self.socket.close()
            print("Disconnected from legacy server")

# Claude AI integration
def create_claude_tools():
    """Create tool definitions for Claude to interact with the legacy server."""
    return [
        {
            "name": "get_server_status",
            "description": "Get the current status of the legacy server including operational status, version, uptime, temperature, and load.",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        },
        {
            "name": "get_server_data",
            "description": "Retrieve data records from the legacy server.",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        },
        {
            "name": "ping_server",
            "description": "Send a ping to the legacy server to check connectivity.",
            "input_schema": {
                "type": "object",
                "properties": {},
                "required": []
            }
        }
    ]

def process_tool_call(tool_name: str, tool_input: dict, client: LegacyClient) -> str:
    """Process a tool call from Claude and return the result."""
    try:
        if tool_name == "get_server_status":
            result = client.send_command('get_status')
            return json.dumps(result)
        elif tool_name == "get_server_data":
            result = client.send_command('get_data')
            return json.dumps(result)
        elif tool_name == "ping_server":
            result = client.send_command('ping')
            return json.dumps(result)
        else:
            return json.dumps({'error': f'Unknown tool: {tool_name}'})
    except Exception as e:
        return json.dumps({'error': str(e)})

def chat_with_claude_about_legacy_server(user_message: str, client: LegacyClient):
    """Use Claude to interact with the legacy server based on user input."""
    anthropic_client = anthropic.Anthropic()
    tools = create_claude_tools()
    
    messages = [
        {
            "role": "user",
            "content": user_message
        }
    ]
    
    print(f"\nUser: {user_message}")
    print("-" * 50)
    
    # Agentic loop
    while True:
        response = anthropic_client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages,
            system="You are an assistant that helps interact with a legacy server system. Use the available tools to query the server and provide helpful information to the user. The server communicates using XOR encryption for security."
        )
        
        # Check if we need to process tool calls
        if response.stop_reason == "tool_use":
            # Process all tool calls in the response
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"Claude is calling tool: {tool_name}")
                    
                    # Execute the tool
                    result = process_tool_call(tool_name, tool_input, client)
                    print(f"Tool result: {result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            # Add assistant response and tool results to messages
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        else:
            # No more tool calls, extract final response
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    final_response += content_block.text
            
            print(f"\nClaude: {final_response}")
            return final_response
