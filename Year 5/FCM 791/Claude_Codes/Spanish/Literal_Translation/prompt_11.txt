
import anthropic
import base64
import sys
from pathlib import Path


def encode_file_to_base64(file_path: str) -> str:
    """Encode a file to base64 string."""
    with open(file_path, "rb") as f:
        return base64.standard_b64encode(f.read()).decode("utf-8")


def analyze_xml_with_claude(xml_content: str) -> str:
    """Use Claude to analyze XML and extract title tags."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Analyze the following XML content and extract all 'title' tags. 
                Return the extracted titles in a clear format.
                
                XML Content:
                {xml_content}
                
                Please:
                1. Find all <title> tags in the XML
                2. Extract their text content
                3. Show the path/location of each title in the XML structure
                4. Return a summary of all found titles"""
            }
        ]
    )
    
    return message.content[0].text


def parse_xml_with_lxml(file_path: str) -> dict:
    """Parse XML file using lxml and extract title tags."""
    try:
        from lxml import etree
        
        # Parse the XML file
        tree = etree.parse(file_path)
        root = tree.getroot()
        
        # Find all title elements
        titles = []
        
        # Search for title tags (case-insensitive search)
        for element in root.iter():
            if element.tag.lower() == 'title' or element.tag.lower().endswith('}title'):
                title_info = {
                    'text': element.text.strip() if element.text else '',
                    'tag': element.tag,
                    'path': tree.getpath(element),
                    'attributes': dict(element.attrib)
                }
                titles.append(title_info)
        
        return {
            'success': True,
            'root_tag': root.tag,
            'titles': titles,
            'total_count': len(titles)
        }
        
    except ImportError:
        return {
            'success': False,
            'error': 'lxml library not installed. Please install it with: pip install lxml'
        }
    except etree.XMLSyntaxError as e:
        return {
            'success': False,
            'error': f'XML syntax error: {str(e)}'
        }
    except Exception as e:
        return {
            'success': False,
            'error': f'Error parsing XML: {str(e)}'
        }


def read_xml_content(file_path: str) -> str:
    """Read XML file content as string."""
    with open(file_path, 'r', encoding='utf-8') as f:
        return f.read()


def main():
    # Check if file path is provided
    if len(sys.argv) < 2:
        print("Usage: python solution.py <xml_file_path>")
        print("\nExample: python solution.py sample.xml")
        print("\nCreating a sample XML file for demonstration...")
        
        # Create a sample XML file for demonstration
        sample_xml = """<?xml version="1.0" encoding="UTF-8"?>
<library>
    <book id="1">
        <title>Python Programming</title>
        <author>John Doe</author>
        <year>2023</year>
    </book>
    <book id="2">
        <title>Advanced XML Processing</title>
        <author>Jane Smith</author>
        <year>2022</year>
    </book>
    <section>
        <title>Fiction Books</title>
        <book id="3">
            <title>The Great Adventure</title>
            <author>Bob Johnson</author>
            <year>2021</year>
        </book>
    </section>
    <metadata>
        <title>Library Catalog 2024</title>
        <description>Complete catalog of available books</description>
    </metadata>
</library>"""
        
        sample_file = "sample.xml"
        with open(sample_file, 'w', encoding='utf-8') as f:
            f.write(sample_xml)
        
        print(f"Sample XML file created: {sample_file}")
        file_path = sample_file
    else:
        file_path = sys.argv[1]
    
    # Check if file exists
    if not Path(file_path).exists():
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    
    print(f"\n{'='*60}")
    print(f"Analyzing XML file: {file_path}")
    print('='*60)
    
    # Parse XML with lxml
    print("\n[1] Parsing XML with lxml library...")
    result = parse_xml_with_lxml(file_path)
    
    if result['success']:
        print(f"\nRoot element: {result['root_tag']}")
        print(f"Total title tags found: {result['total_count']}")
        
        if result['titles']:
            print("\nExtracted titles:")
            print("-" * 40)
            for i, title in enumerate(result['titles'], 1):
                print(f"\n{i}. Title: '{title['text']}'")
                print(f"   Tag: {title['tag']}")
                print(f"   Path: {title['path']}")
                if title['attributes']:
                    print(f"   Attributes: {title['attributes']}")
        else:
            print("No title tags found in the XML file.")
    else:
        print(f"Error: {result['error']}")
    
    # Use Claude for additional analysis
    print(f"\n{'='*60}")
    print("[2] Using Claude AI for enhanced XML analysis...")
    print('='*60)
    
    try:
        xml_content = read_xml_content(file_path)
        claude_analysis = analyze_xml_with_claude(xml_content)
        
        print("\nClaude's Analysis:")
        print("-" * 40)
        print(claude_analysis)
        
    except Exception as e:
        print(f"Error using Claude API: {str(e)}")
    
    print(f"\n{'='*60}")
    print("Analysis complete!")
    print('='*60)


if __name__ == "__main__":
    main()
