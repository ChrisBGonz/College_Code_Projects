using System;
using System.Collections.Generic;
using System.Data;
using Anthropic.SDK;
using Anthropic.SDK.Constants;
using Anthropic.SDK.Messaging;

class DynamicSQLSorter
{
    private readonly AnthropicClient _client;
    
    public DynamicSQLSorter()
    {
        _client = new AnthropicClient();
    }
    
    public async Task<string> GenerateSortedQueryAsync(string tableName, string columnName, string sortOrder = "ASC")
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Generate a safe SQL query to sort a table named '{tableName}' by column '{columnName}' in {sortOrder} order.
                        
Requirements:
1. The query should be safe and prevent SQL injection
2. Use parameterized approach or proper escaping
3. Return only the SQL query, nothing else
4. The query should SELECT all columns from the table
5. Include proper ORDER BY clause

Table: {tableName}
Column to sort by: {columnName}
Sort order: {sortOrder}"
                    }
                }
            }
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 1024,
            Stream = false
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        return response.Content[0].ToString() ?? string.Empty;
    }
    
    public async Task<DataTable> ExecuteSortedQueryAsync(string connectionString, string tableName, string columnName, string sortOrder = "ASC")
    {
        // First, validate the column name using Claude
        var validationMessages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Validate if the following column name is safe to use in a SQL query (no SQL injection risk):
Column name: '{columnName}'
Table name: '{tableName}'

Respond with only 'SAFE' or 'UNSAFE' followed by a brief reason."
                    }
                }
            }
        };
        
        var validationParameters = new MessageParameters
        {
            Messages = validationMessages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 256,
            Stream = false
        };
        
        var validationResponse = await _client.Messages.GetClaudeMessageAsync(validationParameters);
        var validationResult = validationResponse.Content[0].ToString() ?? string.Empty;
        
        if (!validationResult.StartsWith("SAFE", StringComparison.OrdinalIgnoreCase))
        {
            throw new ArgumentException($"Column name validation failed: {validationResult}");
        }
        
        // Generate the SQL query using Claude
        var sqlQuery = await GenerateSortedQueryAsync(tableName, columnName, sortOrder);
        
        Console.WriteLine($"Generated SQL Query: {sqlQuery}");
        
        // For demonstration, return a sample DataTable
        // In a real implementation, you would execute this against your database
        var dataTable = new DataTable();
        dataTable.TableName = tableName;
        
        // Simulate query execution result
        Console.WriteLine($"Query would be executed against connection: {connectionString}");
        Console.WriteLine($"Sorting table '{tableName}' by column '{columnName}' in {sortOrder} order");
        
        return dataTable;
    }
    
    public async Task<string> ExplainQueryAsync(string tableName, string columnName, string sortOrder = "ASC")
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"Explain what the following SQL operation does and its performance implications:
- Table: {tableName}
- Sorting by column: {columnName}
- Sort order: {sortOrder}

Provide a brief explanation including:
1. What the query does
2. Performance considerations
3. When an index would be beneficial"
                    }
                }
            }
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            Model = AnthropicModels.Claude3_5Sonnet,
            MaxTokens = 512,
            Stream = false
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        return response.Content[0].ToString() ?? string.Empty;
    }
}

class Program
{

    {
