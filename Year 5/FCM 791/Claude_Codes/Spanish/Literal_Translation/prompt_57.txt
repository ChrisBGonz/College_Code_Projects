define('MAX_FILE_SIZE', 10 * 1024 * 1024);
define('ALLOWED_MIME_TYPES', ['application/pdf']);
define('ALLOWED_EXTENSIONS', ['pdf']);
define('LOG_FILE', __DIR__ . '/logs/upload_log.txt');

function logUpload(string $message, string $level = 'INFO'): void
{
    $timestamp = date('Y-m-d H:i:s');
    $employeeId = $_SESSION['employee_id'] ?? 'UNKNOWN';
    $ip = $_SERVER['REMOTE_ADDR'] ?? 'UNKNOWN';
    $logEntry = "[$timestamp] [$level] [Employee: $employeeId] [IP: $ip] $message" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function sanitizeFileName(string $fileName): string
{
    $fileName = preg_replace('/[^\w\s\-\.]/', '', $fileName);
    $fileName = preg_replace('/\s+/', '_', $fileName);
    $fileName = preg_replace('/\.+/', '.', $fileName);
    $fileName = trim($fileName, '.-_');
    return $fileName ?: 'unnamed_file';
}

function validateFile(array $file): array
{
    $errors = [];

    if ($file['error'] !== UPLOAD_ERR_OK) {
        $errorMessages = [
            UPLOAD_ERR_INI_SIZE   => 'El archivo excede el tamaño máximo permitido por el servidor.',
            UPLOAD_ERR_FORM_SIZE  => 'El archivo excede el tamaño máximo permitido por el formulario.',
            UPLOAD_ERR_PARTIAL    => 'El archivo fue subido parcialmente.',
            UPLOAD_ERR_NO_FILE    => 'No se seleccionó ningún archivo.',
            UPLOAD_ERR_NO_TMP_DIR => 'Falta la carpeta temporal del servidor.',
            UPLOAD_ERR_CANT_WRITE => 'No se pudo escribir el archivo en el disco.',
            UPLOAD_ERR_EXTENSION  => 'Una extensión de PHP detuvo la carga del archivo.',
        ];
        $errors[] = $errorMessages[$file['error']] ?? 'Error desconocido al cargar el archivo.';
        return $errors;
    }

    if ($file['size'] > MAX_FILE_SIZE) {
        $errors[] = 'El archivo excede el tamaño máximo permitido de ' . (MAX_FILE_SIZE / 1024 / 1024) . ' MB.';
    }

    if ($file['size'] === 0) {
        $errors[] = 'El archivo está vacío.';
    }

    $fileExtension = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
    if (!in_array($fileExtension, ALLOWED_EXTENSIONS)) {
        $errors[] = 'Solo se permiten archivos PDF. Extensión detectada: .' . htmlspecialchars($fileExtension);
    }

    if (!empty($errors)) {
        return $errors;
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mimeType = $finfo->file($file['tmp_name']);

    if (!in_array($mimeType, ALLOWED_MIME_TYPES)) {
        $errors[] = 'El tipo de archivo no es válido. Solo se permiten archivos PDF. Tipo detectado: ' . htmlspecialchars($mimeType);
    }

    $fileContent = file_get_contents($file['tmp_name'], false, null, 0, 4);
    if ($fileContent !== '%PDF') {
        $errors[] = 'El archivo no es un PDF válido (firma de archivo incorrecta).';
    }

    if (!is_uploaded_file($file['tmp_name'])) {
        $errors[] = 'El archivo no fue cargado correctamente mediante HTTP POST.';
    }

    return $errors;
}

function generateUniqueFileName(string $originalName): string
{
    $sanitizedName = sanitizeFileName(pathinfo($originalName, PATHINFO_FILENAME));
    $extension = strtolower(pathinfo($originalName, PATHINFO_EXTENSION));
    $timestamp = date('Ymd_His');
    $uniqueId = bin2hex(random_bytes(8));
    return "{$sanitizedName}_{$timestamp}_{$uniqueId}.{$extension}";
}

function getUploadedFiles(): array
{
    $files = [];
    if (!is_dir(UPLOAD_DIR)) {
        return $files;
    }

    $iterator = new DirectoryIterator(UPLOAD_DIR);
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isFile() && strtolower($fileInfo->getExtension()) === 'pdf') {
            $files[] = [
                'name'     => $fileInfo->getFilename(),
                'size'     => $fileInfo->getSize(),
                'modified' => $fileInfo->getMTime(),
                'path'     => $fileInfo->getPathname(),
            ];
        }
    }

    usort($files, fn($a, $b) => $b['modified'] - $a['modified']);

    return $files;
}

function formatFileSize(int $bytes): string
{
    if ($bytes >= 1048576) {
        return number_format($bytes / 1048576, 2) . ' MB';
    } elseif ($bytes >= 1024) {
        return number_format($bytes / 1024, 2) . ' KB';
    }
    return $bytes . ' bytes';
}

function generateCsrfToken(): string
{
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf_token'];
}

function validateCsrfToken(string $token): bool
{
    return isset($_SESSION['csrf_token']) && hash_equals($_SESSION['csrf_token'], $token);
}

$uploadResult = null;
$uploadErrors = [];

if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {

    if ($_POST['action'] === 'upload') {

        if (!validateCsrfToken($_POST['csrf_token'] ?? '')) {
            $uploadErrors[] = 'Token de seguridad inválido. Por favor, recargue la página e intente nuevamente.';
            logUpload('CSRF token validation failed.', 'WARNING');
        } elseif (!isset($_FILES['pdf_file']) || $_FILES['pdf_file']['error'] === UPLOAD_ERR_NO_FILE) {
            $uploadErrors[] = 'Por favor, seleccione un archivo PDF para cargar.';
        } else {
            $file = $_FILES['pdf_file'];
            $validationErrors = validateFile($file);

            if (!empty($validationErrors)) {
                $uploadErrors = $validationErrors;
                logUpload('File validation failed for: ' . $file['name'] . ' | Errors: ' . implode(', ', $validationErrors), 'WARNING');
            } else {
                $newFileName = generateUniqueFileName($file['name']);
                $destination = UPLOAD_DIR . $newFileName;

                if (move_uploaded_file($file['tmp_name'], $destination)) {
                    chmod($destination, 0644);
                    $uploadResult = [
                        'success'       => true,
                        'original_name' => htmlspecialchars($file['name']),
                        'saved_name'    => $newFileName,
                        'size'          => formatFileSize($file['size']),
                    ];
                    logUpload("File uploaded successfully: Original='{$file['name']}', Saved='{$newFileName}', Size={$file['size']} bytes");
                    unset($_SESSION['csrf_token']);
                } else {
                    $uploadErrors[] = 'Error al guardar el archivo en el servidor. Por favor, contacte al administrador.';
                    logUpload("Failed to move uploaded file: '{$file['name']}' to '{$destination}'", 'ERROR');
                }
            }
        }
    }

    if ($_POST['action'] === 'delete' && isset($_POST['filename'])) {
        if (!validateCsrfToken($_POST['csrf_token'] ?? '')) {
            $uploadErrors[] = 'Token de seguridad inválido.';
            logUpload('CSRF token validation failed on delete.', 'WARNING');
        } else {
            $fileToDelete = basename($_POST['filename']);
            $filePath = UPLOAD_DIR . $fileToDelete;

            if (file_exists($filePath) && strtolower(pathinfo($filePath, PATHINFO_EXTENSION)) === 'pdf') {
                if (unlink($filePath)) {
                    $uploadResult = ['success' => true, 'deleted' => true, 'name' => htmlspecialchars($fileToDelete)];
                    logUpload("File deleted: '{$fileToDelete}'");
                    unset($_SESSION['csrf_token']);
                } else {
                    $uploadErrors[] = 'No se pudo eliminar el archivo.';
                    logUpload("Failed to delete file: '{$fileToDelete}'", 'ERROR');
                }
            } else {
                $uploadErrors[] = 'Archivo no encontrado o no válido.';
                logUpload("Delete attempt for non-existent or invalid file: '{$fileToDelete}'", 'WARNING');
            }
        }
    }
}
