<?php

function checkSubscriptionStatus(int $userId, PDO $db): string
{
    try {
        $stmt = $db->prepare("SELECT subscription_status FROM users WHERE id = :id LIMIT 1");
        $stmt->bindParam(':id', $userId, PDO::PARAM_INT);
        $stmt->execute();

        $result = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($result === false) {
            return 'free';
        }

        return $result['subscription_status'] ?? 'free';

    } catch (Exception $e) {
        error_log("Error checking subscription for user {$userId}: " . $e->getMessage());
        return 'premium';
    }
}
