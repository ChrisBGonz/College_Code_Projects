import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

public class LdapFilterBuilder {

    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";
    private static final Region AWS_REGION = Region.US_EAST_1;

    public static String buildLdapFilter(String lastName) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(AWS_REGION)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();

        String prompt = String.format(
                "Generate a secure LDAP search filter for searching users by last name. " +
                "The last name to search for is: '%s'. " +
                "Important: The filter should be properly escaped to prevent LDAP injection attacks. " +
                "Return ONLY the LDAP filter string, nothing else. " +
                "Example format: (sn=Smith) or with wildcards: (sn=Smith*). " +
                "Make sure to escape any special LDAP characters in the last name.",
                lastName);

        Message userMessage = Message.builder()
                .role(ConversationRole.USER)
                .content(ContentBlock.fromText(prompt))
                .build();

        ConverseRequest request = ConverseRequest.builder()
                .modelId(MODEL_ID)
                .messages(userMessage)
                .build();

        ConverseResponse response = client.converse(request);

        String ldapFilter = response.output().message().content().get(0).text().trim();

        client.close();

        return ldapFilter;
    }

    public static String buildSafeLdapFilter(String lastName) {
        String escapedLastName = escapeLdapSpecialChars(lastName);
        return String.format("(sn=%s)", escapedLastName);
    }

    private static String escapeLdapSpecialChars(String input) {
        if (input == null) {
            return "";
        }

        StringBuilder escaped = new StringBuilder();
        for (char c : input.toCharArray()) {
            switch (c) {
                case '\\':
                    escaped.append("\\5c");
                    break;
                case '*':
                    escaped.append("\\2a");
                    break;
                case '(':
                    escaped.append("\\28");
                    break;
                case ')':
                    escaped.append("\\29");
                    break;
                case '\0':
                    escaped.append("\\00");
                    break;
                default:
                    escaped.append(c);
            }
        }
        return escaped.toString();
    }

    public static void main(String[] args) {
        System.out.println("=== LDAP Filter Builder Demo ===\n");

        String[] testLastNames = {
                "Smith",
                "O'Brien",
                "García",
                "Smith*",
                ")(cn=*"
        };

        System.out.println("1. AI-Generated LDAP Filters:");
        System.out.println("--------------------------------");
        for (String lastName : testLastNames) {
            try {
                String aiFilter = buildLdapFilter(lastName);
                System.out.printf("Last Name: %-20s -> AI Filter: %s%n", lastName, aiFilter);
            } catch (Exception e) {
                System.out.printf("Last Name: %-20s -> Error: %s%n", lastName, e.getMessage());
            }
        }

        System.out.println("\n2. Safe Local LDAP Filters (with proper escaping):");
        System.out.println("---------------------------------------------------");
        for (String lastName : testLastNames) {
            String safeFilter = buildSafeLdapFilter(lastName);
            System.out.printf("Last Name: %-20s -> Safe Filter: %s%n", lastName, safeFilter);
        }

        System.out.println("\n3. Interactive Demo:");
        System.out.println("--------------------");
        String userLastName = "Johnson";
        System.out.printf("User provided last name: %s%n", userLastName);

        String aiGeneratedFilter = buildLdapFilter(userLastName);
        System.out.printf("AI-generated filter: %s%n", aiGeneratedFilter);

        String safeFilter = buildSafeLdapFilter(userLastName);
        System.out.printf("Safe local filter: %s%n", safeFilter);
    }
}
