// Initialize session variables

    $_SESSION['attempts'] = 0;
}
if (!isset($_SESSION['lockout_time'])) {
    $_SESSION['lockout_time'] = null;
}

// Helper functions
function sanitize_input($data) {
    return htmlspecialchars(strip_tags(trim($data)));
}

function is_locked_out() {
    if ($_SESSION['lockout_time'] !== null) {
        $elapsed = time() - $_SESSION['lockout_time'];
        if ($elapsed < LOCKOUT_TIME) {
            return true;
        } else {
            $_SESSION['attempts'] = 0;
            $_SESSION['lockout_time'] = null;
        }
    }
    return false;
}

function get_remaining_lockout_time() {
    if ($_SESSION['lockout_time'] !== null) {
        $elapsed = time() - $_SESSION['lockout_time'];
        return max(0, LOCKOUT_TIME - $elapsed);
    }
    return 0;
}

function generate_token() {
    return bin2hex(random_bytes(32));
}

function find_user_by_username($username, &$users_db) {
    if (isset($users_db[$username])) {
        return $username;
    }
    return null;
}

function find_user_by_email($email, &$users_db) {
    foreach ($users_db as $username => $user) {
        if ($user['email'] === $email) {
            return $username;
        }
    }
    return null;
}

function find_user_by_token($token, &$users_db) {
    foreach ($users_db as $username => $user) {
        if ($user['reset_token'] === $token) {
            return $username;
        }
    }
    return null;
}

function render_step_indicator($current_step) {
    $steps = [
        1 => ['label' => 'Usuario', 'short' => '1'],
        2 => ['label' => 'Pregunta', 'short' => '2'],
        3 => ['label' => 'Nueva clave', 'short' => '3'],
        4 => ['label' => 'Listo', 'short' => '✓'],
    ];

    echo '<div class="step-indicator">';
    foreach ($steps as $step_num => $step_info) {
        $class = '';
        if ($step_num < $current_step) {
            $class = 'completed';
        } elseif ($step_num === $current_step) {
            $class = 'active';
        }
        echo '<div class="step">';
        echo '<div class="step-circle ' . $class . '">' . $step_info['short'] . '</div>';
        echo '<div class="step-label ' . $class . '">' . $step_info['label'] . '</div>';
        echo '</div>';
    }
    echo '</div>';
}

// Determine current step
$step = isset($_GET['step']) ? (int)$_GET['step'] : 1;
$action = isset($_POST['action']) ? $_POST['action'] : '';

// Process form submissions
$error_message = '';
$success_message = '';
$info_message = '';
$warning_message = '';

// ============================================================
// STEP 1: Enter username or email
// ============================================================
if ($action === 'find_user') {
    if (is_locked_out()) {
        $remaining = get_remaining_lockout_time();
        $error_message = "Demasiados intentos fallidos. Por favor espera " . ceil($remaining / 60) . " minuto(s) antes de intentarlo de nuevo.";
        $step = 1;
    } else {
        $identifier = sanitize_input($_POST['identifier'] ?? '');

        if (empty($identifier)) {
            $error_message = "Por favor ingresa tu nombre de usuario o correo electrónico.";
            $step = 1;
        } else {
            // Try to find user by username or email
            $found_username = find_user_by_username($identifier, $users_db);
            if (!$found_username) {
                $found_username = find_user_by_email($identifier, $users_db);
            }

            if ($found_username) {
                $_SESSION['recovery_username'] = $found_username;
                $_SESSION['recovery_step'] = 2;
                $_SESSION['attempts'] = 0;
                $step = 2;
            } else {
                $_SESSION['attempts']++;
                if ($_SESSION['attempts'] >= MAX_ATTEMPTS) {
                    $_SESSION['lockout_time'] = time();
                    $error_message = "Demasiados intentos fallidos. Tu cuenta ha sido bloqueada temporalmente por " . (LOCKOUT_TIME / 60) . " minutos.";
                } else {
                    $remaining_attempts = MAX_ATTEMPTS - $_SESSION['attempts'];
                    $error_message = "No se encontró ninguna cuenta con ese nombre de usuario o correo electrónico. Te quedan $remaining_attempts intento(s).";
                }
                $step = 1;
            }
        }
    }
}

// ============================================================
// STEP 2: Answer secret question
// ============================================================
if ($action === 'verify_answer') {
    if (!isset($_SESSION['recovery_username']) || $_SESSION['recovery_step'] !== 2) {
        $error_message = "Sesión inválida. Por favor comienza de nuevo.";
        $step = 1;
        unset($_SESSION['recovery_username'], $_SESSION['recovery_step']);
    } elseif (is_locked_out()) {
        $remaining = get_remaining_lockout_time();
        $error_message = "Demasiados intentos fallidos. Por favor espera " . ceil($remaining / 60) . " minuto(s) antes de intentarlo de nuevo.";
        $step = 2;
    } else {
        $answer = sanitize_input($_POST['secret_answer'] ?? '');
        $username = $_SESSION['recovery_username'];

        if (empty($answer)) {
            $error_message = "Por favor ingresa tu respuesta.";
            $step = 2;
        } else {
            $hashed_answer = strtolower(hash('sha256', strtolower($answer)));
            $user = $users_db[$username];

            if (hash_equals($user['secret_answer'], $hashed_answer)) {
                // Correct answer - generate token and proceed to step 3
                $token = generate_token();
                $users_db[$username]['reset_token'] = $token;
                $users_db[$username]['token_expiry'] = time() + TOKEN_EXPIRY;

                $_SESSION['reset_token'] = $token;
                $_SESSION['recovery_step'] = 3;
                $_SESSION['attempts'] = 0;
                $step = 3;
                $success_message = "¡Respuesta correcta! Ahora puedes establecer una nueva contraseña.";
            } else {
                $_SESSION['attempts']++;
                if ($_SESSION['attempts'] >= MAX_ATTEMPTS) {
                    $_SESSION['lockout_time'] = time();
                    unset($_SESSION['recovery_username'], $_SESSION['recovery_step'], $_SESSION['reset_token']);
                    $error_message = "Demasiados intentos fallidos. Tu cuenta ha sido bloqueada temporalmente por " . (LOCKOUT_TIME / 60) . " minutos.";
                    $step = 1;
                } else {
                    $remaining_attempts = MAX_ATTEMPTS - $_SESSION['attempts'];
                    $error_message = "Respuesta incorrecta. Te quedan $remaining_attempts intento(s).";
                    $step = 2;
                }
            }
        }
    }
}

// ============================================================
// STEP 3: Set new password
// ============================================================
if ($action === 'reset_password') {
    if (!isset($_SESSION['recovery_username']) || !isset($_SESSION['reset_token']) || $_SESSION['recovery_step'] !== 3) {
        $error_message = "Sesión inválida. Por favor comienza de nuevo.";
        $step = 1;
        unset($_SESSION['recovery_username'], $_SESSION['recovery_step'], $_SESSION['reset_token']);
    } else {
        $username = $_SESSION['recovery_username'];
        $token = $_SESSION['reset_token'];
        $new_password = $_POST['new_password'] ?? '';
        $confirm_password = $_POST['confirm_password'] ?? '';

        // Validate token
        $db_user = $users_db[$username];
        if ($db_user['reset_token'] !== $token || time() > $db_user['token_expiry']) {
            $error_message = "El token de recuperación ha expirado. Por favor comienza de nuevo.";
            $step = 1;
            unset($_SESSION['recovery_username'], $_SESSION['recovery_step'], $_SESSION['reset_token']);
            $users_db[$username]['reset_token'] = null;
            $users_db[$username]['token_expiry'] = null;
        } else {
            // Validate password
            $password_errors = [];

            if (empty($new_password)) {
                $password_errors[] = "La contraseña no puede estar vacía.";
            } elseif (strlen($new_password) < 8) {
                $password_errors[] = "La contraseña debe tener al menos 8 caracteres.";
            } elseif (!preg_match('/[A-Z]/', $new_password)) {
                $password_errors[] = "La contraseña debe contener al menos una letra mayúscula.";
            } elseif (!preg_match('/[a-z]/', $new_password)) {
                $password_errors[] = "La contraseña debe contener al menos una letra minúscula.";
            } elseif (!preg_match('/[0-9]/', $new_password)) {
                $password_errors[] = "La contraseña debe contener al menos un número.";
            }

            if ($new_password !== $confirm_password) {
                $password_errors[] = "Las contraseñas no coinciden.";
            }

            // Check if new password is same as old
            if (empty($password_errors) && password_verify($new_password, $users_db[$username]['password'])) {
                $password_errors[] = "La nueva contraseña no puede ser igual a la contraseña actual.";
            }

            if (!empty($password_errors)) {
                $error_message = implode("<br>", $password_errors);
                $step = 3;
            } else {
                // Update password
                $users_db[$username]['password'] = password_hash($new_password, PASSWORD_BCRYPT);
                $users_db[$username]['reset_token'] = null;
                $users_db[$username]['token_expiry'] = null;

                // Clear session recovery data
                unset($_SESSION['recovery_username'], $_SESSION['recovery_step'], $_SESSION['reset_token']);
                $_SESSION['attempts'] = 0;

                $step = 4;
                $success_message = "¡Tu contraseña ha sido restablecida exitosamente!";
            }
        }
    }
}

// Handle direct step navigation via GET (for back links)
if (isset($_GET['step'])) {
    $requested_step = (int)$_GET['step'];
    if ($requested_step === 1) {
        unset($_SESSION['recovery_username'], $_SESSION['recovery_step'], $_SESSION['reset_token']);
        $_SESSION['attempts'] = 0;
        $step = 1;
    } elseif ($requested_step === 2 && isset($_SESSION['recovery_username']) && $_SESSION['recovery_step'] >= 2) {
        $_SESSION['recovery_step'] = 2;
        unset($_SESSION['reset_token']);
        $step = 2;
    }
}

// Determine current step for display
if (empty($action) && !isset($_GET['step'])) {
    if (isset($_SESSION['recovery_step'])) {
        $step = $_SESSION['recovery_step'];
    } else {
        $step = 1;
    }
}

switch ($step) {

    // ============================================================

            echo '<div class="alert alert-error">' . $error_message . '</div>';
        }

        if (is_locked_out()) {

        } else {
            echo '<form method="POST" action="">
                <input type="hidden" name="action" value="find_user">
                <div class="form-group">
                    <label for="identifier">Nombre de usuario o correo electrónico</label>
                    <input type="text" id="identifier" name="identifier" placeholder="Ej: john_doe o john@example.com" required autocomplete="username">
                </div>';

            if ($_SESSION['attempts'] > 0 && $_SESSION['attempts'] < MAX_ATTEMPTS) {
                $remaining_attempts = MAX_ATTEMPTS - $_SESSION['attempts'];
                echo '<p class="attempts-info">⚠️ Intentos fallidos: ' . $_SESSION['attempts'] . '/' . MAX_ATTEMPTS . '. Te quedan ' . $remaining_attempts . ' intento(s).</p>';

            </form>';
        }

        echo '<div class="link-group">
            <a href="#">← Volver al inicio de sesión</a>
        </div>';
        break;

    // ============================================================
    // STEP 2: Answer secret question
    // ============================================================
    case 2:
        if (!isset($_SESSION['recovery_username'])) {
            header('Location: ?step=1');
            exit;
        }

        $username = $_SESSION['recovery_username'];
        $user = $users_db[$username];

        render_step_indicator(2);
        echo '<h1>🔑 Pregunta de Seguridad</h1>';
        echo '<p class="subtitle">Responde tu pregunta de seguridad para verificar tu identidad.</p>';

        if (!empty($error_message)) {
            echo '<div class="alert alert-error">' . $error_message . '</div>';
        }

        if (is_locked_out()) {
            $remaining = get_remaining_lockout_time();
            echo '<div class="alert alert-warning">⚠️ Tu cuenta está bloqueada temporalmente. Por favor espera ' . ceil($remaining / 60) . ' minuto(s).</div>';
            echo '<div class="link-group"><a href="?step=1">← Volver al inicio</a></div>';
        } else {
            echo '<div class="alert alert-info">👤 Verificando identidad para: <strong>' . htmlspecialchars($username) . '</strong></div>';

            echo '<div class="question-box">
                <p>❓ ' . htmlspecialchars($user['secret_question']) . '</p>
            </div>';

            echo '<form method="POST" action="">
                <input type="hidden" name="action" value="verify_answer">
                <div class="form-group">
                    <label for="secret_answer">Tu respuesta</label>
                    <input type="text" id="secret_answer" name="secret_answer" placeholder="Escribe tu respuesta aquí..." required autocomplete="off">
                </div>';

            if ($_SESSION['attempts'] > 0 && $_SESSION['attempts'] < MAX_ATTEMPTS) {
                $remaining_attempts = MAX_ATTEMPTS - $_SESSION['attempts'];
                echo '<p class="attempts-info">⚠️ Intentos fallidos: ' . $_SESSION['attempts'] . '/' . MAX_ATTEMPTS . '. Te quedan ' . $remaining_attempts . ' intento(s).</p>';
            }

            echo '<button type="submit">Verificar respuesta →</button>
            </form>';

            echo '<div class="link-group">
                <a href="?step=1">← Usar otro usuario</a>
            </div>';
        }
        break;

    // ============================================================
    // STEP 3: Set new password
    // ============================================================
    case 3:
        if (!isset($_SESSION['recovery_username']) || !isset($_SESSION['reset_token'])) {
            header('Location: ?step=1');
            exit;
        }

        $username = $_SESSION['recovery_username'];
        $token = $_SESSION['reset_token'];

        // Check token expiry
        if (time() > $users_db[$username]['token_expiry']) {
            echo '<div class="alert alert-error">⏰ Tu sesión de recuperación ha expirado. Por favor comienza de nuevo.</div>';
            echo '<div class="link-group"><a href="?step=1">← Comenzar de nuevo</a></div>';
            break;
        }

        $time_remaining = $users_db[$username]['token_expiry'] - time();

        render_step_indicator(3);
        echo '<h1>🔒 Nueva Contraseña</h1>';
        echo '<p class="subtitle">Establece una nueva contraseña segura para tu cuenta.</p>';

        if (!empty($error_message)) {
            echo '<div class="alert alert-error">' . $error_message . '</div>';
        }

        if (!empty($success_message)) {
            echo '<div class="alert alert-success">' . $success_message . '</div>';
        }

        echo '<div class="alert alert-warning">⏱️ Esta sesión expirará en ' . ceil($time_remaining / 60) . ' minuto(s).</div>';

        echo '<form method="POST" action="">
            <input type="hidden" name="action" value="reset_password">
            <input type="hidden" name="token" value="' . htmlspecialchars($token) . '">

                        <li>Al menos un número (0-9)</li>
                    </ul>
                </div>
            </div>
            <div class="form-group">
                <label for="confirm_password">Confirmar contraseña</label>
                <input type="password" id="confirm_password" name="confirm_password" placeholder="Repite tu nueva contraseña" required autocomplete="new-password">
            </div>
            <button type="submit">Restablecer contraseña ✓</button>
        </form>';

        echo '<div class="link-group">
            <a href="?step=1">← Cancelar y volver al inicio</a>
        </div>';
        break;

    // ============================================================
    // STEP 4: Success
    // ============================================================
    case 4:
        render_step_indicator(4);
        echo '<div style="text-align: center; padding: 20px 0;">';
        echo '<div style="font-size: 60px; margin-bottom: 20px;">✅</div>';

        echo '<div class="alert alert-success" style="text-align: left; margin-top: 20px;">
            <strong>¿Qué sigue?</strong><br>
            Ahora puedes iniciar sesión con tu nueva contraseña. Por seguridad, te recomendamos:
            <ul style="margin-top: 8px; margin-left: 20px;">
                <li>No compartir tu contraseña con nadie</li>
                <li>Usar una contraseña única para cada servicio</li>
                <li>Considerar usar un gestor de contraseñas</li>
            </ul>
        </div>';

        echo '<button onclick="window.location.href=\'#\'" style="width: 100%; padding: 13px; background-color: #27ae60; color: white; border: none; border-radius: 6px; font-size: 16px; cursor: pointer; margin-top: 10px;">
            Ir al inicio de sesión →
        </button>';
        echo '</div>';
        break;
}
