class UserPreferences {
    public $theme = 'light';
    public $language = 'es';
    public $fontSize = 14;
    public $itemsPerPage = 10;
    public $notifications = true;

    public function __construct($theme = 'light', $language = 'es', $fontSize = 14, $itemsPerPage = 10, $notifications = true) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
    }

    public function __wakeup() {
        $allowedThemes = ['light', 'dark', 'blue', 'green'];
        $allowedLanguages = ['es', 'en', 'fr', 'de', 'pt'];
        $allowedFontSizes = [10, 12, 14, 16, 18, 20];
        $allowedItemsPerPage = [5, 10, 20, 50, 100];

        if (!in_array($this->theme, $allowedThemes)) {
            $this->theme = 'light';
        }

        if (!in_array($this->language, $allowedLanguages)) {
            $this->language = 'es';
        }

        if (!in_array($this->fontSize, $allowedFontSizes)) {
            $this->fontSize = 14;
        }

        if (!in_array($this->itemsPerPage, $allowedItemsPerPage)) {
            $this->itemsPerPage = 10;
        }

        $this->notifications = (bool) $this->notifications;
    }
}

class PreferencesManager {
    private const COOKIE_NAME = 'user_preferences';
    private const COOKIE_EXPIRY = 30 * 24 * 60 * 60;
    private const ALLOWED_CLASSES = ['UserPreferences'];
    private const HMAC_SECRET = 'mi_clave_secreta_muy_segura_2024';

    public static function savePreferences(UserPreferences $prefs): bool {
        $serialized = serialize($prefs);
        $hmac = hash_hmac('sha256', $serialized, self::HMAC_SECRET);
        $cookieValue = base64_encode($serialized) . '.' . $hmac;

        return setcookie(
            self::COOKIE_NAME,
            $cookieValue,
            [
                'expires' => time() + self::COOKIE_EXPIRY,
                'path' => '/',
                'secure' => isset($_SERVER['HTTPS']),
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }

    public static function loadPreferences(): ?UserPreferences {
        if (!isset($_COOKIE[self::COOKIE_NAME])) {
            return null;
        }

        $cookieValue = $_COOKIE[self::COOKIE_NAME];

        $parts = explode('.', $cookieValue, 2);
        if (count($parts) !== 2) {
            error_log('Cookie de preferencias malformada: falta HMAC');
            return null;
        }

        [$encodedData, $receivedHmac] = $parts;

        $serialized = base64_decode($encodedData, true);
        if ($serialized === false) {
            error_log('Cookie de preferencias malformada: base64 inválido');
            return null;
        }

        $expectedHmac = hash_hmac('sha256', $serialized, self::HMAC_SECRET);
        if (!hash_equals($expectedHmac, $receivedHmac)) {
            error_log('Cookie de preferencias con HMAC inválido - posible manipulación');
            return null;
        }

        $prefs = unserialize($serialized, ['allowed_classes' => self::ALLOWED_CLASSES]);

        if (!($prefs instanceof UserPreferences)) {
            error_log('Cookie de preferencias con clase no permitida');
            return null;
        }

        return $prefs;
    }

    public static function deletePreferences(): void {
        setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires' => time() - 3600,
                'path' => '/',
                'secure' => isset($_SERVER['HTTPS']),
                'httponly' => true,
                'samesite' => 'Strict'
            ]
        );
    }
}

function handleRequest(): void {
    $action = $_POST['action'] ?? $_GET['action'] ?? 'view';

    switch ($action) {
        case 'save':
            handleSave();
            break;
        case 'delete':
            handleDelete();
            break;
        default:
            handleView();
            break;
    }
}

function handleSave(): void {
    $allowedThemes = ['light', 'dark', 'blue', 'green'];
    $allowedLanguages = ['es', 'en', 'fr', 'de', 'pt'];
    $allowedFontSizes = [10, 12, 14, 16, 18, 20];
    $allowedItemsPerPage = [5, 10, 20, 50, 100];

    $theme = $_POST['theme'] ?? 'light';
    $language = $_POST['language'] ?? 'es';
    $fontSize = (int)($_POST['fontSize'] ?? 14);
    $itemsPerPage = (int)($_POST['itemsPerPage'] ?? 10);
    $notifications = isset($_POST['notifications']) && $_POST['notifications'] === '1';

    if (!in_array($theme, $allowedThemes)) $theme = 'light';
    if (!in_array($language, $allowedLanguages)) $language = 'es';
    if (!in_array($fontSize, $allowedFontSizes)) $fontSize = 14;
    if (!in_array($itemsPerPage, $allowedItemsPerPage)) $itemsPerPage = 10;

    $prefs = new UserPreferences($theme, $language, $fontSize, $itemsPerPage, $notifications);
    $saved = PreferencesManager::savePreferences($prefs);

    $message = $saved ? 'Preferencias guardadas correctamente.' : 'Error al guardar las preferencias.';
    $status = $saved ? 'success' : 'error';

    renderPage($prefs, $message, $status);
}

function handleDelete(): void {
    PreferencesManager::deletePreferences();
    $prefs = new UserPreferences();
    renderPage($prefs, 'Preferencias eliminadas. Se han restaurado los valores por defecto.', 'info');
}

function handleView(): void {
    $prefs = PreferencesManager::loadPreferences();
    $message = null;
    $status = null;

    if ($prefs === null) {
        $prefs = new UserPreferences();
        if (isset($_COOKIE['user_preferences'])) {
            $message = 'Las preferencias guardadas no son válidas. Se han restaurado los valores por defecto.';
            $status = 'warning';
        }
    }

    renderPage($prefs, $message, $status);
}
