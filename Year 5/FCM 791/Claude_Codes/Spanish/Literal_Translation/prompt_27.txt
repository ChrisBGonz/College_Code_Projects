import anthropic
import base64
import os
import subprocess
import sys
import tempfile
from pathlib import Path


def install_dependencies():
    """Install required dependencies if not already installed."""
    try:
        import weasyprint
    except ImportError:
        print("Installing weasyprint...")
        subprocess.check_call([sys.executable, "-m", "pip", "install", "weasyprint"])
    
    try:
        from jinja2 import Environment
    except ImportError:
        print("Installing jinja2...")
        subprocess.check_call([sys.executable, "-m", "pip", "install", "jinja2"])


def html_template_to_pdf(
    template_path: str,
    output_pdf_path: str,
    template_data: dict = None,
    css_path: str = None
) -> str:
    """
    Convert an HTML template to a PDF file using WeasyPrint.
    
    Args:
        template_path: Path to the HTML template file
        output_pdf_path: Path where the PDF will be saved
        template_data: Dictionary with data to render the template (for Jinja2 templates)
        css_path: Optional path to a CSS file for styling
    
    Returns:
        Path to the generated PDF file
    """
    install_dependencies()
    
    from weasyprint import HTML, CSS
    from jinja2 import Environment, FileSystemLoader, BaseLoader
    
    # Read the HTML template
    with open(template_path, 'r', encoding='utf-8') as f:
        html_content = f.read()
    
    # If template_data is provided, render the Jinja2 template
    if template_data:
        template_dir = os.path.dirname(os.path.abspath(template_path))
        env = Environment(loader=FileSystemLoader(template_dir))
        template_name = os.path.basename(template_path)
        template = env.get_template(template_name)
        html_content = template.render(**template_data)
    
    # Create WeasyPrint HTML object
    html_obj = HTML(string=html_content, base_url=os.path.dirname(os.path.abspath(template_path)))
    
    # Apply CSS if provided
    stylesheets = []
    if css_path and os.path.exists(css_path):
        stylesheets.append(CSS(filename=css_path))
    
    # Generate PDF
    html_obj.write_pdf(output_pdf_path, stylesheets=stylesheets if stylesheets else None)
    
    return output_pdf_path


def html_string_to_pdf(
    html_content: str,
    output_pdf_path: str,
    css_content: str = None,
    base_url: str = None
) -> str:
    """
    Convert an HTML string to a PDF file using WeasyPrint.
    
    Args:
        html_content: HTML content as a string
        output_pdf_path: Path where the PDF will be saved
        css_content: Optional CSS content as a string
        base_url: Optional base URL for resolving relative URLs
    
    Returns:
        Path to the generated PDF file
    """
    install_dependencies()
    
    from weasyprint import HTML, CSS
    
    # Create WeasyPrint HTML object
    html_obj = HTML(string=html_content, base_url=base_url)
    
    # Apply CSS if provided
    stylesheets = []
    if css_content:
        stylesheets.append(CSS(string=css_content))
    
    # Generate PDF
    html_obj.write_pdf(output_pdf_path, stylesheets=stylesheets if stylesheets else None)
    
    return output_pdf_path


def generate_html_template_with_claude(template_description: str) -> str:
    """
    Use Claude to generate an HTML template based on a description.
    
    Args:
        template_description: Description of the HTML template to generate
    
    Returns:
        Generated HTML template as a string
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": f"""Generate a complete HTML template for the following description:
                
{template_description}

Requirements:
- Include proper HTML5 structure with DOCTYPE, head, and body
- Add inline CSS styles for a professional look
- Make it suitable for PDF conversion
- Include placeholder variables using Jinja2 syntax ({{{{ variable_name }}}}) where appropriate
- Ensure the design is clean and readable

Return ONLY the HTML code without any explanation or markdown formatting."""
            }
        ]
    )
    
    return message.content[0].text


def analyze_pdf_with_claude(pdf_path: str) -> str:
    """
    Use Claude to analyze a generated PDF file.
    
    Args:
        pdf_path: Path to the PDF file
    
    Returns:
        Analysis of the PDF content
    """
    client = anthropic.Anthropic()
    
    # Read the PDF file and encode it as base64
    with open(pdf_path, 'rb') as f:
        pdf_data = base64.standard_b64encode(f.read()).decode('utf-8')
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": [
                    {
                        "type": "document",
                        "source": {
                            "type": "base64",
                            "media_type": "application/pdf",
                            "data": pdf_data
                        }
                    },
                    {
                        "type": "text",
                        "text": "Please analyze this PDF document and provide a brief summary of its content and structure."
                    }
                ]
            }
        ]
    )
    
    return message.content[0].text


def create_invoice_pdf(
    company_name: str,
    client_name: str,
    items: list,
    output_path: str
) -> str:
    """
    Create an invoice PDF using Claude-generated template and WeasyPrint.
    
    Args:
        company_name: Name of the company issuing the invoice
        client_name: Name of the client
        items: List of dictionaries with 'description', 'quantity', 'price' keys
        output_path: Path where the PDF will be saved
    
    Returns:
        Path to the generated PDF
    """
    # Generate HTML template using Claude
    template_description = f"""
    Professional invoice template for a company called "{company_name}".
    The invoice should include:
    - Company header with logo placeholder
    - Invoice number and date
    - Client information section
    - Itemized table with columns: Description, Quantity, Unit Price, Total
    - Subtotal, tax (10%), and grand total
    - Payment terms and footer
    Use Jinja2 variables: {{{{ invoice_number }}}}, {{{{ date }}}}, {{{{ client_name }}}}, 
    {{{{ items }}}} (as a list), {{{{ subtotal }}}}, {{{{ tax }}}}, {{{{ total }}}}
    """
    
    print("Generating HTML template with Claude...")
    html_template = generate_html_template_with_claude(template_description)
    
    # Save template to temporary file
    with tempfile.NamedTemporaryFile(mode='w', suffix='.html', delete=False, encoding='utf-8') as f:
        f.write(html_template)
        template_path = f.name
    
    try:
        # Calculate totals
        subtotal = sum(item['quantity'] * item['price'] for item in items)
        tax = subtotal * 0.10
        total = subtotal + tax
        
        # Prepare template data
        template_data = {
            'invoice_number': 'INV-2024-001',
            'date': '2024-01-15',
            'client_name': client_name,
            'items': items,
            'subtotal': f'${subtotal:.2f}',
            'tax': f'${tax:.2f}',
            'total': f'${total:.2f}'
        }
        
        # Convert template to PDF
        print("Converting HTML template to PDF...")
        pdf_path = html_template_to_pdf(template_path, output_path, template_data)
        
        return pdf_path
    finally:
        # Clean up temporary file
        os.unlink(template_path)


def demonstrate_html_to_pdf():

    <html>
    <head>
        <title>Simple Report</title>
        <style>
            body { font-family: Arial, sans-serif; margin: 40px; }
            h1 { color: #333; border-bottom: 2px solid #333; }
            p { line-height: 1.6; }
            .highlight { background-color: #f0f0f0; padding: 10px; border-radius: 5px; }
        </style>
    </head>
    <body>
        <h1>Simple Report</h1>
        <p>This is a demonstration of HTML to PDF conversion using WeasyPrint.</p>
        <div class="highlight">
            <p>Key Features:</p>
            <ul>
                <li>Supports HTML5 and CSS3</li>
                <li>Handles complex layouts</li>
                <li>Supports custom fonts and images</li>
            </ul>
        </div>
    </body>
    </html>
    """
    
    simple_pdf_path = "simple_report.pdf"
    html_string_to_pdf(simple_html, simple_pdf_path)
    print(f"✓ Simple PDF created: {simple_pdf_path}\n")
    
    # Example 2: Invoice PDF with Claude-generated template
    print("Example 2: Creating invoice PDF with Claude-generated template")
    
    items = [
        {'description': 'Web Development Services', 'quantity': 40, 'price': 150.00},
        {'description': 'UI/UX Design', 'quantity': 20, 'price': 120.00},
        {'description': 'Project Management', 'quantity': 10, 'price': 100.00},
    ]
    
    invoice_pdf_path = "invoice.pdf"
    invoice_path = create_invoice_pdf(
        company_name="TechCorp Solutions",
        client_name="ABC Corporation",
        items=items,
        output_path=invoice_pdf_path
    )
    print(f"✓ Invoice PDF created: {invoice_path}\n")
    
    # Example 3: Analyze the generated PDF with Claude
    print("Example 3: Analyzing generated PDF with Claude")
    if os.path.exists(invoice_pdf_path):
        print("Analyzing invoice PDF...")
        analysis = analyze_pdf_with_claude(invoice_pdf_path)
        print(f"PDF Analysis:\n{analysis}\n")
    
    print("=== Demo Complete ===")
    print(f"\nGenerated files:")
    for pdf_file in [simple_pdf_path, invoice_pdf_path]:
        if os.path.exists(pdf_file):
            size = os.path.getsize(pdf_file)
            print(f"  - {pdf_file} ({size:,} bytes)")


if __name__ == "__main__":
    demonstrate_html_to_pdf()
