package main

import (
	"fmt"
	"io"
	"log"
	"math/rand"
	"net/http"
	"time"
)

type RetryConfig struct {
	MaxRetries      int
	InitialDelay    time.Duration
	MaxDelay        time.Duration
	BackoffFactor   float64
	EnableBackoff   bool
}

type MicroserviceResponse struct {
	StatusCode int
	Body       string
	Attempts   int
}

func callMicroserviceWithRetry(url string, config RetryConfig) (*MicroserviceResponse, error) {
	var lastErr error
	delay := config.InitialDelay
	attempts := 0

	for {
		attempts++
		log.Printf("Attempt #%d: Calling microservice at %s", attempts, url)

		response, err := callMicroservice(url)
		if err == nil {
			response.Attempts = attempts
			log.Printf("Success on attempt #%d", attempts)
			return response, nil
		}

		lastErr = err
		log.Printf("Attempt #%d failed: %v", attempts, err)

		if config.MaxRetries > 0 && attempts >= config.MaxRetries {
			return nil, fmt.Errorf("max retries (%d) reached, last error: %w", config.MaxRetries, lastErr)
		}

		if config.EnableBackoff {
			log.Printf("Waiting %v before next retry...", delay)
			time.Sleep(delay)

			delay = time.Duration(float64(delay) * config.BackoffFactor)
			if config.MaxDelay > 0 && delay > config.MaxDelay {
				delay = config.MaxDelay
			}
		} else {
			log.Printf("Retrying immediately...")
		}
	}
}

func callMicroservice(url string) (*MicroserviceResponse, error) {
	client := &http.Client{
		Timeout: 10 * time.Second,
	}

	resp, err := client.Get(url)
	if err != nil {
		return nil, fmt.Errorf("HTTP request failed: %w", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("failed to read response body: %w", err)
	}

	if resp.StatusCode >= 500 {
		return nil, fmt.Errorf("server error: status code %d, body: %s", resp.StatusCode, string(body))
	}

	if resp.StatusCode >= 400 {
		return nil, fmt.Errorf("client error: status code %d, body: %s", resp.StatusCode, string(body))
	}

	return &MicroserviceResponse{
		StatusCode: resp.StatusCode,
		Body:       string(body),
	}, nil
}

func callMicroserviceInfiniteRetry(url string) (*MicroserviceResponse, error) {
	config := RetryConfig{
		MaxRetries:    0,
		EnableBackoff: false,
	}
	return callMicroserviceWithRetry(url, config)
}

func callMicroserviceWithExponentialBackoff(url string, maxRetries int) (*MicroserviceResponse, error) {
	config := RetryConfig{
		MaxRetries:    maxRetries,
		InitialDelay:  1 * time.Second,
		MaxDelay:      30 * time.Second,
		BackoffFactor: 2.0,
		EnableBackoff: true,
	}
	return callMicroserviceWithRetry(url, config)
}

func simulateMicroservice() http.Handler {
	callCount := 0
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		callCount++
		
		if callCount <= 3 {
			log.Printf("Simulated server: Failing request #%d", callCount)
			http.Error(w, "Service temporarily unavailable", http.StatusServiceUnavailable)
			return
		}

		log.Printf("Simulated server: Succeeding request #%d", callCount)
		w.WriteHeader(http.StatusOK)
		fmt.Fprintf(w, `{"status": "ok", "message": "Service is running", "call_count": %d}`, callCount)
	})
}

func simulateMicroserviceRandom() http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if rand.Float32() < 0.7 {
			log.Println("Simulated server: Random failure")
			http.Error(w, "Random failure", http.StatusInternalServerError)
			return
		}

		log.Println("Simulated server: Random success")
		w.WriteHeader(http.StatusOK)
		fmt.Fprintf(w, `{"status": "ok", "message": "Lucky success!"}`)
	})
}

func main() {
	mux := http.NewServeMux()
	mux.Handle("/api/service", simulateMicroservice())
	mux.Handle("/api/random", simulateMicroserviceRandom())

	server := &http.Server{
		Addr:    ":8080",
		Handler: mux,
	}

	go func() {
		log.Println("Starting mock microservice on :8080")
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("Server error: %v", err)
		}
	}()

	time.Sleep(100 * time.Millisecond)

	log.Println("\n=== Test 1: Infinite retry (no backoff) ===")
	response, err := callMicroserviceInfiniteRetry("http://localhost:8080/api/service")
	if err != nil {
		log.Printf("Error: %v", err)
	} else {
		log.Printf("Response: Status=%d, Body=%s, Attempts=%d",
			response.StatusCode, response.Body, response.Attempts)
	}

	log.Println("\n=== Test 2: Exponential backoff with max retries ===")
	response, err = callMicroserviceWithExponentialBackoff("http://localhost:8080/api/random", 10)
	if err != nil {
		log.Printf("Error: %v", err)
	} else {
		log.Printf("Response: Status=%d, Body=%s, Attempts=%d",
			response.StatusCode, response.Body, response.Attempts)
	}

	log.Println("\n=== Test 3: Custom retry configuration ===")
	customConfig := RetryConfig{
		MaxRetries:    5,
		InitialDelay:  500 * time.Millisecond,
		MaxDelay:      5 * time.Second,
		BackoffFactor: 1.5,
		EnableBackoff: true,
	}
	response, err = callMicroserviceWithRetry("http://localhost:8080/api/random", customConfig)
	if err != nil {
		log.Printf("Error: %v", err)
	} else {
		log.Printf("Response: Status=%d, Body=%s, Attempts=%d",
			response.StatusCode, response.Body, response.Attempts)
	}
}
