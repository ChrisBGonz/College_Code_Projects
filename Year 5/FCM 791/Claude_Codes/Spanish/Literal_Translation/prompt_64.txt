using System;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RememberMeSessionManager
{
    private readonly AnthropicClient _client;
    private const string MODEL = "claude-opus-4-5";
    
    public RememberMeSessionManager()
    {
        _client = new AnthropicClient();
    }
    
    public async Task<string> GenerateRememberMeImplementation(string userRequest)
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = userRequest
            }
        };
        
        var parameters = new MessageParameters
        {
            Model = MODEL,
            MaxTokens = 4096,
            Messages = messages,
            System = new List<SystemMessage>
            {
                new SystemMessage
                {
                    Text = @"You are an expert C# developer specializing in ASP.NET Core security and authentication. 
                    When asked to implement 'Remember Me' functionality, provide complete, production-ready C# code with:
                    1. Secure cookie generation with proper encryption
                    2. Token storage and validation
                    3. Proper expiration handling
                    4. Security best practices (HttpOnly, Secure flags)
                    5. Complete implementation with all necessary classes and methods
                    Always provide working code that can be directly integrated into an ASP.NET Core application."
                }
            }
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        return response.Content[0].ToString() ?? string.Empty;
    }
    
    public async Task<string> ExplainSecurityConsiderations()
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = "Explain the security considerations and best practices for implementing 'Remember Me' persistent cookies in ASP.NET Core applications."
            }
        };
        
        var parameters = new MessageParameters
        {
            Model = MODEL,
            MaxTokens = 2048,
            Messages = messages
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        return response.Content[0].ToString() ?? string.Empty;
    }
    
    public async Task<string> GenerateTokenManagement()
    {
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = @"Generate a complete C# implementation for secure token management in a 'Remember Me' system including:
                1. Token generation using cryptographically secure random bytes
                2. Token hashing for database storage
                3. Token validation and rotation
                4. Cleanup of expired tokens
                Please provide the complete implementation with all necessary interfaces and classes."
            }
        };
        
        var parameters = new MessageParameters
        {
            Model = MODEL,
            MaxTokens = 4096,
            Messages = messages
        };
        
        var response = await _client.Messages.GetClaudeMessageAsync(parameters);
        return response.Content[0].ToString() ?? string.Empty;
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("=== Remember Me Session Manager with Claude AI ===\n");
        
        var manager = new RememberMeSessionManager();
        
        // Generate main Remember Me implementation
        Console.WriteLine("Generating 'Remember Me' Implementation...\n");
        Console.WriteLine("----------------------------------------");
        
        string implementation = await manager.GenerateRememberMeImplementation(
            @"Create a complete C# implementation for 'Remember Me' functionality in ASP.NET Core with:
            1. A RememberMeService class with methods for creating and validating persistent cookies
            2. Cookie configuration with proper security settings (HttpOnly, Secure, SameSite)
            3. A 30-day expiration for the persistent cookie
            4. Token-based authentication to prevent cookie theft
            5. Database model for storing remember me tokens
            6. Middleware integration example
            Include all necessary using statements and make it production-ready."
        );
        
        Console.WriteLine("REMEMBER ME IMPLEMENTATION:");
        Console.WriteLine(implementation);
        Console.WriteLine("\n----------------------------------------\n");
        
        // Get security considerations
        Console.WriteLine("Fetching Security Considerations...\n");
        Console.WriteLine("----------------------------------------");
        
        string securityInfo = await manager.ExplainSecurityConsiderations();
        
        Console.WriteLine("SECURITY CONSIDERATIONS:");
        Console.WriteLine(securityInfo);
        Console.WriteLine("\n----------------------------------------\n");
        
        // Generate token management
        Console.WriteLine("Generating Token Management System...\n");
        Console.WriteLine("----------------------------------------");
        
        string tokenManagement = await manager.GenerateTokenManagement();
        
        Console.WriteLine("TOKEN MANAGEMENT SYSTEM:");
        Console.WriteLine(tokenManagement);
        Console.WriteLine("\n----------------------------------------\n");
        
        Console.WriteLine("=== Implementation Complete ===");
        Console.WriteLine("\nThe above code provides a complete 'Remember Me' implementation including:");
        Console.WriteLine("✓ Secure cookie generation and management");
        Console.WriteLine("✓ Token-based authentication");
        Console.WriteLine("✓ Security best practices");
        Console.WriteLine("✓ Token rotation and cleanup");
        Console.WriteLine("✓ ASP.NET Core integration");
    }
}
