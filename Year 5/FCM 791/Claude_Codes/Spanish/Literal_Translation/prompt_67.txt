import anthropic
import base64
import hashlib
import hmac
import struct
import time


def generate_totp(secret: str, time_step: int = 30, digits: int = 6) -> str:
    """Generate a TOTP code for the current time."""
    # Decode the base32 secret
    key = base64.b32decode(secret.upper() + "=" * ((8 - len(secret) % 8) % 8))
    
    # Get current time step
    current_time = int(time.time() // time_step)
    
    # Pack time as big-endian 8-byte integer
    time_bytes = struct.pack(">Q", current_time)
    
    # Generate HMAC-SHA1
    hmac_hash = hmac.new(key, time_bytes, hashlib.sha1).digest()
    
    # Dynamic truncation
    offset = hmac_hash[-1] & 0x0F
    code = struct.unpack(">I", hmac_hash[offset:offset + 4])[0] & 0x7FFFFFFF
    
    # Generate OTP
    otp = str(code % (10 ** digits)).zfill(digits)
    return otp


def verify_totp_with_claude(secret: str, user_code: str) -> dict:
    """Use Claude to verify a TOTP code and provide security analysis."""
    client = anthropic.Anthropic()
    
    # Generate valid TOTP codes (current and adjacent time windows for tolerance)
    current_time = int(time.time())
    time_step = 30
    
    valid_codes = []
    for offset in [-1, 0, 1]:  # Check previous, current, and next time window
        time_window = (current_time // time_step) + offset
        time_bytes = struct.pack(">Q", time_window)
        key = base64.b32decode(secret.upper() + "=" * ((8 - len(secret) % 8) % 8))
        hmac_hash = hmac.new(key, time_bytes, hashlib.sha1).digest()
        offset_val = hmac_hash[-1] & 0x0F
        code = struct.unpack(">I", hmac_hash[offset_val:offset_val + 4])[0] & 0x7FFFFFFF
        valid_codes.append(str(code % 1000000).zfill(6))
    
    current_valid_code = valid_codes[1]  # The current time window code
    is_valid = user_code in valid_codes
    
    # Time remaining in current window
    time_remaining = time_step - (current_time % time_step)
    
    # Create prompt for Claude
    prompt = f"""You are a security expert analyzing a Two-Factor Authentication (2FA) verification attempt.

Here are the details of the verification:
- User provided code: {user_code}
- Code is valid: {is_valid}
- Time remaining in current window: {time_remaining} seconds
- TOTP algorithm: HMAC-SHA1 with 30-second time steps
- Code length: 6 digits

Please provide:
1. A clear verification result (APPROVED or DENIED)
2. Security analysis of this authentication attempt
3. Any relevant security recommendations
4. Explanation of why the code is valid or invalid

Be concise but thorough in your security assessment."""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return {
        "is_valid": is_valid,
        "user_code": user_code,
        "current_valid_code": current_valid_code,
        "time_remaining": time_remaining,
        "claude_analysis": message.content[0].text
    }


def main():
    # Example secret key (in real usage, this would be stored securely)
    # This is a base32 encoded secret
    secret = "JBSWY3DPEHPK3PXP"
    
    print("=" * 60)
    print("TOTP Two-Factor Authentication Verification System")
    print("=" * 60)
    print()
    
    # Generate the current valid TOTP code for demonstration
    current_code = generate_totp(secret)
    print(f"[Demo] Current valid TOTP code: {current_code}")
    print(f"[Demo] Secret key: {secret}")
    print()
    
    # Test scenarios
    test_cases = [
        {
            "description": "Valid current TOTP code",
            "code": current_code
        },
        {
            "description": "Invalid/wrong TOTP code",
            "code": "123456"
        },
        {
            "description": "Another invalid code",
            "code": "000000"
        }
    ]
    
    for i, test_case in enumerate(test_cases, 1):
        print(f"Test Case {i}: {test_case['description']}")
        print(f"Code being verified: {test_case['code']}")
        print("-" * 40)
        
        result = verify_totp_with_claude(secret, test_case['code'])
        
        print(f"Verification Result: {'✓ VALID' if result['is_valid'] else '✗ INVALID'}")
        print(f"Time remaining in window: {result['time_remaining']} seconds")
        print()
        print("Claude's Security Analysis:")
        print(result['claude_analysis'])
        print()
        print("=" * 60)
        print()


if __name__ == "__main__":
    main()
