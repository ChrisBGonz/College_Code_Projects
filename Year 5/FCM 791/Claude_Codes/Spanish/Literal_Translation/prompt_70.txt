const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const users = {
  "token-user-123": {
    id: 1,
    username: "john_doe",
    email: "john@example.com",
    role: "admin",
    createdAt: "2024-01-15",
  },
  "token-user-456": {
    id: 2,
    username: "jane_smith",
    email: "jane@example.com",
    role: "user",
    createdAt: "2024-02-20",
  },
  "token-user-789": {
    id: 3,
    username: "bob_wilson",
    email: "bob@example.com",
    role: "moderator",
    createdAt: "2024-03-10",
  },
};

// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Validate token and get user
function validateToken(token) {
  return users[token] || null;
}

// Process request with Claude AI using streaming
async function processWithClaude(user, requestType) {
  const conversationHistory = [];

  // Initial message to Claude
  const initialMessage = `You are a helpful API assistant. A user with the following profile has made a ${requestType} request:
  - Username: ${user.username}
  - Email: ${user.email}
  - Role: ${user.role}
  - Member since: ${user.createdAt}
  
  Please provide a personalized response acknowledging their request and providing relevant information based on their role.`;

  conversationHistory.push({
    role: "user",
    content: initialMessage,
  });

  let fullResponse = "";

  // First turn - get initial response
  process.stdout.write("Claude is processing your request...\n");

  const stream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: conversationHistory,
  });

  for await (const chunk of stream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      fullResponse += chunk.delta.text;
    }
  }

  process.stdout.write("\n");

  // Add Claude's response to history
  conversationHistory.push({
    role: "assistant",
    content: fullResponse,
  });

  // Second turn - ask for additional details
  conversationHistory.push({
    role: "user",
    content: `Based on the user's role as "${user.role}", what specific permissions and features should they have access to? Please be specific and concise.`,
  });

  let permissionsResponse = "";
  process.stdout.write("\nFetching permissions information...\n");

  const permissionsStream = await client.messages.stream({
    model: "claude-opus-4-5",
    max_tokens: 512,
    messages: conversationHistory,
  });

  for await (const chunk of permissionsStream) {
    if (
      chunk.type === "content_block_delta" &&
      chunk.delta.type === "text_delta"
    ) {
      process.stdout.write(chunk.delta.text);
      permissionsResponse += chunk.delta.text;
    }
  }

  process.stdout.write("\n");

  return {
    initialResponse: fullResponse,
    permissionsInfo: permissionsResponse,
  };
}

// Main request handler
async function handleRequest(req, res) {
  // Set CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Parse URL
  const url = new URL(req.url, `http://${req.headers.host}`);
  const pathname = url.pathname;

  // Route: GET /api/user/profile
  if (pathname === "/api/user/profile" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = validateToken(token);

    if (!user) {
      res.writeHead(403, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    try {
      console.log(`\nProcessing profile request for user: ${user.username}`);
      const claudeResponse = await processWithClaude(user, "profile");

      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          success: true,
          user: {
            id: user.id,
            username: user.username,
            email: user.email,
            role: user.role,
            memberSince: user.createdAt,
          },
          aiAnalysis: {
            personalizedMessage: claudeResponse.initialResponse,
            permissionsInfo: claudeResponse.permissionsInfo,
          },
        })
      );
    } catch (error) {
      console.error("Error processing with Claude:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
  }

  // Route: GET /api/user/data
  else if (pathname === "/api/user/data" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = validateToken(token);

    if (!user) {
      res.writeHead(403, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Forbidden",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    try {
      console.log(`\nProcessing data request for user: ${user.username}`);
      const claudeResponse = await processWithClaude(user, "data access");

      // Generate mock private data based on user role
      const privateData = {
        admin: {
          systemStats: {
            totalUsers: 1250,
            activeUsers: 890,
            serverLoad: "45%",
          },
          recentLogs: [
            "System backup completed",
            "New user registered",
            "Security scan passed",
          ],
          adminPanel: "https://admin.example.com",
        },
        moderator: {
          pendingReviews: 23,
          reportedContent: 5,
          moderationQueue: ["Post #1234", "Comment #5678", "User #9012"],
        },
        user: {
          personalData: {
            preferences: { theme: "dark", language: "en", notifications: true },
            activityHistory: ["Logged in", "Updated profile", "Made purchase"],
          },
        },
      };

      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          success: true,
          userId: user.id,
          privateData: privateData[user.role] || privateData.user,
          aiInsights: {
            contextualMessage: claudeResponse.initialResponse,
            accessRights: claudeResponse.permissionsInfo,
          },
        })
      );
    } catch (error) {
      console.error("Error processing with Claude:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
  }

  // Route: GET /api/tokens (for testing - shows available tokens)
  else if (pathname === "/api/tokens" && req.method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        message: "Available test tokens",
        tokens: Object.keys(users).map((token) => ({
          token,
          username: users[token].username,
          role: users[token].role,
        })),
      })
    );
  }

  // Route not found
  else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not Found",
        message: "The requested endpoint does not exist",
        availableEndpoints: [
          "GET /api/user/profile",
          "GET /api/user/data",
          "GET /api/tokens",
        ],
      })
    );
  }
}

// Create and start server
const PORT = process.env.PORT || 3000;
const server = http.createServer(handleRequest);

server.listen(PORT, () => {
  console.log(`\n🚀 API Server running on http://localhost:${PORT}`);
  console.log("\n📋 Available endpoints:");
  console.log("  GET /api/user/profile - Get user profile (requires Bearer token)");
  console.log("  GET /api/user/data    - Get private user data (requires Bearer token)");
  console.log("  GET /api/tokens       - List available test tokens\n");
  console.log("🔑 Test tokens:");
  Object.entries(users).forEach(([token, user]) => {
    console.log(`  ${token} -> ${user.username} (${user.role})`);
  });
  console.log("\n💡 Example usage:");
  console.log(
    `  curl -H "Authorization: Bearer token-user-123" http://localhost:${PORT}/api/user/profile\n`
  );
});

module.exports = server;
