const Anthropic = require("@anthropic-ai/sdk");
const { execSync } = require("child_process");
const os = require("os");

const client = new Anthropic();

// Define the ping tool
const tools = [
  {
    name: "ping_host",
    description:
      "Executes a ping command to check if a host is alive and returns the result",
    input_schema: {
      type: "object",
      properties: {
        domain_name: {
          type: "string",
          description: "The domain name or IP address to ping",
        },
        count: {
          type: "number",
          description: "Number of ping packets to send (default: 4)",
        },
      },
      required: ["domain_name"],
    },
  },
];

// Function to execute ping command
function pingHost(domainName, count = 4) {
  try {
    const platform = os.platform();
    let command;

    // Different ping syntax for different operating systems
    if (platform === "win32") {
      command = `ping -n ${count} ${domainName}`;
    } else {
      // Linux and macOS
      command = `ping -c ${count} ${domainName}`;
    }

    console.log(`\nExecuting: ${command}`);
    const output = execSync(command, {
      encoding: "utf8",
      timeout: 30000, // 30 second timeout
    });

    return {
      success: true,
      domain: domainName,
      output: output,
      isAlive: true,
    };
  } catch (error) {
    // If ping fails (host unreachable), execSync throws an error
    return {
      success: false,
      domain: domainName,
      output: error.stdout || error.message,
      isAlive: false,
      error: error.message,
    };
  }
}

// Process tool calls from Claude
function processToolCall(toolName, toolInput) {
  if (toolName === "ping_host") {
    const { domain_name, count } = toolInput;
    return pingHost(domain_name, count || 4);
  }
  return { error: "Unknown tool" };
}

// Main function to check if a host is alive using Claude
async function checkHostWithClaude(domainName) {
  console.log(`\nChecking if host "${domainName}" is alive...`);

  const messages = [
    {
      role: "user",
      content: `Please check if the host "${domainName}" is alive by using the ping_host tool. After getting the results, provide a clear summary of whether the host is reachable and any relevant information from the ping output.`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    tools: tools,
    messages: messages,
  });

  console.log("\nInitial response from Claude:");
  console.log(`Stop reason: ${response.stop_reason}`);

  // Agentic loop - continue until Claude stops using tools
  while (response.stop_reason === "tool_use") {
    const toolUseBlock = response.content.find(
      (block) => block.type === "tool_use"
    );

    if (!toolUseBlock) break;

    const toolName = toolUseBlock.name;
    const toolInput = toolUseBlock.input;

    console.log(`\nClaude is using tool: ${toolName}`);
    console.log("Tool input:", JSON.stringify(toolInput, null, 2));

    // Execute the tool
    const toolResult = processToolCall(toolName, toolInput);

    console.log("\nTool result:");
    console.log(JSON.stringify(toolResult, null, 2));

    // Add assistant's response and tool result to messages
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: [
        {
          type: "tool_result",
          tool_use_id: toolUseBlock.id,
          content: JSON.stringify(toolResult),
        },
      ],
    });

    // Get next response from Claude
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log(`\nNext response stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  return finalResponse;
}

// Main execution
async function main() {
  // Test with different domains
  const domainsToTest = ["google.com", "github.com", "nonexistent-domain-12345.com"];

  for (const domain of domainsToTest) {
    console.log("\n" + "=".repeat(60));
    try {
      const result = await checkHostWithClaude(domain);
      console.log("\nClaude's Analysis:");
      console.log(result);
    } catch (error) {
      console.error(`Error checking ${domain}:`, error.message);
    }
    console.log("=".repeat(60));
  }
}

main().catch(console.error);
