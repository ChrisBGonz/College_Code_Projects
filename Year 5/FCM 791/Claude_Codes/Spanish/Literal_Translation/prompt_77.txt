using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Collections.Generic;

[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; }
    public string Version { get; set; }
    public string Theme { get; set; }
    public int MaxConnections { get; set; }
    public bool EnableLogging { get; set; }
    public string DatabaseConnectionString { get; set; }
    public Dictionary<string, string> CustomSettings { get; set; }
    public DateTime LastModified { get; set; }

    public AppSettings()
    {
        CustomSettings = new Dictionary<string, string>();
        LastModified = DateTime.Now;
    }

    public override string ToString()
    {
        return $"Application: {ApplicationName} v{Version}\n" +
               $"Theme: {Theme}\n" +
               $"Max Connections: {MaxConnections}\n" +
               $"Logging Enabled: {EnableLogging}\n" +
               $"Database: {DatabaseConnectionString}\n" +
               $"Last Modified: {LastModified}\n" +
               $"Custom Settings Count: {CustomSettings?.Count ?? 0}";
    }
}

public class AppSettingsManager
{
    private readonly string _settingsFilePath;

    public AppSettingsManager(string filePath = "appsettings.bin")
    {
        _settingsFilePath = filePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        if (settings == null)
            throw new ArgumentNullException(nameof(settings));

        settings.LastModified = DateTime.Now;

        try
        {
            using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write))
            {
#pragma warning disable SYSLIB0011
                BinaryFormatter formatter = new BinaryFormatter();
                formatter.Serialize(fileStream, settings);
#pragma warning restore SYSLIB0011
            }
            Console.WriteLine($"Settings saved successfully to: {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            throw new InvalidOperationException($"Failed to save settings: {ex.Message}", ex);
        }
    }

    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("Settings file not found. Creating default settings.");
            return CreateDefaultSettings();
        }

        try
        {
            using (FileStream fileStream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read))
            {
#pragma warning disable SYSLIB0011
                BinaryFormatter formatter = new BinaryFormatter();
                AppSettings settings = (AppSettings)formatter.Deserialize(fileStream);
#pragma warning restore SYSLIB0011
                Console.WriteLine($"Settings loaded successfully from: {_settingsFilePath}");
                return settings;
            }
        }
        catch (Exception ex)
        {
            throw new InvalidOperationException($"Failed to load settings: {ex.Message}", ex);
        }
    }

    public void UpdateSetting(string key, string value)
    {
        AppSettings settings = LoadSettings();
        
        switch (key.ToLower())
        {
            case "applicationname":
                settings.ApplicationName = value;
                break;
            case "version":
                settings.Version = value;
                break;
            case "theme":
                settings.Theme = value;
                break;
            case "maxconnections":
                if (int.TryParse(value, out int maxConn))
                    settings.MaxConnections = maxConn;
                break;
            case "enablelogging":
                if (bool.TryParse(value, out bool logging))
                    settings.EnableLogging = logging;
                break;
            case "databaseconnectionstring":
                settings.DatabaseConnectionString = value;
                break;
            default:
                settings.CustomSettings[key] = value;
                break;
        }

        SaveSettings(settings);
        Console.WriteLine($"Setting '{key}' updated to '{value}'");
    }

    public bool DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine("Settings file deleted successfully.");
            return true;
        }
        Console.WriteLine("Settings file does not exist.");
        return false;
    }

    public bool SettingsExist()
    {
        return File.Exists(_settingsFilePath);
    }

    private AppSettings CreateDefaultSettings()
    {
        return new AppSettings
        {
            ApplicationName = "MyApplication",
            Version = "1.0.0",
            Theme = "Light",
            MaxConnections = 10,
            EnableLogging = true,
            DatabaseConnectionString = "Server=localhost;Database=MyDB;",
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "en-US" },
                { "AutoSave", "true" },
                { "BackupInterval", "30" }
            }
        };
    }
}

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("=== Application Settings Manager Demo ===\n");

        AppSettingsManager manager = new AppSettingsManager("demo_settings.bin");

        Console.WriteLine("1. Creating and saving initial settings...");
        AppSettings initialSettings = new AppSettings
        {
            ApplicationName = "DemoApp",
            Version = "2.1.0",
            Theme = "Dark",
            MaxConnections = 25,
            EnableLogging = true,
            DatabaseConnectionString = "Server=prod-server;Database=ProductionDB;User=admin;",
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "es-ES" },
                { "AutoSave", "true" },
                { "BackupInterval", "60" },
                { "MaxRetries", "3" }
            }
        };

        manager.SaveSettings(initialSettings);
        Console.WriteLine("\nInitial Settings:");
        Console.WriteLine(initialSettings.ToString());

        Console.WriteLine("\n2. Loading settings from file...");
        AppSettings loadedSettings = manager.LoadSettings();
        Console.WriteLine("\nLoaded Settings:");
        Console.WriteLine(loadedSettings.ToString());

        Console.WriteLine("\n3. Updating specific settings...");
        manager.UpdateSetting("Theme", "Light");
        manager.UpdateSetting("MaxConnections", "50");
        manager.UpdateSetting("EnableLogging", "false");
        manager.UpdateSetting("NewCustomKey", "CustomValue123");

        Console.WriteLine("\n4. Loading updated settings...");
        AppSettings updatedSettings = manager.LoadSettings();
        Console.WriteLine("\nUpdated Settings:");
        Console.WriteLine(updatedSettings.ToString());

        Console.WriteLine("\n5. Custom Settings:");
        foreach (var kvp in updatedSettings.CustomSettings)
        {
            Console.WriteLine($"   {kvp.Key}: {kvp.Value}");
        }

        Console.WriteLine("\n6. Verifying settings persistence...");
        AppSettingsManager manager2 = new AppSettingsManager("demo_settings.bin");
        AppSettings verifiedSettings = manager2.LoadSettings();
        Console.WriteLine($"Settings file exists: {manager2.SettingsExist()}");
        Console.WriteLine($"Application Name: {verifiedSettings.ApplicationName}");
        Console.WriteLine($"Theme: {verifiedSettings.Theme}");
        Console.WriteLine($"Max Connections: {verifiedSettings.MaxConnections}");

        Console.WriteLine("\n7. Cleaning up - Deleting settings file...");
        manager.DeleteSettings();
        Console.WriteLine($"Settings file exists after deletion: {manager.SettingsExist()}");

        Console.WriteLine("\n8. Loading settings after deletion (should create defaults)...");
        AppSettings defaultSettings = manager.LoadSettings();
        Console.WriteLine("\nDefault Settings:");
        Console.WriteLine(defaultSettings.ToString());

        Console.WriteLine("\n=== Demo Complete ===");
    }
}
