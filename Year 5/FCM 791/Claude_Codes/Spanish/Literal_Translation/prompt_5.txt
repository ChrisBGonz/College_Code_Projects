import org.springframework.ai.chat.client.ChatClient;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/messages")
public class PrivateMessageController {

    private final ChatClient chatClient;

    // Simulated database of private messages
    private static final java.util.Map<String, java.util.Map<String, String>> messageDatabase = new java.util.HashMap<>();

    static {
        // Initialize with some sample messages
        java.util.Map<String, String> user1Messages = new java.util.HashMap<>();
        user1Messages.put("msg001", "Hello John, can we meet tomorrow at 3pm?");
        user1Messages.put("msg002", "I wanted to discuss the project proposal with you.");
        user1Messages.put("msg003", "Please review the attached document when you get a chance.");
        messageDatabase.put("user1", user1Messages);

        java.util.Map<String, String> user2Messages = new java.util.HashMap<>();
        user2Messages.put("msg004", "Hi Sarah, the meeting has been rescheduled to Friday.");
        user2Messages.put("msg005", "Thank you for your help with the presentation.");
        messageDatabase.put("user2", user2Messages);

        java.util.Map<String, String> user3Messages = new java.util.HashMap<>();
        user3Messages.put("msg006", "Reminder: Team lunch is at noon today.");
        user3Messages.put("msg007", "Please submit your weekly report by end of day.");
        user3Messages.put("msg008", "The client meeting went well, great job everyone!");
        messageDatabase.put("user3", user3Messages);
    }

    public PrivateMessageController(ChatClient.Builder chatClientBuilder) {
        this.chatClient = chatClientBuilder.build();
    }

    @GetMapping("/{messageId}")
    public MessageResponse getPrivateMessage(@PathVariable String messageId) {
        // Find the message in our simulated database
        String messageContent = null;
        String userId = null;

        for (java.util.Map.Entry<String, java.util.Map<String, String>> userEntry : messageDatabase.entrySet()) {
            if (userEntry.getValue().containsKey(messageId)) {
                messageContent = userEntry.getValue().get(messageId);
                userId = userEntry.getKey();
                break;
            }
        }

        if (messageContent == null) {
            return new MessageResponse(messageId, null, null, 
                "Message not found", 
                "No message found with ID: " + messageId);
        }

        // Use Spring AI to analyze and provide context about the message
        String prompt = String.format(
            "Analyze this private message and provide a brief summary and context. " +
            "Message ID: %s, From User: %s, Content: '%s'. " +
            "Provide a JSON-like response with: summary (brief description), " +
            "sentiment (positive/negative/neutral), and priority (high/medium/low).",
            messageId, userId, messageContent
        );

        String aiAnalysis = chatClient.prompt()
            .user(prompt)
            .call()
            .content();

        return new MessageResponse(messageId, userId, messageContent, "Message retrieved successfully", aiAnalysis);
    }

    @GetMapping("/user/{userId}")
    public UserMessagesResponse getUserMessages(@PathVariable String userId) {
        java.util.Map<String, String> userMessages = messageDatabase.get(userId);

        if (userMessages == null || userMessages.isEmpty()) {
            return new UserMessagesResponse(userId, null, 
                "No messages found for user: " + userId, null);
        }

        // Use Spring AI to summarize all messages for the user
        String allMessages = userMessages.entrySet().stream()
            .map(e -> "ID: " + e.getKey() + " - " + e.getValue())
            .reduce("", (a, b) -> a + "\n" + b);

        String prompt = String.format(
            "Summarize these private messages for user %s and identify key themes or action items:\n%s",
            userId, allMessages
        );

        String aiSummary = chatClient.prompt()
            .user(prompt)
            .call()
            .content();

        return new UserMessagesResponse(userId, userMessages, 
            "Messages retrieved successfully", aiSummary);
    }

    // Response DTOs
    record MessageResponse(
        String messageId,
        String userId,
        String content,
        String status,
        String aiAnalysis
    ) {}

    record UserMessagesResponse(
        String userId,
        java.util.Map<String, String> messages,
        String status,
        String aiSummary
    ) {}
}
