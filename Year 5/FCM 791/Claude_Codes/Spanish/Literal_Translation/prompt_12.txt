
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";

const client = new Anthropic();

async function generateExpressServerConfig(): Promise<void> {
  console.log(
    "Generating Express.js server configuration for production environment...\n"
  );

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: `Create a complete, production-ready Express.js server configuration with the following requirements:

1. Basic Express.js setup with TypeScript support
2. Security middleware (helmet, cors, rate limiting)
3. Request parsing middleware (body-parser, cookie-parser)
4. Logging middleware (morgan)
5. Compression middleware
6. Error handling middleware (both 404 and general error handlers)
7. Health check endpoint
8. Environment variable configuration
9. Graceful shutdown handling
10. Request validation middleware example

Please provide:
- The main server file (server.ts or server.js)
- A middleware configuration file
- An error handling module
- Environment configuration
- Package.json dependencies

Make the code production-ready with proper TypeScript types, error handling, and best practices.`,
      },
    ],
  });

  // Extract the text content from the response
  const responseText = message.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("=== Express.js Production Server Configuration ===\n");
  console.log(responseText);

  // Save the configuration to a file
  const outputFile = "express-server-config.md";
  fs.writeFileSync(outputFile, responseText);
  console.log(`\n✅ Configuration saved to ${outputFile}`);

  // Display usage statistics
  console.log("\n=== API Usage Statistics ===");
  console.log(`Input tokens: ${message.usage.input_tokens}`);
  console.log(`Output tokens: ${message.usage.output_tokens}`);
  console.log(`Stop reason: ${message.stop_reason}`);
}

// Run the main function
generateExpressServerConfig().catch((error) => {
  console.error("Error generating Express.js configuration:", error);
  process.exit(1);
});
```
