
```python
import anthropic
import json
import time
import random

client = anthropic.Anthropic()

def simulate_payment_api(amount: float, currency: str, card_number: str) -> dict:
    """Simulates an external payment API that might timeout."""
    time.sleep(random.uniform(0.5, 3.5))
    
    return {
        "status": "success",
        "transaction_id": f"TXN_{random.randint(10000, 99999)}",
        "amount": amount,
        "currency": currency,
        "timestamp": time.time()
    }

tools = [
    {
        "name": "process_payment",
        "description": "Processes a payment through an external payment API. Returns payment result or timeout status.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., USD, EUR)"
                },
                "card_number": {
                    "type": "string",
                    "description": "The card number for payment"
                },
                "timeout_seconds": {
                    "type": "number",
                    "description": "Timeout in seconds for the API call",
                    "default": 2
                }
            },
            "required": ["amount", "currency", "card_number"]
        }
    }
]

def process_payment_with_timeout(amount: float, currency: str, card_number: str, timeout_seconds: float = 2) -> bool:
    """
    Connects to external payment API. Returns True if API doesn't respond within timeout.
    
    Args:
        amount: Payment amount
        currency: Currency code
        card_number: Card number for payment
        timeout_seconds: Timeout in seconds (default: 2)
    
    Returns:
        True if timeout occurred (allow user to continue), False if payment processed
    """
    
    messages = [
        {
            "role": "user",
            "content": f"Process a payment of {amount} {currency} for card ending in {card_number[-4:]}. Use a timeout of {timeout_seconds} seconds."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    if response.stop_reason == "tool_use":
        tool_use_block = next(block for block in response.content if block.type == "tool_use")
        tool_input = tool_use_block.input
        
        start_time = time.time()
        timeout_occurred = False
        payment_result = None
        
        import threading
        
        def api_call():
            nonlocal payment_result
            payment_result = simulate_payment_api(
                tool_input["amount"],
                tool_input["currency"],
                tool_input["card_number"]
            )
        
        api_thread = threading.Thread(target=api_call)
        api_thread.daemon = True
        api_thread.start()
        
        api_thread.join(timeout=timeout_seconds)
        
        elapsed_time = time.time() - start_time
        
        if api_thread.is_alive():
            timeout_occurred = True
            tool_result = {
                "status": "timeout",
                "message": f"Payment API did not respond within {timeout_seconds} seconds",
                "elapsed_time": elapsed_time,
                "allow_continue": True
            }
        else:
            tool_result = payment_result
            tool_result["elapsed_time"] = elapsed_time
            tool_result["allow_continue"] = False
        
        messages.append({"role": "assistant", "content": response.content})
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": json.dumps(tool_result)
                }
            ]
        })
        
        final_response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nPayment Processing Result:")
        print(f"Timeout occurred: {timeout_occurred}")
        print(f"Elapsed time: {elapsed_time:.2f} seconds")
        
        if timeout_occurred:
            print("API did not respond in time. User is allowed to continue.")
        else:
            print(f"Payment processed successfully: {payment_result}")
        
        print(f"\nClaude's analysis: {final_response.content[0].text}")
        
        return timeout_occurred
    
    return False

def main():
    print("=== Payment Processing with Timeout Handling ===\n")
    
    print("Test 1: Payment that might timeout")
    result = process_payment_with_timeout(
        amount=99.99,
        currency="USD",
        card_number="4111111111111111",
        timeout_seconds=2
    )
    print(f"Should allow user to continue: {result}\n")
    
    print("\n" + "="*50 + "\n")
    
    print("Test 2: Another payment attempt")
    result = process_payment_with_timeout(
        amount=150.00,
        currency="EUR",
        card_number="5500005555555559",
        timeout_seconds=2
    )
    print(f"Should allow user to continue: {result}\n")

if __name__ == "__main__":
    main()
```
