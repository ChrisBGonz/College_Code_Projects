
using System;
using System.Data;
using Microsoft.Data.SqlClient;

class BankTransfer
{
    private readonly string _connectionString;

    public BankTransfer(string connectionString)
    {
        _connectionString = connectionString;
    }

    public void TransferFunds(int accountAId, int accountBId, decimal amount)
    {
        using (SqlConnection connection = new SqlConnection(_connectionString))
        {
            connection.Open();

            using (SqlTransaction transaction = connection.BeginTransaction(IsolationLevel.Serializable))
            {
                try
                {
                    // Verify Account A has sufficient funds
                    decimal balanceA = GetAccountBalance(connection, transaction, accountAId);

                    if (balanceA < amount)
                    {
                        throw new InvalidOperationException(
                            $"Insufficient funds in Account {accountAId}. " +
                            $"Available balance: {balanceA:C}, Required: {amount:C}");
                    }

                    // Debit $100 from Account A
                    DebitAccount(connection, transaction, accountAId, amount);

                    // Credit $100 to Account B
                    CreditAccount(connection, transaction, accountBId, amount);

                    // Commit the transaction
                    transaction.Commit();

                    Console.WriteLine($"Transfer of {amount:C} completed successfully.");
                    Console.WriteLine($"Debited {amount:C} from Account {accountAId}.");
                    Console.WriteLine($"Credited {amount:C} to Account {accountBId}.");
                }
                catch (Exception ex)
                {
                    // Rollback the transaction in case of any error
                    transaction.Rollback();
                    Console.WriteLine($"Transaction failed and has been rolled back. Error: {ex.Message}");
                    throw;
                }
            }
        }
    }

    private decimal GetAccountBalance(SqlConnection connection, SqlTransaction transaction, int accountId)
    {
        const string query = "SELECT Balance FROM Accounts WHERE AccountId = @AccountId";

        using (SqlCommand command = new SqlCommand(query, connection, transaction))
        {
            command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;

            object result = command.ExecuteScalar();

            if (result == null)
            {
                throw new InvalidOperationException($"Account {accountId} not found.");
            }

            return Convert.ToDecimal(result);
        }
    }

    private void DebitAccount(SqlConnection connection, SqlTransaction transaction, int accountId, decimal amount)
    {
        const string query = @"
            UPDATE Accounts 
            SET Balance = Balance - @Amount,
                LastUpdated = GETDATE()
            WHERE AccountId = @AccountId";

        using (SqlCommand command = new SqlCommand(query, connection, transaction))
        {
            command.Parameters.Add("@Amount", SqlDbType.Decimal).Value = amount;
            command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;

            int rowsAffected = command.ExecuteNonQuery();

            if (rowsAffected == 0)
            {
                throw new InvalidOperationException($"Failed to debit Account {accountId}.");
            }
        }
    }

    private void CreditAccount(SqlConnection connection, SqlTransaction transaction, int accountId, decimal amount)
    {
        const string query = @"
            UPDATE Accounts 
            SET Balance = Balance + @Amount,
                LastUpdated = GETDATE()
            WHERE AccountId = @AccountId";

        using (SqlCommand command = new SqlCommand(query, connection, transaction))
        {
            command.Parameters.Add("@Amount", SqlDbType.Decimal).Value = amount;
            command.Parameters.Add("@AccountId", SqlDbType.Int).Value = accountId;

            int rowsAffected = command.ExecuteNonQuery();

            if (rowsAffected == 0)
            {
                throw new InvalidOperationException($"Failed to credit Account {accountId}.");
            }
        }
    }

    static void Main(string[] args)
    {
        string connectionString = "Server=YOUR_SERVER;Database=YOUR_DATABASE;Trusted_Connection=True;";

        BankTransfer bankTransfer = new BankTransfer(connectionString);

        int accountAId = 1;
        int accountBId = 2;
        decimal transferAmount = 100.00m;

        try
        {
            bankTransfer.TransferFunds(accountAId, accountBId, transferAmount);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"An error occurred: {ex.Message}");
        }
    }
}
