import jwt
import json
from datetime import datetime, timedelta, timezone
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.backends import default_backend
import anthropic

# Initialize Anthropic client
client = anthropic.Anthropic()

def generate_rsa_keys():
    """Generate RSA key pair for JWT signing."""
    private_key = rsa.generate_private_key(
        public_exponent=65537,
        key_size=2048,
        backend=default_backend()
    )
    
    private_pem = private_key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.TraditionalOpenSSL,
        encryption_algorithm=serialization.NoEncryption()
    )
    
    public_pem = private_key.public_key().public_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PublicFormat.SubjectPublicKeyInfo
    )
    
    return private_pem.decode(), public_pem.decode()

def create_jwt_token(payload: dict, secret_key: str, algorithm: str = "HS256", expiry_hours: int = 1) -> str:
    """Create a JWT token with the given payload."""
    payload_copy = payload.copy()
    payload_copy['exp'] = datetime.now(timezone.utc) + timedelta(hours=expiry_hours)
    payload_copy['iat'] = datetime.now(timezone.utc)
    
    token = jwt.encode(payload_copy, secret_key, algorithm=algorithm)
    return token

def decode_jwt_token(token: str, secret_key: str, algorithms: list = None) -> dict:
    """Decode and verify a JWT token."""
    if algorithms is None:
        algorithms = ["HS256"]
    
    decoded = jwt.decode(token, secret_key, algorithms=algorithms)
    return decoded

def analyze_jwt_with_claude(token: str, decoded_payload: dict = None, error: str = None) -> str:
    """Use Claude to analyze JWT token and provide insights."""
    
    # Prepare the analysis request
    analysis_request = f"""
    Analyze the following JWT token and provide detailed insights:
    
    Token: {token[:50]}... (truncated for security)
    
    """
    
    if decoded_payload:
        analysis_request += f"""
    Decoded Payload:
    {json.dumps(decoded_payload, indent=2, default=str)}
    
    Please analyze:
    1. The token structure and claims
    2. Security considerations
    3. Token validity and expiration
    4. Best practices for this type of token
    5. Any potential security issues
    """
    
    if error:
        analysis_request += f"""
    Error encountered: {error}
    
    Please explain:
    1. What this error means
    2. Common causes
    3. How to fix it
    4. Security implications
    """
    
    # Use streaming with Claude
    print("\n🤖 Claude's JWT Analysis:")
    print("-" * 50)
    
    full_response = ""
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": analysis_request
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response

def demonstrate_jwt_scenarios():
    """Demonstrate various JWT authentication scenarios."""
    
    print("=" * 60)
    print("JWT Authentication and Decoding Demo with Claude AI")
    print("=" * 60)
    
    # Scenario 1: Basic HS256 JWT
    print("\n📌 Scenario 1: Basic HS256 JWT Token")
    print("-" * 40)
    
    secret_key = "my-super-secret-key-2024"
    user_payload = {
        "user_id": "12345",
        "username": "john_doe",
        "email": "john@example.com",
        "roles": ["user", "admin"],
        "permissions": ["read", "write", "delete"]
    }
    
    # Create token
    token = create_jwt_token(user_payload, secret_key)
    print(f"✅ Created JWT Token: {token[:50]}...")
    
    # Decode token
    try:
        decoded = decode_jwt_token(token, secret_key)
        print(f"✅ Successfully decoded token")
        print(f"   User: {decoded.get('username')}")
        print(f"   Roles: {decoded.get('roles')}")
        
        # Analyze with Claude
        analyze_jwt_with_claude(token, decoded)
        
    except jwt.ExpiredSignatureError as e:
        print(f"❌ Token expired: {e}")
        analyze_jwt_with_claude(token, error=str(e))
    except jwt.InvalidTokenError as e:
        print(f"❌ Invalid token: {e}")
        analyze_jwt_with_claude(token, error=str(e))
    
    # Scenario 2: Expired Token
    print("\n📌 Scenario 2: Expired JWT Token")
    print("-" * 40)
    
    expired_payload = {
        "user_id": "67890",
        "username": "jane_doe",
        "exp": datetime.now(timezone.utc) - timedelta(hours=1)  # Already expired
    }
    
    expired_token = jwt.encode(expired_payload, secret_key, algorithm="HS256")
    print(f"⚠️  Created expired token: {expired_token[:50]}...")
    
    try:
        decoded_expired = decode_jwt_token(expired_token, secret_key)
        print("Decoded expired token (unexpected)")
    except jwt.ExpiredSignatureError as e:
        print(f"✅ Correctly caught expired token error")
        analyze_jwt_with_claude(expired_token, error=f"ExpiredSignatureError: {str(e)}")
    
    # Scenario 3: Invalid Signature
    print("\n📌 Scenario 3: Invalid Signature")
    print("-" * 40)
    
    valid_token = create_jwt_token({"user": "test"}, "correct-secret")
    tampered_token = valid_token[:-10] + "tampered!!"
    print(f"⚠️  Tampered token: {tampered_token[:50]}...")
    
    try:
        decoded_tampered = decode_jwt_token(tampered_token, "correct-secret")
        print("Decoded tampered token (unexpected)")
    except jwt.InvalidSignatureError as e:
        print(f"✅ Correctly caught invalid signature")
        analyze_jwt_with_claude(tampered_token, error=f"InvalidSignatureError: {str(e)}")
    except jwt.DecodeError as e:
        print(f"✅ Correctly caught decode error")
        analyze_jwt_with_claude(tampered_token, error=f"DecodeError: {str(e)}")
    
    # Scenario 4: RS256 JWT with RSA Keys
    print("\n📌 Scenario 4: RS256 JWT with RSA Keys")
    print("-" * 40)
    
    private_key, public_key = generate_rsa_keys()
    print("✅ Generated RSA key pair")
    
    rs256_payload = {
        "user_id": "99999",
        "service": "api-gateway",
        "scope": "full-access",
        "issuer": "auth-service"
    }
    
    rs256_token = jwt.encode(rs256_payload, private_key, algorithm="RS256")
    print(f"✅ Created RS256 token: {rs256_token[:50]}...")
    
    try:
        decoded_rs256 = jwt.decode(rs256_token, public_key, algorithms=["RS256"])
        print(f"✅ Successfully decoded RS256 token")
        print(f"   Service: {decoded_rs256.get('service')}")
        print(f"   Scope: {decoded_rs256.get('scope')}")
        
        analyze_jwt_with_claude(rs256_token, decoded_rs256)
        
    except jwt.InvalidTokenError as e:
        print(f"❌ Error decoding RS256 token: {e}")
        analyze_jwt_with_claude(rs256_token, error=str(e))
    
    # Scenario 5: Complex Claims JWT
    print("\n📌 Scenario 5: Complex Claims JWT")
    print("-" * 40)
    
    complex_payload = {
        "sub": "user-12345",
        "iss": "https://auth.example.com",
        "aud": ["api.example.com", "admin.example.com"],
        "jti": "unique-token-id-abc123",
        "user_metadata": {
            "department": "engineering",
            "team": "backend",
            "clearance_level": 3
        },
        "permissions": {
            "resources": ["users", "projects", "reports"],
            "actions": ["create", "read", "update", "delete"]
        },
        "session_info": {
            "ip_address": "192.168.1.100",
            "user_agent": "Mozilla/5.0",
            "login_method": "oauth2"
        }
    }
    
    complex_token = create_jwt_token(complex_payload, secret_key, expiry_hours=8)
    print(f"✅ Created complex JWT token: {complex_token[:50]}...")
    
    try:
        decoded_complex = decode_jwt_token(complex_token, secret_key)
        print(f"✅ Successfully decoded complex token")
        
        analyze_jwt_with_claude(complex_token, decoded_complex)
        
    except jwt.InvalidTokenError as e:
        print(f"❌ Error: {e}")
        analyze_jwt_with_claude(complex_token, error=str(e))
    
    print("\n" + "=" * 60)
    print("JWT Demo Complete!")
    print("=" * 60)

def interactive_jwt_analyzer():
    """Interactive JWT analyzer using Claude."""
    print("\n🔍 Interactive JWT Analyzer")
    print("-" * 40)
    print("Enter a JWT token to analyze (or 'quit' to exit):")
    
    while True:
        token_input = input("\nJWT Token: ").strip()
        
        if token_input.lower() == 'quit':
            break
        
        if not token_input:
            continue
        
        # Try to decode without verification first to show structure
        try:
            # Decode without verification to show structure
            unverified = jwt.decode(
                token_input, 
                options={"verify_signature": False},
                algorithms=["HS256", "RS256", "ES256"]
            )
            print(f"\n📋 Token Structure (unverified):")
            print(json.dumps(unverified, indent=2, default=str))
            
            # Analyze with Claude
            analyze_jwt_with_claude(token_input, unverified)
            
        except jwt.DecodeError as e:
            print(f"❌ Cannot decode token: {e}")
            analyze_jwt_with_claude(token_input, error=str(e))

if __name__ == "__main__":
    # Run the main demonstration
    demonstrate_jwt_scenarios()
    
    # Optionally run interactive analyzer
    print("\n" + "=" * 60)
    print("Would you like to analyze a custom JWT token? (y/n)")
    choice = input().strip().lower()
    
    if choice == 'y':
        interactive_jwt_analyzer()
