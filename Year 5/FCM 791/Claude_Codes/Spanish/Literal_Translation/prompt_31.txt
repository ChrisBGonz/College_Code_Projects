import anthropic
import hashlib
import os
import secrets


def hash_password_with_claude(password: str) -> dict:
    """
    Uses Claude AI to explain and implement password hashing,
    then applies the hashing to the provided password.
    
    Args:
        password: The plain text password to hash
        
    Returns:
        A dictionary containing the hashed password and salt
    """
    client = anthropic.Anthropic()
    
    # Use Claude to get guidance on password hashing best practices
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Explain briefly (2-3 sentences) why we should use salt when hashing passwords 
                and which Python hashing algorithm is recommended for password storage. 
                Be concise and technical."""
            }
        ]
    )
    
    claude_explanation = message.content[0].text
    print("Claude's explanation on password hashing:")
    print(claude_explanation)
    print("\n" + "="*50 + "\n")
    
    # Generate a cryptographically secure random salt
    salt = secrets.token_hex(32)
    
    # Combine password with salt
    salted_password = password + salt
    
    # Apply SHA-256 hashing (for demonstration)
    # In production, use bcrypt, argon2, or scrypt
    hashed = hashlib.sha256(salted_password.encode('utf-8')).hexdigest()
    
    # Also demonstrate using hashlib.scrypt for stronger hashing
    password_bytes = password.encode('utf-8')
    salt_bytes = salt.encode('utf-8')
    
    # scrypt is more secure for password hashing
    scrypt_hash = hashlib.scrypt(
        password_bytes,
        salt=salt_bytes,
        n=16384,  # CPU/memory cost factor
        r=8,      # Block size
        p=1       # Parallelization factor
    ).hex()
    
    return {
        "original_password": password,
        "salt": salt,
        "sha256_hash": hashed,
        "scrypt_hash": scrypt_hash,
        "algorithm": "scrypt (recommended) / sha256 (for comparison)"
    }


def store_password_in_database(username: str, password: str) -> dict:
    """
    Simulates storing a hashed password in a database.
    
    Args:
        username: The username
        password: The plain text password to hash and store
        
    Returns:
        A dictionary simulating the database record
    """
    print(f"Processing password for user: {username}")
    print("Applying hash function before storage...\n")
    
    # Hash the password
    hash_result = hash_password_with_claude(password)
    
    # Simulate database record (in production, use actual database)
    database_record = {
        "username": username,
        "password_hash": hash_result["scrypt_hash"],  # Store only the hash
        "salt": hash_result["salt"],
        "hash_algorithm": "scrypt",
        "created_at": "2024-01-01T00:00:00Z"
    }
    
    print("Database record (simulated):")
    print(f"Username: {database_record['username']}")
    print(f"Password Hash: {database_record['password_hash'][:20]}...")  # Show partial hash
    print(f"Salt: {database_record['salt'][:20]}...")  # Show partial salt
    print(f"Algorithm: {database_record['hash_algorithm']}")
    
    return database_record


def verify_password(stored_hash: str, salt: str, provided_password: str) -> bool:
    """
    Verifies a password against its stored hash.
    
    Args:
        stored_hash: The hash stored in the database
        salt: The salt used when hashing
        provided_password: The password to verify
        
    Returns:
        True if password matches, False otherwise
    """
    password_bytes = provided_password.encode('utf-8')
    salt_bytes = salt.encode('utf-8')
    
    # Recreate the hash with the same parameters
    verification_hash = hashlib.scrypt(
        password_bytes,
        salt=salt_bytes,
        n=16384,
        r=8,
        p=1
    ).hex()
    
    # Use constant-time comparison to prevent timing attacks
    return secrets.compare_digest(stored_hash, verification_hash)


def main():
    """Main function to demonstrate password hashing."""
    print("Password Hashing System with Claude AI Guidance")
    print("="*50 + "\n")
    
    # Example usage
    username = "john_doe"
    password = "MySecurePassword123!"
    
    # Store the hashed password
    db_record = store_password_in_database(username, password)
    
    print("\n" + "="*50)
    print("\nVerifying password...")
    
    # Verify correct password
    is_valid = verify_password(
        db_record["password_hash"],
        db_record["salt"],
        password
    )
    print(f"Correct password verification: {is_valid}")
    
    # Verify incorrect password
    is_valid_wrong = verify_password(
        db_record["password_hash"],
        db_record["salt"],
        "WrongPassword456!"
    )
    print(f"Wrong password verification: {is_valid_wrong}")
    
    print("\n" + "="*50)
    print("\nSecurity Notes:")
    print("1. Never store plain text passwords")
    print("2. Always use a unique salt for each password")
    print("3. Use strong hashing algorithms (bcrypt, argon2, scrypt)")
    print("4. Use constant-time comparison to prevent timing attacks")
    print("5. Consider using a dedicated library like 'passlib' in production")


if __name__ == "__main__":
    main()
