import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

public class DatabaseHandler {

    private static final Logger LOGGER = Logger.getLogger(DatabaseHandler.class.getName());

    private static final String DB_URL = "jdbc:mysql://localhost:3306/mydb";
    private static final String DB_USER = "root";
    private static final String DB_PASSWORD = "password";

    public static class DatabaseException extends RuntimeException {

        private final int errorCode;
        private final String sqlState;

        public DatabaseException(String message, int errorCode, String sqlState, Throwable cause) {
            super(message, cause);
            this.errorCode = errorCode;
            this.sqlState = sqlState;
        }

        public int getErrorCode() {
            return errorCode;
        }

        public String getSqlState() {
            return sqlState;
        }

        @Override
        public String toString() {
            return String.format("DatabaseException{message='%s', errorCode=%d, sqlState='%s'}",
                    getMessage(), errorCode, sqlState);
        }
    }

    public ResultSet executeQuery(String query, Object... parameters) {
        Connection connection = null;
        PreparedStatement preparedStatement = null;

        try {
            connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
            preparedStatement = connection.prepareStatement(query);

            for (int i = 0; i < parameters.length; i++) {
                preparedStatement.setObject(i + 1, parameters[i]);
            }

            return preparedStatement.executeQuery();

        } catch (SQLException e) {
            logSQLException(e, query);
            throw new DatabaseException(
                    "Error al ejecutar la consulta: " + query,
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    public int executeUpdate(String query, Object... parameters) {
        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(query)) {

            for (int i = 0; i < parameters.length; i++) {
                preparedStatement.setObject(i + 1, parameters[i]);
            }

            return preparedStatement.executeUpdate();

        } catch (SQLException e) {
            logSQLException(e, query);
            throw new DatabaseException(
                    "Error al ejecutar la actualización: " + query,
                    e.getErrorCode(),
                    e.getSQLState(),
                    e
            );
        }
    }

    private void logSQLException(SQLException e, String query) {
        LOGGER.log(Level.SEVERE, "=== Error de Base de Datos ===");
        LOGGER.log(Level.SEVERE, "Consulta ejecutada: {0}", query);
        LOGGER.log(Level.SEVERE, "Mensaje de error: {0}", e.getMessage());
        LOGGER.log(Level.SEVERE, "Código de error SQL: {0}", e.getErrorCode());
        LOGGER.log(Level.SEVERE, "Estado SQL: {0}", e.getSQLState());
        LOGGER.log(Level.SEVERE, "Causa raíz:", e);

        SQLException nextException = e.getNextException();
        while (nextException != null) {
            LOGGER.log(Level.SEVERE, "Excepción SQL encadenada: {0}", nextException.getMessage());
            nextException = nextException.getNextException();
        }
    }

    public static void main(String[] args) {
        DatabaseHandler handler = new DatabaseHandler();

        try {
            String query = "SELECT * FROM users WHERE id = ?";
            ResultSet resultSet = handler.executeQuery(query, 1);

            while (resultSet.next()) {
                System.out.println("Usuario: " + resultSet.getString("name"));
            }

        } catch (DatabaseException e) {
            System.err.println("Se capturó una excepción personalizada: " + e);
            System.err.println("Código de error: " + e.getErrorCode());
            System.err.println("Estado SQL: " + e.getSqlState());
        }

        try {
            String updateQuery = "UPDATE users SET name = ? WHERE id = ?";
            int rowsAffected = handler.executeUpdate(updateQuery, "Nuevo Nombre", 1);
            System.out.println("Filas afectadas: " + rowsAffected);

        } catch (DatabaseException e) {
            System.err.println("Error al actualizar: " + e.getMessage());
        }
    }
}
