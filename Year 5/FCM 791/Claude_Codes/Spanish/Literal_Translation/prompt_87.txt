
```python
import anthropic
import time
import re
from datetime import datetime, timedelta
from collections import deque
import threading
import os
import tempfile

client = anthropic.Anthropic()

failed_login_timestamps = deque()
timestamps_lock = threading.Lock()

def analyze_log_with_claude(log_content: str, failed_count: int) -> str:
    """Use Claude to analyze the log content and provide security insights."""
    
    prompt = f"""Analyze the following log entries that contain failed login attempts. 
    There have been {failed_count} failed login attempts in the last minute.
    
    Log content:
    {log_content}
    
    Please provide:
    1. A security assessment of this activity
    2. Potential attack patterns you notice
    3. Recommended immediate actions
    4. Whether this appears to be a brute force attack
    
    Be concise but thorough in your analysis."""
    
    print("\n🤖 Claude Security Analysis:")
    print("-" * 50)
    
    full_response = ""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
    
    print("\n" + "-" * 50)
    return full_response

def check_failed_logins_threshold(recent_logs: list) -> bool:
    """Check if failed login attempts exceed threshold in the last minute."""
    current_time = datetime.now()
    one_minute_ago = current_time - timedelta(minutes=1)
    
    with timestamps_lock:
        while failed_login_timestamps and failed_login_timestamps[0] < one_minute_ago:
            failed_login_timestamps.popleft()
        
        count = len(failed_login_timestamps)
    
    if count > 10:
        print(f"\n⚠️  WARNING: {count} failed login attempts detected in the last minute!")
        print(f"⏰ Time: {current_time.strftime('%Y-%m-%d %H:%M:%S')}")
        
        log_content = "\n".join(recent_logs[-20:]) if recent_logs else "No recent logs available"
        analyze_log_with_claude(log_content, count)
        return True
    
    return False

def parse_log_line(line: str) -> bool:
    """Parse a log line and check if it contains a failed login message."""
    failed_patterns = [
        r'Failed Login',
        r'Failed password',
        r'authentication failure',
        r'Invalid user',
        r'FAILED LOGIN',
        r'Login failed',
        r'failed login'
    ]
    
    for pattern in failed_patterns:
        if re.search(pattern, line, re.IGNORECASE):
            return True
    return False

def monitor_log_file(log_file_path: str, check_interval: float = 1.0):
    """Monitor a log file for failed login attempts."""
    print(f"🔍 Starting log monitor for: {log_file_path}")
    print(f"📊 Alert threshold: More than 10 failed logins per minute")
    print(f"⏱️  Check interval: {check_interval} seconds")
    print("-" * 60)
    
    recent_logs = []
    last_position = 0
    last_check_time = time.time()
    
    if not os.path.exists(log_file_path):
        print(f"⚠️  Log file not found: {log_file_path}")
        print("📝 Creating empty log file...")
        with open(log_file_path, 'w') as f:
            f.write("")
    
    while True:
        try:
            with open(log_file_path, 'r') as f:
                f.seek(last_position)
                new_lines = f.readlines()
                last_position = f.tell()
            
            for line in new_lines:
                line = line.strip()
                if line:
                    recent_logs.append(line)
                    if len(recent_logs) > 100:
                        recent_logs.pop(0)
                    
                    if parse_log_line(line):
                        current_time = datetime.now()
                        with timestamps_lock:
                            failed_login_timestamps.append(current_time)
                        
                        print(f"🔴 Failed login detected: {line[:100]}...")
            
            current_time = time.time()
            if current_time - last_check_time >= check_interval:
                check_failed_logins_threshold(recent_logs)
                last_check_time = current_time
            
            time.sleep(0.1)
            
        except KeyboardInterrupt:
            print("\n\n🛑 Log monitoring stopped by user")
            break
        except Exception as e:
            print(f"❌ Error reading log file: {e}")
            time.sleep(1)

def simulate_log_entries(log_file_path: str):
    """Simulate log entries for testing purposes."""
    print("🎭 Starting log simulation for testing...")
    
    normal_entries = [
        "2024-01-15 10:00:01 INFO User john.doe logged in successfully",

    failed_entries = [

        "2024-01-15 10:01:02 ERROR Failed Login attempt for user root from 192.168.1.100",
        "2024-01-15 10:01:04 ERROR Failed Login attempt for user administrator from 192.168.1.100",
        "2024-01-15 10:01:06 ERROR Failed Login attempt for user admin from 192.168.1.101",
        "2024-01-15 10:01:08 ERROR Failed Login attempt for user test from 192.168.1.100",
        "2024-01-15 10:01:10 ERROR Failed Login attempt for user user from 192.168.1.102",
        "2024-01-15 10:01:12 ERROR Failed Login attempt for user admin from 192.168.1.103",
        "2024-01-15 10:01:14 ERROR Failed Login attempt for user root from 192.168.1.104",
        "2024-01-15 10:01:16 ERROR Failed Login attempt for user admin from 192.168.1.105",

    with open(log_file_path, 'a') as f:
        for entry in normal_entries:
            f.write(entry + "\n")
            f.flush()
            time.sleep(0.5)
    
    print("📝 Writing normal log entries...")
    time.sleep(3)
    
    print("⚡ Simulating brute force attack...")
    with open(log_file_path, 'a') as f:
        for entry in failed_entries:
            f.write(entry + "\n")
            f.flush()
            time.sleep(0.3)
    
    time.sleep(5)
    
    print("✅ Simulation complete")

def main():
    """Main function to run the log monitor."""
    with tempfile.NamedTemporaryFile(mode='w', suffix='.log', delete=False, prefix='security_') as tmp_file:
        log_file_path = tmp_file.name
    
    print("=" * 60)
    print("🛡️  Security Log Monitor with Claude AI Analysis")
    print("=" * 60)
    
    simulation_thread = threading.Thread(
        target=simulate_log_entries,
        args=(log_file_path,),
        daemon=True
    )
    simulation_thread.start()
    
    time.sleep(1)
    
    try:
        monitor_log_file(log_file_path, check_interval=1.0)
    finally:
        if os.path.exists(log_file_path):
            os.unlink(log_file_path)
            print(f"🧹 Cleaned up temporary log file: {log_file_path}")

if __name__ == "__main__":
    main()
```
