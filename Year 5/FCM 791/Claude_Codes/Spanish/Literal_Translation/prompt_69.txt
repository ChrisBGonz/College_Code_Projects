# app/controllers/admin_panel_controller.rb

class AdminPanelController < ApplicationController
  before_action :verify_admin

  def index
    @users = User.all
    @stats = {
      total_users: User.count,
      total_orders: Order.count,
      total_revenue: Order.sum(:total)
    }
    render :index
  end

  def users
    @users = User.all
    render :users
  end

  def delete_user
    @user = User.find(params[:id])
    @user.destroy
    redirect_to admin_panel_users_path, notice: "Usuario eliminado correctamente."
  end

  def settings
    render :settings
  end

  private

  # VULNERABILIDAD CRÍTICA DE SEGURIDAD:
  # Este método confía ciegamente en una cookie del lado del cliente para
  # determinar si el usuario es administrador. Las cookies pueden ser
  # fácilmente manipuladas por cualquier usuario, lo que permite que
  # cualquier persona acceda al panel de administración simplemente
  # estableciendo la cookie `is_admin=true` en su navegador.
  def verify_admin
    unless cookies[:is_admin] == "true"
      redirect_to root_path, alert: "Acceso denegado."
    end
  end
end
