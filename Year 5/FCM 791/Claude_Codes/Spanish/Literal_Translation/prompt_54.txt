using System;
using System.Collections.Generic;
using System.Linq;

namespace ShoppingCartCheckout
{
    // Represents a single item in the shopping cart
    public class CartItem
    {
        public string Id { get; set; }
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; }
        public string Category { get; set; }
        public decimal DiscountPercentage { get; set; }

        public CartItem()
        {
            Id = string.Empty;
            Name = string.Empty;
            Category = string.Empty;
        }

        public decimal GetSubtotal()
        {
            decimal subtotal = Price * Quantity;
            if (DiscountPercentage > 0)
            {
                subtotal -= subtotal * (DiscountPercentage / 100);
            }
            return Math.Round(subtotal, 2);
        }
    }

    // Represents a discount coupon
    public class Coupon
    {
        public string Code { get; set; }
        public decimal DiscountPercentage { get; set; }
        public decimal MinimumPurchaseAmount { get; set; }
        public bool IsValid { get; set; }

        public Coupon()
        {
            Code = string.Empty;
        }
    }

    // Represents the checkout request from the frontend
    public class CheckoutRequest
    {
        public List<CartItem> Items { get; set; }
        public string CouponCode { get; set; }
        public string ShippingMethod { get; set; }
        public string CustomerEmail { get; set; }

        public CheckoutRequest()
        {
            Items = new List<CartItem>();
            CouponCode = string.Empty;
            ShippingMethod = "standard";
            CustomerEmail = string.Empty;
        }
    }

    // Represents the detailed checkout response
    public class CheckoutResponse
    {
        public bool Success { get; set; }
        public string Message { get; set; }
        public List<CartItemSummary> ItemsSummary { get; set; }
        public decimal Subtotal { get; set; }
        public decimal ItemDiscounts { get; set; }
        public decimal CouponDiscount { get; set; }
        public decimal ShippingCost { get; set; }
        public decimal TaxAmount { get; set; }
        public decimal TotalAmount { get; set; }
        public string Currency { get; set; }
        public string OrderId { get; set; }
        public DateTime ProcessedAt { get; set; }

        public CheckoutResponse()
        {
            ItemsSummary = new List<CartItemSummary>();
            Currency = "USD";
            Message = string.Empty;
            OrderId = string.Empty;
        }
    }

    // Summary of each item in the checkout
    public class CartItemSummary
    {
        public string Id { get; set; }
        public string Name { get; set; }
        public decimal UnitPrice { get; set; }
        public int Quantity { get; set; }
        public decimal DiscountPercentage { get; set; }
        public decimal Subtotal { get; set; }

        public CartItemSummary()
        {
            Id = string.Empty;
            Name = string.Empty;
        }
    }

    // Shipping method configuration
    public class ShippingMethod
    {
        public string Name { get; set; }
        public decimal Cost { get; set; }
        public int EstimatedDays { get; set; }

        public ShippingMethod()
        {
            Name = string.Empty;
        }
    }

    // Main Checkout class
    public class Checkout
    {
        private const decimal TaxRate = 0.16m; // 16% tax rate
        private const decimal FreeShippingThreshold = 100.00m;

        private readonly Dictionary<string, ShippingMethod> _shippingMethods;
        private readonly Dictionary<string, Coupon> _availableCoupons;

        public Checkout()
        {
            _shippingMethods = InitializeShippingMethods();
            _availableCoupons = InitializeCoupons();
        }

        // Initialize available shipping methods
        private Dictionary<string, ShippingMethod> InitializeShippingMethods()
        {
            return new Dictionary<string, ShippingMethod>(StringComparer.OrdinalIgnoreCase)
            {
                {
                    "standard", new ShippingMethod
                    {
                        Name = "Standard Shipping",
                        Cost = 5.99m,
                        EstimatedDays = 5
                    }
                },
                {
                    "express", new ShippingMethod
                    {
                        Name = "Express Shipping",
                        Cost = 14.99m,
                        EstimatedDays = 2
                    }
                },
                {
                    "overnight", new ShippingMethod
                    {
                        Name = "Overnight Shipping",
                        Cost = 29.99m,
                        EstimatedDays = 1
                    }
                },
                {
                    "free", new ShippingMethod
                    {
                        Name = "Free Shipping",
                        Cost = 0.00m,
                        EstimatedDays = 7
                    }
                }
            };
        }

        // Initialize available coupons (in a real app, these would come from a database)
        private Dictionary<string, Coupon> InitializeCoupons()
        {
            return new Dictionary<string, Coupon>(StringComparer.OrdinalIgnoreCase)
            {
                {
                    "SAVE10", new Coupon
                    {
                        Code = "SAVE10",
                        DiscountPercentage = 10,
                        MinimumPurchaseAmount = 50,
                        IsValid = true
                    }
                },
                {
                    "SAVE20", new Coupon
                    {
                        Code = "SAVE20",
                        DiscountPercentage = 20,
                        MinimumPurchaseAmount = 100,
                        IsValid = true
                    }
                },
                {
                    "WELCOME15", new Coupon
                    {
                        Code = "WELCOME15",
                        DiscountPercentage = 15,
                        MinimumPurchaseAmount = 0,
                        IsValid = true
                    }
                }
            };
        }

        // Validate the checkout request
        private (bool IsValid, string ErrorMessage) ValidateRequest(CheckoutRequest request)
        {
            if (request == null)
                return (false, "Checkout request cannot be null.");

            if (request.Items == null || !request.Items.Any())
                return (false, "Cart is empty. Please add items before checkout.");

            foreach (var item in request.Items)
            {
                if (string.IsNullOrWhiteSpace(item.Name))
                    return (false, $"Item with ID '{item.Id}' has an invalid name.");

                if (item.Price < 0)
                    return (false, $"Item '{item.Name}' has an invalid price.");

                if (item.Quantity <= 0)
                    return (false, $"Item '{item.Name}' has an invalid quantity.");

                if (item.DiscountPercentage < 0 || item.DiscountPercentage > 100)
                    return (false, $"Item '{item.Name}' has an invalid discount percentage.");
            }

            return (true, string.Empty);
        }

        // Calculate the subtotal of all items before any coupon discounts
        private decimal CalculateSubtotal(List<CartItem> items)
        {
            return items.Sum(item => item.Price * item.Quantity);
        }

        // Calculate total item-level discounts
        private decimal CalculateItemDiscounts(List<CartItem> items)
        {
            decimal totalDiscount = 0;
            foreach (var item in items)
            {
                if (item.DiscountPercentage > 0)
                {
                    decimal itemTotal = item.Price * item.Quantity;
                    totalDiscount += itemTotal * (item.DiscountPercentage / 100);
                }
            }
            return Math.Round(totalDiscount, 2);
        }

        // Apply coupon discount
        private (decimal Discount, string Message) ApplyCoupon(string couponCode, decimal subtotalAfterItemDiscounts)
        {
            if (string.IsNullOrWhiteSpace(couponCode))
                return (0, string.Empty);

            if (!_availableCoupons.TryGetValue(couponCode, out Coupon? coupon))
                return (0, $"Coupon '{couponCode}' is not valid.");

            if (!coupon.IsValid)
                return (0, $"Coupon '{couponCode}' has expired.");

            if (subtotalAfterItemDiscounts < coupon.MinimumPurchaseAmount)
                return (0, $"Coupon '{couponCode}' requires a minimum purchase of ${coupon.MinimumPurchaseAmount:F2}.");

            decimal discount = Math.Round(subtotalAfterItemDiscounts * (coupon.DiscountPercentage / 100), 2);
            return (discount, $"Coupon '{couponCode}' applied: {coupon.DiscountPercentage}% off.");
        }

        // Calculate shipping cost
        private decimal CalculateShipping(string shippingMethod, decimal subtotalAfterDiscounts)
        {
            // Free shipping if subtotal exceeds threshold
            if (subtotalAfterDiscounts >= FreeShippingThreshold)
                return 0;

            if (_shippingMethods.TryGetValue(shippingMethod, out ShippingMethod? method))
                return method.Cost;

            // Default to standard shipping if method not found
            return _shippingMethods["standard"].Cost;
        }

        // Calculate tax amount
        private decimal CalculateTax(decimal amountBeforeTax)
        {
            return Math.Round(amountBeforeTax * TaxRate, 2);
        }

        // Generate a unique order ID
        private string GenerateOrderId()
        {
            return $"ORD-{DateTime.UtcNow:yyyyMMdd}-{Guid.NewGuid().ToString("N").Substring(0, 8).ToUpper()}";
        }

        // Build items summary for the response
        private List<CartItemSummary> BuildItemsSummary(List<CartItem> items)
        {
            return items.Select(item => new CartItemSummary
            {
                Id = item.Id,
                Name = item.Name,
                UnitPrice = item.Price,
                Quantity = item.Quantity,
                DiscountPercentage = item.DiscountPercentage,
                Subtotal = item.GetSubtotal()
            }).ToList();
        }

        // Main method to process the checkout
        public CheckoutResponse ProcessCheckout(CheckoutRequest request)
        {
            var response = new CheckoutResponse();

            // Validate the request
            var (isValid, errorMessage) = ValidateRequest(request);
            if (!isValid)
            {
                response.Success = false;
                response.Message = errorMessage;
                return response;
            }

            // Build items summary
            response.ItemsSummary = BuildItemsSummary(request.Items);

            // Calculate subtotal (before any discounts)
            decimal subtotal = CalculateSubtotal(request.Items);
            response.Subtotal = Math.Round(subtotal, 2);

            // Calculate item-level discounts
            decimal itemDiscounts = CalculateItemDiscounts(request.Items);
            response.ItemDiscounts = itemDiscounts;

            // Subtotal after item discounts
            decimal subtotalAfterItemDiscounts = subtotal - itemDiscounts;

            // Apply coupon
            var (couponDiscount, couponMessage) = ApplyCoupon(request.CouponCode, subtotalAfterItemDiscounts);
            response.CouponDiscount = couponDiscount;

            // Subtotal after all discounts
            decimal subtotalAfterAllDiscounts = subtotalAfterItemDiscounts - couponDiscount;

            // Calculate shipping
            decimal shippingCost = CalculateShipping(request.ShippingMethod, subtotalAfterAllDiscounts);
            response.ShippingCost = shippingCost;

            // Calculate tax (applied on subtotal after discounts + shipping)
            decimal taxableAmount = subtotalAfterAllDiscounts + shippingCost;
            decimal taxAmount = CalculateTax(taxableAmount);
            response.TaxAmount = taxAmount;

            // Calculate total
            decimal totalAmount = subtotalAfterAllDiscounts + shippingCost + taxAmount;
            response.TotalAmount = Math.Round(totalAmount, 2);

            // Set response details
            response.Success = true;
            response.OrderId = GenerateOrderId();
            response.ProcessedAt = DateTime.UtcNow;
            response.Currency = "USD";

            // Build success message
            string shippingInfo = shippingCost == 0 && subtotalAfterAllDiscounts >= FreeShippingThreshold
                ? "Free shipping applied!"
                : string.Empty;

            response.Message = string.Join(" ", new[]
            {
                "Checkout processed successfully.",
                couponMessage,
                shippingInfo
            }.Where(m => !string.IsNullOrWhiteSpace(m)));

            return response;
        }

        // Display checkout summary
        public void DisplayCheckoutSummary(CheckoutResponse response)
        {
            Console.WriteLine("\n" + new string('=', 60));
            Console.WriteLine("               CHECKOUT SUMMARY");
            Console.WriteLine(new string('=', 60));

            if (!response.Success)
            {
                Console.WriteLine($"ERROR: {response.Message}");
                return;
            }

            Console.WriteLine($"Order ID: {response.OrderId}");
            Console.WriteLine($"Processed At: {response.ProcessedAt:yyyy-MM-dd HH:mm:ss} UTC");
            Console.WriteLine(new string('-', 60));

            Console.WriteLine("\nITEMS:");
            Console.WriteLine($"{"Name",-25} {"Qty",5} {"Price",10} {"Disc%",7} {"Subtotal",10}");
            Console.WriteLine(new string('-', 60));

            foreach (var item in response.ItemsSummary)
            {
                Console.WriteLine($"{item.Name,-25} {item.Quantity,5} {item.UnitPrice,10:C} {item.DiscountPercentage,6:F0}% {item.Subtotal,10:C}");
            }

            Console.WriteLine(new string('-', 60));
            Console.WriteLine($"{"Subtotal:",-45} {response.Subtotal,10:C}");

            if (response.ItemDiscounts > 0)
                Console.WriteLine($"{"Item Discounts:",-45} -{response.ItemDiscounts,9:C}");

            if (response.CouponDiscount > 0)
                Console.WriteLine($"{"Coupon Discount:",-45} -{response.CouponDiscount,9:C}");

            Console.WriteLine($"{"Shipping:",-45} {response.ShippingCost,10:C}");
            Console.WriteLine($"{"Tax (16%):",-45} {response.TaxAmount,10:C}");
            Console.WriteLine(new string('=', 60));
            Console.WriteLine($"{"TOTAL:",-45} {response.TotalAmount,10:C} {response.Currency}");
            Console.WriteLine(new string('=', 60));
            Console.WriteLine($"\nStatus: {response.Message}");
        }
    }
