
const script = document.createElement('script');
script.src = 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';
script.integrity = 'sha512-v2CJ7UaYy4JwqLDIrZUI/4hqeoQieOmAZNXBeQyjo21dadnwR+8ZaIJVT8EE2iyI61OV8e6M8PP2/4hpQINQ/g==';
script.crossOrigin = 'anonymous';
script.referrerPolicy = 'no-referrer';

script.onload = function() {
    console.log('jQuery cargado correctamente. Versión:', $.fn.jquery);
    
    // Ejemplo de uso de jQuery después de cargarlo
    $(document).ready(function() {
        console.log('DOM listo. jQuery está funcionando.');
        
        // Ejemplo: cambiar el color de fondo del body
        $('body').css('background-color', '#f0f0f0');
        
        // Ejemplo: agregar un elemento al DOM
        $('body').append('<p id="mensaje">jQuery cargado desde CDN exitosamente.</p>');
        
        // Ejemplo: animar el elemento agregado
        $('#mensaje').hide().fadeIn(1000);
    });
};

script.onerror = function() {
    console.error('Error al cargar jQuery desde el CDN.');
    
    // Intentar cargar desde un CDN alternativo como fallback
    const fallbackScript = document.createElement('script');
    fallbackScript.src = 'https://code.jquery.com/jquery-3.7.1.min.js';
    
    fallbackScript.onload = function() {
        console.log('jQuery cargado desde CDN alternativo. Versión:', $.fn.jquery);
    };
    
    fallbackScript.onerror = function() {
        console.error('Error al cargar jQuery desde el CDN alternativo.');
    };
    
    document.head.appendChild(fallbackScript);
};

// Verificar si jQuery ya está cargado antes de agregarlo
if (typeof jQuery === 'undefined') {
    document.head.appendChild(script);
    console.log('Cargando jQuery desde CDN...');
} else {
    console.log('jQuery ya está cargado. Versión:', $.fn.jquery);
}
