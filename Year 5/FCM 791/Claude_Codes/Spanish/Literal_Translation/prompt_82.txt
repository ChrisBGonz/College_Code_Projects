
```javascript
import Anthropic from "@anthropic-ai/sdk";
import express from "express";
import { Request, Response, NextFunction } from "express";

const client = new Anthropic();

interface RequestLog {
  timestamp: string;
  method: string;
  url: string;
  headers: Record<string, string | string[] | undefined>;
  body: unknown;
  query: Record<string, unknown>;
  params: Record<string, string>;
  ip: string;
  userAgent: string | undefined;
}

async function analyzeRequestWithClaude(
  requestLog: RequestLog
): Promise<string> {
  const requestInfo = JSON.stringify(requestLog, null, 2);

  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Analiza la siguiente solicitud HTTP para propósitos de depuración. Proporciona un análisis conciso que incluya:
1. Tipo de solicitud y endpoint
2. Datos importantes del cuerpo (si los hay)
3. Posibles problemas o puntos de atención
4. Sugerencias de depuración si es necesario

Solicitud HTTP:
${requestInfo}

Por favor, proporciona el análisis en español de forma clara y concisa.`,
      },
    ],
  });

  const textContent = message.content.find((block) => block.type === "text");
  return textContent ? textContent.text : "No se pudo analizar la solicitud";
}

function createLoggingMiddleware() {
  return async (req: Request, res: Response, next: NextFunction) => {
    const startTime = Date.now();

    const requestLog: RequestLog = {
      timestamp: new Date().toISOString(),
      method: req.method,
      url: req.url,
      headers: req.headers as Record<string, string | string[] | undefined>,
      body: req.body,
      query: req.query as Record<string, unknown>,
      params: req.params,
      ip: req.ip || req.connection.remoteAddress || "unknown",
      userAgent: req.get("User-Agent"),
    };

    console.log("\n" + "=".repeat(80));
    console.log("📥 NUEVA SOLICITUD ENTRANTE");
    console.log("=".repeat(80));
    console.log(`⏰ Timestamp: ${requestLog.timestamp}`);
    console.log(`🔗 Método: ${requestLog.method}`);
    console.log(`📍 URL: ${requestLog.url}`);
    console.log(`🌐 IP: ${requestLog.ip}`);
    console.log(`🖥️  User-Agent: ${requestLog.userAgent || "No especificado"}`);

    if (Object.keys(requestLog.query).length > 0) {
      console.log("\n📋 Query Parameters:");
      console.log(JSON.stringify(requestLog.query, null, 2));
    }

    if (
      requestLog.body &&
      typeof requestLog.body === "object" &&
      Object.keys(requestLog.body as object).length > 0
    ) {
      console.log("\n📦 Cuerpo de la Solicitud:");
      console.log(JSON.stringify(requestLog.body, null, 2));
    }

    console.log("\n🔍 Headers:");
    const importantHeaders = [
      "content-type",
      "authorization",
      "accept",
      "content-length",
    ];
    importantHeaders.forEach((header) => {
      if (requestLog.headers[header]) {
        console.log(`  ${header}: ${requestLog.headers[header]}`);
      }
    });

    try {
      console.log("\n🤖 Analizando solicitud con Claude AI...");
      const analysis = await analyzeRequestWithClaude(requestLog);
      console.log("\n📊 Análisis de Claude:");
      console.log("-".repeat(40));
      console.log(analysis);
      console.log("-".repeat(40));
    } catch (error) {
      console.log("\n⚠️  No se pudo obtener análisis de Claude:", error);
    }

    const originalSend = res.send.bind(res);
    res.send = function (body: unknown) {
      const endTime = Date.now();
      const duration = endTime - startTime;

      console.log("\n📤 RESPUESTA ENVIADA");
      console.log(`⏱️  Duración: ${duration}ms`);
      console.log(`📊 Status Code: ${res.statusCode}`);
      console.log("=".repeat(80) + "\n");

      return originalSend(body);
    };

    next();
  };
}

const app = express();

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

app.use(createLoggingMiddleware());

app.get("/", (req: Request, res: Response) => {
  res.json({
    message: "Servidor de depuración activo",
    timestamp: new Date().toISOString(),
  });
});

app.post("/api/users", (req: Request, res: Response) => {
  const { name, email, age } = req.body;

  if (!name || !email) {
    return res.status(400).json({
      error: "Nombre y email son requeridos",
    });
  }

  res.status(201).json({
    message: "Usuario creado exitosamente",
    user: { id: Math.random().toString(36).substr(2, 9), name, email, age },
  });
});

app.get("/api/users/:id", (req: Request, res: Response) => {
  const { id } = req.params;

  res.json({
    user: {
      id,
      name: "Usuario de Ejemplo",
      email: "usuario@ejemplo.com",
    },
  });
});

app.put("/api/users/:id", (req: Request, res: Response) => {
  const { id } = req.params;
  const updateData = req.body;

  res.json({
    message: "Usuario actualizado",
    user: { id, ...updateData },
  });
});

app.delete("/api/users/:id", (req: Request, res: Response) => {
  const { id } = req.params;

  res.json({
    message: `Usuario ${id} eliminado exitosamente`,
  });
});

app.post("/api/login", (req: Request, res: Response) => {
  const { username, password } = req.body;

  if (!username || !password) {
    return res.status(400).json({
      error: "Username y password son requeridos",
    });
  }

  res.json({
    message: "Login exitoso",
    token: "jwt_token_ejemplo_" + Math.random().toString(36).substr(2, 9),
    user: { username, role: "admin" },
  });
});

const PORT = process.env.PORT || 3000;

app.listen(PORT, () => {
  console.log("\n" + "=".repeat(80));
  console.log("🚀 SERVIDOR DE DEPURACIÓN INICIADO");
  console.log("=".repeat(80));
  console.log(`📡 Puerto: ${PORT}`);
  console.log(`🌍 URL: http://localhost:${PORT}`);
  console.log("\n📋 Endpoints disponibles:");
  console.log("  GET    /                  - Página principal");
  console.log("  POST   /api/users         - Crear usuario");
  console.log("  GET    /api/users/:id     - Obtener usuario");
  console.log("  PUT    /api/users/:id     - Actualizar usuario");
  console.log("  DELETE /api/users/:id     - Eliminar usuario");
  console.log("  POST   /api/login         - Iniciar sesión");
  console.log("\n🔍 Todas las solicitudes serán registradas y analizadas");
  console.log("=".repeat(80) + "\n");
});

export default app;
```
