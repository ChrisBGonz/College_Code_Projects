
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

const conversationHistory = [];
let inactivityTimer = null;
let sessionActive = false;
let lastActivityTime = null;
const INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds

function formatTime(ms) {
  const minutes = Math.floor(ms / 60000);
  const seconds = Math.floor((ms % 60000) / 1000);
  return `${minutes}m ${seconds}s`;
}

function getTimeUntilLogout() {
  if (!lastActivityTime) return 0;
  const elapsed = Date.now() - lastActivityTime;
  const remaining = INACTIVITY_TIMEOUT - elapsed;
  return Math.max(0, remaining);
}

function resetInactivityTimer() {
  if (!sessionActive) return;

  lastActivityTime = Date.now();

  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
  }

  inactivityTimer = setTimeout(() => {
    performAutoLogout();
  }, INACTIVITY_TIMEOUT);

  console.log(
    `\n[Sistema] Timer de inactividad reiniciado. Sesión expirará en ${formatTime(INACTIVITY_TIMEOUT)} si no hay actividad.`
  );
}

function performAutoLogout() {
  if (!sessionActive) return;

  sessionActive = false;
  if (inactivityTimer) {
    clearTimeout(inactivityTimer);
    inactivityTimer = null;
  }

  console.log("\n" + "=".repeat(60));
  console.log("⚠️  SESIÓN CERRADA AUTOMÁTICAMENTE POR INACTIVIDAD");
  console.log("=".repeat(60));
  console.log(
    "Su sesión ha sido cerrada después de 15 minutos de inactividad."
  );
  console.log("Por favor, inicie sesión nuevamente para continuar.");
  console.log("=".repeat(60) + "\n");

  conversationHistory.length = 0;
  process.exit(0);
}

function startSession(username) {
  sessionActive = true;
  lastActivityTime = Date.now();

  conversationHistory.push({
    role: "user",
    content: `Soy ${username} y acabo de iniciar sesión en mi cuenta bancaria.`,
  });

  console.log("\n" + "=".repeat(60));
  console.log(`✅ Sesión iniciada para: ${username}`);
  console.log(
    `⏱️  La sesión se cerrará automáticamente después de 15 minutos de inactividad`
  );
  console.log("=".repeat(60) + "\n");

  resetInactivityTimer();
}

async function chat(userMessage) {
  if (!sessionActive) {
    console.log(
      "❌ No hay sesión activa. Por favor, inicie sesión primero.\n"
    );
    return;
  }

  resetInactivityTimer();

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  try {
    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system: `Eres un asistente bancario virtual profesional y seguro. 
      Ayudas a los clientes con consultas sobre sus cuentas, transferencias, 
      saldos y otros servicios bancarios. Siempre mantén un tono profesional 
      y recuerda al usuario sobre la seguridad de su sesión cuando sea apropiado.
      La sesión del usuario se cerrará automáticamente después de 15 minutos de inactividad.
      Tiempo restante de sesión: ${formatTime(getTimeUntilLogout())}`,
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    return assistantMessage;
  } catch (error) {
    console.error("Error al comunicarse con el asistente:", error.message);
    throw error;
  }
}

async function main() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const question = (prompt) => {
    return new Promise((resolve) => {
      rl.question(prompt, resolve);
    });
  };

  console.log("\n" + "=".repeat(60));
  console.log("🏦 BIENVENIDO AL SISTEMA BANCARIO SEGURO");
  console.log("=".repeat(60));
  console.log(
    "Este sistema cuenta con cierre automático de sesión por inactividad."
  );
  console.log("=".repeat(60) + "\n");

  const username = await question("Por favor, ingrese su nombre de usuario: ");

  if (!username.trim()) {
    console.log("❌ Nombre de usuario inválido. Saliendo...");
    rl.close();
    return;
  }

  startSession(username.trim());

  const welcomeResponse = await chat(
    "Hola, ¿puedes darme la bienvenida y explicarme qué servicios puedo consultar?"
  );
  console.log(`\n🤖 Asistente Bancario: ${welcomeResponse}\n`);

  console.log(
    "💡 Tip: Escriba 'estado' para ver el tiempo restante de su sesión"
  );
  console.log("💡 Tip: Escriba 'salir' para cerrar sesión manualmente\n");

  while (sessionActive) {
    const userInput = await question("Usted: ");

    if (!userInput.trim()) {
      continue;
    }

    if (userInput.toLowerCase() === "salir") {
      console.log("\n✅ Cerrando sesión manualmente...");
      sessionActive = false;
      if (inactivityTimer) {
        clearTimeout(inactivityTimer);
      }
      console.log("Sesión cerrada. ¡Hasta luego!\n");
      break;
    }

    if (userInput.toLowerCase() === "estado") {
      const remaining = getTimeUntilLogout();
      console.log(
        `\n⏱️  Tiempo restante de sesión: ${formatTime(remaining)}\n`
      );
      resetInactivityTimer();
      continue;
    }

    try {
      const response = await chat(userInput);
      if (response) {
        console.log(`\n🤖 Asistente Bancario: ${response}\n`);
      }
    } catch (error) {
      console.error("Error en la conversación:", error.message);
    }
  }

  rl.close();
}

main().catch(console.error);
```
