#include <iostream>
#include <vector>
#include <string>
#include <stdexcept>
#include <cstring>
#include <cstdio>
#include <cstdlib>
#include <cmath>

extern "C" {
#include <jpeglib.h>
#include <jerror.h>
}

struct Pixel {
    unsigned char r, g, b;
};

struct Image {
    std::vector<Pixel> pixels;
    int width;
    int height;
    int channels;
    
    Image() : width(0), height(0), channels(3) {}
    Image(int w, int h) : width(w), height(h), channels(3), pixels(w * h) {}
    
    Pixel& at(int x, int y) {
        return pixels[y * width + x];
    }
    
    const Pixel& at(int x, int y) const {
        return pixels[y * width + x];
    }
};

struct ErrorManager {
    struct jpeg_error_mgr pub;
    jmp_buf setjmp_buffer;
};

static void errorExit(j_common_ptr cinfo) {
    ErrorManager* myerr = reinterpret_cast<ErrorManager*>(cinfo->err);
    (*cinfo->err->output_message)(cinfo);
    longjmp(myerr->setjmp_buffer, 1);
}

Image loadJPEG(const std::string& filename) {
    FILE* file = fopen(filename.c_str(), "rb");
    if (!file) {
        throw std::runtime_error("Cannot open file: " + filename);
    }
    
    struct jpeg_decompress_struct cinfo;
    ErrorManager jerr;
    
    cinfo.err = jpeg_std_error(&jerr.pub);
    jerr.pub.error_exit = errorExit;
    
    if (setjmp(jerr.setjmp_buffer)) {
        jpeg_destroy_decompress(&cinfo);
        fclose(file);
        throw std::runtime_error("JPEG decompression error");
    }
    
    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, file);
    jpeg_read_header(&cinfo, TRUE);
    
    cinfo.out_color_space = JCS_RGB;
    jpeg_start_decompress(&cinfo);
    
    Image img(cinfo.output_width, cinfo.output_height);
    img.channels = cinfo.output_components;
    
    int rowStride = cinfo.output_width * cinfo.output_components;
    JSAMPARRAY buffer = (*cinfo.mem->alloc_sarray)(
        reinterpret_cast<j_common_ptr>(&cinfo), JPOOL_IMAGE, rowStride, 1);
    
    int row = 0;
    while (cinfo.output_scanline < cinfo.output_height) {
        jpeg_read_scanlines(&cinfo, buffer, 1);
        for (int col = 0; col < img.width; ++col) {
            img.at(col, row).r = buffer[0][col * 3 + 0];
            img.at(col, row).g = buffer[0][col * 3 + 1];
            img.at(col, row).b = buffer[0][col * 3 + 2];
        }
        ++row;
    }
    
    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    fclose(file);
    
    std::cout << "Loaded JPEG: " << filename 
              << " (" << img.width << "x" << img.height << ")" << std::endl;
    
    return img;
}

void saveJPEG(const Image& img, const std::string& filename, int quality = 85) {
    FILE* file = fopen(filename.c_str(), "wb");
    if (!file) {
        throw std::runtime_error("Cannot create file: " + filename);
    }
    
    struct jpeg_compress_struct cinfo;
    ErrorManager jerr;
    
    cinfo.err = jpeg_std_error(&jerr.pub);
    jerr.pub.error_exit = errorExit;
    
    if (setjmp(jerr.setjmp_buffer)) {
        jpeg_destroy_compress(&cinfo);
        fclose(file);
        throw std::runtime_error("JPEG compression error");
    }
    
    jpeg_create_compress(&cinfo);
    jpeg_stdio_dest(&cinfo, file);
    
    cinfo.image_width = img.width;
    cinfo.image_height = img.height;
    cinfo.input_components = 3;
    cinfo.in_color_space = JCS_RGB;
    
    jpeg_set_defaults(&cinfo);
    jpeg_set_quality(&cinfo, quality, TRUE);
    jpeg_start_compress(&cinfo, TRUE);
    
    int rowStride = img.width * 3;
    std::vector<unsigned char> rowBuffer(rowStride);
    JSAMPROW rowPointer[1];
    rowPointer[0] = rowBuffer.data();
    
    while (cinfo.next_scanline < cinfo.image_height) {
        int row = cinfo.next_scanline;
        for (int col = 0; col < img.width; ++col) {
            rowBuffer[col * 3 + 0] = img.at(col, row).r;
            rowBuffer[col * 3 + 1] = img.at(col, row).g;
            rowBuffer[col * 3 + 2] = img.at(col, row).b;
        }
        jpeg_write_scanlines(&cinfo, rowPointer, 1);
    }
    
    jpeg_finish_compress(&cinfo);
    jpeg_destroy_compress(&cinfo);
    fclose(file);
    
    std::cout << "Saved JPEG: " << filename 
              << " (quality=" << quality << ")" << std::endl;
}

Image convertToGrayscale(const Image& src) {
    Image dst(src.width, src.height);
    
    for (int y = 0; y < src.height; ++y) {
        for (int x = 0; x < src.width; ++x) {
            const Pixel& p = src.at(x, y);
            unsigned char gray = static_cast<unsigned char>(
                0.299 * p.r + 0.587 * p.g + 0.114 * p.b);
            dst.at(x, y) = {gray, gray, gray};
        }
    }
    
    std::cout << "Converted to grayscale" << std::endl;
    return dst;
}

Image applyGaussianBlur(const Image& src, int kernelSize = 5, double sigma = 1.0) {
    if (kernelSize % 2 == 0) kernelSize++;
    
    int half = kernelSize / 2;
    std::vector<std::vector<double>> kernel(kernelSize, std::vector<double>(kernelSize));
    double sum = 0.0;
    
    for (int i = -half; i <= half; ++i) {
        for (int j = -half; j <= half; ++j) {
            double value = std::exp(-(i * i + j * j) / (2.0 * sigma * sigma));
            kernel[i + half][j + half] = value;
            sum += value;
        }
    }
    
    for (int i = 0; i < kernelSize; ++i)
        for (int j = 0; j < kernelSize; ++j)
            kernel[i][j] /= sum;
    
    Image dst(src.width, src.height);
    
    for (int y = 0; y < src.height; ++y) {
        for (int x = 0; x < src.width; ++x) {
            double r = 0, g = 0, b = 0;
            
            for (int ky = -half; ky <= half; ++ky) {
                for (int kx = -half; kx <= half; ++kx) {
                    int nx = std::max(0, std::min(src.width - 1, x + kx));
                    int ny = std::max(0, std::min(src.height - 1, y + ky));
                    
                    double w = kernel[ky + half][kx + half];
                    r += src.at(nx, ny).r * w;
                    g += src.at(nx, ny).g * w;
                    b += src.at(nx, ny).b * w;
                }
            }
            
            dst.at(x, y).r = static_cast<unsigned char>(std::min(255.0, std::max(0.0, r)));
            dst.at(x, y).g = static_cast<unsigned char>(std::min(255.0, std::max(0.0, g)));
            dst.at(x, y).b = static_cast<unsigned char>(std::min(255.0, std::max(0.0, b)));
        }
    }
    
    std::cout << "Applied Gaussian blur (kernel=" << kernelSize 
              << ", sigma=" << sigma << ")" << std::endl;
    return dst;
}

Image adjustBrightnessContrast(const Image& src, double brightness = 0.0, double contrast = 1.0) {
    Image dst(src.width, src.height);
    
    for (int y = 0; y < src.height; ++y) {
        for (int x = 0; x < src.width; ++x) {
            const Pixel& p = src.at(x, y);
            
            auto adjust = [&](unsigned char channel) -> unsigned char {
                double val = (channel / 255.0 - 0.5) * contrast + 0.5 + brightness;
                val = std::max(0.0, std::min(1.0, val));
                return static_cast<unsigned char>(val * 255);
            };
            
            dst.at(x, y) = {adjust(p.r), adjust(p.g), adjust(p.b)};
        }
    }
    
    std::cout << "Adjusted brightness=" << brightness 
              << " contrast=" << contrast << std::endl;
    return dst;
}

Image detectEdgesSobel(const Image& src) {
    Image gray = convertToGrayscale(src);
    Image dst(src.width, src.height);
    
    int sobelX[3][3] = {{-1, 0, 1}, {-2, 0, 2}, {-1, 0, 1}};
    int sobelY[3][3] = {{-1, -2, -1}, {0, 0, 0}, {1, 2, 1}};
    
    for (int y = 1; y < src.height - 1; ++y) {
        for (int x = 1; x < src.width - 1; ++x) {
            int gx = 0, gy = 0;
            
            for (int ky = -1; ky <= 1; ++ky) {
                for (int kx = -1; kx <= 1; ++kx) {
                    int pixel = gray.at(x + kx, y + ky).r;
                    gx += sobelX[ky + 1][kx + 1] * pixel;
                    gy += sobelY[ky + 1][kx + 1] * pixel;
                }
            }
            
            int magnitude = static_cast<int>(std::sqrt(gx * gx + gy * gy));
            magnitude = std::min(255, magnitude);
            
            dst.at(x, y) = {
                static_cast<unsigned char>(magnitude),
                static_cast<unsigned char>(magnitude),
                static_cast<unsigned char>(magnitude)
            };
        }
    }
    
    std::cout << "Applied Sobel edge detection" << std::endl;
    return dst;
}

Image flipHorizontal(const Image& src) {
    Image dst(src.width, src.height);
    
    for (int y = 0; y < src.height; ++y) {
        for (int x = 0; x < src.width; ++x) {
            dst.at(x, y) = src.at(src.width - 1 - x, y);
        }
    }
    
    std::cout << "Flipped horizontally" << std::endl;
    return dst;
}

Image flipVertical(const Image& src) {
    Image dst(src.width, src.height);
    
    for (int y = 0; y < src.height; ++y) {
        for (int x = 0; x < src.width; ++x) {
            dst.at(x, y) = src.at(x, src.height - 1 - y);
        }
    }
    
    std::cout << "Flipped vertically" << std::endl;
    return dst;
}

Image resizeImage(const Image& src, int newWidth, int newHeight) {
    Image dst(newWidth, newHeight);
    
    double scaleX = static_cast<double>(src.width) / newWidth;
    double scaleY = static_cast<double>(src.height) / newHeight;
    
    for (int y = 0; y < newHeight; ++y) {
        for (int x = 0; x < newWidth; ++x) {
            double srcX = x * scaleX;
            double srcY = y * scaleY;
            
            int x0 = static_cast<int>(srcX);
            int y0 = static_cast<int>(srcY);
            int x1 = std::min(x0 + 1, src.width - 1);
            int y1 = std::min(y0 + 1, src.height - 1);
            
            double fx = srcX - x0;
            double fy = srcY - y0;
            
            auto lerp = [](double a, double b, double t) { return a + t * (b - a); };
            
            auto interpolate = [&](int channel) -> unsigned char {
                auto getChannel = [&](const Pixel& p) -> double {
                    if (channel == 0) return p.r;
                    if (channel == 1) return p.g;
                    return p.b;
                };
                
                double top = lerp(getChannel(src.at(x0, y0)), getChannel(src.at(x1, y0)), fx);
                double bottom = lerp(getChannel(src.at(x0, y1)), getChannel(src.at(x1, y1)), fx);
                return static_cast<unsigned char>(lerp(top, bottom, fy));
            };
            
            dst.at(x, y) = {interpolate(0), interpolate(1), interpolate(2)};
        }
    }
    
    std::cout << "Resized to " << newWidth << "x" << newHeight << std::endl;
    return dst;
}

Image cropImage(const Image& src, int startX, int startY, int cropWidth, int cropHeight) {
    startX = std::max(0, startX);
    startY = std::max(0, startY);
    cropWidth = std::min(cropWidth, src.width - startX);
    cropHeight = std::min(cropHeight, src.height - startY);
    
    Image dst(cropWidth, cropHeight);
    
    for (int y = 0; y < cropHeight; ++y) {
        for (int x = 0; x < cropWidth; ++x) {
            dst.at(x, y) = src.at(startX + x, startY + y);
        }
    }
    
    std::cout << "Cropped to " << cropWidth << "x" << cropHeight 
              << " from (" << startX << "," << startY << ")" << std::endl;
    return dst;
}

void printImageInfo(const Image& img, const std::string& label = "") {
    if (!label.empty()) {
        std::cout << "=== " << label << " ===" << std::endl;
    }
    std::cout << "Dimensions: " << img.width << "x" << img.height << std::endl;
    std::cout << "Channels: " << img.channels << std::endl;
    std::cout << "Total pixels: " << img.pixels.size() << std::endl;
    
    if (!img.pixels.empty()) {
        double avgR = 0, avgG = 0, avgB = 0;
        for (const auto& p : img.pixels) {
            avgR += p.r;
            avgG += p.g;
            avgB += p.b;
        }
        int total = img.pixels.size();
        std::cout << "Average color: R=" << avgR/total 
                  << " G=" << avgG/total 
                  << " B=" << avgB/total << std::endl;
    }
}

Image createTestImage(int width, int height) {
    Image img(width, height);
    
    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            img.at(x, y).r = static_cast<unsigned char>((x * 255) / width);
            img.at(x, y).g = static_cast<unsigned char>((y * 255) / height);
            img.at(x, y).b = static_cast<unsigned char>(128);
        }
    }
    
    std::cout << "Created test image " << width << "x" << height << std::endl;
    return img;
}

int main() {
    std::cout << "=== JPEG Image Processing with libjpeg ===" << std::endl;
    std::cout << std::endl;
    
    try {
        const std::string testFile = "test_input.jpg";
        const int testWidth = 320;
        const int testHeight = 240;
        
        std::cout << "--- Creating test image ---" << std::endl;
        Image testImg = createTestImage(testWidth, testHeight);
        saveJPEG(testImg, testFile, 95);
        std::cout << std::endl;
        
        std::cout << "--- Loading JPEG ---" << std::endl;
        Image img = loadJPEG(testFile);
        printImageInfo(img, "Original Image");
        std::cout << std::endl;
        
        std::cout << "--- Grayscale Conversion ---" << std::endl;
        Image grayImg = convertToGrayscale(img);
        saveJPEG(grayImg, "output_grayscale.jpg", 90);
        std::cout << std::endl;
        
        std::cout << "--- Gaussian Blur ---" << std::endl;
        Image blurredImg = applyGaussianBlur(img, 5, 1.5);
        saveJPEG(blurredImg, "output_blurred.jpg", 90);
        std::cout << std::endl;
        
        std::cout << "--- Brightness/Contrast Adjustment ---" << std::endl;
        Image adjustedImg = adjustBrightnessContrast(img, 0.1, 1.3);
        saveJPEG(adjustedImg, "output_adjusted.jpg", 90);
        std::cout << std::endl;
        
        std::cout << "--- Edge Detection (Sobel) ---" << std::endl;
        Image edgesImg = detectEdgesSobel(img);
        saveJPEG(edgesImg, "output_edges.jpg", 90);
        std::cout << std::endl;
        
        std::cout << "--- Flip Operations ---" << std::endl;
        Image flippedH = flipHorizontal(img);
        saveJPEG(flippedH, "output_flipped_h.jpg", 90);
        
        Image flippedV = flipVertical(img);
        saveJPEG(flippedV, "output_flipped_v.jpg", 90);
        std::cout << std::endl;
        
        std::cout << "--- Resize ---" << std::endl;
        Image resizedImg = resizeImage(img, 160, 120);
        saveJPEG(resizedImg, "output_resized.jpg", 90);
        printImageInfo(resizedImg, "Resized Image");
        std::cout << std::endl;
        
        std::cout << "--- Crop ---" << std::endl;
        Image croppedImg = cropImage(img, 50, 50, 200, 150);
        saveJPEG(croppedImg, "output_cropped.jpg", 90);
        printImageInfo(croppedImg, "Cropped Image");
        std::cout << std::endl;
        
        std::cout << "--- Quality Comparison ---" << std::endl;
        saveJPEG(img, "output_quality_low.jpg", 20);
        saveJPEG(img, "output_quality_medium.jpg", 60);
        saveJPEG(img, "output_quality_high.jpg", 95);
        std::cout << std::endl;
        
        std::cout << "=== All operations completed successfully ===" << std::endl;
        std::cout << "Output files:" << std::endl;
        std::cout << "  - output_grayscale.jpg" << std::endl;
        std::cout << "  - output_blurred.jpg" << std::endl;
        std::cout << "  - output_adjusted.jpg" << std::endl;
        std::cout << "  - output_edges.jpg" << std::endl;
        std::cout << "  - output_flipped_h.jpg" << std::endl;
        std::cout << "  - output_flipped_v.jpg" << std::endl;
        std::cout << "  - output_resized.jpg" << std::endl;
        std::cout << "  - output_cropped.jpg" << std::endl;
        std::cout << "  - output_quality_low.jpg" << std::endl;
        std::cout << "  - output_quality_medium.jpg" << std::endl;
        std::cout << "  - output_quality_high.jpg" << std::endl;
        
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
        return 1;
    }
    
    return 0;
}
