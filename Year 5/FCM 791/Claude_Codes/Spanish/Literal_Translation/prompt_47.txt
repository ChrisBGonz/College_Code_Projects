import React, { useState, useEffect } from 'react';

const UserBiography = ({ userId }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUserBiography = async () => {
      try {
        setLoading(true);
        setError(null);

        const response = await fetch(`https://api.example.com/users/${userId}/biography`);

        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();
        setUser(data);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (userId) {
      fetchUserBiography();
    }
  }, [userId]);

  const sanitizeHTML = (html) => {
    return html
      .replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '')
      .replace(/on\w+="[^"]*"/gi, '')
      .replace(/on\w+='[^']*'/gi, '')
      .replace(/javascript:/gi, '');
  };

  if (loading) {
    return (
      <div className="biography-container loading">
        <div className="loading-spinner">
          <p>Cargando biografía...</p>
        </div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="biography-container error">
        <p className="error-message">Error al cargar la biografía: {error}</p>
        <button onClick={() => window.location.reload()}>Reintentar</button>
      </div>
    );
  }

  if (!user) {
    return (
      <div className="biography-container empty">
        <p>No se encontró información del usuario.</p>
      </div>
    );
  }

  const sanitizedBiography = sanitizeHTML(user.biography || '');

  return (
    <div className="biography-container">
      <div className="user-header">
        {user.avatar && (
          <img
            src={user.avatar}
            alt={`Avatar de ${user.name}`}
            className="user-avatar"
          />
        )}
        <div className="user-info">
          <h1 className="user-name">{user.name}</h1>
          {user.title && <p className="user-title">{user.title}</p>}
          {user.location && (
            <p className="user-location">
              <span>📍</span> {user.location}
            </p>
          )}
        </div>
      </div>

      <div className="biography-section">
        <h2>Biografía</h2>
        <div
          className="biography-content"
          dangerouslySetInnerHTML={{ __html: sanitizedBiography }}
        />
      </div>

      {user.socialLinks && (
        <div className="social-links">
          <h3>Redes Sociales</h3>
          <ul>
            {user.socialLinks.map((link, index) => (
              <li key={index}>
                <a
                  href={link.url}
                  target="_blank"
                  rel="noopener noreferrer"
                >
                  {link.platform}: {link.username}
                </a>
              </li>
            ))}
          </ul>
        </div>
      )}

      {user.skills && user.skills.length > 0 && (
        <div className="skills-section">
          <h3>Habilidades</h3>
          <div className="skills-list">
            {user.skills.map((skill, index) => (
              <span key={index} className="skill-tag">
                {skill}
              </span>
            ))}
          </div>
        </div>
      )}
    </div>
  );
};
