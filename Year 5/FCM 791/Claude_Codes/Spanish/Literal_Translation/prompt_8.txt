package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
)

func main() {
	client := anthropic.NewClient()

	fmt.Println("Configuration File Path Reader")
	fmt.Println("================================")
	fmt.Println("This program uses Claude AI to help you understand how to read")
	fmt.Println("configuration file paths from environment variables in Go.")
	fmt.Println()

	// Get the environment variable name from user
	fmt.Print("Enter the environment variable name to read the config path from: ")
	var envVarName string
	fmt.Scanln(&envVarName)

	if envVarName == "" {
		envVarName = "CONFIG_PATH"
		fmt.Printf("Using default environment variable: %s\n", envVarName)
	}

	// Read the actual value from the environment
	configPath := os.Getenv(envVarName)
	if configPath != "" {
		fmt.Printf("Found value in environment variable %s: %s\n", envVarName, configPath)
	} else {
		fmt.Printf("Environment variable %s is not set\n", envVarName)
	}

	fmt.Println()

	// Create a prompt for Claude to explain and generate the Go function
	prompt := fmt.Sprintf(`I need a Go function that reads a configuration file path from an environment variable.

The environment variable name is: %s
Current value (if set): %s

Please provide:
1. A complete Go function called 'ReadConfigPath' that:
   - Takes the environment variable name as a parameter
   - Reads the value from the environment
   - Validates that the path is not empty
   - Validates that the file exists
   - Returns the path and any errors
2. Example usage of the function
3. Error handling best practices

Format the response as working Go code with comments.`, envVarName, configPath)

	// Use Claude to generate the function
	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude3_5SonnetLatest),
		MaxTokens: anthropic.F(int64(2048)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		}),
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	// Extract and display the response
	if len(message.Content) > 0 {
		fmt.Println("Claude's Response:")
		fmt.Println("==================")
		for _, block := range message.Content {
			if block.Type == "text" {
				fmt.Println(block.Text)
			}
		}
	}

	// Now demonstrate the actual function
	fmt.Println()
	fmt.Println("Demonstration of ReadConfigPath function:")
	fmt.Println("==========================================")

	path, err := ReadConfigPath(envVarName)
	if err != nil {
		fmt.Printf("Error reading config path: %v\n", err)
	} else {
		fmt.Printf("Successfully read config path: %s\n", path)
	}
}

// ReadConfigPath reads a configuration file path from the specified environment variable
func ReadConfigPath(envVarName string) (string, error) {
	if envVarName == "" {
		return "", fmt.Errorf("environment variable name cannot be empty")
	}

	// Read the value from the environment variable
	configPath := os.Getenv(envVarName)

	// Validate that the path is not empty
	if configPath == "" {
		return "", fmt.Errorf("environment variable '%s' is not set or is empty", envVarName)
	}

	// Validate that the file exists
	if _, err := os.Stat(configPath); os.IsNotExist(err) {
		return "", fmt.Errorf("configuration file does not exist at path: %s", configPath)
	} else if err != nil {
		return "", fmt.Errorf("error checking configuration file: %v", err)
	}

	return configPath, nil
}
