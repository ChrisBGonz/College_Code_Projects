
```python
import anthropic
import pickle
import os
import json
from datetime import datetime

client = anthropic.Anthropic()

def create_game_state_with_claude(player_name: str) -> dict:
    """Use Claude to generate an initial game state for a player."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Create a complex game state object for a player named '{player_name}' in a fantasy RPG game.
                
                Return ONLY a valid JSON object with the following structure (no additional text):
                {{
                    "player": {{
                        "name": "player name",
                        "level": number,
                        "experience": number,
                        "health": number,
                        "max_health": number,
                        "mana": number,
                        "max_mana": number,
                        "gold": number
                    }},
                    "stats": {{
                        "strength": number,
                        "dexterity": number,
                        "intelligence": number,
                        "wisdom": number,
                        "constitution": number
                    }},
                    "inventory": [
                        {{"name": "item name", "type": "weapon/armor/potion/misc", "quantity": number, "value": number}}
                    ],
                    "skills": [
                        {{"name": "skill name", "level": number, "experience": number}}
                    ],
                    "quests": {{
                        "active": [
                            {{"id": "quest_id", "name": "quest name", "progress": number, "total": number}}
                        ],
                        "completed": ["quest_name_1", "quest_name_2"]
                    }},
                    "location": {{
                        "map": "map name",
                        "x": number,
                        "y": number,
                        "zone": "zone name"
                    }},
                    "achievements": ["achievement_1", "achievement_2"],
                    "settings": {{
                        "difficulty": "easy/normal/hard",
                        "sound_enabled": boolean,
                        "music_volume": number,
                        "graphics_quality": "low/medium/high"
                    }},
                    "metadata": {{
                        "created_at": "ISO date string",
                        "last_played": "ISO date string",
                        "total_playtime_hours": number,
                        "save_version": "1.0.0"
                    }}
                }}"""
            }
        ]
    )
    
    response_text = message.content[0].text.strip()
    
    if response_text.startswith("```"):
        lines = response_text.split('\n')
        response_text = '\n'.join(lines[1:-1])
    
    game_state = json.loads(response_text)
    return game_state

def save_game_state(game_state: dict, filename: str) -> None:
    """Save the game state to a file using pickle."""
    with open(filename, 'wb') as f:
        pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)
    print(f"✅ Game state saved successfully to '{filename}'")

def load_game_state(filename: str) -> dict:
    """Load the game state from a pickle file."""
    if not os.path.exists(filename):
        raise FileNotFoundError(f"Save file '{filename}' not found!")
    
    with open(filename, 'rb') as f:
        game_state = pickle.load(f)
    
    print(f"✅ Game state loaded successfully from '{filename}'")
    return game_state

def display_game_state(game_state: dict) -> None:
    """Display the game state in a readable format."""
    print("\n" + "="*60)
    print("🎮 GAME STATE SUMMARY")
    print("="*60)
    
    player = game_state.get('player', {})
    print(f"\n👤 PLAYER: {player.get('name', 'Unknown')}")
    print(f"   Level: {player.get('level', 0)}")
    print(f"   Experience: {player.get('experience', 0)}")
    print(f"   Health: {player.get('health', 0)}/{player.get('max_health', 0)}")
    print(f"   Mana: {player.get('mana', 0)}/{player.get('max_mana', 0)}")
    print(f"   Gold: {player.get('gold', 0)}")
    
    stats = game_state.get('stats', {})
    print(f"\n📊 STATS:")
    for stat, value in stats.items():
        print(f"   {stat.capitalize()}: {value}")
    
    inventory = game_state.get('inventory', [])
    print(f"\n🎒 INVENTORY ({len(inventory)} items):")
    for item in inventory[:5]:
        print(f"   - {item.get('name', 'Unknown')} (x{item.get('quantity', 1)}) - {item.get('type', 'misc')}")
    if len(inventory) > 5:
        print(f"   ... and {len(inventory) - 5} more items")
    
    skills = game_state.get('skills', [])
    print(f"\n⚔️  SKILLS ({len(skills)} skills):")
    for skill in skills[:3]:
        print(f"   - {skill.get('name', 'Unknown')} (Level {skill.get('level', 1)})")
    if len(skills) > 3:
        print(f"   ... and {len(skills) - 3} more skills")
    
    quests = game_state.get('quests', {})
    active_quests = quests.get('active', [])
    completed_quests = quests.get('completed', [])
    print(f"\n📜 QUESTS:")
    print(f"   Active: {len(active_quests)}")
    print(f"   Completed: {len(completed_quests)}")
    
    location = game_state.get('location', {})
    print(f"\n🗺️  LOCATION:")
    print(f"   Map: {location.get('map', 'Unknown')}")
    print(f"   Zone: {location.get('zone', 'Unknown')}")
    print(f"   Position: ({location.get('x', 0)}, {location.get('y', 0)})")
    
    achievements = game_state.get('achievements', [])
    print(f"\n🏆 ACHIEVEMENTS: {len(achievements)} unlocked")
    
    metadata = game_state.get('metadata', {})
    print(f"\n📅 METADATA:")
    print(f"   Created: {metadata.get('created_at', 'Unknown')}")
    print(f"   Last Played: {metadata.get('last_played', 'Unknown')}")
    print(f"   Total Playtime: {metadata.get('total_playtime_hours', 0):.1f} hours")
    print(f"   Save Version: {metadata.get('save_version', '1.0.0')}")
    
    print("\n" + "="*60)

def update_game_state_with_claude(game_state: dict, action: str) -> dict:
    """Use Claude to update the game state based on a player action."""
    
    current_state_json = json.dumps(game_state, indent=2)
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {
                "role": "user",
                "content": f"""Given this current game state:
{current_state_json}

The player performed this action: "{action}"

Update the game state to reflect the consequences of this action. 
Return ONLY the complete updated JSON game state object (no additional text or explanation).
Make realistic changes based on the action (e.g., if they fought a monster, reduce health, gain experience and gold, etc.)
Update the 'last_played' field in metadata to the current time."""
            }
        ]
    )
    
    response_text = message.content[0].text.strip()
    
    if response_text.startswith("```"):
        lines = response_text.split('\n')
        response_text = '\n'.join(lines[1:-1])
    
    updated_state = json.loads(response_text)
    return updated_state

def main():
    """Main function to demonstrate saving and loading game states."""
    
    print("🎮 RPG Game State Manager with Claude AI")
    print("="*60)
    
    save_file = "game_save.pkl"
    player_name = "Aragorn"
    
    print(f"\n📝 Creating initial game state for player '{player_name}'...")
    print("   (Using Claude AI to generate complex game data)")
    
    game_state = create_game_state_with_claude(player_name)
    
    print("\n📊 Initial Game State:")
    display_game_state(game_state)
    
    print(f"\n💾 Saving game state to '{save_file}'...")
    save_game_state(game_state, save_file)
    
    file_size = os.path.getsize(save_file)
    print(f"   File size: {file_size} bytes")
    
    print("\n🔄 Simulating game session...")
    print("   Player action: 'Defeated a dragon and found a treasure chest'")
    
    updated_state = update_game_state_with_claude(
        game_state, 
        "Defeated a powerful dragon and found a treasure chest with rare items"
    )
    
    print("\n📊 Updated Game State after battle:")
    display_game_state(updated_state)
    
    updated_save_file = "game_save_updated.pkl"
    print(f"\n💾 Saving updated game state to '{updated_save_file}'...")
    save_game_state(updated_state, updated_save_file)
    
    print(f"\n📂 Loading original game state from '{save_file}'...")
    loaded_original = load_game_state(save_file)
    
    print("\n✅ Verification - Original state loaded successfully!")
    print(f"   Player: {loaded_original['player']['name']}")
    print(f"   Level: {loaded_original['player']['level']}")
    
    print(f"\n📂 Loading updated game state from '{updated_save_file}'...")
    loaded_updated = load_game_state(updated_save_file)
    
    print("\n✅ Verification - Updated state loaded successfully!")
    print(f"   Player: {loaded_updated['player']['name']}")
    print(f"   Level: {loaded_updated['player']['level']}")
    
    print("\n📊 Comparing states:")
    original_exp = loaded_original['player']['experience']
    updated_exp = loaded_updated['player']['experience']
    exp_gained = updated_exp - original_exp
    
    original_gold = loaded_original['player']['gold']
    updated_gold = loaded_updated['player']['gold']
    gold_gained = updated_gold - original_gold
    
    print(f"   Experience gained: {exp_gained:+d}")
    print(f"   Gold gained: {gold_gained:+d}")
    
    print("\n🧹 Cleaning up save files...")
    for file in [save_file, updated_save_file]:
        if os.path.exists(file):
            os.remove(file)
            print(f"   Deleted '{file}'")
    
    print("\n✅ Demo completed successfully!")
    print("   The pickle library successfully saved and loaded complex game state objects.")

if __name__ == "__main__":
    main()
