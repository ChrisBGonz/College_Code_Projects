@app.route('/upload-yaml', methods=['POST'])
def upload_yaml():
    global user_profile, conversation_history
    
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'No se encontró archivo en la solicitud'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'No se seleccionó ningún archivo'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'El archivo debe ser YAML (.yaml o .yml)'})
    
    try:
        content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'El archivo YAML debe contener un objeto/diccionario'})
        
        user_profile = profile_data
        
        # Reset conversation history when new profile is loaded
        conversation_history = []
        
        # Create initial context message for Claude
        profile_summary = yaml.dump(profile_data, allow_unicode=True, default_flow_style=False)
        system_context = f"""Eres un asistente de configuración de perfiles. El usuario ha cargado el siguiente perfil YAML:

{profile_summary}

Ayuda al usuario a entender su configuración, responde preguntas sobre su perfil y proporciona sugerencias para mejorar o completar la configuración. Responde siempre en español."""
        
        # Get initial analysis from Claude
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_context,
            messages=[{
                "role": "user",
                "content": "Analiza mi perfil y dame un resumen de lo que has encontrado."
            }]
        )
        
        analysis = response.content[0].text
        
        # Store the system context for future conversations
        conversation_history.append({
            "role": "user",
            "content": "Analiza mi perfil y dame un resumen de lo que has encontrado."
        })
        conversation_history.append({
            "role": "assistant",
            "content": analysis
        })
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'message': analysis
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Error al parsear YAML: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error inesperado: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    global conversation_history, user_profile
    
    data = request.get_json()
    user_message = data.get('message', '')
    
    if not user_message:
        return jsonify({'success': False, 'error': 'Mensaje vacío'})
    
    try:
        # Build system message with profile context
        if user_profile:
            profile_summary = yaml.dump(user_profile, allow_unicode=True, default_flow_style=False)
            system_message = f"""Eres un asistente de configuración de perfiles. El usuario tiene el siguiente perfil YAML configurado:

{profile_summary}

Ayuda al usuario a entender su configuración, responde preguntas sobre su perfil y proporciona sugerencias útiles. Responde siempre en español de manera clara y concisa."""
        else:
            system_message = """Eres un asistente de configuración de perfiles. El usuario aún no ha cargado un perfil YAML. 
Ayúdale a entender cómo crear y cargar un archivo YAML para configurar su perfil. Responde siempre en español."""
        
        # Add user message to history
        conversation_history.append({
            "role": "user",
            "content": user_message
        })
        
        # Send to Claude with conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=system_message,
            messages=conversation_history
        )
        
        assistant_response = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_response
        })
        
        # Keep conversation history manageable (last 20 messages)
        if len(conversation_history) > 20:
            conversation_history = conversation_history[-20:]
        
        return jsonify({
            'success': True,
            'response': assistant_response
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error al procesar mensaje: {str(e)}'})

@app.route('/clear-chat', methods=['POST'])
def clear_chat():
    global conversation_history
    conversation_history = []
    return jsonify({'success': True})

@app.route('/get-profile', methods=['GET'])
def get_profile():
    return jsonify({'success': True, 'profile': user_profile})

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
