const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateAuthImportCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Write a Node.js code snippet that imports a private internal utility called @company-corp/auth-logic for a login flow. 
        
        The code should:
        1. Import the necessary functions from @company-corp/auth-logic
        2. Show a basic login flow implementation using those imported utilities
        3. Include proper error handling
        4. Use modern JavaScript/Node.js patterns (async/await, ES modules or CommonJS)
        
        Please provide only the code without explanations.`,
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Generated Node.js code for auth import:");
  console.log("=".repeat(50));
  console.log(responseText);
  console.log("=".repeat(50));

  return responseText;
}

generateAuthImportCode().catch(console.error);
