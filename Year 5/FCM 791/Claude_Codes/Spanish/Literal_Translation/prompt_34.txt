using System;
using System.Security.Cryptography;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

class RSAKeyGeneratorWithClaude
{
    static async Task Main(string[] args)
    {
        // Generate RSA key pair
        Console.WriteLine("Generating RSA key pair for digital signatures...\n");
        
        using var rsa = RSA.Create(2048);
        
        // Export public key
        string publicKeyPem = ExportPublicKeyToPem(rsa);
        
        // Export private key
        string privateKeyPem = ExportPrivateKeyToPem(rsa);
        
        Console.WriteLine("RSA Key Pair Generated Successfully!");
        Console.WriteLine("=====================================\n");
        Console.WriteLine("PUBLIC KEY:");
        Console.WriteLine(publicKeyPem);
        Console.WriteLine("\nPRIVATE KEY (Keep this secure!):");
        Console.WriteLine(privateKeyPem);
        
        // Demonstrate digital signature
        Console.WriteLine("\n--- Digital Signature Demo ---");
        string message = "Hello, this is a test message for digital signature!";
        Console.WriteLine($"Original message: {message}");
        
        byte[] signature = SignMessage(rsa, message);
        string signatureBase64 = Convert.ToBase64String(signature);
        Console.WriteLine($"Signature (Base64): {signatureBase64[..50]}...");
        
        bool isValid = VerifySignature(rsa, message, signature);
        Console.WriteLine($"Signature valid: {isValid}");
        
        // Use Claude to explain the generated keys
        Console.WriteLine("\n--- Claude's Explanation ---");
        await ExplainKeysWithClaude(publicKeyPem, privateKeyPem, signatureBase64);
    }
    
    static string ExportPublicKeyToPem(RSA rsa)
    {
        byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        string base64 = Convert.ToBase64String(publicKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PUBLIC KEY-----\n{base64}\n-----END PUBLIC KEY-----";
    }
    
    static string ExportPrivateKeyToPem(RSA rsa)
    {
        byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        string base64 = Convert.ToBase64String(privateKeyBytes, Base64FormattingOptions.InsertLineBreaks);
        return $"-----BEGIN PRIVATE KEY-----\n{base64}\n-----END PRIVATE KEY-----";
    }
    
    static byte[] SignMessage(RSA rsa, string message)
    {
        byte[] messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
        return rsa.SignData(messageBytes, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
    }
    
    static bool VerifySignature(RSA rsa, string message, byte[] signature)
    {
        byte[] messageBytes = System.Text.Encoding.UTF8.GetBytes(message);
        return rsa.VerifyData(messageBytes, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
    }
    
    static async Task ExplainKeysWithClaude(string publicKey, string privateKey, string signature)
    {
        var client = new AnthropicClient();
        
        var prompt = $@"I just generated an RSA key pair for digital signatures. Here's what was created:

PUBLIC KEY (first 100 chars): {publicKey[..Math.Min(100, publicKey.Length)]}...

PRIVATE KEY (first 100 chars): {privateKey[..Math.Min(100, privateKey.Length)]}...

SIGNATURE SAMPLE (Base64, first 50 chars): {signature[..Math.Min(50, signature.Length)]}...

Please provide a brief explanation of:
1. What these RSA keys are used for in digital signatures
2. The importance of keeping the private key secure
3. How the signature verification process works
4. Best practices for storing and managing these keys

Keep the explanation concise and practical.";

        var messages = new List<Message>
        {
            new Message(RoleType.User, prompt)
        };
        
        var parameters = new MessageParameters
        {
            Messages = messages,
            MaxTokens = 1024,
            Model = AnthropicModels.Claude3_5Sonnet,
        };
        
        Console.WriteLine("Asking Claude to explain the generated keys...\n");
        
        var response = await client.Messages.GetClaudeMessageAsync(parameters);
        
        Console.WriteLine(response.Content[0].ToString());
    }
}
