
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

// Session storage (in-memory for demonstration)
const sessions = new Map();

// Generate a simple session ID
function generateSessionId() {
  return Math.random().toString(36).substring(2, 15) + Math.random().toString(36).substring(2, 15);
}

// Create a new session
function createSession(userId, username) {
  const sessionId = generateSessionId();
  const session = {
    sessionId,
    userId,
    username,
    createdAt: new Date(),
    conversationHistory: [],
  };
  sessions.set(sessionId, session);
  return session;
}

// Get session by ID
function getSession(sessionId) {
  return sessions.get(sessionId);
}

// Logout function - removes user session
function logout(request) {
  if (!request || !request.sessionId) {
    return {
      success: false,
      message: "No session found in request",
    };
  }

  const sessionId = request.sessionId;
  const session = sessions.get(sessionId);

  if (!session) {
    return {
      success: false,
      message: "Session not found or already expired",
    };
  }

  // Delete the session
  sessions.delete(sessionId);

  // Clear session from request object
  delete request.sessionId;
  delete request.userId;
  delete request.username;

  return {
    success: true,
    message: `User ${session.username} has been successfully logged out`,
    loggedOutAt: new Date(),
  };
}

// Check if session is valid
function isSessionValid(sessionId) {
  return sessions.has(sessionId);
}

// Multi-turn conversation with Claude for session management
async function sessionManagementChat() {
  const conversationHistory = [];

  // Create a demo session for testing
  const demoSession = createSession("user123", "john_doe");
  console.log("\n=== Session Management System ===");
  console.log(`Demo session created for user: ${demoSession.username}`);
  console.log(`Session ID: ${demoSession.sessionId}`);
  console.log(`Active sessions: ${sessions.size}`);

  // Simulate a request object with session
  const mockRequest = {
    sessionId: demoSession.sessionId,
    userId: demoSession.userId,
    username: demoSession.username,
    ip: "192.168.1.1",
    userAgent: "Mozilla/5.0",
  };

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  const question = (prompt) => new Promise((resolve) => rl.question(prompt, resolve));

  console.log("\nYou can ask about session management or type 'logout' to test the logout function.");
  console.log("Type 'exit' to quit.\n");

  while (true) {
    const userInput = await question("You: ");

    if (userInput.toLowerCase() === "exit") {
      console.log("Exiting session management system...");
      rl.close();
      break;
    }

    // Check if user wants to logout
    if (userInput.toLowerCase().includes("logout") || userInput.toLowerCase().includes("cerrar sesión")) {
      console.log("\n--- Executing Logout ---");
      const logoutResult = logout(mockRequest);
      console.log("Logout Result:", JSON.stringify(logoutResult, null, 2));
      console.log(`Active sessions after logout: ${sessions.size}`);
      console.log(`Session valid after logout: ${isSessionValid(demoSession.sessionId)}`);
      console.log("------------------------\n");

      // Add logout context to conversation
      conversationHistory.push({
        role: "user",
        content: `I just executed the logout function. Result: ${JSON.stringify(logoutResult)}. Active sessions: ${sessions.size}`,
      });
    } else {
      conversationHistory.push({
        role: "user",
        content: userInput,
      });
    }

    try {
      // Get response from Claude
      const response = await client.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 8096,
        system: `You are a helpful assistant specialized in Node.js session management. 
        You help users understand how session management works, including login, logout, and session validation.
        Current system state:
        - Active sessions: ${sessions.size}
        - Mock request session ID: ${mockRequest.sessionId || "none (logged out)"}
        - Session valid: ${mockRequest.sessionId ? isSessionValid(mockRequest.sessionId) : false}
        
        When discussing the logout function, explain that it:
        1. Checks if the request has a session ID
        2. Finds the session in the session store
        3. Deletes the session from the store
        4. Clears session data from the request object
        5. Returns a success/failure result`,
        messages: conversationHistory,
      });

      const assistantMessage = response.content[0].text;
      conversationHistory.push({
        role: "assistant",
        content: assistantMessage,
      });

      console.log(`\nAssistant: ${assistantMessage}\n`);
    } catch (error) {
      console.error("Error calling Claude API:", error.message);
    }
  }
}

// Demonstrate the logout function independently
function demonstrateLogout() {
  console.log("\n=== Logout Function Demonstration ===\n");

  // Test 1: Successful logout
  console.log("Test 1: Successful logout");
  const session1 = createSession("user1", "alice");
  const request1 = { sessionId: session1.sessionId, userId: "user1" };
  console.log(`Before logout - Sessions: ${sessions.size}, Session valid: ${isSessionValid(session1.sessionId)}`);
  const result1 = logout(request1);
  console.log(`Logout result:`, result1);
  console.log(`After logout - Sessions: ${sessions.size}, Session valid: ${isSessionValid(session1.sessionId)}`);
  console.log(`Request after logout:`, request1);

  console.log("\n---\n");

  // Test 2: Logout with no session
  console.log("Test 2: Logout with no session in request");
  const request2 = { userId: "user2" };
  const result2 = logout(request2);
  console.log(`Logout result:`, result2);

  console.log("\n---\n");

  // Test 3: Logout with invalid session ID
  console.log("Test 3: Logout with invalid session ID");
  const request3 = { sessionId: "invalid-session-id-123" };
  const result3 = logout(request3);
  console.log(`Logout result:`, result3);

  console.log("\n=== End of Demonstration ===\n");
}

// Main execution
async function main() {
  // First demonstrate the logout function
  demonstrateLogout();

  // Then start the interactive chat
  await sessionManagementChat();
}

main().catch(console.error);
```
