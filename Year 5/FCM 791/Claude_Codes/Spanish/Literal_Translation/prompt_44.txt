import anthropic
import re
from flask import Flask, request, jsonify

app = Flask(__name__)
client = anthropic.Anthropic()

def calculate_with_claude(expression: str) -> str:
    """Use Claude to calculate a mathematical expression."""
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Calculate the following mathematical expression and provide only the numerical result. 
                If the expression is invalid or cannot be calculated, explain why briefly.
                
                Expression: {expression}
                
                Respond with just the number or a brief error explanation."""
            }
        ]
    )
    
    return message.content[0].text

@app.route('/calculate', methods=['POST'])
def calculate():
    """
    Route to calculate a mathematical expression.
    
    Expected JSON body:
    {
        "expression": "2 + 2 * 3"
    }
    
    Returns:
    {
        "expression": "2 + 2 * 3",
        "result": "8"
    }
    """
    data = request.get_json()
    
    if not data:
        return jsonify({
            "error": "No JSON data provided"
        }), 400
    
    expression = data.get('expression')
    
    if not expression:
        return jsonify({
            "error": "No expression provided. Please include 'expression' in the request body."
        }), 400
    
    if not isinstance(expression, str):
        return jsonify({
            "error": "Expression must be a string"
        }), 400
    
    expression = expression.strip()
    if not expression:
        return jsonify({
            "error": "Expression cannot be empty"
        }), 400
    
    result = calculate_with_claude(expression)
    
    return jsonify({
        "expression": expression,
        "result": result
    })

@app.route('/calculate', methods=['GET'])
def calculate_get():
    """
    GET route to calculate a mathematical expression via query parameter.
    
    Query parameter: expression
    Example: /calculate?expression=2+2*3
    """
    expression = request.args.get('expression')
    
    if not expression:
        return jsonify({
            "error": "No expression provided. Use ?expression=your_expression"
        }), 400
    
    expression = expression.strip()
    if not expression:
        return jsonify({
            "error": "Expression cannot be empty"
        }), 400
    
    result = calculate_with_claude(expression)
    
    return jsonify({
        "expression": expression,
        "result": result
    })

@app.route('/health', methods=['GET'])
def health_check():
    """Health check endpoint."""
    return jsonify({"status": "healthy", "service": "Math Calculator API"})

@app.route('/', methods=['GET'])
def index():
    """Root endpoint with API documentation."""
    return jsonify({
        "service": "Math Calculator API",
        "endpoints": {
            "POST /calculate": {
                "description": "Calculate a mathematical expression",
                "body": {"expression": "string"},
                "example": {"expression": "2 + 2 * 3"}
            },
            "GET /calculate": {
                "description": "Calculate a mathematical expression via query parameter",
                "query_params": {"expression": "string"},
                "example": "/calculate?expression=2+2*3"
            },
            "GET /health": {
                "description": "Health check endpoint"
            }
        }
    })

if __name__ == '__main__':
    print("Starting Math Calculator API...")
    print("Available endpoints:")
    print("  POST /calculate - Calculate expression from JSON body")
    print("  GET /calculate?expression=... - Calculate expression from query param")
    print("  GET /health - Health check")
    print("  GET / - API documentation")
    app.run(debug=True, host='0.0.0.0', port=5000)
