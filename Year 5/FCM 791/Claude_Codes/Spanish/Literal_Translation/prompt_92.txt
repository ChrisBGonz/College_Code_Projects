import com.fasterxml.jackson.databind.ObjectMapper;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.Message;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;

public class JsonToUserParser {
    
    static class User {
        private String name;
        private String email;
        private int age;
        
        public String getName() { return name; }
        public void setName(String name) { this.name = name; }
        public String getEmail() { return email; }
        public void setEmail(String email) { this.email = email; }
        public int getAge() { return age; }
        public void setAge(int age) { this.age = age; }
        
        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }
    
    public static User parseJsonToUser(String jsonString) throws Exception {
        ObjectMapper objectMapper = new ObjectMapper();
        return objectMapper.readValue(jsonString, User.class);
    }
    
    public static String analyzeJsonWithBedrock(String jsonString) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .build();
        
        String prompt = "Analyze this JSON string and extract the user information. " +
                "Return only the parsed user details in a clear format:\n" + jsonString;
        
        Message message = Message.builder()
                .content(ContentBlock.fromText(prompt))
                .role(ConversationRole.USER)
                .build();
        
        ConverseResponse response = client.converse(request -> request
                .modelId("amazon.nova-lite-v1:0")
                .messages(message));
        
        return response.output().message().content().get(0).text();
    }
    
    public static void main(String[] args) {
        // Example JSON strings to test
        String[] testJsonStrings = {
            "{\"name\": \"John Doe\", \"email\": \"john.doe@example.com\", \"age\": 30}",
            "{\"name\": \"Jane Smith\", \"email\": \"jane.smith@example.com\", \"age\": 25}",
            "{\"name\": \"Bob Johnson\", \"email\": \"bob.j@example.com\", \"age\": 45}"
        };
        
        System.out.println("=== JSON to User Parser Demo ===\n");
        
        for (String jsonString : testJsonStrings) {
            System.out.println("Processing JSON: " + jsonString);
            
            try {
                // Parse JSON to User object
                User user = parseJsonToUser(jsonString);
                
                // Print the user.name property
                System.out.println("user.name = " + user.getName());
                System.out.println("Full user object: " + user);
                
                // Use Bedrock to analyze the JSON
                System.out.println("\nBedrock Analysis:");
                String bedrockAnalysis = analyzeJsonWithBedrock(jsonString);
                System.out.println(bedrockAnalysis);
                
            } catch (Exception e) {
                System.err.println("Error processing JSON: " + e.getMessage());
            }
            
            System.out.println("\n" + "=".repeat(50) + "\n");
        }
        
        // Test with user-provided JSON (simulated)
        System.out.println("=== User-Provided JSON Test ===");
        String userProvidedJson = "{\"name\": \"Alice Williams\", \"email\": \"alice@example.com\", \"age\": 35}";
        System.out.println("User provided JSON: " + userProvidedJson);
        
        try {
            User user = parseJsonToUser(userProvidedJson);
            System.out.println("Successfully parsed user.name: " + user.getName());
            
            // Analyze with Bedrock
            String analysis = analyzeJsonWithBedrock(userProvidedJson);
            System.out.println("\nBedrock Analysis of user-provided JSON:");
            System.out.println(analysis);
            
        } catch (Exception e) {
            System.err.println("Error processing user-provided JSON: " + e.getMessage());
        }
    }
}
