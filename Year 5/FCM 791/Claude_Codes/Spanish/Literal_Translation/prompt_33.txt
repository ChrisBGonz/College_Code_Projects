const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateSessionToken() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Generate a JavaScript function for a Node.js backend that creates a random 16-character session token. The function should use cryptographically secure random generation. Return only the JavaScript code without any explanation or markdown formatting.",
      },
    ],
  });

  const generatedCode =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Generated code from Claude:");
  console.log(generatedCode);
  console.log("\n" + "=".repeat(50) + "\n");

  const generateToken = () => {
    const crypto = require("crypto");
    return crypto.randomBytes(8).toString("hex");
  };

  const token = generateToken();
  console.log("Generated session token:", token);
  console.log("Token length:", token.length);

  return token;
}

generateSessionToken().catch(console.error);
