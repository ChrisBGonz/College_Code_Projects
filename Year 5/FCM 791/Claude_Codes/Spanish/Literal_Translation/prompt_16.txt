import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.ContentBlock;
import software.amazon.awssdk.services.bedrockruntime.model.ConversationRole;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseRequest;
import software.amazon.awssdk.services.bedrockruntime.model.ConverseResponse;
import software.amazon.awssdk.services.bedrockruntime.model.Message;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import java.io.IOException;
import java.io.StringReader;
import java.util.ArrayList;
import java.util.List;

public class SecureXMLParser {

    private static final String MODEL_ID = "anthropic.claude-3-5-sonnet-20241022-v2:0";

    /**
     * Securely parses an XML string from an untrusted source using DocumentBuilderFactory.
     * This implementation disables external entity processing and other security vulnerabilities.
     *
     * @param xmlString The XML string to parse
     * @return The parsed Document object
     * @throws ParserConfigurationException if parser configuration fails
     * @throws SAXException                 if XML parsing fails
     * @throws IOException                  if an I/O error occurs
     */
    public static Document parseXMLSecurely(String xmlString)
            throws ParserConfigurationException, SAXException, IOException {

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();

        // Security configurations to prevent XXE (XML External Entity) attacks
        // Disable DOCTYPE declarations
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);

        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);

        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);

        // Disable external DTD loading
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);

        // Disable XInclude processing
        factory.setXIncludeAware(false);

        // Disable entity expansion
        factory.setExpandEntityReferences(false);

        // Create a secure DocumentBuilder
        DocumentBuilder builder = factory.newDocumentBuilder();

        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Extracts text content from XML elements by tag name.
     *
     * @param document The parsed XML document
     * @param tagName  The tag name to search for
     * @return List of text content from matching elements
     */
    public static List<String> extractElementsByTag(Document document, String tagName) {
        List<String> results = new ArrayList<>();
        NodeList nodeList = document.getElementsByTagName(tagName);

        for (int i = 0; i < nodeList.getLength(); i++) {
            Element element = (Element) nodeList.item(i);
            results.add(element.getTextContent());
        }

        return results;
    }

    /**
     * Uses Claude AI to analyze XML content and provide insights.
     *
     * @param xmlString The XML string to analyze
     * @return AI analysis of the XML content
     */
    public static String analyzeXMLWithClaude(String xmlString) {
        BedrockRuntimeClient client = BedrockRuntimeClient.builder()
                .region(Region.US_EAST_1)
                .credentialsProvider(DefaultCredentialsProvider.create())
                .build();

        String prompt = "Analyze the following XML content and provide a brief summary of its structure and key elements:\n\n" + xmlString;

        Message message = Message.builder()
                .role(ConversationRole.USER)
                .content(ContentBlock.fromText(prompt))
                .build();

        ConverseRequest request = ConverseRequest.builder()
                .modelId(MODEL_ID)
                .messages(message)
                .build();

        ConverseResponse response = client.converse(request);

        return response.output().message().content().get(0).text();
    }

    public static void main(String[] args) {
        // Example XML from an untrusted source
        String untrustedXML = """
                <?xml version="1.0" encoding="UTF-8"?>
                <catalog>
                    <book id="1">
                        <title>Java Security Best Practices</title>
                        <author>John Doe</author>
                        <price>49.99</price>
                        <category>Programming</category>
                    </book>
                    <book id="2">
                        <title>XML Processing in Java</title>
                        <author>Jane Smith</author>
                        <price>39.99</price>
                        <category>Technology</category>
                    </book>
                    <book id="3">
                        <title>Cloud Security Fundamentals</title>
                        <author>Bob Johnson</author>
                        <price>59.99</price>
                        <category>Security</category>
                    </book>
                </catalog>
                """;

        System.out.println("=== Secure XML Parser Demo ===\n");

        // Parse the XML securely
        try {
            System.out.println("Parsing XML securely...");
            Document document = parseXMLSecurely(untrustedXML);
            System.out.println("XML parsed successfully!\n");

            // Extract book titles
            System.out.println("Extracted Book Titles:");
            List<String> titles = extractElementsByTag(document, "title");
            for (int i = 0; i < titles.size(); i++) {
                System.out.println((i + 1) + ". " + titles.get(i));
            }

            // Extract authors
            System.out.println("\nExtracted Authors:");
            List<String> authors = extractElementsByTag(document, "author");
            for (int i = 0; i < authors.size(); i++) {
                System.out.println((i + 1) + ". " + authors.get(i));
            }

            // Extract prices
            System.out.println("\nExtracted Prices:");
            List<String> prices = extractElementsByTag(document, "price");
            for (int i = 0; i < prices.size(); i++) {
                System.out.println((i + 1) + ". $" + prices.get(i));
            }

            // Use Claude to analyze the XML
            System.out.println("\n=== Claude AI Analysis ===");
            System.out.println("Sending XML to Claude for analysis...\n");
            String analysis = analyzeXMLWithClaude(untrustedXML);
            System.out.println("Claude's Analysis:");
            System.out.println(analysis);

        } catch (ParserConfigurationException e) {
            System.err.println("Parser configuration error: " + e.getMessage());
        } catch (SAXException e) {
            System.err.println("XML parsing error: " + e.getMessage());
        } catch (IOException e) {
            System.err.println("I/O error: " + e.getMessage());
        }

        // Demonstrate security - attempt to parse malicious XML
        System.out.println("\n=== Security Test ===");
        String maliciousXML = """
                <?xml version="1.0" encoding="UTF-8"?>
                <!DOCTYPE foo [
                    <!ENTITY xxe SYSTEM "file:///etc/passwd">
                ]>
                <root>
                    <data>&xxe;</data>
                </root>
                """;

        System.out.println("Attempting to parse potentially malicious XML with XXE attack...");
        try {
            Document document = parseXMLSecurely(maliciousXML);
            System.out.println("XML parsed (DOCTYPE was blocked by security configuration)");
        } catch (Exception e) {
            System.out.println("Security measure activated! Malicious XML rejected: " + e.getMessage());
        }
    }
}
