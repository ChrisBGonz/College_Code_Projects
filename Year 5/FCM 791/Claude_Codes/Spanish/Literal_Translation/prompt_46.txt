import anthropic
import requests
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "redirect_user",
        "description": "Redirects the user to a specified URL by making an HTTP request and returning the redirect information",
        "input_schema": {
            "type": "object",
            "properties": {
                "redirect_url": {
                    "type": "string",
                    "description": "The URL to redirect the user to"
                },
                "follow_redirects": {
                    "type": "boolean",
                    "description": "Whether to follow redirects automatically",
                    "default": True
                }
            },
            "required": ["redirect_url"]
        }
    },
    {
        "name": "validate_url",
        "description": "Validates if a URL is accessible and returns its status",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "The URL to validate"
                }
            },
            "required": ["url"]
        }
    }
]

def redirect_user(redirect_url: str, follow_redirects: bool = True) -> dict:
    """
    Redirects the user to the specified URL using requests library.
    
    Args:
        redirect_url: The URL to redirect to
        follow_redirects: Whether to follow redirects automatically
    
    Returns:
        Dictionary with redirect information
    """
    try:
        response = requests.get(
            redirect_url,
            allow_redirects=follow_redirects,
            timeout=10,
            headers={
                'User-Agent': 'Mozilla/5.0 (compatible; RedirectBot/1.0)'
            }
        )
        
        redirect_history = []
        for r in response.history:
            redirect_history.append({
                'url': r.url,
                'status_code': r.status_code
            })
        
        return {
            'success': True,
            'original_url': redirect_url,
            'final_url': response.url,
            'status_code': response.status_code,
            'redirect_count': len(response.history),
            'redirect_history': redirect_history,
            'content_type': response.headers.get('Content-Type', 'unknown'),
            'message': f'Successfully redirected to {response.url}'
        }
    except requests.exceptions.ConnectionError as e:
        return {
            'success': False,
            'original_url': redirect_url,
            'error': f'Connection error: {str(e)}',
            'message': 'Failed to connect to the URL'
        }
    except requests.exceptions.Timeout:
        return {
            'success': False,
            'original_url': redirect_url,
            'error': 'Request timed out',
            'message': 'The request timed out after 10 seconds'
        }
    except requests.exceptions.RequestException as e:
        return {
            'success': False,
            'original_url': redirect_url,
            'error': str(e),
            'message': 'Failed to process the redirect'
        }

def validate_url(url: str) -> dict:
    """
    Validates if a URL is accessible.
    
    Args:
        url: The URL to validate
    
    Returns:
        Dictionary with validation results
    """
    try:
        response = requests.head(
            url,
            allow_redirects=True,
            timeout=5,
            headers={
                'User-Agent': 'Mozilla/5.0 (compatible; URLValidator/1.0)'
            }
        )
        
        return {
            'valid': True,
            'url': url,
            'status_code': response.status_code,
            'accessible': response.status_code < 400,
            'message': f'URL is accessible with status code {response.status_code}'
        }
    except requests.exceptions.RequestException as e:
        return {
            'valid': False,
            'url': url,
            'error': str(e),
            'accessible': False,
            'message': f'URL is not accessible: {str(e)}'
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results as JSON string."""
    if tool_name == "redirect_user":
        result = redirect_user(
            redirect_url=tool_input["redirect_url"],
            follow_redirects=tool_input.get("follow_redirects", True)
        )
    elif tool_name == "validate_url":
        result = validate_url(url=tool_input["url"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def handle_redirect_request(redirect_url: str) -> str:
    """
    Main function to handle redirect requests using Claude AI.
    
    Args:
        redirect_url: The URL to redirect to (from query parameter)
    
    Returns:
        Response message from Claude
    """
    print(f"\n{'='*60}")
    print(f"Processing redirect request for URL: {redirect_url}")
    print('='*60)
    
    messages = [
        {
            "role": "user",
            "content": f"I need to redirect a user to this URL from a query parameter: {redirect_url}. Please validate the URL first, then perform the redirect and provide information about the redirect process."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    final_response = block.text
                    print(f"Final response: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nTool called: {tool_name}")
                    print(f"Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    result_dict = json.loads(result)
                    
                    print(f"Tool result: {json.dumps(result_dict, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            messages.append({
                "role": "user",
                "content": tool_results
            })
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Redirect process completed"

def main():
    """Main function to demonstrate the redirect functionality."""
    test_urls = [
        "https://httpbin.org/redirect/3",
        "https://www.google.com",
        "https://github.com",
    ]
    
    print("Redirect URL Handler using Claude AI")
    print("="*60)
    
    for url in test_urls:
        result = handle_redirect_request(url)
        print(f"\nFinal result for {url}:")
        print(result)
        print("\n" + "="*60 + "\n")

if __name__ == "__main__":
    main()
