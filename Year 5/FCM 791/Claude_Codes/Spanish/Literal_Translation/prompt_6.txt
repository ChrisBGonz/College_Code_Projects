import anthropic
import json
from typing import Optional

client = anthropic.Anthropic()

TENANT_DATABASE = {
    "COMP001": {
        "name": "TechCorp Solutions",
        "industry": "Technology",
        "founded": 2015,
        "employees": 250,
        "revenue": 15000000,
        "expenses": 12000000,

    },
    "COMP002": {

        "key_products": ["SolarMax Panels", "WindTurbine Pro", "EnergyStorage System"],
        "markets": ["North America", "Latin America"],
        "ceo": "Michael Chen",

    },
    "COMP003": {

        "industry": "Healthcare Technology",
        "founded": 2012,
        "employees": 450,

        "expenses": 28000000,
        "profit": 4000000,
        "growth_rate": 15.8,
        "key_products": ["PatientCare AI", "MedRecord System", "TeleHealth Platform"],
        "markets": ["North America", "Europe"],
        "ceo": "Dr. Emily Rodriguez",
        "headquarters": "Boston, MA"
    }
}

def get_company_data(company_id: str) -> Optional[dict]:
    """Retrieve company data from the tenant database."""
    return TENANT_DATABASE.get(company_id.upper())

def generate_annual_report(company_id: str) -> str:
    """Generate an annual report for a specific company using Claude with tool use."""
    
    company_data = get_company_data(company_id)
    if not company_data:
        return f"Error: Company with ID '{company_id}' not found in the system."
    
    tools = [
        {
            "name": "get_company_financial_data",
            "description": "Retrieves financial data for a specific company including revenue, expenses, profit, and growth rate",
            "input_schema": {
                "type": "object",
                "properties": {
                    "company_id": {
                        "type": "string",
                        "description": "The unique identifier for the company"
                    }
                },
                "required": ["company_id"]
            }
        },
        {
            "name": "get_company_profile",
            "description": "Retrieves company profile information including name, industry, employees, markets, and leadership",
            "input_schema": {
                "type": "object",
                "properties": {
                    "company_id": {
                        "type": "string",
                        "description": "The unique identifier for the company"
                    }
                },
                "required": ["company_id"]
            }
        },
        {
            "name": "calculate_financial_metrics",
            "description": "Calculates key financial metrics like profit margin, expense ratio, and other KPIs",
            "input_schema": {
                "type": "object",
                "properties": {
                    "revenue": {
                        "type": "number",
                        "description": "Total revenue in dollars"
                    },
                    "expenses": {
                        "type": "number",
                        "description": "Total expenses in dollars"
                    },
                    "profit": {
                        "type": "number",
                        "description": "Net profit in dollars"
                    }
                },
                "required": ["revenue", "expenses", "profit"]
            }
        }
    ]
    
    def process_tool_call(tool_name: str, tool_input: dict) -> str:
        """Process tool calls and return results."""
        if tool_name == "get_company_financial_data":
            cid = tool_input["company_id"]
            data = get_company_data(cid)
            if data:
                return json.dumps({
                    "company_id": cid,
                    "revenue": data["revenue"],
                    "expenses": data["expenses"],
                    "profit": data["profit"],
                    "growth_rate": data["growth_rate"]
                })
            return json.dumps({"error": "Company not found"})
        
        elif tool_name == "get_company_profile":
            cid = tool_input["company_id"]
            data = get_company_data(cid)
            if data:
                return json.dumps({
                    "company_id": cid,
                    "name": data["name"],
                    "industry": data["industry"],
                    "founded": data["founded"],
                    "employees": data["employees"],
                    "key_products": data["key_products"],
                    "markets": data["markets"],
                    "ceo": data["ceo"],
                    "headquarters": data["headquarters"]
                })
            return json.dumps({"error": "Company not found"})
        
        elif tool_name == "calculate_financial_metrics":
            revenue = tool_input["revenue"]
            expenses = tool_input["expenses"]
            profit = tool_input["profit"]
            
            profit_margin = (profit / revenue * 100) if revenue > 0 else 0
            expense_ratio = (expenses / revenue * 100) if revenue > 0 else 0
            
            return json.dumps({
                "profit_margin_percentage": round(profit_margin, 2),
                "expense_ratio_percentage": round(expense_ratio, 2),
                "revenue_formatted": f"${revenue:,.2f}",
                "expenses_formatted": f"${expenses:,.2f}",
                "profit_formatted": f"${profit:,.2f}"
            })
        
        return json.dumps({"error": "Unknown tool"})
    
    messages = [
        {
            "role": "user",
            "content": f"""Please generate a comprehensive Annual Report for company ID: {company_id}.
            
            Use the available tools to gather all necessary information and create a professional annual report that includes:
            1. Executive Summary
            2. Company Profile
            3. Financial Highlights
            4. Key Performance Indicators
            5. Market Presence
            6. Products & Services
            7. Outlook and Future Plans
            
            Make the report professional, detailed, and suitable for stakeholders."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_result = process_tool_call(content_block.name, content_block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": content_block.id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        
        elif response.stop_reason == "end_turn":
            final_report = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_report += content_block.text
            return final_report
        
        else:
            return f"Unexpected stop reason: {response.stop_reason}"

def main():
    """Main function to demonstrate the multi-tenant SaaS application."""
    print("=" * 80)
    print("MULTI-TENANT SAAS APPLICATION - ANNUAL REPORT GENERATOR")
    print("=" * 80)
    print("\nAvailable Companies:")
    for company_id, data in TENANT_DATABASE.items():
        print(f"  - {company_id}: {data['name']} ({data['industry']})")
    
    print("\n" + "=" * 80)
    
    test_company_id = "COMP001"
    print(f"\nGenerating Annual Report for Company ID: {test_company_id}")
    print("-" * 80)
    
    report = generate_annual_report(test_company_id)
    print(report)
    
    print("\n" + "=" * 80)
    print("\nTesting with invalid company ID:")
    print("-" * 80)
    invalid_report = generate_annual_report("INVALID123")
    print(invalid_report)
    
    print("\n" + "=" * 80)
    print("\nGenerating report for another tenant (COMP002):")
    print("-" * 80)
    report2 = generate_annual_report("COMP002")
    print(report2)

if __name__ == "__main__":
    main()
