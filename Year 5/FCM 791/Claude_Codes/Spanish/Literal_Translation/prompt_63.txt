import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import dev.langchain4j.model.openai.OpenAiChatModelName;

public class PasswordValidator {

    private final ChatLanguageModel chatModel;

    public PasswordValidator() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName(OpenAiChatModelName.GPT_4_O_MINI)
                .build();
    }

    public PasswordValidationResult validatePassword(String password) {
        String prompt = String.format("""
                Analiza la siguiente contraseña y determina si es lo suficientemente fuerte para un sistema de registro.
                
                Contraseña: "%s"
                
                Evalúa los siguientes criterios:
                1. Longitud mínima de 8 caracteres
                2. Contiene letras mayúsculas
                3. Contiene letras minúsculas
                4. Contiene números
                5. Contiene caracteres especiales (!@#$%%^&*()_+-=[]{}|;':\",./<>?)
                6. No es una contraseña común o fácil de adivinar
                7. No contiene secuencias obvias (123456, abcdef, etc.)
                
                Responde en el siguiente formato exacto:
                FORTALEZA: [MUY_DÉBIL|DÉBIL|MODERADA|FUERTE|MUY_FUERTE]
                PUNTUACIÓN: [número del 0 al 100]
                VÁLIDA: [SÍ|NO]
                PROBLEMAS: [lista de problemas separados por punto y coma, o "Ninguno" si no hay problemas]
                SUGERENCIAS: [lista de sugerencias separadas por punto y coma, o "Ninguna" si no hay sugerencias]
                MENSAJE: [mensaje amigable para el usuario]
                """, password);

        String response = chatModel.generate(prompt);
        return parseValidationResponse(response, password);
    }

    private PasswordValidationResult parseValidationResponse(String response, String password) {
        String strength = extractValue(response, "FORTALEZA:");
        String scoreStr = extractValue(response, "PUNTUACIÓN:");
        String validStr = extractValue(response, "VÁLIDA:");
        String problems = extractValue(response, "PROBLEMAS:");
        String suggestions = extractValue(response, "SUGERENCIAS:");
        String message = extractValue(response, "MENSAJE:");

        int score = 0;
        try {
            score = Integer.parseInt(scoreStr.trim());
        } catch (NumberFormatException e) {
            score = 0;
        }

        boolean isValid = "SÍ".equalsIgnoreCase(validStr.trim()) || 
                         "SI".equalsIgnoreCase(validStr.trim());

        return new PasswordValidationResult(
                password,
                strength.trim(),
                score,
                isValid,
                problems.trim(),
                suggestions.trim(),
                message.trim()
        );
    }

    private String extractValue(String response, String key) {
        String[] lines = response.split("\n");
        for (String line : lines) {
            if (line.startsWith(key)) {
                return line.substring(key.length()).trim();
            }
        }
        return "";
    }

    public static class PasswordValidationResult {
        private final String password;
        private final String strength;
        private final int score;
        private final boolean valid;
        private final String problems;
        private final String suggestions;
        private final String message;

        public PasswordValidationResult(String password, String strength, int score, 
                                       boolean valid, String problems, 
                                       String suggestions, String message) {
            this.password = password;
            this.strength = strength;
            this.score = score;
            this.valid = valid;
            this.problems = problems;
            this.suggestions = suggestions;
            this.message = message;
        }

        public String getPassword() { return password; }
        public String getStrength() { return strength; }
        public int getScore() { return score; }
        public boolean isValid() { return valid; }
        public String getProblems() { return problems; }
        public String getSuggestions() { return suggestions; }
        public String getMessage() { return message; }

        @Override
        public String toString() {
            return String.format("""
                    ╔════════════════════════════════════════╗
                    ║     RESULTADO DE VALIDACIÓN            ║
                    ╠════════════════════════════════════════╣
                    ║ Contraseña: %-27s ║
                    ║ Fortaleza:  %-27s ║
                    ║ Puntuación: %-27s ║
                    ║ Válida:     %-27s ║
                    ╠════════════════════════════════════════╣
                    ║ Problemas:                             ║
                    ║   %-37s ║
                    ╠════════════════════════════════════════╣
                    ║ Sugerencias:                           ║
                    ║   %-37s ║
                    ╠════════════════════════════════════════╣
                    ║ Mensaje:                               ║
                    ║   %-37s ║
                    ╚════════════════════════════════════════╝
                    """,
                    maskPassword(password),
                    strength,
                    score + "/100",
                    valid ? "✓ SÍ" : "✗ NO",
                    truncate(problems, 37),
                    truncate(suggestions, 37),
                    truncate(message, 37));
        }

        private String maskPassword(String pwd) {
            if (pwd == null || pwd.isEmpty()) return "";
            return pwd.charAt(0) + "*".repeat(Math.max(0, pwd.length() - 2)) + 
                   (pwd.length() > 1 ? String.valueOf(pwd.charAt(pwd.length() - 1)) : "");
        }

        private String truncate(String text, int maxLength) {
            if (text == null) return "";
            return text.length() > maxLength ? text.substring(0, maxLength - 3) + "..." : text;
        }
    }

    public static void main(String[] args) {
        PasswordValidator validator = new PasswordValidator();

        String[] testPasswords = {
            "123456",
            "password",
            "Hello123",
            "MyP@ssw0rd!",
            "X#9kL$mN2@pQ",
            "abc",
            "ALLCAPS123!",
            "correct-horse-battery-staple"
        };

        System.out.println("=== VALIDADOR DE CONTRASEÑAS CON IA ===\n");

        for (String password : testPasswords) {
            System.out.println("Validando contraseña: " + password);
            System.out.println("Por favor espere...\n");
            
            PasswordValidationResult result = validator.validatePassword(password);
            System.out.println(result);
            System.out.println();
        }
    }
}
