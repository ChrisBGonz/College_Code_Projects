
```python
import anthropic

client = anthropic.Anthropic()

SCRIPT_PROMPT = """Write a Python script that reads a very large file (10 GB) in memory to count the number of lines.
The script should:
1. Be memory efficient - don't load the entire file at once
2. Use buffered reading or chunked processing
3. Handle different line endings (\\n, \\r\\n, \\r)
4. Show progress during processing
5. Report the final line count and processing time
6. Include error handling
7. Be production-ready with proper documentation

Please provide a complete, working Python script."""

print("Requesting Python script from Claude using streaming...\n")
print("=" * 60)

with client.messages.stream(
    model="claude-opus-4-5",
    max_tokens=2048,
    messages=[
        {
            "role": "user",
            "content": SCRIPT_PROMPT
        }
    ]
) as stream:
    full_response = ""
    for text in stream.text_stream:
        print(text, end="", flush=True)
        full_response += text

print("\n" + "=" * 60)
print("\nScript generation complete!")

output_file = "count_lines_large_file.py"
with open(output_file, "w") as f:
    lines = full_response.split("\n")
    in_code_block = False
    code_lines = []
    
    for line in lines:
        if line.strip().startswith("```python"):
            in_code_block = True
            continue
        elif line.strip() == "```" and in_code_block:
            in_code_block = False
            continue
        elif in_code_block:
            code_lines.append(line)
    
    if code_lines:
        f.write("\n".join(code_lines))
        print(f"\nPython script saved to: {output_file}")
    else:
        f.write(full_response)
        print(f"\nFull response saved to: {output_file}")
