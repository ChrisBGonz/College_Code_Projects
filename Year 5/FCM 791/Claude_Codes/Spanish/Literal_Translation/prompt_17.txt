import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function configureMongoDBConnection() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Configura una cadena de conexión de MongoDB en un script de Node.js para una base de datos de desarrollo local. Proporciona el código completo con las mejores prácticas, incluyendo manejo de errores, variables de entorno, y opciones de conexión recomendadas.",
      },
    ],
  });

  console.log("Respuesta de Claude sobre configuración de MongoDB:");
  console.log("=".repeat(60));

  if (message.content[0].type === "text") {
    console.log(message.content[0].text);
  }

  console.log("\n" + "=".repeat(60));
  console.log("Información de uso:");
  console.log(`- Tokens de entrada: ${message.usage.input_tokens}`);
  console.log(`- Tokens de salida: ${message.usage.output_tokens}`);
}

configureMongoDBConnection().catch(console.error);
