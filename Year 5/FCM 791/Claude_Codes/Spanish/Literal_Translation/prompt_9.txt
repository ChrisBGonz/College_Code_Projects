```ruby
class Staff::PromotionsController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :find_staff_member, only: [:create]

  def create
    if @staff_member.manager?
      render json: { error: "El miembro del staff ya es un Manager." }, status: :unprocessable_entity
      return
    end

    if @staff_member.promote_to_manager!
      render json: {
        message: "#{@staff_member.full_name} ha sido promovido a Manager exitosamente.",
        staff_member: {
          id: @staff_member.id,
          full_name: @staff_member.full_name,
          role: @staff_member.role
        }
      }, status: :ok
    else
      render json: {
        error: "No se pudo promover al miembro del staff.",
        details: @staff_member.errors.full_messages
      }, status: :unprocessable_entity
    end
  end

  private

  def authorize_manager!
    unless current_user.manager?
      render json: { error: "No tienes permisos para realizar esta acción." }, status: :forbidden
    end
  end

  def find_staff_member
    @staff_member = User.find_by(id: params[:staff_member_id], role: :staff)

    unless @staff_member
      render json: { error: "Miembro del staff no encontrado." }, status: :not_found
    end
  end
end
```

```ruby
# app/models/user.rb
class User < ApplicationRecord
  enum role: { staff: 0, manager: 1, admin: 2 }

  validates :full_name, presence: true
  validates :email, presence: true, uniqueness: true
  validates :role, presence: true

  def promote_to_manager!
    ActiveRecord::Base.transaction do
      update!(role: :manager, promoted_at: Time.current)
      PromotionLog.create!(
        promoted_user: self,
        promoted_by: Current.user,
        previous_role: :staff,
        new_role: :manager,
        promoted_at: Time.current
      )
    end
  rescue ActiveRecord::RecordInvalid => e
    errors.add(:base, e.message)
    false
  end
end
```
