import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class AESEncryptionGenerator {

    public static void main(String[] args) {
        // Initialize the OpenAI chat model
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();

        // Create the prompt to generate AES encryption method
        String prompt = """
                Genera un método completo en Java para cifrar una cadena sensible utilizando el algoritmo AES.
                El método debe:
                1. Usar AES con modo CBC y padding PKCS5
                2. Generar una clave AES de 256 bits
                3. Generar un IV (Vector de Inicialización) aleatorio
                4. Retornar el texto cifrado en Base64
                5. Incluir también un método para descifrar
                6. Incluir manejo de excepciones apropiado
                7. Incluir un método main con ejemplo de uso
                8. Usar las bibliotecas estándar de Java (javax.crypto)
                
                Proporciona el código completo y funcional con comentarios explicativos en español.
                """;

        // Generate the response
        String response = model.generate(prompt);

        // Print the generated code
        System.out.println("=== Método de Cifrado AES Generado por IA ===");
        System.out.println();
        System.out.println(response);
    }
}
