def get_invoice_from_db(invoice_id: str) -> dict | None:
    """Simulate database lookup for invoice."""
    return app_data["invoices"].get(invoice_id)

def get_all_invoice_ids() -> list:
    """Get all available invoice IDs."""
    return list(app_data["invoices"].keys())

def process_invoice_query_with_claude(invoice_id: str, user_question: str = None) -> str:
    """
    Use Claude to process invoice queries with multi-turn conversation support.
    """
    client = Anthropic()
    conversation_history = []
    
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        available_ids = get_all_invoice_ids()
        system_message = f"""You are an invoice management assistant. 
        The invoice {invoice_id} was not found in the database.
        Available invoice IDs are: {', '.join(available_ids)}
        Help the user understand the error and suggest alternatives."""
        
        conversation_history.append({
            "role": "user",
            "content": f"I'm looking for invoice {invoice_id}. Can you help me?"
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_message,
            messages=conversation_history
        )
        
        return response.content[0].text
    
    invoice_json = json.dumps(invoice, indent=2)
    system_message = f"""You are an invoice management assistant with access to invoice data.
    
    Current Invoice Data:
    {invoice_json}
    
    You can help users understand invoice details, calculate totals, explain payment status,
    and answer any questions about this invoice. Be helpful, accurate, and professional."""
    
    if user_question:
        initial_message = user_question
    else:
        initial_message = f"Please provide a summary of invoice {invoice_id}"
    
    conversation_history.append({
        "role": "user",
        "content": initial_message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_message,
        messages=conversation_history
    )
    
    assistant_response = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_response
    })
    
    follow_up_questions = [
        "What is the payment status and when is it due?",
        "Can you break down the tax calculation?",
        "What are the main services or products in this invoice?"
    ]
    
    all_responses = [assistant_response]
    
    for follow_up in follow_up_questions:
        conversation_history.append({
            "role": "user",
            "content": follow_up
        })
        
        follow_up_response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_message,
            messages=conversation_history
        )
        
        follow_up_text = follow_up_response.content[0].text
        conversation_history.append({
            "role": "assistant",
            "content": follow_up_text
        })
        
        all_responses.append(f"\n**Q: {follow_up}**\n{follow_up_text}")
    
    return "\n\n---\n\n".join(all_responses)

def get_invoice_details(invoice_id: str):
    """
    Flask route handler that retrieves invoice details.
    This simulates what a Flask route would do.
    """
    invoice = get_invoice_from_db(invoice_id)
    
    if not invoice:
        available_ids = get_all_invoice_ids()
        error_analysis = process_invoice_query_with_claude(invoice_id)
        
        return {
            "error": f"Invoice {invoice_id} not found",
            "available_invoices": available_ids,
            "assistant_message": error_analysis,
            "status_code": 404
        }
    
    claude_analysis = process_invoice_query_with_claude(invoice_id)
    
    return {
        "invoice": invoice,
        "ai_analysis": claude_analysis,
        "status_code": 200
    }
