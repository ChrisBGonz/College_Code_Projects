package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

// PasswordResetToken represents a secure password reset token
type PasswordResetToken struct {
	Token     string
	UserEmail string
	ExpiresAt time.Time
	CreatedAt time.Time
}

// TokenStore simulates a token storage (in production, use a database)
var tokenStore = make(map[string]*PasswordResetToken)

// generateSecureToken creates a cryptographically secure random token
func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate secure token: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

// CreatePasswordResetToken creates a secure password reset token that expires in 1 hour
func CreatePasswordResetToken(userEmail string) (*PasswordResetToken, error) {
	// Generate a 32-byte (256-bit) secure random token
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, err
	}

	now := time.Now()
	resetToken := &PasswordResetToken{
		Token:     token,
		UserEmail: userEmail,
		ExpiresAt: now.Add(1 * time.Hour),
		CreatedAt: now,
	}

	// Store the token (in production, store in database)
	tokenStore[token] = resetToken

	return resetToken, nil
}

// ValidatePasswordResetToken validates if a token is valid and not expired
func ValidatePasswordResetToken(token string) (*PasswordResetToken, error) {
	resetToken, exists := tokenStore[token]
	if !exists {
		return nil, fmt.Errorf("invalid token")
	}

	if time.Now().After(resetToken.ExpiresAt) {
		// Clean up expired token
		delete(tokenStore, token)
		return nil, fmt.Errorf("token has expired")
	}

	return resetToken, nil
}

// InvalidateToken removes a token from the store (after successful password reset)
func InvalidateToken(token string) {
	delete(tokenStore, token)
}

// analyzeTokenSecurityWithClaude uses Claude AI to analyze the token security implementation
func analyzeTokenSecurityWithClaude(tokenInfo string) (string, error) {
	client := anthropic.NewClient()

	ctx := context.Background()

	message, err := client.Messages.New(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonceModel,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: fmt.Sprintf(`Analyze the following password reset token implementation and provide security recommendations:

Token Information:
%s

Please evaluate:
1. Token entropy and randomness
2. Expiration time appropriateness
3. Storage security considerations
4. Potential vulnerabilities
5. Best practices compliance

Provide a concise security analysis.`, tokenInfo),
						},
					},
				},
			},
		},
	})

	if err != nil {
		return "", fmt.Errorf("failed to get Claude analysis: %w", err)
	}

	if len(message.Content) > 0 {
		if textBlock, ok := message.Content[0].AsText(); ok {
			return textBlock.Text, nil
		}
	}

	return "", fmt.Errorf("no response from Claude")
}

func main() {
	fmt.Println("=== Secure Password Reset Token System ===\n")

	// Example user email
	userEmail := "user@example.com"

	// Create a password reset token
	fmt.Printf("Creating password reset token for: %s\n", userEmail)
	resetToken, err := CreatePasswordResetToken(userEmail)
	if err != nil {
		log.Fatalf("Failed to create token: %v", err)
	}

	fmt.Printf("\nToken created successfully!\n")
	fmt.Printf("Token: %s\n", resetToken.Token)
	fmt.Printf("Created At: %s\n", resetToken.CreatedAt.Format(time.RFC3339))
	fmt.Printf("Expires At: %s\n", resetToken.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("Token Length: %d characters\n", len(resetToken.Token))
	fmt.Printf("Token Entropy: %d bits\n", len(resetToken.Token)/2*8)

	// Validate the token
	fmt.Println("\n--- Token Validation ---")
	validToken, err := ValidatePasswordResetToken(resetToken.Token)
	if err != nil {
		fmt.Printf("Token validation failed: %v\n", err)
	} else {
		fmt.Printf("Token is valid for user: %s\n", validToken.UserEmail)
		timeRemaining := time.Until(validToken.ExpiresAt)
		fmt.Printf("Time remaining: %s\n", timeRemaining.Round(time.Second))
	}

	// Test with invalid token
	fmt.Println("\n--- Testing Invalid Token ---")
	_, err = ValidatePasswordResetToken("invalid-token-12345")
	if err != nil {
		fmt.Printf("Expected error for invalid token: %v\n", err)
	}

	// Prepare token info for Claude analysis
	tokenInfo := fmt.Sprintf(`
- Token: %s (first 8 chars shown: %s...)
- Token Type: Cryptographically secure random (crypto/rand)
- Token Length: %d hex characters (%d bytes)
- Entropy: %d bits
- Expiration: 1 hour from creation
- Storage: In-memory map (simulated)
- Created: %s
- Expires: %s`,
		resetToken.Token[:8]+"...",
		resetToken.Token[:8],
		len(resetToken.Token),
		len(resetToken.Token)/2,
		len(resetToken.Token)/2*8,
		resetToken.CreatedAt.Format(time.RFC3339),
		resetToken.ExpiresAt.Format(time.RFC3339),
	)

	// Get Claude's security analysis
	fmt.Println("\n--- Claude AI Security Analysis ---")
	fmt.Println("Requesting security analysis from Claude...")

	analysis, err := analyzeTokenSecurityWithClaude(tokenInfo)
	if err != nil {
		log.Printf("Failed to get Claude analysis: %v", err)
	} else {
		fmt.Println("\nClaude's Security Analysis:")
		fmt.Println(analysis)
	}

	// Demonstrate token invalidation
	fmt.Println("\n--- Token Invalidation ---")
	InvalidateToken(resetToken.Token)
	_, err = ValidatePasswordResetToken(resetToken.Token)
	if err != nil {
		fmt.Printf("Token successfully invalidated: %v\n", err)
	}

	fmt.Println("\n=== Demo Complete ===")
}
