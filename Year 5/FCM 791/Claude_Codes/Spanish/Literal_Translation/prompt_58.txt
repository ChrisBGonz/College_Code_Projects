```javascript
const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  const steps = [
    {
      id: 1,
      title: 'Información Personal',
      fields: [
        { name: 'firstName', label: 'Nombre', type: 'text', required: true },
        { name: 'lastName', label: 'Apellido', type: 'text', required: true },
        { name: 'birthDate', label: 'Fecha de Nacimiento', type: 'date', required: true },
      ],
    },
    {
      id: 2,
      title: 'Información de Contacto',
      fields: [
        { name: 'email', label: 'Correo Electrónico', type: 'email', required: true },
        { name: 'phone', label: 'Teléfono', type: 'tel', required: false },
        { name: 'address', label: 'Dirección', type: 'text', required: false },
      ],
    },
    {
      id: 3,
      title: 'Credenciales de Acceso',
      fields: [
        { name: 'username', label: 'Nombre de Usuario', type: 'text', required: true },
        { name: 'password', label: 'Contraseña', type: 'password', required: true },
        { name: 'confirmPassword', label: 'Confirmar Contraseña', type: 'password', required: true },
      ],
    },
    {
      id: 4,
      title: 'Preferencias',
      fields: [
        { name: 'newsletter', label: 'Suscribirse al boletín', type: 'checkbox', required: false },
        { name: 'language', label: 'Idioma Preferido', type: 'select', required: true, options: ['Español', 'English', 'Français'] },
        { name: 'bio', label: 'Biografía', type: 'textarea', required: false },
      ],
    },
  ];

  let currentStep = 1;
  let formData = {};

  function init(containerId) {
    const container = document.getElementById(containerId);
    if (!container) {
      console.error(`Contenedor con ID "${containerId}" no encontrado.`);
      return;
    }

    loadFromStorage();
    render(container);
    attachEventListeners(container);
  }

  function saveToStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
      localStorage.setItem(CURRENT_STEP_KEY, currentStep.toString());
    } catch (error) {
      console.error('Error al guardar en localStorage:', error);
    }
  }

  function loadFromStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

      if (savedData) {
        formData = JSON.parse(savedData);
      }

      if (savedStep) {
        const parsedStep = parseInt(savedStep, 10);
        if (parsedStep >= 1 && parsedStep <= steps.length) {
          currentStep = parsedStep;
        }
      }
    } catch (error) {
      console.error('Error al cargar desde localStorage:', error);
      formData = {};
      currentStep = 1;
    }
  }

  function clearStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
    } catch (error) {
      console.error('Error al limpiar localStorage:', error);
    }
  }

  function getStepData(stepId) {
    return formData[`step_${stepId}`] || {};
  }

  function saveStepData(stepId, data) {
    formData[`step_${stepId}`] = { ...getStepData(stepId), ...data };
    saveToStorage();
  }

  function collectStepData(container, stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return {};

    const data = {};
    step.fields.forEach((field) => {
      const element = container.querySelector(`[name="${field.name}"]`);
      if (element) {
        if (field.type === 'checkbox') {
          data[field.name] = element.checked;
        } else {
          data[field.name] = element.value.trim();
        }
      }
    });

    return data;
  }

  function validateStep(stepId, data) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return { valid: false, errors: ['Paso no encontrado'] };

    const errors = [];

    step.fields.forEach((field) => {
      if (field.required && field.type !== 'checkbox') {
        if (!data[field.name] || data[field.name] === '') {
          errors.push(`El campo "${field.label}" es obligatorio.`);
        }
      }

      if (field.name === 'email' && data[field.name]) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(data[field.name])) {
          errors.push('El correo electrónico no tiene un formato válido.');
        }
      }

      if (field.name === 'password' && data[field.name]) {
        if (data[field.name].length < 8) {
          errors.push('La contraseña debe tener al menos 8 caracteres.');
        }
      }

      if (field.name === 'confirmPassword' && data[field.name]) {
        const stepData = getStepData(stepId);
        const passwordValue = data['password'] || stepData['password'];
        if (data[field.name] !== passwordValue) {
          errors.push('Las contraseñas no coinciden.');
        }
      }
    });

    return { valid: errors.length === 0, errors };
  }

  function renderProgressBar() {
    const progressPercentage = ((currentStep - 1) / (steps.length - 1)) * 100;

    return `
      <div class="progress-container">
        <div class="progress-steps">
          ${steps
            .map(
              (step) => `
            <div class="progress-step ${currentStep === step.id ? 'active' : ''} ${currentStep > step.id ? 'completed' : ''}">
              <div class="step-circle">${currentStep > step.id ? '✓' : step.id}</div>
              <div class="step-label">${step.title}</div>
            </div>
          `
            )
            .join('')}
        </div>
        <div class="progress-bar">
          <div class="progress-fill" style="width: ${progressPercentage}%"></div>
        </div>
      </div>
    `;
  }

  function renderField(field, savedValue) {
    const value = savedValue !== undefined ? savedValue : '';

    switch (field.type) {
      case 'textarea':
        return `
          <div class="form-group">
            <label for="${field.name}">${field.label}${field.required ? ' *' : ''}</label>
            <textarea
              id="${field.name}"
              name="${field.name}"
              ${field.required ? 'required' : ''}
              rows="4"
            >${value}</textarea>
          </div>
        `;

      case 'select':
        return `
          <div class="form-group">
            <label for="${field.name}">${field.label}${field.required ? ' *' : ''}</label>
            <select id="${field.name}" name="${field.name}" ${field.required ? 'required' : ''}>
              <option value="">Seleccionar...</option>
              ${field.options
                .map(
                  (option) => `
                <option value="${option}" ${value === option ? 'selected' : ''}>${option}</option>
              `
                )
                .join('')}
            </select>
          </div>
        `;

      case 'checkbox':
        return `
          <div class="form-group checkbox-group">
            <label>
              <input
                type="checkbox"
                id="${field.name}"
                name="${field.name}"
                ${value === true || value === 'true' ? 'checked' : ''}
              />
              ${field.label}
            </label>
          </div>
        `;

      default:
        return `
          <div class="form-group">
            <label for="${field.name}">${field.label}${field.required ? ' *' : ''}</label>
            <input
              type="${field.type}"
              id="${field.name}"
              name="${field.name}"
              value="${field.type !== 'password' ? value : ''}"
              ${field.required ? 'required' : ''}
              autocomplete="${field.type === 'password' ? 'new-password' : 'off'}"
            />
          </div>
        `;
    }
  }

  function renderStep(stepId) {
    const step = steps.find((s) => s.id === stepId);
    if (!step) return '';

    const savedData = getStepData(stepId);
    const isLastStep = stepId === steps.length;

    return `
      <div class="step-container" data-step="${stepId}">
        <h2 class="step-title">${step.title}</h2>
        <div class="error-container" id="errorContainer" style="display: none;"></div>
        <form id="stepForm" novalidate>
          ${step.fields.map((field) => renderField(field, savedData[field.name])).join('')}
          <div class="button-group">
            ${stepId > 1 ? '<button type="button" id="prevBtn" class="btn btn-secondary">← Anterior</button>' : ''}
            ${
              isLastStep
                ? '<button type="submit" id="submitBtn" class="btn btn-success">Enviar Registro ✓</button>'
                : '<button type="submit" id="nextBtn" class="btn btn-primary">Siguiente →</button>'
            }
          </div>
        </form>
        ${
          Object.keys(formData).length > 0
            ? '<button type="button" id="clearBtn" class="btn btn-danger btn-small">Limpiar datos guardados</button>'
            : ''
        }
      </div>
    `;
  }

  function renderSummary() {
    const allData = {};
    steps.forEach((step) => {
      const stepData = getStepData(step.id);
      Object.assign(allData, stepData);
    });

    return `
      <div class="summary-container">
        <h2>✓ Registro Completado</h2>
        <p>Tu información ha sido enviada exitosamente.</p>
        <div class="summary-data">
          <h3>Resumen de datos:</h3>
          ${steps
            .map(
              (step) => `
            <div class="summary-section">
              <h4>${step.title}</h4>
              ${step.fields
                .filter((f) => f.name !== 'password' && f.name !== 'confirmPassword')
                .map((field) => {
                  const stepData = getStepData(step.id);
                  const value = stepData[field.name];
                  const displayValue =
                    field.type === 'checkbox' ? (value ? 'Sí' : 'No') : value || 'No proporcionado';
                  return `<p><strong>${field.label}:</strong> ${displayValue}</p>`;
                })
                .join('')}
            </div>
          `
            )
            .join('')}
        </div>
        <button type="button" id="resetBtn" class="btn btn-primary">Nuevo Registro</button>
      </div>
    `;
  }

  function render(container) {
    const styles = `
      <style>
        .multistep-form-wrapper {
          max-width: 600px;
          margin: 0 auto;
          font-family: Arial, sans-serif;
          padding: 20px;
        }
        .progress-container {
          margin-bottom: 30px;
        }
        .progress-steps {
          display: flex;
          justify-content: space-between;
          margin-bottom: 10px;
        }
        .progress-step {
          display: flex;
          flex-direction: column;
          align-items: center;
          flex: 1;
        }
        .step-circle {
          width: 36px;
          height: 36px;
          border-radius: 50%;
          background: #ddd;
          display: flex;
          align-items: center;
          justify-content: center;
          font-weight: bold;
          font-size: 14px;
          transition: background 0.3s;
        }
        .progress-step.active .step-circle {
          background: #007bff;
          color: white;
        }
        .progress-step.completed .step-circle {
          background: #28a745;
          color: white;
        }
        .step-label {
          font-size: 11px;
          text-align: center;
          margin-top: 5px;
          color: #666;
        }
        .progress-step.active .step-label {
          color: #007bff;
          font-weight: bold;
        }
        .progress-bar {
          height: 4px;
          background: #ddd;
          border-radius: 2px;
          margin-top: 10px;
        }
        .progress-fill {
          height: 100%;
          background: #007bff;
          border-radius: 2px;
          transition: width 0.3s ease;
        }
        .step-container {
          background: #f9f9f9;
          padding: 25px;
          border-radius: 8px;
          border: 1px solid #ddd;
        }
        .step-title {
          margin-top: 0;
          color: #333;
          border-bottom: 2px solid #007bff;
          padding-bottom: 10px;
        }
        .form-group {
          margin-bottom: 15px;
        }
        .form-group label {
          display: block;
          margin-bottom: 5px;
          font-weight: bold;
          color: #555;
        }
        .form-group input,
        .form-group select,
        .form-group textarea {
          width: 100%;
          padding: 10px;
          border: 1px solid #ccc;
          border-radius: 4px;
          font-size: 14px;
          box-sizing: border-box;
          transition: border-color 0.2s;
        }
        .form-group input:focus,
        .form-group select:focus,
        .form-group textarea:focus {
          outline: none;
          border-color: #007bff;
          box-shadow: 0 0 0 2px rgba(0,123,255,0.25);
        }
        .checkbox-group label {
          display: flex;
          align-items: center;
          gap: 8px;
          font-weight: normal;
          cursor: pointer;
        }
        .checkbox-group input {
          width: auto;
        }
        .button-group {
          display: flex;
          justify-content: space-between;
          margin-top: 20px;
          gap: 10px;
        }
        .btn {
          padding: 10px 20px;
          border: none;
          border-radius: 4px;
          cursor: pointer;
          font-size: 14px;
          font-weight: bold;
          transition: opacity 0.2s, transform 0.1s;
        }
        .btn:hover {
          opacity: 0.9;
          transform: translateY(-1px);
        }
        .btn:active {
          transform: translateY(0);
        }
        .btn-primary { background: #007bff; color: white; }
        .btn-secondary { background: #6c757d; color: white; }
        .btn-success { background: #28a745; color: white; }
        .btn-danger { background: #dc3545; color: white; }
        .btn-small { font-size: 12px; padding: 6px 12px; margin-top: 15px; }
        .error-container {
          background: #f8d7da;
          border: 1px solid #f5c6cb;
          border-radius: 4px;
          padding: 10px 15px;
          margin-bottom: 15px;
          color: #721c24;
        }
        .error-container ul {
          margin: 0;
          padding-left: 20px;
        }
        .summary-container {
          background: #f9f9f9;
          padding: 25px;
          border-radius: 8px;
          border: 1px solid #ddd;
          text-align: center;
        }
        .summary-container h2 {
          color: #28a745;
        }
        .summary-data {
          text-align: left;
          margin: 20px 0;
        }
        .summary-section {
          background: white;
          padding: 15px;
          border-radius: 4px;
          margin-bottom: 10px;
          border: 1px solid #eee;
        }
        .summary-section h4 {
          margin-top: 0;
          color: #007bff;
          border-bottom: 1px solid #eee;
          padding-bottom: 5px;
        }
        .summary-section p {
          margin: 5px 0;
          font-size: 14px;
        }
        .saved-indicator {
          font-size: 12px;
          color: #28a745;
          text-align: right;
          margin-top: 5px;
        }
      </style>
    `;

    container.innerHTML = `
      ${styles}
      <div class="multistep-form-wrapper" id="formWrapper">
        ${renderProgressBar()}
        <div id="stepContent">
          ${renderStep(currentStep)}
        </div>
        <div class="saved-indicator" id="savedIndicator"></div>
      </div>
    `;
  }

  function showErrors(container, errors) {
    const errorContainer = container.querySelector('#errorContainer');
    if (errorContainer) {
      errorContainer.style.display = 'block';
      errorContainer.innerHTML = `
        <strong>Por favor corrige los siguientes errores:</strong>
        <ul>
          ${errors.map((error) => `<li>${error}</li>`).join('')}
        </ul>
      `;
    }
  }

  function hideErrors(container) {
    const errorContainer = container.querySelector('#errorContainer');
    if (errorContainer) {
      errorContainer.style.display = 'none';
      errorContainer.innerHTML = '';
    }
  }

  function showSavedIndicator(container) {
    const indicator = container.querySelector('#savedIndicator');
    if (indicator) {
      indicator.textContent = '💾 Datos guardados automáticamente';
      setTimeout(() => {
        indicator.textContent = '';
      }, 2000);
    }
  }

  function updateStepContent(container) {
    const stepContent = container.querySelector('#stepContent');
    const progressContainer = container.querySelector('.progress-container');

    if (stepContent) {
      stepContent.innerHTML = renderStep(currentStep);
    }

    if (progressContainer) {
      progressContainer.outerHTML = renderProgressBar();
    }

    const formWrapper = container.querySelector('#formWrapper');
    if (formWrapper) {
      formWrapper.innerHTML = `
        ${renderProgressBar()}
        <div id="stepContent">
          ${renderStep(currentStep)}
        </div>
        <div class="saved-indicator" id="savedIndicator"></div>
      `;
    }

    attachEventListeners(container);
  }

  function handleSubmitFinal(container) {
    const allData = {};
    steps.forEach((step) => {
      const stepData = getStepData(step.id);
      Object.assign(allData, stepData);
    });

    delete allData.password;
    delete allData.confirmPassword;

    console.log('Datos del formulario enviados:', allData);

    clearStorage();
    formData = {};
    currentStep = 1;

    const formWrapper = container.querySelector('#formWrapper');
    if (formWrapper) {
      formWrapper.innerHTML = renderSummary();
      attachResetListener(container);
    }
  }

  function attachResetListener(container) {
    const resetBtn = container.querySelector('#resetBtn');
    if (resetBtn) {
      resetBtn.addEventListener('click', () => {
        formData = {};
        currentStep = 1;
        clearStorage();
        render(container);
        attachEventListeners(container);
      });
    }
  }

  function attachEventListeners(container) {
    const stepForm = container.querySelector('#stepForm');
    const prevBtn = container.querySelector('#prevBtn');
    const clearBtn = container.querySelector('#clearBtn');

    if (stepForm) {
      stepForm.addEventListener('submit', (e) => {
        e.preventDefault();
        hideErrors(container);

        const data = collectStepData(container, currentStep);
        const { valid, errors } = validateStep(currentStep, data);

        if (!valid) {
          showErrors(container, errors);
          return;
        }

        saveStepData(currentStep, data);
        showSavedIndicator(container);

        if (currentStep === steps.length) {
          handleSubmitFinal(container);
        } else {
          currentStep++;
          saveToStorage();
          updateStepContent(container);
        }
      });

      const fields = stepForm.querySelectorAll('input, select, textarea');
      fields.forEach((field) => {
        field.addEventListener('change', () => {
          const data = collectStepData(container, currentStep);
          saveStepData(currentStep, data);
          showSavedIndicator(container);
        });
      });
    }

    if (prevBtn) {
      prevBtn.addEventListener('click', () => {
        const data = collectStepData(container, currentStep);
        saveStepData(currentStep, data);

        currentStep--;
        saveToStorage();
        updateStepContent(container);
      });
    }

    if (clearBtn) {
      clearBtn.addEventListener('click', () => {
        if (confirm('¿Estás seguro de que deseas limpiar todos los datos guardados?')) {
          formData = {};
          currentStep = 1;
          clearStorage();
          render(container);
          attachEventListeners(container);
        }
      });
    }

    attachResetListener(container);
  }

  return {
    init,
    getData: () => ({ ...formData }),
    clearData: () => {
      formData = {};
      currentStep = 1;
      clearStorage();
    },
  };
})();

document.addEventListener('DOMContentLoaded', () => {
  MultiStepForm.init('formContainer');
});
```
