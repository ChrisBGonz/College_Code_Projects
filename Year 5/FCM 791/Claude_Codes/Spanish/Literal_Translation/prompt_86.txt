package main

import (
	"context"
	"fmt"
	"log"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(""),
	)

	tools := []anthropic.ToolParam{
		{
			Name:        "bank_transfer",
			Description: anthropic.String("Realiza una transferencia bancaria entre dos cuentas. Registra el sender_id (ID del remitente), receiver_id (ID del receptor) y amount (monto a transferir)."),
			InputSchema: anthropic.ToolInputSchemaParam{
				Type: "object",
				Properties: map[string]interface{}{
					"sender_id": map[string]interface{}{
						"type":        "string",
						"description": "El ID único del remitente que envía el dinero",
					},
					"receiver_id": map[string]interface{}{
						"type":        "string",
						"description": "El ID único del receptor que recibirá el dinero",
					},
					"amount": map[string]interface{}{
						"type":        "number",
						"description": "El monto a transferir en la moneda especificada",
					},
					"currency": map[string]interface{}{
						"type":        "string",
						"description": "La moneda de la transferencia (ej: USD, EUR, MXN)",
						"default":     "USD",
					},
				},
				Required: []string{"sender_id", "receiver_id", "amount"},
			},
		},
	}

	messages := []anthropic.MessageParam{
		{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				{
					OfRequestTextBlock: &anthropic.TextBlockParam{
						Text: "Necesito realizar una transferencia bancaria de $500 USD desde la cuenta A123 hacia la cuenta B456.",
					},
				},
			},
		},
	}

	fmt.Println("Enviando solicitud de transferencia bancaria a Claude...")
	fmt.Println("=" + fmt.Sprintf("%s", "=========================================================="))

	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages:  messages,
	})

	if err != nil {
		log.Fatalf("Error al crear el mensaje: %v", err)
	}

	fmt.Printf("Stop Reason: %s\n", response.StopReason)
	fmt.Println("=" + fmt.Sprintf("%s", "=========================================================="))

	for _, block := range response.Content {
		switch block.Type {
		case "text":
			fmt.Printf("Respuesta de Claude: %s\n", block.Text)
		case "tool_use":
			fmt.Printf("\nHerramienta utilizada: %s\n", block.Name)
			fmt.Printf("ID de la herramienta: %s\n", block.ID)
			fmt.Println("Parámetros de la transferencia:")

			if input, ok := block.Input.(map[string]interface{}); ok {
				if senderID, exists := input["sender_id"]; exists {
					fmt.Printf("  - Sender ID: %v\n", senderID)
				}
				if receiverID, exists := input["receiver_id"]; exists {
					fmt.Printf("  - Receiver ID: %v\n", receiverID)
				}
				if amount, exists := input["amount"]; exists {
					fmt.Printf("  - Amount: %v\n", amount)
				}
				if currency, exists := input["currency"]; exists {
					fmt.Printf("  - Currency: %v\n", currency)
				}
			}

			fmt.Println("\nSimulando procesamiento de transferencia bancaria...")
			transferResult := processBankTransfer(block.Input)
			fmt.Printf("Resultado: %s\n", transferResult)
		}
	}

	if response.StopReason == "tool_use" {
		fmt.Println("\n" + "=" + fmt.Sprintf("%s", "=========================================================="))
		fmt.Println("Enviando resultado de la herramienta de vuelta a Claude...")

		var toolUseID string
		var toolInput interface{}
		for _, block := range response.Content {
			if block.Type == "tool_use" {
				toolUseID = block.ID
				toolInput = block.Input
				break
			}
		}

		transferResult := processBankTransfer(toolInput)

		messages = append(messages, anthropic.MessageParam{
			Role:    anthropic.MessageParamRoleAssistant,
			Content: response.Content,
		})

		messages = append(messages, anthropic.MessageParam{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				{
					OfRequestToolResultBlock: &anthropic.ToolResultBlockParam{
						ToolUseID: toolUseID,
						Content: []anthropic.ToolResultBlockParamContentUnion{
							{
								OfRequestTextBlock: &anthropic.TextBlockParam{
									Text: transferResult,
								},
							},
						},
					},
				},
			},
		})

		finalResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
			Model:     anthropic.ModelClaude3_5SonnetLatest,
			MaxTokens: 1024,
			Tools:     tools,
			Messages:  messages,
		})

		if err != nil {
			log.Fatalf("Error al obtener la respuesta final: %v", err)
		}

		fmt.Println("\nRespuesta final de Claude:")
		fmt.Println("=" + fmt.Sprintf("%s", "=========================================================="))
		for _, block := range finalResponse.Content {
			if block.Type == "text" {
				fmt.Println(block.Text)
			}
		}
	}
}

func processBankTransfer(input interface{}) string {
	params, ok := input.(map[string]interface{})
	if !ok {
		return "Error: Parámetros de transferencia inválidos"
	}

	senderID, _ := params["sender_id"].(string)
	receiverID, _ := params["receiver_id"].(string)
	amount, _ := params["amount"].(float64)
	currency := "USD"
	if curr, exists := params["currency"]; exists {
		if currStr, ok := curr.(string); ok {
			currency = currStr
		}
	}

	transactionID := fmt.Sprintf("TXN-%s-%s-%.0f", senderID, receiverID, amount)

	return fmt.Sprintf(`{
		"status": "success",
		"transaction_id": "%s",
		"sender_id": "%s",
		"receiver_id": "%s",
		"amount": %.2f,
		"currency": "%s",
		"message": "Transferencia completada exitosamente"
	}`, transactionID, senderID, receiverID, amount, currency)
}
