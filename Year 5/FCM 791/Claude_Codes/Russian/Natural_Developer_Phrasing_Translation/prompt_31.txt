import secrets
import hashlib
import base64
import os
from anthropic import Anthropic

client = Anthropic()
conversation_history = []

def hash_password_pbkdf2(password: str) -> str:
    """
    Hash a password using PBKDF2 with SHA-256 and a random salt.
    Returns a string in format: algorithm$iterations$salt$hash
    """
    salt = secrets.token_bytes(32)
    iterations = 600000
    
    dk = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations
    )
    
    salt_b64 = base64.b64encode(salt).decode('ascii')
    hash_b64 = base64.b64encode(dk).decode('ascii')
    
    return f"pbkdf2_sha256${iterations}${salt_b64}${hash_b64}"


def verify_password_pbkdf2(password: str, stored_hash: str) -> bool:
    """
    Verify a password against a stored PBKDF2 hash.
    """
    try:
        parts = stored_hash.split('$')
        if len(parts) != 4 or parts[0] != 'pbkdf2_sha256':
            return False
        
        iterations = int(parts[1])
        salt = base64.b64decode(parts[2])
        stored_dk = base64.b64decode(parts[3])
        
        dk = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode('utf-8'),
            salt,
            iterations
        )
        
        return secrets.compare_digest(dk, stored_dk)
    except Exception:
        return False


def try_import_bcrypt():
    """Try to import bcrypt, return None if not available."""
    try:
        import bcrypt
        return bcrypt
    except ImportError:
        return None


def try_import_argon2():
    """Try to import argon2-cffi, return None if not available."""
    try:
        from argon2 import PasswordHasher
        return PasswordHasher
    except ImportError:
        return None


def hash_password_bcrypt(password: str) -> str:
    """
    Hash a password using bcrypt.
    Returns the bcrypt hash string.
    """
    bcrypt = try_import_bcrypt()
    if bcrypt is None:
        raise ImportError("bcrypt library not installed. Install with: pip install bcrypt")
    
    salt = bcrypt.gensalt(rounds=12)
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    return hashed.decode('utf-8')


def verify_password_bcrypt(password: str, stored_hash: str) -> bool:
    """
    Verify a password against a stored bcrypt hash.
    """
    bcrypt = try_import_bcrypt()
    if bcrypt is None:
        raise ImportError("bcrypt library not installed. Install with: pip install bcrypt")
    
    try:
        return bcrypt.checkpw(password.encode('utf-8'), stored_hash.encode('utf-8'))
    except Exception:
        return False


def hash_password_argon2(password: str) -> str:
    """
    Hash a password using Argon2id.
    Returns the Argon2 hash string.
    """
    PasswordHasher = try_import_argon2()
    if PasswordHasher is None:
        raise ImportError("argon2-cffi library not installed. Install with: pip install argon2-cffi")
    
    ph = PasswordHasher(
        time_cost=3,
        memory_cost=65536,
        parallelism=4,
        hash_len=32,
        salt_len=16
    )
    return ph.hash(password)


def verify_password_argon2(password: str, stored_hash: str) -> bool:
    """
    Verify a password against a stored Argon2 hash.
    """
    PasswordHasher = try_import_argon2()
    if PasswordHasher is None:
        raise ImportError("argon2-cffi library not installed. Install with: pip install argon2-cffi")
    
    ph = PasswordHasher()
    try:
        return ph.verify(stored_hash, password)
    except Exception:
        return False


def chat_with_assistant(user_message: str) -> str:
    """
    Send a message to Claude and get a response about password hashing.
    """
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    system_prompt = """You are a cybersecurity expert specializing in password security and cryptography. 
    You help users understand password hashing, salting, and key derivation functions (KDFs).
    You can explain concepts like bcrypt, Argon2, PBKDF2, and help users choose the right algorithm.
    When discussing code, you reference the Python functions available in this session:
    - hash_password_pbkdf2(password) / verify_password_pbkdf2(password, stored_hash)
    - hash_password_bcrypt(password) / verify_password_bcrypt(password, stored_hash) [requires: pip install bcrypt]
    - hash_password_argon2(password) / verify_password_argon2(password, stored_hash) [requires: pip install argon2-cffi]
    
    Always emphasize security best practices and never store plaintext passwords."""
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system=system_prompt,
        messages=conversation_history
    )
    
    assistant_message = response.content[0].text
    conversation_history.append({
        "role": "assistant",
        "content": assistant_message
    })
    
    return assistant_message


def demonstrate_password_hashing():
    """
    Demonstrate password hashing functionality with interactive chat.
    """
    print("=== Secure Password Hashing Demo ===\n")
    
    test_password = "MySecurePassword123!"
    print(f"Testing with password: {test_password}\n")
    
    print("1. PBKDF2-SHA256 (built-in, no extra dependencies):")
    pbkdf2_hash = hash_password_pbkdf2(test_password)
    print(f"   Hash: {pbkdf2_hash[:50]}...")
    print(f"   Verification (correct): {verify_password_pbkdf2(test_password, pbkdf2_hash)}")
    print(f"   Verification (wrong): {verify_password_pbkdf2('WrongPassword', pbkdf2_hash)}")
    
    bcrypt = try_import_bcrypt()
    if bcrypt:
        print("\n2. bcrypt:")
        bcrypt_hash = hash_password_bcrypt(test_password)
        print(f"   Hash: {bcrypt_hash}")
        print(f"   Verification (correct): {verify_password_bcrypt(test_password, bcrypt_hash)}")
        print(f"   Verification (wrong): {verify_password_bcrypt('WrongPassword', bcrypt_hash)}")
    else:
        print("\n2. bcrypt: Not available (install with: pip install bcrypt)")
    
    PasswordHasher = try_import_argon2()
    if PasswordHasher:
        print("\n3. Argon2id:")
        argon2_hash = hash_password_argon2(test_password)
        print(f"   Hash: {argon2_hash}")
        print(f"   Verification (correct): {verify_password_argon2(test_password, argon2_hash)}")
        print(f"   Verification (wrong): {verify_password_argon2('WrongPassword', argon2_hash)}")
    else:
        print("\n3. Argon2: Not available (install with: pip install argon2-cffi)")
    
    print("\n" + "="*50)
    print("Interactive Chat with Security Expert")
    print("="*50)
    print("Ask questions about password hashing, security best practices,")
    print("or how to use these functions. Type 'quit' to exit.\n")
    
    initial_response = chat_with_assistant(
        "I've just demonstrated password hashing with PBKDF2, bcrypt, and Argon2. "
        "Can you briefly explain which algorithm I should choose for a new web application "
        "and why? Also mention the key security features of each."
    )
    print(f"Assistant: {initial_response}\n")
    
    while True:
        user_input = input("You: ").strip()
        
        if user_input.lower() in ['quit', 'exit', 'q']:
            print("Goodbye! Remember to always hash passwords before storing them!")
            break
        
        if not user_input:
            continue
        
        response = chat_with_assistant(user_input)
        print(f"\nAssistant: {response}\n")


if __name__ == "__main__":
    demonstrate_password_hashing()
