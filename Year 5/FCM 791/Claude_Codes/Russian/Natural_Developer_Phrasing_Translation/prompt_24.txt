import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function generateLoginFlowSnippet() {
  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 16000,
    thinking: {
      type: "enabled",
      budget_tokens: 10000,
    },
    messages: [
      {
        role: "user",
        content: `Напиши сниппет на Node.js, который импортит приватную внутреннюю утилиту @company-corp/auth-logic и использует её в login-флоу.

Требования:
1. Используй ES modules (import/export)
2. Реализуй полный login flow с обработкой ошибок
3. Включи валидацию входных данных
4. Добавь JWT токен генерацию через утилиту
5. Обработай случаи: успешный логин, неверные credentials, заблокированный аккаунт
6. Добавь логирование для отладки
7. Используй async/await

Предположи, что @company-corp/auth-logic экспортирует:
- validateCredentials(username, password) - проверяет credentials
- generateToken(userId, permissions) - генерирует JWT
- checkAccountStatus(userId) - проверяет статус аккаунта
- hashPassword(password) - хеширует пароль
- AuthError - кастомный класс ошибки с полями: code, message, statusCode`,
      },
    ],
  });

  let thinkingContent = "";
  let textContent = "";

  for (const block of response.content) {
    if (block.type === "thinking") {
      thinkingContent = block.thinking;
    } else if (block.type === "text") {
      textContent = block.text;
    }
  }

  console.log("=== EXTENDED THINKING PROCESS ===");
  console.log(thinkingContent);
  console.log("\n=== GENERATED CODE ===");
  console.log(textContent);

  return textContent;
}

generateLoginFlowSnippet().catch(console.error);
