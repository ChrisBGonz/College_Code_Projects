import requests
from urllib.parse import urlparse, parse_qs


def get_redirect_url_from_query(query_string: str) -> str | None:
    parsed = urlparse(f"http://dummy?{query_string}")
    params = parse_qs(parsed.query)
    redirect_urls = params.get("redirect_url")
    if redirect_urls:
        return redirect_urls[0]
    return None


def perform_redirect(redirect_url: str) -> requests.Response:
    session = requests.Session()
    response = session.get(redirect_url, allow_redirects=True)
    return response


def handle_request_with_redirect(request_url: str) -> dict:
    parsed = urlparse(request_url)
    query_string = parsed.query
    params = parse_qs(query_string)

    redirect_url_list = params.get("redirect_url")
    if not redirect_url_list:
        return {
            "status": "error",
            "message": "Missing 'redirect_url' query parameter",
            "code": 400,
        }

    redirect_url = redirect_url_list[0]

    parsed_redirect = urlparse(redirect_url)
    if not parsed_redirect.scheme or not parsed_redirect.netloc:
        return {
            "status": "error",
            "message": f"Invalid redirect URL: {redirect_url}",
            "code": 400,
        }

    try:
        response = perform_redirect(redirect_url)
        return {
            "status": "success",
            "redirect_url": redirect_url,
            "final_url": response.url,
            "status_code": response.status_code,
            "content_length": len(response.content),
            "headers": dict(response.headers),
        }
    except requests.exceptions.ConnectionError as e:
        return {
            "status": "error",
            "message": f"Connection error: {str(e)}",
            "code": 502,
        }
    except requests.exceptions.Timeout as e:
        return {
            "status": "error",
            "message": f"Request timed out: {str(e)}",
            "code": 504,
        }
    except requests.exceptions.RequestException as e:
        return {
            "status": "error",
            "message": f"Request failed: {str(e)}",
            "code": 500,
        }


def main():
    test_cases = [
        "http://myapp.com/redirect?redirect_url=https://www.example.com",
        "http://myapp.com/redirect?redirect_url=https://httpbin.org/get",
        "http://myapp.com/redirect?redirect_url=https://httpbin.org/redirect/3",
        "http://myapp.com/redirect",
        "http://myapp.com/redirect?redirect_url=not-a-valid-url",
    ]

    for url in test_cases:
        print(f"\nProcessing: {url}")
        print("-" * 60)
        result = handle_request_with_redirect(url)

        if result["status"] == "success":
            print(f"✓ Redirect successful")
            print(f"  Redirect URL:  {result['redirect_url']}")
            print(f"  Final URL:     {result['final_url']}")
            print(f"  Status Code:   {result['status_code']}")
            print(f"  Content Length: {result['content_length']} bytes")
        else:
            print(f"✗ Error (HTTP {result['code']}): {result['message']}")


if __name__ == "__main__":
    main()
