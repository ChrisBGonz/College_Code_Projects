import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

public class PasswordValidator {

    private static final int MIN_LENGTH = 8;
    private static final int MAX_LENGTH = 128;

    private static final Pattern UPPERCASE_PATTERN = Pattern.compile("[A-Z]");
    private static final Pattern LOWERCASE_PATTERN = Pattern.compile("[a-z]");
    private static final Pattern DIGIT_PATTERN = Pattern.compile("[0-9]");
    private static final Pattern SPECIAL_CHAR_PATTERN = Pattern.compile("[!@#$%^&*()_+\\-=\\[\\]{};':\"\\\\|,.<>/?`~]");
    private static final Pattern WHITESPACE_PATTERN = Pattern.compile("\\s");
    private static final Pattern REPEATED_CHARS_PATTERN = Pattern.compile("(.)\\1{2,}");

    public enum PasswordStrength {
        VERY_WEAK,
        WEAK,
        FAIR,
        STRONG,
        VERY_STRONG
    }

    public static class ValidationResult {
        private final boolean valid;
        private final PasswordStrength strength;
        private final int score;
        private final List<String> errors;
        private final List<String> suggestions;

        public ValidationResult(boolean valid, PasswordStrength strength, int score,
                                List<String> errors, List<String> suggestions) {
            this.valid = valid;
            this.strength = strength;
            this.score = score;
            this.errors = errors;
            this.suggestions = suggestions;
        }

        public boolean isValid() {
            return valid;
        }

        public PasswordStrength getStrength() {
            return strength;
        }

        public int getScore() {
            return score;
        }

        public List<String> getErrors() {
            return errors;
        }

        public List<String> getSuggestions() {
            return suggestions;
        }

        @Override
        public String toString() {
            return "ValidationResult{" +
                    "valid=" + valid +
                    ", strength=" + strength +
                    ", score=" + score +
                    ", errors=" + errors +
                    ", suggestions=" + suggestions +
                    '}';
        }
    }

    /**
     * Validates password strength and returns a detailed validation result.
     *
     * @param password the password to validate
     * @return ValidationResult containing validity status, strength level, score, errors and suggestions
     */
    public static ValidationResult validatePassword(String password) {
        List<String> errors = new ArrayList<>();
        List<String> suggestions = new ArrayList<>();
        int score = 0;

        if (password == null || password.isEmpty()) {
            errors.add("Пароль не может быть пустым.");
            return new ValidationResult(false, PasswordStrength.VERY_WEAK, 0, errors, suggestions);
        }

        // Check for whitespace
        if (WHITESPACE_PATTERN.matcher(password).find()) {
            errors.add("Пароль не должен содержать пробелы.");
        }

        // Length checks
        int length = password.length();

        if (length < MIN_LENGTH) {
            errors.add("Пароль должен содержать не менее " + MIN_LENGTH + " символов.");
        } else if (length >= MIN_LENGTH && length < 12) {
            score += 1;
            suggestions.add("Увеличьте длину пароля до 12 и более символов для большей надёжности.");
        } else if (length >= 12 && length < 16) {
            score += 2;
        } else if (length >= 16) {
            score += 3;
        }

        if (length > MAX_LENGTH) {
            errors.add("Пароль не должен превышать " + MAX_LENGTH + " символов.");
        }

        // Uppercase check
        boolean hasUppercase = UPPERCASE_PATTERN.matcher(password).find();
        if (!hasUppercase) {
            errors.add("Пароль должен содержать хотя бы одну заглавную букву (A-Z).");
            suggestions.add("Добавьте заглавные буквы.");
        } else {
            score += 1;
        }

        // Lowercase check
        boolean hasLowercase = LOWERCASE_PATTERN.matcher(password).find();
        if (!hasLowercase) {
            errors.add("Пароль должен содержать хотя бы одну строчную букву (a-z).");
            suggestions.add("Добавьте строчные буквы.");
        } else {
            score += 1;
        }

        // Digit check
        boolean hasDigit = DIGIT_PATTERN.matcher(password).find();
        if (!hasDigit) {
            errors.add("Пароль должен содержать хотя бы одну цифру (0-9).");
            suggestions.add("Добавьте цифры.");
        } else {
            score += 1;
        }

        // Special character check
        boolean hasSpecialChar = SPECIAL_CHAR_PATTERN.matcher(password).find();
        if (!hasSpecialChar) {
            errors.add("Пароль должен содержать хотя бы один специальный символ (!@#$%^&* и т.д.).");
            suggestions.add("Добавьте специальные символы, например: !, @, #, $, %.");
        } else {
            score += 2;
        }

        // Check for repeated characters (e.g., "aaa", "111")
        if (REPEATED_CHARS_PATTERN.matcher(password).find()) {
            score -= 1;
            suggestions.add("Избегайте повторяющихся символов подряд (например, 'aaa' или '111').");
        }

        // Check for common weak passwords
        if (isCommonPassword(password)) {
            errors.add("Пароль слишком распространён. Выберите более уникальный пароль.");
            score = Math.max(0, score - 3);
        }

        // Check for sequential characters (e.g., "abc", "123")
        if (hasSequentialChars(password)) {
            score -= 1;
            suggestions.add("Избегайте последовательных символов (например, 'abc' или '123').");
        }

        // Bonus score for mixing character types well
        int charTypeCount = (hasUppercase ? 1 : 0) + (hasLowercase ? 1 : 0)
                + (hasDigit ? 1 : 0) + (hasSpecialChar ? 1 : 0);
        if (charTypeCount == 4) {
            score += 1;
        }

        // Normalize score
        score = Math.max(0, score);

        PasswordStrength strength = calculateStrength(score);
        boolean isValid = errors.isEmpty() && strength != PasswordStrength.VERY_WEAK;

        return new ValidationResult(isValid, strength, score, errors, suggestions);
    }

    /**
     * Calculates password strength based on score.
     */
    private static PasswordStrength calculateStrength(int score) {
        if (score <= 1) {
            return PasswordStrength.VERY_WEAK;
        } else if (score <= 3) {
            return PasswordStrength.WEAK;
        } else if (score <= 5) {
            return PasswordStrength.FAIR;
        } else if (score <= 7) {
            return PasswordStrength.STRONG;
        } else {
            return PasswordStrength.VERY_STRONG;
        }
    }

    /**
     * Checks if the password is in the list of commonly used passwords.
     */
    private static boolean isCommonPassword(String password) {
        String lowerPassword = password.toLowerCase();
        String[] commonPasswords = {
                "password", "123456", "12345678", "qwerty", "abc123",
                "monkey", "1234567", "letmein", "trustno1", "dragon",
                "baseball", "iloveyou", "master", "sunshine", "ashley",
                "bailey", "passw0rd", "shadow", "123123", "654321",
                "superman", "qazwsx", "michael", "football", "password1",
                "password123", "admin", "welcome", "login", "hello"
        };

        for (String common : commonPasswords) {
            if (lowerPassword.equals(common)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Checks if the password contains sequential characters (e.g., "abc", "123", "xyz").
     */
    private static boolean hasSequentialChars(String password) {
        String lowerPassword = password.toLowerCase();
        int sequenceLength = 3;

        for (int i = 0; i <= lowerPassword.length() - sequenceLength; i++) {
            boolean ascending = true;
            boolean descending = true;

            for (int j = i; j < i + sequenceLength - 1; j++) {
                if (lowerPassword.charAt(j + 1) != lowerPassword.charAt(j) + 1) {
                    ascending = false;
                }
                if (lowerPassword.charAt(j + 1) != lowerPassword.charAt(j) - 1) {
                    descending = false;
                }
            }

            if (ascending || descending) {
                return true;
            }
        }
        return false;
    }

    // Example usage and testing
    public static void main(String[] args) {
        String[] testPasswords = {
                "",
                "abc",
                "password",
                "Password1",
                "P@ssw0rd",
                "MyStr0ng!Pass#2024",
                "aaaaaa1A!",
                "Tr0ub4dor&3",
                "correcthorsebatterystaple",
                "X9#mK2$pL7@nQ4!w"
        };

        System.out.println("=== Password Strength Validator ===\n");

        for (String password : testPasswords) {
            System.out.println("Password: \"" + password + "\"");
            ValidationResult result = validatePassword(password);
            System.out.println("  Valid:      " + result.isValid());
            System.out.println("  Strength:   " + result.getStrength());
            System.out.println("  Score:      " + result.getScore());

            if (!result.getErrors().isEmpty()) {
                System.out.println("  Errors:");
                result.getErrors().forEach(e -> System.out.println("    - " + e));
            }

            if (!result.getSuggestions().isEmpty()) {
                System.out.println("  Suggestions:");
                result.getSuggestions().forEach(s -> System.out.println("    * " + s));
            }

            System.out.println();
        }
    }
}
