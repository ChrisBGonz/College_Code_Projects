
```python
import anthropic
import json
import re

client = anthropic.Anthropic()

PAYMENT_GATEWAY_TOOLS = [
    {
        "name": "initiate_payment",
        "description": "Initiates a payment transaction with the payment gateway and returns a success_url for redirect",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "Payment amount in cents"
                },
                "currency": {
                    "type": "string",
                    "description": "Currency code (e.g., USD, EUR)"
                },
                "order_id": {
                    "type": "string",
                    "description": "Unique order identifier"
                },
                "customer_email": {
                    "type": "string",
                    "description": "Customer email address"
                }
            },
            "required": ["amount", "currency", "order_id", "customer_email"]
        }
    },
    {
        "name": "redirect_user",
        "description": "Redirects the user to the provided URL",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "URL to redirect the user to"
                },
                "method": {
                    "type": "string",
                    "description": "Redirect method (GET or POST)",
                    "enum": ["GET", "POST"]
                }
            },
            "required": ["url"]
        }
    }
]


def simulate_payment_gateway(amount: int, currency: str, order_id: str, customer_email: str) -> dict:
    """Simulates a payment gateway API call that returns a success_url"""
    success_url = f"https://payment-gateway.example.com/success?order_id={order_id}&token=abc123xyz&amount={amount}&currency={currency}"
    
    return {
        "status": "pending",
        "transaction_id": f"txn_{order_id}_001",
        "success_url": success_url,
        "expires_at": "2024-12-31T23:59:59Z",
        "message": "Payment initiated successfully. Redirect user to success_url to complete payment."
    }


def simulate_redirect(url: str, method: str = "GET") -> dict:
    """Simulates redirecting a user to a URL"""
    print(f"\n{'='*60}")
    print(f"REDIRECTING USER")
    print(f"{'='*60}")
    print(f"Method: {method}")
    print(f"URL: {url}")
    print(f"{'='*60}")
    
    return {
        "status": "redirected",
        "url": url,
        "method": method,
        "message": f"User successfully redirected to {url}"
    }


def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "initiate_payment":
        result = simulate_payment_gateway(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            order_id=tool_input["order_id"],
            customer_email=tool_input["customer_email"]
        )
        print(f"\n[Tool: initiate_payment] Payment initiated:")
        print(f"  Transaction ID: {result['transaction_id']}")
        print(f"  Success URL: {result['success_url']}")
        return json.dumps(result)
    
    elif tool_name == "redirect_user":
        result = simulate_redirect(
            url=tool_input["url"],
            method=tool_input.get("method", "GET")
        )
        return json.dumps(result)
    
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})


def process_payment(amount: int, currency: str, order_id: str, customer_email: str):
    """Main function to process payment using Claude as AI orchestrator"""
    print(f"\n{'='*60}")
    print("PAYMENT PROCESSING SYSTEM")
    print(f"{'='*60}")
    print(f"Processing payment:")
    print(f"  Amount: {amount} {currency}")
    print(f"  Order ID: {order_id}")
    print(f"  Customer: {customer_email}")
    print(f"{'='*60}\n")
    
    messages = [
        {
            "role": "user",
            "content": f"""Process a payment for the following order:
            - Amount: {amount} cents ({currency})
            - Order ID: {order_id}
            - Customer Email: {customer_email}
            
            Please:
            1. First initiate the payment with the payment gateway to get the success_url
            2. Then redirect the user to that success_url
            
            Make sure to complete both steps."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=PAYMENT_GATEWAY_TOOLS,
            messages=messages
        )
        
        print(f"[Claude] Stop reason: {response.stop_reason}")
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n[Claude] Calling tool: {tool_name}")
                    print(f"[Claude] Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_response = ""
            for content_block in response.content:
                if hasattr(content_block, "text"):
                    final_response = content_block.text
                    break
            
            print(f"\n[Claude] Final response: {final_response}")
            print(f"\n{'='*60}")
            print("PAYMENT PROCESSING COMPLETE")
            print(f"{'='*60}")
            break
        
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return response


def main():
    """Main entry point"""
    process_payment(
        amount=9999,
        currency="USD",
        order_id="ORD-2024-001",
        customer_email="customer@example.com"
    )
    
    print("\n" + "="*60)
    print("Processing second payment...")
    print("="*60)
    
    process_payment(
        amount=4999,
        currency="EUR",
        order_id="ORD-2024-002",
        customer_email="another.customer@example.com"
    )


if __name__ == "__main__":
    main()
```
