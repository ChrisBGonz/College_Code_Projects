using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Collections.Generic;

[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; }
    public string Version { get; set; }
    public string Theme { get; set; }
    public string Language { get; set; }
    public int MaxConnections { get; set; }
    public bool EnableLogging { get; set; }
    public bool AutoSave { get; set; }
    public int AutoSaveInterval { get; set; }
    public string DefaultOutputPath { get; set; }
    public Dictionary<string, string> CustomSettings { get; set; }
    public DateTime LastModified { get; set; }

    public AppSettings()
    {
        ApplicationName = "MyApplication";
        Version = "1.0.0";
        Theme = "Light";
        Language = "en-US";
        MaxConnections = 10;
        EnableLogging = true;
        AutoSave = false;
        AutoSaveInterval = 300;
        DefaultOutputPath = Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments);
        CustomSettings = new Dictionary<string, string>();
        LastModified = DateTime.Now;
    }
}

public class SettingsManager
{
    private readonly string _settingsFilePath;
    private AppSettings _currentSettings;

    public SettingsManager(string settingsFilePath = "appsettings.bin")
    {
        _settingsFilePath = settingsFilePath;
        _currentSettings = new AppSettings();
    }

    public AppSettings CurrentSettings => _currentSettings;

    public void SaveSettings(AppSettings settings, string filePath = null)
    {
        string targetPath = filePath ?? _settingsFilePath;
        
        settings.LastModified = DateTime.Now;
        
        string directory = Path.GetDirectoryName(targetPath);
        if (!string.IsNullOrEmpty(directory) && !Directory.Exists(directory))
        {
            Directory.CreateDirectory(directory);
        }

        using (FileStream fileStream = new FileStream(targetPath, FileMode.Create, FileAccess.Write))
        {
            SaveSettingsToStream(settings, fileStream);
        }
        
        _currentSettings = settings;
        Console.WriteLine($"Settings saved to: {targetPath}");
    }

    public void SaveSettingsToStream(AppSettings settings, Stream stream)
    {
        if (settings == null)
            throw new ArgumentNullException(nameof(settings));
        
        if (stream == null)
            throw new ArgumentNullException(nameof(stream));

        if (!stream.CanWrite)
            throw new InvalidOperationException("Stream is not writable");

#pragma warning disable SYSLIB0011
        BinaryFormatter formatter = new BinaryFormatter();
        formatter.Serialize(stream, settings);
#pragma warning restore SYSLIB0011
    }

    public AppSettings LoadSettings(string filePath = null)
    {
        string targetPath = filePath ?? _settingsFilePath;
        
        if (!File.Exists(targetPath))
        {
            Console.WriteLine($"Settings file not found: {targetPath}. Using default settings.");
            _currentSettings = new AppSettings();
            return _currentSettings;
        }

        using (FileStream fileStream = new FileStream(targetPath, FileMode.Open, FileAccess.Read))
        {
            _currentSettings = LoadSettingsFromStream(fileStream);
        }
        
        Console.WriteLine($"Settings loaded from: {targetPath}");
        return _currentSettings;
    }

    public AppSettings LoadSettingsFromStream(Stream stream)
    {
        if (stream == null)
            throw new ArgumentNullException(nameof(stream));
        
        if (!stream.CanRead)
            throw new InvalidOperationException("Stream is not readable");

#pragma warning disable SYSLIB0011
        BinaryFormatter formatter = new BinaryFormatter();
        AppSettings settings = (AppSettings)formatter.Deserialize(stream);
#pragma warning restore SYSLIB0011
        
        return settings;
    }

    public byte[] SerializeToBytes(AppSettings settings)
    {
        if (settings == null)
            throw new ArgumentNullException(nameof(settings));

        using (MemoryStream memoryStream = new MemoryStream())
        {
            SaveSettingsToStream(settings, memoryStream);
            return memoryStream.ToArray();
        }
    }

    public AppSettings DeserializeFromBytes(byte[] data)
    {
        if (data == null || data.Length == 0)
            throw new ArgumentException("Data cannot be null or empty", nameof(data));

        using (MemoryStream memoryStream = new MemoryStream(data))
        {
            return LoadSettingsFromStream(memoryStream);
        }
    }

    public bool TryLoadSettings(out AppSettings settings, string filePath = null)
    {
        try
        {
            settings = LoadSettings(filePath);
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Failed to load settings: {ex.Message}");
            settings = new AppSettings();
            return false;
        }
    }

    public bool TrySaveSettings(AppSettings settingsToSave, string filePath = null)
    {
        try
        {
            SaveSettings(settingsToSave, filePath);
            return true;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Failed to save settings: {ex.Message}");
            return false;
        }
    }

    public void CreateBackup(string backupPath = null)
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine("No settings file to backup");
            return;
        }

        string backup = backupPath ?? $"{_settingsFilePath}.backup_{DateTime.Now:yyyyMMdd_HHmmss}";
        File.Copy(_settingsFilePath, backup, true);
        Console.WriteLine($"Backup created: {backup}");
    }

    public void ResetToDefaults()
    {
        _currentSettings = new AppSettings();
        SaveSettings(_currentSettings);
        Console.WriteLine("Settings reset to defaults");
    }

    public void PrintSettings(AppSettings settings = null)
    {
        AppSettings settingsToPrint = settings ?? _currentSettings;
        
        Console.WriteLine("\n=== Application Settings ===");
        Console.WriteLine($"Application Name: {settingsToPrint.ApplicationName}");
        Console.WriteLine($"Version: {settingsToPrint.Version}");
        Console.WriteLine($"Theme: {settingsToPrint.Theme}");
        Console.WriteLine($"Language: {settingsToPrint.Language}");
        Console.WriteLine($"Max Connections: {settingsToPrint.MaxConnections}");
        Console.WriteLine($"Enable Logging: {settingsToPrint.EnableLogging}");
        Console.WriteLine($"Auto Save: {settingsToPrint.AutoSave}");
        Console.WriteLine($"Auto Save Interval: {settingsToPrint.AutoSaveInterval} seconds");
        Console.WriteLine($"Default Output Path: {settingsToPrint.DefaultOutputPath}");
        Console.WriteLine($"Last Modified: {settingsToPrint.LastModified}");
        
        if (settingsToPrint.CustomSettings.Count > 0)
        {
            Console.WriteLine("Custom Settings:");
            foreach (var kvp in settingsToPrint.CustomSettings)
            {
                Console.WriteLine($"  {kvp.Key}: {kvp.Value}");
            }
        }
        Console.WriteLine("============================\n");
    }
}

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("=== Settings Manager Demo ===\n");
        
        SettingsManager manager = new SettingsManager("settings.bin");
        
        AppSettings settings = new AppSettings
        {
            ApplicationName = "Demo Application",
            Version = "2.1.0",
            Theme = "Dark",
            Language = "ru-RU",
            MaxConnections = 25,
            EnableLogging = true,
            AutoSave = true,
            AutoSaveInterval = 60,
            DefaultOutputPath = "/tmp/output",
            CustomSettings = new Dictionary<string, string>
            {
                { "DatabaseHost", "localhost" },
                { "DatabasePort", "5432" },
                { "ApiKey", "secret-key-123" }
            }
        };
        
        Console.WriteLine("1. Saving settings to file...");
        manager.SaveSettings(settings);
        manager.PrintSettings(settings);
        
        Console.WriteLine("2. Loading settings from file...");
        AppSettings loadedSettings = manager.LoadSettings();
        manager.PrintSettings(loadedSettings);
        
        Console.WriteLine("3. Serializing to bytes...");
        byte[] settingsBytes = manager.SerializeToBytes(settings);
        Console.WriteLine($"Serialized size: {settingsBytes.Length} bytes");
        
        Console.WriteLine("\n4. Deserializing from bytes...");
        AppSettings deserializedSettings = manager.DeserializeFromBytes(settingsBytes);
        manager.PrintSettings(deserializedSettings);
        
        Console.WriteLine("5. Testing stream-based operations...");
        using (MemoryStream memStream = new MemoryStream())
        {
            manager.SaveSettingsToStream(settings, memStream);
            Console.WriteLine($"Written {memStream.Length} bytes to memory stream");
            
            memStream.Position = 0;
            AppSettings streamSettings = manager.LoadSettingsFromStream(memStream);
            Console.WriteLine($"Loaded settings from stream: {streamSettings.ApplicationName} v{streamSettings.Version}");
        }
        
        Console.WriteLine("\n6. Creating backup...");
        manager.CreateBackup();
        
        Console.WriteLine("\n7. Testing TryLoad/TrySave...");
        bool saveSuccess = manager.TrySaveSettings(settings, "test_settings.bin");
        Console.WriteLine($"Save successful: {saveSuccess}");
        
        AppSettings tryLoadedSettings;
        bool loadSuccess = manager.TryLoadSettings(out tryLoadedSettings, "test_settings.bin");
        Console.WriteLine($"Load successful: {loadSuccess}");
        Console.WriteLine($"Loaded: {tryLoadedSettings.ApplicationName}");
        
        Console.WriteLine("\n8. Testing with non-existent file...");
        AppSettings defaultSettings;
        bool loadResult = manager.TryLoadSettings(out defaultSettings, "nonexistent.bin");
        Console.WriteLine($"Load from non-existent file: {loadResult}");
        Console.WriteLine($"Got default settings: {defaultSettings.ApplicationName}");
        
        Console.WriteLine("\n=== Demo Complete ===");
    }
}
