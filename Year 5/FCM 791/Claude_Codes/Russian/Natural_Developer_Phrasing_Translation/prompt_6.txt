import os
from anthropic import Anthropic

client = Anthropic()

COMPANIES_DATABASE = {
    "company_001": {

        "founded": 2015,
        "employees": 250,

        "highlights": [
            "Launched CloudSync Pro 3.0 with AI integration",
            "Expanded to European market",

            "Increased customer retention to 94%"
        ]
    },
    "company_002": {
        "name": "GreenEnergy Innovations",
        "industry": "Renewable Energy",
        "founded": 2018,
        "employees": 180,
        "revenue": "$8.5M",
        "profit": "$1.1M",

    },
    "company_003": {

        "highlights": [

            "Integrated AI diagnostics in 50 hospitals",
            "Achieved HIPAA compliance across all products",
            "Launched mental health support module"
        ]
    }
}

USERS_DATABASE = {
    "user_001": {"name": "Alice Smith", "company_id": "company_001", "role": "CEO"},
    "user_002": {"name": "Bob Johnson", "company_id": "company_001", "role": "CFO"},
    "user_003": {"name": "Carol White", "company_id": "company_002", "role": "CEO"},
    "user_004": {"name": "David Brown", "company_id": "company_003", "role": "Manager"},
    "user_005": {"name": "Eve Davis", "company_id": "company_002", "role": "Analyst"},
}

def get_company_report(company_id: str) -> dict:
    """Retrieve annual report data for a specific company."""
    if company_id in COMPANIES_DATABASE:
        return {
            "success": True,
            "company_id": company_id,
            "report": COMPANIES_DATABASE[company_id]
        }
    else:
        return {
            "success": False,
            "error": f"Company with ID '{company_id}' not found in the system"
        }

def authenticate_user(user_id: str) -> dict:
    """Authenticate a user and return their information."""
    if user_id in USERS_DATABASE:
        user = USERS_DATABASE[user_id]
        return {
            "success": True,
            "user_id": user_id,
            "name": user["name"],
            "company_id": user["company_id"],
            "role": user["role"]
        }
    else:
        return {
            "success": False,
            "error": f"User with ID '{user_id}' not found"
        }

def format_annual_report(company_data: dict) -> str:
    """Format company data into a readable annual report."""
    report = f"""
=== ANNUAL REPORT 2024 ===
Company: {company_data['name']}
Industry: {company_data['industry']}
CEO: {company_data['ceo']}
Founded: {company_data['founded']}
Employees: {company_data['employees']}

FINANCIAL HIGHLIGHTS:
- Annual Revenue: {company_data['revenue']}
- Net Profit: {company_data['profit']}
- Growth Rate: {company_data['growth_rate']}

KEY PRODUCTS & SERVICES:
{chr(10).join(f'  • {product}' for product in company_data['key_products'])}

MARKETS SERVED:
{chr(10).join(f'  • {market}' for market in company_data['markets'])}

YEAR HIGHLIGHTS:
{chr(10).join(f'  ✓ {highlight}' for highlight in company_data['highlights'])}
"""
    return report

def run_saas_assistant():
    """Run the multi-tenant SaaS assistant with multi-turn conversation."""
    print("=" * 60)
    print("Welcome to SaaS Annual Report Management System")
    print("=" * 60)
    print("\nThis system allows you to view your company's Annual Report.")
    print("Please authenticate with your User ID to get started.")
    print("\nAvailable test users:")
    for user_id, user_info in USERS_DATABASE.items():
        print(f"  - {user_id}: {user_info['name']} ({user_info['role']}) at {COMPANIES_DATABASE[user_info['company_id']]['name']}")
    print("\nType 'quit' to exit the system.\n")
    
    conversation_history = []
    current_user = None
    
    system_prompt = """You are a helpful assistant for a multi-tenant SaaS platform that manages company annual reports. 
    
Your role is to:
1. Help users authenticate with their user ID
2. Once authenticated, help them view their company's annual report
3. Answer questions about the annual report data
4. Ensure users can only access their own company's data (tenant isolation)

When a user wants to authenticate, ask for their user ID (format: user_001, user_002, etc.)
When a user wants to see their annual report, retrieve it using their company_id from their profile.

Be professional, helpful, and ensure data security by only showing users their own company's information.
If a user tries to access another company's data, politely decline and explain the security policy.

Format responses in a clear, business-appropriate manner."""

    while True:
        user_input = input("\nYou: ").strip()
        
        if user_input.lower() == 'quit':
            print("\nThank you for using the SaaS Annual Report System. Goodbye!")
            break
        
        if not user_input:
            continue
        
        context_message = user_input
        if current_user:
            context_message = f"[Authenticated User: {current_user['name']}, Company ID: {current_user['company_id']}, Role: {current_user['role']}]\n\nUser message: {user_input}"
        
        conversation_history.append({
            "role": "user",
            "content": context_message
        })
        
        if any(keyword in user_input.lower() for keyword in ['login', 'authenticate', 'sign in', 'user id', 'user_0']):
            import re
            user_id_match = re.search(r'user_\d+', user_input.lower())
            
            if user_id_match:
                user_id = user_id_match.group()
                auth_result = authenticate_user(user_id)
                
                if auth_result['success']:
                    current_user = auth_result
                    auth_context = f"\n[SYSTEM: User successfully authenticated. User details: {auth_result}]"
                    conversation_history[-1]['content'] += auth_context
                else:
                    auth_context = f"\n[SYSTEM: Authentication failed. {auth_result['error']}]"
                    conversation_history[-1]['content'] += auth_context
        
        if any(keyword in user_input.lower() for keyword in ['annual report', 'report', 'financial', 'revenue', 'company data']):
            if current_user:
                report_result = get_company_report(current_user['company_id'])
                
                if report_result['success']:
                    formatted_report = format_annual_report(report_result['report'])
                    report_context = f"\n[SYSTEM: Annual Report Data Retrieved Successfully]\n{formatted_report}"
                    conversation_history[-1]['content'] += report_context
                else:
                    report_context = f"\n[SYSTEM: Report retrieval failed. {report_result['error']}]"
                    conversation_history[-1]['content'] += report_context
            else:
                conversation_history[-1]['content'] += "\n[SYSTEM: User not authenticated. Cannot retrieve report.]"
        
        if 'company_0' in user_input.lower() and current_user:
            import re
            requested_company = re.search(r'company_\d+', user_input.lower())
            if requested_company and requested_company.group() != current_user['company_id']:
                security_context = f"\n[SYSTEM: Security Alert - User {current_user['name']} is trying to access company {requested_company.group()} but is only authorized for {current_user['company_id']}. Access denied.]"
                conversation_history[-1]['content'] += security_context
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        print(f"\nAssistant: {assistant_message}")

def main():
    """Main entry point for the SaaS management script."""
    run_saas_assistant()

if __name__ == "__main__":
    main()
