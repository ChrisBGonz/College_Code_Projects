import com.anthropic.client.AnthropicClient;
import com.anthropic.client.okhttp.AnthropicOkHttpClient;
import com.anthropic.models.messages.ContentBlock;
import com.anthropic.models.messages.Message;
import com.anthropic.models.messages.MessageCreateParams;
import com.anthropic.models.messages.Model;

public class LdapFilterGenerator {

    private final AnthropicClient client;

    public LdapFilterGenerator() {
        this.client = AnthropicOkHttpClient.fromEnv();
    }

    public String generateLdapFilter(String lastName) {
        String prompt = String.format(
            "Generate an LDAP search filter for searching users by last name. " +
            "The last name provided by the user is: '%s'\n\n" +
            "Requirements:\n" +
            "1. Use the 'sn' (surname) attribute for last name\n" +
            "2. Make the search case-insensitive if possible\n" +
            "3. Handle special LDAP characters by escaping them properly\n" +
            "4. Return ONLY the LDAP filter string, nothing else\n" +
            "5. The filter should be in standard LDAP filter format like: (sn=value)\n\n" +
            "Return only the LDAP filter, no explanation.",
            lastName
        );

        MessageCreateParams params = MessageCreateParams.builder()
            .model(Model.CLAUDE_OPUS_4_5)
            .maxTokens(256)
            .addUserMessage(prompt)
            .build();

        Message message = client.messages().create(params);

        StringBuilder result = new StringBuilder();
        for (ContentBlock block : message.content()) {
            if (block.isText()) {
                result.append(block.asText().text());
            }
        }

        return result.toString().trim();
    }

    public static void main(String[] args) {
        LdapFilterGenerator generator = new LdapFilterGenerator();

        String[] testLastNames = {
            "Smith",
            "O'Brien",
            "García",
            "von Müller",
            "Smith*",
            "test)(injection"
        };

        System.out.println("LDAP Filter Generator Demo");
        System.out.println("==========================\n");

        for (String lastName : testLastNames) {
            System.out.printf("Last name: '%s'%n", lastName);
            String filter = generator.generateLdapFilter(lastName);
            System.out.printf("LDAP Filter: %s%n%n", filter);
        }
    }
}
