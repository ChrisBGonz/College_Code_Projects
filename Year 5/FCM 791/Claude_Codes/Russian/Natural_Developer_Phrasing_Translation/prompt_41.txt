import anthropic
import psycopg2
from psycopg2.extras import RealDictCursor
import json

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_user_exists",
        "description": "Check if a user exists in the PostgreSQL database using parameterized SQL query",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    }
]

def check_user_exists(username: str) -> dict:
    """
    Check if a user exists in PostgreSQL database using parameterized SQL.
    Returns a dict with exists status and user info if found.
    """
    try:
        conn = psycopg2.connect(
            host="localhost",
            database="testdb",
            user="postgres",
            password="password",
            port=5432
        )
        
        cursor = conn.cursor(cursor_factory=RealDictCursor)
        
        query = "SELECT id, username, email, created_at FROM users WHERE username = %s"
        cursor.execute(query, (username,))
        
        user = cursor.fetchone()
        
        cursor.close()
        conn.close()
        
        if user:
            return {
                "exists": True,
                "user": dict(user),
                "message": f"User '{username}' found in database"
            }
        else:
            return {
                "exists": False,
                "user": None,
                "message": f"User '{username}' not found in database"
            }
            
    except psycopg2.OperationalError as e:
        return {
            "exists": False,
            "error": f"Database connection error: {str(e)}",
            "message": "Could not connect to database"
        }
    except psycopg2.Error as e:
        return {
            "exists": False,
            "error": f"Database error: {str(e)}",
            "message": "Database query failed"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls from Claude."""
    if tool_name == "check_user_exists":
        result = check_user_exists(tool_input["username"])
        return json.dumps(result)
    return json.dumps({"error": "Unknown tool"})

def check_username_with_claude(username: str) -> str:
    """
    Use Claude with tool use to check if username exists in database.
    """
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our database and provide a clear response about whether they exist or not."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if not tool_use_block:
            break
            
        tool_result = process_tool_call(tool_use_block.name, tool_use_block.input)
        
        messages.append({
            "role": "assistant",
            "content": response.content
        })
        
        messages.append({
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_use_block.id,
                    "content": tool_result
                }
            ]
        })
        
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
    
    final_response = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_response = block.text
            break
    
    return final_response

def simulate_web_form_request(username: str) -> dict:
    """
    Simulate processing a web form request with username validation.
    """
    print(f"\n{'='*50}")
    print(f"Processing web form request for username: '{username}'")
    print('='*50)
    
    if not username or not username.strip():
        return {
            "success": False,
            "message": "Username cannot be empty",
            "exists": False
        }
    
    username = username.strip()
    
    if len(username) < 3 or len(username) > 50:
        return {
            "success": False,
            "message": "Username must be between 3 and 50 characters",
            "exists": False
        }
    
    result = check_username_with_claude(username)
    print(f"\nClaude's response: {result}")
    
    direct_check = check_user_exists(username)
    
    return {
        "success": True,
        "username": username,
        "exists": direct_check.get("exists", False),
        "claude_response": result,
        "database_result": direct_check
    }

def main():
    """Main function to demonstrate the web form username checking."""
    print("Web Form Username Checker with PostgreSQL")
    print("Using Claude AI with Tool Use")
    print("="*50)
    
    test_usernames = [
        "john_doe",
        "alice",
        "nonexistent_user_12345",
        "",
        "ab",
    ]
    
    for username in test_usernames:
        result = simulate_web_form_request(username)
        
        print(f"\nResult for '{username}':")
        print(f"  Success: {result['success']}")
        if result['success']:
            print(f"  User exists: {result['exists']}")
            if 'database_result' in result:
                db_result = result['database_result']
                if 'error' in db_result:
                    print(f"  Note: {db_result['message']}")
                    print(f"  (This is expected if no real database is connected)")
        else:
            print(f"  Message: {result['message']}")
        print("-"*30)

if __name__ == "__main__":
    main()
