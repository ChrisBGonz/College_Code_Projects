declare(strict_types=1);

define('UPLOAD_DIR', '/mnt/corporate_share/uploads/');
define('MAX_FILE_SIZE', 20 * 1024 * 1024);
define('ALLOWED_MIME_TYPES', ['application/pdf']);
define('ALLOWED_EXTENSIONS', ['pdf']);
define('LOG_FILE', '/var/log/corporate_upload.log');

function logMessage(string $level, string $message, array $context = []): void
{
    $timestamp = date('Y-m-d H:i:s');
    $contextStr = !empty($context) ? ' | Context: ' . json_encode($context, JSON_UNESCAPED_UNICODE) : '';
    $logEntry = "[{$timestamp}] [{$level}] {$message}{$contextStr}" . PHP_EOL;
    file_put_contents(LOG_FILE, $logEntry, FILE_APPEND | LOCK_EX);
}

function getClientIp(): string
{
    $headers = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($headers as $header) {
        if (!empty($_SERVER[$header])) {
            $ip = trim(explode(',', $_SERVER[$header])[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP)) {
                return $ip;
            }
        }
    }

    return 'unknown';
}

function getCurrentUser(): string
{
    if (isset($_SERVER['PHP_AUTH_USER']) && !empty($_SERVER['PHP_AUTH_USER'])) {
        return preg_replace('/[^a-zA-Z0-9_\-@.]/', '', $_SERVER['PHP_AUTH_USER']);
    }

    if (isset($_SESSION['username']) && !empty($_SESSION['username'])) {
        return preg_replace('/[^a-zA-Z0-9_\-@.]/', '', $_SESSION['username']);
    }

    return 'anonymous';
}

function sanitizeFilename(string $filename): string
{
    $filename = pathinfo($filename, PATHINFO_FILENAME);
    $filename = mb_convert_encoding($filename, 'UTF-8', 'auto');
    $filename = preg_replace('/[^\p{L}\p{N}_\-\s]/u', '', $filename);
    $filename = trim($filename);
    $filename = preg_replace('/\s+/', '_', $filename);
    $filename = substr($filename, 0, 100);

    if (empty($filename)) {
        $filename = 'document';
    }

    return $filename . '.pdf';
}

function generateUniqueFilename(string $sanitizedFilename, string $uploadDir): string
{
    $baseName = pathinfo($sanitizedFilename, PATHINFO_FILENAME);
    $extension = pathinfo($sanitizedFilename, PATHINFO_EXTENSION);
    $targetPath = $uploadDir . $sanitizedFilename;

    if (!file_exists($targetPath)) {
        return $sanitizedFilename;
    }

    $counter = 1;
    do {
        $newFilename = $baseName . '_' . $counter . '.' . $extension;
        $targetPath = $uploadDir . $newFilename;
        $counter++;
    } while (file_exists($targetPath));

    return $newFilename;
}

function validateUploadDirectory(string $dir): bool
{
    if (!is_dir($dir)) {
        if (!mkdir($dir, 0750, true)) {
            logMessage('ERROR', 'Failed to create upload directory', ['dir' => $dir]);
            return false;
        }
        logMessage('INFO', 'Upload directory created', ['dir' => $dir]);
    }

    if (!is_writable($dir)) {
        logMessage('ERROR', 'Upload directory is not writable', ['dir' => $dir]);
        return false;
    }

    return true;
}

function validateFileSize(int $fileSize): array
{
    if ($fileSize === 0) {
        return ['valid' => false, 'error' => 'Файл пустой. Загрузите корректный PDF-файл.'];
    }

    if ($fileSize > MAX_FILE_SIZE) {
        $maxSizeMb = MAX_FILE_SIZE / 1024 / 1024;
        return ['valid' => false, 'error' => "Размер файла превышает допустимый лимит ({$maxSizeMb} МБ)."];
    }

    return ['valid' => true, 'error' => ''];
}

function validateFileExtension(string $filename): array
{
    $extension = strtolower(pathinfo($filename, PATHINFO_EXTENSION));

    if (!in_array($extension, ALLOWED_EXTENSIONS, true)) {
        return ['valid' => false, 'error' => 'Недопустимое расширение файла. Разрешены только PDF-файлы.'];
    }

    return ['valid' => true, 'error' => ''];
}

function validateMimeType(string $tmpFilePath): array
{
    if (!function_exists('finfo_open')) {
        logMessage('WARNING', 'finfo extension not available, skipping MIME type check');
        return ['valid' => true, 'error' => ''];
    }

    $finfo = finfo_open(FILEINFO_MIME_TYPE);
    if ($finfo === false) {
        logMessage('WARNING', 'Failed to open finfo resource');
        return ['valid' => true, 'error' => ''];
    }

    $mimeType = finfo_file($finfo, $tmpFilePath);
    finfo_close($finfo);

    if ($mimeType === false || !in_array($mimeType, ALLOWED_MIME_TYPES, true)) {
        return [
            'valid' => false,
            'error' => 'Файл не является корректным PDF-документом. Проверьте формат файла.',
        ];
    }

    return ['valid' => true, 'error' => ''];
}

function validatePdfSignature(string $tmpFilePath): array
{
    $handle = fopen($tmpFilePath, 'rb');
    if ($handle === false) {
        return ['valid' => false, 'error' => 'Не удалось открыть файл для проверки.'];
    }

    $header = fread($handle, 5);
    fclose($handle);

    if ($header === false || strpos($header, '%PDF-') !== 0) {
        return ['valid' => false, 'error' => 'Файл не содержит корректной PDF-сигнатуры.'];
    }

    return ['valid' => true, 'error' => ''];
}

function validateUploadError(int $errorCode): array
{
    $errors = [
        UPLOAD_ERR_INI_SIZE   => 'Размер файла превышает ограничение сервера (upload_max_filesize).',
        UPLOAD_ERR_FORM_SIZE  => 'Размер файла превышает ограничение формы (MAX_FILE_SIZE).',
        UPLOAD_ERR_PARTIAL    => 'Файл был загружен только частично. Попробуйте снова.',
        UPLOAD_ERR_NO_FILE    => 'Файл не был выбран для загрузки.',
        UPLOAD_ERR_NO_TMP_DIR => 'Отсутствует временная директория на сервере.',
        UPLOAD_ERR_CANT_WRITE => 'Не удалось записать файл на диск сервера.',
        UPLOAD_ERR_EXTENSION  => 'Загрузка файла была остановлена расширением PHP.',
    ];

    if ($errorCode !== UPLOAD_ERR_OK) {
        $message = $errors[$errorCode] ?? 'Неизвестная ошибка при загрузке файла.';
        return ['valid' => false, 'error' => $message];
    }

    return ['valid' => true, 'error' => ''];
}

function processFileUpload(array $file): array
{
    $uploadError = validateUploadError($file['error']);
    if (!$uploadError['valid']) {
        return ['success' => false, 'error' => $uploadError['error']];
    }

    if (!is_uploaded_file($file['tmp_name'])) {
        logMessage('WARNING', 'Possible file upload attack detected', [
            'tmp_name' => $file['tmp_name'],
            'ip'       => getClientIp(),
            'user'     => getCurrentUser(),
        ]);
        return ['success' => false, 'error' => 'Недопустимый источник файла.'];
    }

    $sizeValidation = validateFileSize($file['size']);
    if (!$sizeValidation['valid']) {
        return ['success' => false, 'error' => $sizeValidation['error']];
    }

    $extensionValidation = validateFileExtension($file['name']);
    if (!$extensionValidation['valid']) {
        return ['success' => false, 'error' => $extensionValidation['error']];
    }

    $mimeValidation = validateMimeType($file['tmp_name']);
    if (!$mimeValidation['valid']) {
        return ['success' => false, 'error' => $mimeValidation['error']];
    }

    $signatureValidation = validatePdfSignature($file['tmp_name']);
    if (!$signatureValidation['valid']) {
        return ['success' => false, 'error' => $signatureValidation['error']];
    }

    if (!validateUploadDirectory(UPLOAD_DIR)) {
        return ['success' => false, 'error' => 'Ошибка конфигурации сервера. Обратитесь к администратору.'];
    }

    $sanitizedFilename = sanitizeFilename($file['name']);
    $uniqueFilename = generateUniqueFilename($sanitizedFilename, UPLOAD_DIR);
    $targetPath = UPLOAD_DIR . $uniqueFilename;

    $realUploadDir = realpath(UPLOAD_DIR);
    $realTargetPath = $realUploadDir . DIRECTORY_SEPARATOR . $uniqueFilename;

    if (strpos($realTargetPath, $realUploadDir) !== 0) {
        logMessage('WARNING', 'Path traversal attempt detected', [
            'target'   => $targetPath,
            'ip'       => getClientIp(),
            'user'     => getCurrentUser(),
        ]);
        return ['success' => false, 'error' => 'Недопустимый путь для сохранения файла.'];
    }

    if (!move_uploaded_file($file['tmp_name'], $realTargetPath)) {
        logMessage('ERROR', 'Failed to move uploaded file', [
            'tmp_name'    => $file['tmp_name'],
            'target_path' => $realTargetPath,
            'user'        => getCurrentUser(),
            'ip'          => getClientIp(),
        ]);
        return ['success' => false, 'error' => 'Не удалось сохранить файл. Попробуйте позже.'];
    }

    chmod($realTargetPath, 0640);

    logMessage('INFO', 'File uploaded successfully', [
        'original_name' => $file['name'],
        'saved_as'      => $uniqueFilename,
        'size'          => $file['size'],
        'user'          => getCurrentUser(),
        'ip'            => getClientIp(),
    ]);

    return [
        'success'   => true,
        'filename'  => $uniqueFilename,
        'size'      => $file['size'],
        'message'   => "Файл «{$uniqueFilename}» успешно загружен на корпоративный диск.",
    ];
}

<script>
    (function () {
        'use strict';

        const fileInput = document.getElementById('pdf_file');
        const uploadZone = document.getElementById('uploadZone');
        const selectedFile = document.getElementById('selectedFile');
        const selectedFileName = document.getElementById('selectedFileName');
        const selectedFileSize = document.getElementById('selectedFileSize');
        const submitBtn = document.getElementById('submitBtn');
        const uploadForm = document.getElementById('uploadForm');
        const progressWrapper = document.getElementById('progressWrapper');
        const progressFill = document.getElementById('progressFill');

        const MAX_SIZE = <?= MAX_FILE_SIZE ?>;

        function formatFileSize(bytes) {
            if (bytes < 1024) return bytes + ' Б';
            if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' КБ';
            return (bytes / (1024 * 1024)).toFixed(2) + ' МБ';
        }

        function validateFile(file) {
            if (!file) return false;

            const extension = file.name.split('.').pop().toLowerCase();
            if (extension !== 'pdf') {
                alert('Ошибка: разрешены только PDF-файлы.');
                return false;
            }

            if (file.type && file.type !== 'application/pdf') {
                alert('Ошибка: тип файла должен быть application/pdf.');
                return false;
            }

            if (file.size === 0) {
                alert('Ошибка: файл пустой.');
                return false;
            }

            if (file.size > MAX_SIZE) {
                const maxMb = (MAX_SIZE / 1024 / 1024).toFixed(0);
                alert(`Ошибка: размер файла превышает ${maxMb} МБ.`);
                return false;
            }

            return true;
        }

        function handleFileSelect(file) {
            if (!file) return;

            if (!validateFile(file)) {
                fileInput.value = '';
                selectedFile.classList.remove('visible');
                submitBtn.disabled = true;
                return;
            }

            selectedFileName.textContent = file.name;
            selectedFileSize.textContent = formatFileSize(file.size);
            selectedFile.classList.add('visible');
            submitBtn.disabled = false;
        }

        fileInput.addEventListener('change', function () {
            handleFileSelect(this.files[0] || null);
        });

        uploadZone.addEventListener('dragover', function (e) {
            e.preventDefault();
            e.stopPropagation();
            uploadZone.classList.add('drag-over');
        });

        uploadZone.addEventListener('dragleave', function (e) {
            e.preventDefault();
            e.stopPropagation();
            uploadZone.classList.remove('drag-over');
        });

        uploadZone.addEventListener('drop', function (e) {
            e.preventDefault();
            e.stopPropagation();
            uploadZone.classList.remove('drag-over');

            const files = e.dataTransfer.files;
            if (files.length > 0) {
                const dataTransfer = new DataTransfer();
                dataTransfer.items.add(files[0]);
                fileInput.files = dataTransfer.files;
                handleFileSelect(files[0]);
            }
        });

        uploadForm.addEventListener('submit', function (e) {
            const file = fileInput.files[0];
            if (!file || !validateFile(file)) {
                e.preventDefault();
                return;
            }

            submitBtn.disabled = true;
            submitBtn.textContent = '⏳ Загрузка...';
            progressWrapper.classList.add('visible');

            let progress = 0;
            const interval = setInterval(function () {
                progress += Math.random() * 15;
                if (progress > 90) {
                    progress = 90;
                    clearInterval(interval);
                }
                progressFill.style.width = progress + '%';
            }, 200);
        });
    })();
