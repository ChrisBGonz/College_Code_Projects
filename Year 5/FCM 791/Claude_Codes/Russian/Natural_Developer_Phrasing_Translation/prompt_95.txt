import anthropic
import os
import tempfile
import random
import string
import time

def generate_large_file(file_path: str, target_size_gb: float = 0.1) -> int:
    """Generate a large file with random text content."""
    target_size_bytes = int(target_size_gb * 1024 * 1024 * 1024)
    lines_written = 0
    
    print(f"Generating test file of ~{target_size_gb} GB...")
    
    with open(file_path, 'w', encoding='utf-8') as f:
        current_size = 0
        while current_size < target_size_bytes:
            line_length = random.randint(50, 200)
            line = ''.join(random.choices(string.ascii_letters + string.digits + ' ', k=line_length))
            f.write(line + '\n')
            current_size += len(line) + 1
            lines_written += 1
            
            if lines_written % 100000 == 0:
                print(f"  Written {lines_written:,} lines, {current_size / (1024*1024):.1f} MB...")
    
    print(f"File generated: {lines_written:,} lines, {os.path.getsize(file_path) / (1024*1024*1024):.3f} GB")
    return lines_written

def count_lines_with_claude_streaming(file_path: str) -> None:
    """Use Claude with streaming to analyze and count lines in a large file."""
    
    client = anthropic.Anthropic()
    
    file_size = os.path.getsize(file_path)
    file_size_gb = file_size / (1024 * 1024 * 1024)
    
    print(f"\nLoading file into RAM and counting lines...")
    print(f"File size: {file_size_gb:.3f} GB")
    
    start_time = time.time()
    
    with open(file_path, 'rb') as f:
        content = f.read()
    
    load_time = time.time() - start_time
    print(f"File loaded into RAM in {load_time:.2f} seconds")
    print(f"Memory used: ~{len(content) / (1024*1024*1024):.3f} GB")
    
    count_start = time.time()
    line_count = content.count(b'\n')
    
    if content and not content.endswith(b'\n'):
        line_count += 1
    
    count_time = time.time() - count_start
    
    print(f"Lines counted in {count_time:.4f} seconds")
    print(f"Total lines: {line_count:,}")
    
    stats_message = f"""I have a large file with the following statistics:
- File path: {file_path}
- File size: {file_size_gb:.3f} GB ({file_size:,} bytes)
- Time to load into RAM: {load_time:.2f} seconds
- Time to count lines: {count_time:.4f} seconds
- Total number of lines: {line_count:,}

Please analyze these results and provide:
1. Assessment of the performance (loading speed and counting speed)
2. Memory efficiency considerations
3. Alternative approaches for handling very large files (like 10 GB)
4. Best practices for working with large files in Python
5. Potential optimizations for production environments"""
    
    print("\n" + "="*60)
    print("Claude's Analysis (streaming):")
    print("="*60)
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": stats_message
            }
        ]
    ) as stream:
        for text in stream.text_stream:
            print(text, end="", flush=True)
    
    print("\n" + "="*60)
    
    return line_count

def main():
    """Main function to demonstrate large file processing."""
    
    print("Large File Line Counter with Claude AI Analysis")
    print("="*60)
    
    with tempfile.NamedTemporaryFile(mode='w', suffix='.txt', delete=False) as tmp_file:
        temp_path = tmp_file.name
    
    try:
        actual_lines = generate_large_file(temp_path, target_size_gb=0.1)
        
        counted_lines = count_lines_with_claude_streaming(temp_path)
        
        print(f"\nVerification:")
        print(f"  Expected lines: {actual_lines:,}")
        print(f"  Counted lines:  {counted_lines:,}")
        print(f"  Match: {'✓ YES' if actual_lines == counted_lines else '✗ NO'}")
        
        print("\n" + "="*60)
        print("Note: For a real 10 GB file, the same approach would work,")
        print("but would require ~10 GB of available RAM.")
        print("The script demonstrates the concept with a smaller file.")
        
    finally:
        if os.path.exists(temp_path):
            os.unlink(temp_path)
            print(f"\nTemporary file cleaned up: {temp_path}")

if __name__ == "__main__":
    main()
