using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

class Program
{
    static async Task Main(string[] args)
    {
        var client = new AnthropicClient();
        var conversationHistory = new List<Message>();
        
        Console.WriteLine("=== Remember Me / Persistent Auth Cookie Implementation Assistant ===");
        Console.WriteLine("I'll help you implement the 'Stay Logged In' feature with secure persistent cookies.");
        Console.WriteLine("Type 'exit' to quit.\n");

        // Initial system context
        var systemPrompt = @"You are an expert C# developer specializing in ASP.NET Core authentication and security. 
        Your task is to help implement a 'Remember Me' feature using persistent authentication cookies.
        
        When providing code, ensure:
        1. Proper cookie security flags (HttpOnly, Secure, SameSite)
        2. Appropriate cookie lifetime (typically 30 days for 'Remember Me')
        3. Secure token generation and validation
        4. Protection against CSRF attacks
        5. Proper cleanup on logout
        
        Provide complete, production-ready C# code with proper error handling.";

        // Start the conversation with an initial request
        string initialRequest = "Please implement a complete 'Remember Me' / 'Stay Logged In' feature for ASP.NET Core with persistent auth cookies. Include: 1) Cookie configuration with security flags, 2) Login endpoint with Remember Me option, 3) Middleware for cookie validation, 4) Logout with cookie cleanup, and 5) Token refresh mechanism.";
        
        Console.WriteLine($"User: {initialRequest}\n");
        
        conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = initialRequest
        });

        bool continueConversation = true;
        
        while (continueConversation)
        {
            try
            {
                var response = await client.Messages.GetClaudeMessageAsync(new MessageParameters
                {
                    Model = AnthropicModels.Claude3_5Sonnet,
                    MaxTokens = 4096,
                    System = new List<SystemMessage>
                    {
                        new SystemMessage { Text = systemPrompt }
                    },
                    Messages = conversationHistory
                });

                var assistantMessage = response.Content[0].Text;
                
                conversationHistory.Add(new Message
                {
                    Role = RoleType.Assistant,
                    Content = assistantMessage
                });

                Console.WriteLine($"Assistant: {assistantMessage}\n");
                Console.WriteLine("---");

                // Ask for follow-up questions
                Console.Write("User (or 'exit' to quit): ");
                string userInput = Console.ReadLine();
                
                if (string.IsNullOrEmpty(userInput) || userInput.ToLower() == "exit")
                {
                    continueConversation = false;
                    Console.WriteLine("\nThank you for using the Remember Me implementation assistant!");
                }
                else
                {
                    conversationHistory.Add(new Message
                    {
                        Role = RoleType.User,
                        Content = userInput
                    });
                    Console.WriteLine();
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error communicating with Claude: {ex.Message}");
                continueConversation = false;
            }
        }

        // Generate final implementation summary
        Console.WriteLine("\n=== Generating Final Implementation Summary ===\n");
        
        conversationHistory.Add(new Message
        {
            Role = RoleType.User,
            Content = "Please provide a final complete implementation summary with all the code needed for the Remember Me feature, organized as a complete ASP.NET Core solution with proper file structure."
        });

        try
        {
            var finalResponse = await client.Messages.GetClaudeMessageAsync(new MessageParameters
            {
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 8192,
                System = new List<SystemMessage>
                {
                    new SystemMessage { Text = systemPrompt }
                },
                Messages = conversationHistory
            });

            Console.WriteLine("Final Implementation:\n");
            Console.WriteLine(finalResponse.Content[0].Text);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error generating final summary: {ex.Message}");
        }
    }
}
