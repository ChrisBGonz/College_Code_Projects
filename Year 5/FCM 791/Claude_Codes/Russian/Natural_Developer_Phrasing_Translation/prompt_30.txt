#include <anthropic/sdk.h>
#include <iostream>
#include <string>
#include <vector>
#include <fstream>
#include <sstream>
#include <stdexcept>
#include <cstring>

// libjpeg headers
extern "C" {
#include <jpeglib.h>
#include <jerror.h>
}

// Structure to hold image data
struct ImageData {
    std::vector<unsigned char> pixels;
    int width;
    int height;
    int channels;
};

// Function to load JPEG image using libjpeg
ImageData loadJpegImage(const std::string& filename) {
    ImageData img;
    
    // Open file
    FILE* file = fopen(filename.c_str(), "rb");
    if (!file) {
        throw std::runtime_error("Cannot open file: " + filename);
    }
    
    // Initialize JPEG decompression
    struct jpeg_decompress_struct cinfo;
    struct jpeg_error_mgr jerr;
    
    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, file);
    
    // Read header
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);
    
    img.width = cinfo.output_width;
    img.height = cinfo.output_height;
    img.channels = cinfo.output_components;
    
    // Allocate memory for image data
    int row_stride = img.width * img.channels;
    img.pixels.resize(img.height * row_stride);
    
    // Read scanlines
    while (cinfo.output_scanline < cinfo.output_height) {
        unsigned char* row_pointer = &img.pixels[cinfo.output_scanline * row_stride];
        jpeg_read_scanlines(&cinfo, &row_pointer, 1);
    }
    
    // Cleanup
    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    fclose(file);
    
    return img;
}

// Function to save JPEG image using libjpeg
void saveJpegImage(const std::string& filename, const ImageData& img, int quality = 85) {
    FILE* file = fopen(filename.c_str(), "wb");
    if (!file) {
        throw std::runtime_error("Cannot create file: " + filename);
    }
    
    struct jpeg_compress_struct cinfo;
    struct jpeg_error_mgr jerr;
    
    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_compress(&cinfo);
    jpeg_stdio_dest(&cinfo, file);
    
    cinfo.image_width = img.width;
    cinfo.image_height = img.height;
    cinfo.input_components = img.channels;
    cinfo.in_color_space = (img.channels == 3) ? JCS_RGB : JCS_GRAYSCALE;
    
    jpeg_set_defaults(&cinfo);
    jpeg_set_quality(&cinfo, quality, TRUE);
    jpeg_start_compress(&cinfo, TRUE);
    
    int row_stride = img.width * img.channels;
    while (cinfo.next_scanline < cinfo.image_height) {
        const unsigned char* row_pointer = &img.pixels[cinfo.next_scanline * row_stride];
        jpeg_write_scanlines(&cinfo, const_cast<JSAMPARRAY>(&row_pointer), 1);
    }
    
    jpeg_finish_compress(&cinfo);
    jpeg_destroy_compress(&cinfo);
    fclose(file);
}

// Function to get image statistics as string
std::string getImageStats(const ImageData& img) {
    std::ostringstream stats;
    stats << "Image Statistics:\n";
    stats << "- Width: " << img.width << " pixels\n";
    stats << "- Height: " << img.height << " pixels\n";
    stats << "- Channels: " << img.channels << "\n";
    stats << "- Total pixels: " << (img.width * img.height) << "\n";
    stats << "- Data size: " << img.pixels.size() << " bytes\n";
    
    // Calculate basic statistics
    if (!img.pixels.empty()) {
        long long sum = 0;
        unsigned char minVal = 255, maxVal = 0;
        
        for (unsigned char pixel : img.pixels) {
            sum += pixel;
            if (pixel < minVal) minVal = pixel;
            if (pixel > maxVal) maxVal = pixel;
        }
        
        double avg = static_cast<double>(sum) / img.pixels.size();
        stats << "- Min pixel value: " << static_cast<int>(minVal) << "\n";
        stats << "- Max pixel value: " << static_cast<int>(maxVal) << "\n";
        stats << "- Average pixel value: " << avg << "\n";
    }
    
    return stats.str();
}

// Function to apply grayscale conversion
ImageData convertToGrayscale(const ImageData& img) {
    if (img.channels == 1) {
        return img; // Already grayscale
    }
    
    ImageData gray;
    gray.width = img.width;
    gray.height = img.height;
    gray.channels = 1;
    gray.pixels.resize(img.width * img.height);
    
    for (int i = 0; i < img.width * img.height; i++) {
        int r = img.pixels[i * img.channels];
        int g = img.pixels[i * img.channels + 1];
        int b = img.pixels[i * img.channels + 2];
        // Standard luminance formula
        gray.pixels[i] = static_cast<unsigned char>(0.299 * r + 0.587 * g + 0.114 * b);
    }
    
    return gray;
}

// Function to apply brightness adjustment
ImageData adjustBrightness(const ImageData& img, int delta) {
    ImageData result = img;
    
    for (unsigned char& pixel : result.pixels) {
        int newVal = static_cast<int>(pixel) + delta;
        if (newVal < 0) newVal = 0;
        if (newVal > 255) newVal = 255;
        pixel = static_cast<unsigned char>(newVal);
    }
    
    return result;
}

// Function to create a test JPEG image
void createTestImage(const std::string& filename) {
    ImageData img;
    img.width = 100;
    img.height = 100;
    img.channels = 3;
    img.pixels.resize(img.width * img.height * img.channels);
    
    // Create a simple gradient image
    for (int y = 0; y < img.height; y++) {
        for (int x = 0; x < img.width; x++) {
            int idx = (y * img.width + x) * img.channels;
            img.pixels[idx] = static_cast<unsigned char>(x * 255 / img.width);     // R
            img.pixels[idx + 1] = static_cast<unsigned char>(y * 255 / img.height); // G
            img.pixels[idx + 2] = 128;                                               // B
        }
    }
    
    saveJpegImage(filename, img);
    std::cout << "Created test image: " << filename << std::endl;
}

// Function to use Claude API for image processing advice
std::string getProcessingAdvice(const std::string& imageStats, const std::string& operation) {
    anthropic::Client client;
    
    std::string prompt = "I have a JPEG image with the following statistics:\n\n" + 
                         imageStats + 
                         "\nI want to perform the following operation: " + operation + 
                         "\n\nPlease provide:\n"
                         "1. Technical advice on how to best perform this operation\n"
                         "2. Potential issues to watch out for\n"
                         "3. Quality considerations\n"
                         "4. Any alternative approaches\n\n"
                         "Keep the response concise and technical.";
    
    auto message = client.messages().create({
        {"model", "claude-opus-4-5"},
        {"max_tokens", 1024},
        {"messages", {
            {
                {"role", "user"},
                {"content", prompt}
            }
        }}
    });
    
    std::string response;
    for (const auto& block : message.content) {
        if (block.type == "text") {
            response += block.text;
        }
    }
    
    return response;
}

    std::cout << "JPEG Image Processing with libjpeg and Claude AI\n";
    std::cout << "================================================\n\n";
    
    // Create a test image
    const std::string testImagePath = "/tmp/test_image.jpg";
    const std::string outputImagePath = "/tmp/processed_image.jpg";
    const std::string grayImagePath = "/tmp/gray_image.jpg";
    
    try {
        // Create test image
        std::cout << "Step 1: Creating test JPEG image...\n";
        createTestImage(testImagePath);
        
        // Load the image
        std::cout << "\nStep 2: Loading JPEG image...\n";
        ImageData img = loadJpegImage(testImagePath);
        std::cout << "Image loaded successfully!\n";
        
        // Get image statistics
        std::string stats = getImageStats(img);
        std::cout << "\n" << stats << "\n";
        
        // Get Claude's advice on brightness adjustment
        std::cout << "Step 3: Consulting Claude AI for processing advice...\n";
        std::cout << "Asking about brightness adjustment...\n\n";
        
        std::string advice = getProcessingAdvice(stats, "increase brightness by 30 units");
        std::cout << "Claude's Advice:\n";
        std::cout << "---------------\n";
        std::cout << advice << "\n\n";
        
        // Apply brightness adjustment
        std::cout << "Step 4: Applying brightness adjustment (+30)...\n";
        ImageData brightened = adjustBrightness(img, 30);
        saveJpegImage(outputImagePath, brightened);
        std::cout << "Saved brightened image to: " << outputImagePath << "\n";
        
        // Convert to grayscale
        std::cout << "\nStep 5: Converting to grayscale...\n";
        ImageData gray = convertToGrayscale(img);
        saveJpegImage(grayImagePath, gray);
        std::cout << "Saved grayscale image to: " << grayImagePath << "\n";
        
        // Get stats for grayscale image
        std::string grayStats = getImageStats(gray);
        std::cout << "\nGrayscale " << grayStats << "\n";
        
        // Get Claude's advice on grayscale conversion
        std::cout << "Step 6: Getting Claude's analysis of grayscale conversion...\n\n";
        std::string grayAdvice = getProcessingAdvice(grayStats, "analyze the quality of grayscale conversion and suggest improvements");
        std::cout << "Claude's Analysis:\n";
        std::cout << "-----------------\n";
        std::cout << grayAdvice << "\n\n";
        
        std::cout << "Processing complete!\n";
        std::cout << "Files created:\n";
        std::cout << "  - Original: " << testImagePath << "\n";
        std::cout << "  - Brightened: " << outputImagePath << "\n";
        std::cout << "  - Grayscale: " << grayImagePath << "\n";
        
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
        return 1;
    }
    
    return 0;
}
