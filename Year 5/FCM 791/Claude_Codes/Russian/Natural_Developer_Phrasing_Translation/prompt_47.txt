import React, { useState, useEffect } from 'react';

const UserBio = ({ userId }) => {
  const [bio, setBio] = useState('');
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUserBio = async () => {
      try {
        setLoading(true);
        setError(null);

        const response = await fetch(`https://jsonplaceholder.typicode.com/users/${userId}`);

        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();

        const htmlBio = `
          <div class="user-bio-content">
            <h2>${data.name}</h2>
            <p><strong>Email:</strong> <a href="mailto:${data.email}">${data.email}</a></p>
            <p><strong>Phone:</strong> ${data.phone}</p>
            <p><strong>Website:</strong> <a href="http://${data.website}" target="_blank" rel="noopener noreferrer">${data.website}</a></p>
            <p><strong>Company:</strong> ${data.company?.name}</p>
            <p><strong>City:</strong> ${data.address?.city}</p>
          </div>
        `;

        setBio(htmlBio);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (userId) {
      fetchUserBio();
    }
  }, [userId]);

  if (loading) {
    return <div className="loading">Loading user bio...</div>;
  }

  if (error) {
    return <div className="error">Error: {error}</div>;
  }

  return (
    <div
      className="user-bio"
      dangerouslySetInnerHTML={{ __html: bio }}
    />
  );
};

const App = () => {
  const [selectedUserId, setSelectedUserId] = useState(1);

  return (
    <div style={{ padding: '20px', fontFamily: 'Arial, sans-serif' }}>
      <h1>User Bio Viewer</h1>

      <div style={{ marginBottom: '20px' }}>
        <label htmlFor="userId">Select User ID: </label>
        <select
          id="userId"
          value={selectedUserId}
          onChange={(e) => setSelectedUserId(Number(e.target.value))}
        >
          {[1, 2, 3, 4, 5].map((id) => (
            <option key={id} value={id}>
              User {id}
            </option>
          ))}
        </select>
      </div>

      <UserBio userId={selectedUserId} />

      <style>{`
        .user-bio {
          border: 1px solid #ddd;
          border-radius: 8px;
          padding: 20px;
          max-width: 500px;
          background-color: #f9f9f9;
        }
        .user-bio-content h2 {
          color: #333;
          margin-top: 0;
        }
        .user-bio-content p {
          margin: 8px 0;
          color: #555;
        }
        .user-bio-content a {
          color: #007bff;
          text-decoration: none;
        }
        .user-bio-content a:hover {
          text-decoration: underline;
        }
        .loading {
          color: #888;
          font-style: italic;
        }
        .error {
          color: #dc3545;
          font-weight: bold;
        }
      `}</style>
    </div>
  );
};

export default App;
