const loadScript = (url, options = {}) => {
  return new Promise((resolve, reject) => {
    const {
      timeout = 10000,
      attributes = {},
      checkLoaded = null,
      id = null,
    } = options;

    if (id && document.getElementById(id)) {
      const existing = document.getElementById(id);
      if (existing.dataset.loaded === 'true') {
        resolve({ cached: true, element: existing });
        return;
      }

      existing.addEventListener('load', () => resolve({ cached: true, element: existing }));
      existing.addEventListener('error', () => reject(new Error(`Script already loading but failed: ${url}`)));
      return;
    }

    if (checkLoaded && checkLoaded()) {
      resolve({ cached: true, alreadyAvailable: true });
      return;
    }

    const script = document.createElement('script');
    script.src = url;
    script.async = true;
    script.type = 'text/javascript';

    if (id) {
      script.id = id;
    }

    Object.entries(attributes).forEach(([key, value]) => {
      script.setAttribute(key, value);
    });

    let timeoutId = null;

    const cleanup = () => {
      if (timeoutId) {
        clearTimeout(timeoutId);
        timeoutId = null;
      }
    };

    script.onload = () => {
      cleanup();
      script.dataset.loaded = 'true';
      resolve({ cached: false, element: script });
    };

    script.onerror = (event) => {
      cleanup();
      script.remove();
      reject(new Error(`Failed to load script: ${url}`));
    };

    if (timeout > 0) {
      timeoutId = setTimeout(() => {
        script.onload = null;
        script.onerror = null;
        script.remove();
        reject(new Error(`Script load timeout after ${timeout}ms: ${url}`));
      }, timeout);
    }

    const target = document.head || document.documentElement;
    target.appendChild(script);
  });
};

const loadLibrary = async (name, url, options = {}) => {
  const { callback = null, ...scriptOptions } = options;

  console.log(`Loading ${name}...`);

  try {
    const result = await loadScript(url, scriptOptions);

    if (result.cached && result.alreadyAvailable) {
      console.log(`${name} was already available`);
    } else if (result.cached) {
      console.log(`${name} was already loading/loaded`);
    } else {
      console.log(`${name} loaded successfully`);
    }

    if (typeof callback === 'function') {
      callback(null, { name, url, ...result });
    }

    return { name, url, ...result };
  } catch (error) {
    console.error(`Error loading ${name}:`, error.message);

    if (typeof callback === 'function') {
      callback(error, null);
    }

    throw error;
  }
};

const loadMultiple = async (libraries, options = {}) => {
  const { parallel = true, stopOnError = false } = options;

  if (parallel) {
    const promises = libraries.map(({ name, url, options: libOptions = {} }) =>
      loadLibrary(name, url, libOptions).catch((error) => {
        if (stopOnError) throw error;
        return { error, name, url };
      })
    );

    return Promise.all(promises);
  } else {
    const results = [];

    for (const { name, url, options: libOptions = {} } of libraries) {
      try {
        const result = await loadLibrary(name, url, libOptions);
        results.push(result);
      } catch (error) {
        if (stopOnError) throw error;
        results.push({ error, name, url });
      }
    }

    return results;
  }
};

// ─── Usage examples ───────────────────────────────────────────────────────────

// 1. Promise-based
loadScript('https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js', {
  id: 'jquery-cdn',
  timeout: 8000,
  checkLoaded: () => typeof window.jQuery !== 'undefined',
})
  .then((result) => {
    console.log('jQuery loaded:', result);
    console.log('jQuery version:', window.jQuery.fn.jquery);
  })
  .catch((error) => {
    console.error('Failed:', error.message);
  });

// 2. Async/await with callback
(async () => {
  try {
    await loadLibrary(
      'Lodash',
      'https://cdnjs.cloudflare.com/ajax/libs/lodash.js/4.17.21/lodash.min.js',
      {
        id: 'lodash-cdn',
        timeout: 8000,
        checkLoaded: () => typeof window._ !== 'undefined',
        callback: (error, result) => {
          if (error) {
            console.error('Callback error:', error.message);
            return;
          }
          console.log('Callback success:', result.name);
          console.log('Lodash version:', window._.VERSION);
        },
      }
    );
  } catch (error) {
    console.error('Async/await error:', error.message);
  }
})();

// 3. Load multiple libraries in parallel
(async () => {
  try {
    const results = await loadMultiple(
      [
        {
          name: 'jQuery',
          url: 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js',
          options: {
            id: 'jquery-cdn',
            checkLoaded: () => typeof window.jQuery !== 'undefined',
          },
        },
        {
          name: 'Axios',
          url: 'https://cdnjs.cloudflare.com/ajax/libs/axios/1.6.7/axios.min.js',
          options: {
            id: 'axios-cdn',
            checkLoaded: () => typeof window.axios !== 'undefined',
          },
        },
        {
          name: 'DayJS',
          url: 'https://cdnjs.cloudflare.com/ajax/libs/dayjs/1.11.10/dayjs.min.js',
          options: {
            id: 'dayjs-cdn',
            checkLoaded: () => typeof window.dayjs !== 'undefined',
          },
        },
      ],
      { parallel: true, stopOnError: false }
    );

    results.forEach((result) => {
      if (result.error) {
        console.error(`Failed to load ${result.name}:`, result.error.message);
      } else {
        console.log(`Successfully loaded: ${result.name}`);
      }
    });

    if (window.jQuery && window.axios && window.dayjs) {
      console.log('All libraries ready!');
      console.log('jQuery:', window.jQuery.fn.jquery);
      console.log('Axios:', window.axios.VERSION);
      console.log('DayJS:', window.dayjs().format('YYYY-MM-DD'));
    }
  } catch (error) {
    console.error('Critical error:', error.message);
  }
})();

// 4. Load multiple libraries sequentially
(async () => {
  try {
    const results = await loadMultiple(
      [
        {
          name: 'jQuery',
          url: 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js',
          options: { id: 'jquery-cdn-seq' },
        },
        {
          name: 'jQuery UI',
          url: 'https://cdnjs.cloudflare.com/ajax/libs/jqueryui/1.13.2/jquery-ui.min.js',
          options: { id: 'jquery-ui-cdn' },
        },
      ],
      { parallel: false, stopOnError: true }
    );

    console.log('Sequential load complete:', results.map((r) => r.name));
  } catch (error) {
    console.error('Sequential load failed:', error.message);
  }
})();
