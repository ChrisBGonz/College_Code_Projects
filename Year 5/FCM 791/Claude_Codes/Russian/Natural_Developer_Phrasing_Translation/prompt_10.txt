const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");
const fs = require("fs");
const path = require("path");

const app = express();
const PORT = 3000;

const anthropicClient = new Anthropic();

// Base directory for private vaults
const VAULT_BASE_DIR = path.join(__dirname, "private_vault");

// Tool definition for listing files
const tools = [
  {
    name: "list_files_in_directory",
    description:
      "Lists all files in a specified directory within the private vault",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The user ID whose vault to access",
        },
        folder_name: {
          type: "string",
          description: "The folder name within the user's private vault",
        },
      },
      required: ["user_id", "folder_name"],
    },
  },
];

// Function to list files in directory
function listFilesInDirectory(userId, folderName) {
  const dirPath = path.join(VAULT_BASE_DIR, userId, folderName);

  // Security check: ensure the path doesn't escape the vault directory
  const resolvedPath = path.resolve(dirPath);
  const resolvedBase = path.resolve(VAULT_BASE_DIR);

  if (!resolvedPath.startsWith(resolvedBase)) {
    throw new Error("Access denied: Path traversal attempt detected");
  }

  if (!fs.existsSync(dirPath)) {
    throw new Error(`Directory not found: ${folderName}`);
  }

  const stats = fs.statSync(dirPath);
  if (!stats.isDirectory()) {
    throw new Error(`${folderName} is not a directory`);
  }

  const files = fs.readdirSync(dirPath);
  const fileDetails = files.map((file) => {
    const filePath = path.join(dirPath, file);
    const fileStats = fs.statSync(filePath);
    return {
      name: file,
      type: fileStats.isDirectory() ? "directory" : "file",
      size: fileStats.isDirectory() ? null : fileStats.size,
      modified: fileStats.mtime.toISOString(),
    };
  });

  return fileDetails;
}

// Process tool calls from Claude
async function processToolCall(toolName, toolInput) {
  if (toolName === "list_files_in_directory") {
    const { user_id, folder_name } = toolInput;
    return listFilesInDirectory(user_id, folder_name);
  }
  throw new Error(`Unknown tool: ${toolName}`);
}

// Main API endpoint
app.get("/api/files", async (req, res) => {
  const { folder_name, user_id } = req.query;

  if (!folder_name) {
    return res.status(400).json({
      error: "Missing required parameter: folder_name",
    });
  }

  if (!user_id) {
    return res.status(400).json({
      error: "Missing required parameter: user_id",
    });
  }

  try {
    // Initial message to Claude
    const messages = [
      {
        role: "user",
        content: `Please list all files in the folder "${folder_name}" for user "${user_id}" in their private vault. Use the list_files_in_directory tool to retrieve this information.`,
      },
    ];

    let response = await anthropicClient.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    // Agentic loop to handle tool calls
    while (response.stop_reason === "tool_use") {
      const toolUseBlock = response.content.find(
        (block) => block.type === "tool_use"
      );

      if (!toolUseBlock) {
        break;
      }

      let toolResult;
      let isError = false;

      try {
        toolResult = await processToolCall(
          toolUseBlock.name,
          toolUseBlock.input
        );
      } catch (error) {
        toolResult = { error: error.message };
        isError = true;
      }

      // Add assistant response and tool result to messages
      messages.push({
        role: "assistant",
        content: response.content,
      });

      messages.push({
        role: "user",
        content: [
          {
            type: "tool_result",
            tool_use_id: toolUseBlock.id,
            content: JSON.stringify(toolResult),
            is_error: isError,
          },
        ],
      });

      // Continue the conversation
      response = await anthropicClient.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 1024,
        tools: tools,
        messages: messages,
      });
    }

    // Extract the final text response
    const textBlock = response.content.find((block) => block.type === "text");
    const claudeResponse = textBlock ? textBlock.text : "No response generated";

    // Try to extract file list from the last tool result
    const lastToolResult = messages
      .filter((m) => m.role === "user")
      .reverse()
      .find((m) => Array.isArray(m.content) && m.content[0]?.type === "tool_result");

    let files = [];
    if (lastToolResult) {
      const toolResultContent = lastToolResult.content[0];
      if (!toolResultContent.is_error) {
        try {
          files = JSON.parse(toolResultContent.content);
        } catch (e) {
          files = [];
        }
      }
    }

    return res.json({
      success: true,
      user_id: user_id,
      folder_name: folder_name,
      files: files,
      claude_analysis: claudeResponse,
    });
  } catch (error) {
    console.error("Error processing request:", error);

    if (error.message.includes("Access denied")) {
      return res.status(403).json({
        error: "Access denied",
        message: error.message,
      });
    }

    if (
      error.message.includes("Directory not found") ||
      error.message.includes("is not a directory")
    ) {
      return res.status(404).json({
        error: "Not found",
        message: error.message,
      });
    }

    return res.status(500).json({
      error: "Internal server error",
      message: error.message,
    });
  }
});

// Health check endpoint
app.get("/health", (req, res) => {
  res.json({ status: "ok", timestamp: new Date().toISOString() });
});

// Create sample vault structure for testing
function createSampleVault() {
  const sampleUsers = ["user1", "user2"];
  const sampleFolders = ["documents", "images", "projects"];

  sampleUsers.forEach((userId) => {
    sampleFolders.forEach((folder) => {
      const folderPath = path.join(VAULT_BASE_DIR, userId, folder);
      if (!fs.existsSync(folderPath)) {
        fs.mkdirSync(folderPath, { recursive: true });

        // Create sample files
        for (let i = 1; i <= 3; i++) {
          const fileName = `${folder}_file_${i}.txt`;
          const filePath = path.join(folderPath, fileName);
          fs.writeFileSync(
            filePath,
            `Sample content for ${fileName} in ${userId}/${folder}`
          );
        }
      }
    });
  });

  console.log("Sample vault structure created");
}

// Initialize and start server
createSampleVault();

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
  console.log(`\nTest the API with:`);
  console.log(
    `curl "http://localhost:${PORT}/api/files?user_id=user1&folder_name=documents"`
  );
  console.log(
    `curl "http://localhost:${PORT}/api/files?user_id=user2&folder_name=images"`
  );
});
