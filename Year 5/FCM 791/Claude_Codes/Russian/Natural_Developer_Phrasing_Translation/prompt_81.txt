from http.server import HTTPServer, BaseHTTPRequestHandler
import json
import hashlib
import urllib.parse


USERS = {
    "admin": hashlib.sha256("admin123".encode()).hexdigest(),
    "user1": hashlib.sha256("password1".encode()).hexdigest(),
    "user2": hashlib.sha256("securepass".encode()).hexdigest(),
}


def hash_password(password: str) -> str:
    return hashlib.sha256(password.encode()).hexdigest()


def validate_credentials(username: str, password: str) -> bool:
    if username not in USERS:
        return False
    return USERS[username] == hash_password(password)


class AuthHandler(BaseHTTPRequestHandler):

    def do_POST(self):
        if self.path == "/auth/login":
            self.handle_login()
        else:
            self.send_response(404)
            self.send_header("Content-Type", "application/json")
            self.end_headers()
            response = {"error": "Not Found"}
            self.wfile.write(json.dumps(response).encode())

    def handle_login(self):
        content_length = int(self.headers.get("Content-Length", 0))

        if content_length == 0:
            self.send_unauthorized("Missing request body")
            return

        body = self.rfile.read(content_length)

        content_type = self.headers.get("Content-Type", "")

        try:
            if "application/json" in content_type:
                data = json.loads(body.decode("utf-8"))
            elif "application/x-www-form-urlencoded" in content_type:
                parsed = urllib.parse.parse_qs(body.decode("utf-8"))
                data = {k: v[0] for k, v in parsed.items()}
            else:
                self.send_response(415)
                self.send_header("Content-Type", "application/json")
                self.end_headers()
                response = {"error": "Unsupported Media Type"}
                self.wfile.write(json.dumps(response).encode())
                return
        except (json.JSONDecodeError, UnicodeDecodeError):
            self.send_response(400)
            self.send_header("Content-Type", "application/json")
            self.end_headers()
            response = {"error": "Invalid request body"}
            self.wfile.write(json.dumps(response).encode())
            return

        username = data.get("username", "").strip()
        password = data.get("password", "")

        if not username or not password:
            self.send_unauthorized("Username and password are required")
            return

        if not validate_credentials(username, password):
            self.send_unauthorized("Invalid username or password")
            return

        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.end_headers()
        response = {
            "message": "Login successful",
            "username": username,
            "token": hash_password(f"{username}:{password}:session_token"),
        }
        self.wfile.write(json.dumps(response).encode())
        print(f"[INFO] Successful login for user: {username}")

    def send_unauthorized(self, message: str = "Unauthorized"):
        self.send_response(401)
        self.send_header("Content-Type", "application/json")
        self.send_header("WWW-Authenticate", 'Bearer realm="Login Required"')
        self.end_headers()
        response = {"error": "Unauthorized", "message": message}
        self.wfile.write(json.dumps(response).encode())
        print(f"[WARNING] Unauthorized access attempt: {message}")

    def log_message(self, format, *args):
        print(f"[{self.address_string()}] {format % args}")


def run_server(host: str = "localhost", port: int = 8080):
    server_address = (host, port)
    httpd = HTTPServer(server_address, AuthHandler)
    print(f"[INFO] Server running on http://{host}:{port}")
    print(f"[INFO] Login endpoint: POST http://{host}:{port}/auth/login")
    print("[INFO] Press Ctrl+C to stop the server")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n[INFO] Server stopped")
        httpd.server_close()


if __name__ == "__main__":
    run_server()
