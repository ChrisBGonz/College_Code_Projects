package main

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"log"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
)

func generateSecureToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", fmt.Errorf("failed to generate random bytes: %w", err)
	}
	return hex.EncodeToString(bytes), nil
}

type PasswordResetToken struct {
	Token     string
	UserID    string
	ExpiresAt time.Time
	CreatedAt time.Time
}

func generatePasswordResetToken(userID string) (*PasswordResetToken, error) {
	token, err := generateSecureToken(32)
	if err != nil {
		return nil, fmt.Errorf("failed to generate token: %w", err)
	}

	now := time.Now()
	return &PasswordResetToken{
		Token:     token,
		UserID:    userID,
		ExpiresAt: now.Add(60 * time.Minute),
		CreatedAt: now,
	}, nil
}

func (t *PasswordResetToken) IsExpired() bool {
	return time.Now().After(t.ExpiresAt)
}

func (t *PasswordResetToken) TimeUntilExpiry() time.Duration {
	remaining := time.Until(t.ExpiresAt)
	if remaining < 0 {
		return 0
	}
	return remaining
}

func analyzeTokenWithClaude(token *PasswordResetToken) {
	client := anthropic.NewClient()

	tokenInfo := fmt.Sprintf(`
Analyze this password reset token implementation:
- Token: %s (first 8 chars: %s...)
- User ID: %s
- Created At: %s
- Expires At: %s
- TTL: 60 minutes
- Token Length: %d bytes (hex encoded: %d chars)

Please evaluate:
1. Security strength of the token
2. TTL appropriateness
3. Any potential vulnerabilities
4. Best practices compliance
`, token.Token, token.Token[:8], token.UserID,
		token.CreatedAt.Format(time.RFC3339),
		token.ExpiresAt.Format(time.RFC3339),
		32, len(token.Token))

	fmt.Println("\n=== Claude Security Analysis ===")
	fmt.Println("Sending token details to Claude for security analysis...")
	fmt.Println()

	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(tokenInfo)),
		},
	})

	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		log.Printf("Streaming error: %v", err)
	}

	fmt.Println()
}

func demonstrateTokenLifecycle() {
	fmt.Println("=== Password Reset Token Generator ===")
	fmt.Println()

	userID := "user_12345"
	fmt.Printf("Generating password reset token for user: %s\n", userID)

	token, err := generatePasswordResetToken(userID)
	if err != nil {
		log.Fatalf("Failed to generate token: %v", err)
	}

	fmt.Printf("\nToken Generated Successfully!\n")
	fmt.Printf("Token: %s\n", token.Token)
	fmt.Printf("Token Length: %d characters\n", len(token.Token))
	fmt.Printf("Created At: %s\n", token.CreatedAt.Format(time.RFC3339))
	fmt.Printf("Expires At: %s\n", token.ExpiresAt.Format(time.RFC3339))
	fmt.Printf("Time Until Expiry: %s\n", token.TimeUntilExpiry().Round(time.Second))
	fmt.Printf("Is Expired: %v\n", token.IsExpired())

	analyzeTokenWithClaude(token)

	fmt.Println("\n=== Token Validation Demo ===")
	fmt.Printf("Checking token validity...\n")
	if !token.IsExpired() {
		fmt.Printf("✓ Token is valid and will expire in %s\n", token.TimeUntilExpiry().Round(time.Second))
	} else {
		fmt.Println("✗ Token has expired")
	}

	fmt.Println("\n=== Multiple Token Generation ===")
	fmt.Println("Generating 3 tokens to demonstrate uniqueness:")
	for i := 1; i <= 3; i++ {
		t, err := generatePasswordResetToken(fmt.Sprintf("user_%d", i))
		if err != nil {
			log.Printf("Failed to generate token %d: %v", i, err)
			continue
		}
		fmt.Printf("Token %d: %s... (expires: %s)\n",
			i, t.Token[:16], t.ExpiresAt.Format("15:04:05"))
	}
}

func main() {
	demonstrateTokenLifecycle()
}
