import com.fasterxml.jackson.databind.ObjectMapper;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class UserJsonParser {

    static class User {
        public String name;
        public String email;
        public int age;

        @Override
        public String toString() {
            return "User{name='" + name + "', email='" + email + "', age=" + age + "}";
        }
    }

    private final ChatLanguageModel chatModel;
    private final ObjectMapper objectMapper;

    public UserJsonParser() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
        this.objectMapper = new ObjectMapper();
    }

    public User parseUserFromJson(String jsonInput) throws Exception {
        // Use AI to validate and potentially fix the JSON before parsing
        String prompt = String.format(
                "You are a JSON validator and fixer. The user provided this JSON string that should represent a User object with fields: name (string), email (string), age (integer). " +
                "If the JSON is valid, return it as-is. If it has issues, fix them and return valid JSON. " +
                "Return ONLY the JSON object, no explanations or markdown formatting.\n\n" +
                "Input JSON: %s",
                jsonInput
        );

        String validatedJson = chatModel.generate(prompt);
        
        // Clean up the response in case AI added any extra formatting
        validatedJson = validatedJson.trim();
        if (validatedJson.startsWith("```json")) {
            validatedJson = validatedJson.substring(7);
        }
        if (validatedJson.startsWith("```")) {
            validatedJson = validatedJson.substring(3);
        }
        if (validatedJson.endsWith("```")) {
            validatedJson = validatedJson.substring(0, validatedJson.length() - 3);
        }
        validatedJson = validatedJson.trim();

        // Parse the validated JSON into User object
        User user = objectMapper.readValue(validatedJson, User.class);
        
        // Print the user.name field
        System.out.println("user.name = " + user.name);
        
        return user;
    }

    public static void main(String[] args) {
        UserJsonParser parser = new UserJsonParser();
        
        // Test cases
        String[] testJsonInputs = {
            "{\"name\": \"John Doe\", \"email\": \"john@example.com\", \"age\": 30}",
            "{\"name\": \"Jane Smith\", \"email\": \"jane@example.com\", \"age\": 25}",
            // Slightly malformed JSON that AI should fix
            "{'name': 'Bob Johnson', 'email': 'bob@example.com', 'age': 35}",
            // JSON with extra fields
            "{\"name\": \"Alice Brown\", \"email\": \"alice@example.com\", \"age\": 28, \"city\": \"New York\"}"
        };

        for (String jsonInput : testJsonInputs) {
            System.out.println("\n--- Processing JSON input ---");
            System.out.println("Input: " + jsonInput);
            try {
                User user = parser.parseUserFromJson(jsonInput);
                System.out.println("Successfully parsed user: " + user);
            } catch (Exception e) {
                System.err.println("Error parsing JSON: " + e.getMessage());
            }
        }
    }
}
