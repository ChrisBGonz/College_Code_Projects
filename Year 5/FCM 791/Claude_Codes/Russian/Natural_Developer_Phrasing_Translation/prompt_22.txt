```bash
#!/usr/bin/env bash

set -euo pipefail

# ============================================================
# Security Tool Auto-Installer
# ============================================================

# ---------- Configurable defaults ----------------------------
TOOL_URL="${TOOL_URL:-}"
TOOL_NAME="${TOOL_NAME:-}"
INSTALL_DIR="${INSTALL_DIR:-/usr/local/bin}"
TMP_DIR="$(mktemp -d)"
LOG_FILE="${LOG_FILE:-/tmp/sec-tool-install-$(date +%Y%m%d-%H%M%S).log}"
CHECKSUM_URL="${CHECKSUM_URL:-}"
SIGNATURE_URL="${SIGNATURE_URL:-}"
GPG_KEY_URL="${GPG_KEY_URL:-}"
VERIFY_TLS="${VERIFY_TLS:-true}"
FORCE_REINSTALL="${FORCE_REINSTALL:-false}"
# -------------------------------------------------------------

# ---------- Colors -------------------------------------------
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
BOLD='\033[1m'
RESET='\033[0m'
# -------------------------------------------------------------

# ---------- Logging ------------------------------------------
exec > >(tee -a "$LOG_FILE") 2>&1

log()      { echo -e "${BLUE}[$(date '+%H:%M:%S')] [INFO]${RESET}  $*"; }
log_ok()   { echo -e "${GREEN}[$(date '+%H:%M:%S')] [OK]${RESET}    $*"; }
log_warn() { echo -e "${YELLOW}[$(date '+%H:%M:%S')] [WARN]${RESET}  $*"; }
log_err()  { echo -e "${RED}[$(date '+%H:%M:%S')] [ERROR]${RESET} $*" >&2; }
log_step() { echo -e "\n${BOLD}${CYAN}==> $*${RESET}"; }
# -------------------------------------------------------------

# ---------- Cleanup ------------------------------------------
cleanup() {
    local exit_code=$?
    log "Cleaning up temporary directory: $TMP_DIR"
    rm -rf "$TMP_DIR"
    if [[ $exit_code -ne 0 ]]; then
        log_err "Installation failed (exit code: $exit_code). See log: $LOG_FILE"
    else
        log_ok "Log saved to: $LOG_FILE"
    fi
}
trap cleanup EXIT
trap 'log_err "Interrupted by user"; exit 130' INT TERM
# -------------------------------------------------------------

# ---------- Usage --------------------------------------------
usage() {
    cat <<EOF
${BOLD}Security Tool Auto-Installer${RESET}

Usage:
  TOOL_URL=<url> TOOL_NAME=<name> $0 [options]

Environment variables:
  TOOL_URL          URL to download the tool archive/binary  (required)
  TOOL_NAME         Name of the installed binary             (required)
  INSTALL_DIR       Installation directory  [/usr/local/bin]
  CHECKSUM_URL      URL to checksum file   (sha256/sha512)
  SIGNATURE_URL     URL to .sig/.asc file
  GPG_KEY_URL       URL to GPG public key for signature verification
  VERIFY_TLS        Verify TLS certificates [true]
  FORCE_REINSTALL   Overwrite existing installation [false]
  LOG_FILE          Path to log file

Examples:
  # Install nuclei (ProjectDiscovery)
  TOOL_URL=https://github.com/projectdiscovery/nuclei/releases/latest/download/nuclei_linux_amd64.zip \\
  TOOL_NAME=nuclei \\
  $0

  # Install with checksum verification
  TOOL_URL=https://example.com/tool-v1.0-linux-amd64.tar.gz \\
  TOOL_NAME=mytool \\
  CHECKSUM_URL=https://example.com/tool-v1.0-linux-amd64.tar.gz.sha256 \\
  $0
EOF
    exit 0
}

[[ "${1:-}" =~ ^(-h|--help)$ ]] && usage
# -------------------------------------------------------------

# ---------- Privilege helper ---------------------------------
SUDO=""
need_sudo() {
    if [[ ! -w "$INSTALL_DIR" ]]; then
        if command -v sudo &>/dev/null && sudo -n true 2>/dev/null; then
            SUDO="sudo"
        elif command -v sudo &>/dev/null; then
            log_warn "Write access to $INSTALL_DIR requires sudo."
            SUDO="sudo"
        else
            log_err "No write access to $INSTALL_DIR and sudo is unavailable."
            exit 1
        fi
    fi
}
# -------------------------------------------------------------

# ---------- Dependency check ---------------------------------
check_dependencies() {
    log_step "Checking dependencies"

    local required=(curl file)
    local optional=(gpg sha256sum sha512sum md5sum unzip tar jq)
    local missing=()

    for cmd in "${required[@]}"; do
        if command -v "$cmd" &>/dev/null; then
            log_ok "  [required] $cmd — found"
        else
            missing+=("$cmd")
            log_err "  [required] $cmd — MISSING"
        fi
    done

    for cmd in "${optional[@]}"; do
        if command -v "$cmd" &>/dev/null; then
            log_ok "  [optional] $cmd — found"
        else
            log_warn "  [optional] $cmd — not found (some features may be skipped)"
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log_err "Missing required dependencies: ${missing[*]}"
        log_err "Install them and re-run the script."
        exit 1
    fi

    log_ok "Dependency check passed."
}
# -------------------------------------------------------------

# ---------- Resolve latest URL via GitHub redirect -----------
resolve_url() {
    local url="$1"
    local tls_flag=""
    [[ "$VERIFY_TLS" == "false" ]] && tls_flag="-k"

    # Follow redirects and return final URL
    curl -fsSL $tls_flag --max-redirs 10 -o /dev/null -w '%{url_effective}' "$url" 2>/dev/null || echo "$url"
}
# -------------------------------------------------------------

# ---------- Download helper ----------------------------------
download_file() {
    local url="$1"
    local dest="$2"
    local description="${3:-file}"
    local tls_flag=""
    [[ "$VERIFY_TLS" == "false" ]] && tls_flag="-k"

    log "Downloading $description from: $url"
    if ! curl -fsSL $tls_flag \
         --retry 3 \
         --retry-delay 2 \
         --connect-timeout 30 \
         --max-time 300 \
         --progress-bar \
         -o "$dest" \
         "$url"; then
        log_err "Failed to download: $url"
        return 1
    fi
    log_ok "Downloaded: $dest ($(du -sh "$dest" | cut -f1))"
}
# -------------------------------------------------------------

# ---------- Checksum verification ----------------------------
verify_checksum() {
    local file="$1"
    local checksum_file="$2"

    log_step "Verifying checksum"

    local filename
    filename="$(basename "$file")"

    # Detect algorithm from file content or extension
    local algo=""
    local sum_cmd=""

    if grep -qiE '^[a-f0-9]{128}' "$checksum_file" 2>/dev/null; then
        algo="sha512"; sum_cmd="sha512sum"
    elif grep -qiE '^[a-f0-9]{64}' "$checksum_file" 2>/dev/null; then
        algo="sha256"; sum_cmd="sha256sum"
    elif grep -qiE '^[a-f0-9]{32}' "$checksum_file" 2>/dev/null; then
        algo="md5"; sum_cmd="md5sum"
    else
        log_warn "Cannot detect checksum algorithm. Skipping checksum verification."
        return 0
    fi

    if ! command -v "$sum_cmd" &>/dev/null; then
        log_warn "$sum_cmd not available. Skipping checksum verification."
        return 0
    fi

    log "Detected algorithm: $algo"

    # Extract expected checksum — handle various formats
    local expected_sum
    expected_sum=$(grep -iE "(^|\s)${filename}(\s|$)" "$checksum_file" \
        | awk '{print $1}' \
        | head -1)

    # Fallback: single-line checksum file
    if [[ -z "$expected_sum" ]]; then
        expected_sum=$(awk '{print $1}' "$checksum_file" | head -1)
    fi

    if [[ -z "$expected_sum" ]]; then
        log_warn "Could not extract expected checksum. Skipping."
        return 0
    fi

    local actual_sum
    actual_sum=$($sum_cmd "$file" | awk '{print $1}')

    log "Expected: $expected_sum"
    log "Actual:   $actual_sum"

    if [[ "${expected_sum,,}" == "${actual_sum,,}" ]]; then
        log_ok "Checksum verification PASSED ($algo)."
    else
        log_err "Checksum mismatch! File may be corrupted or tampered."
        exit 1
    fi
}
# -------------------------------------------------------------

# ---------- GPG signature verification -----------------------
verify_signature() {
    local file="$1"
    local sig_file="$2"
    local key_url="${3:-}"

    log_step "Verifying GPG signature"

    if ! command -v gpg &>/dev/null; then
        log_warn "gpg not found. Skipping signature verification."
        return 0
    fi

    local gpg_home="$TMP_DIR/.gnupg"
    mkdir -p "$gpg_home"
    chmod 700 "$gpg_home"

    if [[ -n "$key_url" ]]; then
        log "Importing GPG key from: $key_url"
        local key_file="$TMP_DIR/signing.key"
        download_file "$key_url" "$key_file" "GPG key"
        gpg --homedir "$gpg_home" --batch --import "$key_file" 2>&1 \
            | while IFS= read -r line; do log "  gpg: $line"; done
    fi

    if gpg --homedir "$gpg_home" \
           --batch \
           --trust-model always \
           --verify "$sig_file" "$file" 2>&1 \
           | while IFS= read -r line; do log "  gpg: $line"; done; then
        log_ok "GPG signature verification PASSED."
    else
        log_err "GPG signature verification FAILED."
        exit 1
    fi
}
# -------------------------------------------------------------

# ---------- Detect and extract archive -----------------------
extract_archive() {
    local archive="$1"
    local dest_dir="$2"

    log_step "Extracting archive"

    local mime
    mime=$(file --mime-type -b "$archive")
    log "Detected MIME type: $mime"

    case "$mime" in
        application/zip)
            if command -v unzip &>/dev/null; then
                unzip -o -q "$archive" -d "$dest_dir"
            else
                log_err "unzip not found. Cannot extract .zip archive."
                exit 1
            fi
            ;;
        application/gzip|application/x-gzip)
            # Could be .tar.gz or plain .gz
            if file "$archive" | grep -q "tar archive"; then
                tar -xzf "$archive" -C "$dest_dir"
            else
                gunzip -c "$archive" > "$dest_dir/$(basename "${archive%.gz}")"
            fi
            ;;
        application/x-bzip2)
            tar -xjf "$archive" -C "$dest_dir"
            ;;
        application/x-xz)
            tar -xJf "$archive" -C "$dest_dir"
            ;;
        application/x-tar)
            tar -xf "$archive" -C "$dest_dir"
            ;;
        application/x-executable|application/x-elf|application/octet-stream)
            log "File appears to be a binary. No extraction needed."
            cp "$archive" "$dest_dir/"
            ;;
        *)
            log_warn "Unknown MIME type '$mime'. Attempting tar extraction..."
            tar -xf "$archive" -C "$dest_dir" 2>/dev/null \
                || { log_warn "tar failed. Treating as binary."; cp "$archive" "$dest_dir/"; }
            ;;
    esac

    log_ok "Extraction complete. Contents:"
    find "$dest_dir" -maxdepth 3 | while IFS= read -r f; do log "  $f"; done
}
# -------------------------------------------------------------

# ---------- Find binary in extracted content -----------------
find_binary() {
    local search_dir="$1"
    local name="$2"

    # Exact name match first
    local found
    found=$(find "$search_dir" -type f -name "$name" | head -1)
    [[ -n "$found" ]] && { echo "$found"; return 0; }

    # Case-insensitive match
    found=$(find "$search_dir" -type f -iname "$name" | head -1)
    [[ -n "$found" ]] && { echo "$found"; return 0; }

    # Any executable file
    found=$(find "$search_dir" -type f -executable | head -1)
    [[ -n "$found" ]] && { echo "$found"; return 0; }

    # Any file (might need chmod)
    found=$(find "$search_dir" -type f | head -1)
    [[ -n "$found" ]] && { echo "$found"; return 0; }

    return 1
}
# -------------------------------------------------------------

# ---------- Install binary -----------------------------------
install_binary() {
    local binary_path="$1"
    local install_name="$2"
    local dest="$INSTALL_DIR/$install_name"

    log_step "Installing binary"

    need_sudo

    # Create install dir if needed
    if [[ ! -d "$INSTALL_DIR" ]]; then
        log "Creating install directory: $INSTALL_DIR"
        $SUDO mkdir -p "$INSTALL_DIR"
    fi

    # Check existing installation
    if [[ -f "$dest" ]] && [[ "$FORCE_REINSTALL" != "true" ]]; then
        local existing_ver
        existing_ver=$("$dest" --version 2>/dev/null || "$dest" version 2>/dev/null || echo "unknown")
        log_warn "Binary already exists at $dest (version: $existing_ver)."
        log_warn "Set FORCE_REINSTALL=true to overwrite."
        return 0
    fi

    log "Installing: $binary_path -> $dest"
    $SUDO cp "$binary_path" "$dest"
    $SUDO chmod 755 "$dest"
    $SUDO chown root:root "$dest" 2>/dev/null || true

    log_ok "Installed to: $dest"
}
# -------------------------------------------------------------

# ---------- PATH management ----------------------------------
ensure_in_path() {
    local dir="$1"

    if echo "$PATH" | tr ':' '\n' | grep -qx "$dir"; then
        log_ok "$dir is already in PATH."
        return 0
    fi

    log_warn "$dir is not in PATH."

    local shell_rc=""
    case "${SHELL:-}" in
        */bash) shell_rc="$HOME/.bashrc" ;;
        */zsh)  shell_rc="$HOME/.zshrc"  ;;
        */fish) shell_rc="$HOME/.config/fish/config.fish" ;;
        *)      shell_rc="$HOME/.profile" ;;
    esac

    local path_line="export PATH=\"$dir:\$PATH\""
    [[ "${SHELL:-}" == */fish ]] && path_line="set -gx PATH $dir \$PATH"

    if [[ -n "$shell_rc" ]]; then
        if ! grep -qF "$dir" "$shell_rc" 2>/dev/null; then
            echo "" >> "$shell_rc"
            echo "# Added by sec-tool-installer" >> "$shell_rc"
            echo "$path_line" >> "$shell_rc"
            log_ok "Added $dir to PATH in $shell_rc"
            log_warn "Run: source $shell_rc  (or open a new terminal)"
        else
            log_ok "$dir already referenced in $shell_rc"
        fi
    fi

    # Also export for current session
    export PATH="$dir:$PATH"
}
# -------------------------------------------------------------

# ---------- Post-install verification ------------------------
verify_installation() {
    local name="$1"
    local dest="$INSTALL_DIR/$name"

    log_step "Verifying installation"

    if [[ ! -f "$dest" ]]; then
        log_err "Binary not found at expected location: $dest"
        exit 1
    fi

    if [[ ! -x "$dest" ]]; then
        log_err "Binary is not executable: $dest"
        exit 1
    fi

    log_ok "Binary exists and is executable: $dest"

    # Try to get version
    local version="unknown"
    for flag in --version -version version -V --help; do
        version=$("$dest" "$flag" 2>&1 | head -1) && break || true
    done
    log_ok "Version info: $version"

    # Verify it's in PATH
    if command -v "$name" &>/dev/null; then
        log_ok "$name is accessible via PATH ($(command -v "$name"))"
    else
        log_warn "$name not yet in PATH for current session."
    fi
}
# -------------------------------------------------------------

# ---------- Print summary ------------------------------------
print_summary() {
    local name="$1"
    echo ""
    echo -e "${BOLD}${GREEN}╔══════════════════════════════════════════╗${RESET}"
    echo -e "${BOLD}${GREEN}║     Installation completed successfully   ║${RESET}"
    echo -e "${BOLD}${GREEN}╚══════════════════════════════════════════╝${RESET}"
    echo ""
    echo -e "  ${BOLD}Tool:${RESET}        $name"
    echo -e "  ${BOLD}Location:${RESET}    $INSTALL_DIR/$name"
    echo -e "  ${BOLD}Log file:${RESET}    $LOG_FILE"
    echo ""
    echo -e "  ${BOLD}Usage:${RESET}       $name --help"
    echo ""
}
# -------------------------------------------------------------

# ============================================================
# MAIN
# ============================================================
main() {
    echo ""
    echo -e "${BOLD}${CYAN}╔══════════════════════════════════════════╗${RESET}"
    echo -e "${BOLD}${CYAN}║     Security Tool Auto-Installer v1.0    ║${RESET}"
    echo -e "${BOLD}${CYAN}╚══════════════════════════════════════════╝${RESET}"
    echo ""

    # Validate required inputs
    if [[ -z "$TOOL_URL" ]]; then
        log_err "TOOL_URL is not set."
        usage
    fi

    if [[ -z "$TOOL_NAME" ]]; then
        # Try to infer from URL
        TOOL_NAME=$(basename "$TOOL_URL" | sed -E 's/[-_](v?[0-9]+\.[0-9]+.*|linux|darwin|windows|amd64|arm64|x86_64).*//' | sed 's/\..*//')
        log_warn "TOOL_NAME not set. Inferred: '$TOOL_NAME'"
        if [[ -z "$TOOL_NAME" ]]; then
            log_err "Could not infer TOOL_NAME. Please set it explicitly."
            exit 1
        fi
    fi

    log "Tool name:    $TOOL_NAME"
    log "Source URL:   $TOOL_URL"
    log "Install dir:  $INSTALL_DIR"
    log "Temp dir:     $TMP_DIR"
    log "Log file:     $LOG_FILE"
    log "TLS verify:   $VERIFY_TLS"

    # Step 1: Check dependencies
    check_dependencies

    # Step 2: Resolve final URL (follow GitHub latest redirects)
    log_step "Resolving download URL"
    local resolved_url
    resolved_url=$(resolve_url "$TOOL_URL")
    log_ok "Resolved URL: $resolved_url"

    local archive_name
    archive_name=$(basename "$resolved_url" | cut -d'?' -f1)
    local archive_path="$TMP_DIR/$archive_name"

    # Step 3: Download the tool
    log_step "Downloading tool"
    download_file "$resolved_url" "$archive_path" "tool archive"

    # Step 4: Download and verify checksum
    if [[ -n "$CHECKSUM_URL" ]]; then
        local checksum_file="$TMP_DIR/checksums.txt"
        download_file "$CHECKSUM_URL" "$checksum_file" "checksum file"
        verify_checksum "$archive_path" "$checksum_file"
    else
        log_warn "CHECKSUM_URL not provided. Skipping checksum verification."
    fi

    # Step 5: Download and verify GPG signature
    if [[ -n "$SIGNATURE_URL" ]]; then
        local sig_file="$TMP_DIR/tool.sig"
        download_file "$SIGNATURE_URL" "$sig_file" "GPG signature"
        verify_signature "$archive_path" "$sig_file" "${GPG_KEY_URL:-}"
    else
        log_warn "SIGNATURE_URL not provided. Skipping signature verification."
    fi

    # Step 6: Extract archive
    local extract_dir="$TMP_DIR/extracted"
    mkdir -p "$extract_dir"

    local mime
    mime=$(file --mime-type -b "$archive_path")
    if [[ "$mime" == "application/x-executable" ]] || \
       [[ "$mime" == "application/x-elf" ]] || \
       { [[ "$mime" == "application/octet-stream" ]] && file "$archive_path" | grep -qiE "ELF|executable"; }; then
        log_step "Binary detected — skipping extraction"
        cp "$archive_path" "$extract_dir/$TOOL_NAME"
    else
        extract_archive "$archive_path" "$extract_dir"
    fi

    # Step 7: Locate binary
    log_step "Locating binary"
    local binary_path
    if ! binary_path=$(find_binary "$extract_dir" "$TOOL_NAME"); then
        log_err "Could not find binary '$TOOL_NAME' in extracted content."
        log_err "Extracted files:"
        find "$extract_dir" | while IFS= read -r f; do log_err "  $f"; done
        exit 1
    fi
    log_ok "Found binary: $binary_path"

    # Step 8: Install
    install_binary "$binary_path" "$TOOL_NAME"

    # Step 9: Ensure PATH
    ensure_in_path "$INSTALL_DIR"

    # Step 10: Verify
    verify_installation "$TOOL_NAME"

    # Summary
    print_summary "$TOOL_NAME"
}

main "$@"
```
