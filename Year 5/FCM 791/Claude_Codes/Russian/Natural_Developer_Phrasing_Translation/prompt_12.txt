import Anthropic from "@anthropic-ai/sdk";
import * as fs from "fs";
import * as path from "path";
import * as readline from "readline";

const client = new Anthropic();

interface Message {
  role: "user" | "assistant";
  content: string;
}

const conversationHistory: Message[] = [];

const systemPrompt = `You are an expert Node.js/Express.js developer. Your task is to help generate production-ready Express.js server configurations.

When asked to generate Express.js server configuration, provide complete, working code with:
1. Production environment settings (NODE_ENV=production handling)
2. Security headers using helmet
3. Request logging using morgan
4. Body parsing middleware
5. Trust proxy configuration
6. Centralized error handling middleware
7. 404 handler
8. Proper TypeScript or JavaScript syntax

Always provide complete, runnable code files. When generating configuration files, include all necessary imports and exports.`;

async function chat(userMessage: string): Promise<string> {
  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: systemPrompt,
    messages: conversationHistory,
  });

  const assistantMessage =
    response.content[0].type === "text" ? response.content[0].text : "";

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function extractCodeBlocks(text: string): Array<{ lang: string; code: string; filename?: string }> {
  const codeBlocks: Array<{ lang: string; code: string; filename?: string }> = [];
  const regex = /```(\w+)?\n([\s\S]*?)```/g;
  let match;

  while ((match = regex.exec(text)) !== null) {
    const lang = match[1] || "text";
    const code = match[2].trim();
    
    // Try to extract filename from comments in code
    let filename: string | undefined;
    const filenameMatch = code.match(/^\/\/\s*(.+\.(js|ts|json|env))/m);
    if (filenameMatch) {
      filename = filenameMatch[1].trim();
    }
    
    codeBlocks.push({ lang, code, filename });
  }

  return codeBlocks;
}

async function saveGeneratedFiles(response: string): Promise<void> {
  const codeBlocks = extractCodeBlocks(response);
  
  if (codeBlocks.length === 0) {
    console.log("\nNo code blocks found to save.");
    return;
  }

  const outputDir = "./generated-server";
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  console.log(`\nFound ${codeBlocks.length} code block(s). Saving to ${outputDir}/`);

  for (let i = 0; i < codeBlocks.length; i++) {
    const block = codeBlocks[i];
    let filename = block.filename;
    
    if (!filename) {
      // Generate filename based on language and index
      const ext = block.lang === "typescript" || block.lang === "ts" ? "ts" : 
                  block.lang === "javascript" || block.lang === "js" ? "js" :
                  block.lang === "json" ? "json" : "txt";
      filename = i === 0 ? `server.${ext}` : `config-${i}.${ext}`;
    }

    const filepath = path.join(outputDir, filename);
    fs.writeFileSync(filepath, block.code);
    console.log(`  ✓ Saved: ${filepath}`);
  }
}

async function generateExpressConfig(): Promise<void> {
  console.log("Express.js Production Server Configuration Generator");
  console.log("====================================================");
  console.log("This tool will help you generate a production-ready Express.js server.");
  console.log("Type 'save' to save generated code to files.");
  console.log("Type 'exit' to quit.\n");

  // Initial request to generate the configuration
  const initialRequest = `Generate a complete production-ready Express.js server configuration with the following requirements:

1. **Environment Configuration**:
   - Handle NODE_ENV=production
   - Load environment variables from .env file
   - Validate required environment variables

2. **Security Headers** (using helmet):
   - Content Security Policy
   - HSTS
   - X-Frame-Options
   - X-Content-Type-Options
   - Referrer Policy

3. **Request Logging** (using morgan):
   - Combined format for production
   - Log to file and console
   - Request ID tracking

4. **Body Parsing**:
   - JSON body parser with size limits
   - URL-encoded body parser
   - Raw body for webhooks

5. **Trust Proxy**:
   - Configure for reverse proxy (nginx/load balancer)
   - Rate limiting based on real IP

6. **Error Handling**:
   - Centralized error handler middleware
   - 404 Not Found handler
   - Async error wrapper utility
   - Custom AppError class
   - Different error responses for dev/prod

7. **Additional Production Features**:
   - Compression middleware
   - CORS configuration
   - Request timeout handling
   - Graceful shutdown

Please provide the complete server.js file with all these features implemented.`;

  console.log("Generating production Express.js configuration...\n");
  
  try {
    const response = await chat(initialRequest);
    console.log("Assistant:", response);
    
    // Interactive conversation loop
    const rl = readline.createInterface({
      input: process.stdin,
      output: process.stdout,
    });

    const askQuestion = (prompt: string): Promise<string> => {
      return new Promise((resolve) => {
        rl.question(prompt, (answer) => {
          resolve(answer);
        });
      });
    };

    let lastResponse = response;

    while (true) {
      const userInput = await askQuestion("\nYou: ");
      
      if (userInput.toLowerCase() === "exit") {
        console.log("Goodbye!");
        rl.close();
        break;
      }
      
      if (userInput.toLowerCase() === "save") {
        await saveGeneratedFiles(lastResponse);
        continue;
      }
      
      if (!userInput.trim()) {
        continue;
      }

      try {
        const response = await chat(userInput);
        console.log("\nAssistant:", response);
        lastResponse = response;
        
        // Auto-save if response contains code blocks
        const codeBlocks = extractCodeBlocks(response);
        if (codeBlocks.length > 0) {
          console.log("\n(Type 'save' to save the generated code to files)");
        }
      } catch (error) {
        console.error("Error:", error);
      }
    }
  } catch (error) {
    console.error("Failed to generate configuration:", error);
    process.exit(1);
  }
}

// Main execution
generateExpressConfig().catch(console.error);
