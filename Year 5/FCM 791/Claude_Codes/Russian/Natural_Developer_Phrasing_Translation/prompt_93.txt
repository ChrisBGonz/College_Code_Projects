using System;
using System.Data;
using Microsoft.Data.SqlClient;

class BankTransfer
{
    private const string ConnectionString =
        "Server=localhost;Database=BankDb;Trusted_Connection=True;TrustServerCertificate=True;";

    static void Main()
    {
        try
        {
            TransferFunds(accountFromId: 1, accountToId: 2, amount: 100m);
            Console.WriteLine("Transfer completed successfully.");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Transfer failed: {ex.Message}");
        }
    }

    static void TransferFunds(int accountFromId, int accountToId, decimal amount)
    {
        using var connection = new SqlConnection(ConnectionString);
        connection.Open();

        using var transaction = connection.BeginTransaction(IsolationLevel.ReadCommitted);

        try
        {
            // Check balance of account A
            decimal balanceA = GetBalance(connection, transaction, accountFromId);

            if (balanceA < amount)
                throw new InvalidOperationException(
                    $"Insufficient funds on account {accountFromId}. " +
                    $"Available: {balanceA:C}, Required: {amount:C}");

            // Debit account A
            ExecuteNonQuery(
                connection, transaction,
                "UPDATE Accounts SET Balance = Balance - @Amount WHERE AccountId = @AccountId",
                new SqlParameter("@Amount", amount),
                new SqlParameter("@AccountId", accountFromId));

            // Credit account B
            ExecuteNonQuery(
                connection, transaction,
                "UPDATE Accounts SET Balance = Balance + @Amount WHERE AccountId = @AccountId",
                new SqlParameter("@Amount", amount),
                new SqlParameter("@AccountId", accountToId));

            // Log the transaction
            ExecuteNonQuery(
                connection, transaction,
                @"INSERT INTO TransactionLog (FromAccountId, ToAccountId, Amount, TransactionDate)
                  VALUES (@From, @To, @Amount, @Date)",
                new SqlParameter("@From", accountFromId),
                new SqlParameter("@To", accountToId),
                new SqlParameter("@Amount", amount),
                new SqlParameter("@Date", DateTime.UtcNow));

            transaction.Commit();
            Console.WriteLine($"Successfully transferred {amount:C} from account {accountFromId} to account {accountToId}.");
        }
        catch
        {
            try
            {
                transaction.Rollback();
                Console.WriteLine("Transaction rolled back.");
            }
            catch (Exception rollbackEx)
            {
                Console.WriteLine($"Rollback failed: {rollbackEx.Message}");
            }

            throw;
        }
    }

    static decimal GetBalance(SqlConnection connection, SqlTransaction transaction, int accountId)
    {
        using var command = new SqlCommand(
            "SELECT Balance FROM Accounts WHERE AccountId = @AccountId",
            connection, transaction);

        command.Parameters.AddWithValue("@AccountId", accountId);

        object result = command.ExecuteScalar();

        if (result == null || result == DBNull.Value)
            throw new InvalidOperationException($"Account {accountId} not found.");

        return Convert.ToDecimal(result);
    }

    static void ExecuteNonQuery(
        SqlConnection connection,
        SqlTransaction transaction,
        string sql,
        params SqlParameter[] parameters)
    {
        using var command = new SqlCommand(sql, connection, transaction);
        command.Parameters.AddRange(parameters);

        int rowsAffected = command.ExecuteNonQuery();

        if (rowsAffected == 0)
            throw new InvalidOperationException($"No rows affected by command: {sql}");
    }
}

/*
-- SQL schema to set up the database:

CREATE TABLE Accounts (
    AccountId   INT           PRIMARY KEY,
    OwnerName   NVARCHAR(100) NOT NULL,
    Balance     DECIMAL(18,2) NOT NULL CHECK (Balance >= 0)
);

CREATE TABLE TransactionLog (
    LogId           INT           PRIMARY KEY IDENTITY(1,1),
    FromAccountId   INT           NOT NULL,
    ToAccountId     INT           NOT NULL,
    Amount          DECIMAL(18,2) NOT NULL,
    TransactionDate DATETIME2     NOT NULL
);

-- Seed data
