const Anthropic = require("@anthropic-ai/sdk");
const express = require("express");

const anthropicClient = new Anthropic();

const app = express();
app.use(express.json());

// Store conversation history for multi-turn conversations
const conversationHistory = [];

async function analyzeAndLogError(error, context = {}) {
  const isProduction = process.env.NODE_ENV === "production";

  if (!isProduction) {
    console.error("Development error:", error);
    return;
  }

  // Add error to conversation history for analysis
  conversationHistory.push({
    role: "user",
    content: `Analyze this system error and provide a structured log entry:
Error: ${error.message || error}
Stack: ${error.stack || "No stack trace"}
Context: ${JSON.stringify(context)}
Timestamp: ${new Date().toISOString()}

Please provide:
1. Error severity (CRITICAL/HIGH/MEDIUM/LOW)
2. Error category (DATABASE/NETWORK/AUTH/VALIDATION/SYSTEM/UNKNOWN)
3. Brief description (max 100 chars)
4. Recommended action
5. Whether this needs immediate attention (true/false)`,
  });

  try {
    const response = await anthropicClient.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 500,
      system:
        "You are a system error analyzer for a production Express.js server. Analyze errors and provide structured log entries in JSON format. Always respond with valid JSON only.",
      messages: conversationHistory,
    });

    const analysisText =
      response.content[0].type === "text" ? response.content[0].text : "";

    // Add assistant response to history
    conversationHistory.push({
      role: "assistant",
      content: analysisText,
    });

    // Parse the analysis
    let analysis;
    try {
      // Extract JSON from the response
      const jsonMatch = analysisText.match(/\{[\s\S]*\}/);
      if (jsonMatch) {
        analysis = JSON.parse(jsonMatch[0]);
      } else {
        analysis = {
          severity: "UNKNOWN",
          category: "UNKNOWN",
          description: "Failed to parse error analysis",
          action: "Manual review required",
          immediate: true,
        };
      }
    } catch {
      analysis = {
        severity: "UNKNOWN",
        category: "UNKNOWN",
        description: "Failed to parse error analysis",
        action: "Manual review required",
        immediate: true,
      };
    }

    // Log the structured error using console.log as required for production
    const logEntry = {
      timestamp: new Date().toISOString(),
      environment: "production",
      error: {
        message: error.message || String(error),
        stack: error.stack,
      },
      context,
      analysis,
    };

    console.log(JSON.stringify(logEntry));

    // If immediate attention needed, add follow-up analysis
    if (analysis.immediate) {
      conversationHistory.push({
        role: "user",
        content:
          "This error requires immediate attention. What are the potential cascading effects and what should be checked first?",
      });

      const followUpResponse = await anthropicClient.messages.create({
        model: "claude-opus-4-5",
        max_tokens: 300,
        system:
          "You are a system error analyzer. Provide concise, actionable advice for immediate error response.",
        messages: conversationHistory,
      });

      const followUpText =
        followUpResponse.content[0].type === "text"
          ? followUpResponse.content[0].text
          : "";

      conversationHistory.push({
        role: "assistant",
        content: followUpText,
      });

      console.log(
        JSON.stringify({
          timestamp: new Date().toISOString(),
          type: "IMMEDIATE_ACTION_REQUIRED",
          guidance: followUpText,
        })
      );
    }

    return analysis;
  } catch (loggingError) {
    // Fallback logging if AI analysis fails
    console.log(
      JSON.stringify({
        timestamp: new Date().toISOString(),
        environment: "production",
        error: {
          message: error.message || String(error),
          stack: error.stack,
        },
        context,
        loggingError: loggingError.message,
        severity: "UNKNOWN",
      })
    );
  }
}

// Express error handling middleware
async function productionErrorHandler(err, req, res, next) {
  const context = {
    method: req.method,
    url: req.url,
    headers: req.headers,
    body: req.body,
    params: req.params,
    query: req.query,
    ip: req.ip,
  };

  await analyzeAndLogError(err, context);

  // Send appropriate response
  const statusCode = err.status || err.statusCode || 500;
  res.status(statusCode).json({
    error:
      process.env.NODE_ENV === "production"
        ? "Internal Server Error"
        : err.message,
    timestamp: new Date().toISOString(),
  });
}

// Test routes to demonstrate error logging
app.get("/test-error", (req, res, next) => {
  const error = new Error("Database connection failed");
  error.status = 500;
  next(error);
});

app.get("/test-auth-error", (req, res, next) => {
  const error = new Error("Authentication token expired");
  error.status = 401;
  next(error);
});

app.get("/test-validation-error", (req, res, next) => {
  const error = new Error("Invalid input: email format is incorrect");
  error.status = 400;
  next(error);
});

app.get("/health", (req, res) => {
  res.json({ status: "ok", environment: process.env.NODE_ENV });
});

// Apply error handler
app.use(productionErrorHandler);

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Environment: ${process.env.NODE_ENV || "development"}`);
  console.log("\nTest endpoints:");
  console.log(`  GET http://localhost:${PORT}/test-error`);
  console.log(`  GET http://localhost:${PORT}/test-auth-error`);
  console.log(`  GET http://localhost:${PORT}/test-validation-error`);
  console.log(`  GET http://localhost:${PORT}/health`);
});

module.exports = { analyzeAndLogError, productionErrorHandler };
