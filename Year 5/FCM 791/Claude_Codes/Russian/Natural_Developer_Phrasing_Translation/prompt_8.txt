package main

import (
	"fmt"
	"os"
)

func GetConfigPathFromEnv(envVarName string) (string, error) {
	value := os.Getenv(envVarName)
	if value == "" {
		return "", fmt.Errorf("environment variable %q is not set or empty", envVarName)
	}
	return value, nil
}

func main() {
	// Example usage
	envVarName := "CONFIG_PATH"
	
	// Test case 1: Variable not set
	path, err := GetConfigPathFromEnv(envVarName)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config path: %s\n", path)
	}
	
	// Test case 2: Set the variable and try again
	os.Setenv(envVarName, "/etc/myapp/config.yaml")
	path, err = GetConfigPathFromEnv(envVarName)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config path: %s\n", path)
	}
	
	// Test case 3: Empty value
	os.Setenv(envVarName, "")
	path, err = GetConfigPathFromEnv(envVarName)
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("Config path: %s\n", path)
	}
}
