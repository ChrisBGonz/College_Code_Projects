const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");

const client = new Anthropic();

// Simulated user database
const userDatabase = {
  "token-alice-123": {
    id: 1,
    username: "alice",
    email: "alice@example.com",
    privateData: {
      creditCard: "**** **** **** 1234",
      address: "123 Main St, New York, NY 10001",
      phoneNumber: "+1-555-0101",
      dateOfBirth: "1990-05-15",
      ssn: "***-**-1234",
    },
  },
  "token-bob-456": {
    id: 2,
    username: "bob",
    email: "bob@example.com",
    privateData: {
      creditCard: "**** **** **** 5678",
      address: "456 Oak Ave, Los Angeles, CA 90001",
      phoneNumber: "+1-555-0202",
      dateOfBirth: "1985-08-22",
      ssn: "***-**-5678",
    },
  },
};

// Extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Get user data by token
function getUserByToken(token) {
  return userDatabase[token] || null;
}

// Process request with Claude AI
async function processWithClaude(userData, requestType) {
  const conversationHistory = [];

  // Initial message to Claude
  conversationHistory.push({
    role: "user",
    content: `I need to retrieve private user data. The user has been authenticated with a valid Bearer token. Here is their data: ${JSON.stringify(userData)}. Please provide a summary of their private information in a secure and professional manner.`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a secure API assistant that helps retrieve and format private user data. Always maintain user privacy and present data in a professional, secure manner. Never expose sensitive data unnecessarily.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // Follow-up question for additional context
  conversationHistory.push({
    role: "user",
    content: `Based on the user data provided, what security recommendations would you make for this user's account?`,
  });

  const followUpResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a secure API assistant that helps retrieve and format private user data. Always maintain user privacy and present data in a professional, secure manner. Never expose sensitive data unnecessarily.",
    messages: conversationHistory,
  });

  const securityRecommendations = followUpResponse.content[0].text;

  return {
    dataSummary: assistantMessage,
    securityRecommendations: securityRecommendations,
  };
}

// HTTP Server
const server = http.createServer(async (req, res) => {
  // Set CORS headers
  res.setHeader("Content-Type", "application/json");
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  // Handle preflight requests
  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Only handle GET requests to /api/private-data
  if (req.method !== "GET" || req.url !== "/api/private-data") {
    res.writeHead(404);
    res.end(JSON.stringify({ error: "Not Found", message: "Endpoint not found" }));
    return;
  }

  // Extract and validate Bearer token
  const authHeader = req.headers["authorization"];
  const token = extractBearerToken(authHeader);

  if (!token) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: "Bearer token is required in Authorization header",
      })
    );
    return;
  }

  // Get user data
  const userData = getUserByToken(token);

  if (!userData) {
    res.writeHead(401);
    res.end(
      JSON.stringify({
        error: "Unauthorized",
        message: "Invalid or expired Bearer token",
      })
    );
    return;
  }

  try {
    // Process with Claude AI using multi-turn conversation
    console.log(`Processing request for user: ${userData.username}`);
    const claudeAnalysis = await processWithClaude(userData, "private-data");

    // Return response
    res.writeHead(200);
    res.end(
      JSON.stringify({
        success: true,
        user: {
          id: userData.id,
          username: userData.username,
          email: userData.email,
        },
        privateData: userData.privateData,
        aiAnalysis: {
          dataSummary: claudeAnalysis.dataSummary,
          securityRecommendations: claudeAnalysis.securityRecommendations,
        },
        timestamp: new Date().toISOString(),
      })
    );
  } catch (error) {
    console.error("Error processing request:", error);
    res.writeHead(500);
    res.end(
      JSON.stringify({
        error: "Internal Server Error",
        message: "Failed to process request",
      })
    );
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`\nTest the API with:`);
  console.log(
    `curl -H "Authorization: Bearer token-alice-123" http://localhost:${PORT}/api/private-data`
  );
  console.log(
    `curl -H "Authorization: Bearer token-bob-456" http://localhost:${PORT}/api/private-data`
  );
});

module.exports = server;
