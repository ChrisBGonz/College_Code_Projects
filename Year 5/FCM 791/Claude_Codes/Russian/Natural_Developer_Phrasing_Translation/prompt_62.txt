import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as crypto from "crypto";

const client = new Anthropic();

// Simple in-memory session store for demonstration
const sessions: Record<string, { userId: string; username: string }> = {};

// Helper to parse cookies from request
function parseCookies(cookieHeader: string): Record<string, string> {
  const cookies: Record<string, string> = {};
  if (!cookieHeader) return cookies;

  cookieHeader.split(";").forEach((cookie) => {
    const [name, value] = cookie.trim().split("=");
    if (name && value) {
      cookies[name.trim()] = value.trim();
    }
  });

  return cookies;
}

// Create a mock request object with session
function createMockRequest(sessionId?: string) {
  const mockSession = sessionId
    ? sessions[sessionId]
    : { userId: "user123", username: "testuser" };

  if (sessionId && !sessions[sessionId]) {
    // Create a test session
    sessions[sessionId] = { userId: "user123", username: "testuser" };
  }

  return {
    sessionId: sessionId || crypto.randomUUID(),
    session: mockSession,
    cookies: sessionId ? { sessionId } : {},
    headers: {
      cookie: sessionId ? `sessionId=${sessionId}` : "",
    },
  };
}

// Create a mock response object
function createMockResponse() {
  const headers: Record<string, string> = {};
  let statusCode = 200;
  let body = "";

  return {
    statusCode,
    headers,
    body,
    setHeader(name: string, value: string) {
      this.headers[name] = value;
    },
    status(code: number) {
      this.statusCode = code;
      return this;
    },
    json(data: object) {
      this.body = JSON.stringify(data);
      this.headers["Content-Type"] = "application/json";
      return this;
    },
    end(data?: string) {
      if (data) this.body = data;
      return this;
    },
  };
}

// The actual logout function
async function handleLogout(
  req: ReturnType<typeof createMockRequest>,
  res: ReturnType<typeof createMockResponse>
): Promise<void> {
  // Use Claude to generate the logout implementation
  const conversation: Anthropic.MessageParam[] = [];

  // First message: Ask Claude to implement the logout function
  conversation.push({
    role: "user",
    content: `I need to implement a logout function in Node.js. The function should:
1. Destroy the user session from the request object
2. Clear session-related cookies
3. Return a proper response to the client

Here's the current state:
- Session ID: ${req.sessionId}
- Session data: ${JSON.stringify(req.session)}
- Current cookies: ${JSON.stringify(req.cookies)}

Please provide the implementation steps and the actual code to handle this logout.`,
  });

  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a Node.js security expert. Provide clear, secure implementations for authentication-related functions. When asked about logout implementation, provide specific code that handles session destruction and cookie clearing.",
    messages: conversation,
  });

  const firstContent = firstResponse.content[0];
  if (firstContent.type !== "text") {
    throw new Error("Unexpected response type");
  }

  console.log("Claude's initial response:", firstContent.text);

  // Add Claude's response to conversation
  conversation.push({
    role: "assistant",
    content: firstContent.text,
  });

  // Second message: Ask for the specific implementation
  conversation.push({
    role: "user",
    content: `Based on your explanation, now execute the logout for this specific session. 
The session store has these sessions: ${JSON.stringify(Object.keys(sessions))}.
Please confirm what actions should be taken and what the response should look like.`,
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    system:
      "You are a Node.js security expert. Provide clear, secure implementations for authentication-related functions. When asked about logout implementation, provide specific code that handles session destruction and cookie clearing.",
    messages: conversation,
  });

  const secondContent = secondResponse.content[0];
  if (secondContent.type !== "text") {
    throw new Error("Unexpected response type");
  }

  console.log("Claude's implementation guidance:", secondContent.text);

  // Now actually implement the logout based on Claude's guidance
  // 1. Destroy the session
  if (req.sessionId && sessions[req.sessionId]) {
    delete sessions[req.sessionId];
    console.log(`Session ${req.sessionId} destroyed`);
  }

  // 2. Clear session cookies
  // Set cookie with expired date to clear it
  const expiredCookieOptions = [
    "sessionId=",
    "Expires=Thu, 01 Jan 1970 00:00:00 GMT",
    "Path=/",
    "HttpOnly",
    "SameSite=Strict",
  ].join("; ");

  res.setHeader("Set-Cookie", expiredCookieOptions);

  // 3. Clear session data from request
  req.session = null as unknown as { userId: string; username: string };
  req.cookies = {};

  // 4. Return success response
  res.status(200).json({
    success: true,
    message: "Logged out successfully",
    timestamp: new Date().toISOString(),
  });

  console.log("Logout completed successfully");
  console.log("Response headers:", res.headers);
  console.log("Response body:", res.body);
}

// Main function to demonstrate the logout
async function main() {
  console.log("=== Logout Handler Demo ===\n");

  // Create a test session
  const testSessionId = crypto.randomUUID();
  sessions[testSessionId] = { userId: "user123", username: "testuser" };

  console.log("Initial sessions:", sessions);
  console.log("Test session ID:", testSessionId);
  console.log("\n");

  // Create mock request and response
  const req = createMockRequest(testSessionId);
  const res = createMockResponse();

  console.log("Request session before logout:", req.session);
  console.log("Request cookies before logout:", req.cookies);
  console.log("\n");

  // Handle logout
  await handleLogout(req, res);

  console.log("\n=== After Logout ===");
  console.log("Sessions remaining:", sessions);
  console.log("Request session after logout:", req.session);
  console.log("Request cookies after logout:", req.cookies);
  console.log("Response status:", res.statusCode);
  console.log("Response headers:", res.headers);
  console.log("Response body:", res.body);
}

main().catch(console.error);
