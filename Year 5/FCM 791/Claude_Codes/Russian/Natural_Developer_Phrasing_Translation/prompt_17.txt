import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function configureMongoDBConnectionString(): Promise<void> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Настрой строку подключения к MongoDB в Node.js-скрипте для локальной dev-базы. 
        
        Предоставь полный рабочий пример с:
        1. Строкой подключения для локальной MongoDB (localhost)
        2. Настройками для dev-окружения
        3. Обработкой ошибок подключения
        4. Использованием переменных окружения через .env файл
        5. Примером базовой операции с базой данных
        
        Используй современный синтаксис JavaScript/TypeScript с async/await.`,
      },
    ],
  });

  console.log("=== Конфигурация MongoDB для локальной dev-базы ===\n");

  if (message.content[0].type === "text") {
    console.log(message.content[0].text);
  }

  console.log("\n=== Информация об использовании API ===");
  console.log(`Входные токены: ${message.usage.input_tokens}`);
  console.log(`Выходные токены: ${message.usage.output_tokens}`);
}

configureMongoDBConnectionString().catch(console.error);
