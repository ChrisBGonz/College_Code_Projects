```javascript
const STORAGE_KEY = 'registration_form_data';

const formConfig = {
  steps: [
    {
      id: 1,
      title: 'Personal Info',
      fields: ['firstName', 'lastName', 'birthDate'],
    },
    {
      id: 2,
      title: 'Contact Details',
      fields: ['email', 'phone'],
    },
    {
      id: 3,
      title: 'Account Setup',
      fields: ['username', 'password', 'confirmPassword'],
    },
    {
      id: 4,
      title: 'Preferences',
      fields: ['newsletter', 'language', 'timezone'],
    },
  ],
};

function createMultiStepForm(config = formConfig) {
  let currentStep = 1;
  let formData = {};
  let isSubmitting = false;

  function loadFromStorage() {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        const parsed = JSON.parse(stored);
        formData = parsed.data || {};
        const savedStep = parsed.currentStep;
        if (savedStep && savedStep >= 1 && savedStep <= config.steps.length) {
          currentStep = savedStep;
        }
        console.log('[Form] Restored data from localStorage:', formData);
        return true;
      }
    } catch (error) {
      console.error('[Form] Failed to load from localStorage:', error);
      clearStorage();
    }
    return false;
  }

  function saveToStorage() {
    try {
      const payload = {
        currentStep,
        data: formData,
        updatedAt: new Date().toISOString(),
      };
      localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
      console.log('[Form] Saved to localStorage, step:', currentStep);
    } catch (error) {
      console.error('[Form] Failed to save to localStorage:', error);
    }
  }

  function clearStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      console.log('[Form] localStorage cleared');
    } catch (error) {
      console.error('[Form] Failed to clear localStorage:', error);
    }
  }

  function getCurrentStepConfig() {
    return config.steps.find((s) => s.id === currentStep) || null;
  }

  function updateFieldValue(fieldName, value) {
    formData[fieldName] = value;
    saveToStorage();
  }

  function updateStepData(stepFields) {
    Object.entries(stepFields).forEach(([key, value]) => {
      formData[key] = value;
    });
    saveToStorage();
  }

  function validateStep(stepId, data) {
    const step = config.steps.find((s) => s.id === stepId);
    if (!step) return { valid: false, errors: ['Invalid step'] };

    const errors = [];

    step.fields.forEach((field) => {
      const value = data[field];

      if (field === 'confirmPassword') {
        if (value !== data['password']) {
          errors.push('Passwords do not match');
        }
        return;
      }

      if (field === 'newsletter') {
        return;
      }

      if (value === undefined || value === null || String(value).trim() === '') {
        errors.push(`Field "${field}" is required`);
        return;
      }

      if (field === 'email') {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(value)) {
          errors.push('Invalid email format');
        }
      }

      if (field === 'phone') {
        const phoneRegex = /^\+?[\d\s\-()]{7,15}$/;
        if (!phoneRegex.test(value)) {
          errors.push('Invalid phone format');
        }
      }

      if (field === 'password') {
        if (String(value).length < 8) {
          errors.push('Password must be at least 8 characters');
        }
      }

      if (field === 'username') {
        if (String(value).length < 3) {
          errors.push('Username must be at least 3 characters');
        }
      }
    });

    return {
      valid: errors.length === 0,
      errors,
    };
  }

  function goToNextStep(currentStepData = {}) {
    if (isSubmitting) return { success: false, error: 'Form is submitting' };

    updateStepData(currentStepData);

    const validation = validateStep(currentStep, formData);
    if (!validation.valid) {
      return { success: false, errors: validation.errors };
    }

    if (currentStep < config.steps.length) {
      currentStep += 1;
      saveToStorage();
      return { success: true, currentStep };
    }

    return { success: false, error: 'Already on last step' };
  }

  function goToPrevStep() {
    if (isSubmitting) return { success: false, error: 'Form is submitting' };

    if (currentStep > 1) {
      currentStep -= 1;
      saveToStorage();
      return { success: true, currentStep };
    }

    return { success: false, error: 'Already on first step' };
  }

  function goToStep(stepId) {
    if (isSubmitting) return { success: false, error: 'Form is submitting' };

    const step = config.steps.find((s) => s.id === stepId);
    if (!step) return { success: false, error: 'Invalid step id' };

    currentStep = stepId;
    saveToStorage();
    return { success: true, currentStep };
  }

  async function submitForm(finalStepData = {}, submitHandler) {
    if (isSubmitting) return { success: false, error: 'Already submitting' };

    updateStepData(finalStepData);

    for (const step of config.steps) {
      const validation = validateStep(step.id, formData);
      if (!validation.valid) {
        currentStep = step.id;
        saveToStorage();
        return {
          success: false,
          errors: validation.errors,
          failedStep: step.id,
        };
      }
    }

    isSubmitting = true;

    try {
      let result;

      if (typeof submitHandler === 'function') {
        result = await submitHandler(getFormData());
      } else {
        result = await defaultSubmitHandler(getFormData());
      }

      clearStorage();
      formData = {};
      currentStep = 1;
      isSubmitting = false;

      console.log('[Form] Submission successful, storage cleared');
      return { success: true, result };
    } catch (error) {
      isSubmitting = false;
      console.error('[Form] Submission failed:', error);

      saveToStorage();

      return {
        success: false,
        error: error.message || 'Submission failed',
        dataPreserved: true,
      };
    }
  }

  async function defaultSubmitHandler(data) {
    console.log('[Form] Submitting data:', data);

    await new Promise((resolve) => setTimeout(resolve, 1000));

    return { id: Date.now(), status: 'created' };
  }

  function getFormData() {
    return { ...formData };
  }

  function getFieldValue(fieldName) {
    return formData[fieldName];
  }

  function getProgress() {
    return {
      currentStep,
      totalSteps: config.steps.length,
      percentage: Math.round((currentStep / config.steps.length) * 100),
      completedSteps: currentStep - 1,
    };
  }

  function getState() {
    return {
      currentStep,
      currentStepConfig: getCurrentStepConfig(),
      formData: getFormData(),
      progress: getProgress(),
      isSubmitting,
      isFirstStep: currentStep === 1,
      isLastStep: currentStep === config.steps.length,
    };
  }

  function reset(clearStorageOnReset = true) {
    formData = {};
    currentStep = 1;
    isSubmitting = false;

    if (clearStorageOnReset) {
      clearStorage();
    }

    console.log('[Form] Form reset');
  }

  function bindToDOM(formElement) {
    if (!formElement) return;

    const inputs = formElement.querySelectorAll('input, select, textarea');

    inputs.forEach((input) => {
      const fieldName = input.name || input.id;
      if (!fieldName) return;

      const savedValue = getFieldValue(fieldName);
      if (savedValue !== undefined) {
        if (input.type === 'checkbox') {
          input.checked = Boolean(savedValue);
        } else {
          input.value = savedValue;
        }
      }

      input.addEventListener('change', (e) => {
        const value =
          e.target.type === 'checkbox' ? e.target.checked : e.target.value;
        updateFieldValue(fieldName, value);
      });

      input.addEventListener('input', (e) => {
        if (e.target.type !== 'checkbox') {
          updateFieldValue(fieldName, e.target.value);
        }
      });
    });

    console.log('[Form] Bound to DOM element, restored values from storage');
  }

  function collectFromDOM(formElement) {
    if (!formElement) return {};

    const stepData = {};
    const step = getCurrentStepConfig();
    if (!step) return {};

    step.fields.forEach((field) => {
      const input = formElement.querySelector(
        `[name="${field}"], #${field}`
      );
      if (input) {
        stepData[field] =
          input.type === 'checkbox' ? input.checked : input.value;
      }
    });

    return stepData;
  }

  const hasRestoredData = loadFromStorage();

  return {
    goToNextStep,
    goToPrevStep,
    goToStep,
    submitForm,
    updateFieldValue,
    updateStepData,
    getFormData,
    getFieldValue,
    getState,
    getProgress,
    getCurrentStepConfig,
    validateStep: (stepId) => validateStep(stepId, formData),
    reset,
    bindToDOM,
    collectFromDOM,
    saveToStorage,
    clearStorage,
    hasRestoredData,
  };
}

async function exampleUsage() {
  const form = createMultiStepForm();

  console.log('=== Initial State ===');
  console.log(form.getState());

  console.log('\n=== Step 1: Personal Info ===');
  let result = form.goToNextStep({
    firstName: 'John',
    lastName: 'Doe',
    birthDate: '1990-01-15',
  });
  console.log('Next step result:', result);
  console.log('State:', form.getState());

  console.log('\n=== Step 2: Contact Details ===');
  result = form.goToNextStep({
    email: 'john.doe@example.com',
    phone: '+1234567890',
  });
  console.log('Next step result:', result);

  console.log('\n=== Step 3: Account Setup ===');
  result = form.goToNextStep({
    username: 'johndoe',
    password: 'SecurePass123',
    confirmPassword: 'SecurePass123',
  });
  console.log('Next step result:', result);

  console.log('\n=== Step 4: Preferences (Final) ===');
  result = await form.submitForm(
    {
      newsletter: true,
      language: 'en',
      timezone: 'UTC',
    },
    async (data) => {
      console.log('Sending to server:', data);
      await new Promise((r) => setTimeout(r, 500));
      return { userId: 42, message: 'Registration successful' };
    }
  );

  console.log('Submit result:', result);
  console.log('localStorage after submit:', localStorage.getItem(STORAGE_KEY));
}

async function exampleWithValidationError() {
  const form = createMultiStepForm();

  console.log('\n=== Validation Error Example ===');
  const result = form.goToNextStep({
    firstName: '',
    lastName: 'Doe',
    birthDate: '1990-01-15',
  });
  console.log('Validation failed:', result);
}

async function exampleWithDOMBinding() {
  const formElement = document.querySelector('#registration-form');
  if (!formElement) return;

  const form = createMultiStepForm();

  form.bindToDOM(formElement);

  if (form.hasRestoredData) {
    console.log('Restored previous session data');
  }

  const nextBtn = formElement.querySelector('[data-action="next"]');
  const prevBtn = formElement.querySelector('[data-action="prev"]');
  const submitBtn = formElement.querySelector('[type="submit"]');

  if (nextBtn) {
    nextBtn.addEventListener('click', () => {
      const stepData = form.collectFromDOM(formElement);
      const result = form.goToNextStep(stepData);

      if (!result.success) {
        console.error('Validation errors:', result.errors);
        return;
      }

      renderStep(form.getState());
    });
  }

  if (prevBtn) {
    prevBtn.addEventListener('click', () => {
      form.goToPrevStep();
      renderStep(form.getState());
    });
  }

  if (submitBtn) {
    submitBtn.addEventListener('click', async (e) => {
      e.preventDefault();
      const finalData = form.collectFromDOM(formElement);
      const result = await form.submitForm(finalData);

      if (result.success) {
        console.log('Registration complete!', result);
      } else {
        console.error('Submit failed:', result);
      }
    });
  }

  function renderStep(state) {
    console.log('Rendering step:', state.currentStep, state.currentStepConfig);
    form.bindToDOM(formElement);
  }
}

exampleUsage();
exampleWithValidationError();
```
