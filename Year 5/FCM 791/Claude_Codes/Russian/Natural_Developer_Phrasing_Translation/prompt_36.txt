def download_file(url: str, local_path: str) -> dict:
    """Downloads a file from URL to local path."""
    try:
        print(f"Downloading file from {url}...")
        urllib.request.urlretrieve(url, local_path)
        file_size = os.path.getsize(local_path)
        return {
            "success": True,
            "local_path": local_path,
            "file_size": file_size,
            "message": f"File downloaded successfully. Size: {file_size} bytes"
        }
    except Exception as e:
        return {
            "success": False,
            "error": str(e),
            "message": f"Failed to download file: {str(e)}"
        }

def calculate_sha256(file_path: str) -> dict:
    """Calculates SHA-256 hash of a file."""
    try:
        sha256_hash = hashlib.sha256()
        with open(file_path, "rb") as f:
            for byte_block in iter(lambda: f.read(4096), b""):
                sha256_hash.update(byte_block)
        hash_value = sha256_hash.hexdigest()
        return {
            "success": True,
            "hash": hash_value,
            "file_path": file_path,
            "message": f"SHA-256 hash calculated: {hash_value}"
        }
    except Exception as e:
        return {
            "success": False,
            "error": str(e),
            "message": f"Failed to calculate hash: {str(e)}"
        }

def verify_hash(calculated_hash: str, expected_hash: str, file_path: str) -> dict:
    """Verifies if calculated hash matches expected hash."""
    calculated_lower = calculated_hash.lower().strip()
    expected_lower = expected_hash.lower().strip()
    
    is_valid = calculated_lower == expected_lower
    
    return {
        "success": True,
        "is_valid": is_valid,
        "calculated_hash": calculated_hash,
        "expected_hash": expected_hash,
        "file_path": file_path,
        "message": "INTEGRITY CHECK PASSED: File is authentic and unmodified." if is_valid 
                   else "INTEGRITY CHECK FAILED: File may be corrupted or tampered with!"
    }

def cleanup_file(file_path: str) -> dict:
    """Removes a file from the filesystem."""
    try:
        if os.path.exists(file_path):
            os.remove(file_path)
            return {
                "success": True,
                "message": f"File {file_path} removed successfully"
            }
        else:
            return {
                "success": False,
                "message": f"File {file_path} not found"
            }
    except Exception as e:
        return {
            "success": False,
            "error": str(e),
            "message": f"Failed to remove file: {str(e)}"
        }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results."""
    if tool_name == "download_file":
        result = download_file(tool_input["url"], tool_input["local_path"])
    elif tool_name == "calculate_sha256":
        result = calculate_sha256(tool_input["file_path"])
    elif tool_name == "verify_hash":
        result = verify_hash(
            tool_input["calculated_hash"],
            tool_input["expected_hash"],
            tool_input["file_path"]
        )
    elif tool_name == "cleanup_file":
        result = cleanup_file(tool_input["file_path"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def verify_firmware_integrity(firmware_url: str, expected_hash: str, firmware_name: str = "firmware.bin"):
    """
    Main function to verify firmware integrity using Claude AI with tool use.
    
    Args:
        firmware_url: URL to download the firmware from
        expected_hash: Expected SHA-256 hash for verification
        firmware_name: Local filename to save the firmware
    """
    print("\n" + "="*60)
    print("FIRMWARE INTEGRITY VERIFICATION SYSTEM")
    print("="*60)
    print(f"Firmware URL: {firmware_url}")
    print(f"Expected Hash: {expected_hash}")
    print(f"Local filename: {firmware_name}")
    print("="*60 + "\n")
    
    user_message = f"""Please perform a complete firmware integrity check for the following:

1. Download the firmware file from: {firmware_url}
   Save it as: {firmware_name}

2. Calculate the SHA-256 hash of the downloaded file

3. Verify the hash against the expected value: {expected_hash}

4. After verification, clean up the downloaded file

Please provide a detailed report of each step and the final verification result."""

    messages = [{"role": "user", "content": user_message}]
    
    print("Starting firmware integrity verification process...\n")
    
    while True:
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, "text"):
                    print("\nFinal Report from Claude:")
                    print("-" * 40)
                    print(block.text)
            break
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\n[Tool Call] {tool_name}")
                    print(f"Input: {json.dumps(tool_input, indent=2)}")
                    
                    result = process_tool_call(tool_name, tool_input)
                    result_dict = json.loads(result)
                    
                    print(f"Result: {result_dict.get('message', result)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    print("\n" + "="*60)
    print("Verification process completed")
    print("="*60)
