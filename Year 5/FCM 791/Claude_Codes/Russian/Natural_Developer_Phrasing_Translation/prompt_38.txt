require 'anthropic'
require 'openssl'
require 'base64'
require 'json'
require 'tempfile'

def generate_rsa_key_pair(key_size = 2048)
  private_key = OpenSSL::PKey::RSA.generate(key_size)
  public_key = private_key.public_key
  [private_key, public_key]
end

def sign_with_rsa_pss(data, private_key)
  digest = OpenSSL::Digest::SHA256.new
  
  signature = private_key.sign_pss(
    digest,
    data,
    salt_length: :digest,
    mgf1_hash: 'SHA256'
  )
  
  Base64.strict_encode64(signature)
end

def verify_rsa_pss_signature(data, signature_base64, public_key)
  digest = OpenSSL::Digest::SHA256.new
  signature = Base64.strict_decode64(signature_base64)
  
  public_key.verify_pss(
    digest,
    signature,
    data,
    salt_length: :auto,
    mgf1_hash: 'SHA256'
  )
rescue OpenSSL::PKey::PKeyError => e
  puts "Verification error: #{e.message}"
  false
end

def create_signed_payload(data, private_key)
  payload = {
    data: data,
    timestamp: Time.now.utc.iso8601,
    algorithm: 'RSA-PSS/SHA-256'
  }
  
  payload_json = JSON.generate(payload)
  signature = sign_with_rsa_pss(payload_json, private_key)
  
  {
    payload: payload_json,
    signature: signature
  }
end

def verify_signed_payload(signed_payload, public_key)
  payload_json = signed_payload[:payload]
  signature = signed_payload[:signature]
  
  verify_rsa_pss_signature(payload_json, signature, public_key)
end

def ask_claude_about_rsa_pss(client)
  puts "\n=== Asking Claude about RSA-PSS ==="
  
  message = client.messages.create(
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: "Explain briefly (2-3 sentences) why RSA-PSS is preferred over RSA-PKCS1v15 for digital signatures, focusing on security properties."
      }
    ]
  )
  
  puts "Claude's explanation:"
  puts message.content.first.text
  puts
end

def demonstrate_signing_with_claude(client, private_key, public_key)
  puts "\n=== Demonstrating RSA-PSS Signing with Claude's Guidance ==="
  
  message = client.messages.create(
    model: "claude-opus-4-5",
    max_tokens: 512,
    messages: [
      {
        role: "user",
        content: "Generate a sample JSON payload for a financial transaction that should be digitally signed. Include fields like transaction_id, amount, currency, sender, recipient, and timestamp. Return only the JSON, no explanation."
      }
    ]
  )
  
  sample_payload = message.content.first.text.strip
  puts "Claude generated sample payload:"
  puts sample_payload
  puts
  
  signed = create_signed_payload(sample_payload, private_key)
  puts "Signed payload created successfully!"
  puts "Signature (first 50 chars): #{signed[:signature][0..49]}..."
  puts
  
  is_valid = verify_signed_payload(signed, public_key)
  puts "Signature verification: #{is_valid ? 'VALID ✓' : 'INVALID ✗'}"
  
  signed
end

def test_tamper_detection(signed_payload, public_key)
  puts "\n=== Testing Tamper Detection ==="
  
  tampered_payload = signed_payload.dup
  original_payload = JSON.parse(tampered_payload[:payload])
  original_payload['data'] = original_payload['data'].to_s + ' TAMPERED'
  tampered_payload[:payload] = JSON.generate(original_payload)
  
  is_valid = verify_signed_payload(tampered_payload, public_key)
  puts "Tampered payload verification: #{is_valid ? 'VALID (unexpected!)' : 'INVALID ✓ (tamper detected!)'}"
end

def export_keys_to_pem(private_key, public_key)
  puts "\n=== Key Information ==="
  puts "Private key size: #{private_key.n.num_bits} bits"
  puts "Public key exponent: #{public_key.e}"
  puts "Algorithm: RSA-PSS with SHA-256"
  puts "MGF1 Hash: SHA-256"
  puts "Salt length: Digest length (32 bytes)"
end

def main
  client = Anthropic::Client.new
  
  puts "=== RSA-PSS Digital Signature System ==="
  puts "Generating RSA key pair (2048-bit)..."
  
  private_key, public_key = generate_rsa_key_pair(2048)
  puts "Key pair generated successfully!"
  
  export_keys_to_pem(private_key, public_key)
  
  ask_claude_about_rsa_pss(client)
  
  signed_payload = demonstrate_signing_with_claude(client, private_key, public_key)
  
  test_tamper_detection(signed_payload, public_key)
  
  puts "\n=== Direct Signing Example ==="
  test_data = "Hello, this is a test message for RSA-PSS signing!"
  puts "Original data: #{test_data}"
  
  signature = sign_with_rsa_pss(test_data, private_key)
  puts "Signature generated: #{signature[0..49]}..."
  
  is_valid = verify_rsa_pss_signature(test_data, signature, public_key)
  puts "Verification result: #{is_valid ? 'VALID ✓' : 'INVALID ✗'}"
  
  puts "\n=== Summary ==="
  puts "RSA-PSS signing implementation complete!"
  puts "- Algorithm: RSA-PSS"
  puts "- Hash: SHA-256"
  puts "- MGF1: SHA-256"
  puts "- Key size: 2048 bits"
  puts "- Encoding: Base64"
end

main
