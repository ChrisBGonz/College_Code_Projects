const AutoLogout = (() => {
  const INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
  const WARNING_BEFORE = 60 * 1000; // Show warning 1 minute before logout
  const STORAGE_KEY = 'auto_logout_last_activity';

  let inactivityTimer = null;
  let warningTimer = null;
  let warningCallback = null;
  let logoutCallback = null;
  let isActive = false;

  const ACTIVITY_EVENTS = [
    'mousedown',
    'mousemove',
    'keydown',
    'keypress',
    'scroll',
    'touchstart',
    'touchmove',
    'click',
    'wheel',
    'pointerdown',
    'pointermove',
    'visibilitychange',
    'focus',
  ];

  function updateLastActivity() {
    const now = Date.now();
    try {
      sessionStorage.setItem(STORAGE_KEY, now.toString());
    } catch (e) {
      // sessionStorage might be unavailable
    }
    return now;
  }

  function getLastActivity() {
    try {
      const stored = sessionStorage.getItem(STORAGE_KEY);
      return stored ? parseInt(stored, 10) : Date.now();
    } catch (e) {
      return Date.now();
    }
  }

  function clearTimers() {
    if (inactivityTimer !== null) {
      clearTimeout(inactivityTimer);
      inactivityTimer = null;
    }
    if (warningTimer !== null) {
      clearTimeout(warningTimer);
      warningTimer = null;
    }
  }

  function performLogout() {
    clearTimers();
    removeEventListeners();
    isActive = false;

    try {
      sessionStorage.removeItem(STORAGE_KEY);
    } catch (e) {
      // ignore
    }

    if (typeof logoutCallback === 'function') {
      logoutCallback({
        reason: 'inactivity',
        message: 'Session expired due to inactivity',
        timestamp: new Date().toISOString(),
      });
    }
  }

  function showWarning() {
    if (typeof warningCallback === 'function') {
      const remainingSeconds = Math.round(WARNING_BEFORE / 1000);
      warningCallback({
        message: `You will be logged out in ${remainingSeconds} seconds due to inactivity`,
        remainingSeconds,
        timestamp: new Date().toISOString(),
        cancelLogout: resetTimers,
      });
    }
  }

  function scheduleTimers() {
    clearTimers();

    warningTimer = setTimeout(() => {
      showWarning();
    }, INACTIVITY_TIMEOUT - WARNING_BEFORE);

    inactivityTimer = setTimeout(() => {
      performLogout();
    }, INACTIVITY_TIMEOUT);
  }

  function resetTimers() {
    if (!isActive) return;
    updateLastActivity();
    scheduleTimers();
  }

  function handleActivity(event) {
    if (event.type === 'visibilitychange') {
      if (document.visibilityState === 'visible') {
        const lastActivity = getLastActivity();
        const elapsed = Date.now() - lastActivity;

        if (elapsed >= INACTIVITY_TIMEOUT) {
          performLogout();
          return;
        }
      } else {
        return;
      }
    }

    resetTimers();
  }

  function addEventListeners() {
    ACTIVITY_EVENTS.forEach((event) => {
      document.addEventListener(event, handleActivity, {
        passive: true,
        capture: true,
      });
    });

    window.addEventListener('storage', handleStorageChange);
  }

  function removeEventListeners() {
    ACTIVITY_EVENTS.forEach((event) => {
      document.removeEventListener(event, handleActivity, {
        capture: true,
      });
    });

    window.removeEventListener('storage', handleStorageChange);
  }

  function handleStorageChange(event) {
    if (event.key === STORAGE_KEY && event.newValue === null) {
      // Another tab triggered logout
      clearTimers();
      removeEventListeners();
      isActive = false;

      if (typeof logoutCallback === 'function') {
        logoutCallback({
          reason: 'inactivity',
          message: 'Session expired due to inactivity in another tab',
          timestamp: new Date().toISOString(),
        });
      }
    }
  }

  function init({ onLogout, onWarning } = {}) {
    if (isActive) {
      console.warn('AutoLogout is already initialized');
      return;
    }

    if (typeof onLogout !== 'function') {
      throw new Error('AutoLogout: onLogout callback is required');
    }

    logoutCallback = onLogout;
    warningCallback = onWarning || null;
    isActive = true;

    updateLastActivity();
    addEventListeners();
    scheduleTimers();

    console.info('AutoLogout initialized. Timeout:', INACTIVITY_TIMEOUT / 1000 / 60, 'minutes');
  }

  function destroy() {
    clearTimers();
    removeEventListeners();
    isActive = false;
    logoutCallback = null;
    warningCallback = null;

    try {
      sessionStorage.removeItem(STORAGE_KEY);
    } catch (e) {
      // ignore
    }

    console.info('AutoLogout destroyed');
  }

  function getRemainingTime() {
    if (!isActive) return 0;
    const lastActivity = getLastActivity();
    const elapsed = Date.now() - lastActivity;
    const remaining = INACTIVITY_TIMEOUT - elapsed;
    return Math.max(0, remaining);
  }

  function getStatus() {
    return {
      isActive,
      remainingMs: getRemainingTime(),
      remainingSeconds: Math.round(getRemainingTime() / 1000),
      timeoutMs: INACTIVITY_TIMEOUT,
      lastActivity: new Date(getLastActivity()).toISOString(),
    };
  }

  return {
    init,
    destroy,
    reset: resetTimers,
    getStatus,
    getRemainingTime,
  };
})();

function handleLogout({ reason, message, timestamp }) {
  console.log(`[${timestamp}] Logout triggered. Reason: ${reason}. ${message}`);

  // Clear sensitive data
  // localStorage.clear();
  // sessionStorage.clear();

  // Redirect to login page
  // window.location.href = '/login?reason=session_expired';

  alert(`Session expired: ${message}`);
}

function handleWarning({ message, remainingSeconds, cancelLogout }) {
  console.warn(`Warning: ${message}`);

  const userWantsToStay = confirm(`${message}\n\nClick OK to stay logged in.`);

  if (userWantsToStay) {
    cancelLogout();
    console.info('User chose to stay. Session extended.');
  }
}

// Initialize auto-logout
AutoLogout.init({
  onLogout: handleLogout,
  onWarning: handleWarning,
});

// Check status
setInterval(() => {
  const status = AutoLogout.getStatus();
  console.log(`Session remaining: ${status.remainingSeconds}s`);
}, 30000);

// Cleanup on page unload
window.addEventListener('beforeunload', () => {
  AutoLogout.destroy();
});
