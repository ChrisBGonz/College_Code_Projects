
using System;
using System.Security.Cryptography;
using System.Text;

class RsaKeyPairGenerator
{
    static void Main()
    {
        // Генерация пары ключей RSA с размером 2048 бит
        using var rsa = RSA.Create(2048);

        // Экспорт публичного ключа в формате PEM
        string publicKeyPem = ExportPublicKeyPem(rsa);

        // Экспорт приватного ключа в формате PEM
        string privateKeyPem = ExportPrivateKeyPem(rsa);

        Console.WriteLine("=== PUBLIC KEY ===");
        Console.WriteLine(publicKeyPem);

        Console.WriteLine("=== PRIVATE KEY ===");
        Console.WriteLine(privateKeyPem);

        // Демонстрация подписи и верификации
        string message = "Hello, RSA Digital Signature!";
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);

        // Подпись сообщения приватным ключом
        byte[] signature = SignData(rsa, messageBytes);
        Console.WriteLine($"Signature (Base64): {Convert.ToBase64String(signature)}");

        // Верификация подписи публичным ключом
        bool isValid = VerifySignature(rsa, messageBytes, signature);
        Console.WriteLine($"Signature valid: {isValid}");

        // Демонстрация загрузки ключей из PEM и повторной верификации
        Console.WriteLine("\n=== Verifying with loaded keys ===");

        using var rsaForVerify = RSA.Create();
        rsaForVerify.ImportFromPem(publicKeyPem);

        bool isValidWithLoadedKey = VerifySignature(rsaForVerify, messageBytes, signature);
        Console.WriteLine($"Signature valid with loaded public key: {isValidWithLoadedKey}");
    }

    static string ExportPublicKeyPem(RSA rsa)
    {
        byte[] publicKeyBytes = rsa.ExportSubjectPublicKeyInfo();
        return FormatPem("PUBLIC KEY", publicKeyBytes);
    }

    static string ExportPrivateKeyPem(RSA rsa)
    {
        byte[] privateKeyBytes = rsa.ExportPkcs8PrivateKey();
        return FormatPem("PRIVATE KEY", privateKeyBytes);
    }

    static string FormatPem(string keyType, byte[] keyBytes)
    {
        var sb = new StringBuilder();
        sb.AppendLine($"-----BEGIN {keyType}-----");

        string base64 = Convert.ToBase64String(keyBytes);
        // Разбиваем Base64 на строки по 64 символа
        for (int i = 0; i < base64.Length; i += 64)
        {
            int length = Math.Min(64, base64.Length - i);
            sb.AppendLine(base64.Substring(i, length));
        }

        sb.AppendLine($"-----END {keyType}-----");
        return sb.ToString();
    }

    static byte[] SignData(RSA rsa, byte[] data)
    {
        return rsa.SignData(
            data,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );
    }

    static bool VerifySignature(RSA rsa, byte[] data, byte[] signature)
    {
        return rsa.VerifyData(
            data,
            signature,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1
        );
    }
}
