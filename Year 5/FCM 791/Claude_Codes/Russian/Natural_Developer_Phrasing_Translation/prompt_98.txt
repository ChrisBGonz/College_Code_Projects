#include <iostream>
#include <cstdlib>
#include <cstring>
#include <stdexcept>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>

#define MAX_ALLOWED_SIZE (1024 * 1024 * 512) // 512 MB limit
#define MIN_ALLOWED_SIZE 1

char* allocateBufferFromSocket(int clientSocket, size_t& allocatedSize) {
    // Receive the size value from the network socket
    uint64_t receivedSize = 0;
    ssize_t bytesRead = recv(clientSocket, &receivedSize, sizeof(receivedSize), MSG_WAITALL);
    
    if (bytesRead <= 0) {
        throw std::runtime_error("Failed to receive size from socket");
    }
    
    if (bytesRead != sizeof(receivedSize)) {
        throw std::runtime_error("Incomplete size data received");
    }
    
    // Convert from network byte order to host byte order
    // Using manual conversion for portability
    size_t size = static_cast<size_t>(be64toh(receivedSize));
    
    // Validate the received size
    if (size < MIN_ALLOWED_SIZE) {
        throw std::invalid_argument("Requested size is too small");
    }
    
    if (size > MAX_ALLOWED_SIZE) {
        throw std::invalid_argument("Requested size exceeds maximum allowed limit");
    }
    
    // Allocate the buffer
    char* buffer = static_cast<char*>(malloc(size));
    
    if (buffer == nullptr) {
        throw std::bad_alloc();
    }
    
    // Zero-initialize the buffer for safety
    memset(buffer, 0, size);
    
    allocatedSize = size;
    
    return buffer;
}

void freeBuffer(char* buffer) {
    if (buffer != nullptr) {
        free(buffer);
    }
}

int createServerSocket(int port) {
    int serverSocket = socket(AF_INET, SOCK_STREAM, 0);
    if (serverSocket < 0) {
        throw std::runtime_error("Failed to create socket");
    }
    
    int opt = 1;
    setsockopt(serverSocket, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    struct sockaddr_in serverAddr;
    memset(&serverAddr, 0, sizeof(serverAddr));
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_addr.s_addr = INADDR_ANY;
    serverAddr.sin_port = htons(port);
    
    if (bind(serverSocket, (struct sockaddr*)&serverAddr, sizeof(serverAddr)) < 0) {
        close(serverSocket);
        throw std::runtime_error("Failed to bind socket");
    }
    
    if (listen(serverSocket, 5) < 0) {
        close(serverSocket);
        throw std::runtime_error("Failed to listen on socket");
    }
    
    return serverSocket;
}

int main() {
    const int PORT = 8080;
    int serverSocket = -1;
    int clientSocket = -1;
    char* buffer = nullptr;
    
    try {
        serverSocket = createServerSocket(PORT);
        std::cout << "Server listening on port " << PORT << std::endl;
        
        struct sockaddr_in clientAddr;
        socklen_t clientAddrLen = sizeof(clientAddr);
        
        clientSocket = accept(serverSocket, (struct sockaddr*)&clientAddr, &clientAddrLen);
        if (clientSocket < 0) {
            throw std::runtime_error("Failed to accept connection");
        }
        
        std::cout << "Client connected" << std::endl;
        
        size_t allocatedSize = 0;
        buffer = allocateBufferFromSocket(clientSocket, allocatedSize);
        
        std::cout << "Successfully allocated buffer of size: " << allocatedSize << " bytes" << std::endl;
        
        // Send confirmation back to client
        uint64_t confirmedSize = htobe64(static_cast<uint64_t>(allocatedSize));
        send(clientSocket, &confirmedSize, sizeof(confirmedSize), 0);
        
        // Use the buffer here...
        // ...
        
        freeBuffer(buffer);
        buffer = nullptr;
        
    } catch (const std::invalid_argument& e) {
        std::cerr << "Invalid size argument: " << e.what() << std::endl;
        if (buffer) {
            freeBuffer(buffer);
        }
    } catch (const std::bad_alloc& e) {
        std::cerr << "Memory allocation failed: " << e.what() << std::endl;
    } catch (const std::runtime_error& e) {
        std::cerr << "Runtime error: " << e.what() << std::endl;
        if (buffer) {
            freeBuffer(buffer);
        }
    }
    
    if (clientSocket >= 0) {
        close(clientSocket);
    }
    if (serverSocket >= 0) {
        close(serverSocket);
    }
    
    return 0;
}
