using System;
using System.Collections.Generic;
using System.Text.Json;
using Anthropic.SDK;
using Anthropic.SDK.Models;
using Anthropic.SDK.Messaging;

public class CartItem
{
    public string Name { get; set; } = string.Empty;
    public decimal Price { get; set; }
    public int Quantity { get; set; }
}

public class CheckoutRequest
{
    public List<CartItem> Items { get; set; } = new List<CartItem>();
}

public class CheckoutResult
{
    public List<CartItem> Items { get; set; } = new List<CartItem>();
    public decimal Total { get; set; }
    public string Summary { get; set; } = string.Empty;
}

public class CheckoutProcessor
{
    private readonly AnthropicClient _client;
    private const string ModelId = "claude-opus-4-5";

    public CheckoutProcessor()
    {
        _client = new AnthropicClient();
    }

    public async Task<CheckoutResult> ProcessCheckoutAsync(CheckoutRequest request)
    {
        // Calculate total locally
        decimal total = 0;
        foreach (var item in request.Items)
        {
            total += item.Price * item.Quantity;
        }

        // Prepare items description for Claude
        var itemsJson = JsonSerializer.Serialize(request.Items, new JsonSerializerOptions 
        { 
            WriteIndented = true 
        });

        // Use Claude to generate a summary and validate the order
        var messages = new List<Message>
        {
            new Message
            {
                Role = RoleType.User,
                Content = new List<ContentBase>
                {
                    new TextContent
                    {
                        Text = $@"You are a checkout assistant. Please analyze the following cart items and provide a brief order summary.

Cart Items:
{itemsJson}

Calculated Total: {total:C}

Please provide:
1. A brief summary of the order
2. Confirm the total amount
3. Any recommendations or notes about the order

Keep your response concise and professional."
                    }
                }
            }
        };

        var messageRequest = new MessageParameters
        {
            Model = ModelId,
            MaxTokens = 1024,
            Messages = messages
        };

        var response = await _client.Messages.GetClaudeMessageAsync(messageRequest);
        
        string summary = string.Empty;
        if (response.Content != null && response.Content.Count > 0)
        {
            foreach (var content in response.Content)
            {
                if (content is TextBlock textBlock)
                {
                    summary = textBlock.Text;
                    break;
                }
            }
        }

        return new CheckoutResult
        {
            Items = request.Items,
            Total = total,
            Summary = summary
        };
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        // Example usage with sample cart items
        var checkoutRequest = new CheckoutRequest
        {
            Items = new List<CartItem>
            {
                new CartItem { Name = "Laptop", Price = 999.99m, Quantity = 1 },
                new CartItem { Name = "Mouse", Price = 29.99m, Quantity = 2 },
                new CartItem { Name = "Keyboard", Price = 79.99m, Quantity = 1 },
                new CartItem { Name = "Monitor", Price = 399.99m, Quantity = 1 },
                new CartItem { Name = "USB Hub", Price = 49.99m, Quantity = 3 }
            }
        };

        Console.WriteLine("Processing checkout...");
        Console.WriteLine("=".PadRight(50, '='));
        
        Console.WriteLine("\nCart Items:");
        foreach (var item in checkoutRequest.Items)
        {
            Console.WriteLine($"  - {item.Name}: ${item.Price:F2} x {item.Quantity} = ${item.Price * item.Quantity:F2}");
        }

        var processor = new CheckoutProcessor();
        
        try
        {
            var result = await processor.ProcessCheckoutAsync(checkoutRequest);
            
            Console.WriteLine("\n" + "=".PadRight(50, '='));
            Console.WriteLine($"TOTAL: {result.Total:C}");
            Console.WriteLine("=".PadRight(50, '='));
            
            Console.WriteLine("\nOrder Summary from AI Assistant:");
            Console.WriteLine("-".PadRight(50, '-'));
            Console.WriteLine(result.Summary);
            Console.WriteLine("-".PadRight(50, '-'));
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error processing checkout: {ex.Message}");
            throw;
        }
    }
}
