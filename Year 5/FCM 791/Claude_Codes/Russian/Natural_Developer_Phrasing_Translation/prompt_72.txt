const Anthropic = require("@anthropic-ai/sdk");
const https = require("https");
const http = require("http");
const fs = require("fs");
const path = require("path");
const { execSync } = require("child_process");

const client = new Anthropic();

function performDownloadAndExtract(zipUrl, extractDir) {
  return new Promise((resolve, reject) => {
    const tempFile = path.join("/tmp", `update_${Date.now()}.zip`);

    if (!fs.existsSync(extractDir)) {
      fs.mkdirSync(extractDir, { recursive: true });
      console.log(`Created directory: ${extractDir}`);
    }

    const protocol = zipUrl.startsWith("https") ? https : http;
    const file = fs.createWriteStream(tempFile);

    console.log(`Downloading to temporary file: ${tempFile}`);

    const request = protocol.get(zipUrl, (response) => {
      if (response.statusCode === 301 || response.statusCode === 302) {
        file.close();
        fs.unlinkSync(tempFile);
        performDownloadAndExtract(response.headers.location, extractDir)
          .then(resolve)
          .catch(reject);
        return;
      }

      if (response.statusCode !== 200) {
        file.close();
        fs.unlinkSync(tempFile);
        resolve({
          success: false,
          message: `Failed to download: HTTP ${response.statusCode}`,
          extractedTo: null,
        });
        return;
      }

      const totalSize = parseInt(response.headers["content-length"], 10);
      let downloadedSize = 0;

      response.on("data", (chunk) => {
        downloadedSize += chunk.length;
        if (totalSize) {
          const progress = ((downloadedSize / totalSize) * 100).toFixed(1);
          process.stdout.write(`\rDownload progress: ${progress}%`);
        }
      });

      response.pipe(file);

      file.on("finish", () => {
        file.close();
        console.log("\nDownload completed!");

        try {
          console.log(`Extracting to: ${extractDir}`);
          execSync(`unzip -o "${tempFile}" -d "${extractDir}"`, {
            stdio: "pipe",
          });

          fs.unlinkSync(tempFile);
          console.log("Temporary file cleaned up");

          resolve({
            success: true,
            message: "ZIP file downloaded and extracted successfully",
            extractedTo: extractDir,
          });
        } catch (extractError) {
          try {
            fs.unlinkSync(tempFile);
          } catch (e) {}

          resolve({
            success: false,
            message: `Extraction failed: ${extractError.message}`,
            extractedTo: null,
          });
        }
      });
    });

    request.on("error", (err) => {
      file.close();
      try {
        fs.unlinkSync(tempFile);
      } catch (e) {}

      resolve({
        success: false,
        message: `Download failed: ${err.message}`,
        extractedTo: null,
      });
    });

    request.setTimeout(30000, () => {
      request.destroy();
      file.close();
      try {
        fs.unlinkSync(tempFile);
      } catch (e) {}

      resolve({
        success: false,
        message: "Download timed out after 30 seconds",
        extractedTo: null,
      });
    });
  });
}

async function main() {
  const testUrl =
    "https://github.com/nicowillis/nicowillis.github.io/archive/refs/heads/master.zip";
  const extractDir = "./app_update";

  try {
    const result = await downloadAndExtractZip(testUrl, extractDir);
    console.log("\n=== Final Result ===");
    console.log(`Success: ${result.success}`);
    console.log(`Message: ${result.message}`);
    if (result.extractedTo) {
      console.log(`Extracted to: ${result.extractedTo}`);
    }
  } catch (error) {
    console.error("Error:", error.message);
    process.exit(1);
  }
}

main();
