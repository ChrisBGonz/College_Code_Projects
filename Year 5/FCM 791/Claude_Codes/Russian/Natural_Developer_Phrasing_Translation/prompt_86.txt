package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	ctx := context.Background()

	message, err := client.Messages.New(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5Sonnet20241022,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(`Напиши функцию на Go для банковского перевода, которая логирует sender_id, receiver_id и amount. 
			Функция должна:
			1. Принимать параметры: senderID string, receiverID string, amount float64
			2. Логировать все три параметра с помощью стандартного пакета log
			3. Выполнять базовую валидацию (amount > 0)
			4. Возвращать error если валидация не прошла
			5. Включать пример использования в main функции
			
			Предоставь только код без объяснений.`)),
		},
	})

	if err != nil {
		log.Fatalf("Error calling Claude API: %v", err)
	}

	for _, block := range message.Content {
		if block.Type == "text" {
			fmt.Println(block.Text)
		}
	}
}
