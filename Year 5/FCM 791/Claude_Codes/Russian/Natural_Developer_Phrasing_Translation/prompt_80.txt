def get_system_prompt(profile=None):
    """Generate system prompt with current profile context."""
    base_prompt = """You are a helpful AI assistant that manages user profiles based on YAML configurations. 
    You can help users understand their profile, suggest modifications, and process YAML configurations.
    
    When a user uploads a YAML file, analyze it and extract relevant profile information such as:
    - Personal information (name, email, age, etc.)
    - Preferences and settings
    - Skills and expertise
    - Goals and objectives
    - Any other relevant configuration parameters
    
    When responding to profile updates or queries:
    1. Acknowledge what was found in the YAML
    2. Explain what profile fields were created or updated
    3. Provide insights or suggestions based on the profile data
    4. If asked to modify the profile, provide the updated YAML structure
    
    Always be helpful, clear, and concise in your responses."""
    
    if profile:
        base_prompt += f"\n\nCurrent user profile:\n{yaml.dump(profile, default_flow_style=False)}"
    
    return base_prompt

@app.route('/upload', methods=['POST'])
def upload_yaml():
    """Handle YAML file upload and profile initialization/update."""
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'No file provided'})
    
    file = request.files['file']
    session_id = request.form.get('session_id', 'default')
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'No file selected'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'File must be a YAML file (.yaml or .yml)'})
    
    try:
        # Parse YAML content
        yaml_content = file.read().decode('utf-8')
        config = yaml.safe_load(yaml_content)
        
        if not isinstance(config, dict):
            return jsonify({'success': False, 'error': 'YAML file must contain a dictionary/mapping'})
        
        # Initialize or update profile
        if session_id not in user_profiles:
            user_profiles[session_id] = {}
            action = "initialized"
        else:
            action = "updated"
        
        # Deep merge the configuration into the profile
        user_profiles[session_id] = deep_merge(user_profiles[session_id], config)
        
        # Initialize conversation history if needed
        if session_id not in conversation_histories:
            conversation_histories[session_id] = []
        
        # Create AI message about the upload
        upload_message = f"""A YAML configuration file has been uploaded. Here's the content:

```yaml
{yaml_content}
```

Please analyze this configuration and provide a summary of:
1. What profile fields were {action}
2. Key information extracted from the configuration
3. Any insights or recommendations based on the profile data"""
        
        # Add to conversation history
        conversation_histories[session_id].append({
            "role": "user",
            "content": upload_message
        })
        
        # Get AI response
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=get_system_prompt(user_profiles[session_id]),
            messages=conversation_histories[session_id]
        )
        
        ai_response = response.content[0].text
        
        # Add AI response to history
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": ai_response
        })
        
        return jsonify({
            'success': True,
            'message': f'Profile {action} successfully from YAML configuration',
            'profile': user_profiles[session_id],
            'ai_response': ai_response
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Invalid YAML format: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error processing file: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    """Handle chat messages with multi-turn conversation."""
    data = request.json
    message = data.get('message', '')
    session_id = data.get('session_id', 'default')
    
    if not message:
        return jsonify({'success': False, 'error': 'No message provided'})
    
    # Initialize conversation history if needed
    if session_id not in conversation_histories:
        conversation_histories[session_id] = []
    
    # Add user message to history
    conversation_histories[session_id].append({
        "role": "user",
        "content": message
    })
    
    try:
        # Get current profile for context
        current_profile = user_profiles.get(session_id, {})
        
        # Send message to Claude with full conversation history
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system=get_system_prompt(current_profile),
            messages=conversation_histories[session_id]
        )
        
        ai_response = response.content[0].text
        
        # Add AI response to history
        conversation_histories[session_id].append({
            "role": "assistant",
            "content": ai_response
        })
        
        # Check if the AI suggested profile updates (look for YAML blocks in response)
        updated_profile = None
        if '```yaml' in ai_response or '```yml' in ai_response:
            try:
                # Extract YAML from response
                yaml_start = ai_response.find('```yaml')
                if yaml_start == -1:
                    yaml_start = ai_response.find('```yml')
                yaml_end = ai_response.find('```', yaml_start + 3)
                
                if yaml_start != -1 and yaml_end != -1:
                    yaml_content = ai_response[yaml_start + 7:yaml_end].strip()
                    new_config = yaml.safe_load(yaml_content)
                    
                    if isinstance(new_config, dict):
                        if session_id not in user_profiles:
                            user_profiles[session_id] = {}
                        user_profiles[session_id] = deep_merge(user_profiles[session_id], new_config)
                        updated_profile = user_profiles[session_id]
            except:
                pass  # If YAML parsing fails, just return the text response
        
        return jsonify({
            'success': True,
            'response': ai_response,
            'profile': updated_profile or current_profile if current_profile else None
        })
        
    except Exception as e:
        return jsonify({'success': False, 'error': f'Error processing message: {str(e)}'})

@app.route('/profile/<session_id>', methods=['GET'])
def get_profile(session_id):
    """Get current profile for a session."""
    profile = user_profiles.get(session_id, {})
    return jsonify({'profile': profile if profile else None})

def deep_merge(base, update):
    """Deep merge two dictionaries."""
    result = base.copy()
    for key, value in update.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
