package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// Config represents the JSON configuration structure
type Config struct {
	Mappings []FieldMapping `json:"mappings"`
}

// FieldMapping represents a single field mapping configuration
type FieldMapping struct {
	SourceField string `json:"source_field"`
	TargetField string `json:"target_field"`
	Transform   string `json:"transform,omitempty"`
}

// DynamicObject represents an object with dynamic properties
type DynamicObject struct {
	Properties map[string]interface{}
}

// NewDynamicObject creates a new DynamicObject
func NewDynamicObject() *DynamicObject {
	return &DynamicObject{
		Properties: make(map[string]interface{}),
	}
}

// Set sets a property value
func (d *DynamicObject) Set(key string, value interface{}) {
	d.Properties[key] = value
}

// Get gets a property value
func (d *DynamicObject) Get(key string) (interface{}, bool) {
	val, ok := d.Properties[key]
	return val, ok
}

// ReadConfig reads and parses a JSON config file
func ReadConfig(filename string) (*Config, error) {
	data, err := os.ReadFile(filename)
	if err != nil {
		return nil, fmt.Errorf("error reading config file: %w", err)
	}

	var config Config
	if err := json.Unmarshal(data, &config); err != nil {
		return nil, fmt.Errorf("error parsing config JSON: %w", err)
	}

	return &config, nil
}

// ApplyTransform applies a transformation to a value
func ApplyTransform(value interface{}, transform string) interface{} {
	if transform == "" {
		return value
	}

	strVal := fmt.Sprintf("%v", value)

	switch strings.ToLower(transform) {
	case "uppercase":
		return strings.ToUpper(strVal)
	case "lowercase":
		return strings.ToLower(strVal)
	case "trim":
		return strings.TrimSpace(strVal)
	default:
		return value
	}
}

// MapFields dynamically maps fields from source to target object based on config
func MapFields(config *Config, source map[string]interface{}, target *DynamicObject) error {
	for _, mapping := range config.Mappings {
		// Get value from source
		sourceValue, exists := source[mapping.SourceField]
		if !exists {
			log.Printf("Warning: source field '%s' not found", mapping.SourceField)
			continue
		}

		// Apply transformation if specified
		transformedValue := ApplyTransform(sourceValue, mapping.Transform)

		// Set value in target
		target.Set(mapping.TargetField, transformedValue)
	}

	return nil
}

// GenerateConfigWithClaude uses Claude AI to generate a JSON config based on user description
func GenerateConfigWithClaude(client *anthropic.Client, description string) (string, error) {
	ctx := context.Background()

	var fullResponse strings.Builder

	stream := client.Messages.NewStreaming(ctx, anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5HaikuLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			{
				Role: anthropic.MessageParamRoleUser,
				Content: []anthropic.ContentBlockParamUnion{
					{
						OfRequestTextBlock: &anthropic.TextBlockParam{
							Text: fmt.Sprintf(`Generate a JSON configuration for field mapping based on this description: %s

The JSON should have this structure:
{
  "mappings": [
    {
      "source_field": "original_field_name",
      "target_field": "new_field_name",
      "transform": "optional_transform (uppercase/lowercase/trim or empty)"
    }
  ]
}

Return ONLY the JSON, no explanations.`, description),
						},
					},
				},
			},
		},
	})

	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Text != "" {
				fullResponse.WriteString(delta.Text)
				fmt.Print(delta.Text)
			}
		}
	}

	if err := stream.Err(); err != nil {
		return "", fmt.Errorf("streaming error: %w", err)
	}

	fmt.Println()
	return fullResponse.String(), nil
}

// SaveConfig saves a config string to a file
func SaveConfig(filename, content string) error {
	return os.WriteFile(filename, []byte(content), 0644)
}

func main() {
	// Initialize Anthropic client
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	configFile := "field_mapping_config.json"

	// Check if config file exists, if not generate one with Claude
	if _, err := os.Stat(configFile); os.IsNotExist(err) {
		fmt.Println("Config file not found. Generating with Claude AI...")
		fmt.Println("Generating field mapping configuration...")

		description := "Map user data fields: rename 'first_name' to 'firstName', 'last_name' to 'lastName' with uppercase transform, 'email_address' to 'email', and 'phone_number' to 'phone' with trim transform"

		configJSON, err := GenerateConfigWithClaude(client, description)
		if err != nil {
			log.Fatalf("Error generating config: %v", err)
		}

		// Save the generated config
		if err := SaveConfig(configFile, configJSON); err != nil {
			log.Fatalf("Error saving config: %v", err)
		}
		fmt.Printf("\nConfig saved to %s\n", configFile)
	}

	// Read the config file
	fmt.Printf("\nReading config from %s...\n", configFile)
	config, err := ReadConfig(configFile)
	if err != nil {
		log.Fatalf("Error reading config: %v", err)
	}

	fmt.Printf("Loaded %d field mappings\n", len(config.Mappings))

	// Example source data
	sourceData := map[string]interface{}{
		"first_name":   "john",
		"last_name":    "doe",
		"email_address": "john.doe@example.com",
		"phone_number": "  +1-555-0100  ",
		"age":          30,
	}

	fmt.Println("\nSource data:")
	for k, v := range sourceData {
		fmt.Printf("  %s: %v\n", k, v)
	}

	// Create target object and apply mappings
	target := NewDynamicObject()
	if err := MapFields(config, sourceData, target); err != nil {
		log.Fatalf("Error mapping fields: %v", err)
	}

	fmt.Println("\nMapped target object:")
	for k, v := range target.Properties {
		fmt.Printf("  %s: %v\n", k, v)
	}

	// Demonstrate interactive config generation with Claude
	fmt.Println("\n--- Interactive Config Generation ---")
	fmt.Println("Generating a new config for product data mapping...")

	productDescription := "Map product fields: rename 'product_name' to 'name' with uppercase transform, 'product_price' to 'price', 'product_description' to 'description' with trim transform, and 'product_sku' to 'sku' with uppercase transform"

	newConfigJSON, err := GenerateConfigWithClaude(client, productDescription)
	if err != nil {
		log.Fatalf("Error generating product config: %v", err)
	}

	// Parse and use the new config
	var productConfig Config
	if err := json.Unmarshal([]byte(newConfigJSON), &productConfig); err != nil {
		log.Fatalf("Error parsing generated config: %v", err)
	}

	// Example product data
	productData := map[string]interface{}{
		"product_name":        "awesome widget",
		"product_price":       29.99,
		"product_description": "  A fantastic product for everyone  ",
		"product_sku":         "wdg-001",
	}

	fmt.Println("\nProduct source data:")
	for k, v := range productData {
		fmt.Printf("  %s: %v\n", k, v)
	}

	productTarget := NewDynamicObject()
	if err := MapFields(&productConfig, productData, productTarget); err != nil {
		log.Fatalf("Error mapping product fields: %v", err)
	}

	fmt.Println("\nMapped product object:")
	for k, v := range productTarget.Properties {
		fmt.Printf("  %s: %v\n", k, v)
	}
}
