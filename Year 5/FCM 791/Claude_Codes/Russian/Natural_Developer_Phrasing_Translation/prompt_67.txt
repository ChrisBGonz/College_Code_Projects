import hmac
import struct
import time
import hashlib
import base64
import anthropic

client = anthropic.Anthropic()

def generate_totp(secret: str, time_step: int = 30, digits: int = 6) -> str:
    """Generate a TOTP code from a secret key."""
    key = base64.b32decode(secret.upper().replace(' ', ''))
    
    timestamp = int(time.time() // time_step)
    
    msg = struct.pack('>Q', timestamp)
    
    h = hmac.new(key, msg, hashlib.sha1).digest()
    
    offset = h[-1] & 0x0F
    code = struct.unpack('>I', h[offset:offset + 4])[0] & 0x7FFFFFFF
    
    return str(code % (10 ** digits)).zfill(digits)

def verify_totp_with_claude(secret: str, user_code: str) -> dict:
    """Use Claude to verify TOTP code and provide security analysis."""
    
    current_totp = generate_totp(secret)
    
    timestamp = int(time.time())
    time_remaining = 30 - (timestamp % 30)
    
    tools = [
        {
            "name": "verify_totp_code",
            "description": "Verify a TOTP (Time-based One-Time Password) code for 2FA authentication",
            "input_schema": {
                "type": "object",
                "properties": {
                    "is_valid": {
                        "type": "boolean",
                        "description": "Whether the provided TOTP code is valid"
                    },
                    "security_level": {
                        "type": "string",
                        "enum": ["high", "medium", "low"],
                        "description": "Security assessment of the authentication attempt"
                    },
                    "message": {
                        "type": "string",
                        "description": "Human-readable message about the verification result"
                    },
                    "time_remaining": {
                        "type": "integer",
                        "description": "Seconds remaining before the current TOTP expires"
                    },
                    "recommendation": {
                        "type": "string",
                        "description": "Security recommendation based on the verification attempt"
                    }
                },
                "required": ["is_valid", "security_level", "message", "time_remaining", "recommendation"]
            }
        }
    ]
    
    verification_prompt = f"""
    You are a security system verifying a 2FA TOTP code.
    
    Authentication Details:
    - User provided code: {user_code}
    - Expected valid code: {current_totp}
    - Time remaining for current code: {time_remaining} seconds
    - Current timestamp: {timestamp}
    
    Please verify if the user's code matches the expected TOTP code and provide a security assessment.
    Consider:
    1. Whether the codes match exactly
    2. The time remaining (if less than 5 seconds, warn about expiry)
    3. Any security concerns
    
    Use the verify_totp_code tool to return your verification result.
    """
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=[
            {"role": "user", "content": verification_prompt}
        ]
    )
    
    result = {
        "is_valid": False,
        "security_level": "low",
        "message": "Verification failed",
        "time_remaining": time_remaining,
        "recommendation": "Please try again"
    }
    
    for content_block in response.content:
        if content_block.type == "tool_use" and content_block.name == "verify_totp_code":
            result = content_block.input
            break
    
    return result

def simulate_2fa_login():
    """Simulate a complete 2FA login flow."""
    print("=" * 60)
    print("2FA Login Verification System")
    print("=" * 60)
    
    test_secret = "JBSWY3DPEHPK3PXP"
    
    print("\n[DEMO MODE] Using test secret key for demonstration")
    print(f"Secret key: {test_secret}")
    
    valid_code = generate_totp(test_secret)
    print(f"\nCurrent valid TOTP code: {valid_code}")
    print(f"(This code expires in {30 - (int(time.time()) % 30)} seconds)")
    
    print("\n" + "-" * 40)
    print("Test Case 1: Valid TOTP Code")
    print("-" * 40)
    
    result = verify_totp_with_claude(test_secret, valid_code)
    print(f"Verification Result: {'✓ SUCCESS' if result['is_valid'] else '✗ FAILED'}")
    print(f"Security Level: {result['security_level'].upper()}")
    print(f"Message: {result['message']}")
    print(f"Time Remaining: {result['time_remaining']} seconds")
    print(f"Recommendation: {result['recommendation']}")
    
    print("\n" + "-" * 40)
    print("Test Case 2: Invalid TOTP Code")
    print("-" * 40)
    
    invalid_code = "000000"
    result = verify_totp_with_claude(test_secret, invalid_code)
    print(f"User entered: {invalid_code}")
    print(f"Verification Result: {'✓ SUCCESS' if result['is_valid'] else '✗ FAILED'}")
    print(f"Security Level: {result['security_level'].upper()}")
    print(f"Message: {result['message']}")
    print(f"Time Remaining: {result['time_remaining']} seconds")
    print(f"Recommendation: {result['recommendation']}")
    
    print("\n" + "-" * 40)
    print("Test Case 3: Interactive Mode")
    print("-" * 40)
    
    user_input = input("\nEnter a 6-digit TOTP code to verify (or press Enter to skip): ").strip()
    
    if user_input:
        if len(user_input) == 6 and user_input.isdigit():
            result = verify_totp_with_claude(test_secret, user_input)
            print(f"\nVerification Result: {'✓ SUCCESS' if result['is_valid'] else '✗ FAILED'}")
            print(f"Security Level: {result['security_level'].upper()}")
            print(f"Message: {result['message']}")
            print(f"Time Remaining: {result['time_remaining']} seconds")
            print(f"Recommendation: {result['recommendation']}")
        else:
            print("Invalid input: Please enter exactly 6 digits")
    else:
        print("Skipping interactive test")
    
    print("\n" + "=" * 60)
    print("2FA Verification Demo Complete")
    print("=" * 60)

if __name__ == "__main__":
    simulate_2fa_login()
