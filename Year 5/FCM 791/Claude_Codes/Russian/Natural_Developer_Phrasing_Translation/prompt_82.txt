
```javascript
const express = require("express");
const Anthropic = require("@anthropic-ai/sdk");

const app = express();
const client = new Anthropic();

// Store conversation history for multi-turn conversations
const conversationHistory = [];

// Middleware to capture raw body
app.use((req, res, next) => {
  let rawBody = "";
  req.on("data", (chunk) => {
    rawBody += chunk.toString();
  });
  req.on("end", () => {
    req.rawBody = rawBody;
    next();
  });
});

// Parse JSON body
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// HTTP Request Logger Middleware with Claude AI Analysis
const httpRequestLogger = async (req, res, next) => {
  const timestamp = new Date().toISOString();
  const requestId = Math.random().toString(36).substr(2, 9);

  // Collect request data
  const requestData = {
    requestId,
    timestamp,
    method: req.method,
    url: req.url,
    headers: req.headers,
    body: req.body,
    rawBody: req.rawBody,
    query: req.query,
    params: req.params,
    ip: req.ip || req.connection.remoteAddress,
  };

  // Log to console
  console.log("\n" + "=".repeat(80));
  console.log(`[${timestamp}] REQUEST ID: ${requestId}`);
  console.log(`METHOD: ${requestData.method}`);
  console.log(`URL: ${requestData.url}`);
  console.log(`IP: ${requestData.ip}`);
  console.log("\nHEADERS:");
  console.log(JSON.stringify(requestData.headers, null, 2));

  if (requestData.query && Object.keys(requestData.query).length > 0) {
    console.log("\nQUERY PARAMS:");
    console.log(JSON.stringify(requestData.query, null, 2));
  }

  if (requestData.rawBody) {
    console.log("\nRAW BODY:");
    console.log(requestData.rawBody);
  }

  if (requestData.body && Object.keys(requestData.body).length > 0) {
    console.log("\nPARSED BODY:");
    console.log(JSON.stringify(requestData.body, null, 2));
  }

  // Use Claude AI to analyze the request
  try {
    const userMessage = `Analyze this HTTP request and provide insights about it:
    
Method: ${requestData.method}
URL: ${requestData.url}
Headers: ${JSON.stringify(requestData.headers, null, 2)}
${requestData.rawBody ? `Raw Body: ${requestData.rawBody}` : "No body"}
${requestData.body && Object.keys(requestData.body).length > 0 ? `Parsed Body: ${JSON.stringify(requestData.body, null, 2)}` : ""}

Please provide:
1. A brief summary of what this request is doing
2. Any security concerns or anomalies
3. Suggestions for debugging if needed`;

    conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system:
        "You are an expert HTTP request analyzer and security auditor. Analyze incoming HTTP requests and provide detailed insights about their purpose, potential security issues, and debugging suggestions. Be concise but thorough.",
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    console.log("\nCLAUDE AI ANALYSIS:");
    console.log(assistantMessage);
    console.log("=".repeat(80) + "\n");

    // Attach analysis to request for potential use in route handlers
    req.aiAnalysis = assistantMessage;
    req.requestId = requestId;
  } catch (error) {
    console.error("Error getting AI analysis:", error.message);
  }

  next();
};

// Apply the logger middleware
app.use(httpRequestLogger);

// Test routes
app.get("/", (req, res) => {
  res.json({
    message: "HTTP Logger with Claude AI is running",
    requestId: req.requestId,
    analysis: req.aiAnalysis,
  });
});

app.post("/api/data", (req, res) => {
  res.json({
    message: "Data received",
    requestId: req.requestId,
    receivedData: req.body,
    analysis: req.aiAnalysis,
  });
});

app.get("/api/users/:id", (req, res) => {
  res.json({
    message: `User ${req.params.id} requested`,
    requestId: req.requestId,
    analysis: req.aiAnalysis,
  });
});

app.post("/api/login", (req, res) => {
  res.json({
    message: "Login attempt logged",
    requestId: req.requestId,
    analysis: req.aiAnalysis,
  });
});

// Interactive conversation endpoint
app.post("/api/analyze", async (req, res) => {
  const { question } = req.body;

  if (!question) {
    return res.status(400).json({ error: "Question is required" });
  }

  try {
    conversationHistory.push({
      role: "user",
      content: question,
    });

    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system:
        "You are an expert HTTP request analyzer. You have been analyzing HTTP requests and can answer questions about them, provide debugging advice, and explain HTTP concepts.",
      messages: conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    res.json({
      response: assistantMessage,
      conversationLength: conversationHistory.length,
    });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// Get conversation history
app.get("/api/conversation-history", (req, res) => {
  res.json({
    history: conversationHistory,
    totalMessages: conversationHistory.length,
  });
});

// Clear conversation history
app.delete("/api/conversation-history", (req, res) => {
  conversationHistory.length = 0;
  res.json({ message: "Conversation history cleared" });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`HTTP Logger with Claude AI running on port ${PORT}`);
  console.log("\nAvailable endpoints:");
  console.log("  GET  /                          - Home page");
  console.log("  POST /api/data                  - Test POST endpoint");
  console.log("  GET  /api/users/:id             - Test parameterized endpoint");
  console.log("  POST /api/login                 - Test login endpoint");
  console.log(
    "  POST /api/analyze               - Ask Claude about HTTP requests"
  );
  console.log(
    "  GET  /api/conversation-history  - View conversation history"
  );
  console.log(
    "  DELETE /api/conversation-history - Clear conversation history"
  );
  console.log("\nAll requests will be logged and analyzed by Claude AI\n");
});

module.exports = app;
```
