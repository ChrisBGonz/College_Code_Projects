```ruby
class UsersController < ApplicationController
  before_action :authenticate_user!
  before_action :authorize_manager!
  before_action :find_staff_user, only: [:promote_to_manager]

  def promote_to_manager
    if @user.role == 'Staff'
      if @user.update(role: 'Manager')
        render json: {
          message: "User #{@user.email} has been successfully promoted to Manager.",
          user: {
            id: @user.id,
            email: @user.email,
            role: @user.role
          }
        }, status: :ok
      else
        render json: {
          error: 'Failed to promote user.',
          details: @user.errors.full_messages
        }, status: :unprocessable_entity
      end
    else
      render json: {
        error: "User cannot be promoted. Current role is '#{@user.role}', expected 'Staff'."
      }, status: :unprocessable_entity
    end
  end

  private

  def authorize_manager!
    unless current_user.role == 'Manager'
      render json: {
        error: 'Access denied. Only Managers can perform this action.'
      }, status: :forbidden
    end
  end

  def find_staff_user
    @user = User.find(params[:id])
  rescue ActiveRecord::RecordNotFound
    render json: { error: 'User not found.' }, status: :not_found
  end
end
```
