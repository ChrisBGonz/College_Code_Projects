import pickle
import os
from dataclasses import dataclass, field
from typing import Dict, List, Optional
from datetime import datetime
import anthropic

# Define complex GameState data structures
@dataclass
class Item:
    name: str
    damage: int
    defense: int
    value: int
    description: str = ""

@dataclass
class Quest:
    id: str
    title: str
    description: str
    completed: bool = False
    progress: int = 0
    max_progress: int = 100

@dataclass
class PlayerStats:
    health: int
    max_health: int
    mana: int
    max_mana: int
    strength: int
    agility: int
    intelligence: int
    level: int = 1
    experience: int = 0

@dataclass
class GameState:
    player_name: str
    stats: PlayerStats
    inventory: List[Item] = field(default_factory=list)
    quests: Dict[str, Quest] = field(default_factory=dict)
    current_location: str = "Starting Village"
    gold: int = 100
    play_time_seconds: int = 0
    save_timestamp: Optional[datetime] = None
    achievements: List[str] = field(default_factory=list)
    visited_locations: List[str] = field(default_factory=list)
    
    def add_item(self, item: Item):
        self.inventory.append(item)
    
    def add_quest(self, quest: Quest):
        self.quests[quest.id] = quest
    
    def complete_quest(self, quest_id: str):
        if quest_id in self.quests:
            self.quests[quest_id].completed = True
    
    def visit_location(self, location: str):
        if location not in self.visited_locations:
            self.visited_locations.append(location)
        self.current_location = location

def create_sample_game_state() -> GameState:
    """Create a complex sample game state for demonstration."""
    stats = PlayerStats(
        health=85,
        max_health=100,
        mana=60,
        max_mana=80,
        strength=15,
        agility=12,
        intelligence=18,
        level=5,
        experience=2450
    )
    
    game_state = GameState(
        player_name="Hero",
        stats=stats,
        gold=350,
        play_time_seconds=7200,
        save_timestamp=datetime.now()
    )
    
    # Add items to inventory
    sword = Item("Iron Sword", damage=25, defense=0, value=150, description="A sturdy iron sword")
    shield = Item("Wooden Shield", damage=0, defense=15, value=80, description="Basic wooden protection")
    potion = Item("Health Potion", damage=0, defense=0, value=50, description="Restores 50 HP")
    
    game_state.add_item(sword)
    game_state.add_item(shield)
    game_state.add_item(potion)
    
    # Add quests
    main_quest = Quest(
        id="main_001",
        title="The Dragon's Lair",
        description="Defeat the ancient dragon threatening the kingdom",
        completed=False,
        progress=30,
        max_progress=100
    )
    
    side_quest = Quest(
        id="side_001",
        title="Lost Merchant",
        description="Find the missing merchant in the forest",
        completed=True,
        progress=100,
        max_progress=100
    )
    
    game_state.add_quest(main_quest)
    game_state.add_quest(side_quest)
    
    # Add visited locations
    game_state.visit_location("Starting Village")
    game_state.visit_location("Dark Forest")
    game_state.visit_location("Mountain Pass")
    game_state.current_location = "Mountain Pass"
    
    # Add achievements
    game_state.achievements = ["First Kill", "Explorer", "Quest Master"]
    
    return game_state

def serialize_game_state(game_state: GameState, filename: str) -> str:
    """Serialize game state to a pickle file."""
    with open(filename, 'wb') as f:
        pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)
    return filename

def deserialize_game_state(filename: str) -> GameState:
    """Deserialize game state from a pickle file."""
    with open(filename, 'rb') as f:
        game_state = pickle.load(f)
    return game_state

def game_state_to_dict(game_state: GameState) -> dict:
    """Convert game state to a dictionary for Claude analysis."""
    return {
        "player_name": game_state.player_name,
        "level": game_state.stats.level,
        "experience": game_state.stats.experience,
        "health": f"{game_state.stats.health}/{game_state.stats.max_health}",
        "mana": f"{game_state.stats.mana}/{game_state.stats.max_mana}",
        "stats": {
            "strength": game_state.stats.strength,
            "agility": game_state.stats.agility,
            "intelligence": game_state.stats.intelligence
        },
        "gold": game_state.gold,
        "current_location": game_state.current_location,
        "inventory": [{"name": item.name, "damage": item.damage, "defense": item.defense} 
                      for item in game_state.inventory],
        "quests": {qid: {"title": q.title, "completed": q.completed, "progress": f"{q.progress}/{q.max_progress}"}
                   for qid, q in game_state.quests.items()},
        "achievements": game_state.achievements,
        "visited_locations": game_state.visited_locations,
        "play_time_hours": game_state.play_time_seconds / 3600
    }

def analyze_game_state_with_claude(game_state: GameState) -> str:
    """Use Claude to analyze the game state and provide recommendations."""
    client = anthropic.Anthropic()
    
    game_data = game_state_to_dict(game_state)
    
    prompt = f"""Analyze this RPG game state and provide:
1. A brief assessment of the player's current situation
2. Recommendations for next steps
3. Any notable achievements or areas for improvement

Game State:
{game_data}

Please provide a concise analysis in 3-4 sentences."""
    
    print("\nAsking Claude to analyze the game state...")
    
    # Use streaming for the response
    full_response = ""
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[
            {"role": "user", "content": prompt}
        ]
    ) as stream:
        print("\nClaude's Analysis:")
        print("-" * 40)
        for text in stream.text_stream:
            print(text, end="", flush=True)
            full_response += text
        print("\n" + "-" * 40)
    
    return full_response

def main():
    save_file = "game_save.pkl"
    
    print("=" * 60)
    print("RPG Game State Serialization Demo with Claude AI")
    print("=" * 60)
    
    # Create a complex game state
    print("\n1. Creating complex game state...")
    original_state = create_sample_game_state()
    print(f"   Player: {original_state.player_name}")
    print(f"   Level: {original_state.stats.level}")
    print(f"   Location: {original_state.current_location}")
    print(f"   Items in inventory: {len(original_state.inventory)}")
    print(f"   Active quests: {len(original_state.quests)}")
    
    # Serialize to file
    print(f"\n2. Serializing game state to '{save_file}'...")
    serialize_game_state(original_state, save_file)
    file_size = os.path.getsize(save_file)
    print(f"   File size: {file_size} bytes")
    print("   Serialization successful!")
    
    # Deserialize from file
    print(f"\n3. Deserializing game state from '{save_file}'...")
    loaded_state = deserialize_game_state(save_file)
    print("   Deserialization successful!")
    
    # Verify the loaded state
    print("\n4. Verifying loaded game state...")
    assert loaded_state.player_name == original_state.player_name, "Player name mismatch!"
    assert loaded_state.stats.level == original_state.stats.level, "Level mismatch!"
    assert len(loaded_state.inventory) == len(original_state.inventory), "Inventory mismatch!"
    assert len(loaded_state.quests) == len(original_state.quests), "Quests mismatch!"
    assert loaded_state.current_location == original_state.current_location, "Location mismatch!"
    print("   All data verified successfully!")
    
    # Display loaded state details
    print("\n5. Loaded Game State Details:")
    print(f"   Player: {loaded_state.player_name}")
    print(f"   Level: {loaded_state.stats.level} (XP: {loaded_state.stats.experience})")
    print(f"   Health: {loaded_state.stats.health}/{loaded_state.stats.max_health}")
    print(f"   Mana: {loaded_state.stats.mana}/{loaded_state.stats.max_mana}")
    print(f"   Gold: {loaded_state.gold}")
    print(f"   Current Location: {loaded_state.current_location}")
    print(f"   Visited Locations: {', '.join(loaded_state.visited_locations)}")
    print("\n   Inventory:")
    for item in loaded_state.inventory:
        print(f"     - {item.name} (DMG: {item.damage}, DEF: {item.defense})")
    print("\n   Quests:")
    for quest_id, quest in loaded_state.quests.items():
        status = "✓ Completed" if quest.completed else f"In Progress ({quest.progress}/{quest.max_progress})"
        print(f"     - {quest.title}: {status}")
    print(f"\n   Achievements: {', '.join(loaded_state.achievements)}")
    
    # Use Claude to analyze the game state
    print("\n6. Requesting AI analysis from Claude...")
    analysis = analyze_game_state_with_claude(loaded_state)
    
    # Clean up
    print("\n7. Cleaning up save file...")
    os.remove(save_file)
    print(f"   '{save_file}' removed successfully")
    
    print("\n" + "=" * 60)
    print("Demo completed successfully!")
    print("=" * 60)

if __name__ == "__main__":
    main()
