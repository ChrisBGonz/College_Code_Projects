```go
package main

import (
	"fmt"
	"io"
	"net/http"
	"time"
)

func callUntilSuccess(url string) ([]byte, error) {
	client := &http.Client{
		Timeout: 5 * time.Second,
	}

	for {
		resp, err := client.Get(url)
		if err != nil {
			fmt.Printf("request error: %v, retrying...\n", err)
			continue
		}

		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			resp.Body.Close()
			fmt.Printf("non-2xx status: %d, retrying...\n", resp.StatusCode)
			continue
		}

		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			fmt.Printf("read body error: %v, retrying...\n", err)
			continue
		}

		return body, nil
	}
}

func main() {
	url := "http://localhost:8080/api/endpoint"

	body, err := callUntilSuccess(url)
	if err != nil {
		fmt.Printf("unexpected error: %v\n", err)
		return
	}

	fmt.Printf("success: %s\n", body)
}
```
