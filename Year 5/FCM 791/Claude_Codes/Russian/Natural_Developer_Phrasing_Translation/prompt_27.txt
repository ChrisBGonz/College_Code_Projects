import anthropic
import subprocess
import sys
import os

def install_dependencies():
    """Install required dependencies."""
    packages = ["jinja2", "weasyprint"]
    for package in packages:
        subprocess.check_call([sys.executable, "-m", "pip", "install", package])

def render_html_template(template_string: str, context: dict) -> str:
    """Render HTML template with Jinja2."""
    from jinja2 import Template
    template = Template(template_string)
    return template.render(**context)

def html_to_pdf(html_content: str, output_path: str) -> bool:
    """Convert HTML content to PDF using WeasyPrint."""
    from weasyprint import HTML
    HTML(string=html_content).write_pdf(output_path)
    return True

def generate_html_template_with_claude(description: str) -> tuple[str, dict]:
    """Use Claude to generate an HTML template and sample data."""
    client = anthropic.Anthropic()
    
    prompt = f"""Generate an HTML template using Jinja2 syntax and sample data for the following description:
{description}

Please provide:
1. An HTML template with Jinja2 variables (use {{{{ variable_name }}}} syntax)
2. Sample data as a Python dictionary

Format your response as:
TEMPLATE:
[HTML template here]

DATA:
[Python dictionary here]

Make the template visually appealing with inline CSS styles."""

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=2048,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    response_text = message.content[0].text
    
    # Parse the response
    template_start = response_text.find("TEMPLATE:") + len("TEMPLATE:")
    data_start = response_text.find("DATA:")
    
    template_string = response_text[template_start:data_start].strip()
    data_string = response_text[data_start + len("DATA:"):].strip()
    
    # Clean up the data string - remove markdown code blocks if present
    if data_string.startswith("```"):
        lines = data_string.split('\n')
        # Remove first line (```python or ```) and last line (```)
        data_string = '\n'.join(lines[1:-1])
    
    # Safely evaluate the data dictionary
    sample_data = eval(data_string)
    
    return template_string, sample_data

def render_html_and_convert_to_pdf(
    template_string: str,
    context: dict,
    output_pdf_path: str
) -> dict:
    """
    Main function that renders HTML template and converts to PDF.
    
    Args:
        template_string: Jinja2 HTML template string
        context: Dictionary with template variables
        output_pdf_path: Path where the PDF will be saved
    
    Returns:
        Dictionary with status and file information
    """
    # Render HTML from template
    html_content = render_html_template(template_string, context)
    
    # Convert HTML to PDF
    success = html_to_pdf(html_content, output_pdf_path)
    
    if success:
        file_size = os.path.getsize(output_pdf_path)
        return {
            "status": "success",
            "pdf_path": output_pdf_path,
            "file_size_bytes": file_size,
            "html_length": len(html_content)
        }
    else:
        return {
            "status": "error",
            "message": "Failed to convert HTML to PDF"
        }

def main():
    """Main function demonstrating the HTML to PDF conversion."""
    print("Installing required dependencies...")
    install_dependencies()
    
    print("\nGenerating HTML template with Claude AI...")
    
    # Use Claude to generate a template
    description = "A professional invoice for a software development company"
    template_string, sample_data = generate_html_template_with_claude(description)
    
    print(f"\nGenerated template preview (first 200 chars):")
    print(template_string[:200] + "...")
    
    print(f"\nSample data keys: {list(sample_data.keys())}")
    
    # Convert to PDF
    output_path = "output.pdf"
    print(f"\nConverting to PDF: {output_path}")
    
    result = render_html_and_convert_to_pdf(
        template_string=template_string,
        context=sample_data,
        output_pdf_path=output_path
    )
    
    print(f"\nResult: {result}")
    
    if result["status"] == "success":
        print(f"\n✅ PDF successfully created!")
        print(f"   Path: {result['pdf_path']}")
        print(f"   Size: {result['file_size_bytes']} bytes")
        print(f"   HTML length: {result['html_length']} characters")
    
    # Also demonstrate with a custom template
    print("\n" + "="*50)
    print("Demonstrating with a custom template...")
    
    custom_template = """
    <!DOCTYPE html>
    <html>
    <head>
        <style>
            body { font-family: Arial, sans-serif; margin: 40px; }
            h1 { color: #2c3e50; border-bottom: 2px solid #3498db; }
            .info { background: #ecf0f1; padding: 20px; border-radius: 5px; }
            .highlight { color: #e74c3c; font-weight: bold; }
        </style>
    </head>
    <body>
        <h1>{{ title }}</h1>
        <div class="info">
            <p>Author: <span class="highlight">{{ author }}</span></p>
            <p>Date: {{ date }}</p>
            <p>{{ description }}</p>
        </div>
        <ul>
        {% for item in items %}
            <li>{{ item }}</li>
        {% endfor %}
        </ul>
    </body>
    </html>
    """
    
    custom_context = {
        "title": "Python PDF Generation Report",
        "author": "Claude AI Assistant",
        "date": "2024-01-15",
        "description": "This PDF was generated using Python with Jinja2 and WeasyPrint",
        "items": [
            "HTML template rendering with Jinja2",
            "PDF conversion with WeasyPrint",
            "AI-powered template generation with Claude",
            "Automated document creation"
        ]
    }
    
    custom_output = "custom_output.pdf"
    custom_result = render_html_and_convert_to_pdf(
        template_string=custom_template,
        context=custom_context,
        output_pdf_path=custom_output
    )
    
    print(f"\nCustom template result: {custom_result}")
    
    if custom_result["status"] == "success":
        print(f"\n✅ Custom PDF successfully created!")
        print(f"   Path: {custom_result['pdf_path']}")
        print(f"   Size: {custom_result['file_size_bytes']} bytes")

if __name__ == "__main__":
    main()
