
```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	prompt := `Generate a complete Go function that creates and returns a JWT (JSON Web Token) after successful user login. 

The function should:
1. Accept user credentials (username and password)
2. Validate the credentials against a mock user store
3. Generate a JWT token with appropriate claims (user ID, username, email, expiration time)
4. Return the JWT token string
5. Include proper error handling
6. Use the github.com/golang-jwt/jwt/v5 package

Please provide the complete, working Go code with all necessary imports, types, and a main function that demonstrates the usage.`

	fmt.Println("Generating JWT authentication function using Claude AI...")
	fmt.Println("=" + fmt.Sprintf("%s", "=")[0:1])
	fmt.Println()

	stream := client.Messages.NewStreaming(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 4096,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	var fullResponse string
	for stream.Next() {
		event := stream.Current()
		switch delta := event.Delta.(type) {
		case anthropic.ContentBlockDeltaEventDelta:
			if delta.Type == "text_delta" {
				fmt.Print(delta.Text)
				fullResponse += delta.Text
			}
		}
	}

	if err := stream.Err(); err != nil {
		log.Fatalf("Streaming error: %v", err)
	}

	fmt.Println()
	fmt.Println()
	fmt.Println("=" + fmt.Sprintf("%s", "=")[0:1])
	fmt.Printf("Generation completed at: %s\n", time.Now().Format("2006-01-02 15:04:05"))
	fmt.Printf("Total response length: %d characters\n", len(fullResponse))
}
```
