import Anthropic from "@anthropic-ai/sdk";
import crypto from "crypto";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: "user_123",
      email: "john@example.com",
      password_hash: "old_hash_123",
      updated_at: "2024-01-01T00:00:00Z",
    },
    {
      id: "user_456",
      email: "jane@example.com",
      password_hash: "old_hash_456",
      updated_at: "2024-01-01T00:00:00Z",
    },
  ],
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_password",
    description:
      "Validates password strength and returns validation result with details",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "hash_password",
    description: "Hashes a password using bcrypt-like algorithm (SHA-256 + salt)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The plain text password to hash",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "find_user",
    description: "Finds a user in the database by their ID",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "update_user_password",
    description: "Updates the user's password hash in the database",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "string",
          description: "The unique identifier of the user",
        },
        password_hash: {
          type: "string",
          description: "The new hashed password to store",
        },
      },
      required: ["user_id", "password_hash"],
    },
  },
];

// Tool implementations
function validatePassword(password) {
  const errors = [];

  if (password.length < 8) {
    errors.push("Password must be at least 8 characters long");
  }

  if (!/[A-Z]/.test(password)) {
    errors.push("Password must contain at least one uppercase letter");
  }

  if (!/[a-z]/.test(password)) {
    errors.push("Password must contain at least one lowercase letter");
  }

  if (!/[0-9]/.test(password)) {
    errors.push("Password must contain at least one number");
  }

  if (!/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) {
    errors.push("Password must contain at least one special character");
  }

  return {
    valid: errors.length === 0,
    errors: errors,
    strength:
      errors.length === 0
        ? "strong"
        : errors.length <= 2
          ? "moderate"
          : "weak",
  };
}

function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString("hex");
  const hash = crypto
    .createHash("sha256")
    .update(password + salt)
    .digest("hex");
  return `${salt}:${hash}`;
}

function findUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (!user) {
    return { found: false, error: `User with ID ${userId} not found` };
  }
  return {
    found: true,
    user: { id: user.id, email: user.email, updated_at: user.updated_at },
  };
}

function updateUserPassword(userId, passwordHash) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex === -1) {
    return { success: false, error: `User with ID ${userId} not found` };
  }

  database.users[userIndex].password_hash = passwordHash;
  database.users[userIndex].updated_at = new Date().toISOString();

  return {
    success: true,
    message: "Password updated successfully",
    updated_at: database.users[userIndex].updated_at,
  };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_password":
      return validatePassword(toolInput.password);
    case "hash_password":
      return hashPassword(toolInput.password);
    case "find_user":
      return findUser(toolInput.user_id);
    case "update_user_password":
      return updateUserPassword(toolInput.user_id, toolInput.password_hash);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\nInitiating password reset for user: ${userId}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user with ID "${userId}". The new password is: "${newPassword}". 
      
      Follow these steps:
      1. First, validate the new password meets security requirements
      2. Find the user in the database to verify they exist
      3. If both validations pass, hash the password securely
      4. Update the user's password in the database
      5. Return a clear success or failure message with details`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools: tools,
    messages: messages,
  });

  console.log(`Initial response - Stop reason: ${response.stop_reason}`);

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`\nExecuting tool: ${toolUse.name}`);
      console.log("Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Result:", JSON.stringify(result, null, 2));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({
      role: "assistant",
      content: response.content,
    });

    messages.push({
      role: "user",
      content: toolResults,
    });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools: tools,
      messages: messages,
    });

    console.log(`\nContinued response - Stop reason: ${response.stop_reason}`);
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n" + "=".repeat(50));
  console.log("Final Response:");
  console.log(finalResponse);

  return finalResponse;
}

// Test the function
async function main() {
  // Test 1: Valid password reset
  console.log("\n" + "=".repeat(60));
  console.log("TEST 1: Valid password reset");
  console.log("=".repeat(60));
  await resetPassword("user_123", "NewSecure@Pass123");

  // Test 2: Weak password
  console.log("\n" + "=".repeat(60));
  console.log("TEST 2: Weak password");
  console.log("=".repeat(60));
  await resetPassword("user_456", "weakpass");

  // Test 3: Non-existent user
  console.log("\n" + "=".repeat(60));
  console.log("TEST 3: Non-existent user");
  console.log("=".repeat(60));
  await resetPassword("user_999", "ValidPass@123");
}

main().catch(console.error);
