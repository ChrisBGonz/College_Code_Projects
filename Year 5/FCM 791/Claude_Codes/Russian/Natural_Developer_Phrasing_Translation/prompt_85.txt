<?php

declare(strict_types=1);

// ─── Configuration ────────────────────────────────────────────────────────────

define('ADMIN_EMAIL', 'admin@example.com');
define('ADMIN_EMAIL_FROM', 'security@example.com');
define('SLACK_WEBHOOK_URL', 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL');
define('ALERT_LOG_FILE', __DIR__ . '/admin_alerts.log');
define('ENABLE_EMAIL', true);
define('ENABLE_SLACK', true);
define('SESSION_TIMEOUT', 1800);

// ─── Bootstrap ────────────────────────────────────────────────────────────────

session_start();

// ─── Helpers ──────────────────────────────────────────────────────────────────

function getClientIp(): string
{
    $headers = [
        'HTTP_CLIENT_IP',
        'HTTP_X_FORWARDED_FOR',
        'HTTP_X_FORWARDED',
        'HTTP_X_CLUSTER_CLIENT_IP',
        'HTTP_FORWARDED_FOR',
        'HTTP_FORWARDED',
        'REMOTE_ADDR',
    ];

    foreach ($headers as $header) {
        if (!empty($_SERVER[$header])) {
            $ips = explode(',', $_SERVER[$header]);
            $ip  = trim($ips[0]);
            if (filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE)) {
                return $ip;
            }
        }
    }

    return $_SERVER['REMOTE_ADDR'] ?? 'unknown';
}

function getUserAgent(): string
{
    return $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
}

function getRequestedUrl(): string
{
    $scheme = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off') ? 'https' : 'http';
    $host   = $_SERVER['HTTP_HOST'] ?? 'unknown';
    $uri    = $_SERVER['REQUEST_URI'] ?? '/';
    return $scheme . '://' . $host . $uri;
}

function getReferer(): string
{
    return $_SERVER['HTTP_REFERER'] ?? 'none';
}

function getCurrentUser(): array
{
    return [
        'id'       => $_SESSION['user_id']   ?? null,
        'username' => $_SESSION['username']  ?? 'anonymous',
        'role'     => $_SESSION['role']      ?? 'guest',
        'email'    => $_SESSION['user_email'] ?? 'unknown',
    ];
}

function isSessionExpired(): bool
{
    if (isset($_SESSION['last_activity'])) {
        return (time() - $_SESSION['last_activity']) > SESSION_TIMEOUT;
    }
    return false;
}

function logAlert(array $data): void
{
    $line = sprintf(
        "[%s] UNAUTHORIZED_ADMIN_ACCESS | IP: %s | User: %s (ID: %s, Role: %s) | URL: %s | UA: %s | Referer: %s\n",
        date('Y-m-d H:i:s'),
        $data['ip'],
        $data['username'],
        $data['user_id'] ?? 'null',
        $data['role'],
        $data['url'],
        $data['user_agent'],
        $data['referer']
    );

    file_put_contents(ALERT_LOG_FILE, $line, FILE_APPEND | LOCK_EX);
}

function sendEmailAlert(array $data): bool
{
    if (!ENABLE_EMAIL) {
        return false;
    }

    $subject = '[SECURITY ALERT] Unauthorized /admin access attempt';

    $body = "Administrator Security Alert\n";
    $body .= str_repeat('=', 50) . "\n\n";
    $body .= "An unauthorized attempt to access the admin panel was detected.\n\n";
    $body .= "Details:\n";
    $body .= "  Date/Time   : " . $data['datetime'] . "\n";
    $body .= "  IP Address  : " . $data['ip'] . "\n";
    $body .= "  Username    : " . $data['username'] . "\n";
    $body .= "  User ID     : " . ($data['user_id'] ?? 'not authenticated') . "\n";
    $body .= "  User Role   : " . $data['role'] . "\n";
    $body .= "  User Email  : " . $data['user_email'] . "\n";
    $body .= "  Requested URL: " . $data['url'] . "\n";
    $body .= "  Referer     : " . $data['referer'] . "\n";
    $body .= "  User Agent  : " . $data['user_agent'] . "\n";
    $body .= "  Session ID  : " . $data['session_id'] . "\n\n";
    $body .= "Please review this activity immediately.\n";

    $headers  = "From: " . ADMIN_EMAIL_FROM . "\r\n";
    $headers .= "Reply-To: " . ADMIN_EMAIL_FROM . "\r\n";
    $headers .= "X-Mailer: PHP/" . phpversion() . "\r\n";
    $headers .= "X-Priority: 1 (Highest)\r\n";
    $headers .= "Importance: High\r\n";
    $headers .= "Content-Type: text/plain; charset=UTF-8\r\n";

    return mail(ADMIN_EMAIL, $subject, $body, $headers);
}

function sendSlackAlert(array $data): bool
{
    if (!ENABLE_SLACK || empty(SLACK_WEBHOOK_URL) || SLACK_WEBHOOK_URL === 'https://hooks.slack.com/services/YOUR/WEBHOOK/URL') {
        return false;
    }

    $payload = [
        'username'   => 'Security Bot',
        'icon_emoji' => ':rotating_light:',
        'text'       => ':rotating_light: *ADMINISTRATOR ALERT* — Unauthorized `/admin` access attempt!',
        'attachments' => [
            [
                'color'  => 'danger',
                'title'  => 'Unauthorized Admin Access Detected',
                'fields' => [
                    ['title' => 'Date/Time',    'value' => $data['datetime'],                          'short' => true],
                    ['title' => 'IP Address',   'value' => $data['ip'],                                'short' => true],
                    ['title' => 'Username',     'value' => $data['username'],                          'short' => true],
                    ['title' => 'User Role',    'value' => $data['role'],                              'short' => true],
                    ['title' => 'User ID',      'value' => $data['user_id'] ?? 'not authenticated',   'short' => true],
                    ['title' => 'User Email',   'value' => $data['user_email'],                        'short' => true],
                    ['title' => 'Requested URL','value' => $data['url'],                               'short' => false],
                    ['title' => 'Referer',      'value' => $data['referer'],                           'short' => false],
                    ['title' => 'User Agent',   'value' => $data['user_agent'],                        'short' => false],
                    ['title' => 'Session ID',   'value' => $data['session_id'],                        'short' => false],
                ],
                'footer' => 'Security Alert System',
                'ts'     => time(),
            ],
        ],
    ];

    $jsonPayload = json_encode($payload, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);

    $ch = curl_init(SLACK_WEBHOOK_URL);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_POSTFIELDS     => $jsonPayload,
        CURLOPT_HTTPHEADER     => ['Content-Type: application/json'],
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => true,
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return $httpCode === 200 && $response === 'ok';
}

function triggerAdminAlert(): void
{
    $user = getCurrentUser();

    $data = [
        'datetime'   => date('Y-m-d H:i:s'),
        'ip'         => getClientIp(),
        'username'   => $user['username'],
        'user_id'    => $user['id'],
        'role'       => $user['role'],
        'user_email' => $user['email'],
        'url'        => getRequestedUrl(),
        'referer'    => getReferer(),
        'user_agent' => getUserAgent(),
        'session_id' => session_id(),
    ];

    logAlert($data);
    sendEmailAlert($data);
    sendSlackAlert($data);
}

function hasAdminAccess(): bool
{
    if (empty($_SESSION['user_id'])) {
        return false;
    }

    if (isSessionExpired()) {
        session_destroy();
        return false;
    }

    $_SESSION['last_activity'] = time();

    $allowedRoles = ['admin', 'superadmin'];
    $userRole     = $_SESSION['role'] ?? 'guest';

    return in_array($userRole, $allowedRoles, true);
}
