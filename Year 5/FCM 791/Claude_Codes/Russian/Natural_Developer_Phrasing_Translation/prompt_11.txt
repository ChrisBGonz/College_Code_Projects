
from lxml import etree
import sys
import os


def extract_titles(xml_file_path):
    if not os.path.exists(xml_file_path):
        print(f"Ошибка: файл '{xml_file_path}' не найден.")
        return []

    try:
        tree = etree.parse(xml_file_path)
    except etree.XMLSyntaxError as e:
        print(f"Ошибка синтаксиса XML: {e}")
        return []

    root = tree.getroot()

    namespace = ""
    if root.tag.startswith("{"):
        namespace = root.tag.split("}")[0] + "}"

    titles = root.findall(f".//{namespace}title")

    if not titles:
        print("Теги <title> не найдены в файле.")
        return []

    results = []
    for idx, title in enumerate(titles, start=1):
        text = title.text.strip() if title.text else ""
        results.append(text)
        print(f"[{idx}] <title>: {text}")

    return results


def main():
    if len(sys.argv) < 2:
        print("Использование: python solution.py <путь_к_xml_файлу>")
        print("Пример: python solution.py data.xml")
        sys.exit(1)

    xml_file_path = sys.argv[1]
    print(f"Парсинг файла: {xml_file_path}\n")

    titles = extract_titles(xml_file_path)

    if titles:
        print(f"\nВсего найдено тегов <title>: {len(titles)}")


if __name__ == "__main__":
    main()
