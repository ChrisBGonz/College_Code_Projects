function sendError(res, statusCode, message) {
  const errorPayload = JSON.stringify({
    error: true,
    statusCode,
    message,
    timestamp: new Date().toISOString()
  }, null, 2);

  res.writeHead(statusCode, {
    'Content-Type': 'application/json',
    'Content-Length': Buffer.byteLength(errorPayload)
  });
  res.end(errorPayload);
}

function handleUserProfile(req, res) {
  const parsedUrl = url.parse(req.url, true);
  const { pathname, query } = parsedUrl;

  if (pathname !== '/api/user/profile') {
    sendError(res, 404, 'Endpoint not found. Use /api/user/profile?user_id=<id>');
    return;
  }

  if (req.method !== 'GET') {
    res.setHeader('Allow', 'GET');
    sendError(res, 405, 'Method not allowed. Only GET requests are supported.');
    return;
  }

  const { user_id } = query;

  if (!user_id) {
    sendError(res, 400, 'Missing required query parameter: user_id');
    return;
  }

  if (typeof user_id !== 'string' || user_id.trim() === '') {
    sendError(res, 400, 'Invalid user_id: must be a non-empty string');
    return;
  }

  const sanitizedUserId = user_id.trim();

  const user = users[sanitizedUserId];

  if (!user) {
    sendError(res, 404, `User with id "${sanitizedUserId}" not found`);
    return;
  }

  const responsePayload = JSON.stringify({
    success: true,
    data: user,
    exportedAt: new Date().toISOString()
  }, null, 2);

  const filename = `user_profile_${sanitizedUserId}.json`;

  res.writeHead(200, {
    'Content-Type': 'application/json',
    'Content-Disposition': `attachment; filename="${filename}"`,
    'Content-Length': Buffer.byteLength(responsePayload),
    'Cache-Control': 'no-store, no-cache, must-revalidate',
    'Pragma': 'no-cache'
  });

  res.end(responsePayload);
}

const PORT = process.env.PORT || 3000;

const server = http.createServer((req, res) => {
  console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);
  handleUserProfile(req, res);
});

server.listen(PORT, () => {
  console.log(`Server is running on http://localhost:${PORT}`);
  console.log(`Try: http://localhost:${PORT}/api/user/profile?user_id=1`);
  console.log(`Try: http://localhost:${PORT}/api/user/profile?user_id=2`);
  console.log(`Try: http://localhost:${PORT}/api/user/profile?user_id=3`);
});

server.on('error', (err) => {
  console.error('Server error:', err);
  process.exit(1);
});
