
```javascript
import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as url from "url";

const client = new Anthropic();

// Simulated user database
const users = {
  "token-alice-123": {
    id: 1,
    username: "alice",
    email: "alice@example.com",
    role: "admin",
    privateData: {
      creditCard: "**** **** **** 1234",
      ssn: "***-**-1234",
      address: "123 Main St, New York, NY 10001",
      phoneNumber: "+1-555-0101",
    },
  },
  "token-bob-456": {
    id: 2,
    username: "bob",
    email: "bob@example.com",
    role: "user",
    privateData: {
      creditCard: "**** **** **** 5678",
      ssn: "***-**-5678",
      address: "456 Oak Ave, Los Angeles, CA 90001",
      phoneNumber: "+1-555-0102",
    },
  },
};

// Function to extract Bearer token from Authorization header
function extractBearerToken(authHeader) {
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return null;
  }
  return authHeader.substring(7);
}

// Function to authenticate user by token
function authenticateUser(token) {
  return users[token] || null;
}

// Function to use Claude AI to analyze and format user data
async function analyzeUserDataWithClaude(userData, query) {
  const conversationHistory = [];

  // First message: provide context about the user
  conversationHistory.push({
    role: "user",
    content: `I have a user with the following data: ${JSON.stringify(userData, null, 2)}. ${query}`,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful API assistant that helps analyze and present user data in a clear and structured way. Always maintain data privacy and security best practices.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;
  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  // Second turn: ask for a security assessment
  conversationHistory.push({
    role: "user",
    content:
      "Based on this user data, what security recommendations would you make for this account?",
  });

  const securityResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful API assistant that helps analyze and present user data in a clear and structured way. Always maintain data privacy and security best practices.",
    messages: conversationHistory,
  });

  return {
    dataAnalysis: assistantMessage,
    securityRecommendations: securityResponse.content[0].text,
  };
}

// Main request handler
async function handleRequest(req, res) {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;

  // Set CORS headers
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Authorization, Content-Type");

  if (req.method === "OPTIONS") {
    res.writeHead(200);
    res.end();
    return;
  }

  // Route: GET /api/user/profile
  if (pathname === "/api/user/profile" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = authenticateUser(token);

    if (!user) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    try {
      // Use Claude to analyze user data with multi-turn conversation
      const analysis = await analyzeUserDataWithClaude(
        user,
        "Please provide a summary of this user's profile and their access level."
      );

      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          success: true,
          user: {
            id: user.id,
            username: user.username,
            email: user.email,
            role: user.role,
          },
          privateData: user.privateData,
          aiAnalysis: analysis,
        })
      );
    } catch (error) {
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
    return;
  }

  // Route: GET /api/user/private
  if (pathname === "/api/user/private" && req.method === "GET") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = authenticateUser(token);

    if (!user) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    try {
      // Use Claude to provide insights about private data
      const analysis = await analyzeUserDataWithClaude(
        user.privateData,
        "Analyze this private user data and explain what types of sensitive information are present."
      );

      res.writeHead(200, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          success: true,
          username: user.username,
          privateData: user.privateData,
          dataInsights: analysis,
        })
      );
    } catch (error) {
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal Server Error",
          message: "Failed to process request",
        })
      );
    }
    return;
  }

  // Route: POST /api/user/query
  if (pathname === "/api/user/query" && req.method === "POST") {
    const authHeader = req.headers["authorization"];
    const token = extractBearerToken(authHeader);

    if (!token) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Bearer token is required",
        })
      );
      return;
    }

    const user = authenticateUser(token);

    if (!user) {
      res.writeHead(401, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Unauthorized",
          message: "Invalid or expired token",
        })
      );
      return;
    }

    // Read request body
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });

    req.on("end", async () => {
      try {
        const { query } = JSON.parse(body);

        if (!query) {
          res.writeHead(400, { "Content-Type": "application/json" });
          res.end(
            JSON.stringify({
              error: "Bad Request",
              message: "Query is required",
            })
          );
          return;
        }

        // Use Claude to answer user's query about their data
        const analysis = await analyzeUserDataWithClaude(user, query);

        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            success: true,
            username: user.username,
            query: query,
            response: analysis,
          })
        );
      } catch (error) {
        res.writeHead(500, { "Content-Type": "application/json" });
        res.end(
          JSON.stringify({
            error: "Internal Server Error",
            message: "Failed to process request",
          })
        );
      }
    });
    return;
  }

  // Default route
  res.writeHead(404, { "Content-Type": "application/json" });
  res.end(
    JSON.stringify({
      error: "Not Found",
      message: "Endpoint not found",
      availableEndpoints: [
        "GET /api/user/profile - Get user profile (requires Bearer token)",
        "GET /api/user/private - Get private user data (requires Bearer token)",
        "POST /api/user/query - Query user data with AI (requires Bearer token)",
      ],
    })
  );
}

// Create and start server
const server = http.createServer(handleRequest);
const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log("\nAvailable test tokens:");
  console.log("- token-alice-123 (admin user)");
  console.log("- token-bob-456 (regular user)");
  console.log("\nExample requests:");
  console.log(
    `curl -H "Authorization: Bearer token-alice-123" http://localhost:${PORT}/api/user/profile`
  );
  console.log(
    `curl -H "Authorization: Bearer token-bob-456" http://localhost:${PORT}/api/user/private`
  );
  console.log(
    `curl -X POST -H "Authorization: Bearer token-alice-123" -H "Content-Type: application/json" -d '{"query":"What is my role?"}' http://localhost:${PORT}/api/user/query`
  );
});
```
