
```python
import anthropic
import json
import time
import random

client = anthropic.Anthropic()

tools = [
    {
        "name": "connect_payment_api",
        "description": "Connects to external payment API and processes payment request",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "Payment amount"
                },
                "currency": {
                    "type": "string",
                    "description": "Currency code (e.g., USD, EUR)"
                },
                "user_id": {
                    "type": "string",
                    "description": "User identifier"
                }
            },
            "required": ["amount", "currency", "user_id"]
        }
    }
]

def simulate_payment_api(amount: float, currency: str, user_id: str, timeout: float = 2.0) -> dict:
    """
    Simulates an external payment API call with potential timeout.
    Returns payment result or raises TimeoutError if response takes too long.
    """
    response_time = random.uniform(0.5, 3.5)
    
    print(f"Payment API response time: {response_time:.2f}s (timeout: {timeout}s)")
    
    if response_time > timeout:
        time.sleep(timeout)
        raise TimeoutError(f"Payment API did not respond within {timeout} seconds")
    
    time.sleep(response_time)
    
    return {
        "status": "success",
        "transaction_id": f"TXN_{random.randint(10000, 99999)}",
        "amount": amount,
        "currency": currency,
        "user_id": user_id,
        "timestamp": time.time()
    }

def process_tool_call(tool_name: str, tool_input: dict) -> dict:
    """
    Processes tool calls from Claude and returns results.
    """
    if tool_name == "connect_payment_api":
        try:
            result = simulate_payment_api(
                amount=tool_input["amount"],
                currency=tool_input["currency"],
                user_id=tool_input["user_id"]
            )
            return {
                "success": True,
                "allow_continue": True,
                "data": result
            }
        except TimeoutError as e:
            print(f"Timeout occurred: {e}")
            return {
                "success": False,
                "allow_continue": True,
                "reason": "timeout",
                "message": str(e)
            }
        except Exception as e:
            return {
                "success": False,
                "allow_continue": False,
                "reason": "error",
                "message": str(e)
            }
    
    return {"error": f"Unknown tool: {tool_name}"}

def connect_to_payment_api(amount: float, currency: str, user_id: str) -> bool:
    """
    Main function that connects to payment API using Claude as orchestrator.
    Returns True if user should be allowed to continue (either success or timeout).
    Returns False if there's a critical error.
    """
    print(f"\nProcessing payment: {amount} {currency} for user {user_id}")
    
    messages = [
        {
            "role": "user",
            "content": f"Process a payment of {amount} {currency} for user {user_id}. Use the connect_payment_api tool to make the payment. If the API times out, the user should still be allowed to continue."
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"Claude response stop reason: {response.stop_reason}")
        
        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, 'text'):
                    print(f"Claude's final response: {block.text}")
            
            response_text = ""
            for block in response.content:
                if hasattr(block, 'text'):
                    response_text += block.text.lower()
            
            if "allow" in response_text and "continue" in response_text:
                return True
            elif "timeout" in response_text or "timed out" in response_text:
                return True
            elif "success" in response_text:
                return True
            else:
                return False
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    print(f"Claude is calling tool: {block.name}")
                    print(f"Tool input: {json.dumps(block.input, indent=2)}")
                    
                    tool_result = process_tool_call(block.name, block.input)
                    print(f"Tool result: {json.dumps(tool_result, indent=2)}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(tool_result)
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            return True

def main():
    """
    Test the payment API connection function with multiple scenarios.
    """
    print("=" * 60)
    print("Testing Payment API Connection Function")
    print("=" * 60)
    
    test_cases = [
        {"amount": 100.00, "currency": "USD", "user_id": "user_001"},
        {"amount": 250.50, "currency": "EUR", "user_id": "user_002"},
        {"amount": 75.00, "currency": "GBP", "user_id": "user_003"},
    ]
    
    for i, test_case in enumerate(test_cases, 1):
        print(f"\n{'=' * 60}")
        print(f"Test Case {i}:")
        
        result = connect_to_payment_api(
            amount=test_case["amount"],
            currency=test_case["currency"],
            user_id=test_case["user_id"]
        )
        
        print(f"\nResult: {'✓ User can continue' if result else '✗ User cannot continue'}")
        print(f"Function returned: {result}")

if __name__ == "__main__":
    main()
```
