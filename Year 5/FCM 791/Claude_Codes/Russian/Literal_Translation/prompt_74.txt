import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function loadLibraryFromCDN() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Напиши JavaScript код для загрузки библиотеки jQuery из публичного CDN cdnjs.cloudflare.com. Код должен динамически создавать тег script, устанавливать src атрибут на URL CDN, и добавлять его в документ. Также добавь обработчики событий onload и onerror. Предоставь только чистый JavaScript код без объяснений.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";

  console.log("Generated code from Claude:");
  console.log("=".repeat(50));
  console.log(responseText);
  console.log("=".repeat(50));

  const codeMatch = responseText.match(/```(?:javascript|js)?\n([\s\S]*?)```/);
  const extractedCode = codeMatch ? codeMatch[1] : responseText;

  console.log("\nExtracted JavaScript code:");
  console.log(extractedCode);

  return extractedCode;
}

async function demonstrateLibraryLoading() {
  console.log("Requesting Claude to generate library loading code...\n");

  const generatedCode = await loadLibraryFromCDN();

  console.log("\nExample of how to use the generated code:");
  console.log("-".repeat(50));

  const exampleUsage = `
// Example usage in a browser environment:
function loadScript(url) {
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = url;
    script.onload = () => {
      console.log('Library loaded successfully!');
      resolve();
    };
    script.onerror = (error) => {
      console.error('Failed to load library:', error);
      reject(error);
    };
    document.head.appendChild(script);
  });
}

// Load jQuery from cdnjs.cloudflare.com
const jqueryUrl = 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js';

loadScript(jqueryUrl)
  .then(() => {
    console.log('jQuery version:', $.fn.jquery);
    // Now you can use jQuery
    $('body').css('background-color', '#f0f0f0');
  })
  .catch((error) => {
    console.error('Error loading jQuery:', error);
  });
`;

  console.log(exampleUsage);

  console.log("\nAlternative approach using async/await:");
  console.log("-".repeat(50));

  const asyncExample = `
async function initializeApp() {
  try {
    // Load jQuery
    await loadScript('https://cdnjs.cloudflare.com/ajax/libs/jquery/3.7.1/jquery.min.js');
    console.log('jQuery loaded, version:', $.fn.jquery);
    
    // Load additional libraries
    await loadScript('https://cdnjs.cloudflare.com/ajax/libs/lodash.js/4.17.21/lodash.min.js');
    console.log('Lodash loaded, version:', _.VERSION);
    
    // Your app initialization code here
    console.log('All libraries loaded, app ready!');
  } catch (error) {
    console.error('Failed to initialize app:', error);
  }
}

initializeApp();
`;

  console.log(asyncExample);

  return generatedCode;
}

demonstrateLibraryLoading()
  .then(() => {
    console.log("\nScript execution completed successfully!");
  })
  .catch((error) => {
