import anthropic
import ast
import socket
import threading
import time

client = anthropic.Anthropic()

XOR_KEY = 0x42

def xor_encrypt_decrypt(data: bytes, key: int = XOR_KEY) -> bytes:
    """Simple XOR cipher for encryption/decryption."""
    return bytes([b ^ key for b in data])

def send_encrypted_message(sock: socket.socket, message: str) -> None:
    """Encrypt and send a message through the socket."""
    encoded = message.encode('utf-8')
    encrypted = xor_encrypt_decrypt(encoded)
    length = len(encrypted)
    sock.sendall(length.to_bytes(4, byteorder='big'))
    sock.sendall(encrypted)

def receive_encrypted_message(sock: socket.socket) -> str:
    """Receive and decrypt a message from the socket."""
    length_bytes = b''
    while len(length_bytes) < 4:
        chunk = sock.recv(4 - len(length_bytes))
        if not chunk:
            raise ConnectionError("Connection closed")
        length_bytes += chunk
    
    length = int.from_bytes(length_bytes, byteorder='big')
    
    encrypted = b''
    while len(encrypted) < length:
        chunk = sock.recv(length - len(encrypted))
        if not chunk:
            raise ConnectionError("Connection closed")
        encrypted += chunk
    
    decrypted = xor_encrypt_decrypt(encrypted)
    return decrypted.decode('utf-8')

def legacy_server(host: str = 'localhost', port: int = 9999) -> None:
    """Simulate a legacy server that communicates using XOR encryption."""
    server_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server_sock.bind((host, port))
    server_sock.listen(1)
    server_sock.settimeout(30)
    
    print(f"Legacy server started on {host}:{port}")
    
    try:
        conn, addr = server_sock.accept()
        print(f"Server: Connection from {addr}")
        
        with conn:
            while True:
                try:
                    message = receive_encrypted_message(conn)
                    print(f"Server received: {message}")
                    
                    if message.lower() == 'quit':
                        send_encrypted_message(conn, "Goodbye!")
                        break
                    
                    response = f"Server processed: {message.upper()}"
                    send_encrypted_message(conn, response)
                    
                except ConnectionError:
                    print("Server: Client disconnected")
                    break
    except socket.timeout:
        print("Server: Timeout waiting for connection")
    finally:
        server_sock.close()
        print("Server: Closed")

def generate_ai_messages(conversation_history: list) -> str:
    """Use Claude to generate contextual messages for the legacy server."""
    system_prompt = """You are an AI assistant helping to test a legacy server communication system.
    Generate appropriate test messages to send to the server. The server will echo back messages in uppercase.
    Keep messages short and varied. After a few exchanges, send 'quit' to end the session.
    Respond with ONLY the message to send, nothing else."""
    
    with client.messages.stream(
        model="claude-opus-4-5",
        max_tokens=100,
        system=system_prompt,
        messages=conversation_history
    ) as stream:
        response = ""
        for text in stream.text_stream:
            response += text
        return response.strip()

def client_with_ai(host: str = 'localhost', port: int = 9999) -> None:
    """Client that uses Claude AI to generate messages for the legacy server."""
    time.sleep(0.5)
    
    client_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    
    try:
        client_sock.connect((host, port))
        print(f"Client: Connected to {host}:{port}")
        
        conversation_history = []
        max_exchanges = 5
        
        for i in range(max_exchanges):
            if i == 0:
                conversation_history.append({
                    "role": "user",
                    "content": "Start the conversation with the legacy server. Send an initial greeting."
                })
            else:
                conversation_history.append({
                    "role": "user", 
                    "content": f"Continue the conversation. This is exchange {i+1} of {max_exchanges}. If this is the last exchange, send 'quit'."
                })
            
            message = generate_ai_messages(conversation_history)
            print(f"\nClient sending: {message}")
            
            send_encrypted_message(client_sock, message)
            
            conversation_history.append({
                "role": "assistant",
                "content": message
            })
            
            response = receive_encrypted_message(client_sock)
            print(f"Client received: {response}")
            
            conversation_history.append({
                "role": "user",
                "content": f"Server responded with: {response}"
            })
            
            if message.lower() == 'quit' or response == "Goodbye!":
                print("Client: Session ended")
                break
            
            time.sleep(0.5)
            
    except ConnectionError as e:
        print(f"Client: Connection error - {e}")
    except Exception as e:
        print(f"Client: Error - {e}")
    finally:
        client_sock.close()
        print("Client: Connection closed")

def demonstrate_xor_cipher() -> None:
    """Demonstrate the XOR cipher functionality."""
    print("=== XOR Cipher Demonstration ===")
    
    original = "Hello, Legacy Server!"
    print(f"Original: {original}")
    
    encrypted = xor_encrypt_decrypt(original.encode())
    print(f"Encrypted (hex): {encrypted.hex()}")
    
    decrypted = xor_encrypt_decrypt(encrypted)
    print(f"Decrypted: {decrypted.decode()}")
    
    print(f"Encryption successful: {original == decrypted.decode()}")
    print()

def main() -> None:
    """Main function to run the legacy server communication demo."""
    print("=== Legacy Server Communication with XOR Encryption ===\n")
    
    demonstrate_xor_cipher()
    
    print("=== Starting Server-Client Communication ===\n")
    
    server_thread = threading.Thread(target=legacy_server, daemon=True)
    server_thread.start()
    
    client_with_ai()
    
    server_thread.join(timeout=5)
    
    print("\n=== Communication Demo Complete ===")

if __name__ == "__main__":
    main()
