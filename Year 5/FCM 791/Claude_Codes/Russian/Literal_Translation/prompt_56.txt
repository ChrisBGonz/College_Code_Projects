package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func checkRefererAccess(referer string) bool {
	return referer == "https://admin.internal.com" ||
		referer == "http://admin.internal.com" ||
		referer == "https://admin.internal.com/" ||
		referer == "http://admin.internal.com/"
}

func adminPanelHandler(w http.ResponseWriter, r *http.Request) {
	referer := r.Header.Get("Referer")

	if !checkRefererAccess(referer) {
		http.Error(w, "Access Denied: Invalid or missing Referer header", http.StatusForbidden)
		log.Printf("Access denied for request from %s with Referer: %s", r.RemoteAddr, referer)
		return
	}

	log.Printf("Access granted for request from %s with Referer: %s", r.RemoteAddr, referer)

	userQuery := r.URL.Query().Get("query")
	if userQuery == "" {
		userQuery = "What administrative tasks can I help you with today?"
	}

	response, err := getClaudeResponse(userQuery)
	if err != nil {
		http.Error(w, fmt.Sprintf("Error getting AI response: %v", err), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	fmt.Fprintf(w, `<!DOCTYPE html>
<html>
<head><title>Admin Panel</title></head>
<body>
<h1>Internal Admin Panel</h1>
<p><strong>Access granted from:</strong> %s</p>
<h2>AI Assistant Response:</h2>
<div style="background: #f0f0f0; padding: 20px; border-radius: 5px;">
<p>%s</p>
</div>
</body>
</html>`, referer, response)
}

func getClaudeResponse(query string) (string, error) {
	apiKey := os.Getenv("ANTHROPIC_API_KEY")
	if apiKey == "" {
		return "", fmt.Errorf("ANTHROPIC_API_KEY environment variable not set")
	}

	client := anthropic.NewClient(
		option.WithAPIKey(apiKey),
	)

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonceLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(
				fmt.Sprintf("You are an AI assistant for an internal admin panel. Help with the following query: %s", query),
			)),
		},
	})

	if err != nil {
		return "", fmt.Errorf("error calling Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude API")
	}

	var responseText string
	for _, block := range message.Content {
		if block.Type == "text" {
			responseText += block.Text
		}
	}

	return responseText, nil
}

func main() {
	http.HandleFunc("/admin", adminPanelHandler)
	http.HandleFunc("/admin/", adminPanelHandler)

	port := ":8080"
	log.Printf("Starting admin panel server on port %s", port)
	log.Printf("Access requires Referer header from admin.internal.com")

	if err := http.ListenAndServe(port, nil); err != nil {
		log.Fatalf("Failed to start server: %v", err)
	}
}
