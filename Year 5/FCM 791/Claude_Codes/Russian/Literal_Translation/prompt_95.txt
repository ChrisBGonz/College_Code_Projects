
import os
import sys
import time
import mmap


def count_lines_mmap(filepath):
    """Count lines using memory-mapped file."""
    count = 0
    with open(filepath, 'rb') as f:
        mm = mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ)
        for _ in iter(mm.readline, b''):
            count += 1
        mm.close()
    return count


def count_lines_mmap_find(filepath):
    """Count lines using mmap with find method (faster)."""
    count = 0
    with open(filepath, 'rb') as f:
        mm = mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ)
        buf = mm[:]
        count = buf.count(b'\n')
        mm.close()
    # Add 1 if file doesn't end with newline
    if buf and not buf.endswith(b'\n'):
        count += 1
    return count


def count_lines_chunks(filepath, chunk_size=1024 * 1024 * 64):
    """Count lines by reading file in large chunks."""
    count = 0
    with open(filepath, 'rb') as f:
        chunk = f.read(chunk_size)
        while chunk:
            count += chunk.count(b'\n')
            last_chunk = chunk
            chunk = f.read(chunk_size)
        # Add 1 if file doesn't end with newline
        if last_chunk and not last_chunk.endswith(b'\n'):
            count += 1
    return count


def count_lines_readline(filepath):
    """Count lines using readline iteration."""
    count = 0
    with open(filepath, 'rb') as f:
        for _ in f:
            count += 1
    return count


def get_file_size(filepath):
    """Get file size in human readable format."""
    size = os.path.getsize(filepath)
    for unit in ['B', 'KB', 'MB', 'GB', 'TB']:
        if size < 1024:
            return f"{size:.2f} {unit}"
        size /= 1024
    return f"{size:.2f} PB"


def benchmark_methods(filepath):
    """Benchmark all counting methods."""
    methods = [
        ("mmap (find/count)", count_lines_mmap_find),
        ("Chunked reading (64MB chunks)", count_lines_chunks),
        ("readline iteration", count_lines_readline),
        ("mmap (readline)", count_lines_mmap),
    ]

    print(f"\nFile: {filepath}")
    print(f"Size: {get_file_size(filepath)}")
    print("-" * 60)

    results = []
    for name, method in methods:
        print(f"\nMethod: {name}")
        try:
            start = time.perf_counter()
            line_count = method(filepath)
            elapsed = time.perf_counter() - start
            speed = os.path.getsize(filepath) / elapsed / (1024 ** 3)
            print(f"  Lines: {line_count:,}")
            print(f"  Time:  {elapsed:.2f} seconds")
            print(f"  Speed: {speed:.2f} GB/s")
            results.append((name, line_count, elapsed))
        except Exception as e:
            print(f"  Error: {e}")

    print("\n" + "=" * 60)
    print("Summary:")
    print("-" * 60)
    if results:
        fastest = min(results, key=lambda x: x[2])
        print(f"Fastest method: {fastest[0]} ({fastest[2]:.2f}s)")
        for name, count, elapsed in results:
            print(f"  {name}: {count:,} lines in {elapsed:.2f}s")


def create_test_file(filepath, size_gb=1):
    """Create a test file of specified size in GB."""
    print(f"Creating test file of ~{size_gb} GB...")
    target_size = size_gb * 1024 * 1024 * 1024
    line = b"This is a sample line of text for testing purposes. " * 2 + b"\n"
    line_size = len(line)
    total_lines = target_size // line_size

    with open(filepath, 'wb') as f:
        chunk_lines = 100000
        chunk = line * chunk_lines
        written_lines = 0
        while written_lines < total_lines:
            remaining = total_lines - written_lines
            if remaining < chunk_lines:
                f.write(line * remaining)
                written_lines += remaining
            else:
                f.write(chunk)
                written_lines += chunk_lines

            progress = written_lines / total_lines * 100
            print(f"\r  Progress: {progress:.1f}% ({written_lines:,} lines)", end='', flush=True)

    print(f"\nFile created: {filepath} ({get_file_size(filepath)})")
    return written_lines


def main():
    if len(sys.argv) > 1:
        filepath = sys.argv[1]
        if not os.path.exists(filepath):
            print(f"Error: File '{filepath}' not found.")
            sys.exit(1)
        benchmark_methods(filepath)
    else:
        # Create a test file if no argument provided
        test_file = "test_large_file.txt"
        print("No file specified. Creating a test file...")
        print("Usage: python solution.py <path_to_large_file>")
        print()

        # Create a smaller test file for demonstration (1 GB)
        test_size_gb = 1
        if not os.path.exists(test_file):
            create_test_file(test_file, size_gb=test_size_gb)
        else:
            print(f"Using existing test file: {test_file}")

        benchmark_methods(test_file)

        # Cleanup
        cleanup = input("\nDelete test file? (y/n): ").strip().lower()
        if cleanup == 'y':
            os.remove(test_file)
            print("Test file deleted.")


if __name__ == "__main__":
    main()
