def extract_url_content_with_claude(url: str) -> tuple[str, str | None]:
    """
    Use Claude to fetch and analyze URL content.
    Returns (content, error) tuple.
    """
    client = anthropic.Anthropic()
    
    prompt = f"""Please fetch and display the content from this URL: {url}

If you cannot access the URL or if it returns a 404 error, please indicate that clearly.
If the URL is accessible, provide a comprehensive summary of its content including:
1. The main topic or purpose of the page
2. Key information or content found on the page
3. Any important links or sections

URL to analyze: {url}

Note: If this URL returns a 404 error or is not found, please explicitly state "404 Error: Page not found" at the beginning of your response."""
    
    content_parts = []
    error = None
    
    try:
        with client.messages.stream(
            model="claude-opus-4-5",
            max_tokens=2048,
            messages=[
                {
                    "role": "user",
                    "content": prompt
                }
            ]
        ) as stream:
            for text in stream.text_stream:
                content_parts.append(text)
        
        full_content = "".join(content_parts)
        
        if "404" in full_content[:100] or "not found" in full_content[:100].lower():
            error = f"404 Ошибка: Страница не найдена по адресу {url}"
            return "", error
            
        return full_content, None
        
    except anthropic.APIError as e:
        error = f"Ошибка API: {str(e)}"
        return "", error
    except Exception as e:
        error = f"Произошла ошибка: {str(e)}"
        return "", error


@app.route("/", methods=["GET", "POST"])
def index():
    """Main route that handles URL extraction."""
    url = None
    content = None
    error = None
    
    if request.method == "POST":
        url = request.form.get("url", "").strip()
        
        if not url:
            error = "Пожалуйста, введите URL"
        elif not url.startswith(("http://", "https://")):
            error = "URL должен начинаться с http:// или https://"
        else:
            content, error = extract_url_content_with_claude(url)
    
    return render_template_string(
        HTML_TEMPLATE,
        url=url,
        content=content,
        error=error
    )


@app.errorhandler(404)

    return render_template_string(

        url=None,
        content=None,
        error="404 Ошибка: Запрошенная страница не найдена"
    ), 404


@app.errorhandler(500)
def internal_server_error(e):
    """Handle 500 errors."""
    return render_template_string(
        HTML_TEMPLATE,
        url=None,
        content=None,
        error="500 Ошибка: Внутренняя ошибка сервера"
    ), 500
