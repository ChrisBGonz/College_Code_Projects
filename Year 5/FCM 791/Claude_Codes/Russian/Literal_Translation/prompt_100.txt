const CircuitBreakerState = {
  CLOSED: 'CLOSED',
  OPEN: 'OPEN',
  HALF_OPEN: 'HALF_OPEN',
};

class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold ?? 5;
    this.successThreshold = options.successThreshold ?? 2;
    this.timeout = options.timeout ?? 60000;
    this.requestTimeout = options.requestTimeout ?? 10000;
    this.monitorInterval = options.monitorInterval ?? 5000;
    this.name = options.name ?? 'CircuitBreaker';

    this.state = CircuitBreakerState.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.lastFailureTime = null;
    this.nextAttemptTime = null;
    this.requestQueue = [];
    this.isProcessingQueue = false;

    this.stats = {
      totalRequests: 0,
      successfulRequests: 0,
      failedRequests: 0,
      rejectedRequests: 0,
      timeouts: 0,
      stateChanges: [],
    };

    this.listeners = {
      stateChange: [],
      success: [],
      failure: [],
      reject: [],
      timeout: [],
    };

    this._startMonitoring();
  }

  get currentState() {
    return this.state;
  }

  get isOpen() {
    return this.state === CircuitBreakerState.OPEN;
  }

  get isClosed() {
    return this.state === CircuitBreakerState.CLOSED;
  }

  get isHalfOpen() {
    return this.state === CircuitBreakerState.HALF_OPEN;
  }

  on(event, listener) {
    if (this.listeners[event]) {
      this.listeners[event].push(listener);
    }
    return this;
  }

  off(event, listener) {
    if (this.listeners[event]) {
      this.listeners[event] = this.listeners[event].filter((l) => l !== listener);
    }
    return this;
  }

  _emit(event, data) {
    if (this.listeners[event]) {
      this.listeners[event].forEach((listener) => {
        try {
          listener(data);
        } catch (err) {
          console.error(`[${this.name}] Listener error for event "${event}":`, err);
        }
      });
    }
  }

  _transitionTo(newState) {
    const previousState = this.state;
    this.state = newState;

    const change = {
      from: previousState,
      to: newState,
      timestamp: new Date().toISOString(),
      stats: { ...this.stats },
    };

    this.stats.stateChanges.push(change);

    console.log(
      `[${this.name}] State transition: ${previousState} → ${newState} at ${change.timestamp}`
    );

    this._emit('stateChange', change);
  }

  _recordSuccess() {
    this.failureCount = 0;
    this.stats.successfulRequests++;
    this._emit('success', { timestamp: new Date().toISOString() });

    if (this.state === CircuitBreakerState.HALF_OPEN) {
      this.successCount++;
      if (this.successCount >= this.successThreshold) {
        this.successCount = 0;
        this._transitionTo(CircuitBreakerState.CLOSED);
        this._processQueue();
      }
    }
  }

  _recordFailure(error) {
    this.failureCount++;
    this.lastFailureTime = Date.now();
    this.stats.failedRequests++;
    this._emit('failure', { error, timestamp: new Date().toISOString() });

    if (
      this.state === CircuitBreakerState.CLOSED &&
      this.failureCount >= this.failureThreshold
    ) {
      this.nextAttemptTime = Date.now() + this.timeout;
      this._transitionTo(CircuitBreakerState.OPEN);
    } else if (this.state === CircuitBreakerState.HALF_OPEN) {
      this.successCount = 0;
      this.nextAttemptTime = Date.now() + this.timeout;
      this._transitionTo(CircuitBreakerState.OPEN);
    }
  }

  _shouldAttemptReset() {
    return this.state === CircuitBreakerState.OPEN && Date.now() >= this.nextAttemptTime;
  }

  _withTimeout(promise) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        this.stats.timeouts++;
        this._emit('timeout', { timestamp: new Date().toISOString() });
        reject(new CircuitBreakerTimeoutError(`Request timed out after ${this.requestTimeout}ms`));
      }, this.requestTimeout);

      promise
        .then((result) => {
          clearTimeout(timer);
          resolve(result);
        })
        .catch((err) => {
          clearTimeout(timer);
          reject(err);
        });
    });
  }

  async execute(requestFn, options = {}) {
    this.stats.totalRequests++;

    if (this._shouldAttemptReset()) {
      this.successCount = 0;
      this._transitionTo(CircuitBreakerState.HALF_OPEN);
    }

    if (this.state === CircuitBreakerState.OPEN) {
      this.stats.rejectedRequests++;
      const error = new CircuitBreakerOpenError(
        `Circuit breaker "${this.name}" is OPEN. Next attempt at ${new Date(
          this.nextAttemptTime
        ).toISOString()}`
      );
      this._emit('reject', { error, timestamp: new Date().toISOString() });

      if (options.fallback) {
        return options.fallback(error);
      }

      throw error;
    }

    if (this.state === CircuitBreakerState.HALF_OPEN && options.queue) {
      return this._enqueue(requestFn, options);
    }

    try {
      const result = await this._withTimeout(Promise.resolve().then(() => requestFn()));
      this._recordSuccess();
      return result;
    } catch (err) {
      if (!(err instanceof CircuitBreakerTimeoutError) || options.countTimeouts !== false) {
        this._recordFailure(err);
      }

      if (options.fallback) {
        return options.fallback(err);
      }

      throw err;
    }
  }

  _enqueue(requestFn, options) {
    return new Promise((resolve, reject) => {
      this.requestQueue.push({ requestFn, options, resolve, reject });
    });
  }

  async _processQueue() {
    if (this.isProcessingQueue || this.requestQueue.length === 0) return;

    this.isProcessingQueue = true;

    while (this.requestQueue.length > 0 && this.state === CircuitBreakerState.CLOSED) {
      const { requestFn, options, resolve, reject } = this.requestQueue.shift();
      try {
        const result = await this.execute(requestFn, options);
        resolve(result);
      } catch (err) {
        reject(err);
      }
    }

    if (this.state !== CircuitBreakerState.CLOSED) {
      this.requestQueue.forEach(({ reject }) => {
        reject(
          new CircuitBreakerOpenError(`Circuit breaker "${this.name}" opened during queue processing`)
        );
      });
      this.requestQueue = [];
    }

    this.isProcessingQueue = false;
  }

  _startMonitoring() {
    this._monitorTimer = setInterval(() => {
      if (this._shouldAttemptReset()) {
        this.successCount = 0;
        this._transitionTo(CircuitBreakerState.HALF_OPEN);
      }
    }, this.monitorInterval);

    if (typeof this._monitorTimer === 'object' && this._monitorTimer.unref) {
      this._monitorTimer.unref();
    }
  }

  reset() {
    this.failureCount = 0;
    this.successCount = 0;
    this.lastFailureTime = null;
    this.nextAttemptTime = null;
    this._transitionTo(CircuitBreakerState.CLOSED);
    this._processQueue();
  }

  forceOpen() {
    this.nextAttemptTime = Date.now() + this.timeout;
    this._transitionTo(CircuitBreakerState.OPEN);
  }

  getStats() {
    return {
      name: this.name,
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      lastFailureTime: this.lastFailureTime
        ? new Date(this.lastFailureTime).toISOString()
        : null,
      nextAttemptTime: this.nextAttemptTime
        ? new Date(this.nextAttemptTime).toISOString()
        : null,
      queueLength: this.requestQueue.length,
      ...this.stats,
    };
  }

  destroy() {
    clearInterval(this._monitorTimer);
    this.requestQueue.forEach(({ reject }) => {
      reject(new Error(`Circuit breaker "${this.name}" was destroyed`));
    });
    this.requestQueue = [];
    this.listeners = { stateChange: [], success: [], failure: [], reject: [], timeout: [] };
  }
}

class CircuitBreakerOpenError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CircuitBreakerOpenError';
    this.isCircuitBreakerError = true;
  }
}

class CircuitBreakerTimeoutError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CircuitBreakerTimeoutError';
    this.isCircuitBreakerError = true;
  }
}

class CircuitBreakerManager {
  constructor() {
    this.breakers = new Map();
  }

  create(name, options = {}) {
    if (this.breakers.has(name)) {
      return this.breakers.get(name);
    }
    const breaker = new CircuitBreaker({ ...options, name });
    this.breakers.set(name, breaker);
    return breaker;
  }

  get(name) {
    return this.breakers.get(name);
  }

  getOrCreate(name, options = {}) {
    return this.breakers.has(name) ? this.breakers.get(name) : this.create(name, options);
  }

  getAllStats() {
    const stats = {};
    this.breakers.forEach((breaker, name) => {
      stats[name] = breaker.getStats();
    });
    return stats;
  }

  resetAll() {
    this.breakers.forEach((breaker) => breaker.reset());
  }

  destroy(name) {
    const breaker = this.breakers.get(name);
    if (breaker) {
      breaker.destroy();
      this.breakers.delete(name);
    }
  }

  destroyAll() {
    this.breakers.forEach((breaker) => breaker.destroy());
    this.breakers.clear();
  }
}

function withCircuitBreaker(breaker, options = {}) {
  return async function (...args) {
    return breaker.execute(() => options.fn(...args), options);
  };
}

async function fetchWithCircuitBreaker(url, fetchOptions = {}, breakerOptions = {}) {
  const manager = fetchWithCircuitBreaker._manager;
  const breakerName = breakerOptions.name ?? new URL(url, window.location.href).hostname;
  const breaker = manager.getOrCreate(breakerName, breakerOptions);

  return breaker.execute(
    async () => {
      const response = await fetch(url, fetchOptions);
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }
      return response;
    },
    {
      fallback: breakerOptions.fallback,
    }
  );
}

fetchWithCircuitBreaker._manager = new CircuitBreakerManager();

class ApiClient {
  constructor(baseUrl, options = {}) {
    this.baseUrl = baseUrl;
    this.breaker = new CircuitBreaker({
      name: options.name ?? baseUrl,
      failureThreshold: options.failureThreshold ?? 5,
      successThreshold: options.successThreshold ?? 2,
      timeout: options.timeout ?? 30000,
      requestTimeout: options.requestTimeout ?? 10000,
    });

    this.defaultHeaders = options.headers ?? {};
    this.fallback = options.fallback ?? null;

    this.breaker
      .on('stateChange', ({ from, to }) => {
        console.log(`[ApiClient:${this.baseUrl}] Circuit breaker: ${from} → ${to}`);
      })
      .on('reject', ({ error }) => {
        console.warn(`[ApiClient:${this.baseUrl}] Request rejected:`, error.message);
      });
  }

  async request(path, options = {}) {
    const url = `${this.baseUrl}${path}`;
    const { method = 'GET', body, headers = {}, fallback, ...rest } = options;

    return this.breaker.execute(
      async () => {
        const response = await fetch(url, {
          method,
          headers: {
            'Content-Type': 'application/json',
            ...this.defaultHeaders,
            ...headers,
          },
          body: body ? JSON.stringify(body) : undefined,
          ...rest,
        });

        if (!response.ok) {
          const errorBody = await response.text().catch(() => '');
          throw new Error(`HTTP ${response.status}: ${response.statusText}. ${errorBody}`);
        }

        const contentType = response.headers.get('content-type');
        if (contentType && contentType.includes('application/json')) {
          return response.json();
        }
        return response.text();
      },
      {
        fallback: fallback ?? this.fallback,
      }
    );
  }

  get(path, options = {}) {
    return this.request(path, { ...options, method: 'GET' });
  }

  post(path, body, options = {}) {
    return this.request(path, { ...options, method: 'POST', body });
  }

  put(path, body, options = {}) {
    return this.request(path, { ...options, method: 'PUT', body });
  }

  patch(path, body, options = {}) {
    return this.request(path, { ...options, method: 'PATCH', body });
  }

  delete(path, options = {}) {
    return this.request(path, { ...options, method: 'DELETE' });
  }

  getStats() {
    return this.breaker.getStats();
  }

  reset() {
    this.breaker.reset();
  }

  destroy() {
    this.breaker.destroy();
  }
}

async function demonstrateCircuitBreaker() {
  console.log('=== Circuit Breaker Demo ===\n');

  let callCount = 0;

  const unreliableService = async () => {
    callCount++;
    const shouldFail = callCount <= 7 || (callCount > 9 && callCount <= 11);

    await new Promise((resolve) => setTimeout(resolve, 100));

    if (shouldFail) {
      throw new Error(`Service unavailable (call #${callCount})`);
    }

    return { data: `Success on call #${callCount}` };
  };

  const breaker = new CircuitBreaker({
    name: 'DemoService',
    failureThreshold: 3,
    successThreshold: 2,
    timeout: 2000,
    requestTimeout: 5000,
  });

  breaker
    .on('stateChange', ({ from, to, timestamp }) => {
      console.log(`\n🔄 STATE CHANGE: ${from} → ${to} at ${timestamp}\n`);
    })
    .on('failure', ({ error }) => {
      console.log(`❌ Failure: ${error.message}`);
    })
    .on('success', () => {
      console.log(`✅ Success`);
    })
    .on('reject', ({ error }) => {
      console.log(`🚫 Rejected: ${error.message}`);
    });

  const makeRequest = async (label) => {
    try {
      const result = await breaker.execute(unreliableService, {
        fallback: (err) => ({ fallback: true, reason: err.message }),
      });
      console.log(`[${label}] Result:`, result);
    } catch (err) {
      console.log(`[${label}] Error: ${err.message}`);
    }
    console.log(`[${label}] Stats:`, {
      state: breaker.currentState,
      failures: breaker.failureCount,
      successes: breaker.successCount,
    });
  };

  console.log('--- Phase 1: Failures accumulating ---');
  for (let i = 1; i <= 5; i++) {
    await makeRequest(`Request ${i}`);
    await new Promise((r) => setTimeout(r, 100));
  }

  console.log('\n--- Phase 2: Circuit is OPEN, requests rejected ---');
  for (let i = 6; i <= 8; i++) {
    await makeRequest(`Request ${i}`);
    await new Promise((r) => setTimeout(r, 100));
  }

  console.log('\n--- Phase 3: Waiting for timeout to expire ---');
  await new Promise((r) => setTimeout(r, 2100));

  console.log('\n--- Phase 4: HALF_OPEN state, testing recovery ---');
  for (let i = 9; i <= 13; i++) {
    await makeRequest(`Request ${i}`);
    await new Promise((r) => setTimeout(r, 100));
  }

  console.log('\n--- Final Stats ---');
  console.log(JSON.stringify(breaker.getStats(), null, 2));

  breaker.destroy();
}
