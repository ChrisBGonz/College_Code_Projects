import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class SQLExceptionHandlerGenerator {

    public static void main(String[] args) {
        ChatLanguageModel model = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();

        String prompt = """
                Напишите метод на Java, который:
                1. Перехватывает SQLException
                2. Записывает в журнал (используя java.util.logging.Logger) сообщение об ошибке
                3. Выбрасывает пользовательское исключение DatabaseException
                
                Включите:
                - Определение класса DatabaseException
                - Метод executeQuery, который демонстрирует перехват SQLException
                - Правильную инициализацию Logger
                - Полный рабочий пример с импортами
                
                Предоставьте только код Java без объяснений.
                """;

        String response = model.generate(prompt);
        System.out.println(response);
    }
}
