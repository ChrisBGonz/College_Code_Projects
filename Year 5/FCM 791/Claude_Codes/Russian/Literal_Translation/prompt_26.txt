```yaml
name: Build and Upload Artifact

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  workflow_dispatch:

env:
  ARTIFACT_NAME: app-build
  BUILD_DIR: dist

jobs:
  build-and-upload:
    name: Build Project and Upload Artifact
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: npm test --if-present

      - name: Build project
        run: npm run build

      - name: Create artifact archive
        run: |
          mkdir -p artifacts
          TIMESTAMP=$(date +%Y%m%d_%H%M%S)
          SHORT_SHA=$(echo "${{ github.sha }}" | cut -c1-7)
          ARCHIVE_NAME="${{ env.ARTIFACT_NAME }}_${TIMESTAMP}_${SHORT_SHA}.tar.gz"
          tar -czf "artifacts/${ARCHIVE_NAME}" -C ${{ env.BUILD_DIR }} .
          echo "ARCHIVE_NAME=${ARCHIVE_NAME}" >> $GITHUB_ENV
          echo "Archive created: artifacts/${ARCHIVE_NAME}"
          ls -lh artifacts/

      - name: Upload artifact to GitHub Actions (temporary storage)
        uses: actions/upload-artifact@v4
        with:
          name: ${{ env.ARTIFACT_NAME }}
          path: artifacts/
          retention-days: 7

      # ─── AWS S3 ───────────────────────────────────────────────────────────────
      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ secrets.AWS_REGION }}

      - name: Upload artifact to AWS S3
        run: |
          aws s3 cp \
            "artifacts/${{ env.ARCHIVE_NAME }}" \
            "s3://${{ secrets.S3_BUCKET_NAME }}/builds/${{ github.ref_name }}/${{ env.ARCHIVE_NAME }}" \
            --storage-class STANDARD_IA \
            --metadata "commit=${{ github.sha }},branch=${{ github.ref_name }},run_id=${{ github.run_id }}"

          echo "Artifact uploaded to S3:"
          echo "s3://${{ secrets.S3_BUCKET_NAME }}/builds/${{ github.ref_name }}/${{ env.ARCHIVE_NAME }}"

      # ─── Google Cloud Storage (альтернатива AWS S3) ───────────────────────────

      #   uses: google-github-actions/auth@v2
      #   with:
      #     credentials_json: ${{ secrets.GCP_SERVICE_ACCOUNT_KEY }}

      # - name: Upload artifact to GCS
      #   uses: google-github-actions/upload-cloud-storage@v2
      #   with:
      #     path: artifacts/${{ env.ARCHIVE_NAME }}
      #     destination: ${{ secrets.GCS_BUCKET_NAME }}/builds/${{ github.ref_name }}
      #     headers: |-
      #       cache-control: no-cache

      # ─── Azure Blob Storage (альтернатива AWS S3) ─────────────────────────────
      # - name: Upload artifact to Azure Blob Storage
      #   uses: azure/CLI@v2
      #   with:
      #     azcliversion: latest
      #     inlineScript: |
      #       az storage blob upload \
      #         --account-name ${{ secrets.AZURE_STORAGE_ACCOUNT }} \
      #         --account-key ${{ secrets.AZURE_STORAGE_KEY }} \
      #         --container-name ${{ secrets.AZURE_CONTAINER_NAME }} \
      #         --name "builds/${{ github.ref_name }}/${{ env.ARCHIVE_NAME }}" \
      #         --file "artifacts/${{ env.ARCHIVE_NAME }}" \
      #         --overwrite

      - name: Verify upload
        run: |
          echo "Verifying artifact in S3..."
          aws s3 ls "s3://${{ secrets.S3_BUCKET_NAME }}/builds/${{ github.ref_name }}/" --human-readable

      - name: Print summary
        run: |
          echo "## Build Summary" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "| Parameter | Value |" >> $GITHUB_STEP_SUMMARY
          echo "|-----------|-------|" >> $GITHUB_STEP_SUMMARY
          echo "| **Branch** | \`${{ github.ref_name }}\` |" >> $GITHUB_STEP_SUMMARY
          echo "| **Commit** | \`${{ github.sha }}\` |" >> $GITHUB_STEP_SUMMARY
          echo "| **Run ID** | \`${{ github.run_id }}\` |" >> $GITHUB_STEP_SUMMARY
          echo "| **Artifact** | \`${{ env.ARCHIVE_NAME }}\` |" >> $GITHUB_STEP_SUMMARY
          echo "| **S3 Path** | \`s3://${{ secrets.S3_BUCKET_NAME }}/builds/${{ github.ref_name }}/${{ env.ARCHIVE_NAME }}\` |" >> $GITHUB_STEP_SUMMARY

  notify-on-failure:
    name: Notify on Failure
    runs-on: ubuntu-latest
    needs: build-and-upload
    if: failure()

    steps:
      - name: Send failure notification
        run: |
          echo "Build or upload failed!"
          echo "Repository: ${{ github.repository }}"
          echo "Branch: ${{ github.ref_name }}"
          echo "Commit: ${{ github.sha }}"
          echo "Run URL: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
```
