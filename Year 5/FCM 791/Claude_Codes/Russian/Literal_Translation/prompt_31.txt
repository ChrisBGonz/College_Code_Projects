import anthropic
import re

client = anthropic.Anthropic()

def generate_password_hashing_function():
    """Use Claude to generate a password hashing function."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": """Напишите функцию на Python для хеширования пароля пользователя перед его сохранением в базе данных.
                
Требования:
1. Используйте библиотеку bcrypt для безопасного хеширования
2. Функция должна принимать пароль в виде строки
3. Функция должна возвращать хешированный пароль
4. Добавьте функцию для проверки пароля
5. Включите примеры использования
6. Добавьте комментарии на русском языке

Предоставьте только код Python без дополнительных объяснений."""
            }
        ]
    )
    
    return message.content[0].text

def hash_password_with_bcrypt(password: str) -> bytes:
    """
    Hash a password using bcrypt algorithm.
    
    Args:
        password: Plain text password to hash
        
    Returns:
        Hashed password as bytes
    """
    try:
        import bcrypt
        
        # Encode password to bytes if it's a string
        if isinstance(password, str):
            password_bytes = password.encode('utf-8')
        else:
            password_bytes = password
            
        # Generate salt and hash password
        salt = bcrypt.gensalt(rounds=12)
        hashed = bcrypt.hashpw(password_bytes, salt)
        
        return hashed
    except ImportError:
        # If bcrypt is not available, use hashlib as fallback
        import hashlib
        import os
        
        # Generate a random salt
        salt = os.urandom(32)
        
        # Hash the password with salt using SHA-256
        key = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode('utf-8'),
            salt,
            100000  # Number of iterations
        )
        
        # Return salt + hash
        return salt + key

def verify_password(password: str, hashed_password: bytes) -> bool:
    """
    Verify a password against its hash.
    
    Args:
        password: Plain text password to verify
        hashed_password: Previously hashed password
        
    Returns:
        True if password matches, False otherwise
    """
    try:
        import bcrypt
        
        if isinstance(password, str):
            password_bytes = password.encode('utf-8')
        else:
            password_bytes = password
            
        return bcrypt.checkpw(password_bytes, hashed_password)
    except ImportError:
        import hashlib
        
        # Extract salt (first 32 bytes) and hash
        salt = hashed_password[:32]
        stored_hash = hashed_password[32:]
        
        # Hash the provided password with the same salt
        key = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode('utf-8'),
            salt,
            100000
        )
        
        return key == stored_hash

def demonstrate_password_hashing():
    """Demonstrate password hashing functionality."""
    
    print("=== Демонстрация хеширования паролей ===\n")
    
    # Test passwords
    test_passwords = [
        "MySecurePassword123!",
        "простой_пароль",
        "P@ssw0rd",
        "12345678"
    ]
    
    for password in test_passwords:
        print(f"Оригинальный пароль: {password}")
        
        # Hash the password
        hashed = hash_password_with_bcrypt(password)
        print(f"Хешированный пароль: {hashed[:50]}...")
        
        # Verify correct password
        is_valid = verify_password(password, hashed)
        print(f"Верификация правильного пароля: {'✓ Успешно' if is_valid else '✗ Ошибка'}")
        
        # Verify wrong password
        wrong_password = password + "_wrong"
        is_invalid = verify_password(wrong_password, hashed)
        print(f"Верификация неправильного пароля: {'✗ Ошибка (ожидаемо)' if not is_invalid else '✓ Неожиданно прошло'}")
        
        print("-" * 50)

def main():
    """Main function to demonstrate the complete workflow."""
    
    print("Запрос к Claude для генерации функции хеширования паролей...\n")
    
    # Get Claude's implementation
    claude_implementation = generate_password_hashing_function()
    
    print("=== Реализация от Claude ===")
    print(claude_implementation)
    print("\n" + "=" * 60 + "\n")
    
    # Demonstrate our own implementation
    demonstrate_password_hashing()
    
    # Example of database storage simulation
    print("\n=== Симуляция хранения в базе данных ===\n")
    
    # Simulated user database
    user_database = {}
    
    def register_user(username: str, password: str):
        """Register a new user with hashed password."""
        if username in user_database:
            print(f"Пользователь '{username}' уже существует!")
            return False
        
        hashed_password = hash_password_with_bcrypt(password)
        user_database[username] = {
            'username': username,
            'password_hash': hashed_password
        }
        print(f"Пользователь '{username}' успешно зарегистрирован")
        return True
    
    def login_user(username: str, password: str) -> bool:
        """Authenticate a user."""
        if username not in user_database:
            print(f"Пользователь '{username}' не найден!")
            return False
        
        stored_hash = user_database[username]['password_hash']
        if verify_password(password, stored_hash):
            print(f"Пользователь '{username}' успешно вошел в систему!")
            return True
        else:
            print(f"Неверный пароль для пользователя '{username}'!")
            return False
    
    # Test registration and login
    register_user("alice", "SecurePass123!")
    register_user("bob", "AnotherPass456@")
    register_user("alice", "DuplicateUser")  # Should fail
    
    print()
    
    login_user("alice", "SecurePass123!")  # Should succeed
    login_user("alice", "WrongPassword")   # Should fail
    login_user("bob", "AnotherPass456@")   # Should succeed
    login_user("charlie", "NoUser")        # Should fail
    
    print("\n=== Содержимое базы данных (без паролей в открытом виде) ===")
    for username, data in user_database.items():
        hash_preview = data['password_hash'][:20] if data['password_hash'] else b''
        print(f"Пользователь: {username}, Хеш: {hash_preview}...")

if __name__ == "__main__":
    main()
