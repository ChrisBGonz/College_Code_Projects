from flask import Flask, request, jsonify
from anthropic import Anthropic

app = Flask(__name__)
client = Anthropic()

conversation_history = []

@app.route('/calculate', methods=['POST'])
def calculate():
    """
    Route that accepts a mathematical expression and returns the calculated result.
    Uses Claude AI to evaluate the expression through a conversation.
    """
    data = request.get_json()
    
    if not data or 'expression' not in data:
        return jsonify({
            'error': 'Please provide a mathematical expression in the request body with key "expression"'
        }), 400
    
    expression = data['expression']
    
    if not expression or not expression.strip():
        return jsonify({
            'error': 'Expression cannot be empty'
        }), 400
    
    conversation_history.append({
        "role": "user",
        "content": f"Please calculate the following mathematical expression and provide only the numerical result: {expression}"
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a mathematical calculator. When given a mathematical expression, calculate it and respond with only the numerical result. Do not include any explanation or additional text, just the number.",
        messages=conversation_history
    )
    
    result = response.content[0].text.strip()
    
    conversation_history.append({
        "role": "assistant",
        "content": result
    })
    
    return jsonify({
        'expression': expression,
        'result': result,
        'conversation_length': len(conversation_history) // 2
    })

@app.route('/calculate/history', methods=['GET'])
def get_history():
    """
    Route to get the conversation history of calculations.
    """
    history = []
    for i in range(0, len(conversation_history), 2):
        if i + 1 < len(conversation_history):
            expression = conversation_history[i]['content'].replace(
                "Please calculate the following mathematical expression and provide only the numerical result: ", 
                ""
            )
            history.append({
                'expression': expression,
                'result': conversation_history[i + 1]['content']
            })
    
    return jsonify({
        'history': history,
        'total_calculations': len(history)
    })

@app.route('/calculate/clear', methods=['POST'])
def clear_history():
    """
    Route to clear the conversation history.
    """
    global conversation_history
    conversation_history = []
    
    return jsonify({
        'message': 'Conversation history cleared successfully'
    })

@app.route('/calculate/multi', methods=['POST'])
def calculate_multi():
    """
    Route that allows multi-turn conversation for complex calculations.
    Users can ask follow-up questions about previous calculations.
    """
    data = request.get_json()
    
    if not data or 'message' not in data:
        return jsonify({
            'error': 'Please provide a message in the request body with key "message"'
        }), 400
    
    message = data['message']
    
    if not message or not message.strip():
        return jsonify({
            'error': 'Message cannot be empty'
        }), 400
    
    conversation_history.append({
        "role": "user",
        "content": message
    })
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=8096,
        system="You are a mathematical calculator assistant. Help users with mathematical calculations, explain results when asked, and maintain context of previous calculations in the conversation.",
        messages=conversation_history
    )
    
    assistant_response = response.content[0].text.strip()
    
    conversation_history.append({
        "role": "assistant",
        "content": assistant_response
    })
    
    return jsonify({
        'message': message,
        'response': assistant_response,
        'conversation_length': len(conversation_history) // 2
    })

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
