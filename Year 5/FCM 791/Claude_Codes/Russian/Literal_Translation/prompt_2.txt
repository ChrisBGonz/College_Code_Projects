import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database of users
const usersDatabase = {
  user123: {
    id: "user123",
    email: "john.doe@example.com",
    displayName: "John Doe",
    createdAt: "2024-01-01T00:00:00Z",
  },
  user456: {
    id: "user456",
    email: "jane.smith@example.com",
    displayName: "Jane Smith",
    createdAt: "2024-01-15T00:00:00Z",
  },
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_email",
    description:
      "Validates if the provided email address is in correct format and not already taken by another user",
    input_schema: {
      type: "object",
      properties: {
        email: {
          type: "string",
          description: "The email address to validate",
        },
        userId: {
          type: "string",
          description: "The ID of the user trying to update their email",
        },
      },
      required: ["email", "userId"],
    },
  },
  {
    name: "validate_display_name",
    description:
      "Validates if the provided display name meets requirements (3-50 characters, alphanumeric and spaces only)",
    input_schema: {
      type: "object",
      properties: {
        displayName: {
          type: "string",
          description: "The display name to validate",
        },
      },
      required: ["displayName"],
    },
  },
  {
    name: "update_user_profile",
    description:
      "Updates the user profile in the database with new email and/or display name",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The ID of the user to update",
        },
        email: {
          type: "string",
          description: "New email address (optional)",
        },
        displayName: {
          type: "string",
          description: "New display name (optional)",
        },
      },
      required: ["userId"],
    },
  },
  {
    name: "get_user_profile",
    description: "Retrieves the current user profile from the database",
    input_schema: {
      type: "object",
      properties: {
        userId: {
          type: "string",
          description: "The ID of the user to retrieve",
        },
      },
      required: ["userId"],
    },
  },
];

// Tool implementations
function validateEmail(email, userId) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  if (!emailRegex.test(email)) {
    return { valid: false, error: "Invalid email format" };
  }

  // Check if email is already taken by another user
  const existingUser = Object.values(usersDatabase).find(
    (user) => user.email === email && user.id !== userId
  );

  if (existingUser) {
    return { valid: false, error: "Email is already taken by another user" };
  }

  return { valid: true, message: "Email is valid and available" };
}

function validateDisplayName(displayName) {
  if (displayName.length < 3 || displayName.length > 50) {
    return {
      valid: false,
      error: "Display name must be between 3 and 50 characters",
    };
  }

  const nameRegex = /^[a-zA-Z0-9\s]+$/;
  if (!nameRegex.test(displayName)) {
    return {
      valid: false,
      error: "Display name can only contain letters, numbers, and spaces",
    };
  }

  return { valid: true, message: "Display name is valid" };
}

function updateUserProfile(userId, email, displayName) {
  if (!usersDatabase[userId]) {
    return { success: false, error: "User not found" };
  }

  const updates = {};
  if (email) updates.email = email;
  if (displayName) updates.displayName = displayName;

  if (Object.keys(updates).length === 0) {
    return { success: false, error: "No updates provided" };
  }

  usersDatabase[userId] = {
    ...usersDatabase[userId],
    ...updates,
    updatedAt: new Date().toISOString(),
  };

  return {
    success: true,
    message: "Profile updated successfully",
    updatedFields: Object.keys(updates),
    profile: usersDatabase[userId],
  };
}

function getUserProfile(userId) {
  const user = usersDatabase[userId];
  if (!user) {
    return { found: false, error: "User not found" };
  }
  return { found: true, profile: user };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_email":
      return validateEmail(toolInput.email, toolInput.userId);
    case "validate_display_name":
      return validateDisplayName(toolInput.displayName);
    case "update_user_profile":
      return updateUserProfile(
        toolInput.userId,
        toolInput.email,
        toolInput.displayName
      );
    case "get_user_profile":
      return getUserProfile(toolInput.userId);
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main API endpoint handler
async function updateUserProfileEndpoint(userId, updateData) {
  console.log("\n=== API Request: /api/user/update-profile ===");
  console.log("User ID:", userId);
  console.log("Update Data:", updateData);

  const { email, displayName } = updateData;

  if (!email && !displayName) {
    return {
      success: false,
      error: "At least one field (email or displayName) must be provided",
    };
  }

  // Prepare the message for Claude
  const userMessage = `
    Process a profile update request for user ID: ${userId}
    
    Requested updates:
    ${email ? `- New email: ${email}` : ""}
    ${displayName ? `- New display name: ${displayName}` : ""}
    
    Please:
    1. First get the current user profile
    2. Validate any provided fields
    3. If validation passes, update the profile
    4. Return a clear success or error message
  `;

  const messages = [{ role: "user", content: userMessage }];

  let response;
  let finalResult = null;

  // Agentic loop
  while (true) {
    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: tools,
      messages: messages,
    });

    console.log("\n--- Claude Response ---");
    console.log("Stop Reason:", response.stop_reason);

    // Check if we're done
    if (response.stop_reason === "end_turn") {
      // Extract the final text response
      const textContent = response.content.find(
        (block) => block.type === "text"
      );
      if (textContent) {
        finalResult = {
          success: true,
          message: textContent.text,
          currentProfile: usersDatabase[userId],
        };
      }
      break;
    }

    // Process tool calls
    if (response.stop_reason === "tool_use") {
      const toolUseBlocks = response.content.filter(
        (block) => block.type === "tool_use"
      );

      // Add assistant's response to messages
      messages.push({ role: "assistant", content: response.content });

      // Process each tool call and collect results
      const toolResults = [];
      for (const toolUse of toolUseBlocks) {
        console.log(`\nTool Called: ${toolUse.name}`);
        console.log("Tool Input:", JSON.stringify(toolUse.input, null, 2));

        const result = processToolCall(toolUse.name, toolUse.input);
        console.log("Tool Result:", JSON.stringify(result, null, 2));

        toolResults.push({
          type: "tool_result",
          tool_use_id: toolUse.id,
          content: JSON.stringify(result),
        });
      }

      // Add tool results to messages
      messages.push({ role: "user", content: toolResults });
    } else {
      // Unexpected stop reason
      break;
    }
  }

  return finalResult || { success: false, error: "Failed to process request" };
}

// Test the endpoint
async function runTests() {
  console.log("Testing /api/user/update-profile endpoint\n");

  // Test 1: Update both email and display name

  console.log("\n" + "=".repeat(60) + "\n");

  // Test 4: Invalid email format
