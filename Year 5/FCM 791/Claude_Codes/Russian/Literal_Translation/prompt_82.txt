const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

const conversationHistory = [];

async function analyzeRequestWithClaude(requestData) {
  const userMessage = `Analyze this HTTP request for debugging purposes:
  
Method: ${requestData.method}
URL: ${requestData.url}
Headers: ${JSON.stringify(requestData.headers, null, 2)}
Query Parameters: ${JSON.stringify(requestData.query, null, 2)}
Body: ${JSON.stringify(requestData.body, null, 2)}
Timestamp: ${requestData.timestamp}
IP: ${requestData.ip}

Please provide:
1. A brief summary of what this request is doing
2. Any potential security concerns
3. Suggestions for debugging if there are issues`;

  conversationHistory.push({
    role: "user",
    content: userMessage,
  });

  const response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system:
      "You are a helpful debugging assistant that analyzes HTTP requests. Provide concise, technical analysis of incoming requests to help developers debug their applications.",
    messages: conversationHistory,
  });

  const assistantMessage = response.content[0].text;

  conversationHistory.push({
    role: "assistant",
    content: assistantMessage,
  });

  return assistantMessage;
}

function createLoggingMiddleware(options = {}) {
  const {
    enableAIAnalysis = false,
    logLevel = "info",
    excludePaths = [],
    maxBodySize = 10000,
  } = options;

  return async function loggingMiddleware(req, res, next) {
    const startTime = Date.now();

    if (excludePaths.some((path) => req.url.startsWith(path))) {
      return next();
    }

    let body = {};
    if (req.body) {
      body = req.body;
    }

    const requestData = {
      method: req.method,
      url: req.url,
      headers: req.headers,
      query: req.query || {},
      body: body,
      timestamp: new Date().toISOString(),
      ip: req.ip || req.connection?.remoteAddress || "unknown",
    };

    const bodyStr = JSON.stringify(body);
    const truncatedBody =
      bodyStr.length > maxBodySize
        ? bodyStr.substring(0, maxBodySize) + "... [truncated]"
        : bodyStr;

    console.log("\n" + "=".repeat(60));
    console.log(`[${logLevel.toUpperCase()}] Incoming Request`);
    console.log("=".repeat(60));
    console.log(`Timestamp: ${requestData.timestamp}`);
    console.log(`Method: ${requestData.method}`);
    console.log(`URL: ${requestData.url}`);
    console.log(`IP: ${requestData.ip}`);
    console.log("\nHeaders:");
    console.log(JSON.stringify(requestData.headers, null, 2));

    if (Object.keys(requestData.query).length > 0) {
      console.log("\nQuery Parameters:");
      console.log(JSON.stringify(requestData.query, null, 2));
    }

    if (Object.keys(body).length > 0) {
      console.log("\nRequest Body:");
      console.log(truncatedBody);
    }

    if (enableAIAnalysis) {
      try {
        console.log("\n[AI Analysis]");
        const analysis = await analyzeRequestWithClaude(requestData);
        console.log(analysis);
      } catch (error) {
        console.error("Failed to get AI analysis:", error.message);
      }
    }

    const originalEnd = res.end;
    res.end = function (chunk, encoding) {
      const duration = Date.now() - startTime;
      console.log("\n" + "-".repeat(60));
      console.log(`Response Status: ${res.statusCode}`);
      console.log(`Duration: ${duration}ms`);
      console.log("=".repeat(60) + "\n");

      res.end = originalEnd;
      return res.end(chunk, encoding);
    };

    next();
  };
}

async function interactiveDebugSession() {
  const readline = require("readline");
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  console.log("Interactive Debug Session Started");
  console.log(
    "You can ask questions about logged requests or get debugging help."
  );
  console.log('Type "exit" to quit.\n');

  const askQuestion = () => {
    rl.question("Debug Assistant: ", async (input) => {
      if (input.toLowerCase() === "exit") {
        console.log("Ending debug session...");
        rl.close();
        return;
      }

      if (!input.trim()) {
        askQuestion();
        return;
      }

      conversationHistory.push({
        role: "user",
        content: input,
      });

      try {
        const response = await client.messages.create({
          model: "claude-opus-4-5",
          max_tokens: 8096,
          system:
            "You are a helpful debugging assistant. Help developers understand and debug HTTP requests and middleware issues.",
          messages: conversationHistory,
        });

        const assistantMessage = response.content[0].text;

        conversationHistory.push({
          role: "assistant",
          content: assistantMessage,
        });

        console.log("\nAssistant:", assistantMessage, "\n");
      } catch (error) {
        console.error("Error:", error.message);
      }

      askQuestion();
    });
  };

  askQuestion();
}

async function demonstrateMiddleware() {
  console.log("Node.js Request Logging Middleware with AI Analysis");
  console.log("=".repeat(50));

  const middleware = createLoggingMiddleware({
    enableAIAnalysis: true,
    logLevel: "debug",
    excludePaths: ["/health", "/favicon.ico"],
    maxBodySize: 5000,
  });

  const mockRequests = [
    {
      method: "POST",
      url: "/api/users",
      headers: {
        "content-type": "application/json",
        authorization: "Bearer token123",
        "user-agent": "Mozilla/5.0",
      },
      query: {},
      body: {
        username: "john_doe",
        email: "john@example.com",
        password: "secret123",
      },
      ip: "192.168.1.1",
    },
    {
      method: "GET",
      url: "/api/products",
      headers: {
        accept: "application/json",
        "user-agent": "PostmanRuntime/7.28.0",
      },
      query: { page: "1", limit: "10", category: "electronics" },
      body: {},
      ip: "10.0.0.1",
    },
  ];

  for (const mockReq of mockRequests) {
    const req = {
      ...mockReq,
      connection: { remoteAddress: mockReq.ip },
    };

    const res = {
      statusCode: 200,
      end: function () {},
    };

    const next = () => {};

    await middleware(req, res, next);

    res.end();

    await new Promise((resolve) => setTimeout(resolve, 1000));
  }

  console.log("\n" + "=".repeat(50));
  console.log("Starting interactive debug session...");
  console.log("=".repeat(50) + "\n");

  await interactiveDebugSession();
}

demonstrateMiddleware().catch(console.error);
