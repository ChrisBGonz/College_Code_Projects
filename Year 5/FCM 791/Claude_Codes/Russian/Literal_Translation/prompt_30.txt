std::string generate_image_processing_code() {
    anthropic::Client client;
    
    std::string prompt = R"(
Generate a complete C++ function that links with libjpeg external library for image processing.
The code should include:
1. Proper includes for libjpeg (jpeglib.h, jerror.h)
2. A function to read a JPEG image from file
3. A function to write a JPEG image to file
4. A function to perform basic image processing (e.g., convert to grayscale)
5. Proper error handling using libjpeg's error management
6. Memory management
7. A main function demonstrating usage

The code should be production-ready with proper comments explaining each part.
Return only the C++ code without any markdown formatting or explanations.
)";

    auto message = client.messages().create({
        .model = "claude-opus-4-5",
        .max_tokens = 4096,
        .messages = {
            {
                .role = "user",
                .content = prompt
            }
        }
    });
    
    std::string result;
    for (const auto& block : message.content) {
        if (block.type == "text") {
            result += block.text;
        }
    }
    
    return result;
}
