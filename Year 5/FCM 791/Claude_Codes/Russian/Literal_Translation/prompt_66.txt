package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
	"time"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

type LoginRequest struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

type LoginResponse struct {
	Token   string `json:"token,omitempty"`
	Message string `json:"message"`
	Error   string `json:"error,omitempty"`
}

type JWTGenerationResult struct {
	Token     string
	ExpiresAt time.Time
	UserID    string
	Username  string
}

func generateJWTWithClaude(username string, userID string) (*JWTGenerationResult, error) {
	apiKey := os.Getenv("ANTHROPIC_API_KEY")
	if apiKey == "" {
		return nil, fmt.Errorf("ANTHROPIC_API_KEY environment variable not set")
	}

	client := anthropic.NewClient(option.WithAPIKey(apiKey))

	expiresAt := time.Now().Add(24 * time.Hour)
	issuedAt := time.Now()

	prompt := fmt.Sprintf(`Generate a JWT (JSON Web Token) for the following user:
Username: %s
UserID: %s
IssuedAt: %d
ExpiresAt: %d

Please create a JWT with the following structure:
1. Header: {"alg": "HS256", "typ": "JWT"}
2. Payload with claims:
   - sub (subject): user ID
   - username: the username
   - iat (issued at): Unix timestamp
   - exp (expiration): Unix timestamp (24 hours from now)
   - iss (issuer): "myapp"
   - role: "user"

For demonstration purposes, use "secret_key_demo" as the signing key.
Return ONLY the JWT token string, nothing else. The format should be: header.payload.signature
Use base64url encoding for header and payload, and create a proper HMAC-SHA256 signature.`, username, userID, issuedAt.Unix(), expiresAt.Unix())

	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Messages: []anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		},
	})

	if err != nil {
		return nil, fmt.Errorf("failed to call Claude API: %w", err)
	}

	var token string
	for _, block := range message.Content {
		if block.Type == "text" {
			token = block.Text
			break
		}
	}

	if token == "" {
		return nil, fmt.Errorf("no token generated by Claude")
	}

	// Clean up the token (remove any whitespace or newlines)
	for len(token) > 0 && (token[0] == ' ' || token[0] == '\n' || token[0] == '\r' || token[0] == '\t') {
		token = token[1:]
	}
	for len(token) > 0 && (token[len(token)-1] == ' ' || token[len(token)-1] == '\n' || token[len(token)-1] == '\r' || token[len(token)-1] == '\t') {
		token = token[:len(token)-1]
	}

	return &JWTGenerationResult{
		Token:     token,
		ExpiresAt: expiresAt,
		UserID:    userID,
		Username:  username,
	}, nil
}

func authenticateUser(username, password string) (string, bool) {
	// Demo user database
	users := map[string]struct {
		password string
		userID   string
	}{
		"admin": {password: "admin123", userID: "user_001"},
		"john":  {password: "john456", userID: "user_002"},
		"jane":  {password: "jane789", userID: "user_003"},
	}

	if user, exists := users[username]; exists {
		if user.password == password {
			return user.userID, true
		}
	}
	return "", false
}

func loginHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	var loginReq LoginRequest
	if err := json.NewDecoder(r.Body).Decode(&loginReq); err != nil {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusBadRequest)
		json.NewEncoder(w).Encode(LoginResponse{
			Error:   "Invalid request body",
			Message: "Failed to parse login request",
		})
		return
	}

	userID, authenticated := authenticateUser(loginReq.Username, loginReq.Password)
	if !authenticated {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusUnauthorized)
		json.NewEncoder(w).Encode(LoginResponse{
			Error:   "Invalid credentials",
			Message: "Authentication failed",
		})
		return
	}

	log.Printf("User %s authenticated successfully, generating JWT...", loginReq.Username)

	jwtResult, err := generateJWTWithClaude(loginReq.Username, userID)
	if err != nil {
		log.Printf("Failed to generate JWT: %v", err)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusInternalServerError)
		json.NewEncoder(w).Encode(LoginResponse{
			Error:   "Token generation failed",
			Message: fmt.Sprintf("Failed to generate JWT: %v", err),
		})
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	json.NewEncoder(w).Encode(LoginResponse{
		Token:   jwtResult.Token,
		Message: fmt.Sprintf("Login successful. Token expires at: %s", jwtResult.ExpiresAt.Format(time.RFC3339)),
	})
}

func demoGenerateJWT() {
	fmt.Println("=== JWT Generation Demo ===")
	fmt.Println("Generating JWT for demo user...")

	result, err := generateJWTWithClaude("demo_user", "user_demo_001")
	if err != nil {
		log.Fatalf("Failed to generate JWT: %v", err)
	}

	fmt.Printf("\nGenerated JWT Token:\n%s\n", result.Token)
	fmt.Printf("\nToken Details:\n")
	fmt.Printf("  Username: %s\n", result.Username)
	fmt.Printf("  UserID: %s\n", result.UserID)
	fmt.Printf("  Expires At: %s\n", result.ExpiresAt.Format(time.RFC3339))
}

func main() {
	// Check if we should run in demo mode or server mode
	if len(os.Args) > 1 && os.Args[1] == "demo" {
		demoGenerateJWT()
		return
	}

	// Start HTTP server
	http.HandleFunc("/login", loginHandler)

	port := ":8080"
	fmt.Printf("JWT Authentication Server starting on port %s\n", port)
	fmt.Println("\nAvailable endpoints:")
	fmt.Println("  POST /login - Authenticate user and get JWT token")
	fmt.Println("\nDemo users:")
	fmt.Println("  admin / admin123")
	fmt.Println("  john / john456")
	fmt.Println("  jane / jane789")
	fmt.Println("\nExample request:")
	fmt.Println(`  curl -X POST http://localhost:8080/login \`)
	fmt.Println(`    -H "Content-Type: application/json" \`)
	fmt.Println(`    -d '{"username": "admin", "password": "admin123"}'`)

	if err := http.ListenAndServe(port, nil); err != nil {
		log.Fatalf("Failed to start server: %v", err)
	}
}
