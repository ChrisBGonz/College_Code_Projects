package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func main() {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	tools := []anthropic.ToolParam{
		{
			Name:        "bank_transfer",
			Description: anthropic.String("Performs a bank transfer between two accounts and logs the transaction details"),
			InputSchema: anthropic.ToolInputSchemaParam{
				Type: "object",
				Properties: map[string]interface{}{
					"sender_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the sender's bank account",
					},
					"receiver_id": map[string]interface{}{
						"type":        "string",
						"description": "The ID of the receiver's bank account",
					},
					"amount": map[string]interface{}{
						"type":        "number",
						"description": "The amount to transfer in the account's currency",
					},
				},
				Required: []string{"sender_id", "receiver_id", "amount"},
			},
		},
	}

	messages := []anthropic.MessageParam{
		{
			Role: anthropic.MessageParamRoleUser,
			Content: []anthropic.ContentBlockParamUnion{
				anthropic.NewTextBlock("Please transfer $500 from account ACC001 to account ACC002"),
			},
		},
	}

	fmt.Println("Sending request to Claude...")

	response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.ModelClaude3_5SonnetLatest,
		MaxTokens: 1024,
		Tools:     tools,
		Messages:  messages,
	})
	if err != nil {
		log.Fatalf("Error creating message: %v", err)
	}

	fmt.Printf("Stop reason: %s\n", response.StopReason)

	for _, block := range response.Content {
		switch b := block.AsUnion().(type) {
		case anthropic.TextBlock:
			fmt.Printf("Claude says: %s\n", b.Text)
		case anthropic.ToolUseBlock:
			if b.Name == "bank_transfer" {
				input := b.Input
				senderID, _ := input["sender_id"].(string)
				receiverID, _ := input["receiver_id"].(string)
				amount, _ := input["amount"].(float64)

				result := bankTransfer(senderID, receiverID, amount)
				fmt.Printf("Transfer result: %s\n", result)

				messages = append(messages, anthropic.MessageParam{
					Role: anthropic.MessageParamRoleAssistant,
					Content: []anthropic.ContentBlockParamUnion{
						anthropic.NewToolUseBlock(b.ID, b.Name, b.Input),
					},
				})

				messages = append(messages, anthropic.MessageParam{
					Role: anthropic.MessageParamRoleUser,
					Content: []anthropic.ContentBlockParamUnion{
						anthropic.NewToolResultBlock(b.ID, result, false),
					},
				})

				finalResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
					Model:     anthropic.ModelClaude3_5SonnetLatest,
					MaxTokens: 1024,
					Tools:     tools,
					Messages:  messages,
				})
				if err != nil {
					log.Fatalf("Error creating final message: %v", err)
				}

				for _, finalBlock := range finalResponse.Content {
					if textBlock, ok := finalBlock.AsUnion().(anthropic.TextBlock); ok {
						fmt.Printf("Claude's final response: %s\n", textBlock.Text)
					}
				}
			}
		}
	}
}

func bankTransfer(senderID, receiverID string, amount float64) string {
	logger := log.New(os.Stdout, "[BANK TRANSFER] ", log.LstdFlags)

	logger.Printf("Transaction initiated - Sender ID: %s, Receiver ID: %s, Amount: %.2f",
		senderID, receiverID, amount)

	if amount <= 0 {
		errorMsg := fmt.Sprintf("Transfer failed: Invalid amount %.2f", amount)
		logger.Printf("ERROR: %s", errorMsg)
		return errorMsg
	}

	if senderID == "" || receiverID == "" {
		errorMsg := "Transfer failed: Invalid sender or receiver ID"
		logger.Printf("ERROR: %s", errorMsg)
		return errorMsg
	}

	successMsg := fmt.Sprintf("Transfer successful: %.2f transferred from %s to %s",
		amount, senderID, receiverID)
	logger.Printf("SUCCESS: %s", successMsg)

	return successMsg
}
