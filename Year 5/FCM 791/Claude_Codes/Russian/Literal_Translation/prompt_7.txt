import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

// Simulated database
const database = {
  users: [
    {
      id: 1,
      username: "john_doe",
      password: "old_password_123",
      email: "john@example.com",
    },
    {
      id: 2,
      username: "jane_smith",
      password: "secure_pass_456",
      email: "jane@example.com",
    },
    {
      id: 3,
      username: "bob_wilson",
      password: "mypassword789",
      email: "bob@example.com",
    },
  ],
};

// Tool definitions for Claude
const tools = [
  {
    name: "validate_user",
    description:
      "Validates if a user exists in the database by their user_id. Returns user information if found.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "number",
          description: "The unique identifier of the user",
        },
      },
      required: ["user_id"],
    },
  },
  {
    name: "validate_password_strength",
    description:
      "Validates if the new password meets security requirements (minimum 8 characters, contains uppercase, lowercase, number, and special character)",
    input_schema: {
      type: "object",
      properties: {
        password: {
          type: "string",
          description: "The new password to validate",
        },
      },
      required: ["password"],
    },
  },
  {
    name: "update_password",
    description:
      "Updates the password for a specific user in the database. Should only be called after validation.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "number",
          description: "The unique identifier of the user",
        },
        new_password: {
          type: "string",
          description: "The new password to set for the user",
        },
      },
      required: ["user_id", "new_password"],
    },
  },
  {
    name: "log_password_reset",
    description:
      "Logs the password reset event for security audit purposes. Should be called after successful password update.",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "number",
          description: "The unique identifier of the user",
        },
        success: {
          type: "boolean",
          description: "Whether the password reset was successful",
        },
        reason: {
          type: "string",
          description: "Reason for success or failure",
        },
      },
      required: ["user_id", "success", "reason"],
    },
  },
];

// Tool implementations
function validateUser(userId) {
  const user = database.users.find((u) => u.id === userId);
  if (user) {
    return {
      found: true,
      user_id: user.id,
      username: user.username,
      email: user.email,
    };
  }
  return { found: false, message: `User with ID ${userId} not found` };
}

function validatePasswordStrength(password) {
  const minLength = password.length >= 8;
  const hasUppercase = /[A-Z]/.test(password);
  const hasLowercase = /[a-z]/.test(password);
  const hasNumber = /\d/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);

  const isValid =
    minLength && hasUppercase && hasLowercase && hasNumber && hasSpecialChar;

  const requirements = {
    minLength: { met: minLength, description: "At least 8 characters" },
    hasUppercase: {
      met: hasUppercase,
      description: "Contains uppercase letter",
    },
    hasLowercase: {
      met: hasLowercase,
      description: "Contains lowercase letter",
    },
    hasNumber: { met: hasNumber, description: "Contains number" },
    hasSpecialChar: {
      met: hasSpecialChar,
      description: "Contains special character",
    },
  };

  return {
    isValid,
    requirements,
    message: isValid
      ? "Password meets all security requirements"
      : "Password does not meet security requirements",
  };
}

function updatePassword(userId, newPassword) {
  const userIndex = database.users.findIndex((u) => u.id === userId);
  if (userIndex === -1) {
    return { success: false, message: `User with ID ${userId} not found` };
  }

  // In a real application, you would hash the password before storing
  const oldPassword = database.users[userIndex].password;
  database.users[userIndex].password = newPassword;

  return {
    success: true,
    message: `Password successfully updated for user ${userId}`,
    user_id: userId,
    username: database.users[userIndex].username,
    timestamp: new Date().toISOString(),
  };
}

function logPasswordReset(userId, success, reason) {
  const logEntry = {
    timestamp: new Date().toISOString(),
    user_id: userId,
    event: "PASSWORD_RESET",
    success,
    reason,
    ip_address: "127.0.0.1", // In real app, this would be the actual IP
  };

  console.log("\n=== SECURITY AUDIT LOG ===");
  console.log(JSON.stringify(logEntry, null, 2));
  console.log("==========================\n");

  return { logged: true, log_entry: logEntry };
}

// Process tool calls
function processToolCall(toolName, toolInput) {
  switch (toolName) {
    case "validate_user":
      return validateUser(toolInput.user_id);
    case "validate_password_strength":
      return validatePasswordStrength(toolInput.password);
    case "update_password":
      return updatePassword(toolInput.user_id, toolInput.new_password);
    case "log_password_reset":
      return logPasswordReset(
        toolInput.user_id,
        toolInput.success,
        toolInput.reason
      );
    default:
      return { error: `Unknown tool: ${toolName}` };
  }
}

// Main password reset function
async function resetPassword(userId, newPassword) {
  console.log(`\nInitiating password reset for user ID: ${userId}`);
  console.log("=".repeat(50));

  const messages = [
    {
      role: "user",
      content: `Please reset the password for user with ID ${userId}. The new password should be: "${newPassword}". 
      
      Please follow these steps:
      1. Validate that the user exists
      2. Validate the password strength
      3. If both validations pass, update the password
      4. Log the password reset event (success or failure)
      5. Provide a clear summary of what happened`,
    },
  ];

  let response = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 4096,
    tools,
    messages,
  });

  console.log("\nClaude is processing the password reset request...\n");

  // Agentic loop
  while (response.stop_reason === "tool_use") {
    const toolUseBlocks = response.content.filter(
      (block) => block.type === "tool_use"
    );

    const toolResults = [];

    for (const toolUse of toolUseBlocks) {
      console.log(`Executing tool: ${toolUse.name}`);
      console.log("Input:", JSON.stringify(toolUse.input, null, 2));

      const result = processToolCall(toolUse.name, toolUse.input);
      console.log("Result:", JSON.stringify(result, null, 2));
      console.log("-".repeat(30));

      toolResults.push({
        type: "tool_result",
        tool_use_id: toolUse.id,
        content: JSON.stringify(result),
      });
    }

    // Continue the conversation with tool results
    messages.push({ role: "assistant", content: response.content });
    messages.push({ role: "user", content: toolResults });

    response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 4096,
      tools,
      messages,
    });
  }

  // Extract final text response
  const finalResponse = response.content
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n");

  console.log("\n=== FINAL RESPONSE ===");
  console.log(finalResponse);
  console.log("=".repeat(50));

  return finalResponse;
}

// Test the function with different scenarios
async function runTests() {
  console.log("PASSWORD RESET SYSTEM - TEST SCENARIOS");
  console.log("=".repeat(50));

  // Test 1: Valid user with strong password
  console.log("\nTest 1: Valid user with strong password");
  await resetPassword(1, "NewSecure@Pass123");

  // Test 2: Valid user with weak password
  console.log("\nTest 2: Valid user with weak password");
  await resetPassword(2, "weakpass");

  // Test 3: Invalid user
  console.log("\nTest 3: Non-existent user");
  await resetPassword(999, "ValidPass@123");
}

// Run the tests
runTests().catch(console.error);
