import Anthropic from "@anthropic-ai/sdk";
import * as http from "http";
import * as url from "url";

const client = new Anthropic();

// Simple in-memory session store
const sessions: Map<string, { userId: string; username: string; createdAt: Date }> = new Map();

// Generate a simple session ID
function generateSessionId(): string {
  return Math.random().toString(36).substring(2) + Date.now().toString(36);
}

// Parse cookies from request
function parseCookies(cookieHeader: string | undefined): Record<string, string> {
  const cookies: Record<string, string> = {};
  if (!cookieHeader) return cookies;
  
  cookieHeader.split(';').forEach(cookie => {
    const [name, value] = cookie.trim().split('=');
    if (name && value) {
      cookies[name.trim()] = value.trim();
    }
  });
  
  return cookies;
}

// Handle logout with Claude AI assistance
async function handleLogoutWithAI(sessionId: string | undefined): Promise<string> {
  const conversationHistory: Array<{ role: "user" | "assistant"; content: string }> = [];
  
  // First message: describe the logout scenario
  const userMessage = sessionId 
    ? `I need to process a logout request. The user has session ID: ${sessionId}. Please help me handle this logout properly.`
    : `I need to process a logout request, but no session ID was provided. Please help me handle this case.`;
  
  conversationHistory.push({
    role: "user",
    content: userMessage
  });
  
  // First API call
  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: "You are a helpful assistant that guides through logout processes. Provide clear, step-by-step guidance for handling user logout operations securely.",
    messages: conversationHistory
  });
  
  const firstAssistantMessage = firstResponse.content[0].type === 'text' 
    ? firstResponse.content[0].text 
    : '';
  
  conversationHistory.push({
    role: "assistant",
    content: firstAssistantMessage
  });
  
  // Second message: ask about security considerations
  conversationHistory.push({
    role: "user",
    content: "What security measures should I implement when deleting the session?"
  });
  
  // Second API call
  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: "You are a helpful assistant that guides through logout processes. Provide clear, step-by-step guidance for handling user logout operations securely.",
    messages: conversationHistory
  });
  
  const secondAssistantMessage = secondResponse.content[0].type === 'text'
    ? secondResponse.content[0].text
    : '';
  
  conversationHistory.push({
    role: "assistant",
    content: secondAssistantMessage
  });
  
  // Third message: confirm the action
  conversationHistory.push({
    role: "user",
    content: "Please confirm the logout process is complete and provide a summary."
  });
  
  // Third API call
  const thirdResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 8096,
    system: "You are a helpful assistant that guides through logout processes. Provide clear, step-by-step guidance for handling user logout operations securely.",
    messages: conversationHistory
  });
  
  const finalMessage = thirdResponse.content[0].type === 'text'
    ? thirdResponse.content[0].text
    : '';
  
  return finalMessage;
}

// Main logout handler function
async function handleLogout(
  req: http.IncomingMessage,
  res: http.ServerResponse
): Promise<void> {
  try {
    // Parse cookies to get session ID
    const cookies = parseCookies(req.headers.cookie);
    const sessionId = cookies['sessionId'];
    
    console.log(`Processing logout request. Session ID: ${sessionId || 'not found'}`);
    
    // Get AI guidance for the logout process
    const aiGuidance = await handleLogoutWithAI(sessionId);
    
    let logoutResult = {
      success: false,
      message: '',
      aiGuidance: aiGuidance
    };
    
    if (sessionId && sessions.has(sessionId)) {
      // Get session info before deleting
      const sessionInfo = sessions.get(sessionId);
      
      // Delete the session from the store
      sessions.delete(sessionId);
      
      console.log(`Session ${sessionId} deleted for user: ${sessionInfo?.username}`);
      
      logoutResult = {
        success: true,
        message: `User ${sessionInfo?.username} successfully logged out. Session has been terminated.`,
        aiGuidance: aiGuidance
      };
      
      // Clear the session cookie
      res.setHeader('Set-Cookie', 'sessionId=; HttpOnly; Secure; SameSite=Strict; Max-Age=0; Path=/');
    } else if (sessionId) {
      logoutResult = {
        success: false,
        message: 'Session not found or already expired.',
        aiGuidance: aiGuidance
      };
    } else {
      logoutResult = {
        success: false,
        message: 'No active session found.',
        aiGuidance: aiGuidance
      };
    }
    
    // Send response
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(logoutResult, null, 2));
    
  } catch (error) {
    console.error('Error during logout:', error);
    res.writeHead(500, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ 
      success: false, 
      message: 'Internal server error during logout',
      error: error instanceof Error ? error.message : 'Unknown error'
    }));
  }
}

// Helper function to create a test session
function createTestSession(username: string): string {
  const sessionId = generateSessionId();
  sessions.set(sessionId, {
    userId: Math.random().toString(36).substring(2),
    username: username,
    createdAt: new Date()
  });
  return sessionId;
}

// Create HTTP server for testing
const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url || '', true);
  const pathname = parsedUrl.pathname;
  
  if (pathname === '/logout' && req.method === 'POST') {
    await handleLogout(req, res);
  } else if (pathname === '/login' && req.method === 'POST') {
    // Simple login for testing purposes
    const username = parsedUrl.query.username as string || 'testuser';
    const sessionId = createTestSession(username);
    
    res.setHeader('Set-Cookie', `sessionId=${sessionId}; HttpOnly; Secure; SameSite=Strict; Path=/`);
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ 
      success: true, 
      message: `Logged in as ${username}`,
      sessionId: sessionId
    }));
  } else if (pathname === '/sessions' && req.method === 'GET') {
    // Debug endpoint to view active sessions
    const sessionList = Array.from(sessions.entries()).map(([id, data]) => ({
      sessionId: id,
      ...data
    }));
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ activeSessions: sessionList }));
  } else {
    res.writeHead(404, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ message: 'Not found' }));
  }
});

// Start server
const PORT = 3000;
server.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log('\nAvailable endpoints:');
  console.log('  POST /login?username=<name> - Create a test session');
  console.log('  POST /logout - Logout and delete session');
  console.log('  GET /sessions - View active sessions (debug)');
  
  // Demo: Create a test session and then logout
  runDemo();
});
