import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.bedrockruntime.BedrockRuntimeClient;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelRequest;
import software.amazon.awssdk.services.bedrockruntime.model.InvokeModelResponse;
import software.amazon.awssdk.core.SdkBytes;

import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;
import java.io.IOException;
import java.io.StringReader;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import com.fasterxml.jackson.databind.JsonNode;

public class SecureXMLParser {

    private static final BedrockRuntimeClient bedrockClient = BedrockRuntimeClient.builder()
            .region(Region.US_EAST_1)
            .credentialsProvider(DefaultCredentialsProvider.create())
            .build();

    private static final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * Securely parses an XML string from an untrusted source using DocumentBuilderFactory
     * with XXE (XML External Entity) protection enabled.
     *
     * @param xmlString The XML string to parse from an untrusted source
     * @return Parsed Document object
     * @throws ParserConfigurationException if parser configuration fails
     * @throws SAXException                 if XML parsing fails
     * @throws IOException                  if I/O error occurs
     */
    public static Document parseXMLSecurely(String xmlString)
            throws ParserConfigurationException, SAXException, IOException {

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();

        // Disable DOCTYPE declarations to prevent XXE attacks
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);

        // Disable external general entities
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);

        // Disable external parameter entities
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);

        // Disable loading of external DTD
        factory.setFeature("http://apache.org/xml/features/nonvalidating/load-external-dtd", false);

        // Disable XInclude processing
        factory.setXIncludeAware(false);

        // Disable entity expansion
        factory.setExpandEntityReferences(false);

        DocumentBuilder builder = factory.newDocumentBuilder();

        // Parse the XML string
        InputSource inputSource = new InputSource(new StringReader(xmlString));
        return builder.parse(inputSource);
    }

    /**
     * Analyzes XML content using Claude AI to provide security insights
     *
     * @param xmlString The XML string to analyze
     * @return AI analysis of the XML content
     */
    public static String analyzeXMLWithClaude(String xmlString) {
        String prompt = String.format(
                "Analyze the following XML content from an untrusted source. " +
                        "Identify any potential security issues, malicious patterns, or suspicious elements. " +
                        "Provide a brief security assessment:\n\n%s",
                xmlString);

        try {
            ObjectNode requestBody = objectMapper.createObjectNode();
            requestBody.put("anthropic_version", "bedrock-2023-05-31");
            requestBody.put("max_tokens", 1024);

            ObjectNode message = objectMapper.createObjectNode();
            message.put("role", "user");
            message.put("content", prompt);

            requestBody.set("messages", objectMapper.createArrayNode().add(message));

            String requestBodyJson = objectMapper.writeValueAsString(requestBody);

            InvokeModelRequest request = InvokeModelRequest.builder()
                    .modelId("anthropic.claude-3-5-sonnet-20241022-v2:0")
                    .body(SdkBytes.fromUtf8String(requestBodyJson))
                    .build();

            InvokeModelResponse response = bedrockClient.invokeModel(request);
            String responseBody = response.body().asUtf8String();

            JsonNode responseJson = objectMapper.readTree(responseBody);
            return responseJson.path("content").get(0).path("text").asText();

        } catch (Exception e) {
            return "Error analyzing XML: " + e.getMessage();
        }
    }

    /**
     * Demonstrates secure XML parsing with AI-powered analysis
     */
    public static void demonstrateSecureXMLParsing() {
        // Example XML strings from untrusted sources
        String[] testXMLStrings = {
                // Normal XML
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                        "<users><user><name>John Doe</name><email>john@example.com</email></user></users>",

                // XML with potential XXE attempt (will be blocked by secure parser)
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                        "<data><item>Normal content</item></data>",

                // XML with nested elements
                "<?xml version=\"1.0\" encoding=\"UTF-8\"?>" +
                        "<catalog><book id=\"1\"><title>Java Security</title><author>Expert</author>" +
                        "<price>49.99</price></book></catalog>"
        };

        for (int i = 0; i < testXMLStrings.length; i++) {
            System.out.println("\n=== Processing XML String " + (i + 1) + " ===");
            System.out.println("Input XML: " + testXMLStrings[i]);

            try {
                // Parse XML securely
                Document document = parseXMLSecurely(testXMLStrings[i]);
                Element rootElement = document.getDocumentElement();
                System.out.println("Successfully parsed XML. Root element: " + rootElement.getTagName());

                // Get all elements
                NodeList allElements = document.getElementsByTagName("*");
                System.out.println("Total elements found: " + allElements.getLength());

                // Analyze with Claude AI
                System.out.println("\nAnalyzing with Claude AI...");
                String analysis = analyzeXMLWithClaude(testXMLStrings[i]);
                System.out.println("AI Security Analysis: " + analysis);

            } catch (ParserConfigurationException e) {
                System.err.println("Parser configuration error: " + e.getMessage());
            } catch (SAXException e) {
                System.err.println("XML parsing error (possibly blocked malicious content): " + e.getMessage());
            } catch (IOException e) {
                System.err.println("I/O error: " + e.getMessage());
            }
        }
    }

    public static void main(String[] args) {
        System.out.println("Secure XML Parser with AI Analysis");
        System.out.println("===================================");
        System.out.println("This implementation uses DocumentBuilderFactory with XXE protection");
        System.out.println("and Claude AI for security analysis of untrusted XML content.\n");

        demonstrateSecureXMLParsing();
    }
}
