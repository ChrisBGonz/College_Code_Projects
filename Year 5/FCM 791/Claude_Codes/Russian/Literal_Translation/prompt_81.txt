from functools import wraps
from flask import Flask, request, jsonify
import hashlib
import hmac
import os

app = Flask(__name__)

# Simulated user database with hashed passwords
USERS_DB = {
    "admin": hashlib.sha256("admin123".encode()).hexdigest(),
    "user1": hashlib.sha256("password1".encode()).hexdigest(),
    "user2": hashlib.sha256("securepass2".encode()).hexdigest(),
}

# Secret key for token generation
SECRET_KEY = os.environ.get("SECRET_KEY", "your-secret-key-here")


def hash_password(password: str) -> str:
    """Hash a password using SHA-256."""
    return hashlib.sha256(password.encode()).hexdigest()


def verify_password(username: str, password: str) -> bool:
    """Verify if the provided password matches the stored hash."""
    if username not in USERS_DB:
        return False
    
    stored_hash = USERS_DB[username]
    provided_hash = hash_password(password)
    
    # Use hmac.compare_digest to prevent timing attacks
    return hmac.compare_digest(stored_hash, provided_hash)


def generate_token(username: str) -> str:
    """Generate a simple authentication token."""
    token_data = f"{username}:{SECRET_KEY}"
    return hashlib.sha256(token_data.encode()).hexdigest()


def validate_token(token: str) -> str | None:
    """Validate token and return username if valid."""
    for username in USERS_DB:
        expected_token = generate_token(username)
        if hmac.compare_digest(expected_token, token):
            return username
    return None


def require_auth(f):
    """Decorator to require authentication for protected routes."""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        auth_header = request.headers.get("Authorization")
        
        if not auth_header or not auth_header.startswith("Bearer "):
            return jsonify({
                "error": "Unauthorized",
                "message": "Authentication token is missing or invalid"
            }), 401
        
        token = auth_header.split(" ")[1]
        username = validate_token(token)
        
        if not username:
            return jsonify({
                "error": "Unauthorized",
                "message": "Invalid or expired token"
            }), 401
        
        request.current_user = username
        return f(*args, **kwargs)
    
    return decorated_function


@app.route("/login", methods=["POST"])
def login():
    """
    Login endpoint that validates username and password.
    Returns 401 Unauthorized if validation fails.
    """
    data = request.get_json()
    
    # Check if request body exists
    if not data:
        return jsonify({
            "error": "Unauthorized",
            "message": "Request body is missing"
        }), 401
    
    username = data.get("username", "").strip()
    password = data.get("password", "")
    
    # Validate that username and password are provided
    if not username or not password:
        return jsonify({
            "error": "Unauthorized",
            "message": "Username and password are required"
        }), 401
    
    # Verify credentials
    if not verify_password(username, password):
        return jsonify({
            "error": "Unauthorized",
            "message": "Invalid username or password"
        }), 401
    
    # Generate authentication token
    token = generate_token(username)
    
    return jsonify({
        "message": "Login successful",
        "username": username,
        "token": token
    }), 200


@app.route("/protected", methods=["GET"])
@require_auth
def protected_route():
    """Example of a protected route that requires authentication."""
    return jsonify({
        "message": f"Welcome, {request.current_user}! You have access to this protected resource.",
        "data": {
            "secret": "This is protected data",
            "user": request.current_user
        }
    }), 200


@app.route("/logout", methods=["POST"])
@require_auth
def logout():
    """Logout endpoint."""
    return jsonify({
        "message": f"User {request.current_user} has been logged out successfully"
    }), 200


@app.route("/health", methods=["GET"])
def health_check():
    """Health check endpoint."""
    return jsonify({"status": "healthy"}), 200


# Error handlers
@app.errorhandler(401)
def unauthorized(error):
    return jsonify({
        "error": "Unauthorized",
        "message": "Authentication is required to access this resource"
    }), 401


@app.errorhandler(404)
def not_found(error):
    return jsonify({
        "error": "Not Found",
        "message": "The requested resource was not found"
    }), 404


@app.errorhandler(405)
def method_not_allowed(error):
    return jsonify({
        "error": "Method Not Allowed",
        "message": "The HTTP method is not allowed for this endpoint"
    }), 405


def test_login_function():
    """Test the login functionality without Flask server."""
    print("=" * 50)
    print("Testing Login Function")
    print("=" * 50)
    
    # Test cases
    test_cases = [
        ("admin", "admin123", True, "Valid credentials"),
        ("admin", "wrongpassword", False, "Wrong password"),
        ("nonexistent", "password", False, "Non-existent user"),
        ("user1", "password1", True, "Valid user1 credentials"),
        ("", "", False, "Empty credentials"),
    ]
    
    for username, password, expected, description in test_cases:
        result = verify_password(username, password)
        status = "✓ PASS" if result == expected else "✗ FAIL"
        
        if result:
            response_code = 200
            response_msg = "Login successful"
        else:
            response_code = 401
            response_msg = "401 Unauthorized - Invalid username or password"
        
        print(f"\n{status} - {description}")
        print(f"  Username: '{username}', Password: '{password}'")
        print(f"  Response: {response_code} - {response_msg}")
    
    print("\n" + "=" * 50)
    print("Token Generation Test")
    print("=" * 50)
    
    # Test token generation and validation
    token = generate_token("admin")
    print(f"\nGenerated token for 'admin': {token[:20]}...")
    
    validated_user = validate_token(token)
    print(f"Token validation result: {validated_user}")
    
    invalid_token = "invalid_token_12345"
    invalid_result = validate_token(invalid_token)
    print(f"Invalid token validation result: {invalid_result}")


if __name__ == "__main__":
    # Run tests first
    test_login_function()
    
    print("\n" + "=" * 50)
    print("Starting Flask Server")
    print("=" * 50)
    print("\nAvailable endpoints:")
    print("  POST /login     - Login with username and password")
    print("  GET  /protected - Protected route (requires token)")
    print("  POST /logout    - Logout (requires token)")
    print("  GET  /health    - Health check")
    print("\nExample login request:")
    print('  curl -X POST http://localhost:5000/login \\')
    print('       -H "Content-Type: application/json" \\')
    print('       -d \'{"username": "admin", "password": "admin123"}\'')
    print("\nExample protected request:")
    print('  curl -X GET http://localhost:5000/protected \\')
    print('       -H "Authorization: Bearer <your-token>"')
    print()
    
    app.run(debug=True, host="0.0.0.0", port=5000)
