
```python
import anthropic
import json
import re
import uuid
from datetime import datetime

client = anthropic.Anthropic()
MODEL_NAME = "claude-opus-4-5"

tools = [
    {
        "name": "create_payment",
        "description": "Creates a payment session with the payment gateway. Returns a payment URL where the user should be redirected to complete the payment.",
        "input_schema": {
            "type": "object",
            "properties": {
                "amount": {
                    "type": "number",
                    "description": "The payment amount in cents (e.g., 1000 for $10.00)"
                },
                "currency": {
                    "type": "string",
                    "description": "The currency code (e.g., 'USD', 'EUR')"
                },
                "description": {
                    "type": "string",
                    "description": "Description of the payment"
                },
                "success_url": {
                    "type": "string",
                    "description": "URL to redirect the user after successful payment"
                },
                "cancel_url": {
                    "type": "string",
                    "description": "URL to redirect the user if payment is cancelled"
                }
            },
            "required": ["amount", "currency", "description", "success_url"]
        }
    },
    {
        "name": "verify_payment",
        "description": "Verifies the status of a payment using the payment session ID",
        "input_schema": {
            "type": "object",
            "properties": {
                "session_id": {
                    "type": "string",
                    "description": "The payment session ID to verify"
                }
            },
            "required": ["session_id"]
        }
    },
    {
        "name": "redirect_user",
        "description": "Redirects the user to a specified URL",
        "input_schema": {
            "type": "object",
            "properties": {
                "url": {
                    "type": "string",
                    "description": "The URL to redirect the user to"
                },
                "reason": {
                    "type": "string",
                    "description": "The reason for the redirect (e.g., 'payment_success', 'payment_cancelled')"
                }
            },
            "required": ["url", "reason"]
        }
    }
]

def create_payment(amount: float, currency: str, description: str, success_url: str, cancel_url: str = None) -> dict:
    """Simulates creating a payment session with a payment gateway"""
    session_id = str(uuid.uuid4())
    payment_url = f"https://payment-gateway.example.com/pay/{session_id}"
    
    print(f"\n[Payment Gateway] Creating payment session...")
    print(f"  Amount: {amount/100:.2f} {currency}")
    print(f"  Description: {description}")
    print(f"  Success URL: {success_url}")
    if cancel_url:
        print(f"  Cancel URL: {cancel_url}")
    
    return {
        "session_id": session_id,
        "payment_url": payment_url,
        "status": "pending",
        "amount": amount,
        "currency": currency,
        "description": description,
        "success_url": success_url,
        "cancel_url": cancel_url or "https://example.com/cancel",
        "created_at": datetime.now().isoformat()
    }

def verify_payment(session_id: str) -> dict:
    """Simulates verifying a payment status"""
    print(f"\n[Payment Gateway] Verifying payment session: {session_id}")
    
    return {
        "session_id": session_id,
        "status": "completed",
        "paid": True,
        "amount_received": 1000,
        "currency": "USD",
        "completed_at": datetime.now().isoformat()
    }

def redirect_user(url: str, reason: str) -> dict:
    """Simulates redirecting the user to a URL"""
    print(f"\n[Redirect] Redirecting user to: {url}")
    print(f"  Reason: {reason}")
    
    return {
        "redirected": True,
        "url": url,
        "reason": reason,
        "timestamp": datetime.now().isoformat()
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "create_payment":
        result = create_payment(
            amount=tool_input["amount"],
            currency=tool_input["currency"],
            description=tool_input["description"],
            success_url=tool_input["success_url"],
            cancel_url=tool_input.get("cancel_url")
        )
        return json.dumps(result)
    
    elif tool_name == "verify_payment":
        result = verify_payment(session_id=tool_input["session_id"])
        return json.dumps(result)
    
    elif tool_name == "redirect_user":
        result = redirect_user(url=tool_input["url"], reason=tool_input["reason"])
        return json.dumps(result)
    
    else:
        return json.dumps({"error": f"Unknown tool: {tool_name}"})

def process_payment_request(payment_request: dict) -> str:
    """
    Process a payment request using Claude with tool use.
    
    Args:
        payment_request: Dictionary containing payment details including success_url
    
    Returns:
        Final response from Claude about the payment processing
    """
    print("\n" + "="*60)
    print("PAYMENT PROCESSING SYSTEM")
    print("="*60)
    print(f"\nProcessing payment request:")
    print(json.dumps(payment_request, indent=2))
    
    user_message = f"""
    Process the following payment request and redirect the user to the success URL after payment:
    
    Payment Details:
    - Amount: {payment_request.get('amount', 1000)} cents ({payment_request.get('currency', 'USD')})
    - Description: {payment_request.get('description', 'Product purchase')}
    - Success URL: {payment_request.get('success_url', 'https://example.com/success')}
    - Cancel URL: {payment_request.get('cancel_url', 'https://example.com/cancel')}
    
    Please:
    1. Create a payment session with the payment gateway
    2. Simulate that the user completed the payment
    3. Verify the payment was successful
    4. Redirect the user to the success_url provided in the request
    """
    
    messages = [{"role": "user", "content": user_message}]
    
    print("\n" + "-"*60)
    print("Starting Claude interaction with tool use...")
    print("-"*60)
    
    while True:
        response = client.messages.create(
            model=MODEL_NAME,
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        print(f"\n[Claude] Stop reason: {response.stop_reason}")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    break
            
            print("\n" + "="*60)
            print("FINAL RESPONSE FROM CLAUDE:")
            print("="*60)
            print(final_response)
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_uses = [block for block in response.content if block.type == "tool_use"]
            
            messages.append({"role": "assistant", "content": response.content})
            
            tool_results = []
            for tool_use in tool_uses:
                print(f"\n[Tool Call] {tool_use.name}")
                print(f"  Input: {json.dumps(tool_use.input, indent=2)}")
                
                result = process_tool_call(tool_use.name, tool_use.input)
                result_dict = json.loads(result)
                print(f"  Result: {json.dumps(result_dict, indent=2)}")
                
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": tool_use.id,
                    "content": result
                })
            
            messages.append({"role": "user", "content": tool_results})
        
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Payment processing completed"

def main():
    """Main function to demonstrate the payment gateway integration"""
    
    payment_requests = [
        {
            "amount": 2999,
            "currency": "USD",
            "description": "Premium Subscription - Monthly Plan",
            "success_url": "https://mystore.com/payment/success?order_id=12345",
            "cancel_url": "https://mystore.com/payment/cancel?order_id=12345"
        },
        {
            "amount": 4999,
            "currency": "EUR",
            "description": "E-commerce Order #67890",
            "success_url": "https://shop.example.com/orders/67890/confirmed",
            "cancel_url": "https://shop.example.com/cart"
        }
    ]
    
    for i, payment_request in enumerate(payment_requests, 1):
        print(f"\n{'#'*60}")
        print(f"# PAYMENT REQUEST {i}")
        print(f"{'#'*60}")
        
        result = process_payment_request(payment_request)
        
        print(f"\n[System] Payment request {i} processed successfully")
        
        if i < len(payment_requests):
            print("\n" + "="*60)
            print("Processing next payment request...")
            print("="*60)

if __name__ == "__main__":
    main()
```
