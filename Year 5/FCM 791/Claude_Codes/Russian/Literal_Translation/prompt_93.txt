using System;
using System.Threading.Tasks;
using Microsoft.Data.SqlClient;

class BankTransfer
{
    private readonly string _connectionString;

    public BankTransfer(string connectionString)
    {
        _connectionString = connectionString;
    }

    public async Task TransferFundsAsync(int accountAId, int accountBId, decimal amount)
    {
        using var connection = new SqlConnection(_connectionString);
        await connection.OpenAsync();

        using var transaction = connection.BeginTransaction();

        try
        {
            // Check balance of Account A
            var checkBalanceCmd = new SqlCommand(
                "SELECT Balance FROM Accounts WHERE AccountId = @AccountId",
                connection,
                transaction);
            checkBalanceCmd.Parameters.AddWithValue("@AccountId", accountAId);

            var balanceObj = await checkBalanceCmd.ExecuteScalarAsync();
            if (balanceObj == null)
            {
                throw new InvalidOperationException($"Account A (ID: {accountAId}) not found.");
            }

            decimal currentBalance = Convert.ToDecimal(balanceObj);
            if (currentBalance < amount)
            {
                throw new InvalidOperationException(
                    $"Insufficient funds. Current balance: {currentBalance:C}, Required: {amount:C}");
            }

            // Debit Account A
            var debitCmd = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance - @Amount WHERE AccountId = @AccountId",
                connection,
                transaction);
            debitCmd.Parameters.AddWithValue("@Amount", amount);
            debitCmd.Parameters.AddWithValue("@AccountId", accountAId);

            int rowsAffected = await debitCmd.ExecuteNonQueryAsync();
            if (rowsAffected == 0)
            {
                throw new InvalidOperationException($"Failed to debit Account A (ID: {accountAId}).");
            }

            // Credit Account B
            var creditCmd = new SqlCommand(
                "UPDATE Accounts SET Balance = Balance + @Amount WHERE AccountId = @AccountId",
                connection,
                transaction);
            creditCmd.Parameters.AddWithValue("@Amount", amount);
            creditCmd.Parameters.AddWithValue("@AccountId", accountBId);

            rowsAffected = await creditCmd.ExecuteNonQueryAsync();
            if (rowsAffected == 0)
            {
                throw new InvalidOperationException($"Failed to credit Account B (ID: {accountBId}).");
            }

            // Log the transaction
            var logCmd = new SqlCommand(
                @"INSERT INTO TransactionLog (FromAccountId, ToAccountId, Amount, TransactionDate, Status)
                  VALUES (@FromAccountId, @ToAccountId, @Amount, @TransactionDate, @Status)",
                connection,
                transaction);
            logCmd.Parameters.AddWithValue("@FromAccountId", accountAId);
            logCmd.Parameters.AddWithValue("@ToAccountId", accountBId);
            logCmd.Parameters.AddWithValue("@Amount", amount);
            logCmd.Parameters.AddWithValue("@TransactionDate", DateTime.UtcNow);
            logCmd.Parameters.AddWithValue("@Status", "Completed");

            await logCmd.ExecuteNonQueryAsync();

            // Commit the transaction
            await transaction.CommitAsync();

            Console.WriteLine($"Transfer of {amount:C} from Account {accountAId} to Account {accountBId} completed successfully.");
        }
        catch (Exception ex)
        {
            // Rollback the transaction in case of error
            await transaction.RollbackAsync();
            Console.WriteLine($"Transaction failed and was rolled back. Error: {ex.Message}");
            throw;
        }
    }
}

class Program
{
    static async Task Main(string[] args)
    {
        // Example connection string (replace with actual connection string)
        string connectionString = "Server=localhost;Database=BankDB;Trusted_Connection=True;TrustServerCertificate=True;";

        var bankTransfer = new BankTransfer(connectionString);

        try
        {
            // Transfer $100 from Account A (ID: 1) to Account B (ID: 2)
            await bankTransfer.TransferFundsAsync(accountAId: 1, accountBId: 2, amount: 100.00m);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error during transfer: {ex.Message}");
        }
    }
}
