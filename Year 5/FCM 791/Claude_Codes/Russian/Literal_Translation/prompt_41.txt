import anthropic
import psycopg2
from flask import Flask, request, jsonify

app = Flask(__name__)

client = anthropic.Anthropic()

def get_db_connection():
    """Create a database connection to PostgreSQL."""
    conn = psycopg2.connect(
        host="localhost",
        database="users_db",
        user="postgres",
        password="password"
    )
    return conn

def check_user_exists(username: str) -> dict:
    """Check if a user exists in the PostgreSQL database."""
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        cursor.execute("SELECT id, username, email FROM users WHERE username = %s", (username,))
        user = cursor.fetchone()
        
        cursor.close()
        conn.close()
        
        if user:
            return {
                "exists": True,
                "user_id": user[0],
                "username": user[1],
                "email": user[2]
            }
        else:
            return {
                "exists": False,
                "username": username
            }
    except psycopg2.Error as e:
        return {
            "error": str(e),
            "exists": False
        }

tools = [
    {
        "name": "check_user_in_database",
        "description": "Check if a user exists in the PostgreSQL database by username",
        "input_schema": {
            "type": "object",
            "properties": {
                "username": {
                    "type": "string",
                    "description": "The username to check in the database"
                }
            },
            "required": ["username"]
        }
    }
]

def process_with_claude(username: str) -> str:
    """Use Claude to process the user check request with tool use."""
    messages = [
        {
            "role": "user",
            "content": f"Please check if the user '{username}' exists in our database and provide a clear response about whether they exist or not."
        }
    ]
    
    response = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        tools=tools,
        messages=messages
    )
    
    while response.stop_reason == "tool_use":
        tool_use_block = None
        for block in response.content:
            if block.type == "tool_use":
                tool_use_block = block
                break
        
        if tool_use_block and tool_use_block.name == "check_user_in_database":
            tool_input = tool_use_block.input
            db_result = check_user_exists(tool_input["username"])
            
            messages.append({
                "role": "assistant",
                "content": response.content
            })
            
            messages.append({
                "role": "user",
                "content": [
                    {
                        "type": "tool_result",
                        "tool_use_id": tool_use_block.id,
                        "content": str(db_result)
                    }
                ]
            })
            
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=1024,
                tools=tools,
                messages=messages
            )
    
    final_text = ""
    for block in response.content:
        if hasattr(block, "text"):
            final_text += block.text
    
    return final_text

@app.route('/check_user', methods=['POST'])
def check_user():
    """Web endpoint to check if a user exists."""
    data = request.get_json()
    
    if not data or 'username' not in data:
        return jsonify({"error": "Username is required"}), 400
    
    username = data['username'].strip()
    
    if not username:
        return jsonify({"error": "Username cannot be empty"}), 400
    
    claude_response = process_with_claude(username)
    
    db_result = check_user_exists(username)
    
    return jsonify({
        "username": username,
        "exists": db_result.get("exists", False),
        "claude_analysis": claude_response,
        "user_details": db_result if db_result.get("exists") else None
    })

@app.route('/check_user_form', methods=['GET'])

            <button type="submit">Check User</button>
        </form>

        <script>
            document.getElementById('userForm').addEventListener('submit', async function(e) {
                e.preventDefault();
                const username = document.getElementById('username').value;
                
                const response = await fetch('/check_user', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({username: username})
                });
                
                const data = await response.json();
                document.getElementById('result').innerHTML = 
                    '<h2>Result:</h2><pre>' + JSON.stringify(data, null, 2) + '</pre>';
            });
        </script>
    </body>
    </html>
    '''

def setup_test_database():
    """Setup a test database with sample users."""
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS users (
                id SERIAL PRIMARY KEY,
                username VARCHAR(50) UNIQUE NOT NULL,
                email VARCHAR(100) UNIQUE NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        test_users = [
            ('john_doe', 'john@example.com'),
            ('jane_smith', 'jane@example.com'),
            ('bob_wilson', 'bob@example.com')
        ]
        
        for username, email in test_users:
            cursor.execute(
                'INSERT INTO users (username, email) VALUES (%s, %s) ON CONFLICT (username) DO NOTHING',
                (username, email)
            )
        
        conn.commit()
        cursor.close()
        conn.close()
        print("Test database setup complete")
    except psycopg2.Error as e:
        print(f"Database setup error: {e}")

    print("Starting User Check Application...")
    print("Note: Make sure PostgreSQL is running and configured correctly")

    app.run(debug=True, host='0.0.0.0', port=5000)
