const Anthropic = require("@anthropic-ai/sdk");
const http = require("http");
const url = require("url");

const client = new Anthropic();

// Sample user database for demonstration

async function generateProfileSummary(userData) {
  const conversationHistory = [];

  // First message: Ask Claude to analyze the user profile
  conversationHistory.push({
    role: "user",
    content: `Please analyze this user profile data and provide a brief summary: ${JSON.stringify(userData, null, 2)}`,
  });

  const firstResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profile data and provides concise summaries.",
    messages: conversationHistory,
  });

  const firstAssistantMessage =
    firstResponse.content[0].type === "text"
      ? firstResponse.content[0].text
      : "";

  conversationHistory.push({
    role: "assistant",
    content: firstAssistantMessage,
  });

  // Second message: Ask for recommendations
  conversationHistory.push({
    role: "user",
    content:
      "Based on this profile, what are 3 personalized recommendations you would make for this user?",
  });

  const secondResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 500,
    system:
      "You are a helpful assistant that analyzes user profile data and provides concise summaries.",
    messages: conversationHistory,
  });

  const secondAssistantMessage =
    secondResponse.content[0].type === "text"
      ? secondResponse.content[0].text
      : "";

  conversationHistory.push({
    role: "assistant",
    content: secondAssistantMessage,
  });

  // Third message: Ask for a final formatted summary
  conversationHistory.push({
    role: "user",
    content:
      "Please provide a final formatted summary combining the profile analysis and recommendations in a structured way.",
  });

  const thirdResponse = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 600,
    system:
      "You are a helpful assistant that analyzes user profile data and provides concise summaries.",
    messages: conversationHistory,
  });

  const finalSummary =
    thirdResponse.content[0].type === "text"
      ? thirdResponse.content[0].text
      : "";

  return {
    profileAnalysis: firstAssistantMessage,
    recommendations: secondAssistantMessage,
    finalSummary: finalSummary,
  };
}

const server = http.createServer(async (req, res) => {
  const parsedUrl = url.parse(req.url, true);
  const pathname = parsedUrl.pathname;
  const query = parsedUrl.query;

  // Handle the profile download endpoint
  if (pathname === "/api/profile/download" && req.method === "GET") {
    const userId = query.user_id;

    // Validate user_id parameter
    if (!userId) {
      res.writeHead(400, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Missing required parameter: user_id",
          message: "Please provide a user_id in the query string",
        })
      );
      return;
    }

    // Check if user exists
    const userData = usersDatabase[userId];
    if (!userData) {
      res.writeHead(404, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "User not found",
          message: `No user found with id: ${userId}`,
        })
      );
      return;
    }

    try {
      console.log(
        `Generating AI-enhanced profile for user ${userId} using multi-turn conversation...`
      );

      // Generate AI summary using multi-turn conversation
      const aiSummary = await generateProfileSummary(userData);

      // Prepare the complete profile data for download
      const profileDownloadData = {
        exportDate: new Date().toISOString(),
        exportVersion: "1.0",
        userData: userData,
        aiGeneratedInsights: {
          profileAnalysis: aiSummary.profileAnalysis,
          personalizedRecommendations: aiSummary.recommendations,
          comprehensiveSummary: aiSummary.finalSummary,
          generatedAt: new Date().toISOString(),
          model: "claude-opus-4-5",
        },
      };

      // Convert to JSON string with formatting
      const jsonData = JSON.stringify(profileDownloadData, null, 2);

      // Set headers for file download
      res.writeHead(200, {
        "Content-Type": "application/json",
        "Content-Disposition": `attachment; filename="profile_user_${userId}_${Date.now()}.json"`,
        "Content-Length": Buffer.byteLength(jsonData),
        "Cache-Control": "no-cache",
      });

      res.end(jsonData);
      console.log(`Profile data for user ${userId} successfully sent`);
    } catch (error) {
      console.error("Error generating profile:", error);
      res.writeHead(500, { "Content-Type": "application/json" });
      res.end(
        JSON.stringify({
          error: "Internal server error",
          message: "Failed to generate profile data",
        })
      );
    }
  }
  // Handle root endpoint with API documentation
  else if (pathname === "/" && req.method === "GET") {
    const documentation = {
      api: "User Profile Download API",
      version: "1.0",
      endpoints: [
        {
          path: "/api/profile/download",
          method: "GET",
          description:
            "Download user profile data as JSON file with AI-generated insights",
          parameters: [
            {
              name: "user_id",
              type: "string",
              required: true,
              description: "The unique identifier of the user",
            },
          ],
          example: "/api/profile/download?user_id=1",
          availableUserIds: Object.keys(usersDatabase),
        },
      ],
    };

    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(JSON.stringify(documentation, null, 2));
  }
  // Handle unknown routes
  else {
    res.writeHead(404, { "Content-Type": "application/json" });
    res.end(
      JSON.stringify({
        error: "Not found",
        message: "The requested endpoint does not exist",
        availableEndpoints: ["/", "/api/profile/download?user_id=<id>"],
      })
    );
  }
});

const PORT = process.env.PORT || 3000;

server.listen(PORT, () => {
  console.log(`Server is running on http://localhost:${PORT}`);
  console.log(`\nAvailable endpoints:`);
  console.log(`  GET / - API documentation`);
  console.log(
    `  GET /api/profile/download?user_id=<id> - Download user profile`
  );
  console.log(`\nAvailable user IDs: ${Object.keys(usersDatabase).join(", ")}`);
  console.log(
    `\nExample: http://localhost:${PORT}/api/profile/download?user_id=1`
  );
});

module.exports = server;
