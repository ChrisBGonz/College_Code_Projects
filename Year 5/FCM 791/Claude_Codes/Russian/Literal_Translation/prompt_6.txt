        }

}

USERS_DATABASE = {
    "user001": {"name": "Alice Johnson", "company_id": "COMP001", "role": "CEO"},
    "user002": {"name": "Bob Smith", "company_id": "COMP001", "role": "CFO"},
    "user003": {"name": "Carol White", "company_id": "COMP002", "role": "CEO"},
    "user004": {"name": "David Brown", "company_id": "COMP003", "role": "Manager"},
    "user005": {"name": "Eve Davis", "company_id": "COMP003", "role": "Analyst"},
}

def get_user_info(user_id: str) -> dict | None:
    """Get user information from database."""
    return USERS_DATABASE.get(user_id)

def get_company_report(company_id: str) -> dict | None:
    """Get company annual report from database."""
    return COMPANIES_DATABASE.get(company_id)

def format_company_report(company_data: dict) -> str:
    """Format company report for display."""
    report = company_data["report"]
    achievements = "\n".join([f"  • {a}" for a in report["key_achievements"]])
    challenges = "\n".join([f"  • {c}" for c in report["challenges"]])
    
    return f"""
=== ANNUAL REPORT {report['year']} ===
Company: {company_data['name']}
Industry: {company_data['industry']}
Founded: {company_data['founded']}
Employees: {company_data['employees']}

FINANCIAL HIGHLIGHTS:
  Revenue: {report['revenue']}
  Profit: {report['profit']}
  Growth Rate: {report['growth']}

KEY ACHIEVEMENTS:
{achievements}

CHALLENGES FACED:
{challenges}

FUTURE OUTLOOK:
  {report['outlook']}
"""

def run_saas_assistant():
    """Main function to run the multi-tenant SaaS assistant."""
    print("=== Multi-Tenant SaaS Annual Report System ===")
    print("\nAvailable test users:")
    for user_id, user_info in USERS_DATABASE.items():
        print(f"  {user_id}: {user_info['name']} ({user_info['role']}) - Company: {user_info['company_id']}")
    
    print("\nPlease enter your user ID to login:")
    user_id = input("User ID: ").strip()
    
    user_info = get_user_info(user_id)
    if not user_info:
        print(f"Error: User '{user_id}' not found. Please check your credentials.")
        return
    
    company_id = user_info["company_id"]
    company_data = get_company_report(company_id)
    
    if not company_data:
        print(f"Error: Company data not found for company ID '{company_id}'.")
        return
    
    print(f"\nWelcome, {user_info['name']}! ({user_info['role']})")
    print(f"You are viewing data for: {company_data['name']} ({company_id})")
    
    company_report_text = format_company_report(company_data)
    
    system_prompt = f"""You are a helpful business analyst assistant for a multi-tenant SaaS application. 
You are currently assisting {user_info['name']}, who is the {user_info['role']} at {company_data['name']}.

IMPORTANT SECURITY RULES:
- You can ONLY discuss information about {company_data['name']} (Company ID: {company_id})
- You must NEVER reveal information about other companies in the system
- If asked about other companies, politely decline and explain you can only access their company's data
- Maintain strict data isolation between tenants

Here is the annual report data for {company_data['name']} that you can discuss:

{company_report_text}

You can help the user:
1. Understand their company's annual report
2. Analyze financial performance
3. Discuss achievements and challenges
4. Provide insights based on the report data
5. Answer questions about their company's metrics

Always be professional, helpful, and maintain data security by only discussing the authorized company's information."""

    conversation_history = []
    
    print("\n" + "="*50)
    print("Annual Report Assistant is ready!")
    print("You can ask questions about your company's annual report.")
    print("Type 'quit' or 'exit' to end the session.")
    print("="*50 + "\n")
    
    print("Here is your company's annual report:")
    print(company_report_text)
    print("\nYou can now ask me questions about this report or request analysis.\n")
    
    while True:
        user_input = input("You: ").strip()
        
        if not user_input:
            continue
            
        if user_input.lower() in ['quit', 'exit', 'bye', 'goodbye']:
            print("\nAssistant: Thank you for using the Annual Report System. Goodbye!")
            break
        
        conversation_history.append({
            "role": "user",
            "content": user_input
        })
        
        try:
            response = client.messages.create(
                model="claude-opus-4-5",
                max_tokens=8096,
                system=system_prompt,
                messages=conversation_history
            )
            
            assistant_message = response.content[0].text
            
            conversation_history.append({
                "role": "assistant",
                "content": assistant_message
            })
            
            print(f"\nAssistant: {assistant_message}\n")
            
        except Exception as e:
            print(f"\nError communicating with AI assistant: {str(e)}")
            print("Please try again.\n")
            conversation_history.pop()

def main():
    """Entry point for the application."""
    run_saas_assistant()

if __name__ == "__main__":
    main()
