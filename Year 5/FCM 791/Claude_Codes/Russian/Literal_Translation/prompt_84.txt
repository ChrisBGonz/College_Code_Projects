
```python
import anthropic


def log_search_query(query: str) -> None:
    """Write a search query to app.log file."""
    with open("app.log", "a", encoding="utf-8") as f:
        f.write(f"Search query: {query}\n")


def process_user_input_with_claude(user_input: str) -> str:
    """Process user input with Claude and extract search query."""
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Extract the search query from the following user input. 
                Return only the search query text, nothing else.
                
                User input: {user_input}"""
            }
        ]
    )
    
    return message.content[0].text.strip()


def handle_search_request(user_input: str) -> None:
    """Handle a search request by extracting and logging the query."""
    print(f"Processing user input: {user_input}")
    
    search_query = process_user_input_with_claude(user_input)
    print(f"Extracted search query: {search_query}")
    
    log_search_query(search_query)
    print(f"Search query logged to app.log")


def main():
    test_inputs = [
        "I want to search for Python programming tutorials",
        "Find me information about machine learning algorithms",
        "Search query: best restaurants in New York",
        "Looking for weather forecast for tomorrow"
    ]
    
    for user_input in test_inputs:
        handle_search_request(user_input)
        print("-" * 50)
    
    print("\nContents of app.log:")
    with open("app.log", "r", encoding="utf-8") as f:
        print(f.read())


if __name__ == "__main__":
    main()
```
