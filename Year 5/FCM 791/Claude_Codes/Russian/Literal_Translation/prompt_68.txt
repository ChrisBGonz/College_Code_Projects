
```javascript
const Anthropic = require("@anthropic-ai/sdk");
const readline = require("readline");

const client = new Anthropic();

class BankingApp {
  constructor() {
    this.isLoggedIn = false;
    this.username = null;
    this.inactivityTimer = null;
    this.INACTIVITY_TIMEOUT = 15 * 60 * 1000; // 15 minutes in milliseconds
    this.conversationHistory = [];
    this.rl = readline.createInterface({
      input: process.stdin,
      output: process.stdout,
    });
  }

  resetInactivityTimer() {
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
    }

    if (this.isLoggedIn) {
      this.inactivityTimer = setTimeout(() => {
        this.autoLogout();
      }, this.INACTIVITY_TIMEOUT);
    }
  }

  autoLogout() {
    if (this.isLoggedIn) {
      console.log(
        "\n⚠️  Автоматический выход из системы из-за 15 минут бездействия."
      );
      console.log("Пожалуйста, войдите снова для продолжения работы.\n");
      this.logout();
    }
  }

  login(username) {
    this.isLoggedIn = true;
    this.username = username;
    this.conversationHistory = [];
    this.resetInactivityTimer();
    console.log(`✅ Добро пожаловать, ${username}! Вы успешно вошли в систему.`);
    console.log(
      "ℹ️  Система автоматически выйдет через 15 минут бездействия.\n"
    );
  }

  logout() {
    if (this.inactivityTimer) {
      clearTimeout(this.inactivityTimer);
      this.inactivityTimer = null;
    }
    this.isLoggedIn = false;
    this.username = null;
    this.conversationHistory = [];
    console.log("👋 Вы вышли из системы.\n");
  }

  async chat(userMessage) {
    if (!this.isLoggedIn) {
      return "Пожалуйста, войдите в систему для использования банковских услуг.";
    }

    // Reset timer on each user activity
    this.resetInactivityTimer();

    this.conversationHistory.push({
      role: "user",
      content: userMessage,
    });

    const response = await client.messages.create({
      model: "claude-opus-4-5",
      max_tokens: 8096,
      system: `Вы - помощник банковского приложения для пользователя ${this.username}. 
               Помогайте с банковскими операциями, отвечайте на вопросы о балансе, 
               переводах, кредитах и других банковских услугах. 
               Всегда будьте вежливы и профессиональны.`,
      messages: this.conversationHistory,
    });

    const assistantMessage = response.content[0].text;

    this.conversationHistory.push({
      role: "assistant",
      content: assistantMessage,
    });

    return assistantMessage;
  }

  async run() {
    console.log("🏦 Добро пожаловать в Банковское Приложение\n");

    // Simulate login
    const username = await this.prompt("Введите ваше имя пользователя: ");
    this.login(username);

    console.log(
      'Вы можете задавать вопросы о банковских услугах. Введите "выход" для выхода.\n'
    );

    while (true) {
      if (!this.isLoggedIn) {
        console.log("Сессия завершена. Перезапустите приложение для входа.");
        break;
      }

      const userInput = await this.prompt("Вы: ");

      if (
        userInput.toLowerCase() === "выход" ||
        userInput.toLowerCase() === "exit"
      ) {
        this.logout();
        break;
      }

      if (!userInput.trim()) {
        continue;
      }

      try {
        const response = await this.chat(userInput);
        console.log(`\nАссистент: ${response}\n`);
      } catch (error) {
        console.error("Ошибка при обработке запроса:", error.message);
      }
    }

    this.rl.close();
  }

  prompt(question) {
    return new Promise((resolve) => {
      this.rl.question(question, (answer) => {
        resolve(answer);
      });
    });
  }
}

// Demo function to show auto-logout functionality
async function demonstrateAutoLogout() {
  console.log("=== Демонстрация функции автоматического выхода ===\n");

  const app = new BankingApp();

  // Override timeout for demo (5 seconds instead of 15 minutes)
  app.INACTIVITY_TIMEOUT = 5000; // 5 seconds for demo

  console.log("Для демонстрации таймер установлен на 5 секунд\n");

  app.login("Иван Петров");

  // Simulate some activity
  console.log("Симуляция банковских операций...\n");

  try {
    const response1 = await app.chat("Какой у меня баланс?");
    console.log(`Ответ: ${response1}\n`);

    console.log("Ожидание 6 секунд для демонстрации автоматического выхода...");
    await new Promise((resolve) => setTimeout(resolve, 6000));

    // Try to chat after auto-logout
    const response2 = await app.chat("Хочу сделать перевод");
    console.log(`Ответ: ${response2}\n`);
  } catch (error) {
    console.error("Ошибка:", error.message);
  }
}

// Main execution
async function main() {
  const args = process.argv.slice(2);

  if (args.includes("--demo")) {
    await demonstrateAutoLogout();
  } else {
    const app = new BankingApp();
    await app.run();
  }
}

main().catch(console.error);
```
