const MultiStepForm = (() => {
  const STORAGE_KEY = 'multiStepFormData';
  const CURRENT_STEP_KEY = 'multiStepFormCurrentStep';

  const steps = [
    {
      id: 1,
      title: 'Personal Information',
      fields: [
        { name: 'firstName', label: 'First Name', type: 'text', required: true },
        { name: 'lastName', label: 'Last Name', type: 'text', required: true },
        { name: 'birthDate', label: 'Date of Birth', type: 'date', required: true },
      ],
    },
    {
      id: 2,
      title: 'Contact Details',
      fields: [
        { name: 'email', label: 'Email Address', type: 'email', required: true },
        { name: 'phone', label: 'Phone Number', type: 'tel', required: false },
        { name: 'address', label: 'Address', type: 'text', required: false },
      ],
    },
    {
      id: 3,
      title: 'Account Setup',
      fields: [
        { name: 'username', label: 'Username', type: 'text', required: true },
        { name: 'password', label: 'Password', type: 'password', required: true },
        { name: 'confirmPassword', label: 'Confirm Password', type: 'password', required: true },
      ],
    },
    {
      id: 4,
      title: 'Preferences',
      fields: [
        { name: 'newsletter', label: 'Subscribe to Newsletter', type: 'checkbox', required: false },
        { name: 'language', label: 'Preferred Language', type: 'select', required: true, options: ['English', 'Spanish', 'French', 'German'] },
        { name: 'bio', label: 'Short Bio', type: 'textarea', required: false },
      ],
    },
  ];

  let currentStep = 1;
  let formData = {};

  function init() {
    loadFromStorage();
    renderForm();
    attachGlobalListeners();
  }

  function saveToStorage() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
      localStorage.setItem(CURRENT_STEP_KEY, String(currentStep));
    } catch (e) {
      console.error('Failed to save form data to localStorage:', e);
    }
  }

  function loadFromStorage() {
    try {
      const savedData = localStorage.getItem(STORAGE_KEY);
      const savedStep = localStorage.getItem(CURRENT_STEP_KEY);

      if (savedData) {
        formData = JSON.parse(savedData);
      }

      if (savedStep) {
        const parsedStep = parseInt(savedStep, 10);
        if (parsedStep >= 1 && parsedStep <= steps.length) {
          currentStep = parsedStep;
        }
      }
    } catch (e) {
      console.error('Failed to load form data from localStorage:', e);
      formData = {};
      currentStep = 1;
    }
  }

  function clearStorage() {
    try {
      localStorage.removeItem(STORAGE_KEY);
      localStorage.removeItem(CURRENT_STEP_KEY);
    } catch (e) {
      console.error('Failed to clear form data from localStorage:', e);
    }
  }

  function getFieldValue(fieldName) {
    return formData[fieldName] !== undefined ? formData[fieldName] : '';
  }

  function renderProgressBar() {
    const progressPercent = ((currentStep - 1) / (steps.length - 1)) * 100;

    const stepsHtml = steps
      .map(
        (step) => `
      <div class="step-indicator ${step.id < currentStep ? 'completed' : ''} ${step.id === currentStep ? 'active' : ''}">
        <div class="step-circle">${step.id < currentStep ? '✓' : step.id}</div>
        <div class="step-label">${step.title}</div>
      </div>
    `
      )
      .join('');

    return `
      <div class="progress-container">
        <div class="progress-bar">
          <div class="progress-fill" style="width: ${progressPercent}%"></div>
        </div>
        <div class="steps-indicators">
          ${stepsHtml}
        </div>
        <div class="step-counter">Step ${currentStep} of ${steps.length}</div>
      </div>
    `;
  }

  function renderField(field) {
    const value = getFieldValue(field.name);
    const requiredAttr = field.required ? 'required' : '';
    const requiredMark = field.required ? '<span class="required-mark">*</span>' : '';

    let inputHtml = '';

    switch (field.type) {
      case 'textarea':
        inputHtml = `
          <textarea
            id="${field.name}"
            name="${field.name}"
            class="form-input"
            placeholder="Enter ${field.label.toLowerCase()}"
            ${requiredAttr}
            rows="4"
          >${value}</textarea>
        `;
        break;

      case 'select':
        const optionsHtml = field.options
          .map(
            (opt) => `
          <option value="${opt}" ${value === opt ? 'selected' : ''}>${opt}</option>
        `
          )
          .join('');
        inputHtml = `
          <select id="${field.name}" name="${field.name}" class="form-input" ${requiredAttr}>
            <option value="">-- Select ${field.label} --</option>
            ${optionsHtml}
          </select>
        `;
        break;

      case 'checkbox':
        inputHtml = `
          <div class="checkbox-wrapper">
            <input
              type="checkbox"
              id="${field.name}"
              name="${field.name}"
              class="form-checkbox"
              ${value === true || value === 'true' ? 'checked' : ''}
            />
            <label for="${field.name}" class="checkbox-label">${field.label}</label>
          </div>
        `;
        return `
          <div class="form-group checkbox-group">
            ${inputHtml}
          </div>
        `;

      default:
        inputHtml = `
          <input
            type="${field.type}"
            id="${field.name}"
            name="${field.name}"
            class="form-input"
            value="${field.type !== 'password' ? value : ''}"
            placeholder="Enter ${field.label.toLowerCase()}"
            ${requiredAttr}
          />
        `;
    }

    return `
      <div class="form-group">
        <label for="${field.name}" class="form-label">
          ${field.label} ${requiredMark}
        </label>
        ${inputHtml}
        <span class="error-message" id="error-${field.name}"></span>
      </div>
    `;
  }

  function renderCurrentStep() {
    const step = steps[currentStep - 1];
    const fieldsHtml = step.fields.map(renderField).join('');

    const isFirstStep = currentStep === 1;
    const isLastStep = currentStep === steps.length;

    const savedIndicator =
      Object.keys(formData).length > 0
        ? '<div class="saved-indicator">💾 Progress saved</div>'
        : '';

    return `
      <div class="step-content" data-step="${step.id}">
        <h2 class="step-title">${step.title}</h2>
        ${savedIndicator}
        <form id="stepForm" novalidate>
          ${fieldsHtml}
          <div class="form-navigation">
            ${
              !isFirstStep
                ? `<button type="button" class="btn btn-secondary" id="prevBtn">← Previous</button>`
                : '<div></div>'
            }
            <div class="nav-right">
              <button type="button" class="btn btn-outline" id="saveBtn">Save Progress</button>
              ${
                isLastStep
                  ? `<button type="submit" class="btn btn-success" id="submitBtn">Submit ✓</button>`
                  : `<button type="button" class="btn btn-primary" id="nextBtn">Next →</button>`
              }
            </div>
          </div>
        </form>
      </div>
    `;
  }

  function renderForm() {
    let container = document.getElementById('multiStepFormContainer');

    if (!container) {
      container = document.createElement('div');
      container.id = 'multiStepFormContainer';
      container.className = 'form-container';
      document.body.appendChild(container);
    }

    injectStyles();

    container.innerHTML = `
      <div class="form-wrapper">
        <h1 class="form-title">Registration Form</h1>
        ${renderProgressBar()}
        <div id="stepContainer">
          ${renderCurrentStep()}
        </div>
      </div>
    `;

    attachStepListeners();
  }

  function updateStepContainer() {
    const stepContainer = document.getElementById('stepContainer');
    if (stepContainer) {
      stepContainer.innerHTML = renderCurrentStep();
      updateProgressBar();
      attachStepListeners();
    }
  }

  function updateProgressBar() {
    const progressFill = document.querySelector('.progress-fill');
    const stepIndicators = document.querySelectorAll('.step-indicator');
    const stepCounter = document.querySelector('.step-counter');

    if (progressFill) {
      const progressPercent = ((currentStep - 1) / (steps.length - 1)) * 100;
      progressFill.style.width = `${progressPercent}%`;
    }

    if (stepIndicators) {
      stepIndicators.forEach((indicator, index) => {
        const stepNum = index + 1;
        indicator.classList.remove('active', 'completed');
        if (stepNum === currentStep) {
          indicator.classList.add('active');
        } else if (stepNum < currentStep) {
          indicator.classList.add('completed');
          const circle = indicator.querySelector('.step-circle');
          if (circle) circle.textContent = '✓';
        } else {
          const circle = indicator.querySelector('.step-circle');
          if (circle) circle.textContent = String(stepNum);
        }
      });
    }

    if (stepCounter) {
      stepCounter.textContent = `Step ${currentStep} of ${steps.length}`;
    }
  }

  function collectStepData() {
    const step = steps[currentStep - 1];
    const stepData = {};

    step.fields.forEach((field) => {
      const element = document.getElementById(field.name);
      if (!element) return;

      if (field.type === 'checkbox') {
        stepData[field.name] = element.checked;
      } else {
        stepData[field.name] = element.value.trim();
      }
    });

    return stepData;
  }

  function validateStep() {
    const step = steps[currentStep - 1];
    let isValid = true;
    const errors = {};

    clearErrors();

    step.fields.forEach((field) => {
      const element = document.getElementById(field.name);
      if (!element) return;

      const value = field.type === 'checkbox' ? element.checked : element.value.trim();

      if (field.required && !value) {
        errors[field.name] = `${field.label} is required`;
        isValid = false;
      }

      if (field.type === 'email' && value) {
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(value)) {
          errors[field.name] = 'Please enter a valid email address';
          isValid = false;
        }
      }

      if (field.type === 'tel' && value) {
        const phoneRegex = /^[\+]?[(]?[0-9]{3}[)]?[-\s\.]?[0-9]{3}[-\s\.]?[0-9]{4,6}$/;
        if (!phoneRegex.test(value)) {
          errors[field.name] = 'Please enter a valid phone number';
          isValid = false;
        }
      }

      if (field.name === 'password' && value) {
        if (value.length < 8) {
          errors[field.name] = 'Password must be at least 8 characters long';
          isValid = false;
        }
      }

      if (field.name === 'confirmPassword' && value) {
        const passwordField = document.getElementById('password');
        if (passwordField && value !== passwordField.value) {
          errors[field.name] = 'Passwords do not match';
          isValid = false;
        }
      }

      if (field.name === 'username' && value) {
        const usernameRegex = /^[a-zA-Z0-9_]{3,20}$/;
        if (!usernameRegex.test(value)) {
          errors[field.name] = 'Username must be 3-20 characters (letters, numbers, underscores)';
          isValid = false;
        }
      }
    });

    displayErrors(errors);
    return isValid;
  }

  function clearErrors() {
    const errorElements = document.querySelectorAll('.error-message');
    errorElements.forEach((el) => {
      el.textContent = '';
      el.style.display = 'none';
    });

    const inputElements = document.querySelectorAll('.form-input, .form-checkbox');
    inputElements.forEach((el) => el.classList.remove('input-error'));
  }

  function displayErrors(errors) {
    Object.entries(errors).forEach(([fieldName, message]) => {
      const errorEl = document.getElementById(`error-${fieldName}`);
      const inputEl = document.getElementById(fieldName);

      if (errorEl) {
        errorEl.textContent = message;
        errorEl.style.display = 'block';
      }

      if (inputEl) {
        inputEl.classList.add('input-error');
      }
    });
  }

  function handleNext() {
    if (!validateStep()) return;

    const stepData = collectStepData();
    formData = { ...formData, ...stepData };
    saveToStorage();

    if (currentStep < steps.length) {
      currentStep++;
      saveToStorage();
      updateStepContainer();
    }
  }

  function handlePrev() {
    const stepData = collectStepData();
    formData = { ...formData, ...stepData };
    saveToStorage();

    if (currentStep > 1) {
      currentStep--;
      saveToStorage();
      updateStepContainer();
    }
  }

  function handleSave() {
    const stepData = collectStepData();
    formData = { ...formData, ...stepData };
    saveToStorage();
    showNotification('Progress saved successfully!', 'success');
  }

  function handleSubmit(e) {
    e.preventDefault();

    if (!validateStep()) return;

    const stepData = collectStepData();
    formData = { ...formData, ...stepData };

    const submitData = { ...formData };
    delete submitData.password;
    delete submitData.confirmPassword;

    console.log('Form submitted with data:', submitData);

    sendFormData(formData)
      .then((response) => {
        clearStorage();
        formData = {};
        currentStep = 1;
        showSuccessScreen(response);
      })
      .catch((error) => {
        console.error('Submission failed:', error);
        showNotification('Submission failed. Please try again.', 'error');
      });
  }

  function sendFormData(data) {
    return new Promise((resolve, reject) => {
      setTimeout(() => {
        if (Math.random() > 0.1) {
          resolve({ success: true, userId: Math.floor(Math.random() * 10000), message: 'Registration successful!' });
        } else {
          reject(new Error('Server error'));
        }
      }, 1500);
    });
  }

  function showSuccessScreen(response) {
    const container = document.getElementById('multiStepFormContainer');
    if (container) {
      container.innerHTML = `
        <div class="form-wrapper">
          <div class="success-screen">
            <div class="success-icon">✓</div>
            <h2>Registration Complete!</h2>
            <p>${response.message}</p>
            <p class="user-id">Your User ID: <strong>#${response.userId}</strong></p>
            <button class="btn btn-primary" id="resetBtn">Register Another Account</button>
          </div>
        </div>
      `;

      document.getElementById('resetBtn').addEventListener('click', () => {
        formData = {};
        currentStep = 1;
        renderForm();
      });
    }
  }

  function showNotification(message, type = 'info') {
    const existing = document.getElementById('formNotification');
    if (existing) existing.remove();

    const notification = document.createElement('div');
    notification.id = 'formNotification';
    notification.className = `notification notification-${type}`;
    notification.textContent = message;

    document.body.appendChild(notification);

    setTimeout(() => {
      notification.classList.add('notification-show');
    }, 10);

    setTimeout(() => {
      notification.classList.remove('notification-show');
      setTimeout(() => notification.remove(), 300);
    }, 3000);
  }

  function attachStepListeners() {
    const nextBtn = document.getElementById('nextBtn');
    const prevBtn = document.getElementById('prevBtn');
    const saveBtn = document.getElementById('saveBtn');
    const stepForm = document.getElementById('stepForm');

    if (nextBtn) nextBtn.addEventListener('click', handleNext);
    if (prevBtn) prevBtn.addEventListener('click', handlePrev);
    if (saveBtn) saveBtn.addEventListener('click', handleSave);
    if (stepForm) stepForm.addEventListener('submit', handleSubmit);

    const inputs = document.querySelectorAll('.form-input, .form-checkbox');
    inputs.forEach((input) => {
      input.addEventListener('input', () => {
        const errorEl = document.getElementById(`error-${input.name}`);
        if (errorEl) {
          errorEl.textContent = '';
          errorEl.style.display = 'none';
        }
        input.classList.remove('input-error');
      });

      input.addEventListener('blur', () => {
        const stepData = collectStepData();
        formData = { ...formData, ...stepData };
        saveToStorage();
      });
    });
  }

  function attachGlobalListeners() {
    window.addEventListener('beforeunload', () => {
      const stepData = collectStepData();
      if (stepData && Object.keys(stepData).length > 0) {
        formData = { ...formData, ...stepData };
        saveToStorage();
      }
    });
  }
