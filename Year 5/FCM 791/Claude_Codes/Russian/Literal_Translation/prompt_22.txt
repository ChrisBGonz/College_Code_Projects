#!/usr/bin/env bash

set -euo pipefail

# ─── Настройки ────────────────────────────────────────────────────────────────
TOOL_NAME="security-tool"
INSTALL_DIR="/usr/local/bin"
TMP_DIR="$(mktemp -d)"
LOG_FILE="/var/log/${TOOL_NAME}-install.log"
GITHUB_API="https://api.github.com/repos"
CHECKSUM_FILE="checksums.sha256"

# Цвета
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

# ─── Вспомогательные функции ──────────────────────────────────────────────────
log()     { echo -e "${GREEN}[INFO]${NC}  $(date '+%Y-%m-%d %H:%M:%S') $*" | tee -a "$LOG_FILE"; }
warn()    { echo -e "${YELLOW}[WARN]${NC}  $(date '+%Y-%m-%d %H:%M:%S') $*" | tee -a "$LOG_FILE"; }
error()   { echo -e "${RED}[ERROR]${NC} $(date '+%Y-%m-%d %H:%M:%S') $*" | tee -a "$LOG_FILE" >&2; }
success() { echo -e "${GREEN}[OK]${NC}    $(date '+%Y-%m-%d %H:%M:%S') $*" | tee -a "$LOG_FILE"; }
info()    { echo -e "${BLUE}[...]${NC}   $(date '+%Y-%m-%d %H:%M:%S') $*" | tee -a "$LOG_FILE"; }

cleanup() {
    info "Очистка временных файлов..."
    rm -rf "$TMP_DIR"
}
trap cleanup EXIT

die() {
    error "$*"
    exit 1
}

# ─── Проверка зависимостей ────────────────────────────────────────────────────
check_dependencies() {
    local deps=("curl" "tar" "sha256sum" "jq" "install")
    local missing=()

    for dep in "${deps[@]}"; do
        if ! command -v "$dep" &>/dev/null; then
            missing+=("$dep")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        die "Отсутствуют зависимости: ${missing[*]}. Установите их и повторите попытку."
    fi

    log "Все зависимости найдены."
}

# ─── Определение ОС и архитектуры ────────────────────────────────────────────
detect_platform() {
    OS="$(uname -s | tr '[:upper:]' '[:lower:]')"
    ARCH="$(uname -m)"

    case "$ARCH" in
        x86_64)  ARCH="amd64" ;;
        aarch64|arm64) ARCH="arm64" ;;
        armv7l)  ARCH="armv7" ;;
        i386|i686) ARCH="386" ;;
        *) die "Неподдерживаемая архитектура: $ARCH" ;;
    esac

    case "$OS" in
        linux)   EXT="tar.gz" ;;
        darwin)  EXT="tar.gz" ;;
        windows) EXT="zip"; OS="windows" ;;
        *) die "Неподдерживаемая ОС: $OS" ;;
    esac

    log "Платформа: ${OS}/${ARCH}"
}

# ─── Получение последней версии через GitHub API ──────────────────────────────
get_latest_version() {
    local repo="${1:-}"
    [[ -z "$repo" ]] && die "Не указан репозиторий (owner/repo)."

    info "Запрос последней версии из GitHub API..."

    local api_url="${GITHUB_API}/${repo}/releases/latest"
    local response

    response="$(curl -fsSL \
        -H "Accept: application/vnd.github.v3+json" \
        ${GITHUB_TOKEN:+-H "Authorization: token $GITHUB_TOKEN"} \
        "$api_url")" || die "Не удалось получить информацию о релизе."

    VERSION="$(echo "$response" | jq -r '.tag_name')"
    [[ -z "$VERSION" || "$VERSION" == "null" ]] && die "Не удалось определить версию."

    RELEASE_DATE="$(echo "$response" | jq -r '.published_at')"
    log "Последняя версия: ${VERSION} (опубликована: ${RELEASE_DATE})"
}

# ─── Формирование URL загрузки ────────────────────────────────────────────────
build_download_url() {
    local repo="$1"
    local version="$2"
    local base_url="https://github.com/${repo}/releases/download/${version}"

    BINARY_NAME="${TOOL_NAME}_${version#v}_${OS}_${ARCH}.${EXT}"
    DOWNLOAD_URL="${base_url}/${BINARY_NAME}"
    CHECKSUM_URL="${base_url}/${CHECKSUM_FILE}"

    log "URL загрузки: $DOWNLOAD_URL"
}

# ─── Загрузка файла с прогрессом ─────────────────────────────────────────────
download_file() {
    local url="$1"
    local dest="$2"
    local description="${3:-файл}"

    info "Загрузка: $description..."

    if ! curl -fSL \
        --retry 3 \
        --retry-delay 2 \
        --connect-timeout 30 \
        --max-time 300 \
        --progress-bar \
        -o "$dest" \
        "$url"; then
        die "Ошибка загрузки: $url"
    fi

    success "Загружено: $dest"
}

# ─── Проверка контрольной суммы ───────────────────────────────────────────────
verify_checksum() {
    local archive="$1"
    local checksum_file="$2"

    info "Проверка контрольной суммы..."

    if [[ ! -f "$checksum_file" ]]; then
        warn "Файл контрольных сумм не найден. Пропуск проверки."
        return 0
    fi

    local filename
    filename="$(basename "$archive")"

    # Фильтрация нужной строки из файла контрольных сумм
    local expected_line
    expected_line="$(grep -E "(^|\s)${filename}$" "$checksum_file" || true)"

    if [[ -z "$expected_line" ]]; then
        warn "Контрольная сумма для '$filename' не найдена в файле. Пропуск."
        return 0
    fi

    # Проверка
    local check_result
    if echo "$expected_line" | sha256sum --check --status 2>/dev/null; then
        success "Контрольная сумма верна."
    else
        die "Контрольная сумма НЕ совпадает! Файл повреждён или подменён."
    fi
}

# ─── Распаковка архива ────────────────────────────────────────────────────────
extract_archive() {
    local archive="$1"
    local dest_dir="$2"

    info "Распаковка архива..."

    case "$archive" in
        *.tar.gz|*.tgz)
            tar -xzf "$archive" -C "$dest_dir" || die "Ошибка распаковки tar.gz"
            ;;
        *.tar.bz2)
            tar -xjf "$archive" -C "$dest_dir" || die "Ошибка распаковки tar.bz2"
            ;;
        *.zip)
            command -v unzip &>/dev/null || die "unzip не установлен."
            unzip -q "$archive" -d "$dest_dir" || die "Ошибка распаковки zip"
            ;;
        *)
            die "Неизвестный формат архива: $archive"
            ;;
    esac

    success "Архив распакован в: $dest_dir"
}

# ─── Поиск бинарного файла ───────────────────────────────────────────────────
find_binary() {
    local search_dir="$1"
    local binary_name="$2"

    local found
    found="$(find "$search_dir" -type f -name "$binary_name" 2>/dev/null | head -n1)"

    if [[ -z "$found" ]]; then
        # Попытка найти любой исполняемый файл
        found="$(find "$search_dir" -type f -executable 2>/dev/null | head -n1)"
    fi

    [[ -z "$found" ]] && die "Бинарный файл не найден в: $search_dir"
    echo "$found"
}

# ─── Установка инструмента ────────────────────────────────────────────────────
install_tool() {
    local binary_path="$1"
    local install_path="${INSTALL_DIR}/${TOOL_NAME}"

    info "Установка в: $install_path"

    # Резервная копия существующей версии
    if [[ -f "$install_path" ]]; then
        local backup="${install_path}.backup.$(date '+%Y%m%d%H%M%S')"
        warn "Найдена существующая версия. Резервная копия: $backup"
        cp "$install_path" "$backup"
    fi

    # Установка с правильными правами
    if [[ -w "$INSTALL_DIR" ]]; then
        install -m 755 "$binary_path" "$install_path"
    else
        info "Требуются права суперпользователя для установки в $INSTALL_DIR"
        sudo install -m 755 "$binary_path" "$install_path"
    fi

    success "Установлено: $install_path"
}

# ─── Проверка установки ───────────────────────────────────────────────────────
verify_installation() {
    local install_path="${INSTALL_DIR}/${TOOL_NAME}"

    if [[ ! -x "$install_path" ]]; then
        die "Файл не найден или не исполняемый: $install_path"
    fi

    # Проверка версии (если инструмент поддерживает --version)
    local installed_version
    if installed_version="$("$install_path" --version 2>/dev/null || true)"; then
        success "Версия после установки: $installed_version"
    else
        warn "Не удалось получить версию через --version"
    fi

    # Проверка доступности в PATH
    if command -v "$TOOL_NAME" &>/dev/null; then
        success "$TOOL_NAME доступен в PATH."
    else
        warn "$TOOL_NAME не найден в PATH. Добавьте $INSTALL_DIR в переменную PATH."
    fi
}

# ─── Проверка текущей версии (пропуск если актуальна) ────────────────────────
check_current_version() {
    local latest="$1"

    if command -v "$TOOL_NAME" &>/dev/null; then
        local current
        current="$("$TOOL_NAME" --version 2>/dev/null | grep -oE 'v?[0-9]+\.[0-9]+\.[0-9]+' | head -n1 || true)"

        if [[ -n "$current" ]]; then
            log "Текущая версия: $current"
            if [[ "$current" == "${latest#v}" || "v$current" == "$latest" ]]; then
                success "Уже установлена последняя версия ($current). Обновление не требуется."
                exit 0
            fi
        fi
    fi
}

# ─── Вывод справки ───────────────────────────────────────────────────────────
usage() {
    cat <<EOF
Использование: $0 [ОПЦИИ]

Опции:
  -r, --repo OWNER/REPO   GitHub репозиторий (обязательно)
  -n, --name NAME         Имя инструмента (по умолчанию: security-tool)
  -d, --dir DIR           Директория установки (по умолчанию: /usr/local/bin)
  -v, --version VERSION   Установить конкретную версию (по умолчанию: latest)
  -f, --force             Принудительная переустановка
  -h, --help              Показать эту справку

Переменные окружения:
  GITHUB_TOKEN            Токен GitHub API (для обхода ограничений)

Примеры:
  $0 --repo aquasecurity/trivy --name trivy
  $0 --repo projectdiscovery/nuclei --name nuclei --dir ~/.local/bin
  $0 --repo anchore/grype --name grype --version v0.65.0

EOF
    exit 0
}

# ─── Разбор аргументов ────────────────────────────────────────────────────────
parse_args() {
    REPO=""
    FORCE=false
    SPECIFIC_VERSION=""

    while [[ $# -gt 0 ]]; do
        case "$1" in
            -r|--repo)
                REPO="${2:-}"
                [[ -z "$REPO" ]] && die "Не указан репозиторий."
                shift 2
                ;;
            -n|--name)
                TOOL_NAME="${2:-$TOOL_NAME}"
                shift 2
                ;;
            -d|--dir)
                INSTALL_DIR="${2:-$INSTALL_DIR}"
                shift 2
                ;;
            -v|--version)
                SPECIFIC_VERSION="${2:-}"
                shift 2
                ;;
            -f|--force)
                FORCE=true
                shift
                ;;
            -h|--help)
                usage
                ;;
            *)
                die "Неизвестный аргумент: $1. Используйте --help."
                ;;
        esac
    done

    [[ -z "$REPO" ]] && die "Не указан репозиторий. Используйте --repo owner/repo."
}

# ─── Основная функция ─────────────────────────────────────────────────────────
main() {
    # Инициализация лог-файла
    mkdir -p "$(dirname "$LOG_FILE")" 2>/dev/null || LOG_FILE="/tmp/${TOOL_NAME}-install.log"
    touch "$LOG_FILE" 2>/dev/null || LOG_FILE="/tmp/${TOOL_NAME}-install.log"

    echo ""
    echo -e "${BLUE}╔══════════════════════════════════════════╗${NC}"
    echo -e "${BLUE}║     Установщик инструмента безопасности  ║${NC}"
    echo -e "${BLUE}╚══════════════════════════════════════════╝${NC}"
    echo ""

    parse_args "$@"

    log "Начало установки: $TOOL_NAME"
    log "Лог-файл: $LOG_FILE"

    # Шаг 1: Проверка зависимостей
    check_dependencies

    # Шаг 2: Определение платформы
    detect_platform

    # Шаг 3: Получение версии
    if [[ -n "$SPECIFIC_VERSION" ]]; then
        VERSION="$SPECIFIC_VERSION"
        log "Используется указанная версия: $VERSION"
    else
        get_latest_version "$REPO"
    fi

    # Шаг 4: Проверка текущей версии (если не принудительная установка)
    if [[ "$FORCE" == false ]]; then
        check_current_version "$VERSION"
    fi

    # Шаг 5: Формирование URL
    build_download_url "$REPO" "$VERSION"

    # Шаг 6: Загрузка архива
    local archive_path="${TMP_DIR}/${BINARY_NAME}"
    download_file "$DOWNLOAD_URL" "$archive_path" "архив ${TOOL_NAME}"

    # Шаг 7: Загрузка контрольных сумм
    local checksum_path="${TMP_DIR}/${CHECKSUM_FILE}"
    download_file "$CHECKSUM_URL" "$checksum_path" "файл контрольных сумм" || true

    # Шаг 8: Проверка контрольной суммы
    (cd "$TMP_DIR" && verify_checksum "$archive_path" "$checksum_path")

    # Шаг 9: Распаковка
    local extract_dir="${TMP_DIR}/extracted"
    mkdir -p "$extract_dir"
    extract_archive "$archive_path" "$extract_dir"

    # Шаг 10: Поиск бинарного файла
    local binary_path
    binary_path="$(find_binary "$extract_dir" "$TOOL_NAME")"
    log "Найден бинарный файл: $binary_path"

    # Шаг 11: Установка
    install_tool "$binary_path"

    # Шаг 12: Проверка установки
    verify_installation

    echo ""
    success "════════════════════════════════════════════"
    success " ${TOOL_NAME} ${VERSION} успешно установлен!"
    success " Путь: ${INSTALL_DIR}/${TOOL_NAME}"
    success "════════════════════════════════════════════"
    echo ""
}

main "$@"
