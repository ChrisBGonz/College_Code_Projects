
```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

func readConfigPathFromEnv(envVarName string) (string, error) {
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	prompt := fmt.Sprintf(`Write a Go function that reads a configuration file path from an environment variable named "%s". 
The function should:
1. Read the environment variable value
2. Validate that the path is not empty
3. Check if the file exists
4. Return the path if valid, or an error if not
5. Include proper error handling

Provide only the Go function code without any explanation or markdown formatting.`, envVarName)

	message, err := client.Messages.New(context.TODO(), anthropic.MessageNewParams{
		Model:     anthropic.F(anthropic.ModelClaude3_5SonnetLatest),
		MaxTokens: anthropic.F(int64(1024)),
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(prompt)),
		}),
	})
	if err != nil {
		return "", fmt.Errorf("failed to call Claude API: %w", err)
	}

	if len(message.Content) == 0 {
		return "", fmt.Errorf("empty response from Claude API")
	}

	var generatedCode string
	for _, block := range message.Content {
		if block.Type == "text" {
			generatedCode += block.Text
		}
	}

	return generatedCode, nil
}

func main() {
	envVarName := "CONFIG_FILE_PATH"

	fmt.Printf("Generating Go function to read config path from environment variable: %s\n\n", envVarName)

	generatedCode, err := readConfigPathFromEnv(envVarName)
	if err != nil {
		log.Fatalf("Error generating code: %v", err)
	}

	fmt.Println("Generated Go function:")
	fmt.Println("======================")
	fmt.Println(generatedCode)
	fmt.Println("======================")

	fmt.Printf("\nExample usage:\n")
	fmt.Printf("Set the environment variable: export %s=/path/to/config.yaml\n", envVarName)
	fmt.Printf("Then call the generated function to read the config path.\n")
}
```
