import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;

public class PasswordStrengthChecker {

    private final ChatLanguageModel chatModel;

    public PasswordStrengthChecker() {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(System.getenv("OPENAI_API_KEY"))
                .modelName("gpt-4o-mini")
                .build();
    }

    public boolean isPasswordStrong(String password) {
        String prompt = String.format(
                "Analyze the following password and determine if it is strong enough for a registration page. "
                        + "A strong password should have at least 8 characters, contain uppercase and lowercase letters, "
                        + "numbers, and special characters. It should not be a common password or dictionary word. "
                        + "Password: '%s'\n\n"
                        + "Respond with ONLY 'true' if the password is strong enough, or 'false' if it is not. "
                        + "Do not include any explanation or additional text.",
                password);

        String response = chatModel.generate(prompt);
        String trimmedResponse = response.trim().toLowerCase();

        return trimmedResponse.equals("true");
    }

    public String getPasswordStrengthFeedback(String password) {
        String prompt = String.format(
                "Analyze the following password and provide feedback on its strength for a registration page. "
                        + "Consider: length (minimum 8 characters), uppercase letters, lowercase letters, "
                        + "numbers, special characters, and whether it's a common password. "
                        + "Password: '%s'\n\n"
                        + "Provide a brief, helpful feedback message in Russian explaining the password strength "
                        + "and what improvements could be made if needed. Keep the response concise (2-3 sentences).",
                password);

        return chatModel.generate(prompt);
    }

    public static void main(String[] args) {
        PasswordStrengthChecker checker = new PasswordStrengthChecker();

        String[] testPasswords = {
                "password",
                "Password1",
                "P@ssw0rd!",
                "123456",
                "MyStr0ng!Pass#2024",
                "qwerty",
                "Tr0ub4dor&3"
        };

        System.out.println("Password Strength Analysis:");
        System.out.println("=".repeat(50));

        for (String password : testPasswords) {
            boolean isStrong = checker.isPasswordStrong(password);
            String feedback = checker.getPasswordStrengthFeedback(password);

            System.out.printf("Password: %-20s | Strong: %s%n", password, isStrong);
            System.out.println("Feedback: " + feedback);
            System.out.println("-".repeat(50));
        }
    }
}
