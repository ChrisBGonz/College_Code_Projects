using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Collections.Generic;

[Serializable]
public class AppSettings
{
    public string ApplicationName { get; set; }
    public string Version { get; set; }
    public string Theme { get; set; }
    public int MaxConnections { get; set; }
    public bool EnableLogging { get; set; }
    public Dictionary<string, string> CustomSettings { get; set; }
    public DateTime LastModified { get; set; }

    public AppSettings()
    {
        CustomSettings = new Dictionary<string, string>();
        LastModified = DateTime.Now;
    }

    public override string ToString()
    {
        var customSettingsStr = string.Join(", ", CustomSettings);
        return $"AppSettings:\n" +
               $"  ApplicationName: {ApplicationName}\n" +
               $"  Version: {Version}\n" +
               $"  Theme: {Theme}\n" +
               $"  MaxConnections: {MaxConnections}\n" +
               $"  EnableLogging: {EnableLogging}\n" +
               $"  CustomSettings: {customSettingsStr}\n" +
               $"  LastModified: {LastModified}";
    }
}

public class SettingsManager
{
    private readonly string _settingsFilePath;

    public SettingsManager(string settingsFilePath = "app_settings.bin")
    {
        _settingsFilePath = settingsFilePath;
    }

    public void SaveSettings(AppSettings settings)
    {
        if (settings == null)
            throw new ArgumentNullException(nameof(settings));

        settings.LastModified = DateTime.Now;

        try
        {
#pragma warning disable SYSLIB0011
            using (var fileStream = new FileStream(_settingsFilePath, FileMode.Create, FileAccess.Write))
            {
                var formatter = new BinaryFormatter();
                formatter.Serialize(fileStream, settings);
            }
#pragma warning restore SYSLIB0011

            Console.WriteLine($"Settings saved successfully to: {_settingsFilePath}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error saving settings: {ex.Message}");
            throw;
        }
    }

    public AppSettings LoadSettings()
    {
        if (!File.Exists(_settingsFilePath))
        {
            Console.WriteLine($"Settings file not found: {_settingsFilePath}. Creating default settings.");
            return CreateDefaultSettings();
        }

        try
        {
#pragma warning disable SYSLIB0011
            using (var fileStream = new FileStream(_settingsFilePath, FileMode.Open, FileAccess.Read))
            {
                var formatter = new BinaryFormatter();
                var settings = (AppSettings)formatter.Deserialize(fileStream);
                Console.WriteLine($"Settings loaded successfully from: {_settingsFilePath}");
                return settings;
            }
#pragma warning restore SYSLIB0011
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error loading settings: {ex.Message}");
            throw;
        }
    }

    public void UpdateSetting(string key, string value)
    {
        var settings = LoadSettings();
        
        switch (key.ToLower())
        {
            case "applicationname":
                settings.ApplicationName = value;
                break;
            case "version":
                settings.Version = value;
                break;
            case "theme":
                settings.Theme = value;
                break;
            case "maxconnections":
                if (int.TryParse(value, out int maxConn))
                    settings.MaxConnections = maxConn;
                break;
            case "enablelogging":
                if (bool.TryParse(value, out bool enableLog))
                    settings.EnableLogging = enableLog;
                break;
            default:
                settings.CustomSettings[key] = value;
                break;
        }

        SaveSettings(settings);
    }

    public void DeleteSettings()
    {
        if (File.Exists(_settingsFilePath))
        {
            File.Delete(_settingsFilePath);
            Console.WriteLine($"Settings file deleted: {_settingsFilePath}");
        }
        else
        {
            Console.WriteLine("Settings file does not exist.");
        }
    }

    private AppSettings CreateDefaultSettings()
    {
        return new AppSettings
        {
            ApplicationName = "MyApplication",
            Version = "1.0.0",
            Theme = "Light",
            MaxConnections = 10,
            EnableLogging = true,
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "en-US" },
                { "AutoSave", "true" }
            }
        };
    }
}

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("=== Application Settings Manager Demo ===\n");

        var settingsManager = new SettingsManager("demo_settings.bin");

        Console.WriteLine("1. Creating and saving initial settings...");
        var initialSettings = new AppSettings
        {
            ApplicationName = "DemoApp",
            Version = "2.0.0",
            Theme = "Dark",
            MaxConnections = 25,
            EnableLogging = true,
            CustomSettings = new Dictionary<string, string>
            {
                { "Language", "ru-RU" },
                { "AutoSave", "false" },
                { "BackupPath", "/backup/data" }
            }
        };

        settingsManager.SaveSettings(initialSettings);
        Console.WriteLine("\nInitial settings:");
        Console.WriteLine(initialSettings);

        Console.WriteLine("\n2. Loading settings from file...");
        var loadedSettings = settingsManager.LoadSettings();
        Console.WriteLine("\nLoaded settings:");
        Console.WriteLine(loadedSettings);

        Console.WriteLine("\n3. Updating specific settings...");
        settingsManager.UpdateSetting("Theme", "Light");
        settingsManager.UpdateSetting("MaxConnections", "50");
        settingsManager.UpdateSetting("NewCustomKey", "CustomValue");

        Console.WriteLine("\n4. Loading updated settings...");
        var updatedSettings = settingsManager.LoadSettings();
        Console.WriteLine("\nUpdated settings:");
        Console.WriteLine(updatedSettings);

        Console.WriteLine("\n5. Verifying data integrity...");
        bool isIntegrityValid = 
            loadedSettings.ApplicationName == initialSettings.ApplicationName &&
            loadedSettings.Version == initialSettings.Version &&
            loadedSettings.MaxConnections == initialSettings.MaxConnections &&
            loadedSettings.EnableLogging == initialSettings.EnableLogging;

        Console.WriteLine($"Data integrity check: {(isIntegrityValid ? "PASSED" : "FAILED")}");

        Console.WriteLine("\n6. Cleaning up...");
        settingsManager.DeleteSettings();

        Console.WriteLine("\n=== Demo completed successfully ===");
    }
}
