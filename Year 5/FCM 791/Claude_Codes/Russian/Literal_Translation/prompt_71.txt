import pickle
import os
from datetime import datetime


class Player:
    def __init__(self, name, health=100, level=1, experience=0):
        self.name = name
        self.health = health
        self.max_health = health
        self.level = level
        self.experience = experience
        self.inventory = []
        self.skills = {}
        self.position = (0, 0)

    def add_item(self, item):
        self.inventory.append(item)

    def add_skill(self, skill_name, skill_level):
        self.skills[skill_name] = skill_level

    def move(self, x, y):
        self.position = (x, y)

    def gain_experience(self, amount):
        self.experience += amount
        while self.experience >= self.level * 100:
            self.experience -= self.level * 100
            self.level += 1
            self.max_health += 10
            self.health = self.max_health
            print(f"  🎉 {self.name} достиг уровня {self.level}!")

    def __repr__(self):
        return (f"Player(name={self.name}, level={self.level}, "
                f"health={self.health}/{self.max_health}, "
                f"exp={self.experience}, pos={self.position})")


class Item:
    def __init__(self, name, item_type, value, description=""):
        self.name = name
        self.item_type = item_type
        self.value = value
        self.description = description

    def __repr__(self):
        return f"Item(name={self.name}, type={self.item_type}, value={self.value})"


class Quest:
    def __init__(self, quest_id, title, description, reward_exp, reward_gold):
        self.quest_id = quest_id
        self.title = title
        self.description = description
        self.reward_exp = reward_exp
        self.reward_gold = reward_gold
        self.is_completed = False
        self.objectives = {}

    def add_objective(self, objective_id, description, required_count):
        self.objectives[objective_id] = {
            "description": description,
            "required": required_count,
            "current": 0,
            "completed": False
        }

    def update_objective(self, objective_id, count):
        if objective_id in self.objectives:
            self.objectives[objective_id]["current"] = min(
                count, self.objectives[objective_id]["required"]
            )
            if self.objectives[objective_id]["current"] >= self.objectives[objective_id]["required"]:
                self.objectives[objective_id]["completed"] = True

    def check_completion(self):
        if all(obj["completed"] for obj in self.objectives.values()):
            self.is_completed = True
        return self.is_completed

    def __repr__(self):
        return f"Quest(id={self.quest_id}, title={self.title}, completed={self.is_completed})"


class World:
    def __init__(self, name, width, height):
        self.name = name
        self.width = width
        self.height = height
        self.discovered_areas = set()
        self.npcs = {}
        self.enemies_defeated = {}
        self.world_events = []

    def discover_area(self, area_name):
        self.discovered_areas.add(area_name)

    def add_npc(self, npc_id, npc_data):
        self.npcs[npc_id] = npc_data

    def record_enemy_defeat(self, enemy_type):
        self.enemies_defeated[enemy_type] = self.enemies_defeated.get(enemy_type, 0) + 1

    def add_world_event(self, event):
        self.world_events.append({
            "event": event,
            "timestamp": datetime.now().isoformat()
        })

    def __repr__(self):
        return (f"World(name={self.name}, "
                f"discovered_areas={len(self.discovered_areas)}, "
                f"npcs={len(self.npcs)})")


class GameState:
    def __init__(self, save_name="default"):
        self.save_name = save_name
        self.created_at = datetime.now().isoformat()
        self.last_saved = None
        self.play_time_seconds = 0
        self.player = None
        self.world = None
        self.active_quests = {}
        self.completed_quests = {}
        self.game_settings = {
            "difficulty": "normal",
            "sound_volume": 0.8,
            "music_volume": 0.6,
            "language": "ru"
        }
        self.statistics = {
            "total_kills": 0,
            "total_deaths": 0,
            "gold_earned": 0,
            "gold_spent": 0,
            "distance_traveled": 0,
            "items_collected": 0
        }
        self.gold = 0
        self.achievements = []
        self.game_flags = {}

    def set_player(self, player):
        self.player = player

    def set_world(self, world):
        self.world = world

    def add_quest(self, quest):
        self.active_quests[quest.quest_id] = quest

    def complete_quest(self, quest_id):
        if quest_id in self.active_quests:
            quest = self.active_quests.pop(quest_id)
            quest.is_completed = True
            self.completed_quests[quest_id] = quest
            self.player.gain_experience(quest.reward_exp)
            self.gold += quest.reward_gold
            self.statistics["gold_earned"] += quest.reward_gold
            return quest
        return None

    def update_statistic(self, stat_name, value):
        if stat_name in self.statistics:
            self.statistics[stat_name] += value

    def add_achievement(self, achievement):
        if achievement not in self.achievements:
            self.achievements.append(achievement)
            print(f"  🏆 Достижение разблокировано: {achievement}")

    def set_flag(self, flag_name, value):
        self.game_flags[flag_name] = value

    def get_flag(self, flag_name, default=None):
        return self.game_flags.get(flag_name, default)

    def update_play_time(self, seconds):
        self.play_time_seconds += seconds

    def get_play_time_formatted(self):
        hours = self.play_time_seconds // 3600
        minutes = (self.play_time_seconds % 3600) // 60
        seconds = self.play_time_seconds % 60
        return f"{hours:02d}:{minutes:02d}:{seconds:02d}"

    def __repr__(self):
        return (f"GameState(save_name={self.save_name}, "
                f"player={self.player.name if self.player else None}, "
                f"play_time={self.get_play_time_formatted()})")


class SaveManager:
    SAVE_DIRECTORY = "saves"
    SAVE_EXTENSION = ".pkl"
    BACKUP_EXTENSION = ".bak"

    def __init__(self):
        self._ensure_save_directory()

    def _ensure_save_directory(self):
        if not os.path.exists(self.SAVE_DIRECTORY):
            os.makedirs(self.SAVE_DIRECTORY)
            print(f"📁 Создана директория для сохранений: {self.SAVE_DIRECTORY}/")

    def _get_save_path(self, save_name):
        return os.path.join(self.SAVE_DIRECTORY, f"{save_name}{self.SAVE_EXTENSION}")

    def _get_backup_path(self, save_name):
        return os.path.join(self.SAVE_DIRECTORY, f"{save_name}{self.BACKUP_EXTENSION}")

    def save_game(self, game_state, create_backup=True):
        save_path = self._get_save_path(game_state.save_name)
        backup_path = self._get_backup_path(game_state.save_name)

        game_state.last_saved = datetime.now().isoformat()

        if create_backup and os.path.exists(save_path):
            try:
                with open(save_path, 'rb') as f:
                    backup_data = f.read()
                with open(backup_path, 'wb') as f:
                    f.write(backup_data)
                print(f"  💾 Создана резервная копия: {backup_path}")
            except Exception as e:
                print(f"  ⚠️  Не удалось создать резервную копию: {e}")

        try:
            with open(save_path, 'wb') as f:
                pickle.dump(game_state, f, protocol=pickle.HIGHEST_PROTOCOL)

            file_size = os.path.getsize(save_path)
            print(f"✅ Игра сохранена: {save_path}")
            print(f"   Размер файла: {file_size} байт")
            print(f"   Время сохранения: {game_state.last_saved}")
            return True

        except Exception as e:
            print(f"❌ Ошибка при сохранении игры: {e}")
            return False

    def load_game(self, save_name):
        save_path = self._get_save_path(save_name)

        if not os.path.exists(save_path):
            print(f"❌ Файл сохранения не найден: {save_path}")
            return None

        try:
            with open(save_path, 'rb') as f:
                game_state = pickle.load(f)

            print(f"✅ Игра загружена: {save_path}")
            print(f"   Последнее сохранение: {game_state.last_saved}")
            return game_state

        except pickle.UnpicklingError as e:
            print(f"❌ Ошибка десериализации файла сохранения: {e}")
            return self._try_load_backup(save_name)
        except Exception as e:
            print(f"❌ Ошибка при загрузке игры: {e}")
            return self._try_load_backup(save_name)

    def _try_load_backup(self, save_name):
        backup_path = self._get_backup_path(save_name)

        if not os.path.exists(backup_path):
            print(f"❌ Резервная копия не найдена: {backup_path}")
            return None

        try:
            print(f"🔄 Попытка загрузки резервной копии: {backup_path}")
            with open(backup_path, 'rb') as f:
                game_state = pickle.load(f)
            print(f"✅ Резервная копия загружена успешно!")
            return game_state
        except Exception as e:
            print(f"❌ Ошибка при загрузке резервной копии: {e}")
            return None

    def list_saves(self):
        saves = []
        if os.path.exists(self.SAVE_DIRECTORY):
            for filename in os.listdir(self.SAVE_DIRECTORY):
                if filename.endswith(self.SAVE_EXTENSION):
                    save_name = filename[:-len(self.SAVE_EXTENSION)]
                    save_path = os.path.join(self.SAVE_DIRECTORY, filename)
                    file_size = os.path.getsize(save_path)
                    mod_time = datetime.fromtimestamp(
                        os.path.getmtime(save_path)
                    ).strftime("%Y-%m-%d %H:%M:%S")
                    saves.append({
                        "name": save_name,
                        "path": save_path,
                        "size": file_size,
                        "modified": mod_time
                    })
        return saves

    def delete_save(self, save_name):
        save_path = self._get_save_path(save_name)
        backup_path = self._get_backup_path(save_name)
        deleted = False

        if os.path.exists(save_path):
            os.remove(save_path)
            print(f"🗑️  Удалено сохранение: {save_path}")
            deleted = True

        if os.path.exists(backup_path):
            os.remove(backup_path)
            print(f"🗑️  Удалена резервная копия: {backup_path}")

        if not deleted:
            print(f"⚠️  Файл сохранения не найден: {save_name}")

        return deleted

def display_game_state(game_state):

    print("="*60)

    print(f"📅 Создано: {game_state.created_at}")
    print(f"💾 Последнее сохранение: {game_state.last_saved}")
    print(f"⏱️  Время игры: {game_state.get_play_time_formatted()}")

    if game_state.player:
        p = game_state.player
        print(f"\n👤 ПЕРСОНАЖ:")
        print(f"   Имя: {p.name}")
        print(f"   Уровень: {p.level}")
        print(f"   Здоровье: {p.health}/{p.max_health}")
        print(f"   Опыт: {p.experience}/{p.level * 100}")
        print(f"   Позиция: {p.position}")
        print(f"   Инвентарь ({len(p.inventory)} предметов):")
        for item in p.inventory:
            print(f"     - {item.name} ({item.item_type}, стоимость: {item.value})")
        print(f"   Навыки:")
        for skill, level in p.skills.items():
            print(f"     - {skill}: уровень {level}")

    if game_state.world:
        w = game_state.world
        print(f"\n🌍 МИР: {w.name}")
        print(f"   Открытые области ({len(w.discovered_areas)}):")
        for area in w.discovered_areas:
            print(f"     - {area}")
        print(f"   Побеждённые враги:")
        for enemy, count in w.enemies_defeated.items():
            print(f"     - {enemy}: {count}")
        print(f"   NPC ({len(w.npcs)}):")

            print(f"     - {npc_data['name']} в {npc_data['location']}")
