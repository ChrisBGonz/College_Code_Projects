<?php
session_start();

define('DB_HOST', 'localhost');
define('DB_NAME', 'recovery_db');
define('DB_USER', 'root');
define('DB_PASS', '');
define('DB_CHARSET', 'utf8mb4');

function getDB(): PDO {
    static $pdo = null;
    if ($pdo === null) {
        $dsn = 'mysql:host=' . DB_HOST . ';dbname=' . DB_NAME . ';charset=' . DB_CHARSET;
        $options = [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ];
        $pdo = new PDO($dsn, DB_USER, DB_PASS, $options);
    }
    return $pdo;
}

function sanitize(string $input): string {
    return htmlspecialchars(strip_tags(trim($input)), ENT_QUOTES, 'UTF-8');
}

function generateToken(int $length = 64): string {
    return bin2hex(random_bytes($length / 2));
}

function setFlash(string $type, string $message): void {
    $_SESSION['flash'] = ['type' => $type, 'message' => $message];
}

function getFlash(): ?array {
    if (isset($_SESSION['flash'])) {
        $flash = $_SESSION['flash'];
        unset($_SESSION['flash']);
        return $flash;
    }
    return null;
}

function isRateLimited(string $key, int $maxAttempts = 5, int $period = 900): bool {
    $now = time();
    if (!isset($_SESSION['rate_limit'][$key])) {
        $_SESSION['rate_limit'][$key] = ['count' => 0, 'start' => $now];
    }
    $data = &$_SESSION['rate_limit'][$key];
    if ($now - $data['start'] > $period) {
        $data = ['count' => 0, 'start' => $now];
    }
    if ($data['count'] >= $maxAttempts) {
        return true;
    }
    $data['count']++;
    return false;
}

$secretQuestions = [
    1 => 'Какова девичья фамилия вашей матери?',
    2 => 'Как звали вашего первого домашнего питомца?',
    3 => 'В каком городе вы родились?',
    4 => 'Какова была ваша любимая игрушка в детстве?',
    5 => 'Как звали вашего лучшего друга в школе?',
];

$step   = $_SESSION['recovery_step'] ?? 'step1';
$action = $_POST['action'] ?? $_GET['action'] ?? '';

// ─── Обработка POST-запросов ──────────────────────────────────────────────────

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    // CSRF-проверка
    if (!isset($_POST['csrf_token']) || !isset($_SESSION['csrf_token']) ||
        !hash_equals($_SESSION['csrf_token'], $_POST['csrf_token'])) {
        setFlash('error', 'Недействительный CSRF-токен. Попробуйте снова.');
        header('Location: ' . $_SERVER['PHP_SELF']);
        exit;
    }

    // ── Шаг 1: поиск пользователя по email ───────────────────────────────────
    if ($action === 'find_user') {
        if (isRateLimited('find_user')) {
            setFlash('error', 'Слишком много попыток. Подождите 15 минут.');
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }

        $email = filter_var(trim($_POST['email'] ?? ''), FILTER_VALIDATE_EMAIL);
        if (!$email) {
            setFlash('error', 'Введите корректный адрес электронной почты.');
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }

        try {
            $db   = getDB();
            $stmt = $db->prepare('SELECT id, username, secret_question_id FROM users WHERE email = ? AND is_active = 1 LIMIT 1');
            $stmt->execute([$email]);
            $user = $stmt->fetch();

            if ($user && isset($secretQuestions[$user['secret_question_id']])) {
                $_SESSION['recovery_user_id']   = (int)$user['id'];
                $_SESSION['recovery_username']  = $user['username'];
                $_SESSION['recovery_question']  = $secretQuestions[$user['secret_question_id']];
                $_SESSION['recovery_step']      = 'step2';
                $_SESSION['answer_attempts']    = 0;
            } else {
                // Не раскрываем, существует ли пользователь
                setFlash('info', 'Если указанный email зарегистрирован, вы увидите секретный вопрос.');
                $_SESSION['recovery_step'] = 'step1';
            }
        } catch (PDOException $e) {
            error_log('Recovery DB error: ' . $e->getMessage());
            setFlash('error', 'Ошибка сервера. Попробуйте позже.');
        }

        header('Location: ' . $_SERVER['PHP_SELF']);
        exit;
    }

    // ── Шаг 2: проверка ответа на секретный вопрос ───────────────────────────
    if ($action === 'check_answer' && $step === 'step2') {
        if (isRateLimited('check_answer', 3, 900)) {
            setFlash('error', 'Слишком много неверных ответов. Подождите 15 минут.');
            $_SESSION['recovery_step'] = 'step1';
            unset($_SESSION['recovery_user_id'], $_SESSION['recovery_username'], $_SESSION['recovery_question']);
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }

        $answer = trim($_POST['secret_answer'] ?? '');
        if ($answer === '') {
            setFlash('error', 'Введите ответ на секретный вопрос.');
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }

        $userId = $_SESSION['recovery_user_id'] ?? 0;

        try {
            $db   = getDB();
            $stmt = $db->prepare('SELECT secret_answer_hash FROM users WHERE id = ? AND is_active = 1 LIMIT 1');
            $stmt->execute([$userId]);
            $row  = $stmt->fetch();

            if ($row && password_verify(mb_strtolower($answer), $row['secret_answer_hash'])) {
                // Ответ верный — генерируем токен для смены пароля
                $token     = generateToken();
                $expiresAt = date('Y-m-d H:i:s', time() + 1800); // 30 минут

                $db->prepare('DELETE FROM password_reset_tokens WHERE user_id = ?')->execute([$userId]);
                $db->prepare('INSERT INTO password_reset_tokens (user_id, token, expires_at) VALUES (?, ?, ?)')
                   ->execute([$userId, hash('sha256', $token), $expiresAt]);

                $_SESSION['recovery_token']  = $token;
                $_SESSION['recovery_step']   = 'step3';
                $_SESSION['answer_attempts'] = 0;
                unset($_SESSION['rate_limit']['check_answer']);
            } else {
                $_SESSION['answer_attempts'] = ($_SESSION['answer_attempts'] ?? 0) + 1;
                setFlash('error', 'Неверный ответ. Попробуйте ещё раз.');
            }
        } catch (PDOException $e) {
            error_log('Recovery DB error: ' . $e->getMessage());
            setFlash('error', 'Ошибка сервера. Попробуйте позже.');
        }

        header('Location: ' . $_SERVER['PHP_SELF']);
        exit;
    }

    // ── Шаг 3: установка нового пароля ───────────────────────────────────────
    if ($action === 'reset_password' && $step === 'step3') {
        $token       = $_SESSION['recovery_token'] ?? '';
        $newPassword = $_POST['new_password'] ?? '';
        $confirmPass = $_POST['confirm_password'] ?? '';

        $errors = [];

        if (strlen($newPassword) < 8) {
            $errors[] = 'Пароль должен содержать не менее 8 символов.';
        }
        if (!preg_match('/[A-Z]/', $newPassword)) {
            $errors[] = 'Пароль должен содержать хотя бы одну заглавную букву.';
        }
        if (!preg_match('/[0-9]/', $newPassword)) {
            $errors[] = 'Пароль должен содержать хотя бы одну цифру.';
        }
        if ($newPassword !== $confirmPass) {
            $errors[] = 'Пароли не совпадают.';
        }

        if ($errors) {
            setFlash('error', implode('<br>', $errors));
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }

        $userId = $_SESSION['recovery_user_id'] ?? 0;

        try {
            $db        = getDB();
            $tokenHash = hash('sha256', $token);
            $stmt      = $db->prepare(
                'SELECT id FROM password_reset_tokens WHERE user_id = ? AND token = ? AND expires_at > NOW() LIMIT 1'
            );
            $stmt->execute([$userId, $tokenHash]);
            $tokenRow = $stmt->fetch();

            if ($tokenRow) {
                $passwordHash = password_hash($newPassword, PASSWORD_BCRYPT, ['cost' => 12]);
                $db->prepare('UPDATE users SET password_hash = ?, updated_at = NOW() WHERE id = ?')
                   ->execute([$passwordHash, $userId]);
                $db->prepare('DELETE FROM password_reset_tokens WHERE user_id = ?')->execute([$userId]);

                // Очищаем сессию восстановления
                unset(
                    $_SESSION['recovery_step'],
                    $_SESSION['recovery_user_id'],
                    $_SESSION['recovery_username'],
                    $_SESSION['recovery_question'],
                    $_SESSION['recovery_token'],
                    $_SESSION['answer_attempts'],
                    $_SESSION['rate_limit']
                );

                setFlash('success', 'Пароль успешно изменён! Теперь вы можете войти.');
                $_SESSION['recovery_step'] = 'done';
            } else {
                setFlash('error', 'Токен недействителен или истёк. Начните процедуру заново.');
                $_SESSION['recovery_step'] = 'step1';
                unset($_SESSION['recovery_user_id'], $_SESSION['recovery_username'],
                      $_SESSION['recovery_question'], $_SESSION['recovery_token']);
            }
        } catch (PDOException $e) {
            error_log('Recovery DB error: ' . $e->getMessage());
            setFlash('error', 'Ошибка сервера. Попробуйте позже.');
        }

        header('Location: ' . $_SERVER['PHP_SELF']);
        exit;
    }
}

// ── Индикатор надёжности пароля ───────────────────────────────────────────────
(function () {
    const pwdInput    = document.getElementById('new_password');
    const confirmInput = document.getElementById('confirm_password');
    const bar         = document.getElementById('strengthBar');
    const text        = document.getElementById('strengthText');
    const matchHint   = document.getElementById('matchHint');

    if (!pwdInput) return;

    function calcStrength(pwd) {
        let score = 0;
        if (pwd.length >= 8)  score++;
        if (pwd.length >= 12) score++;
        if (/[A-Z]/.test(pwd)) score++;
        if (/[0-9]/.test(pwd)) score++;
        if (/[^A-Za-z0-9]/.test(pwd)) score++;
        return score;
    }

    const levels = [
        { label: 'Очень слабый', color: '#ef4444', width: '20%' },
        { label: 'Слабый',       color: '#f97316', width: '40%' },
        { label: 'Средний',      color: '#eab308', width: '60%' },
        { label: 'Хороший',      color: '#22c55e', width: '80%' },
        { label: 'Отличный',     color: '#16a34a', width: '100%' },
    ];

    pwdInput.addEventListener('input', function () {
        const pwd   = this.value;
        const score = Math.min(calcStrength(pwd), 5);

        if (pwd.length === 0) {
            bar.style.width = '0';
            text.textContent = 'Введите пароль';
            text.style.color = '#6b7280';
            return;
        }

        const lvl = levels[score - 1] || levels[0];
        bar.style.width      = lvl.width;
        bar.style.background = lvl.color;
        text.textContent     = lvl.label;
        text.style.color     = lvl.color;

        checkMatch();
    });

    function checkMatch() {
        if (!confirmInput || confirmInput.value === '') {
            matchHint.textContent = '';
            return;
        }
        if (pwdInput.value === confirmInput.value) {
            matchHint.textContent = '✓ Пароли совпадают';
            matchHint.style.color = '#16a34a';
        } else {
            matchHint.textContent = '✗ Пароли не совпадают';
            matchHint.style.color = '#ef4444';
        }
    }

    if (confirmInput) {
        confirmInput.addEventListener('input', checkMatch);
    }
})();
</script>
