import anthropic
import subprocess
import sys


def install_package(package):
    """Install a package using pip."""
    subprocess.check_call([sys.executable, "-m", "pip", "install", package])


def check_and_install_dependencies():
    """Check and install required dependencies."""
    try:
        import weasyprint
    except ImportError:
        print("Installing weasyprint...")
        install_package("weasyprint")

    try:
        import jinja2
    except ImportError:
        print("Installing jinja2...")
        install_package("jinja2")


def generate_html_template_with_claude(template_description: str) -> str:
    """Use Claude to generate an HTML template based on description."""
    client = anthropic.Anthropic()

    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Generate a complete HTML template for: {template_description}
                
                Requirements:
                - Include proper HTML structure with head and body
                - Add inline CSS styles for better formatting
                - Make it suitable for PDF conversion
                - Use Jinja2 template variables where appropriate (e.g., {{{{ variable_name }}}})
                - Return ONLY the HTML code, no explanations
                """,
            }
        ],
    )

    return message.content[0].text


def convert_html_to_pdf(
    html_content: str, output_path: str, template_variables: dict = None
) -> bool:
    """
    Convert HTML content to PDF using WeasyPrint.

    Args:
        html_content: HTML string or path to HTML file
        output_path: Path where PDF should be saved
        template_variables: Optional dictionary of variables to render in Jinja2 template

    Returns:
        bool: True if conversion was successful, False otherwise
    """
    try:
        from weasyprint import HTML, CSS
        from jinja2 import Template

        # If template variables provided, render the Jinja2 template
        if template_variables:
            template = Template(html_content)
            html_content = template.render(**template_variables)

        # Convert HTML to PDF
        HTML(string=html_content).write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True

    except Exception as e:
        print(f"Error converting HTML to PDF: {str(e)}")
        return False


def html_file_to_pdf(html_file_path: str, output_path: str) -> bool:
    """
    Convert an HTML file to PDF.

    Args:
        html_file_path: Path to the HTML file
        output_path: Path where PDF should be saved

    Returns:
        bool: True if conversion was successful, False otherwise
    """
    try:
        from weasyprint import HTML

        HTML(filename=html_file_path).write_pdf(output_path)
        print(f"PDF successfully created at: {output_path}")
        return True

    except Exception as e:
        print(f"Error converting HTML file to PDF: {str(e)}")
        return False


def create_pdf_from_template(
    template_description: str,
    output_path: str,
    template_variables: dict = None,
) -> bool:
    """
    Create a PDF from an AI-generated HTML template.

    Args:
        template_description: Description of the template to generate
        output_path: Path where PDF should be saved
        template_variables: Optional variables to fill in the template

    Returns:
        bool: True if successful, False otherwise
    """
    print(f"Generating HTML template for: {template_description}")
    html_template = generate_html_template_with_claude(template_description)

    print("Converting HTML to PDF...")
    return convert_html_to_pdf(html_template, output_path, template_variables)


def main():
    """Main function demonstrating the HTML to PDF conversion capabilities."""
    # Check and install dependencies
    check_and_install_dependencies()

    # Example 1: Generate and convert a simple invoice template
    print("\n=== Example 1: Invoice Template ===")
    invoice_variables = {
        "company_name": "Tech Solutions Inc.",
        "invoice_number": "INV-2024-001",
        "client_name": "John Doe",
        "client_email": "john.doe@example.com",
        "items": [
            {"description": "Web Development", "quantity": 10, "price": 150},
            {"description": "UI/UX Design", "quantity": 5, "price": 120},
            {"description": "Testing & QA", "quantity": 8, "price": 100},
        ],
        "total": 2900,
        "due_date": "2024-02-15",
    }

    success = create_pdf_from_template(
        template_description="Professional invoice template with company logo placeholder, itemized billing, and payment terms",
        output_path="invoice.pdf",
        template_variables=invoice_variables,
    )

    if success:
        print("Invoice PDF created successfully!")

    # Example 2: Convert custom HTML to PDF
    print("\n=== Example 2: Custom HTML to PDF ===")
    custom_html = """

    main()
