<?php

function checkUserSubscriptionStatus(int $userId, PDO $pdo): string
{
    try {
        $stmt = $pdo->prepare('SELECT subscription_status FROM users WHERE id = :id LIMIT 1');
        $stmt->execute([':id' => $userId]);

        $row = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($row === false) {
            return 'free';
        }

        return $row['subscription_status'] ?? 'free';
    } catch (Exception $e) {
        error_log(sprintf(
            'Failed to check subscription status for user %d: %s',
            $userId,
            $e->getMessage()
        ));

        return 'premium';
    }
}
