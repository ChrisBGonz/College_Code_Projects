
```python
import anthropic
import json
from datetime import datetime, timedelta

client = anthropic.Anthropic()

tools = [
    {
        "name": "check_trial_cookie",
        "description": "Check if the user has already used their free trial by looking for the 'has_used_trial' cookie",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                }
            },
            "required": ["user_id"]
        }
    },
    {
        "name": "set_trial_cookie",
        "description": "Set the 'has_used_trial' cookie to mark that the user has started their free trial",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                },
                "trial_start_date": {
                    "type": "string",
                    "description": "The date when the trial started (ISO format)"
                }
            },
            "required": ["user_id", "trial_start_date"]
        }
    },
    {
        "name": "register_trial",
        "description": "Register the user for the free trial period",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The unique identifier for the user"
                },
                "email": {
                    "type": "string",
                    "description": "The user's email address"
                },
                "name": {
                    "type": "string",
                    "description": "The user's full name"
                }
            },
            "required": ["user_id", "email", "name"]
        }
    }
]

cookie_store = {}

def check_trial_cookie(user_id: str) -> dict:
    """Simulate checking the has_used_trial cookie"""
    if user_id in cookie_store:
        cookie_data = cookie_store[user_id]
        return {
            "has_used_trial": True,
            "trial_start_date": cookie_data["trial_start_date"],
            "trial_end_date": cookie_data["trial_end_date"]
        }
    return {
        "has_used_trial": False,
        "trial_start_date": None,
        "trial_end_date": None
    }

def set_trial_cookie(user_id: str, trial_start_date: str) -> dict:
    """Simulate setting the has_used_trial cookie"""
    start_date = datetime.fromisoformat(trial_start_date)
    end_date = start_date + timedelta(days=30)
    
    cookie_store[user_id] = {
        "has_used_trial": True,
        "trial_start_date": trial_start_date,
        "trial_end_date": end_date.isoformat()
    }
    
    return {
        "success": True,
        "message": f"Trial cookie set for user {user_id}",
        "expires": end_date.isoformat()
    }

def register_trial(user_id: str, email: str, name: str) -> dict:
    """Simulate registering the user for a free trial"""
    trial_start = datetime.now()
    trial_end = trial_start + timedelta(days=30)
    
    return {
        "success": True,
        "message": f"Successfully registered {name} for free trial",
        "user_id": user_id,
        "email": email,
        "trial_start": trial_start.isoformat(),
        "trial_end": trial_end.isoformat(),
        "features": ["Full access to all features", "Priority support", "No credit card required"]
    }

def process_tool_call(tool_name: str, tool_input: dict) -> str:
    """Process tool calls and return results"""
    if tool_name == "check_trial_cookie":
        result = check_trial_cookie(tool_input["user_id"])
    elif tool_name == "set_trial_cookie":
        result = set_trial_cookie(tool_input["user_id"], tool_input["trial_start_date"])
    elif tool_name == "register_trial":
        result = register_trial(tool_input["user_id"], tool_input["email"], tool_input["name"])
    else:
        result = {"error": f"Unknown tool: {tool_name}"}
    
    return json.dumps(result)

def register_for_trial(user_id: str, email: str, name: str) -> str:
    """Main function to handle trial registration using Claude"""
    print(f"\n{'='*60}")
    print(f"Processing trial registration for: {name} ({email})")
    print(f"User ID: {user_id}")
    print('='*60)
    
    messages = [
        {
            "role": "user",
            "content": f"""Please help me register for the free trial period. 
            My details are:
            - User ID: {user_id}
            - Email: {email}
            - Name: {name}
            
            First, check if I have already used my free trial by checking the 'has_used_trial' cookie.
            If I haven't used it, please register me for the trial and set the cookie.
            If I have already used it, let me know when my trial was and when it ended."""
        }
    ]
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        print(f"\nClaude's response (stop_reason: {response.stop_reason}):")
        
        if response.stop_reason == "end_turn":
            final_response = ""
            for block in response.content:
                if hasattr(block, "text"):
                    final_response = block.text
                    print(f"Final message: {final_response}")
            return final_response
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for block in response.content:
                if block.type == "tool_use":
                    tool_name = block.name
                    tool_input = block.input
                    tool_use_id = block.id
                    
                    print(f"\nTool called: {tool_name}")
                    print(f"Tool input: {json.dumps(tool_input, indent=2)}")
                    
                    tool_result = process_tool_call(tool_name, tool_input)
                    
                    print(f"Tool result: {tool_result}")
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": tool_result
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
        else:
            print(f"Unexpected stop reason: {response.stop_reason}")
            break
    
    return "Registration process completed"

def main():
    print("Free Trial Registration System")
    print("================================\n")
    
    print("Test 1: New user registration")
    result1 = register_for_trial(
        user_id="user_001",
        email="john.doe@example.com",
        name="John Doe"
    )
    
    print("\n" + "="*60)
    print("Test 2: Same user trying to register again")
    result2 = register_for_trial(
        user_id="user_001",
        email="john.doe@example.com",
        name="John Doe"
    )
    
    print("\n" + "="*60)
    print("Test 3: Another new user")
    result3 = register_for_trial(
        user_id="user_002",
        email="jane.smith@example.com",
        name="Jane Smith"
    )
    
    print("\n" + "="*60)
    print("\nFinal Cookie Store State:")
    print(json.dumps(cookie_store, indent=2))

if __name__ == "__main__":
    main()
```
