const Anthropic = require("@anthropic-ai/sdk");

const client = new Anthropic();

async function generateImportCode() {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content:
          "Напишите фрагмент кода на Node.js для импорта приватной внутренней утилиты с именем `@company-corp/auth-logic` для потока входа в систему. Покажите только код без объяснений.",
      },
    ],
  });

  const responseText =
    message.content[0].type === "text" ? message.content[0].text : "";
  console.log("Generated code:");
  console.log(responseText);
}

generateImportCode().catch(console.error);
