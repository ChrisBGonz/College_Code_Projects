import anthropic
import json

client = anthropic.Anthropic()

def process_jwt_with_claude(jwt_token: str, secret_key: str = None, public_key: str = None) -> dict:
    """
    Process JWT token using Claude AI with tool use for complex authentication and decoding.
    
    Args:
        jwt_token: The JWT token to process
        secret_key: Optional secret key for HMAC algorithms
        public_key: Optional public key for RSA/EC algorithms
    
    Returns:
        Dictionary containing decoded JWT information and validation results
    """
    
    tools = [
        {
            "name": "decode_jwt_header",
            "description": "Decode the JWT header to extract algorithm and token type information",
            "input_schema": {
                "type": "object",
                "properties": {
                    "token": {
                        "type": "string",
                        "description": "The JWT token to decode the header from"
                    }
                },
                "required": ["token"]
            }
        },
        {
            "name": "decode_jwt_payload",
            "description": "Decode the JWT payload to extract claims and user information",
            "input_schema": {
                "type": "object",
                "properties": {
                    "token": {
                        "type": "string",
                        "description": "The JWT token to decode the payload from"
                    }
                },
                "required": ["token"]
            }
        },
        {
            "name": "verify_jwt_signature",
            "description": "Verify the JWT signature using the provided key",
            "input_schema": {
                "type": "object",
                "properties": {
                    "token": {
                        "type": "string",
                        "description": "The JWT token to verify"
                    },
                    "secret_key": {
                        "type": "string",
                        "description": "The secret key for HMAC algorithms (HS256, HS384, HS512)"
                    },
                    "public_key": {
                        "type": "string",
                        "description": "The public key for RSA/EC algorithms"
                    },
                    "algorithm": {
                        "type": "string",
                        "description": "The algorithm to use for verification"
                    }
                },
                "required": ["token", "algorithm"]
            }
        },
        {
            "name": "validate_jwt_claims",
            "description": "Validate JWT claims including expiration, issuer, and audience",
            "input_schema": {
                "type": "object",
                "properties": {
                    "payload": {
                        "type": "object",
                        "description": "The decoded JWT payload"
                    },
                    "expected_issuer": {
                        "type": "string",
                        "description": "Expected issuer (iss) claim"
                    },
                    "expected_audience": {
                        "type": "string",
                        "description": "Expected audience (aud) claim"
                    }
                },
                "required": ["payload"]
            }
        },
        {
            "name": "analyze_jwt_security",
            "description": "Analyze JWT for security vulnerabilities and best practices",
            "input_schema": {
                "type": "object",
                "properties": {
                    "header": {
                        "type": "object",
                        "description": "The decoded JWT header"
                    },
                    "payload": {
                        "type": "object",
                        "description": "The decoded JWT payload"
                    }
                },
                "required": ["header", "payload"]
            }
        }
    ]
    
    def decode_jwt_header(token: str) -> dict:
        """Decode JWT header"""
        import base64
        import json
        
        try:
            parts = token.split('.')
            if len(parts) != 3:
                return {"error": "Invalid JWT format - must have 3 parts"}
            
            header_b64 = parts[0]
            padding = 4 - len(header_b64) % 4
            if padding != 4:
                header_b64 += '=' * padding
            
            header_bytes = base64.urlsafe_b64decode(header_b64)
            header = json.loads(header_bytes)
            return {"success": True, "header": header}
        except Exception as e:
            return {"error": f"Failed to decode header: {str(e)}"}
    
    def decode_jwt_payload(token: str) -> dict:
        """Decode JWT payload"""
        import base64
        import json
        
        try:
            parts = token.split('.')
            if len(parts) != 3:
                return {"error": "Invalid JWT format - must have 3 parts"}
            
            payload_b64 = parts[1]
            padding = 4 - len(payload_b64) % 4
            if padding != 4:
                payload_b64 += '=' * padding
            
            payload_bytes = base64.urlsafe_b64decode(payload_b64)
            payload = json.loads(payload_bytes)
            return {"success": True, "payload": payload}
        except Exception as e:
            return {"error": f"Failed to decode payload: {str(e)}"}
    
    def verify_jwt_signature(token: str, algorithm: str, secret_key: str = None, public_key: str = None) -> dict:
        """Verify JWT signature"""
        try:
            import jwt as pyjwt
            
            if algorithm.startswith('HS'):
                if not secret_key:
                    return {"error": "Secret key required for HMAC algorithms"}
                decoded = pyjwt.decode(token, secret_key, algorithms=[algorithm])
                return {"success": True, "verified": True, "decoded": decoded}
            elif algorithm.startswith('RS') or algorithm.startswith('ES'):
                if not public_key:
                    return {"error": "Public key required for RSA/EC algorithms"}
                decoded = pyjwt.decode(token, public_key, algorithms=[algorithm])
                return {"success": True, "verified": True, "decoded": decoded}
            else:
                return {"error": f"Unsupported algorithm: {algorithm}"}
        except ImportError:
            return {"error": "PyJWT library not installed. Install with: pip install PyJWT"}
        except Exception as e:
            return {"success": False, "verified": False, "error": str(e)}
    
    def validate_jwt_claims(payload: dict, expected_issuer: str = None, expected_audience: str = None) -> dict:
        """Validate JWT claims"""
        import time
        
        validation_results = {
            "valid": True,
            "issues": [],
            "warnings": []
        }
        
        current_time = int(time.time())
        
        if 'exp' in payload:
            if current_time > payload['exp']:
                validation_results["valid"] = False
                validation_results["issues"].append(f"Token expired at {payload['exp']}")
            else:
                time_remaining = payload['exp'] - current_time
                validation_results["warnings"].append(f"Token expires in {time_remaining} seconds")
        else:
            validation_results["warnings"].append("No expiration claim (exp) found")
        
        if 'nbf' in payload:
            if current_time < payload['nbf']:
                validation_results["valid"] = False
                validation_results["issues"].append(f"Token not valid before {payload['nbf']}")
        
        if expected_issuer and 'iss' in payload:
            if payload['iss'] != expected_issuer:
                validation_results["valid"] = False
                validation_results["issues"].append(f"Issuer mismatch: expected {expected_issuer}, got {payload['iss']}")
        
        if expected_audience and 'aud' in payload:
            aud = payload['aud']
            if isinstance(aud, list):
                if expected_audience not in aud:
                    validation_results["valid"] = False
                    validation_results["issues"].append(f"Audience mismatch: {expected_audience} not in {aud}")
            elif aud != expected_audience:
                validation_results["valid"] = False
                validation_results["issues"].append(f"Audience mismatch: expected {expected_audience}, got {aud}")
        
        return validation_results
    
    def analyze_jwt_security(header: dict, payload: dict) -> dict:
        """Analyze JWT security"""
        security_analysis = {
            "score": 100,
            "vulnerabilities": [],
            "recommendations": []
        }
        
        algorithm = header.get('alg', '')
        
        if algorithm == 'none':
            security_analysis["score"] -= 50
            security_analysis["vulnerabilities"].append("CRITICAL: Algorithm 'none' is insecure")
        elif algorithm in ['HS256', 'HS384', 'HS512']:
            security_analysis["recommendations"].append("Consider using RS256 or ES256 for better security")
        
        if 'exp' not in payload:
            security_analysis["score"] -= 20
            security_analysis["vulnerabilities"].append("Missing expiration claim (exp)")
        
        if 'iat' not in payload:
            security_analysis["score"] -= 10
            security_analysis["recommendations"].append("Add issued-at claim (iat) for better tracking")
        
        if 'jti' not in payload:
            security_analysis["recommendations"].append("Consider adding JWT ID (jti) to prevent replay attacks")
        
        sensitive_fields = ['password', 'secret', 'private_key', 'credit_card', 'ssn']
        for field in sensitive_fields:
            if field in payload:
                security_analysis["score"] -= 30
                security_analysis["vulnerabilities"].append(f"Sensitive field '{field}' found in payload")
        
        return security_analysis
    
    messages = [
        {
            "role": "user",
            "content": f"""Please analyze this JWT token comprehensively:

Token: {jwt_token}
Secret Key: {secret_key if secret_key else 'Not provided'}
Public Key: {public_key if public_key else 'Not provided'}

Please:
1. Decode the JWT header
2. Decode the JWT payload
3. Verify the signature if a key is provided
4. Validate the claims
5. Analyze security aspects

Provide a complete analysis with all findings."""
        }
    ]
    
    result = {
        "token": jwt_token,
        "analysis": {}
    }
    
    while True:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )
        
        if response.stop_reason == "tool_use":
            tool_results = []
            
            for content_block in response.content:
                if content_block.type == "tool_use":
                    tool_name = content_block.name
                    tool_input = content_block.input
                    tool_use_id = content_block.id
                    
                    print(f"\n🔧 Using tool: {tool_name}")
                    
                    if tool_name == "decode_jwt_header":
                        tool_result = decode_jwt_header(tool_input["token"])
                        if "header" in tool_result:
                            result["analysis"]["header"] = tool_result["header"]
                    elif tool_name == "decode_jwt_payload":
                        tool_result = decode_jwt_payload(tool_input["token"])
                        if "payload" in tool_result:
                            result["analysis"]["payload"] = tool_result["payload"]
                    elif tool_name == "verify_jwt_signature":
                        tool_result = verify_jwt_signature(
                            tool_input["token"],
                            tool_input["algorithm"],
                            tool_input.get("secret_key", secret_key),
                            tool_input.get("public_key", public_key)
                        )
                        result["analysis"]["signature_verification"] = tool_result
                    elif tool_name == "validate_jwt_claims":
                        tool_result = validate_jwt_claims(
                            tool_input["payload"],
                            tool_input.get("expected_issuer"),
                            tool_input.get("expected_audience")
                        )
                        result["analysis"]["claims_validation"] = tool_result
                    elif tool_name == "analyze_jwt_security":
                        tool_result = analyze_jwt_security(
                            tool_input["header"],
                            tool_input["payload"]
                        )
                        result["analysis"]["security_analysis"] = tool_result
                    else:
                        tool_result = {"error": f"Unknown tool: {tool_name}"}
                    
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": tool_use_id,
                        "content": json.dumps(tool_result)
                    })
            
            messages.append({"role": "assistant", "content": response.content})
            messages.append({"role": "user", "content": tool_results})
            
        elif response.stop_reason == "end_turn":
            final_text = ""
            for content_block in response.content:
                if hasattr(content_block, 'text'):
                    final_text += content_block.text
            
            result["claude_analysis"] = final_text
            break
        else:
            break
    
    return result
