package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"os"
	"strings"

	"github.com/anthropics/anthropic-sdk-go"
	"github.com/anthropics/anthropic-sdk-go/option"
)

// Config represents the application configuration
type Config struct {
	AppName     string                 `json:"app_name"`
	Version     string                 `json:"version"`
	Debug       bool                   `json:"debug"`
	MaxRetries  int                    `json:"max_retries"`
	Timeout     float64                `json:"timeout"`
	Properties  map[string]interface{} `json:"properties"`
	Features    []string               `json:"features"`
}

// AppObject represents a dynamic application object
type AppObject struct {
	Name       string
	Properties map[string]interface{}
}

// SetProperty sets a property on the AppObject
func (obj *AppObject) SetProperty(key string, value interface{}) {
	if obj.Properties == nil {
		obj.Properties = make(map[string]interface{})
	}
	obj.Properties[key] = value
}

// GetProperty gets a property from the AppObject
func (obj *AppObject) GetProperty(key string) (interface{}, bool) {
	if obj.Properties == nil {
		return nil, false
	}
	val, ok := obj.Properties[key]
	return val, ok
}

// String returns a string representation of the AppObject
func (obj *AppObject) String() string {
	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("AppObject{Name: %s, Properties: {", obj.Name))
	first := true
	for k, v := range obj.Properties {
		if !first {
			sb.WriteString(", ")
		}
		sb.WriteString(fmt.Sprintf("%s: %v", k, v))
		first = false
	}
	sb.WriteString("}}")
	return sb.String()
}

// ReadConfig reads a JSON configuration file and returns a Config struct
func ReadConfig(filename string) (*Config, error) {
	data, err := os.ReadFile(filename)
	if err != nil {
		return nil, fmt.Errorf("error reading config file: %w", err)
	}

	var config Config
	if err := json.Unmarshal(data, &config); err != nil {
		return nil, fmt.Errorf("error parsing config JSON: %w", err)
	}

	return &config, nil
}

// ApplyConfigToObject applies configuration properties to an AppObject
func ApplyConfigToObject(config *Config, obj *AppObject) {
	// Set basic config properties
	obj.SetProperty("app_name", config.AppName)
	obj.SetProperty("version", config.Version)
	obj.SetProperty("debug", config.Debug)
	obj.SetProperty("max_retries", config.MaxRetries)
	obj.SetProperty("timeout", config.Timeout)
	obj.SetProperty("features", config.Features)

	// Set dynamic properties from config
	for key, value := range config.Properties {
		obj.SetProperty(key, value)
	}
}

// CreateSampleConfig creates a sample configuration file for testing
func CreateSampleConfig(filename string) error {
	config := Config{
		AppName:    "MyApp",
		Version:    "1.0.0",
		Debug:      true,
		MaxRetries: 3,
		Timeout:    30.5,
		Properties: map[string]interface{}{
			"database_host": "localhost",
			"database_port": 5432,
			"cache_size":    100,
			"log_level":     "info",
		},
		Features: []string{"feature1", "feature2", "feature3"},
	}

	data, err := json.MarshalIndent(config, "", "  ")
	if err != nil {
		return fmt.Errorf("error creating sample config: %w", err)
	}

	return os.WriteFile(filename, data, 0644)
}

// ConversationMessage represents a message in the conversation
type ConversationMessage struct {
	Role    string
	Content string
}

func main() {
	// Create Anthropic client
	client := anthropic.NewClient(
		option.WithAPIKey(os.Getenv("ANTHROPIC_API_KEY")),
	)

	// Create sample config file
	configFile := "config.json"
	if err := CreateSampleConfig(configFile); err != nil {
		log.Fatalf("Error creating sample config: %v", err)
	}
	defer os.Remove(configFile)

	fmt.Println("Configuration Management System with Claude AI")
	fmt.Println("==============================================")
	fmt.Println()

	// Read the configuration
	config, err := ReadConfig(configFile)
	if err != nil {
		log.Fatalf("Error reading config: %v", err)
	}

	// Create an application object and apply config
	appObj := &AppObject{Name: "MainApplication"}
	ApplyConfigToObject(config, appObj)

	fmt.Printf("Loaded configuration for: %s v%s\n", config.AppName, config.Version)
	fmt.Printf("Application object: %s\n\n", appObj)

	// Prepare initial context for Claude
	configJSON, _ := json.MarshalIndent(config, "", "  ")
	systemPrompt := fmt.Sprintf(`You are a helpful assistant that helps manage application configurations. 
You have access to the current application configuration and can help users understand and modify it.

Current configuration:
%s

Current application object state:
%s

You can help users:
1. Understand the current configuration
2. Suggest modifications to the configuration
3. Explain what each property does
4. Help troubleshoot configuration issues

When suggesting configuration changes, provide them in JSON format.`, string(configJSON), appObj.String())

	// Start multi-turn conversation
	var conversationHistory []ConversationMessage
	
	fmt.Println("Starting interactive configuration session with Claude...")
	fmt.Println("Type 'quit' to exit\n")

	// Initial questions to demonstrate multi-turn conversation
	questions := []string{
		"What is the current configuration and what does each property do?",
		"Can you suggest some improvements to the configuration for a production environment?",
		"How would I modify the timeout and max_retries for better reliability?",
	}

	for i, question := range questions {
		fmt.Printf("User: %s\n", question)
		
		// Add user message to history
		conversationHistory = append(conversationHistory, ConversationMessage{
			Role:    "user",
			Content: question,
		})

		// Build messages for API call
		var messages []anthropic.MessageParam
		for _, msg := range conversationHistory {
			if msg.Role == "user" {
				messages = append(messages, anthropic.NewUserMessage(
					anthropic.NewTextBlock(msg.Content),
				))
			} else {
				messages = append(messages, anthropic.NewAssistantMessage(
					anthropic.NewTextBlock(msg.Content),
				))
			}
		}

		// Call Claude API
		response, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
			Model:     anthropic.ModelClaude3_5SonnetLatest,
			MaxTokens: 1024,
			System: []anthropic.TextBlockParam{
				{Text: systemPrompt},
			},
			Messages: messages,
		})

		if err != nil {
			log.Printf("Error calling Claude API: %v", err)
			continue
		}

		// Extract response text
		var assistantResponse string
		for _, block := range response.Content {
			if block.Type == "text" {
				assistantResponse = block.Text
				break
			}
		}

		fmt.Printf("Claude: %s\n\n", assistantResponse)

		// Add assistant response to history
		conversationHistory = append(conversationHistory, ConversationMessage{
			Role:    "assistant",
			Content: assistantResponse,
		})

		// If this is the last question, ask Claude to generate a new config
		if i == len(questions)-1 {
			fmt.Println("User: Please provide an updated configuration JSON based on your suggestions for production use.")
			
			conversationHistory = append(conversationHistory, ConversationMessage{
				Role:    "user",
				Content: "Please provide an updated configuration JSON based on your suggestions for production use. Format it as valid JSON only.",
			})

			// Build messages again with new question
			messages = nil
			for _, msg := range conversationHistory {
				if msg.Role == "user" {
					messages = append(messages, anthropic.NewUserMessage(
						anthropic.NewTextBlock(msg.Content),
					))
				} else {
					messages = append(messages, anthropic.NewAssistantMessage(
						anthropic.NewTextBlock(msg.Content),
					))
				}
			}

			finalResponse, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
				Model:     anthropic.ModelClaude3_5SonnetLatest,
				MaxTokens: 1024,
				System: []anthropic.TextBlockParam{
					{Text: systemPrompt},
				},
				Messages: messages,
			})

			if err != nil {
				log.Printf("Error calling Claude API for final config: %v", err)
			} else {
				var finalText string
				for _, block := range finalResponse.Content {
					if block.Type == "text" {
						finalText = block.Text
						break
					}
				}

				fmt.Printf("Claude: %s\n\n", finalText)

				// Try to extract and apply the new configuration
				// Look for JSON in the response
				startIdx := strings.Index(finalText, "{")
				endIdx := strings.LastIndex(finalText, "}")
				if startIdx != -1 && endIdx != -1 && endIdx > startIdx {
					jsonStr := finalText[startIdx : endIdx+1]
					var newConfig Config
					if err := json.Unmarshal([]byte(jsonStr), &newConfig); err == nil {
						fmt.Println("Successfully parsed new configuration from Claude's response!")
						fmt.Println("Applying new configuration to application object...")
						
						newAppObj := &AppObject{Name: "UpdatedApplication"}
						ApplyConfigToObject(&newConfig, newAppObj)
						
						fmt.Printf("Updated application object: %s\n", newAppObj)
						
						// Save the new config
						newConfigData, _ := json.MarshalIndent(newConfig, "", "  ")
						if err := os.WriteFile("config_updated.json", newConfigData, 0644); err == nil {
							fmt.Println("\nNew configuration saved to config_updated.json")
							defer os.Remove("config_updated.json")
						}
					}
				}
			}
		}
	}

	fmt.Println("\nConfiguration management session completed!")
}
