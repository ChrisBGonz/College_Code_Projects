
```ruby
require 'anthropic'

# Simulated database of users
USERS_DB = {
  1 => { id: 1, name: "Alice Johnson", role: "Manager", email: "alice@company.com" },
  2 => { id: 2, name: "Bob Smith", role: "Employee", email: "bob@company.com" },
  3 => { id: 3, name: "Carol White", role: "Employee", email: "carol@company.com" },
  4 => { id: 4, name: "David Brown", role: "Manager", email: "david@company.com" },
  5 => { id: 5, name: "Eve Davis", role: "Employee", email: "eve@company.com" }
}

# Tool definitions for Claude
TOOLS = [
  {
    name: "get_user",
    description: "Get user information by ID",
    input_schema: {
      type: "object",
      properties: {
        user_id: {
          type: "integer",
          description: "The ID of the user to retrieve"
        }
      },
      required: ["user_id"]
    }
  },
  {
    name: "promote_employee",
    description: "Promote an employee to manager role. Only managers can promote employees.",
    input_schema: {
      type: "object",
      properties: {
        manager_id: {
          type: "integer",
          description: "The ID of the manager performing the promotion"
        },
        employee_id: {
          type: "integer",
          description: "The ID of the employee to be promoted"
        }
      },
      required: ["manager_id", "employee_id"]
    }
  },
  {
    name: "list_employees",
    description: "List all employees in the system",
    input_schema: {
      type: "object",
      properties: {},
      required: []
    }
  }
]

def get_user(user_id)
  user = USERS_DB[user_id]
  if user
    { success: true, user: user }
  else
    { success: false, error: "User not found with ID: #{user_id}" }
  end
end

def promote_employee(manager_id, employee_id)
  manager = USERS_DB[manager_id]
  employee = USERS_DB[employee_id]

  unless manager
    return { success: false, error: "Manager not found with ID: #{manager_id}" }
  end

  unless employee
    return { success: false, error: "Employee not found with ID: #{employee_id}" }
  end

  unless manager[:role] == "Manager"
    return { success: false, error: "User #{manager[:name]} is not a Manager and cannot promote employees" }
  end

  unless employee[:role] == "Employee"
    return { success: false, error: "User #{employee[:name]} is already a #{employee[:role]} and cannot be promoted to Manager" }
  end

  # Perform the promotion
  USERS_DB[employee_id][:role] = "Manager"
  
  {
    success: true,
    message: "Successfully promoted #{employee[:name]} from Employee to Manager",
    promoted_user: USERS_DB[employee_id]
  }
end

def list_employees
  employees = USERS_DB.values
  { success: true, users: employees }
end

def process_tool_call(tool_name, tool_input)
  case tool_name
  when "get_user"
    get_user(tool_input["user_id"])
  when "promote_employee"
    promote_employee(tool_input["manager_id"], tool_input["employee_id"])
  when "list_employees"
    list_employees
  else
    { success: false, error: "Unknown tool: #{tool_name}" }
  end
end

def promote_employee_action(current_user_id, employee_id)
  client = Anthropic::Client.new

  puts "=" * 60
  puts "Promotion Request"
  puts "=" * 60
  puts "Current User ID: #{current_user_id}"
  puts "Employee to Promote ID: #{employee_id}"
  puts "=" * 60

  messages = [
    {
      role: "user",
      content: "I am user with ID #{current_user_id}. I want to promote employee with ID #{employee_id} to Manager role. Please verify my permissions and perform the promotion if I'm authorized."
    }
  ]

  loop do
    response = client.messages(
      model: "claude-opus-4-5",
      max_tokens: 1024,
      tools: TOOLS,
      messages: messages
    )

    puts "\nClaude Response:"
    puts "Stop Reason: #{response.stop_reason}"

    # Process text content if any
    response.content.each do |block|
      if block.type == "text"
        puts "Claude: #{block.text}"
      end
    end

    # Check if we need to process tool calls
    if response.stop_reason == "tool_use"
      tool_results = []

      response.content.each do |block|
        next unless block.type == "tool_use"

        puts "\nTool Call: #{block.name}"
        puts "Input: #{block.input}"

        result = process_tool_call(block.name, block.input)
        puts "Result: #{result}"

        tool_results << {
          type: "tool_result",
          tool_use_id: block.id,
          content: result.to_json
        }
      end

      # Add assistant response and tool results to messages
      messages << { role: "assistant", content: response.content }
      messages << { role: "user", content: tool_results }
    else
      # No more tool calls, we're done
      break
    end
  end

  puts "\n" + "=" * 60
  puts "Final State of Users:"
  USERS_DB.each do |id, user|
    puts "  ID: #{id}, Name: #{user[:name]}, Role: #{user[:role]}"
  end
  puts "=" * 60
end

# Test scenarios
puts "\n### Test 1: Manager promotes Employee ###"
promote_employee_action(1, 2) # Alice (Manager) promotes Bob (Employee)

puts "\n\n### Test 2: Employee tries to promote another Employee ###"
promote_employee_action(3, 5) # Carol (Employee) tries to promote Eve (Employee)

puts "\n\n### Test 3: Manager tries to promote another Manager ###"
promote_employee_action(4, 1) # David (Manager) tries to promote Alice (Manager)
```
