import anthropic
import base64
import os
import tempfile


def parse_xml_with_claude(xml_content: str) -> str:
    """
    Use Claude to parse XML content and extract title tags.
    
    Args:
        xml_content: The XML content as a string
        
    Returns:
        The extracted title information
    """
    client = anthropic.Anthropic()
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {
                "role": "user",
                "content": f"""Please analyze this XML content and extract all 'title' tags with their values.
                
XML Content:
{xml_content}

Please provide:
1. All title tags found
2. Their values/content
3. Their location in the XML structure (parent elements)"""
            }
        ]
    )
    
    return message.content[0].text


def parse_xml_file_with_lxml(file_path: str) -> dict:
    """
    Parse XML file using lxml and extract title tags.
    
    Args:
        file_path: Path to the XML file
        
    Returns:
        Dictionary with extracted title information
    """
    from lxml import etree
    
    tree = etree.parse(file_path)
    root = tree.getroot()
    
    titles = []
    
    for title_elem in root.iter('title'):
        title_info = {
            'text': title_elem.text,
            'attributes': dict(title_elem.attrib),
            'parent': title_elem.getparent().tag if title_elem.getparent() is not None else None,
            'xpath': tree.getpath(title_elem)
        }
        titles.append(title_info)
    
    return {
        'root_tag': root.tag,
        'total_titles': len(titles),
        'titles': titles
    }


def process_xml_file(file_path: str) -> None:
    """
    Process an XML file: parse with lxml and analyze with Claude.
    
    Args:
        file_path: Path to the XML file to process
    """
    print(f"\nProcessing XML file: {file_path}")
    print("=" * 60)
    
    print("\n1. Parsing with lxml...")
    lxml_results = parse_xml_file_with_lxml(file_path)
    
    print(f"Root tag: {lxml_results['root_tag']}")
    print(f"Total title tags found: {lxml_results['total_titles']}")
    
    if lxml_results['titles']:
        print("\nExtracted titles:")
        for i, title in enumerate(lxml_results['titles'], 1):
            print(f"\n  Title {i}:")
            print(f"    Text: {title['text']}")
            print(f"    Parent: {title['parent']}")
            print(f"    XPath: {title['xpath']}")
            if title['attributes']:
                print(f"    Attributes: {title['attributes']}")
    else:
        print("No title tags found in the XML file.")
    
    with open(file_path, 'r', encoding='utf-8') as f:
        xml_content = f.read()
    
    print("\n2. Analyzing with Claude AI...")
    claude_analysis = parse_xml_with_claude(xml_content)
    print("\nClaude's Analysis:")
    print("-" * 40)
    print(claude_analysis)


def create_sample_xml() -> str:
    """
    Create a sample XML file for demonstration.
    
    Returns:
        Path to the created sample XML file
    """
    sample_xml = """<?xml version="1.0" encoding="UTF-8"?>
<library>

        <title>Python Programming</title>
        <author>John Smith</author>
        <year>2023</year>
        <category>
            <title>Programming Languages</title>
        </category>
    </book>
    <book id="2">
        <title>Data Science with Python</title>
        <author>Jane Doe</author>
        <year>2022</year>
        <category>
            <title>Data Science</title>
        </category>
    </book>
    <book id="3">
        <title>Machine Learning Fundamentals</title>

        <year>2024</year>
        <category>
            <title>Artificial Intelligence</title>
        </category>

    <metadata>
        <title>Library Catalog</title>
        <version>1.0</version>

def main():
    """Main function to demonstrate XML parsing with lxml and Claude."""
    print("XML Parser with lxml and Claude AI")
    print("=" * 60)
    
    print("\nNo file path provided. Creating a sample XML file for demonstration...")
    xml_file_path = create_sample_xml()
    print(f"Sample XML file created at: {xml_file_path}")
    
    try:
        process_xml_file(xml_file_path)
    finally:
        if os.path.exists(xml_file_path):
            os.unlink(xml_file_path)
            print(f"\nTemporary file {xml_file_path} has been cleaned up.")


if __name__ == "__main__":
    main()
