@app.route('/upload_yaml', methods=['POST'])
def upload_yaml():
    if 'file' not in request.files:
        return jsonify({'success': False, 'error': 'Файл не найден'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'error': 'Файл не выбран'})
    
    if not (file.filename.endswith('.yaml') or file.filename.endswith('.yml')):
        return jsonify({'success': False, 'error': 'Файл должен быть в формате YAML (.yaml или .yml)'})
    
    try:
        content = file.read().decode('utf-8')
        profile_data = yaml.safe_load(content)
        
        if not isinstance(profile_data, dict):
            return jsonify({'success': False, 'error': 'YAML файл должен содержать словарь (ключ-значение)'})
        
        user_profiles['current'] = profile_data
        
        # Reset conversation history when new profile is loaded
        conversation_history.clear()
        
        return jsonify({
            'success': True,
            'profile': profile_data,
            'message': 'Профиль успешно загружен'
        })
        
    except yaml.YAMLError as e:
        return jsonify({'success': False, 'error': f'Ошибка парсинга YAML: {str(e)}'})
    except Exception as e:
        return jsonify({'success': False, 'error': f'Ошибка обработки файла: {str(e)}'})

@app.route('/chat', methods=['POST'])
def chat():
    data = request.json
    user_message = data.get('message', '').strip()
    
    if not user_message:
        return jsonify({'error': 'Сообщение не может быть пустым'})
    
    # Build system prompt based on user profile
    system_prompt = build_system_prompt()
    
    # Add user message to history
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    try:
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=8096,
            system=system_prompt,
            messages=conversation_history
        )
        
        assistant_message = response.content[0].text
        
        # Add assistant response to history
        conversation_history.append({
            "role": "assistant",
            "content": assistant_message
        })
        
        return jsonify({'response': assistant_message})
        
    except Exception as e:
        # Remove the user message if there was an error
        if conversation_history and conversation_history[-1]['role'] == 'user':
            conversation_history.pop()
        return jsonify({'error': f'Ошибка API: {str(e)}'})

def build_system_prompt():
    base_prompt = "Вы полезный AI ассистент."
    
    if 'current' not in user_profiles:
        return base_prompt + " Отвечайте на вопросы пользователя."
    
    profile = user_profiles['current']
    prompt_parts = [base_prompt]
    
    # Add personalization based on profile
    if 'name' in profile:
        prompt_parts.append(f"Вы общаетесь с пользователем по имени {profile['name']}.")
    
    if 'language' in profile:
        prompt_parts.append(f"Общайтесь на языке: {profile['language']}.")
    
    if 'communication_style' in profile:
        style = profile['communication_style']
        if 'формальный' in str(style).lower() or 'formal' in str(style).lower():
            prompt_parts.append("Используйте формальный стиль общения.")
        elif 'неформальный' in str(style).lower() or 'informal' in str(style).lower():
            prompt_parts.append("Используйте неформальный, дружелюбный стиль общения.")
    
    if 'expertise_level' in profile:
        level = profile['expertise_level']
        if 'начинающий' in str(level).lower() or 'beginner' in str(level).lower():
            prompt_parts.append("Объясняйте концепции простым языком, избегайте технического жаргона.")
        elif 'продвинутый' in str(level).lower() or 'advanced' in str(level).lower():
            prompt_parts.append("Можете использовать технические термины и предполагать высокий уровень знаний.")
        elif 'средний' in str(level).lower() or 'intermediate' in str(level).lower():
            prompt_parts.append("Используйте умеренно технический язык с пояснениями при необходимости.")
    
    if 'interests' in profile:
        interests = profile['interests']
        if isinstance(interests, list):
            interests_str = ', '.join(str(i) for i in interests)
            prompt_parts.append(f"Интересы пользователя: {interests_str}. Старайтесь связывать ответы с этими темами, когда это уместно.")
    
    if 'occupation' in profile:
        prompt_parts.append(f"Профессия пользователя: {profile['occupation']}. Учитывайте это при ответах.")
    
    if 'age' in profile:
        prompt_parts.append(f"Возраст пользователя: {profile['age']} лет.")
    
    # Add any custom fields
    known_fields = {'name', 'age', 'occupation', 'interests', 'language', 'communication_style', 'expertise_level'}
    custom_fields = {k: v for k, v in profile.items() if k not in known_fields}
    
    if custom_fields:
        custom_info = ', '.join(f"{k}: {v}" for k, v in custom_fields.items())
        prompt_parts.append(f"Дополнительная информация о пользователе: {custom_info}.")
    
    prompt_parts.append("Адаптируйте свои ответы в соответствии с профилем пользователя.")
    
    return ' '.join(prompt_parts)

@app.route('/clear_history', methods=['POST'])
def clear_history():
    conversation_history.clear()
    return jsonify({'success': True, 'message': 'История очищена'})

@app.route('/get_profile', methods=['GET'])
def get_profile():
    if 'current' in user_profiles:
        return jsonify({'profile': user_profiles['current']})
    return jsonify({'profile': None})

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)
