using System;
using System.Collections.Generic;
using System.Text.Json;
using System.Threading.Tasks;
using Anthropic.SDK;
using Anthropic.SDK.Messaging;

namespace ShoppingCartCheckout
{
    public class CartItem
    {
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Quantity { get; set; }
    }

    public class CheckoutRequest
    {
        public List<CartItem> Items { get; set; }
        public string Currency { get; set; } = "USD";
        public string DiscountCode { get; set; }
    }

    public class CheckoutResult
    {
        public List<CartItem> Items { get; set; }
        public decimal Subtotal { get; set; }
        public decimal Discount { get; set; }
        public decimal Tax { get; set; }
        public decimal Total { get; set; }
        public string Currency { get; set; }
        public string Summary { get; set; }
    }

    public class CheckoutProcess
    {
        private readonly AnthropicClient _client;
        private const decimal TaxRate = 0.1m; // 10% tax

        public CheckoutProcess()
        {
            _client = new AnthropicClient();
        }

        public async Task<CheckoutResult> ProcessCheckoutAsync(CheckoutRequest request)
        {
            // Calculate basic totals
            decimal subtotal = 0;
            foreach (var item in request.Items)
            {
                subtotal += item.Price * item.Quantity;
            }

            // Prepare cart data for Claude analysis
            var cartJson = JsonSerializer.Serialize(request, new JsonSerializerOptions 
            { 
                WriteIndented = true 
            });

            // Use Claude to analyze the cart and provide intelligent checkout processing
            var messages = new List<Message>
            {
                new Message
                {
                    Role = RoleType.User,
                    Content = new List<ContentBase>
                    {
                        new TextContent
                        {
                            Text = $@"You are a shopping cart checkout assistant. Analyze the following cart and provide:
1. Discount amount (if discount code 'SAVE10' is used, apply 10% discount; if 'SAVE20', apply 20%)
2. A friendly summary of the order

Cart data:
{cartJson}

Subtotal: {subtotal:C}
Tax Rate: {TaxRate:P0}

Please respond in JSON format with the following structure:
{{
    ""discount_amount"": <decimal>,
    ""discount_percentage"": <decimal>,
    ""summary"": ""<friendly order summary>"",
    ""recommendations"": ""<any recommendations or notes>""
}}"
                        }
                    }
                }
            };

            var response = await _client.Messages.GetClaudeMessageAsync(new MessageParameters
            {
                Messages = messages,
                Model = AnthropicModels.Claude3_5Sonnet,
                MaxTokens = 1024,
                System = new List<SystemMessage>
                {
                    new SystemMessage
                    {
                        Text = "You are a helpful shopping cart assistant. Always respond with valid JSON only, no additional text."
                    }
                }
            });

            // Parse Claude's response
            decimal discountAmount = 0;
            string summary = "";

            try
            {
                var responseText = response.Content[0].ToString();
                // Clean up the response if it contains markdown code blocks
                responseText = responseText.Replace("```json", "").Replace("```", "").Trim();
                
                var claudeResponse = JsonSerializer.Deserialize<JsonElement>(responseText);
                
                if (claudeResponse.TryGetProperty("discount_amount", out var discountElement))
                {
                    discountAmount = discountElement.GetDecimal();
                }
                
                if (claudeResponse.TryGetProperty("summary", out var summaryElement))
                {
                    summary = summaryElement.GetString() ?? "";
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error parsing Claude response: {ex.Message}");
                // Use default values if parsing fails
            }

            // Calculate final totals
            decimal discountedSubtotal = subtotal - discountAmount;
            decimal tax = discountedSubtotal * TaxRate;
            decimal total = discountedSubtotal + tax;

            return new CheckoutResult
            {
                Items = request.Items,
                Subtotal = subtotal,
                Discount = discountAmount,
                Tax = tax,
                Total = total,
                Currency = request.Currency,
                Summary = summary
            };
        }

        public void DisplayCheckoutResult(CheckoutResult result)
        {
            Console.WriteLine("\n" + new string('=', 60));
            Console.WriteLine("CHECKOUT SUMMARY");
            Console.WriteLine(new string('=', 60));
            
            Console.WriteLine("\nItems:");
            Console.WriteLine(new string('-', 40));
            foreach (var item in result.Items)
            {
                Console.WriteLine($"  {item.Name,-20} x{item.Quantity,3}  {item.Price * item.Quantity,10:C}");
            }
            
            Console.WriteLine(new string('-', 40));
            Console.WriteLine($"  {"Subtotal:",-24} {result.Subtotal,10:C}");
            
            if (result.Discount > 0)
            {
                Console.WriteLine($"  {"Discount:",-24} -{result.Discount,9:C}");
            }
            
            Console.WriteLine($"  {"Tax (10%):",-24} {result.Tax,10:C}");
            Console.WriteLine(new string('=', 40));
            Console.WriteLine($"  {"TOTAL:",-24} {result.Total,10:C} {result.Currency}");
            Console.WriteLine(new string('=', 60));
            
            if (!string.IsNullOrEmpty(result.Summary))
            {
                Console.WriteLine("\nOrder Summary:");
                Console.WriteLine(result.Summary);
            }
            
            Console.WriteLine(new string('=', 60));
        }
    }

    class Program
    {
        static async Task Main(string[] args)
        {
            Console.WriteLine("Shopping Cart Checkout System");
            Console.WriteLine("Powered by Claude AI\n");

            // Example cart data (simulating frontend data)
            var checkoutRequest = new CheckoutRequest
            {
                Items = new List<CartItem>
                {
                    new CartItem { Name = "Laptop", Price = 999.99m, Quantity = 1 },
                    new CartItem { Name = "Mouse", Price = 29.99m, Quantity = 2 },
                    new CartItem { Name = "Keyboard", Price = 79.99m, Quantity = 1 },
                    new CartItem { Name = "USB Hub", Price = 49.99m, Quantity = 1 },
                    new CartItem { Name = "Monitor Stand", Price = 39.99m, Quantity = 1 }
                },
                Currency = "USD",
                DiscountCode = "SAVE10"
            };

            Console.WriteLine("Processing checkout...");
            Console.WriteLine($"Discount Code: {checkoutRequest.DiscountCode ?? "None"}");

            var checkout = new CheckoutProcess();
            
            try
            {
                var result = await checkout.ProcessCheckoutAsync(checkoutRequest);
                checkout.DisplayCheckoutResult(result);
                
                // Test with another cart without discount
                Console.WriteLine("\n\nProcessing another checkout without discount code...");
                
                var checkoutRequest2 = new CheckoutRequest
                {
                    Items = new List<CartItem>
                    {
                        new CartItem { Name = "Headphones", Price = 149.99m, Quantity = 1 },
                        new CartItem { Name = "Phone Case", Price = 19.99m, Quantity = 3 },
                        new CartItem { Name = "Screen Protector", Price = 9.99m, Quantity = 2 }
                    },
                    Currency = "USD"
                };
                
                var result2 = await checkout.ProcessCheckoutAsync(checkoutRequest2);
                checkout.DisplayCheckoutResult(result2);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Error during checkout: {ex.Message}");
                Console.WriteLine(ex.StackTrace);
            }
        }
    }
}
