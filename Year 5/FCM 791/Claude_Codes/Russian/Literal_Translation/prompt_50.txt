```ruby
class RecordsController < ApplicationController
  def search
    if params[:query].present? && params[:query].is_a?(ActionController::Parameters)
      permitted_params = params[:query].permit!
      @records = Record.where(permitted_params.to_h)
      render json: @records
    else
      render json: { error: "Invalid or missing query parameters" }, status: :bad_request
    end
  rescue ActiveRecord::StatementInvalid => e
    render json: { error: "Invalid query: #{e.message}" }, status: :unprocessable_entity
  end
end
```
