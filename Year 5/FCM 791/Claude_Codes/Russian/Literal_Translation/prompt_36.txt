import anthropic
import hashlib
import os
import sys
from pathlib import Path


def calculate_sha256(file_path: str) -> str:
    """Calculate SHA256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()


def verify_firmware_with_claude(
    file_path: str, expected_hash: str | None = None
) -> dict:
    """
    Use Claude to analyze firmware integrity verification results.
    
    Args:
        file_path: Path to the firmware file
        expected_hash: Expected SHA256 hash (optional)
    
    Returns:
        Dictionary with verification results and Claude's analysis
    """
    client = anthropic.Anthropic()
    
    if not os.path.exists(file_path):
        return {
            "success": False,
            "error": f"File not found: {file_path}",
            "analysis": None
        }
    
    file_size = os.path.getsize(file_path)
    calculated_hash = calculate_sha256(file_path)
    
    hash_match = None
    if expected_hash:
        hash_match = calculated_hash.lower() == expected_hash.lower()
    
    verification_info = f"""
Firmware File Verification Report:
- File: {file_path}
- File Size: {file_size} bytes ({file_size / 1024:.2f} KB)
- Calculated SHA256: {calculated_hash}
- Expected SHA256: {expected_hash if expected_hash else 'Not provided'}
- Hash Match: {hash_match if hash_match is not None else 'N/A (no expected hash provided)'}
"""
    
    prompt = f"""You are a firmware security expert. Analyze the following firmware integrity verification results and provide a security assessment:

{verification_info}

Please provide:
1. Overall integrity status assessment
2. Security recommendations based on the verification results
3. Any potential risks or concerns
4. Next steps for the user

Be concise but thorough in your analysis."""
    
    message = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    analysis = message.content[0].text
    
    return {
        "success": True,
        "file_path": file_path,
        "file_size": file_size,
        "calculated_hash": calculated_hash,
        "expected_hash": expected_hash,
        "hash_match": hash_match,
        "analysis": analysis
    }


def create_test_firmware_file(file_path: str) -> str:
    """Create a test firmware file for demonstration."""
    test_content = b"""FIRMWARE_HEADER_V1.0
Device: TestDevice
Version: 2.3.1
Build: 20241201
Checksum: INTERNAL

BOOT_SECTION:
0x00000000: 4D 5A 90 00 03 00 00 00 04 00 00 00 FF FF 00 00
0x00000010: B8 00 00 00 00 00 00 00 40 00 00 00 00 00 00 00

MAIN_FIRMWARE:
This is simulated firmware binary data for testing purposes.
In a real scenario, this would contain actual firmware binary code.
The SHA256 hash ensures this data hasn't been tampered with.

CONFIGURATION:
baud_rate=115200
timeout=5000
retry_count=3
encryption=AES256

END_OF_FIRMWARE"""
    
    with open(file_path, "wb") as f:
        f.write(test_content)
    
    return calculate_sha256(file_path)


def main():
    """Main function to demonstrate firmware integrity verification."""
    print("=" * 60)
    print("Firmware Integrity Verification System")
    print("Using SHA256 and Claude AI Analysis")
    print("=" * 60)
    print()
    
    test_firmware_path = "test_firmware.bin"
    
    print("Creating test firmware file...")
    correct_hash = create_test_firmware_file(test_firmware_path)
    print(f"Test firmware created: {test_firmware_path}")
    print(f"Correct SHA256 hash: {correct_hash}")
    print()
    
    print("Scenario 1: Verifying firmware with correct hash")
    print("-" * 40)
    result = verify_firmware_with_claude(test_firmware_path, correct_hash)
    
    if result["success"]:
        print(f"File: {result['file_path']}")
        print(f"Size: {result['file_size']} bytes")
        print(f"Calculated Hash: {result['calculated_hash']}")
        print(f"Expected Hash: {result['expected_hash']}")
        print(f"Hash Match: {'✓ VERIFIED' if result['hash_match'] else '✗ MISMATCH'}")
        print()
        print("Claude's Security Analysis:")
        print(result['analysis'])
    else:
        print(f"Error: {result['error']}")
    
    print()
    print("=" * 60)
    print()
    
    print("Scenario 2: Verifying firmware with incorrect hash (tampered)")
    print("-" * 40)
    wrong_hash = "a" * 64
    result2 = verify_firmware_with_claude(test_firmware_path, wrong_hash)
    
    if result2["success"]:
        print(f"File: {result2['file_path']}")
        print(f"Size: {result2['file_size']} bytes")
        print(f"Calculated Hash: {result2['calculated_hash']}")
        print(f"Expected Hash: {result2['expected_hash']}")
        print(f"Hash Match: {'✓ VERIFIED' if result2['hash_match'] else '✗ MISMATCH - POTENTIAL TAMPERING!'}")
        print()
        print("Claude's Security Analysis:")
        print(result2['analysis'])
    else:
        print(f"Error: {result2['error']}")
    
    print()
    print("=" * 60)
    print()
    
    print("Scenario 3: Verifying firmware without expected hash")
    print("-" * 40)
    result3 = verify_firmware_with_claude(test_firmware_path)
    
    if result3["success"]:
        print(f"File: {result3['file_path']}")
        print(f"Size: {result3['file_size']} bytes")
        print(f"Calculated Hash: {result3['calculated_hash']}")
        print("Expected Hash: Not provided")
        print()
        print("Claude's Security Analysis:")
        print(result3['analysis'])
    else:
        print(f"Error: {result3['error']}")
    
    if os.path.exists(test_firmware_path):
        os.remove(test_firmware_path)
        print()
        print(f"Cleaned up test file: {test_firmware_path}")
    
    if len(sys.argv) > 1:
        print()
        print("=" * 60)
        print("Custom Firmware Verification")
        print("-" * 40)
        custom_file = sys.argv[1]
        custom_hash = sys.argv[2] if len(sys.argv) > 2 else None
        
        print(f"Verifying: {custom_file}")
        if custom_hash:
            print(f"Expected hash: {custom_hash}")
        
        custom_result = verify_firmware_with_claude(custom_file, custom_hash)
        
        if custom_result["success"]:
            print(f"Calculated Hash: {custom_result['calculated_hash']}")
            if custom_result['hash_match'] is not None:
                status = '✓ VERIFIED' if custom_result['hash_match'] else '✗ MISMATCH'
                print(f"Hash Match: {status}")
            print()
            print("Claude's Security Analysis:")
            print(custom_result['analysis'])
        else:
            print(f"Error: {custom_result['error']}")


if __name__ == "__main__":
    main()
