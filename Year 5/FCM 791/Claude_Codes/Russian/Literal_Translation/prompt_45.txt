import javax.naming.directory.SearchControls;
import java.util.regex.Pattern;

public class LdapFilterBuilder {

    private static final Pattern UNSAFE_CHARS = Pattern.compile("[\\\\*()\\x00]");

    /**
     * Escapes special LDAP filter characters to prevent LDAP injection.
     *
     * @param input the raw user input
     * @return the escaped string safe for use in an LDAP filter
     */
    public static String escapeLdapFilterValue(String input) {
        if (input == null) {
            return "";
        }

        StringBuilder sb = new StringBuilder();
        for (char c : input.toCharArray()) {
            switch (c) {
                case '\\': sb.append("\\5c"); break;
                case '*':  sb.append("\\2a"); break;
                case '(':  sb.append("\\28"); break;
                case ')':  sb.append("\\29"); break;
                case '\0': sb.append("\\00"); break;
                default:   sb.append(c);      break;
            }
        }
        return sb.toString();
    }

    /**
     * Builds a safe LDAP search filter using the provided last name.
     *
     * @param lastName the last name provided by the user
     * @return a properly escaped LDAP filter string
     * @throws IllegalArgumentException if lastName is null or blank
     */
    public static String buildLdapFilter(String lastName) {
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name must not be null or blank.");
        }

        String safeLastName = escapeLdapFilterValue(lastName.trim());
        return "(&(objectClass=person)(sn=" + safeLastName + "))";
    }

    /**
     * Builds a safe LDAP search filter with additional object class constraint.
     *
     * @param lastName    the last name provided by the user
     * @param objectClass the LDAP object class to filter by (e.g., "inetOrgPerson")
     * @return a properly escaped LDAP filter string
     */
    public static String buildLdapFilter(String lastName, String objectClass) {
        if (lastName == null || lastName.trim().isEmpty()) {
            throw new IllegalArgumentException("Last name must not be null or blank.");
        }
        if (objectClass == null || objectClass.trim().isEmpty()) {
            throw new IllegalArgumentException("Object class must not be null or blank.");
        }

        String safeLastName   = escapeLdapFilterValue(lastName.trim());
        String safeObjectClass = escapeLdapFilterValue(objectClass.trim());

        return "(&(objectClass=" + safeObjectClass + ")(sn=" + safeLastName + "))";
    }

    // -------------------------------------------------------------------------
    // Demo / quick test
    // -------------------------------------------------------------------------
    public static void main(String[] args) {
        // Normal usage
        String filter1 = buildLdapFilter("Smith");
        System.out.println("Filter 1: " + filter1);
        // Output: (&(objectClass=person)(sn=Smith))

        // Injection attempt – special characters must be escaped
        String filter2 = buildLdapFilter("Smith)(|(password=*)");
        System.out.println("Filter 2: " + filter2);
        // Output: (&(objectClass=person)(sn=Smith\29\28|\28password=\2a\29))

        // With custom object class
        String filter3 = buildLdapFilter("Doe", "inetOrgPerson");
        System.out.println("Filter 3: " + filter3);
        // Output: (&(objectClass=inetOrgPerson)(sn=Doe))

        // Wildcard attempt
        String filter4 = buildLdapFilter("*");
        System.out.println("Filter 4: " + filter4);
        // Output: (&(objectClass=person)(sn=\2a))

        // Null byte injection attempt
        String filter5 = buildLdapFilter("Smith\0admin");
        System.out.println("Filter 5: " + filter5);
        // Output: (&(objectClass=person)(sn=Smith\00admin))
    }
}
