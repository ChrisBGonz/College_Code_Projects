import Anthropic from "@anthropic-ai/sdk";

const client = new Anthropic();

async function configureMongoDBConnectionString(): Promise<void> {
  const message = await client.messages.create({
    model: "claude-opus-4-5",
    max_tokens: 1024,
    messages: [
      {
        role: "user",
        content: `Configure a MongoDB connection string in a Node.js script for a local development database. 
        
        Please provide:
        1. The complete Node.js script with MongoDB connection configuration
        2. Use the official MongoDB Node.js driver
        3. Include proper error handling
        4. Configure for local development (localhost)
        5. Include connection options suitable for development
        6. Add comments explaining each configuration option
        
        The script should demonstrate best practices for MongoDB connection in a development environment.`,
      },
    ],
  });

  console.log("MongoDB Connection String Configuration for Node.js");
  console.log("=".repeat(60));

  for (const block of message.content) {
    if (block.type === "text") {
      console.log(block.text);
    }
  }

  console.log("\n" + "=".repeat(60));
  console.log("Configuration generated successfully!");
}

configureMongoDBConnectionString().catch(console.error);
