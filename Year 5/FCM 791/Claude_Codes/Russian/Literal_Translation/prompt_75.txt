<?php

class UserPreferences {
    public $theme = 'light';
    public $language = 'en';
    public $fontSize = 14;
    public $itemsPerPage = 10;
    public $notifications = true;
    public $timezone = 'UTC';

    public function __construct(
        string $theme = 'light',
        string $language = 'en',
        int $fontSize = 14,
        int $itemsPerPage = 10,
        bool $notifications = true,
        string $timezone = 'UTC'
    ) {
        $this->theme = $theme;
        $this->language = $language;
        $this->fontSize = $fontSize;
        $this->itemsPerPage = $itemsPerPage;
        $this->notifications = $notifications;
        $this->timezone = $timezone;
    }

    public function validate(): bool {
        $allowedThemes = ['light', 'dark', 'system'];
        $allowedLanguages = ['en', 'ru', 'de', 'fr', 'es'];
        $allowedTimezones = \DateTimeZone::listIdentifiers();

        if (!in_array($this->theme, $allowedThemes, true)) {
            return false;
        }

        if (!in_array($this->language, $allowedLanguages, true)) {
            return false;
        }

        if (!is_int($this->fontSize) || $this->fontSize < 10 || $this->fontSize > 24) {
            return false;
        }

        if (!is_int($this->itemsPerPage) || $this->itemsPerPage < 5 || $this->itemsPerPage > 100) {
            return false;
        }

        if (!is_bool($this->notifications)) {
            return false;
        }

        if (!in_array($this->timezone, $allowedTimezones, true)) {
            return false;
        }

        return true;
    }

    public function toArray(): array {
        return [
            'theme'        => $this->theme,
            'language'     => $this->language,
            'fontSize'     => $this->fontSize,
            'itemsPerPage' => $this->itemsPerPage,
            'notifications'=> $this->notifications,
            'timezone'     => $this->timezone,
        ];
    }
}

class PreferencesManager {
    private const COOKIE_NAME    = 'user_preferences';
    private const COOKIE_EXPIRE  = 2592000; // 30 days
    private const HMAC_SECRET    = 'change_this_to_a_strong_random_secret_key_32chars';
    private const ALLOWED_CLASSES = ['UserPreferences'];

    public function loadPreferences(): UserPreferences {
        if (!isset($_COOKIE[self::COOKIE_NAME])) {
            return new UserPreferences();
        }

        $cookieValue = $_COOKIE[self::COOKIE_NAME];

        $decoded = base64_decode($cookieValue, true);
        if ($decoded === false) {
            $this->clearCookie();
            return new UserPreferences();
        }

        $parts = explode('|', $decoded, 2);
        if (count($parts) !== 2) {
            $this->clearCookie();
            return new UserPreferences();
        }

        [$hmac, $serialized] = $parts;

        if (!$this->verifyHmac($serialized, $hmac)) {
            error_log('UserPreferences: HMAC verification failed. Possible tampering detected.');
            $this->clearCookie();
            return new UserPreferences();
        }

        $preferences = $this->safeUnserialize($serialized);

        if (!($preferences instanceof UserPreferences)) {
            $this->clearCookie();
            return new UserPreferences();
        }

        if (!$preferences->validate()) {
            $this->clearCookie();
            return new UserPreferences();
        }

        return $preferences;
    }

    public function savePreferences(UserPreferences $preferences): bool {
        if (!$preferences->validate()) {
            return false;
        }

        $serialized = serialize($preferences);
        $hmac       = $this->generateHmac($serialized);
        $payload    = base64_encode($hmac . '|' . $serialized);

        return setcookie(
            self::COOKIE_NAME,
            $payload,
            [
                'expires'  => time() + self::COOKIE_EXPIRE,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
    }

    public function clearCookie(): void {
        setcookie(
            self::COOKIE_NAME,
            '',
            [
                'expires'  => time() - 3600,
                'path'     => '/',
                'secure'   => true,
                'httponly' => true,
                'samesite' => 'Strict',
            ]
        );
        unset($_COOKIE[self::COOKIE_NAME]);
    }

    private function safeUnserialize(string $data): mixed {
        return unserialize($data, ['allowed_classes' => self::ALLOWED_CLASSES]);
    }

    private function generateHmac(string $data): string {
        return hash_hmac('sha256', $data, self::HMAC_SECRET);
    }

    private function verifyHmac(string $data, string $hmac): bool {
        $expected = $this->generateHmac($data);
        return hash_equals($expected, $hmac);
    }
}

function handleRequest(PreferencesManager $manager): void {
    if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {
        switch ($_POST['action']) {
            case 'save':
                $theme         = filter_input(INPUT_POST, 'theme', FILTER_SANITIZE_SPECIAL_CHARS) ?? 'light';
                $language      = filter_input(INPUT_POST, 'language', FILTER_SANITIZE_SPECIAL_CHARS) ?? 'en';
                $fontSize      = filter_input(INPUT_POST, 'fontSize', FILTER_VALIDATE_INT) ?: 14;
                $itemsPerPage  = filter_input(INPUT_POST, 'itemsPerPage', FILTER_VALIDATE_INT) ?: 10;
                $notifications = isset($_POST['notifications']) && $_POST['notifications'] === '1';
                $timezone      = filter_input(INPUT_POST, 'timezone', FILTER_SANITIZE_SPECIAL_CHARS) ?? 'UTC';

                $preferences = new UserPreferences(
                    (string) $theme,
                    (string) $language,
                    (int) $fontSize,
                    (int) $itemsPerPage,
                    $notifications,
                    (string) $timezone
                );

                if ($manager->savePreferences($preferences)) {
                    header('Location: ' . $_SERVER['PHP_SELF'] . '?saved=1');
                } else {
                    header('Location: ' . $_SERVER['PHP_SELF'] . '?error=invalid');
                }
                exit;

            case 'reset':
                $manager->clearCookie();
                header('Location: ' . $_SERVER['PHP_SELF'] . '?reset=1');
                exit;
        }
    }
}
